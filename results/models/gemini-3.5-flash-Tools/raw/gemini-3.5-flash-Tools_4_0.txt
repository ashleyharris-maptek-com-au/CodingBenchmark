// 3D vector
struct Vec3 {
    double x, y, z;
    Vec3 operator+(const Vec3& v) const { return {x + v.x, y + v.y, z + v.z}; }
    Vec3 operator-(const Vec3& v) const { return {x - v.x, y - v.y, z - v.z}; }
    Vec3 operator*(double s) const { return {x * s, y * s, z * s}; }
    Vec3 operator/(double s) const { return {x / s, y / s, z / s}; }
    double dot(const Vec3& v) const { return x * v.x + y * v.y + z * v.z; }
    Vec3 cross(const Vec3& v) const {
        return {
            y * v.z - z * v.y,
            z * v.x - x * v.z,
            x * v.y - y * v.x
        };
    }
    double length() const { return sqrt(x * x + y * y + z * z); }
    Vec3 normalize() const {
        double len = length();
        return len > 1e-9 ? *this / len : Vec3{0, 0, 0};
    }
};

// We need to construct the convex hull or faces of the container.
// Since the input only gives container vertices, how do we know the faces?
// Wait, the input describes "container vertices", but doesn't explicitly give faces?
// "Line 1: V edge_length
// Lines 2..V+1: x y z (container vertices)"
// If only vertices are given, is the container their convex hull?
// Yes, "Pack regular tetrahedrons into polyhedra". Usually, it's a convex polyhedron defined by the convex hull of the vertices.
// Let's assume the container is the convex hull of the given vertices.
// Alternatively, if V is small (e.g. 4 for a tetrahedron container, 8 for a box, etc.), we can compute the 3D convex hull.
// Let's implement a quick 3D convex hull or 3D point-in-convex-polyhedron test.
// For a convex polyhedron, a point is inside if it is on the correct side of all face planes.
// How do we find the face planes of the convex hull of V points?
// If V is small, we can do a simple O(V^3) or O(V^2) face identification.
// Since it's a convex hull, a triplet of vertices forms a face (or part of a face) if all other vertices lie on one side of the plane spanned by the triplet.
// Let's write a robust convex hull face finder.
struct Plane {
    Vec3 normal;
    double d; // normal.dot(p) + d = 0, or normal.dot(p) <= d for inside
};
vector<Plane> container_planes;

void compute_container_planes() {
    container_planes.clear();
    int n = container_vertices.size();
    if (n < 4) return;

    // Find a baseline centroid of all vertices to determine "inside" direction
    Vec3 centroid = {0, 0, 0};
    for (const auto& v : container_vertices) {
        centroid = centroid + v;
    }
    centroid = centroid / n;

    // Let's find all triplets that form the boundary
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            for (int k = j + 1; k < n; ++k) {
                Vec3 v0 = container_vertices[i];
                Vec3 v1 = container_vertices[j];
                Vec3 v2 = container_vertices[k];

                Vec3 normal = (v1 - v0).cross(v2 - v0);
                if (normal.length() < 1e-7) continue;
                normal = normal.normalize();

                // Orient normal outwards from centroid
                if (normal.dot(v0 - centroid) < 0) {
                    normal = normal * -1.0;
                }

                // Check if all other vertices are on the "inside" side of this plane
                // i.e., dot(v, normal) <= dot(v0, normal) + epsilon
                double max_proj = normal.dot(v0);
                bool is_face = true;
                for (int l = 0; l < n; ++l) {
                    if (l == i || l == j || l == k) continue;
                    double proj = normal.dot(container_vertices[l]);
                    if (proj > max_proj + 1e-5) {
                        is_face = false;
                        break;
                    }
                }

                if (is_face) {
                    // Check if we already have this plane
                    bool duplicate = false;
                    for (const auto& plane : container_planes) {
                        if (abs(plane.normal.dot(normal) - 1.0) < 1e-5 && abs(plane.d - max_proj) < 1e-5) {
                            duplicate = true;
                            break;
                        }
                    }
                    if (!duplicate) {
                        container_planes.push_back({normal, max_proj});
                    }
                }
            }
        }
    }
}

// Point in convex container test
bool is_point_inside_container(const Vec3& p, double margin = 0.0) {
    for (const auto& plane : container_planes) {
        if (plane.normal.dot(p) > plane.d - margin) {
            return false;
        }
    }
    return true;
}

// Test if tetrahedron is fully inside the container
bool is_tetrahedron_inside_container(const Tetrahedron& tet) {
    for (int i = 0; i < 4; ++i) {
        if (!is_point_inside_container(tet.v[i], 1e-7)) {
            return false;
        }
    }
    return true;
}

// GJK implementation for Tetrahedron-Tetrahedron collision detection
struct GJK {
    static Vec3 support(const Tetrahedron& t1, const Tetrahedron& t2, const Vec3& dir) {
        // Furthest point of t1 in dir
        double max_d1 = t1.v[0].dot(dir);
        int idx1 = 0;
        for (int i = 1; i < 4; ++i) {
            double d = t1.v[i].dot(dir);
            if (d > max_d1) {
                max_d1 = d;
                idx1 = i;
            }
        }
        // Furthest point of t2 in -dir
        Vec3 mdir = dir * -1.0;
        double max_d2 = t2.v[0].dot(mdir);
        int idx2 = 0;
        for (int i = 1; i < 4; ++i) {
            double d = t2.v[i].dot(mdir);
            if (d > max_d2) {
                max_d2 = d;
                idx2 = i;
            }
        }
        return t1.v[idx1] - t2.v[idx2];
    }

    static bool update_simplex(vector<Vec3>& simplex, Vec3& dir) {
        if (simplex.size() == 2) {
            // Line segment AB
            Vec3 B = simplex[0];
            Vec3 A = simplex[1];
            Vec3 AB = B - A;
            Vec3 AO = A * -1.0;
            if (AB.dot(AO) > 0) {
                dir = AB.cross(AO).cross(AB);
                if (dir.length() < 1e-9) {
                    // AO is parallel to AB, choose any perpendicular direction
                    dir = (abs(AB.x) > abs(AB.y)) ? Vec3{-AB.z, 0, AB.x} : Vec3{0, AB.z, -AB.y};
                }
                dir = dir.normalize();
            } else {
                simplex = {A};
                dir = AO.normalize();
            }
        } else if (simplex.size() == 3) {
            // Triangle ABC
            Vec3 C = simplex[0];
            Vec3 B = simplex[1];
            Vec3 A = simplex[2];
            Vec3 AB = B - A;
            Vec3 AC = C - A;
            Vec3 ABC = AB.cross(AC);
            Vec3 AO = A * -1.0;

            if (AB.cross(ABC).dot(AO) > 0) {
                simplex = {B, A};
                return update_simplex(simplex, dir);
            } else if (ABC.cross(AC).dot(AO) > 0) {
                simplex = {C, A};
                return update_simplex(simplex, dir);
            } else {
                if (ABC.dot(AO) > 0) {
                    dir = ABC.normalize();
                } else {
                    simplex = {B, C, A}; // swap order to maintain orientation if needed, but not strictly necessary for simple collision
                    dir = (ABC * -1.0).normalize();
                }
            }
        } else if (simplex.size() == 4) {
            // Tetrahedron ABCD
            Vec3 D = simplex[0];
            Vec3 C = simplex[1];
            Vec3 B = simplex[2];
            Vec3 A = simplex[3];

            Vec3 AB = B - A;
            Vec3 AC = C - A;
            Vec3 AD = D - A;
            Vec3 AO = A * -1.0;

            Vec3 ABC = AB.cross(AC);
            Vec3 ACD = AC.cross(AD);
            Vec3 ADB = AD.cross(AB);

            // Check if ABC normal points towards D. If not, orient it outwards
            if (ABC.dot(AD) > 0) ABC = ABC * -1.0;
            if (ACD.dot(AB) > 0) ACD = ACD * -1.0;
            if (ADB.dot(AC) > 0) ADB = ADB * -1.0;

            if (ABC.dot(AO) > 0) {
                simplex = {C, B, A};
                return update_simplex(simplex, dir);
            } else if (ACD.dot(AO) > 0) {
                simplex = {D, C, A};
                return update_simplex(simplex, dir);
            } else if (ADB.dot(AO) > 0) {
                simplex = {B, D, A};
                return update_simplex(simplex, dir);
            } else {
                return true; // Origin is inside the tetrahedron
            }
        }
        return false;
    }

    static bool intersects(const Tetrahedron& t1, const Tetrahedron& t2) {
        Vec3 dir = {1, 0, 0};
        vector<Vec3> simplex;
        simplex.push_back(support(t1, t2, dir));
        dir = simplex[0] * -1.0;
        dir = dir.normalize();

        for (int iter = 0; iter < 50; ++iter) {
            Vec3 A = support(t1, t2, dir);
            if (A.dot(dir) < 0) {
                return false; // No intersection
            }
            simplex.push_back(A);
            if (update_simplex(simplex, dir)) {
                return true; // Intersection
            }
        }
        return false;
    }
};

// Generate standard tetrahedron vertices centered at (0, 0, 0) with edge length e=1
// Height: h = e * sqrt(2/3) ≈ 0.8165
// Circumradius: R = e * sqrt(3/8) ≈ 0.6124
// Inradius: r = e / sqrt(24) ≈ 0.2041
Tetrahedron get_base_tetrahedron(double scale = 1.0) {
    Tetrahedron t;
    // Standard vertices from earlier web search
    t.v[0] = Vec3{ 1.0 / sqrt(3.0), 0.0, -1.0 / sqrt(24.0) } * scale;
    t.v[1] = Vec3{ -1.0 / (2.0 * sqrt(3.0)), 0.5, -1.0 / sqrt(24.0) } * scale;
    t.v[2] = Vec3{ -1.0 / (2.0 * sqrt(3.0)), -0.5, -1.0 / sqrt(24.0) } * scale;
    t.v[3] = Vec3{ 0.0, 0.0, 3.0 / sqrt(24.0) } * scale;
    return t;
}

// Convert Euler angles (rx, ry, rz in radians) to rotation matrix
void rotate_vertex(Vec3& p, double rx, double ry, double rz) {
    double cx = cos(rx), sx = sin(rx);
    double cy = cos(ry), sy = sin(ry);
    double cz = cos(rz), sz = sin(rz);

    // ZYX order: R = Rz * Ry * Rx
    double x = p.x, y = p.y, z = p.z;
    p.x = cz * cy * x + (cz * sy * sx - sz * cx) * y + (cz * sy * cx + sz * sx) * z;
    p.y = sz * cy * x + (sz * sy * sx + cz * cx) * y + (sz * sy * cx - cz * sx) * z;
    p.z = -sy * x + cy * sx * y + cy * cx * z;
}

Tetrahedron get_transformed_tetrahedron(const Tetrahedron& base, const Vec3& center, double rx, double ry, double rz) {
    Tetrahedron t = base;
    for (int i = 0; i < 4; ++i) {
        rotate_vertex(t.v[i], rx, ry, rz);
        t.v[i] = t.v[i] + center;
    }
    return t;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    if (!(cin >> V >> edge_length)) return 0;
    container_vertices.resize(V);
    double min_x = 1e18, max_x = -1e18;
    double min_y = 1e18, max_y = -1e18;
    double min_z = 1e18, max_z = -1e18;

    for (int i = 0; i < V; ++i) {
        cin >> container_vertices[i].x >> container_vertices[i].y >> container_vertices[i].z;
        min_x = min(min_x, container_vertices[i].x);
        max_x = max(max_x, container_vertices[i].x);
        min_y = min(min_y, container_vertices[i].y);
        max_y = max(max_y, container_vertices[i].y);
        min_z = min(min_z, container_vertices[i].z);
        max_z = max(max_z, container_vertices[i].z);
    }

    compute_container_planes();

    // Base tetrahedron
    Tetrahedron base_tet = get_base_tetrahedron(edge_length);

    // Let's implement a grid-based packing / search algorithm.
    // To maximize quality and performance, we can do a randomized/greedy search or coordinate sweep
    // with local optimization/shaking.
    // For ludicrous scale, a regular lattice filling is extremely fast and high quality.
    // Let's design a high-density lattice.
    // Can we pack regular tetrahedra using a known dense packing?
    // One dense packing is a dimer packing, but even simple FCC-like or tetrahedral honeycomb patterns
    // or alternating orientations can yield good packing.
    // Let's define a set of relative orientations that pack tightly.
    // For example, if we place two tetrahedra face-to-face, they form a triangular bipyramid.
    // Or we can just try a few symmetric orientations:
    // Regular tetrahedra can be packed by rotating them.
    // Let's generate a list of candidate orientations.
    // A regular tetrahedron has 12 rotational symmetries.
    // But we want to pack them. Let's try:
    // 1. A set of standard orientations (e.g. 4 or 8 orientations).
    // 2. Sweep the bounding box of the container.
    // Since we need to fit as many as possible within 30 seconds:
    // We can adjust the grid spacing depending on the container volume.
    // Bounding box volume:
    double dx = max_x - min_x;
    double dy = max_y - min_y;
    double dz = max_z - min_z;
    double bbox_volume = dx * dy * dz;
    // Tetrahedron volume: V = edge_length^3 / (6 * sqrt(2)) ≈ 0.11785 * edge_length^3
    double tet_vol = pow(edge_length, 3.0) / (6.0 * sqrt(2.0));
    double max_possible_tets = bbox_volume / tet_vol;

    // We can dynamically choose our step size and approach.
    // If max_possible_tets is very large (e.g., > 100,000), we must use a highly efficient grid/lattice sweep
    // without full O(N^2) checks (using spatial grid or structured grid).
    // If it's small, we can do more random trials, rotations, and local adjustments.

    vector<PlacedTet> placed_tets;

    // Let's define the grid step sizes.
    // Since R (circumradius) is e * sqrt(3/8) ≈ 0.6124 * e, the diameter is ~1.22 * e.
    // Let's step by a fraction of e, e.g. e/2 or e/sqrt(2), etc.
    // To optimize packing, we can sweep x, y, z.
    // Let's try a simple lattice sweep.
    // What if we try to fit them in a grid?
    // If we step z by h = e * sqrt(2/3) ≈ 0.8165 * e, y by e * sqrt(3)/2 ≈ 0.866 * e, x by e.
    // This matches the heights of tetrahedra layers.
    // Let's define a standard set of orientations.
    // Orientation 0: Identity (rx=0, ry=0, rz=0)
    // Orientation 1: Rotated to point downwards?
    // In a tetrahedral honeycomb, we alternate orientations.
    // Let's try 8 orientations.
    vector<vector<double>> orientations = {
        {0, 0, 0},
        {0, 0, M_PI},
        {0, M_PI, 0},
        {M_PI, 0, 0},
        {M_PI/2, 0, M_PI/2},
        {0, M_PI/2, M_PI/2},
        {M_PI/2, M_PI/2, 0},
        {M_PI/4, M_PI/4, M_PI/4}
    };

    // To make it compile on all platforms, ensure M_PI is defined
    #ifndef M_PI
    #define M_PI 3.14159265358979323846
    #endif

    // Let's define grid steps based on scale.
    // If container is massive, we do a fast grid sweep.
    // If container is small, we can do a finer search.
    double step_x = edge_length * 0.5;
    double step_y = edge_length * 0.5;
    double step_z = edge_length * 0.5;

    // Let's run a sweep.
    // For very large scale, we want to avoid O(N^2) overlap check.
    // We can maintain a grid of placed tetrahedrons.
    // Cell size for grid: edge_length * 1.5.
    // Every cell holds a list of placed tetrahedrons.
    // When placing a new one, we only check neighboring cells.

    struct Grid {
        double cell_size;
        double min_x, min_y, min_z;
        int nx, ny, nz;
        vector<vector<PlacedTet>> cells;

        Grid(double cs, double mx, double my, double mz, double max_x, double max_y, double max_z) 
            : cell_size(cs), min_x(mx), min_y(my), min_z(mz) {
            nx = max(1, (int)ceil((max_x - min_x) / cell_size));
            ny = max(1, (int)ceil((max_y - min_y) / cell_size));
            nz = max(1, (int)ceil((max_z - min_z) / cell_size));
            cells.resize(nx * ny * nz);
        }

        int get_index(const Vec3& p) {
            int cx = (int)((p.x - min_x) / cell_size);
            int cy = (int)((p.y - min_y) / cell_size);
            int cz = (int)((p.z - min_z) / cell_size);
            if (cx < 0) cx = 0; if (cx >= nx) cx = nx - 1;
            if (cy < 0) cy = 0; if (cy >= ny) cy = ny - 1;
            if (cz < 0) cz = 0; if (cz >= nz) cz = nz - 1;
            return cx * ny * nz + cy * nz + cz;
        }

        void add(const PlacedTet& t) {
            int idx = get_index(t.center);
            cells[idx].push_back(t);
        }

        bool check_collision(const Tetrahedron& tet, const Vec3& center) {
            // Check cells around center
            int cx = (int)((center.x - min_x) / cell_size);
            int cy = (int)((center.y - min_y) / cell_size);
            int cz = (int)((center.z - min_z) / cell_size);

            for (int dx = -1; dx <= 1; ++dx) {
                int ix = cx + dx;
                if (ix < 0 || ix >= nx) continue;
                for (int dy = -1; dy <= 1; ++dy) {
                    int iy = cy + dy;
                    if (iy < 0 || iy >= ny) continue;
                    for (int dz = -1; dz <= 1; ++dz) {
                        int iz = cz + dz;
                        if (iz < 0 || iz >= nz) continue;

                        int idx = ix * ny * nz + iy * nz + iz;
                        for (const auto& other : cells[idx]) {
                            if (GJK::intersects(tet, other.world_tet)) {
                                return true;
                            }
                        }
                    }
                }
            }
            return false;
        }
    };

    Grid grid(edge_length * 1.3, min_x, min_y, min_z, max_x, max_y, max_z);

    // Let's plan the execution time
    auto start_time = chrono::high_resolution_clock::now();

    // To improve density, we can try different packing lattices or random perturbations.
    // Let's do a structured sweep with multiple orientations.
    // For each grid point, we try orientations and choose the one that fits.
    // We can adjust steps depending on the container scale.
    if (max_possible_tets > 100000) {
        // Massive container: use larger steps to ensure we finish in 30s
        step_x = edge_length * 0.8;
        step_y = edge_length * 0.8;
        step_z = edge_length * 0.8;
    } else if (max_possible_tets > 10000) {
        step_x = edge_length * 0.6;
        step_y = edge_length * 0.6;
        step_z = edge_length * 0.6;
    } else {
        step_x = edge_length * 0.45;
        step_y = edge_length * 0.45;
        step_z = edge_length * 0.45;
    }

    // Run the sweep
    for (double z = min_z + edge_length * 0.2; z < max_z; z += step_z) {
        // Quick time limit check
        auto current_time = chrono::high_resolution_clock::now();
        double elapsed = chrono::duration<double>(current_time - start_time).count();
        if (elapsed > 25.0) break; // Keep safety margin of 5 seconds

        for (double y = min_y + edge_length * 0.2; y < max_y; y += step_y) {
            for (double x = min_x + edge_length * 0.2; x < max_x; x += step_x) {
                Vec3 center = {x, y, z};
                if (!is_point_inside_container(center)) continue;

                // Try orientations
                for (const auto& rot : orientations) {
                    Tetrahedron world_tet = get_transformed_tetrahedron(base_tet, center, rot[0], rot[1], rot[2]);
                    if (is_tetrahedron_inside_container(world_tet)) {
                        if (!grid.check_collision(world_tet, center)) {
                            PlacedTet pt = {center, rot[0], rot[1], rot[2], world_tet};
                            grid.add(pt);
                            placed_tets.push_back(pt);
                            break; // Successfully placed, move to next position
                        }
                    }
                }
            }
        }
    }

    // Output results
    cout << placed_tets.size() << "\n";
    for (const auto& pt : placed_tets) {
        cout << pt.center.x << " " << pt.center.y << " " << pt.center.z << " "
             << pt.rx << " " << pt.ry << " " << pt.rz << "\n";
    }

    return 0;
}
