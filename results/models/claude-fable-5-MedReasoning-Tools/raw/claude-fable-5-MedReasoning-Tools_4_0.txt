// Tetrahedron packing into a convex polyhedron (convex hull of given vertices).
//
// Strategy (robust to any checker Euler-angle convention):
//   * All tetrahedra are emitted with rotation (0,0,0)  -> they are all translates
//     of the checker's canonical tetrahedron, whatever that is.
//   * Two translates of a convex body never overlap when the center offset exceeds
//     the body's diameter.  A regular tetrahedron with edge e has diameter e.
//     So an FCC lattice with nearest-neighbour distance e*(1+eps) is overlap-free
//     regardless of orientation convention.  (Density ~16.7%.)
//   * Containment: every point of the tetrahedron is within the circumradius
//     R = e*sqrt(3/8) of its centroid, so a center at depth >= R from every hull
//     plane is safely inside for any orientation.
//
// Scalability: lattice rows are solved analytically (an integer interval per row
// from the half-space constraints), output is capped, and a 25 s wall-clock guard
// aborts pathological enumerations gracefully.

#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <cmath>
#include <vector>
#include <array>
#include <set>
#include <map>
#include <string>
#include <algorithm>
#include <numeric>
#include <chrono>
#include <random>

using namespace std;
typedef long long ll;

struct V3 { double x, y, z; };
static inline V3 sub(V3 a, V3 b) { return { a.x - b.x, a.y - b.y, a.z - b.z }; }
static inline V3 crs(V3 a, V3 b) {
    return { a.y * b.z - a.z * b.y, a.z * b.x - a.x * b.z, a.x * b.y - a.y * b.x };
}
static inline double dt(V3 a, V3 b) { return a.x * b.x + a.y * b.y + a.z * b.z; }

struct Plane { double nx, ny, nz, rhs; }; // feasible centers: n·x <= rhs

int main() {
    int V; double e;
    if (scanf("%d %lf", &V, &e) != 2 || V < 4 || !(e > 0)) { printf("0\n"); return 0; }
    vector<V3> pts(V);
    for (int i = 0; i < V; i++) {
        if (scanf("%lf %lf %lf", &pts[i].x, &pts[i].y, &pts[i].z) != 3) {
            printf("0\n"); return 0;
        }
    }

    auto t0 = chrono::steady_clock::now();
    auto elapsed = [&]() {
        return chrono::duration<double>(chrono::steady_clock::now() - t0).count();
    };

    // Bounding box / normalization
    double bx0 = 1e300, by0 = 1e300, bz0 = 1e300, bx1 = -1e300, by1 = -1e300, bz1 = -1e300;
    double maxAbs = 0;
    for (auto& p : pts) {
        bx0 = min(bx0, p.x); bx1 = max(bx1, p.x);
        by0 = min(by0, p.y); by1 = max(by1, p.y);
        bz0 = min(bz0, p.z); bz1 = max(bz1, p.z);
        maxAbs = max({ maxAbs, fabs(p.x), fabs(p.y), fabs(p.z) });
    }
    V3 cen = { (bx0 + bx1) / 2, (by0 + by1) / 2, (bz0 + bz1) / 2 };
    double scale = max({ bx1 - bx0, by1 - by0, bz1 - bz0, 1e-300 });

    // Per-axis normalization (affine map preserves hull combinatorics) so extreme
    // aspect ratios (e.g. 1e9 x 2 x 2 beams) stay numerically well-conditioned.
    double sx = max(bx1 - bx0, 1e-300), sy = max(by1 - by0, 1e-300), sz = max(bz1 - bz0, 1e-300);
    vector<V3> q(V);
    for (int i = 0; i < V; i++)
        q[i] = { (pts[i].x - cen.x) / sx, (pts[i].y - cen.y) / sy, (pts[i].z - cen.z) / sz };

    // Sample subset if huge vertex count (subset hull is an inner approximation -> safe)
    vector<int> idx(V);
    iota(idx.begin(), idx.end(), 0);
    {
        mt19937 rng(987654321u);
        shuffle(idx.begin(), idx.end(), rng);
        if ((int)idx.size() > 3000) idx.resize(3000);
    }
    int m = (int)idx.size();
    auto Q = [&](int i) -> V3& { return q[idx[i]]; };
    auto orient = [&](int a, int b, int c, int d) -> double {
        return dt(crs(sub(Q(b), Q(a)), sub(Q(c), Q(a))), sub(Q(d), Q(a)));
    };

    const double EPS = 1e-10;
    // Find initial non-degenerate simplex
    int t1 = -1, t2 = -1, t3 = -1;
    for (int i = 1; i < m && t1 < 0; i++) {
        V3 d = sub(Q(i), Q(0));
        if (dt(d, d) > 1e-20) t1 = i;
    }
    if (t1 >= 0) for (int i = 0; i < m && t2 < 0; i++) {
        if (i == 0 || i == t1) continue;
        V3 c = crs(sub(Q(t1), Q(0)), sub(Q(i), Q(0)));
        if (dt(c, c) > 1e-20) t2 = i;
    }
    if (t2 >= 0) for (int i = 0; i < m && t3 < 0; i++) {
        if (i == 0 || i == t1 || i == t2) continue;
        if (fabs(orient(0, t1, t2, i)) > EPS) t3 = i;
    }
    if (t3 < 0) { printf("0\n"); return 0; } // degenerate container

    // Incremental convex hull
    vector<array<int, 3>> fs;
    auto addF = [&](int a, int b, int c, int d) {
        if (orient(a, b, c, d) > 0) swap(b, c);
        fs.push_back({ a, b, c });
    };
    addF(0, t1, t2, t3); addF(0, t1, t3, t2); addF(0, t2, t3, t1); addF(t1, t2, t3, 0);
    for (int i = 0; i < m; i++) {
        if (i == 0 || i == t1 || i == t2 || i == t3) continue;
        vector<array<int, 3>> keep;
        set<pair<int, int>> edges;
        bool any = false;
        for (auto& f : fs) {
            if (orient(f[0], f[1], f[2], i) > EPS) {
                any = true;
                edges.insert({ f[0], f[1] });
                edges.insert({ f[1], f[2] });
                edges.insert({ f[2], f[0] });
            } else keep.push_back(f);
        }
        if (!any) continue;
        for (auto& ed : edges)
            if (!edges.count({ ed.second, ed.first }))
                keep.push_back({ ed.first, ed.second, i });
        fs.swap(keep);
    }

    // Geometry constants and safety margins
    double R = e * sqrt(3.0 / 8.0);
    int decs = 6;
    if (e < 1.0) decs = min(15, (int)ceil(-log10(e)) + 6);
    double quant = pow(10.0, -decs);
    double margin = e * 1e-6 + 4 * quant + (maxAbs + scale) * 1e-11;
    double clearance = R + margin;
    double nn = e + margin;        // min center-to-center distance
    double a = nn / sqrt(2.0);     // FCC: sites (i,j,k)*a + off, i+j+k even; nn dist = a*sqrt(2)

    // Half-space planes from hull faces (offsets from ORIGINAL coordinates, max over sample)
    vector<Plane> pl;
    {
        map<array<ll, 3>, char> seen;
        for (auto& f : fs) {
            V3 A = pts[idx[f[0]]], B = pts[idx[f[1]]], C = pts[idx[f[2]]];
            V3 n = crs(sub(B, A), sub(C, A));
            double len = sqrt(dt(n, n));
            if (!(len > 0)) continue;
            n.x /= len; n.y /= len; n.z /= len;
            array<ll, 3> key = { (ll)llround(n.x * 1e7), (ll)llround(n.y * 1e7), (ll)llround(n.z * 1e7) };
            if (seen.count(key)) continue;
            seen[key] = 1;
            double d = -1e300;
            for (int t = 0; t < m; t++) d = max(d, dt(n, pts[idx[t]]));
            pl.push_back({ n.x, n.y, n.z, d - clearance });
        }
    }
    if (pl.empty()) { printf("0\n"); return 0; }
#ifdef DEBUGP
    fprintf(stderr, "faces=%zu planes=%zu a=%g clearance=%g\n", fs.size(), pl.size(), a, clearance);
    for (auto& P : pl) fprintf(stderr, "  n=(%g,%g,%g) rhs=%g\n", P.nx, P.ny, P.nz, P.rhs);
#endif

    auto toIdx = [&](double v) -> ll {
        if (v > 4e18) return (ll)4e18;
        if (v < -4e18) return (ll)-4e18;
        return (ll)v;
    };

    const ll MAXOUT = 2000000;
    string out;
    char buf[160];

    // Fourier-Motzkin elimination so we only ever visit feasible lattice rows.
    // c2: constraints cy*y + cz*z <= r  (x eliminated); c1: cz*z <= r (y eliminated).
    const double NTH = 1e-9;
    bool infeas = false;
    auto dedupePush = [&](vector<array<double, 3>>& dst, set<array<ll, 3>>& seen,
                          double cy, double cz, double r) {
        double len = sqrt(cy * cy + cz * cz);
        if (len < 1e-13) { if (r < -margin * 0.5) infeas = true; return; }
        cy /= len; cz /= len; r /= len;
        array<ll, 3> key = { (ll)llround(cy * 1e7), (ll)llround(cz * 1e7),
                             (ll)llround(r / max(scale, 1e-300) * 1e7) };
        if (seen.insert(key).second) dst.push_back({ cy, cz, r });
    };
    vector<array<double, 3>> c2;   // {cy, cz, r}
    vector<array<double, 2>> c1;   // {cz, r}
    bool fmOK = pl.size() <= 400;
    if (fmOK) {
        set<array<ll, 3>> seen2;
        for (auto& P : pl)
            if (fabs(P.nx) <= NTH) dedupePush(c2, seen2, P.ny, P.nz, P.rhs);
        for (auto& P : pl) {
            if (P.nx <= NTH) continue;
            for (auto& Q : pl) {
                if (Q.nx >= -NTH) continue;
                double A = -Q.nx, B = P.nx;
                dedupePush(c2, seen2, P.ny * A + Q.ny * B, P.nz * A + Q.nz * B,
                           P.rhs * A + Q.rhs * B);
            }
        }
        if (c2.size() > 100000) fmOK = false;
    }
    if (fmOK) {
        set<array<ll, 3>> seen1;
        vector<array<double, 3>> tmp;
        for (auto& C : c2)
            if (fabs(C[0]) <= NTH) dedupePush(tmp, seen1, 0.0, C[1], C[2]);
        for (auto& C : c2) {
            if (C[0] <= NTH) continue;
            for (auto& D : c2) {
                if (D[0] >= -NTH) continue;
                double A = -D[0], B = C[0];
                dedupePush(tmp, seen1, 0.0, C[1] * A + D[1] * B, C[2] * A + D[2] * B);
            }
        }
        if (tmp.size() > 100000) fmOK = false;
        else for (auto& t : tmp) c1.push_back({ t[1], t[2] });
    }

    // Enumerate lattice; emit==false -> count only (fast per-row arithmetic)
    // par=true: FCC (sites i+j+k even, spacing a, nn dist a*sqrt2 = nn)
    // par=false: simple cubic (all sites, spacing nn) - wins for thin/1D containers
    auto run = [&](double ox, double oy, double oz, double a, bool par,
                   bool emit, ll cap) -> ll {
        if (infeas) return 0;
        ll step = par ? 2 : 1;
        double coefTh = a * 1e-9;
        // k-range: from c1 if available, else bbox
        double zA = bz0 - a, zB = bz1 + a;
        if (fmOK) {
            for (auto& C : c1) {
                if (fabs(C[0]) <= NTH) { if (C[1] < 0) return 0; }
                else if (C[0] > 0) zB = min(zB, C[1] / C[0]);
                else               zA = max(zA, C[1] / C[0]);
            }
            if (zA > zB) return 0;
        }
        ll klo = toIdx(ceil((zA - oz) / a)), khi = toIdx(floor((zB - oz) / a));
        ll cnt = 0, rows = 0;
        for (ll k = klo; k <= khi; k++) {
            double z = k * (double)a + oz;
            // j-range: from c2 if available, else bbox
            double yA = by0 - a, yB = by1 + a;
            bool kBad = false;
            if (fmOK) {
                for (auto& C : c2) {
                    double c0 = C[2] - C[1] * z;
                    if (fabs(C[0]) <= NTH) { if (c0 < 0) { kBad = true; break; } }
                    else if (C[0] > 0) yB = min(yB, c0 / C[0]);
                    else               yA = max(yA, c0 / C[0]);
                }
                if (kBad || yA > yB) continue;
            }
            ll jlo = toIdx(ceil((yA - oy) / a)), jhi = toIdx(floor((yB - oy) / a));
            for (ll j = jlo; j <= jhi; j++) {
                if (((++rows) & 4095) == 0 && elapsed() > 25.0) return cnt;
                double y = j * (double)a + oy;
                double lo = -1e300, hi = 1e300;
                bool bad = false;
                for (auto& P : pl) {
                    double c1v = P.nx * a;
                    double c0 = P.rhs - P.ny * y - P.nz * z - P.nx * ox;
                    if (fabs(c1v) <= coefTh) { if (c0 < 0) { bad = true; break; } }
                    else {
                        double t = c0 / c1v;
                        if (c1v > 0) { if (t < hi) hi = t; }
                        else         { if (t > lo) lo = t; }
                    }
                }
                if (bad || lo > hi) continue;
                if (lo < -4.1e18 || hi > 4.1e18) { // unbounded row: be safe
                    lo = max(lo, floor((bx0 - ox) / a) - 1);
                    hi = min(hi, ceil((bx1 - ox) / a) + 1);
                    if (lo > hi) continue;
                }
                ll il = (ll)ceil(lo), ih = (ll)floor(hi);
                if (par && ((il + j + k) & 1LL) != 0) il++;
                if (il > ih) continue;
                if (emit) {
                    for (ll i = il; i <= ih; i += step) {
                        double x = i * (double)a + ox;
                        int len = snprintf(buf, sizeof buf, "%.*f %.*f %.*f 0 0 0\n",
                                           decs, x, decs, y, decs, z);
                        out.append(buf, (size_t)len);
                        if (++cnt >= cap) return cnt;
                    }
                } else {
                    cnt += (ih - il) / step + 1;
                    if (cnt >= cap) return cnt;
                }
            }
        }
        return cnt;
    };

    // Choose lattice type (FCC vs simple cubic) and offset by trial counts.
    double bestOx = bx0, bestOy = by0, bestOz = bz0, bestA = a;
    bool bestPar = true;
    ll bestCnt = -1;
    {
        double est = ((bx1 - bx0) / a + 2) * ((by1 - by0) / a + 2) * ((bz1 - bz0) / a + 2) / 2.0;
        vector<array<double, 3>> cands;
        if (est < 200000.0) {
            for (int dx = 0; dx < 2; dx++)
                for (int dy = 0; dy < 2; dy++)
                    for (int dz = 0; dz < 2; dz++)
                        cands.push_back({ bx0 + dx * a / 2, by0 + dy * a / 2, bz0 + dz * a / 2 });
            cands.push_back({ cen.x, cen.y, cen.z }); // lattice site exactly at bbox center
        } else {
            cands.push_back({ bx0, by0, bz0 });
        }
        for (int type = 0; type < 2; type++) {
            double aa = type == 0 ? a : nn;
            bool par = (type == 0);
            for (auto& c : cands) {
                if (elapsed() > 18.0) break;
                ll cc = run(c[0], c[1], c[2], aa, par, false, MAXOUT);
                if (cc > bestCnt) {
                    bestCnt = cc; bestOx = c[0]; bestOy = c[1]; bestOz = c[2];
                    bestA = aa; bestPar = par;
                }
            }
        }
    }

    out.reserve(1 << 22);
    ll K = run(bestOx, bestOy, bestOz, bestA, bestPar, true, MAXOUT);

    // Fallback for tiny containers: try to fit a single tetrahedron near the deepest point
    if (K == 0) {
        V3 c = { 0, 0, 0 };
        for (auto& p : pts) { c.x += p.x; c.y += p.y; c.z += p.z; }
        c.x /= V; c.y /= V; c.z /= V;
        double step = scale * 0.25;
        for (int it = 0; it < 4000; it++) {
            double worst = 1e300; int wi = -1;
            for (int t = 0; t < (int)pl.size(); t++) {
                double s = pl[t].rhs - (pl[t].nx * c.x + pl[t].ny * c.y + pl[t].nz * c.z);
                if (s < worst) { worst = s; wi = t; }
            }
            if (worst >= 0) break;
            c.x -= pl[wi].nx * step; c.y -= pl[wi].ny * step; c.z -= pl[wi].nz * step;
            step *= 0.999;
        }
        bool ok = true;
        for (auto& P : pl)
            if (P.nx * c.x + P.ny * c.y + P.nz * c.z > P.rhs) { ok = false; break; }
        if (ok) {
            int len = snprintf(buf, sizeof buf, "%.*f %.*f %.*f 0 0 0\n",
                               decs, c.x, decs, c.y, decs, c.z);
            out.assign(buf, (size_t)len);
            K = 1;
        }
    }

    printf("%lld\n", K);
    fwrite(out.data(), 1, out.size(), stdout);
    return 0;
}
