```javascript
const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { width, height, grid, ignitions, towns, cutBudget } = data;

if (cutBudget <= 0 || towns.length === 0) {
    console.log(JSON.stringify({ firebreaks: [] }));
    process.exit(0);
}

// Precompute town locations and values
const townMap = new Map();
const townValue = {};
let totalTownValue = 0;
for (const t of towns) {
    const key = `${t.x},${t.y}`;
    townMap.set(key, t.id);
    townValue[t.id] = t.value;
    totalTownValue += t.value;
}

// Direction vectors for 4-way spread
const dirs = [[0, 1], [1, 0], [0, -1], [-1, 0]];

// Simulate fire spread given a set of cut cells (as a Set of "x,y" strings)
function simulateFire(cutSet) {
    const burned = new Set();
    const queue = [];

    // Initialize with ignitions (if not blocked by water/rock or cut)
    for (const [x, y] of ignitions) {
        if (x < 0 || x >= width || y < 0 || y >= height) continue;
        const cell = grid[y][x];
        if (cell === '~') continue; // water/rock blocks
        const key = `${x},${y}`;
        if (cutSet.has(key)) continue; // cut blocks
        burned.add(key);
        queue.push([x, y]);
    }

    // BFS for fire spread
    while (queue.length > 0) {
        const [x, y] = queue.shift();
        for (const [dx, dy] of dirs) {
            const nx = x + dx, ny = y + dy;
            if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;
            const nkey = `${nx},${ny}`;
            if (burned.has(nkey)) continue;
            const ncell = grid[ny][nx];
            if (ncell === '~') continue; // water/rock
            if (cutSet.has(nkey)) continue; // cut blocks
            burned.add(nkey);
            queue.push([nx, ny]);
        }
    }

    return burned;
}

// Compute initial burned area without any cuts
const initialBurned = simulateFire(new Set());

// Compute which towns are burned initially and their value
let initialTownLoss = 0;
for (const t of towns) {
    const key = `${t.x},${t.y}`;
    if (initialBurned.has(key)) {
        initialTownLoss += t.value;
    }
}

// If no towns are burned initially, no need to cut
if (initialTownLoss === 0) {
    console.log(JSON.stringify({ firebreaks: [] }));
    process.exit(0);
}

// Greedy: iteratively select the cut that saves the most additional town value
const cutSet = new Set();
const firebreaks = [];

// Precompute: for each burnable forest cell, how much town value it currently "protects" if cut
// We'll recompute marginal gain each iteration by simulating with current cuts + candidate

for (let iter = 0; iter < cutBudget; iter++) {
    let bestGain = 0;
    let bestCell = null;

    // Try every burnable forest cell that is not water/rock and not already cut
    for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
            if (grid[y][x] !== '.') continue; // only cut forest
            const key = `${x},${y}`;
            if (cutSet.has(key)) continue; // already cut

            // Tentatively add this cut
            cutSet.add(key);
            const burnedWithCut = simulateFire(cutSet);
            cutSet.delete(key); // revert

            // Compute town value saved by this cut: towns burned without cut but not with cut
            let gain = 0;
            for (const t of towns) {
                const tkey = `${t.x},${t.y}`;
                const burnedInitially = initialBurned.has(tkey);
                const burnedWithCutNow = burnedWithCut.has(tkey);
                if (burnedInitially && !burnedWithCutNow) {
                    gain += t.value;
                }
            }

            if (gain > bestGain) {
                bestGain = gain;
                bestCell = [x, y];
            }
        }
    }

    // If no positive gain, break early
    if (bestGain <= 0 || bestCell === null) break;

    // Apply the best cut
    const [bx, by] = bestCell;
    const bkey = `${bx},${by}`;
    cutSet.add(bkey);
    firebreaks.push(bestCell);
}

console.log(JSON.stringify({ firebreaks: firebreaks }));
```
