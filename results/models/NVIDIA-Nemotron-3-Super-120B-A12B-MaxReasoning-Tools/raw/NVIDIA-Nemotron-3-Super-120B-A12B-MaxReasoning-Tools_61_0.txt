declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

function dist(a: number[], b: number[]): number {
    const dx = a[0] - b[0];
    const dy = a[1] - b[1];
    return Math.sqrt(dx * dx + dy * dy);
}

function nearestNeighbor(depot: number[], packages: any[], used: boolean[]): number[] {
    const route: number[] = [];
    let current = depot;
    const remaining = packages.filter((_, i) => !used[i]);

    while (remaining.length > 0) {
        let bestIdx = -1;
        let bestDist = Infinity;
        for (let i = 0; i < remaining.length; i++) {
            const d = dist(current, [remaining[i].x, remaining[i].y]);
            if (d < bestDist) {
                bestDist = d;
                bestIdx = i;
            }
        }
        const next = remaining[bestIdx];
        route.push(next.id);
        used[packages.findIndex(p => p.id === next.id)] = true;
        current = [next.x, next.y];
        remaining.splice(bestIdx, 1);
    }

    return route;
}

function main() {
    const depot = data.depot;
    const truckCount = data.truckCount;
    const packages = data.packages;

    const used = new Array(packages.length).fill(false);
    const routes: number[][] = [];

    // Sort packages by distance from depot for better initial assignment
    const sortedPackages = [...packages].sort((a, b) => 
        dist(depot, [a.x, a.y]) - dist(depot, [b.x, b.y])
    );

    // Assign packages to routes using round-robin after sorting by proximity
    for (let i = 0; i < sortedPackages.length; i++) {
        const routeIndex = i % truckCount;
        if (!routes[routeIndex]) {
            routes[routeIndex] = [];
        }
        routes[routeIndex].push(sortedPackages[i].id);
    }

    // Mark used packages
    for (const route of routes) {
        for (const id of route) {
            const idx = packages.findIndex(p => p.id === id);
            if (idx !== -1) used[idx] = true;
        }
    }

    // Apply nearest neighbor to each route to improve order
    for (let i = 0; i < routes.length; i++) {
        if (routes[i].length === 0) continue;
        const routePackages = routes[i].map(id => 
            packages.find(p => p.id === id)!
        );
        const ordered = nearestNeighbor(depot, routePackages, used.slice()); // passed copy for marking, but we rebuild used
        routes[i] = ordered;
    }

    // Ensure all packages are used (fallback: assign any missing to first route)
    const allIds = packages.map(p => p.id);
    const assignedIds = routes.flat();
    const missing = allIds.filter(id => !assignedIds.includes(id));
    if (missing.length > 0 && routes.length > 0) {
        routes[0].push(...missing);
    }

    // Final output
    const output = { routes };
    console.log(JSON.stringify(output));
}

main();
