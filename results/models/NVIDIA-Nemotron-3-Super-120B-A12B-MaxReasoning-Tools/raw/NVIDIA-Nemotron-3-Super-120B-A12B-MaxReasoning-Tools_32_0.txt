To solve the Steiner Tree problem, we need to find a minimum-weight tree in a weighted undirected graph that connects all given terminal vertices. The tree may include non-terminal (Steiner) vertices to minimize the total weight. Given the NP-hard nature of the problem, we employ the Dreyfus-Wagner dynamic programming algorithm, which is efficient for small numbers of terminals (up to around 15-20), leveraging bitmask DP to explore subsets of terminals.

### Approach
1. **Problem Analysis**: The Steiner Tree problem requires connecting all terminal vertices with the minimum total edge weight, possibly using non-terminal vertices. The solution involves:
   - **Input Handling**: Read the number of vertices, edges, and terminals, followed by the edges and terminal list.
   - **Special Case Handling**: 
     - If there's only one terminal, the solution is trivial (no edges needed).
     - If there are no terminals, similarly, no edges are needed.
   - **Dreyfus-Wagner Algorithm**:
     - **Initialization**: For each terminal, initialize the DP state where the mask represents the terminal itself with zero cost.
     - **DP State Update**: For each subset of terminals (represented by a bitmask), compute the minimum cost to connect the subset by:
       - **Combining Subsets**: Split the mask into two non-overlapping submasks and combine their solutions.
       - **Relaxation Step**: Use the shortest paths (via Floyd-Warshall) to propagate the solution through the graph, allowing the use of non-terminal vertices as intermediate nodes.
     - **Result Extraction**: The solution for the full mask (all terminals) is found in `dp[full_mask][v]` for any vertex `v`, and we backtrack to reconstruct the tree edges.
   - **Output**: Print the total weight and the edges of the Steiner tree.

2. **Complexity Analysis**: The Dreyfus-Wagner algorithm runs in O(3^k * n + 2^k * n^2) time, where `k` is the number of terminals and `n` is the number of vertices. This is feasible for small `k` (up to 15-20) but becomes impractical for larger `k`. The space complexity is O(2^k * n) for storing DP states and parent pointers.

### Solution Code
```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <climits>
#include <algorithm>
#include <utility>
#include <tuple>
using namespace std;

const long long INF = 1e18;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int V, E, K;
    if (!(cin >> V >> E >> K)) {
        return 0;
    }

    vector<vector<long long>> dist(V, vector<long long>(V, INF));
    for (int i = 0; i < V; i++) {
        dist[i][i] = 0;
    }

    vector<tuple<int, int, long long>> edge_list;
    for (int i = 0; i < E; i++) {
        int u, v;
        long long w;
        cin >> u >> v >> w;
        if (w < dist[u][v]) {
            dist[u][v] = w;
            dist[v][u] = w;
        }
        edge_list.push_back(make_tuple(u, v, w));
    }

    vector<int> terminals(K);
    for (int i = 0; i < K; i++) {
        cin >> terminals[i];
    }

    if (K == 0) {
        cout << "0 0\n";
        return 0;
    }

    if (K == 1) {
        cout << "0 0\n";
        return 0;
    }

    vector<int> term_index(V, -1);
    for (int i = 0; i < K; i++) {
        term_index[terminals[i]] = i;
    }

    for (int k = 0; k < K; k++) {
        for (int i = 0; i < V; i++) {
            for (int j = 0; j < V; j++) {
                if (dist[i][k] != INF && dist[k][j] != INF) {
                    if (dist[i][j] > dist[i][k] + dist[k][j]) {
                        dist[i][j] = dist[i][k] + dist[k][j];
                    }
                }
            }
        }
    }

    int total_masks = 1 << K;
    vector<vector<long long>> dp(total_masks, vector<long long>(V, INF));
    vector<vector<int>> parent_mask(total_masks, vector<int>(V, -1));
    vector<vector<int>> parent_node(total_masks, vector<int>(V, -1));
    vector<vector<int>> split_mask(total_masks, vector<int>(V, 0));

    for (int i = 0; i < K; i++) {
        int t = terminals[i];
        dp[1 << i][t] = 0;
    }

    for (int mask = 1; mask < total_masks; mask++) {
        int sub = (mask - 1) & mask;
        while (sub > 0) {
            int other = mask ^ sub;
            if (other < sub) {
                sub = (sub - 1) & mask;
                continue;
            }
            for (int v = 0; v < V; v++) {
                if (dp[sub][v] < INF && dp[other][v] < INF) {
                    long long new_val = dp[sub][v] + dp[other][v];
                    if (new_val < dp[mask][v]) {
                        dp[mask][v] = new_val;
                        parent_mask[mask][v] = -1;
                        parent_node[mask][v] = v;
                        split_mask[mask][v] = sub;
                    }
                }
            }
            sub = (sub - 1) & mask;
        }

        bool updated;
        do {
            updated = false;
            for (int u = 0; u < V; u++) {
                for (int v = 0; v < V; v++) {
                    if (dp[mask][u] == INF) continue;
                    if (dist[u][v] == INF) continue;
                    long long new_val = dp[mask][u] + dist[u][v];
                    if (new_val < dp[mask][v]) {
                        dp[mask][v] = new_val;
                        parent_mask[mask][v] = mask;
                        parent_node[mask][v] = u;
                        split_mask[mask][v] = 0;
                        updated = true;
                    }
                }
            }
        } while (updated);
    }

    int full_mask = total_masks - 1;
    long long best = INF;
    int best_vertex = -1;
    for (int v = 0; v < V; v++) {
        if (dp[full_mask][v] < best) {
            best = dp[full_mask][v];
            best_vertex = v;
        }
    }

    if (best >= INF) {
        best = 0;
        best_vertex = 0;
    }

    vector<pair<int, int>> tree_edges;

    function<void(int, int)> dfs = [&](int mask, int v) {
        if (split_mask[mask][v] != 0) {
            int sub = split_mask[mask][v];
            int other = mask ^ sub;
            dfs(sub, v);
            dfs(other, v);
        } else if (parent_mask[mask][v] == mask) {
            int u = parent_node[mask][v];
            if (u != v) {
                tree_edges.push_back({min(u, v), max(u, v)});
                dfs(mask, u);
            }
        }
    };

    dfs(full_mask, best_vertex);

    sort(tree_edges.begin(), tree_edges.end());
    auto last = unique(tree_edges.begin(), tree_edges.end());
    tree_edges.erase(last, tree_edges.end());

    cout << best << " " << tree_edges.size() << "\n";
    for (auto edge : tree_edges) {
        cout << edge.first << " " << edge.second << "\n";
    }

    return 0;
}
```

### Explanation
1. **Input Handling**: The code reads the graph details (vertices, edges, terminals) and initializes the distance matrix with infinity (except diagonal entries set to zero).
2. **Floyd-Warshall Algorithm**: Computes the shortest paths between all pairs of vertices to facilitate the relaxation step in the DP.
3. **DP Initialization**: For each terminal, the DP state is initialized to zero cost at the terminal's vertex.
4. **DP State Update**:
   - **Subset Combination**: For each subset of terminals (bitmask), the solution is updated by combining solutions from two non-overlapping submasks.
   - **Relaxation Step**: Uses the precomputed shortest paths to propagate the solution through the graph, allowing intermediate (non-terminal) vertices to reduce the total cost.
5. **Result Extraction**: The minimum cost to connect all terminals is found by checking `dp[full_mask][v]` for all vertices `v`. The solution is then backtracked to reconstruct the edges of the Steiner tree.
6. **Output**: The total weight and the edges of the Steiner tree are printed, ensuring the edges are unique and formatted correctly.

This approach efficiently solves the Steiner Tree problem for small terminal sets using dynamic programming and shortest path precomputation, ensuring optimality and correctness within feasible computational limits.
