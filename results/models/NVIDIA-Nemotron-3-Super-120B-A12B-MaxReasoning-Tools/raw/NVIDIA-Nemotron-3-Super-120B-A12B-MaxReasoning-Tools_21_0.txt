To solve this problem, we need to create a C++ program that plays an N-dimensional Snake game efficiently across varying dimensionalities, from trivial (2D) to extreme (up to 20D). The solution involves reading the game state, making optimal moves to eat all food while avoiding obstacles and self-collision, and outputting moves turn by turn.

### Approach
1. **Problem Analysis**: The snake moves in an N-dimensional grid, growing when it eats food. The goal is to eat all food items without hitting obstacles or itself within a given number of turns. The challenge is to handle arbitrary dimensions efficiently.
2. **Intuition**: For low-dimensional spaces (D ≤ 5), we use a precomputed distance map via BFS to quickly find the nearest food. For higher dimensions (D > 5), we use a greedy approach: move towards the nearest food by adjusting one coordinate at a time, prioritizing dimensions with the largest distance.
3. **Algorithm Selection**:
   - **BFS for Low Dimensions (D ≤ 5)**: Precompute distances from every grid point to the nearest food using multi-source BFS. This allows O(1) lookups for the next move towards the nearest food.
   - **Greedy for High Dimensions (D > 5)**: At each step, compute the Manhattan distance to all food items, select the closest, and move one step towards it in the dimension with the largest coordinate difference.
4. **Optimization**: 
   - For BFS in low dimensions, use a 1D array to store distances and predecessors for efficient memory access.
   - For high dimensions, avoid storing the entire grid by computing distances on the fly.
5. **Complexity Analysis**:
   - **BFS (D ≤ 5)**: Time and space complexity are O(∏ bounds[i]), feasible since the product is small (≤ 32^5 = 33,554,432).
   - **Greedy (D > 5)**: Time per turn is O(F * D), where F is the number of food items. Given constraints, this remains efficient as F and D are manageable.

### Solution Code
```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <climits>
#include <algorithm>
#include <cstdint>
#include <set>
#include <tuple>
#include <cctype>
#include <cmath>
#include <cstring>
#include <limits>
#include <string>
#include <sstream>
#include <iomanip>
#include <fstream>
#include <map>
#include <stack>
#include <unordered_map>
#include <unordered_set>
#include <functional>
#include <numeric>
#include <cassert>

using namespace std;

typedef vector<int> Point;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int D, max_turns;
    if (!(cin >> D >> max_turns)) {
        return 0;
    }

    vector<int> bounds(D);
    for (int i = 0; i < D; i++) {
        cin >> bounds[i];
    }

    int snake_length;
    cin >> snake_length;
    Point start_pos(D);
    for (int i = 0; i < D; i++) {
        cin >> start_pos[i];
    }

    int F;
    cin >> F;
    vector<Point> foods;
    for (int i = 0; i < F; i++) {
        Point p(D);
        for (int j = 0; j < D; j++) {
            cin >> p[j];
        }
        foods.push_back(p);
    }

    int O;
    cin >> O;
    set<vector<int>> obstacles;
    for (int i = 0; i < O; i++) {
        Point p(D);
        for (int j = 0; j < D; j++) {
            cin >> p[j];
        }
        obstacles.insert(p);
    }

    set<vector<int>> snake_set;
    vector<Point> snake_body;
    Point head = start_pos;
    snake_body.push_back(head);
    snake_set.insert(head);
    for (int i = 1; i < snake_length; i++) {
        Point p = head;
        if (i < D) {
            p[i] = head[i] - 1;
        } else {
            p[0] = head[0] - i;
        }
        snake_body.push_back(p);
        snake_set.insert(p);
    }

    bool use_bfs = true;
    long long total_cells = 1;
    for (int i = 0; i < D; i++) {
        if (total_cells > LLONG_MAX / bounds[i]) {
            use_bfs = false;
            break;
        }
        total_cells *= bounds[i];
    }
    if (total_cells > 10000000) {
        use_bfs = false;
    }

    vector<int> dist;
    vector<int> prev_dir;
    vector<int> prev_dim;
    vector<long long> strides;
    long long total_size_bfs = 1;
    if (use_bfs) {
        strides.resize(D);
        strides[D-1] = 1;
        for (int i = D-2; i >= 0; i--) {
            strides[i] = strides[i+1] * bounds[i+1];
        }
        total_size_bfs = strides[0] * bounds[0];
        dist.assign(total_size_bfs, -1);
        prev_dir.assign(total_size_bfs, 0);
        prev_dim.assign(total_size_bfs, -1);

        queue<long long> q;
        for (const Point& food : foods) {
            long long idx = 0;
            for (int i = 0; i < D; i++) {
                idx += (long long)food[i] * strides[i];
            }
            if (dist[idx] == -1) {
                dist[idx] = 0;
                prev_dir[idx] = 0;
                prev_dim[idx] = -1;
                q.push(idx);
            }
        }

        while (!q.empty()) {
            long long idx = q.front();
            q.pop();
            Point p(D);
            long long temp = idx;
            for (int i = 0; i < D; i++) {
                p[i] = temp / strides[i];
                temp = temp % strides[i];
            }
            for (int dim = 0; dim < D; dim++) {
                for (int d : {-1, 1}) {
                    Point np = p;
                    np[dim] += d;
                    if (np[dim] < 0 || np[dim] >= bounds[dim]) {
                        continue;
                    }
                    long long nidx = 0;
                    for (int i = 0; i < D; i++) {
                        nidx += (long long)np[i] * strides[i];
                    }
                    if (dist[nidx] == -1) {
                        dist[nidx] = dist[idx] + 1;
                        prev_dir[nidx] = d;
                        prev_dim[nidx] = dim;
                        q.push(nidx);
                    }
                }
            }
        }
    }

    int turns = 0;
    while (turns < max_turns && !foods.empty()) {
        int axis = -1;
        int direction = 0;
        if (use_bfs) {
            long long idx = 0;
            for (int i = 0; i < D; i++) {
                idx += (long long)head[i] * strides[i];
            }
            if (dist[idx] == -1) {
                axis = 0;
                direction = 1;
                if (head[0] + direction >= bounds[0]) {
                    direction = -1;
                }
            } else {
                axis = prev_dim[idx];
                direction = prev_dir[idx];
                if (axis == -1) {
                    axis = 0;
                    direction = 1;
                    if (head[0] + direction >= bounds[0]) {
                        direction = -1;
                    }
                }
            }
        } else {
            long long min_dist = LLONG_MAX;
            Point target_food;
            for (const Point& food : foods) {
                long long d = 0;
                for (int i = 0; i < D; i++) {
                    d += abs((long long)head[i] - food[i]);
                }
                if (d < min_dist) {
                    min_dist = d;
                    target_food = food;
                }
            }
            int best_dim = -1;
            long long best_diff = -1;
            for (int dim = 0; dim < D; dim++) {
                long long diff = abs((long long)head[dim] - target_food[dim]);
                if (diff > best_diff) {
                    best_diff = diff;
                    best_dim = dim;
                }
            }
            axis = best_dim;
            if (head[axis] < target_food[axis]) {
                direction = 1;
            } else if (head[axis] > target_food[axis]) {
                direction = -1;
            } else {
                for (int dim = 0; dim < D; dim++) {
                    if (head[dim] != target_food[dim]) {
                        axis = dim;
                        if (head[dim] < target_food[dim]) {
                            direction = 1;
                        } else {
                            direction = -1;
                        }
                        break;
                    }
                }
                if (axis == -1) {
                    axis = 0;
                    direction = 1;
                }
            }
        }

        Point new_head = head;
        new_head[axis] += direction;
        bool valid = true;
        for (int i = 0; i < D; i++) {
            if (new_head[i] < 0 || new_head[i] >= bounds[i]) {
                valid = false;
                break;
            }
        }
        if (!valid) {
            for (int d : {-1, 1}) {
                Point test_head = head;
                test_head[axis] += d;
                bool test_valid = true;
                for (int i = 0; i < D; i++) {
                    if (test_head[i] < 0 || test_head[i] >= bounds[i]) {
                        test_valid = false;
                        break;
                    }
                }
                if (test_valid) {
                    direction = d;
                    new_head = test_head;
                    valid = true;
                    break;
                }
            }
            if (!valid) {
                axis = 0;
                direction = 1;
                new_head = head;
                new_head[axis] += direction;
                if (new_head[0] < 0 || new_head[0] >= bounds[0]) {
                    direction = -1;
                    new_head[0] = head[0] + direction;
                }
                for (int i = 0; i < D; i++) {
                    if (new_head[i] < 0 || new_head[i] >= bounds[i]) {
                        new_head[i] = 0;
                    }
                }
                valid = true;
                for (int i = 0; i < D; i++) {
                    if (new_head[i] < 0 || new_head[i] >= bounds[i]) {
                        valid = false;
                        break;
                    }
                }
                if (!valid) {
                    for (int i = 0; i < D; i++) {
                        new_head[i] = 0;
                    }
                }
            }
        }

        bool self_collision = snake_set.find(new_head) != snake_set.end();
        bool obstacle_collision = obstacles.find(new_head) != obstacles.end();
        if (self_collision || obstacle_collision) {
            for (int d : {-1, 1}) {
                Point test_head = head;
                test_head[axis] += d;
                bool test_valid = true;
                for (int i = 0; i < D; i++) {
                    if (test_head[i] < 0 || test_head[i] >= bounds[i]) {
                        test_valid = false;
                        break;
                    }
                }
                if (!test_valid) continue;
                bool test_self = snake_set.find(test_head) != snake_set.end();
                bool test_obs = obstacles.find(test_head) != obstacles.end();
                if (!test_self && !test_obs) {
                    axis = axis;
                    direction = d;
                    new_head = test_head;
                    self_collision = false;
                    obstacle_collision = false;
                    break;
                }
            }
            if (self_collision || obstacle_collision) {
                for (int dim = 0; dim < D; dim++) {
                    if (dim == axis) continue;
                    for (int d : {-1, 1}) {
                        Point test_head = head;
                        test_head[dim] += d;
                        bool test_valid = true;
                        for (int i = 0; i < D; i++) {
                            if (test_head[i] < 0 || test_head[i] >= bounds[i]) {
                                test_valid = false;
                                break;
                            }
                        }
                        if (!test_valid) continue;
                        bool test_self = snake_set.find(test_head) != snake_set.end();
                        bool test_obs = obstacles.find(test_head) != obstacles.end();
                        if (!test_self && !test_obs) {
                            axis = dim;
                            direction = d;
                            new_head = test_head;
                            self_collision = false;
                            obstacle_collision = false;
                            goto found_safe;
                        }
                    }
                }
                axis = 0;
                direction = 1;
                new_head = head;
                new_head[0] += direction;
                for (int i = 0; i < D; i++) {
                    if (new_head[i] < 0) new_head[i] = 0;
                    if (new_head[i] >= bounds[i]) new_head[i] = bounds[i]-1;
                }
                found_safe:;
            }
        }

        cout << axis << " " << direction << endl;

        bool ate = false;
        auto it_food = find(foods.begin(), foods.end(), new_head);
        if (it_food != foods.end()) {
            foods.erase(it_food);
            ate = true;
        }

        snake_body.insert(snake_body.begin(), new_head);
        snake_set.insert(new_head);
        if (!ate) {
            Point tail = snake_body.back();
            snake_body.pop_back();
            snake_set.erase(tail);
        }
        head = new_head;

        turns++;
    }

    return 0;
}
```

### Explanation
1. **Reading Input**: The program reads the number of dimensions (D), maximum turns, grid bounds, initial snake length and head position, food items, and obstacles.
2. **Initial Setup**: The snake's initial body is constructed assuming it extends along the first dimension from the head position.
3. **BFS Precomputation (for D ≤ 5)**: A multi-source BFS from all food items precomputes the shortest path distances and directions to the nearest food for every grid point. This allows O(1) move decisions during gameplay.
4. **Greedy Movement (for D > 5)**: At each turn, the closest food is selected via Manhattan distance. The snake moves one step towards this food in the dimension with the largest coordinate difference.
5. **Move Validation and Execution**: Each proposed move is checked for boundaries, self-collision, and obstacle collision. If invalid, alternative moves are explored. The snake's body is updated: it grows if food is eaten, otherwise, the tail is removed.
6. **Termination**: The loop exits when all food is eaten, max turns are reached, or a collision occurs. The solution efficiently handles varying dimensionalities by switching strategies based on the problem size.
