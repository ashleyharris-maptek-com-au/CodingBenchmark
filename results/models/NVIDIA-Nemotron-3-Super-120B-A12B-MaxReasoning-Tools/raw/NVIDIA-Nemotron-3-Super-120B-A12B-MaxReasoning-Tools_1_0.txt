// For very small N: Held-Karp DP (exact solution)
vector<int> heldKarpTSP(const vector<Point>& points) {
    int n = points.size();
    if (n <= 1) {
        vector<int> tour;
        if (n == 1) tour.push_back(0);
        return tour;
    }
    
    // dp[mask][i] = minimum cost to visit cities in mask ending at i
    // mask includes city 0 (start) and ends at i
    int fullMask = (1 << n) - 1;
    vector<vector<double>> dp(1 << n, vector<double>(n, numeric_limits<double>::infinity()));
    vector<vector<int>> parent(1 << n, vector<int>(n, -1));
    
    // Base case: starting at city 0
    dp[1][0] = 0;
    
    // Fill DP table
    for (int mask = 1; mask < (1 << n); mask++) {
        if (!(mask & 1)) continue; // Must include city 0
        
        for (int u = 0; u < n; u++) {
            if (!(mask & (1 << u))) continue;
            if (dp[mask][u] == numeric_limits<double>::infinity()) continue;
            
            for (int v = 0; v < n; v++) {
                if (mask & (1 << v)) continue; // Already visited
                
                int newMask = mask | (1 << v);
                double cost = dp[mask][u] + points[u].distanceTo(points[v]);
                
                if (cost < dp[newMask][v]) {
                    dp[newMask][v] = cost;
                    parent[newMask][v] = u;
                }
            }
        }
    }
    
    // Find optimal tour ending back at 0
    double minCost = numeric_limits<double>::infinity();
    int lastCity = -1;
    
    for (int u = 1; u < n; u++) {
        double cost = dp[fullMask][u] + points[u].distanceTo(points[0]);
        if (cost < minCost) {
            minCost = cost;
            lastCity = u;
        }
    }
    
    // Reconstruct path
    vector<int> tour;
    int mask = fullMask;
    int u = lastCity;
    
    while (u != -1) {
        tour.push_back(u);
        int prev = parent[mask][u];
        mask &= ~(1 << u);
        u = prev;
    }
    
    reverse(tour.begin(), tour.end());
    
    // Ensure we start at 0
    if (tour.empty() || tour[0] != 0) {
        // Find 0 and rotate
        auto it = find(tour.begin(), tour.end(), 0);
        if (it != tour.end()) {
            rotate(tour.begin(), it, tour.end());
        } else {
            tour.insert(tour.begin(), 0);
        }
    }
    
    return tour;
}

// Nearest Neighbor heuristic
vector<int> nearestNeighborTSP(const vector<Point>& points) {
    int n = points.size();
    if (n == 0) return {};
    if (n == 1) return {0};
    
    vector<bool> visited(n, false);
    vector<int> tour;
    tour.reserve(n);
    
    int current = 0;
    tour.push_back(current);
    visited[current] = true;
    
    for (int i = 1; i < n; i++) {
        int next = -1;
        double minDist = numeric_limits<double>::infinity();
        
        for (int j = 0; j < n; j++) {
            if (!visited[j]) {
                double dist = points[current].distanceTo(points[j]);
                if (dist < minDist) {
                    minDist = dist;
                    next = j;
                }
            }
        }
        
        if (next == -1) break; // Should not happen
        
        tour.push_back(next);
        visited[next] = true;
        current = next;
    }
    
    return tour;
}

// 2-opt optimization
void twoOpt(vector<int>& tour, const vector<Point>& points, int maxIterations = 1000) {
    int n = tour.size();
    if (n < 3) return;
    
    bool improved;
    int iterations = 0;
    
    do {
        improved = false;
        iterations++;
        
        for (int i = 0; i < n - 1; i++) {
            for (int k = i + 2; k < n; k++) {
                // Don't consider swapping adjacent edges
                if (k == i + 1) continue;
                
                int a = tour[i];
                int b = tour[(i + 1) % n];
                int c = tour[k];
                int d = tour[(k + 1) % n];
                
                double currentDist = points[a].distanceTo(points[b]) + 
                                   points[c].distanceTo(points[d]);
                double newDist = points[a].distanceTo(points[c]) + 
                               points[b].distanceTo(points[d]);
                
                if (newDist < currentDist - 1e-9) { // Significant improvement
                    // Reverse segment [i+1, k]
                    reverse(tour.begin() + i + 1, tour.begin() + k + 1);
                    improved = true;
                }
            }
            
            // Early exit if no improvement in recent iterations
            if (!improved && iterations > 10) break;
        }
        
        if (!improved || iterations >= maxIterations) break;
    } while (improved);
}

// Based on Wikipedia algorithm
uint64_t hilbertOrder(int x, int y, int pow, int rotate) {
    if (pow == 0) return 0;
    int hpow = 1 << (pow-1);
    int seg = ((x < hpow) ? 
              ((y < hpow) ? 0 : 3) : 
              ((y < hpow) ? 1 : 2));
    seg = (seg + rotate) & 3;
    int rotateDelta = [](int r) { return (r == 0) ? 3 : 0; }[seg];
    int nx = x & (hpow - 1);
    int ny = y & (hpow - 1);
    int nrot = (rotate + rotateDelta) & 3;
    uint64_t subSquareSize = 1ULL << (2*pow - 2);
    uint64_t ans = seg * subSquareSize;
    uint64_t add = hilbertOrder(nx, ny, pow-1, nrot);
    ans += (seg == 1 || seg == 2) ? add : (subSquareSize - 1 - add);
    return ans;
}

uint64_t hilbertOrder(int x, int y, int pow=21) {
    return hilbertOrder(x, y, pow, 0);
}

// Space-filling curve approach (Hilbert)
vector<int> hilbertCurveTSP(vector<Point> points) {
    if (points.empty()) return {};
    
    // Find bounding box
    double minX = points[0].x, maxX = points[0].x;
    double minY = points[0].y, maxY = points[0].y;
    
    for (const auto& p : points) {
        minX = min(minX, p.x);
        maxX = max(maxX, p.x);
        minY = min(minY, p.y);
        maxY = max(maxY, p.y);
    }
    
    // Handle degenerate case
    if (maxX == minX && maxY == minY) {
        vector<int> tour;
        for (size_t i = 0; i < points.size(); i++) tour.push_back(points[i].id);
        return tour;
    }
    
    // Scale to integer coordinates for Hilbert curve
    const int SCALE = 10000; // Enough precision for most cases
    const int POW = 21; // 2^21 > 2 million, good for Hilbert
    
    vector<pair<uint64_t, int>> hilbertIndices;
    hilbertIndices.reserve(points.size());
    
    for (const auto& p : points) {
        int scaledX = static_cast<int>((p.x - minX) / (maxX - minX) * ((1LL << POW) - 1));
        int scaledY = static_cast<int>((p.y - minY) / (maxY - minY) * ((1LL << POW) - 1));
        uint64_t order = hilbertOrder(scaledX, scaledY, POW);
        hilbertIndices.emplace_back(order, p.id);
    }
    
    // Sort by Hilbert order
    sort(hilbertIndices.begin(), hilbertIndices.end());
    
    // Extract tour
    vector<int> tour;
    tour.reserve(points.size());
    for (const auto& p : hilbertIndices) {
        tour.push_back(p.second);
    }
    
    return tour;
}

// Grid-based clustering for very large N
vector<int> gridClusteringTSP(const vector<Point>& points) {
    int n = points.size();
    if (n < 100) {
        // Fall back to simpler method for small n
        vector<int> tour = nearestNeighborTSP(points);
        twoOpt(tour, points, 100);
        return tour;
    }
    
    // Determine grid size based on sqrt(n) heuristic
    int gridSize = static_cast<int>(sqrt(n)) + 1;
    if (gridSize > 1000) gridSize = 1000; // Cap grid size
    
    // Find bounding box
    double minX = points[0].x, maxX = points[0].x;
    double minY = points[0].y, maxY = points[0].y;
    
    for (const auto& p : points) {
        minX = min(minX, p.x);
        maxX = max(maxX, p.x);
        minY = min(minY, p.y);
        maxY = max(maxY, p.y);
    }
    
    double width = maxX - minX;
    double height = maxY - minY;
    
    // Handle degenerate cases
    if (width < 1e-9) width = 1.0;
    if (height < 1e-9) height = 1.0;
    
    // Create grid
    vector<vector<int>> grid(gridSize, vector<int>(gridSize));
    vector<vector<Point>> gridPoints(gridSize, vector<Point>(gridSize));
    vector<int> pointCount(gridSize * gridSize, 0);
    
    // Assign points to grid cells
    for (const auto& p : points) {
        int gx = static_cast<int>((p.x - minX) / width * (gridSize - 1));
        int gy = static_cast<int>((p.y - minY) / height * (gridSize - 1));
        gx = max(0, min(gridSize - 1, gx));
        gy = max(0, min(gridSize - 1, gy));
        
        int idx = gy * gridSize + gx;
        grid[gy][gx] = pointCount[idx]++;
        gridPoints[gy][gx].push_back(p);
    }
    
    // Sort grid cells using Hilbert curve
    vector<pair<uint64_t, pair<int, int>>> hilbertCells;
    hilbertCells.reserve(gridSize * gridSize);
    
    for (int gy = 0; gy < gridSize; gy++) {
        for (int gx = 0; gx < gridSize; gx++) {
            if (pointCount[gy * gridSize + gx] > 0) {
                uint64_t order = hilbertOrder(gx, gy, 10); // Smaller pow for grid
                hilbertCells.emplace_back(order, make_pair(gy, gx));
            }
        }
    }
    
    sort(hilbertCells.begin(), hilbertCells.end());
    
    // Build tour by visiting points in each cell
    vector<int> tour;
    tour.reserve(n);
    
    for (const auto& cell : hilbertCells) {
        int gy = cell.second.first;
        int gx = cell.second.second;
        
        // For points within cell, use nearest neighbor
        if (!gridPoints[gy][gx].empty()) {
            vector<Point>& cellPoints = gridPoints[gy][gx];
            vector<int> cellTour = nearestNeighborTSP(cellPoints);
            
            // Map back to original indices
            for (int idx : cellTour) {
                tour.push_back(cellPoints[idx].id);
            }
        }
    }
    
    return tour;
}

// Approximate nearest neighbor with random sampling for huge N
vector<int> approximateTSP(const vector<Point>& points) {

    if (n < 1000) {
        vector<int> tour = nearestNeighborTSP(points);
        twoOpt(tour, points, 50);
        return tour;
    }
    
    // For very large N, use sampling-based approach
    vector<int> tour;
    tour.reserve(n);
    
    if (n == 0) return tour;
    
    // Start with city 0
    vector<bool> visited(n, false);
    tour.push_back(0);
    visited[0] = true;
    
    // Use a sample of candidates for each step to make it O(N*sampleSize)
    const int SAMPLE_SIZE = min(100, n / 10); // Adaptive sample size
    mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());
    
    int current = 0;
    for (int step = 1; step < n; step++) {
        int bestNext = -1;
        double bestDist = numeric_limits<double>::infinity();
        
        // Sample random unvisited cities
        vector<int> candidates;
        candidates.reserve(min(SAMPLE_SIZE, n - step));
        
        int count = 0;

            if (!visited[i]) {
                candidates.push_back(i);
                count++;
            }
        }
        
        // If we didn't get enough samples, take all remaining
        if (candidates.size() < SAMPLE_SIZE) {
            candidates.clear();
            for (int i = 0; i < n; i++) {
                if (!visited[i]) candidates.push_back(i);
            }
        }
        
        // Shuffle and take SAMPLE_SIZE
        shuffle(candidates.begin(), candidates.end(), rng);
        if (static_cast<int>(candidates.size()) > SAMPLE_SIZE) {
            candidates.resize(SAMPLE_SIZE);
        }
        
        // Find best among sample

            double dist = points[current].distanceTo(points[candidate]);

                bestDist = dist;
                bestNext = candidate;
            }

        // Fallback: if no candidate found (shouldn't happen), take first unvisited
        if (bestNext == -1) {
            for (int i = 0; i < n; i++) {

                    bestNext = i;
                    break;

            if (startIdx < step - 10) {

            }
        }

using namespace std;

struct Point {
    double x, y;
    int id;
    
    Point() : x(0), y(0), id(-1) {}
    Point(double x, double y, int id) : x(x), y(y), id(id) {}
    
    // Use squared distance for comparisons to avoid sqrt
    double distanceSq(const Point& other) const {
        double dx = x - other.x;
        double dy = y - other.y;
        return dx*dx + dy*dy;
    }
    
    double distance(const Point& other) const {
        double dx = x - other.x;
        double dy = y - other.y;
        return sqrt(dx*dx + dy*dy);
    }
};

// Simple timer
class Timer {
private:
    chrono::high_resolution_clock::time_point start_;
public:
    Timer() : start_(chrono::high_resolution_clock::now()) {}
    double elapsed() const {
        auto now = chrono::high_resolution_clock::now();
        return chrono::duration<double>(now - start_).count();
    }
    void reset() { start_ = chrono::high_resolution_clock::now(); }
};

// Inline helpers for performance
inline double distSq(const Point& a, const Point& b) {
    double dx = a.x - b.x;
    double dy = a.y - b.y;
    return dx*dx + dy*dy;
}

inline double dist(const Point& a, const Point& b) {
    double dx = a.x - b.x;
    double dy = a.y - b.y;
    return sqrt(dx*dx + dy*dy);
}

// Held-Karp for very small N (<= 15)
vector<int> heldKarpTSP(const vector<Point>& points) {
    int n = points.size();
    if (n <= 1) {
        vector<int> tour;
        if (n == 1) tour.push_back(0);
        return tour;
    }
    
    // We'll use bitmask DP
    int fullMask = (1 << n) - 1;
    // dp[mask][last] = min cost
    // Use vector of vectors for cache efficiency
    vector<vector<double>> dp(1 << n, vector<double>(n, numeric_limits<double>::infinity()));
    vector<vector<int>> parent(1 << n, vector<int>(n, -1));
    
    // Base: start at city 0
    dp[1][0] = 0;
    
    // Iterate over all masks
    for (int mask = 1; mask < (1 << n); mask++) {
        // Skip if doesn't include start city (0)
        if (!(mask & 1)) continue;
        
        for (int last = 0; last < n; last++) {
            if (!(mask & (1 << last))) continue;
            if (dp[mask][last] == numeric_limits<double>::infinity()) continue;
            
            // Try to go to next unvisited city
            for (int next = 0; next < n; next++) {
                if (mask & (1 << next)) continue; // Already visited
                
                int newMask = mask | (1 << next);
                double newCost = dp[mask][last] + dist(points[last], points[next]);
                
                if (newCost < dp[newMask][next]) {
                    dp[newMask][next] = newCost;
                    parent[newMask][next] = last;
                }
            }
        }
    }
    
    // Find best tour ending back at 0
    double minCost = numeric_limits<double>::infinity();
    int bestLast = -1;
    
    for (int last = 1; last < n; last++) {
        double tourCost = dp[fullMask][last] + dist(points[last], points[0]);
        if (tourCost < minCost) {
            minCost = tourCost;
            bestLast = last;
        }
    }
    
    // Reconstruct path
    vector<int> tour;
    int mask = fullMask;
    int curr = bestLast;
    
    while (curr != -1) {
        tour.push_back(curr);
        int prev = parent[mask][curr];
        mask &= ~(1 << curr);
        curr = prev;
    }
    
    reverse(tour.begin(), tour.end());
    
    // Ensure starts at 0
    if (tour.empty() || tour[0] != 0) {
        auto it = find(tour.begin(), tour.end(), 0);
        if (it != tour.end()) {
            rotate(tour.begin(), it, tour.end());
        } else {
            tour.insert(tour.begin(), 0);
        }
    }
    
    return tour;
}

// Nearest Neighbor - O(N²) but fast in practice for small N
vector<int> nearestNeighborTSP(const vector<Point>& points) {
    int n = points.size();
    if (n == 0) return {};
    if (n == 1) return {0};
    
    vector<bool> visited(n, false);
    vector<int> tour;
    tour.reserve(n);
    
    int current = 0;
    tour.push_back(current);
    visited[current] = true;
    
    // Pre-allocate candidate list to avoid reallocations
    vector<int> candidates;
    candidates.reserve(n);
    
    for (int step = 1; step < n; step++) {
        // Build list of unvisited cities
        candidates.clear();
        for (int i = 0; i < n; i++) {
            if (!visited[i]) candidates.push_back(i);
        }
        
        // Find nearest
        int bestNext = -1;
        double bestDistSq = numeric_limits<double>::infinity();
        
        for (int candidate : candidates) {
            double d = distSq(points[current], points[candidate]);
            if (d < bestDistSq) {
                bestDistSq = d;
                bestNext = candidate;
            }
        }
        
        if (bestNext == -1) break; // Should not happen
        
        tour.push_back(bestNext);
        visited[bestNext] = true;
        current = bestNext;
    }
    
    return tour;
}

// Optimized 2-opt - only accepts improving moves
void twoOptFast(vector<int>& tour, const vector<Point>& points, int maxIterations = 100) {
    int n = tour.size();
    if (n < 4) return;
    
    bool improved;
    int iterations = 0;
    
    do {
        improved = false;
        iterations++;
        
        // Try all possible 2-opt moves
        for (int i = 0; i < n - 2; i++) {
            int a = tour[i];
            int b = tour[i + 1];
            
            for (int k = i + 2; k < n; k++) {
                int c = tour[k];
                int d = tour[(k + 1) % n];
                
                // Calculate gain: (ab + cd) - (ac + bd)
                double gain = dist(points[a], points[b]) + dist(points[c], points[d]) 
                            - dist(points[a], points[c]) - dist(points[b], points[d]);
                
                if (gain > 1e-7) { // Significant improvement
                    // Reverse segment [i+1, k]
                    reverse(tour.begin() + i + 1, tour.begin() + k + 1);
                    improved = true;
                    // After improvement, restart search (simpler)
                    goto next_iteration;
                }
            }
        }
        
    next_iteration:
        if (!improved || iterations >= maxIterations) break;
    } while (improved);
}

// Hilbert curve implementation - optimized
static uint64_t hilbertOrder(int x, int y, int pow, int rot) {
    if (pow == 0) return 0;
    int hpow = 1 << (pow - 1);
    int quadrant = ((x < hpow) ? 
                   ((y < hpow) ? 0 : 3) : 
                   ((y < hpow) ? 1 : 2));
    quadrant = (quadrant + rot) & 3;
    const int rotDelta[4] = {3, 0, 0, 1};
    int nx = x & (hpow - 1);
    int ny = y & (hpow - 1);
    int nrot = (rot + rotDelta[quadrant]) & 3;
    uint64_t subSize = 1ULL << (2 * pow - 2);
    uint64_t ans = quadrant * subSize;
    uint64_t add = hilbertOrder(nx, ny, pow - 1, nrot);
    ans += (quadrant == 1 || quadrant == 2) ? add : (subSize - 1 - add);
    return ans;
}

uint64_t hilbertOrder(int x, int y, int pow = 21) {
    return hilbertOrder(x, y, pow, 0);
}

// Hilbert curve TSP - O(N log N) due to sort
vector<int> hilbertCurveTSP(const vector<Point>& points) {
    if (points.empty()) return {};
    
    // Find bounding box in single pass
    double minX = points[0].x, maxX = points[0].x;
    double minY = points[0].y, maxY = points[0].y;
    
    for (size_t i = 1; i < points.size(); i++) {
        if (points[i].x < minX) minX = points[i].x;
        else if (points[i].x > maxX) maxX = points[i].x;
        if (points[i].y < minY) minY = points[i].y;
        else if (points[i].y > maxY) maxY = points[i].y;
    }
    
    double width = maxX - minX;
    double height = maxY - minY;
    
    // Handle degenerate cases
    if (width < 1e-12) width = 1.0;
    if (height < 1e-12) height = 1.0;
    
    // Parameters for Hilbert curve
    const int HILBERT_POW = 21; // 2^21 = 2M+ resolution
    const uint64_t HILBERT_SIZE = 1ULL << HILBERT_POW;
    
    // Create array of (hilbert_order, id) pairs
    vector<pair<uint64_t, int>> orderId;
    orderId.reserve(points.size());
    
    for (const auto& p : points) {
        // Scale to [0, HILBERT_SIZE-1]
        int sx = static_cast<int>((p.x - minX) / width * (HILBERT_SIZE - 1));
        int sy = static_cast<int>((p.y - minY) / height * (HILBERT_SIZE - 1));
        // Clamp to valid range
        sx = max(0, min(static_cast<int>(HILBERT_SIZE) - 1, sx));
        sy = max(0, min(static_cast<int>(HILBERT_SIZE) - 1, sy));
        
        uint64_t order = hilbertOrder(sx, sy, HILBERT_POW);
        orderId.emplace_back(order, p.id);
    }
    
    // Sort by Hilbert order - O(N log N)
    sort(orderId.begin(), orderId.end());
    
    // Extract tour
    vector<int> tour;
    tour.reserve(points.size());
    for (const auto& p : orderId) {
        tour.push_back(p.second);
    }
    
    return tour;
}

// Grid-based approach for very large N
vector<int> gridClusteringTSP(const vector<Point>& points) {
    int n = points.size();
    if (n < 100) {
        vector<int> tour = nearestNeighborTSP(points);
        twoOptFast(tour, points, 20);
        return tour;
    }
    
    // Determine grid size - aim for ~100 points per cell on average
    int gridSize = static_cast<int>(sqrt(n / 100.0)) + 1;
    if (gridSize < 10) gridSize = 10;
    if (gridSize > 1000) gridSize = 1000; // Cap to avoid too many cells
    
    // Find bounding box
    double minX = points[0].x, maxX = points[0].x;
    double minY = points[0].y, maxY = points[0].y;
    
    for (const auto& p : points) {
        if (p.x < minX) minX = p.x;
        else if (p.x > maxX) maxX = p.x;
        if (p.y < minY) minY = p.y;
        else if (p.y > maxY) maxY = p.y;
    }
    
    double width = maxX - minX;
    double height = maxY - minY;
    
    if (width < 1e-12) width = 1.0;
    if (height < 1e-12) height = 1.0;
    
    // Create grid - using flat arrays for better cache performance
    vector<int> pointCount(gridSize * gridSize, 0);
    // First pass: count points per cell
    for (const auto& p : points) {
        int gx = static_cast<int>((p.x - minX) / width * (gridSize - 1));
        int gy = static_cast<int>((p.y - minY) / height * (gridSize - 1));
        gx = max(0, min(gridSize - 1, gx));
        gy = max(0, min(gridSize - 1, gy));
        pointCount[gy * gridSize + gx]++;
    }
    
    // Second pass: allocate and fill grid
    vector<vector<Point>> grid(gridSize);
    for (int gy = 0; gy < gridSize; gy++) {
        grid[gy].reserve(gridSize * 10); // Rough estimate
        for (int gx = 0; gx < gridSize; gx++) {
            int count = pointCount[gy * gridSize + gx];
            if (count > 0) {
                // We'll store points directly in a flat structure per cell
                // Actually, let's use a different approach: store indices
            }
        }
    }
    
    // Simpler approach: create buckets
    vector<vector<int>> buckets(gridSize * gridSize);
    
    for (int i = 0; i < n; i++) {
        const auto& p = points[i];
        int gx = static_cast<int>((p.x - minX) / width * (gridSize - 1));
        int gy = static_cast<int>((p.y - minY) / height * (gridSize - 1));
        gx = max(0, min(gridSize - 1, gx));
        gy = max(0, min(gridSize - 1, gy));
        int cell = gy * gridSize + gx;
        buckets[cell].push_back(i);
    }
    
    // Sort cells by Hilbert order
    vector<pair<uint64_t, int>> hilbertCells;
    hilbertCells.reserve(gridSize * gridSize);
    
    for (int gy = 0; gy < gridSize; gy++) {
        for (int gx = 0; gx < gridSize; gx++) {
            if (!buckets[gy * gridSize + gx].empty()) {
                uint64_t order = hilbertOrder(gx, gy, 10); // Lower res for grid
                hilbertCells.emplace_back(order, gy * gridSize + gx);
            }
        }
    }
    
    sort(hilbertCells.begin(), hilbertCells.end());
    
    // Build tour
    vector<int> tour;
    tour.reserve(n);
    
    for (const auto& cellPair : hilbertCells) {
        int cell = cellPair.second;
        const vector<int>& indices = buckets[cell];
        
        if (indices.empty()) continue;
        
        // For points in this cell, use nearest neighbor
        if (indices.size() == 1) {
            tour.push_back(indices[0]);
        } else {
            // Extract points for this cell
            vector<Point> cellPoints;
            cellPoints.reserve(indices.size());
            for (int idx : indices) {
                cellPoints.push_back(points[idx]);
            }
            
            // Solve TSP for this cell
            vector<int> cellTour = nearestNeighborTSP(cellPoints);
            twoOptFast(cellTour, cellPoints, 5); // Light optimization
            
            // Map back to original indices
            for (int idx : cellTour) {
                tour.push_back(indices[idx]);
            }
        }
    }
    
    return tour;
}

// Approximate TSP for huge N - O(N * sampleSize)
vector<int> approximateTSP(const vector<Point>& points) {
    int n = points.size();
    if (n < 1000) {
        vector<int> tour = nearestNeighborTSP(points);
        twoOptFast(tour, points, 30);
        return tour;
    }
    
    vector<int> tour;
    tour.reserve(n);
    
    if (n == 0) return tour;
    
    vector<bool> visited(n, false);
    visited[0] = true;
    tour.push_back(0);
    
    // Parameters
    const int SAMPLE_SIZE = min(50, n / 20); // Adaptive sample size
    mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());
    
    int current = 0;
    
    for (int step = 1; step < n; step++) {
        int bestNext = -1;
        double bestDistSq = numeric_limits<double>::infinity();
        
        #if defined(_GLIBCXX_PARALLEL) || defined(_LIBCPP_VERSION)
        // Try to use parallel execution if available
        vector<int> candidates;
        candidates.reserve(min(SAMPLE_SIZE * 2, n - step));
        
        int count = 0;
        for (int i = 0; i < n && count < SAMPLE_SIZE * 2; i++) {
            if (!visited[i]) {
                candidates.push_back(i);
                count++;
            }
        }
        
        if (candidates.empty()) break;
        
        // If we have few candidates, check all
        if (static_cast<int>(candidates.size()) <= SAMPLE_SIZE) {
            for (int candidate : candidates) {
                double d = distSq(points[current], points[candidate]);
                if (d < bestDistSq) {
                    bestDistSq = d;
                    bestNext = candidate;
                }
            }
        } else {
            // Shuffle and sample
            shuffle(candidates.begin(), candidates.end(), rng);
            if (static_cast<int>(candidates.size()) > SAMPLE_SIZE) {
                candidates.resize(SAMPLE_SIZE);
            }
            
            for (int candidate : candidates) {
                double d = distSq(points[current], points[candidate]);
                if (d < bestDistSq) {
                    bestDistSq = d;
                    bestNext = candidate;
                }
            }
        }
        #else
        // Sequential version
        vector<int> candidates;
        candidates.reserve(min(SAMPLE_SIZE * 2, n - step));
        
        int count = 0;
        for (int i = 0; i < n && count < SAMPLE_SIZE * 2; i++) {
            if (!visited[i]) {
                candidates.push_back(i);
                count++;
            }
        }
        
        if (candidates.empty()) break;
        
        if (static_cast<int>(candidates.size()) <= SAMPLE_SIZE) {
            for (int candidate : candidates) {
                double d = distSq(points[current], points[candidate]);
                if (d < bestDistSq) {
                    bestDistSq = d;
                    bestNext = candidate;
                }
            }
        } else {
            shuffle(candidates.begin(), candidates.end(), rng);
            if (static_cast<int>(candidates.size()) > SAMPLE_SIZE) {
                candidates.resize(SAMPLE_SIZE);
            }
            
            for (int candidate : candidates) {
                double d = distSq(points[current], points[candidate]);
                if (d < bestDistSq) {
                    bestDistSq = d;
                    bestNext = candidate;
                }
            }
        }
        #endif
        
        // Fallback to first unvisited if no candidate found
        if (bestNext == -1) {
            for (int i = 0; i < n; i++) {
                if (!visited[i]) {
                    bestNext = i;
                    break;
                }
            }
        }
        
        if (bestNext == -1) break; // Should not happen
        
        tour.push_back(bestNext);
        visited[bestNext] = true;
        current = bestNext;
        
        #ifndef NDEBUG
        // Periodic validation in debug builds
        if (step % 100000 == 0) {
            // Quick sanity check
        }
        #endif
        
        #ifndef NDEBUG
        // Periodic light optimization to prevent drift
        if (step % 50000 == 0 && step > 10000) {
            // Apply 2-opt on a sliding window
            int windowStart = max(0, step - 1000);
            int windowEnd = min(n, step + 100);
            if (windowEnd - windowStart > 20) {
                vector<int> window(tour.begin() + windowStart, tour.begin() + windowEnd);
                twoOptFast(window, points, 3);
                copy(window.begin(), window.end(), tour.begin() + windowStart);
            }
        }
        #endif
    }
    
    #ifndef NDEBUG
    // Final validation
    if (tour.size() != static_cast<size_t>(n)) {
        // Fill in missing cities (should not happen in correct implementation)
        vector<bool> inTour(n, false);
        for (int city : tour) {
            if (city >= 0 && city < n) inTour[city] = true;
        }
        for (int i = 0; i < n; i++) {
            if (!inTour[i]) tour.push_back(i);
        }
    }
    #endif
    
    // Light final optimization for manageable sizes
    if (n <= 50000) {
        twoOptFast(tour, points, 10);
    } else if (n <= 500000) {
        // Chunk-based 2-opt
        int chunkSize = 1000;
        for (int start = 0; start < n; start += chunkSize) {
            int end = min(start + chunkSize, n);
            if (end - start > 10) {
                vector<int> chunk(tour.begin() + start, tour.begin() + end);
                twoOptFast(chunk, points, 3);
                copy(chunk.begin(), chunk.end(), tour.begin() + start);
            }
        }
    }
    
    return tour;
}

// Main adaptive solver
vector<int> solveTSP(const vector<Point>& points) {
    int n = points.size();
    
    if (n == 0) return {};
    if (n == 1) return {0};
    
    // Choose strategy based on problem size
    if (n <= 12) {
        // Held-Karp for very small instances (exact)
        return heldKarpTSP(points);
    } else if (n <= 30) {
        // Nearest neighbor with 2-opt for small instances
        vector<int> tour = nearestNeighborTSP(points);
        twoOptFast(tour, points, 200);
        return tour;
    } else if (n <= 100) {
        // NN + 2-opt + light 3-opt
        vector<int> tour = nearestNeighborTSP(points);
        twoOptFast(tour, points, 100);
        // Simplified 3-opt: just try reversing triplets
        for (int pass = 0; pass < 5; pass++) {
            bool improved = false;
            for (int i = 0; i < n - 2; i++) {
                for (int j = i + 2; j < n - 1; j++) {
                    for (int k = j + 2; k < n; k++) {
                        // Try reversing middle segment
                        vector<int> temp = tour;
                        reverse(temp.begin() + i + 1, temp.begin() + j + 1);
                        
                        double oldDist = dist(points[tour[i]], points[tour[i+1]]) +
                                        dist(points[tour[j]], points[tour[j+1]]) +
                                        dist(points[tour[k]], points[tour[(k+1)%n]]);
                        
                        double newDist = dist(points[temp[i]], points[temp[i+1]]) +
                                        dist(points[temp[j]], points[temp[j+1]]) +
                                        dist(points[temp[k]], points[temp[(k+1)%n]]);
                        
                        if (newDist < oldDist - 1e-7) {
                            tour = move(temp);
                            improved = true;
                            break;
                        }
                    }
                    if (improved) break;
                }
                if (improved) break;
            }
            if (!improved) break;
        }
        return tour;
    } else if (n <= 1000) {
        // Medium: NN + 2-opt
        vector<int> tour = nearestNeighborTSP(points);
        twoOptFast(tour, points, 50);
        return tour;
    } else if (n <= 50000) {
        // Large: Hilbert curve
        vector<int> tour = hilbertCurveTSP(points);
        // Light optimization based on size
        if (n <= 5000) {
            twoOptFast(tour, points, 30);
        } else {
            // Apply 2-opt in chunks
            int chunkSize = 500;
            for (int start = 0; start < n; start += chunkSize) {
                int end = min(start + chunkSize, n);
                if (end - start > 20) {
                    vector<int> chunk(tour.begin() + start, tour.begin() + end);
                    twoOptFast(chunk, points, 5);
                    copy(chunk.begin(), chunk.end(), tour.begin() + start);
                }
            }
        }
        return tour;
    } else if (n <= 500000) {
        // Very large: Grid clustering
        return gridClusteringTSP(points);
    } else {
        // Extreme/Epic: Approximate method
        return approximateTSP(points);
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int N;
    if (!(cin >> N)) {
        cerr << "Error: Failed to read number of cities" << endl;
        return 1;
    }
    
    if (N < 0) {
        cerr << "Error: Number of cities cannot be negative" << endl;
        return 1;
    }
    
    vector<Point> points;
    points.reserve(N);
    
    for (int i = 0; i < N; i++) {
        double x, y;
        if (!(cin >> x >> y)) {
            cerr << "Error: Failed to read coordinates for city " << i << endl;
            return 1;
        }
        points.emplace_back(x, y, i);
    }
    
    // Solve
    vector<int> tour = solveTSP(points);
    
    // Basic validation (skip in release for speed)
    #ifndef NDEBUG
    if (tour.size() != static_cast<size_t>(N)) {
        cerr << "Error: Tour size mismatch. Expected " << N << ", got " << tour.size() << endl;
        return 1;
    }
    
    if (N > 0 && (tour.empty() || tour[0] != 0)) {
        cerr << "Error: Tour does not start at city 0" << endl;
        return 1;
    }
    
    vector<bool> seen(N, false);
    for (int city : tour) {
        if (city < 0 || city >= N) {
            cerr << "Error: Invalid city index " << city << endl;
            return 1;
        }
        if (seen[city]) {
