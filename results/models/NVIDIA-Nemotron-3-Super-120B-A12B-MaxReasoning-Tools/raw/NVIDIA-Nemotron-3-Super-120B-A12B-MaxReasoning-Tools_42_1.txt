; SPIR-V Geometry Shader
; Vulkan 1.0
; Input: InputPoints (1 vertex per invocation)
; Output: TriangleStrip, max 4 vertices
; Expands each point into a diamond (4 vertices) using point's color

OpCapability Geometry

OpExecutionMode %main InputPoints
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1

%ptr_output_v3float = OpTypePointer Output %v3float
%ptr_output_v4float = OpTypePointer Output %v4float

%float_h = OpConstant %f32 0.05

; Input interface: per-vertex arrays
%gl_PerVertex_in = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_1
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input
%vec3_array_type = OpTypeArray %v3float %uint_1
%ptr_in_arr_vec3 = OpTypePointer Input %vec3_array_type
%worldPos = OpVariable %ptr_in_arr_vec3 Input
OpDecorate %worldPos Location 0
%normal = OpVariable %ptr_in_arr_vec3 Input
OpDecorate %normal Location 1
%color = OpVariable %ptr_in_arr_vec3 Input
OpDecorate %color Location 2

; Output interface: per-emitted-vertex
%gl_PerVertex_out = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output
%ptr_out_vec3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_vec3 Output
OpDecorate %outColor Location 0

; Function: main
%void_func = OpTypeFunction %void
%main = OpFunction %void None %void_func
%entry = OpLabel

; Load input data for this point (index 0)
%ptr_gl_in_0 = OpAccessChain %ptr_in_arr_pv %gl_in %int_0
%ptr_gl_in_0_v4 = OpAccessChain %ptr_in_arr_pv %gl_in %int_0 %int_0
%gl_in_0_v4 = OpLoad %v4float %ptr_gl_in_0_v4
%ptr_worldPos_0 = OpAccessChain %ptr_in_arr_vec3 %worldPos %int_0
%worldPos_0 = OpLoad %v3float %ptr_worldPos_0
%ptr_color_0 = OpAccessChain %ptr_in_arr_vec3 %color %int_0
%color_0 = OpLoad %v3float %ptr_color_0

; Extract x, y from gl_in[0].gl_Position (we assume it's already in clip space or we just use x,y as-is for simplicity)
; Note: In a real shader, you might transform, but per task we use the point position directly.
%x = OpCompositeExtract %f32 %gl_in_0_v4 0
%y = OpCompositeExtract %f32 %gl_in_0_v4 1

; Compute diamond vertices
; Top: (x, y + h)
%y_plus_h = OpFAdd %f32 %y %float_h
%top_pos = OpCompositeConstruct %v4float %x %y_plus_h %float_0 %float_1
; Right: (x + h, y)
%x_plus_h = OpFAdd %f32 %x %float_h
%right_pos = OpCompositeConstruct %v4float %x_plus_h %y %float_0 %float_1
; Left: (x - h, y)
%x_minus_h = OpFSub %f32 %x %float_h
%left_pos = OpCompositeConstruct %v4float %x_minus_h %y %float_0 %float_1
; Bottom: (x, y - h)
%y_minus_h = OpFSub %f32 %y %float_h
%bottom_pos = OpCompositeConstruct %v4float %x %y_minus_h %float_0 %float_1

; Emit triangle strip: top, right, left, bottom

; Emit top vertex
%ptr_gl_out_v4 = OpAccessChain %ptr_out_pv %gl_out %int_0
OpStore %ptr_gl_out_v4 %top_pos
%ptr_outColor = OpAccessChain %ptr_out_vec3 %outColor %int_0
OpStore %ptr_outColor %color_0
OpEmitVertex

; Emit right vertex
OpStore %ptr_gl_out_v4 %right_pos
OpStore %ptr_outColor %color_0
OpEmitVertex

; Emit left vertex
OpStore %ptr_gl_out_v4 %left_pos
OpStore %ptr_outColor %color_0
OpEmitVertex

; Emit bottom vertex
OpStore %ptr_gl_out_v4 %bottom_pos
OpStore %ptr_outColor %color_0
OpEmitVertex

; End primitive (triangle strip of 4 vertices makes 2 triangles)
OpEndPrimitive

OpReturn
OpFunctionEnd
