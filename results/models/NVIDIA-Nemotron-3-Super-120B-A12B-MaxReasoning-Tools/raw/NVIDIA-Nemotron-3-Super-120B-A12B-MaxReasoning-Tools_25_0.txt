        private static bool IsOnCollisionCourse(CelestialBody asteroid, CelestialBody earth, double warningDays)
        {
            // Simple check: if asteroid is approaching Earth within warning time
            var relativePos = asteroid.Position - earth.Position;
            var relativeVel = asteroid.Velocity - earth.Velocity;
            
            // Time to closest approach (simplified)
            double tca = -Vector3.Dot(relativePos, relativeVel) / Vector3.Dot(relativeVel, relativeVel);
            double minDistance = Vector3.Length(relativePos + relativeVel * (float)tca);
            
            return tca > 0 && tca <= warningDays * DAY_IN_SECONDS && minDistance < 2 * AU;
        }

        private static InterceptionSolution SolveInterception(

            // If no solution found, use fallback

        private static double EstimateTimeToImpact(CelestialBody asteroid, CelestialBody earth, double warningDays)
        {
            var relativePos = asteroid.Position - earth.Position;
            var relativeVel = asteroid.Velocity - earth.Velocity;
            
            double velSq = Vector3.Dot(relativeVel, relativeVel);
            if (velSq < 1e-10) return warningDays * DAY_IN_SECONDS; // Avoid division by zero
            
            double tca = -Vector3.Dot(relativePos, relativeVel) / velSq;
            return Math.Max(0, tca);
        }

        private static TrajectoryAttempt TryInterceptTrajectory(
            CelestialBody asteroid,
            CelestialBody earth,
            double impactorMass,
            double launchTime,
            double maxInterceptTime,
            double deltaVBudget,
            List<CelestialBody> bodies,
            string de421Path)
        {
            var attempt = new TrajectoryAttempt();
            
            // Get states at launch time
            var scStateAtLaunch = PropagateState(earth, launchTime, bodies, de421Path);
            var asteroidStateAtLaunch = PropagateState(asteroid, launchTime, bodies, de421Path);
            
            // Simple Lambert-like solution for interception
            var asteroidStateAtImpact = PropagateState(asteroid, maxInterceptTime, bodies, de421Path);
            
            // Calculate required velocity change for interception
            var r0 = scStateAtLaunch.Position;
            var v0 = scStateAtLaunch.Velocity;
            var rf = asteroidStateAtImpact.Position;
            var tof = maxInterceptTime - launchTime;
            
            if (tof <= MIN_DT) return attempt; // Too short time of flight
            
            // Simplified two-impulse solution
            var vfNeeded = (rf - r0) / (float)tof; // Simplified
            
            // Launch burn: from Earth orbit to transfer trajectory
            var launchBurn = vfNeeded - v0;
            double launchDeltaV = Vector3.Length(launchBurn);
            
            // Impact burn: match asteroid velocity
            var asteroidVelocityAtImpact = PropagateState(asteroid, maxInterceptTime, bodies, de421Path).Velocity;
            var impactBurn = asteroidVelocityAtImpact - vfNeeded;
            double impactDeltaV = Vector3.Length(impactBurn);
            
            attempt.TotalDeltaV = launchDeltaV + impactDeltaV;
            attempt.Success = attempt.TotalDeltaV <= deltaVBudget;
            
            if (attempt.Success)
            {
                attempt.LaunchBurn = launchBurn;
                attempt.ImpactBurn = impactBurn;
                attempt.ImpactTime = maxInterceptTime;
            }
            
            return attempt;
        }

        private static CelestialBodyState PropagateState(
            CelestialBody body, 
            double timeSeconds,

        private static InterceptionSolution GenerateFallbackSolution(
            CelestialBody asteroid,
            CelestialBody earth,

            double warningDays,
            double t0UnixUtc)

            var solution = new InterceptionSolution();

            // Simple radial outward burn to deflect
            var earthToAsteroid = asteroid.Position - earth.Position;
            if (Vector3.Length(earthToAsteroid) > 1e-6)

                var direction = Vector3.Normalize(earthToAsteroid);
                // Use half the budget for impact burn
                double impactSpeed = deltaVBudget * 0.5 / impactorMass; // Simplified

            solution.Burns = new List<InterceptionBurn>

                new InterceptionBurn { TimeDays = 0, DeltaV = solution.LaunchBurn },
                new InterceptionBurn { TimeDays = solution.ImpactTimeDays, DeltaV = solution.ImpactBurn }
