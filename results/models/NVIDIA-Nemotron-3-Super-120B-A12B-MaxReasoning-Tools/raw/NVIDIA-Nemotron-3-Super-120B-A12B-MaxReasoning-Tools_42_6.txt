; SPIR-V Geometry Shader: Identity passthrough for triangle input
; Input: Triangles (3 vertices)
; Output: TriangleStrip (3 vertices emitted, 1 primitive)
; No transformation — just passthrough of gl_Position and color

; Magic number and version
OpCode 0x07230203
OpCapability Shader
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main InputTriangles
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 3
OpExecutionMode %main Invocations 1

; Source language (optional but good practice)
OpSource GLSL 450

; Type definitions
%void = OpTypeVoid
%bool = OpTypeBool
%i32 = OpTypeInt 32 1
%u32 = OpTypeInt 32 0
%f32 = OpTypeFloat 32
%v2float = OpTypeVector %f32 2
%v3float = OpTypeVector %f32 3
%v4float = OpTypeVector %f32 4

; ==== Input: gl_PerVertex (per-vertex array) ====
%gl_PerVertex_in = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position

; Array size = 3 for triangles
%uint_3 = OpConstant %u32 3
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_3
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

; ==== User-defined inputs (per-vertex arrays, location 0,1,2) ====
%ptr_input_worldPos = OpTypePointer Input %v3float
%worldPos = OpVariable %ptr_input_worldPos Input
OpDecorate %worldPos Location 0

%ptr_input_normal = OpTypePointer Input %v3float
%normal = OpVariable %ptr_input_normal Input
OpDecorate %normal Location 1

%ptr_input_color = OpTypePointer Input %v3float
%color = OpVariable %ptr_input_color Input
OpDecorate %color Location 2

; ==== Output: gl_PerVertex (single emitted vertex) ====
%gl_PerVertex_out = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

%ptr_output_outColor = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_output_outColor Output
OpDecorate %outColor Location 0

; ==== Constants for indexing ====
%int_0 = OpConstant %i32 0
%int_1 = OpConstant %i32 1
%int_2 = OpConstant %i32 2

; ==== Main function ====
%main = OpFunction %void None %void
%entry = OpLabel

; === Emit Vertex 0: i = 0 ===
; gl_in[0].gl_Position -> gl_out.gl_Position
%ptr_gl_in_0 = OpAccessChain %ptr_in_arr_pv %gl_in %int_0 %int_0
%gl_in_0_pos = OpLoad %v4float %ptr_gl_in_0
%ptr_gl_out_pos = OpAccessChain %ptr_out_pv %gl_out %int_0
OpStore %ptr_gl_out_pos %gl_in_0_pos

; color[0] -> outColor
%ptr_color_0 = OpAccessChain %ptr_input_color %color %int_0
%color_0 = OpLoad %v3float %ptr_color_0
%ptr_outColor = OpAccessChain %ptr_output_outColor %outColor %int_0
OpStore %ptr_outColor %color_0

OpEmitVertex

; === Emit Vertex 1: i = 1 ===
; gl_in[1].gl_Position -> gl_out.gl_Position
%ptr_gl_in_1 = OpAccessChain %ptr_in_arr_pv %gl_in %int_0 %int_1
%gl_in_1_pos = OpLoad %v4float %ptr_gl_in_1
%ptr_gl_out_pos = OpAccessChain %ptr_out_pv %gl_out %int_0
OpStore %ptr_gl_out_pos %gl_in_1_pos

; color[1] -> outColor
%ptr_color_1 = OpAccessChain %ptr_input_color %color %int_1
%color_1 = OpLoad %v3float %ptr_color_1
%ptr_outColor = OpAccessChain %ptr_output_outColor %outColor %int_0
OpStore %ptr_outColor %color_1

OpEmitVertex

; === Emit Vertex 2: i = 2 ===
; gl_in[2].gl_Position -> gl_out.gl_Position
%ptr_gl_in_2 = OpAccessChain %ptr_in_arr_pv %gl_in %int_0 %int_2
%gl_in_2_pos = OpLoad %v4float %ptr_gl_in_2
%ptr_gl_out_pos = OpAccessChain %ptr_out_pv %gl_out %int_0
OpStore %ptr_gl_out_pos %gl_in_2_pos

; color[2] -> outColor
%ptr_color_2 = OpAccessChain %ptr_input_color %color %int_2
%color_2 = OpLoad %v3float %ptr_color_2
%ptr_outColor = OpAccessChain %ptr_output_outColor %outColor %int_0
OpStore %ptr_outColor %color_2

OpEmitVertex

; End the triangle strip (1 primitive: 3 vertices)
OpEndPrimitive

OpReturn
OpFunctionEnd
