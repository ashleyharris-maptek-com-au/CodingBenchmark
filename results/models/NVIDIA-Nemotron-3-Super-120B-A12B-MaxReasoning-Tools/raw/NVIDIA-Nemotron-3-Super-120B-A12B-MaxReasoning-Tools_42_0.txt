; SPIR-V Geometry Shader
; Vulkan 1.0, InputPoints -> OutputTriangleStrip, 4 vertices emitted
; Expands each input point into a colored quad (triangle strip)

OpCapability Shader
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main InputPoints
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1

; Type definitions
%void = OpTypeVoid
%bool = OpTypeBool
%i32 = OpTypeInt 32 1
%f32 = OpTypeFloat 32
%v2float = OpTypeVector %f32 2
%v3float = OpTypeVector %f32 3
%v4float = OpTypeVector %f32 4

; gl_PerVertex struct (input and output)
%gl_PerVertex_in = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position

%gl_PerVertex_out = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

; Array types for input (size = 1 for InputPoints)
%arr_pv_in = OpTypeArray %gl_PerVertex_in %i32 1
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3float = OpTypeArray %v3float %i32 1
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3float
%worldPos = OpVariable %ptr_in_arr_v3 Input
OpDecorate %worldPos Location 0
%normal = OpVariable %ptr_in_arr_v3 Input
OpDecorate %normal Location 1
%color = OpVariable %ptr_in_arr_v3 Input
OpDecorate %color Location 2

; Output: per-vertex gl_Position and outColor
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output
%outColor = OpVariable %ptr_in_arr_v3 Output  ; Reuse pointer type for vec3 output
OpDecorate %outColor Location 0

; Constants
%int_0 = OpConstant %i32 0
%int_1 = OpConstant %i32 1
%uint_0 = OpConstant %i32 0
%f_0 = OpConstant %f32 0.0
%f_1 = OpConstant %f32 1.0
%f_h = OpConstant %f32 0.04  ; half-size

; Main function
%void_functype = OpTypeFunction %void
%main = OpFunction %void %void_functype %void
%entry = OpLabel

; --- Load input data (index 0 only, since array size = 1) ---
; gl_in[0]
%ptr_gl_in_0 = OpAccessChain %ptr_in_arr_pv %gl_in %int_0
%gl_in_0 = OpLoad %gl_PerVertex_in %ptr_gl_in_0
%ptr_gl_in_0_pos = OpAccessChain %v4float %gl_in_0 %int_0
%gl_in_pos = OpLoad %v4float %ptr_gl_in_0_pos

; worldPos[0] (unused in position calc, but loaded if needed; we use gl_in_pos)
%ptr_worldPos_0 = OpAccessChain %ptr_in_arr_v3 %worldPos %int_0
%worldPos_0 = OpLoad %v3float %ptr_worldPos_0

; color[0]
%ptr_color_0 = OpAccessChain %ptr_in_arr_v3 %color %int_0
%color_0 = OpLoad %v3float %ptr_color_0

; Extract x, y from gl_in[0].gl_Position (assume z and w are valid; we override z=0, w=1)
%ptr_x = OpAccessChain %f32 %gl_in_pos %int_0
%x = OpLoad %f32 %ptr_x
%ptr_y = OpAccessChain %f32 %gl_in_pos %int_1
%y = OpLoad %f32 %ptr_y

; Compute quad vertices: (x±h, y±h, 0, 1)
; Vertex 0: (x-h, y-h)
%x_h = OpFSub %f32 %x %f_h
%y_h = OpFSub %f32 %y %f_h
%v0 = OpCompositeConstruct %v4float %x_h %y_h %f_0 %f_1

; Vertex 1: (x+h, y-h)
%x_plus_h = OpFAdd %f32 %x %f_h
%v1 = OpCompositeConstruct %v4float %x_plus_h %y_h %f_0 %f_1

; Vertex 2: (x-h, y+h)
%y_plus_h = OpFAdd %f32 %y %f_h
%v2 = OpCompositeConstruct %v4float %x_h %y_plus_h %f_0 %f_1

; Vertex 3: (x+h, y+h)
%v3 = OpCompositeConstruct %v4float %x_plus_h %y_plus_h %f_0 %f_1

; --- Emit vertices as triangle strip ---
; Emit vertex 0
%ptr_gl_out_0 = OpAccessChain %gl_PerVertex_out %gl_out %int_0
%ptr_gl_out_0_pos = OpAccessChain %v4float %ptr_gl_out_0 %int_0
OpStore %ptr_gl_out_0_pos %v0
%ptr_outColor_0 = OpAccessChain %v3float %outColor %int_0
OpStore %ptr_outColor_0 %color_0
OpEmitVertex

; Emit vertex 1
%ptr_gl_out_1 = OpAccessChain %gl_PerVertex_out %gl_out %int_0  ; Output is per-vertex, re-use base pointer
%ptr_gl_out_1_pos = OpAccessChain %v4float %ptr_gl_out_1 %int_0
OpStore %ptr_gl_out_1_pos %v1
%ptr_outColor_1 = OpAccessChain %v3float %outColor %int_0
OpStore %ptr_outColor_1 %color_0
OpEmitVertex

; Emit vertex 2
%ptr_gl_out_2 = OpAccessChain %gl_PerVertex_out %gl_out %int_0
%ptr_gl_out_2_pos = OpAccessChain %v4float %ptr_gl_out_2 %int_0
OpStore %ptr_gl_out_2_pos %v2
%ptr_outColor_2 = OpAccessChain %v3float %outColor %int_0
OpStore %ptr_outColor_2 %color_0
OpEmitVertex

; Emit vertex 3
%ptr_gl_out_3 = OpAccessChain %gl_PerVertex_out %gl_out %int_0
%ptr_gl_out_3_pos = OpAccessChain %v4float %ptr_gl_out_3 %int_0
OpStore %ptr_gl_out_3_pos %v3
%ptr_outColor_3 = OpAccessChain %v3float %outColor %int_0
OpStore %ptr_outColor_3 %color_0
OpEmitVertex

; End primitive (triangle strip of 4 vertices = 2 triangles)
OpEndPrimitive

OpReturn
OpFunctionEnd
