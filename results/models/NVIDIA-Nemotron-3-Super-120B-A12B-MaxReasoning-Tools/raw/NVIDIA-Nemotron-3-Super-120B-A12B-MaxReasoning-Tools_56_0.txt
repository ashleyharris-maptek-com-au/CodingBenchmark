```javascript
const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const depot = data.depot;
const truckCount = data.truckCount;
const packages = data.packages;

function dist(a, b) {
  return Math.hypot(a[0] - b[0], a[1] - b[1]);
}

// Create array of packages with id and coordinates
const packageList = packages.map(p => ({ id: p.id, x: p.x, y: p.y }));
const n = packageList.length;

// Handle edge cases
if (n === 0) {
  console.log(JSON.stringify({ routes: Array.from({ length: truckCount }, () => []) }));
  process.exit(0);
}

if (truckCount <= 0) {
  console.log(JSON.stringify({ routes: [] }));
  process.exit(0);
}

// If we have more trucks than packages, assign one package per route (if any)
if (truckCount >= n) {
  const routes = packageList.map(p => [p.id]);
  // Fill remaining routes with empty arrays
  while (routes.length < truckCount) {
    routes.push([]);
  }
  console.log(JSON.stringify({ routes }));
  process.exit(0);
}

// Greedy clustering by nearest depot, then nearest neighbor routing
const clusters = Array.from({ length: truckCount }, () => []);

// Assign each package to the nearest cluster centroid (initially depot)
for (const p of packageList) {
  let bestIdx = 0;
  let bestDist = Infinity;
  for (let i = 0; i < truckCount; i++) {
    // Use depot as reference for initial assignment
    const d = dist([p.x, p.y], depot);
    if (d < bestDist) {
      bestDist = d;
      bestIdx = i;
    }
  }
  clusters[bestIdx].push(p);
}

// Rebalance: move packages from overloaded clusters to underloaded ones
let changed;
do {
  changed = false;
  const sizes = clusters.map(c => c.length);
  const avg = n / truckCount;
  const overloaded = [];
  const underloaded = [];

  for (let i = 0; i < truckCount; i++) {
    if (sizes[i] > avg + 1) overloaded.push(i);
    if (sizes[i] < avg - 1) underloaded.push(i);
  }

  // Sort overloaded by farthest package from depot (to move outliers)
  const overloadedWithFar = overloaded
    .map(i => ({
      idx: i,
      farthest: clusters[i].reduce((max, p) => 
        (dist([p.x, p.y], depot) > dist([max.x, max.y], depot) ? p : max), 
      clusters[i][0] || {x:0,y:0}
    }))
    .sort((a, b) => dist([b.farthest.x, b.farthest.y], depot) - dist([a.farthest.x, a.farthest.y], depot));

  // Sort underloaded by nearest to depot
  const underloadedWithNear = underloaded
    .map(i => ({ idx: i, nearest: clusters[i].reduce((min, p) => 
      (dist([p.x, p.y], depot) < dist([min.x, min.y], depot) ? p : min), 
    clusters[i][0] || {x:depot[0], y:depot[1]}) }))
    .sort((a, b) => dist([a.nearest.x, a.nearest.y], depot) - dist([b.nearest.x, b.nearest.y], depot));

  // Try to move one package from each overloaded to each underloaded
  for (const ov of overloadedWithFar) {
    if (underloadedWithNear.length === 0) break;
    const pkgToMove = ov.farthest;
    const fromIdx = ov.idx;
    const toIdx = underloadedWithNear[0].idx;

    // Remove from source
    clusters[fromIdx] = clusters[fromIdx].filter(p => p.id !== pkgToMove.id);
    // Add to target
    clusters[toIdx].push(pkgToMove);
    changed = true;

    // Recompute sizes and break inner loop to restart
    break;
  }
} while (changed);

// For each cluster, compute route using nearest neighbor from depot
const routes = clusters.map(cluster => {
  if (cluster.length === 0) return [];

  let current = { x: depot[0], y: depot[1] };
  const visited = new Set();
  const route = [];

  while (route.length < cluster.length) {
    let nearest = null;
    let minDist = Infinity;

    for (const p of cluster) {
      if (visited.has(p.id)) continue;
      const d = dist([current.x, current.y], [p.x, p.y]);
      if (d < minDist) {
        minDist = d;
        nearest = p;
      }
    }

    if (nearest === null) break; // should not happen

    visited.add(nearest.id);
    route.push(nearest.id);
    current = { x: nearest.x, y: nearest.y };
  }

  return route;
});

// Ensure every package is assigned exactly once
const assigned = new Set();
routes.forEach(route => route.forEach(id => assigned.add(id)));
const unassigned = packageList.filter(p => !assigned.has(p.id)).map(p => p.id);

// Distribute unassigned packages to shortest routes
for (const pid of unassigned) {
  let bestIdx = 0;
  let bestLen = routes[0].length;
  for (let i = 1; i < truckCount; i++) {
    if (routes[i].length < bestLen) {
      bestLen = routes[i].length;
      bestIdx = i;
    }
  }
  routes[bestIdx].push(pid);
}

// Final validation: flatten and check
const flat = routes.flat();
if (new Set(flat).size !== n || flat.length !== n) {
  // Fallback: assign each package to its own route, fill rest empty
  const fallbackRoutes = packageList.map(p => [p.id]);
  while (fallbackRoutes.length < truckCount) fallbackRoutes.push([]);
  console.log(JSON.stringify({ routes: fallbackRoutes }));
} else {
  console.log(JSON.stringify({ routes }));
}
```
