```ts
declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { width, height, grid, ignitions, towns, cutBudget } = data;

// Validate and parse grid
const gridArray: string[][] = grid.map((row: string) => row.split(''));

// Directions for 4-way spread: up, down, left, right
const directions: [number, number][] = [[0, -1], [0, 1], [-1, 0], [1, 0]];

// Helper: Check if a cell is within bounds
const inBounds = (x: number, y: number): boolean =>
  x >= 0 && x < width && y >= 0 && y < height;

// Helper: Check if a cell is burnable forest ('.') and not water/rock ('~')
const isForest = (x: number, y: number): boolean =>
  inBounds(x, y) && gridArray[y][x] === '.';

// Helper: Simulate fire spread given a set of cut cells and return burned towns
const simulateFire = (cuts: Set<string>): { burned: boolean[]; burnedCount: number } => {
  const burned: boolean[] = new Array(towns.length).fill(false);
  const visited: boolean[][] = Array.from({ length: height }, () => Array(width).fill(false));
  const queue: [number, number][] = [];

  // Initialize queue with ignitions that are on forest and not cut
  for (const [x, y] of ignitions) {
    if (!inBounds(x, y)) continue;
    const key = `${x},${y}`;
    if (isForest(x, y) && !cuts.has(key)) {
      visited[y][x] = true;
      queue.push([x, y]);
    }
  }

  // BFS for fire spread
  while (queue.length > 0) {
    const [x, y] = queue.shift()!;
    for (const [dx, dy] of directions) {
      const nx = x + dx, ny = y + dy;
      if (!inBounds(nx, ny)) continue;
      const nkey = `${nx},${ny}`;
      if (visited[ny][nx]) continue;
      if (!isForest(nx, ny)) continue; // Only spread through forest
      if (cuts.has(nkey)) continue; // Blocked by cut
      visited[ny][nx] = true;
      queue.push([nx, ny]);
    }
  }

  // Check which towns are burned (i.e., their cell was reached by fire)
  for (let i = 0; i < towns.length; i++) {
    const { x, y } = towns[i];
    if (inBounds(x, y) && visited[y][x]) {
      burned[i] = true;
    }
  }

  const burnedCount = burned.filter(b => b).length;
  return { burned, burnedCount };
};

// Main greedy algorithm: iteratively select the cut that saves the most town value
const cuts = new Set<string>();

for (let step = 0; step < cutBudget; step++) {
  let bestGain = -1;
  let bestCut: [number, number] | null = null;

  // Try every possible forest cell that is not already cut
  for (let y = 0; y < height; y++) {
    for (let x = 0; x < width; x++) {
      if (!isForest(x, y)) continue;
      const key = `${x},${y}`;
      if (cuts.has(key)) continue;

      // Simulate adding this cut
      cuts.add(key);
      const { burned: burnedWithCut } = simulateFire(cuts);
      cuts.delete(key); // undo

      // Calculate gain: number of towns saved by this cut (compared to current cuts)
      const { burned: burnedCurrent } = simulateFire(cuts);
      let gain = 0;
      for (let i = 0; i < towns.length; i++) {
        if (burnedCurrent[i] && !burnedWithCut[i]) {
          gain += towns[i].value;
        }
      }

      if (gain > bestGain) {
        bestGain = gain;
        bestCut = [x, y];
      }
    }
  }

  // If no beneficial cut found, break early
  if (bestCut === null || bestGain <= 0) break;

  // Apply the best cut
  cuts.add(`${bestCut[0]},${bestCut[1]}`);
}

// Output the firebreaks
const firebreaks: [number, number][] = Array.from(cuts).map(key => {
  const [x, y] = key.split(',').map(Number);
  return [x, y];
});

console.log(JSON.stringify({ firebreaks }));
```
