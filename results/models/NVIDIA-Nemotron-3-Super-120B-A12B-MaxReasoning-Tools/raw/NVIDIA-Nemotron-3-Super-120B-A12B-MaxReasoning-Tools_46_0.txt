```glsl
#version 450

layout(local_size_x = 256) in;

// Input buffer: read-only
layout(set = 0, binding = 0) readonly buffer InputBodies {
    vec4 inData[];   // [pos.x, pos.y, pos.z, mass], [vel.x, vel.y, vel.z, 0.0] per body
};

// Output buffer: read-write
layout(set = 0, binding = 1) buffer OutputBodies {
    vec4 outData[];  // Same layout as inData
};

// Uniform parameters
layout(set = 0, binding = 2) uniform Params {
    uvec4 counts;    // counts.x = N (number of bodies)
    vec4 physics;    // physics.x = dt, physics.y = softening, physics.z = G
};

void main() {
    uint N = counts.x;
    float dt = physics.x;
    float softening = physics.y;
    float G = physics.z;

    uint i = gl_GlobalInvocationID.x;
    if (i >= N) {
        return;
    }

    // Read body i: position and mass
    vec4 posMass_i = inData[i * 2];
    vec3 pos_i = posMass_i.xyz;
    float mass_i = posMass_i.w;  // mass is stored in w component

    // Read body i: velocity
    vec4 vel_i = inData[i * 2 + 1];
    vec3 vel = vel_i.xyz;

    // Compute acceleration from all other bodies
    vec3 acc = vec3(0.0);
    for (uint j = 0; j < N; ++j) {
        if (j == i) continue;  // Skip self-interaction

        vec4 posMass_j = inData[j * 2];
        vec3 pos_j = posMass_j.xyz;
        float mass_j = posMass_j.w;

        vec3 r = pos_j - pos_i;
        float r2 = dot(r, r);
        float r2_soft = r2 + softening * softening;
        float inv_r3 = inversesqrt(r2_soft * r2_soft * r2_soft);  // 1 / (r^2 + eps^2)^(3/2)
        acc += G * mass_j * r * inv_r3;
    }

    // Leapfrog/Euler integration: v_new = v + a * dt
    vec3 vel_new = vel + acc * dt;

    // Position update: x_new = x + v_new * dt
    vec3 pos_new = pos_i + vel_new * dt;

    // Write updated position and mass to output
    outData[i * 2] = vec4(pos_new, mass_i);

    // Write updated velocity to output (w = 0.0 as per format)
    outData[i * 2 + 1] = vec4(vel_new, 0.0);
}
```
