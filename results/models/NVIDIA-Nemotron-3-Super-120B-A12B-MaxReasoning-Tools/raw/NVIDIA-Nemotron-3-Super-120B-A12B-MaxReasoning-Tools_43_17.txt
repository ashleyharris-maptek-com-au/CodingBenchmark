[[vk::binding(0, 0)]]
cbuffer UBO : register(b0) {
    float4x4 model;
    float4x4 view;
    float4x4 projection;
    float4 lightPos;
    float4 cameraPos;
    float4 params;
};

struct PSInput {
    [[vk::location(0)]] float3 worldPos : TEXCOORD0;
    [[vk::location(1)]] float3 normal   : TEXCOORD1;
    [[vk::location(2)]] float2 uv       : TEXCOORD2;
    [[vk::location(3)]] float3 tangent  : TEXCOORD3;
    [[vk::location(4)]] float3 color    : TEXCOORD4;
};

float4 main(PSInput input) : SV_Target0 {
    // Normalize interpolated normal
    float3 N = normalize(input.normal);

    // Lambertian shading
    float3 L = normalize(lightPos.xyz - input.worldPos);
    float diffuse = max(0.0, dot(N, L));
    float shadingFactor = 0.4 + 0.6 * diffuse;

    // Procedural brick pattern: 8x16 UV scale, staggered rows
    float2 brickUV = input.uv * float2(8.0, 16.0);

    // Stagger every other row: offset X by 0.5 on odd rows
    float row = floor(brickUV.y);
    float2 offsetUV = brickUV;
    offsetUV.x += floor(row * 0.5) * 0.5; // half-brick offset per two rows? Actually: offset by 0.5 on odd rows

    // Correct: offset by 0.5 on odd integer rows
    offsetUV.x += (fmod(floor(brickUV.y), 2.0) * 0.5);

    // Brick size: 1.0 in U, 0.5 in V (since 8x16 over 1x1 UV => brick is 1/8 wide, 1/16 high? No)
    // Wait: UV scaled by 8x16 => one brick is 1/8 in U, 1/16 in V? But we want 8 bricks across, 16 tall => each brick 1/8 x 1/16 in UV
    // But mortar thickness given in UV? Likely in UV space after scaling.

    // Actually: after scaling UV by 8x16, the pattern repeats every 1 brick in U, 1 brick in V? No.
    // We want 8 bricks horizontally => each brick width in UV: 1/8
    // 16 bricks vertically => each brick height in UV: 1/16
    // But mortar thickness: 0.05 in X, 0.1 in Y -> likely in UV space after scaling? Or in texture space?

    // Re-interpret: "8x16 UV scale" means we multiply UV by 8 and 16 so that 8 bricks fit in U, 16 in V over [0,1] UV.
    // So brick width in UV space: 1/8, height: 1/16.
    // Mortar thickness: 0.05 in X, 0.1 in Y -> likely as fraction of brick? Or absolute in UV?

    // Standard approach: after scaling, check if fragment is in mortar.
    // Let brick width = 1.0, height = 0.5 in scaled UV (so that 8 bricks fit: 8*1.0 = 8? No.)

    // Correction: If we want 8 bricks across [0,1] UV, then each brick is 1/8 wide in UV.
    // So after scaling UV by 8, each brick is 1 unit wide in scaled space.
    // Similarly, scale V by 16 => each brick is 1/16 * 16 = 1 unit high in scaled V? No.
    // Actually: scaledUV = uv * float2(8,16) => then one brick spans 1.0 in U and 1.0 in V? Only if we intend 8x16 bricks to occupy [0,8]x[0,16] in scaled space, but we want to repeat.

    // Standard brick shader: scale UV, then use fract to get position within brick, mortar if near edge.

    // Let's define: after scaling UV by 8 and 16, the brick pattern repeats every 1.0 in U and 0.5 in V? Because staggered rows of height 0.5?

    // Actually, common approach: 
    //   float2 brickUV = input.uv * float2(8.0, 16.0);
    //   float2 brick = fract(brickUV);
    //   float2 brickOffset = brickUV * 0.5; // for stagger? No.

    // Better: 
    //   float2 brickUV = input.uv * float2(8.0, 16.0);
    //   float2 brick = fract(brickUV);
    //   // Stagger: shift odd rows by 0.5
    //   float2 staggeredUV = brickUV;
    //   staggeredUV.x -= 0.5 * floor(brickUV.y); // not quite

    // Correct stagger: 
    //   float2 staggeredUV = brickUV;
    //   staggeredUV.x -= 0.5 * floor(brickUV.y); // but then mod 2?

    // Actually: 
    //   float row = floor(brickUV.y);
    //   float2 offsetUV = brickUV;
    //   offsetUV.x += 0.5 * (fmod(row, 2.0)); // if row odd, add 0.5

    // Then use fract on offsetUV to get position within brick.

    // Let's do:
    float2 offsetUV = brickUV;
    offsetUV.x += 0.5 * floor(brickUV.y * 0.5); // No.

    // Standard: 
    //   offsetUV.x += 0.5 * (floor(brickUV.y) % 2);
    // In HLSL: use fmod

    float2 staggeredUV = brickUV;
    staggeredUV.x += 0.5 * fmod(floor(brickUV.y), 2.0);

    // Now, brick is 1x1 in staggeredUV space? We want mortar thickness as fraction of brick.
    // Mortar thickness: 0.05 in X, 0.1 in Y -> interpret as fraction of brick size.
    // So if brick is 1x1, mortar is 0.05 wide, 0.1 high.

    float2 brick = fract(staggeredUV);
    float2 mortar = step(brick, float2(0.05, 0.1)); // 1 if in mortar (left/bottom edge), but we want mortar as border.

    // Actually, mortar is where u < thickness_u OR v < thickness_v? But also need to check right/top? 
    // Standard: mortar if fract < thickness OR (1 - fract) < thickness? But usually only left and bottom for simplicity, or use smoothstep.

    // Simpler: mortar if brick.x < 0.05 || brick.y < 0.1
    bool isMortar = (brick.x < 0.05) || (brick.y < 0.1);

    // Brick color and mortar color
    float3 brickColor = float3(0.7, 0.2, 0.1);
    float3 mortarColor = float3(0.7, 0.7, 0.7);
    float3 baseColor = lerp(brickColor, mortarColor, isMortar ? 1.0 : 0.0); // if mortar, use mortarColor

    // Apply shading
    float3 shadedColor = baseColor * shadingFactor;

    return float4(shadedColor, 1.0);
}
