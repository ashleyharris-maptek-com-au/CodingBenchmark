[[vk::binding(0, 0)]]
cbuffer UBO : register(b0) {
    float4x4 model;
    float4x4 view;
    float4x4 projection;
    float4 lightPos;
    float4 cameraPos;
    float4 params;
};

float4 main(PSInput input) : SV_Target0 {
    float3 N = normalize(input.normal);
    float3 L = normalize(lightPos.xyz - input.worldPos);
    float diffuse = saturate(dot(N, L));
    float lambert = 0.5 + 0.5 * diffuse;

    float2 uvScaled = input.uv * 8.0;
    float checker = floor(uvScaled.x) + floor(uvScaled.y);
    float pattern = fmod(checker, 2.0);
    float checkerColor = lerp(0.2, 0.9, pattern);

    float3 baseColor = input.color * checkerColor;
    float3 litColor = baseColor * lambert;

    return float4(litColor, 1.0);
}
