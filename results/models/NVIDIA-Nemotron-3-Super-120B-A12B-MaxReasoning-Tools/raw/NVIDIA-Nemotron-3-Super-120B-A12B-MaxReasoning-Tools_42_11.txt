; SPIR-V Geometry Shader
; Vulkan 1.0, InputLinesAdjacency (4 vertices), OutputTriangleStrip, 8 vertices
; Dashed thick curve: draw 1st and 3rd thirds of segment [gl_in[1], gl_in[2]]
; Skip middle third (gap). Each drawn part: 4-vertex triangle strip quad.
; Color interpolated from color[1] and color[2]
; Thickness = 0.02, using adjacency for smooth tangents

; Capabilities
OpCapability Shader
OpCapability Geometry
OpCapability Matrix

; Memory model
OpMemoryModel Logical GLSL450

; Entry point and execution modes
OpEntryPoint Geometry %main "main" %gl_in %worldPos_in %normal_in %color_in %gl_out %outColor
OpExecutionMode %main InputLinesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 8
OpExecutionMode %main Invocations 1

; Types
%void = OpTypeVoid
%bool = OpTypeBool
%float = OpTypeFloat 32
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%m4float = OpTypeMatrix %float 4

; Constants
%uint_1 = OpConstant %uint 1
%uint_2 = OpConstant %uint 2
%uint_3 = OpConstant %uint 3
%uint_4 = OpConstant %uint 4
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3
%float_0 = OpConstant %float 0.0
%float_1 = OpConstant %float 1.0
%float_2 = OpConstant %float 2.0
%float_0_02 = OpConstant %float 0.02
%float_1_0_div_3 = OpConstant %float 0.33333334
%float_2_0_div_3 = OpConstant %float 0.6666667

; Input per-vertex block (gl_PerVertex)
%gl_PerVertex_in = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position

; Input arrays (size 4 for InputLinesAdjacency)
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

; User inputs: arrays of size 4
%ptr_in_vec3 = OpTypePointer Input %v3float
%arr_worldPos_in = OpTypeArray %v3float %uint_4
%worldPos_in = OpVariable (OpTypePointer Input %arr_worldPos_in) Input
%arr_normal_in = OpTypeArray %v3float %uint_4
%normal_in = OpVariable (OpTypePointer Input %arr_normal_in) Input
%arr_color_in = OpTypeArray %v3float %uint_4
%color_in = OpVariable (OpTypePointer Input %arr_color_in) Input

; Decorate user inputs with Location
OpDecorate %worldPos_in Location 0
OpDecorate %normal_in Location 1
OpDecorate %color_in Location 2

; Output per-vertex block (gl_PerVertex)
%gl_PerVertex_out = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

; User output
%ptr_out_vec3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_vec3 Output
OpDecorate %outColor Location 0

; Uniform Buffer Object (set=0, binding=0)
%ptr_uniform = OpTypePointer UniformConstant %v4float
%ubo_arr = OpTypeArray %v4float %uint_3
%ptr_ubo_arr = OpTypePointer UniformConstant %ubo_arr
%ubo = OpVariable %ptr_ubo_arr Uniform
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0

; Helper function: cross product (%v3float x %v3float -> %v3float)
%type_cross_func = OpTypeFunction %v3float (%v3float %v3float)
%ptr_params_cross = OpTypePointer Function %v3float
%ptr_params_cross_2 = OpTypePointer Function %v3float
%func_cross = OpFunction %v3float None %type_cross_func
%param_a_cross = OpFunctionParameter %v3float None
%param_b_cross = OpFunctionParameter %v3float None
OpLabel %label_cross_entry
%a_cross = OpVariable %ptr_params_cross Function
%b_cross = OpVariable %ptr_params_cross_2 Function
OpStore %a_cross %param_a_cross
OpStore %b_cross %param_b_cross
%a_val = OpLoad %v3float %a_cross
%b_val = OpLoad %v3float %b_cross
%ax = OpCompositeExtract %float %a_val 0
%ay = OpCompositeExtract %float %a_val 1
%az = OpCompositeExtract %float %a_val 2
%bx = OpCompositeExtract %float %b_val 0
%by = OpCompositeExtract %float %b_val 1
%bz = OpCompositeExtract %float %b_val 2
%cx = OpFMul %float %ay %bz
%cx = OpFSub %float %cx OpFMul %float %az %by
%cy = OpFMul %float %az %bx
%cy = OpFSub %float %cy OpFMul %float %ax %bz
%cz = OpFMul %float %ax %by
%cz = OpFSub %float %cz OpFMul %float %ay %bx
%c_vec = OpCompositeConstruct %v3float %cx %cy %cz
OpReturnValue %c_vec
OpFunctionEnd

; Helper function: normalize (%v3float -> %v3float)
%type_norm_func = OpTypeFunction %v3float (%v3float)
%ptr_param_norm = OpTypePointer Function %v3float
%func_norm = OpFunction %v3float None %type_norm_func
%param_norm = OpFunctionParameter %v3float None
OpLabel %label_norm_entry
%v_norm = OpVariable %ptr_param_norm Function
OpStore %v_norm %param_norm
%v_val = OpLoad %v3float %v_norm
%xx = OpCompositeExtract %float %v_val 0
%yy = OpCompositeExtract %float %v_val 1
%zz = OpCompositeExtract %float %v_val 2
%dp = OpFMA %float %xx %xx OpFMA %float %yy %yy OpFMul %float %zz %zz
%rsq = OpInverseSqrt %float %dp
%xn = OpFMul %float %xx %rsq
%yn = OpFMul %float %yy %rsq
%zn = OpFMul %float %zz %rsq
%vn = OpCompositeConstruct %v3float %xn %yn %zn
OpReturnValue %vn
OpFunctionEnd

; Main function
%type_main = OpTypeFunction %void
%func_main = OpFunction %void None %type_main
OpLabel %label_main

;--- Load input data for indices 1 and 2 (inner segment) ---
; gl_in[1].gl_Position
%ptr_gl_in_1 = OpAccessChain %ptr_in_arr_pv %gl_in %int_1
%ptr_gl_in_1_pv = OpAccessChain %v4float %ptr_gl_in_1 %int_0
%p1 = OpLoad %v4float %ptr_gl_in_1_pv
; gl_in[2].gl_Position
%ptr_gl_in_2 = OpAccessChain %ptr_in_arr_pv %gl_in %int_2
%ptr_gl_in_2_pv = OpAccessChain %v4float %ptr_gl_in_2 %int_0
%p2 = OpLoad %v4float %ptr_gl_in_2_pv

; worldPos[1], worldPos[2]
%ptr_wp_1 = OpAccessChain %ptr_in_vec3 %worldPos_in %int_1
%wp1 = OpLoad %v3float %ptr_wp_1
%ptr_wp_2 = OpAccessChain %ptr_in_vec3 %worldPos_in %int_2
%wp2 = OpLoad %v3float %ptr_wp_2

; normal[1], normal[2] (for tangent approximation)
%ptr_n_1 = OpAccessChain %ptr_in_vec3 %normal_in %int_1
%n1 = OpLoad %v3float %ptr_n_1
%ptr_n_2 = OpAccessChain %ptr_in_vec3 %normal_in %int_2
%n2 = OpLoad %v3float %ptr_n_2

; color[1], color[2]
%ptr_c_1 = OpAccessChain %ptr_in_vec3 %color_in %int_1
%c1 = OpLoad %v3float %ptr_c_1
%ptr_c_2 = OpAccessChain %ptr_in_vec3 %color_in %int_2
%c2 = OpLoad %v3float %ptr_c_2

;--- Compute segment vector and tangents for smooth perpendicular ---
; seg = p2 - p1 (xyz only)
%p1_xyz = OpCompositeExtract %v3float %p1 0 1 2
%p2_xyz = OpCompositeExtract %v3float %p2 0 1 2
%seg = OpFSub %v3float %p2_xyz %p1_xyz

; Tangents: use normals as approximation (or average)
%T1 = %n1
%T2 = %n2
; Average tangent for smoothness
%T_avg = OpFAdd %v3float %T1 %T2
%T_avg = OpFMul %v3float %T_avg %float_0_5
; Normalize average tangent
%T_avg_norm = OpCall %v3float %func_norm %T_avg

; Perpendicular in view plane: assume camera looks along -Z, so use ( -T_y, T_x, 0 )
; For true view-aligned, would need camera, but approximate with world Y-up fallback
; Use arbitrary perp: if T_avg not near Z, use ( -T_z, 0, T_x ) else (0, -T_z, T_y)
; Simpler: use world Y (0,1,0) x T_avg to get perp in XZ plane
%world_up = OpConstant %v3float 0.0 1.0 0.0
%perp = OpCall %v3float %func_cross %world_up %T_avg_norm
; If perp is zero (T_avg parallel to Y), fallback to X axis cross
%perp_x = OpCompositeExtract %float %perp 0
%perp_y = OpCompositeExtract %float %perp 1
%perp_z = OpCompositeExtract %float %perp 2
%len_sq = OpFMA %float %perp_x %perp_x OpFMA %float %perp_y %perp_y OpFMul %float %perp_z %perp_z
%is_zero = OpFOrdLessEqual %float %len_sq %float_0_000001
%fallback_perp = OpConstant %v3float 1.0 0.0 0.0  ; X axis
%perp_sel = OpSelect %bool %is_zero %fallback_perp %perp
; Normalize perp
%perp_norm = OpCall %v3float %func_norm %perp_sel

; Scale by thickness/2
%thick_half = OpFMul %float %float_0_02 %float_0_5
%offset = OpFMul %v3float %perp_norm %thick_half

;--- Compute points for dashed segment ---
; Only draw t in [0, 1/3] and [2/3, 1]
; Precompute: p1, p2, offset
; For each part: emit 4 vertices as triangle strip: 
;   v0 = p1 + offset * t0
;   v1 = p1 - offset * t0
;   v2 = p2 + offset * t1
;   v3 = p2 - offset * t1
; Where t0, t1 are start/end of sub-segment

; First part: t in [0, 1/3]
%t0_1 = OpConstant %float 0.0
%t1_1 = OpConstant %float 0.33333334
%p_a1_1 = OpFMul %v3float %seg %t0_1
%p_a1_1 = OpFAdd %v3float %p1_xyz %p_a1_1  ; p1
%p_b1_1 = OpFMul %v3float %seg %t1_1
%p_b1_1 = OpFAdd %v3float %p1_xyz %p_b1_1  ; p1 + 1/3*seg
%offset_scaled1 = OpFMul %v3float %offset %t0_1  ; zero
%offset_scaled2 = OpFMul %v3float %offset %t1_1
; Vertices for first quad:
;   v0 = p_a1_1 + offset
;   v1 = p_a1_1 - offset
;   v2 = p_b1_1 + offset_scaled2
;   v3 = p_b1_1 - offset_scaled2
%v0_1 = OpFAdd %v3float %p_a1_1 %offset
%v1_1 = OpFSub %v3float %p_a1_1 %offset
%v2_1 = OpFAdd %v3float %p_b1_1 %offset_scaled2
%v3_1 = OpFSub %v3float %p_b1_1 %offset_scaled2
; Convert to v4 (w=1.0)
%v0_1_4 = OpInsertComposite %v4float %v0_1 %float_1 3
%v1_1_4 = OpInsertComposite %v4float %v1_1 %float_1 3
%v2_1_4 = OpInsertComposite %v4float %v2_1 %float_1 3
%v3_1_4 = OpInsertComposite %v4float %v3_1 %float_1 3
; Color: interpolate at t0_1 and t1_1 (same for whole quad? No, per vertex)
; Actually, color varies along segment. For quad, we need color per vertex.
; But task says: "Color from color[1] and color[2], interpolated."
; For simplicity, use constant color per quad? Or interpolate per vertex.
; We'll compute color at each vertex position.
; For v0_1, v1_1: at t=0 -> c1
; For v2_1, v3_1: at t=1/3 -> mix(c1,c2,1/3)
%c_at_t0 = %c1
%c_at_t1_1 = OpFMul %v3float %c2 %float_0_33333334
%c_at_t1_1 = OpFAdd %c_at_t1_1 OpFMul %v3float %c1 %float_0_66666666
; Store and emit first quad
; v0
%ptr_out_v4 = OpAccessChain %v4float %gl_out %int_0
OpStore %ptr_out_v4 %v0_1_4
%ptr_out_color = OpAccessChain %v3float %outColor %int_0
OpStore %ptr_out_color %c_at_t0
OpEmitVertex
; v1
%ptr_out_v4 = OpAccessChain %v4float %gl_out %int_1
OpStore %ptr_out_v4 %v1_1_4
%ptr_out_color = OpAccessChain %v3float %outColor %int_1
OpStore %ptr_out_color %c_at_t0
OpEmitVertex
; v2
%ptr_out_v4 = OpAccessChain %v4float %gl_out %int_2
OpStore %ptr_out_v4 %v2_1_4
%ptr_out_color = OpAccessChain %v3float %outColor %int_2
OpStore %ptr_out_color %c_at_t1_1
OpEmitVertex
; v3
%ptr_out_v4 = OpAccessChain %v4float %gl_out %int_3
OpStore %ptr_out_v4 %v3_1_4
%ptr_out_color = OpAccessChain %v3float %outColor %int_3
OpStore %ptr_out_color %c_at_t1_1
OpEmitVertex
OpEndPrimitive

; Second part: t in [2/3, 1]
%t0_2 = OpConstant %float 0.6666667
%t1_2 = OpConstant %float 1.0
%p_a1_2 = OpFMul %v3float %seg %t0_2
%p_a1_2 = OpFAdd %v3float %p1_xyz %p_a1_2  ; p1 + 2/3*seg
%p_b1_2 = OpFMul %v3float %seg %t1_2
%p_b1_2 = OpFAdd %v3float %p1_xyz %p_b1_2  ; p2
%offset_scaled1 = OpFMul %v3float %offset %t0_2
%offset_scaled2 = OpFMul %v3float %offset %t1_2
; Vertices:
%v0_2 = OpFSub %v3float %p_a1_2 %offset_scaled1  ; p_a1_2 - offset
%v1_2 = OpFAdd %v3float %p_a1_2 %offset_scaled1  ; p_a1_2 + offset
%v2_2 = OpFSub %v3float %p_b1_2 %offset_scaled2  ; p2 - offset
%v3_2 = OpFAdd %v3float %p_b1_2 %offset_scaled2  ; p2 + offset
; Note: triangle strip order matters. After EndPrimitive, we restart.
; For new strip, we want: v0, v1, v2, v3
%v0_2_4 = OpInsertComposite %v4float %v0_2 %float_1 3
%v1_2_4 = OpInsertComposite %v4float %v1_2 %float_1 3
%v2_2_4 = OpInsertComposite %v4float %v2_2 %float_1 3
%v3_2_4 = OpInsertComposite %v4float %v3_2 %float_1 3
; Color: at t=2/3 and t=1
%c_at_t0_2 = OpFMul %v3float %c2 %float_0_6666667
%c_at_t0_2 = OpFAdd %c_at_t0_2 OpFMul %v3float %c1 %float_0_3333333
%c_at_t1_2 = %c2
; Emit second quad
; v0
%ptr_out_v4 = OpAccessChain %v4float %gl_out %int_4
OpStore %ptr_out_v4 %v0_2_4
%ptr_out_color = OpAccessChain %v3float %outColor %int_4
OpStore %ptr_out_color %c_at_t0_2
OpEmitVertex
; v1
%ptr_out_v4 = OpAccessChain %v4float %gl_out %int_5
OpStore %ptr_out_v4 %v1_2_4
%ptr_out_color = OpAccessChain %v3float %outColor %int_5
OpStore %ptr_out_color %c_at_t0_2
OpEmitVertex
; v2
%ptr_out_v4 = OpAccessChain %v4float %gl_out %int_6
OpStore %ptr_out_v4 %v2_2_4
%ptr_out_color = OpAccessChain %v3float %outColor %int_6
OpStore %ptr_out_color %c_at_t1_2
OpEmitVertex
; v3
%ptr_out_v4 = OpAccessChain %v4float %gl_out %int_7
OpStore %ptr_out_v4 %v3_2_4
%ptr_out_color = OpAccessChain %v3float %outColor %int_7
OpStore %ptr_out_color %c_at_t1_2
OpEmitVertex
OpEndPrimitive

OpReturn
OpFunctionEnd
