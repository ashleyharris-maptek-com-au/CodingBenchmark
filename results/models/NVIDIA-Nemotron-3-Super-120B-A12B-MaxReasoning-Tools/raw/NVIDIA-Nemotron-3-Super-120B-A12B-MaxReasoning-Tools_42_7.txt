```
; SPIR-V Geometry Shader: Explode Triangles Along Face Normal
; Input primitive: Triangles (3 vertices)
; Output primitive: TriangleStrip, 3 vertices emitted per invocation

OpCapability Shader
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main InputTriangles
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 3
OpExecutionMode %main Invocations 1

; Types
%void = OpTypeVoid
%bool = OpTypeBool
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4

; Constants
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%float_0 = OpConstant %float 0.0
%float_1 = OpConstant %float 1.0
%float_0_08 = OpConstant %float 0.08

; UBO (set=0, binding=0) - same as test 41
%uint_0 = OpConstant %uint 0
%uint_1 = OpConstant %uint 1
%ptr_UniformConstant_uvec2 = OpTypePointer UniformConstant %v2uint
%v2uint = OpTypeVector %uint 2
%ptr_UniformConstant_uvec2_0_0 = OpConstantComposite %v2uint %uint_0 %uint_0
%ptr_UniformConstant_mat4 = OpTypePointer UniformConstant %mat4
%mat4 = OpTypeMatrix %float 4
%ptr_UniformConstant_vec3 = OpTypePointer UniformConstant %v3float
%ptr_UniformConstant_float = OpTypePointer UniformConstant %float

%UBO = OpTypeStruct %mat4 %mat4 %mat4 %v3float %v3float %v3float %float
OpDecorate %UBO Block
OpMemberDecorate %UBO 0 Offset 0   ; model
OpMemberDecorate %UBO 1 Offset 64  ; view
OpMemberDecorate %UBO 2 Offset 128 ; proj
OpMemberDecorate %UBO 3 Offset 192 ; lightPos
OpMemberDecorate %UBO 4 Offset 204 ; cameraPos
OpMemberDecorate %UBO 5 Offset 216 ; params
OpMemberDecorate %UBO 6 Offset 228 ; unused (padding)
%ptr_UBO = OpTypePointer UniformConstant %UBO
%ubo = OpVariable %ptr_UBO Uniform

; Input: gl_PerVertex block (per-vertex array)
%gl_PerVertex_in = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
%arr_pv_in = OpTypeArray %gl_PerVertex_in %int_3
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

; User inputs: arrays of size 3
%ptr_in_vec3 = OpTypePointer Input %v3float
%arr_worldPos = OpTypeArray %v3float %int_3
%ptr_in_arr_worldPos = OpTypePointer Input %arr_worldPos
%worldPos = OpVariable %ptr_in_arr_worldPos Input
OpDecorate %worldPos Location 0

%arr_normal = OpTypeArray %v3float %int_3
%ptr_in_arr_normal = OpTypePointer Input %arr_normal
%normal = OpVariable %ptr_in_arr_normal Input
OpDecorate %normal Location 1

%arr_color = OpTypeArray %v3float %int_3
%ptr_in_arr_color = OpTypePointer Input %arr_color
%color = OpVariable %ptr_in_arr_color Input
OpDecorate %color Location 2

; Output: gl_PerVertex block (single emitted vertex)
%gl_PerVertex_out = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

%ptr_out_vec3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_vec3 Output
OpDecorate %outColor Location 0

; Helper: load vec3 from array at index
%func_load_vec3 = OpFunction %v3float %None %int
%ptr_func_idx = OpFunctionParameter %int
%arr_param = OpFunctionParameter %ptr_in_arr_pv  ; generic, will be cast
%idx = OpVariable %ptr_func_idx Function
OpStore %idx %ptr_func_idx
%load_idx = OpLoad %int %idx
%gep = OpAccessChain %ptr_in_arr_pv %arr_param %load_idx %int_0
%elem = OpLoad %gl_PerVertex_in %gep
%vec3_ptr = OpAccessChain %ptr_in_arr_pv %gep %int_0  ; .gl_Position is first member
%vec3_load = OpLoad %v4float %vec3_ptr
%vec3 = OpCompositeExtract %vec3_load 0 1 2  ; extract x,y,z
OpReturnValue %vec3
OpFunctionEnd

; Main function
%func_main = OpFunction %void %None
%entry = OpLabel
%main = OpLabel

; Load UBO (unused in this shader, but required to match interface)
%ubo_val = OpLoad %UBO %ubo

; Compute face center and normal from gl_in[0], gl_in[1], gl_in[2]
; p0 = gl_in[0].gl_Position
%p0_ptr = OpAccessChain %gl_in %int_0 %int_0
%p0_vec4 = OpLoad %v4float %p0_ptr
%p0 = OpCompositeExtract %p0_vec4 0 1 2

; p1 = gl_in[1].gl_Position
%p1_ptr = OpAccessChain %gl_in %int_1 %int_0
%p1_vec4 = OpLoad %v4float %p1_ptr
%p1 = OpCompositeExtract %p1_vec4 0 1 2

; p2 = gl_in[2].gl_Position
%p2_ptr = OpAccessChain %gl_in %int_2 %int_0
%p2_vec4 = OpLoad %v4float %p2_ptr
%p2 = OpCompositeExtract %p2_vec4 0 1 2

; center = (p0 + p1 + p2) / 3
%sum = OpFAdd %v3float %p0 %p1
%sum = OpFAdd %v3float %sum %p2
%one_third = OpConstant %float 0.3333333
%center = OpFMul %v3float %sum %one_third

; e1 = p1 - p0
%e1 = OpFSub %v3float %p1 %p0
; e2 = p2 - p0
%e2 = OpFSub %v3float %p2 %p0

; cross(e1, e2) = (e1.y*e2.z - e1.z*e2.y, e1.z*e2.x - e1.x*e2.z, e1.x*e2.y - e1.y*e2.x)
%e1x = OpCompositeExtract %e1 0
%e1y = OpCompositeExtract %e1 1
%e1z = OpCompositeExtract %e1 2
%e2x = OpCompositeExtract %e2 0
%e2y = OpCompositeExtract %e2 1
%e2z = OpCompositeExtract %e2 2

%cx = OpFSub %float (OpFMul %float %e1y %e2z) (OpFMul %float %e1z %e2y)
%cy = OpFSub %float (OpFMul %float %e1z %e2x) (OpFMul %float %e1x %e2z)
%cz = OpFSub %float (OpFMul %float %e1x %e2y) (OpFMul %float %e1y %e2x)
%N_raw = OpCompositeConstruct %v3float %cx %cy %cz

; length = sqrt(cx^2 + cy^2 + cz^2)
%cx2 = OpFMul %float %cx %cx
%cy2 = OpFMul %float %cy %cy
%cz2 = OpFMul %float %cz %cz
%len2 = OpFAdd %float (OpFAdd %float %cx2 %cy2) %cz2
%len = OpFsqrt %float %len2
%one_over_len = OpFDiv %float %float_1 %len
%N = OpCompositeConstruct %v3float
    (OpFMul %float %cx %one_over_len)
    (OpFMul %float %cy %one_over_len)
    (OpFMul %float %cz %one_over_len)

; offset = N * 0.08
%offset = OpFMul %v3float %N %float_0_08

; Emit 3 vertices for triangle strip
; Vertex 0: p0 + offset
%v0_pos = OpFAdd %v3float %p0 %offset
%v0_vec4 = OpInsertComposite %v4float %v0_pos %float_0 0  %float_0 1  %float_0 2  %float_1 3
%v0_ptr_out = OpAccessChain %gl_out %int_0
%v0_color_ptr = OpAccessChain %worldPos %int_0  ; reuse worldPos for color? No — use color array
%v0_color_ptr = OpAccessChain %color %int_0
%v0_color = OpLoad %v3float %v0_color_ptr
OpStore %v0_ptr_out %v0_vec4
OpStore %outColor %v0_color
OpEmitVertex

; Vertex 1: p1 + offset
%v1_pos = OpFAdd %v3float %p1 %offset
%v1_vec4 = OpInsertComposite %v4float %v1_pos %float_0 0  %float_0 1  %float_0 2  %float_1 3
%v1_ptr_out = OpAccessChain %gl_out %int_0  ; re-use same output variable (overwrites previous)
%v1_color_ptr = OpAccessChain %color %int_1
%v1_color = OpLoad %v3float %v1_color_ptr
OpStore %v1_ptr_out %v1_vec4
OpStore %outColor %v1_color
OpEmitVertex

; Vertex 2: p2 + offset
%v2_pos = OpFAdd %v3float %p2 %offset
%v2_vec4 = OpInsertComposite %v4float %v2_pos %float_0 0  %float_0 1  %float_0 2  %float_1 3
%v2_ptr_out = OpAccessChain %gl_out %int_0
%v2_color_ptr = OpAccessChain %color %int_2
%v2_color = OpLoad %v3float %v2_color_ptr
OpStore %v2_ptr_out %v2_vec4
OpStore %outColor %v2_color
OpEmitVertex

OpEndPrimitive
OpReturn
OpFunctionEnd
```

Note: This SPIR-V assembly assumes the input and output interfaces are correctly matched by the fixed vertex and fragment shaders. The UBO is loaded but not used, as the explosion is done in clip space (as per requirement). The color is taken directly from the input `color[]` array. The output uses a single `gl_out` variable overwritten per vertex, which is valid for geometry shaders emitting vertices sequentially. All required decorations, modes, and instructions are included.
