import sys

def solve() -> None:
    data = sys.stdin.read().splitlines()
    # Remove possible leading empty lines
    i = 0
    while i < len(data) and not data[i].strip():
        i += 1
    if i >= len(data):
        return
    first = data[i].strip()
    data = data[i:]  # start from the first meaningful line

    parts = first.split()
    if len(parts) < 2:
        return
    U, S = map(int, parts[:2])

    set_masks = []
    # cover[e] = list of set indices that contain element e
    cover = [[] for _ in range(U)]

    for idx in range(S):
        if idx + 1 >= len(data):
            # missing line, treat as empty set
            line = ''
        else:
            line = data[idx + 1].strip()
        if line == '':
            mask = 0
        else:
            elems = list(map(int, line.split()))
            mask = 0
            for e in elems:
                if 0 <= e < U:
                    mask |= 1 << e
        set_masks.append(mask)
        if mask != 0:
            m = mask
            while m:
                low = m & -m
                e = low.bit_length() - 1
                cover[e].append(idx)
                m ^= low   # remove the lowest set bit

    full_mask = (1 << U) - 1 if U > 0 else 0

    # Trivial case: empty universe
    if U == 0:
        print("SOLUTION")
        print()
        return

    sys.setrecursionlimit(10000)
    solution = None

    def search(current_mask: int, chosen: list) -> bool:
        nonlocal solution
        if solution is not None:
            return True
        if current_mask == full_mask:
            solution = chosen.copy()
            return True

        # Choose an uncovered element with the fewest remaining viable sets
        uncovered = (~current_mask) & full_mask
        best_elem = -1
        best_opts = None
        min_len = None

        m = uncovered
        while m:
            low = m & -m
            e = low.bit_length() - 1
            opts = []
            for si in cover[e]:
                if (set_masks[si] & current_mask) == 0:
                    opts.append(si)
            if not opts:  # dead end
                return False
            if min_len is None or len(opts) < min_len:
                min_len = len(opts)
                best_elem = e
                best_opts = opts
                if min_len == 1:
                    break
            m ^= low

        for si in best_opts:
            chosen.append(si)
            if search(current_mask | set_masks[si], chosen):
                return True
            chosen.pop()
        return False

    if search(0, []):
        solution.sort()
        print("SOLUTION")
        print(" ".join(map(str, solution)))
    else:
        print("NO SOLUTION")

if __name__ == "__main__":
    solve()
