    void expand(const Point3D& p) {

        max.x = std::max(max.x, p.x); max.y = std::max(max.y, p.y); max.z = std::max(max.z, p.z);

    void expand(const BBox& other) {

        expand(other.max);

        double dy = max.y - min.y;
