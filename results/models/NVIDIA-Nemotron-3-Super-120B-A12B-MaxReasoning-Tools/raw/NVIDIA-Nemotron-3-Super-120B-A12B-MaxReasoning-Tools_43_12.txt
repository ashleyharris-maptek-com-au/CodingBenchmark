Implement:

float4 main(PSInput input) : SV_Target0
{
    float3 N = normalize(input.normal);
    float factor = N.y * 0.5 + 0.5;
    float3 sky = float3(0.4,0.6,1.0);
    float3 ground = float3(0.3,0.15,0.05);
    float3 ambient = lerp(ground, sky, factor); // ground*(1-factor)+sky*factor

    float3 lightDir = normalize(UBO.lightPos.xyz - input.worldPos);
    float d = max(dot(N, lightDir), 0.0);
    float3 diffuse = 0.3 * d;

    float3 col = ambient + diffuse;
    // optionally add input.color? Not required.
    return float4(col, 1.0);
}
