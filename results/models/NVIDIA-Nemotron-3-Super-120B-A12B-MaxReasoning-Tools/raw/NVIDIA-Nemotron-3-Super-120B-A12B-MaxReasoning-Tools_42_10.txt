; SPIR-V Geometry Shader for thick smooth curve (InputLinesAdjacency)
; Subdivides segment P1→P2 into 4 sub-segments (5 points)
; Emits triangle strip of 10 vertices (pos±perp) for thick curve
; Uses Catmull-Rom-inspired tangents for smooth normals
; Outputs interpolated color from color[1] to color[2]

OpCapability Shader

OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPosIn %normalIn %colorIn %gl_out %outColor
OpExecutionMode %main InputLinesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 10
OpExecutionMode %main Invocations 1

%gl_PerVertex_in = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_in Block

; Main function
%void_func = OpTypeFunction %void
%main = OpFunction %void None %void_func
%entry = OpLabel

; Load P0, P1, P2, P3 from gl_in[0..3]
%ptr_p0 = OpAccessChain %ptr_in_arr_pv %gl_in %int_0
%p0_struct = OpLoad %gl_PerVertex_in %ptr_p0
%ptr_p0_pos = OpAccessChain %v4float %p0_struct %int_0
%P0 = OpLoad %v4float %ptr_p0_pos

%ptr_p1 = OpAccessChain %ptr_in_arr_pv %gl_in %int_1
%p1_struct = OpLoad %gl_PerVertex_in %ptr_p1
%ptr_p1_pos = OpAccessChain %v4float %p1_struct %int_0
%P1 = OpLoad %v4float %ptr_p1_pos

%ptr_p2 = OpAccessChain %ptr_in_arr_pv %gl_in %int_2
%p2_struct = OpLoad %gl_PerVertex_in %ptr_p2
%ptr_p2_pos = OpAccessChain %v4float %p2_struct %int_0
%P2 = OpLoad %v4float %ptr_p2_pos

%ptr_p3 = OpAccessChain %ptr_in_arr_pv %gl_in %int_3
%p3_struct = OpLoad %gl_PerVertex_in %ptr_p3
%ptr_p3_pos = OpAccessChain %v4float %p3_struct %int_0
%P3 = OpLoad %v4float %ptr_p3_pos

; Extract xyz components (ignore w for position math)
%P0_xyz = OpCompositeExtract %P0 %int_0 %int_1 %int_2
%P1_xyz = OpCompositeExtract %P1 %int_0 %int_1 %int_2
%P2_xyz = OpCompositeExtract %P2 %int_0 %int_1 %int_2
%P3_xyz = OpCompositeExtract %P3 %int_0 %int_1 %int_2

; Compute tangents for smooth perpendiculars (Catmull-Rom inspired)
; T_start = (P2 - P0) * 0.5
%sub_P2_P0 = OpVectorSubtract %P2_xyz %P0_xyz
%T_start = OpVectorTimesScalar %sub_P2_P0 %float_0_5

; T_end = (P3 - P1) * 0.5
%sub_P3_P1 = OpVectorSubtract %P3_xyz %P1_xyz
%T_end = OpVectorTimesScalar %sub_P3_P1 %float_0_5

; Subdivide P1→P2 into 4 segments → t = 0, 0.25, 0.5, 0.75, 1.0
; We'll loop 5 times (i=0 to 4)
; Since SPIR-V lacks loops in asm, we unroll manually

; i=0: t=0.0 → point = P1
%t0 = OpConstant %float 0.0
%delta_P1P2 = OpVectorSubtract %P2_xyz %P1_xyz
%interp0 = OpVectorTimesScalar %delta_P1P2 %t0
%point0 = OpVectorAdd %P1_xyz %interp0

; Tangent at t=0: blend T_start and T_end? Use T_start for start
%tan0 = %T_start

; i=1: t=0.25
%t1 = OpConstant %float 0.25
%interp1 = OpVectorTimesScalar %delta_P1P2 %t1
%point1 = OpVectorAdd %P1_xyz %interp1
%tan1 = OpVectorAdd %T_start (OpVectorTimesScalar (OpVectorSubtract %T_end %T_start) %t1)

; i=2: t=0.5
%t2 = OpConstant %float 0.5
%interp2 = OpVectorTimesScalar %delta_P1P2 %t2
%point2 = OpVectorAdd %P1_xyz %interp2
%tan2 = OpVectorAdd %T_start (OpVectorTimesScalar (OpVectorSubtract %T_end %T_start) %t2)

; i=3: t=0.75
%t3 = OpConstant %float 0.75
%interp3 = OpVectorTimesScalar %delta_P1P2 %t3
%point3 = OpVectorAdd %P1_xyz %interp3
%tan3 = OpVectorAdd %T_start (OpVectorTimesScalar (OpVectorSubtract %T_end %T_start) %t3)

; i=4: t=1.0 → point = P2
%t4 = OpConstant %float 1.0
%interp4 = OpVectorTimesScalar %delta_P1P2 %t4
%point4 = OpVectorAdd %P1_xyz %interp4
%tan4 = %T_end

; Compute perpendicular in view space? 
; Since we don't have camera matrix here, approximate perpendicular in world space.
; For a curve in arbitrary orientation, we need a reference vector.
; Use world-up (0,1,0) if available, but we don't have it.
; Fallback: use tangent × (0,0,1) or similar, but may degenerate.
; Instead, use a fixed approximate method: perp = normalize(cross(tan, vec3(0,1,0)))
; If tan is near vertical, use cross(tan, vec3(1,0,0))
; But we lack branching and state. Simplify: assume curve is mostly in XY plane.
; Use perp = (-tan_y, tan_x, 0)  → rotate tangent 90° in XY plane.
; This works if curve is in XY plane (z=0). For general case, not perfect but acceptable for demo.
; We'll compute 2D perp in XY and keep z=0.

; Function to compute 2D perpendicular: (x,y) → (-y, x)
%perp2D = OpFunction %v3float None %void_func  ; placeholder, we'll inline

; For each point, compute perp in XY plane: (-y, x, 0)
; Then scale by thickness/2 = 0.01

%thickness_half = OpConstant %float 0.01  ; 0.02 / 2

; Point 0
%perp0 = OpCompositeExtract %tan0 %int_1 %int_0  ; (y, x)
%perp0 = OpCompositeConstruct %v2float %perp0.x %perp0.y  ; actually: we want (-y, x)
; Better: extract x,y, then compute (-y, x)
%tan0_xy = OpCompositeExtract %tan0 %int_0 %int_1
%neg_y0 = OpVectorTimesScalar (OpCompositeExtract %tan0_xy %int_1) %float_neg_1  ; -y
%x0 = OpCompositeExtract %tan0_xy %int_0
%perp0_xy = OpCompositeConstruct %v2float %neg_y0 %x0
%perp0_xyz = OpVectorInsert %perp0_xy %float_0 %int_2  ; z=0
%len0 = OpLength %perp0_xyz
%perp0_norm = OpVectorTimesScalar %perp0_xyz (OpFDiv %float_1 %len0)
%offset0 = OpVectorTimesScalar %perp0_norm %thickness_half
%pos0_plus = OpVectorAdd %point0 %offset0
%pos0_minus = OpVectorSubtract %point0 %offset0

; Point 1
%tan1_xy = OpCompositeExtract %tan1 %int_0 %int_1
%neg_y1 = OpVectorTimesScalar (OpCompositeExtract %tan1_xy %int_1) %float_neg_1
%x1 = OpCompositeExtract %tan1_xy %int_0
%perp1_xy = OpCompositeConstruct %v2float %neg_y1 %x1
%perp1_xyz = OpVectorInsert %perp1_xy %float_0 %int_2
%len1 = OpLength %perp1_xyz
%perp1_norm = OpVectorTimesScalar %perp1_xyz (OpFDiv %float_1 %len1)
%offset1 = OpVectorTimesScalar %perp1_norm %thickness_half
%pos1_plus = OpVectorAdd %point1 %offset1
%pos1_minus = OpVectorSubtract %point1 %offset1

; Point 2
%tan2_xy = OpCompositeExtract %tan2 %int_0 %int_1
%neg_y2 = OpVectorTimesScalar (OpCompositeExtract %tan2_xy %int_1) %float_neg_1
%x2 = OpCompositeExtract %tan2_xy %int_0
%perp2_xy = OpCompositeConstruct %v2float %neg_y2 %x2
%perp2_xyz = OpVectorInsert %perp2_xy %float_0 %int_2
%len2 = OpLength %perp2_xyz
%perp2_norm = OpVectorTimesScalar %perp2_xyz (OpFDiv %float_1 %len2)
%offset2 = OpVectorTimesScalar %perp2_norm %thickness_half
%pos2_plus = OpVectorAdd %point2 %offset2
%pos2_minus = OpVectorSubtract %point2 %offset2

; Point 3
%tan3_xy = OpCompositeExtract %tan3 %int_0 %int_1
%neg_y3 = OpVectorTimesScalar (OpCompositeExtract %tan3_xy %int_1) %float_neg_1
%x3 = OpCompositeExtract %tan3_xy %int_0
%perp3_xy = OpCompositeConstruct %v2float %neg_y3 %x3
%perp3_xyz = OpVectorInsert %perp3_xy %float_0 %int_2
%len3 = OpLength %perp3_xyz
%perp3_norm = OpVectorTimesScalar %perp3_xyz (OpFDiv %float_1 %len3)
%offset3 = OpVectorTimesScalar %perp3_norm %thickness_half
%pos3_plus = OpVectorAdd %point3 %offset3
%pos3_minus = OpVectorSubtract %point3 %offset3

; Point 4
%tan4_xy = OpCompositeExtract %tan4 %int_0 %int_1
%neg_y4 = OpVectorTimesScalar (OpCompositeExtract %tan4_xy %int_1) %float_neg_1
%x4 = OpCompositeExtract %tan4_xy %int_0
%perp4_xy = OpCompositeConstruct %v2float %neg_y4 %x4
%perp4_xyz = OpVectorInsert %perp4_xy %float_0 %int_2
%len4 = OpLength %perp4_xyz
%perp4_norm = OpVectorTimesScalar %perp4_xyz (OpFDiv %float_1 %len4)
%offset4 = OpVectorTimesScalar %perp4_norm %thickness_half
%pos4_plus = OpVectorAdd %point4 %offset4
%pos4_minus = OpVectorSubtract %point4 %offset4

; Interpolate color from color[1] to color[2] (linear over 5 points)
%ptr_c1 = OpAccessChain %ptr_in_color %colorInVar %int_1
%color1 = OpLoad %v3float %ptr_c1
%ptr_c2 = OpAccessChain %ptr_in_color %colorInVar %int_2
%color2 = OpLoad %v3float %ptr_c2
%delta_color = OpVectorSubtract %color2 %color1

; For i=0 to 4: color = color1 + (i/4.0) * delta_color
%float_4 = OpConstant %float 4.0

; i=0
%color_out0 = %color1
; i=1
%t_color1 = OpConstant %float 0.25
%color_out1 = OpVectorAdd %color1 (OpVectorTimesScalar %delta_color %t_color1)
; i=2
%t_color2 = OpConstant %float 0.5
%color_out2 = OpVectorAdd %color1 (OpVectorTimesScalar %delta_color %t_color2)
; i=3
%t_color3 = OpConstant %float 0.75
%color_out3 = OpVectorAdd %color1 (OpVectorTimesScalar %delta_color %t_color3)
; i=4
%t_color4 = OpConstant %float 1.0
%color_out4 = OpVectorAdd %color1 (OpVectorTimesScalar %delta_color %t_color4)

; Now emit vertices in triangle strip order:
; For each point i: emit (point+perp), then (point-perp)
; So: 0+, 0-, 1+, 1-, 2+, 2-, 3+, 3-, 4+, 4-

; Vertex 0: point0 + perp0, color_out0
%ptr_out_pos0 = OpAccessChain %gl_out %int_0
%ptr_out_pos0_vec4 = OpAccessChain %v4float %ptr_out_pos0 %int_0
%out_pos0_vec4 = OpVectorInsert %pos0_plus %float_1.0 %int_3  ; w=1.0
OpStore %ptr_out_pos0_vec4 %out_pos0_vec4
%ptr_out_color0 = OpAccessChain %outColorVar %int_0
OpStore %ptr_out_color0 %color_out0
OpEmitVertex

; Vertex 1: point0 - perp0, color_out0
%ptr_out_pos1 = OpAccessChain %gl_out %int_1
%ptr_out_pos1_vec4 = OpAccessChain %v4float %ptr_out_pos1 %int_0
%out_pos1_vec4 = OpVectorInsert %pos0_minus %float_1.0 %int_3
OpStore %ptr_out_pos1_vec4 %out_pos1_vec4
%ptr_out_color1 = OpAccessChain %outColorVar %int_1
OpStore %ptr_out_color1 %color_out0
OpEmitVertex

; Vertex 2: point1 + perp1, color_out1
%ptr_out_pos2 = OpAccessChain %gl_out %int_2
%ptr_out_pos2_vec4 = OpAccessChain %v4float %ptr_out_pos2 %int_0
%out_pos2_vec4 = OpVectorInsert %pos1_plus %float_1.0 %int_3
OpStore %ptr_out_pos2_vec4 %out_pos2_vec4
%ptr_out_color2 = OpAccessChain %outColorVar %int_2
OpStore %ptr_out_color2 %color_out1
OpEmitVertex

; Vertex 3: point1 - perp1, color_out1
%ptr_out_pos3 = OpAccessChain %gl_out %int_3
%ptr_out_pos3_vec4 = OpAccessChain %v4float %ptr_out_pos3 %int_0
%out_pos3_vec4 = OpVectorInsert %pos1_minus %float_1.0 %int_3
OpStore %ptr_out_pos3_vec4 %out_pos3_vec4
%ptr_out_color3 = OpAccessChain %outColorVar %int_3
OpStore %ptr_out_color3 %color_out1
OpEmitVertex

; Vertex 4: point2 + perp2, color_out2
%ptr_out_pos4 = OpAccessChain %gl_out %int_4
%ptr_out_pos4_vec4 = OpAccessChain %v4float %ptr_out_pos4 %int_0
%out_pos4_vec4 = OpVectorInsert %pos2_plus %float_1.0 %int_3
OpStore %ptr_out_pos4_vec4 %out_pos4_vec4
%ptr_out_color4 = OpAccessChain %outColorVar %int_4
OpStore %ptr_out_color4 %color_out2
OpEmitVertex

; Vertex 5: point2 - perp2, color_out2
%ptr_out_pos5 = OpAccessChain %gl_out %int_5
%ptr_out_pos5_vec4 = OpAccessChain %v4float %ptr_out_pos5 %int_0
%out_pos5_vec4 = OpVectorInsert %pos2_minus %float_1.0 %int_3
OpStore %ptr_out_pos5_vec4 %out_pos5_vec4
%ptr_out_color5 = OpAccessChain %outColorVar %int_5
OpStore %ptr_out_color5 %color_out2
OpEmitVertex

; Vertex 6: point3 + perp3, color_out3
%ptr_out_pos6 = OpAccessChain %gl_out %int_6
%ptr_out_pos6_vec4 = OpAccessChain %v4float %ptr_out_pos6 %int_0
%out_pos6_vec4 = OpVectorInsert %pos3_plus %float_1.0 %int_3
OpStore %ptr_out_pos6_vec4 %out_pos6_vec4
%ptr_out_color6 = OpAccessChain %outColorVar %int_6
OpStore %ptr_out_color6 %color_out3
OpEmitVertex

; Vertex 7: point3 - perp3, color_out3
%ptr_out_pos7 = OpAccessChain %gl_out %int_7
%ptr_out_pos7_vec4 = OpAccessChain %v4float %ptr_out_pos7 %int_0
%out_pos7_vec4 = OpVectorInsert %pos3_minus %float_1.0 %int_3
OpStore %ptr_out_pos7_vec4 %out_pos7_vec4
%ptr_out_color7 = OpAccessChain %outColorVar %int_7
OpStore %ptr_out_color7 %color_out3
OpEmitVertex

; Vertex 8: point4 + perp4, color_out4
%ptr_out_pos8 = OpAccessChain %gl_out %int_8
%ptr_out_pos8_vec4 = OpAccessChain %v4float %ptr_out_pos8 %int_0
%out_pos8_vec4 = OpVectorInsert %pos4_plus %float_1.0 %int_3
OpStore %ptr_out_pos8_vec4 %out_pos8_vec4
%ptr_out_color8 = OpAccessChain %outColorVar %int_8
OpStore %ptr_out_color8 %color_out4
OpEmitVertex

; Vertex 9: point4 - perp4, color_out4
%ptr_out_pos9 = OpAccessChain %gl_out %int_9
%ptr_out_pos9_vec4 = OpAccessChain %v4float %ptr_out_pos9 %int_0
%out_pos9_vec4 = OpVectorInsert %pos4_minus %float_1.0 %int_3
OpStore %ptr_out_pos9_vec4 %out_pos9_vec4
%ptr_out_color9 = OpAccessChain %outColorVar %int_9
OpStore %ptr_out_color9 %color_out4
OpEmitVertex

OpEndPrimitive
OpReturn
OpFunctionEnd
