@group(0) @binding(0) var<storage, read>       baseData : array<u32, 16>;
@group(0) @binding(1) var<storage, read_write>  result   : array<u32, 4>;
@group(0) @binding(2) var<uniform>              params   : vec4<u32>;

// params.x = difficulty, params.y = range_start, params.z = range_size

@compute @workgroup_size(256)
fn main(@builtin(global_invocation_id) gid : vec3<u32>) {
    let idx = gid.x;
    if (idx >= params.z) {
        return;
    }

    let nonce = params.y + idx;

    var state: array<u32, 16>;
    for (i in 0u..16u) {
        state[i] = baseData[i] ^ (nonce + i);
    }

    // 8 rounds of quarter-round mixing (explicitly unrolled)
    for (_ in 0u..8u) {
        // Group 0: state[0], state[1], state[2], state[3]
        let a0 = state[0] + state[1];
        let d0 = state[3] ^ a0;
        state[3] = (d0 << 16u) | (d0 >> (32u - 16u));
        let c0 = state[2] + state[3];
        let b0 = state[1] ^ c0;
        state[1] = (b0 << 12u) | (b0 >> (32u - 12u));
        let a1 = state[0] + state[1];
        let d1 = state[3] ^ a1;
        state[3] = (d1 << 8u) | (d1 >> (32u - 8u));
        let c1 = state[2] + state[3];
        let b1 = state[1] ^ c1;
        state[1] = (b1 << 7u) | (b1 >> (32u - 7u));

        // Group 1: state[4], state[5], state[6], state[7]
        let a2 = state[4] + state[5];
        let d2 = state[7] ^ a2;
        state[7] = (d2 << 16u) | (d2 >> (32u - 16u));
        let c2 = state[6] + state[7];
        let b2 = state[5] ^ c2;
        state[5] = (b2 << 12u) | (b2 >> (32u - 12u));
        let a3 = state[4] + state[5];
        let d3 = state[7] ^ a3;
        state[7] = (d3 << 8u) | (d3 >> (32u - 8u));
        let c3 = state[6] + state[7];
        let b3 = state[5] ^ c3;
        state[5] = (b3 << 7u) | (b3 >> (32u - 7u));

        // Group 2: state[8], state[9], state[10], state[11]
        let a4 = state[8] + state[9];
        let d4 = state[11] ^ a4;
        state[11] = (d4 << 16u) | (d4 >> (32u - 16u));
        let c4 = state[10] + state[11];
        let b4 = state[9] ^ c4;
        state[9] = (b4 << 12u) | (b4 >> (32u - 12u));
        let a5 = state[8] + state[9];
        let d5 = state[11] ^ a5;
        state[11] = (d5 << 8u) | (d5 >> (32u - 8u));
        let c5 = state[10] + state[11];
        let b5 = state[9] ^ c5;
        state[9] = (b5 << 7u) | (b5 >> (32u - 7u));

        // Group 3: state[12], state[13], state[14], state[15]
        let a6 = state[12] + state[13];
        let d6 = state[15] ^ a6;
        state[15] = (d6 << 16u) | (d6 >> (32u - 16u));
        let c6 = state[14] + state[15];
        let b6 = state[13] ^ c6;
        state[13] = (b6 << 12u) | (b6 >> (32u - 12u));
        let a7 = state[12] + state[13];
        let d7 = state[15] ^ a7;
        state[15] = (d7 << 8u) | (d7 >> (32u - 8u));
        let c7 = state[14] + state[15];
        let b7 = state[13] ^ c7;
        state[13] = (b7 << 7u) | (b7 >> (32u - 7u));
    }

    // Fold state to h0 and h1
    let h0 = state[0] ^ state[2] ^ state[4] ^ state[6] ^ state[8] ^ state[10] ^ state[12] ^ state[14];
    let h1 = state[1] ^ state[3] ^ state[5] ^ state[7] ^ state[9] ^ state[11] ^ state[13] ^ state[15];

    // Count trailing zeros in 64-bit value [h0 | (h1 << 32)]
    let tz_h0 = count_trailing_zeros(h0);
    let tz = select(tz_h0, 32u + count_trailing_zeros(h1), tz_h0 == 32u);

    if (tz >= params.x) {
        result[0] = 1u;
        result[1] = nonce;
        result[2] = h0;
        result[3] = h1;
    }
}
