; SPIR-V Geometry Shader
; Vulkan 1.0, Geometry Shader, InputPoints -> OutputTriangleStrip
; Expands each input point into a hexagon (6-triangle fan) using 6 separate triangle strips
; Each strip: center -> edge_i -> edge_{i+1} -> EndPrimitive
; Total emitted vertices per invocation: 18 (6 strips × 3 vertices)
; Output primitive: TriangleStrip, max vertices: 18

; Magic number and version
OpCapability Shader
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos_in %normal_in %color_in %gl_out %outColor
OpExecutionMode %main InputPoints
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 18
OpExecutionMode %main Invocations 1

; Source language (optional but good practice)
OpSource GLSL 450

; ============================================================
; Type definitions
; ============================================================

; Float and vector types
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4

; Integer types
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1

; Boolean
%bool = OpTypeBool

; ============================================================
; UBO (set=0, binding=0) - same as test 41
; ============================================================
%_struct_UBO = OpTypeStruct
  %float   ; model (column 0)
  %float   ; model (column 1)
  %float   ; model (column 2)
  %float   ; model (column 3)
  %float   ; view (column 0)
  %float   ; view (column 1)
  %float   ; view (column 2)
  %float   ; view (column 3)
  %float   ; proj (column 0)
  %float   ; proj (column 1)
  %float   ; proj (column 2)
  %float   ; proj (column 3)
  %v3float ; lightPos
  %v3float ; cameraPos
  %float   ; params.x (time)
  %float   ; params.y
  %float   ; params.z
  %float   ; params.w
%ptr_UBO_uniform = OpTypePointer Uniform %_struct_UBO
%UBO = OpVariable %ptr_UBO_uniform Uniform

; ============================================================
; Input: gl_PerVertex (built-in)
; ============================================================
%gl_PerVertex_in = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint 1   ; InputPoints: size 1
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

; ============================================================
; User-defined inputs (per-vertex arrays, size=1 for InputPoints)
; ============================================================
%ptr_in_vec3 = OpTypePointer Input %v3float
%worldPos_in = OpVariable %ptr_in_vec3 Input
  OpDecorate %worldPos_in Location 0
%normal_in = OpVariable %ptr_in_vec3 Input
  OpDecorate %normal_in Location 1
%color_in = OpVariable %ptr_in_vec3 Input
  OpDecorate %color_in Location 2

; ============================================================
; Output: gl_PerVertex (built-in)
; ============================================================
%gl_PerVertex_out = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

; ============================================================
; User-defined output (per emitted vertex)
; ============================================================
%ptr_out_vec3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_vec3 Output
  OpDecorate %outColor Location 0

; ============================================================
; Constants
; ============================================================
%uint_0 = OpConstant %uint 0
%uint_1 = OpConstant %uint 1
%uint_6 = OpConstant %uint 6
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%float_0 = OpConstant %float 0.0
%float_1 = OpConstant %float 1.0
%float_0_04 = OpConstant %float 0.04   ; radius
%float_0_5 = OpConstant %float 0.5     ; for darkening
%float_2pi = OpConstant %float 6.283185307179586 ; 2*pi

; ============================================================
; Helper function: cos(angle) - approximated via inline (no func call for simplicity)
; We'll compute sin/cos directly in main using OpFAdd, OpFMul, etc.
; Since no trig instructions, we approximate using known values at 60° increments.
; cos(0) = 1, cos(60)=0.5, cos(120)=-0.5, cos(180)=-1, cos(240)=-0.5, cos(300)=0.5
; sin(0)=0, sin(60)=√3/2≈0.866, sin(120)=√3/2≈0.866, sin(180)=0, sin(240)=-√3/2≈-0.866, sin(300)=-√3/2≈-0.866
%float_0_5 = OpConstant %float 0.5
%float_neg_0_5 = OpConstant %float -0.5
%float_neg_1 = OpConstant %float -1.0
%float_sqrt3_2 = OpConstant %float 0.8660254037844386 ; sqrt(3)/2
%float_neg_sqrt3_2 = OpConstant %float -0.8660254037844386

; ============================================================
; Main function
; ============================================================
%void = OpTypeVoid
%func_main = OpTypeFunction %void
%main = OpFunction %void None %func_main
%label_start = OpLabel

; ============================================================
; Load input data (index 0 only, since InputPoints, array size=1)
; ============================================================
; gl_in[0].gl_Position
%ptr_gl_in_0 = OpAccessChain %ptr_in_arr_pv %gl_in %uint_0
%ptr_gl_in_0_v4 = OpAccessChain %ptr_in_arr_pv %ptr_gl_in_0 %int_0
%gl_in_0_pos = OpLoad %v4float %ptr_gl_in_0_v4

; worldPos_in[0]
%worldPos_0 = OpLoad %v3float %worldPos_in

; normal_in[0] (unused but loaded for completeness)
%normal_0 = OpLoad %v3float %normal_in

; color_in[0]
%color_0 = OpLoad %v3float %color_in

; ============================================================
; Compute center position (in clip space, already transformed)
; For hexagon, we offset in model/view/proj space? 
; But input worldPos is already in clip space? 
; Per task: "layout(location = 0) in vec3 worldPos[]; ; positions in clip space"
; So worldPos is clip-space position. We'll treat it as center in clip space.
; Offset in clip space: but clip space [-1,1] not uniform. 
; However, for small radius and assuming near origin, we approximate.
; Better: assume worldPos is in view space? But says clip space.
; Since radius is small (0.04), and clip space is [-1,1], we can use directly if point is not near edge.
; We'll use worldPos as center in clip space and add offset in clip space.
; This is approximate but acceptable for task.
%center = %worldPos_0

; ============================================================
; Precompute edge offsets (6 directions) in clip space
; Using precomputed cos/sin for 0°,60°,120°,180°,240°,300°
; offset = radius * (cos, sin, 0)
; We'll compute 6 edge positions: center + offset
; ============================================================
; Angle 0°: cos=1, sin=0
%offset0 = OpCompositeConstruct %v3float %float_0_04 %float_0 %float_0
%edge0 = OpFAdd %v3float %center %offset0

; Angle 60°: cos=0.5, sin=√3/2
%offset1 = OpCompositeConstruct %v3float 
  %float_0_02  ; 0.04 * 0.5 = 0.02
  %float_0_034641  ; 0.04 * 0.8660254 ≈ 0.034641
  %float_0
%edge1 = OpFAdd %v3float %center %offset1

; Angle 120°: cos=-0.5, sin=√3/2
%offset2 = OpCompositeConstruct %v3float
  %OpFNeg %float_0_02   ; -0.02
  %float_0_034641
  %float_0
%edge2 = OpFAdd %v3float %center %offset2

; Angle 180°: cos=-1, sin=0
%offset3 = OpCompositeConstruct %v3float
  %OpFNeg %float_0_04   ; -0.04
  %float_0
  %float_0
%edge3 = OpFAdd %v3float %center %offset3

; Angle 240°: cos=-0.5, sin=-√3/2
%offset4 = OpCompositeConstruct %v3float
  %OpFNeg %float_0_02
  %OpFNeg %float_0_034641
  %float_0
%edge4 = OpFAdd %v3float %center %offset4

; Angle 300°: cos=0.5, sin=-√3/2
%offset5 = OpCompositeConstruct %v3float
  %float_0_02
  %OpFNeg %float_0_034641
  %float_0
%edge5 = OpFAdd %v3float %center %offset5

; ============================================================
; Emit 6 triangle strips: each strip = center, edge_i, edge_{i+1}
; For i=0 to 5, edge_{i+1} wraps to edge0 when i=5
; After each strip: OpEndPrimitive
; ============================================================
; We'll emit vertices in order:
; Strip 0: center, edge0, edge1 -> EndPrimitive
; Strip 1: center, edge1, edge2 -> EndPrimitive
; Strip 2: center, edge2, edge3 -> EndPrimitive
; Strip 3: center, edge3, edge4 -> EndPrimitive
; Strip 4: center, edge4, edge5 -> EndPrimitive
; Strip 5: center, edge5, edge0 -> EndPrimitive

; Helper: compute darkened color (50% brightness)
%dark_color = OpFMul %v3float %color_0 %float_0_5

; ============================================================
; Strip 0: center, edge0, edge1
; ============================================================
; Vertex 0: center
%ptr_gl_out_v4 = OpAccessChain %ptr_out_pv %gl_out %int_0
%ptr_out_color = OpAccessChain %ptr_out_vec3 %outColor %int_0
OpStore %ptr_gl_out_v4 %gl_PerVertex_in  ; Wait, no: we need to construct gl_PerVertex_out from center
; Actually: gl_PerVertex_out is struct with vec4 pos. We have center as vec3 -> vec3(center, 1.0)
%center_vec4 = OpCompositeConstruct %v4float %center %float_1
OpStore %ptr_gl_out_v4 %center_vec4
OpStore %ptr_out_color %color_0  ; center: full color
OpEmitVertex

; Vertex 1: edge0
OpStore %ptr_gl_out_v4 %OpCompositeConstruct %v4float %edge0 %float_1
OpStore %ptr_out_color %dark_color  ; edge: dark
OpEmitVertex

; Vertex 2: edge1
OpStore %ptr_gl_out_v4 %OpCompositeConstruct %v4float %edge1 %float_1
OpStore %ptr_out_color %dark_color
OpEmitVertex
OpEndPrimitive

; ============================================================
; Strip 1: center, edge1, edge2
; ============================================================
; Vertex 0: center
OpStore %ptr_gl_out_v4 %center_vec4
OpStore %ptr_out_color %color_0
OpEmitVertex

; Vertex 1: edge1
OpStore %ptr_gl_out_v4 %OpCompositeConstruct %v4float %edge1 %float_1
OpStore %ptr_out_color %dark_color
OpEmitVertex

; Vertex 2: edge2
OpStore %ptr_gl_out_v4 %OpCompositeConstruct %v4float %edge2 %float_1
OpStore %ptr_out_color %dark_color
OpEmitVertex
OpEndPrimitive

; ============================================================
; Strip 2: center, edge2, edge3
; ============================================================
; Vertex 0: center
OpStore %ptr_gl_out_v4 %center_vec4
OpStore %ptr_out_color %color_0
OpEmitVertex

; Vertex 1: edge2
OpStore %ptr_gl_out_v4 %OpCompositeConstruct %v4float %edge2 %float_1
OpStore %ptr_out_color %dark_color
OpEmitVertex

; Vertex 2: edge3
OpStore %ptr_gl_out_v4 %OpCompositeConstruct %v4float %edge3 %float_1
OpStore %ptr_out_color %dark_color
OpEmitVertex
OpEndPrimitive

; ============================================================
; Strip 3: center, edge3, edge4
; ============================================================
; Vertex 0: center
OpStore %ptr_gl_out_v4 %center_vec4
OpStore %ptr_out_color %color_0
OpEmitVertex

; Vertex 1: edge3
OpStore %ptr_gl_out_v4 %OpCompositeConstruct %v4float %edge3 %float_1
OpStore %ptr_out_color %dark_color
OpEmitVertex

; Vertex 2: edge4
OpStore %ptr_gl_out_v4 %OpCompositeConstruct %v4float %edge4 %float_1
OpStore %ptr_out_color %dark_color
OpEmitVertex
OpEndPrimitive

; ============================================================
; Strip 4: center, edge4, edge5
; ============================================================
; Vertex 0: center
OpStore %ptr_gl_out_v4 %center_vec4
OpStore %ptr_out_color %color_0
OpEmitVertex

; Vertex 1: edge4
OpStore %ptr_gl_out_v4 %OpCompositeConstruct %v4float %edge4 %float_1
OpStore %ptr_out_color %dark_color
OpEmitVertex

; Vertex 2: edge5
OpStore %ptr_gl_out_v4 %OpCompositeConstruct %v4float %edge5 %float_1
OpStore %ptr_out_color %dark_color
OpEmitVertex
OpEndPrimitive

; ============================================================
; Strip 5: center, edge5, edge0
; ============================================================
; Vertex 0: center
OpStore %ptr_gl_out_v4 %center_vec4
OpStore %ptr_out_color %color_0
OpEmitVertex

; Vertex 1: edge5
OpStore %ptr_gl_out_v4 %OpCompositeConstruct %v4float %edge5 %float_1
OpStore %ptr_out_color %dark_color
OpEmitVertex

; Vertex 2: edge0
OpStore %ptr_gl_out_v4 %OpCompositeConstruct %v4float %edge0 %float_1
OpStore %ptr_out_color %dark_color
OpEmitVertex
OpEndPrimitive

; ============================================================
; End function
; ============================================================
OpReturn
OpFunctionEnd
