```javascript
const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { machineCount, tasks } = data;
const n = tasks.length;

// Map task id to index
const idToIndex = {};
tasks.forEach((task, idx) => {
    idToIndex[task.id] = idx;
});

// Build dependency graph and in-degree array
const graph = Array.from({ length: n }, () => []);
const inDegree = new Array(n).fill(0);
tasks.forEach((task, idx) => {
    if (task.deps) {
        task.deps.forEach(depId => {
            const depIdx = idToIndex[depId];
            graph[depIdx].push(idx);
            inDegree[idx]++;
        });
    }
});

// Topological sort (Kahn's algorithm)
const queue = [];
for (let i = 0; i < n; i++) {
    if (inDegree[i] === 0) {
        queue.push(i);
    }
}

const topoOrder = [];
while (queue.length) {
    const u = queue.shift();
    topoOrder.push(u);
    for (const v of graph[u]) {
        inDegree[v]--;
        if (inDegree[v] === 0) {
            queue.push(v);
        }
    }
}

// If cycle exists (should not happen per problem), fallback to original order
if (topoOrder.length !== n) {
    for (let i = 0; i < n; i++) topoOrder.push(i);
}

// Greedy list scheduling: assign tasks in topo order to earliest available machine
const machineAvailable = new Array(machineCount).fill(0);
const schedule = [];

for (const taskIdx of topoOrder) {
    const task = tasks[taskIdx];
    // Find machine that becomes available the earliest
    let bestMachine = 0;
    let minAvailable = machineAvailable[0];
    for (let m = 1; m < machineCount; m++) {
        if (machineAvailable[m] < minAvailable) {
            minAvailable = machineAvailable[m];
            bestMachine = m;
        }
    }
    const startTime = minAvailable;
    schedule.push({
        id: task.id,
        start: startTime,
        machine: bestMachine
    });
    machineAvailable[bestMachine] = startTime + task.duration;
}

// Output schedule
const output = { schedule };
process.stdout.write(JSON.stringify(output));
```
