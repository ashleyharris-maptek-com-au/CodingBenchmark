```ts
declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

interface DroneStart { id: number; x: number; y: number; }

const probability: number[][] = data.probability;
const drones: DroneStart[] = data.drones;
const maxStepsPerDrone: number = data.maxStepsPerDrone;

const rows = probability.length;
const cols = rows > 0 ? probability[0].length : 0;

const directions = [
  [0, 0],   // stay
  [0, 1],   // right
  [0, -1],  // left
  [1, 0],   // down
  [-1, 0]   // up
];

function isValid(x: number, y: number): boolean {
  return x >= 0 && x < rows && y >= 0 && y < cols && probability[x][y] !== -1;
}

const paths: number[][][] = [];

for (const drone of drones) {
  const path: number[][] = [[drone.x, drone.y]];
  let x = drone.x;
  let y = drone.y;
  const visited = new Set<string>();
  visited.add(`${x},${y}`);

  for (let step = 0; step < maxStepsPerDrone; step++) {
    let bestNext: [number, number] | null = null;
    let bestValue = -1;

    for (const [dx, dy] of directions) {
      const nx = x + dx;
      const ny = y + dy;
      if (!isValid(nx, ny)) continue;
      const key = `${nx},${ny}`;
      if (visited.has(key)) continue;
      const val = probability[nx][ny];
      if (val > bestValue) {
        bestValue = val;
        bestNext = [nx, ny];
      }
    }

    if (bestNext === null) {
      // No unvisited valid moves; stay in place
      path.push([x, y]);
    } else {
      [x, y] = bestNext;
      visited.add(`${x},${y}`);
      path.push([x, y]);
    }
  }

  paths.push(path);
}

console.log(JSON.stringify({ paths }));
```
