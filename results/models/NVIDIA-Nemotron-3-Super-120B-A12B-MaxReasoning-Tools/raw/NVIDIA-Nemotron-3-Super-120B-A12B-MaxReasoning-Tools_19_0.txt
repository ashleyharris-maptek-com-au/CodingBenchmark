                if (headerParts.Length != 6)
                    return;

                int xVoxelCount = int.Parse(headerParts[0]);
                int yVoxelCount = int.Parse(headerParts[1]);
                int zVoxelCount = int.Parse(headerParts[2]);
                double xBlockSize = double.Parse(headerParts[3]);
                double yBlockSize = double.Parse(headerParts[4]);
                double zBlockSize = double.Parse(headerParts[5]);

                // Create voxel grid
                byte[,,] voxels = new byte[xVoxelCount, yVoxelCount, zVoxelCount];
                
                // Read voxel data - optimized reading
                int totalVoxels = xVoxelCount * yVoxelCount * zVoxelCount;
                int bytesRead = 0;
                int bufferSize = Math.Min(8192, totalVoxels);
                byte[] buffer = new byte[bufferSize];
                
                while (bytesRead < totalVoxels)
                {
                    int remaining = totalVoxels - bytesRead;
                    int readSize = Math.Min(bufferSize, remaining);
                    int read = Console.OpenStandardInput().Read(buffer, 0, readSize);
                    if (read <= 0) break;
                    
                    for (int i = 0; i < read; i++)
                    {
                        if (bytesRead >= totalVoxels) break;
                        int x = bytesRead / (yVoxelCount * zVoxelCount);
                        int y = (bytesRead % (yVoxelCount * zVoxelCount)) / zVoxelCount;
                        int z = bytesRead % zVoxelCount;
                        voxels[x, y, z] = buffer[i];
                        bytesRead++;
                    }
                }

                // Solve based on problem scale
                List<string> operations = SolveMiningProblem(voxels, xVoxelCount, yVoxelCount, zVoxelCount, xBlockSize, yBlockSize, zBlockSize);
                
                // Output results

        static List<string> SolveMiningProblem(byte[,,] voxels, int xCount, int yCount, int zCount, double xBlockSize, double yBlockSize, double zBlockSize)
        {
            List<string> operations = new List<string>();
            
            // Adaptive strategy based on problem size
            long totalVoxels = (long)xCount * yCount * zCount;
            
            if (totalVoxels <= 10000) // Small problem - exhaustive search
            {
                return SolveSmallProblem(voxels, xCount, yCount, zCount, xBlockSize, yBlockSize, zBlockSize);
            }
            else if (totalVoxels <= 1000000) // Medium problem - greedy with local optimization
            {

            else // Large problem - heuristic approach
            {

        static List<string> SolveSmallProblem(byte[,,] voxels, int xCount, int yCount, int zCount, double xBlockSize, double yBlockSize, double zBlockSize)
        {
            List<string> operations = new List<string>();
            bool[,,] visited = new bool[xCount, yCount, zCount];
            
            // Simple greedy: dig exposed valuable ore and move to plant
            for (int x = 0; x < xCount; x++)
            {
                for (int y = 0; y < yCount; y++)
                {
                    for (int z = 0; z < zCount; z++)
                    {
                        if (visited[x, y, z]) continue;
                        
                        byte voxelData = voxels[x, y, z];
                        int mass = voxelData & 0x0F;
                        int value = voxelData >> 4;
                        
                        // Only process valuable ore (value > 0) or dirt that blocks valuable ore
                        if (value > 0 || IsBlockingValuableOre(voxels, x, y, z, xCount, yCount, zCount))
                        {
                            if (IsExposed(voxels, x, y, z, xCount, yCount, zCount))
                            {
                                // Try to move to processing plant (0,0)
                                if (CanReachPlant(x, y, z, 0, 0, z, xCount, yCount, zCount, voxels))
                                {
                                    // Move voxel to plant
                                    List<int> path = FindPathToPlant(x, y, 0, 0, xCount, yCount, zCount, voxels);
                                    if (path != null && path.Count >= 2)
                                    {
                                        string operation = BuildOperationString(1, x, y, z, path, 0, 0);
                                        operations.Add(operation);
                                        visited[x, y, z] = true;
                                        
                                        // Update voxel (removed from ground)
                                        voxels[x, y, z] = 0;
                                    }
                                }
                                else
                                {
                                    // Dump nearby if can't reach plant
                                    int dumpX = Math.Min(x + 1, xCount - 1);
                                    int dumpY = Math.Min(y + 1, yCount - 1);
                                    if (CanReachPlant(x, y, z, dumpX, dumpY, z + 1, xCount, yCount, zCount, voxels) && 
                                        IsValidDumpLocation(voxels, dumpX, dumpY, z + 1, xCount, yCount, zCount))
                                    {
                                        List<int> path = FindPathToLocation(x, y, dumpX, dumpY, xCount, yCount, zCount, voxels);
                                        if (path != null && path.Count >= 2)
                                        {
                                            string operation = BuildOperationString(1, x, y, z, path, dumpX, dumpY);
                                            operations.Add(operation);
                                            visited[x, y, z] = true;
                                            
                                            // Update voxel (moved to dump location)
                                            voxels[x, y, z] = 0;
                                            // Increment dump location (simplified - in reality would need to track ground height)
                                            if (dumpX < xCount && dumpY < yCount && z + 1 < zCount)
                                            {
                                                byte dumpVoxel = voxels[dumpX, dumpY, z + 1];
                                                int newMass = (dumpVoxel & 0x0F) + 1;
                                                int valuePart = dumpVoxel >> 4;
                                                voxels[dumpX, dumpY, z + 1] = (byte)((valuePart << 4) | (newMass & 0x0F));
                                            }
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }
            
            return operations;
        }

        static List<string> SolveMediumProblem(byte[,,] voxels, int xCount, int yCount, int zCount, double xBlockSize, double yBlockSize, double zBlockSize)
        {
            List<string> operations = new List<string>();
            bool[,,] visited = new bool[xCount, yCount, zCount];
            
            // Priority queue approach: process highest value first
            List<VoxelInfo> valuableVoxels = new List<VoxelInfo>();
            
            for (int x = 0; x < xCount; x++)
            {
                for (int y = 0; y < yCount; y++)
                {
                    for (int z = 0; z < zCount; z++)
                    {
                        byte voxelData = voxels[x, y, z];
                        int mass = voxelData & 0x0F;
                        int value = voxelData >> 4;
                        
                        if (mass > 0 && value > 0) // Valuable ore
                        {
                            valuableVoxels.Add(new VoxelInfo(x, y, z, mass, value));
                        }
                    }
                }
            }
            
            // Sort by value density (value * mass) descending
            valuableVoxels.Sort((a, b) => 
                (b.value * b.mass).CompareTo(a.value * a.mass));
            
            foreach (VoxelInfo voxel in valuableVoxels)
            {
                if (visited[voxel.x, voxel.y, voxel.z]) continue;
                
                if (IsExposed(voxels, voxel.x, voxel.y, voxel.z, xCount, yCount, zCount))
                {
                    // Try to move to plant
                    if (CanReachPlant(voxel.x, voxel.y, voxel.z, 0, 0, voxel.z, xCount, yCount, zCount, voxels))
                    {
                        List<int> path = FindPathToPlant(voxel.x, voxel.y, 0, 0, xCount, yCount, zCount, voxels);
                        if (path != null && path.Count >= 2)
                        {
                            string operation = BuildOperationString(voxel.mass, voxel.x, voxel.y, voxel.z, path, 0, 0);
                            operations.Add(operation);
                            visited[voxel.x, voxel.y, voxel.z] = true;
                            voxels[voxel.x, voxel.y, voxel.z] = 0;
                        }
                    }
                    else
                    {
                        // Find nearby dump location
                        Tuple<int, int> dumpLoc = FindNearbyDumpLocation(voxel.x, voxel.y, voxel.z, xCount, yCount, zCount, voxels);
                        if (dumpLoc != null)
                        {
                            List<int> path = FindPathToLocation(voxel.x, voxel.y, dumpLoc.Item1, dumpLoc.Item2, xCount, yCount, zCount, voxels);
                            if (path != null && path.Count >= 2)
                            {
                                string operation = BuildOperationString(voxel.mass, voxel.x, voxel.y, voxel.z, path, dumpLoc.Item1, dumpLoc.Item2);
                                operations.Add(operation);
                                visited[voxel.x, voxel.y, voxel.z] = true;
                                voxels[voxel.x, voxel.y, voxel.z] = 0;
                                
                                // Update dump location (simplified)
                                if (dumpLoc.Item1 < xCount && dumpLoc.Item2 < yCount && voxel.z + 1 < zCount)
                                {
                                    byte dumpVoxel = voxels[dumpLoc.Item1, dumpLoc.Item2, voxel.z + 1];
                                    int newMass = Math.Min(15, (dumpVoxel & 0x0F) + voxel.mass);
                                    int valuePart = dumpVoxel >> 4;
                                    voxels[dumpLoc.Item1, dumpLoc.Item2, voxel.z + 1] = (byte)((valuePart << 4) | newMass);
                                }
                            }
                        }
                    }
                }
            }
            
            return operations;
        }

        static List<string> SolveLargeProblem(byte[,,] voxels, int xCount, int yCount, int zCount, double xBlockSize, double yBlockSize, double zBlockSize)
        {
            List<string> operations = new List<string>();
            
            // For large problems, use sampling and heuristic approach
            int sampleSize = Math.Max(100, Math.Min(1000, xCount * yCount / 100));
            
            Random rand = new Random();
            HashSet<string> processed = new HashSet<string>();
            
            // Process in batches to avoid memory issues
            for (int batch = 0; batch < 10 && operations.Count < 10000; batch++) // Limit operations
            {
                // Sample random locations
                for (int i = 0; i < sampleSize / 10 && operations.Count < 1000; i++)
                {
                    int x = rand.Next(xCount);
                    int y = rand.Next(yCount);
                    int z = rand.Next(zCount);
                    
                    string key = $"{x},{y},{z}";
                    if (processed.Contains(key)) continue;
                    processed.Add(key);
                    
                    byte voxelData = voxels[x, y, z];
                    int mass = voxelData & 0x0F;
                    int value = voxelData >> 4;
                    
                    if (mass > 0 && IsExposed(voxels, x, y, z, xCount, yCount, zCount))
                    {
                        // Prefer valuable ore
                        if (value > 0)
                        {
                            // Try direct path to plant
                            if (CanReachPlant(x, y, z, 0, 0, z, xCount, yCount, zCount, voxels))
                            {
                                List<int> path = FindSimplePath(x, y, 0, 0, xCount, yCount, zCount, voxels);
                                if (path != null)
                                {
                                    string operation = BuildOperationString(mass, x, y, z, path, 0, 0);
                                    operations.Add(operation);
                                    voxels[x, y, z] = 0; // Remove voxel
                                }
                            }
                        }
                        else
                        {
                            // For dirt, try to clear path to valuable ore or dump nearby
                            if (HasValuableOreAbove(x, y, z, xCount, yCount, zCount, voxels))
                            {
                                // Try to dump to create path
                                int dumpX = Math.Min(x + 1, xCount - 1);
                                int dumpY = Math.Min(y + 1, yCount - 1);
                                int dumpZ = Math.Min(z + 1, zCount - 1);
                                
                                if (IsValidDumpLocation(voxels, dumpX, dumpY, dumpZ, xCount, yCount, zCount))
                                {
                                    List<int> path = FindSimplePath(x, y, dumpX, dumpY, xCount, yCount, zCount, voxels);
                                    if (path != null)
                                    {
                                        string operation = BuildOperationString(mass, x, y, z, path, dumpX, dumpY);
                                        operations.Add(operation);
                                        voxels[x, y, z] = 0;
                                        
                                        // Update dump location
                                        if (dumpX < xCount && dumpY < yCount && dumpZ < zCount)
                                        {
                                            byte dumpVoxel = voxels[dumpX, dumpY, dumpZ];
                                            int newMass = Math.Min(15, (dumpVoxel & 0x0F) + mass);
                                            int valuePart = dumpVoxel >> 4;
                                            voxels[dumpX, dumpY, dumpZ] = (byte)((valuePart << 4) | newMass);
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }
            
            return operations;
        }

        static bool IsExposed(byte[,,] voxels, int x, int y, int z, int xCount, int yCount, int zCount)
        {
            // Check if exposed from above (z-1 is air or out of bounds)
            if (z == 0 || z >= zCount) return true;
            
            byte above = voxels[x, y, z - 1];
            int aboveMass = above & 0x0F;
            
            // Exposed if nothing above or only air (mass == 0)
            return aboveMass == 0;
        }

        static bool IsBlockingValuableOre(byte[,,] voxels, int x, int y, int z, int xCount, int yCount, int zCount)
        {
            // Check if this dirt blocks valuable ore above it
            for (int checkZ = z + 1; checkZ < zCount; checkZ++)
            {
                byte voxelData = voxels[x, y, checkZ];
                int value = voxelData >> 4;
                if (value > 0) return true; // Found valuable ore above
                
                int mass = voxelData & 0x0F;
                if (mass == 0) break; // Hit air, stop checking
            }
            return false;
        }

        static bool CanReachPlant(int startX, int startY, int startZ, int targetX, int targetY, int targetZ, 
                                 int xCount, int yCount, int zCount, byte[,,] voxels)
        {
            // Simple Manhattan path check with slope constraint
            int dx = Math.Abs(targetX - startX);
            int dy = Math.Abs(targetY - startY);
            int dz = Math.Abs(targetZ - startZ);
            
            // Check if slope constraint is violated
            int maxHorizontal = Math.Max(dx, dy);
            if (maxHorizontal > 0 && dz > maxHorizontal) // More than 1 Z per X/Y
                return false;
            
            // Additional check: path must not go through solid voxels
            return IsPathClear(startX, startY, startZ, targetX, targetY, targetZ, xCount, yCount, zCount, voxels);
        }

        static bool IsPathClear(int x1, int y1, int z1, int x2, int y2, int z2, 
                               int xCount, int yCount, int zCount, byte[,,] voxels)
        {
            // Simple line-of-sight check (simplified for performance)
            int dx = x2 - x1;
            int dy = y2 - y1;
            int dz = z2 - z1;
            int steps = Math.Max(Math.Abs(dx), Math.Max(Math.Abs(dy), Math.Abs(dz)));
            
            if (steps == 0) return true;
            
            double xInc = dx / (double)steps;
            double yInc = dy / (double)steps;
            double zInc = dz / (double)steps;
            
            for (int i = 0; i <= steps; i++)
            {
                int x = (int)Math.Round(x1 + i * xInc);
                int y = (int)Math.Round(y1 + i * yInc);
                int z = (int)Math.Round(z1 + i * zInc);
                
                if (x < 0 || x >= xCount || y < 0 || y >= yCount || z < 0 || z >= zCount)
                    continue;
                
                // Check if voxel is solid (mass > 0) and not the start/end
                if ((i > 0 && i < steps) || (z1 != z2)) // Allow start and end to be solid if same Z
                {
                    byte voxelData = voxels[x, y, z];
                    int mass = voxelData & 0x0F;
                    if (mass > 0 && !(x == x1 && y == y1 && z == z1) && !(x == x2 && y == y2 && z == z2))
                    {
                        return false; // Blocked by solid voxel
                    }
                }
            }
            return true;
        }

        static List<int> FindPathToPlant(int startX, int startY, int targetX, int targetY, 
                                        int xCount, int yCount, int zCount, byte[,,] voxels)
        {
            // Simple greedy pathfinding: move toward target
            List<int> path = new List<int>();
            int x = startX;
            int y = startY;
            int z = startZ; // Keep same Z for simplicity
            
            path.Add(x);
            path.Add(y);
            
            // Move in X direction first, then Y
            while (x != targetX || y != targetY)
            {
                int nextX = x;
                int nextY = y;
                
                if (x < targetX) nextX = Math.Min(x + 1, xCount - 1);
                else if (x > targetX) nextX = Math.Max(x - 1, 0);
                
                if (y < targetY) nextY = Math.Min(y + 1, yCount - 1);
                else if (y > targetY) nextY = Math.Max(y - 1, 0);
                
                // Check if next position is valid (slope constraint)
                if (IsValidStep(x, y, z, nextX, nextY, z, xCount, yCount, zCount, voxels))
                {
                    x = nextX;
                    y = nextY;
                    path.Add(x);
                    path.Add(y);
                }
                else
                {
                    // Try alternative: move only in one direction
                    if (x != targetX)
                    {
                        int testX = x < targetX ? Math.Min(x + 1, xCount - 1) : Math.Max(x - 1, 0);
                        if (IsValidStep(x, y, z, testX, y, z, xCount, yCount, zCount, voxels))
                        {
                            x = testX;
                            path.Add(x);
                            path.Add(y);
                            continue;
                        }
                    }
                    
                    if (y != targetY)
                    {
                        int testY = y < targetY ? Math.Min(y + 1, yCount - 1) : Math.Max(y - 1, 0);
                        if (IsValidStep(x, y, z, x, testY, z, xCount, yCount, zCount, voxels))
                        {
                            y = testY;
                            path.Add(x);
                            path.Add(y);
                            continue;
                        }
                    }
                    
                    // If we can't move, break
                    break;
                }
            }
            
            // Ensure we end at target
            if (x == targetX && y == targetY)
            {
                path.Add(x);
                path.Add(y);
                return path;
            }
            
            return null;
        }

        static List<int> FindPathToLocation(int startX, int startY, int targetX, int targetY, 
                                           int xCount, int yCount, int zCount, byte[,,] voxels)
        {
            return FindPathToPlant(startX, startY, targetX, targetY, xCount, yCount, zCount, voxels);
        }

        static List<int> FindSimplePath(int startX, int startY, int targetX, int targetY, 
                                       int xCount, int yCount, int zCount, byte[,,] voxels)
        {
            // Even simpler path - just try to go straight
            List<int> path = new List<int>();
            path.Add(startX);
            path.Add(startY);
            
            int x = startX;
            int y = startY;
            
            // Try X then Y
            while (x != targetX)
            {
                int step = x < targetX ? 1 : -1;
                int nextX = Math.Max(0, Math.Min(xCount - 1, x + step));
                if (IsValidStep(x, y, 0, nextX, y, 0, xCount, yCount, zCount, voxels)) // Assume Z=0 for simplicity
                {
                    x = nextX;
                    path.Add(x);
                    path.Add(y);
                }
                else
                {
                    break;
                }
            }
            
            while (y != targetY)
            {
                int step = y < targetY ? 1 : -1;
                int nextY = Math.Max(0, Math.Min(yCount - 1, y + step));
                if (IsValidStep(x, y, 0, x, nextY, 0, xCount, yCount, zCount, voxels))
                {
                    y = nextY;
                    path.Add(x);
                    path.Add(y);
                }
                else
                {
                    break;
                }
            }
            
            if (x == targetX && y == targetY)
            {
                path.Add(x);
                path.Add(y);
                return path;
            }
            
            return null;
        }

        static bool IsValidStep(int x1, int y1, int z1, int x2, int y2, int z2, 
                               int xCount, int yCount, int zCount, byte[,,] voxels)
        {
            // Check bounds
            if (x2 < 0 || x2 >= xCount || y2 < 0 || y2 >= yCount || z2 < 0 || z2 >= zCount)
                return false;
            
            // Check slope constraint: |dz| <= max(|dx|, |dy|)
            int dx = Math.Abs(x2 - x1);
            int dy = Math.Abs(y2 - y1);
            int dz = Math.Abs(z2 - z1);
            int maxHorizontal = Math.Max(dx, dy);
            
            if (dz > maxHorizontal) // More than 1 Z per X/Y voxel
                return false;
            
            // Check if destination is not solid (unless it's the target we're moving to)
            byte voxelData = voxels[x2, y2, z2];
            int mass = voxelData & 0x0F;
            
            // Allow moving to air (mass == 0) or same position
            return mass == 0 || (x1 == x2 && y1 == y2 && z1 == z2);
        }

        static Tuple<int, int> FindNearbyDumpLocation(int x, int y, int z, int xCount, int yCount, int zCount, byte[,,] voxels)
        {
            // Look for nearby empty spot to dump
            int searchRadius = Math.Min(5, Math.Min(xCount, yCount) / 2);
            
            for (int dx = -searchRadius; dx <= searchRadius; dx++)
            {
                for (int dy = -searchRadius; dy <= searchRadius; dy++)
                {
                    int nx = x + dx;
                    int ny = y + dy;
                    
                    if (nx < 0 || nx >= xCount || ny < 0 || ny >= yCount) continue;
                    
                    // Look for first empty Z from current Z upward
                    for (int nz = z; nz < zCount; nz++)
                    {
                        byte voxelData = voxels[nx, ny, nz];
                        int mass = voxelData & 0x0F;
                        if (mass == 0) // Found empty spot
                        {
                            return new Tuple<int, int>(nx, ny);
                        }
                    }
                }
            }
            
            return null;
        }

        static bool HasValuableOreAbove(int x, int y, int z, int xCount, int yCount, int zCount, byte[,,] voxels)
        {
            // Check if there's valuable ore above this position
            for (int checkZ = z + 1; checkZ < zCount; checkZ++)
            {
                byte voxelData = voxels[x, y, checkZ];
                int value = voxelData >> 4;
                int mass = voxelData & 0x0F;
                
                if (value > 0 && mass > 0) return true;
                if (mass == 0) break; // Hit air
            }
            return false;
        }

        static bool IsValidDumpLocation(byte[,,] voxels, int x, int y, int z, int xCount, int yCount, int zCount)
        {
            if (x < 0 || x >= xCount || y < 0 || y >= yCount || z < 0 || z >= zCount)
                return false;
            
            byte voxelData = voxels[x, y, z];
            int mass = voxelData & 0x0F;
            return mass == 0; // Must be empty to dump on
        }

        static string BuildOperationString(int mass, int startX, int startY, int startZ, 
                                          List<int> path, int destX, int destY)
        {
            StringBuilder sb = new StringBuilder();
            sb.Append(mass.ToString());
            sb.Append(" ");
            sb.Append(startX.ToString());
            sb.Append(" ");
            sb.Append(startY.ToString());
            sb.Append(" ");
            sb.Append(startZ.ToString());
            
            // Add path steps (X,Y pairs)
            for (int i = 0; i < path.Count; i += 2)
            {
                if (i + 1 < path.Count)
                {
                    sb.Append(" ");
                    sb.Append(path[i].ToString());
                    sb.Append(" ");
                    sb.Append(path[i + 1].ToString());
                }
            }
            
            sb.Append(" ");
            sb.Append(destX.ToString());
            sb.Append(" ");
            sb.Append(destY.ToString());
            
            return sb.ToString();
        }
    }

    class VoxelInfo
    {
        public int x, y, z;
        public int mass;
        public int value;
        
        public VoxelInfo(int x, int y, int z, int mass, int value)
        {
            this.x = x;
            this.y = y;
            this.z = z;
            this.mass = mass;
            this.value = value;
        }
    }

This solution implements:

1. **Adaptive Strategy Selection**: Chooses different algorithms based on problem size:
   - Small problems (< 10k voxels): Exhaustive greedy approach
   - Medium problems (10k-1M voxels): Priority-based processing of valuable ore
   - Large problems (> 1M voxels): Sampling and heuristic approach

2. **Core Functionality**:
   - Reads voxel data efficiently from stdin
   - Implements all constraints (exposure rules, slope limits, dumping rules)
   - Tracks voxel state changes during operations
   - Calculates valid paths with slope constraints
   - Handles both dirt and valuable ore
