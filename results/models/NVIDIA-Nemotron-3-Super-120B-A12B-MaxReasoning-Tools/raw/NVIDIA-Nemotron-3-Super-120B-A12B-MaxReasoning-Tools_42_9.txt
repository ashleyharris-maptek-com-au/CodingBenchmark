```spirv
; SPIR-V Geometry Shader for thick smooth curve using adjacency (InputLinesAdjacency)
; Capabilities
OpCapability Shader
OpCapability Geometry

; Memory model
OpMemoryModel Logical GLSL450

; Entry point with all interface variables
OpEntryPoint Geometry %main "main" %gl_in %gl_out %ubo %outColor

; Execution modes
OpExecutionMode %main InputLinesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1

; Types
%void = OpTypeVoid
%bool = OpTypeBool
%float = OpTypeFloat 32
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4

; Constants
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3
%float_0 = OpConstant %float 0.0
%float_1 = OpConstant %float 1.0
%float_0_025 = OpConstant %float 0.025

; UBO layout (set=0, binding=0) - from test 41
%float4 = OpTypeVector %float 4
%array_float4 = OpTypeArray %float4 %int_6
%ptr_ubo_uniform = OpTypePointer Uniform %array_float4
%ubo = OpVariable %ptr_ubo_uniform Uniform

; Input per-vertex block (gl_PerVertex)
%gl_PerVertex_in = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position

; Input array: 4 vertices (InputLinesAdjacency)
%arr_pv_in = OpTypeArray %gl_PerVertex_in %int_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

; User inputs: arrays of size 4
%ptr_in_worldPos = OpTypePointer Input %v3float
%worldPos = OpVariable %ptr_in_worldPos Input
OpDecorate %worldPos Location 0

%ptr_in_normal = OpTypePointer Input %v3float
%normal = OpVariable %ptr_in_normal Input
OpDecorate %normal Location 1

%ptr_in_color = OpTypePointer Input %v3float
%color = OpVariable %ptr_in_color Input
OpDecorate %color Location 2

; Output per-vertex block (gl_PerVertex)
%gl_PerVertex_out = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

; User output
%ptr_out_outColor = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_outColor Output
OpDecorate %outColor Location 0

; Function: main
%void_fn = OpTypeFunction %void
%main = OpFunction %void None %void_fn
%entry = OpLabel

; === Compute tangent and perpendiculars ===
; t0 = normalize(gl_in[2].pos - gl_in[0].pos)
; Access gl_in[0]
%ptr_gl_in_0 = OpAccessChain %ptr_in_arr_pv %gl_in %int_0 %int_0
%gl_in_0_pos = OpLoad %v4float %ptr_gl_in_0
%ptr_gl_in_2 = OpAccessChain %ptr_in_arr_pv %gl_in %int_2 %int_0
%gl_in_2_pos = OpLoad %v4float %ptr_gl_in_2
; diff0 = gl_in[2].xyz - gl_in[0].xyz
%diff0_x = OpFSub %float OpCompositeExtract %gl_in_2_pos 0 OpCompositeExtract %gl_in_0_pos 0
%diff0_y = OpFSub %float OpCompositeExtract %gl_in_2_pos 1 OpCompositeExtract %gl_in_0_pos 1
%diff0_z = OpFSub %float OpCompositeExtract %gl_in_2_pos 2 OpCompositeExtract %gl_in_0_pos 2
%diff0 = OpCompositeConstruct %v3float %diff0_x %diff0_y %diff0_z
; length0 = sqrt(dot(diff0, diff0))
%dot0 = OpFDot %float %diff0 %diff0
%length0 = OpFsqrt %float %dot0
; t0 = diff0 / length0
%t0_x = OpFDiv %float %diff0_x %length0
%t0_y = OpFDiv %float %diff0_y %length0
%t0_z = OpFDiv %float %diff0_z %length0
%t0 = OpCompositeConstruct %v3float %t0_x %t0_y %t0_z
; perp0 = (-t0.y, t0.x) in 2D
%perp0_x = OpFNeg %float %t0_y
%perp0_y = OpFSub %float %t0_x %float_0  ; t0.x
%perp0 = OpCompositeConstruct %v2float %perp0_x %perp0_y

; t1 = normalize(gl_in[3].pos - gl_in[1].pos)
; Access gl_in[1]
%ptr_gl_in_1 = OpAccessChain %ptr_in_arr_pv %gl_in %int_1 %int_0
%gl_in_1_pos = OpLoad %v4float %ptr_gl_in_1
; Access gl_in[3]
%ptr_gl_in_3 = OpAccessChain %ptr_in_arr_pv %gl_in %int_3 %int_0
%gl_in_3_pos = OpLoad %v4float %ptr_gl_in_3
; diff1 = gl_in[3].xyz - gl_in[1].xyz
%diff1_x = OpFSub %float OpCompositeExtract %gl_in_3_pos 0 OpCompositeExtract %gl_in_1_pos 0
%diff1_y = OpFSub %float OpCompositeExtract %gl_in_3_pos 1 OpCompositeExtract %gl_in_1_pos 1
%diff1_z = OpFSub %float OpCompositeExtract %gl_in_3_pos 2 OpCompositeExtract %gl_in_1_pos 2
%diff1 = OpCompositeConstruct %v3float %diff1_x %diff1_y %diff1_z
; length1 = sqrt(dot(diff1, diff1))
%dot1 = OpFDot %float %diff1 %diff1
%length1 = OpFsqrt %float %dot1
; t1 = diff1 / length1
%t1_x = OpFDiv %float %diff1_x %length1
%t1_y = OpFDiv %float %diff1_y %length1
%t1_z = OpFDiv %float %diff1_z %length1
%t1 = OpCompositeConstruct %v3float %t1_x %t1_y %t1_z
; perp1 = (-t1.y, t1.x) in 2D
%perp1_x = OpFNeg %float %t1_y
%perp1_y = OpFSub %float %t1_x %float_0  ; t1.x
%perp1 = OpCompositeConstruct %v2float %perp1_x %perp1_y

; === Emit 4 vertices as triangle strip ===
; Vertex 0: start + perp0 * thickness (gl_in[1])
%start_pos = OpCompositeConstruct %v3float OpCompositeExtract %gl_in_1_pos 0 OpCompositeExtract %gl_in_1_pos 1 OpCompositeExtract %gl_in_1_pos 2
%perp0_scaled_x = OpFMul %float %perp0_x %float_0_025
%perp0_scaled_y = OpFMul %float %perp0_y %float_0_025
%offset0 = OpCompositeConstruct %v2float %perp0_scaled_x %perp0_scaled_y
%v0_pos_xy = OpFAdd %v2float OpCompositeConstruct %v2float OpCompositeExtract %start_pos 0 OpCompositeExtract %start_pos 1 %offset0
%v0_pos = OpCompositeConstruct %v4float %v0_pos_xy OpCompositeExtract %start_pos 2 %float_1
; Access chain for output
%ptr_gl_out = OpAccessChain %ptr_out_pv %gl_out %int_0
%ptr_out_v4 = OpAccessChain %ptr_gl_out %ptr_gl_out %int_0
%ptr_out_pos = OpAccessChain %ptr_out_v4 %ptr_out_v4 %int_0
OpStore %ptr_out_pos %v0_pos
; outColor = color[1]
%ptr_color_1 = OpAccessChain %ptr_in_color %color %int_1
%color1 = OpLoad %v3float %ptr_color_1
OpStore %outColor %color1
OpEmitVertex

; Vertex 1: start - perp0 * thickness (gl_in[1])
%v1_pos_xy = OpFSub %v2float OpCompositeConstruct %v2float OpCompositeExtract %start_pos 0 OpCompositeExtract %start_pos 1 %offset0
%v1_pos = OpCompositeConstruct %v4float %v1_pos_xy OpCompositeExtract %start_pos 2 %float_1
OpStore %ptr_out_pos %v1_pos
OpStore %outColor %color1
OpEmitVertex

; Vertex 2: end + perp1 * thickness (gl_in[2])
%end_pos = OpCompositeConstruct %v3float OpCompositeExtract %gl_in_2_pos 0 OpCompositeExtract %gl_in_2_pos 1 OpCompositeExtract %gl_in_2_pos 2
%perp1_scaled_x = OpFMul %float %perp1_x %float_0_025
%perp1_scaled_y = OpFMul %float %perp1_y %float_0_025
%offset1 = OpCompositeConstruct %v2float %perp1_scaled_x %perp1_scaled_y
%v2_pos_xy = OpFAdd %v2float OpCompositeConstruct %v2float OpCompositeExtract %end_pos 0 OpCompositeExtract %end_pos 1 %offset1
%v2_pos = OpCompositeConstruct %v4float %v2_pos_xy OpCompositeExtract %end_pos 2 %float_1
OpStore %ptr_out_pos %v2_pos
; outColor = color[2]
%ptr_color_2 = OpAccessChain %ptr_in_color %color %int_2
%color2 = OpLoad %v3float %ptr_color_2
OpStore %outColor %color2
OpEmitVertex

; Vertex 3: end - perp1 * thickness (gl_in[2])
%v3_pos_xy = OpFSub %v2float OpCompositeConstruct %v2float OpCompositeExtract %end_pos 0 OpCompositeExtract %end_pos 1 %offset1
%v3_pos = OpCompositeConstruct %v4float %v3_pos_xy OpCompositeExtract %end_pos 2 %float_1
OpStore %ptr_out_pos %v3_pos
OpStore %outColor %color2
OpEmitVertex

; End primitive
OpEndPrimitive
OpReturn
OpFunctionEnd
```
