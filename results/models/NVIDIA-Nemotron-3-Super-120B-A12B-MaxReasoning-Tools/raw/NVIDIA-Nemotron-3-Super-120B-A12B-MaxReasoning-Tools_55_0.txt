To solve this problem, we need to create a chess bot that, given a chess position in FEN notation and a timeout in seconds, returns the best move possible within the given time. The bot should use a combination of iterative deepening, alpha-beta pruning with transposition tables, and move ordering to efficiently search the game tree and find the best move.

### Approach
1. **FEN Parsing**: Parse the input FEN string to reconstruct the chessboard state, including piece positions, castling rights, en passant target, and whose turn it is.
2. **Move Generation**: Generate all legal moves for the current player, considering all piece movements, captures, promotions, castling, and en passant.
3. **Move Ordering**: Prioritize moves that are likely to be good (captures, checks, promotions, castling) to improve the efficiency of alpha-beta pruning.
4. **Alpha-Beta Search with Transposition Table**: Use iterative deepening to search deeper into the game tree within the time limit. The transposition table stores previously evaluated positions to avoid redundant calculations.
5. **Evaluation Function**: Evaluate the board state based on material balance, piece-square tables (to encourage central control and development), and king safety (penalizing exposed kings).
6. **Time Management**: Monitor the elapsed time during the search and terminate iterations if the timeout is exceeded, returning the best move found so far.

void init_board(BoardState& state, const string& fen) {
    int row = 0, col = 0;
    for (char c : fen) {
        if (c == ' ') break;
        if (c == '/') {
            row++;
            col = 0;
        } else if (isdigit(c)) {
            int num = c - '0';
            for (int i = 0; i < num; i++) {
                state.board[row][col++] = '.';
            }
        } else {
            state.board[row][col++] = c;
        }
    }

    while (fen[0] != ' ') fen.erase(fen.begin());
    fen.erase(fen.begin());

    state.white_to_move = (fen[0] == 'w');
    fen.erase(0, 2);

    state.castling_rights = 0;
    if (fen[0] != '-') {
        for (int i = 0; fen[i] != ' '; i++) {
            switch (fen[i]) {
                case 'K': state.castling_rights |= 1; break;
                case 'Q': state.castling_rights |= 2; break;
                case 'k': state.castling_rights |= 4; break;
                case 'q': state.castling_rights |= 8; break;
            }
        }
    }
    fen = fen.substr(fen.find(' ') + 1);

    if (fen[0] == '-') {
        state.en_passant = -1;
        fen = fen.substr(2);
    } else {
        int file = fen[0] - 'a';
        int rank = fen[1] - '1';
        state.en_passant = rank * 8 + file;
        fen = fen.substr(3);
    }

    state.halfmove_clock = stoi(fen.substr(0, fen.find(' ')));
    fen = fen.substr(fen.find(' ') + 1);
    state.fullmove_number = stoi(fen);

    state.zobrist_key = 0;
    for (int i = 0; i < 8; i++) {
        for (int j = 0; j < 8; j++) {
            char c = state.board[i][j];
            if (c != '.') {
                int idx = piece_to_index(c);
                state.zobrist_key ^= piece_keys[idx][i*8+j];
            }
        }
    }
    if (!state.white_to_move) {
        state.zobrist_key ^= side_key;
    }
    state.zobrist_key ^= castling_keys[state.castling_rights];
    if (state.en_passant != -1) {
        state.zobrist_key ^= en_passant_keys[state.en_passant % 8];
    }
}

bool is_square_attacked(const BoardState& state, int square, bool by_white) {
    int r = square / 8, c = square % 8;
    for (int dr = -1; dr <= 1; dr++) {
        for (int dc = -1; dc <= 1; dc++) {
            if (dr == 0 && dc == 0) continue;
            int nr = r + dr, nc = c + dc;
            while (nr >= 0 && nr < 8 && nc >= 0 && nc < 8) {
                char piece = state.board[nr][nc];
                if (piece != '.') {
                    bool is_white = isupper(piece);
                    if (is_white == by_white) {
                        int piece_type = tolower(piece);
                        if ((dr != 0 && dc != 0) && (piece_type == 'b' || piece_type == 'q')) {
                            return true;
                        }
                        if ((dr == 0 || dc == 0) && (piece_type == 'r' || piece_type == 'q')) {
                            return true;
                        }
                        if (abs(dr) == 1 && abs(dc) == 1 && piece_type == 'k') {
                            return true;
                        }
                    }
                    break;
                }
                nr += dr;
                nc += dc;
            }
        }
    }

    int knight_moves[8][2] = {{-2,-1},{-2,1},{-1,-2},{-1,2},{1,-2},{1,2},{2,-1},{2,1}};
    for (int i = 0; i < 8; i++) {
        int nr = r + knight_moves[i][0];
        int nc = c + knight_moves[i][1];
        if (nr >= 0 && nr < 8 && nc >= 0 && nc < 8) {
            char piece = state.board[nr][nc];
            if (piece != '.') {
                bool is_white = isupper(piece);
                if (is_white == by_white && tolower(piece) == 'n') {
                    return true;
                }
            }
        }
    }

    int pawn_dir = by_white ? -1 : 1;
    int start_rank = by_white ? 6 : 1;
    int capture_cols[2] = {-1, 1};
    for (int dc : capture_cols) {
        int nr = r + pawn_dir;
        int nc = c + dc;
        if (nr >= 0 && nr < 8 && nc >= 0 && nc < 8) {
            char piece = state.board[nr][nc];
            if (piece != '.') {
                bool is_white = isupper(piece);
                if (is_white == by_white && tolower(piece) == 'p') {
                    return true;
                }
            }
        }
    }

    return false;
}

bool is_in_check(const BoardState& state) {
    int king_sq = -1;
    char king_piece = state.white_to_move ? 'K' : 'k';
    for (int i = 0; i < 8; i++) {
        for (int j = 0; j < 8; j++) {
            if (state.board[i][j] == king_piece) {
                king_sq = i*8+j;
                break;
            }
        }
        if (king_sq != -1) break;
    }
    if (king_sq == -1) return false;
    return is_square_attacked(state, king_sq, !state.white_to_move);
}

void generate_pawn_moves(const BoardState& state, vector<Move>& moves, int from) {
    int r = from / 8, c = from % 8;
    char piece = state.board[r][c];
    bool is_white = isupper(piece);
    int dir = is_white ? -1 : 1;
    int start_row = is_white ? 6 : 1;
    int promotion_row = is_white ? 0 : 7;

    int nr = r + dir;
    if (nr >= 0 && nr < 8) {
        if (state.board[nr][c] == '.') {
            if (nr == promotion_row) {
                for (char promo : {'q','r','b','n'}) {
                    Move m;
                    m.from = from;
                    m.to = nr*8+c;
                    m.piece = piece;
                    m.promotion = true;
                    m.promote_to = toupper(promo);
                    moves.push_back(m);
                }
            } else {
                Move m;
                m.from = from;
                m.to = nr*8+c;
                m.piece = piece;
                moves.push_back(m);
            }
            if (r == start_row) {
                int nr2 = r + 2*dir;
                if (state.board[nr2][c] == '.') {
                    Move m;
                    m.from = from;
                    m.to = nr2*8+c;
                    m.piece = piece;
                    moves.push_back(m);
                }
            }
        }
    }

    int capture_cols[2] = {-1, 1};
    for (int dc : capture_cols) {
        int nc = c + dc;
        int nr = r + dir;
        if (nr >= 0 && nr < 8 && nc >= 0 && nc < 8) {
            if (state.board[nr][nc] != '.' && isupper(state.board[nr][nc]) != is_white) {
                if (nr == promotion_row) {
                    for (char promo : {'q','r','b','n'}) {
                        Move m;
                        m.from = from;
                        m.to = nr*8+nc;
                        m.piece = piece;
                        m.captured = state.board[nr][nc];
                        m.promotion = true;
                        m.promote_to = toupper(promo);
                        moves.push_back(m);
                    }
                } else {
                    Move m;
                    m.from = from;
                    m.to = nr*8+nc;
                    m.piece = piece;
                    m.captured = state.board[nr][nc];
                    moves.push_back(m);
                }
            }
        }
    }

    if (state.en_passant != -1) {
        int ep_r = state.en_passant / 8;
        int ep_c = state.en_passant % 8;
        if (abs(r - ep_r) == 1 && abs(c - ep_c) == 1) {
            bool valid = false;
            if (is_white) {
                if (r == 3 && ep_r == 2) valid = true;
            } else {
                if (r == 4 && ep_r == 5) valid = true;
            }
            if (valid) {
                Move m;
                m.from = from;
                m.to = state.en_passant;
                m.piece = piece;
                m.captured = state.board[ep_r][ep_c];
                m.en_passant = true;
                moves.push_back(m);
            }
        }
    }
}

void generate_slider_moves(const BoardState& state, vector<Move>& moves, int from, const int* dirs, int dir_count) {

    for (int i = 0; i < dir_count; i++) {
        int dr = dirs[2*i];
        int dc = dirs[2*i+1];
        int nr = r + dr, nc = c + dc;
        while (nr >= 0 && nr < 8 && nc >= 0 && nc < 8) {
            char target = state.board[nr][nc];
            if (target == '.') {
                Move m;
                m.from = from;
                m.to = nr*8+nc;
                m.piece = piece;
                moves.push_back(m);
            } else if (isupper(target) != is_white) {
                Move m;
                m.from = from;
                m.to = nr*8+nc;
                m.piece = piece;
                m.captured = target;
                moves.push_back(m);
                break;
            } else {
                break;
            }
            nr += dr;
            nc += dc;
        }
    }
}

void generate_all_moves(const BoardState& state, vector<Move>& moves) {
    moves.clear();
    for (int i = 0; i < 64; i++) {
        char piece = state.board[i/8][i%8];
        if (piece == '.') continue;
        bool is_white = isupper(piece);
        if (state.white_to_move != is_white) continue;
        switch (tolower(piece)) {
            case 'p': generate_pawn_moves(state, moves, i); break;
            case 'n': generate_knight_moves(state, moves, i); break;
            case 'b': generate_bishop_moves(state, moves, i); break;
            case 'r': generate_rook_moves(state, moves, i); break;
            case 'q': generate_queen_moves(state, moves, i); break;
            case 'k': generate_king_moves(state, moves, i); break;
        }
    }
}

int evaluate_board(const BoardState& state) {
    int score = 0;
    int material[2] = {0,0};
    int piece_count[2] = {0,0};

    for (int i = 0; i < 8; i++) {
        for (int j = 0; j < 8; j++) {
            char c = state.board[i][j];
            if (c == '.') continue;
            bool is_white = isupper(c);
            int idx = piece_to_index(c);
            int value = piece_value[c];
            score += is_white ? -value : value;
            if (is_white) {
                material[0] += value;
                piece_count[0]++;
            } else {
                material[1] += value;
                piece_count[1]++;
            }

            int table_value = 0;
            switch (tolower(c)) {
                case 'p': table_value = pawn_table[i*8+j]; break;
                case 'n': table_value = knight_table[i*8+j]; break;
                case 'b': table_value = bishop_table[i*8+j]; break;
                case 'r': table_value = rook_table[i*8+j]; break;
                case 'q': table_value = queen_table[i*8+j]; break;
                case 'k': 
                    int total_material = material[0] + material[1];
                    if (total_material < 1500) {
                        table_value = king_table_endgame[i*8+j];
                    } else {
                        table_value = king_table[i*8+j];
                    }
                    break;
            }
            score += is_white ? -table_value : table_value;
        }
    }

    int king_sq[2] = {-1,-1};
    for (int i = 0; i < 8; i++) {
        for (int j = 0; j < 8; j++) {
            if (state.board[i][j] == 'K') king_sq[0] = i*8+j;
            if (state.board[i][j] == 'k') king_sq[1] = i*8+j;
        }
    }
    for (int side = 0; side < 2; side++) {
        if (king_sq[side] == -1) continue;
        bool is_white = (side == 0);
        int attackers = 0;
        for (int i = 0; i < 8; i++) {
            for (int j = 0; j < 8; j++) {
                char c = state.board[i][j];
                if (c == '.') continue;
                bool piece_white = isupper(c);
                if (piece_white == is_white) continue;
                if (is_square_attacked(state, king_sq[side], !is_white)) {
                    attackers++;
                }
            }
        }
        if (attackers > 0) {
            score += is_white ? -50 * attackers : 50 * attackers;
        }
    }

    return score;
}

bool make_move(BoardState& state, const Move& move) {
    char piece = state.board[move.from/8][move.from%8];
    state.board[move.from/8][move.from%8] = '.';

    int captured_piece = 0;
    if (move.en_passant) {
        int ep_r = move.to/8, ep_c = move.to%8;
        int pawn_r = move.from/8, pawn_c = move.from%8;
        captured_piece = state.board[ep_r][ep_c];
        state.board[ep_r][ep_c] = '.';
    } else if (move.captured != 0) {
        captured_piece = move.captured;
    }

    if (move.promotion) {
        state.board[move.to/8][move.to%8] = move.promote_to;
    } else if (move.kingside_castle) {
        state.board[move.to/8][move.to%8] = piece;
        int rook_from = move.from/8*8 + (state.white_to_move ? 7 : 0);
        int rook_to = move.to/8*8 + (state.white_to_move ? 5 : 5);
        char rook = state.board[rook_from/8][rook_from%8];
        state.board[rook_from/8][rook_from%8] = '.';
        state.board[rook_to/8][rook_to%8] = rook;
    } else if (move.queenside_castle) {
        state.board[move.to/8][move.to%8] = piece;
        int rook_from = move.from/8*8 + (state.white_to_move ? 0 : 0);
        int rook_to = move.to/8*8 + (state.white_to_move ? 3 : 3);
        char rook = state.board[rook_from/8][rook_from%8];
        state.board[rook_from/8][rook_from%8] = '.';
        state.board[rook_to/8][rook_to%8] = rook;
    } else {
        state.board[move.to/8][move.to%8] = piece;
    }

    state.white_to_move = !state.white_to_move;
    state.halfmove_clock++;

    if (tolower(piece) == 'p' || captured_piece != 0) {
        state.halfmove_clock = 0;
    }

    uint64_t new_key = state.zobrist_key;
    new_key ^= piece_keys[piece_to_index(piece)][move.from];
    new_key ^= piece_keys[piece_to_index(piece)][move.to];
    if (move.captured != 0) {
        new_key ^= piece_keys[piece_to_index(move.captured)][move.to];
    }
    if (move.en_passant) {
        int ep_r = move.to/8, ep_c = move.to%8;
        int pawn_r = move.from/8, pawn_c = move.from%8;
        new_key ^= piece_keys[piece_to_index(captured_piece)][ep_r*8+ep_c];
    }
    if (move.promotion) {
        new_key ^= piece_keys[piece_to_index(piece)][move.to];
        new_key ^= piece_keys[piece_to_index(move.promote_to)][move.to];
    }
    if (move.kingside_castle) {
        int rook_from = move.from/8*8 + (state.white_to_move ? 7 : 0);
        int rook_to = move.to/8*8 + (state.white_to_move ? 5 : 5);
        char rook = state.board[rook_from/8][rook_from%8];
        new_key ^= piece_keys[piece_to_index(rook)][rook_from];
        new_key ^= piece_keys[piece_to_index(rook)][rook_to];
    }
    if (move.queenside_castle) {
        int rook_from = move.from/8*8 + (state.white_to_move ? 0 : 0);
        int rook_to = move.to/8*8 + (state.white_to_move ? 3 : 3);
        char rook = state.board[rook_from/8][rook_from%8];
        new_key ^= piece_keys[piece_to_index(rook)][rook_from];
        new_key ^= piece_keys[piece_to_index(rook)][rook_to];
    }

    uint64_t old_castling = state.castling_rights;
    if (state.white_to_move) {
        if (piece == 'K') {
            state.castling_rights &= ~3;
        } else if (piece == 'R') {
            if (move.from == 7*8+0) state.castling_rights &= ~2;
            if (move.from == 7*8+7) state.castling_rights &= ~1;
        }
    } else {
        if (piece == 'k') {
            state.castling_rights &= ~12;
        } else if (piece == 'r') {
            if (move.from == 0*8+0) state.castling_rights &= ~8;
            if (move.from == 0*8+7) state.castling_rights &= ~4;
        }
    }
    if (state.castling_rights != old_castling) {
        new_key ^= castling_keys[old_castling];
        new_key ^= castling_keys[state.castling_rights];
    }

    int old_ep = state.en_passant;
    state.en_passant = -1;
    if (tolower(piece) == 'p' && abs(move.from/8 - move.to/8) == 2) {
        state.en_passant = (move.from/8 + move.to/8)/2 * 8 + move.from%8;
    }
    if (old_ep != -1) {
        new_key ^= en_passant_keys[old_ep % 8];
    }
    if (state.en_passant != -1) {
        new_key ^= en_passant_keys[state.en_passant % 8];
    }

    state.zobrist_key = new_key;
    return true;
}

void unmake_move(BoardState& state, const Move& move) {
    state.white_to_move = !state.white_to_move;
    state.halfmove_clock--;

    char piece = state.board[move.to/8][move.to%8];
    if (move.promotion) {
        piece = move.piece;
    }

    state.board[move.to/8][move.to%8] = '.';

    if (move.kingside_castle) {
        state.board[move.from/8][move.from%8] = piece;
        int rook_from = move.to/8*8 + (state.white_to_move ? 5 : 5);
        int rook_to = move.from/8*8 + (state.white_to_move ? 7 : 7);
        char rook = state.board[rook_from/8][rook_from%8];
        state.board[rook_from/8][rook_from%8] = '.';
        state.board[rook_to/8][rook_to%8] = rook;
    } else if (move.queenside_castle) {
        state.board[move.from/8][move.from%8] = piece;
        int rook_from = move.to/8*8 + (state.white_to_move ? 3 : 3);
        int rook_to = move.from/8*8 + (state.white_to_move ? 0 : 0);
        char rook = state.board[rook_from/8][rook_from%8];
        state.board[rook_from/8][rook_from%8] = '.';
        state.board[rook_to/8][rook_to%8] = rook;
    } else {
        state.board[move.from/8][move.from%8] = piece;
    }

    if (move.en_passant) {
        int ep_r = move.to/8, ep_c = move.to%8;
        int pawn_r = move.from/8, pawn_c = move.from%8;
        state.board[ep_r][ep_c] = (state.white_to_move ? 'p' : 'P');
        state.board[move.to/8][move.to%8] = '.';
    } else if (move.captured != 0) {
        state.board[move.to/8][move.to%8] = move.captured;
    }

    state.zobrist_key = 0;
    for (int i = 0; i < 8; i++) {
        for (int j = 0; j < 8; j++) {
            char c = state.board[i][j];
            if (c != '.') {
                int idx = piece_to_index(c);
                state.zobrist_key ^= piece_keys[idx][i*8+j];
            }
        }
    }
    if (!state.white_to_move) {
        state.zobrist_key ^= side_key;
    }
    state.zobrist_key ^= castling_keys[state.castling_rights];
    if (state.en_passant != -1) {
        state.zobrist_key ^= en_passant_keys[state.en_passant % 8];
    }
}

int alphabeta(BoardState& state, int depth, int alpha, int beta, bool maximizingPlayer, unordered_map<uint64_t, TTEntry>& tt, int& nodes_searched) {
    nodes_searched++;

    if (depth == 0) {
        return evaluate_board(state);
    }

    uint64_t key = state.zobrist_key;
    auto it = tt.find(key);
    if (it != tt.end() && it->second.depth >= depth) {
        if (it->second.flag == 0) {
            return it->second.score;
        } else if (it->second.flag == 1) {
            alpha = max(alpha, it->second.score);
        } else if (it->second.flag == 2) {
            beta = min(beta, it->second.score);
        }
        if (alpha >= beta) {
            return it->second.score;
        }
    }

    vector<Move> moves;
    generate_all_moves(state, moves);
    if (moves.empty()) {
        if (is_in_check(state)) {
            return maximizingPlayer ? -CHECKMATE : CHECKMATE;
        } else {
            return DRAW_SCORE;
        }
    }

    for (Move& m : moves) {
        m.score = 0;
        if (m.captured != 0) {
            m.score += 10 * piece_value[m.captured] - piece_value[m.piece];
        }
        if (m.promotion) {
            m.score += 900;
        }
        if (m.kingside_castle || m.queenside_castle) {
            m.score += 50;
        }
        BoardState temp_state = state;
        make_move(temp_state, m);
        if (is_in_check(temp_state)) {
            m.score += 50;
        }
    }

    sort(moves.begin(), moves.end(), [](const Move& a, const Move& b) {
        return a.score > b.score;
    });

    int best_score = maximizingPlayer ? -INF : INF;
    Move best_move;

    for (Move& m : moves) {
        BoardState new_state = state;
        make_move(new_state, m);
        int score = alphabeta(new_state, depth-1, alpha, beta, !maximizingPlayer, tt, nodes_searched);
        if (maximizingPlayer) {
            if (score > best_score) {
                best_score = score;
                best_move = m;
                alpha = max(alpha, score);
            }
        } else {
            if (score < best_score) {
                best_score = score;
                best_move = m;
                beta = min(beta, score);
            }
        }
        if (alpha >= beta) {
            break;
        }
    }

    int flag = 0;
    if (best_score <= alpha) {
        flag = 2;
    } else if (best_score >= beta) {
        flag = 1;
    } else {
        flag = 0;
    }
    tt[key] = {key, depth, best_score, flag, best_move.from, best_move.to, best_move.piece, best_move.promote_to, best_move.en_passant, best_move.kingside_castle, best_move.queenside_castle};

    return best_score;
}

string move_to_string(const Move& move) {
    string s;
    s += move.piece;
    char file1 = 'a' + (move.from % 8);
    char rank1 = '1' + (7 - move.from / 8);
    s += file1;
    s += rank1;
    if (move.en_passant) {
        s += "x";
    } else if (move.captured != 0) {
        s += "x";
    }
    char file2 = 'a' + (move.to % 8);
    char rank2 = '1' + (7 - move.to / 8);
    s += file2;
    s += rank2;
    if (move.promotion) {
        s += "/";
        s += move.promote_to;
    }
    if (move.kingside_castle) {
        return "O-O";
    }
    if (move.queenside_castle) {
        return "O-O-O";
    }
    if (move.en_passant) {
        s += "ep";
    }
    BoardState temp;
    return s;
}

### Explanation
1. **FEN Parsing**: The input FEN string is parsed to set up the initial board state, including piece positions, castling rights, en passant target, and turn information.
2. **Move Generation**: Legal moves are generated for all pieces, considering captures, promotions, castling, and en passant. Moves are ordered to prioritize captures, checks, promotions, and castling for better alpha-beta pruning efficiency.
3. **Alpha-Beta Search**: The search uses iterative deepening to explore deeper levels within the time limit. A transposition table stores evaluated positions to avoid redundant calculations.
4. **Evaluation Function**: The board is evaluated based on material balance, piece-square tables (to encourage central control and development), and king safety (penalizing exposed kings).
5. **Time Management**: The search monitors elapsed time and terminates iterations if the timeout is exceeded, returning the best move found so far in disambiguated algebraic notation. The solution efficiently leverages multi-core processing and optimizations for strong performance.
