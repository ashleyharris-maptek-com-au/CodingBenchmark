using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;
using System.Threading.Tasks;

namespace MaximumCliqueSolver
{
    class Program
    {
        static int[] bestClique = Array.Empty<int>();
        static int bestCliqueSize = 0;
        static Stopwatch stopwatch;
        static long timeLimitMs = 290_000; // 290 seconds, leave 10s for output

        static void Main(string[] args)
        {
            stopwatch = Stopwatch.StartNew();

            // Read input
            string firstLine = Console.ReadLine();
            if (firstLine == null) return;
            var parts = firstLine.Split(' ');
            int numVertices = int.Parse(parts[0]);
            int numEdges = int.Parse(parts[1]);

            // Build adjacency list: HashSet for O(1) lookup
            var adj = new HashSet<int>[numVertices];
            for (int i = 0; i < numVertices; i++)
                adj[i] = new HashSet<int>();

            for (int i = 0; i < numEdges; i++)
            {
                var edge = Console.ReadLine()?.Split(' ');
                if (edge == null || edge.Length < 2) continue;
                int u = int.Parse(edge[0]);
                int v = int.Parse(edge[1]);
                if (u < 0 || u >= numVertices || v < 0 || v >= numVertices) continue;
                adj[u].Add(v);
                adj[v].Add(u);
            }

            // Check time
            if (stopwatch.ElapsedMilliseconds > timeLimitMs)
            {
                OutputResult();
                return;
            }

            // Choose strategy based on graph size
            if (numVertices <= 100)
            {
                ExactBronKerboschWithPivot(adj, numVertices);
            }
            else if (numVertices <= 10000)
            {
                GreedyWithLocalSearchParallel(adj, numVertices);
            }
            else
            {
                LargeGraphHeuristicParallel(adj, numVertices);
            }

            // Final safety: if no clique found, output single vertex
            if (bestCliqueSize == 0 && numVertices > 0)
            {
                bestClique = new int[] { 0 };
                bestCliqueSize = 1;
            }

            OutputResult();
        }

        static void ExactBronKerboschWithPivot(HashSet<int>[] adj, int n)
        {
            var R = new HashSet<int>();
            var P = new HashSet<int>();
            var X = new HashSet<int>();
            for (int i = 0; i < n; i++) P.Add(i);

            BronKerboschPivot(R, P, X, adj);
        }

        static void BronKerboschPivot(HashSet<int> R, HashSet<int> P, HashSet<int> X, HashSet<int>[] adj)
        {
            if (stopwatch.ElapsedMilliseconds > timeLimitMs) return;

            if (P.Count == 0 && X.Count == 0)
            {
                if (R.Count > bestCliqueSize)
                {
                    bestClique = R.ToArray();
                    bestCliqueSize = R.Count;
                }
                return;
            }

            // Choose pivot u from P ∪ X to minimize |P ∩ N(u)|
            int u = -1;
            int minCount = int.MaxValue;
            var union = new HashSet<int>(P);
            union.UnionWith(X);
            foreach (int vertex in union)
            {
                var intersection = new HashSet<int>(P);
                intersection.IntersectWith(adj[vertex]);
                if (intersection.Count < minCount)
                {
                    minCount = intersection.Count;
                    u = vertex;
                }
            }

            // If no pivot found (shouldn't happen), fallback
            if (u == -1 && P.Count > 0)
                u = P.First();

            // Candidates: P \ N(u)
            var candidates = new HashSet<int>(P);
            if (u != -1)
            {
                candidates.ExceptWith(adj[u]);
            }

            foreach (int v in candidates.ToList()) // ToList to avoid modification during iteration
            {
                if (stopwatch.ElapsedMilliseconds > timeLimitMs) return;

                var Rnew = new HashSet<int>(R) { v };
                var Pnew = new HashSet<int>(P);
                Pnew.IntersectWith(adj[v]);
                var Xnew = new HashSet<int>(X);
                Xnew.IntersectWith(adj[v]);

                BronKerboschPivot(Rnew, Pnew, Xnew, adj);

                P.Remove(v);
                X.Add(v);
            }
        }

        static void GreedyWithLocalSearchParallel(HashSet<int>[] adj, int n)
        {
            var seeds = Enumerable.Range(0, n)
                .OrderByDescending(i => adj[i].Count)
                .Take(Math.Min(50, n))
                .ToArray();

            var tasks = new List<Task>();
            var lockObj = new object();

            foreach (int seed in seeds)
            {
                if (stopwatch.ElapsedMilliseconds > timeLimitMs) break;
                tasks.Add(Task.Run(() =>
                {
                    var clique = GreedyGrow(adj, seed);
                    var improved = LocalSearchImprove(adj, clique);
                    lock (lockObj)
                    {
                        if (improved.Length > bestCliqueSize)
                        {
                            bestClique = improved;
                            bestCliqueSize = improved.Length;
                        }
                    }
                }));
            }

            Task.WaitAll(tasks.ToArray());
        }

        static int[] GreedyGrow(HashSet<int>[] adj, int start)
        {
            var clique = new HashSet<int> { start };
            var candidates = new HashSet<int>(adj[start]);

            while (candidates.Count > 0)
            {
                // Pick vertex with most connections to current clique
                int bestV = -1;
                int maxScore = -1;
                foreach (int v in candidates)
                {
                    int score = 0;
                    foreach (int c in clique)
                    {
                        if (adj[v].Contains(c)) score++;
                    }
                    if (score > maxScore)
                    {
                        maxScore = score;
                        bestV = v;
                    }
                }

                if (bestV == -1) break;

                clique.Add(bestV);
                // Update candidates: intersection with neighbors of bestV
                candidates.IntersectWith(adj[bestV]);
            }

            return clique.ToArray();
        }

        static int[] LocalSearchImprove(HashSet<int>[] adj, int[] clique)
        {
            var current = new HashSet<int>(clique);
            bool improved;

            do
            {
                improved = false;
                var nonClique = Enumerable.Range(0, adj.Length)
                    .Where(v => !current.Contains(v))
                    .ToArray();

                foreach (int v in nonClique)
                {
                    // Check if v is connected to all in current clique
                    bool isConnectedToAll = true;
                    foreach (int c in current)
                    {
                        if (!adj[v].Contains(c))
                        {
                            isConnectedToAll = false;
                            break;
                        }
                    }

                    if (isConnectedToAll)
                    {
                        // Try to replace one vertex in clique with v to keep size same or increase
                        // We try to find a vertex u in clique such that (clique - {u}) + {v} is still a clique
                        foreach (int u in current.ToArray())
                        {
                            var test = new HashSet<int>(current);
                            test.Remove(u);
                            test.Add(v);

                            bool valid = true;
                            var list = test.ToArray();
                            for (int i = 0; i < list.Length; i++)
                            {
                                for (int j = i + 1; j < list.Length; j++)
                                {
                                    if (!adj[list[i]].Contains(list[j]))
                                    {
                                        valid = false;
                                        break;
                                    }
                                }
                                if (!valid) break;
                            }

                            if (valid && test.Count > current.Count)
                            {
                                current = test;
                                improved = true;
                                break;
                            }
                        }
                        if (improved) break;
                    }
                }
            } while (improved && stopwatch.ElapsedMilliseconds < timeLimitMs);

            return current.ToArray();
        }

        static void LargeGraphHeuristicParallel(HashSet<int>[] adj, int n)
        {
            // Step 1: Find high-degree seeds
            var degreeList = Enumerable.Range(0, n)
                .Select(i => new { Vertex = i, Degree = adj[i].Count })
                .OrderByDescending(x => x.Degree)
                .Take(Math.Min(100, n))
                .ToArray();

            var tasks = new List<Task>();
            var lockObj = new object();

            // Parallel seed evaluation
            foreach (var item in degreeList)
            {
                if (stopwatch.ElapsedMilliseconds > timeLimitMs) break;
                int seed = item.Vertex;
                tasks.Add(Task.Run(() =>
                {
                    var clique = GreedyGrow(adj, seed);
                    var improved = LocalSearchImprove(adj, clique);
                    lock (lockObj)
                    {
                        if (improved.Length > bestCliqueSize)
                        {
                            bestClique = improved;
                            bestCliqueSize = improved.Length;
                        }
                    }
                }));
            }

            Task.WaitAll(tasks.ToArray());

            // Step 2: Triangle counting heuristic for refinement (optional, lightweight)
            if (stopwatch.ElapsedMilliseconds < timeLimitMs * 0.8) // Only if we have time
            {
                var topVertices = degreeList.Take(Math.Min(20, n)).Select(x => x.Vertex).ToArray();
                foreach (int v in topVertices)
                {
                    if (stopwatch.ElapsedMilliseconds > timeLimitMs) break;
                    var neighbors = adj[v].ToArray();
                    // Look for triangles: u-w where u,w in neighbors and edge(u,w) exists
                    for (int i = 0; i < neighbors.Length && stopwatch.ElapsedMilliseconds < timeLimitMs; i++)
                    {
                        for (int j = i + 1; j < neighbors.Length && stopwatch.ElapsedMilliseconds < timeLimitMs; j++)
                        {
                            int u = neighbors[i];
                            int w = neighbors[j];
                            if (adj[u].Contains(w))
                            {
                                // Found triangle v-u-w, try to grow from here
                                var seedClique = new int[] { v, u, w };
                                var grown = GreedyGrowFromSeed(adj, seedClique);
                                var improved = LocalSearchImprove(adj, grown);
                                lock (lockObj)
                                {
                                    if (improved.Length > bestCliqueSize)
                                    {
                                        bestClique = improved;
                                        bestCliqueSize = improved.Length;
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }

        static int[] GreedyGrowFromSeed(HashSet<int>[] adj, int[] seed)
        {
            var clique = new HashSet<int>(seed);
            var candidates = new HashSet<int>(adj[seed[0]]);
            for (int i = 1; i < seed.Length; i++)
            {
                candidates.IntersectWith(adj[seed[i]]);
            }

            while (candidates.Count > 0)
            {
                int bestV = -1;
                int maxScore = -1;
                foreach (int v in candidates)
                {
                    int score = 0;
                    foreach (int c in clique)
                    {
                        if (adj[v].Contains(c)) score++;
                    }
                    if (score > maxScore)
                    {
                        maxScore = score;
                        bestV = v;
                    }
                }

                if (bestV == -1) break;

                clique.Add(bestV);
                candidates.IntersectWith(adj[bestV]);
            }

            return clique.ToArray();
        }

        static void OutputResult()
        {
            Console.WriteLine(bestCliqueSize);
            if (bestCliqueSize > 0)
            {
                Console.WriteLine(string.Join(" ", bestClique.OrderBy(v => v)));
            }
            else
            {
                Console.WriteLine(); // Empty line if no vertices (shouldn't happen)
            }
        }
    }
