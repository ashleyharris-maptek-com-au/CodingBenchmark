use std::io::{self, BufRead, Write};
use std::sync::{Arc, Mutex};
use std::thread;
use std::time::Duration;

#[derive(Default, Clone, Copy)]
struct State {
    x: f32,
    y: f32,
    vx: f32,
    vy: f32,
    angle: f32,
    angular_vel: f32,
    fuel_remaining: f32,
    updated: bool,
}

fn main() {
    let mut reader = io::BufReader::new(io::stdin());
    
    let mut line = String::new();
    if reader.read_line(&mut line).unwrap_or(0) == 0 { return; }
    let parts: Vec<f32> = line.split_whitespace().filter_map(|s| s.parse().ok()).collect();
    if parts.len() < 5 { return; }
    let _width = parts[0] as usize;
    let height = parts[1] as usize;
    let gravity = parts[2];
    let max_thrust = parts[3];
    let _fuel = parts[4];
    
    line.clear();
    if reader.read_line(&mut line).unwrap_or(0) == 0 { return; }
    let parts: Vec<f32> = line.split_whitespace().filter_map(|s| s.parse().ok()).collect();
    if parts.len() < 4 { return; }
    let _start_x = parts[0];
    let _start_y = parts[1];
    let target_x = parts[2];
    let target_y = parts[3];
    
    // Consume map
    for _ in 0..height {
        line.clear();
        if reader.read_line(&mut line).unwrap_or(0) == 0 { return; }
    }
    
    let state = Arc::new(Mutex::new(State::default()));
    let state_clone = Arc::clone(&state);
    
    thread::spawn(move || {
        let mut local_line = String::new();
        while let Ok(n) = reader.read_line(&mut local_line) {
            if n == 0 { break; }
            let parts: Vec<f32> = local_line.split_whitespace().filter_map(|s| s.parse().ok()).collect();
            if parts.len() >= 7 {
                let mut s = state_clone.lock().unwrap();
                s.x = parts[0];
                s.y = parts[1];
                s.vx = parts[2];
                s.vy = parts[3];
                s.angle = parts[4];
                s.angular_vel = parts[5];
                s.fuel_remaining = parts[6];
                s.updated = true;
            }
            local_line.clear();
        }
    });
    
    loop {
        let current_state = {
            let s = state.lock().unwrap();
            *s
        };
        
        let dx = target_x - current_state.x;
        let dy = target_y - current_state.y;
        let dist = (dx * dx + dy * dy).sqrt();
        
        // Approach logic
        let mut target_vx = dx.clamp(-15.0, 15.0);
        if dist < 50.0 {
            target_vx = dx.clamp(-3.0, 3.0);
        }
        if dist < 10.0 {
            target_vx = dx.clamp(-1.0, 1.0);
        }
        
        let vx_err = target_vx - current_state.vx;
        
        // Calculate desired angle to achieve target_vx
        let target_angle = (vx_err * 0.08).clamp(-0.8, 0.8);
        let a_err = target_angle - current_state.angle;
        let turn = (a_err * 4.0 - current_state.angular_vel * 1.5).clamp(-1.0, 1.0);
        
        // Altitude control
        let mut target_vy = (dy * 0.15).clamp(-15.0, 2.5);
        if dist < 50.0 {
            target_vy = (dy * 0.15).clamp(-5.0, 1.5);
        }
        if dist < 15.0 {
            target_vy = (dy * 0.2).clamp(-2.0, 1.0);
        }
        
        let vy_err = target_vy - current_state.vy;
        let target_accel_y = vy_err * 2.0;
        
        let mut thrust = 0.0;
        let cos_angle = current_state.angle.cos();
        if max_thrust > 0.0 && cos_angle > 0.01 {
            thrust = (gravity - target_accel_y) / (max_thrust * cos_angle);
        }
        
        if !current_state.updated && max_thrust > 0.0 {
            thrust = gravity / max_thrust; // Hover if no updates yet
        }
        
        let final_thrust = thrust.clamp(0.0, 1.0);
        let final_turn = turn.clamp(-1.0, 1.0);
        
        println!("{} {}", final_thrust, final_turn);
        io::stdout().flush().unwrap_or(());
        
        thread::sleep(Duration::from_millis(20)); // Output at 50Hz
    }
}
