using System;
using System.Collections.Generic;
using System.Linq;

class FastScanner {
    private System.IO.Stream stream;
    private byte[] buffer = new byte[1024 * 1024];
    private int head = 0;
    private int tail = 0;

    public FastScanner() {
        stream = System.Console.OpenStandardInput();
    }

    public int NextInt() {
        int c = Read();
        while (c <= 32) {
            if (c == -1) return -1;
            c = Read();
        }
        int res = 0;
        while (c > 32) {
            res = res * 10 + c - '0';
            c = Read();
        }
        return res;
    }

    private int Read() {
        if (head >= tail) {
            head = 0;
            tail = stream.Read(buffer, 0, buffer.Length);
            if (tail == 0) return -1;
        }
        return buffer[head++];
    }
}

class Graph {
    public HashSet<int> Vertices = new HashSet<int>();
    public Dictionary<int, HashSet<int>> InEdges = new Dictionary<int, HashSet<int>>();
    public Dictionary<int, HashSet<int>> OutEdges = new Dictionary<int, HashSet<int>>();

    public void AddVertex(int v) {
        if (Vertices.Add(v)) {
            InEdges[v] = new HashSet<int>();
            OutEdges[v] = new HashSet<int>();
        }
    }

    public void AddEdge(int u, int v) {
        OutEdges[u].Add(v);
        InEdges[v].Add(u);
    }

    public void RemoveVertex(int v) {
        if (!Vertices.Contains(v)) return;
        foreach (var u in InEdges[v]) OutEdges[u].Remove(v);
        foreach (var w in OutEdges[v]) InEdges[w].Remove(v);
        InEdges.Remove(v);
        OutEdges.Remove(v);
        Vertices.Remove(v);
    }

    public void RemoveEdge(int u, int v) {
        OutEdges[u].Remove(v);
        InEdges[v].Remove(u);
    }

    public Graph Copy() {
        Graph g = new Graph();
        foreach (var v in Vertices) {
            g.AddVertex(v);
        }
        foreach (var v in Vertices) {
            foreach (var w in OutEdges[v]) {
                g.AddEdge(v, w);
            }
        }
        return g;
    }
}

class Program {
    public static System.Diagnostics.Stopwatch sw = System.Diagnostics.Stopwatch.StartNew();

    static List<List<int>> FindSCCs(Graph G) {
        var sccs = new List<List<int>>();
        var visited = new HashSet<int>();
        var order = new List<int>();
        
        var edgeIndex = new Dictionary<int, int>();
        var outEdgesList = new Dictionary<int, List<int>>();

        foreach (var startNode in G.Vertices) {
            if (visited.Contains(startNode)) continue;
            
            var stack = new Stack<int>();
            stack.Push(startNode);
            
            while (stack.Count > 0) {
                int v = stack.Peek();
                if (!visited.Contains(v)) {
                    visited.Add(v);
                    outEdgesList[v] = G.OutEdges[v].ToList();
                    edgeIndex[v] = 0;
                }
                
                bool pushed = false;
                var list = outEdgesList[v];
                int idx = edgeIndex[v];
                
                while (idx < list.Count) {
                    int w = list[idx++];
                    if (!visited.Contains(w)) {
                        edgeIndex[v] = idx;
                        stack.Push(w);
                        pushed = true;
                        break;
                    }
                }
                
                if (!pushed) {
                    stack.Pop();
                    order.Add(v);
                }
            }
        }

        visited.Clear();
        order.Reverse();

        foreach (var startNode in order) {
            if (visited.Contains(startNode)) continue;
            
            var scc = new List<int>();
            var stack = new Stack<int>();
            stack.Push(startNode);
            visited.Add(startNode);
            
            while (stack.Count > 0) {
                int v = stack.Pop();
                scc.Add(v);
                
                foreach (var w in G.InEdges[v]) {
                    if (!visited.Contains(w)) {
                        visited.Add(w);
                        stack.Push(w);
                    }
                }
            }
            sccs.Add(scc);
        }

        return sccs;
    }

    static void Reduce(Graph G, List<int> currentFVS) {
        bool changed = true;
        while (changed) {
            changed = false;
            
            bool localChanged = true;
            while (localChanged) {
                localChanged = false;
                var vertices = G.Vertices.ToList();
                foreach (var v in vertices) {
                    if (!G.Vertices.Contains(v)) continue;
                    
                    if (G.OutEdges[v].Contains(v)) {
                        currentFVS.Add(v);
                        G.RemoveVertex(v);
                        localChanged = true;
                        continue;
                    }
                    
                    if (G.InEdges[v].Count == 0 || G.OutEdges[v].Count == 0) {
                        G.RemoveVertex(v);
                        localChanged = true;
                        continue;
                    }
                    
                    if (G.InEdges[v].Count == 1 || G.OutEdges[v].Count == 1) {
                        var in_edges = G.InEdges[v].ToList();
                        var out_edges = G.OutEdges[v].ToList();
                        G.RemoveVertex(v);
                        foreach (var u in in_edges) {
                            foreach (var w in out_edges) {
                                G.AddEdge(u, w);
                            }
                        }
                        localChanged = true;
                    }
                }
                if (localChanged) changed = true;
            }
            
            if (G.Vertices.Count == 0) break;
            
            var sccs = FindSCCs(G);
            if (sccs.Count > 1) {
                var sccMap = new Dictionary<int, int>();
                for(int i = 0; i < sccs.Count; i++) {
                    foreach(var v in sccs[i]) sccMap[v] = i;
                }
                
                int removedEdges = 0;
                var vertices = G.Vertices.ToList();
                foreach (var v in vertices) {
                    if (!G.Vertices.Contains(v)) continue;
                    var outEdges = G.OutEdges[v].ToList();
                    foreach (var w in outEdges) {
                        if (sccMap[v] != sccMap[w]) {
                            G.RemoveEdge(v, w);
                            removedEdges++;
                        }
                    }
                }
                if (removedEdges > 0) changed = true;
            }
        }
    }

    static int ComputeLowerBound(Graph G) {
        var visited = new HashSet<int>();
        int lb = 0;
        foreach(var v in G.Vertices) {
            if (visited.Contains(v)) continue;
            foreach(var u in G.OutEdges[v]) {
                if (visited.Contains(u)) continue;
                if (G.OutEdges[u].Contains(v)) {
                    lb++;
                    visited.Add(u);
                    visited.Add(v);
                    break;
                }
            }
        }
        return lb;
    }

    static List<int> GreedyFVS(Graph G) {
        Graph g = G.Copy();
        List<int> fvs = new List<int>();
        while(g.Vertices.Count > 0) {
            bool changed = true;
            while(changed) {
                changed = false;
                foreach(var v in g.Vertices.ToList()) {
                    if (!g.Vertices.Contains(v)) continue;
                    if (g.OutEdges[v].Contains(v)) {
                        fvs.Add(v);
                        g.RemoveVertex(v);
                        changed = true;
                    } else if (g.InEdges[v].Count == 0 || g.OutEdges[v].Count == 0) {
                        g.RemoveVertex(v);
                        changed = true;
                    } else if (g.InEdges[v].Count == 1 || g.OutEdges[v].Count == 1) {
                        var in_edges = g.InEdges[v].ToList();
                        var out_edges = g.OutEdges[v].ToList();
                        g.RemoveVertex(v);
                        foreach(var u in in_edges) foreach(var w in out_edges) g.AddEdge(u, w);
                        changed = true;
                    }
                }
            }
            if (g.Vertices.Count == 0) break;
            int max_deg = -1;
            int best_v = -1;
            foreach(var v in g.Vertices) {
                int deg = g.InEdges[v].Count + g.OutEdges[v].Count;
                if (deg > max_deg) { max_deg = deg; best_v = v; }
            }
            fvs.Add(best_v);
            g.RemoveVertex(best_v);
        }
        return fvs;
    }

    static List<int> Solve(Graph G, int current_best) {
        if (sw.ElapsedMilliseconds > 4500) throw new System.TimeoutException();

        List<int> fvs = new List<int>();
        Reduce(G, fvs);
        
        int lb = ComputeLowerBound(G);
        if (fvs.Count + lb >= current_best) return null;
        
        if (G.Vertices.Count == 0) return fvs;
        
        var sccs = FindSCCs(G);
        if (sccs.Count > 1) {
            List<int> total_fvs = new List<int>(fvs);
            int remaining_budget = current_best - fvs.Count;
            foreach (var scc in sccs) {
                Graph subG = new Graph();
                foreach (var v in scc) subG.AddVertex(v);
                foreach (var v in scc) {
                    foreach (var w in G.OutEdges[v]) {
                        subG.AddEdge(v, w);
                    }
                }
                var subFVS = Solve(subG, remaining_budget);
                if (subFVS == null) return null;
                total_fvs.AddRange(subFVS);
                remaining_budget -= subFVS.Count;
            }
            return total_fvs;
        }
        
        int branch_v = -1;
        foreach(var v in G.Vertices) {
            foreach(var u in G.OutEdges[v]) {
                if (G.OutEdges[u].Contains(v)) {
                    branch_v = v;
                    break;
                }
            }
            if (branch_v != -1) break;
        }

        if (branch_v == -1) {
            int max_degree = -1;
            foreach(var v in G.Vertices) {
                int deg = G.InEdges[v].Count + G.OutEdges[v].Count;
                if (deg > max_degree) {
                    max_degree = deg;
                    branch_v = v;
                }
            }
        }
        
        List<int> best_local = null;
        
        Graph G1 = G.Copy();
        G1.RemoveVertex(branch_v);
        var fvs1 = Solve(G1, current_best - fvs.Count - 1);
        
        if (fvs1 != null) {
            current_best = fvs.Count + 1 + fvs1.Count;
            best_local = new List<int>(fvs);
            best_local.Add(branch_v);
            best_local.AddRange(fvs1);
        }
        
        Graph G2 = G.Copy();
        var in_edges = G2.InEdges[branch_v].ToList();
        var out_edges = G2.OutEdges[branch_v].ToList();
        G2.RemoveVertex(branch_v);
        foreach(var u in in_edges) foreach(var w in out_edges) G2.AddEdge(u, w);
        
        var fvs2 = Solve(G2, current_best - fvs.Count);
        if (fvs2 != null) {
            best_local = new List<int>(fvs);
            best_local.AddRange(fvs2);
        }
        
        return best_local;
    }

    static void Main() {
        var scanner = new FastScanner();
        int n = scanner.NextInt();
        if (n == -1) return;
        int m = scanner.NextInt();
        
        Graph G = new Graph();
        for(int i=0; i<n; i++) G.AddVertex(i);
        
        for(int i=0; i<m; i++) {
            int u = scanner.NextInt();
            int v = scanner.NextInt();
            G.AddEdge(u, v);
        }
        
        var greedy = GreedyFVS(G);
        List<int> ans;
        try {
            ans = Solve(G, greedy.Count);
            if (ans == null) ans = greedy;
        } catch (System.TimeoutException) {
            ans = greedy;
        }
        
        Console.WriteLine(ans.Count);
        Console.WriteLine(string.Join(" ", ans));
    }
}
