const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    if (!input.trim()) return;
    const data = JSON.parse(input);

    const probability = data.probability;
    const drones = data.drones;
    const maxStepsPerDrone = data.maxStepsPerDrone;

    const R = probability.length;
    if (R === 0) return;
    const C = probability[0].length;

    const paths = drones.map(d => [[d.x, d.y]]);
    const pos = drones.map(d => ({ x: d.x, y: d.y }));

    const visited = Array.from({ length: R }, () => new Array(C).fill(false));
    for (const p of pos) {
        visited[p.y][p.x] = true;
        probability[p.y][p.x] = 0;
    }

    const dx = [0, 1, 0, -1, 0];
    const dy = [-1, 0, 1, 0, 0];

    let lookahead = 5;
    const totalSimulations = maxStepsPerDrone * drones.length;
    if (totalSimulations > 5000) {
        lookahead = 4;
    }
    if (totalSimulations > 20000) {
        lookahead = 3;
    }

    function getBestMove(x, y) {
        let bestMove = 4;
        let maxVal = -1;

        function dfs(cx, cy, steps, currentVal, currentVisited) {
            if (steps === 0) return currentVal;
            let maxSub = currentVal;
            for (let i = 0; i < 5; i++) {
                const nx = cx + dx[i];
                const ny = cy + dy[i];
                if (nx >= 0 && nx < C && ny >= 0 && ny < R && probability[ny][nx] !== -1) {
                    const key = ny * C + nx;
                    const wasVisited = currentVisited.has(key) || visited[ny][nx];
                    const val = wasVisited ? 0 : probability[ny][nx];
                    if (!wasVisited && val > 0) {
                        currentVisited.add(key);
                    }
                    const subVal = dfs(nx, ny, steps - 1, currentVal + val, currentVisited);
                    if (subVal > maxSub) {
                        maxSub = subVal;
                    }
                    if (!wasVisited && val > 0) {
                        currentVisited.delete(key);
                    }
                }
            }
            return maxSub;
        }

        for (let i = 0; i < 5; i++) {
            const nx = x + dx[i];
            const ny = y + dy[i];
            if (nx >= 0 && nx < C && ny >= 0 && ny < R && probability[ny][nx] !== -1) {
                const currentVisited = new Set();
                const key = ny * C + nx;
                currentVisited.add(key);
                const val = visited[ny][nx] ? 0 : probability[ny][nx];
                const score = dfs(nx, ny, lookahead - 1, val, currentVisited);
                if (score > maxVal) {
                    maxVal = score;
                    bestMove = i;
                }
            }
        }
        
        return bestMove;
    }

    for (let step = 0; step < maxStepsPerDrone; step++) {
        for (let d = 0; d < drones.length; d++) {
            const move = getBestMove(pos[d].x, pos[d].y);
            pos[d].x += dx[move];
            pos[d].y += dy[move];
            paths[d].push([pos[d].x, pos[d].y]);
            
            visited[pos[d].y][pos[d].x] = true;
            probability[pos[d].y][pos[d].x] = 0;
        }
    }

    console.log(JSON.stringify({ paths }));
}

solve();
