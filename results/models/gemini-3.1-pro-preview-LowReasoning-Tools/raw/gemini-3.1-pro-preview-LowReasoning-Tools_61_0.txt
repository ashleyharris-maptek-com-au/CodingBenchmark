declare const require: any; 
const fs = require('fs'); 
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const depot = data.depot;
const packages = data.packages;
const n = packages.length;

if (n === 0) {
    console.log(JSON.stringify({ routes: [] }));
    process.exit(0);
}

const xs = new Float64Array(n);
const ys = new Float64Array(n);
for (let i = 0; i < n; i++) {
    xs[i] = packages[i].x;
    ys[i] = packages[i].y;
}
const depotX = depot[0];
const depotY = depot[1];

let unvisited = new Uint8Array(n);
unvisited.fill(1);

let currentX = depotX;
let currentY = depotY;
let tour = new Int32Array(n);

// 1. Nearest Neighbor Initialization
for (let step = 0; step < n; step++) {
    let bestIdx = -1;
    let bestDist = Infinity;
    for (let i = 0; i < n; i++) {
        if (unvisited[i]) {
            const dx = currentX - xs[i];
            const dy = currentY - ys[i];
            const d = dx * dx + dy * dy;
            if (d < bestDist) {
                bestDist = d;
                bestIdx = i;
            }
        }
    }
    tour[step] = bestIdx;
    unvisited[bestIdx] = 0;
    currentX = xs[bestIdx];
    currentY = ys[bestIdx];
}

// 2. 2-opt Refinement
let improvement = true;
let passes = 0;
const MAX_PASSES = 50;
const start_time = Date.now();

while (improvement && passes < MAX_PASSES) {
    if (Date.now() - start_time > 8000) break; // 8-second time limit to ensure fast termination
    improvement = false;
    passes++;
    
    for (let i = 0; i < n - 1; i++) {
        const x_i_minus_1 = i === 0 ? depotX : xs[tour[i - 1]];
        const y_i_minus_1 = i === 0 ? depotY : ys[tour[i - 1]];
        const x_i = xs[tour[i]];
        const y_i = ys[tour[i]];
        
        for (let k = i + 1; k < n; k++) {
            const x_k = xs[tour[k]];
            const y_k = ys[tour[k]];
            const x_k_plus_1 = k === n - 1 ? depotX : xs[tour[k + 1]];
            const y_k_plus_1 = k === n - 1 ? depotY : ys[tour[k + 1]];
            
            const dx1 = x_i_minus_1 - x_i;
            const dy1 = y_i_minus_1 - y_i;
            const d1 = Math.sqrt(dx1*dx1 + dy1*dy1);
            
            const dx2 = x_k - x_k_plus_1;
            const dy2 = y_k - y_k_plus_1;
            const d2 = Math.sqrt(dx2*dx2 + dy2*dy2);
            
            const dx3 = x_i_minus_1 - x_k;
            const dy3 = y_i_minus_1 - y_k;
            const d3 = Math.sqrt(dx3*dx3 + dy3*dy3);
            
            const dx4 = x_i - x_k_plus_1;
            const dy4 = y_i - y_k_plus_1;
            const d4 = Math.sqrt(dx4*dx4 + dy4*dy4);
            
            if (d3 + d4 < d1 + d2 - 1e-9) {
                let left = i;
                let right = k;
                while (left < right) {
                    let temp = tour[left];
                    tour[left] = tour[right];
                    tour[right] = temp;
                    left++;
                    right--;
                }
                improvement = true;
                break; 
            }
        }
    }
}

// 3. Output Preparation
const routeIds = [];
for (let i = 0; i < n; i++) {
    routeIds.push(packages[tour[i]].id);
}

console.log(JSON.stringify({ routes: [routeIds] }));
