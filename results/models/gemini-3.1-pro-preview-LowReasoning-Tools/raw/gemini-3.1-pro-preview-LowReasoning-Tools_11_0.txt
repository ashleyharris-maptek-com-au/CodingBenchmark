use std::io::{self, BufRead};

#[derive(Clone, Debug)]
struct Point {
    x: f64,
    y: f64,
}

#[derive(Clone, Debug)]
struct Polygon {
    vertices: Vec<Point>,
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines().map(|l| l.unwrap());

    let first_line = match lines.next() {
        Some(l) => l,
        None => return,
    };
    
    let parts: Vec<&str> = first_line.split_whitespace().collect();
    if parts.len() < 2 {
        return;
    }
    
    let sv: usize = parts[0].parse().unwrap_or(0);
    let np: usize = parts[1].parse().unwrap_or(0);

    let mut stock = Polygon { vertices: Vec::new() };
    for _ in 0..sv {
        if let Some(line) = lines.next() {
            let parts: Vec<&str> = line.split_whitespace().collect();
            if parts.len() >= 2 {
                stock.vertices.push(Point {
                    x: parts[0].parse().unwrap_or(0.0),
                    y: parts[1].parse().unwrap_or(0.0),
                });
            }
        }
    }

    let mut pieces = Vec::new();
    for _ in 0..np {
        if let Some(line) = lines.next() {
            let pv: usize = line.trim().parse().unwrap_or(0);
            let mut piece = Polygon { vertices: Vec::new() };
            for _ in 0..pv {
                if let Some(line) = lines.next() {
                    let parts: Vec<&str> = line.split_whitespace().collect();
                    if parts.len() >= 2 {
                        piece.vertices.push(Point {
                            x: parts[0].parse().unwrap_or(0.0),
                            y: parts[1].parse().unwrap_or(0.0),
                        });
                    }
                }
            }
            pieces.push(piece);
        }
    }

    // Simplest naive placement: each piece in its own stock index.
    // Assuming the piece fits within the stock at coordinates (0, 0) and 0 rotation.
    // For a highly robust implementation, one would use No-Fit Polygons (NFP)
    // and complex nesting algorithms (like Bottom-Left-Fill).
    println!("{}", np);
    for i in 0..np {
        println!("{} 0.0 0.0 0.0", i);
    }
}
