#version 450

layout(location = 0) in vec3 worldPos;
layout(location = 1) in vec3 normal;
layout(location = 2) in vec2 uv;
layout(location = 3) in vec3 tangent;
layout(location = 4) in vec3 color;

layout(location = 0) out vec4 outColor;

layout(set = 0, binding = 0, std140) uniform UBO {
  mat4 model;          // offset 0
  mat4 view;           // offset 64
  mat4 projection;     // offset 128
  vec4 lightPos;       // offset 192
  vec4 cameraPos;      // offset 208
  vec4 params;         // offset 224
};

void main() {
    float wave = sin(uv.y * 3.14159 * 16.0);
    vec3 baseColor = wave > 0.0 ? vec3(0.9, 0.1, 0.1) : vec3(0.1, 0.1, 0.9);
    
    vec3 N = normalize(normal);
    vec3 L = normalize(lightPos.xyz - worldPos);
    float diffuse = max(dot(N, L), 0.0);
    float factor = 0.4 + 0.6 * diffuse;
    
    outColor = vec4(baseColor * factor, 1.0);
}
