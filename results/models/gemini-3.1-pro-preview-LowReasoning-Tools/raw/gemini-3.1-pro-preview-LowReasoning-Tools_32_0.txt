#include <iostream>
#include <vector>
#include <queue>
#include <numeric>
#include <algorithm>

using namespace std;

const long long INF = 1e18;

struct Edge {
    int to;
    long long weight;
    int id;
};

struct EdgeData {
    int u, v;
    long long weight;
    int id;
};

struct GPrimeEdge {
    int u_term;
    int v_term;
    long long weight;
    int original_u;
    int original_v;
    int original_edge_id;
};

struct DSU {
    vector<int> parent;
    DSU(int n) {
        parent.resize(n);
        iota(parent.begin(), parent.end(), 0);
    }
    int find(int i) {
        if (parent[i] == i)
            return i;
        return parent[i] = find(parent[i]);
    }
    bool unite(int i, int j) {
        int root_i = find(i);
        int root_j = find(j);
        if (root_i != root_j) {
            parent[root_i] = root_j;
            return true;
        }
        return false;
    }
};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int V, E, K;
    if (!(cin >> V >> E >> K)) return 0;

    vector<EdgeData> edges(E);
    vector<vector<Edge>> adj(V);

    for (int i = 0; i < E; ++i) {
        cin >> edges[i].u >> edges[i].v >> edges[i].weight;
        edges[i].id = i;
        adj[edges[i].u].push_back({edges[i].v, edges[i].weight, i});
        adj[edges[i].v].push_back({edges[i].u, edges[i].weight, i});
    }

    vector<int> terminals(K);
    vector<bool> is_terminal(V, false);
    for (int i = 0; i < K; ++i) {
        cin >> terminals[i];
        is_terminal[terminals[i]] = true;
    }

    if (K <= 1) {
        cout << "0 0\n";
        return 0;
    }

    bool use_exact = false;
    if (K <= 15) {
        long long p3 = 1;
        for(int i=0; i<K; ++i) p3 *= 3;
        long long op1 = p3 * V;
        long long op2 = (1LL << K) * (V * 20LL + E);
        long long states = (1LL << K) * V;
        if (states <= 5000000 && op1 + op2 <= 100000000) {
            use_exact = true;
        }
    }

    vector<int> final_edges;

    if (use_exact) {
        vector<long long> dp((1 << K) * V, INF);
        vector<int> parent_type((1 << K) * V, 0);
        vector<int> parent_info((1 << K) * V, 0);
        vector<int> parent_edge_arr((1 << K) * V, -1);

        for (int i = 0; i < K; ++i) {
            dp[(1 << i) * V + terminals[i]] = 0;
        }

        for (int mask = 1; mask < (1 << K); ++mask) {
            if ((mask & (mask - 1)) != 0) { 
                int lowest_bit = mask & -mask;
                for (int submask = (mask - 1) & mask; submask > 0; submask = (submask - 1) & mask) {
                    if ((submask & lowest_bit) == 0) continue;
                    int other = mask ^ submask;
                    
                    const long long* dp_sub = &dp[submask * V];
                    const long long* dp_oth = &dp[other * V];
                    long long* dp_msk = &dp[mask * V];
                    int* ptype_msk = &parent_type[mask * V];
                    int* pinfo_msk = &parent_info[mask * V];

                    for (int v = 0; v < V; ++v) {
                        long long cost = dp_sub[v] + dp_oth[v];
                        if (cost < dp_msk[v]) {
                            dp_msk[v] = cost;
                            ptype_msk[v] = 1;
                            pinfo_msk[v] = submask;
                        }
                    }
                }
            }

            priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;
            long long* dp_msk = &dp[mask * V];
            int* ptype_msk = &parent_type[mask * V];
            int* pinfo_msk = &parent_info[mask * V];
            int* pedge_msk = &parent_edge_arr[mask * V];

            for (int v = 0; v < V; ++v) {
                if (dp_msk[v] != INF) {
                    pq.push({dp_msk[v], v});
                }
            }

            while (!pq.empty()) {
                auto [d, u] = pq.top();
                pq.pop();
                if (d > dp_msk[u]) continue;

                for (auto& edge : adj[u]) {
                    int v = edge.to;
                    if (dp_msk[v] > d + edge.weight) {
                        dp_msk[v] = d + edge.weight;
                        ptype_msk[v] = 2;
                        pinfo_msk[v] = u;
                        pedge_msk[v] = edge.id;
                        pq.push({dp_msk[v], v});
                    }
                }
            }
        }

        long long best_cost = INF;
        int best_root = -1;
        int full_mask = (1 << K) - 1;
        for (int v = 0; v < V; ++v) {
            if (dp[full_mask * V + v] < best_cost) {
                best_cost = dp[full_mask * V + v];
                best_root = v;
            }
        }

        vector<bool> edge_used(E, false);
        queue<pair<int, int>> q;
        if (best_root != -1) {
            q.push({full_mask, best_root});
        }

        while (!q.empty()) {
            auto [mask, v] = q.front();
            q.pop();

            int ptype = parent_type[mask * V + v];
            if (ptype == 1) {
                int submask = parent_info[mask * V + v];
                int other = mask ^ submask;
                q.push({submask, v});
                q.push({other, v});
            } else if (ptype == 2) {
                int u = parent_info[mask * V + v];
                int e_id = parent_edge_arr[mask * V + v];
                if (!edge_used[e_id]) {
                    edge_used[e_id] = true;
                    final_edges.push_back(e_id);
                }
                q.push({mask, u});
            }
        }
    } else {
        vector<long long> dist(V, INF);
        vector<int> origin(V, -1);
        vector<int> parent_edge_arr(V, -1);
        vector<int> parent_node(V, -1);

        priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;

        for (int t : terminals) {
            dist[t] = 0;
            origin[t] = t;
            pq.push({0, t});
        }

        while (!pq.empty()) {
            auto [d, u] = pq.top();
            pq.pop();

            if (d > dist[u]) continue;

            for (auto& edge : adj[u]) {
                int v = edge.to;
                if (dist[v] > d + edge.weight) {
                    dist[v] = d + edge.weight;
                    origin[v] = origin[u];
                    parent_node[v] = u;
                    parent_edge_arr[v] = edge.id;
                    pq.push({dist[v], v});
                }
            }
        }

        vector<GPrimeEdge> gp_edges;
        for (int i = 0; i < E; ++i) {
            int u = edges[i].u;
            int v = edges[i].v;
            if (origin[u] != -1 && origin[v] != -1 && origin[u] != origin[v]) {
                long long w = dist[u] + dist[v] + edges[i].weight;
                gp_edges.push_back({origin[u], origin[v], w, u, v, i});
            }
        }

        sort(gp_edges.begin(), gp_edges.end(), [](const GPrimeEdge& a, const GPrimeEdge& b) {
            return a.weight < b.weight;
        });

        DSU dsu(V);
        vector<bool> subgraph_edges(E, false);
        int edges_added = 0;

        for (auto& gpe : gp_edges) {
            if (dsu.unite(gpe.u_term, gpe.v_term)) {
                subgraph_edges[gpe.original_edge_id] = true;
                
                int curr = gpe.original_u;
                while (parent_edge_arr[curr] != -1) {
                    subgraph_edges[parent_edge_arr[curr]] = true;
                    curr = parent_node[curr];
                }
                
                curr = gpe.original_v;
                while (parent_edge_arr[curr] != -1) {
                    subgraph_edges[parent_edge_arr[curr]] = true;
                    curr = parent_node[curr];
                }
                
                edges_added++;
                if (edges_added == K - 1) break;
            }
        }

        vector<EdgeData> H_edges;
        for (int i = 0; i < E; ++i) {
            if (subgraph_edges[i]) {
                H_edges.push_back(edges[i]);
            }
        }

        sort(H_edges.begin(), H_edges.end(), [](const EdgeData& a, const EdgeData& b) {
            return a.weight < b.weight;
        });

        DSU dsu_H(V);
        vector<int> mst_edges;
        for (auto& edge : H_edges) {
            if (dsu_H.unite(edge.u, edge.v)) {
                mst_edges.push_back(edge.id);
            }
        }

        vector<vector<pair<int, int>>> mst_adj(V);
        vector<int> degree(V, 0);

        for (int e_id : mst_edges) {
            int u = edges[e_id].u;
            int v = edges[e_id].v;
            mst_adj[u].push_back({v, e_id});
            mst_adj[v].push_back({u, e_id});
            degree[u]++;
            degree[v]++;
        }

        queue<int> leaves;
        for (int i = 0; i < V; ++i) {
            if (degree[i] == 1 && !is_terminal[i]) {
                leaves.push(i);
            }
        }

        vector<bool> pruned_edges(E, false);

        while (!leaves.empty()) {
            int u = leaves.front();
            leaves.pop();

            for (auto& edge : mst_adj[u]) {
                int v = edge.first;
                int e_id = edge.second;
                if (!pruned_edges[e_id]) {
                    pruned_edges[e_id] = true;
                    degree[u]--;
                    degree[v]--;
                    if (degree[v] == 1 && !is_terminal[v]) {
                        leaves.push(v);
                    }
                }
            }
        }

        for (int e_id : mst_edges) {
            if (!pruned_edges[e_id]) {
                final_edges.push_back(e_id);
            }
        }
    }

    long long total_weight = 0;
    for (int e_id : final_edges) {
        total_weight += edges[e_id].weight;
    }

    cout << total_weight << " " << final_edges.size() << "\n";
    for (int e_id : final_edges) {
        cout << edges[e_id].u << " " << edges[e_id].v << "\n";
    }

    return 0;
}
