```rust
use std::io::{self, BufRead};
use std::collections::{HashMap, BinaryHeap};
use std::cmp::Ordering;

#[derive(Debug, Clone)]
struct Sample {
    hole_id: String,
    sample_idx: usize,
    x: f64,
    y: f64,
    z: f64,
    values: Vec<f64>,
}

#[derive(Debug)]
struct Anomaly {
    hole_id: String,
    sample_idx: usize,
    property_name: String,
    confidence: f64,
}

impl PartialEq for Anomaly {
    fn eq(&self, other: &Self) -> bool {
        self.confidence == other.confidence
    }
}

impl Eq for Anomaly {}

impl PartialOrd for Anomaly {
    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        self.confidence.partial_cmp(&other.confidence)
    }
}

impl Ord for Anomaly {
    fn cmp(&self, other: &Self) -> Ordering {
        self.partial_cmp(other).unwrap_or(Ordering::Equal)
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines().filter_map(|l| l.ok());

    let first_line = match lines.next() {
        Some(l) => l,
        None => return,
    };
    
    let parts: Vec<&str> = first_line.split_whitespace().collect();
    if parts.len() < 2 { return; }
    
    let _num_holes: usize = parts[0].parse().unwrap_or(0);
    let num_properties: usize = parts[1].parse().unwrap_or(0);

    let prop_line = lines.next().unwrap_or_default();
    let properties: Vec<String> = prop_line.split_whitespace().map(|s| s.to_string()).collect();

    let mut all_samples = Vec::new();

    while let Some(hole_line) = lines.next() {
        if hole_line.trim().is_empty() { continue; }
        let h_parts: Vec<&str> = hole_line.split_whitespace().collect();
        if h_parts.len() < 9 { continue; }

        let hole_id = h_parts[0].to_string();
        let start_x: f64 = h_parts[1].parse().unwrap_or(0.0);
        let start_y: f64 = h_parts[2].parse().unwrap_or(0.0);
        let start_z: f64 = h_parts[3].parse().unwrap_or(0.0);
        let dir_x: f64 = h_parts[4].parse().unwrap_or(0.0);
        let dir_y: f64 = h_parts[5].parse().unwrap_or(0.0);
        let dir_z: f64 = h_parts[6].parse().unwrap_or(0.0);
        let _length: f64 = h_parts[7].parse().unwrap_or(0.0);
        let num_samples: usize = h_parts[8].parse().unwrap_or(0);

        // Normalize direction
        let dir_len = (dir_x*dir_x + dir_y*dir_y + dir_z*dir_z).sqrt();
        let (dx, dy, dz) = if dir_len > 0.0 {
            (dir_x / dir_len, dir_y / dir_len, dir_z / dir_len)
        } else {
            (0.0, 0.0, 0.0)
        };

        for i in 0..num_samples {
            if let Some(s_line) = lines.next() {
                let s_parts: Vec<&str> = s_line.split_whitespace().collect();
                if s_parts.is_empty() { continue; }
                let depth: f64 = s_parts[0].parse().unwrap_or(0.0);
                let mut values = Vec::new();
                for j in 0..num_properties {
                    if j + 1 < s_parts.len() {
                        values.push(s_parts[j+1].parse().unwrap_or(f64::NAN));
                    } else {
                        values.push(f64::NAN);
                    }
                }
                
                all_samples.push(Sample {
                    hole_id: hole_id.clone(),
                    sample_idx: i,
                    x: start_x + dx * depth,
                    y: start_y + dy * depth,
                    z: start_z + dz * depth,
                    values,
                });
            }
        }
    }

    let n = all_samples.len();
    let mut anomalies = Vec::new();

    // Adaptive strategy based on dataset size
    if n < 10000 {
        // Naive O(N^2) for small datasets
        for i in 0..n {
            let s1 = &all_samples[i];
            for p in 0..num_properties {
                let v1 = s1.values[p];
                if v1.is_nan() { continue; }

                let mut local_vals = Vec::new();
                for j in 0..n {
                    if i == j { continue; }
                    let s2 = &all_samples[j];
                    let dist2 = (s1.x - s2.x).powi(2) + (s1.y - s2.y).powi(2) + (s1.z - s2.z).powi(2);
                    if dist2 < 10000.0 { // 100 units search radius
                        let v2 = s2.values[p];
                        if !v2.is_nan() {
                            local_vals.push(v2);
                        }
                    }
                }
                
                if local_vals.len() > 5 {
                    let mean: f64 = local_vals.iter().sum::<f64>() / local_vals.len() as f64;
                    let var: f64 = local_vals.iter().map(|&x| (x - mean).powi(2)).sum::<f64>() / local_vals.len() as f64;
                    let std_dev = var.sqrt();
                    
                    if std_dev > 1e-6 {
                        let z_score = ((v1 - mean) / std_dev).abs();
                        if z_score > 3.0 { // 3 sigma
                            anomalies.push(Anomaly {
                                hole_id: s1.hole_id.clone(),
                                sample_idx: s1.sample_idx,
                                property_name: properties[p].clone(),
                                confidence: z_score,
                            });
                        }
                    }
                }
            }
        }
    } else {
        // Spatial hashing (grid) for larger datasets
        let grid_size = 50.0; // Cells of 50x50x50
        let mut grid: HashMap<(i32, i32, i32), Vec<usize>> = HashMap::new();
        
        for (i, s) in all_samples.iter().enumerate() {
            let gx = (s.x / grid_size).floor() as i32;
            let gy = (s.y / grid_size).floor() as i32;
            let gz = (s.z / grid_size).floor() as i32;
            grid.entry((gx, gy, gz)).or_default().push(i);
        }

        for i in 0..n {
            let s1 = &all_samples[i];
            let gx = (s1.x / grid_size).floor() as i32;
            let gy = (s1.y / grid_size).floor() as i32;
            let gz = (s1.z / grid_size).floor() as i32;

            let mut neighbors = Vec::new();
            for dx in -1..=1 {
                for dy in -1..=1 {
                    for dz in -1..=1 {
                        if let Some(cell) = grid.get(&(gx + dx, gy + dy, gz + dz)) {
                            for &idx in cell {
                                if idx != i {
                                    let s2 = &all_samples[idx];
                                    let dist2 = (s1.x - s2.x).powi(2) + (s1.y - s2.y).powi(2) + (s1.z - s2.z).powi(2);
                                    if dist2 < 10000.0 {
                                        neighbors.push(idx);
                                    }
                                }
                            }
                        }
                    }
                }
            }

            for p in 0..num_properties {
                let v1 = s1.values[p];
                if v1.is_nan() { continue; }

                let mut local_vals = Vec::new();
                for &idx in &neighbors {
                    let v2 = all_samples[idx].values[p];
                    if !v2.is_nan() {
                        local_vals.push(v2);
                    }
                }

                if local_vals.len() > 5 {
                    let mean: f64 = local_vals.iter().sum::<f64>() / local_vals.len() as f64;
                    let var: f64 = local_vals.iter().map(|&x| (x - mean).powi(2)).sum::<f64>() / local_vals.len() as f64;
                    let std_dev = var.sqrt();

                    if std_dev > 1e-6 {
                        let z_score = ((v1 - mean) / std_dev).abs();
                        if z_score > 3.0 {
                            anomalies.push(Anomaly {
                                hole_id: s1.hole_id.clone(),
                                sample_idx: s1.sample_idx,
                                property_name: properties[p].clone(),
                                confidence: z_score,
                            });
                        }
                    }
                }
            }
        }
    }

    anomalies.sort_by(|a, b| b.cmp(a));

    println!("{}", anomalies.len());
    for a in anomalies {
        println!("{} {} {} {:.2}", a.hole_id, a.sample_idx, a.property_name, a.confidence);
    }
}
```
