#version 450
layout(lines) in;
layout(triangle_strip, max_vertices = 7) out;

layout(location = 0) in vec3 in_worldPos[];
layout(location = 1) in vec3 in_normal[];
layout(location = 2) in vec3 in_color[];

layout(location = 0) out vec3 outColor;

void main() {
    vec4 p0 = gl_in[0].gl_Position;
    vec4 p1 = gl_in[1].gl_Position;
    vec3 c0 = in_color[0];
    vec3 c1 = in_color[1];

    vec2 dir = p1.xy - p0.xy;
    vec2 perp = normalize(vec2(-dir.y, dir.x));

    vec2 mid = p0.xy + 0.75 * dir;

    vec2 p0_p = p0.xy + perp * 0.015;
    vec2 p0_m = p0.xy - perp * 0.015;
    vec2 mid_p = mid + perp * 0.015;
    vec2 mid_m = mid - perp * 0.015;

    // Shaft
    outColor = c0;
    gl_Position = vec4(p0_p, p0.z, p0.w); EmitVertex();
    gl_Position = vec4(p0_m, p0.z, p0.w); EmitVertex();
    gl_Position = vec4(mid_p, p0.z, p0.w); EmitVertex();
    gl_Position = vec4(mid_m, p0.z, p0.w); EmitVertex();
    EndPrimitive();

    // Arrowhead
    vec2 tip_p = mid + perp * 0.05;
    vec2 tip_m = mid - perp * 0.05;
    outColor = c1;
    gl_Position = vec4(tip_p, p1.z, p1.w); EmitVertex();
    gl_Position = vec4(tip_m, p1.z, p1.w); EmitVertex();
    gl_Position = vec4(p1.xy, p1.z, p1.w); EmitVertex();
    EndPrimitive();
}
