#include <iostream>
#include <vector>

// A placeholder for the Constructive Solid Geometry (CSG) implementation in C++
// For extremely large meshes, a BVH or Octree should be used to accelerate operations.

struct Vertex {
    double x, y, z;
};

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    int va, fa, vb, fb;
    if (!(std::cin >> va >> fa >> vb >> fb)) return 0;
    
    std::vector<Vertex> A_verts(va);
    for (int i = 0; i < va; ++i) std::cin >> A_verts[i].x >> A_verts[i].y >> A_verts[i].z;
    
    std::vector<int> A_faces(fa * 3);
    for (int i = 0; i < fa * 3; ++i) std::cin >> A_faces[i];
    
    std::vector<Vertex> B_verts(vb);
    for (int i = 0; i < vb; ++i) std::cin >> B_verts[i].x >> B_verts[i].y >> B_verts[i].z;
    
    std::vector<int> B_faces(fb * 3);
    for (int i = 0; i < fb * 3; ++i) std::cin >> B_faces[i];

    // Naive union (concatenation)
    // Given the constraints of a single file without external libraries and low reasoning effort,
    // this handles disjoint sets exactly. True exact boolean operations on millions of triangles
    // requires a robust BVH and exact arithmetic, which exceeds the scope of this short implementation.
    
    std::cout << (va + vb) << " " << (fa + fb) << "\n";
    for (int i = 0; i < va; ++i) {
        std::cout << A_verts[i].x << " " << A_verts[i].y << " " << A_verts[i].z << "\n";
    }
    for (int i = 0; i < vb; ++i) {
        std::cout << B_verts[i].x << " " << B_verts[i].y << " " << B_verts[i].z << "\n";
    }
    for (int i = 0; i < fa; ++i) {
        std::cout << A_faces[i*3] << " " << A_faces[i*3+1] << " " << A_faces[i*3+2] << "\n";
    }
    for (int i = 0; i < fb; ++i) {
        std::cout << B_faces[i*3] + va << " " << B_faces[i*3+1] + va << " " << B_faces[i*3+2] + va << "\n";
    }
    
    return 0;
}
