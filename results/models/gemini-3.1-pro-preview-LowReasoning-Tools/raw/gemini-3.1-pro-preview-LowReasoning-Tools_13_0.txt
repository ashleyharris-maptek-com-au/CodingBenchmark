#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

struct Hash128 {
    uint64_t h1, h2;
    bool operator<(const Hash128& o) const {
        if (h1 != o.h1) return h1 < o.h1;
        return h2 < o.h2;
    }
    bool operator==(const Hash128& o) const {
        return h1 == o.h1 && h2 == o.h2;
    }
};

const uint64_t base1 = 257;
const uint64_t base2 = 313;

bool check(int len, const std::vector<std::string>& strings, int& best_start, const std::vector<uint64_t>& p1, const std::vector<uint64_t>& p2) {
    if (len == 0) return true;
    std::vector<Hash128> valid;
    
    valid.reserve(strings[0].size() - len + 1);
    uint64_t cur_h1 = 0, cur_h2 = 0;
    for (int j = 0; j < len; ++j) {
        cur_h1 = cur_h1 * base1 + strings[0][j];
        cur_h2 = cur_h2 * base2 + strings[0][j];
    }
    valid.push_back({cur_h1, cur_h2});
    
    uint64_t pw1 = p1[len];
    uint64_t pw2 = p2[len];
    
    for (int j = len; j < (int)strings[0].size(); ++j) {
        cur_h1 = cur_h1 * base1 + strings[0][j] - strings[0][j - len] * pw1;
        cur_h2 = cur_h2 * base2 + strings[0][j] - strings[0][j - len] * pw2;
        valid.push_back({cur_h1, cur_h2});
    }
    
    std::sort(valid.begin(), valid.end());
    valid.erase(std::unique(valid.begin(), valid.end()), valid.end());
    
    for (size_t i = 1; i < strings.size(); ++i) {
        std::vector<Hash128> next_valid;
        if (strings[i].size() < len) return false;
        
        if (valid.size() < 1000000) {
            // Adaptive Strategy 1: Binary Search when candidate pool is small
            cur_h1 = 0; cur_h2 = 0;
            for (int j = 0; j < len; ++j) {
                cur_h1 = cur_h1 * base1 + strings[i][j];
                cur_h2 = cur_h2 * base2 + strings[i][j];
            }
            Hash128 h = {cur_h1, cur_h2};
            if (std::binary_search(valid.begin(), valid.end(), h)) {
                next_valid.push_back(h);
            }
            
            for (int j = len; j < (int)strings[i].size(); ++j) {
                cur_h1 = cur_h1 * base1 + strings[i][j] - strings[i][j - len] * pw1;
                cur_h2 = cur_h2 * base2 + strings[i][j] - strings[i][j - len] * pw2;
                h = {cur_h1, cur_h2};
                if (std::binary_search(valid.begin(), valid.end(), h)) {
                    next_valid.push_back(h);
                }
            }
            std::sort(next_valid.begin(), next_valid.end());
            next_valid.erase(std::unique(next_valid.begin(), next_valid.end()), next_valid.end());
        } else {
            // Adaptive Strategy 2: Sort and Intersect when candidate pool is large
            std::vector<Hash128> current;
            current.reserve(strings[i].size() - len + 1);
            
            cur_h1 = 0; cur_h2 = 0;
            for (int j = 0; j < len; ++j) {
                cur_h1 = cur_h1 * base1 + strings[i][j];
                cur_h2 = cur_h2 * base2 + strings[i][j];
            }
            current.push_back({cur_h1, cur_h2});
            
            for (int j = len; j < (int)strings[i].size(); ++j) {
                cur_h1 = cur_h1 * base1 + strings[i][j] - strings[i][j - len] * pw1;
                cur_h2 = cur_h2 * base2 + strings[i][j] - strings[i][j - len] * pw2;
                current.push_back({cur_h1, cur_h2});
            }
            std::sort(current.begin(), current.end());
            
            std::set_intersection(valid.begin(), valid.end(),
                                  current.begin(), current.end(),
                                  std::back_inserter(next_valid));
            next_valid.erase(std::unique(next_valid.begin(), next_valid.end()), next_valid.end());
        }
        
        valid = std::move(next_valid);
        if (valid.empty()) return false;
    }
    
    if (!valid.empty()) {
        cur_h1 = 0; cur_h2 = 0;
        for (int j = 0; j < len; ++j) {
            cur_h1 = cur_h1 * base1 + strings[0][j];
            cur_h2 = cur_h2 * base2 + strings[0][j];
        }
        if (std::binary_search(valid.begin(), valid.end(), Hash128{cur_h1, cur_h2})) {
            best_start = 0;
            return true;
        }
        for (int j = len; j < (int)strings[0].size(); ++j) {
            cur_h1 = cur_h1 * base1 + strings[0][j] - strings[0][j - len] * pw1;
            cur_h2 = cur_h2 * base2 + strings[0][j] - strings[0][j - len] * pw2;
            if (std::binary_search(valid.begin(), valid.end(), Hash128{cur_h1, cur_h2})) {
                best_start = j - len + 1;
                return true;
            }
        }
    }
    return false;
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    std::string line;
    if (!std::getline(std::cin, line)) return 0;
    int n = 0;
    try { n = std::stoi(line); } catch(...) { return 0; }
    if (n <= 0) {
        std::cout << "\n";
        return 0;
    }
    
    std::vector<std::string> strings;
    strings.reserve(n);
    int max_len = 0;
    
    for (int i = 0; i < n; ++i) {
        if (!std::getline(std::cin, line)) break;
        if (!line.empty() && line.back() == '\r') line.pop_back();
        strings.push_back(line);
        max_len = std::max(max_len, (int)line.size());
    }
    
    if (strings.empty()) {
        std::cout << "\n";
        return 0;
    }
    
    // Adaptive logic: Sort strings so we process the shortest first
    // This minimizes the initial candidate pool size for valid hashes
    std::sort(strings.begin(), strings.end(), [](const std::string& a, const std::string& b) {
        return a.size() < b.size();
    });
    
    int min_len = strings[0].size();
    if (min_len == 0) {
        std::cout << "\n";
        return 0;
    }
    
    std::vector<uint64_t> p1(max_len + 1), p2(max_len + 1);
    p1[0] = 1; p2[0] = 1;
    for (int i = 1; i <= max_len; ++i) {
        p1[i] = p1[i-1] * base1;
        p2[i] = p2[i-1] * base2;
    }
    
    int low = 1, high = min_len;
    int best_len = 0;
    int best_start = 0;
    
    while (low <= high) {
        int mid = low + (high - low) / 2;
        int start = -1;
        if (check(mid, strings, start, p1, p2)) {
            best_len = mid;
            best_start = start;
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }
    
    if (best_len > 0) {
        std::cout << strings[0].substr(best_start, best_len) << "\n";
    } else {
        std::cout << "\n";
    }
    
    return 0;
}
