import os

asm = """
OpCapability Geometry
%1 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main InputTrianglesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 3
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_in Location 0

OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2

OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %gl_out Location 0

OpDecorate %outColor Location 0

%void = OpTypeVoid
%func = OpTypeFunction %void
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%int_0 = OpConstant %int 0
%int_2 = OpConstant %int 2
%int_4 = OpConstant %int 4
%uint_6 = OpConstant %uint 6
%float_0 = OpConstant %float 0.0
%float_0_2 = OpConstant %float 0.2
%float_0_8 = OpConstant %float 0.8
%float_0_3333333 = OpConstant %float 0.333333333
%v3float_L = OpConstantComposite %v3float %float_0 %float_0 %float_0 ; We'll override this

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_6
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3_in = OpTypeArray %v3float %uint_6
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%worldPos = OpVariable %ptr_in_arr_v3 Input
%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_v4 = OpTypePointer Output %v4float

%float_0_57735 = OpConstant %float 0.577350269
%v3float_lightDir = OpConstantComposite %v3float %float_0_57735 %float_0_57735 %float_0_57735

%main = OpFunction %void None %func
%label = OpLabel

; Get worldPos[0], worldPos[2], worldPos[4]
%wp0_ptr = OpAccessChain %ptr_in_v3 %worldPos %int_0
%wp0 = OpLoad %v3float %wp0_ptr
%wp2_ptr = OpAccessChain %ptr_in_v3 %worldPos %int_2
%wp2 = OpLoad %v3float %wp2_ptr
%wp4_ptr = OpAccessChain %ptr_in_v3 %worldPos %int_4
%wp4 = OpLoad %v3float %wp4_ptr

; p2 - p0
%v1 = OpFSub %v3float %wp2 %wp0
; p4 - p0
%v2 = OpFSub %v3float %wp4 %wp0

; Extract components of v1
%v1_x = OpCompositeExtract %float %v1 0
%v1_y = OpCompositeExtract %float %v1 1
%v1_z = OpCompositeExtract %float %v1 2

; Extract components of v2
%v2_x = OpCompositeExtract %float %v2 0
%v2_y = OpCompositeExtract %float %v2 1
%v2_z = OpCompositeExtract %float %v2 2

; c.x = v1.y * v2.z - v1.z * v2.y
%c_x_1 = OpFMul %float %v1_y %v2_z
%c_x_2 = OpFMul %float %v1_z %v2_y
%c_x = OpFSub %float %c_x_1 %c_x_2

; c.y = v1.z * v2.x - v1.x * v2.z
%c_y_1 = OpFMul %float %v1_z %v2_x
%c_y_2 = OpFMul %float %v1_x %v2_z
%c_y = OpFSub %float %c_y_1 %c_y_2

; c.z = v1.x * v2.y - v1.y * v2.x
%c_z_1 = OpFMul %float %v1_x %v2_y
%c_z_2 = OpFMul %float %v1_y %v2_x
%c_z = OpFSub %float %c_z_1 %c_z_2

%cross_unnorm = OpCompositeConstruct %v3float %c_x %c_y %c_z

; N = normalize(cross_unnorm)
%N = OpExtInst %v3float %1 Normalize %cross_unnorm

; dot(N, L)
%dotNL = OpDot %float %N %v3float_lightDir

; max(dotNL, 0.0)
%diffuse = OpExtInst %float %1 FMax %dotNL %float_0

; color[0], color[2], color[4]
%col0_ptr = OpAccessChain %ptr_in_v3 %color %int_0
%col0 = OpLoad %v3float %col0_ptr
%col2_ptr = OpAccessChain %ptr_in_v3 %color %int_2
%col2 = OpLoad %v3float %col2_ptr
%col4_ptr = OpAccessChain %ptr_in_v3 %color %int_4
%col4 = OpLoad %v3float %col4_ptr

; base = (col0 + col2 + col4) * 0.333333333
%sum1 = OpFAdd %v3float %col0 %col2
%sum2 = OpFAdd %v3float %sum1 %col4
%base = OpVectorTimesScalar %v3float %sum2 %float_0_3333333

; factor = 0.2 + 0.8 * diffuse
%diffuse_0_8 = OpFMul %float %float_0_8 %diffuse
%factor = OpFAdd %float %float_0_2 %diffuse_0_8

; final_color = base * factor
%final_color = OpVectorTimesScalar %v3float %base %factor

; -- Vertex 0 --
%gl_in_0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%pos0 = OpLoad %v4float %gl_in_0_ptr

%gl_out_pos_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_out_pos_ptr %pos0
OpStore %outColor %final_color
OpEmitVertex

; -- Vertex 2 --
%gl_in_2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%pos2 = OpLoad %v4float %gl_in_2_ptr

OpStore %gl_out_pos_ptr %pos2
OpStore %outColor %final_color
OpEmitVertex

; -- Vertex 4 --
%gl_in_4_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_4 %int_0
%pos4 = OpLoad %v4float %gl_in_4_ptr

OpStore %gl_out_pos_ptr %pos4
OpStore %outColor %final_color
OpEmitVertex

OpEndPrimitive
OpReturn
OpFunctionEnd
"""

with open("answer.txt", "w") as f:
    f.write(asm.strip() + "\n")
