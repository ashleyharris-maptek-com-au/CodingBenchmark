OpCapability Geometry
               OpExtension "SPV_KHR_storage_buffer_storage_class"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %color %gl_out %outColor
               OpExecutionMode %main InputLinesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 4
               OpExecutionMode %main Invocations 1

               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

               OpDecorate %color Location 2
               OpDecorate %outColor Location 0

       %void = OpTypeVoid
          %3 = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v4float = OpTypeVector %float 4
    %v3float = OpTypeVector %float 3
    %v2float = OpTypeVector %float 2
       %uint = OpTypeInt 32 0
     %uint_4 = OpConstant %uint 4
%gl_PerVertex_in = OpTypeStruct %v4float
  %arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
      %gl_in = OpVariable %ptr_in_arr_pv Input

        %int = OpTypeInt 32 1
      %int_0 = OpConstant %int 0
      %int_1 = OpConstant %int 1
      %int_2 = OpConstant %int 2
      %int_3 = OpConstant %int 3

 %ptr_in_v4 = OpTypePointer Input %v4float

   %arr_v3_4 = OpTypeArray %v3float %uint_4
 %ptr_in_arr_v3 = OpTypePointer Input %arr_v3_4
      %color = OpVariable %ptr_in_arr_v3 Input
  %ptr_in_v3 = OpTypePointer Input %v3float

%gl_PerVertex_out = OpTypeStruct %v4float
 %ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
     %gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v4 = OpTypePointer Output %v4float

 %ptr_out_v3 = OpTypePointer Output %v3float
   %outColor = OpVariable %ptr_out_v3 Output

%float_0_025 = OpConstant %float 0.025
    %float_0 = OpConstant %float 0.0

   %ext_glsl = OpExtInstImport "GLSL.std.450"

       %main = OpFunction %void None %3
          %5 = OpLabel

          ; t0 = normalize(gl_in[2].pos - gl_in[0].pos)
     %ptr_p0 = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
         %p0 = OpLoad %v4float %ptr_p0
     %ptr_p2 = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
         %p2 = OpLoad %v4float %ptr_p2
   %diff_p20 = OpFSub %v4float %p2 %p0
 %diff20_xy = OpVectorShuffle %v2float %diff_p20 %diff_p20 0 1
         %t0 = OpExtInst %v2float %ext_glsl 69 %diff20_xy  ; 69 is Normalize

          ; t1 = normalize(gl_in[3].pos - gl_in[1].pos)
     %ptr_p1 = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
         %p1 = OpLoad %v4float %ptr_p1
     %ptr_p3 = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
         %p3 = OpLoad %v4float %ptr_p3
   %diff_p31 = OpFSub %v4float %p3 %p1
 %diff31_xy = OpVectorShuffle %v2float %diff_p31 %diff_p31 0 1
         %t1 = OpExtInst %v2float %ext_glsl 69 %diff31_xy

          ; perp0 = vec2(-t0.y, t0.x)
       %t0_x = OpCompositeExtract %float %t0 0
       %t0_y = OpCompositeExtract %float %t0 1
      %nt0_y = OpFNegate %float %t0_y
      %perp0 = OpCompositeConstruct %v2float %nt0_y %t0_x

          ; perp1 = vec2(-t1.y, t1.x)
       %t1_x = OpCompositeExtract %float %t1 0
       %t1_y = OpCompositeExtract %float %t1 1
      %nt1_y = OpFNegate %float %t1_y
      %perp1 = OpCompositeConstruct %v2float %nt1_y %t1_x

          ; offset0 = perp0 * thickness
    %offset0 = OpVectorTimesScalar %v2float %perp0 %float_0_025
    %off0_v4 = OpCompositeConstruct %v4float %offset0 %float_0 %float_0

          ; offset1 = perp1 * thickness
    %offset1 = OpVectorTimesScalar %v2float %perp1 %float_0_025
    %off1_v4 = OpCompositeConstruct %v4float %offset1 %float_0 %float_0

          ; V0: start + offset0, color[1]
   %pos_v0   = OpFAdd %v4float %p1 %off0_v4
  %ptr_col_1 = OpAccessChain %ptr_in_v3 %color %int_1
      %col_1 = OpLoad %v3float %ptr_col_1
 %ptr_out_pos = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %ptr_out_pos %pos_v0
               OpStore %outColor %col_1
               OpEmitVertex

          ; V1: start - offset0, color[1]
   %pos_v1   = OpFSub %v4float %p1 %off0_v4
               OpStore %ptr_out_pos %pos_v1
               OpStore %outColor %col_1
               OpEmitVertex

          ; V2: end + offset1, color[2]
   %pos_v2   = OpFAdd %v4float %p2 %off1_v4
  %ptr_col_2 = OpAccessChain %ptr_in_v3 %color %int_2
      %col_2 = OpLoad %v3float %ptr_col_2
               OpStore %ptr_out_pos %pos_v2
               OpStore %outColor %col_2
               OpEmitVertex

          ; V3: end - offset1, color[2]
   %pos_v3   = OpFSub %v4float %p2 %off1_v4
               OpStore %ptr_out_pos %pos_v3
               OpStore %outColor %col_2
               OpEmitVertex

               OpEndPrimitive
               OpReturn
               OpFunctionEnd
