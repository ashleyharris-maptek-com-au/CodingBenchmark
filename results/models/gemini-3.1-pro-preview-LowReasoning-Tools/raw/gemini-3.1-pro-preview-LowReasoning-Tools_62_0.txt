declare const require: any;
const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);
    const { width, height, grid, ignitions, towns, cutBudget } = data;

    if (cutBudget === 0) {
        console.log(JSON.stringify({ firebreaks: [] }));
        return;
    }

    const isBurnable = new Uint8Array(width * height);
    const isTown = new Uint8Array(width * height);
    let totalTownValue = 0;
    
    for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
            if (grid[y][x] === '.') {
                isBurnable[y * width + x] = 1;
            }
        }
    }

    for (const t of towns) {
        isTown[t.y * width + t.x] = 1;
        totalTownValue += t.value;
    }
    
    // We cannot cut ignition cells
    for (const ig of ignitions) {
        isBurnable[ig[1] * width + ig[0]] = 0;
    }
    // We shouldn't cut town cells
    for (const t of towns) {
        isBurnable[t.y * width + t.x] = 0;
    }

    let candidates: number[] = [];
    for (let i = 0; i < width * height; i++) {
        if (isBurnable[i]) candidates.push(i);
    }

    function evaluate(cuts: number[]): number {
        const burned = new Uint8Array(width * height);
        const q = new Int32Array(width * height);
        let head = 0;
        let tail = 0;

        for (const ig of ignitions) {
            const idx = ig[1] * width + ig[0];
            burned[idx] = 1;
            q[tail++] = idx;
        }

        const cutSet = new Uint8Array(width * height);
        for (let i = 0; i < cuts.length; i++) cutSet[cuts[i]] = 1;

        while (head < tail) {
            const curr = q[head++];
            const cx = curr % width;
            const cy = Math.floor(curr / width);

            const neighbors = [
                curr - width, // up
                curr + width, // down
                curr - 1,     // left
                curr + 1      // right
            ];

            if (cy === 0) neighbors[0] = -1;
            if (cy === height - 1) neighbors[1] = -1;
            if (cx === 0) neighbors[2] = -1;
            if (cx === width - 1) neighbors[3] = -1;

            for (let i = 0; i < 4; i++) {
                const n = neighbors[i];
                if (n >= 0 && !burned[n] && !cutSet[n] && (grid[Math.floor(n/width)][n%width] === '.' || isTown[n])) {
                    burned[n] = 1;
                    q[tail++] = n;
                }
            }
        }

        let savedValue = 0;
        for (const t of towns) {
            if (!burned[t.y * width + t.x]) {
                savedValue += t.value;
            }
        }
        return savedValue;
    }

    let bestCuts: number[] = [];
    let bestScore = -1;

    // Simple randomized hill climbing
    const tStart = Date.now();
    let currentCuts: number[] = [];
    
    // Initialize with random valid candidates
    for (let i = 0; i < Math.min(cutBudget, candidates.length); i++) {
        let r = Math.floor(Math.random() * candidates.length);
        currentCuts.push(candidates[r]);
    }

    bestCuts = [...currentCuts];
    bestScore = evaluate(currentCuts);
    let currentScore = bestScore;

    let iters = 0;
    while (Date.now() - tStart < 8000) {
        iters++;
        if (currentCuts.length === 0) break;
        
        const replaceIdx = Math.floor(Math.random() * currentCuts.length);
        const oldCut = currentCuts[replaceIdx];
        const newCut = candidates[Math.floor(Math.random() * candidates.length)];
        
        currentCuts[replaceIdx] = newCut;
        const score = evaluate(currentCuts);
        
        if (score >= currentScore) {
            currentScore = score;
            if (score > bestScore) {
                bestScore = score;
                bestCuts = [...currentCuts];
            }
        } else {
            // Revert
            currentCuts[replaceIdx] = oldCut;
        }
    }

    const firebreaks = bestCuts.map(idx => [idx % width, Math.floor(idx / width)]);
    console.log(JSON.stringify({ firebreaks }));
}

solve();
