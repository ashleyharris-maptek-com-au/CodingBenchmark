```glsl
#version 450

layout(local_size_x = 256) in;

// Each boid stored as two vec4s: [pos.xyz, 0], [vel.xyz, 0]
// Boid i is at index i*2 and i*2+1
layout(set = 0, binding = 0) readonly buffer InputBoids {
    vec4 inData[];   // length = N * 2
};

layout(set = 0, binding = 1) buffer OutputBoids {
    vec4 outData[];  // length = N * 2
};

layout(set = 0, binding = 2) uniform Params {
    uvec4 counts;      // counts.x = N
    vec4 radii;        // radii.x=dt, radii.y=sepRadius, radii.z=alignRadius, radii.w=cohRadius
    vec4 weights;      // weights.x=sepWeight, weights.y=alignWeight, weights.z=cohWeight, weights.w=maxSpeed
    vec4 bounds;       // bounds.x=boundSize, bounds.y=boundForce
};

void main() {
    uint index = gl_GlobalInvocationID.x;
    uint N = counts.x;
    
    // Guard against out-of-bounds invocations
    if (index >= N) {
        return;
    }
    
    // Extract parameters
    float dt = radii.x;
    float sepRadius = radii.y;
    float alignRadius = radii.z;
    float cohRadius = radii.w;
    
    float sepWeight = weights.x;
    float alignWeight = weights.y;
    float cohWeight = weights.z;
    float maxSpeed = weights.w;
    
    float boundSize = bounds.x;
    float boundForce = bounds.y;
    
    // Read current boid data
    vec3 pos = inData[index * 2].xyz;
    vec3 vel = inData[index * 2 + 1].xyz;
    
    // Accumulators for rules
    vec3 sepAcc = vec3(0.0);
    vec3 alignVel = vec3(0.0);
    vec3 cohPos = vec3(0.0);
    
    int sepCount = 0;
    int alignCount = 0;
    int cohCount = 0;
    
    for (uint i = 0; i < N; ++i) {
        if (i == index) {
            continue; // Skip self
        }
        
        vec3 otherPos = inData[i * 2].xyz;
        vec3 otherVel = inData[i * 2 + 1].xyz;
        
        vec3 offset = pos - otherPos;
        float dist = length(offset);
        
        if (dist > 0.0) { // Skip zero-distance neighbors
            if (dist < sepRadius) {
                // Separation: steer away from neighbors
                sepAcc += normalize(offset);
                sepCount++;
            }
            if (dist < alignRadius) {
                // Alignment: steer towards average velocity of neighbors
                alignVel += otherVel;
                alignCount++;
            }
            if (dist < cohRadius) {
                // Cohesion: steer towards average position of neighbors
                cohPos += otherPos;
                cohCount++;
            }
        }
    }
    
    vec3 accel = vec3(0.0);
    
    // Average the accumulated forces by neighbor count to avoid division by zero
    if (sepCount > 0) {
        sepAcc /= float(sepCount);
        accel += sepAcc * sepWeight;
    }
    
    if (alignCount > 0) {
        alignVel /= float(alignCount);
        accel += (alignVel - vel) * alignWeight;
    }
    
    if (cohCount > 0) {
        cohPos /= float(cohCount);
        accel += (cohPos - pos) * cohWeight;
    }
    
    // Boundary force
    // apply the boundary force when a component exceeds boundSize*0.4 
    // (push toward center using boundForce and boundSize*0.5 scaling)
    for (int j = 0; j < 3; ++j) {
        if (abs(pos[j]) > boundSize * 0.4) {
            accel[j] -= (pos[j] / (boundSize * 0.5)) * boundForce;
        }
    }
    
    // Update velocity
    vel += accel * dt;
    
    // Clamp velocity magnitude to maxSpeed
    float speed = length(vel);
    if (speed > maxSpeed) {
        vel = (vel / speed) * maxSpeed;
    }
    
    // Update position
    pos += vel * dt;
    
    // Write back updated data
    outData[index * 2] = vec4(pos, 0.0);
    outData[index * 2 + 1] = vec4(vel, 0.0);
}
```
