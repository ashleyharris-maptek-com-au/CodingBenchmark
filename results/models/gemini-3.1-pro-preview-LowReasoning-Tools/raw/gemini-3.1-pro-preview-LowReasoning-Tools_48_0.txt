#version 450

layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;

layout(std430, binding = 0) readonly buffer InBuffer {
    float f_in[];
};

layout(std430, binding = 1) buffer OutBuffer {
    float f_out[];
};

layout(std140, binding = 2) uniform Params {
    uvec4 dims; // W, H, 0, 0
    vec4 params; // omega, 0, 0, 0
};

const int cx[9] = int[9](0, 1, 0, -1, 0, 1, -1, -1, 1);
const int cy[9] = int[9](0, 0, 1, 0, -1, 1, 1, -1, -1);
const float w[9] = float[9](
    4.0/9.0,
    1.0/9.0, 1.0/9.0, 1.0/9.0, 1.0/9.0,
    1.0/36.0, 1.0/36.0, 1.0/36.0, 1.0/36.0
);

void main() {
    uint W = dims.x;
    uint H = dims.y;
    float omega = params.x;

    uint x = gl_GlobalInvocationID.x;
    uint y = gl_GlobalInvocationID.y;

    if (x >= W || y >= H) {
        return;
    }

    uint base_idx = (y * W + x) * 9;
    
    float f[9];
    float rho = 0.0;
    float ux = 0.0;
    float uy = 0.0;
    
    for (int k = 0; k < 9; ++k) {
        f[k] = f_in[base_idx + k];
        rho += f[k];
        ux += f[k] * float(cx[k]);
        uy += f[k] * float(cy[k]);
    }
    
    if (rho > 0.0) {
        ux /= rho;
        uy /= rho;
    }

    float u2 = ux * ux + uy * uy;
    
    for (int k = 0; k < 9; ++k) {
        float cu = float(cx[k]) * ux + float(cy[k]) * uy;
        float feq = w[k] * rho * (1.0 + 3.0 * cu + 4.5 * cu * cu - 1.5 * u2);
        float fout_k = f[k] + omega * (feq - f[k]);
        
        int nx = int(x) + cx[k];
        if (nx < 0) nx += int(W);
        else if (nx >= int(W)) nx -= int(W);
        
        int ny = int(y) + cy[k];
        if (ny < 0) ny += int(H);
        else if (ny >= int(H)) ny -= int(H);
        
        uint n_idx = (uint(ny) * W + uint(nx)) * 9 + k;
        f_out[n_idx] = fout_k;
    }
}
