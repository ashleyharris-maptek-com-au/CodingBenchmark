#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

struct Rect {
    int id;
    int w, h;
};

struct Placed {
    int board_index;
    int x, y;
    int rotated;
};

struct FreeRect {
    int x, y, w, h;
};

struct Board {
    vector<FreeRect> free_rects;
    int max_dim;
    Board(int W, int H) {
        free_rects.push_back({0, 0, W, H});
        max_dim = max(W, H);
    }
    void update_max() {
        max_dim = 0;
        for (const auto& fr : free_rects) {
            max_dim = max(max_dim, max(fr.w, fr.h));
        }
    }
};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int N, W, H;
    if (!(cin >> N >> W >> H)) return 0;

    vector<Rect> rects(N);
    for (int i = 0; i < N; ++i) {
        rects[i].id = i;
        cin >> rects[i].w >> rects[i].h;
    }

    // Sort rectangles by area descending, then by maximum side
    sort(rects.begin(), rects.end(), [](const Rect& a, const Rect& b) {
        long long areaA = (long long)a.w * a.h;
        long long areaB = (long long)b.w * b.h;
        if (areaA != areaB) return areaA > areaB;
        return max(a.w, a.h) > max(b.w, b.h);
    });

    vector<Board> boards;
    vector<Placed> result(N);

    for (const auto& r : rects) {
        int best_board = -1;
        int best_free_idx = -1;
        bool best_rotated = false;
        int best_score1 = 2e9; // min short side
        int best_score2 = 2e9; // min long side
        
        int min_r_dim = min(r.w, r.h);
        
        // Find Best Short Side Fit
        for (int b = 0; b < boards.size(); ++b) {
            if (boards[b].max_dim < min_r_dim) continue;
            
            for (int f = 0; f < boards[b].free_rects.size(); ++f) {
                const auto& fr = boards[b].free_rects[f];
                
                // Try not rotated
                if (r.w <= fr.w && r.h <= fr.h) {
                    int short_side = min(fr.w - r.w, fr.h - r.h);
                    int long_side = max(fr.w - r.w, fr.h - r.h);
                    if (short_side < best_score1 || (short_side == best_score1 && long_side < best_score2)) {
                        best_score1 = short_side;
                        best_score2 = long_side;
                        best_board = b;
                        best_free_idx = f;
                        best_rotated = false;
                    }
                }
                
                // Try rotated
                if (r.h <= fr.w && r.w <= fr.h) {
                    int short_side = min(fr.w - r.h, fr.h - r.w);
                    int long_side = max(fr.w - r.h, fr.h - r.w);
                    if (short_side < best_score1 || (short_side == best_score1 && long_side < best_score2)) {
                        best_score1 = short_side;
                        best_score2 = long_side;
                        best_board = b;
                        best_free_idx = f;
                        best_rotated = true;
                    }
                }
            }
            if (best_score1 == 0 && best_score2 == 0) break; // perfect fit, prune search
        }
        
        // No fit found, add a new board
        if (best_board == -1) {
            boards.push_back(Board(W, H));
            best_board = boards.size() - 1;
            best_free_idx = 0;
            
            const auto& fr = boards[best_board].free_rects[best_free_idx];
            if (r.w <= fr.w && r.h <= fr.h) {
                best_rotated = false;
            } else {
                best_rotated = true;
            }
        }
        
        int bw = best_rotated ? r.h : r.w;
        int bh = best_rotated ? r.w : r.h;
        
        FreeRect fr = boards[best_board].free_rects[best_free_idx];
        boards[best_board].free_rects.erase(boards[best_board].free_rects.begin() + best_free_idx);
        
        // Split the remaining space (Guillotine split, Minimize Area Split)
        int dw = fr.w - bw;
        int dh = fr.h - bh;
        
        if (dw > 0 || dh > 0) {
            FreeRect f1, f2;
            if (dw > dh) { 
                f1 = {fr.x + bw, fr.y, dw, fr.h};
                f2 = {fr.x, fr.y + bh, bw, dh};
            } else {
                f1 = {fr.x + bw, fr.y, dw, bh};
                f2 = {fr.x, fr.y + bh, fr.w, dh};
            }
            if (f1.w > 0 && f1.h > 0) boards[best_board].free_rects.push_back(f1);
            if (f2.w > 0 && f2.h > 0) boards[best_board].free_rects.push_back(f2);
        }
        
        boards[best_board].update_max();
        result[r.id] = {best_board, fr.x, fr.y, best_rotated ? 1 : 0};
    }

    // Output
    cout << boards.size() << "\n";
    for (int i = 0; i < N; ++i) {
        cout << result[i].board_index << " " << result[i].x << " " << result[i].y << " " << result[i].rotated << "\n";
    }

    return 0;
}
