const fs = require('fs');

class PriorityQueue {
    constructor(compare) {
        this.data = [];
        this.compare = compare;
    }
    push(val) {
        this.data.push(val);
        this.up(this.data.length - 1);
    }
    pop() {
        if (this.data.length === 0) return null;
        if (this.data.length === 1) return this.data.pop();
        const top = this.data[0];
        this.data[0] = this.data.pop();
        this.down(0);
        return top;
    }
    up(i) {
        while (i > 0) {
            const p = Math.floor((i - 1) / 2);
            if (this.compare(this.data[i], this.data[p]) < 0) {
                const tmp = this.data[i];
                this.data[i] = this.data[p];
                this.data[p] = tmp;
                i = p;
            } else {
                break;
            }
        }
    }
    down(i) {
        const n = this.data.length;
        while (true) {
            let left = 2 * i + 1;
            let right = 2 * i + 2;
            let best = i;
            if (left < n && this.compare(this.data[left], this.data[best]) < 0) {
                best = left;
            }
            if (right < n && this.compare(this.data[right], this.data[best]) < 0) {
                best = right;
            }
            if (best !== i) {
                const tmp = this.data[i];
                this.data[i] = this.data[best];
                this.data[best] = tmp;
                i = best;
            } else {
                break;
            }
        }
    }
    get length() {
        return this.data.length;
    }
}

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    if (!input.trim()) return;
    const data = JSON.parse(input);
    const tasks = data.tasks;
    if (!tasks || tasks.length === 0) {
        console.log(JSON.stringify({ order: [] }));
        return;
    }
    
    const adj = new Map();
    const taskMap = new Map();
    const inDegree = new Map();
    
    for (const t of tasks) {
        taskMap.set(t.id, t);
        adj.set(t.id, []);
        inDegree.set(t.id, 0);
    }
    
    for (const t of tasks) {
        if (t.deps) {
            for (const dep of t.deps) {
                if (!adj.has(dep)) {
                    adj.set(dep, []);
                    inDegree.set(dep, 0);
                }
                adj.get(dep).push(t.id);
                inDegree.set(t.id, (inDegree.get(t.id) || 0) + 1);
            }
        }
    }
    
    const bottomLevel = new Map();
    function getBottomLevel(u) {
        if (bottomLevel.has(u)) return bottomLevel.get(u);
        let maxV = 0;
        const neighbors = adj.get(u) || [];
        for (const v of neighbors) {
            maxV = Math.max(maxV, getBottomLevel(v));
        }
        const t = taskMap.get(u);
        const val = (t ? t.duration : 0) + maxV;
        bottomLevel.set(u, val);
        return val;
    }
    
    for (const t of tasks) {
        getBottomLevel(t.id);
    }
    
    const compare = (a, b) => {
        const ta = taskMap.get(a);
        const tb = taskMap.get(b);
        const bla = bottomLevel.get(a);
        const blb = bottomLevel.get(b);
        
        if (bla !== blb) return blb - bla; // descending bottom level
        
        const da = ta && ta.deadline !== undefined ? ta.deadline : Infinity;
        const db = tb && tb.deadline !== undefined ? tb.deadline : Infinity;
        if (da !== db) return da - db; // ascending deadline
        
        const wa = ta && ta.weight !== undefined ? ta.weight : 0;
        const wb = tb && tb.weight !== undefined ? tb.weight : 0;
        if (wa !== wb) return wb - wa; // descending weight
        
        return a.localeCompare ? a.localeCompare(b) : (a < b ? -1 : (a > b ? 1 : 0));
    };
    
    const pq = new PriorityQueue(compare);
    
    for (const t of tasks) {
        if (inDegree.get(t.id) === 0) {
            pq.push(t.id);
        }
    }
    
    const order = [];
    while (pq.length > 0) {
        const nxt = pq.pop();
        order.push(nxt);
        
        const neighbors = adj.get(nxt) || [];
        for (const v of neighbors) {
            const ind = inDegree.get(v) - 1;
            inDegree.set(v, ind);
            if (ind === 0) {
                pq.push(v);
            }
        }
    }
    
    console.log(JSON.stringify({ order: order }));
}

solve();
