declare const require: any;
const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);
    
    const prob: number[][] = data.probability;
    const height = prob.length;
    const width = prob[0].length;
    const maxSteps = data.maxStepsPerDrone;
    const drones: {id: any, x: number, y: number}[] = data.drones;
    
    const visited = new Set<string>();
    
    const paths: [number, number][][] = [];
    for (const d of drones) {
        paths.push([[d.x, d.y]]);
        visited.add(`${d.x},${d.y}`);
    }
    
    const dx = [0, 1, 0, -1, 0];
    const dy = [1, 0, -1, 0, 0];
    
    const currPos = drones.map(d => ({x: d.x, y: d.y}));
    
    // To avoid multiple drones targeting the same far away cell simultaneously
    const claimed = new Set<string>();
    
    for (let step = 0; step < maxSteps; step++) {
        for (let i = 0; i < drones.length; i++) {
            const pos = currPos[i];
            
            let bestTarget: {x: number, y: number} | null = null;
            let bestScore = -1;
            let bestTargetKey = "";
            
            const q: {x: number, y: number, dist: number, fx: number, fy: number}[] = [];
            const bfsVisited = new Set<string>();
            
            for (let dir = 0; dir < 5; dir++) {
                const nx = pos.x + dx[dir];
                const ny = pos.y + dy[dir];
                if (nx >= 0 && nx < width && ny >= 0 && ny < height && prob[ny][nx] !== -1) {
                    q.push({x: nx, y: ny, dist: 1, fx: nx, fy: ny});
                    bfsVisited.add(`${nx},${ny}`);
                }
            }
            
            let qIdx = 0;
            const maxBfsDepth = 20;
            
            while (qIdx < q.length) {
                const curr = q[qIdx++];
                
                if (curr.dist > maxBfsDepth) break;
                
                const key = `${curr.x},${curr.y}`;
                if (!visited.has(key) && !claimed.has(key) && prob[curr.y][curr.x] > 0) {
                    const score = prob[curr.y][curr.x] / (curr.dist * curr.dist);
                    if (score > bestScore) {
                        bestScore = score;
                        bestTarget = {x: curr.fx, y: curr.fy};
                        bestTargetKey = key;
                    }
                }
                
                if (curr.dist < maxBfsDepth) {
                    for (let dir = 0; dir < 4; dir++) {
                        const nx = curr.x + dx[dir];
                        const ny = curr.y + dy[dir];
                        if (nx >= 0 && nx < width && ny >= 0 && ny < height && prob[ny][nx] !== -1) {
                            const nkey = `${nx},${ny}`;
                            if (!bfsVisited.has(nkey)) {
                                bfsVisited.add(nkey);
                                q.push({x: nx, y: ny, dist: curr.dist + 1, fx: curr.fx, fy: curr.fy});
                            }
                        }
                    }
                }
            }
            
            if (bestTarget) {
                pos.x = bestTarget.x;
                pos.y = bestTarget.y;
                if (bestTargetKey !== `${pos.x},${pos.y}`) {
                    claimed.add(bestTargetKey); // claim it so others don't go for it
                }
            } else {
                // If no target found, just pick any valid unvisited neighbor, or just valid
                let moved = false;
                for (let dir = 0; dir < 4; dir++) {
                    const nx = pos.x + dx[dir];
                    const ny = pos.y + dy[dir];
                    if (nx >= 0 && nx < width && ny >= 0 && ny < height && prob[ny][nx] !== -1) {
                        const key = `${nx},${ny}`;
                        if (!visited.has(key)) {
                            pos.x = nx;
                            pos.y = ny;
                            moved = true;
                            break;
                        }
                    }
                }
                if (!moved) {
                    for (let dir = 0; dir < 5; dir++) {
                        const nx = pos.x + dx[dir];
                        const ny = pos.y + dy[dir];
                        if (nx >= 0 && nx < width && ny >= 0 && ny < height && prob[ny][nx] !== -1) {
                            pos.x = nx;
                            pos.y = ny;
                            break;
                        }
                    }
                }
            }
            
            paths[i].push([pos.x, pos.y]);
            visited.add(`${pos.x},${pos.y}`);
        }
        claimed.clear(); // Clear claims each step to allow re-evaluating
    }
    
    console.log(JSON.stringify({paths}));
}

solve();
