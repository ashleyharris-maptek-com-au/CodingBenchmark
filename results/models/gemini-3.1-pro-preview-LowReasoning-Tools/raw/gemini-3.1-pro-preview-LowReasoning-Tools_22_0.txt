using System;
using System.Collections.Generic;

class TetrisPlayer
{
    static int width;
    static int height;
    static int numPieces;
    static bool[,] board; // true if filled, x is 0..width-1, y is 0..height-1

    struct Placement
    {
        public int rotation;
        public int column;
        public double score;
    }

    // Piece shapes for each rotation. A shape is a list of (dx, dy) where dx is right, dy is up.
    // Rotations are 90 degrees clockwise.
    static Dictionary<string, List<List<(int x, int y)>>> pieces = new Dictionary<string, List<List<(int x, int y)>>>();

    static void InitPieces()
    {
        // I
        pieces["I"] = new List<List<(int x, int y)>> {
            new List<(int, int)> { (0, 0), (1, 0), (2, 0), (3, 0) }, // Rot 0: horizontal
            new List<(int, int)> { (0, 0), (0, 1), (0, 2), (0, 3) }, // Rot 1: vertical
            new List<(int, int)> { (0, 0), (1, 0), (2, 0), (3, 0) }, // Rot 2
            new List<(int, int)> { (0, 0), (0, 1), (0, 2), (0, 3) }  // Rot 3
        };
        // O
        pieces["O"] = new List<List<(int x, int y)>> {
            new List<(int, int)> { (0, 0), (1, 0), (0, 1), (1, 1) },
            new List<(int, int)> { (0, 0), (1, 0), (0, 1), (1, 1) },
            new List<(int, int)> { (0, 0), (1, 0), (0, 1), (1, 1) },
            new List<(int, int)> { (0, 0), (1, 0), (0, 1), (1, 1) }
        };
        // T
        pieces["T"] = new List<List<(int x, int y)>> {
            new List<(int, int)> { (0, 0), (1, 0), (2, 0), (1, 1) }, // _|_
            new List<(int, int)> { (0, 1), (1, 0), (1, 1), (1, 2) }, // |-
            new List<(int, int)> { (1, 0), (0, 1), (1, 1), (2, 1) }, // T
            new List<(int, int)> { (0, 0), (0, 1), (0, 2), (1, 1) }  // -|
        };
        // S
        pieces["S"] = new List<List<(int x, int y)>> {
            new List<(int, int)> { (0, 0), (1, 0), (1, 1), (2, 1) }, // bottom left to top right
            new List<(int, int)> { (1, 0), (1, 1), (0, 1), (0, 2) },
            new List<(int, int)> { (0, 0), (1, 0), (1, 1), (2, 1) },
            new List<(int, int)> { (1, 0), (1, 1), (0, 1), (0, 2) }
        };
        // Z
        pieces["Z"] = new List<List<(int x, int y)>> {
            new List<(int, int)> { (0, 1), (1, 1), (1, 0), (2, 0) }, // top left to bottom right
            new List<(int, int)> { (0, 0), (0, 1), (1, 1), (1, 2) },
            new List<(int, int)> { (0, 1), (1, 1), (1, 0), (2, 0) },
            new List<(int, int)> { (0, 0), (0, 1), (1, 1), (1, 2) }
        };
        // J
        pieces["J"] = new List<List<(int x, int y)>> {
            new List<(int, int)> { (0, 0), (1, 0), (2, 0), (0, 1) }, 
            new List<(int, int)> { (0, 0), (1, 0), (1, 1), (1, 2) }, 
            new List<(int, int)> { (2, 0), (0, 1), (1, 1), (2, 1) }, 
            new List<(int, int)> { (0, 0), (0, 1), (0, 2), (1, 2) }  
        };
        // L
        pieces["L"] = new List<List<(int x, int y)>> {
            new List<(int, int)> { (0, 0), (1, 0), (2, 0), (2, 1) }, 
            new List<(int, int)> { (0, 0), (1, 0), (0, 1), (0, 2) }, 
            new List<(int, int)> { (0, 0), (0, 1), (1, 1), (2, 1) }, 
            new List<(int, int)> { (1, 0), (1, 1), (1, 2), (0, 2) }  
        };
    }

    static void Main(string[] args)
    {
        InitPieces();
        string[] initParams = Console.ReadLine().Split(new char[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        if (initParams.Length < 3) return;
        width = int.Parse(initParams[0]);
        height = int.Parse(initParams[1]);
        numPieces = int.Parse(initParams[2]);

        board = new bool[width, height];

        for (int p = 0; p < numPieces; p++)
        {
            string pieceType = Console.ReadLine()?.Trim();
            if (string.IsNullOrEmpty(pieceType)) break;

            Placement best = FindBestPlacement(pieceType, p);
            Console.WriteLine($"{best.rotation} {best.column}");

            ApplyPlacement(pieceType, best.rotation, best.column);

            string result = Console.ReadLine()?.Trim();
            if (result == "gameover" || result == null)
            {
                break;
            }
        }
    }

    static void ApplyPlacement(string type, int rotation, int column)
    {
        var shape = pieces[type][rotation];
        int bottomY = height;
        for (int y = height - 1; y >= 0; y--)
        {
            bool collision = false;
            foreach (var block in shape)
            {
                int bx = column + block.x;
                int by = y + block.y;
                if (by < 0 || by >= height || board[bx, by])
                {
                    collision = true;
                    break;
                }
            }
            if (collision)
            {
                bottomY = y + 1;
                break;
            }
            if (y == 0) bottomY = 0;
        }

        foreach (var block in shape)
        {
            int bx = column + block.x;
            int by = bottomY + block.y;
            if (by < height) board[bx, by] = true;
        }

        // Clear lines
        int yCheck = 0;
        while (yCheck < height)
        {
            bool full = true;
            for (int x = 0; x < width; x++)
            {
                if (!board[x, yCheck])
                {
                    full = false;
                    break;
                }
            }
            if (full)
            {
                for (int y2 = yCheck; y2 < height - 1; y2++)
                {
                    for (int x = 0; x < width; x++)
                    {
                        board[x, y2] = board[x, y2 + 1];
                    }
                }
                for (int x = 0; x < width; x++) board[x, height - 1] = false;
            }
            else
            {
                yCheck++;
            }
        }
    }

    static Placement FindBestPlacement(string type, int pieceIndex)
    {
        Placement best = new Placement { rotation = 0, column = 0, score = double.MaxValue };

        // Adaptive strategy based on game complexity (piece count)
        double a = 0.510066;
        double b = -0.760666;
        double c = 0.356630;
        double d = 0.184483;

        if (pieceIndex > 1000)
        {
            // More aggressive hole penalty later in the game
            c = 0.5;
            a = 0.6;
        }

        for (int r = 0; r < 4; r++)
        {
            var shape = pieces[type][r];
            int pieceWidth = 0;
            foreach (var block in shape) if (block.x + 1 > pieceWidth) pieceWidth = block.x + 1;

            for (int c_idx = 0; c_idx <= width - pieceWidth; c_idx++)
            {
                double score = EvaluatePlacement(shape, c_idx, a, b, c, d);
                if (score < best.score)
                {
                    best.score = score;
                    best.rotation = r;
                    best.column = c_idx;
                }
            }
        }
        return best;
    }

    static double EvaluatePlacement(List<(int x, int y)> shape, int column, double w_h, double w_l, double w_hole, double w_bump)
    {
        bool[,] tempBoard = (bool[,])board.Clone();

        int bottomY = height;
        for (int y = height - 1; y >= 0; y--)
        {
            bool collision = false;
            foreach (var block in shape)
            {
                int bx = column + block.x;
                int by = y + block.y;
                if (by < 0 || by >= height || tempBoard[bx, by])
                {
                    collision = true;
                    break;
                }
            }
            if (collision)
            {
                bottomY = y + 1;
                break;
            }
            if (y == 0) bottomY = 0;
        }

        foreach (var block in shape)
        {
            int bx = column + block.x;
            int by = bottomY + block.y;
            if (by >= height) return double.MaxValue; // Invalid placement
            tempBoard[bx, by] = true;
        }

        int linesCleared = 0;
        for (int y = 0; y < height; y++)
        {
            bool full = true;
            for (int x = 0; x < width; x++)
            {
                if (!tempBoard[x, y])
                {
                    full = false;
                    break;
                }
            }
            if (full) linesCleared++;
        }

        int[] heights = new int[width];
        for (int x = 0; x < width; x++)
        {
            for (int y = height - 1; y >= 0; y--)
            {
                if (tempBoard[x, y])
                {
                    heights[x] = y + 1;
                    break;
                }
            }
        }

        int aggregateHeight = 0;
        for (int x = 0; x < width; x++) aggregateHeight += heights[x];

        int holes = 0;
        for (int x = 0; x < width; x++)
        {
            bool blockAbove = false;
            for (int y = height - 1; y >= 0; y--)
            {
                if (tempBoard[x, y]) blockAbove = true;
                else if (blockAbove) holes++;
            }
        }

        int bumpiness = 0;
        for (int x = 0; x < width - 1; x++)
        {
            bumpiness += Math.Abs(heights[x] - heights[x + 1]);
        }

        return (w_h * aggregateHeight) + (w_l * linesCleared) + (w_hole * holes) + (w_bump * bumpiness);
    }
}
