#version 450
layout(location = 0) in vec3 worldPos;
layout(location = 1) in vec3 normal;
layout(location = 2) in vec2 uv;
layout(location = 3) in vec3 tangent;
layout(location = 4) in vec3 color;

layout(location = 0) out vec4 outColor;

layout(std140, set = 0, binding = 0) uniform UBO {
  mat4 model;
  mat4 view;
  mat4 projection;
  vec4 lightPos;
  vec4 cameraPos;
  vec4 params;
} ubo;

void main() {
    // Lambertian diffuse
    vec3 n = normalize(normal);
    vec3 l = normalize(ubo.lightPos.xyz - worldPos);
    float diffuse = max(dot(n, l), 0.0);
    float lighting = 0.4 + 0.6 * diffuse;

    // Brick pattern
    vec2 scale = vec2(8.0, 16.0);
    vec2 scaled_uv = uv * scale;

    // Staggered rows (half-brick offset every other row)
    float offset = step(0.5, fract(scaled_uv.y * 0.5)) * 0.5;
    vec2 offset_uv = vec2(scaled_uv.x + offset, scaled_uv.y);

    // Mortar mask
    vec2 fract_uv = fract(offset_uv);
    // mortar thickness is 0.05 in X and 0.1 in Y
    float x_mortar = step(fract_uv.x, 0.05);
    float y_mortar = step(fract_uv.y, 0.1);
    float is_mortar = max(x_mortar, y_mortar);

    // Colors
    vec3 brickColor = vec3(0.7, 0.2, 0.1);
    vec3 mortarColor = vec3(0.7, 0.7, 0.7);
    vec3 texColor = mix(brickColor, mortarColor, is_mortar);

    vec3 finalColor = texColor * lighting;
    outColor = vec4(finalColor, 1.0);
}
