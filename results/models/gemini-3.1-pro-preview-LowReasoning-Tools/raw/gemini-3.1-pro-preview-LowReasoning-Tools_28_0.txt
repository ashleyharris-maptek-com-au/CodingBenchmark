use std::io::{self, Read, Write};

struct Scanner<R: Read> {
    reader: R,
    buffer: Vec<u8>,
    pos: usize,
    len: usize,
}

impl<R: Read> Scanner<R> {
    fn new(reader: R) -> Self {
        Self {
            reader,
            buffer: vec![0; 1 << 16],
            pos: 0,
            len: 0,
        }
    }

    fn next_u32(&mut self) -> Option<u32> {
        let mut res = 0;
        let mut found = false;
        loop {
            if self.pos >= self.len {
                self.len = self.reader.read(&mut self.buffer).unwrap_or(0);
                self.pos = 0;
                if self.len == 0 {
                    break;
                }
            }
            let b = self.buffer[self.pos];
            self.pos += 1;
            if b >= b'0' && b <= b'9' {
                res = res * 10 + (b - b'0') as u32;
                found = true;
            } else if found {
                break;
            }
        }
        if found {
            Some(res)
        } else {
            None
        }
    }
}

fn remove_from_bucket(
    v: u32,
    d: u32,
    bucket_head: &mut [u32],
    bucket_next: &mut [u32],
    bucket_prev: &mut [u32],
) {
    let nxt = bucket_next[v as usize];
    let prv = bucket_prev[v as usize];
    if prv != !0 {
        bucket_next[prv as usize] = nxt;
    } else {
        bucket_head[d as usize] = nxt;
    }
    if nxt != !0 {
        bucket_prev[nxt as usize] = prv;
    }
    bucket_next[v as usize] = !0;
    bucket_prev[v as usize] = !0;
}

fn add_to_bucket(
    v: u32,
    d: u32,
    bucket_head: &mut [u32],
    bucket_next: &mut [u32],
    bucket_prev: &mut [u32],
) {
    let nxt = bucket_head[d as usize];
    bucket_next[v as usize] = nxt;
    bucket_prev[v as usize] = !0;
    if nxt != !0 {
        bucket_prev[nxt as usize] = v;
    }
    bucket_head[d as usize] = v;
}

fn main() {
    let stdin = io::stdin();
    let mut scanner = Scanner::new(stdin.lock());

    let num_vertices = match scanner.next_u32() {
        Some(v) => v,
        None => return,
    };
    let num_edges = scanner.next_u32().unwrap();

    let mut edges = Vec::with_capacity(num_edges as usize);
    let mut degree = vec![0u32; num_vertices as usize];

    for _ in 0..num_edges {
        let u = scanner.next_u32().unwrap();
        let v = scanner.next_u32().unwrap();
        if u != v {
            edges.push((u, v));
            degree[u as usize] += 1;
            degree[v as usize] += 1;
        }
    }

    let actual_edges = edges.len() as u32;

    let mut head = vec![!0u32; num_vertices as usize];
    let mut next = vec![!0u32; (actual_edges * 2) as usize];
    let mut to = vec![0u32; (actual_edges * 2) as usize];
    let mut edge_cnt = 0;

    for &(u, v) in &edges {
        to[edge_cnt] = v;
        next[edge_cnt] = head[u as usize];
        head[u as usize] = edge_cnt as u32;
        edge_cnt += 1;

        to[edge_cnt] = u;
        next[edge_cnt] = head[v as usize];
        head[v as usize] = edge_cnt as u32;
        edge_cnt += 1;
    }

    let mut bucket_head = vec![!0u32; (num_vertices + 1) as usize];
    let mut bucket_next = vec![!0u32; num_vertices as usize];
    let mut bucket_prev = vec![!0u32; num_vertices as usize];

    let mut active = vec![true; num_vertices as usize];
    let mut in_cover = vec![false; num_vertices as usize];
    let mut cur_degree = degree.clone();

    for v in 0..num_vertices {
        let d = cur_degree[v as usize];
        if d > 0 {
            add_to_bucket(v, d, &mut bucket_head, &mut bucket_next, &mut bucket_prev);
        } else {
            active[v as usize] = false;
        }
    }

    let mut max_d = num_vertices;

    loop {
        let mut chosen = !0;
        
        if bucket_head[1] != !0 {
            let u = bucket_head[1];
            let mut e = head[u as usize];
            while e != !0 {
                let neighbor = to[e as usize];
                if active[neighbor as usize] {
                    chosen = neighbor;
                    break;
                }
                e = next[e as usize];
            }
            if chosen == !0 {
                active[u as usize] = false;
                remove_from_bucket(u, 1, &mut bucket_head, &mut bucket_next, &mut bucket_prev);
                continue;
            }
        } else {
            while max_d > 0 && bucket_head[max_d as usize] == !0 {
                max_d -= 1;
            }
            if max_d == 0 {
                break; 
            }
            chosen = bucket_head[max_d as usize];
        }

        in_cover[chosen as usize] = true;
        active[chosen as usize] = false;
        let d_chosen = cur_degree[chosen as usize];
        remove_from_bucket(chosen, d_chosen, &mut bucket_head, &mut bucket_next, &mut bucket_prev);

        let mut e = head[chosen as usize];
        while e != !0 {
            let neighbor = to[e as usize];
            if active[neighbor as usize] {
                let d_neighbor = cur_degree[neighbor as usize];
                remove_from_bucket(neighbor, d_neighbor, &mut bucket_head, &mut bucket_next, &mut bucket_prev);
                cur_degree[neighbor as usize] -= 1;
                let new_d = cur_degree[neighbor as usize];
                if new_d > 0 {
                    add_to_bucket(neighbor, new_d, &mut bucket_head, &mut bucket_next, &mut bucket_prev);
                } else {
                    active[neighbor as usize] = false; 
                }
            }
            e = next[e as usize];
        }
    }

    for v in 0..num_vertices {
        if in_cover[v as usize] {
            let mut redundant = true;
            let mut e = head[v as usize];
            while e != !0 {
                let neighbor = to[e as usize];
                if !in_cover[neighbor as usize] {
                    redundant = false;
                    break;
                }
                e = next[e as usize];
            }
            if redundant {
                in_cover[v as usize] = false;
            }
        }
    }

    let mut cover_vertices = Vec::new();
    for v in 0..num_vertices {
        if in_cover[v as usize] {
            cover_vertices.push(v);
        }
    }

    let stdout = io::stdout();
    let mut out = io::BufWriter::new(stdout.lock());
    
    writeln!(out, "{}", cover_vertices.len()).unwrap();
    for (i, &v) in cover_vertices.iter().enumerate() {
        if i > 0 {
            write!(out, " ").unwrap();
        }
        write!(out, "{}", v).unwrap();
    }
    writeln!(out).unwrap();
}
