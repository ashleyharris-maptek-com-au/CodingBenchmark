using System;
using System.Collections.Generic;
using System.Linq;

namespace MinesweeperSolver
{
    class Program
    {
        static void Main(string[] args)
        {
            var lines = new List<string>();
            string line;
            while ((line = Console.ReadLine()) != null)
            {
                lines.Add(line);
            }

            if (lines.Count == 0) return;

            string[] dims = lines[0].Split(new[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
            int h = int.Parse(dims[0]);
            int w = int.Parse(dims[1]);

            char[][] board = new char[h][];
            for (int i = 0; i < h; i++)
            {
                board[i] = lines[i + 1].ToCharArray();
            }

            Solver solver = new Solver(h, w, board);
            var safeCells = solver.Solve();

            Console.WriteLine(safeCells.Count);
            foreach (var cell in safeCells)
            {
                Console.WriteLine($"{cell.X} {cell.Y}");
            }
        }
    }

    class Solver
    {
        int H, W;
        char[][] board;
        bool[,] isSafe;
        bool[,] isMine;

        public Solver(int h, int w, char[][] board)
        {
            H = h;
            W = w;
            this.board = board;
            isSafe = new bool[H, W];
            isMine = new bool[H, W];

            for (int y = 0; y < H; y++)
            {
                for (int x = 0; x < W; x++)
                {
                    if (board[y][x] == '*')
                        isMine[y, x] = true;
                }
            }
        }

        public List<(int X, int Y)> Solve()
        {
            bool changed = true;
            while (changed)
            {
                changed = false;
                changed |= SimpleDeduction();
                if (!changed)
                {
                    changed |= SubsetDeduction();
                }
            }

            List<(int X, int Y)> result = new List<(int, int)>();
            for (int y = 0; y < H; y++)
            {
                for (int x = 0; x < W; x++)
                {
                    if (board[y][x] == ' ' && isSafe[y, x])
                    {
                        result.Add((x, y));
                    }
                }
            }
            
            return result.OrderBy(c => c.Y).ThenBy(c => c.X).ToList();
        }

        bool SimpleDeduction()
        {
            bool changed = false;
            for (int y = 0; y < H; y++)
            {
                for (int x = 0; x < W; x++)
                {
                    if (board[y][x] >= '0' && board[y][x] <= '8')
                    {
                        int targetMines = board[y][x] - '0';
                        var neighbors = GetNeighbors(x, y);
                        
                        int knownMines = 0;
                        int unknownCount = 0;
                        List<(int X, int Y)> unknowns = new List<(int, int)>();

                        foreach (var n in neighbors)
                        {
                            if (isMine[n.Y, n.X]) knownMines++;
                            else if (board[n.Y][n.X] == ' ' && !isSafe[n.Y, n.X])
                            {
                                unknownCount++;
                                unknowns.Add(n);
                            }
                        }

                        if (unknownCount > 0)
                        {
                            if (knownMines == targetMines)
                            {
                                foreach (var u in unknowns)
                                {
                                    isSafe[u.Y, u.X] = true;
                                    changed = true;
                                }
                            }
                            else if (knownMines + unknownCount == targetMines)
                            {
                                foreach (var u in unknowns)
                                {
                                    isMine[u.Y, u.X] = true;
                                    changed = true;
                                }
                            }
                        }
                    }
                }
            }
            return changed;
        }

        bool SubsetDeduction()
        {
            bool changed = false;
            var equations = new List<Equation>();

            for (int y = 0; y < H; y++)
            {
                for (int x = 0; x < W; x++)
                {
                    if (board[y][x] >= '0' && board[y][x] <= '8')
                    {
                        int targetMines = board[y][x] - '0';
                        var neighbors = GetNeighbors(x, y);
                        
                        int knownMines = 0;
                        HashSet<(int X, int Y)> unknowns = new HashSet<(int, int)>();

                        foreach (var n in neighbors)
                        {
                            if (isMine[n.Y, n.X]) knownMines++;
                            else if (board[n.Y][n.X] == ' ' && !isSafe[n.Y, n.X])
                            {
                                unknowns.Add(n);
                            }
                        }

                        if (unknowns.Count > 0)
                        {
                            equations.Add(new Equation { Variables = unknowns, Mines = targetMines - knownMines, X = x, Y = y });
                        }
                    }
                }
            }

            for (int i = 0; i < equations.Count; i++)
            {
                for (int j = i + 1; j < equations.Count; j++)
                {
                    var eq1 = equations[i];
                    var eq2 = equations[j];

                    if (Math.Abs(eq1.X - eq2.X) > 2 || Math.Abs(eq1.Y - eq2.Y) > 2) continue;

                    if (eq1.Variables.IsSupersetOf(eq2.Variables) && eq1.Variables.Count > eq2.Variables.Count)
                    {
                        int mineDiff = eq1.Mines - eq2.Mines;
                        var diffSet = new HashSet<(int X, int Y)>(eq1.Variables);
                        diffSet.ExceptWith(eq2.Variables);

                        if (mineDiff == 0)
                        {
                            foreach (var v in diffSet)
                            {
                                if (!isSafe[v.Y, v.X])
                                {
                                    isSafe[v.Y, v.X] = true;
                                    changed = true;
                                }
                            }
                        }
                        else if (mineDiff == diffSet.Count)
                        {
                            foreach (var v in diffSet)
                            {
                                if (!isMine[v.Y, v.X])
                                {
                                    isMine[v.Y, v.X] = true;
                                    changed = true;
                                }
                            }
                        }
                    }
                    else if (eq2.Variables.IsSupersetOf(eq1.Variables) && eq2.Variables.Count > eq1.Variables.Count)
                    {
                        int mineDiff = eq2.Mines - eq1.Mines;
                        var diffSet = new HashSet<(int X, int Y)>(eq2.Variables);
                        diffSet.ExceptWith(eq1.Variables);

                        if (mineDiff == 0)
                        {
                            foreach (var v in diffSet)
                            {
                                if (!isSafe[v.Y, v.X])
                                {
                                    isSafe[v.Y, v.X] = true;
                                    changed = true;
                                }
                            }
                        }
                        else if (mineDiff == diffSet.Count)
                        {
                            foreach (var v in diffSet)
                            {
                                if (!isMine[v.Y, v.X])
                                {
                                    isMine[v.Y, v.X] = true;
                                    changed = true;
                                }
                            }
                        }
                    }
                }
                if (changed) break; 
            }

            return changed;
        }

        class Equation
        {
            public HashSet<(int X, int Y)> Variables;
            public int Mines;
            public int X, Y;
        }

        List<(int X, int Y)> GetNeighbors(int x, int y)
        {
            List<(int, int)> list = new List<(int, int)>();
            for (int dy = -1; dy <= 1; dy++)
            {
                for (int dx = -1; dx <= 1; dx++)
                {
                    if (dx == 0 && dy == 0) continue;
                    int nx = x + dx;
                    int ny = y + dy;
                    if (nx >= 0 && nx < W && ny >= 0 && ny < H)
                    {
                        list.Add((nx, ny));
                    }
                }
            }
            return list;
        }
    }
}
