state_memory = {}

def autopilot_step(state: dict, dt: float) -> dict:
    global state_memory
    
    # Helpers for redundant sensors
    def valid_avg(prefix, suffix='_valid', val_suffix='_dots', count=3):
        vals = []
        for i in range(1, count + 1):
            if state.get(f'{prefix}_{i}{suffix}', False):
                vals.append(state.get(f'{prefix}_{i}{val_suffix}', 0.0))
        return sum(vals)/len(vals) if vals else 0.0
        
    gs_err = valid_avg('glideslope')
    loc_err = valid_avg('localizer')
    
    # Altimeter
    rad_alt = state.get('radio_alt_ft', 10000.0)
    if rad_alt < 0 or rad_alt > 50000:
        baros = []
        for i in range(1, 4):
            if state.get(f'baro_alt_{i}_valid', False):
                baros.append(state.get(f'baro_alt_{i}_ft', 10000.0))
        if baros:
            rad_alt = sum(baros)/len(baros)
            
    # Airspeed
    ias = state.get('target_vref_kt', 137.0)
    for i in range(1, 4):
        if state.get(f'ias_{i}_valid', False):
            ias = state.get(f'ias_{i}_kt', ias)
            break
            
    vref = state.get('target_vref_kt', 137.0)
    
    # Attitude
    pitch = state.get('irs_1_pitch_deg', 0.0)
    roll = state.get('irs_1_roll_deg', 0.0)
    hdg = state.get('irs_1_heading_deg', 0.0)
    track = state.get('irs_1_track_deg', 0.0)
    vs = state.get('irs_1_vs_fpm', 0.0)
    
    on_ground = state.get('weight_on_wheels', False) or state.get('on_ground', False)
    
    e1_run = state.get('engine_1_running', True)
    e2_run = state.get('engine_2_running', True)
    
    # Go-Around Logic
    go_around = state_memory.get('go_around', False)
    
    if not on_ground and rad_alt < 1000:
        # Unstable approach criteria
        if abs(gs_err) > 2.0 or abs(loc_err) > 2.0 or vs < -1500 or ias > vref + 30:
            go_around = True
            
    # Go-around reset if we climbed high enough
    if go_around and rad_alt > 1500:
        go_around = False
        
    state_memory['go_around'] = go_around
    
    # Control commands
    target_pitch = 0.0
    target_roll = 0.0
    throttle_cmd = 0.0
    rudder_cmd = 0.0
    flaps = 30.0
    gear = 1.0
    
    if go_around:
        target_pitch = 15.0
        target_roll = 0.0
        throttle_cmd = 1.0
        flaps = 15.0
        gear = 0.0
    else:
        # Normal Approach
        # Glideslope tracking
        if rad_alt < 40 and not on_ground:
            target_pitch = 3.0 # Flare
        else:
            # gs_err: positive means we are above glideslope -> pitch down
            target_pitch = -gs_err * 3.0 
            target_pitch += max(0, (vref - ias) * 0.1) # speed protection
            
        # Localizer tracking
        target_roll = loc_err * 15.0
        
        # Crosswind and crab
        crab_angle = track - hdg
        # normalize crab_angle
        while crab_angle > 180: crab_angle -= 360
        while crab_angle < -180: crab_angle += 360
        
        if rad_alt < 150 and not on_ground:
            # Decrab
            rudder_cmd = crab_angle * 0.05
            target_roll -= crab_angle * 0.2
            
        # Speed control
        if rad_alt < 30:
            throttle_cmd = 0.0 # Idle thrust
        else:
            target_speed = vref + 5.0
            speed_err = target_speed - ias
            # Basic PI
            integ = state_memory.get('speed_int', 0.0) + speed_err * dt
            state_memory['speed_int'] = max(-5.0, min(5.0, integ))
            throttle_cmd = 0.5 + speed_err * 0.02 + integ * 0.005
            
        if ias > 200: flaps = 0.0
        elif ias > 160: flaps = 15.0
        else: flaps = 30.0
        
        gear = 1.0 if rad_alt < 3000 else 0.0
        
    # PD controllers for pitch/roll
    elev_cmd = (target_pitch - pitch) * 0.1 - state.get('irs_1_pitch_rate', 0.0) * 0.05
    ail_cmd = (target_roll - roll) * 0.05 - state.get('irs_1_roll_rate', 0.0) * 0.02
    
    # Asymmetric thrust compensation
    if e1_run and not e2_run: rudder_cmd += 0.2
    if e2_run and not e1_run: rudder_cmd -= 0.2
    
    # Ground handling
    if on_ground:
        throttle_cmd = 0.0
        elev_cmd = -1.0 # Nose down
        ail_cmd = 0.0
        rudder_cmd = loc_err * 0.5
        spoiler = 1.0
        autobrake = 1.0
        rev = 1.0 if state.get('groundspeed_kt', 100) > 60 else 0.0
    else:
        spoiler = 0.0
        autobrake = 0.0
        rev = 0.0
        
    return {
        'elevator': max(-1.0, min(1.0, elev_cmd)),
        'aileron': max(-1.0, min(1.0, ail_cmd)),
        'rudder': max(-1.0, min(1.0, rudder_cmd)),
        'throttle_1': max(0.0, min(1.0, throttle_cmd)) if e1_run else 0.0,
        'throttle_2': max(0.0, min(1.0, throttle_cmd)) if e2_run else 0.0,
        'flaps': flaps,
        'gear': gear,
        'spoiler': spoiler,
        'autobrake': autobrake,
        'reverse_1': rev if e1_run else 0.0,
        'reverse_2': rev if e2_run else 0.0
    }
