#include <iostream>
#include <vector>
#include <algorithm>
#include <chrono>
#include <random>

using namespace std;

// Fast I/O for large graphs
void fast_io() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
}

int main() {
    fast_io();
    int n, m, k;
    if (!(cin >> n >> m >> k)) return 0;

    vector<vector<int>> adj(n);
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    // Track colors for each vertex
    vector<int> colors(n, -1);
    auto start_time = chrono::high_resolution_clock::now();
    mt19937 rng(1337);

    // ---------------------------------------------------------
    // Strategy 1: Greedy Coloring with Largest Degree First (LDF)
    // ---------------------------------------------------------
    vector<int> order(n);
    for (int i = 0; i < n; ++i) order[i] = i;
    
    // Sort vertices by descending degree (LDF)
    sort(order.begin(), order.end(), [&](int a, int b) {
        return adj[a].size() > adj[b].size();
    });

    for (int u : order) {
        vector<bool> used(k, false);
        for (int v : adj[u]) {
            if (colors[v] != -1 && colors[v] < k) {
                used[colors[v]] = true;
            }
        }
        int c;
        for (c = 0; c < k; ++c) {
            if (!used[c]) break;
        }
        if (c < k) {
            colors[u] = c;
        } else {
            // If we run out of colors in our budget, randomly assign one
            // and rely on Local Search to fix it.
            colors[u] = rng() % k;
        }
    }

    // ---------------------------------------------------------
    // Strategy 2: Min-Conflicts Local Search
    // ---------------------------------------------------------
    auto get_conflicts = [&]() {
        int conflicts = 0;
        for (int u = 0; u < n; ++u) {
            for (int v : adj[u]) {
                if (u < v && colors[u] == colors[v]) {
                    conflicts++;
                }
            }
        }
        return conflicts;
    };

    int current_conflicts = get_conflicts();

    // If Greedy couldn't find a perfect coloring within `k` colors, use Min-Conflicts
    if (current_conflicts > 0) {
        int max_iters = 20000000;
        for (int iter = 0; iter < max_iters && current_conflicts > 0; ++iter) {
            // Check time limit occasionally (stay well within 5 mins, break at 290s)
            if (iter % 50000 == 0) {
                auto now = chrono::high_resolution_clock::now();
                auto duration = chrono::duration_cast<chrono::seconds>(now - start_time).count();
                if (duration > 290) break;
            }

            // Find all conflicted vertices
            vector<int> conflicted_vertices;
            for (int u = 0; u < n; ++u) {
                bool conflict = false;
                for (int v : adj[u]) {
                    if (colors[u] == colors[v]) {
                        conflict = true;
                        break;
                    }
                }
                if (conflict) {
                    conflicted_vertices.push_back(u);
                }
            }
            if (conflicted_vertices.empty()) break;

            // Pick a random conflicted vertex
            int u = conflicted_vertices[rng() % conflicted_vertices.size()];
            
            // Find the color that results in the minimum number of conflicts for u
            vector<int> color_conflicts(k, 0);
            for (int v : adj[u]) {
                color_conflicts[colors[v]]++;
            }

            int min_conf = color_conflicts[colors[u]];
            vector<int> best_colors;
            
            for (int c = 0; c < k; ++c) {
                if (color_conflicts[c] < min_conf) {
                    min_conf = color_conflicts[c];
                    best_colors.clear();
                    best_colors.push_back(c);
                } else if (color_conflicts[c] == min_conf) {
                    best_colors.push_back(c);
                }
            }
            
            if (!best_colors.empty()) {
                colors[u] = best_colors[rng() % best_colors.size()];
                current_conflicts = get_conflicts();
            }
        }
    }

    // Output formatting
    for (int i = 0; i < n; ++i) {
        cout << colors[i] << (i == n - 1 ? "" : " ");
    }
    cout << "\n";

    return 0;
}
