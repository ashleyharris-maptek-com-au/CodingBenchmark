using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;

class Program
{
    static int numVertices;
    static int[] degrees;
    static int[][] adj;
    static HashSet<int>[] adjSet;

    static int[] bestClique;
    static int maxCliqueSize = 0;
    static object lockObj = new object();

    static void Main()
    {
        var sw = Stopwatch.StartNew();
        
        using (Stream stream = Console.OpenStandardInput())
        {
            FastScanner scanner = new FastScanner(stream);
            int vCount = scanner.NextInt();
            int eCount = scanner.NextInt();
            
            if (vCount == -1) return;
            
            numVertices = vCount;
            int numEdges = eCount;

            degrees = new int[numVertices];
            List<int>[] tempAdj = new List<int>[numVertices];
            for (int i = 0; i < numVertices; i++) tempAdj[i] = new List<int>();

            for (int i = 0; i < numEdges; i++)
            {
                int u = scanner.NextInt();
                int v = scanner.NextInt();
                if (u == -1 || v == -1) break;
                
                if (u != v)
                {
                    tempAdj[u].Add(v);
                    tempAdj[v].Add(u);
                    degrees[u]++;
                    degrees[v]++;
                }
            }

            adj = new int[numVertices][];
            adjSet = new HashSet<int>[numVertices];
            for (int i = 0; i < numVertices; i++)
            {
                adj[i] = tempAdj[i].ToArray();
                adjSet[i] = new HashSet<int>(adj[i]);
            }
        }

        int[] initialClique = GreedyClique();
        bestClique = initialClique;
        maxCliqueSize = initialClique.Length;

        if (maxCliqueSize == numVertices || sw.ElapsedMilliseconds > 280000)
        {
            PrintResult();
            return;
        }

        int timeLimitMs = 280000; // 4.6 minutes
        int[] sortedVertices = Enumerable.Range(0, numVertices).OrderByDescending(v => degrees[v]).ToArray();

        Parallel.For(0, Environment.ProcessorCount, new ParallelOptions { MaxDegreeOfParallelism = Environment.ProcessorCount }, (threadId, state) =>
        {
            Random rand = new Random(threadId + Environment.TickCount);
            
            while (sw.ElapsedMilliseconds < timeLimitMs)
            {
                List<int> currentClique = new List<int>();
                List<int> candidates = new List<int>();
                
                int topK = Math.Max(1, Math.Min(100, numVertices / 10));
                int startVertex = sortedVertices[rand.Next(topK)];
                currentClique.Add(startVertex);
                candidates.AddRange(adj[startVertex]);

                while (candidates.Count > 0 && sw.ElapsedMilliseconds < timeLimitMs)
                {
                    int pickIdx = rand.Next(candidates.Count);
                    int u = candidates[pickIdx];
                    
                    currentClique.Add(u);
                    
                    List<int> nextCandidates = new List<int>(candidates.Count);
                    foreach (int c in candidates)
                    {
                        if (c != u && adjSet[u].Contains(c))
                        {
                            nextCandidates.Add(c);
                        }
                    }
                    candidates = nextCandidates;
                }

                if (currentClique.Count > maxCliqueSize)
                {
                    lock (lockObj)
                    {
                        if (currentClique.Count > maxCliqueSize)
                        {
                            maxCliqueSize = currentClique.Count;
                            bestClique = currentClique.ToArray();
                        }
                    }
                }
            }
        });

        PrintResult();
    }

    static int[] GreedyClique()
    {
        List<int> clique = new List<int>();
        int[] sortedByDegree = Enumerable.Range(0, numVertices).OrderByDescending(v => degrees[v]).ToArray();
        
        foreach (int v in sortedByDegree)
        {
            bool canAdd = true;
            foreach (int c in clique)
            {
                if (!adjSet[v].Contains(c))
                {
                    canAdd = false;
                    break;
                }
            }
            if (canAdd)
            {
                clique.Add(v);
            }
        }
        return clique.ToArray();
    }

    static void PrintResult()
    {
        Console.WriteLine(maxCliqueSize);
        Console.WriteLine(string.Join(" ", bestClique));
    }

    class FastScanner
    {
        private Stream stream;
        private byte[] buffer = new byte[1024 * 64];
        private int head;
        private int tail;

        public FastScanner(Stream stream)
        {
            this.stream = stream;
        }

        public int NextInt()
        {
            int c = Read();
            while (c <= 32)
            {
                if (c == -1) return -1;
                c = Read();
            }

            int res = 0;
            while (c > 32)
            {
                if (c >= '0' && c <= '9')
                {
                    res = res * 10 + c - '0';
                }
                c = Read();
            }
            return res;
        }

        private int Read()
        {
            if (head >= tail)
            {
                head = 0;
                tail = stream.Read(buffer, 0, buffer.Length);
                if (tail <= 0) return -1;
            }
            return buffer[head++];
        }
    }
}
