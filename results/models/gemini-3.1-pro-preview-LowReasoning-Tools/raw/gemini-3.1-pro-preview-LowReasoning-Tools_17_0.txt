using System;
using System.Collections.Generic;
using System.Linq;

namespace BinPacking
{
    class Program
    {
        struct Space
        {
            public long X, Y, Z, W, H, D;
            public Space(long x, long y, long z, long w, long h, long d)
            {
                X = x; Y = y; Z = z; W = w; H = h; D = d;
            }
        }

        struct Box
        {
            public int Index;
            public long W, H, D;
            public long Volume => W * H * D;
        }

        struct PlacedBox
        {
            public int Index;
            public long X, Y, Z;
        }

        static void Main(string[] args)
        {
            string line1 = Console.ReadLine();
            if (string.IsNullOrWhiteSpace(line1)) return;
            string[] parts = line1.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
            int N = int.Parse(parts[0]);
            long cW = long.Parse(parts[1]);
            long cH = long.Parse(parts[2]);
            long cD = long.Parse(parts[3]);

            List<Box> boxes = new List<Box>();
            for (int i = 0; i < N; i++)
            {
                string line = Console.ReadLine();
                if (string.IsNullOrWhiteSpace(line)) continue;
                string[] p = line.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
                boxes.Add(new Box { Index = i, W = long.Parse(p[0]), H = long.Parse(p[1]), D = long.Parse(p[2]) });
            }

            // Sort boxes by volume ascending to maximize the number of boxes packed
            boxes = boxes.OrderBy(b => b.Volume).ThenBy(b => b.W + b.H + b.D).ToList();

            List<Space> freeSpaces = new List<Space>();
            freeSpaces.Add(new Space(0, 0, 0, cW, cH, cD));

            List<PlacedBox> placed = new List<PlacedBox>();

            foreach (var box in boxes)
            {
                long bestZ = long.MaxValue, bestY = long.MaxValue, bestX = long.MaxValue;
                int bestSpaceIdx = -1;
                long bestW = 0, bestH = 0, bestD = 0;

                long[] dims = new[] { box.W, box.H, box.D };
                var rotations = GetRotations(dims);

                for (int i = 0; i < freeSpaces.Count; i++)
                {
                    var space = freeSpaces[i];
                    foreach (var rot in rotations)
                    {
                        if (rot[0] <= space.W && rot[1] <= space.H && rot[2] <= space.D)
                        {
                            if (space.Z < bestZ ||
                                (space.Z == bestZ && space.Y < bestY) ||
                                (space.Z == bestZ && space.Y == bestY && space.X < bestX))
                            {
                                bestZ = space.Z;
                                bestY = space.Y;
                                bestX = space.X;
                                bestSpaceIdx = i;
                                bestW = rot[0];
                                bestH = rot[1];
                                bestD = rot[2];
                            }
                        }
                    }
                }

                if (bestSpaceIdx != -1)
                {
                    var space = freeSpaces[bestSpaceIdx];
                    freeSpaces.RemoveAt(bestSpaceIdx);

                    placed.Add(new PlacedBox { Index = box.Index, X = space.X, Y = space.Y, Z = space.Z });

                    var newSpaces = SplitSpace(space, bestW, bestH, bestD);
                    foreach (var ns in newSpaces)
                    {
                        if (ns.W > 0 && ns.H > 0 && ns.D > 0)
                        {
                            freeSpaces.Add(ns);
                        }
                    }
                }
            }

            Console.WriteLine(placed.Count);
            foreach (var p in placed)
            {
                Console.WriteLine($"{p.Index} {p.X} {p.Y} {p.Z}");
            }
        }

        static List<long[]> GetRotations(long[] dims)
        {
            var rots = new List<long[]>();
            rots.Add(new[] { dims[0], dims[1], dims[2] });
            rots.Add(new[] { dims[0], dims[2], dims[1] });
            rots.Add(new[] { dims[1], dims[0], dims[2] });
            rots.Add(new[] { dims[1], dims[2], dims[0] });
            rots.Add(new[] { dims[2], dims[0], dims[1] });
            rots.Add(new[] { dims[2], dims[1], dims[0] });

            var unique = new List<long[]>();
            foreach (var r in rots)
            {
                bool exists = false;
                foreach (var u in unique)
                {
                    if (u[0] == r[0] && u[1] == r[1] && u[2] == r[2])
                    {
                        exists = true;
                        break;
                    }
                }
                if (!exists) unique.Add(r);
            }
            return unique;
        }

        static List<Space> SplitSpace(Space space, long w, long h, long d)
        {
            long X = space.X, Y = space.Y, Z = space.Z;
            long W = space.W, H = space.H, D = space.D;

            var splits = new List<List<Space>>();

            // XYZ
            splits.Add(new List<Space> {
                new Space(X + w, Y, Z, W - w, H, D),
                new Space(X, Y + h, Z, w, H - h, D),
                new Space(X, Y, Z + d, w, h, D - d)
            });
            // XZY
            splits.Add(new List<Space> {
                new Space(X + w, Y, Z, W - w, H, D),
                new Space(X, Y, Z + d, w, H, D - d),
                new Space(X, Y + h, Z, w, H - h, d)
            });
            // YXZ
            splits.Add(new List<Space> {
                new Space(X, Y + h, Z, W, H - h, D),
                new Space(X + w, Y, Z, W - w, h, D),
                new Space(X, Y, Z + d, w, h, D - d)
            });
            // YZX
            splits.Add(new List<Space> {
                new Space(X, Y + h, Z, W, H - h, D),
                new Space(X, Y, Z + d, W, h, D - d),
                new Space(X + w, Y, Z, W - w, h, d)
            });
            // ZXY
            splits.Add(new List<Space> {
                new Space(X, Y, Z + d, W, H, D - d),
                new Space(X + w, Y, Z, W - w, H, d),
                new Space(X, Y + h, Z, w, H - h, d)
            });
            // ZYX
            splits.Add(new List<Space> {
                new Space(X, Y, Z + d, W, H, D - d),
                new Space(X, Y + h, Z, W, H - h, d),
                new Space(X + w, Y, Z, W - w, h, d)
            });

            long maxMaxVol = -1;
            List<Space> bestSplit = null;

            foreach (var split in splits)
            {
                long maxVol = 0;
                foreach (var s in split)
                {
                    long vol = s.W * s.H * s.D;
                    if (vol > maxVol) maxVol = vol;
                }
                if (maxVol > maxMaxVol)
                {
                    maxMaxVol = maxVol;
                    bestSplit = split;
                }
            }

            return bestSplit;
        }
    }
}
