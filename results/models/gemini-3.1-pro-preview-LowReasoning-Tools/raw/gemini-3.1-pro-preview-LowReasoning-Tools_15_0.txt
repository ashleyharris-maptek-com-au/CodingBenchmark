```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines().filter_map(|l| l.ok());

    let nv_line = match lines.next() {
        Some(l) => l,
        None => return,
    };
    let nv: usize = nv_line.trim().parse().unwrap_or(0);

    let mut min_x = f64::MAX;
    let mut max_x = f64::MIN;
    let mut min_y = f64::MAX;
    let mut max_y = f64::MIN;

    for _ in 0..nv {
        let line = lines.next().unwrap_or_default();
        let parts: Vec<f64> = line.split_whitespace().filter_map(|s| s.parse().ok()).collect();
        if parts.len() >= 2 {
            min_x = min_x.min(parts[0]);
            max_x = max_x.max(parts[0]);
            min_y = min_y.min(parts[1]);
            max_y = max_y.max(parts[1]);
        }
    }

    let sun_line = lines.next().unwrap_or_default();
    let sun_parts: Vec<f64> = sun_line.split_whitespace().filter_map(|s| s.parse().ok()).collect();
    let mut sx = 0.0;
    let mut sy = 0.0;
    let mut sz = -1.0;
    if sun_parts.len() >= 3 {
        sx = sun_parts[0];
        sy = sun_parts[1];
        sz = sun_parts[2];
    }

    if nv == 0 {
        println!("0");
        return;
    }
    
    // Adaptive logic based on problem scale
    let scale = if nv > 1000 { 10.0 } else { 5.0 };
    
    let mut tets = Vec::new();
    
    // Rotate 45 degrees around Z to turn diamond projection into a square
    let qw = std::f64::consts::PI / 8.0;
    let qw_val = qw.cos();
    let qz_val = qw.sin();
    
    let square_side = scale * std::f64::consts::SQRT_2;
    
    let mut start_x = min_x - square_side;
    let mut end_x = max_x + square_side;
    let mut start_y = min_y - square_side;
    let mut end_y = max_y + square_side;
    
    if start_x > end_x { start_x = end_x; }
    if start_y > end_y { start_y = end_y; }
    
    let nx = ((end_x - start_x) / square_side).ceil() as i32 + 1;
    let ny = ((end_y - start_y) / square_side).ceil() as i32 + 1;
    
    let mut z_offset = 100.0;
    for i in 0..nx {
        for j in 0..ny {
            let cx = start_x + (i as f64) * square_side;
            let cy = start_y + (j as f64) * square_side;
            
            let t = if sz != 0.0 { -z_offset / sz } else { 0.0 };
            let px = cx - t * sx;
            let py = cy - t * sy;
            let pz = z_offset;
            
            tets.push((px, py, pz, qw_val, 0.0, 0.0, qz_val, scale));
            
            z_offset += 20.0; // Stagger Z to avoid 3D intersection
        }
    }
    
    println!("{}", tets.len());
    for tet in tets {
        println!("{:.6} {:.6} {:.6} {:.6} {:.6} {:.6} {:.6} {:.6}", 
            tet.0, tet.1, tet.2, tet.3, tet.4, tet.5, tet.6, tet.7);
    }
}
```
