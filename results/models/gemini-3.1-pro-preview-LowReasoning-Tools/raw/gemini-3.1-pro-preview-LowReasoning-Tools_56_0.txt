const fs = require('fs');

function distance(p1, p2) {
    const dx = p1.x - p2.x;
    const dy = p1.y - p2.y;
    return Math.sqrt(dx * dx + dy * dy);
}

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);
    const depot = { x: data.depot[0], y: data.depot[1] };
    const maxTrucks = data.truckCount;
    const packages = data.packages;

    if (!packages || packages.length === 0) {
        console.log(JSON.stringify({ routes: [] }));
        return;
    }

    packages.forEach(p => {
        p.angle = Math.atan2(p.y - depot.y, p.x - depot.x);
    });

    packages.sort((a, b) => a.angle - b.angle);

    const routes = [];
    const numRoutes = Math.min(maxTrucks, packages.length);
    const chunkSize = Math.ceil(packages.length / numRoutes);

    for (let i = 0; i < numRoutes; i++) {
        const chunk = packages.slice(i * chunkSize, (i + 1) * chunkSize);
        if (chunk.length === 0) break;

        let unvisited = [...chunk];
        let curr = depot;
        const route = [];

        while (unvisited.length > 0) {
            let bestDist = Infinity;
            let bestIdx = -1;
            for (let j = 0; j < unvisited.length; j++) {
                const dist = distance(curr, unvisited[j]);
                if (dist < bestDist) {
                    bestDist = dist;
                    bestIdx = j;
                }
            }
            curr = unvisited[bestIdx];
            route.push(curr);
            unvisited.splice(bestIdx, 1);
        }

        let improved = true;
        let iter = 0;
        while (improved && iter < 100) {
            improved = false;
            iter++;
            for (let j = 0; j < route.length; j++) {
                for (let k = j + 1; k < route.length; k++) {
                    const p1 = j === 0 ? depot : route[j - 1];
                    const p2 = route[j];
                    const p3 = route[k];
                    const p4 = k === route.length - 1 ? depot : route[k + 1];

                    const dOld = distance(p1, p2) + distance(p3, p4);
                    const dNew = distance(p1, p3) + distance(p2, p4);

                    if (dNew < dOld - 1e-6) {
                        let left = j;
                        let right = k;
                        while (left < right) {
                            const temp = route[left];
                            route[left] = route[right];
                            route[right] = temp;
                            left++;
                            right--;
                        }
                        improved = true;
                    }
                }
            }
        }

        routes.push(route.map(p => p.id));
    }

    console.log(JSON.stringify({ routes }));
}

solve();
