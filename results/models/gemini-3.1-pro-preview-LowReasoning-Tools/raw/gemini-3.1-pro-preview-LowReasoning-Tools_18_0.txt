#include <iostream>

// Stoer-Wagner exact global min-cut for small graphs
int stoer_wagner(int n, const vector<pair<int,int>>& edges, vector<int>& out_cut_set) {
    vector<vector<int>> weight(n, vector<int>(n, 0));
    for (auto& e : edges) {
        weight[e.first][e.second]++;
        weight[e.second][e.first]++;
    }
    vector<int> v(n);
    iota(v.begin(), v.end(), 0);
    vector<vector<int>> nodes(n);
    for (int i = 0; i < n; ++i) nodes[i].push_back(i);
    
    int best_w = 2e9;
    vector<int> best_cut;
    
    while (v.size() > 1) {
        vector<int> w(v.size(), 0);
        vector<bool> added(v.size(), false);
        int prev = -1, curr = -1;
        
        for (size_t i = 0; i < v.size(); ++i) {
            int next = -1;
            for (size_t j = 0; j < v.size(); ++j) {
                if (!added[j] && (next == -1 || w[j] > w[next])) {
                    next = j;
                }
            }
            added[next] = true;
            if (i == v.size() - 1) {
                if (w[next] < best_w) {
                    best_w = w[next];
                    best_cut = nodes[v[next]];
                }
                for (size_t j = 0; j < v.size(); ++j) {
                    if (j != prev && j != next) {
                        weight[v[prev]][v[j]] += weight[v[next]][v[j]];
                        weight[v[j]][v[prev]] += weight[v[next]][v[j]];
                    }
                }
                nodes[v[prev]].insert(nodes[v[prev]].end(), nodes[v[next]].begin(), nodes[v[next]].end());
                v.erase(v.begin() + next);
            } else {
                for (size_t j = 0; j < v.size(); ++j) {
                    if (!added[j]) w[j] += weight[v[next]][v[j]];
                }
                prev = curr;
                curr = next;
            }
        }
    }
    out_cut_set = best_cut;
    return best_w;
}

int main() {
    // Fast I/O
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    uint64_t n, m;
    if (!(cin >> n >> m)) return 0;

    // Adaptive logic: use exact O(V^3) Stoer-Wagner for small graphs
    // Otherwise use minimum degree vertex heuristic for huge scalability
    if (n <= 500 && m <= 100000) {
        vector<pair<int,int>> edges;
        edges.reserve(m);
        for (uint64_t i = 0; i < m; ++i) {
            int u, v;
            cin >> u >> v;
            edges.push_back({u, v});
        }
        vector<int> out_cut_set;
        int min_cut = stoer_wagner(n, edges, out_cut_set);
        vector<bool> in_cut_set(n, false);
        for (int node : out_cut_set) in_cut_set[node] = true;
        
        vector<pair<int,int>> result_edges;
        for (auto& e : edges) {
            if (in_cut_set[e.first] != in_cut_set[e.second]) {
                result_edges.push_back(e);
            }
        }
        cout << result_edges.size() << "\n";
        for (auto& e : result_edges) {
            cout << e.first << " " << e.second << "\n";
        }
    } else {
        // High scalability heuristic: isolate the lowest degree node
        // Fits comfortably in 127GB RAM up to 500 million edges
        const uint64_t MEMORY_LIMIT = 500000000ULL; 
        
        if (m <= MEMORY_LIMIT) {
            vector<pair<int,int>> edges;
            edges.reserve(m);
            vector<int> degree(n, 0);
            for (uint64_t i = 0; i < m; ++i) {
                int u, v;
                cin >> u >> v;
                edges.push_back({u, v});
                degree[u]++;
                degree[v]++;
            }
            int min_deg_node = 0;
            int min_deg = degree[0];
            for (uint64_t i = 1; i < n; ++i) {
                if (degree[i] < min_deg) {
                    min_deg = degree[i];
                    min_deg_node = i;
                }
            }
            
            vector<pair<int,int>> result_edges;
            for (auto& e : edges) {
                if (e.first == min_deg_node || e.second == min_deg_node) {
                    result_edges.push_back(e);
                }
            }
            cout << result_edges.size() << "\n";
            for (auto& e : result_edges) {
                cout << e.first << " " << e.second << "\n";
            }
        } else {
            // Out-of-core fallback: flush edges to disk 
            FILE* tmp = fopen("edges.tmp", "wb+");
            vector<int> degree(n, 0);
            for (uint64_t i = 0; i < m; ++i) {
                int u, v;
                cin >> u >> v;
                int edge[2] = {u, v};
                if (tmp) fwrite(edge, sizeof(int), 2, tmp);
                degree[u]++;
                degree[v]++;
            }
            int min_deg_node = 0;
            int min_deg = degree[0];
            for (uint64_t i = 1; i < n; ++i) {
                if (degree[i] < min_deg) {
                    min_deg = degree[i];
                    min_deg_node = i;
                }
            }
            
            if (tmp) {
                fseek(tmp, 0, SEEK_SET);
                vector<pair<int,int>> result_edges;
                for (uint64_t i = 0; i < m; ++i) {
                    int edge[2];
                    if (fread(edge, sizeof(int), 2, tmp) == 2) {
                        if (edge[0] == min_deg_node || edge[1] == min_deg_node) {
                            result_edges.push_back({edge[0], edge[1]});
                        }
                    }
                }
                cout << result_edges.size() << "\n";
                for (auto& e : result_edges) {
                    cout << e.first << " " << e.second << "\n";
                }
                fclose(tmp);
                remove("edges.tmp");
            }
        }
    }
    return 0;
}
