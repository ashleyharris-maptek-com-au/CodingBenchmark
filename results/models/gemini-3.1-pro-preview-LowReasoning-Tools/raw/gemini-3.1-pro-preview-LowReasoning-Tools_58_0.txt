const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    if (!input.trim()) return;
    const data = JSON.parse(input);
    const grid = data.grid;
    const robots = data.robots;
    const jobs = data.jobs;

    const H = grid.length;
    const W = grid[0].length;

    let maxRobotId = -1;
    for (const r of robots) {
        if (r.id > maxRobotId) maxRobotId = r.id;
    }
    const assignments = [];
    for (let i = 0; i <= maxRobotId; i++) {
        assignments.push([]);
    }

    const points = [];
    for (const r of robots) points.push({x: r.x, y: r.y});
    for (const j of jobs) points.push({x: j.x, y: j.y});

    const pointIdx = new Map();
    const uniquePoints = [];
    for (const p of points) {
        const key = p.y * W + p.x;
        if (!pointIdx.has(key)) {
            pointIdx.set(key, uniquePoints.length);
            uniquePoints.push(p);
        }
    }

    const numPoints = uniquePoints.length;
    const dists = Array(numPoints);
    const dx = [0, 1, 0, -1];
    const dy = [1, 0, -1, 0];

    const q = new Int32Array(H * W * 2);
    const d = new Int32Array(H * W);

    for (let i = 0; i < numPoints; i++) {
        const start = uniquePoints[i];
        let head = 0, tail = 0;
        q[tail++] = start.x;
        q[tail++] = start.y;
        
        d.fill(1e9);
        d[start.y * W + start.x] = 0;
        
        while (head < tail) {
            const cx = q[head++];
            const cy = q[head++];
            const curD = d[cy * W + cx];
            
            for (let dir = 0; dir < 4; dir++) {
                const nx = cx + dx[dir];
                const ny = cy + dy[dir];
                if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
                    if (grid[ny].charCodeAt(nx) !== 35) { // 35 is '#'
                        const nIdx = ny * W + nx;
                        if (d[nIdx] > curD + 1) {
                            d[nIdx] = curD + 1;
                            q[tail++] = nx;
                            q[tail++] = ny;
                        }
                    }
                }
            }
        }
        
        dists[i] = new Int32Array(numPoints);
        for (let j = 0; j < numPoints; j++) {
            const p = uniquePoints[j];
            dists[i][j] = d[p.y * W + p.x];
        }
    }

    function getDist(x1, y1, x2, y2) {
        const k1 = pointIdx.get(y1 * W + x1);
        const k2 = pointIdx.get(y2 * W + x2);
        return dists[k1][k2];
    }

    const unassigned = new Set(jobs);
    const robotPos = new Map();
    for (const r of robots) {
        robotPos.set(r.id, { x: r.x, y: r.y });
    }

    while (unassigned.size > 0) {
        let bestPair = null;
        for (const job of unassigned) {
            for (const r of robots) {
                const pos = robotPos.get(r.id);
                const dist = getDist(pos.x, pos.y, job.x, job.y);
                if (dist > 1e8) continue; 
                
                if (!bestPair) {
                    bestPair = { dist, p: job.priority, rId: r.id, jId: job.id, job };
                } else {
                    if (dist < bestPair.dist) {
                        bestPair = { dist, p: job.priority, rId: r.id, jId: job.id, job };
                    } else if (dist === bestPair.dist) {
                        if (job.priority > bestPair.p) {
                            bestPair = { dist, p: job.priority, rId: r.id, jId: job.id, job };
                        } else if (job.priority === bestPair.p) {
                            if (r.id < bestPair.rId) {
                                bestPair = { dist, p: job.priority, rId: r.id, jId: job.id, job };
                            } else if (r.id === bestPair.rId) {
                                if (job.id < bestPair.jId) {
                                    bestPair = { dist, p: job.priority, rId: r.id, jId: job.id, job };
                                }
                            }
                        }
                    }
                }
            }
        }

        if (!bestPair) break; 
        
        assignments[bestPair.rId].push(bestPair.jId);
        robotPos.set(bestPair.rId, { x: bestPair.job.x, y: bestPair.job.y });
        unassigned.delete(bestPair.job);
    }
    
    console.log(JSON.stringify({ assignments }));
}

solve();
