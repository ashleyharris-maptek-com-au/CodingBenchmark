const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { width, height, grid, ignitions, towns, cutBudget } = data;

const gridArr = grid.map(row => row.split(''));

towns.sort((a, b) => b.value - a.value);

let remainingBudget = cutBudget;
let firebreaks = [];

const isIgnition = (x, y) => {
    return ignitions.some(ig => ig[0] === x && ig[1] === y);
};

for (const town of towns) {
    if (remainingBudget === 0) break;
    
    const neighbors = [
        [town.x, town.y - 1],
        [town.x, town.y + 1],
        [town.x - 1, town.y],
        [town.x + 1, town.y]
    ];
    
    let neededCuts = [];
    let possibleToProtect = true;
    
    for (const [nx, ny] of neighbors) {
        if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
            if (gridArr[ny][nx] === '.') {
                if (isIgnition(nx, ny)) {
                    possibleToProtect = false;
                    break;
                }
                neededCuts.push([nx, ny]);
            }
        }
    }
    
    if (possibleToProtect && neededCuts.length <= remainingBudget) {
        for (const [nx, ny] of neededCuts) {
            firebreaks.push([nx, ny]);
            gridArr[ny][nx] = '~'; // mark as cut
        }
        remainingBudget -= neededCuts.length;
    }
}

console.log(JSON.stringify({ firebreaks }));
