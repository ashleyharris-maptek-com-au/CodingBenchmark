use std::io::{self, BufRead, Read, Write, BufReader, BufWriter};
use std::str::FromStr;

struct Centroid {
    x: f32,
    y: f32,
    z: f32,
    count: f32,
}

fn main() {
    let stdin = io::stdin();
    let mut stdin_lock = stdin.lock();
    
    // Read first line
    let mut header = String::new();
    stdin_lock.read_line(&mut header).expect("Failed to read header");
    let parts: Vec<&str> = header.trim().split_whitespace().collect();
    if parts.len() < 2 {
        return;
    }
    let num_points: usize = parts[0].parse().expect("Invalid num_points");
    let num_clusters: usize = parts[1].parse().expect("Invalid num_clusters");
    
    let mut out = BufWriter::new(io::stdout().lock());
    
    if num_points == 0 || num_clusters == 0 {
        return;
    }

    let mut centroids: Vec<Centroid> = Vec::with_capacity(num_clusters);
    
    let mut point_bytes = [0u8; 12];
    
    // Initialize centroids with the first `num_clusters` points
    for _ in 0..num_clusters {
        if let Ok(12) = stdin_lock.read_exact(&mut point_bytes).map(|_| 12) {
            let x = f32::from_le_bytes(point_bytes[0..4].try_into().unwrap());
            let y = f32::from_le_bytes(point_bytes[4..8].try_into().unwrap());
            let z = f32::from_le_bytes(point_bytes[8..12].try_into().unwrap());
            centroids.push(Centroid { x, y, z, count: 1.0 });
            
            // Output index for the initialized point
            let index = (centroids.len() - 1) as u32;
            out.write_all(&index.to_le_bytes()).unwrap();
        } else {
            break;
        }
    }
    
    // If we haven't reached num_points, process the rest
    let remaining = num_points.saturating_sub(centroids.len());
    if remaining == 0 {
        return;
    }
    
    // Process remaining points in chunks
    let chunk_size = 8192;
    let mut buffer = vec![0u8; chunk_size * 12];
    
    let mut points_read = 0;
    while points_read < remaining {
        let to_read = std::cmp::min(chunk_size, remaining - points_read);
        let bytes_to_read = to_read * 12;
        
        let mut bytes_read = 0;
        while bytes_read < bytes_to_read {
            match stdin_lock.read(&mut buffer[bytes_read..bytes_to_read]) {
                Ok(0) => break,
                Ok(n) => bytes_read += n,
                Err(e) if e.kind() == io::ErrorKind::Interrupted => continue,
                Err(_) => break,
            }
        }
        
        if bytes_read == 0 {
            break;
        }
        
        let actual_points = bytes_read / 12;
        
        for i in 0..actual_points {
            let offset = i * 12;
            let px = f32::from_le_bytes(buffer[offset..offset+4].try_into().unwrap());
            let py = f32::from_le_bytes(buffer[offset+4..offset+8].try_into().unwrap());
            let pz = f32::from_le_bytes(buffer[offset+8..offset+12].try_into().unwrap());
            
            // Find closest centroid
            let mut min_dist = f32::MAX;
            let mut best_idx = 0;
            
            for (idx, c) in centroids.iter().enumerate() {
                let dx = px - c.x;
                let dy = py - c.y;
                let dz = pz - c.z;
                let dist_sq = dx*dx + dy*dy + dz*dz;
                if dist_sq < min_dist {
                    min_dist = dist_sq;
                    best_idx = idx;
                }
            }
            
            // Output the assignment
            out.write_all(&(best_idx as u32).to_le_bytes()).unwrap();
            
            // Update centroid
            let c = &mut centroids[best_idx];
            c.count += 1.0;
            let lr = 1.0 / c.count;
            c.x += lr * (px - c.x);
            c.y += lr * (py - c.y);
            c.z += lr * (pz - c.z);
        }
        
        points_read += actual_points;
    }
    
    out.flush().unwrap();
}
