declare const require: any;
const fs = require('fs');

function solve() {
    const input: string = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);

    const tasks: any[] = data.tasks;

    const outEdges = new Map<string | number, (string | number)[]>();
    const taskMap = new Map<string | number, any>();
    const inDegree = new Map<string | number, number>();

    for (const t of tasks) {
        taskMap.set(t.id, t);
        inDegree.set(t.id, t.deps ? t.deps.length : 0);
        if (!outEdges.has(t.id)) outEdges.set(t.id, []);
    }

    for (const t of tasks) {
        if (t.deps) {
            for (const dep of t.deps) {
                if (!outEdges.has(dep)) outEdges.set(dep, []);
                outEdges.get(dep)!.push(t.id);
            }
        }
    }

    const reverseInDegree = new Map<string | number, number>();
    for (const t of tasks) {
        reverseInDegree.set(t.id, outEdges.get(t.id)!.length);
    }

    const revAvailable: (string | number)[] = [];
    for (const t of tasks) {
        if (reverseInDegree.get(t.id) === 0) {
            revAvailable.push(t.id);
        }
    }

    const revOrder: (string | number)[] = [];
    let head = 0;
    while (head < revAvailable.length) {
        const curr = revAvailable[head++];
        revOrder.push(curr);
        const t = taskMap.get(curr)!;
        if (t.deps) {
            for (const dep of t.deps) {
                const d = reverseInDegree.get(dep)! - 1;
                reverseInDegree.set(dep, d);
                if (d === 0) {
                    revAvailable.push(dep);
                }
            }
        }
    }

    const memo = new Map<string | number, number>();
    for (const id of revOrder) {
        let maxChild = 0;
        const children = outEdges.get(id) || [];
        for (const child of children) {
            maxChild = Math.max(maxChild, memo.get(child) || 0);
        }
        memo.set(id, taskMap.get(id)!.duration + maxChild);
    }

    const available: any[] = [];
    for (const t of tasks) {
        if (inDegree.get(t.id) === 0) {
            available.push(t);
        }
    }

    const order: (string | number)[] = [];

    while (available.length > 0) {
        let bestIdx = 0;
        let bestScore = -Infinity;

        for (let i = 0; i < available.length; i++) {
            const t = available[i];
            const bl = memo.get(t.id)!;
            
            let score = bl;
            score += (t.weight * 1000) / (t.deadline > 0 ? t.deadline : 1);
            
            if (score > bestScore) {
                bestScore = score;
                bestIdx = i;
            }
        }

        const nextTask = available[bestIdx];
        available[bestIdx] = available[available.length - 1];
        available.pop();
        order.push(nextTask.id);

        const children = outEdges.get(nextTask.id) || [];
        for (const child of children) {
            const deg = inDegree.get(child)! - 1;
            inDegree.set(child, deg);
            if (deg === 0) {
                available.push(taskMap.get(child)!);
            }
        }
    }

    console.log(JSON.stringify({ order }));
}

solve();
