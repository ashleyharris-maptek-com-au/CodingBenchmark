using System;
using System.IO;
using System.Diagnostics;

class Program
{
    static void Main()
    {
        using (Stream stream = Console.OpenStandardInput())
        {
            Scanner scanner = new Scanner(stream);
            int N = scanner.NextInt();
            if (N <= 0) return;

            int[] flow = new int[N * N];
            for (int i = 0; i < N * N; i++) flow[i] = scanner.NextInt();

            int[] dist = new int[N * N];
            for (int i = 0; i < N * N; i++) dist[i] = scanner.NextInt();

            int[] p = new int[N];
            for (int i = 0; i < N; i++) p[i] = i;

            if (N == 1)
            {
                long c = CalculateCost(p, flow, dist, N);
                Console.WriteLine(c);
                Console.WriteLine("0");
                return;
            }

            Stopwatch sw = Stopwatch.StartNew();
            Random rand = new Random(42);

            // Initial random shuffle
            for (int i = 0; i < N; i++)
            {
                int j = rand.Next(i, N);
                int temp = p[i]; p[i] = p[j]; p[j] = temp;
            }

            long currentCost = CalculateCost(p, flow, dist, N);
            long bestCost = currentCost;
            int[] bestP = (int[])p.Clone();

            double T = 1000000.0;
            double alpha = 0.9999;
            long iterations = 0;

            // Run for ~280 seconds (4.6 minutes) to stay well within 5 minutes
            while (sw.ElapsedMilliseconds < 280000)
            {
                int u = rand.Next(N);
                int v = rand.Next(N);
                if (u == v) continue;

                long delta = CalculateDelta(p, flow, dist, N, u, v);
                bool accept = false;

                if (delta < 0)
                {
                    accept = true;
                }
                else
                {
                    if (delta < T * 10) // Optimization to avoid Math.Exp for large negatives
                    {
                        if (rand.NextDouble() < Math.Exp(-delta / T))
                        {
                            accept = true;
                        }
                    }
                }

                if (accept)
                {
                    currentCost += delta;
                    int temp = p[u]; p[u] = p[v]; p[v] = temp;

                    if (currentCost < bestCost)
                    {
                        bestCost = currentCost;
                        Array.Copy(p, bestP, N);
                    }
                }

                iterations++;
                if (iterations % 10000 == 0)
                {
                    T *= alpha;
                    if (T < 1e-3) T = 1000000.0; // Reheat
                }
            }

            Console.WriteLine(bestCost);
            Console.WriteLine(string.Join(" ", bestP));
        }
    }

    static long CalculateCost(int[] p, int[] flow, int[] dist, int N)
    {
        long cost = 0;
        for (int i = 0; i < N; i++)
        {
            int rowU = i * N;
            int rowPU = p[i] * N;
            for (int j = 0; j < N; j++)
            {
                cost += (long)flow[rowU + j] * dist[rowPU + p[j]];
            }
        }
        return cost;
    }

    static long CalculateDelta(int[] p, int[] flow, int[] dist, int N, int u, int v)
    {
        long delta = 0;
        int pu = p[u];
        int pv = p[v];

        int rowU = u * N;
        int rowV = v * N;
        int rowPU = pu * N;
        int rowPV = pv * N;

        for (int k = 0; k < N; k++)
        {
            if (k == u || k == v) continue;
            
            int pk = p[k];
            int rowK = k * N;
            int rowPK = pk * N;

            delta += (long)flow[rowU + k] * (dist[rowPV + pk] - dist[rowPU + pk])
                   + (long)flow[rowK + u] * (dist[rowPK + pv] - dist[rowPK + pu])
                   + (long)flow[rowV + k] * (dist[rowPU + pk] - dist[rowPV + pk])
                   + (long)flow[rowK + v] * (dist[rowPK + pu] - dist[rowPK + pv]);
        }

        delta += (long)flow[rowU + v] * (dist[rowPV + pu] - dist[rowPU + pv])
               + (long)flow[rowV + u] * (dist[rowPU + pv] - dist[rowPV + pu]);

        return delta;
    }

    class Scanner
    {
        Stream stream;
        byte[] buffer = new byte[1024 * 1024];
        int head;
        int tail;

        public Scanner(Stream s)
        {
            stream = s;
        }

        public int NextInt()
        {
            int c = Read();
            while (c != '-' && (c < '0' || c > '9'))
            {
                if (c == -1) return -1;
                c = Read();
            }
            if (c == -1) return -1;
            int sign = 1;
            if (c == '-')
            {
                sign = -1;
                c = Read();
            }
            int res = 0;
            while (c >= '0' && c <= '9')
            {
                res = res * 10 + c - '0';
                c = Read();
            }
            return res * sign;
        }

        int Read()
        {
            if (head >= tail)
            {
                head = 0;
                tail = stream.Read(buffer, 0, buffer.Length);
                if (tail <= 0) return -1;
            }
            return buffer[head++];
        }
    }
}
