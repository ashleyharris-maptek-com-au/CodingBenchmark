#version 450

layout(triangles_adjacency) in;
layout(triangle_strip, max_vertices = 12) out;

layout(location = 0) in vec3 worldPos[];
layout(location = 1) in vec3 normal[];
layout(location = 2) in vec3 color[];

layout(location = 0) out vec3 outColor;

void emitEdge(int i, int j) {
    vec4 p0 = gl_in[i].gl_Position;
    vec4 p1 = gl_in[j].gl_Position;
    
    vec3 avg_color = (color[i] + color[j]) * 0.5;

    vec2 ndc0 = p0.xy / p0.w;
    vec2 ndc1 = p1.xy / p1.w;
    vec2 dir = normalize(ndc1 - ndc0);
    vec2 perp = vec2(-dir.y, dir.x) * 0.004;

    vec4 offset0 = vec4(perp * p0.w, 0.0, 0.0);
    vec4 offset1 = vec4(perp * p1.w, 0.0, 0.0);

    outColor = avg_color;
    gl_Position = p0 + offset0;
    EmitVertex();

    outColor = avg_color;
    gl_Position = p0 - offset0;
    EmitVertex();

    outColor = avg_color;
    gl_Position = p1 + offset1;
    EmitVertex();

    outColor = avg_color;
    gl_Position = p1 - offset1;
    EmitVertex();

    EndPrimitive();
}

void main() {
    emitEdge(0, 2);
    emitEdge(2, 4);
    emitEdge(4, 0);
}
