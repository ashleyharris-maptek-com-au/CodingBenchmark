#include <iostream>
#include <vector>
#include <queue>
#include <unordered_set>
#include <unordered_map>
#include <algorithm>

using namespace std;

struct Point {
    vector<int> coords;
    bool operator==(const Point& o) const {
        return coords == o.coords;
    }
    bool operator<(const Point& o) const {
        return coords < o.coords;
    }
};

struct PointHash {
    size_t operator()(const Point& p) const {
        size_t h = 0;
        for (int c : p.coords) {
            h ^= hash<int>()(c) + 0x9e3779b9 + (h << 6) + (h >> 2);
        }
        return h;
    }
};

int D, max_turns;
vector<int> bounds;
int initial_snake_length;
Point start_pos;
int F;
vector<Point> foods;
int O;
unordered_set<Point, PointHash> obstacles;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    if (!(cin >> D >> max_turns)) return 0;
    bounds.resize(D);
    for (int i = 0; i < D; ++i) cin >> bounds[i];
    cin >> initial_snake_length;
    start_pos.coords.resize(D);
    for (int i = 0; i < D; ++i) cin >> start_pos.coords[i];
    
    cin >> F;
    foods.resize(F);
    unordered_set<Point, PointHash> remaining_foods;
    for (int i = 0; i < F; ++i) {
        foods[i].coords.resize(D);
        for (int j = 0; j < D; ++j) cin >> foods[i].coords[j];
        remaining_foods.insert(foods[i]);
    }
    
    cin >> O;
    for (int i = 0; i < O; ++i) {
        Point p; p.coords.resize(D);
        for (int j = 0; j < D; ++j) cin >> p.coords[j];
        obstacles.insert(p);
    }
    
    vector<Point> snake_body;
    snake_body.push_back(start_pos);
    int current_length = initial_snake_length;
    int turn = 0;

    while (!remaining_foods.empty() && turn < max_turns) {
        unordered_map<Point, pair<int, int>, PointHash> parent_move;
        unordered_map<Point, Point, PointHash> parent_point;
        queue<Point> q;
        unordered_set<Point, PointHash> visited;
        
        unordered_set<Point, PointHash> current_snake_set;
        for (const auto& p : snake_body) current_snake_set.insert(p);
        
        q.push(snake_body.back()); 
        visited.insert(snake_body.back());
        
        Point found_food;
        bool found = false;
        
        while (!q.empty()) {
            Point curr = q.front(); q.pop();
            
            if (remaining_foods.count(curr)) {
                found = true;
                found_food = curr;
                break;
            }
            
            for (int i = 0; i < D; ++i) {
                for (int dir : {-1, 1}) {
                    Point next_p = curr;
                    next_p.coords[i] += dir;
                    
                    if (next_p.coords[i] < 0 || next_p.coords[i] >= bounds[i]) continue;
                    if (obstacles.count(next_p)) continue;
                    
                    if (current_snake_set.count(next_p) && !(next_p == snake_body.front())) {
                        continue;
                    }
                    
                    if (!visited.count(next_p)) {
                        visited.insert(next_p);
                        parent_move[next_p] = {i, dir};
                        parent_point[next_p] = curr;
                        q.push(next_p);
                    }
                }
            }
        }
        
        if (found) {
            vector<pair<int, int>> path;
            Point curr = found_food;
            while (!(curr == snake_body.back())) {
                path.push_back(parent_move[curr]);
                curr = parent_point[curr];
            }
            reverse(path.begin(), path.end());
            
            for (auto mv : path) {
                cout << mv.first << " " << mv.second << "\n";
                
                Point next_head = snake_body.back();
                next_head.coords[mv.first] += mv.second;
                snake_body.push_back(next_head);
                
                if (remaining_foods.count(next_head)) {
                    remaining_foods.erase(next_head);
                    current_length++;
                }
                while (snake_body.size() > current_length) {
                    snake_body.erase(snake_body.begin());
                }
                turn++;
                if (turn >= max_turns || remaining_foods.empty()) break;
            }
        } else {
            bool moved = false;
            for (int i = 0; i < D; ++i) {
                for (int dir : {-1, 1}) {
                    Point next_p = snake_body.back();
                    next_p.coords[i] += dir;
                    
                    if (next_p.coords[i] < 0 || next_p.coords[i] >= bounds[i]) continue;
                    if (obstacles.count(next_p)) continue;
                    
                    unordered_set<Point, PointHash> current_snake_set_local;
                    for (const auto& p : snake_body) current_snake_set_local.insert(p);
                    if (current_snake_set_local.count(next_p)) continue;
                    
                    cout << i << " " << dir << "\n";
                    
                    snake_body.push_back(next_p);
                    while (snake_body.size() > current_length) {
                        snake_body.erase(snake_body.begin());
                    }
                    moved = true;
                    turn++;
                    break;
                }
                if (moved) break;
            }
            if (!moved) break;
        }
    }
    
    return 0;
}
