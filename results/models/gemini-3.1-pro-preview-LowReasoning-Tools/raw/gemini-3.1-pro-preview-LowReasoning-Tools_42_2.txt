import math

angles = [i * math.pi / 3 for i in range(7)]
r = 0.04

offsets = [(r * math.cos(a), r * math.sin(a)) for a in angles]

code = """
; SPIR-V
; Version: 1.0
; Generator: Khronos SPIR-V Tools Assembler; 0
; Bound: 1000
; Schema: 0

OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main InputPoints
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 18
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2
OpDecorate %outColor Location 0

%void = OpTypeVoid
%void_fn = OpTypeFunction %void
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1

%uint_1 = OpConstant %uint 1
%int_0 = OpConstant %int 0

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_1
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input
%ptr_in_v4 = OpTypePointer Input %v4float

%arr_v3_in = OpTypeArray %v3float %uint_1
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%worldPos = OpVariable %ptr_in_arr_v3 Input
%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input
%ptr_in_v3 = OpTypePointer Input %v3float

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v4 = OpTypePointer Output %v4float

%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%float_0_5 = OpConstant %float 0.5
%float_0 = OpConstant %float 0.0

%main = OpFunction %void None %void_fn
%label = OpLabel

; Read center pos
%in_pos_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%center_pos = OpLoad %v4float %in_pos_ptr

; Read center color
%in_color_ptr = OpAccessChain %ptr_in_v3 %color %int_0
%center_color = OpLoad %v3float %in_color_ptr

; Compute edge color (center_color * 0.5)
%edge_color = OpVectorTimesScalar %v3float %center_color %float_0_5

; Pointers to output
%out_pos_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
"""

id_counter = 100
def get_id():
    global id_counter
    id_counter += 1
    return f"%val_{id_counter}"

for i in range(6):
    off1 = offsets[i]
    off2 = offsets[i+1]
    
    code += f"\n; --- Triangle {i} ---\n"
    # Vertex 1: Center
    code += f"OpStore %out_pos_ptr %center_pos\n"
    code += f"OpStore %outColor %center_color\n"
    code += f"OpEmitVertex\n"
    
    # Vertex 2: Edge i
    const_x1 = get_id()
    const_y1 = get_id()
    offset_vec1 = get_id()
    pos1 = get_id()
    code += f"{const_x1} = OpConstant %float {off1[0]:.6f}\n"
    code += f"{const_y1} = OpConstant %float {off1[1]:.6f}\n"
    code += f"{offset_vec1} = OpConstantComposite %v4float {const_x1} {const_y1} %float_0 %float_0\n"
    code += f"{pos1} = OpFAdd %v4float %center_pos {offset_vec1}\n"
    code += f"OpStore %out_pos_ptr {pos1}\n"
    code += f"OpStore %outColor %edge_color\n"
    code += f"OpEmitVertex\n"
    
    # Vertex 3: Edge i+1
    const_x2 = get_id()
    const_y2 = get_id()
    offset_vec2 = get_id()
    pos2 = get_id()
    code += f"{const_x2} = OpConstant %float {off2[0]:.6f}\n"
    code += f"{const_y2} = OpConstant %float {off2[1]:.6f}\n"
    code += f"{offset_vec2} = OpConstantComposite %v4float {const_x2} {const_y2} %float_0 %float_0\n"
    code += f"{pos2} = OpFAdd %v4float %center_pos {offset_vec2}\n"
    code += f"OpStore %out_pos_ptr {pos2}\n"
    code += f"OpStore %outColor %edge_color\n"
    code += f"OpEmitVertex\n"
    
    code += f"OpEndPrimitive\n"

code += """
OpReturn
OpFunctionEnd
"""

with open('answer.txt', 'w') as f:
    f.write(code)

print("answer.txt generated!")
