#version 450
layout(triangles_adjacency) in;
layout(triangle_strip, max_vertices = 15) out;

layout(location = 0) in vec3 inWorldPos[6];
layout(location = 1) in vec3 inNormal[6];
layout(location = 2) in vec3 inColor[6];

layout(location = 0) out vec3 outColor;

layout(set=0, binding = 0) uniform UBO {
    mat4 model;
    mat4 view;
    mat4 proj;
    vec3 lightPos;
    vec3 cameraPos;
    vec4 params;
} ubo;

void emitThickLine(int v0, int v1) {
    vec4 p0 = gl_in[v0].gl_Position;
    vec4 p1 = gl_in[v1].gl_Position;

    vec2 dir = normalize((p1.xy / p1.w) - (p0.xy / p0.w));
    vec2 normal = vec2(-dir.y, dir.x);
    vec2 offset = normal * 0.015;

    gl_Position = p0 + vec4(offset * p0.w, 0.0, 0.0);
    outColor = vec3(1.0);
    EmitVertex();

    gl_Position = p0 - vec4(offset * p0.w, 0.0, 0.0);
    outColor = vec3(1.0);
    EmitVertex();

    gl_Position = p1 + vec4(offset * p1.w, 0.0, 0.0);
    outColor = vec3(1.0);
    EmitVertex();

    gl_Position = p1 - vec4(offset * p1.w, 0.0, 0.0);
    outColor = vec3(1.0);
    EmitVertex();

    EndPrimitive();
}

void main() {
    emitThickLine(0, 2);
    emitThickLine(2, 4);
    emitThickLine(4, 0);

    gl_Position = gl_in[0].gl_Position;
    outColor = inColor[0] * 0.3;
    EmitVertex();

    gl_Position = gl_in[2].gl_Position;
    outColor = inColor[2] * 0.3;
    EmitVertex();

    gl_Position = gl_in[4].gl_Position;
    outColor = inColor[4] * 0.3;
    EmitVertex();

    EndPrimitive();
}
