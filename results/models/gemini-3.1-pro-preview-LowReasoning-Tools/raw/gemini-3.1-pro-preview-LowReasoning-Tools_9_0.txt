using System;
using System.IO;

class Program
{
    struct Cut : IComparable<Cut>
    {
        public int len;
        public int id;
        public int CompareTo(Cut other)
        {
            return other.len.CompareTo(this.len); // descending
        }
    }

    static int[] tree;
    static int stock_length;

    static void Build(int node, int start, int end)
    {
        if (start == end)
        {
            tree[node] = stock_length;
            return;
        }
        int mid = (start + end) / 2;
        Build(2 * node, start, mid);
        Build(2 * node + 1, mid + 1, end);
        tree[node] = stock_length;
    }

    static int QueryAndUpdate(int node, int start, int end, int cut)
    {
        if (start == end)
        {
            tree[node] -= cut;
            return start;
        }
        int mid = (start + end) / 2;
        int res;
        if (tree[2 * node] >= cut)
        {
            res = QueryAndUpdate(2 * node, start, mid, cut);
        }
        else
        {
            res = QueryAndUpdate(2 * node + 1, mid + 1, end, cut);
        }
        tree[node] = Math.Max(tree[2 * node], tree[2 * node + 1]);
        return res;
    }

    static void Main()
    {
        using Stream stdin = Console.OpenStandardInput();
        byte[] buffer = new byte[1024 * 1024];
        int head = 0, tail = 0;

        int Read()
        {
            if (head >= tail)
            {
                head = 0;
                tail = stdin.Read(buffer, 0, buffer.Length);
                if (tail == 0) return -1;
            }
            return buffer[head++];
        }

        int NextInt()
        {
            int c = Read();
            while (c <= 32)
            {
                if (c == -1) return -1;
                c = Read();
            }
            int res = 0;
            while (c > 32)
            {
                res = res * 10 + c - '0';
                c = Read();
            }
            return res;
        }

        int n = NextInt();
        if (n <= 0) return;
        stock_length = NextInt();

        Cut[] cuts = new Cut[n];
        for (int i = 0; i < n; i++)
        {
            cuts[i] = new Cut { len = NextInt(), id = i };
        }

        Array.Sort(cuts);

        tree = new int[4 * n];
        Build(1, 0, n - 1);

        int[] bin_head = new int[n];
        int[] cut_next = new int[n];
        for (int i = 0; i < n; i++) bin_head[i] = -1;
        int max_bin = -1;

        for (int i = 0; i < n; i++)
        {
            int bin_idx = QueryAndUpdate(1, 0, n - 1, cuts[i].len);
            if (bin_idx > max_bin) max_bin = bin_idx;
            
            cut_next[cuts[i].id] = bin_head[bin_idx];
            bin_head[bin_idx] = cuts[i].id;
        }

        int num_stocks = max_bin + 1;

        using var sw = new StreamWriter(Console.OpenStandardOutput(), bufferSize: 1024 * 1024);
        sw.WriteLine(num_stocks);

        for (int i = 0; i < num_stocks; i++)
        {
            int curr = bin_head[i];
            bool first = true;
            while (curr != -1)
            {
                if (!first) sw.Write(" ");
                sw.Write(curr);
                first = false;
                curr = cut_next[curr];
            }
            sw.WriteLine();
        }
    }
}
