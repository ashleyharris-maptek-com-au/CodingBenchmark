using System;
using System.IO;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;
using System.Threading.Tasks;
using System.Text;

struct Item {
    public int length;
    public int index;
}

struct NoisyItem : IComparable<NoisyItem> {
    public double score;
    public int idx;
    public int CompareTo(NoisyItem other) {
        return other.score.CompareTo(this.score);
    }
}

class Program {
    class Scanner {
        Stream s;
        byte[] buf = new byte[65536];
        int head = 0, tail = 0;
        public Scanner(Stream s) { this.s = s; }
        public int NextInt() {
            int res = 0;
            bool found = false;
            while (true) {
                if (head == tail) {
                    head = 0;
                    tail = s.Read(buf, 0, buf.Length);
                    if (tail == 0) break;
                }
                byte c = buf[head++];
                if (c >= '0' && c <= '9') {
                    res = res * 10 + (c - '0');
                    found = true;
                } else if (found) {
                    break;
                }
            }
            return res;
        }
        public bool HasNext() {
            while (true) {
                if (head == tail) {
                    head = 0;
                    tail = s.Read(buf, 0, buf.Length);
                    if (tail == 0) return false;
                }
                byte c = buf[head];
                if (c > 32) return true;
                head++;
            }
        }
    }

    class BinComparer : IComparer<(int cap, int id)> {
        public int Compare((int cap, int id) x, (int cap, int id) y) {
            int c = x.cap.CompareTo(y.cap);
            if (c != 0) return c;
            return x.id.CompareTo(y.id);
        }
    }

    static List<List<int>> BFD(Item[] items, int L) {
        var bins = new SortedSet<(int cap, int id)>(new BinComparer());
        var binContents = new List<List<int>>();
        int binCount = 0;
        
        foreach (var item in items) {
            var dummy = (cap: item.length, id: int.MinValue);
            var view = bins.GetViewBetween(dummy, (int.MaxValue, int.MaxValue));
            using var enumerator = view.GetEnumerator();
            if (enumerator.MoveNext()) {
                var bestBin = enumerator.Current;
                bins.Remove(bestBin);
                binContents[bestBin.id].Add(item.index);
                int newCap = bestBin.cap - item.length;
                if (newCap > 0) {
                    bins.Add((newCap, bestBin.id));
                }
            } else {
                int id = binCount++;
                binContents.Add(new List<int> { item.index });
                int newCap = L - item.length;
                if (newCap > 0) {
                    bins.Add((newCap, id));
                }
            }
        }
        return binContents;
    }

    class FFDTree {
        int[] tree;
        int B;
        public FFDTree(int n, int L) {
            tree = new int[4 * n + 1];
            B = n;
            Build(1, 0, B - 1, L);
        }
        void Build(int node, int start, int end, int L) {
            tree[node] = L;
            if (start == end) return;
            int mid = (start + end) / 2;
            Build(2 * node, start, mid, L);
            Build(2 * node + 1, mid + 1, end, L);
        }
        public int Assign(int node, int start, int end, int val) {
            if (tree[node] < val) return -1;
            if (start == end) {
                tree[node] -= val;
                return start;
            }
            int mid = (start + end) / 2;
            if (tree[2 * node] >= val) {
                int res = Assign(2 * node, start, mid, val);
                tree[node] = Math.Max(tree[2 * node], tree[2 * node + 1]);
                return res;
            } else {
                int res = Assign(2 * node + 1, mid + 1, end, val);
                tree[node] = Math.Max(tree[2 * node], tree[2 * node + 1]);
                return res;
            }
        }
    }

    static List<List<int>> FFD(Item[] items, int L) {
        var tree = new FFDTree(items.Length, L);
        var binContents = new List<List<int>>();
        int maxBinId = -1;
        
        foreach (var item in items) {
            int binId = tree.Assign(1, 0, items.Length - 1, item.length);
            if (binId > maxBinId) {
                while (binContents.Count <= binId) {
                    binContents.Add(new List<int>());
                }
                maxBinId = binId;
            }
            binContents[binId].Add(item.index);
        }
        
        var res = new List<List<int>>();
        for (int i = 0; i <= maxBinId; i++) {
            if (binContents[i].Count > 0) {
                res.Add(binContents[i]);
            }
        }
        return res;
    }

    static int GetLowerBound(Item[] items, int L) {
        long sum = 0;
        foreach (var item in items) sum += item.length;
        int l1 = (int)((sum + L - 1) / L);
        
        int max_l2 = l1;
        var distinct_w = new List<int>();
        int last = -1;
        foreach (var item in items) {
            if (item.length <= L / 2) {
                if (item.length != last) {
                    distinct_w.Add(item.length);
                    last = item.length;
                }
            }
        }
        if (distinct_w.Count == 0 || distinct_w.Last() != L / 2) {
            distinct_w.Add(L / 2);
        }
        
        int n = items.Length;
        long[] pref_weights = new long[n + 1];
        for (int i = 0; i < n; i++) {
            pref_weights[i + 1] = pref_weights[i] + items[i].length;
        }
        
        (int count, long weight) Query(int min_w, int max_w) {
            if (min_w > max_w) return (0, 0);
            int left = 0, right = n - 1;
            int start_idx = n;
            while (left <= right) {
                int mid = (left + right) / 2;
                if (items[mid].length <= max_w) {
                    start_idx = mid;
                    right = mid - 1;
                } else {
                    left = mid + 1;
                }
            }
            if (start_idx == n || items[start_idx].length < min_w) return (0, 0);
            
            left = start_idx; right = n - 1;
            int end_idx = start_idx - 1;
            while (left <= right) {
                int mid = (left + right) / 2;
                if (items[mid].length >= min_w) {
                    end_idx = mid;
                    left = mid + 1;
                } else {
                    right = mid - 1;
                }
            }
            if (end_idx < start_idx) return (0, 0);
            int count = end_idx - start_idx + 1;
            long weight = pref_weights[end_idx + 1] - pref_weights[start_idx];
            return (count, weight);
        }
        
        foreach (int C in distinct_w) {
            var j1 = Query(L - C + 1, L);
            var j2 = Query(L / 2 + 1, L - C);
            var j3 = Query(C, L / 2);
            
            long extra = j3.weight - (j2.count * (long)L - j2.weight);
            long extraBins = 0;
            if (extra > 0) {
                extraBins = (extra + L - 1) / L;
            }
            int l2 = j1.count + j2.count + (int)extraBins;
            if (l2 > max_l2) max_l2 = l2;
        }
        return max_l2;
    }

    static void Main() {
        Stopwatch sw = Stopwatch.StartNew();
        using var stream = Console.OpenStandardInput();
        var scanner = new Scanner(stream);
        
        if (!scanner.HasNext()) return;
        int N = scanner.NextInt();
        if (N == 0) {
            Console.WriteLine("0");
            return;
        }
        int L = scanner.NextInt();
        
        Item[] items = new Item[N];
        for (int i = 0; i < N; i++) {
            items[i] = new Item { length = scanner.NextInt(), index = i };
        }
        
        Array.Sort(items, (a, b) => b.length.CompareTo(a.length));
        
        int lowerBound = GetLowerBound(items, L);
        
        var bestSolution = BFD(items, L);
        if (bestSolution.Count == lowerBound) {
            PrintSolution(bestSolution);
            return;
        }
        
        var ffdSolution = FFD(items, L);
        if (ffdSolution.Count < bestSolution.Count) {
            bestSolution = ffdSolution;
        }
        
        if (bestSolution.Count == lowerBound) {
            PrintSolution(bestSolution);
            return;
        }
        
        object lockObj = new object();
        bool foundOptimal = false;
        
        Parallel.For(0, Environment.ProcessorCount, (threadId, state) => {
            var rnd = new Random(42 + threadId);
            Item[] localItems = new Item[N];
            Array.Copy(items, localItems, N);
            
            var noisy = new NoisyItem[N];
            Item[] permuted = new Item[N];
            
            while (sw.ElapsedMilliseconds < 270000 && !state.IsStopped && !foundOptimal) {
                double maxPerturbation = L * 0.05 * rnd.NextDouble();
                for (int i = 0; i < N; i++) {
                    noisy[i].score = localItems[i].length + rnd.NextDouble() * maxPerturbation;
                    noisy[i].idx = i;
                }
                Array.Sort(noisy);
                
                for (int i = 0; i < N; i++) {
                    permuted[i] = localItems[noisy[i].idx];
                }
                
                List<List<int>> sol;
                if (rnd.Next(4) == 0) sol = BFD(permuted, L);
                else sol = FFD(permuted, L);
                
                lock (lockObj) {
                    if (sol.Count < bestSolution.Count) {
                        bestSolution = sol;
                        if (bestSolution.Count == lowerBound) {
                            foundOptimal = true;
                            state.Stop();
                        }
                    }
                }
            }
        });
        
        PrintSolution(bestSolution);
    }
    
    static void PrintSolution(List<List<int>> sol) {
        using var outStream = Console.OpenStandardOutput();
        using var writer = new StreamWriter(outStream, Encoding.ASCII, 65536);
        writer.WriteLine(sol.Count);
        foreach (var bin in sol) {
            writer.WriteLine(string.Join(" ", bin));
        }
    }
}
