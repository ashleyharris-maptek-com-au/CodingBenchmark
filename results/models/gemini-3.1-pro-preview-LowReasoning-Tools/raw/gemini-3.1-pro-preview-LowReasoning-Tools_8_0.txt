use std::collections::VecDeque;
use std::io::{self, BufRead, Write};
use std::str::FromStr;

fn main() {
    let stdin = io::stdin();
    let mut stdin_lock = stdin.lock();
    
    let mut header = String::new();
    if stdin_lock.read_line(&mut header).unwrap() == 0 {
        return;
    }
    
    let parts: Vec<&str> = header.split_whitespace().collect();
    if parts.len() < 2 {
        return;
    }
    let h: usize = usize::from_str(parts[0]).unwrap();
    let w: usize = usize::from_str(parts[1]).unwrap();
    
    let mut grid: Vec<u8> = vec![0; h * w];
    let mut start = None;
    let mut end = None;
    
    let mut line = String::new();
    for y in 0..h {
        line.clear();
        stdin_lock.read_line(&mut line).unwrap();
        let bytes = line.as_bytes();
        let mut x = 0;
        for &b in bytes.iter() {
            if b == b'\n' || b == b'\r' {
                continue;
            }
            if x >= w {
                break;
            }
            let idx = y * w + x;
            match b {
                b'#' => grid[idx] = 1,
                b'A' => {
                    start = Some((x, y));
                    grid[idx] = 0;
                }
                b'B' => {
                    end = Some((x, y));
                    grid[idx] = 0;
                }
                _ => {
                    grid[idx] = 0;
                }
            }
            x += 1;
        }
    }
    
    if start.is_none() || end.is_none() {
        println!("0");
        return;
    }
    
    let (sx, sy) = start.unwrap();
    let (ex, ey) = end.unwrap();
    
    // We search from end to start.
    let mut queue = VecDeque::new();
    queue.push_back((ex, ey));
    grid[ey * w + ex] = 6; // Mark end as visited with special marker
    
    let mut found = false;
    
    while let Some((x, y)) = queue.pop_front() {
        if x == sx && y == sy {
            found = true;
            break;
        }
        
        // Check UP (x, y-1)
        if y > 0 {
            let ny = y - 1;
            let nx = x;
            let idx = ny * w + nx;
            if grid[idx] == 0 {
                grid[idx] = 3; // To go back, go DOWN
                queue.push_back((nx, ny));
            }
        }
        // Check DOWN (x, y+1)
        if y + 1 < h {
            let ny = y + 1;
            let nx = x;
            let idx = ny * w + nx;
            if grid[idx] == 0 {
                grid[idx] = 2; // To go back, go UP
                queue.push_back((nx, ny));
            }
        }
        // Check LEFT (x-1, y)
        if x > 0 {
            let ny = y;
            let nx = x - 1;
            let idx = ny * w + nx;
            if grid[idx] == 0 {
                grid[idx] = 5; // To go back, go RIGHT
                queue.push_back((nx, ny));
            }
        }
        // Check RIGHT (x+1, y)
        if x + 1 < w {
            let ny = y;
            let nx = x + 1;
            let idx = ny * w + nx;
            if grid[idx] == 0 {
                grid[idx] = 4; // To go back, go LEFT
                queue.push_back((nx, ny));
            }
        }
    }
    
    if !found {
        println!("0");
        return;
    }
    
    let mut len = 1;
    let mut curr_x = sx;
    let mut curr_y = sy;
    
    while curr_x != ex || curr_y != ey {
        let marker = grid[curr_y * w + curr_x];
        match marker {
            2 => curr_y -= 1, // go UP
            3 => curr_y += 1, // go DOWN
            4 => curr_x -= 1, // go LEFT
            5 => curr_x += 1, // go RIGHT
            _ => panic!("Invalid path"),
        }
        len += 1;
    }
    
    let stdout = io::stdout();
    let mut out = io::BufWriter::new(stdout.lock());
    
    writeln!(out, "{}", len).unwrap();
    
    curr_x = sx;
    curr_y = sy;
    writeln!(out, "{} {}", curr_x, curr_y).unwrap();
    
    while curr_x != ex || curr_y != ey {
        let marker = grid[curr_y * w + curr_x];
        match marker {
            2 => curr_y -= 1,
            3 => curr_y += 1,
            4 => curr_x -= 1,
            5 => curr_x += 1,
            _ => unreachable!(),
        }
        writeln!(out, "{} {}", curr_x, curr_y).unwrap();
    }
}
