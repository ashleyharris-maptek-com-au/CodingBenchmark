use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines().filter_map(Result::ok);

    let first_line = match lines.next() {
        Some(l) => l,
        None => return,
    };
    let parts: Vec<usize> = first_line.split_whitespace().filter_map(|s| s.parse().ok()).collect();
    if parts.len() < 2 {
        return;
    }
    let n = parts[0];
    let m = parts[1];

    if n == 0 {
        println!("0\n");
        return;
    }

    let mut adj = vec![0u128; n];
    let mut edges_read = 0;
    while edges_read < m {
        if let Some(line) = lines.next() {
            let p: Vec<usize> = line.split_whitespace().filter_map(|s| s.parse().ok()).collect();
            if p.len() >= 2 {
                let u = p[0];
                let v = p[1];
                if u < n && v < n {
                    adj[u] |= 1 << v;
                    adj[v] |= 1 << u;
                }
                edges_read += 1;
            }
        } else {
            break;
        }
    }

    let mut best_cut = u32::MAX;
    let mut best_a = 0u128;

    fn solve(
        index: usize,
        a_count: usize,
        b_count: usize,
        current_cut: u32,
        assigned_a: u128,
        assigned_b: u128,
        best_cut: &mut u32,
        best_a: &mut u128,
        adj: &[u128],
        n: usize,
    ) {
        if current_cut >= *best_cut {
            return;
        }
        
        let mut lower_bound = current_cut;
        for i in index..n {
            let c_a = (adj[i] & assigned_b).count_ones();
            let c_b = (adj[i] & assigned_a).count_ones();
            lower_bound += c_a.min(c_b);
        }
        if lower_bound >= *best_cut {
            return;
        }

        if index == n {
            *best_cut = current_cut;
            *best_a = assigned_a;
            return;
        }

        let half = n / 2;

        let cost_a = (adj[index] & assigned_b).count_ones();
        let cost_b = (adj[index] & assigned_a).count_ones();

        let try_a = a_count < half;
        let try_b = b_count < half && index > 0; // Force node 0 to A to break symmetry

        if try_a && (!try_b || cost_a <= cost_b) {
            solve(index + 1, a_count + 1, b_count, current_cut + cost_a, assigned_a | (1 << index), assigned_b, best_cut, best_a, adj, n);
            if try_b {
                solve(index + 1, a_count, b_count + 1, current_cut + cost_b, assigned_a, assigned_b | (1 << index), best_cut, best_a, adj, n);
            }
        } else if try_b {
            solve(index + 1, a_count, b_count + 1, current_cut + cost_b, assigned_a, assigned_b | (1 << index), best_cut, best_a, adj, n);
            if try_a {
                solve(index + 1, a_count + 1, b_count, current_cut + cost_a, assigned_a | (1 << index), assigned_b, best_cut, best_a, adj, n);
            }
        }
    }

    solve(0, 0, 0, 0, 0, 0, &mut best_cut, &mut best_a, &adj, n);

    let mut a_verts = Vec::new();
    for i in 0..n {
        if (best_a & (1 << i)) != 0 {
            a_verts.push(i);
        }
    }

    println!("{}", best_cut);
    let a_strs: Vec<String> = a_verts.iter().map(|v| v.to_string()).collect();
    println!("{}", a_strs.join(" "));
}
