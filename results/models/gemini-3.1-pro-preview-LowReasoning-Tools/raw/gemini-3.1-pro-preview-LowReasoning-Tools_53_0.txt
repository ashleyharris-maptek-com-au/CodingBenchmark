STATE = {}

def autopilot_step(state: dict, dt: float) -> dict:
    global STATE
    
    if not STATE:
        STATE = {
            'init': True,
            'x': 0.0, 'y': 0.0, 'z': 0.0,
            'vx': 0.0, 'vy': 0.0, 'vz': 0.0,
            'roll': 0.0, 'pitch': 0.0, 'yaw': 0.0,
            'last_cmd': [0.0, 0.0, 0.0],
            'mass': 1000.0,
        }
        
    a_orbit = state.get('ground_semi_major_axis_m', 6771000.0)
    if not state.get('ground_semi_major_axis_m_valid', True):
        a_orbit = (6371.0 + state.get('ground_altitude_km', 400.0)) * 1000.0
    mu = 3.986004418e14
    n = (mu / a_orbit**3)**0.5
    
    # 1. Predict Pos/Vel
    x, y, z = STATE['x'], STATE['y'], STATE['z']
    vx, vy, vz = STATE['vx'], STATE['vy'], STATE['vz']
    
    ax_thrust = STATE['last_cmd'][0] * 220.0 / STATE['mass']
    ay_thrust = STATE['last_cmd'][1] * 220.0 / STATE['mass']
    az_thrust = STATE['last_cmd'][2] * 220.0 / STATE['mass']
    
    ax_pred = 2*n*vy + 3*n**2*x + ax_thrust
    ay_pred = -2*n*vx + ay_thrust
    az_pred = -n**2*z + az_thrust
    
    x += vx * dt + 0.5 * ax_pred * dt**2
    y += vy * dt + 0.5 * ay_pred * dt**2
    z += vz * dt + 0.5 * az_pred * dt**2
    vx += ax_pred * dt
    vy += ay_pred * dt
    vz += az_pred * dt
    
    # 2. Measure Pos/Vel
    def get_val(keys, valid_keys):
        for k, vk in zip(keys, valid_keys):
            if state.get(vk, False):
                return state.get(k)
        return None

    mx = get_val(['lidar_rel_pos_x_m', 'radar_rel_pos_x_m'], ['lidar_rel_x_valid', 'radar_rel_x_valid'])
    my = get_val(['lidar_rel_pos_y_m', 'radar_rel_pos_y_m'], ['lidar_rel_y_valid', 'radar_rel_y_valid'])
    mz = get_val(['lidar_rel_pos_z_m', 'radar_rel_pos_z_m'], ['lidar_rel_z_valid', 'radar_rel_z_valid'])
    
    mvx = get_val(['lidar_rel_vel_x_ms', 'radar_rel_vel_x_ms'], ['lidar_relv_x_valid', 'radar_relv_x_valid'])
    mvy = get_val(['lidar_rel_vel_y_ms', 'radar_rel_vel_y_ms'], ['lidar_relv_y_valid', 'radar_relv_y_valid'])
    mvz = get_val(['lidar_rel_vel_z_ms', 'radar_rel_vel_z_ms'], ['lidar_relv_z_valid', 'radar_relv_z_valid'])
    
    dist_est = (x**2 + y**2 + z**2)**0.5
    
    if mx is None:
        if state.get('dock_camera_active', False):
            rng = state.get('dock_camera_range_m', 0.0)
            if state.get('dock_camera_bearing_x_valid', False): mx = rng * state.get('dock_camera_bearing_x', 0.0)
            if state.get('dock_camera_bearing_y_valid', False): my = rng * state.get('dock_camera_bearing_y', 0.0)
            if state.get('dock_camera_bearing_z_valid', False): mz = rng * state.get('dock_camera_bearing_z', 0.0)
        else:
            r1 = get_val(['range_1_m', 'range_2_m', 'range_3_m'], ['range_1_valid', 'range_2_valid', 'range_3_valid'])
            if r1 is not None and dist_est > 0.1:
                scale = r1 / dist_est
                mx = x * scale
                my = y * scale
                mz = z * scale
                
    alpha_p = 0.5
    alpha_v = 0.5
    
    if STATE['init']:
        x = mx if mx is not None else 0.0
        y = my if my is not None else -200.0
        z = mz if mz is not None else 0.0
        vx = mvx if mvx is not None else 0.0
        vy = mvy if mvy is not None else 0.0
        vz = mvz if mvz is not None else 0.0
        STATE['init'] = False
    else:
        if mx is not None: x += alpha_p * (mx - x)
        if my is not None: y += alpha_p * (my - y)
        if mz is not None: z += alpha_p * (mz - z)
        if mvx is not None: vx += alpha_v * (mvx - vx)
        if mvy is not None: vy += alpha_v * (mvy - vy)
        if mvz is not None: vz += alpha_v * (mvz - vz)
        
    STATE['x'], STATE['y'], STATE['z'] = x, y, z
    STATE['vx'], STATE['vy'], STATE['vz'] = vx, vy, vz
    
    # 3. Attitude Update
    roll, pitch, yaw = STATE['roll'], STATE['pitch'], STATE['yaw']
    
    g_roll = get_val(['gyro_1_roll_dps', 'gyro_2_roll_dps', 'gyro_3_roll_dps'], ['gyro_1_roll_valid', 'gyro_2_roll_valid', 'gyro_3_roll_valid'])
    g_pitch = get_val(['gyro_1_pitch_dps', 'gyro_2_pitch_dps', 'gyro_3_pitch_dps'], ['gyro_1_pitch_valid', 'gyro_2_pitch_valid', 'gyro_3_pitch_valid'])
    g_yaw = get_val(['gyro_1_yaw_dps', 'gyro_2_yaw_dps', 'gyro_3_yaw_dps'], ['gyro_1_yaw_valid', 'gyro_2_yaw_valid', 'gyro_3_yaw_valid'])
    
    if g_roll is not None: roll += g_roll * dt
    if g_pitch is not None: pitch += g_pitch * dt
    if g_yaw is not None: yaw += g_yaw * dt
    
    st_roll = get_val(['star_tracker_1_roll_deg', 'star_tracker_2_roll_deg'], ['star_tracker_1_roll_valid', 'star_tracker_2_roll_valid'])
    st_pitch = get_val(['star_tracker_1_pitch_deg', 'star_tracker_2_pitch_deg'], ['star_tracker_1_pitch_valid', 'star_tracker_2_pitch_valid'])
    st_yaw = get_val(['star_tracker_1_yaw_deg', 'star_tracker_2_yaw_deg'], ['star_tracker_1_yaw_valid', 'star_tracker_2_yaw_valid'])
    
    if st_roll is not None: roll += 0.5 * (st_roll - roll)
    if st_pitch is not None: pitch += 0.5 * (st_pitch - pitch)
    if st_yaw is not None: yaw += 0.5 * (st_yaw - yaw)
    
    STATE['roll'], STATE['pitch'], STATE['yaw'] = roll, pitch, yaw
    
    # 4. Guidance
    dist = (x**2 + y**2 + z**2)**0.5
    
    target_speed = min(5.0, max(0.05, 0.02 * dist))
    if dist < 0.5:
        target_speed = 0.0
        
    if dist > 0.1:
        dir_x, dir_y, dir_z = -x/dist, -y/dist, -z/dist
    else:
        dir_x, dir_y, dir_z = 0.0, 0.0, 0.0
        
    vx_t = dir_x * target_speed
    vy_t = dir_y * target_speed
    vz_t = dir_z * target_speed
    
    # 5. Control (Translation)
    kd = 0.2
    ax_cmd = kd * (vx_t - vx) - 2*n*vy - 3*n**2*x
    ay_cmd = kd * (vy_t - vy) + 2*n*vx
    az_cmd = kd * (vz_t - vz) + n**2*z
    
    max_accel = 220.0 / STATE['mass']
    tx = max(-1.0, min(1.0, ax_cmd / max_accel))
    ty = max(-1.0, min(1.0, ay_cmd / max_accel))
    tz = max(-1.0, min(1.0, az_cmd / max_accel))
    STATE['last_cmd'] = [tx, ty, tz]
    
    # 6. Control (Attitude)
    k_att = 0.02
    rate_roll = max(-0.035, min(0.035, -k_att * roll))
    rate_pitch = max(-0.035, min(0.035, -k_att * pitch))
    rate_yaw = max(-0.035, min(0.035, -k_att * yaw))
    
    return {
        'thrust_x': float(tx),
        'thrust_y': float(ty),
        'thrust_z': float(tz),
        'att_rate_roll': float(rate_roll),
        'att_rate_pitch': float(rate_pitch),
        'att_rate_yaw': float(rate_yaw)
    }
