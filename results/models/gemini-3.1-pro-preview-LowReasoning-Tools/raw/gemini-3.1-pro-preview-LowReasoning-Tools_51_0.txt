import statistics

# Globals for PID
_integral_alt = 0.0
_integral_hdg = 0.0
_integral_spd = 0.0

def get_median(state, keys, valid_keys, fallback_key):
    vals = []
    for k, vk in zip(keys, valid_keys):
        if state.get(vk, True): # Default to valid if flag is missing
            vals.append(state.get(k, 0))
    if vals:
        return statistics.median(vals)
    return state.get(fallback_key, 0)

def autopilot_step(state: dict, dt: float) -> dict:
    global _integral_alt, _integral_hdg, _integral_spd
    
    if not state.get('ap_engaged', False):
        return {}

    # 1. Sensor fusion (median voting)
    alt = get_median(state, 
                     ['baro_alt_1_ft', 'baro_alt_2_ft', 'baro_alt_3_ft'],
                     ['baro_alt_1_valid', 'baro_alt_2_valid', 'baro_alt_3_valid'],
                     'gps_1_alt_ft')
                     
    spd = get_median(state,
                     ['ias_1_kt', 'ias_2_kt', 'ias_3_kt'],
                     ['ias_1_valid', 'ias_2_valid', 'ias_3_valid'],
                     'gps_1_gs_kt')
                     
    hdg = get_median(state,
                     ['irs_1_heading_deg', 'irs_2_heading_deg', 'irs_3_heading_deg'],
                     ['irs_1_valid', 'irs_2_valid', 'irs_3_valid'],
                     'magnetic_heading_deg')
    
    target_alt = state.get('target_alt_ft', alt)
    target_hdg = state.get('target_hdg_deg', hdg)
    target_spd = state.get('target_ias_kt', spd)

    # PID Altitude -> Pitch / Elevator
    alt_err = target_alt - alt
    _integral_alt += alt_err * dt
    
    vs = state.get('vertical_speed_fpm', 0)
    target_vs = max(-2500, min(2500, alt_err * 2.5))
    vs_err = target_vs - vs
    
    elevator = max(-1.0, min(1.0, vs_err * 0.001)) 

    # PID Heading -> Roll / Aileron
    hdg_err = (target_hdg - hdg + 180) % 360 - 180
    _integral_hdg += hdg_err * dt
    
    target_roll = max(-30, min(30, hdg_err * 2.0))
    roll = state.get('irs_1_roll_deg', 0)
    roll_err = target_roll - roll
    
    aileron = max(-1.0, min(1.0, roll_err * 0.05))

    # PID Speed -> Throttle
    spd_err = target_spd - spd
    _integral_spd += spd_err * dt
    
    throttle_base = state.get('throttle_1_pos', 0.5)
    throttle_delta = spd_err * 0.02 + _integral_spd * 0.001
    throttle = max(0.0, min(1.0, throttle_base + throttle_delta))

    # Simple yaw dampening
    yaw_rate = state.get('gyro_1_yaw_rate_dps', 0)
    rudder = max(-1.0, min(1.0, -yaw_rate * 0.1))

    return {
        'elevator': elevator,
        'aileron': aileron,
        'rudder': rudder,
        'throttle_1': throttle,
        'throttle_2': throttle,
        'elevator_trim': elevator * 0.01  # Auto-trim
    }
