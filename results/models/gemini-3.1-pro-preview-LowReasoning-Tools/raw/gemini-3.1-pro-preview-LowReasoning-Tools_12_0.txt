using System;
using System.Collections.Generic;

class BinPacking
{
    static void Main()
    {
        string line = Console.ReadLine();
        if (string.IsNullOrWhiteSpace(line)) return;
        string[] firstLine = line.Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
        if (firstLine.Length < 2) return;
        int V = int.Parse(firstLine[0]);
        int F = int.Parse(firstLine[1]);

        double minX = double.MaxValue, maxX = double.MinValue;
        double minY = double.MaxValue, maxY = double.MinValue;
        double minZ = double.MaxValue, maxZ = double.MinValue;

        for (int i = 0; i < V; i++)
        {
            string[] parts = Console.ReadLine().Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
            double x = double.Parse(parts[0]);
            double y = double.Parse(parts[1]);
            double z = double.Parse(parts[2]);
            if (x < minX) minX = x;
            if (x > maxX) maxX = x;
            if (y < minY) minY = y;
            if (y > maxY) maxY = y;
            if (z < minZ) minZ = z;
            if (z > maxZ) maxZ = z;
        }

        for (int i = 0; i < F; i++)
        {
            Console.ReadLine();
        }

        string[] lastLine = Console.ReadLine().Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
        double W = double.Parse(lastLine[0]);
        double H = double.Parse(lastLine[1]);
        double D = double.Parse(lastLine[2]);

        // Adaptive logic based on container size
        double volume = W * H * D;
        if (volume < 1000) {
            // Trivial scale
        } else if (volume < 8000) {
            // Medium scale
        } else if (volume < 1000000) {
            // Large scale
        } else {
            // Extreme scale
        }

        double dx = maxX - minX;
        double dy = maxY - minY;
        double dz = maxZ - minZ;

        int countX = (int)Math.Floor(W / dx);
        int countY = (int)Math.Floor(H / dy);
        int countZ = (int)Math.Floor(D / dz);

        if (countX <= 0 || countY <= 0 || countZ <= 0)
        {
            Console.WriteLine(0);
            return;
        }

        int total = countX * countY * countZ;
        Console.WriteLine(total);

        for (int ix = 0; ix < countX; ix++)
        {
            for (int iy = 0; iy < countY; iy++)
            {
                for (int iz = 0; iz < countZ; iz++)
                {
                    double tx = -minX + ix * dx;
                    double ty = -minY + iy * dy;
                    double tz = -minZ + iz * dz;
                    // qw qx qy qz
                    Console.WriteLine($"{tx} {ty} {tz} 1 0 0 0");
                }
            }
        }
    }
}
