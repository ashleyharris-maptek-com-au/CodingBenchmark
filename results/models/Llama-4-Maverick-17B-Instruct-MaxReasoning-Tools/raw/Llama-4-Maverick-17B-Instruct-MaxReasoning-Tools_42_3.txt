```spirv
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos_in %normal_in %color_in %gl_out %outColor
OpExecutionMode %main InputLines
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1

OpDecorate %ubo ArrayStride 16
OpMemberDecorate %ubo 0 Offset 0
OpMemberDecorate %ubo 1 Offset 64
OpMemberDecorate %ubo 2 Offset 128
OpMemberDecorate %ubo 3 Offset 192
OpMemberDecorate %ubo 4 Offset 208
OpDecorate %ubo Block
OpDecorate %ubov DescriptorSet 0
OpDecorate %ubov Binding 0

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

OpDecorate %worldPos_in Location 0
OpDecorate %normal_in Location 1
OpDecorate %color_in Location 2
OpDecorate %outColor Location 0

%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_2 = OpConstant %uint 2
%uint_0 = OpConstant %uint 0
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%float_0 = OpConstant %float 0
%float_003 = OpConstant %float 0.03

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3_in = OpTypeArray %v3float %uint_2
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%worldPos_in = OpVariable %ptr_in_arr_v3 Input
%normal_in = OpVariable %ptr_in_arr_v3 Input
%color_in = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%ubo = OpTypeStruct %v4float %v4float %v4float %v4float %v3float %v3float %v4float
%ptr_ubo = OpTypePointer Uniform %ubo
%ubov = OpVariable %ptr_ubo Uniform

%main = OpFunction %void None %void
%entry = OpLabel

; Load input positions and colors
%in_p0 = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%p0 = OpLoad %v4float %in_p0
%in_p1 = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%p1 = OpLoad %v4float %in_p1

%in_c0 = OpAccessChain %ptr_in_arr_v3 %color_in %int_0
%c0 = OpLoad %v3float %in_c0
%in_c1 = OpAccessChain %ptr_in_arr_v3 %color_in %int_1
%c1 = OpLoad %v3float %in_c1

; Compute line direction and perpendicular
%p0_xy = OpVectorShuffle %v2float %p0 %p0 0 1
%p1_xy = OpVectorShuffle %v2float %p1 %p1 0 1
%dir_xy = OpFSub %v2float %p1_xy %p0_xy
%perp_xy = OpVectorShuffle %v2float %dir_xy %dir_xy 1 0
%perp_xy = OpFMul %v2float %perp_xy %float_003 %float_0 %float_0 %float_-1
%length = OpExtInst %float %1 Length %perp_xy
%perp_xy = OpFDiv %v2float %perp_xy %length
%perp = OpVectorShuffle %v4float %perp_xy %float_0 %float_0 0 1

; Emit vertices
%p0_plus_perp = OpFAdd %v4float %p0 %perp
%out_p0_plus_perp = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_p0_plus_perp %p0_plus_perp
%out_c0 = OpAccessChain %ptr_out_v3 %outColor
OpStore %out_c0 %c0
OpEmitVertex

%p0_minus_perp = OpFSub %v4float %p0 %perp
%out_p0_minus_perp = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_p0_minus_perp %p0_minus_perp
%out_c0 = OpAccessChain %ptr_out_v3 %outColor
OpStore %out_c0 %c0
OpEmitVertex

%p1_plus_perp = OpFAdd %v4float %p1 %perp
%out_p1_plus_perp = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_p1_plus_perp %p1_plus_perp
%out_c1 = OpAccessChain %ptr_out_v3 %outColor
OpStore %out_c1 %c1
OpEmitVertex

%p1_minus_perp = OpFSub %v4float %p1 %perp
%out_p1_minus_perp = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_p1_minus_perp %p1_minus_perp
%out_c1 = OpAccessChain %ptr_out_v3 %outColor
OpStore %out_c1 %c1
OpEmitVertex

OpEndPrimitive

OpReturn
OpFunctionEnd
```
