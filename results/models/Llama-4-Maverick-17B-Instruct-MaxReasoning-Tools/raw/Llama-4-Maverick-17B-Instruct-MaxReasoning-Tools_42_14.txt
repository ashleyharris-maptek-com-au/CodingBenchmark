```spirv
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos_in %normal_in %color_in %gl_out %outColor
OpExecutionMode %main InputTrianglesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 3
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %worldPos_in Location 0
OpDecorate %normal_in Location 1
OpDecorate %color_in Location 2
OpDecorate %outColor Location 0
OpDecorate %ubo Binding 0
OpDecorate %ubo DescriptorSet 0

%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%uint_6 = OpConstant %uint 6
%uint_3 = OpConstant %uint 3
%int = OpTypeInt 32 1
%int_0 = OpConstant %int 0
%int_2 = OpConstant %int 2
%int_4 = OpConstant %int 4
%float_0 = OpConstant %float 0
%float_1 = OpConstant %float 1
%float_0_2 = OpConstant %float 0.2
%float_0_8 = OpConstant %float 0.8

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_6
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3float = OpTypeArray %v3float %uint_6
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3float
%worldPos_in = OpVariable %ptr_in_arr_v3 Input
%normal_in = OpVariable %ptr_in_arr_v3 Input
%color_in = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%ubo_type = OpTypeStruct %v4float %v4float %v4float %v4float %v4float %v4float %v4float %v4float
%ptr_ubo = OpTypePointer Uniform %ubo_type
%ubo = OpVariable %ptr_ubo Uniform

%fn_type = OpTypeFunction %void

%main = OpFunction %void None %fn_type
%entry = OpLabel

; Load positions
%ptr_p0 = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%p0 = OpLoad %v4float %ptr_p0
%p0_xyz = OpVectorShuffle %v3float %p0 %p0 0 1 2
%ptr_p2 = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%p2 = OpLoad %v4float %ptr_p2
%p2_xyz = OpVectorShuffle %v3float %p2 %p2 0 1 2
%ptr_p4 = OpAccessChain %ptr_in_v4 %gl_in %int_4 %int_0
%p4 = OpLoad %v4float %ptr_p4
%p4_xyz = OpVectorShuffle %v3float %p4 %p4 0 1 2

; Compute face normal
%p2_p0 = OpFSub %v3float %p2_xyz %p0_xyz
%p4_p0 = OpFSub %v3float %p4_xyz %p0_xyz
%cross_x = OpFMul %float %p2_p0_1 %p4_p0_2
%cross_x = OpFSub %float %cross_x %p2_p0_2 %p4_p0_1
%cross_y = OpFMul %float %p2_p0_2 %p4_p0_0
%cross_y = OpFSub %float %cross_y %p2_p0_0 %p4_p0_2
%cross_z = OpFMul %float %p2_p0_0 %p4_p0_1
%cross_z = OpFSub %float %cross_z %p2_p0_1 %p4_p0_0
%cross = OpCompositeConstruct %v3float %cross_x %cross_y %cross_z
%N = OpExtInst %v3float %1 Normalize %cross

; Compute light direction and diffuse
%L = OpCompositeConstruct %v3float %float_1 %float_1 %float_1
%L = OpExtInst %v3float %1 Normalize %L
%dot_NL = OpDot %float %N %L
%diffuse = OpExtInst %float %1 FMax %dot_NL %float_0

; Compute final color
%ptr_c0 = OpAccessChain %ptr_in_arr_v3 %color_in %int_0
%c0 = OpLoad %v3float %ptr_c0
%ptr_c2 = OpAccessChain %ptr_in_arr_v3 %color_in %int_2
%c2 = OpLoad %v3float %ptr_c2
%ptr_c4 = OpAccessChain %ptr_in_arr_v3 %color_in %int_4
%c4 = OpLoad %v3float %ptr_c4
%c_sum = OpFAdd %v3float %c0 %c2
%c_sum = OpFAdd %v3float %c_sum %c4
%c_avg = OpFMul %v3float %c_sum %float_0_333333
%final_c = OpFMul %v3float %c_avg %float_0_2
%tmp = OpFMul %float %float_0_8 %diffuse
%tmp_v3 = OpCompositeConstruct %v3float %tmp %tmp %tmp
%final_c = OpFAdd %v3float %final_c %tmp_v3

; Emit vertices
%ptr_out_pos0 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out_pos0 %p0
%ptr_out_pos2 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out_pos2 %p2
%ptr_out_pos4 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out_pos4 %p4
OpStore %outColor %final_c
OpEmitVertex
OpStore %ptr_out_v4 %gl_out %int_0 %p2
OpStore %outColor %final_c
OpEmitVertex
OpStore %ptr_out_v4 %gl_out %int_0 %p4
OpStore %outColor %final_c
OpEmitVertex
OpEndPrimitive

OpReturn
OpFunctionEnd
```
