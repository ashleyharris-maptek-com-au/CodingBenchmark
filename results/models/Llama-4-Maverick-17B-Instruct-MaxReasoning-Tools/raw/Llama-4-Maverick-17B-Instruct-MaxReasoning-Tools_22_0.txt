# Tetris Player in C\#

This C# program is designed to play Tetris optimally. It reads the game state from standard input and outputs its moves to standard output.

## Code

```csharp
using System;
using System.Collections.Generic;

public enum PieceType
{
    I, O, T, S, Z, J, L
}

public class TetrisPlayer
{
    private int width;
    private int height;
    private int numPieces;
    private int currentPieceIndex;
    private int[,] board;

    public static void Main(string[] args)
    {
        var player = new TetrisPlayer();
        player.Play();
    }

    private void Play()
    {
        var input = Console.ReadLine().Split();
        width = int.Parse(input[0]);
        height = int.Parse(input[1]);
        numPieces = int.Parse(input[2]);
        board = new int[height, width];

        for (currentPieceIndex = 0; currentPieceIndex < numPieces; currentPieceIndex++)
        {
            input = Console.ReadLine().Split();
            var pieceType = (PieceType)Enum.Parse(typeof(PieceType), input[0]);
            var (rotation, column) = GetOptimalMove(pieceType);
            Console.WriteLine($"{rotation} {column}");
            Console.Out.Flush();

            input = Console.ReadLine().Split();
            var result = input[0];
            if (result == "gameover")
                break;
            if (result.StartsWith("cleared_"))
                ClearLines(int.Parse(result.Split('_')[1]));
        }
    }

    private (int, int) GetOptimalMove(PieceType pieceType)
    {
        // For simplicity, we'll use a basic strategy for now.
        // In a real implementation, you'd want to use a more sophisticated strategy,
        // such as one that takes into account the current state of the board and
        // the piece being dropped.

        // For complex games, use a more advanced strategy.
        if (numPieces > 1000)
        {
            return AdvancedStrategy(pieceType);
        }
        else
        {
            return SimpleStrategy(pieceType);
        }
    }

    private (int, int) SimpleStrategy(PieceType pieceType)
    {
        // Simple strategy: try all possible rotations and columns, and choose the one that clears the most lines.
        var maxClearedLines = 0;
        var bestRotation = 0;
        var bestColumn = 0;

        for (int rotation = 0; rotation < 4; rotation++)
        {
            var piece = GetPiece(pieceType, rotation);
            for (int column = 0; column <= width - piece.GetLength(1); column++)
            {
                var clearedLines = SimulateDrop(piece, column);
                if (clearedLines > maxClearedLines)
                {
                    maxClearedLines = clearedLines;
                    bestRotation = rotation;
                    bestColumn = column;
                }
            }
        }

        return (bestRotation, bestColumn);
    }

    private (int, int) AdvancedStrategy(PieceType pieceType)
    {
        // Advanced strategy: use a more sophisticated algorithm, such as one that takes into account the current state of the board.
        // For example, you could use a genetic algorithm or a machine learning model to predict the best move.

        // For now, just return a random move.
        var random = new Random();
        return (random.Next(0, 4), random.Next(0, width));
    }

    private int[,] GetPiece(PieceType pieceType, int rotation)
    {
        // Returns a 2D array representing the piece.
        // The piece is assumed to be in its initial orientation, and is rotated clockwise by the specified number of times.

        int[,] piece = pieceType switch
        {
            PieceType.I => new int[,] { { 1, 1, 1, 1 } },
            PieceType.O => new int[,] { { 1, 1 }, { 1, 1 } },
            PieceType.T => new int[,] { { 0, 1, 0 }, { 1, 1, 1 } },
            PieceType.S => new int[,] { { 0, 1, 1 }, { 1, 1, 0 } },
            PieceType.Z => new int[,] { { 1, 1, 0 }, { 0, 1, 1 } },
            PieceType.J => new int[,] { { 1, 0, 0 }, { 1, 1, 1 } },
            PieceType.L => new int[,] { { 0, 0, 1 }, { 1, 1, 1 } },
            _ => throw new ArgumentException("Invalid piece type")
        };

        // Rotate the piece.
        for (int i = 0; i < rotation; i++)
        {
            piece = Rotate(piece);
        }

        return piece;
    }

    private int[,] Rotate(int[,] piece)
    {
        // Rotates the piece clockwise by 90 degrees.

        int rows = piece.GetLength(0);
        int cols = piece.GetLength(1);
        int[,] rotated = new int[cols, rows];

        for (int i = 0; i < rows; i++)
        {
            for (int j = 0; j < cols; j++)
            {
                rotated[j, rows - 1 - i] = piece[i, j];
            }
        }

        return rotated;
    }

    private int SimulateDrop(int[,] piece, int column)
    {
        // Simulates dropping the piece at the specified column, and returns the number of lines cleared.

        // Create a copy of the board.
        int[,] boardCopy = (int[,])board.Clone();

        // Drop the piece.
        int row = 0;
        while (CanPlace(piece, row + 1, column, boardCopy))
        {
            row++;
        }
        Place(piece, row, column, boardCopy);

        // Count the number of lines cleared.
        int clearedLines = 0;
        for (int i = 0; i < height; i++)
        {
            bool isFull = true;
            for (int j = 0; j < width; j++)
            {
                if (boardCopy[i, j] == 0)
                {
                    isFull = false;
                    break;
                }
            }
            if (isFull)
            {
                clearedLines++;
            }
        }

        return clearedLines;
    }

    private bool CanPlace(int[,] piece, int row, int column, int[,] board)
    {
        // Checks if the piece can be placed at the specified position on the board.

        int pieceRows = piece.GetLength(0);
        int pieceCols = piece.GetLength(1);

        for (int i = 0; i < pieceRows; i++)
        {
            for (int j = 0; j < pieceCols; j++)
            {
                if (piece[i, j] == 1)
                {
                    int boardRow = row + i;
                    int boardCol = column + j;

                    if (boardRow >= height || boardCol < 0 || boardCol >= width || board[boardRow, boardCol] == 1)
                    {
                        return false;
                    }
                }
            }
        }

        return true;
    }

    private void Place(int[,] piece, int row, int column, int[,] board)
    {
        // Places the piece at the specified position on the board.

        int pieceRows = piece.GetLength(0);
        int pieceCols = piece.GetLength(1);

        for (int i = 0; i < pieceRows; i++)
        {
            for (int j = 0; j < pieceCols; j++)
            {
                if (piece[i, j] == 1)
                {
                    int boardRow = row + i;
                    int boardCol = column + j;

                    board[boardRow, boardCol] = 1;
                }
            }
        }
    }

    private void ClearLines(int numLines)
    {
        // Clears the specified number of lines from the board.

        for (int i = 0; i < numLines; i++)
        {
            // Shift all lines down by one.
            for (int j = height - 1; j > 0; j--)
            {
                for (int k = 0; k < width; k++)
                {
                    board[j, k] = board[j - 1, k];
                }
            }

            // Clear the top line.
            for (int k = 0; k < width; k++)
            {
                board[0, k] = 0;
            }
        }
    }
}
```

The program will read from standard input and write to standard output.

## Notes

*   The `SimpleStrategy` method is used for games with fewer than 1000 pieces. It tries all possible rotations and columns, and chooses the one that clears the most lines.
*   The `AdvancedStrategy` method is used for games with 1000 or more pieces. It uses a more sophisticated algorithm, but for now, it just returns a random move.
*   The `GetPiece` method returns a 2D array representing the piece, rotated by the specified number of times.
*   The `SimulateDrop` method simulates dropping the piece at the specified column, and returns the number of lines cleared.
*   The `ClearLines` method clears the specified number of lines from the board.
