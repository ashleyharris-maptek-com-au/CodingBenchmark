// Function to simulate fire spread and calculate the value of towns that would be burned
function simulateFire(grid, width, height, ignitions, towns, firebreaks) {
    // Create a grid to track visited cells
    const visited = Array.from({ length: height }, () => Array(width).fill(false));
    
    // Mark firebreaks and ignitions as visited
    for (const [x, y] of firebreaks) {
        visited[y][x] = true;
    }
    for (const [x, y] of ignitions) {
        visited[y][x] = true;
    }

    // Perform BFS to simulate fire spread
    const queue = [...ignitions];
    while (queue.length > 0) {
        const [x, y] = queue.shift();
        for (const [dx, dy] of [[-1, 0], [1, 0], [0, -1], [0, 1]]) {
            const nx = x + dx;
            const ny = y + dy;
            if (nx >= 0 && nx < width && ny >= 0 && ny < height && !visited[ny][nx] && grid[ny][nx] === '.') {
                visited[ny][nx] = true;
                queue.push([nx, ny]);
            }
        }
    }

    // Calculate the value of towns that would be burned
    let burnedValue = 0;
    for (const town of towns) {
        if (visited[town.y][town.x]) {
            burnedValue += town.value;
        }
    }
    return burnedValue;
}

// Main function to determine the optimal firebreaks
function determineFirebreaks(data) {
    const { width, height, grid, ignitions, towns, cutBudget } = data;
    const firebreaks = [];

    // Simulate initial fire spread to identify cells that are reachable by fire
    const initialBurnedValue = simulateFire(grid, width, height, ignitions, towns, []);

    // If no towns are burned, return empty firebreaks
    if (initialBurnedValue === 0) {
        return { firebreaks };
    }

    // Create a priority queue to store cells to be considered for firebreaks
    const queue = [];
    for (const [x, y] of ignitions) {
        for (const [dx, dy] of [[-1, 0], [1, 0], [0, -1], [0, 1]]) {
            const nx = x + dx;
            const ny = y + dy;
            if (nx >= 0 && nx < width && ny >= 0 && ny < height && grid[ny][nx] === '.') {
                queue.push([nx, ny, 0]); // [x, y, distance]
            }
        }
    }

    // Perform a greedy search to find the optimal firebreaks
    while (queue.length > 0 && cutBudget > 0) {
        queue.sort((a, b) => {
            // Prioritize cells that are closer to ignitions and have higher town value saved
            const valueSavedA = simulateFire(grid, width, height, ignitions, towns, [...firebreaks, [a[0], a[1]]]) - initialBurnedValue;
            const valueSavedB = simulateFire(grid, width, height, ignitions, towns, [...firebreaks, [b[0], b[1]]]) - initialBurnedValue;
            return valueSavedB - valueSavedA || a[2] - b[2];
        });
        const [x, y, distance] = queue.shift();
        if (grid[y][x] === '.' && !firebreaks.some(([fx, fy]) => fx === x && fy === y)) {
            firebreaks.push([x, y]);
            cutBudget--;
            // Update the queue with new neighbors
            for (const [dx, dy] of [[-1, 0], [1, 0], [0, -1], [0, 1]]) {
                const nx = x + dx;
                const ny = y + dy;
                if (nx >= 0 && nx < width && ny >= 0 && ny < height && grid[ny][nx] === '.') {
                    queue.push([nx, ny, distance + 1]);
                }
            }
        }
    }

    return { firebreaks };
}

// Determine and output the optimal firebreaks
const result = determineFirebreaks(data);
