using System;
using System.Collections.Generic;
using System.Diagnostics;

class Program
{
    static void Main(string[] args)
    {
        var stopwatch = Stopwatch.StartNew();

        // Read input
        var (height, width) = ReadDimensions();
        var board = ReadBoard(height, width);

        // Choose strategy based on board complexity
        var safeCells = height * width > 1000000 ? SimpleStrategy(board, height, width) : AdvancedStrategy(board, height, width);

        // Output results
        OutputSafeCells(safeCells);

        stopwatch.Stop();
        Console.Error.WriteLine($"Execution time: {stopwatch.Elapsed.TotalSeconds} seconds");
    }

    static (int, int) ReadDimensions()
    {
        var line = Console.ReadLine().Split();
        return (int.Parse(line[0]), int.Parse(line[1]));
    }

    static char[,] ReadBoard(int height, int width)
    {
        var board = new char[height, width];
        for (int i = 0; i < height; i++)
        {
            var line = Console.ReadLine();
            for (int j = 0; j < width; j++)
            {
                board[i, j] = line[j];
            }
        }
        return board;
    }

    static List<(int, int)> SimpleStrategy(char[,] board, int height, int width)
    {
        var safeCells = new List<(int, int)>();

        // Iterate over revealed cells
        for (int i = 0; i < height; i++)
        {
            for (int j = 0; j < width; j++)
            {
                if (char.IsDigit(board[i, j]))
                {
                    var count = int.Parse(board[i, j].ToString());
                    var unknownNeighbors = GetUnknownNeighbors(board, i, j, height, width);

                    // If the number of unknown neighbors equals the count, they are all mines
                    if (unknownNeighbors.Count == count)
                    {
                        // Mark neighbors as not safe (not necessary, but for clarity)
                    }
                    // If the count is zero, all neighbors are safe
                    else if (count == 0)
                    {
                        foreach (var neighbor in GetNeighbors(i, j, height, width))
                        {
                            if (board[neighbor.Item1, neighbor.Item2] == ' ')
                            {
                                safeCells.Add((neighbor.Item2, neighbor.Item1));
                            }
                        }
                    }
                }
            }
        }

        return safeCells;
    }

    static List<(int, int)> AdvancedStrategy(char[,] board, int height, int width)
    {
        var safeCells = new List<(int, int)>();

        // Iterate over revealed cells
        for (int i = 0; i < height; i++)
        {
            for (int j = 0; j < width; j++)
            {
                if (char.IsDigit(board[i, j]))
                {
                    var count = int.Parse(board[i, j].ToString());
                    var unknownNeighbors = GetUnknownNeighbors(board, i, j, height, width);
                    var flaggedNeighbors = GetFlaggedNeighbors(board, i, j, height, width);

                    // If the number of unknown neighbors plus flagged neighbors equals the count, unknown neighbors are mines
                    if (unknownNeighbors.Count + flaggedNeighbors.Count == count)
                    {
                        // Mark neighbors as not safe (not necessary, but for clarity)
                    }
                    // If the count equals the number of flagged neighbors, unknown neighbors are safe
                    else if (count == flaggedNeighbors.Count)
                    {
                        foreach (var neighbor in unknownNeighbors)
                        {
                            safeCells.Add((neighbor.Item2, neighbor.Item1));
                        }
                    }
                }
            }
        }

        return safeCells;
    }

    static List<(int, int)> GetUnknownNeighbors(char[,] board, int i, int j, int height, int width)
    {
        var unknownNeighbors = new List<(int, int)>();
        foreach (var neighbor in GetNeighbors(i, j, height, width))
        {
            if (board[neighbor.Item1, neighbor.Item2] == ' ')
            {
                unknownNeighbors.Add(neighbor);
            }
        }
        return unknownNeighbors;
    }

    static int GetFlaggedNeighbors(char[,] board, int i, int j, int height, int width)
    {
        var count = 0;
        foreach (var neighbor in GetNeighbors(i, j, height, width))
        {
            if (board[neighbor.Item1, neighbor.Item2] == '*')
            {
                count++;
            }
        }
        return count;
    }

    static IEnumerable<(int, int)> GetNeighbors(int i, int j, int height, int width)
    {
        for (int x = -1; x <= 1; x++)
        {
            for (int y = -1; y <= 1; y++)
            {
                if (x == 0 && y == 0) continue;
                var ni = i + x;
                var nj = j + y;
                if (ni >= 0 && ni < height && nj >= 0 && nj < width)
                {
                    yield return (ni, nj);
                }
            }
        }
    }

    static void OutputSafeCells(List<(int, int)> safeCells)
    {
        Console.WriteLine(safeCells.Count);
        foreach (var cell in safeCells)
        {
            Console.WriteLine($"{cell.Item1} {cell.Item2}");
        }
    }
}

This C# code reads a Minesweeper board from standard input, applies either a simple or advanced strategy to identify provably safe cells based on the board complexity, and outputs the results to standard output.

1.  **Input Handling**: The program reads the board dimensions and the board state from standard input.
2.  **Strategy Selection**: It chooses between a simple and an advanced strategy based on the board size. For larger boards (over 1,000,000 cells), it uses the simple strategy; otherwise, it uses the advanced strategy.
3.  **Simple Strategy**: This strategy checks revealed cells with a count of adjacent mines. If a cell has a count of zero, all its neighbors are considered safe.
4.  **Advanced Strategy**: In addition to the simple strategy's logic, the advanced strategy also considers cells where the count of adjacent mines equals the number of flagged neighbors, marking unknown neighbors as safe.
5.  **Output**: The program outputs the number of safe cells found, followed by their coordinates (column and row, 0-indexed).

The code is designed to handle varying board sizes and mine densities efficiently, adhering to the constraints of not guessing and only returning cells that are provably safe.
