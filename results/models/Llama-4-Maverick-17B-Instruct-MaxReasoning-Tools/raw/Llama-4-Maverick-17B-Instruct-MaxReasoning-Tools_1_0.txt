import sys
import math
import random

def distance(city1, city2):
    x1, y1 = city1
    x2, y2 = city2
    return math.sqrt((x2-x1)**2 + (y2-y1)**2)

def nearest_neighbor(cities, start):
    num_cities = len(cities)
    visited = [False] * num_cities
    visited[start] = True
    tour = [start]
    current = start
    for _ in range(num_cities - 1):
        min_dist = float('inf')
        next_city = None
        for city in range(num_cities):
            if not visited[city]:
                dist = distance(cities[current], cities[city])
                if dist < min_dist:
                    min_dist = dist
                    next_city = city
        visited[next_city] = True
        tour.append(next_city)
        current = next_city
    return tour

def two_opt(tour, cities):
    num_cities = len(cities)
    improved = True
    while improved:
        improved = False
        for i in range(num_cities - 1):
            for j in range(i + 1, num_cities):
                new_tour = tour[:i] + list(reversed(tour[i:j])) + tour[j:]
                if total_distance(new_tour, cities) < total_distance(tour, cities):
                    tour = new_tour
                    improved = True
    return tour

def total_distance(tour, cities):
    total = 0
    for i in range(len(tour) - 1):
        total += distance(cities[tour[i]], cities[tour[i+1]])
    return total

def main():
    N = int(sys.stdin.readline().strip())
    cities = []
    for _ in range(N):
        x, y = map(float, sys.stdin.readline().strip().split())
        cities.append((x, y))

    if N <= 20:
        # For small N, we could use a brute force approach, but it's not implemented here
        tour = nearest_neighbor(cities, 0)
        tour = two_opt(tour, cities)
    elif N <= 1000:
        tour = nearest_neighbor(cities, 0)
        tour = two_opt(tour, cities)
    else:
        tour = nearest_neighbor(cities, 0)

    # Ensure the tour starts with 0
    if tour[0] != 0:
        index = tour.index(0)
        tour = [0] + tour[index+1:] + tour[1:index]

    print(' '.join(map(str, tour)))

if __name__ == "__main__":
    main()
