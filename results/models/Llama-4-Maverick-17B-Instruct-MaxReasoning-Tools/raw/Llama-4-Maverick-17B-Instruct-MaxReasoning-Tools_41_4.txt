               OpExecutionMode %main OriginUpperLeft

               OpSource GLSL 450
               OpName %main "main"

               OpName %normal "normal"
               OpName %ubo "ubo"
               OpName %outColor "outColor"
               OpMemberName %ubo_type 0 "model"
               OpMemberName %ubo_type 1 "view"
               OpMemberName %ubo_type 2 "projection"

               OpMemberName %ubo_type 4 "cameraPos"
               OpMemberName %ubo_type 5 "params"
               OpDecorate %worldPos Location 0
               OpDecorate %normal Location 1
               OpDecorate %ubo Block
               OpMemberDecorate %ubo_type 0 Offset 0
               OpMemberDecorate %ubo_type 0 ColMajor
               OpMemberDecorate %ubo_type 0 MatrixStride 16
               OpMemberDecorate %ubo_type 1 Offset 64
               OpMemberDecorate %ubo_type 1 ColMajor
               OpMemberDecorate %ubo_type 1 MatrixStride 16
               OpMemberDecorate %ubo_type 2 Offset 128
               OpMemberDecorate %ubo_type 2 ColMajor
               OpMemberDecorate %ubo_type 2 MatrixStride 16
               OpMemberDecorate %ubo_type 3 Offset 192
               OpMemberDecorate %ubo_type 4 Offset 208
               OpMemberDecorate %ubo_type 5 Offset 224
               OpDecorate %ubo DescriptorSet 0
               OpDecorate %ubo Binding 0
               OpDecorate %outColor Location 0

       %void = OpTypeVoid
          %3 = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v3float = OpTypeVector %float 3
    %v4float = OpTypeVector %float 4
       %uint = OpTypeInt 32 0
     %uint_0 = OpConstant %uint 0
     %uint_3 = OpConstant %uint 3
%mat4v4float = OpTypeMatrix %v4float 4
%ubo_type = OpTypeStruct %mat4v4float %mat4v4float %mat4v4float %v4float %v4float %v4float
%_ptr_Uniform_ubo_type = OpTypePointer Uniform %ubo_type
        %ubo = OpVariable %_ptr_Uniform_ubo_type Uniform
%_ptr_Input_v3float = OpTypePointer Input %v3float
   %worldPos = OpVariable %_ptr_Input_v3float Input
      %normal = OpVariable %_ptr_Input_v3float Input
%_ptr_Output_v4float = OpTypePointer Output %v4float
    %outColor = OpVariable %_ptr_Output_v4float Output
%_ptr_Uniform_v4float = OpTypePointer Uniform %v4float
       %bool = OpTypeBool
      %int_0 = OpConstant %int 0
     %float_0 = OpConstant %float 0
     %float_1 = OpConstant %float 1
   %v3float_0 = OpConstantComposite %v3float %float_0 %float_0 %float_0
   %v4float_1 = OpConstantComposite %v4float %float_1 %float_1 %float_1 %float_1

       %main = OpFunction %void None %3
          %5 = OpLabel
   %normal_v = OpLoad %v3float %normal
   %normalize_tmp = OpExtInst %v3float %1 Normalize %normal_v
         %N = OpCopyObject %v3float %normalize_tmp
%lightPos_vec = OpAccessChain %_ptr_Uniform_v4float %ubo %uint_3
   %lightPos = OpLoad %v4float %lightPos_vec
%lightPos_xyz = OpVectorShuffle %v3float %lightPos %lightPos 0 1 2
   %worldPos_v = OpLoad %v3float %worldPos
         %L_v = OpFSub %v3float %lightPos_xyz %worldPos_v
         %L = OpExtInst %v3float %1 Normalize %L_v
       %dot = OpDot %float %N %L
       %d = OpExtInst %float %1 FMax %dot %float_0
       %d_vec = OpCompositeConstruct %v4float %d %d %d %float_1
               OpStore %outColor %d_vec
               OpReturn
               OpFunctionEnd
