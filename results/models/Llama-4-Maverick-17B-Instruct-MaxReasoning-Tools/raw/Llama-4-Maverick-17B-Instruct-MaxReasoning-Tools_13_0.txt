import sys

def rolling_hash(s, length, base=257, mod=2**32):
    if length > len(s):
        return {}
    hashes = {}
    hash_value = 0
    base_pow = 1
    for i in range(length):
        hash_value = (hash_value * base + ord(s[i])) % mod
        base_pow = (base_pow * base) % mod
    hashes[hash_value] = hashes.get(hash_value, []) + [0]
    for i in range(1, len(s) - length + 1):
        hash_value = (hash_value * base - ord(s[i-1]) * base_pow + ord(s[i+length-1])) % mod
        if hash_value < 0:
            hash_value += mod
        hashes[hash_value] = hashes.get(hash_value, []) + [i]
    return hashes

def check_substring_length(strings, length):
    if length == 0:
        return True
    first_string = strings[0]
    hashes = rolling_hash(first_string, length)
    for hash_value, indices in hashes.items():
        is_common = True
        for other_string in strings[1:]:
            other_hashes = rolling_hash(other_string, length)
            if hash_value not in other_hashes:
                is_common = False
                break
        if is_common:
            return True
    return False

def longest_common_substring(strings):
    min_length = min(len(s) for s in strings)
    left, right = 0, min_length
    longest_substring = ""
    while left <= right:
        mid = (left + right) // 2
        if check_substring_length(strings, mid):
            left = mid + 1
            if mid > len(longest_substring):
                first_string = strings[0]
                hashes = rolling_hash(first_string, mid)
                for hash_value, indices in hashes.items():
                    is_common = True
                    for other_string in strings[1:]:
                        other_hashes = rolling_hash(other_string, mid)
                        if hash_value not in other_hashes:
                            is_common = False
                            break
                    if is_common:
                        longest_substring = first_string[indices[0]:indices[0]+mid]
                        break
        else:
            right = mid - 1
    return longest_substring

def main():
    try:
        N = int(sys.stdin.readline().strip())
        strings = []
        for _ in range(N):
            strings.append(sys.stdin.readline().strip())
        if not strings:
            print("")
        else:
            print(longest_common_substring(strings))
    except Exception as e:
        print("", file=sys.stderr)

if __name__ == "__main__":
    main()
