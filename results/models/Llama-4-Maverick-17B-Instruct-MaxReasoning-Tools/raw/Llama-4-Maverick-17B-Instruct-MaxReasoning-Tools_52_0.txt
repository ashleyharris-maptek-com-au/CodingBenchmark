### Step 1: Import Necessary Libraries and Initialize Module-Level Variables

```python
import numpy as np

# Module-level variables for persistence across function calls
class AutopilotState:
    def __init__(self):
        self.prev_steering = 0.0
        self.prev_throttle = 0.0
        self.prev_brake = 0.0
        self.integral_speed_error = 0.0
        self.prev_speed_error = 0.0

autopilot_state = AutopilotState()
```

### Step 2: Implement Sensor Fusion and Data Validation

```python
def validate_and_fuse_sensors(state):
    # Wheel speed sensors
    wheel_speeds = []
    for suffix in ['fl', 'fr', 'rl', 'rr']:
        if state[f'wheel_speed_{suffix}_valid']:
            wheel_speeds.append(state[f'wheel_speed_{suffix}_kmh'])
    vehicle_speed_kmh = np.median(wheel_speeds) if wheel_speeds else state.get('vehicle_speed_1_kmh', 0)

    # Gyroscopes
    yaw_rates = []
    for i in range(1, 4):
        if state[f'gyro_{i}_valid']:
            yaw_rates.append(state[f'gyro_{i}_yaw_rate_dps'])
    yaw_rate_dps = np.median(yaw_rates) if yaw_rates else 0

    # Accelerometers
    longitudinal_accels = []
    for i in range(1, 4):
        if state[f'accel_{i}_valid']:
            longitudinal_accels.append(state[f'accel_{i}_longitudinal_ms2'])
    longitudinal_accel_ms2 = np.median(longitudinal_accels) if longitudinal_accels else 0

    # GPS
    gps_speeds = []
    for i in range(1, 3):
        if state[f'gps_{i}_valid']:
            gps_speeds.append(state[f'gps_{i}_speed_kmh'])
    gps_speed_kmh = np.median(gps_speeds) if gps_speeds else vehicle_speed_kmh

    # Vision systems
    lane_offsets = []
    for i in range(1, 4):
        if state[f'vision_{i}_lane_offset_m'] is not None:
            lane_offsets.append(state[f'vision_{i}_lane_offset_m'])
    lane_offset_m = np.median(lane_offsets) if lane_offsets else 0

    return vehicle_speed_kmh, yaw_rate_dps, longitudinal_accel_ms2, gps_speed_kmh, lane_offset_m
```

### Step 3: Implement Lane Keeping, Speed Control, and Obstacle Avoidance

```python
def control_vehicle(state, dt, vehicle_speed_kmh, lane_offset_m):
    target_speed_kmh = state['target_speed_kmh']
    target_lane = state['target_lane']

    # Lane keeping
    steering = -lane_offset_m * 0.1  # Simple proportional control
    steering = np.clip(steering, -1.0, 1.0)

    # Speed control
    speed_error = target_speed_kmh - vehicle_speed_kmh
    autopilot_state.integral_speed_error += speed_error * dt
    derivative_speed_error = (speed_error - autopilot_state.prev_speed_error) / dt
    autopilot_state.prev_speed_error = speed_error

    throttle = 0.1 * speed_error + 0.01 * autopilot_state.integral_speed_error + 0.05 * derivative_speed_error
    throttle = np.clip(throttle, 0.0, 1.0)

    # Obstacle avoidance
    obstacle_detected = any(state[f'vision_{i}_obstacle_detected'] for i in range(1, 4))
    if obstacle_detected:
        brake = 1.0
        throttle = 0.0
    else:
        brake = 0.0

    return steering, throttle, brake
```

### Step 4: Implement Traction Management and Graceful Degradation

```python
def adjust_for_traction(state, steering, throttle, brake):
    estimated_mu = state['estimated_mu']
    if estimated_mu < 0.5:  # Low traction surface
        throttle *= 0.5
        steering *= 0.5  # Gentler steering inputs

    return steering, throttle, brake
```

### Step 5: Combine All Steps into the Autopilot Function

```python
def autopilot_step(state: dict, dt: float) -> dict:
    vehicle_speed_kmh, _, _, _, lane_offset_m = validate_and_fuse_sensors(state)
    steering, throttle, brake = control_vehicle(state, dt, vehicle_speed_kmh, lane_offset_m)
    steering, throttle, brake = adjust_for_traction(state, steering, throttle, brake)

    # Update module-level variables
    autopilot_state.prev_steering = steering
    autopilot_state.prev_throttle = throttle
    autopilot_state.prev_brake = brake

    return {
        'steering': steering,
        'throttle': throttle,
        'brake': brake
    }
```
