import * as fs from 'fs';

// Function to read input from stdin
function readInput(): string {
    return fs.readFileSync(0, 'utf8');
}

// Function to simulate fire spread and calculate the score
function simulateFireSpread(grid: string[][], width: number, height: number, ignitions: number[][], towns: { id: any; x: number; y: number; value: number }[], firebreaks: number[][]): number {
    // Create a copy of the grid to simulate fire spread
    const fireGrid: boolean[][] = Array.from({ length: height }, () => new Array(width).fill(false));

    // Mark ignition points as burned
    for (const [x, y] of ignitions) {
        fireGrid[y][x] = true;
    }

    // Simulate fire spread
    const directions: number[][] = [[0, 1], [0, -1], [1, 0], [-1, 0]];
    let queue: number[][] = [...ignitions];
    while (queue.length > 0) {
        const newQueue: number[][] = [];
        for (const [x, y] of queue) {
            for (const [dx, dy] of directions) {
                const nx: number = x + dx;
                const ny: number = y + dy;
                if (nx >= 0 && nx < width && ny >= 0 && ny < height && grid[ny][nx] === '.' && !fireGrid[ny][nx]) {
                    let isFirebreak: boolean = false;
                    for (const [fx, fy] of firebreaks) {
                        if (fx === nx && fy === ny) {
                            isFirebreak = true;
                            break;
                        }
                    }
                    if (!isFirebreak) {
                        fireGrid[ny][nx] = true;
                        newQueue.push([nx, ny]);
                    }
                }
            }
        }
        queue = newQueue;
    }

    // Calculate the score
    let savedValue: number = 0;
    let totalBurnedValue: number = 0;
    for (const town of towns) {
        if (fireGrid[town.y][town.x]) {
            totalBurnedValue += town.value;
        } else {
            savedValue += town.value;
        }
    }

    return totalBurnedValue > 0 ? savedValue / (savedValue + totalBurnedValue) : 1;
}

// Function to choose firebreaks
function chooseFirebreaks(data: any): number[][] {
    const width: number = data.width;
    const height: number = data.height;
    const grid: string[] = data.grid;
    const ignitions: number[][] = data.ignitions;
    const towns: { id: any; x: number; y: number; value: number }[] = data.towns;
    const cutBudget: number = data.cutBudget;

    // Convert grid to 2D array
    const grid2D: string[][] = Array.from({ length: height }, (_, y) => grid.slice(y * width, (y + 1) * width));

    // Simulate fire spread without firebreaks to identify cells that will be burned
    const fireGrid: boolean[][] = Array.from({ length: height }, () => new Array(width).fill(false));
    for (const [x, y] of ignitions) {
        fireGrid[y][x] = true;
    }
    const directions: number[][] = [[0, 1], [0, -1], [1, 0], [-1, 0]];
    let queue: number[][] = [...ignitions];
    while (queue.length > 0) {
        const newQueue: number[][] = [];
        for (const [x, y] of queue) {
            for (const [dx, dy] of directions) {
                const nx: number = x + dx;
                const ny: number = y + dy;
                if (nx >= 0 && nx < width && ny >= 0 && ny < height && grid2D[ny][nx] === '.' && !fireGrid[ny][nx]) {
                    fireGrid[ny][nx] = true;
                    newQueue.push([nx, ny]);
                }
            }
        }
        queue = newQueue;
    }

    // Identify cells that are adjacent to burned cells and are burnable
    const candidates: number[][] = [];
    for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
            if (grid2D[y][x] === '.') {
                for (const [dx, dy] of directions) {
                    const nx: number = x + dx;
                    const ny: number = y + dy;
                    if (nx >= 0 && nx < width && ny >= 0 && ny < height && fireGrid[ny][nx]) {
                        candidates.push([x, y]);
                        break;
                    }
                }
            }
        }
    }

    // Choose firebreaks
    const firebreaks: number[][] = [];
    for (let i = 0; i < cutBudget && i < candidates.length; i++) {
        firebreaks.push(candidates[i]);
    }

    return firebreaks;
}

// Main function
function main(): void {
    const input: string = readInput();
    const data = JSON.parse(input);
    const firebreaks: number[][] = chooseFirebreaks(data);
    console.log(JSON.stringify({ firebreaks }));
}

main();
