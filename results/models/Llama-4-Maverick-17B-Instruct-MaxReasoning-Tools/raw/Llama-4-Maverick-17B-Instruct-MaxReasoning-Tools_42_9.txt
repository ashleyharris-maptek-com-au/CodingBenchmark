```spirv-assembly
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %gl_out %outColor
OpExecutionMode %main InputLinesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %worldPos_in Location 0
OpDecorate %normal_in Location 1
OpDecorate %color_in Location 2
OpDecorate %outColor Location 0
OpDecorate %ubo_var DescriptorSet 0
OpDecorate %ubo_var Binding 0

%void = OpTypeVoid
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_4 = OpConstant %uint 4
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3
%float_0_025 = OpConstant %float 0.025

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3_in = OpTypeArray %v3float %uint_4
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%worldPos_in = OpVariable %ptr_in_arr_v3 Input
%normal_in = OpVariable %ptr_in_arr_v3 Input
%color_in = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float

%ubo_type = OpTypeStruct %v4float %v4float %v4float %v4float %v3float %v3float %v4float
%ptr_ubo = OpTypePointer Uniform %ubo_type
%ubo_var = OpVariable %ptr_ubo Uniform

%main = OpFunction %void None %void
%entry = OpLabel

; Read positions
%prev_pos = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%start_pos = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%end_pos = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%next_pos = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
%prev_pos_val = OpLoad %v4float %prev_pos
%start_pos_val = OpLoad %v4float %start_pos
%end_pos_val = OpLoad %v4float %end_pos
%next_pos_val = OpLoad %v4float %next_pos

; Compute tangents
%t0 = OpFSub %v4float %end_pos_val %prev_pos_val
%t1 = OpFSub %v4float %next_pos_val %start_pos_val
%t0_xy = OpVectorShuffle %v2float %t0 %t0 0 1
%t1_xy = OpVectorShuffle %v2float %t1 %t1 0 1
%t0_len = OpExtInst %float %1 Length %t0_xy
%t1_len = OpExtInst %float %1 Length %t1_xy
%t0_norm = OpFDiv %v2float %t0_xy %t0_len
%t1_norm = OpFDiv %v2float %t1_xy %t1_len

; Compute perpendiculars
%perp0_x = OpFMul %float %float_0_025 %t0_norm 1
%perp0_y = OpFMul %float %float_0_025 %t0_norm 0
%perp0 = OpCompositeConstruct %v2float %perp0_y %perp0_x
%perp1_x = OpFMul %float %float_0_025 %t1_norm 1
%perp1_y = OpFMul %float %float_0_025 %t1_norm 0
%perp1 = OpCompositeConstruct %v2float %perp1_y %perp1_x
%perp0_neg = OpFNegate %v2float %perp0
%perp1_neg = OpFNegate %v2float %perp1

; Emit vertices
%start_xy = OpVectorShuffle %v2float %start_pos_val %start_pos_val 0 1
%start_pos0 = OpCompositeConstruct %v4float %start_xy %start_pos_val 2 %start_pos_val 3
%start_pos1 = OpCompositeConstruct %v4float %start_xy %start_pos_val 2 %start_pos_val 3
%end_xy = OpVectorShuffle %v2float %end_pos_val %end_pos_val 0 1
%end_pos0 = OpCompositeConstruct %v4float %end_xy %end_pos_val 2 %end_pos_val 3
%end_pos1 = OpCompositeConstruct %v4float %end_xy %end_pos_val 2 %end_pos_val 3

%start_perp0_pos = OpFAdd %v2float %start_xy %perp0
%start_perp0_neg_pos = OpFAdd %v2float %start_xy %perp0_neg
%end_perp1_pos = OpFAdd %v2float %end_xy %perp1
%end_perp1_neg_pos = OpFAdd %v2float %end_xy %perp1_neg

%start_perp0_pos_v4 = OpCompositeConstruct %v4float %start_perp0_pos %start_pos_val 2 %start_pos_val 3
%start_perp0_neg_pos_v4 = OpCompositeConstruct %v4float %start_perp0_neg_pos %start_pos_val 2 %start_pos_val 3
%end_perp1_pos_v4 = OpCompositeConstruct %v4float %end_perp1_pos %end_pos_val 2 %end_pos_val 3
%end_perp1_neg_pos_v4 = OpCompositeConstruct %v4float %end_perp1_neg_pos %end_pos_val 2 %end_pos_val 3

%color1 = OpAccessChain %ptr_in_v3 %color_in %int_1
%color2 = OpAccessChain %ptr_in_v3 %color_in %int_2
%color1_val = OpLoad %v3float %color1
%color2_val = OpLoad %v3float %color2

OpStore %gl_out %gl_PerVertex_out %start_perp0_pos_v4
OpStore %outColor %color1_val
OpEmitVertex

OpStore %gl_out %gl_PerVertex_out %start_perp0_neg_pos_v4
OpStore %outColor %color1_val
OpEmitVertex

OpStore %gl_out %gl_PerVertex_out %end_perp1_pos_v4
OpStore %outColor %color2_val
OpEmitVertex

OpStore %gl_out %gl_PerVertex_out %end_perp1_neg_pos_v4
OpStore %outColor %color2_val
OpEmitVertex

OpEndPrimitive

OpReturn
OpFunctionEnd
```
