#version 450

layout(location = 0) in vec3 worldPos;
layout(location = 1) in vec3 normal;
layout(location = 2) in vec2 uv;
layout(location = 3) in vec3 tangent;
layout(location = 4) in vec3 color;

layout(set = 0, binding = 0) uniform UBO {
    mat4 model;
    mat4 view;
    mat4 projection;
    vec4 lightPos;
    vec4 cameraPos;
    vec4 params;
} ubo;

layout(location = 0) out vec4 outColor;

void main() {
    vec3 lightDir = normalize(ubo.lightPos.xyz - worldPos);
    vec3 viewDir = normalize(ubo.cameraPos.xyz - worldPos);
    vec3 norm = normalize(normal);

    vec2 scaledUV = uv * vec2(8.0, 16.0);
    vec2 brickCoord = fract(scaledUV);
    vec2 brickSize = vec2(1.0) - vec2(0.05, 0.1);

    float rowOffset = mod(floor(scaledUV.y), 2.0) * 0.5;
    vec2 brickCheck = step(brickCoord, brickSize);

    float brick = brickCheck.x * brickCheck.y;
    vec3 brickColor = vec3(0.7, 0.2, 0.1);
    vec3 mortarColor = vec3(0.7, 0.7, 0.7);

    vec3 finalColor = mix(mortarColor, brickColor, brick);

    float diffuse = max(dot(norm, lightDir), 0.0);
    float lambertian = 0.4 + 0.6 * diffuse;

    outColor = vec4(finalColor * lambertian, 1.0);
}
