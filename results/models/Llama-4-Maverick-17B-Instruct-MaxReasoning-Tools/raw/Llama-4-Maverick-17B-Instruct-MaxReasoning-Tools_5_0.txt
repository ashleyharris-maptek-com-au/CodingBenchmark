# Problem Understanding

The problem requires solving a Hamiltonian Path problem on 2D grids with obstacles in Rust. The task is to find a path that visits every non-obstacle cell exactly once, starting from (0, 0), and moving only to adjacent cells (up/down/left/right).

Key constraints and edge cases:

- The grid size can vary from trivial to ludicrous scale.
- The path must start at (0, 0).
- The path must visit every non-obstacle cell exactly once.
- Movement is only to adjacent cells (up/down/left/right).
- If `require_cycle` is True, the last cell must be adjacent to (0, 0).
- The input format is specific, with the first line containing the width, height, obstacle count, and `require_cycle` flag, followed by the coordinates of the obstacles.
- The output format is also specific, with either a single line containing 0 (if no path exists) or the length of the path followed by the coordinates of the path in order.

Input: width, height, obstacle count, `require_cycle` flag, and obstacle coordinates.
Output: either 0 (if no path exists) or the length of the path and the path coordinates.

Subtle requirements or implicit constraints:

- The algorithm must adapt based on grid size and obstacle density.
- The solution must complete within 5 minutes even for 2500 cell grids.
- The solution must be correct and find valid paths or correctly report impossibility.

2. Mark the obstacles on the grid.
3. Check if (0, 0) is an obstacle or if the number of non-obstacle cells is 0. If so, output 0.
4. Implement a backtracking algorithm with heuristics to find a Hamiltonian Path.
5. Use a data structure (e.g., a Vec) to keep track of the current path being explored.
6. Explore all possible next steps from the current cell, ensuring that the next cell is within the grid, not an obstacle, and not already visited in the current path.

8. Output the result in the required format.

import sys

def main():
    try:
        width, height, obstacle_count, require_cycle = map(int, input().split())
    except ValueError:
        print("Invalid input format")
        return

    grid = [[0 for _ in range(width)] for _ in range(height)]
    for _ in range(obstacle_count):
        try:
            x, y = map(int, input().split())
        except ValueError:
            print("Invalid obstacle format")
            return
        if x < 0 or x >= width or y < 0 or y >= height:
            print("Obstacle out of bounds")
            return
        grid[y][x] = 1

    if grid[0][0] == 1:
        print(0)
        return

    non_obstacle_count = sum(1 for row in grid for cell in row if cell == 0)
    if non_obstacle_count == 0:
        print(0)
        return

    path = []
    visited = [[False for _ in range(width)] for _ in range(height)]
    visited[0][0] = True
    path.append((0, 0))

    if backtrack(grid, visited, path, non_obstacle_count, require_cycle):
        if require_cycle:
            last_x, last_y = path[-1]
            if not is_adjacent(last_x, last_y, 0, 0, width, height):
                print(0)
                return
        print(len(path))
        for x, y in path:
            print(x, y)
    else:
        print(0)

def backtrack(grid, visited, path, non_obstacle_count, require_cycle):
    if len(path) == non_obstacle_count:
        return True
    x, y = path[-1]
    directions = [(0, 1), (0, -1), (1, 0), (-1, 0)]
    for dx, dy in directions:
        nx, ny = x + dx, y + dy
        if is_valid(nx, ny, grid, visited):
            visited[ny][nx] = True
            path.append((nx, ny))
            if backtrack(grid, visited, path, non_obstacle_count, require_cycle):
                return True
            path.pop()
            visited[ny][nx] = False
    return False

def is_valid(x, y, grid, visited):
    height, width = len(grid), len(grid[0])
    if x < 0 or x >= width or y < 0 or y >= height:
        return False
    if grid[y][x] == 1 or visited[y][x]:
        return False
    return True

def is_adjacent(x1, y1, x2, y2, width, height):
    if x1 < 0 or x1 >= width or y1 < 0 or y1 >= height or x2 < 0 or x2 >= width or y2 < 0 or y2 >= height:
        return False
    dx = abs(x1 - x2)
    dy = abs(y1 - y2)
    return (dx == 1 and dy == 0) or (dx == 0 and dy == 1)

if __name__ == "__main__":
    main()

use std::collections::VecDeque;
use std::io::{self, BufRead, BufReader};

fn main() {
    let stdin = io::stdin();
    let mut reader = BufReader::new(stdin.lock());

    let mut line = String::new();
    reader.read_line(&mut line).unwrap();
    let mut parts = line.trim().split_whitespace();
    let width: usize = parts.next().unwrap().parse().unwrap();
    let height: usize = parts.next().unwrap().parse().unwrap();
    let obstacle_count: usize = parts.next().unwrap().parse().unwrap();
    let require_cycle: bool = parts.next().unwrap().parse().unwrap();

    let mut grid = vec![vec![0; width]; height];
    for _ in 0..obstacle_count {
        line.clear();
        reader.read_line(&mut line).unwrap();
        let mut parts = line.trim().split_whitespace();
        let x: usize = parts.next().unwrap().parse().unwrap();
        let y: usize = parts.next().unwrap().parse().unwrap();
        grid[y][x] = 1;
    }

    if grid[0][0] == 1 {
        println!(0);
        return;
    }

    let non_obstacle_count = grid.iter().flatten().filter(|&&x| x == 0).count();
    if non_obstacle_count == 0 {
        println!(0);
        return;
    }

    let mut visited = vec![vec![false; width]; height];
    let mut path = Vec::new();
    visited[0][0] = true;
    path.push((0, 0));

    if backtrack(&grid, &mut visited, &mut path, non_obstacle_count, require_cycle) {
        if require_cycle {
            let (last_x, last_y) = *path.last().unwrap();
            if !is_adjacent(last_x as isize, last_y as isize, 0, 0, width as isize, height as isize) {
                println!(0);
                return;
            }
        }
        println!("{}", path.len());
        for &(x, y) in &path {
            println!("{} {}", x, y);
        }
    } else {
        println!(0);
    }
}

fn backtrack(grid: &Vec<Vec<i32>>, visited: &mut Vec<Vec<bool>>, path: &mut Vec<(usize, usize)>, non_obstacle_count: usize, require_cycle: bool) -> bool {
    if path.len() == non_obstacle_count {
        return true;
    }
    let (x, y) = *path.last().unwrap();
    let directions = [(0, 1), (0, -1), (1, 0), (-1, 0)];
    for &(dx, dy) in &directions {
        let nx = x as isize + dx;
        let ny = y as isize + dy;
        if is_valid(nx, ny, grid, visited) {
            let nx = nx as usize;
            let ny = ny as usize;
            visited[ny][nx] = true;
            path.push((nx, ny));
            if backtrack(grid, visited, path, non_obstacle_count, require_cycle) {
                return true;
            }
            path.pop();
            visited[ny][nx] = false;
        }
    }
    false
}

fn is_valid(x: isize, y: isize, grid: &Vec<Vec<i32>>, visited: &Vec<Vec<bool>>) -> bool {
    let height = grid.len() as isize;
    let width = grid[0].len() as isize;
    if x < 0 || x >= width || y < 0 || y >= height {
        return false;
    }
    let x = x as usize;
    let y = y as usize;
    if grid[y][x] == 1 || visited[y][x] {
        return false;
    }
    true
}

fn is_adjacent(x1: isize, y1: isize, x2: isize, y2: isize, width: isize, height: isize) -> bool {
    if x1 < 0 || x1 >= width || y1 < 0 || y1 >= height || x2 < 0 || x2 >= width || y2 < 0 || y2 >= height {
        return false;
    }
    let dx = (x1 - x2).abs();
    let dy = (y1 - y2).abs();
    (dx == 1 && dy == 0) || (dx == 0 && dy == 1)
}
