```javascript
// Import required modules
const fs = require('fs');

// Function to calculate the Euclidean distance between two points
function distance(x1, y1, x2, y2) {
    return Math.sqrt((x2 - x1) ** 2 + (y2 - y1) ** 2);
}

// Function to solve the vehicle routing problem
function solve(data) {
    const { depot, truckCount, packages } = data;
    const depotX = depot[0];
    const depotY = depot[1];

    // Sort packages by distance from the depot
    packages.sort((a, b) => {
        const distA = distance(depotX, depotY, a.x, a.y);
        const distB = distance(depotX, depotY, b.x, b.y);
        return distA - distB;
    });

    // Assign packages to routes
    const routes = Array(truckCount).fill().map(() => []);
    for (let i = 0; i < packages.length; i++) {
        const packageId = packages[i].id;
        const routeIndex = i % truckCount;
        routes[routeIndex].push(packageId);
    }

    // Filter out empty routes
    const nonEmptyRoutes = routes.filter(route => route.length > 0);

    return { routes: nonEmptyRoutes };
}

// Read input from stdin
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

// Solve the problem and print the result to stdout
const result = solve(data);
console.log(JSON.stringify(result));
```
