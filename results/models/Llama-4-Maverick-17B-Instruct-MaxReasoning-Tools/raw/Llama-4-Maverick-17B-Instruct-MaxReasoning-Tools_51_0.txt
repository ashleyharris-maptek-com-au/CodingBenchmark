# Autopilot implementation for A320-class twin-engine airliner

import numpy as np

# Module-level variables for PID controllers
altitude_integral = 0.0
airspeed_integral = 0.0
heading_integral = 0.0
previous_altitude_error = 0.0
previous_airspeed_error = 0.0
previous_heading_error = 0.0

def autopilot_step(state: dict, dt: float) -> dict:
    """
    Autopilot step function for A320-class twin-engine airliner.

    Args:
    - state (dict): Current state of the aircraft, including sensor data and autopilot targets.
    - dt (float): Timestep (~0.05s = 20 Hz).

    Returns:
    - controls (dict): Control outputs for the aircraft.
    """

    global altitude_integral, airspeed_integral, heading_integral
    global previous_altitude_error, previous_airspeed_error, previous_heading_error

    # Sensor fusion: Select valid and most reliable sensor data
    baro_alt = get_reliable_baro_alt(state)
    radar_alt = get_reliable_radar_alt(state)
    ias = get_reliable_ias(state)
    gps_alt = get_reliable_gps_alt(state)
    irs_gs = get_reliable_irs_gs(state)
    irs_heading = get_reliable_irs_heading(state)

    # Autopilot targets
    target_alt_ft = state['target_alt_ft']
    target_hdg_deg = state['target_hdg_deg']
    target_ias_kt = state['target_ias_kt']

    # PID control for altitude
    altitude_error = target_alt_ft - baro_alt
    altitude_derivative = (altitude_error - previous_altitude_error) / dt
    altitude_integral += altitude_error * dt
    previous_altitude_error = altitude_error
    elevator = 0.01 * altitude_error + 0.001 * altitude_derivative + 0.0001 * altitude_integral

    # PID control for airspeed
    airspeed_error = target_ias_kt - ias
    airspeed_derivative = (airspeed_error - previous_airspeed_error) / dt
    airspeed_integral += airspeed_error * dt
    previous_airspeed_error = airspeed_error
    throttle = 0.1 * airspeed_error + 0.01 * airspeed_derivative + 0.001 * airspeed_integral
    throttle_1 = np.clip(throttle, 0.0, 1.0)
    throttle_2 = np.clip(throttle, 0.0, 1.0)

    # PID control for heading
    heading_error = target_hdg_deg - irs_heading
    heading_derivative = (heading_error - previous_heading_error) / dt
    heading_integral += heading_error * dt
    previous_heading_error = heading_error
    aileron = 0.01 * heading_error + 0.001 * heading_derivative + 0.0001 * heading_integral

    # Rudder control for yaw damping
    rudder = -0.1 * state['gyro_1_yaw_rate_dps']

    # Flaps and gear control
    flaps = 0.0  # Retract flaps for now
    gear = 0 if irs_gs > 50 else 1  # Retract gear when ground speed is high

    # Speedbrake control
    speedbrake = 0.0  # Don't deploy speedbrake for now

    # Elevator trim control
    elevator_trim = 0.0  # Don't adjust trim for now

    controls = {
        'elevator': np.clip(elevator, -1.0, 1.0),
        'aileron': np.clip(aileron, -1.0, 1.0),
        'rudder': np.clip(rudder, -1.0, 1.0),
        'throttle_1': throttle_1,
        'throttle_2': throttle_2,
        'elevator_trim': elevator_trim,
        'flaps': flaps,
        'speedbrake': speedbrake,
        'gear': gear
    }

    return controls

def get_reliable_baro_alt(state: dict) -> float:
    # Implement baro alt sensor fusion
    baro_alt_1 = state['baro_alt_1_ft'] if state['baro_alt_1_valid'] else np.nan
    baro_alt_2 = state['baro_alt_2_ft'] if state['baro_alt_2_valid'] else np.nan
    baro_alt_3 = state['baro_alt_3_ft'] if state['baro_alt_3_valid'] else np.nan
    baro_alts = [baro_alt_1, baro_alt_2, baro_alt_3]
    baro_alts = [alt for alt in baro_alts if not np.isnan(alt)]
    return np.median(baro_alts) if baro_alts else 0.0

def get_reliable_radar_alt(state: dict) -> float:
    # Implement radar alt sensor fusion
    radar_alt_1 = state['radar_alt_1_ft'] if state['radar_alt_1_valid'] else np.nan

    # Implement IAS sensor fusion
    ias_1 = state['ias_1_kt'] if state['ias_1_valid'] else np.nan
    ias_2 = state['ias_2_kt'] if state['ias_2_valid'] else np.nan
    ias_3 = state['ias_3_kt'] if state['ias_3_valid'] else np.nan
    ias_values = [ias_1, ias_2, ias_3]
    ias_values = [ias for ias in ias_values if not np.isnan(ias)]
    return np.median(ias_values) if ias_values else 0.0
