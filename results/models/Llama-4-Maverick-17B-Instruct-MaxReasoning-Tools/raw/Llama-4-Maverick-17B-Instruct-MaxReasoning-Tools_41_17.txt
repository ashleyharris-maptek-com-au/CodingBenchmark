; SPIR-V
; Version: 1.0
OpCapability Shader
OpMemoryModel Logical GLSL450

OpEntryPoint Fragment %main "main" %worldPos %normal %uv %tangent %color %outColor %ubo
OpExecutionMode %main OriginUpperLeft

OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %uv Location 2
OpDecorate %tangent Location 3
OpDecorate %color Location 4
OpDecorate %outColor Location 0
OpDecorate %ubo Block
OpDecorate %_ptr_Uniform_ubo DescriptorSet 0
OpDecorate %_ptr_Uniform_ubo Binding 0
OpMemberDecorate %ubo 0 Offset 0
OpMemberDecorate %ubo 1 Offset 64
OpMemberDecorate %ubo 2 Offset 128
OpMemberDecorate %ubo 3 Offset 192
OpMemberDecorate %ubo 4 Offset 208
OpMemberDecorate %ubo 5 Offset 224

%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%mat4 = OpTypeMatrix %v4float 4
%ubo = OpTypeStruct %mat4 %mat4 %mat4 %v4float %v4float %v4float
%_ptr_Uniform_ubo = OpTypePointer Uniform %ubo
%_ptr_Input_v3float = OpTypePointer Input %v3float
%_ptr_Input_v2float = OpTypePointer Input %v2float
%_ptr_Output_v4float = OpTypePointer Output %v4float
%_ptr_Uniform_v4float = OpTypePointer Uniform %v4float

%worldPos = OpVariable %_ptr_Input_v3float Input
%normal = OpVariable %_ptr_Input_v3float Input
%uv = OpVariable %_ptr_Input_v2float Input
%tangent = OpVariable %_ptr_Input_v3float Input
%color = OpVariable %_ptr_Input_v3float Input
%outColor = OpVariable %_ptr_Output_v4float Output
%ubo = OpVariable %_ptr_Uniform_ubo Uniform

%main = OpFunction %void None %void
%entry = OpLabel

%uv_val = OpLoad %v2float %uv
%uv_scaled = OpFMul %v2float %uv_val %v2float %float_128 %float_8
%row_idx = OpFDiv %float %uv_scaled %float_1
%row_idx_int = OpConvertFToS %int %row_idx
%offset = OpSelect %float %row_idx_int %int_1 %float_0.5 %float_0
%brick_uv = OpFAdd %v2float %uv_scaled %offset
%brick_uv_int = OpConvertFToS %v2float %brick_uv
%in_mortar = OpFOrdLessThan %v2float %brick_uv %v2float %float_0 %float_0
%in_mortar_x = OpCompositeExtract %float %in_mortar 0
%in_mortar_y = OpCompositeExtract %float %in_mortar 1
%in_mortar_x_or_y = OpLogicalOr %bool %in_mortar_x %in_mortar_y
%not_in_mortar = OpLogicalNot %bool %in_mortar_x_or_y
%brick_color = OpCompositeConstruct %v3float %float_0.7 %float_0.2 %float_0.1
%mortar_color = OpCompositeConstruct %v3float %float_0.7 %float_0.7 %float_0.7
%final_color = OpSelect %v3float %not_in_mortar %brick_color %mortar_color

%normal_val = OpLoad %v3float %normal
%normal_len = OpExtInst %float %glsl Length %normal_val
%normal_n = OpFDiv %v3float %normal_val %normal_len
%lightPos = OpAccessChain %_ptr_Uniform_v4float %ubo %int_3
%lightPos_val = OpLoad %v4float %lightPos
%lightDir = OpVectorTimesScalar %v3float %lightPos_val %float_1
%lightDir = OpFSub %v3float %lightDir %worldPos
%lightDir_len = OpExtInst %float %glsl Length %lightDir
%lightDir_n = OpFDiv %v3float %lightDir %lightDir_len
%dot_product = OpDot %float %normal_n %lightDir_n
%dot_product_clamped = OpExtInst %float %glsl FClamp %dot_product %float_0 %float_1
%lambertian = OpFAdd %float %float_0.4 %dot_product_clamped %float_0.6
%final_color_lambertian = OpVectorTimesScalar %v3float %final_color %lambertian
%final_color_rgba = OpCompositeConstruct %v4float %final_color_lambertian %float_1

OpStore %outColor %final_color_rgba
OpReturn
OpFunctionEnd

%float_0 = OpConstant %float 0
%float_1 = OpConstant %float 1
%float_8 = OpConstant %float 8
%float_128 = OpConstant %float 16
%float_0.5 = OpConstant %float 0.5
%float_0.7 = OpConstant %float 0.7
%float_0.2 = OpConstant %float 0.2
%float_0.1 = OpConstant %float 0.1
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_3 = OpConstant %int 3
%bool = OpTypeBool
%glsl = OpExtInstImport "GLSL.std.450"
