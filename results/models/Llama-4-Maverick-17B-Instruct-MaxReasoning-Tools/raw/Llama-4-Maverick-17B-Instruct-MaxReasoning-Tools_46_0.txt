Here's the GLSL compute shader for the gravitational N-body simulation:

```glsl
#version 450

layout(local_size_x = 256) in;

// Each body is stored as two consecutive vec4s:
//   vec4(position.xyz, mass)
//   vec4(velocity.xyz, 0.0)
// So body i starts at index i*2 in the buffer.

layout(set = 0, binding = 0) readonly buffer InputBodies {
    vec4 inData[];   // length = N * 2
};

layout(set = 0, binding = 1) buffer OutputBodies {
    vec4 outData[];  // length = N * 2
};

layout(set = 0, binding = 2) uniform Params {
    uvec4 counts;    // counts.x = N (number of bodies)
    vec4 physics;    // physics.x = dt, physics.y = softening, physics.z = G
};

void main() {
    uint N = counts.x;
    uint idx = gl_GlobalInvocationID.x;

    // Guard against out-of-bounds invocation IDs
    if (idx >= N) {
        return;
    }

    // Load the position, velocity, and mass of the current body
    vec4 pos_mass = inData[idx * 2];
    vec4 vel_zero = inData[idx * 2 + 1];
    vec3 pos = pos_mass.xyz;
    vec3 vel = vel_zero.xyz;
    float mass = pos_mass.w;

    // Initialize acceleration to zero
    vec3 acc = vec3(0.0);

    // Compute the acceleration due to all other bodies
    float G = physics.z;
    float softening = physics.y;
    float softening_squared = softening * softening;
    for (uint j = 0; j < N; ++j) {
        // Skip self-interaction
        if (j == idx) {
            continue;
        }

        // Load the position and mass of the other body
        vec4 other_pos_mass = inData[j * 2];
        vec3 other_pos = other_pos_mass.xyz;
        float other_mass = other_pos_mass.w;

        // Compute the difference vector and its length squared
        vec3 diff = other_pos - pos;
        float diff_squared = dot(diff, diff);

        // Compute the acceleration contribution
        float denominator = pow(diff_squared + softening_squared, 1.5);
        acc += G * other_mass * diff / denominator;
    }

    // Update velocity and position using leapfrog/Euler integration
    float dt = physics.x;
    vel += acc * dt;
    pos += vel * dt;

    // Store the updated position, velocity, and mass
    outData[idx * 2] = vec4(pos, mass);
    outData[idx * 2 + 1] = vec4(vel, 0.0);
}
```

This shader performs one timestep of the gravitational N-body simulation using the leapfrog/Euler integration method. It computes the acceleration of each body due to all other bodies, updates the velocity and position accordingly, and stores the results in the output buffer. The shader is designed to be dispatched with `ceil(N/256)` workgroups in the X dimension, where `N` is the number of bodies. The `Params` uniform buffer provides the necessary parameters, including the number of bodies, timestep, softening parameter, and gravitational constant.
