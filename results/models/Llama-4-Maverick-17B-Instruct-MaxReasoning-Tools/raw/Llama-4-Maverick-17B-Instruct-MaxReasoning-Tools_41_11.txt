```spirv
; SPIR-V
; Version: 1.0

OpCapability Shader
OpMemoryModel Logical GLSL450

OpEntryPoint Fragment %main "main" %worldPos %normal %uv %tangent %color %outColor %ubo
OpExecutionMode %main OriginUpperLeft

OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %uv Location 2
OpDecorate %tangent Location 3
OpDecorate %color Location 4
OpDecorate %outColor Location 0
OpDecorate %ubo Block
OpDecorate %_ubo DescriptorSet 0
OpDecorate %_ubo Binding 0
OpMemberDecorate %ubo 0 Offset 0
OpMemberDecorate %ubo 1 Offset 64
OpMemberDecorate %ubo 2 Offset 128
OpMemberDecorate %ubo 3 Offset 192
OpMemberDecorate %ubo 4 Offset 208
OpMemberDecorate %ubo 5 Offset 224

%void = OpTypeVoid
%voidfn = OpTypeFunction %void
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%v2float = OpTypeVector %float 2
%mat4 = OpTypeMatrix %v4float 4
%ubo = OpTypeStruct %mat4 %mat4 %mat4 %v4float %v4float %v4float
%_ubo = OpTypePointer Uniform %ubo
%ubo_ptr = OpVariable %_ubo Uniform

%worldPos = OpVariable %_v3float Input
%normal = OpVariable %_v3float Input
%uv = OpVariable %_v2float Input
%tangent = OpVariable %_v3float Input
%color = OpVariable %_v3float Input
%outColor = OpVariable %_v4float Output

%_v3float = OpTypePointer Input %v3float
%_v2float = OpTypePointer Input %v2float
%_v4float = OpTypePointer Output %v4float

%float_0 = OpConstant %float 0
%float_1 = OpConstant %float 1
%float_0_1 = OpConstant %float 0.1
%float_0_3 = OpConstant %float 0.3
%float_0_5 = OpConstant %float 0.5
%float_0_7 = OpConstant %float 0.7
%float_0_8 = OpConstant %float 0.8
%float_3 = OpConstant %float 3
%float_5 = OpConstant %float 5
%v3float_0_1_0_3 = OpConstantComposite %v3float %float_0_1 %float_0_1 %float_0_3
%v3float_1_0_8_0 = OpConstantComposite %v3float %float_1 %float_0_8 %float_0
%int_0 = OpConstant %int 3
%int_1 = OpConstant %int 1

%main = OpFunction %void None %voidfn
%entry = OpLabel

%uv_val = OpLoad %v2float %uv
%uv_scaled = OpFMul %v2float %uv_val %float_6 = OpConstant %float 6
%uv_frac = OpFract %v2float %uv_scaled
%uv_dist = OpExtInst %float %glsl450 Distance %uv_frac %v2float %float_0_5 %float_0_5 = OpConstantComposite %v2float %float_0_5 %float_0_5
%uv_check = OpFOrdLessThan %bool = OpTypeBool %v2bool = OpTypeVector %bool 2 %uv_dist %float_0_3
%uv_dot = OpCompositeExtract %bool %uv_check 0

%normal_val = OpLoad %v3float %normal
%normal_n = OpExtInst %v3float %glsl450 Normalize %normal_val

%ubo_val = OpLoad %ubo %ubo_ptr
%lightPos_val = OpCompositeExtract %v4float %ubo_val 3
%lightPos_xyz = OpVectorShuffle %v3float %lightPos_val %lightPos_val 0 1 2
%worldPos_val = OpLoad %v3float %worldPos
%lightDir = OpFSub %v3float %lightPos_xyz %worldPos_val
%lightDir_n = OpExtInst %v3float %glsl450 Normalize %lightDir
%diffuse = OpFMax %float %float_0 %float = OpTypeFloat 32 = OpDot %v3float %normal_n %lightDir_n
%shade = OpFAdd %float %float_0_3 %float = OpFMul %float %float_0_7 %diffuse

%color_uv = OpSelect %v3float %uv_dot %v3float_1_0_8_0 %v3float_0_1_0_3
%color_shaded = OpVectorTimesScalar %v3float %color_uv %shade
%outColor_val = OpCompositeConstruct %v4float %color_shaded %float_1
OpStore %outColor %outColor_val

OpReturn
OpFunctionEnd
```
