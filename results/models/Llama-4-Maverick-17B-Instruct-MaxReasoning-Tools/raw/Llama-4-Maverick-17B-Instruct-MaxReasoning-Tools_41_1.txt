; SPIR-V
; Version: 1.0
               OpCapability Shader
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Fragment %main "main" %normal_in %outColor
               OpExecutionMode %main OriginUpperLeft

               OpSource GLSL 450
               OpName %main "main"
               OpName %normal_in "normal"
               OpName %outColor "outColor"

               OpDecorate %normal_in Location 1
               OpDecorate %outColor Location 0

               OpMemberDecorate %ubo 0 Offset 0
               OpMemberDecorate %ubo 1 Offset 64
               OpMemberDecorate %ubo 2 Offset 128
               OpMemberDecorate %ubo 3 Offset 192
               OpMemberDecorate %ubo 4 Offset 208
               OpMemberDecorate %ubo 5 Offset 224
               OpDecorate %ubo Block
               OpDecorate %ub "DescriptorSet 0"
               OpDecorate %ub "Binding 0"

       %void = OpTypeVoid
          %3 = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v3float = OpTypeVector %float 3
%_ptr_Input_v3float = OpTypePointer Input %v3float
   %normal_in = OpVariable %_ptr_Input_v3float Input
    %v4float = OpTypeVector %float 4
%_ptr_Output_v4float = OpTypePointer Output %v4float
    %outColor = OpVariable %_ptr_Output_v4float Output

       %ubo = OpTypeStruct %v4float %v4float %v4float %v4float %v4float %v4float
%_ptr_Uniform_ubo = OpTypePointer Uniform %ubo
          %ub = OpVariable %_ptr_Uniform_ubo Uniform

       %main = OpFunction %void None %3
          %5 = OpLabel
   %normal = OpLoad %v3float %normal_in
   %normal = OpExtInst %v3float %1 Normalize %normal
   %r = OpCompositeExtract %float %normal 0
   %g = OpCompositeExtract %float %normal 1
   %b = OpCompositeExtract %float %normal 2
   %r = OpFAdd %float %r 1.0
   %g = OpFAdd %float %g 1.0
   %b = OpFAdd %float %b 1.0
   %r = OpFMul %float %r 0.5
   %g = OpFMul %float %g 0.5
   %b = OpFMul %float %b 0.5
   %color = OpCompositeConstruct %v4float %r %g %b %float_1
               OpStore %outColor %color
               OpReturn
               OpFunctionEnd
