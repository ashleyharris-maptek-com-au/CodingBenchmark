```spirv
OpCapability Geometry
OpMemoryModel Logical GLSL450

OpEntryPoint Geometry %main "main" %gl_in %worldPos_in %normal_in %color_in %gl_out %outColor

OpExecutionMode %main InputPoints
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 18
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

OpDecorate %worldPos_in Location 0
OpDecorate %normal_in Location 1
OpDecorate %color_in Location 2
OpDecorate %outColor Location 0

OpDecorate %ubo Binding 0
OpDecorate %ubo DescriptorSet 0

%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_1 = OpConstant %uint 1
%uint_0 = OpConstant %uint 0
%int_0 = OpConstant %int 0
%float_0 = OpConstant %float 0
%float_1 = OpConstant %float 1
%float_0_04 = OpConstant %float 0.04
%float_0_9 = OpConstant %float 0.9

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_1
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3float = OpTypeArray %v3float %uint_1
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3float
%worldPos_in = OpVariable %ptr_in_arr_v3 Input
%normal_in = OpVariable %ptr_in_arr_v3 Input
%color_in = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float

%ubo_type = OpTypeStruct %v4float %v4float %v4float %v4float %v3float %v3float %v4float
%ptr_ubo = OpTypePointer Uniform %ubo_type
%ubo = OpVariable %ptr_ubo Uniform

%main = OpFunction %void None %void
%entry = OpLabel

; Load point position and color
%pos = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%pos_val = OpLoad %v4float %pos
%color = OpAccessChain %ptr_in_v3 %color_in %int_0
%color_val = OpLoad %v3float %color

; Define radius
%radius = %float_0_04

; Center vertex
%center_pos = OpVectorExtractDynamic %v3float %pos_val %uint_0
%out_pos = OpCompositeConstruct %v4float %center_pos %float_1
OpStore %gl_out %out_pos
OpStore %outColor %color_val
OpEmitVertex

; Emit 6 edge vertices
%angle_step = OpFDiv %float %float_6_2831853 %float_6
%angle = OpFDiv %float %float_0 %float_6
%loop_label = OpLabel
OpBranch %loop_entry

%loop_entry = OpLabel
%angle_phi = OpPhi %float %angle %loop_label %angle_next %loop_body
%i_phi = OpPhi %int %int_0 %loop_label %i_next %loop_body

%loop_cond = OpSLessThan %int %i_phi %int_6
OpLoopMerge %loop_merge %loop_body None

OpBranchConditional %loop_cond %loop_body %loop_merge

%loop_body = OpLabel
; Calculate edge position
%sin = OpExtInst %float %1 Sin %angle_phi
%cos = OpExtInst %float %1 Cos %angle_phi
%edge_x = OpFMul %float %radius %cos
%edge_y = OpFMul %float %radius %sin
%edge_z = OpCompositeExtract %float %center_pos 2
%edge_pos = OpCompositeConstruct %v3float %edge_x %edge_y %edge_z
%edge_pos_v4 = OpCompositeConstruct %v4float %edge_pos %float_1
OpStore %gl_out %edge_pos_v4
%darker_color = OpFMul %v3float %color_val %float_0_9
OpStore %outColor %darker_color
OpEmitVertex

; Update angle and counter
%angle_next = OpFAdd %float %angle_phi %angle_step
%i_next = OpIAdd %int %i_phi %int_1
OpBranch %loop_entry

%loop_merge = OpLabel
OpEndPrimitive

OpReturn
OpFunctionEnd
```
