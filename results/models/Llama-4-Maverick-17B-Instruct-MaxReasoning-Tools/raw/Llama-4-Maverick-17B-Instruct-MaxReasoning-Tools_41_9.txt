```spirv
; SPIR-V
; Version: 1.0

OpCapability Shader
OpMemoryModel Logical GLSL450

OpEntryPoint Fragment %main "main" %worldPos %normal %uv %tangent %color %outColor %ubo
OpExecutionMode %main OriginUpperLeft

OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %uv Location 2
OpDecorate %tangent Location 3
OpDecorate %color Location 4
OpDecorate %outColor Location 0
OpDecorate %ubo Block
OpDecorate %ubo_var DescriptorSet 0
OpDecorate %ubo_var Binding 0
OpMemberDecorate %ubo 0 Offset 0
OpMemberDecorate %ubo 1 Offset 64
OpMemberDecorate %ubo 2 Offset 128
OpMemberDecorate %ubo 3 Offset 192
OpMemberDecorate %ubo 4 Offset 208
OpMemberDecorate %ubo 5 Offset 224

%void = OpTypeVoid
%void_fn = OpTypeFunction %void

%float = OpTypeFloat 32
%vec2 = OpTypeVector %float 2
%vec3 = OpTypeVector %float 3
%vec4 = OpTypeVector %float 4
%mat4 = OpTypeMatrix %vec4 4

%ubo = OpTypeStruct %mat4 %mat4 %mat4 %vec4 %vec4 %vec4
%ubo_ptr = OpTypePointer Uniform %ubo
%ubo_var = OpVariable %ubo_ptr Uniform

%worldPos_ptr = OpTypePointer Input %vec3
%worldPos = OpVariable %worldPos_ptr Input

%normal_ptr = OpTypePointer Input %vec3
%normal = OpVariable %normal_ptr Input

%uv_ptr = OpTypePointer Input %vec2
%uv = OpVariable %uv_ptr Input

%tangent_ptr = OpTypePointer Input %vec3
%tangent = OpVariable %tangent_ptr Input

%color_ptr = OpTypePointer Input %vec3
%color = OpVariable %color_ptr Input

%outColor_ptr = OpTypePointer Output %vec4
%outColor = OpVariable %outColor_ptr Output

%main = OpFunction %void None %void_fn
%entry = OpLabel

%normal_val = OpLoad %vec3 %normal
%normal_norm = OpExtInst %vec3 %1 Normalize %normal_val

%uv_val = OpLoad %vec2 %uv
%uv_scaled = OpFMul %vec2 %uv_val %float_8
%uv_int = OpConvertFToS %v2int %uv_scaled
%uv_parity = OpBitwiseAnd %v2int %uv_int %int_1
%parity = OpIEqual %bool %uv_parity %int_0
%color_idx = OpSelect %float %parity %float_0_9 %float_0_2
%color_vec = OpCompositeConstruct %vec3 %color_idx %color_idx %color_idx

%ubo_val = OpLoad %ubo %ubo_var
%lightPos = OpCompositeExtract %vec4 %ubo_val 3
%lightPos_vec3 = OpVectorExtractDynamic %vec3 %lightPos %uint_0
%worldPos_val = OpLoad %vec3 %worldPos
%light_dir = OpFSub %vec3 %lightPos_vec3 %worldPos_val
%light_dist = OpExtInst %float %1 Length %light_dir
%light_dir_norm = OpFDiv %vec3 %light_dir %float_light_dist
%normal_dot_light = OpDot %float %normal_norm %light_dir_norm
%diffuse = OpFMax %float %normal_dot_light %float_0
%lambertian = OpFAdd %float %float_0_5 %float_0_5 %diffuse
%final_color = OpFMul %vec3 %color_vec %lambertian

%outColor_val = OpCompositeConstruct %vec4 %final_color %float_1
OpStore %outColor %outColor_val

OpReturn
OpFunctionEnd

%1 = OpExtInstImport "GLSL.std.450"

%float_0 = OpConstant %float 0
%float_1 = OpConstant %float 1
%float_8 = OpConstant %float 8
%float_0_5 = OpConstant %float 0.5
%float_0_9 = OpConstant %float 0.9
%float_0_2 = OpConstant %float 0.2

%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1

%v2int = OpTypeVector %int 2
%bool = OpTypeBool
%uint_0 = OpConstant %uint 0
```
