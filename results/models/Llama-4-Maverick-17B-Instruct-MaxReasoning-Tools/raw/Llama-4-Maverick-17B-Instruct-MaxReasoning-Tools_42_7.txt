```spirv
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos_in %normal_in %color_in %gl_out %outColor
OpExecutionMode %main Triangles
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 3
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %worldPos_in Location 0
OpDecorate %normal_in Location 1
OpDecorate %color_in Location 2
OpDecorate %outColor Location 0
OpDecorate %ubo Binding 0
OpDecorate %ubo DescriptorSet 0

%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%uint_3 = OpConstant %uint 3
%int = OpTypeInt 32 1
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%float_0_08 = OpConstant %float 0.08
%float_1 = OpConstant %float 1
%float_0 = OpConstant %float 0

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_3
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3float = OpTypeArray %v3float %uint_3
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3float
%worldPos_in = OpVariable %ptr_in_arr_v3 Input
%normal_in = OpVariable %ptr_in_arr_v3 Input
%color_in = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float

%mat4v4float = OpTypeMatrix %v4float 4
%ubo_type = OpTypeStruct %mat4v4float %mat4v4float %mat4v4float %v3float %v3float %v3float
%ptr_ubo_ubo_type = OpTypePointer Uniform %ubo_type
%ubo = OpVariable %ptr_ubo_ubo_type Uniform

%main = OpFunction %void None %void
%entry = OpLabel

; Load positions
%pos0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%pos1_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%pos2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%pos0 = OpLoad %v4float %pos0_ptr
%pos1 = OpLoad %v4float %pos1_ptr
%pos2 = OpLoad %v4float %pos2_ptr

; Compute face center
%center = OpFAdd %v4float %pos0 %pos1
%center = OpFAdd %v4float %center %pos2
%center = OpFMul %v4float %center %float_0_333333 ; 1/3

; Compute face normal
%p0 = OpVectorExtractDynamic %v3float %pos0 %int_0
%p1 = OpVectorExtractDynamic %v3float %pos1 %int_0
%p2 = OpVectorExtractDynamic %v3float %pos2 %int_0
%e1 = OpFSub %v3float %p1 %p0
%e2 = OpFSub %v3float %p2 %p0
%e1x = OpCompositeExtract %float %e1 0
%e1y = OpCompositeExtract %float %e1 1
%e1z = OpCompositeExtract %float %e1 2
%e2x = OpCompositeExtract %float %e2 0
%e2y = OpCompositeExtract %float %e2 1
%e2z = OpCompositeExtract %float %e2 2
%nx = OpFMul %float %e1y %e2z
%ny = OpFMul %float %e1z %e2x
%nz = OpFMul %float %e1x %e2y
%nx = OpFSub %float %nx %ny
%ny = OpFSub %float %ny %nz
%nz = OpFSub %float %nz %nx
%N = OpCompositeConstruct %v3float %nx %ny %nz
%N_len = OpExtInst %float %1 Length %N
%N_invlen = OpFDiv %float %float_1 %N_len
%N = OpVectorTimesScalar %v3float %N %N_invlen

; Compute offset
%offset = OpVectorTimesScalar %v3float %N %float_0_08

; Emit 3 vertices
%color0_ptr = OpAccessChain %ptr_in_v3 %color_in %int_0
%color1_ptr = OpAccessChain %ptr_in_v3 %color_in %int_1
%color2_ptr = OpAccessChain %ptr_in_v3 %color_in %int_2
%color0 = OpLoad %v3float %color0_ptr
%color1 = OpLoad %v3float %color1_ptr
%color2 = OpLoad %v3float %color2_ptr

%new_pos0 = OpFAdd %v4float %pos0 %offset_x0
%new_pos1 = OpFAdd %v4float %pos1 %offset_x1
%new_pos2 = OpFAdd %v4float %pos2 %offset_x2

%offset_x0 = OpVectorExtractDynamic %v4float %offset_v4 %int_0
%offset_x1 = OpVectorExtractDynamic %v4float %offset_v4 %int_0
%offset_x2 = OpVectorExtractDynamic %v4float %offset_v4 %int_0
%offset_v4 = OpCompositeConstruct %v4float %offset %float_0

%gl_pos_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_pos_ptr %new_pos0
OpStore %outColor %color0
OpEmitVertex

OpStore %gl_pos_ptr %new_pos1
OpStore %outColor %color1
OpEmitVertex

OpStore %gl_pos_ptr %new_pos2
OpStore %outColor %color2
OpEmitVertex

OpEndPrimitive

OpReturn
OpFunctionEnd
```
