```spirv
OpCapability Geometry
OpMemoryModel Logical GLSL450

OpEntryPoint Geometry %main "main" %gl_in %worldPos_in %normal_in %color_in %gl_out %outColor

OpExecutionMode %main InputLinesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 10
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

OpDecorate %worldPos_in Location 0
OpDecorate %normal_in Location 1
OpDecorate %color_in Location 2
OpDecorate %outColor Location 0

OpDecorate %ubo Binding 0
OpDecorate %ubo DescriptorSet 0
OpMemberDecorate %ubo 0 Offset 0
OpMemberDecorate %ubo 1 Offset 64
OpMemberDecorate %ubo 2 Offset 128
OpMemberDecorate %ubo 3 Offset 192
OpMemberDecorate %ubo 4 Offset 208
OpMemberDecorate %ubo 5 Offset 224

%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_4 = OpConstant %uint 4
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3
%float_0 = OpConstant %float 0
%float_0_02 = OpConstant %float 0.02
%float_0_25 = OpConstant %float 0.25
%float_0_5 = OpConstant %float 0.5
%float_0_75 = OpConstant %float 0.75
%float_1 = OpConstant %float 1

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3float_4 = OpTypeArray %v3float %uint_4
%ptr_in_arr_v3float = OpTypePointer Input %arr_v3float_4
%worldPos_in = OpVariable %ptr_in_arr_v3float Input
%normal_in = OpVariable %ptr_in_arr_v3float Input
%color_in = OpVariable %ptr_in_arr_v3float Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

%ptr_out_v3float = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3float Output

%ubo_type = OpTypeStruct %v4float %v4float %v4float %v4float %v4float %v4float
%ptr_ubo = OpTypePointer Uniform %ubo_type
%ubo = OpVariable %ptr_ubo Uniform

%ptr_func_v3float = OpTypePointer Function %v3float
%ptr_func_float = OpTypePointer Function %float

%main = OpFunction %void None %void
%entry = OpLabel

%P0 = OpAccessChain %ptr_in_v3float %worldPos_in %int_0
%P1 = OpAccessChain %ptr_in_v3float %worldPos_in %int_1
%P2 = OpAccessChain %ptr_in_v3float %worldPos_in %int_2
%P3 = OpAccessChain %ptr_in_v3float %worldPos_in %int_3
%P0_val = OpLoad %v3float %P0
%P1_val = OpLoad %v3float %P1
%P2_val = OpLoad %v3float %P2
%P3_val = OpLoad %v3float %P3

%tangent1 = OpFSub %v3float %P2_val %P0_val
%tangent2 = OpFSub %v3float %P3_val %P1_val
%tangent1_len = OpExtInst %float %1 Length %tangent1
%tangent2_len = OpExtInst %float %1 Length %tangent2
%tangent1_norm = OpFDiv %v3float %tangent1 %tangent1_len
%tangent2_norm = OpFDiv %v3float %tangent2 %tangent2_len

%color1 = OpAccessChain %ptr_in_v3float %color_in %int_1
%color2 = OpAccessChain %ptr_in_v3float %color_in %int_2
%color1_val = OpLoad %v3float %color1
%color2_val = OpLoad %v3float %color2

%ts = OpVariable %ptr_func_float Function
OpStore %ts %float_0

%loop_label = OpLabel
OpBranch %loop_body

%loop_body = OpLabel
%t = OpLoad %float %ts
%cond = OpFOrdLessThan %bool %t %float_1
OpLoopMerge %loop_merge %loop_continue None
OpBranchConditional %cond %loop_body_true %loop_merge

%loop_body_true = OpLabel
%interp_P = OpFAdd %v3float %P1_val %t %P2_val %float_1 %t %float_-1
%interp_tangent = OpFAdd %v3float %tangent1_norm %t %float_-1 %tangent2_norm %t
%perp = OpVectorTimesScalar %v3float %interp_tangent %float_0_02
%perp = OpVectorShuffle %v4float %perp %perp 0 1 2 3
%w = OpCompositeExtract %float %perp 3
%perp = OpVectorShuffle %v3float %perp %perp 0 1 2
%interp_color = OpFAdd %v3float %color1_val %t %float_-1 %color2_val %t

%pos_plus_perp = OpFAdd %v3float %interp_P %perp
%pos_minus_perp = OpFSub %v3float %interp_P %perp

%pos_plus_perp_v4 = OpCompositeConstruct %v4float %pos_plus_perp %float_1
%pos_minus_perp_v4 = OpCompositeConstruct %v4float %pos_minus_perp %float_1

%out_pos_plus_perp = OpAccessChain %ptr_out_v4 %gl_out %int_0
%out_pos_minus_perp = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_plus_perp %pos_plus_perp_v4
OpStore %out_pos_minus_perp %pos_minus_perp_v4

OpStore %outColor %interp_color

OpEmitVertex

OpBranch %loop_continue

%loop_continue = OpLabel
%t_old = OpLoad %float %ts
%t_new = OpFAdd %float %t_old %float_0_25
OpStore %ts %t_new
OpBranch %loop_body

%loop_merge = OpLabel
OpEndPrimitive

OpReturn
OpFunctionEnd
```
