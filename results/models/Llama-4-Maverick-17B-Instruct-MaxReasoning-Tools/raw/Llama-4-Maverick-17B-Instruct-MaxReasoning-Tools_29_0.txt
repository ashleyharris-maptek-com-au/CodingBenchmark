# Maximum Clique Problem Solver in C#

This C# program solves the Maximum Clique problem for an undirected graph. It reads the graph from standard input and writes the maximum clique to standard output.

## Code

```csharp
using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;
using System.Threading;

class Program
{
    static void Main(string[] args)
    {
        // Read input
        var input = Console.In.ReadToEnd().Split(new[] { "\r\n", "\n" }, StringSplitOptions.None);
        var (numVertices, numEdges) = ParseHeader(input[0]);
        var graph = ParseGraph(input.Skip(1).Take(numEdges), numVertices);

        // Initialize timer
        var sw = Stopwatch.StartNew();

        // Find maximum clique
        var maxClique = FindMaximumClique(graph, sw);

        // Output result
        Console.WriteLine(maxClique.Count);
        Console.WriteLine(string.Join(" ", maxClique.OrderBy(v => v)));
    }

    static (int, int) ParseHeader(string header)
    {
        var parts = header.Split(' ');
        return (int.Parse(parts[0]), int.Parse(parts[1]));
    }

    static IEnumerable<(int, int)> ParseGraph(IEnumerable<string> edges, int numVertices)
    {
        foreach (var edge in edges)
        {
            var parts = edge.Split(' ');
            yield return (int.Parse(parts[0]), int.Parse(parts[1]));
        }
    }

    static HashSet<int>[] BuildGraph(IEnumerable<(int, int)> edges, int numVertices)
    {
        var graph = new HashSet<int>[numVertices];
        for (int i = 0; i < numVertices; i++)
        {
            graph[i] = new HashSet<int>();
        }

        foreach (var (u, v) in edges)
        {
            graph[u].Add(v);
            graph[v].Add(u);
        }

        return graph;
    }

    static List<int> FindMaximumClique(HashSet<int>[] graph, Stopwatch sw)
    {
        var maxClique = new List<int>();
        var candidates = Enumerable.Range(0, graph.Length).ToList();

        // Use a simple greedy algorithm for very large graphs
        if (graph.Length > 10000)
        {
            return GreedyMaximumClique(graph);
        }

        // Use a backtracking algorithm with parallel processing for smaller graphs
        var maxCliqueLock = new object();
        Parallel.ForEach(Enumerable.Range(0, graph.Length), (i) =>
        {
            var localClique = new List<int> { i };
            var localCandidates = graph[i].ToList();

            Backtrack(graph, localClique, localCandidates, sw);

            if (localClique.Count > maxClique.Count)
            {
                lock (maxCliqueLock)
                {
                    if (localClique.Count > maxClique.Count)
                    {
                        maxClique = localClique;
                    }
                }
            }
        });

        return maxClique;

        void Backtrack(HashSet<int>[] graph, List<int> clique, List<int> candidates, Stopwatch sw)
        {
            if (sw.Elapsed.TotalMinutes > 5)
            {
                // Switch to a greedy algorithm if time is running out
                var greedyClique = GreedyMaximumClique(graph);
                if (greedyClique.Count > clique.Count)
                {
                    clique.Clear();
                    clique.AddRange(greedyClique);
                }
                return;
            }

            while (candidates.Count > 0)
            {
                var v = candidates[0];
                candidates.RemoveAt(0);

                if (clique.All(u => graph[v].Contains(u)))
                {
                    clique.Add(v);
                    var newCandidates = candidates.Where(u => graph[v].Contains(u)).ToList();
                    Backtrack(graph, clique, newCandidates, sw);
                    clique.RemoveAt(clique.Count - 1);
                }
            }
        }

        List<int> GreedyMaximumClique(HashSet<int>[] graph)
        {
            var clique = new List<int>();
            var candidates = Enumerable.Range(0, graph.Length).ToList();

            while (candidates.Count > 0)
            {
                var maxDegree = -1;
                var maxVertex = -1;

                foreach (var v in candidates)
                {
                    var degree = graph[v].Count(u => candidates.Contains(u));
                    if (degree > maxDegree)
                    {
                        maxDegree = degree;
                        maxVertex = v;
                    }
                }

                if (maxDegree == clique.Count)
                {
                    clique.Add(maxVertex);
                    candidates = graph[maxVertex].Where(u => candidates.Contains(u)).ToList();
                }
                else
                {
                    break;
                }
            }

            return clique;
        }
    }
}
```

## Explanation

This program first reads the input graph from standard input. It then builds an adjacency list representation of the graph.

For smaller graphs (less than 10,000 vertices), it uses a backtracking algorithm with parallel processing to find the maximum clique. The backtracking algorithm explores all possible cliques by adding vertices one by one, and it uses a lock to update the maximum clique found so far.

If the graph is very large or if the time limit is exceeded during the backtracking algorithm, it switches to a simple greedy algorithm to find a maximal clique.

Finally, it outputs the size and vertices of the maximum clique found.

## Compilation and Execution

To compile the program, use the following command:

```bash
csc MaximumClique.cs
```

To run the program, use the following command:

```bash
MaximumClique.exe < input.txt > output.txt
```

Replace `input.txt` with the input file containing the graph, and `output.txt` with the desired output file.
