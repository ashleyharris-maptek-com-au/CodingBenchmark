Here's the C++ code using CGAL to perform the CSG Union operation:

```cpp
#include <CGAL/Exact_predicates_inexact_constructions_kernel.h>
#include <CGAL/Surface_mesh.h>
#include <CGAL/Polygon_mesh_processing/corefinement.h>
#include <CGAL/Polygon_mesh_processing/IO/polygon_mesh_io.h>
#include <iostream>
#include <vector>

typedef CGAL::Exact_predicates_inexact_constructions_kernel Kernel;
typedef Kernel::Point_3 Point_3;
typedef CGAL::Surface_mesh<Point_3> Mesh;

namespace PMP = CGAL::Polygon_mesh_processing;

int main() {
    Mesh mesh_A, mesh_B;

    // Read counts
    int VA, FA, VB, FB;
    std::cin >> VA >> FA >> VB >> FB;

    // Read mesh A vertices and faces
    std::vector<Point_3> vertices_A(VA);
    for (int i = 0; i < VA; ++i) {
        double x, y, z;
        std::cin >> x >> y >> z;
        vertices_A[i] = Point_3(x, y, z);
    }
    std::vector<std::array<int, 3>> faces_A(FA);
    for (int i = 0; i < FA; ++i) {
        std::cin >> faces_A[i][0] >> faces_A[i][1] >> faces_A[i][2];
    }
    mesh_A.clear();
    for (int i = 0; i < VA; ++i) {
        mesh_A.add_vertex(vertices_A[i]);
    }
    for (const auto& face : faces_A) {
        mesh_A.add_face(mesh_A.vertex_index(vertices_A[face[0]]), mesh_A.vertex_index(vertices_A[face[1]]), mesh_A.vertex_index(vertices_A[face[2]]));
    }

    // Read mesh B vertices and faces
    std::vector<Point_3> vertices_B(VB);
    for (int i = 0; i < VB; ++i) {
        double x, y, z;
        std::cin >> x >> y >> z;
        vertices_B[i] = Point_3(x, y, z);
    }
    std::vector<std::array<int, 3>> faces_B(FB);
    for (int i = 0; i < FB; ++i) {
        std::cin >> faces_B[i][0] >> faces_B[i][1] >> faces_B[i][2];
    }
    mesh_B.clear();
    for (int i = 0; i < VB; ++i) {
        mesh_B.add_vertex(vertices_B[i]);
    }
    for (const auto& face : faces_B) {
        mesh_B.add_face(mesh_B.vertex_index(vertices_B[face[0]]), mesh_B.vertex_index(vertices_B[face[1]]), mesh_B.vertex_index(vertices_B[face[2]]));
    }

    // Perform CSG Union
    Mesh union_mesh;
    PMP::corefine_and_compute_union(mesh_A, mesh_B, union_mesh);

    // Output union mesh
    std::cout << union_mesh.num_vertices() << " " << union_mesh.num_faces() << std::endl;
    std::vector<Point_3> union_vertices;
    for (const auto& v : union_mesh.vertices()) {
        union_vertices.push_back(union_mesh.point(v));
        std::cout << union_mesh.point(v) << std::endl;
    }
    for (const auto& f : union_mesh.faces()) {
        std::cout << union_mesh.vertex_index(union_mesh.vertex(f, 0)) << " "
                  << union_mesh.vertex_index(union_mesh.vertex(f, 1)) << " "
                  << union_mesh.vertex_index(union_mesh.vertex(f, 2)) << std::endl;
    }

    return 0;
}
```
