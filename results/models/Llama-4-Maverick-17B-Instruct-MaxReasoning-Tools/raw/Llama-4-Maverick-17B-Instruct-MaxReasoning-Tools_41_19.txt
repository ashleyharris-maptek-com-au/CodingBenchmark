```spirv
; SPIR-V
; Version: 1.0

OpCapability Shader
OpMemoryModel Logical GLSL450

OpEntryPoint Fragment %main "main" %worldPos %normal %uv %tangent %color %outColor %ubo
OpExecutionMode %main OriginUpperLeft

OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %uv Location 2
OpDecorate %tangent Location 3
OpDecorate %color Location 4
OpDecorate %outColor Location 0
OpDecorate %ubo Block
OpDecorate %_ubo DescriptorSet 0
OpDecorate %_ubo Binding 0
OpMemberDecorate %ubo 0 Offset 0
OpMemberDecorate %ubo 1 Offset 64
OpMemberDecorate %ubo 2 Offset 128
OpMemberDecorate %ubo 3 Offset 192
OpMemberDecorate %ubo 4 Offset 208
OpMemberDecorate %ubo 5 Offset 224

%void = OpTypeVoid
%voidfn = OpTypeFunction %void
%float = OpTypeFloat 32
%vec3 = OpTypeVector %float 3
%vec4 = OpTypeVector %float 4
%mat4 = OpTypeMatrix %vec4 4
%ubo = OpTypeStruct %mat4 %mat4 %mat4 %vec4 %vec4 %vec4
%_ptr_Uniform_ubo = OpTypePointer Uniform %ubo
%_ubo = OpVariable %_ptr_Uniform_ubo Uniform
%_ptr_Input_vec3 = OpTypePointer Input %vec3
%worldPos = OpVariable %_ptr_Input_vec3 Input
%normal = OpVariable %_ptr_Input_vec3 Input
%uv = OpVariable %_ptr_Input_vec2 Input
%vec2 = OpTypeVector %float 2
%tangent = OpVariable %_ptr_Input_vec3 Input
%color = OpVariable %_ptr_Input_vec3 Input
%_ptr_Output_vec4 = OpTypePointer Output %vec4
%outColor = OpVariable %_ptr_Output_vec4 Output
%float_0 = OpConstant %float 0
%float_1 = OpConstant %float 1
%float_0_1 = OpConstantComposite %vec3 %float_0_1 %float_0_05 %float_0_05
%float_32 = OpConstant %float 32
%int = OpTypeInt 32 1
%int_3 = OpConstant %int 3
%int_4 = OpConstant %int 4

%main = OpFunction %void None %voidfn
%entry = OpLabel

; Load normal and normalize
%normal_val = OpLoad %vec3 %normal
%normal_norm = OpNormalize %vec3 %normal_val

; Load light position and compute L
%ubo_val = OpLoad %ubo %_ubo
%lightPos = OpCompositeExtract %vec4 %ubo_val 3
%lightPos_vec3 = OpVectorExtractDynamic %vec3 %lightPos %int_0
%worldPos_val = OpLoad %vec3 %worldPos
%L = OpSub %vec3 %lightPos_vec3 %worldPos_val
%L_norm = OpNormalize %vec3 %L

; Compute N.L
%N_dot_L = OpDot %float %normal_norm %L_norm
%N_dot_L_max = OpFMax %float %float_0 %N_dot_L

; Load color and compute diffuse
%color_val = OpLoad %vec3 %color
%diffuse = OpVectorTimesScalar %vec3 %color_val %N_dot_L_max

; Compute specular
%cameraPos = OpCompositeExtract %vec4 %ubo_val 4
%cameraPos_vec3 = OpVectorExtractDynamic %vec3 %cameraPos %int_0
%V = OpSub %vec3 %cameraPos_vec3 %worldPos_val
%V_norm = OpNormalize %vec3 %V
%R = OpReflect %vec3 %L_norm %normal_norm
%R_dot_V = OpDot %float %R %V_norm
%R_dot_V_max = OpFMax %float %float_0 %R_dot_V
%specular_intensity = OpPow %float %R_dot_V_max %float_32
%specular = OpVectorTimesScalar %vec3 %float_1 %specular_intensity

; Sum ambient, diffuse, and specular
%ambient = OpCompositeConstruct %vec3 %float_0_1 %float_0_05 %float_0_05
%sum_rgb = OpFAdd %vec3 %ambient %diffuse
%sum_rgb = OpFAdd %vec3 %sum_rgb %specular

; Clamp sum to [0,1]
%sum_rgb_clamped = OpFClamp %vec3 %sum_rgb %float_0 %float_1

; Output color with alpha = 1.0
%outColor_val = OpCompositeConstruct %vec4 %sum_rgb_clamped %float_1
OpStore %outColor %outColor_val

OpReturn
OpFunctionEnd
```
