```
; SPIR-V
; Version: 1.0
OpCapability Shader
OpMemoryModel Logical GLSL450

OpEntryPoint Fragment %main "main" %worldPos %normal %uv %tangent %color %outColor %ubo
OpExecutionMode %main OriginUpperLeft

OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %uv Location 2
OpDecorate %tangent Location 3
OpDecorate %color Location 4
OpDecorate %outColor Location 0
OpDecorate %ubo Block
OpDecorate %_ubo DescriptorSet 0
OpDecorate %_ubo Binding 0
OpMemberDecorate %ubo 0 Offset 0
OpMemberDecorate %ubo 1 Offset 64
OpMemberDecorate %ubo 2 Offset 128
OpMemberDecorate %ubo 3 Offset 192
OpMemberDecorate %ubo 4 Offset 208
OpMemberDecorate %ubo 5 Offset 224

%void = OpTypeVoid
%voidfn = OpTypeFunction %void
%float = OpTypeFloat 32
%vec2 = OpTypeVector %float 2
%vec3 = OpTypeVector %float 3
%vec4 = OpTypeVector %float 4
%mat4 = OpTypeMatrix %vec4 4
%ubo = OpTypeStruct %mat4 %mat4 %mat4 %vec4 %vec4 %vec4
%_ubo = OpTypePointer Uniform %ubo
%ubo_ptr = OpVariable %_ubo Uniform

%worldPos_ptr = OpTypePointer Input %vec3
%worldPos = OpVariable %worldPos_ptr Input

%normal_ptr = OpTypePointer Input %vec3
%normal = OpVariable %normal_ptr Input

%uv_ptr = OpTypePointer Input %vec2
%uv = OpVariable %uv_ptr Input

%tangent_ptr = OpTypePointer Input %vec3
%tangent = OpVariable %tangent_ptr Input

%color_ptr = OpTypePointer Input %vec3
%color = OpVariable %color_ptr Input

%outColor_ptr = OpTypePointer Output %vec4
%outColor = OpVariable %outColor_ptr Output

%main = OpFunction %void None %voidfn
%entry = OpLabel

%normal_val = OpLoad %vec3 %normal
%normal_norm = OpExtInst %vec3 %1 Normalize %normal_val

%worldPos_val = OpLoad %vec3 %worldPos
%cameraPos = OpAccessChain %_ubo %ubo_ptr 4
%cameraPos_val = OpLoad %vec4 %cameraPos
%cameraPos_vec3 = OpVectorShuffle %vec3 %cameraPos_val %cameraPos_val 0 1 2
%viewDir = OpFSub %vec3 %cameraPos_vec3 %worldPos_val
%viewDir_norm = OpExtInst %vec3 %1 Normalize %viewDir

%dotNV = OpDot %float %normal_norm %viewDir_norm
%dotNV_clamped = OpFClamp %float %dotNV %float_0 %float_1
%rim = OpFSub %float %float_1 %dotNV_clamped
%rim_pow = OpExtInst %float %1 Pow %rim %float_3

%color_val = OpLoad %vec3 %color
%diffuse = OpFMul %float %float_0p5 %color_val
%diffuse_add = OpFAdd %float %float_0p3 %diffuse
%baseColor = OpVectorTimesScalar %vec3 %vec3_0p2_0p3_0p8 %diffuse_add

%rim_tint = OpVectorTimesScalar %vec3 %vec3_1_0p8_0p5 %rim_pow
%finalColor = OpFAdd %vec3 %baseColor %rim_tint
%finalColor_vec4 = OpCompositeConstruct %vec4 %finalColor %float_1

OpStore %outColor %finalColor_vec4

OpReturn
OpFunctionEnd

%1 = OpExtInstImport "GLSL.std.450"
%float_0 = OpConstant %float 0
%float_1 = OpConstant %float 1
%float_0p3 = OpConstant %float 0.3
%float_0p5 = OpConstant %float 0.5
%float_0p8 = OpConstant %float 0.8
%float_2 = OpConstant %float 2
%float_3 = OpConstant %float 3
%vec3_0p2_0p3_0p8 = OpConstantComposite %vec3 %float_0p2 %float_0p3 %float_0p8
%vec3_1_0p8_0p5 = OpConstantComposite %vec3 %float_1 %float_0p8 %float_0p5
```
