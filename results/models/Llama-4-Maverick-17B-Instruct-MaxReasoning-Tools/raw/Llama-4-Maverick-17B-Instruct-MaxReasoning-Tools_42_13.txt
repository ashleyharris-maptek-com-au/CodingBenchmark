```spirv
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %gl_out %outColor
OpExecutionMode %main InputTrianglesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 12
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %worldPos_in Location 0
OpDecorate %normal_in Location 1
OpDecorate %color_in Location 2
OpDecorate %outColor Location 0
OpDecorate %ubo Binding 0
OpDecorate %ubo DescriptorSet 0

%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%uint_6 = OpConstant %uint 6
%int = OpTypeInt 32 1
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3
%int_4 = OpConstant %int 4
%float_0p008 = OpConstant %float 0.008

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_6
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3float = OpTypeArray %v3float %uint_6
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3float
%worldPos_in = OpVariable %ptr_in_arr_v3 Input
%normal_in = OpVariable %ptr_in_arr_v3 Input
%color_in = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float

%ubo_type = OpTypeStruct %v4float %v4float %v4float %v4float %v3float %v3float %v3float
%ptr_ubo = OpTypePointer Uniform %ubo_type
%ubo = OpVariable %ptr_ubo Uniform

%main = OpFunction %void None %void
%entry = OpLabel

; Compute and emit 3 edges as thin quads
%idx0 = OpConstant %int 0
%idx2 = OpConstant %int 2
%idx4 = OpConstant %int 4

; Edge 0→2
%v0_pos = OpAccessChain %ptr_in_v4 %gl_in %idx0 %int_0
%v0_worldPos = OpAccessChain %ptr_in_v3 %worldPos_in %idx0
%v0_color = OpAccessChain %ptr_in_v3 %color_in %idx0
%v2_pos = OpAccessChain %ptr_in_v4 %gl_in %idx2 %int_0
%v2_worldPos = OpAccessChain %ptr_in_v3 %worldPos_in %idx2
%v2_color = OpAccessChain %ptr_in_v3 %color_in %idx2
%edge_color = OpFAdd %v3float %v0_color %v2_color
%edge_color = OpFMul %v3float %edge_color %float_0p5
%dir = OpFSub %v3float %v2_worldPos %v0_worldPos
%perp = OpVectorTimesScalar %v3float %dir %float_0p008
%perp = OpVectorTimesMatrix %v3float %perp %float_0p008 ; Not needed, use OpCross instead
%norm = OpLoad %v3float %normal_in %idx0
%norm = OpLoad %v3float %normal_in %idx2
%norm = OpFAdd %v3float %norm %norm
%norm = OpNormalize %v3float %norm
%perp = OpCross %v3float %dir %norm
%perp = OpNormalize %v3float %perp
%perp = OpVectorTimesScalar %v3float %perp %float_0p008
%v0_pos_offset = OpFSub %v4float %v0_pos %perp
%v0_pos_offset = OpVectorInsertDynamic %v4float %v0_pos_offset %float_1 %int_3
%v2_pos_offset = OpFSub %v4float %v2_pos %perp
%v2_pos_offset = OpVectorInsertDynamic %v4float %v2_pos_offset %float_1 %int_3
%v0_pos = OpFAdd %v4float %v0_pos %perp
%v0_pos = OpVectorInsertDynamic %v4float %v0_pos %float_1 %int_3
%v2_pos = OpFAdd %v4float %v2_pos %perp
%v2_pos = OpVectorInsertDynamic %v4float %v2_pos %float_1 %int_3

%ptr_out = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out %v0_pos_offset
OpStore %outColor %edge_color
OpEmitVertex
%ptr_out = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out %v0_pos
OpStore %outColor %edge_color
OpEmitVertex
%ptr_out = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out %v2_pos_offset
OpStore %outColor %edge_color
OpEmitVertex
%ptr_out = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out %v2_pos
OpStore %outColor %edge_color
OpEmitVertex
OpEndPrimitive

; Edge 2→4
%v2_pos = OpAccessChain %ptr_in_v4 %gl_in %idx2 %int_0
%v2_worldPos = OpAccessChain %ptr_in_v3 %worldPos_in %idx2
%v2_color = OpAccessChain %ptr_in_v3 %color_in %idx2
%v4_pos = OpAccessChain %ptr_in_v4 %gl_in %idx4 %int_0
%v4_worldPos = OpAccessChain %ptr_in_v3 %worldPos_in %idx4
%v4_color = OpAccessChain %ptr_in_v3 %color_in %idx4
%edge_color = OpFAdd %v3float %v2_color %v4_color
%edge_color = OpFMul %v3float %edge_color %float_0p5
%dir = OpFSub %v3float %v4_worldPos %v2_worldPos
%perp = OpCross %v3float %dir %norm
%perp = OpNormalize %v3float %perp
%perp = OpVectorTimesScalar %v3float %perp %float_0p008
%v2_pos_offset = OpFSub %v4float %v2_pos %perp
%v2_pos_offset = OpVectorInsertDynamic %v4float %v2_pos_offset %float_1 %int_3
%v4_pos_offset = OpFSub %v4float %v4_pos %perp
%v4_pos_offset = OpVectorInsertDynamic %v4float %v4_pos_offset %float_1 %int_3
%v2_pos = OpFAdd %v4float %v2_pos %perp
%v2_pos = OpVectorInsertDynamic %v4float %v2_pos %float_1 %int_3
%v4_pos = OpFAdd %v4float %v4_pos %perp
%v4_pos = OpVectorInsertDynamic %v4float %v4_pos %float_1 %int_3

%ptr_out = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out %v2_pos_offset
OpStore %outColor %edge_color
OpEmitVertex
%ptr_out = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out %v2_pos
OpStore %outColor %edge_color
OpEmitVertex
%ptr_out = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out %v4_pos_offset
OpStore %outColor %edge_color
OpEmitVertex
%ptr_out = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out %v4_pos
OpStore %outColor %edge_color
OpEmitVertex
OpEndPrimitive

; Edge 4→0
%v4_pos = OpAccessChain %ptr_in_v4 %gl_in %idx4 %int_0
%v4_worldPos = OpAccessChain %ptr_in_v3 %worldPos_in %idx4
%v4_color = OpAccessChain %ptr_in_v3 %color_in %idx4
%v0_pos = OpAccessChain %ptr_in_v4 %gl_in %idx0 %int_0
%v0_worldPos = OpAccessChain %ptr_in_v3 %worldPos_in %idx0
%v0_color = OpAccessChain %ptr_in_v3 %color_in %idx0
%edge_color = OpFAdd %v3float %v4_color %v0_color
%edge_color = OpFMul %v3float %edge_color %float_0p5
%dir = OpFSub %v3float %v0_worldPos %v4_worldPos
%perp = OpCross %v3float %dir %norm
%perp = OpNormalize %v3float %perp
%perp = OpVectorTimesScalar %v3float %perp %float_0p008
%v4_pos_offset = OpFSub %v4float %v4_pos %perp
%v4_pos_offset = OpVectorInsertDynamic %v4float %v4_pos_offset %float_1 %int_3
%v0_pos_offset = OpFSub %v4float %v0_pos %perp
%v0_pos_offset = OpVectorInsertDynamic %v4float %v0_pos_offset %float_1 %int_3
%v4_pos = OpFAdd %v4float %v4_pos %perp
%v4_pos = OpVectorInsertDynamic %v4float %v4_pos %float_1 %int_3
%v0_pos = OpFAdd %v4float %v0_pos %perp
%v0_pos = OpVectorInsertDynamic %v4float %v0_pos %float_1 %int_3

%ptr_out = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out %v4_pos_offset
OpStore %outColor %edge_color
OpEmitVertex
%ptr_out = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out %v4_pos
OpStore %outColor %edge_color
OpEmitVertex
%ptr_out = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out %v0_pos_offset
OpStore %outColor %edge_color
OpEmitVertex
%ptr_out = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out %v0_pos
OpStore %outColor %edge_color
OpEmitVertex
OpEndPrimitive

OpReturn
OpFunctionEnd
```
