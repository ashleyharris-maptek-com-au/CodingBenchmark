```spirv
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main Triangles
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 3
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2
OpDecorate %outColor Location 0

%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%uint_3 = OpConstant %uint 3
%int = OpTypeInt 32 1
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%float_0_7 = OpConstant %float 0.7
%float_3 = OpConstant %float 3

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in_3 = OpTypeArray %gl_PerVertex_in %uint_3
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in_3
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3float_3 = OpTypeArray %v3float %uint_3
%ptr_in_arr_v3float = OpTypePointer Input %arr_v3float_3
%worldPos = OpVariable %ptr_in_arr_v3float Input
%normal = OpVariable %ptr_in_arr_v3float Input
%color = OpVariable %ptr_in_arr_v3float Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3float = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3float Output

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3float = OpTypePointer Input %v3float

%void_fn = OpTypeFunction %void

%main = OpFunction %void None %void_fn
%entry = OpLabel

; Load vertex positions
%ptr_p0 = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%p0 = OpLoad %v4float %ptr_p0
%ptr_p1 = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%p1 = OpLoad %v4float %ptr_p1
%ptr_p2 = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%p2 = OpLoad %v4float %ptr_p2

; Compute centroid
%p0_p1 = OpFAdd %v4float %p0 %p1
%p0_p1_p2 = OpFAdd %v4float %p0_p1 %p2
%centroid = OpFDiv %v4float %p0_p1_p2 %float_3

; Shrink factor
%s = OpLoad %float %float_0_7

; Shrink vertices
%p0_c = OpFSub %v4float %p0 %centroid
%p0_c_scaled = OpVectorTimesScalar %v4float %p0_c %s
%new_p0 = OpFAdd %v4float %centroid %p0_c_scaled

%p1_c = OpFSub %v4float %p1 %centroid
%p1_c_scaled = OpVectorTimesScalar %v4float %p1_c %s
%new_p1 = OpFAdd %v4float %centroid %p1_c_scaled

%p2_c = OpFSub %v4float %p2 %centroid
%p2_c_scaled = OpVectorTimesScalar %v4float %p2_c %s
%new_p2 = OpFAdd %v4float %centroid %p2_c_scaled

; Emit vertices
%ptr_out_pos = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out_pos %new_p0
%ptr_c0 = OpAccessChain %ptr_in_v3float %color %int_0
%c0 = OpLoad %v3float %ptr_c0
OpStore %outColor %c0
OpEmitVertex

OpStore %ptr_out_pos %new_p1
%ptr_c1 = OpAccessChain %ptr_in_v3float %color %int_1
%c1 = OpLoad %v3float %ptr_c1
OpStore %outColor %c1
OpEmitVertex

OpStore %ptr_out_pos %new_p2
%ptr_c2 = OpAccessChain %ptr_in_v3float %color %int_2
%c2 = OpLoad %v3float %ptr_c2
OpStore %outColor %c2
OpEmitVertex

OpEndPrimitive

OpReturn
OpFunctionEnd
```
