```spirv
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos_in %normal_in %color_in %gl_out %outColor
OpExecutionMode %main InputLines
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 7
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %worldPos_in Location 0
OpDecorate %normal_in Location 1
OpDecorate %color_in Location 2
OpDecorate %outColor Location 0
OpDecorate %ubo Binding 0
OpDecorate %ubo DescriptorSet 0

%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%uint_2 = OpConstant %uint 2
%int = OpTypeInt 32 1
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%float_0 = OpConstant %float 0
%float_0_015 = OpConstant %float 0.015
%float_0_75 = OpConstant %float 0.75
%float_0_05 = OpConstant %float 0.05
%float_1 = OpConstant %float 1

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3float = OpTypeArray %v3float %uint_2
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3float
%worldPos_in = OpVariable %ptr_in_arr_v3 Input
%normal_in = OpVariable %ptr_in_arr_v3 Input
%color_in = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_v4 = OpTypePointer Output %v4float

%mat4x4float = OpTypeMatrix %v4float 4
%ubo_type = OpTypeStruct %mat4x4float %mat4x4float %mat4x4float %v3float %v3float %v3float
%ptr_ubo = OpTypePointer Uniform %ubo_type
%ubo = OpVariable %ptr_ubo Uniform

%main = OpFunction %void None %void
%entry = OpLabel

; Read line endpoints and colors
%idx0 = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%p0 = OpLoad %v4float %idx0
%idx1 = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%p1 = OpLoad %v4float %idx1
%idx2 = OpAccessChain %ptr_in_v3 %color_in %int_0
%c0 = OpLoad %v3float %idx2
%idx3 = OpAccessChain %ptr_in_v3 %color_in %int_1
%c1 = OpLoad %v3float %idx3

; Compute direction and perpendicular
%dir = OpFSub %v4float %p1 %p0
%dir_x = OpCompositeExtract %float %dir 0
%dir_y = OpCompositeExtract %float %dir 1
%perp_x = OpFNegate %float %dir_y
%perp_y = OpCopyObject %float %dir_x
%perp = OpCompositeConstruct %v4float %perp_x %perp_y %float_0 %float_0
%perp_len = OpExtInst %float %1 Length %perp
%perp = OpFDiv %v4float %perp %perp_len
%perp = OpVectorTimesScalar %v4float %perp %float_0_015

; Compute mid point
%mid = OpFSub %v4float %p1 %p0
%mid = OpVectorTimesScalar %v4float %mid %float_0_75
%mid = OpFAdd %v4float %p0 %mid

; Emit shaft
%shaft_p0_plus_perp = OpFAdd %v4float %p0 %perp
%idx4 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %idx4 %shaft_p0_plus_perp
OpStore %outColor %c0
OpEmitVertex

%shaft_p0_minus_perp = OpFSub %v4float %p0 %perp
%idx5 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %idx5 %shaft_p0_minus_perp
OpStore %outColor %c0
OpEmitVertex

%shaft_mid_plus_perp = OpFAdd %v4float %mid %perp
%idx6 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %idx6 %shaft_mid_plus_perp
OpStore %outColor %c1
OpEmitVertex

%shaft_mid_minus_perp = OpFSub %v4float %mid %perp
%idx7 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %idx7 %shaft_mid_minus_perp
OpStore %outColor %c1
OpEmitVertex
OpEndPrimitive

; Emit arrowhead
%arrow_mid_plus_perp = OpFAdd %v4float %mid %perp
%arrow_mid_plus_perp = OpVectorTimesScalar %v4float %arrow_mid_plus_perp %float_3_333333 ; 0.05/0.015
%idx8 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %idx8 %arrow_mid_plus_perp
OpStore %outColor %c1
OpEmitVertex

%arrow_mid_minus_perp = OpFSub %v4float %mid %perp
%arrow_mid_minus_perp = OpVectorTimesScalar %v4float %arrow_mid_minus_perp %float_3_333333 ; 0.05/0.015
%idx9 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %idx9 %arrow_mid_minus_perp
OpStore %outColor %c1
OpEmitVertex

%idx10 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %idx10 %p1
OpStore %outColor %c1
OpEmitVertex
OpEndPrimitive

OpReturn
OpFunctionEnd
```
