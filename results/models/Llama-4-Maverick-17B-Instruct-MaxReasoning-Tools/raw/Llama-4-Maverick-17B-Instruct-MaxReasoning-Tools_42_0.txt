```spirv
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %color_in %gl_out %outColor
OpExecutionMode %main InputPoints
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %color_in Location 0
OpDecorate %outColor Location 0

%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%uint_1 = OpConstant %uint 1
%int = OpTypeInt 32 1
%int_0 = OpConstant %int 0
%float_0 = OpConstant %float 0
%float_004 = OpConstant %float 0.04
%float_1 = OpConstant %float 1

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_1
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3float = OpTypeArray %v3float %uint_1
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3float
%color_in = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_v4 = OpTypePointer Output %v4float

%main = OpFunction %void None %void
%entry = OpLabel

; Read gl_in[0].gl_Position and color[0]
%ptr_pos = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%pos = OpLoad %v4float %ptr_pos
%ptr_color = OpAccessChain %ptr_in_v3 %color_in %int_0
%color = OpLoad %v3float %ptr_color

; Store color for emitted vertices
OpStore %outColor %color

; Define half-size
%h = OpLoad %float %float_004

; Extract x and y from input position
%x = OpCompositeExtract %float %pos 0
%y = OpCompositeExtract %float %pos 1

; Emit 4 vertices
; Vertex 1: (x-h, y-h)
%pos1_x = OpFSub %float %x %h
%pos1_y = OpFSub %float %y %h
%pos1 = OpCompositeConstruct %v4float %pos1_x %pos1_y %float_0 %float_1
%ptr_pos_out = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_pos_out %pos1
OpEmitVertex

; Vertex 2: (x+h, y-h)
%pos2_x = OpFAdd %float %x %h
%pos2_y = OpFSub %float %y %h
%pos2 = OpCompositeConstruct %v4float %pos2_x %pos2_y %float_0 %float_1
OpStore %ptr_pos_out %pos2
OpEmitVertex

; Vertex 3: (x-h, y+h)
%pos3_x = OpFSub %float %x %h
%pos3_y = OpFAdd %float %y %h
%pos3 = OpCompositeConstruct %v4float %pos3_x %pos3_y %float_0 %float_1
OpStore %ptr_pos_out %pos3
OpEmitVertex

; Vertex 4: (x+h, y+h)
%pos4_x = OpFAdd %float %x %h
%pos4_y = OpFAdd %float %y %h
%pos4 = OpCompositeConstruct %v4float %pos4_x %pos4_y %float_0 %float_1
OpStore %ptr_pos_out %pos4
OpEmitVertex

OpEndPrimitive

OpReturn
OpFunctionEnd
```
