```spirv
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %gl_out %outColor
OpExecutionMode %main InputLines
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %worldPos_in Location 0
OpDecorate %normal_in Location 1
OpDecorate %color_in Location 2
OpDecorate %outColor Location 0
OpDecorate %ubo Binding 0
OpDecorate %ubo DescriptorSet 0

%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%uint_2 = OpConstant %uint 2
%int = OpTypeInt 32 1
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%float_004 = OpConstant %float 0.04
%float_0005 = OpConstant %float 0.005

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3float = OpTypeArray %v3float %uint_2
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3float
%worldPos_in = OpVariable %ptr_in_arr_v3 Input
%normal_in = OpVariable %ptr_in_arr_v3 Input
%color_in = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_v4 = OpTypePointer Output %v4float

%ubo_type = OpTypeStruct %v4float %v4float %v4float %v4float %v4float %v4float %v4float
%ptr_ubo = OpTypePointer Uniform %ubo_type
%ubo = OpVariable %ptr_ubo Uniform

%main = OpFunction %void None %void
%entry = OpLabel

; Load line endpoints and colors
%idx0 = OpAccessChain %ptr_in_v3 %worldPos_in %int_0
%p0 = OpLoad %v3float %idx0
%idx1 = OpAccessChain %ptr_in_v3 %worldPos_in %int_1
%p1 = OpLoad %v3float %idx1
%idxc0 = OpAccessChain %ptr_in_v3 %color_in %int_0
%c0 = OpLoad %v3float %idxc0
%idxc1 = OpAccessChain %ptr_in_v3 %color_in %int_1
%c1 = OpLoad %v3float %idxc1

; Compute direction and perpendicular direction
%direction = OpFSub %v3float %p1 %p0
%length = OpExtInst %float %1 Length %direction
%norm_direction = OpFDiv %v3float %direction %length
%perp = OpExtInst %v3float %1 Cross %norm_direction %v3float_0_0_1
%norm_perp = OpExtInst %v3float %1 Normalize %perp

; Compute vertices
%p0_plus = OpFAdd %v3float %p0 %norm_perp %float_004
%p0_minus = OpFSub %v3float %p0 %norm_perp %float_004
%p1_plus = OpFAdd %v3float %p1 %norm_perp %float_0005
%p1_minus = OpFSub %v3float %p1 %norm_perp %float_0005

; Convert to vec4 and store gl_Position
%v4_p0_plus = OpCompositeConstruct %v4float %p0_plus %float_1
%v4_p0_minus = OpCompositeConstruct %v4float %p0_minus %float_1
%v4_p1_plus = OpCompositeConstruct %v4float %p1_plus %float_1
%v4_p1_minus = OpCompositeConstruct %v4float %p1_minus %float_1

; Emit vertices
OpStore %outColor %c0
%out_pos = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos %v4_p0_plus
OpEmitVertex
OpStore %out_pos %v4_p0_minus
OpEmitVertex
OpStore %outColor %c1
OpStore %out_pos %v4_p1_plus
OpEmitVertex
OpStore %out_pos %v4_p1_minus
OpEmitVertex

OpEndPrimitive

OpReturn
OpFunctionEnd
```
