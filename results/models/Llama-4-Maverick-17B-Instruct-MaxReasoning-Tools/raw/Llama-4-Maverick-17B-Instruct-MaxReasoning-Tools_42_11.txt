```spirv
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %gl_out %outColor %ubo

OpExecutionMode %main InputLinesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 8
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %outColor Location 0
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0

%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_4 = OpConstant %uint 4
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3
%float_0 = OpConstant %float 0
%float_0_02 = OpConstant %float 0.02

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float

%arr_v3_in = OpTypeArray %v3float %uint_4
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%worldPos = OpVariable %ptr_in_arr_v3 Input
%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input

OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%ubo_type = OpTypeStruct %v4float %v4float %v4float %v4float %v3float %v3float %v4float
%ptr_ubo = OpTypePointer Uniform %ubo_type
%ubo = OpVariable %ptr_ubo Uniform

%main = OpFunction %void None %void
%entry = OpLabel

; Read start and end points
%start_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%start = OpLoad %v4float %start_ptr
%end_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%end = OpLoad %v4float %end_ptr

; Compute tangent and perpendicular
%tangent = OpFSub %v4float %end %start
%length = OpExtInst %float %1 Length %tangent
%tangent = OpFDiv %v4float %tangent %length
%perp = OpVectorShuffle %v3float %tangent %tangent 1 2 0
%perp = OpVectorTimesScalar %v3float %perp %float_0_02

; Split segment into 3 parts
%one_third = OpFDiv %float %length %float_3
%part1_end = OpFAdd %v4float %start %tangent %one_third
%part2_start = OpFAdd %v4float %start %tangent %one_third
%part2_end = OpFSub %v4float %end %tangent %one_third
%part3_start = OpFSub %v4float %end %tangent %one_third

; Draw 1st part
%color1_ptr = OpAccessChain %ptr_in_v3 %color %int_1
%color1 = OpLoad %v3float %color1_ptr
%color2_ptr = OpAccessChain %ptr_in_v3 %color %int_2
%color2 = OpLoad %v3float %color2_ptr
%emit_pos1 = OpFSub %v4float %start %perp
OpStore %outColor %color1
%gl_out_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_out_ptr %emit_pos1
OpEmitVertex
%emit_pos2 = OpFAdd %v4float %start %perp
OpStore %outColor %color1
%gl_out_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_out_ptr %emit_pos2
OpEmitVertex
%emit_pos3 = OpFSub %v4float %part1_end %perp
OpStore %outColor %color2
%gl_out_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_out_ptr %emit_pos3
OpEmitVertex
%emit_pos4 = OpFAdd %v4float %part1_end %perp
OpStore %outColor %color2
%gl_out_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_out_ptr %emit_pos4
OpEmitVertex
OpEndPrimitive

; Draw 3rd part
%emit_pos5 = OpFSub %v4float %part3_start %perp
OpStore %outColor %color1
%gl_out_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_out_ptr %emit_pos5
OpEmitVertex
%emit_pos6 = OpFAdd %v4float %part3_start %perp
OpStore %outColor %color1
%gl_out_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_out_ptr %emit_pos6
OpEmitVertex
%emit_pos7 = OpFSub %v4float %end %perp
OpStore %outColor %color2
%gl_out_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_out_ptr %emit_pos7
OpEmitVertex
%emit_pos8 = OpFAdd %v4float %end %perp
OpStore %outColor %color2
%gl_out_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_out_ptr %emit_pos8
OpEmitVertex
OpEndPrimitive

OpReturn
OpFunctionEnd
```
