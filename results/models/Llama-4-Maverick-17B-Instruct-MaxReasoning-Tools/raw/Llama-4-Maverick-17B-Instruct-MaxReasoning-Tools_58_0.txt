const fs = require('fs');

// Read input from stdin
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

// Define a function to calculate Manhattan distance between two points
function manhattanDistance(x1, y1, x2, y2) {
    return Math.abs(x1 - x2) + Math.abs(y1 - y2);
}

// Assign jobs to robots based on the nearest robot
function assignJobs(robots, jobs) {
    // Create a copy of the jobs array to avoid modifying the original array
    const remainingJobs = [...jobs];

    // Sort the jobs by priority in descending order
    remainingJobs.sort((a, b) => b.priority - a.priority);

    // Initialize an array to store the assigned jobs for each robot
    const assignments = Array(robots.length).fill().map(() => []);

    // Assign jobs to the nearest robot
    while (remainingJobs.length > 0) {
        const job = remainingJobs.shift();
        let nearestRobotIndex = -1;
        let minDistance = Infinity;

        // Find the nearest robot to the current job
        for (let i = 0; i < robots.length; i++) {
            const distance = manhattanDistance(robots[i].x, robots[i].y, job.x, job.y);
            if (distance < minDistance) {
                minDistance = distance;
                nearestRobotIndex = i;
            }
        }

        // Assign the job to the nearest robot
        assignments[nearestRobotIndex].push(job.id);
    }

    return assignments;
}

// Process the input data
function processInput(data) {
    const robots = data.robots;
    const jobs = data.jobs;

    // Assign jobs to robots
    const assignments = assignJobs(robots, jobs);

    // Return the assignments in the required format
    return { assignments };
}

// Output the result to stdout
const output = processInput(data);
console.log(JSON.stringify(output));
