// Tetrahedron packing into a convex polyhedron (convex hull of input vertices).
//
// Strategy:
//  - All tetrahedra are emitted with Euler angles (0,0,0). A zero rotation is the
//    identity in every Euler convention, so the verifier reconstructs its own
//    canonical regular tetrahedron (edge e) translated to each center. All placed
//    tetrahedra are therefore translates of one identical convex body whose
//    diameter is exactly e (the edge length).
//  - Centers are placed on an FCC lattice with nearest-neighbor distance
//    slightly greater than e => any two translates are disjoint, no matter what
//    the canonical orientation/reference point is.
//  - Each center is kept at distance >= e (+slack) from every hull face plane,
//    so the whole tetrahedron (contained in a ball of radius <= diameter e
//    around any interior reference point) lies inside the container.
//  - Scales from tiny to enormous containers: per-row analytic x-interval
//    clipping makes enumeration proportional to output size; hard caps on
//    output count and wall time guarantee the 30 s budget.
#include <cstdio>
#include <cmath>
#include <cstdint>
#include <cstring>
#include <vector>
#include <string>
#include <algorithm>
#include <chrono>
#include <unordered_set>
#include <random>

using namespace std;

int main() {
    auto T0 = chrono::steady_clock::now();
    auto elapsed = [&]() {
        return chrono::duration<double>(chrono::steady_clock::now() - T0).count();
    };

    int V = 0; double e = 0.0;
    if (scanf("%d %lf", &V, &e) != 2) { printf("0\n"); return 0; }
    vector<double> px, py, pz;
    px.reserve(V > 0 ? V : 0);
    for (int i = 0; i < V; i++) {
        double x, y, z;
        if (scanf("%lf %lf %lf", &x, &y, &z) != 3) break;
        if (!isfinite(x) || !isfinite(y) || !isfinite(z)) continue;
        px.push_back(x); py.push_back(y); pz.push_back(z);
    }
    size_t N = px.size();
    if (!(e > 0.0) || !isfinite(e) || N < 4) { printf("0\n"); return 0; }

    double scale = e;
    for (size_t i = 0; i < N; i++)
        scale = max(max(scale, fabs(px[i])), max(fabs(py[i]), fabs(pz[i])));
    if (!(scale > 0.0)) scale = 1.0;

    // ---- Reduce vertex count via support-point sampling if huge.
    // Hull of a subset is contained in the true hull, so this stays safe.
    vector<double> hx, hy, hz;
    {
        vector<size_t> idx;
        if (N > 320) {
            unordered_set<size_t> used;
            auto addSupport = [&](double dx, double dy, double dz) {
                double best = -1e300; size_t bi = 0;
                for (size_t i = 0; i < N; i++) {
                    double s = px[i] * dx + py[i] * dy + pz[i] * dz;
                    if (s > best) { best = s; bi = i; }
                }
                if (used.insert(bi).second) idx.push_back(bi);
            };
            for (int s = -1; s <= 1; s += 2) {
                addSupport(s, 0, 0); addSupport(0, s, 0); addSupport(0, 0, s);
                addSupport(s, s, 0); addSupport(s, 0, s); addSupport(0, s, s);
                addSupport(s, s, s); addSupport(s, -s, 0); addSupport(s, 0, -s);
                addSupport(0, s, -s); addSupport(s, s, -s); addSupport(s, -s, s);
            }
            mt19937_64 rng(0x9E3779B97F4A7C15ull);
            normal_distribution<double> nd(0.0, 1.0);
            for (int t = 0; t < 2500 && (int)idx.size() < 320; t++) {
                double dx = nd(rng), dy = nd(rng), dz = nd(rng);
                double L = sqrt(dx * dx + dy * dy + dz * dz);
                if (L < 1e-12) continue;
                addSupport(dx / L, dy / L, dz / L);
            }
        } else {
            for (size_t i = 0; i < N; i++) idx.push_back(i);
        }
        // Shuffle so the face test's early-exit fires fast on average.
        mt19937_64 rng(1234567);
        shuffle(idx.begin(), idx.end(), rng);
        for (size_t i : idx) { hx.push_back(px[i]); hy.push_back(py[i]); hz.push_back(pz[i]); }
    }
    size_t H = hx.size();

    // ---- Enumerate hull face planes by brute force over vertex triples.
    struct Plane { double nx, ny, nz, d; };   // n . p <= d, |n| = 1
    vector<Plane> planes;
    unordered_set<uint64_t> seenPlane;
    double epsF = scale * 1e-9;
    auto planeKey = [&](double nx, double ny, double nz, double d) -> uint64_t {
        uint64_t h = 1469598103934665603ull;
        int64_t q[4] = { llround(nx * 1048576.0), llround(ny * 1048576.0),
                         llround(nz * 1048576.0), llround(d / scale * 1048576.0) };
        for (int i = 0; i < 4; i++) { h ^= (uint64_t)q[i]; h *= 1099511628211ull; }
        return h;
    };
    for (size_t i = 0; i < H; i++)
        for (size_t j = i + 1; j < H; j++) {
            double ax = hx[j] - hx[i], ay = hy[j] - hy[i], az = hz[j] - hz[i];
            for (size_t k = j + 1; k < H; k++) {
                double bx = hx[k] - hx[i], by = hy[k] - hy[i], bz = hz[k] - hz[i];
                double nx = ay * bz - az * by;
                double ny = az * bx - ax * bz;
                double nz = ax * by - ay * bx;
                double L = sqrt(nx * nx + ny * ny + nz * nz);
                if (L < scale * scale * 1e-14) continue;
                nx /= L; ny /= L; nz /= L;
                double d = nx * hx[i] + ny * hy[i] + nz * hz[i];
                uint64_t k1 = planeKey(nx, ny, nz, d);
                uint64_t k2 = planeKey(-nx, -ny, -nz, -d);
                bool need1 = !seenPlane.count(k1);
                bool need2 = !seenPlane.count(k2);
                if (!need1 && !need2) continue;
                bool upper = true, lower = true;   // all pts on <= side / >= side
                for (size_t m = 0; m < H; m++) {
                    double s = nx * hx[m] + ny * hy[m] + nz * hz[m] - d;
                    if (s > epsF) upper = false;
                    if (s < -epsF) lower = false;
                    if (!upper && !lower) break;
                }
                if (upper && need1) { planes.push_back({ nx, ny, nz, d }); seenPlane.insert(k1); }
                if (lower && need2) { planes.push_back({ -nx, -ny, -nz, -d }); seenPlane.insert(k2); }
            }
        }
    if (planes.empty()) { printf("0\n"); return 0; }   // degenerate input

    // ---- Lattice parameters.
    // slack covers FP arithmetic, output rounding and hull-plane tolerance.
    double slack = max(e * 1e-4, scale * 1e-12);
    double margin = e + slack + scale * 2e-9;      // keep centers this far inside
    double a = 1.4142135623730951 * (e + slack);   // FCC cube; nn-dist = e + slack
    double half = a * 0.5;
    int prec = (int)ceil(-log10(slack * 0.005));
    if (prec < 0) prec = 0; if (prec > 17) prec = 17;

    // Inset planes so a center satisfying them keeps the whole tet inside.
    for (auto& pl : planes) pl.d -= margin;

    // Inset bounding box.
    double lox = 1e300, loy = 1e300, loz = 1e300;
    double hix = -1e300, hiy = -1e300, hiz = -1e300;
    for (size_t i = 0; i < N; i++) {
        lox = min(lox, px[i]); hix = max(hix, px[i]);
        loy = min(loy, py[i]); hiy = max(hiy, py[i]);
        loz = min(loz, pz[i]); hiz = max(hiz, pz[i]);
    }
    lox += margin; loy += margin; loz += margin;
    hix -= margin; hiy -= margin; hiz -= margin;
    if (lox > hix || loy > hiy || loz > hiz) { printf("0\n"); return 0; }

    const long long KMAX = 4000000;
    string out;

    // Enumerate FCC points (i+j+k even, step = half) inside the inset region.
    // Per row (fixed y,z) the feasible x range is clipped analytically against
    // every plane, so cost is ~ O(rows + output), independent of bbox volume.
    auto run = [&](double ox, double oy, double oz, bool emit,
                   long long cap, double deadline) -> long long {
        long long cnt = 0, rowTick = 0;
        double x0 = lox + ox, y0 = loy + oy, z0 = loz + oz;
        size_t P = planes.size();
        for (long long kz = 0;; kz++) {
            double z = z0 + kz * half;
            if (z > hiz) break;
            for (long long jy = 0;; jy++) {
                double y = y0 + jy * half;
                if (y > hiy) break;
                if (((++rowTick) & 4095) == 0 && elapsed() > deadline) return cnt;
                double xlo = lox, xhi = hix;
                bool empty = false;
                for (size_t p = 0; p < P; p++) {
                    const Plane& pl = planes[p];
                    double c = pl.d - pl.ny * y - pl.nz * z;
                    if (pl.nx > 1e-12) { double xb = c / pl.nx; if (xb < xhi) xhi = xb; }
                    else if (pl.nx < -1e-12) { double xb = c / pl.nx; if (xb > xlo) xlo = xb; }
                    else if (c < 0.0) { empty = true; break; }
                    if (xlo > xhi) { empty = true; break; }
                }
                if (empty) continue;
                long long i0 = (long long)ceil((xlo - x0) / half - 1e-9);
                while (x0 + (double)i0 * half < xlo) i0++;
                long long i1 = (long long)floor((xhi - x0) / half + 1e-9);
                while (x0 + (double)i1 * half > xhi) i1--;
                long long par = (jy + kz) & 1;
                if ((i0 & 1) != par) i0++;
                if ((i1 & 1) != par) i1--;
                if (i1 < i0) continue;
                if (!emit) {
                    cnt += (i1 - i0) / 2 + 1;
                    if (cnt >= cap) return cnt;
                    continue;
                }
                char ybuf[40], zbuf[40];
                snprintf(ybuf, sizeof(ybuf), "%.*f", prec, y);
                snprintf(zbuf, sizeof(zbuf), "%.*f", prec, z);
                for (long long i = i0; i <= i1; i += 2) {
                    double x = x0 + (double)i * half;
                    char tmp[144];
                    int len = snprintf(tmp, sizeof(tmp), "%.*f %s %s 0 0 0\n",
                                       prec, x, ybuf, zbuf);
                    out.append(tmp, (size_t)len);
                    if (++cnt >= cap) return cnt;
                    if ((cnt & 65535) == 0 && elapsed() > deadline) return cnt;
                }
            }
        }
        return cnt;
    };

    // ---- Small containers: search lattice offsets for the best fit.
    double ox = 0, oy = 0, oz = 0;
    long long est = run(0, 0, 0, false, 300000, min(6.0, 20.0));
    if (est < 250000 && elapsed() < 8.0) {
        int G = (est <= 2000) ? 6 : 4;
        long long best = est;
        bool stop = false;
        for (int ia = 0; ia < G && !stop; ia++)
            for (int ib = 0; ib < G && !stop; ib++)
                for (int ic = 0; ic < G && !stop; ic++) {
                    if (ia == 0 && ib == 0 && ic == 0) continue;
                    if (elapsed() > 14.0) { stop = true; break; }
                    double o1 = ia * (a / G), o2 = ib * (a / G), o3 = ic * (a / G);
                    long long cn = run(o1, o2, o3, false, 300000, 16.0);
                    if (cn > best) { best = cn; ox = o1; oy = o2; oz = o3; }
                }
    }

    // ---- Emit.
    long long reserveN = min(KMAX, max(est, 1024LL));
    out.reserve((size_t)min<long long>(reserveN * 72, 300000000LL));
    long long K = run(ox, oy, oz, true, KMAX, 24.0);

    printf("%lld\n", K);
    fwrite(out.data(), 1, out.size(), stdout);
    fflush(stdout);
    return 0;
}
