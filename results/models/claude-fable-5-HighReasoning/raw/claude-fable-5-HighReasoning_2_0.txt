// Chinese Postman Problem solver.
// Strategy:
//   1. Read graph, compute degrees, find odd-degree vertices.
//   2. Pair odd vertices with minimum-ish total shortest-path cost:
//        - <= 12 odd vertices: exact bitmask-DP matching over full Dijkstra distances.
//        - otherwise: greedy nearest-neighbour matching via truncated Dijkstra
//          (each search stops at the first unmatched odd vertex it settles).
//   3. Duplicate edges along each matching path -> all degrees even.
//   4. Iterative Hierholzer to produce an Euler circuit starting/ending at node 0.
using System;
using System.Collections.Generic;
using System.IO;

public static class Program
{
    // ---------- fast input ----------
    static byte[] inBuf;
    static int inLen, inPos;

    static void ReadAllInput()
    {
        var s = Console.OpenStandardInput();
        var ms = new MemoryStream(1 << 20);
        var tmp = new byte[1 << 16];
        int r;
        while ((r = s.Read(tmp, 0, tmp.Length)) > 0) ms.Write(tmp, 0, r);
        inBuf = ms.GetBuffer();
        inLen = (int)ms.Length;
        inPos = 0;
    }

    static int NextInt()
    {
        while (inPos < inLen && (inBuf[inPos] < '0' || inBuf[inPos] > '9') && inBuf[inPos] != '-') inPos++;
        bool neg = false;
        if (inPos < inLen && inBuf[inPos] == '-') { neg = true; inPos++; }
        int x = 0;
        while (inPos < inLen && inBuf[inPos] >= '0' && inBuf[inPos] <= '9')
        {
            x = x * 10 + (inBuf[inPos] - '0');
            inPos++;
        }
        return neg ? -x : x;
    }

    // ---------- graph ----------
    static int n, m;
    static int[] eu, ev, ew;
    static int[] adjStart, adjTo, adjEid; // CSR over original edges (half-edges)

    // ---------- Dijkstra state (versioned so no O(V) reset per run) ----------
    static long[] dist;
    static int[] distVer;
    static int[] parEdge;
    static int curVer;
    static long[] heapKey;
    static int[] heapVal;
    static int heapSize;

    static void HeapPush(long k, int v)
    {
        if (heapSize == heapKey.Length)
        {
            Array.Resize(ref heapKey, heapSize * 2);
            Array.Resize(ref heapVal, heapSize * 2);
        }
        int i = heapSize++;
        heapKey[i] = k; heapVal[i] = v;
        while (i > 0)
        {
            int p = (i - 1) >> 1;
            if (heapKey[p] <= heapKey[i]) break;
            long tk = heapKey[p]; heapKey[p] = heapKey[i]; heapKey[i] = tk;
            int tv = heapVal[p]; heapVal[p] = heapVal[i]; heapVal[i] = tv;
            i = p;
        }
    }

    static void HeapPop(out long k, out int v)
    {
        k = heapKey[0]; v = heapVal[0];
        heapSize--;
        if (heapSize > 0)
        {
            long lk = heapKey[heapSize]; int lv = heapVal[heapSize];
            int i = 0;
            while (true)
            {
                int c = i * 2 + 1;
                if (c >= heapSize) break;
                if (c + 1 < heapSize && heapKey[c + 1] < heapKey[c]) c++;
                if (heapKey[c] >= lk) break;
                heapKey[i] = heapKey[c]; heapVal[i] = heapVal[c];
                i = c;
            }
            heapKey[i] = lk; heapVal[i] = lv;
        }
    }

    // Dijkstra from src. If stopFlags != null, stops when it settles a node t != src
    // with stopFlags[t] set, returning t. Otherwise runs to completion and returns -1.
    static int Dijkstra(int src, bool[] stopFlags)
    {
        curVer++;
        heapSize = 0;
        dist[src] = 0; distVer[src] = curVer; parEdge[src] = -1;
        HeapPush(0, src);
        while (heapSize > 0)
        {
            long d; int v;
            HeapPop(out d, out v);
            if (distVer[v] != curVer || d != dist[v]) continue;
            if (stopFlags != null && v != src && stopFlags[v]) return v;
            for (int p = adjStart[v]; p < adjStart[v + 1]; p++)
            {
                int e = adjEid[p];
                int to = adjTo[p];
                long nd = d + ew[e];
                if (distVer[to] != curVer || nd < dist[to])
                {
                    dist[to] = nd;
                    distVer[to] = curVer;
                    parEdge[to] = e;
                    HeapPush(nd, to);
                }
            }
        }
        return -1;
    }

    // Walk parent-edge chain from t back to the Dijkstra source, appending edge ids.
    static void TracePath(int t, List<int> dup, int[] parents)
    {
        int v = t;
        while (parents[v] != -1)
        {
            int e = parents[v];
            dup.Add(e);
            v = eu[e] ^ ev[e] ^ v;
        }
    }

    public static void Main()
    {
        ReadAllInput();
        n = NextInt(); m = NextInt();
        eu = new int[m]; ev = new int[m]; ew = new int[m];
        var deg = new int[n];
        for (int i = 0; i < m; i++)
        {
            int u = NextInt(), v = NextInt(), w = NextInt();
            eu[i] = u; ev[i] = v; ew[i] = w < 0 ? 0 : w;
            deg[u]++; deg[v]++;
        }

        var stdout = Console.OpenStandardOutput();
        if (m == 0)
        {
            var b = System.Text.Encoding.ASCII.GetBytes("1\n0\n");
            stdout.Write(b, 0, b.Length);
            stdout.Flush();
            return;
        }

        // CSR adjacency of original graph
        adjStart = new int[n + 1];
        for (int v = 0; v < n; v++) adjStart[v + 1] = adjStart[v] + deg[v];
        adjTo = new int[2 * m];
        adjEid = new int[2 * m];
        var fill = new int[n];
        Array.Copy(adjStart, fill, n);
        for (int i = 0; i < m; i++)
        {
            int u = eu[i], v = ev[i];
            adjTo[fill[u]] = v; adjEid[fill[u]] = i; fill[u]++;
            adjTo[fill[v]] = u; adjEid[fill[v]] = i; fill[v]++;
        }

        // odd-degree vertices
        var odd = new List<int>();
        for (int v = 0; v < n; v++) if ((deg[v] & 1) == 1) odd.Add(v);

        dist = new long[n];
        distVer = new int[n];
        parEdge = new int[n];
        heapKey = new long[1024];
        heapVal = new int[1024];

        var dup = new List<int>(); // edge ids to duplicate

        int k = odd.Count;
        if (k > 0 && k <= 12)
        {
            // exact matching via bitmask DP on all-pairs shortest distances
            var cost = new long[k, k];
            var parRows = new int[k][];
            for (int i = 0; i < k; i++)
            {
                Dijkstra(odd[i], null);
                parRows[i] = new int[n];
                Array.Copy(parEdge, parRows[i], n);
                for (int j = 0; j < k; j++)
                    cost[i, j] = distVer[odd[j]] == curVer ? dist[odd[j]] : long.MaxValue / 4;
            }
            int full = 1 << k;
            var dp = new long[full];
            var choice = new int[full];
            for (int mask = 1; mask < full; mask++) dp[mask] = long.MaxValue / 4;
            for (int mask = 1; mask < full; mask++)
            {
                int i = 0;
                while ((mask & (1 << i)) == 0) i++;
                int rest = mask ^ (1 << i);
                if (rest == 0) continue; // odd popcount masks never used
                for (int j = i + 1; j < k; j++)
                {
                    if ((rest & (1 << j)) == 0) continue;
                    long cand = dp[rest ^ (1 << j)] + cost[i, j];
                    if (cand < dp[mask]) { dp[mask] = cand; choice[mask] = j; }
                }
            }
            int cur = full - 1;
            while (cur != 0)
            {
                int i = 0;
                while ((cur & (1 << i)) == 0) i++;
                int j = choice[cur];
                TracePath(odd[j], dup, parRows[i]);
                cur ^= (1 << i) | (1 << j);
            }
        }
        else if (k > 0)
        {
            // greedy nearest-pair matching via truncated Dijkstra
            var unmatched = new bool[n];
            foreach (int v in odd) unmatched[v] = true;
            foreach (int a in odd)
            {
                if (!unmatched[a]) continue;
                unmatched[a] = false; // exclude self as a stop target
                int t = Dijkstra(a, unmatched);
                if (t < 0) { unmatched[a] = true; continue; } // unreachable partner (disconnected)
                unmatched[t] = false;
                TracePath(t, dup, parEdge);
            }
        }

        // ----- build multigraph (original + duplicated edges) -----
        int extra = dup.Count;
        int T = m + extra;
        var feu = new int[T];
        var fev = new int[T];
        Array.Copy(eu, feu, m);
        Array.Copy(ev, fev, m);
        for (int i = 0; i < extra; i++)
        {
            int e = dup[i];
            feu[m + i] = eu[e];
            fev[m + i] = ev[e];
        }

        var start2 = new int[n + 1];
        for (int i = 0; i < T; i++) { start2[feu[i] + 1]++; start2[fev[i] + 1]++; }
        for (int v = 0; v < n; v++) start2[v + 1] += start2[v];
        var to2 = new int[2 * T];
        var eid2 = new int[2 * T];
        var fill2 = new int[n];
        Array.Copy(start2, fill2, n);
        for (int i = 0; i < T; i++)
        {
            int u = feu[i], v = fev[i];
            to2[fill2[u]] = v; eid2[fill2[u]] = i; fill2[u]++;
            to2[fill2[v]] = u; eid2[fill2[v]] = i; fill2[v]++;
        }

        // ----- Hierholzer (iterative) -----
        var ptr = new int[n];
        Array.Copy(start2, ptr, n);
        var used = new bool[T];
        var stack = new int[T + 2];
        var route = new int[T + 2];
        int sp = 0, rl = 0;
        stack[sp++] = 0;
        while (sp > 0)
        {
            int v = stack[sp - 1];
            int p = ptr[v];
            int end = start2[v + 1];
            while (p < end && used[eid2[p]]) p++;
            ptr[v] = p;
            if (p == end)
            {
                route[rl++] = v;
                sp--;
            }
            else
            {
                used[eid2[p]] = true;
                ptr[v] = p + 1;
                stack[sp++] = to2[p];
            }
        }

        // ----- output -----
        var sb = new System.Text.StringBuilder(rl * 7 + 16);
        sb.Append(rl).Append('\n');
        for (int i = rl - 1; i >= 0; i--)
        {
            sb.Append(route[i]);
            sb.Append(i == 0 ? '\n' : ' ');
        }
        var outBytes = System.Text.Encoding.ASCII.GetBytes(sb.ToString());
        stdout.Write(outBytes, 0, outBytes.Length);
        stdout.Flush();
    }
}
