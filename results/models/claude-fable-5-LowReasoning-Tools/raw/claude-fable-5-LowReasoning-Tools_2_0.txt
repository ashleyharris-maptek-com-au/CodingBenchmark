using System;
using System.IO;
using System.Collections.Generic;

class ChinesePostman
{
    static int N, M;
    static int[] eu, ev;
    static long[] ew;
    // CSR of original graph (for Dijkstra)
    static int[] ostart, oadjTo;
    static long[] oadjW;

    // Dijkstra state, stamped so we never clear whole arrays
    static long[] dist;
    static int[] par;
    static int[] stampArr;
    static int stamp = 0;
    static bool[] settled;

    static long[] heapD;
    static int[] heapV;
    static int heapN;

    static List<int> dupU = new List<int>();
    static List<int> dupV = new List<int>();

    static void HeapPush(long d, int v)
    {
        if (heapN == heapD.Length)
        {
            Array.Resize(ref heapD, heapN * 2);
            Array.Resize(ref heapV, heapN * 2);
        }
        int i = heapN++;
        heapD[i] = d; heapV[i] = v;
        while (i > 0)
        {
            int p = (i - 1) >> 1;
            if (heapD[p] <= heapD[i]) break;
            long td = heapD[p]; heapD[p] = heapD[i]; heapD[i] = td;
            int tv = heapV[p]; heapV[p] = heapV[i]; heapV[i] = tv;
            i = p;
        }
    }

    static void HeapPop(out long d, out int v)
    {
        d = heapD[0]; v = heapV[0];
        heapN--;
        if (heapN > 0)
        {
            heapD[0] = heapD[heapN]; heapV[0] = heapV[heapN];
            int i = 0;
            while (true)
            {
                int l = 2 * i + 1, r = l + 1, s = i;
                if (l < heapN && heapD[l] < heapD[s]) s = l;
                if (r < heapN && heapD[r] < heapD[s]) s = r;
                if (s == i) break;
                long td = heapD[s]; heapD[s] = heapD[i]; heapD[i] = td;
                int tv = heapV[s]; heapV[s] = heapV[i]; heapV[i] = tv;
                i = s;
            }
        }
    }

    static long GetDist(int v) { return stampArr[v] == stamp ? dist[v] : long.MaxValue; }

    static void SetDist(int v, long d, int p)
    {
        if (stampArr[v] != stamp) { stampArr[v] = stamp; settled[v] = false; }
        dist[v] = d; par[v] = p;
    }

    // Dijkstra from s.
    // target >= 0: stop when target settled, return target.
    // target < 0 and wanted != null: stop at first settled v != s with wanted[v], return it.
    // target < 0 and wanted == null: full Dijkstra, return -1.
    static int Dijkstra(int s, int target, bool[] wanted)
    {
        stamp++;
        heapN = 0;
        SetDist(s, 0, -1);
        HeapPush(0, s);
        while (heapN > 0)
        {
            long d; int v;
            HeapPop(out d, out v);
            if (stampArr[v] != stamp || settled[v] || d != dist[v]) continue;
            settled[v] = true;
            if (target >= 0) { if (v == target) return v; }
            else if (wanted != null && v != s && wanted[v]) return v;
            for (int i = ostart[v]; i < ostart[v + 1]; i++)
            {
                int to = oadjTo[i];
                long nd = d + oadjW[i];
                if (stampArr[to] != stamp || (!settled[to] && nd < dist[to]))
                {
                    SetDist(to, nd, v);
                    HeapPush(nd, to);
                }
            }
        }
        return -1;
    }

    static void DuplicatePath(int s, int t)
    {
        int cur = t;
        while (cur != s)
        {
            int p = par[cur];
            dupU.Add(p); dupV.Add(cur);
            cur = p;
        }
    }

    static void Main()
    {
        // ---- fast input ----
        var stdinStream = Console.OpenStandardInput();
        byte[] buf;
        int len;
        using (var ms = new MemoryStream())
        {
            stdinStream.CopyTo(ms, 1 << 20);
            buf = ms.GetBuffer();
            len = (int)ms.Length;
        }
        int pos = 0;
        Func<long> readLong = delegate ()
        {
            while (pos < len && (buf[pos] < (byte)'0' || buf[pos] > (byte)'9') && buf[pos] != (byte)'-') pos++;
            bool neg = false;
            if (pos < len && buf[pos] == (byte)'-') { neg = true; pos++; }
            long x = 0;
            while (pos < len && buf[pos] >= (byte)'0' && buf[pos] <= (byte)'9') { x = x * 10 + (buf[pos] - (byte)'0'); pos++; }
            return neg ? -x : x;
        };

        N = (int)readLong(); M = (int)readLong();
        eu = new int[M]; ev = new int[M]; ew = new long[M];
        int[] deg = new int[N];
        for (int i = 0; i < M; i++)
        {
            eu[i] = (int)readLong(); ev[i] = (int)readLong(); ew[i] = readLong();
            if (ew[i] < 0) ew[i] = 0;
            deg[eu[i]]++; deg[ev[i]]++;
        }

        var outWriter = new StreamWriter(new BufferedStream(Console.OpenStandardOutput(), 1 << 22));
        if (M == 0) { outWriter.Write("1\n0\n"); outWriter.Flush(); return; }

        // ---- CSR original graph ----
        ostart = new int[N + 1];
        for (int v = 0; v < N; v++) ostart[v + 1] = ostart[v] + deg[v];
        oadjTo = new int[2 * M]; oadjW = new long[2 * M];
        int[] fill = new int[N];
        Array.Copy(ostart, fill, N);
        for (int i = 0; i < M; i++)
        {
            oadjTo[fill[eu[i]]] = ev[i]; oadjW[fill[eu[i]]++] = ew[i];
            oadjTo[fill[ev[i]]] = eu[i]; oadjW[fill[ev[i]]++] = ew[i];
        }

        dist = new long[N]; par = new int[N];
        stampArr = new int[N]; settled = new bool[N];
        heapD = new long[1 << 12]; heapV = new int[1 << 12];

        // ---- odd-degree vertices ----
        var oddList = new List<int>();
        for (int v = 0; v < N; v++) if ((deg[v] & 1) != 0) oddList.Add(v);
        int K = oddList.Count;

        if (K > 0)
        {
            if (K <= 16)
            {
                // exact minimum-weight perfect matching via bitmask DP
                long[,] dm = new long[K, K];
                for (int i = 0; i < K; i++)
                {
                    Dijkstra(oddList[i], -1, null);
                    for (int j = 0; j < K; j++) dm[i, j] = GetDist(oddList[j]);
                }
                int full = 1 << K;
                long[] dp = new long[full];
                int[] choice = new int[full];
                dp[0] = 0;
                for (int m2 = 1; m2 < full; m2++)
                {
                    dp[m2] = long.MaxValue; choice[m2] = -1;
                    int i = 0;
                    while (((m2 >> i) & 1) == 0) i++;
                    int pc = m2; pc = pc - ((pc >> 1) & 0x55555555); pc = (pc & 0x33333333) + ((pc >> 2) & 0x33333333); pc = (((pc + (pc >> 4)) & 0x0F0F0F0F) * 0x01010101) >> 24;
                    if ((pc & 1) != 0) continue;
                    for (int j = i + 1; j < K; j++)
                    {
                        if (((m2 >> j) & 1) == 0) continue;
                        int prev = m2 & ~(1 << i) & ~(1 << j);
                        if (dp[prev] == long.MaxValue) continue;
                        if (dm[i, j] == long.MaxValue) continue;
                        long c = dp[prev] + dm[i, j];
                        if (c < dp[m2]) { dp[m2] = c; choice[m2] = (i << 5) | j; }
                    }
                }
                int mask = full - 1;
                while (mask != 0)
                {
                    int ch = choice[mask];
                    int i = ch >> 5, j = ch & 31;
                    int s = oddList[i], t = oddList[j];
                    Dijkstra(s, t, null);
                    DuplicatePath(s, t);
                    mask &= ~(1 << i) & ~(1 << j);
                }
            }
            else
            {
                // greedy nearest-unmatched matching (scales to huge graphs)
                bool[] unmatched = new bool[N];
                foreach (int v in oddList) unmatched[v] = true;
                foreach (int s in oddList)
                {
                    if (!unmatched[s]) continue;
                    unmatched[s] = false;
                    int t = Dijkstra(s, -1, unmatched);
                    unmatched[t] = false;
                    DuplicatePath(s, t);
                }
            }
        }

        // ---- final multigraph: original + duplicated edges ----
        int D = dupU.Count;
        int E2 = M + D;
        int[] fu = new int[E2], fv = new int[E2];
        Array.Copy(eu, fu, M); Array.Copy(ev, fv, M);
        for (int i = 0; i < D; i++) { fu[M + i] = dupU[i]; fv[M + i] = dupV[i]; }

        int[] fdeg = new int[N];
        for (int i = 0; i < E2; i++) { fdeg[fu[i]]++; fdeg[fv[i]]++; }
        int[] fstart = new int[N + 1];
        for (int v = 0; v < N; v++) fstart[v + 1] = fstart[v] + fdeg[v];
        int[] adjTo = new int[2 * E2];
        int[] adjEid = new int[2 * E2];
        int[] ffill = new int[N];
        Array.Copy(fstart, ffill, N);
        for (int i = 0; i < E2; i++)
        {
            adjTo[ffill[fu[i]]] = fv[i]; adjEid[ffill[fu[i]]++] = i;
            adjTo[ffill[fv[i]]] = fu[i]; adjEid[ffill[fv[i]]++] = i;
        }

        // ---- iterative Hierholzer Euler circuit from node 0 ----
        bool[] used = new bool[E2];
        int[] ptr = new int[N];
        Array.Copy(fstart, ptr, N);
        int[] stack = new int[64];
        int sp = 0;
        stack[sp++] = 0;
        int[] circuit = new int[E2 + 2];
        int cn = 0;
        while (sp > 0)
        {
            int v = stack[sp - 1];
            int p = ptr[v];
            int end = fstart[v + 1];
            while (p < end && used[adjEid[p]]) p++;
            ptr[v] = p;
            if (p == end)
            {
                circuit[cn++] = v;
                sp--;
            }
            else
            {
                used[adjEid[p]] = true;
                ptr[v] = p + 1;
                if (sp == stack.Length) Array.Resize(ref stack, sp * 2);
                stack[sp++] = adjTo[p];
            }
        }

        // ---- output ----
        var sb = new System.Text.StringBuilder(cn * 8 + 16);
        sb.Append(cn).Append('\n');
        for (int i = cn - 1; i >= 0; i--)
        {
            sb.Append(circuit[i]);
            sb.Append(i == 0 ? '\n' : ' ');
        }
        outWriter.Write(sb.ToString());
        outWriter.Flush();
    }
}
