// Tetrahedron packing into a convex container (convex hull of input vertices).
// Strategy: all tetrahedra share one orientation (Euler 0,0,0). Any two congruent
// regular tetrahedra whose centers are more than one edge length apart cannot
// overlap (the diameter of a regular tetrahedron equals its edge). So we place
// centers on an FCC lattice with minimum point distance slightly above the edge
// length -- valid no matter which canonical orientation the checker uses.
// Containment: center must be at least the circumradius away from every hull
// plane (conservative, orientation independent).
// Scalability: rows of the lattice are clipped analytically against the hull
// planes, so per emitted tetra work is O(1) and empty space costs one interval
// computation per row. Output is capped and time-guarded.
#include <cstdio>
#include <cstdlib>
#include <cmath>
#include <cstring>
#include <vector>
#include <string>
#include <algorithm>
#include <charconv>
#include <chrono>
#include <random>

using namespace std;
using Clock = chrono::steady_clock;
static Clock::time_point T0;
static double elapsed() { return chrono::duration<double>(Clock::now() - T0).count(); }

struct V3 { double x, y, z; };
static V3 sub(const V3&a,const V3&b){return{a.x-b.x,a.y-b.y,a.z-b.z};}
static V3 cross(const V3&a,const V3&b){return{a.y*b.z-a.z*b.y,a.z*b.x-a.x*b.z,a.x*b.y-a.y*b.x};}
static double dot(const V3&a,const V3&b){return a.x*b.x+a.y*b.y+a.z*b.z;}
static double norm(const V3&a){return sqrt(dot(a,a));}

struct Plane { V3 n; double d; }; // n·x <= d, |n| = 1

static const double TIME_LIMIT = 24.0;
static const long long MAX_OUT = 2000000;

int main() {
    T0 = Clock::now();
    int V; double e;
    if (scanf("%d %lf", &V, &e) != 2 || V < 4 || !(e > 0)) { printf("0\n"); return 0; }
    vector<V3> pts(V);
    for (int i = 0; i < V; i++)
        if (scanf("%lf %lf %lf", &pts[i].x, &pts[i].y, &pts[i].z) != 3) { printf("0\n"); return 0; }

    V3 lo = pts[0], hi = pts[0];
    for (auto&p : pts) {
        lo.x=min(lo.x,p.x); lo.y=min(lo.y,p.y); lo.z=min(lo.z,p.z);
        hi.x=max(hi.x,p.x); hi.y=max(hi.y,p.y); hi.z=max(hi.z,p.z);
    }
    double scale = max({hi.x-lo.x, hi.y-lo.y, hi.z-lo.z, 1e-300});
    double geps = scale * 1e-9;

    // If many vertices, keep only extreme points along many directions (a subset
    // of hull vertices) -> inner approximation of the container, still safe.
    vector<V3> hp = pts;
    if ((int)hp.size() > 220) {
        mt19937_64 rng(12345);
        vector<V3> sel;
        auto addExtreme = [&](V3 d){
            int best = 0; double bv = -1e308;
            for (int i = 0; i < (int)hp.size(); i++) { double v = dot(hp[i], d); if (v > bv) { bv = v; best = i; } }
            sel.push_back(hp[best]);
        };
        for (int s2 = -1; s2 <= 1; s2 += 2) { addExtreme({(double)s2,0,0}); addExtreme({0,(double)s2,0}); addExtreme({0,0,(double)s2}); }
        uniform_real_distribution<double> U(-1.0, 1.0);
        for (int k = 0; k < 400; k++) { V3 d{U(rng),U(rng),U(rng)}; double n = norm(d); if (n > 1e-12) { d.x/=n; d.y/=n; d.z/=n; addExtreme(d); } }
        sort(sel.begin(), sel.end(), [](const V3&a,const V3&b){ return a.x!=b.x?a.x<b.x : a.y!=b.y?a.y<b.y : a.z<b.z; });
        sel.erase(unique(sel.begin(), sel.end(), [](const V3&a,const V3&b){ return a.x==b.x&&a.y==b.y&&a.z==b.z; }), sel.end());
        hp = sel;
    }

    // Hull planes by triple enumeration.
    vector<Plane> planes;
    {
        int n = (int)hp.size();
        for (int i = 0; i < n && elapsed() < TIME_LIMIT * 0.4; i++)
        for (int j = i+1; j < n; j++)
        for (int k = j+1; k < n; k++) {
            V3 nn = cross(sub(hp[j],hp[i]), sub(hp[k],hp[i]));
            double ln = norm(nn);
            if (ln < 1e-14 * scale * scale) continue;
            nn.x/=ln; nn.y/=ln; nn.z/=ln;
            double d = dot(nn, hp[i]);
            bool allBelow = true, allAbove = true;
            for (int m = 0; m < n; m++) {
                double sd = dot(nn, hp[m]) - d;
                if (sd >  geps) allBelow = false;
                if (sd < -geps) allAbove = false;
                if (!allBelow && !allAbove) break;
            }
            if (!allBelow && !allAbove) continue;
            if (!allBelow) { nn.x=-nn.x; nn.y=-nn.y; nn.z=-nn.z; d = -d; }
            bool dup = false;
            for (auto&q : planes)
                if (fabs(q.n.x-nn.x)<1e-7 && fabs(q.n.y-nn.y)<1e-7 && fabs(q.n.z-nn.z)<1e-7 && fabs(q.d-d)<geps*10+1e-12) { dup = true; break; }
            if (!dup) planes.push_back({nn, d});
        }
    }
    if (planes.size() < 4) { printf("0\n"); return 0; }

    const double R = e * sqrt(3.0/8.0);            // circumradius
    const double margin = max(e * 1e-6, scale * 1e-11);
    const double s = e + margin;                    // min center distance
    const double dz = s / sqrt(2.0);                // FCC layer spacing

    // Effective planes for the center: n·c <= d - R
    vector<Plane> pl = planes;
    for (auto&p : pl) p.d -= R;

    double zlo = lo.z + R, zhi = hi.z - R;
    double ylo0 = lo.y + R, yhi0 = hi.y - R;
    double xlo0 = lo.x + R, xhi0 = hi.x - R;

    auto centroidFallback = [&]() {
        V3 c{0,0,0}; for (auto&p:pts){c.x+=p.x;c.y+=p.y;c.z+=p.z;}
        c.x/=V; c.y/=V; c.z/=V;
        bool ok = true; for (auto&p:pl) if (dot(p.n,c) > p.d) { ok=false; break; }
        if (ok) printf("1\n%.17g %.17g %.17g 0 0 0\n", c.x, c.y, c.z);
        else printf("0\n");
    };

    if (zhi < zlo || yhi0 < ylo0 || xhi0 < xlo0) { centroidFallback(); return 0; }

    string out;
    auto emitNum = [&](double v){
        char buf[40];
        auto r = to_chars(buf, buf + sizeof(buf), v);
        out.append(buf, r.ptr);
    };

    // One packing pass with a given lattice origin shift.
    auto runPass = [&](double ox, double oy, double oz, vector<V3>* store, long long cap) -> long long {
        long long cnt = 0;
        double z0 = zlo + oz;
        long long kmax = (long long)floor((zhi - z0) / dz);
        for (long long k = 0; k <= kmax; k++) {
            double z = z0 + k * dz;
            double off = (k & 1) ? s * 0.5 : 0.0;
            double y0 = ylo0 + oy + off;
            long long jmax = (long long)floor((yhi0 - y0) / s);
            if (jmax < 0) continue;
            for (long long j = 0; j <= jmax; j++) {
                double y = y0 + j * s;
                double xstart = xlo0 + ox + off;
                double xl = xstart, xh = xhi0;
                bool feas = true;
                for (auto&p : pl) {
                    double t = p.d - p.n.y * y - p.n.z * z;
                    if (p.n.x > 1e-12)      { double v = t / p.n.x; if (v < xh) xh = v; }
                    else if (p.n.x < -1e-12){ double v = t / p.n.x; if (v > xl) xl = v; }
                    else if (t < 0)         { feas = false; break; }
                    if (xh < xl) { feas = false; break; }
                }
                if (!feas) continue;
                long long i0 = (long long)ceil((xl - xstart) / s - 1e-12);
                if (i0 < 0) i0 = 0;
                long long i1 = (long long)floor((xh - xstart) / s + 1e-12);
                for (long long i = i0; i <= i1; i++) {
                    double x = xstart + i * s;
                    bool ok = true;
                    for (auto&p : pl) if (p.n.x*x + p.n.y*y + p.n.z*z > p.d) { ok = false; break; }
                    if (!ok) continue;
                    if (store) store->push_back({x,y,z});
                    else {
                        emitNum(x); out.push_back(' ');
                        emitNum(y); out.push_back(' ');
                        emitNum(z); out.append(" 0 0 0\n");
                    }
                    if (++cnt >= cap) return cnt;
                }
                if ((j & 1023) == 0 && elapsed() > TIME_LIMIT) return cnt;
            }
            if (elapsed() > TIME_LIMIT) return cnt;
        }
        return cnt;
    };

    // First pass: store points (bounded) to decide if offset search pays off.
    vector<V3> best;
    long long firstCap = min(MAX_OUT, (long long)300000);
    runPass(0, 0, 0, &best, firstCap);

    if ((long long)best.size() >= firstCap) {
        // Big container: single streaming pass, no offset search.
        best.clear(); best.shrink_to_fit();
        out.reserve(1u << 26);
        long long cnt = runPass(0, 0, 0, nullptr, MAX_OUT);
        printf("%lld\n", cnt);
        fwrite(out.data(), 1, out.size(), stdout);
        return 0;
    }

    // Small/medium: try a few lattice shifts, keep the best.
    double sh[3] = {0.0, s/3.0, 2.0*s/3.0};
    for (int a = 0; a < 3; a++)
    for (int b = 0; b < 3; b++)
    for (int c = 0; c < 3; c++) {
        if (a==0 && b==0 && c==0) continue;
        if (elapsed() > TIME_LIMIT * 0.8) goto doneSearch;
        {
            double ozc = sh[c] / sqrt(2.0);
            vector<V3> cur;
            runPass(sh[a], sh[b], ozc, &cur, firstCap);
            if (cur.size() > best.size()) best.swap(cur);
        }
    }
doneSearch:;

    if (best.empty()) { centroidFallback(); return 0; }

    printf("%zu\n", best.size());
    out.reserve(best.size() * 64);
    for (auto&p : best) {
        emitNum(p.x); out.push_back(' ');
        emitNum(p.y); out.push_back(' ');
        emitNum(p.z); out.append(" 0 0 0\n");
    }
    fwrite(out.data(), 1, out.size(), stdout);
    return 0;
}
