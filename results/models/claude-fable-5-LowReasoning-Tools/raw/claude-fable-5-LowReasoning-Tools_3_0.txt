// Graph layout (edge-crossing minimization) via adaptive force-directed layout.
// Fruchterman-Reingold with spatial-grid approximate repulsion => ~O(N) per iteration.
// Iteration count adapts to graph size; components laid out separately and packed.

use std::io::{self, Read, Write};

struct Rng(u64);
impl Rng {
    fn next(&mut self) -> u64 {
        // xorshift64*
        let mut x = self.0;
        x ^= x >> 12;
        x ^= x << 25;
        x ^= x >> 27;
        self.0 = x;
        x.wrapping_mul(0x2545F4914F6CDD1D)
    }
    fn f(&mut self) -> f32 {
        (self.next() >> 11) as f32 / (1u64 << 53) as f32
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input
        .split_ascii_whitespace()
        .map(|t| t.parse::<usize>().unwrap_or(0));
    let n = it.next().unwrap_or(0);
    let m = it.next().unwrap_or(0);

    let stdout = io::stdout();
    let mut out = io::BufWriter::new(stdout.lock());

    if n == 0 {
        return;
    }

    // Read edges, drop self-loops, dedupe parallel edges.
    let mut edges: Vec<(u32, u32)> = Vec::with_capacity(m);
    for _ in 0..m {
        let u = it.next().unwrap_or(0);
        let v = it.next().unwrap_or(0);
        if u == v || u >= n || v >= n {
            continue;
        }
        let (a, b) = if u < v { (u, v) } else { (v, u) };
        edges.push((a as u32, b as u32));
    }
    edges.sort_unstable();
    edges.dedup();
    let me = edges.len();

    // CSR adjacency
    let mut deg = vec![0u32; n];
    for &(u, v) in &edges {
        deg[u as usize] += 1;
        deg[v as usize] += 1;
    }
    let mut off = vec![0usize; n + 1];
    for i in 0..n {
        off[i + 1] = off[i] + deg[i] as usize;
    }
    let mut adj = vec![0u32; 2 * me];
    let mut cur = off.clone();
    for &(u, v) in &edges {
        adj[cur[u as usize]] = v;
        cur[u as usize] += 1;
        adj[cur[v as usize]] = u;
        cur[v as usize] += 1;
    }

    // Connected components via BFS
    let mut comp = vec![u32::MAX; n];
    let mut comps: Vec<Vec<u32>> = Vec::new();
    let mut queue: Vec<u32> = Vec::new();
    for s in 0..n {
        if comp[s] != u32::MAX {
            continue;
        }
        let cid = comps.len() as u32;
        comp[s] = cid;
        queue.clear();
        queue.push(s as u32);
        let mut head = 0;
        while head < queue.len() {
            let u = queue[head] as usize;
            head += 1;
            for &v in &adj[off[u]..off[u + 1]] {
                if comp[v as usize] == u32::MAX {
                    comp[v as usize] = cid;
                    queue.push(v);
                }
            }
        }
        comps.push(queue.clone());
    }

    let mut px = vec![0f32; n];
    let mut py = vec![0f32; n];
    let mut rng = Rng(0x9E3779B97F4A7C15);

    // Iteration budget adapts to total size.
    let iters_total: usize = if n <= 200 {
        800
    } else if n <= 2_000 {
        400
    } else if n <= 20_000 {
        150
    } else if n <= 80_000 {
        70
    } else {
        40
    };

    let mut placements: Vec<(f32, f32)> = Vec::with_capacity(comps.len()); // half-extents
    for members in &comps {
        let cn = members.len();
        let radius = (cn as f32).sqrt().max(1.0);
        if cn == 1 {
            px[members[0] as usize] = 0.0;
            py[members[0] as usize] = 0.0;
            placements.push((1.0, 1.0));
            continue;
        }
        // initial random placement in a disc
        for &v in members {
            let a = rng.f() * std::f32::consts::TAU;
            let r = radius * rng.f().sqrt();
            px[v as usize] = r * a.cos();
            py[v as usize] = r * a.sin();
        }
        layout_component(members, &adj, &off, &mut px, &mut py, radius, iters_total);
        // recenter to bounding-box center
        let (mut minx, mut maxx, mut miny, mut maxy) = (f32::MAX, f32::MIN, f32::MAX, f32::MIN);
        for &v in members {
            let (x, y) = (px[v as usize], py[v as usize]);
            minx = minx.min(x);
            maxx = maxx.max(x);
            miny = miny.min(y);
            maxy = maxy.max(y);
        }
        let cx = (minx + maxx) * 0.5;
        let cy = (miny + maxy) * 0.5;
        for &v in members {
            px[v as usize] -= cx;
            py[v as usize] -= cy;
        }
        placements.push(((maxx - minx) * 0.5 + 1.0, (maxy - miny) * 0.5 + 1.0));
    }

    // Pack components into rows, tallest first.
    if comps.len() > 1 {
        let mut order: Vec<usize> = (0..comps.len()).collect();
        order.sort_unstable_by(|&a, &b| placements[b].1.partial_cmp(&placements[a].1).unwrap());
        let total_area: f32 = placements.iter().map(|p| 4.0 * p.0 * p.1).sum();
        let row_width = total_area
            .sqrt()
            .max(placements.iter().fold(0.0f32, |mx, p| mx.max(2.0 * p.0)));
        let (mut x, mut y, mut row_h) = (0f32, 0f32, 0f32);
        for &ci in &order {
            let (hw, hh) = placements[ci];
            if x + 2.0 * hw > row_width && x > 0.0 {
                x = 0.0;
                y += row_h;
                row_h = 0.0;
            }
            let ox = x + hw;
            let oy = y + hh;
            for &v in &comps[ci] {
                px[v as usize] += ox;
                py[v as usize] += oy;
            }
            x += 2.0 * hw;
            row_h = row_h.max(2.0 * hh);
        }
    }

    let mut buf = String::with_capacity(1 << 20);
    for i in 0..n {
        buf.push_str(&format!("{:.4} {:.4}\n", px[i], py[i]));
        if buf.len() > 1 << 20 {
            out.write_all(buf.as_bytes()).unwrap();
            buf.clear();
        }
    }
    out.write_all(buf.as_bytes()).unwrap();
    out.flush().unwrap();
}

fn layout_component(
    members: &[u32],
    adj: &[u32],
    off: &[usize],
    px: &mut [f32],
    py: &mut [f32],
    radius: f32,
    iters: usize,
) {
    let cn = members.len();
    let area = std::f32::consts::PI * radius * radius;
    let k = (area / cn as f32).sqrt(); // ideal edge length
    let k2 = k * k;

    let mut dx = vec![0f32; cn];
    let mut dy = vec![0f32; cn];

    let mut temp = radius * 0.2;
    let final_temp = (k * 0.01).max(1e-3);
    let cool = (final_temp / temp).powf(1.0 / iters as f32);

    let cell = k * 1.6;
    let inv_cell = 1.0 / cell;

    let mut cell_of = vec![0u32; cn];
    for _ in 0..iters {
        // grid bounds
        let (mut minx, mut miny) = (f32::MAX, f32::MAX);
        let (mut maxx, mut maxy) = (f32::MIN, f32::MIN);
        for &g in members {
            minx = minx.min(px[g as usize]);
            miny = miny.min(py[g as usize]);
            maxx = maxx.max(px[g as usize]);
            maxy = maxy.max(py[g as usize]);
        }
        let gw = (((maxx - minx) * inv_cell) as usize + 2).max(1);
        let gh = (((maxy - miny) * inv_cell) as usize + 2).max(1);
        let ncells = gw * gh;
        let mut counts = vec![0u32; ncells + 1];
        for (li, &g) in members.iter().enumerate() {
            let cxi = (((px[g as usize] - minx) * inv_cell) as usize).min(gw - 1);
            let cyi = (((py[g as usize] - miny) * inv_cell) as usize).min(gh - 1);
            let c = (cyi * gw + cxi) as u32;
            cell_of[li] = c;
            counts[c as usize + 1] += 1;
        }
        for i in 0..ncells {
            counts[i + 1] += counts[i];
        }
        let mut bucket = vec![0u32; cn];
        let mut fill: Vec<u32> = counts[..ncells].to_vec();
        for li in 0..cn {
            let c = cell_of[li] as usize;
            bucket[fill[c] as usize] = li as u32;
            fill[c] += 1;
        }

        for d in dx.iter_mut() {
            *d = 0.0;
        }
        for d in dy.iter_mut() {
            *d = 0.0;
        }

        // Repulsion from nodes in the 3x3 neighbouring cells
        for li in 0..cn {
            let g = members[li] as usize;
            let (x, y) = (px[g], py[g]);
            let c = cell_of[li] as usize;
            let cx = c % gw;
            let cy = c / gw;
            let x0 = cx.saturating_sub(1);
            let x1 = (cx + 1).min(gw - 1);
            let y0 = cy.saturating_sub(1);
            let y1 = (cy + 1).min(gh - 1);
            let (mut fx, mut fy) = (0f32, 0f32);
            for yy in y0..=y1 {
                for xx in x0..=x1 {
                    let cc = yy * gw + xx;
                    let s = counts[cc] as usize;
                    let e = counts[cc + 1] as usize;
                    let cnt = e - s;
                    // Cap per-cell work: sample up to MAXS nodes and scale the force,
                    // so dense blobs stay O(1) per neighbouring cell.
                    const MAXS: usize = 12;
                    let (stride, scale) = if cnt > MAXS {
                        (cnt / MAXS, cnt as f32 / MAXS as f32)
                    } else {
                        (1, 1.0f32)
                    };
                    let mut idx = s + (li % stride.max(1));
                    let mut taken = 0usize;
                    while idx < e && taken < MAXS {
                        let oj = bucket[idx] as usize;
                        idx += stride;
                        taken += 1;
                        if oj == li {
                            continue;
                        }
                        let og = members[oj] as usize;
                        let ddx = x - px[og];
                        let ddy = y - py[og];
                        let d2 = (ddx * ddx + ddy * ddy).max(1e-4);
                        let f = scale * k2 / d2;
                        fx += ddx * f;
                        fy += ddy * f;
                    }
                }
            }
            dx[li] += fx;
            dy[li] += fy;
        }

        // Attraction along edges
        for li in 0..cn {
            let g = members[li] as usize;
            let (x, y) = (px[g], py[g]);
            let (mut fx, mut fy) = (0f32, 0f32);
            for &v in &adj[off[g]..off[g + 1]] {
                let ddx = x - px[v as usize];
                let ddy = y - py[v as usize];
                let dist = (ddx * ddx + ddy * ddy).sqrt().max(1e-3);
                let f = dist / k;
                fx -= ddx * f;
                fy -= ddy * f;
            }
            dx[li] += fx;
            dy[li] += fy;
        }

        // Apply displacements, capped by temperature
        for li in 0..cn {
            let g = members[li] as usize;
            let (fx, fy) = (dx[li], dy[li]);
            let len = (fx * fx + fy * fy).sqrt();
            if len > 1e-9 {
                let step = len.min(temp);
                px[g] += fx / len * step;
                py[g] += fy / len * step;
            }
        }
        temp *= cool;
    }
}
