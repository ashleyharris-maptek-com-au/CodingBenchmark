const double EPS = 1e-7;

struct Vector3 {
    double x, y, z;
    Vector3() : x(0), y(0), z(0) {}
    Vector3(double x, double y, double z) : x(x), y(y), z(z) {}
    Vector3 operator+(const Vector3& o) const { return Vector3(x + o.x, y + o.y, z + o.z); }
    Vector3 operator-(const Vector3& o) const { return Vector3(x - o.x, y - o.y, z - o.z); }
    Vector3 operator*(double s) const { return Vector3(x * s, y * s, z * s); }
    double length() const { return sqrt(x*x + y*y + z*z); }
    void normalize() {
        double l = length();
        if (l > 1e-15) { x /= l; y /= l; z /= l; }
    }
    double operator[](int i) const {
        if (i == 0) return x;
        if (i == 1) return y;
        return z;
    }
};

double dot(const Vector3& a, const Vector3& b) {
    return a.x*b.x + a.y*b.y + a.z*b.z;
}

Vector3 cross(const Vector3& a, const Vector3& b) {
    return Vector3(a.y*b.z - a.z*b.y, a.z*b.x - a.x*b.z, a.x*b.y - a.y*b.x);
}

struct Ray {
    Vector3 origin, dir, inv_dir;
    Ray(Vector3 o, Vector3 d) : origin(o), dir(d) {
        inv_dir.x = 1.0 / (d.x != 0 ? d.x : 1e-15);
        inv_dir.y = 1.0 / (d.y != 0 ? d.y : 1e-15);
        inv_dir.z = 1.0 / (d.z != 0 ? d.z : 1e-15);
    }
};

struct BBox {
    Vector3 min_p, max_p;
    BBox() : min_p(1e15, 1e15, 1e15), max_p(-1e15, -1e15, -1e15) {}
    void expand(const Vector3& p) {
        min_p.x = min(min_p.x, p.x); max_p.x = max(max_p.x, p.x);
        min_p.y = min(min_p.y, p.y); max_p.y = max(max_p.y, p.y);
        min_p.z = min(min_p.z, p.z); max_p.z = max(max_p.z, p.z);
    }
    bool intersect(const BBox& o) const {
        if (max_p.x < o.min_p.x - EPS || min_p.x > o.max_p.x + EPS) return false;
        if (max_p.y < o.min_p.y - EPS || min_p.y > o.max_p.y + EPS) return false;
        if (max_p.z < o.min_p.z - EPS || min_p.z > o.max_p.z + EPS) return false;
        return true;
    }
    bool intersect(const Ray& ray) const {
        double t1 = (min_p.x - ray.origin.x) * ray.inv_dir.x;
        double t2 = (max_p.x - ray.origin.x) * ray.inv_dir.x;
        double tmin = min(t1, t2), tmax = max(t1, t2);
        t1 = (min_p.y - ray.origin.y) * ray.inv_dir.y;
        t2 = (max_p.y - ray.origin.y) * ray.inv_dir.y;
        tmin = max(tmin, min(t1, t2)); tmax = min(tmax, max(t1, t2));
        t1 = (min_p.z - ray.origin.z) * ray.inv_dir.z;
        t2 = (max_p.z - ray.origin.z) * ray.inv_dir.z;
        tmin = max(tmin, min(t1, t2)); tmax = min(tmax, max(t1, t2));
        return tmax >= max(0.0, tmin);
    }
};

struct Plane {
    Vector3 n;
    double d;
    Plane(Vector3 n_, double d_) : n(n_), d(d_) { normalize(); }
    Plane(Vector3 n_, Vector3 p_) : n(n_), d(-dot(n_, p_)) { normalize(); }
    Plane(Vector3 a, Vector3 b, Vector3 c) {
        n = cross(b - a, c - a);
        d = -dot(n, a);
        normalize();
    }
    void normalize() {
        double len = n.length();
        if (len > 1e-15) { n = n * (1.0 / len); d /= len; }
    }
    double dist(const Vector3& p) const { return dot(n, p) + d; }
};

struct Triangle {
    Vector3 a, b, c;
    Vector3 normal;
    Vector3 centroid;
    BBox bbox;
    void init() {
        normal = cross(b - a, c - a);
        normal.normalize();
        centroid = (a + b + c) * (1.0 / 3.0);
        bbox = BBox();
        bbox.expand(a); bbox.expand(b); bbox.expand(c);
        bbox.min_p = bbox.min_p - Vector3(EPS, EPS, EPS);
        bbox.max_p = bbox.max_p + Vector3(EPS, EPS, EPS);
    }
};

struct Mesh {
    int V, F;
    vector<Vector3> vertices;
    vector<int> indices;
    vector<Triangle> triangles;
};

struct BVHNode {
    BBox box;
    int left = -1, right = -1;
    int first_tri = -1, num_tris = 0;
};

struct BVH {
    vector<BVHNode> nodes;
    vector<int> tri_indices;
    int root = -1;

    int build_recursive(int first, int num, const Mesh& mesh) {
        int node_idx = nodes.size();
        nodes.push_back(BVHNode());
        
        BBox box;
        for(int i=0; i<num; ++i) {
            box.expand(mesh.triangles[tri_indices[first+i]].bbox.min_p);
            box.expand(mesh.triangles[tri_indices[first+i]].bbox.max_p);
        }
        nodes[node_idx].box = box;
        
        if (num <= 8) {
            nodes[node_idx].first_tri = first;
            nodes[node_idx].num_tris = num;
            return node_idx;
        }
        
        int axis = 0;
        Vector3 extent = box.max_p - box.min_p;
        if (extent.y > extent.x) axis = 1;
        if (extent.z > extent[axis]) axis = 2;
        
        double split_pos = box.min_p[axis] + extent[axis] * 0.5;
        
        int mid = first;
        for(int i=first; i<first+num; ++i) {
            if (mesh.triangles[tri_indices[i]].centroid[axis] < split_pos) {
                swap(tri_indices[i], tri_indices[mid]);
                mid++;
            }
        }
        
        if (mid == first || mid == first+num) mid = first + num/2;
        
        nodes[node_idx].left = build_recursive(first, mid - first, mesh);
        nodes[node_idx].right = build_recursive(mid, first + num - mid, mesh);
        return node_idx;
    }

    void build(const Mesh& mesh) {
        tri_indices.resize(mesh.F);
        for(int i=0; i<mesh.F; ++i) tri_indices[i] = i;
        if (mesh.F > 0) root = build_recursive(0, mesh.F, mesh);
    }
};

void bvh_query(int node_idx, const BBox& box, const BVH& bvh, vector<int>& out) {
    if (node_idx == -1 || !bvh.nodes[node_idx].box.intersect(box)) return;
    if (bvh.nodes[node_idx].num_tris > 0) {
        for(int i=0; i<bvh.nodes[node_idx].num_tris; ++i) {
            out.push_back(bvh.tri_indices[bvh.nodes[node_idx].first_tri + i]);
        }
    } else {
        bvh_query(bvh.nodes[node_idx].left, box, bvh, out);
        bvh_query(bvh.nodes[node_idx].right, box, bvh, out);
    }
}

bool ray_intersect_tri(const Ray& ray, const Triangle& tri, double& t) {
    Vector3 edge1 = tri.b - tri.a;
    Vector3 edge2 = tri.c - tri.a;
    Vector3 h = cross(ray.dir, edge2);
    double a = dot(edge1, h);
    if (a > -EPS && a < EPS) return false;
    double f = 1.0 / a;
    Vector3 s = ray.origin - tri.a;
    double u = f * dot(s, h);
    if (u < 0.0 || u > 1.0) return false;
    Vector3 q = cross(s, edge1);
    double v = f * dot(ray.dir, q);
    if (v < 0.0 || u + v > 1.0) return false;
    t = f * dot(edge2, q);
    return t > EPS;
}

struct Hit {
    bool valid = false;
    double t = 1e15;
    int tri_idx = -1;
};

void bvh_raycast(int node_idx, const Ray& ray, const BVH& bvh, const Mesh& mesh, Hit& closest) {
    if (node_idx == -1 || !bvh.nodes[node_idx].box.intersect(ray)) return;
    if (bvh.nodes[node_idx].num_tris > 0) {
        for(int i=0; i<bvh.nodes[node_idx].num_tris; ++i) {
            int tri_idx = bvh.tri_indices[bvh.nodes[node_idx].first_tri + i];
            double t;
            if (ray_intersect_tri(ray, mesh.triangles[tri_idx], t)) {
                if (t < closest.t) {
                    closest.t = t;
                    closest.tri_idx = tri_idx;
                    closest.valid = true;
                }
            }
        }
    } else {
        bvh_raycast(bvh.nodes[node_idx].left, ray, bvh, mesh, closest);
        bvh_raycast(bvh.nodes[node_idx].right, ray, bvh, mesh, closest);
    }
}

double dist_point_segment(Vector3 p, Vector3 v, Vector3 w) {
    Vector3 vw = w - v;
    double l2 = dot(vw, vw);
    if (l2 == 0) return (p - v).length();
    double t = dot(p - v, vw) / l2;
    t = max(0.0, min(1.0, t));
    Vector3 proj = v + vw * t;
    return (p - proj).length();
}

Vector3 random_dir() {
    double z = ((double)rand() / RAND_MAX) * 2.0 - 1.0;
    double a = ((double)rand() / RAND_MAX) * 2.0 * 3.14159265359;
    double r = sqrt(max(0.0, 1.0 - z*z));
    Vector3 dir(r * cos(a), r * sin(a), z);
    dir.normalize();
    return dir;
}

bool is_inside(Vector3 p, const Mesh& mesh, const BVH& bvh) {
    int attempts = 0;
    while (attempts++ < 50) {
        Vector3 dir = random_dir();
        Ray ray(p, dir);
        Hit hit;
        bvh_raycast(bvh.root, ray, bvh, mesh, hit);
        if (!hit.valid) return false;
        
        Vector3 hit_p = ray.origin + ray.dir * hit.t;
        const Triangle& tri = mesh.triangles[hit.tri_idx];
        
        double d_edge = min({
            dist_point_segment(hit_p, tri.a, tri.b),
            dist_point_segment(hit_p, tri.b, tri.c),
            dist_point_segment(hit_p, tri.c, tri.a)
        });
        if (d_edge < 1e-4) continue;
        
        return dot(dir, tri.normal) > 0;
    }
    return false;
}

bool is_point_in_triangle_3d(Vector3 p, const Triangle& t) {
    Vector3 u = t.b - t.a;
    Vector3 v = t.c - t.a;
    Vector3 w = p - t.a;
    double uu = dot(u,u), uv = dot(u,v), vv = dot(v,v);
    double wu = dot(w,u), wv = dot(w,v);
    double D = uv * uv - uu * vv;
    if (abs(D) < 1e-12) return false;
    double s = (uv * wv - vv * wu) / D;
    if (s < -1e-5 || s > 1.0 + 1e-5) return false;
    double t_val = (uv * wu - uu * wv) / D;
    if (t_val < -1e-5 || (s + t_val) > 1.0 + 1e-5) return false;
    return true;
}

bool is_coplanar_tri_tri(const Triangle& t1, const Triangle& t2) {
    Plane p2(t2.a, t2.b, t2.c);
    if (abs(p2.dist(t1.a)) > EPS || abs(p2.dist(t1.b)) > EPS || abs(p2.dist(t1.c)) > EPS) return false;
    return true;
}

bool tri_tri_intersect(const Triangle& t1, const Triangle& t2, bool& is_coplanar) {
    Plane p1(t1.a, t1.b, t1.c);
    Plane p2(t2.a, t2.b, t2.c);
    
    double d1[3] = {p2.dist(t1.a), p2.dist(t1.b), p2.dist(t1.c)};
    if ((d1[0]>EPS && d1[1]>EPS && d1[2]>EPS) || (d1[0]<-EPS && d1[1]<-EPS && d1[2]<-EPS)) return false;
    
    double d2[3] = {p1.dist(t2.a), p1.dist(t2.b), p1.dist(t2.c)};
    if ((d2[0]>EPS && d2[1]>EPS && d2[2]>EPS) || (d2[0]<-EPS && d2[1]<-EPS && d2[2]<-EPS)) return false;
    
    if (abs(d1[0])<=EPS && abs(d1[1])<=EPS && abs(d1[2])<=EPS) {
        is_coplanar = true;
        return true; 
    }
    is_coplanar = false;
    
    Vector3 v1[3] = {t1.a, t1.b, t1.c};
    vector<Vector3> seg1;
    for(int i=0; i<3; ++i) {
        int j = (i+1)%3;
        if (d1[i]*d1[j] < -EPS) {
            seg1.push_back(v1[i] + (v1[j]-v1[i]) * (d1[i]/(d1[i]-d1[j])));
        } else if (abs(d1[i])<=EPS) {
            seg1.push_back(v1[i]);
        }
    }
    if (seg1.size() < 2) return false;
    
    Vector3 v2[3] = {t2.a, t2.b, t2.c};
    vector<Vector3> seg2;
    for(int i=0; i<3; ++i) {
        int j = (i+1)%3;
        if (d2[i]*d2[j] < -EPS) {
            seg2.push_back(v2[i] + (v2[j]-v2[i]) * (d2[i]/(d2[i]-d2[j])));
        } else if (abs(d2[i])<=EPS) {
            seg2.push_back(v2[i]);
        }
    }
    if (seg2.size() < 2) return false;
    
    Vector3 D = cross(p1.n, p2.n);
    int axis = 0;
    if (abs(D.y) > abs(D.x)) axis = 1;
    if (abs(D.z) > abs(D[axis])) axis = 2;
    
    double min1 = seg1[0][axis], max1 = seg1[0][axis];
    for(auto& p : seg1) { min1 = min(min1, p[axis]); max1 = max(max1, p[axis]); }
    
    double min2 = seg2[0][axis], max2 = seg2[0][axis];
    for(auto& p : seg2) { min2 = min(min2, p[axis]); max2 = max(max2, p[axis]); }
    
    if (max1 < min2 - EPS || max2 < min1 - EPS) return false;
    return true;
}

struct Fragment {
    vector<Vector3> v;
};

double polygon_area(const Fragment& f) {
    if (f.v.size() < 3) return 0;
    Vector3 normal(0,0,0);
    for(size_t i=0; i<f.v.size(); ++i) {
        normal = normal + cross(f.v[i], f.v[(i+1)%f.v.size()]);
    }
    return normal.length() * 0.5;
}

bool plane_intersects_bbox(const Plane& p, const Fragment& f) {
    BBox box;
    for(const Vector3& pt : f.v) box.expand(pt);
    Vector3 center = (box.min_p + box.max_p) * 0.5;
    Vector3 extents = box.max_p - center;
    double r = extents.x * abs(p.n.x) + extents.y * abs(p.n.y) + extents.z * abs(p.n.z);
    return abs(p.dist(center)) <= r + EPS;
}

void split_fragment(const Fragment& frag, const Plane& plane, vector<Fragment>& out) {
    Fragment front, back;
    int n = frag.v.size();
    if (n == 0) return;
    
    vector<double> d(n);
    for(int i=0; i<n; ++i) d[i] = plane.dist(frag.v[i]);
    
    for(int i=0; i<n; ++i) {
        int j = (i+1)%n;
        Vector3 p1 = frag.v[i], p2 = frag.v[j];
        double d1 = d[i], d2 = d[j];
        
        if (d1 >= -EPS) front.v.push_back(p1);
        if (d1 <= EPS) back.v.push_back(p1);
        
        if ((d1 > EPS && d2 < -EPS) || (d1 < -EPS && d2 > EPS)) {
            Vector3 p = p1 + (p2 - p1) * (d1 / (d1 - d2));
            front.v.push_back(p);
            back.v.push_back(p);
        }
    }
    
    if (front.v.size() >= 3) out.push_back(front);
    if (back.v.size() >= 3) out.push_back(back);
}

struct EdgeInfo {
    Vector3 A, B;
    vector<double> t_vals;
};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    Mesh meshA, meshB;
    if (!(cin >> meshA.V >> meshA.F >> meshB.V >> meshB.F)) return 0;
    
    meshA.vertices.resize(meshA.V);
    for(int i=0; i<meshA.V; ++i) cin >> meshA.vertices[i].x >> meshA.vertices[i].y >> meshA.vertices[i].z;
    meshA.indices.resize(meshA.F * 3);
    for(int i=0; i<meshA.F*3; ++i) cin >> meshA.indices[i];
    
    meshB.vertices.resize(meshB.V);
    for(int i=0; i<meshB.V; ++i) cin >> meshB.vertices[i].x >> meshB.vertices[i].y >> meshB.vertices[i].z;
    meshB.indices.resize(meshB.F * 3);
    for(int i=0; i<meshB.F*3; ++i) cin >> meshB.indices[i];
    
    meshA.triangles.resize(meshA.F);
    for(int i=0; i<meshA.F; ++i) {
        meshA.triangles[i].a = meshA.vertices[meshA.indices[i*3]];
        meshA.triangles[i].b = meshA.vertices[meshA.indices[i*3+1]];
        meshA.triangles[i].c = meshA.vertices[meshA.indices[i*3+2]];
        meshA.triangles[i].init();
    }
    
    meshB.triangles.resize(meshB.F);
    for(int i=0; i<meshB.F; ++i) {
        meshB.triangles[i].a = meshB.vertices[meshB.indices[i*3]];
        meshB.triangles[i].b = meshB.vertices[meshB.indices[i*3+1]];
        meshB.triangles[i].c = meshB.vertices[meshB.indices[i*3+2]];
        meshB.triangles[i].init();
    }
    
    BVH bvhA, bvhB;
    bvhA.build(meshA);
    bvhB.build(meshB);
    
    vector<vector<int>> intersecting_B_for_A(meshA.F);
    vector<vector<int>> intersecting_A_for_B(meshB.F);
    
    for(int i=0; i<meshA.F; ++i) {
        vector<int> b_candidates;
        bvh_query(bvhB.root, meshA.triangles[i].bbox, bvhB, b_candidates);
        for(int b_idx : b_candidates) {
            bool is_copl = false;
            if (tri_tri_intersect(meshA.triangles[i], meshB.triangles[b_idx], is_copl)) {
                intersecting_B_for_A[i].push_back(b_idx);
                intersecting_A_for_B[b_idx].push_back(i);
            }
        }
    }
    
    vector<vector<Fragment>> fragmentsA(meshA.F);
    vector<vector<Fragment>> fragmentsB(meshB.F);
    
    auto process_mesh = [&](Mesh& mesh, Mesh& other_mesh, vector<vector<int>>& intersecting, vector<vector<Fragment>>& fragments) {
        for(int i=0; i<mesh.F; ++i) {
            Triangle T = mesh.triangles[i];
            vector<Plane> planes;
            for(int o_idx : intersecting[i]) {
                const Triangle& o = other_mesh.triangles[o_idx];
                if (is_coplanar_tri_tri(T, o)) {
                    Plane p(o.a, o.b, o.c);
                    Vector3 d1 = o.b - o.a; planes.push_back(Plane(cross(d1, p.n), o.a));
                    Vector3 d2 = o.c - o.b; planes.push_back(Plane(cross(d2, p.n), o.b));
                    Vector3 d3 = o.a - o.c; planes.push_back(Plane(cross(d3, p.n), o.c));
                } else {
                    planes.push_back(Plane(o.a, o.b, o.c));
                }
            }
            vector<Plane> dedup;
            for(const Plane& p : planes) {
                bool dup = false;
                for(const Plane& op : dedup) {
                    if (abs(dot(p.n, op.n)) > 1.0 - 1e-6 && abs(p.d - op.d) < 1e-5) { dup = true; break; }
                    if (abs(dot(p.n, op.n)) < -1.0 + 1e-6 && abs(p.d + op.d) < 1e-5) { dup = true; break; }
                }
                if (!dup) dedup.push_back(p);
            }
            
            vector<Fragment> frags = { {{T.a, T.b, T.c}} };
            for(const Plane& p : dedup) {
                vector<Fragment> next_frags;
                for(const Fragment& f : frags) {
                    if (!plane_intersects_bbox(p, f)) {
                        next_frags.push_back(f);
                    } else {
                        split_fragment(f, p, next_frags);
                    }
                }
                frags = next_frags;
            }
            for(const auto& f : frags) {
                if (polygon_area(f) > 1e-10) fragments[i].push_back(f);
            }
        }
    };
    
    process_mesh(meshA, meshB, intersecting_B_for_A, fragmentsA);
    process_mesh(meshB, meshA, intersecting_A_for_B, fragmentsB);
    
    auto gather_and_inject_edges = [&](Mesh& mesh, vector<vector<Fragment>>& fragments) {
        map<pair<int,int>, EdgeInfo> edges;
        for(int i=0; i<mesh.F; ++i) {
            int v[3] = {mesh.indices[i*3], mesh.indices[i*3+1], mesh.indices[i*3+2]};
            for(int j=0; j<3; ++j) {
                int v1 = v[j], v2 = v[(j+1)%3];
                if (v1 > v2) swap(v1, v2);
                if (edges.find({v1, v2}) == edges.end()) {
                    edges[{v1, v2}] = {mesh.vertices[v1], mesh.vertices[v2], {0.0, 1.0}};
                }
            }
        }
        for(int i=0; i<mesh.F; ++i) {
            int v[3] = {mesh.indices[i*3], mesh.indices[i*3+1], mesh.indices[i*3+2]};
            for(int j=0; j<3; ++j) {
                int v1 = v[j], v2 = v[(j+1)%3];
                if (v1 > v2) swap(v1, v2);
                EdgeInfo& E = edges[{v1, v2}];
                for(const Fragment& f : fragments[i]) {
                    for(Vector3 p : f.v) {
                        double len2 = dot(E.B - E.A, E.B - E.A);
                        if (len2 == 0) continue;
                        double t = dot(p - E.A, E.B - E.A) / len2;
                        if (t >= -1e-5 && t <= 1.0 + 1e-5) {
                            Vector3 proj = E.A + (E.B - E.A) * t;
                            if ((p - proj).length() < 1e-5) E.t_vals.push_back(t);
                        }
                    }
                }
            }
        }
        for(auto& pair : edges) {
            auto& t_vals = pair.second.t_vals;
            sort(t_vals.begin(), t_vals.end());
            vector<double> unique_t;
            for(double t : t_vals) {
                if (unique_t.empty() || t - unique_t.back() > 1e-5) unique_t.push_back(t);
            }
            t_vals = unique_t;
        }
        for(int i=0; i<mesh.F; ++i) {
            int v_idx[3] = {mesh.indices[i*3], mesh.indices[i*3+1], mesh.indices[i*3+2]};
            for(Fragment& f : fragments[i]) {
                vector<Vector3> new_v;
                int n = f.v.size();
                for(int k=0; k<n; ++k) {
                    Vector3 p1 = f.v[k];
                    Vector3 p2 = f.v[(k+1)%n];
                    new_v.push_back(p1);
                    for(int j=0; j<3; ++j) {
                        int v1 = v_idx[j], v2 = v_idx[(j+1)%3];
                        if (v1 > v2) swap(v1, v2);
                        EdgeInfo& E = edges[{v1, v2}];
                        double len2 = dot(E.B - E.A, E.B - E.A);
                        if (len2 == 0) continue;
                        double t1 = dot(p1 - E.A, E.B - E.A) / len2;
                        double t2 = dot(p2 - E.A, E.B - E.A) / len2;
                        bool p1_on = (t1 >= -1e-5 && t1 <= 1.0 + 1e-5 && (p1 - (E.A + (E.B - E.A)*t1)).length() < 1e-5);
                        bool p2_on = (t2 >= -1e-5 && t2 <= 1.0 + 1e-5 && (p2 - (E.A + (E.B - E.A)*t2)).length() < 1e-5);
                        if (p1_on && p2_on) {
                            if (t1 < t2) {
                                for(double t : E.t_vals) {
                                    if (t > t1 + 1e-5 && t < t2 - 1e-5) new_v.push_back(E.A + (E.B - E.A) * t);
                                }
                            } else {
                                for(int m = (int)E.t_vals.size()-1; m >= 0; --m) {
                                    double t = E.t_vals[m];
                                    if (t < t1 - 1e-5 && t > t2 + 1e-5) new_v.push_back(E.A + (E.B - E.A) * t);
                                }
                            }
                            break;
                        }
                    }
                }
                f.v = new_v;
            }
        }
    };
    
    gather_and_inject_edges(meshA, fragmentsA);
    gather_and_inject_edges(meshB, fragmentsB);
    
    vector<Triangle> final_tris;
    
    auto add_fragment = [&](const Fragment& f) {
        for(size_t i=1; i<f.v.size()-1; ++i) {
            Triangle t;
            t.a = f.v[0]; t.b = f.v[i]; t.c = f.v[i+1];
            if (polygon_area({{t.a, t.b, t.c}}) > 1e-10) {
                final_tris.push_back(t);
            }
        }
    };
    
    for(int i=0; i<meshA.F; ++i) {
        for(const Fragment& f : fragmentsA[i]) {
            Vector3 centroid(0,0,0);
            for(Vector3 p : f.v) centroid = centroid + p;
            centroid = centroid * (1.0 / f.v.size());
            
            bool on_boundary = false;
            bool keep = false;
            for(int b_idx : intersecting_B_for_A[i]) {
                if (is_coplanar_tri_tri(meshA.triangles[i], meshB.triangles[b_idx])) {
                    if (is_point_in_triangle_3d(centroid, meshB.triangles[b_idx])) {
                        on_boundary = true;
                        if (dot(meshA.triangles[i].normal, meshB.triangles[b_idx].normal) > 0) keep = true;
                        else keep = false;
                        break;
                    }
                }
            }
            
            if (on_boundary) {
                if (keep) add_fragment(f);
            } else {
                if (!is_inside(centroid, meshB, bvhB)) add_fragment(f);
            }
        }
    }
    
    for(int i=0; i<meshB.F; ++i) {
        for(const Fragment& f : fragmentsB[i]) {
            Vector3 centroid(0,0,0);
            for(Vector3 p : f.v) centroid = centroid + p;
            centroid = centroid * (1.0 / f.v.size());
            
            bool on_boundary = false;
            bool keep = false;
            for(int a_idx : intersecting_A_for_B[i]) {
                if (is_coplanar_tri_tri(meshB.triangles[i], meshA.triangles[a_idx])) {
                    if (is_point_in_triangle_3d(centroid, meshA.triangles[a_idx])) {
                        on_boundary = true;
                        keep = false;
                        break;
                    }
                }
            }
            
            if (on_boundary) {
                if (keep) add_fragment(f);
            } else {
                if (!is_inside(centroid, meshA, bvhA)) add_fragment(f);
            }
        }
    }
    
    struct Vec3Hash {
        size_t operator()(const Vector3& k) const {
            long long x = round(k.x * 10000.0);
            long long y = round(k.y * 10000.0);
            long long z = round(k.z * 10000.0);
            return (x * 73856093) ^ (y * 19349663) ^ (z * 83492791);
        }
    };
    struct Vec3Equal {
        bool operator()(const Vector3& a, const Vector3& b) const {
            return (a - b).length() < 1e-4;
        }
    };
    
    unordered_map<Vector3, int, Vec3Hash, Vec3Equal> vertex_map;
    vector<Vector3> out_vertices;
    vector<int> out_indices;
    
    for(const Triangle& t : final_tris) {
        int idx[3];
        Vector3 v[3] = {t.a, t.b, t.c};
        for(int i=0; i<3; ++i) {
            if (vertex_map.find(v[i]) == vertex_map.end()) {
                vertex_map[v[i]] = out_vertices.size();
                out_vertices.push_back(v[i]);
            }
            idx[i] = vertex_map[v[i]];
        }
        if (idx[0] != idx[1] && idx[1] != idx[2] && idx[2] != idx[0]) {
            out_indices.push_back(idx[0]);
            out_indices.push_back(idx[1]);
            out_indices.push_back(idx[2]);
        }
    }
    
    cout << out_vertices.size() << " " << out_indices.size() / 3 << "\n";
    for(const Vector3& v : out_vertices) {
        cout << fixed << setprecision(6) << v.x << " " << v.y << " " << v.z << "\n";
    }
    for(size_t i=0; i<out_indices.size(); i+=3) {
        cout << out_indices[i] << " " << out_indices[i+1] << " " << out_indices[i+2] << "\n";
    }
    
    return 0;
}
