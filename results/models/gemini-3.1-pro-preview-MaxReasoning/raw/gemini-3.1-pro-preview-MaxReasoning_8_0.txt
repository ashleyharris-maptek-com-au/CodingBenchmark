use std::io::{BufRead, Write};

fn write_num(buf: &mut Vec<u8>, mut n: u32) {
    if n == 0 {
        buf.push(b'0');
        return;
    }
    let mut temp = [0u8; 10];
    let mut i = 0;
    while n > 0 {
        temp[i] = b'0' + (n % 10) as u8;
        n /= 10;
        i += 1;
    }
    while i > 0 {
        i -= 1;
        buf.push(temp[i]);
    }
}

fn main() {
    let stdin = std::io::stdin();
    let mut reader = stdin.lock();

    let mut header = String::new();
    loop {
        header.clear();
        let bytes = reader.read_line(&mut header).expect("Failed to read header");
        if bytes == 0 {
            println!("0");
            return;
        }
        if header.trim().is_empty() {
            continue;
        }
        break;
    }

    let mut parts = header.split_whitespace();
    let h_str = parts.next().expect("Missing H");
    let w_str = parts.next().expect("Missing W");
    let h: usize = h_str.parse().expect("Invalid H");
    let w: usize = w_str.parse().expect("Invalid W");

    if h == 0 || w == 0 {
        println!("0");
        return;
    }

    let s = w + 1;
    let mut grid = Vec::with_capacity(h * s);
    let mut start_idx = u32::MAX;
    let mut end_idx = u32::MAX;

    let mut row_buf = Vec::with_capacity(w + 10);
    for _ in 0..h {
        row_buf.clear();
        let bytes = reader.read_until(b'\n', &mut row_buf).expect("Failed to read row");
        if bytes == 0 {
            break;
        }
        let mut cols = 0;
        for &c in &row_buf {
            if c == b' ' || c == b'#' || c == b'A' || c == b'B' {
                if c == b'A' {
                    start_idx = grid.len() as u32;
                } else if c == b'B' {
                    end_idx = grid.len() as u32;
                }
                grid.push(c);
                cols += 1;
                if cols == w {
                    break;
                }
            }
        }
        while cols < w {
            grid.push(b'#');
            cols += 1;
        }
        grid.push(b'#');
    }

    if start_idx == u32::MAX || end_idx == u32::MAX {
        println!("0");
        return;
    }

    let mut current_frontier = Vec::with_capacity(1_000_000);
    let mut next_frontier = Vec::with_capacity(1_000_000);
    current_frontier.push(start_idx);

    let mut found = false;
    let su32 = s as u32;

    'bfs: while !current_frontier.is_empty() {
        for &curr in &current_frontier {
            if curr == end_idx {
                found = true;
                break 'bfs;
            }

            // Up
            if curr >= su32 {
                let nxt = curr - su32;
                let c = grid[nxt as usize];
                if c == b' ' || c == b'B' {
                    grid[nxt as usize] = b'D';
                    next_frontier.push(nxt);
                    if nxt == end_idx {
                        found = true;
                        break 'bfs;
                    }
                }
            }
            // Down
            let nxt_d = curr + su32;
            if (nxt_d as usize) < grid.len() {
                let c = grid[nxt_d as usize];
                if c == b' ' || c == b'B' {
                    grid[nxt_d as usize] = b'U';
                    next_frontier.push(nxt_d);
                    if nxt_d == end_idx {
                        found = true;
                        break 'bfs;
                    }
                }
            }
            // Left
            if curr > 0 {
                let nxt = curr - 1;
                let c = grid[nxt as usize];
                if c == b' ' || c == b'B' {
                    grid[nxt as usize] = b'R';
                    next_frontier.push(nxt);
                    if nxt == end_idx {
                        found = true;
                        break 'bfs;
                    }
                }
            }
            // Right
            let nxt_r = curr + 1;
            if (nxt_r as usize) < grid.len() {
                let c = grid[nxt_r as usize];
                if c == b' ' || c == b'B' {
                    grid[nxt_r as usize] = b'L';
                    next_frontier.push(nxt_r);
                    if nxt_r == end_idx {
                        found = true;
                        break 'bfs;
                    }
                }
            }
        }
        current_frontier.clear();
        std::mem::swap(&mut current_frontier, &mut next_frontier);
    }

    if !found {
        println!("0");
        return;
    }

    let mut path = Vec::with_capacity(1_000_000);
    let mut curr = end_idx;
    while curr != start_idx {
        path.push(curr);
        match grid[curr as usize] {
            b'D' => curr += su32,
            b'U' => curr -= su32,
            b'R' => curr += 1,
            b'L' => curr -= 1,
            _ => {
                println!("0");
                return;
            }
        }
    }
    path.push(start_idx);
    path.reverse();

    let mut stdout = std::io::stdout().lock();
    let mut out_buf = Vec::with_capacity(65536);

    write_num(&mut out_buf, path.len() as u32);
    out_buf.push(b'\n');

    for idx in path {
        let x = idx % su32;
        let y = idx / su32;
        write_num(&mut out_buf, x);
        out_buf.push(b' ');
        write_num(&mut out_buf, y);
        out_buf.push(b'\n');

        if out_buf.len() > 60000 {
            stdout.write_all(&out_buf).expect("Failed to write to stdout");
            out_buf.clear();
        }
    }
    if !out_buf.is_empty() {
        stdout.write_all(&out_buf).expect("Failed to write to stdout");
    }
}
