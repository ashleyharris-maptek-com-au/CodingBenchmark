use std::io::{self, BufRead};

#[derive(Clone)]
struct Piece {
    id: usize,
    vertices: Vec<(f64, f64)>,
    area: f64,
}

struct Stock {
    id: usize,
    vertices: Vec<(f64, f64)>,
    placed_pieces: Vec<Vec<(f64, f64)>>,
    grid: Vec<bool>,
    free_area: f64,
    min_x: f64, max_x: f64,
    min_y: f64, max_y: f64,
    dx: f64, dy: f64,
    gw: usize, gh: usize,
}

fn polygon_area(poly: &[(f64, f64)]) -> f64 {
    let mut area = 0.0;
    for i in 0..poly.len() {
        let a = poly[i];
        let b = poly[(i+1)%poly.len()];
        area += a.0 * b.1 - a.1 * b.0;
    }
    (area / 2.0).abs()
}

fn ensure_ccw(poly: &mut Vec<(f64, f64)>) {
    let mut area = 0.0;
    for i in 0..poly.len() {
        let a = poly[i];
        let b = poly[(i+1)%poly.len()];
        area += a.0 * b.1 - a.1 * b.0;
    }
    if area < 0.0 {
        poly.reverse();
    }
}

fn point_in_polygon(pt: (f64, f64), poly: &[(f64, f64)]) -> bool {
    let mut inside = false;
    let (x, y) = pt;
    let mut j = poly.len() - 1;
    for i in 0..poly.len() {
        let (xi, yi) = poly[i];
        let (xj, yj) = poly[j];
        
        if (yi > y) != (yj > y) {
            let intersect_x = (xj - xi) * (y - yi) / (yj - yi) + xi;
            if x < intersect_x {
                inside = !inside;
            }
        }
        j = i;
    }
    inside
}

fn ccw(a: (f64, f64), b: (f64, f64), c: (f64, f64)) -> f64 {
    (b.0 - a.0) * (c.1 - a.1) - (b.1 - a.1) * (c.0 - a.0)
}

fn sign(val: f64, max_abs: f64) -> i32 {
    let eps = 1e-9 * max_abs.max(1.0);
    if val > eps { 1 }
    else if val < -eps { -1 }
    else { 0 }
}

fn segments_intersect(a: (f64, f64), b: (f64, f64), c: (f64, f64), d: (f64, f64)) -> bool {
    let cp1 = ccw(a, b, c);
    let cp2 = ccw(a, b, d);
    let cp3 = ccw(c, d, a);
    let cp4 = ccw(c, d, b);
    
    let m1 = a.0.abs().max(b.0.abs()).max(c.0.abs()).max(a.1.abs()).max(b.1.abs()).max(c.1.abs());
    let m2 = a.0.abs().max(b.0.abs()).max(d.0.abs()).max(a.1.abs()).max(b.1.abs()).max(d.1.abs());
    let m3 = c.0.abs().max(d.0.abs()).max(a.0.abs()).max(c.1.abs()).max(d.1.abs()).max(a.1.abs());
    let m4 = c.0.abs().max(d.0.abs()).max(b.0.abs()).max(c.1.abs()).max(d.1.abs()).max(b.1.abs());
    
    let s1 = sign(cp1, m1 * m1);
    let s2 = sign(cp2, m2 * m2);
    let s3 = sign(cp3, m3 * m3);
    let s4 = sign(cp4, m4 * m4);
    
    if s1 != 0 && s2 != 0 && s1 != s2 && s3 != 0 && s4 != 0 && s3 != s4 {
        return true;
    }
    
    false
}

fn aabb_intersect(p1: &[(f64, f64)], p2: &[(f64, f64)]) -> bool {
    let mut min_x1 = f64::MAX; let mut max_x1 = f64::MIN;
    let mut min_y1 = f64::MAX; let mut max_y1 = f64::MIN;
    for &(x, y) in p1 {
        min_x1 = min_x1.min(x); max_x1 = max_x1.max(x);
        min_y1 = min_y1.min(y); max_y1 = max_y1.max(y);
    }
    let mut min_x2 = f64::MAX; let mut max_x2 = f64::MIN;
    let mut min_y2 = f64::MAX; let mut max_y2 = f64::MIN;
    for &(x, y) in p2 {
        min_x2 = min_x2.min(x); max_x2 = max_x2.max(x);
        min_y2 = min_y2.min(y); max_y2 = max_y2.max(y);
    }
    
    let max_abs = max_x1.abs().max(min_x1.abs()).max(max_y1.abs()).max(min_y1.abs());
    let eps = 1e-7 * max_abs.max(1.0);
    max_x1 > min_x2 + eps && max_x2 > min_x1 + eps &&
    max_y1 > min_y2 + eps && max_y2 > min_y1 + eps
}

fn polygons_overlap(p1: &[(f64, f64)], p2: &[(f64, f64)]) -> bool {
    if !aabb_intersect(p1, p2) { return false; }
    
    for i in 0..p1.len() {
        let a = p1[i];
        let b = p1[(i+1)%p1.len()];
        for j in 0..p2.len() {
            let c = p2[j];
            let d = p2[(j+1)%p2.len()];
            if segments_intersect(a, b, c, d) {
                return true;
            }
        }
    }
    
    let pairs = [(p1, p2), (p2, p1)];
    for &(p, other) in pairs.iter() {
        for i in 0..p.len() {
            let a = p[i];
            let b = p[(i+1)%p.len()];
            let mid = ((a.0+b.0)/2.0, (a.1+b.1)/2.0);
            let dx = b.0 - a.0;
            let dy = b.1 - a.1;
            let len = (dx*dx + dy*dy).sqrt();
            if len > 0.0 {
                let nx = -dy / len;
                let ny = dx / len;
                let eps = 1e-5 * len.max(1.0);
                let test_pt = (mid.0 + nx * eps, mid.1 + ny * eps);
                if point_in_polygon(test_pt, other) {
                    return true;
                }
            }
        }
    }
    
    false
}

fn polygon_inside_stock(p: &[(f64, f64)], stock: &[(f64, f64)]) -> bool {
    for i in 0..p.len() {
        let a = p[i];
        let b = p[(i+1)%p.len()];
        for j in 0..stock.len() {
            let c = stock[j];
            let d = stock[(j+1)%stock.len()];
            if segments_intersect(a, b, c, d) {
                return false;
            }
        }
    }
    
    for i in 0..p.len() {
        let a = p[i];
        let b = p[(i+1)%p.len()];
        let mid = ((a.0+b.0)/2.0, (a.1+b.1)/2.0);
        let dx = b.0 - a.0;
        let dy = b.1 - a.1;
        let len = (dx*dx + dy*dy).sqrt();
        if len > 0.0 {
            let nx = -dy / len;
            let ny = dx / len;
            let eps = 1e-5 * len.max(1.0);
            let test_pt = (mid.0 + nx * eps, mid.1 + ny * eps);
            if !point_in_polygon(test_pt, stock) {
                return false;
            }
        }
    }
    
    true
}

fn rotate_polygon(poly: &[(f64, f64)], angle: f64) -> Vec<(f64, f64)> {
    let cos_a = angle.cos();
    let sin_a = angle.sin();
    poly.iter().map(|&(x, y)| {
        (x * cos_a - y * sin_a, x * sin_a + y * cos_a)
    }).collect()
}

fn translate_polygon(poly: &[(f64, f64)], tx: f64, ty: f64) -> Vec<(f64, f64)> {
    poly.iter().map(|&(x, y)| (x + tx, y + ty)).collect()
}

fn create_stock(id: usize, vertices: &[(f64, f64)]) -> Stock {
    let mut min_x = f64::MAX; let mut max_x = f64::MIN;
    let mut min_y = f64::MAX; let mut max_y = f64::MIN;
    for &(x, y) in vertices {
        min_x = min_x.min(x); max_x = max_x.max(x);
        min_y = min_y.min(y); max_y = max_y.max(y);
    }
    let gw = 400;
    let gh = 400;
    let dx = (max_x - min_x) / gw as f64;
    let dy = (max_y - min_y) / gh as f64;
    
    let mut grid = vec![false; gw * gh];
    for r in 0..gh {
        for c in 0..gw {
            let cx = min_x + c as f64 * dx + dx / 2.0;
            let cy = min_y + r as f64 * dy + dy / 2.0;
            if !point_in_polygon((cx, cy), vertices) {
                grid[r * gw + c] = true;
            }
        }
    }
    
    Stock {
        id,
        vertices: vertices.to_vec(),
        placed_pieces: Vec::new(),
        grid,
        free_area: polygon_area(vertices),
        min_x, max_x,
        min_y, max_y,
        dx, dy,
        gw, gh,
    }
}

fn update_grid(stock: &mut Stock, final_poly: &[(f64, f64)]) {
    let mut p_min_x = f64::MAX; let mut p_max_x = f64::MIN;
    let mut p_min_y = f64::MAX; let mut p_max_y = f64::MIN;
    for &(x, y) in final_poly {
        p_min_x = p_min_x.min(x); p_max_x = p_max_x.max(x);
        p_min_y = p_min_y.min(y); p_max_y = p_max_y.max(y);
    }
    
    let min_c = ((p_min_x - stock.min_x) / stock.dx).floor() as isize;
    let max_c = ((p_max_x - stock.min_x) / stock.dx).ceil() as isize;
    let min_r = ((p_min_y - stock.min_y) / stock.dy).floor() as isize;
    let max_r = ((p_max_y - stock.min_y) / stock.dy).ceil() as isize;
    
    let min_c = min_c.max(0) as usize;
    let max_c = (max_c.max(0) as usize).min(stock.gw - 1);
    let min_r = min_r.max(0) as usize;
    let max_r = (max_r.max(0) as usize).min(stock.gh - 1);
    
    for r in min_r..=max_r {
        for c in min_c..=max_c {
            let cx = stock.min_x + c as f64 * stock.dx + stock.dx / 2.0;
            let cy = stock.min_y + r as f64 * stock.dy + stock.dy / 2.0;
            if point_in_polygon((cx, cy), final_poly) {
                stock.grid[r * stock.gw + c] = true;
            }
        }
    }
}

fn find_placement(stock: &Stock, piece: &Piece, rots: &[f64]) -> Option<(f64, f64, f64, Vec<(f64, f64)>)> {
    for &rot in rots {
        let rotated = rotate_polygon(&piece.vertices, rot);
        let mut p_min_x = f64::MAX; let mut p_max_x = f64::MIN;
        let mut p_min_y = f64::MAX; let mut p_max_y = f64::MIN;
        for &(x, y) in &rotated {
            p_min_x = p_min_x.min(x); p_max_x = p_max_x.max(x);
            p_min_y = p_min_y.min(y); p_max_y = p_max_y.max(y);
        }
        
        let pw = p_max_x - p_min_x;
        let ph = p_max_y - p_min_y;
        if pw > stock.max_x - stock.min_x + 1e-5 || ph > stock.max_y - stock.min_y + 1e-5 {
            continue;
        }
        
        let pw_cells = (pw / stock.dx).ceil() as usize + 2;
        let ph_cells = (ph / stock.dy).ceil() as usize + 2;
        let mut piece_cells = Vec::new();
        for dr in 0..ph_cells {
            for dc in 0..pw_cells {
                let lx = p_min_x + dc as f64 * stock.dx + stock.dx / 2.0;
                let ly = p_min_y + dr as f64 * stock.dy + stock.dy / 2.0;
                if point_in_polygon((lx, ly), &rotated) {
                    piece_cells.push((dc, dr));
                }
            }
        }
        if piece_cells.is_empty() {
            piece_cells.push((pw_cells / 2, ph_cells / 2));
        }
        
        for sr in 0..stock.gh {
            for sc in 0..stock.gw {
                let mut ok = true;
                for &(dc, dr) in &piece_cells {
                    let tc = sc + dc;
                    let tr = sr + dr;
                    if tc >= stock.gw || tr >= stock.gh { ok = false; break; }
                    if stock.grid[tr * stock.gw + tc] { ok = false; break; }
                }
                if ok {
                    let tx = stock.min_x + sc as f64 * stock.dx;
                    let ty = stock.min_y + sr as f64 * stock.dy;
                    let t = (tx - p_min_x, ty - p_min_y);
                    let final_poly = translate_polygon(&rotated, t.0, t.1);
                    
                    if !polygon_inside_stock(&final_poly, &stock.vertices) {
                        continue;
                    }
                    
                    let mut overlap = false;
                    for placed in &stock.placed_pieces {
                        if polygons_overlap(&final_poly, placed) {
                            overlap = true;
                            break;
                        }
                    }
                    
                    if !overlap {
                        return Some((t.0, t.1, rot, final_poly));
                    }
                }
            }
        }
    }
    None
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines().map(|l| l.unwrap());
    
    let first_line = match lines.next() {
        Some(l) => l,
        None => return,
    };
    let parts: Vec<usize> = first_line.split_whitespace().map(|s| s.parse().unwrap()).collect();
    if parts.len() < 2 { return; }
    let sv = parts[0];
    let np = parts[1];
    
    let mut stock_vertices = Vec::new();
    for _ in 0..sv {
        let line = lines.next().unwrap();
        let coords: Vec<f64> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
        stock_vertices.push((coords[0], coords[1]));
    }
    ensure_ccw(&mut stock_vertices);
    
    let mut pieces = Vec::new();
    for i in 0..np {
        let line = lines.next().unwrap();
        let pv: usize = line.trim().parse().unwrap();
        let mut p_verts = Vec::new();
        for _ in 0..pv {
            let line = lines.next().unwrap();
            let coords: Vec<f64> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
            p_verts.push((coords[0], coords[1]));
        }
        ensure_ccw(&mut p_verts);
        pieces.push(Piece {
            id: i,
            vertices: p_verts.clone(),
            area: polygon_area(&p_verts),
        });
    }
    
    pieces.sort_by(|a, b| b.area.partial_cmp(&a.area).unwrap());
    
    let mut stocks: Vec<Stock> = Vec::new();
    let mut output = vec![(0, 0.0, 0.0, 0.0); np];
    
    let rots = [
        0.0,
        std::f64::consts::PI / 4.0,
        std::f64::consts::PI / 2.0,
        3.0 * std::f64::consts::PI / 4.0,
        std::f64::consts::PI,
        5.0 * std::f64::consts::PI / 4.0,
        3.0 * std::f64::consts::PI / 2.0,
        7.0 * std::f64::consts::PI / 4.0,
    ];
    
    for piece in pieces {
        let mut placed = false;
        
        for stock in &mut stocks {
            if stock.free_area < piece.area {
                continue;
            }
            
            if let Some((tx, ty, rot, final_poly)) = find_placement(stock, &piece, &rots) {
                stock.placed_pieces.push(final_poly.clone());
                stock.free_area -= piece.area;
                update_grid(stock, &final_poly);
                output[piece.id] = (stock.id, tx, ty, rot);
                placed = true;
                break;
            }
        }
        
        if !placed {
            let mut new_stock = create_stock(stocks.len(), &stock_vertices);
            if let Some((tx, ty, rot, final_poly)) = find_placement(&mut new_stock, &piece, &rots) {
                new_stock.placed_pieces.push(final_poly.clone());
                new_stock.free_area -= piece.area;
                update_grid(&mut new_stock, &final_poly);
                output[piece.id] = (new_stock.id, tx, ty, rot);
                stocks.push(new_stock);
            } else {
                stocks.push(new_stock);
                output[piece.id] = (stocks.len()-1, 0.0, 0.0, 0.0);
            }
        }
    }
    
    println!("{}", stocks.len());
    for i in 0..np {
        let (s_idx, x, y, rot) = output[i];
        println!("{} {} {} {}", s_idx, x, y, rot);
    }
}
