use std::io::{self, BufRead};
use std::collections::HashSet;

#[derive(Clone, Copy)]
struct Point {
    u: f64,
    v: f64,
}

#[derive(Clone, Copy)]
struct Edge {
    v_min: f64,
    v_max: f64,
    u_at_vmin: f64,
    du_dv: f64,
}

fn rasterize_edge(u1: f64, v1: f64, u2: f64, v2: f64, cells: &mut HashSet<(i32, i32)>) {
    let mut x = u1.floor() as i32;
    let mut y = v1.floor() as i32;
    let end_x = u2.floor() as i32;
    let end_y = v2.floor() as i32;

    cells.insert((x, y));

    let step_x = if u2 > u1 { 1 } else if u2 < u1 { -1 } else { 0 };
    let step_y = if v2 > v1 { 1 } else if v2 < v1 { -1 } else { 0 };

    let t_delta_x = if step_x != 0 { (1.0 / (u2 - u1).abs()) } else { f64::INFINITY };
    let t_delta_y = if step_y != 0 { (1.0 / (v2 - v1).abs()) } else { f64::INFINITY };

    let mut t_max_x = if step_x > 0 {
        ((x + 1) as f64 - u1) * t_delta_x
    } else if step_x < 0 {
        (u1 - x as f64) * t_delta_x
    } else {
        f64::INFINITY
    };

    let mut t_max_y = if step_y > 0 {
        ((y + 1) as f64 - v1) * t_delta_y
    } else if step_y < 0 {
        (v1 - y as f64) * t_delta_y
    } else {
        f64::INFINITY
    };

    let mut max_steps = (x - end_x).abs() + (y - end_y).abs() + 2;

    while (x != end_x || y != end_y) && max_steps > 0 {
        max_steps -= 1;
        if t_max_x < t_max_y {
            t_max_x += t_delta_x;
            x += step_x;
        } else {
            t_max_y += t_delta_y;
            y += step_y;
        }
        cells.insert((x, y));
    }
    cells.insert((end_x, end_y));
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines().filter_map(|l| l.ok()).filter(|l| !l.trim().is_empty());

    let nv_line = match lines.next() {
        Some(l) => l,
        None => return,
    };
    let nv: usize = nv_line.trim().parse().unwrap();

    let mut orig_poly = Vec::with_capacity(nv);
    for _ in 0..nv {
        let line = lines.next().unwrap();
        let parts: Vec<f64> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
        orig_poly.push((parts[0], parts[1]));
    }

    let sun_line = lines.next().unwrap();
    let parts: Vec<f64> = sun_line.split_whitespace().map(|s| s.parse().unwrap()).collect();
    let sx = parts[0];
    let sy = parts[1];
    let sz = parts[2];

    let s_len = (sx * sx + sy * sy + sz * sz).sqrt();
    let dx = -sx / s_len;
    let dy = -sy / s_len;
    let dz = -sz / s_len;

    let w = ((1.0 + dz) / 2.0).sqrt();
    let qx = -dy / (2.0 * w);
    let qy = dx / (2.0 * w);
    let qz = 0.0;

    let ux = 1.0 - 2.0 * (qy * qy + qz * qz);
    let uy = 2.0 * (qx * qy + w * qz);
    let uz = 2.0 * (qx * qz - w * qy);

    let vx = 2.0 * (qx * qy - w * qz);
    let vy = 1.0 - 2.0 * (qx * qx + qz * qz);
    let vz = 2.0 * (qy * qz + w * qx);

    let sz_safe = if sz == 0.0 { -1e-9 } else { sz };
    let up_x = ux - sx * (uz / sz_safe);
    let up_y = uy - sy * (uz / sz_safe);

    let vp_x = vx - sx * (vz / sz_safe);
    let vp_y = vy - sy * (vz / sz_safe);

    let scale = 10.0;
    let w1x = scale * (up_x - vp_x);
    let w1y = scale * (up_y - vp_y);

    let w2x = scale * (up_x + vp_x);
    let w2y = scale * (up_y + vp_y);

    let det = w1x * w2y - w1y * w2x;
    let inv_m00 = w2y / det;
    let inv_m01 = -w2x / det;
    let inv_m10 = -w1y / det;
    let inv_m11 = w1x / det;

    let mut poly = Vec::with_capacity(nv);
    for &(x, y) in &orig_poly {
        let u = inv_m00 * x + inv_m01 * y + 0.5;
        let v = inv_m10 * x + inv_m11 * y + 0.5;
        poly.push(Point { u, v });
    }

    let mut cells: HashSet<(i32, i32)> = HashSet::new();

    for k in 0..nv {
        let p1 = &poly[k];
        let p2 = &poly[(k + 1) % nv];
        rasterize_edge(p1.u, p1.v, p2.u, p2.v, &mut cells);
    }

    let mut min_v = f64::INFINITY;
    let mut max_v = f64::NEG_INFINITY;
    for p in &poly {
        if p.v < min_v { min_v = p.v; }
        if p.v > max_v { max_v = p.v; }
    }

    let j_min = min_v.floor() as i32;
    let j_max = max_v.floor() as i32;

    let mut edges = Vec::new();
    for k in 0..nv {
        let p1 = &poly[k];
        let p2 = &poly[(k + 1) % nv];
        if (p1.v - p2.v).abs() > 1e-9 {
            let (p_min, p_max) = if p1.v < p2.v { (p1, p2) } else { (p2, p1) };
            edges.push(Edge {
                v_min: p_min.v,
                v_max: p_max.v,
                u_at_vmin: p_min.u,
                du_dv: (p_max.u - p_min.u) / (p_max.v - p_min.v),
            });
        }
    }

    edges.sort_by(|a, b| a.v_min.partial_cmp(&b.v_min).unwrap());

    let mut active_edges: Vec<Edge> = Vec::new();
    let mut edge_idx = 0;

    for j in j_min..=j_max {
        let v_test = j as f64 + 0.5;

        while edge_idx < edges.len() && edges[edge_idx].v_min < v_test {
            if edges[edge_idx].v_max >= v_test {
                active_edges.push(edges[edge_idx]);
            }
            edge_idx += 1;
        }

        active_edges.retain(|e| e.v_max >= v_test);

        let mut intersections = Vec::with_capacity(active_edges.len());
        for e in &active_edges {
            let u = e.u_at_vmin + (v_test - e.v_min) * e.du_dv;
            intersections.push(u);
        }

        intersections.sort_by(|a, b| a.partial_cmp(b).unwrap());

        for m in 0..(intersections.len() / 2) {
            let u_start = intersections[2 * m];
            let u_end = intersections[2 * m + 1];
            let i_start = u_start.floor() as i32;
            let i_end = u_end.floor() as i32;
            for i in i_start..=i_end {
                cells.insert((i, j));
            }
        }
    }

    println!("{}", cells.len());
    let mut idx = 0;
    for &(i, j) in &cells {
        let cx = i as f64 * w1x + j as f64 * w2x;
        let cy = i as f64 * w1y + j as f64 * w2y;

        let t = idx as f64 * (30.0 / sz_safe.abs().max(1e-9));
        let px = cx + t * sx;
        let py = cy + t * sy;
        let pz = 0.0 + t * sz;

        println!("{:.6} {:.6} {:.6} {:.6} {:.6} {:.6} {:.6} {:.1}", px, py, pz, w, qx, qy, qz, scale);
        idx += 1;
    }
}
