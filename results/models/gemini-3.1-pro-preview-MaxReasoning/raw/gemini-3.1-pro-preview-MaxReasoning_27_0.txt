#include <vector>
#include <numeric>

#include <cstdlib>

#if __cplusplus >= 202002L || defined(_MSVC_LANG) && _MSVC_LANG >= 202002L
#include <bit>

#include <intrin.h>
#endif

using namespace std;

int V, E, K;
vector<int> degree_cnt;
vector<int> offset;
vector<int> adj;
vector<int> color;

vector<int> conflicts;
vector<int> pos_in_conflicting;
vector<int> conflicting_nodes;
int total_conflicts_edges = 0;
int best_total_conflicts_seen = 0;

vector<int> tabu_until;
vector<int> tabu_color;

auto start_time = chrono::steady_clock::now();
inline double get_elapsed_sec() {
    auto now = chrono::steady_clock::now();
    return chrono::duration<double>(now - start_time).count();
}

const int BUF_SIZE = 65536;
char buf[BUF_SIZE];
int buf_pos = 0, buf_len = 0;

inline int get_char() {
    if (buf_pos == buf_len) {
        buf_pos = 0;
        buf_len = fread(buf, 1, BUF_SIZE, stdin);
        if (buf_len == 0) return EOF;
    }
    return (unsigned char)buf[buf_pos++];
}

inline int read_int() {
    int c = get_char();
    while (c != EOF && (c < '0' || c > '9')) {
        c = get_char();
    }
    if (c == EOF) return -1;
    int res = 0;
    while (c != EOF && c >= '0' && c <= '9') {
        res = res * 10 + (c - '0');
        c = get_char();
    }
    return res;
}

mt19937 rng(42);
int get_rand(int max_val) {
    if (max_val <= 1) return 0;
    return uniform_int_distribution<int>(0, max_val - 1)(rng);
}

void print_solution() {
    for (int v = 0; v < V; ++v) {
        cout << color[v] << (v == V - 1 ? "" : " ");
    }
    cout << "\n";
}

bool backtrack(int u_idx, const vector<int>& order) {
    if (u_idx == V) return true;
    static int bt_steps = 0;
    if (++bt_steps % 10000 == 0 && get_elapsed_sec() > 2.0) return false;
    
    int u = order[u_idx];
    vector<bool> used(K, false);
    for (int idx = offset[u]; idx < offset[u+1]; ++idx) {
        int v = adj[idx];
        if (color[v] != -1) {
            used[color[v]] = true;
        }
    }
    
    for (int c = 0; c < K; ++c) {
        if (!used[c]) {
            color[u] = c;
            if (backtrack(u_idx + 1, order)) return true;
            color[u] = -1;
        }
    }
    return false;
}

bool solve_backtracking() {
    vector<int> order(V);
    iota(order.begin(), order.end(), 0);
    sort(order.begin(), order.end(), [&](int a, int b) {
        return degree_cnt[a] > degree_cnt[b];
    });
    fill(color.begin(), color.end(), -1);
    return backtrack(0, order);
}

void run_dsatur() {
    int max_colors = min(K, 1024);
    int words = (max_colors + 63) / 64;
    vector<uint64_t> adj_colors(V * words, 0);
    
    int max_sat = 0;
    vector<vector<int>> buckets(V + 1);
    vector<int> pos_in_bucket(V, -1);
    vector<int> node_bucket(V, 0);
    
    vector<int> initial_nodes(V);
    iota(initial_nodes.begin(), initial_nodes.end(), 0);
    sort(initial_nodes.begin(), initial_nodes.end(), [&](int a, int b) {
        return degree_cnt[a] < degree_cnt[b]; 
    });
    buckets[0] = initial_nodes;
    for(int i=0; i<V; ++i) {
        pos_in_bucket[buckets[0][i]] = i;
    }
    
    auto increase_sat = [&](int u) {
        int old_b = node_bucket[u];
        int new_b = old_b + 1;
        
        int idx = pos_in_bucket[u];
        int last_node = buckets[old_b].back();
        buckets[old_b][idx] = last_node;
        pos_in_bucket[last_node] = idx;
        buckets[old_b].pop_back();
        
        pos_in_bucket[u] = buckets[new_b].size();
        buckets[new_b].push_back(u);
        node_bucket[u] = new_b;
        
        if (new_b > max_sat) max_sat = new_b;
    };
    
    auto pick_next = [&]() {
        while (max_sat >= 0 && buckets[max_sat].empty()) {
            max_sat--;
        }
        if (max_sat < 0) return -1;
        int u = buckets[max_sat].back();
        buckets[max_sat].pop_back();
        pos_in_bucket[u] = -1;
        return u;
    };
    
    for (int step = 0; step < V; ++step) {
        int v = pick_next();
        if (v == -1) break;
        
        int c = -1;
        for (int w = 0; w < words; ++w) {
            uint64_t mask = adj_colors[v * words + w];
            if (mask != ~0ULL) {
#ifdef HAS_BIT_OPS
                int b = std::countr_zero(static_cast<uint64_t>(~mask));
#else
#ifdef _MSC_VER
                unsigned long b;
                _BitScanForward64(&b, ~mask);
#else
                int b = __builtin_ctzll(~mask);
#endif
#endif
                c = w * 64 + b;
                break;
            }
        }
        if (c == -1 || c >= max_colors) {
            vector<bool> used(degree_cnt[v] + max_colors + 1, false);
            for (int idx = offset[v]; idx < offset[v+1]; ++idx) {
                int u = adj[idx];
                if (color[u] != -1 && color[u] < used.size()) {
                    used[color[u]] = true;
                }
            }
            for (int i = 0; i < used.size(); ++i) {
                if (!used[i]) { c = i; break; }
            }
        }
        
        color[v] = c;
        if (c < max_colors) {
            int word_idx = c / 64;
            uint64_t bit = 1ULL << (c % 64);
            for (int idx = offset[v]; idx < offset[v+1]; ++idx) {
                int u = adj[idx];
                if (pos_in_bucket[u] != -1) {
                    int arr_idx = u * words + word_idx;
                    if (!(adj_colors[arr_idx] & bit)) {
                        adj_colors[arr_idx] |= bit;
                        increase_sat(u);
                    }
                }
            }
        }
    }
}

void remove_conflicting(int v) {
    int idx = pos_in_conflicting[v];
    if (idx != -1) {
        int last_v = conflicting_nodes.back();
        conflicting_nodes[idx] = last_v;
        pos_in_conflicting[last_v] = idx;
        conflicting_nodes.pop_back();
        pos_in_conflicting[v] = -1;
    }
}

void add_conflicting(int v) {
    if (pos_in_conflicting[v] == -1) {
        pos_in_conflicting[v] = conflicting_nodes.size();
        conflicting_nodes.push_back(v);
    }
}

bool change_color(int v, int c_new, int iter) {
    int c_old = color[v];
    if (c_old == c_new) return false;
    
    int old_v_conflicts = conflicts[v];
    color[v] = c_new;
    int new_v_conflicts = 0;
    
    for (int idx = offset[v]; idx < offset[v+1]; ++idx) {
        int u = adj[idx];
        int c_u = color[u];
        if (c_u == c_old) {
            conflicts[u]--;
            if (conflicts[u] == 0) remove_conflicting(u);
        } else if (c_u == c_new) {
            if (conflicts[u] == 0) add_conflicting(u);
            conflicts[u]++;
            new_v_conflicts++;
        }
    }
    
    conflicts[v] = new_v_conflicts;
    if (conflicts[v] == 0) {
        remove_conflicting(v);
    } else {
        add_conflicting(v);
    }
    
    total_conflicts_edges += (new_v_conflicts - old_v_conflicts);
    tabu_color[v] = c_old;
    tabu_until[v] = iter + get_rand(15) + 5;
    
    if (total_conflicts_edges < best_total_conflicts_seen) {
        best_total_conflicts_seen = total_conflicts_edges;
        return true;
    }
    return false;
}

void init_conflicts() {
    conflicts.assign(V, 0);
    pos_in_conflicting.assign(V, -1);
    conflicting_nodes.clear();
    total_conflicts_edges = 0;
    
    for (int v = 0; v < V; ++v) {
        int c_v = color[v];
        int count = 0;
        for (int idx = offset[v]; idx < offset[v+1]; ++idx) {
            int u = adj[idx];
            if (color[u] == c_v) {
                count++;
            }
        }
        conflicts[v] = count;
        total_conflicts_edges += count;
        if (count > 0) {
            add_conflicting(v);
        }
    }
    total_conflicts_edges /= 2;
    best_total_conflicts_seen = total_conflicts_edges;
    
    tabu_until.assign(V, 0);
    tabu_color.assign(V, -1);
}

void tabu_search_full() {
    int iter = 0;
    int last_improve = 0;
    while (total_conflicts_edges > 0 && get_elapsed_sec() < 290.0) {
        iter++;
        
        if (iter - last_improve > 10000) {
            int perturb_count = max(1, V / 100);
            for (int i = 0; i < perturb_count; ++i) {
                int rand_v = get_rand(V);
                int rand_c = get_rand(K);
                change_color(rand_v, rand_c, iter);
            }
            last_improve = iter;
            continue;
        }
        
        int best_v = -1;
        int best_c = -1;
        int best_delta = 1e9;
        int best_count = 0;
        
        for (int v : conflicting_nodes) {
            vector<int> counts(K, 0);
            for (int idx = offset[v]; idx < offset[v+1]; ++idx) {
                counts[color[adj[idx]]]++;
            }
            
            for (int c = 0; c < K; ++c) {
                if (c == color[v]) continue;
                int delta = counts[c] - counts[color[v]];
                
                bool is_tabu = (tabu_until[v] >= iter && tabu_color[v] == c);
                bool aspirated = (total_conflicts_edges + delta < best_total_conflicts_seen);
                
                if (is_tabu && !aspirated) continue;
                
                if (delta < best_delta) {
                    best_delta = delta;
                    best_v = v;
                    best_c = c;
                    best_count = 1;
                } else if (delta == best_delta) {
                    best_count++;
                    if (get_rand(best_count) == 0) {
                        best_v = v;
                        best_c = c;
                    }
                }
            }
        }
        
        if (best_v != -1) {
            if (change_color(best_v, best_c, iter)) {
                last_improve = iter;
            }
        } else {
            int v = conflicting_nodes[get_rand(conflicting_nodes.size())];
            int c = get_rand(K);
            while (c == color[v] && K > 1) {
                c = get_rand(K);
            }
            change_color(v, c, iter);
        }
    }
}

void tabu_search_fast() {
    int iter = 0;
    int last_improve = 0;
    while (total_conflicts_edges > 0 && get_elapsed_sec() < 290.0) {
        iter++;
        
        if (iter - last_improve > 1000000) {
            int perturb_count = max(1, V / 100);
            for (int i = 0; i < perturb_count; ++i) {
                int rand_v = get_rand(V);
                int rand_c = get_rand(K);
                change_color(rand_v, rand_c, iter);
            }
            last_improve = iter;
            continue;
        }
        
        int v = conflicting_nodes[get_rand(conflicting_nodes.size())];
        
        if (get_rand(100) < 10) {
            int c = get_rand(K);
            while (c == color[v] && K > 1) {
                c = get_rand(K);
            }
            change_color(v, c, iter);
            continue;
        }
        
        int best_delta = 1e9;
        int best_c = -1;
        int best_count = 0;
        
        vector<int> counts(K, 0);
        for (int idx = offset[v]; idx < offset[v+1]; ++idx) {
            counts[color[adj[idx]]]++;
        }
        
        for (int c = 0; c < K; ++c) {
            if (c == color[v]) continue;
            int delta = counts[c] - counts[color[v]];
            
            bool is_tabu = (tabu_until[v] >= iter && tabu_color[v] == c);
            bool aspirated = (total_conflicts_edges + delta < best_total_conflicts_seen);
            
            if (is_tabu && !aspirated) continue;
            
            if (delta < best_delta) {
                best_delta = delta;
                best_c = c;
                best_count = 1;
            } else if (delta == best_delta) {
                best_count++;
                if (get_rand(best_count) == 0) {
                    best_c = c;
                }
            }
        }
        
        if (best_c != -1) {
            if (change_color(v, best_c, iter)) {
                last_improve = iter;
            }
        } else {
            int c = get_rand(K);
            while (c == color[v] && K > 1) {
                c = get_rand(K);
            }
            change_color(v, c, iter);
        }
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    start_time = chrono::steady_clock::now();
    V = read_int();
    if (V <= 0) return 0;
    E = read_int();
    K = read_int();
    
    if (K <= 0) {
        for(int i=0; i<V; ++i) cout << 0 << (i == V-1 ? "" : " ");
        cout << "\n";
        return 0;
    }
    
    struct Edge { int u, v; };
    vector<Edge> edges(E);
    for(int i=0; i<E; ++i) {
        int u = read_int();
        int v = read_int();
        if (u >= 0 && u < V && v >= 0 && v < V) {
            edges[i].u = u;
            edges[i].v = v;
            if (edges[i].u > edges[i].v) swap(edges[i].u, edges[i].v);
        } else {
            edges[i].u = 0;
            edges[i].v = 0;
        }
    }
    
    sort(edges.begin(), edges.end(), [](const Edge& a, const Edge& b){
        if (a.u != b.u) return a.u < b.u;
        return a.v < b.v;
    });
    
    int valid_E = 0;
    for(int i=0; i<E; ++i) {
        if (edges[i].u == edges[i].v) continue;
        if (i > 0 && edges[i].u == edges[i-1].u && edges[i].v == edges[i-1].v) continue;
        edges[valid_E++] = edges[i];
    }
    E = valid_E;
    
    if (E == 0) {
        for(int i=0; i<V; ++i) cout << 0 << (i == V-1 ? "" : " ");
        cout << "\n";
        return 0;
    }
    
    degree_cnt.assign(V, 0);
    for(int i=0; i<E; ++i) {
        degree_cnt[edges[i].u]++;
        degree_cnt[edges[i].v]++;
    }
    
    offset.assign(V + 1, 0);
    for(int i=0; i<V; ++i) {
        offset[i+1] = offset[i] + degree_cnt[i];
    }
    
    adj.assign(2 * E, 0);
    vector<int> cur_offset = offset;
    for(int i=0; i<E; ++i) {
        int u = edges[i].u;
        int v = edges[i].v;
        adj[cur_offset[u]++] = v;
        adj[cur_offset[v]++] = u;
    }
    
    color.assign(V, -1);
    
    if (V <= 100) {
        if (solve_backtracking()) {
            print_solution();
            return 0;
        }
    }
    
    fill(color.begin(), color.end(), -1);
    run_dsatur();
    
    int max_color_used = 0;
    for(int v=0; v<V; ++v) {
        if (color[v] > max_color_used) max_color_used = color[v];
    }
    
    if (max_color_used < K) {
        print_solution();
        return 0;
    }
    
    for (int v = 0; v < V; ++v) {
        if (color[v] >= K) {
            int best_c = get_rand(K);
            int min_conf = 1e9;
            vector<int> counts(K, 0);
            for(int idx = offset[v]; idx < offset[v+1]; ++idx) {
                int u = adj[idx];
                if (color[u] < K && color[u] >= 0) {
                    counts[color[u]]++;
                }
            }
            int best_count = 0;
            for(int c=0; c<K; ++c) {
                if(counts[c] < min_conf) {
                    min_conf = counts[c];
                    best_c = c;
                    best_count = 1;
                } else if (counts[c] == min_conf) {
                    best_count++;
                    if (get_rand(best_count) == 0) best_c = c;
                }
            }
            color[v] = best_c;
        }
    }
    
    init_conflicts();
    
    if (V <= 5000) {
        tabu_search_full();
    } else {
        tabu_search_fast();
    }
    
    print_solution();
    return 0;
}
