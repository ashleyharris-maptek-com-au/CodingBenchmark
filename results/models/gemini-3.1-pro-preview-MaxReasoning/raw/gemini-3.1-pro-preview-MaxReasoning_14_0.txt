using System;
using System.IO;
using System.Collections.Generic;

class Program {
    static int W, H;
    static byte[] board;
    static bool[] inQueue;
    static Queue<int> queue = new Queue<int>();
    static List<int> deducedSafeOutput = new List<int>();
    static int[] dx = {-1, 0, 1, -1, 1, -1, 0, 1};
    static int[] dy = {-1, -1, -1, 0, 0, 1, 1, 1};

    static int[] activeRevealedArr;
    static int[] activeRevealedIndex;
    static int activeRevealedCount = 0;

    static void Main() {
        Stream input = Console.OpenStandardInput();
        byte[] buffer = new byte[65536];
        int bytesRead = 0;
        int bufferPos = 0;
        
        int ReadByte() {
            if (bufferPos >= bytesRead) {
                bufferPos = 0;
                bytesRead = input.Read(buffer, 0, buffer.Length);
                if (bytesRead == 0) return -1;
            }
            return buffer[bufferPos++];
        }
        
        int ReadInt() {
            int c = ReadByte();
            while (c != -1 && (c < '0' || c > '9')) c = ReadByte();
            if (c == -1) return -1;
            int res = 0;
            while (c >= '0' && c <= '9') {
                res = res * 10 + (c - '0');
                c = ReadByte();
            }
            return res;
        }

        H = ReadInt();
        W = ReadInt();
        if (H <= 0 || W <= 0) return;

        board = new byte[H * W];
        inQueue = new bool[H * W];
        activeRevealedIndex = new int[H * W];
        for(int i=0; i<H*W; i++) activeRevealedIndex[i] = -1;

        int[] tempRevealed = new int[H * W];

        int idx = 0;
        while (idx < H * W) {
            int c = ReadByte();
            if (c == -1) break;
            if (c == ' ' || c == '*' || (c >= '0' && c <= '8')) {
                if (c == ' ') board[idx] = 9;
                else if (c == '*') board[idx] = 10;
                else {
                    board[idx] = (byte)(c - '0');
                    tempRevealed[activeRevealedCount] = idx;
                    activeRevealedIndex[idx] = activeRevealedCount;
                    activeRevealedCount++;
                    queue.Enqueue(idx);
                    inQueue[idx] = true;
                }
                idx++;
            }
        }

        activeRevealedArr = new int[activeRevealedCount];
        Array.Copy(tempRevealed, activeRevealedArr, activeRevealedCount);
        tempRevealed = null;

        while(true) {
            if (Phase1_Queue()) continue;
            List<Constraint> reduced;
            if (Phase2_Pairwise(out reduced)) continue;
            if (Phase3_Backtracking(reduced)) continue;
            break;
        }

        Stream output = Console.OpenStandardOutput();
        StreamWriter writer = new StreamWriter(output, new System.Text.ASCIIEncoding(), 65536);
        writer.WriteLine(deducedSafeOutput.Count);
        foreach(int outIdx in deducedSafeOutput) {
            writer.WriteLine($"{outIdx % W} {outIdx / W}");
        }
        writer.Flush();
    }

    static void RemoveActiveRevealed(int idx) {
        int pos = activeRevealedIndex[idx];
        if (pos == -1) return;
        
        activeRevealedCount--;
        if (pos < activeRevealedCount) {
            int lastIdx = activeRevealedArr[activeRevealedCount];
            activeRevealedArr[pos] = lastIdx;
            activeRevealedIndex[lastIdx] = pos;
        }
        activeRevealedIndex[idx] = -1;
    }

    static void MarkMine(int idx) {
        if (board[idx] == 9) {
            board[idx] = 12; // Deduced Mine
            EnqueueNeighbors(idx);
        }
    }

    static void MarkSafe(int idx) {
        if (board[idx] == 9) {
            board[idx] = 11; // Deduced Safe
            deducedSafeOutput.Add(idx);
            EnqueueNeighbors(idx);
        }
    }

    static void EnqueueNeighbors(int idx) {
        int y = idx / W;
        int x = idx % W;
        for(int i=0; i<8; i++) {
            int nx = x + dx[i];
            int ny = y + dy[i];
            if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
                int nidx = ny * W + nx;
                int st = board[nidx];
                if (st >= 0 && st <= 8) {
                    if (!inQueue[nidx]) {
                        inQueue[nidx] = true;
                        queue.Enqueue(nidx);
                    }
                }
            }
        }
    }

    static bool Phase1_Queue() {
        bool deducedAny = false;
        while(queue.Count > 0) {
            int idx = queue.Dequeue();
            inQueue[idx] = false;
            
            int val = board[idx];
            int y = idx / W;
            int x = idx % W;
            int unkCount = 0;
            int mineCount = 0;
            
            int[] unkIndices = new int[8];
            
            for(int i=0; i<8; i++) {
                int nx = x + dx[i];
                int ny = y + dy[i];
                if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
                    int nidx = ny * W + nx;
                    int st = board[nidx];
                    if (st == 9) {
                        unkIndices[unkCount++] = nidx;
                    } else if (st == 10 || st == 12) {
                        mineCount++;
                    }
                }
            }
            
            if (unkCount > 0) {
                int minesLeft = val - mineCount;
                if (minesLeft == 0) {
                    for(int i=0; i<unkCount; i++) MarkSafe(unkIndices[i]);
                    deducedAny = true;
                } else if (minesLeft == unkCount) {
                    for(int i=0; i<unkCount; i++) MarkMine(unkIndices[i]);
                    deducedAny = true;
                }
            }
        }
        return deducedAny;
    }

    class Constraint {
        public int Id;
        public int[] Vars;
        public int Mines;
        public Constraint(int[] v, int m) { Vars = v; Mines = m; }
    }

    static List<Constraint> GetActiveConstraintsFromBoard() {
        List<Constraint> constraints = new List<Constraint>();
        for(int i = activeRevealedCount - 1; i >= 0; i--) {
            int idx = activeRevealedArr[i];
            int val = board[idx];
            int y = idx / W;
            int x = idx % W;
            int unkCount = 0;
            int mineCount = 0;
            
            List<int> unkVars = null;
            for(int k=0; k<8; k++) {
                int nx = x + dx[k];
                int ny = y + dy[k];
                if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
                    int nidx = ny * W + nx;
                    int st = board[nidx];
                    if (st == 9) {
                        if (unkVars == null) unkVars = new List<int>(8);
                        unkVars.Add(nidx);
                        unkCount++;
                    } else if (st == 10 || st == 12) {
                        mineCount++;
                    }
                }
            }
            
            if (unkCount > 0) {
                int minesLeft = val - mineCount;
                unkVars.Sort();
                constraints.Add(new Constraint(unkVars.ToArray(), minesLeft));
            } else {
                RemoveActiveRevealed(idx);
            }
        }
        return constraints;
    }

    static int CountDiff(Constraint c1, Constraint c2) {
        int count = 0;
        int i = 0, j = 0;
        while(i < c1.Vars.Length && j < c2.Vars.Length) {
            if (c1.Vars[i] == c2.Vars[j]) { i++; j++; }
            else if (c1.Vars[i] < c2.Vars[j]) { count++; i++; }
            else { j++; }
        }
        count += c1.Vars.Length - i;
        return count;
    }

    static int[] SetMinus(int[] v1, int[] v2) {
        List<int> res = new List<int>(v1.Length);
        int i = 0, j = 0;
        while(i < v1.Length && j < v2.Length) {
            if (v1[i] == v2[j]) { i++; j++; }
            else if (v1[i] < v2[j]) { res.Add(v1[i]); i++; }
            else { j++; }
        }
        while(i < v1.Length) { res.Add(v1[i]); i++; }
        return res.ToArray();
    }

    static void ApplyRule(Constraint c1, Constraint c2) {
        int i = 0, j = 0;
        while(i < c1.Vars.Length && j < c2.Vars.Length) {
            if (c1.Vars[i] == c2.Vars[j]) { i++; j++; }
            else if (c1.Vars[i] < c2.Vars[j]) { MarkMine(c1.Vars[i]); i++; }
            else { MarkSafe(c2.Vars[j]); j++; }
        }
        while(i < c1.Vars.Length) { MarkMine(c1.Vars[i]); i++; }
        while(j < c2.Vars.Length) { MarkSafe(c2.Vars[j]); j++; }
    }

    static bool Phase2_Pairwise(out List<Constraint> reducedConstraints) {
        reducedConstraints = GetActiveConstraintsFromBoard();
        bool boardChanged = false;
        bool constraintsChanged = true;
        
        while (constraintsChanged && !boardChanged) {
            constraintsChanged = false;
            
            for(int i=0; i<reducedConstraints.Count; i++) reducedConstraints[i].Id = i;
            
            Dictionary<int, List<Constraint>> varToC = new Dictionary<int, List<Constraint>>();
            foreach(var c in reducedConstraints) {
                foreach(var v in c.Vars) {
                    if (!varToC.ContainsKey(v)) varToC[v] = new List<Constraint>();
                    varToC[v].Add(c);
                }
            }
            
            HashSet<long> checkedPairs = new HashSet<long>();
            
            foreach(var list in varToC.Values) {
                for(int i=0; i<list.Count; i++) {
                    for(int j=i+1; j<list.Count; j++) {
                        var C1 = list[i];
                        var C2 = list[j];
                        if (C1.Id == C2.Id) continue;
                        
                        long pairId = ((long)Math.Min(C1.Id, C2.Id) << 32) | (uint)Math.Max(C1.Id, C2.Id);
                        if (!checkedPairs.Add(pairId)) continue;
                        
                        int diff1 = CountDiff(C1, C2);
                        int diff2 = CountDiff(C2, C1);
                        
                        if (C1.Mines - C2.Mines == diff1) {
                            if (diff1 > 0 || diff2 > 0) {
                                ApplyRule(C1, C2);
                                boardChanged = true; goto OutOfLoops;
                            }
                        }
                        if (C2.Mines - C1.Mines == diff2) {
                            if (diff1 > 0 || diff2 > 0) {
                                ApplyRule(C2, C1);
                                boardChanged = true; goto OutOfLoops;
                            }
                        }
                        
                        if (diff1 == 0 && diff2 > 0) {
                            reducedConstraints[C2.Id] = new Constraint(SetMinus(C2.Vars, C1.Vars), C2.Mines - C1.Mines) { Id = C2.Id };
                            constraintsChanged = true; goto OutOfLoops;
                        } else if (diff2 == 0 && diff1 > 0) {
                            reducedConstraints[C1.Id] = new Constraint(SetMinus(C1.Vars, C2.Vars), C1.Mines - C2.Mines) { Id = C1.Id };
                            constraintsChanged = true; goto OutOfLoops;
                        }
                    }
                }
            }
            
        OutOfLoops:
            if (constraintsChanged && !boardChanged) {
                reducedConstraints.RemoveAll(c => c.Vars.Length == 0);
            }
        }
        
        return boardChanged;
    }

    class Component {
        public List<int> Vars;
        public List<Constraint> Constraints;
    }

    static bool Phase3_Backtracking(List<Constraint> reducedConstraints) {
        if (reducedConstraints.Count == 0) return false;

        HashSet<string> seen = new HashSet<string>();
        List<Constraint> uniqueC = new List<Constraint>();
        foreach(var c in reducedConstraints) {
            if (c.Vars.Length == 0) continue;
            string hash = string.Join(",", c.Vars) + "=" + c.Mines;
            if (seen.Add(hash)) uniqueC.Add(c);
        }
        if (uniqueC.Count == 0) return false;

        Dictionary<int, List<Constraint>> varConstraints = new Dictionary<int, List<Constraint>>();
        foreach(var c in uniqueC) {
            foreach(var v in c.Vars) {
                if (!varConstraints.ContainsKey(v)) varConstraints[v] = new List<Constraint>();
                varConstraints[v].Add(c);
            }
        }

        HashSet<int> visited = new HashSet<int>();
        List<Component> components = new List<Component>();

        foreach(var startVar in varConstraints.Keys) {
            if (visited.Contains(startVar)) continue;
            
            Component comp = new Component();
            comp.Vars = new List<int>();
            comp.Constraints = new List<Constraint>();
            HashSet<Constraint> compConstraints = new HashSet<Constraint>();
            
            Queue<int> q = new Queue<int>();
            q.Enqueue(startVar);
            visited.Add(startVar);
            
            while(q.Count > 0) {
                int curr = q.Dequeue();
                comp.Vars.Add(curr);
                
                foreach(var c in varConstraints[curr]) {
                    if (compConstraints.Add(c)) {
                        comp.Constraints.Add(c);
                        foreach(var nextVar in c.Vars) {
                            if (visited.Add(nextVar)) {
                                q.Enqueue(nextVar);
                            }
                        }
                    }
                }
            }
            components.Add(comp);
        }

        bool anyDeduced = false;
        foreach(var comp in components) {
            if (comp.Vars.Count > 400) continue; // Prevent StackOverflow and extremely long TLE
            if (SolveComponent(comp, varConstraints)) {
                anyDeduced = true;
            }
        }
        return anyDeduced;
    }

    static bool SolveComponent(Component comp, Dictionary<int, List<Constraint>> globalVarConstraints) {
        int[] varArr = comp.Vars.ToArray();
        int N = varArr.Length;
        
        Array.Sort(varArr, (a, b) => globalVarConstraints[b].Count.CompareTo(globalVarConstraints[a].Count));
        
        Dictionary<int, int> varIdxMap = new Dictionary<int, int>();
        for(int i=0; i<N; i++) varIdxMap[varArr[i]] = i;
        
        int C = comp.Constraints.Count;
        int[][] cVars = new int[C][];
        int[] minesLeft = new int[C];
        int[] unassigned = new int[C];
        
        for(int i=0; i<C; i++) {
            var c = comp.Constraints[i];
            cVars[i] = new int[c.Vars.Length];
            for(int j=0; j<c.Vars.Length; j++) {
                cVars[i][j] = varIdxMap[c.Vars[j]];
            }
            minesLeft[i] = c.Mines;
            unassigned[i] = cVars[i].Length;
        }
        
        List<int>[] varToCIdx = new List<int>[N];
        for(int i=0; i<N; i++) varToCIdx[i] = new List<int>();
        for(int i=0; i<C; i++) {
            foreach(var v in cVars[i]) {
                varToCIdx[v].Add(i);
            }
        }
        
        int[] mineCounts = new int[N];
        long validSolutions = 0;
        bool limitReached = false;
        int[] assignment = new int[N];
        
        void Backtrack(int varIdx) {
            if (limitReached) return;
            
            if (varIdx == N) {
                validSolutions++;
                for(int i=0; i<N; i++) {
                    if (assignment[i] == 1) mineCounts[i]++;
                }
                if (validSolutions > 20000) {
                    limitReached = true;
                }
                return;
            }
            
            // Try Safe (0)
            bool canSafe = true;
            foreach(var cIdx in varToCIdx[varIdx]) {
                if (minesLeft[cIdx] > unassigned[cIdx] - 1) {
                    canSafe = false; break;
                }
            }
            if (canSafe) {
                foreach(var cIdx in varToCIdx[varIdx]) unassigned[cIdx]--;
                assignment[varIdx] = 0;
                Backtrack(varIdx + 1);
                foreach(var cIdx in varToCIdx[varIdx]) unassigned[cIdx]++;
            }
            
            // Try Mine (1)
            bool canMine = true;
            foreach(var cIdx in varToCIdx[varIdx]) {
                if (minesLeft[cIdx] - 1 < 0) {
                    canMine = false; break;
                }
            }
            if (canMine) {
                foreach(var cIdx in varToCIdx[varIdx]) {
                    unassigned[cIdx]--;
                    minesLeft[cIdx]--;
                }
                assignment[varIdx] = 1;
                Backtrack(varIdx + 1);
                foreach(var cIdx in varToCIdx[varIdx]) {
                    unassigned[cIdx]++;
                    minesLeft[cIdx]++;
                }
            }
        }
        
        Backtrack(0);
        
        if (limitReached || validSolutions == 0) return false;
        
        bool deducedAny = false;
        for(int i=0; i<N; i++) {
            if (mineCounts[i] == 0) {
                MarkSafe(varArr[i]);
                deducedAny = true;
            } else if (mineCounts[i] == validSolutions) {
                MarkMine(varArr[i]);
                deducedAny = true;
            }
        }
        return deducedAny;
    }
}
