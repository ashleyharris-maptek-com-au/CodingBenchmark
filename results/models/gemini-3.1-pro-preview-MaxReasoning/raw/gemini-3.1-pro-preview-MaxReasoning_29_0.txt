using System;
using System.IO;
using System.Collections.Generic;
using System.Collections.Concurrent;
using System.Threading;
using System.Threading.Tasks;
using System.Diagnostics;

class Program {
    public static int V;
    public static int E;
    public static int[][] edges;
    public static int[] pos;
    public static int[] order;
    
    public static ulong[] adjMatrix;
    public static int ulongsPerRow;
    
    public static volatile bool stopFlag = false;
    public static volatile int globalMaxCliqueSize = 0;
    public static int[] globalMaxClique;
    public static object lockObj = new object();
    
    public static WorkerPool workerPool;

    static void Main() {
        Stopwatch sw = Stopwatch.StartNew();
        Thread timerThread = new Thread(() => {
            while (sw.ElapsedMilliseconds < 285000) { // 4.75 minutes
                Thread.Sleep(100);
                if (stopFlag) return;
            }
            stopFlag = true;
        });
        timerThread.IsBackground = true;
        timerThread.Start();
        
        Scanner scanner = new Scanner(Console.OpenStandardInput());
        V = scanner.NextInt();
        if (V <= 0) {
            Console.WriteLine(0);
            return;
        }
        int E_input = scanner.NextInt();
        
        if (E_input == 0) {
            Console.WriteLine(1);
            Console.WriteLine(0);
            return;
        }

        globalMaxClique = new int[V];
        int[] degree = new int[V];
        const int CHUNK_SIZE = 100_000_000;
        List<int[]> u_chunks = new List<int[]>();
        List<int[]> v_chunks = new List<int[]>();
        
        int edgesRead = 0;
        int initSize = E_input > 0 ? Math.Min(E_input, CHUNK_SIZE) : 1000;
        u_chunks.Add(new int[initSize]);
        v_chunks.Add(new int[initSize]);
        
        for (int i = 0; i < E_input; i++) {
            int u = scanner.NextInt();
            int v = scanner.NextInt();
            if (u == -1 || v == -1) break;
            if (u == v) continue;
            
            int chunkIdx = edgesRead / CHUNK_SIZE;
            int chunkOffset = edgesRead % CHUNK_SIZE;
            if (chunkIdx >= u_chunks.Count) {
                int remain = E_input - i;
                int nextSize = Math.Max(1000, Math.Min(remain, CHUNK_SIZE));
                u_chunks.Add(new int[nextSize]);
                v_chunks.Add(new int[nextSize]);
            }
            
            u_chunks[chunkIdx][chunkOffset] = u;
            v_chunks[chunkIdx][chunkOffset] = v;
            
            degree[u]++;
            degree[v]++;
            edgesRead++;
        }
        E = edgesRead;
        
        if (E == 0) {
            Console.WriteLine(1);
            Console.WriteLine(0);
            return;
        }

        edges = new int[V][];
        for (int i = 0; i < V; i++) {
            edges[i] = new int[degree[i]];
        }
        int[] offset = new int[V];
        
        for (int i = 0; i < E; i++) {
            int chunkIdx = i / CHUNK_SIZE;
            int chunkOffset = i % CHUNK_SIZE;
            int u = u_chunks[chunkIdx][chunkOffset];
            int v = v_chunks[chunkIdx][chunkOffset];
            
            edges[u][offset[u]++] = v;
            edges[v][offset[v]++] = u;
        }
        
        u_chunks = null;
        v_chunks = null;
        GC.Collect();

        for (int i = 0; i < V; i++) {
            Array.Sort(edges[i]);
            int uniqueCount = 0;
            for(int j = 0; j < edges[i].Length; j++) {
                if (j == 0 || edges[i][j] != edges[i][j-1]) {
                    edges[i][uniqueCount++] = edges[i][j];
                }
            }
            if (uniqueCount < edges[i].Length) {
                Array.Resize(ref edges[i], uniqueCount);
            }
        }

        if (V <= 80000) {
            long bits = (long)V * V;
            ulongsPerRow = (V + 63) / 64;
            long totalUlongs = (long)V * ulongsPerRow;
            if (totalUlongs <= 250_000_000L) { // Fits within 2GB single object limit
                try {
                    adjMatrix = new ulong[totalUlongs];
                    for (int i = 0; i < V; i++) {
                        int rowStart = i * ulongsPerRow;
                        foreach (int w in edges[i]) {
                            adjMatrix[rowStart + (w >> 6)] |= (1UL << (w & 63));
                        }
                    }
                } catch (OutOfMemoryException) {
                    adjMatrix = null;
                }
            }
        }

        order = ComputeDegeneracyOrdering();
        workerPool = new WorkerPool(V);
        
        FindInitialClique();

        Parallel.For(0, V, new ParallelOptions { MaxDegreeOfParallelism = Environment.ProcessorCount }, i_rev => {
            if (stopFlag) return;
            int i = V - 1 - i_rev;
            int v = order[i];
            
            int forwardDegree = 0;
            foreach (int u in edges[v]) {
                if (pos[u] > i) forwardDegree++;
            }
            
            if (1 + forwardDegree <= globalMaxCliqueSize) return;
            
            Worker w = workerPool.Get();
            w.EnsureDepth(0, forwardDegree);
            int[] P = w.P_alloc[0];
            int P_size = 0;
            
            foreach (int u in edges[v]) {
                if (pos[u] > i) {
                    P[P_size++] = u;
                }
            }
            
            w.R[0] = v;
            Search(w, 1, P_size);
            
            workerPool.Return(w);
        });

        stopFlag = true;

        Console.WriteLine(globalMaxCliqueSize);
        for (int i = 0; i < globalMaxCliqueSize; i++) {
            Console.Write(globalMaxClique[i] + (i == globalMaxCliqueSize - 1 ? "" : " "));
        }
        Console.WriteLine();
    }

    [System.Runtime.CompilerServices.MethodImpl(System.Runtime.CompilerServices.MethodImplOptions.AggressiveInlining)]
    public static bool IsAdjacent(int u, int v) {
        if (adjMatrix != null) {
            return (adjMatrix[u * ulongsPerRow + (v >> 6)] & (1UL << (v & 63))) != 0;
        }
        
        int[] edges_u = edges[u];
        int[] edges_v = edges[v];
        if (edges_u.Length > edges_v.Length) {
            int[] tmp = edges_u; edges_u = edges_v; edges_v = tmp;
            int tmp_v = u; u = v; v = tmp_v;
        }
        
        int l = 0, r = edges_u.Length - 1;
        while (l <= r) {
            int m = l + (r - l) / 2;
            int w = edges_u[m];
            if (w == v) return true;
            if (w < v) l = m + 1;
            else r = m - 1;
        }
        return false;
    }

    static int[] ComputeDegeneracyOrdering() {
        int maxDeg = 0;
        for (int i = 0; i < V; i++) {
            if (edges[i].Length > maxDeg) maxDeg = edges[i].Length;
        }
        
        int[] deg = new int[V];
        for (int i = 0; i < V; i++) deg[i] = edges[i].Length;
        
        int[] bin = new int[maxDeg + 1];
        for (int i = 0; i < V; i++) bin[deg[i]]++;
        
        int start = 0;
        for (int d = 0; d <= maxDeg; d++) {
            int num = bin[d];
            bin[d] = start;
            start += num;
        }
        
        int[] localPos = new int[V];
        int[] vert = new int[V];
        for (int i = 0; i < V; i++) {
            localPos[i] = bin[deg[i]];
            vert[localPos[i]] = i;
            bin[deg[i]]++;
        }
        
        for (int d = maxDeg; d > 0; d--) {
            bin[d] = bin[d - 1];
        }
        bin[0] = 0;
        
        int[] localOrder = new int[V];
        for (int i = 0; i < V; i++) {
            int v = vert[i];
            localOrder[i] = v;
            
            foreach (int u in edges[v]) {
                if (localPos[u] > i) {
                    int du = deg[u];
                    int pu = localPos[u];
                    int pw = bin[du];
                    int w = vert[pw];
                    
                    if (u != w) {
                        localPos[u] = pw;
                        localPos[w] = pu;
                        vert[pu] = w;
                        vert[pw] = u;
                    }
                    
                    bin[du]++;
                    deg[u]--;
                }
            }
        }
        
        pos = localPos;
        return localOrder;
    }

    static void FindInitialClique() {
        int[] sortedByDeg = new int[V];
        for (int i = 0; i < V; i++) sortedByDeg[i] = i;
        Array.Sort(sortedByDeg, (a, b) => edges[b].Length.CompareTo(edges[a].Length));
        
        int tries = Math.Min(V, 50);
        for (int i = 0; i < tries; i++) {
            if (stopFlag) return;
            int start = sortedByDeg[i];
            List<int> clique = new List<int>();
            clique.Add(start);
            
            List<int> cands = new List<int>(edges[start]);
            bool exact = cands.Count <= 2000;
            
            while (cands.Count > 0) {
                int bestCand = -1;
                int maxDeg = -1;
                
                foreach (int c in cands) {
                    int deg = exact ? 0 : edges[c].Length;
                    if (exact) {
                        foreach (int other in cands) {
                            if (c != other && IsAdjacent(c, other)) deg++;
                        }
                    }
                    if (deg > maxDeg) {
                        maxDeg = deg;
                        bestCand = c;
                    }
                }
                
                clique.Add(bestCand);
                List<int> nextCands = new List<int>();
                foreach (int c in cands) {
                    if (c != bestCand && IsAdjacent(c, bestCand)) {
                        nextCands.Add(c);
                    }
                }
                cands = nextCands;
            }
            
            lock (lockObj) {
                if (clique.Count > globalMaxCliqueSize) {
                    globalMaxCliqueSize = clique.Count;
                    clique.CopyTo(globalMaxClique);
                }
            }
        }
    }

    static void Search(Worker w, int depth, int P_size) {
        if (stopFlag) return;
        
        if (depth > globalMaxCliqueSize) {
            lock (lockObj) {
                if (depth > globalMaxCliqueSize) {
                    globalMaxCliqueSize = depth;
                    Array.Copy(w.R, globalMaxClique, depth);
                }
            }
        }
        
        if (P_size == 0) return;
        
        w.ColorSort(depth, P_size);
        
        int[] P = w.P_alloc[depth];
        int[] P_color = w.P_color_alloc[depth];
        
        for (int i = P_size - 1; i >= 0; i--) {
            if (stopFlag) return;
            
            int v = P[i];
            int c = P_color[i];
            
            if (depth + c + 1 <= globalMaxCliqueSize) {
                break;
            }
            
            w.EnsureDepth(depth + 1, i);
            int[] P_new = w.P_alloc[depth + 1];
            int P_new_size = 0;
            
            for (int j = 0; j < i; j++) {
                int u = P[j];
                if (IsAdjacent(u, v)) {
                    P_new[P_new_size++] = u;
                }
            }
            
            w.R[depth] = v;
            Search(w, depth + 1, P_new_size);
        }
    }
}

class Worker {
    public int[][] P_alloc = new int[100][];
    public int[][] P_color_alloc = new int[100][];
    public int[] R = new int[100];
    public int[] colorConflict;
    
    public int[] temp = new int[0];
    public int[] colorCount = new int[0];
    public int[] colorStarts = new int[0];
    
    public int stepCounter = 0;
    
    public Worker(int V) {
        colorConflict = new int[V];
    }
    
    public void EnsureDepth(int depth, int size) {
        if (depth >= P_alloc.Length) {
            int newLen = depth * 2;
            Array.Resize(ref P_alloc, newLen);
            Array.Resize(ref P_color_alloc, newLen);
            Array.Resize(ref R, newLen);
        }
        if (P_alloc[depth] == null || P_alloc[depth].Length < size) {
            int newSize = Math.Max(size, 16);
            if (P_alloc[depth] != null) newSize = Math.Max(newSize, P_alloc[depth].Length * 2);
            P_alloc[depth] = new int[newSize];
            P_color_alloc[depth] = new int[newSize];
        }
    }
    
    public void EnsureTemp(int size) {
        if (temp.Length < size) temp = new int[Math.Max(size, temp.Length * 2)];
    }
    public void EnsureColorCounts(int size) {
        if (colorCount.Length < size) colorCount = new int[Math.Max(size, colorCount.Length * 2)];
    }
    public void EnsureColorStarts(int size) {
        if (colorStarts.Length < size) colorStarts = new int[Math.Max(size, colorStarts.Length * 2)];
    }
    
    public void ColorSort(int depth, int P_size) {
        int[] P = P_alloc[depth];
        int[] P_color = P_color_alloc[depth];
        
        int numColors = 0;
        for (int i = 0; i < P_size; i++) {
            int v = P[i];
            stepCounter++;
            if (stepCounter == int.MaxValue) {
                Array.Clear(colorConflict, 0, colorConflict.Length);
                stepCounter = 1;
            }
            
            for (int j = 0; j < i; j++) {
                int u = P[j];
                if (Program.IsAdjacent(u, v)) {
                    colorConflict[P_color[j]] = stepCounter;
                }
            }
            
            int c = 0;
            while (colorConflict[c] == stepCounter) {
                c++;
            }
            P_color[i] = c;
            if (c >= numColors) numColors = c + 1;
        }
        
        EnsureTemp(P_size);
        EnsureColorCounts(numColors);
        EnsureColorStarts(numColors);
        
        Array.Clear(colorCount, 0, numColors);
        for (int i = 0; i < P_size; i++) {
            colorCount[P_color[i]]++;
        }
        
        int pos = 0;
        for (int c = 0; c < numColors; c++) {
            colorStarts[c] = pos;
            pos += colorCount[c];
        }
        
        for (int i = 0; i < P_size; i++) {
            int v = P[i];
            int c = P_color[i];
            int idx = colorStarts[c]++;
            temp[idx] = v;
        }
        
        Array.Copy(temp, 0, P, 0, P_size);
        
        int current_c = 0;
        int count = colorCount[0];
        for (int i = 0; i < P_size; i++) {
            while (count == 0) {
                current_c++;
                count = colorCount[current_c];
            }
            P_color[i] = current_c;
            count--;
        }
    }
}

class WorkerPool {
    private ConcurrentBag<Worker> bag = new ConcurrentBag<Worker>();
    private int V;
    public WorkerPool(int V) { this.V = V; }
    public Worker Get() {
        if (bag.TryTake(out Worker w)) return w;
        return new Worker(V);
    }
    public void Return(Worker w) {
        bag.Add(w);
    }
}

class Scanner {
    private Stream stream;
    private byte[] buffer = new byte[1024 * 1024];
    private int head;
    private int tail;

    public Scanner(Stream stream) {
        this.stream = stream;
    }

    public int NextInt() {
        int c = Read();
        while (c <= 32) {
            if (c == -1) return -1;
            c = Read();
        }
        int res = 0;
        while (c > 32) {
            res = res * 10 + c - '0';
            c = Read();
        }
        return res;
    }

    private int Read() {
        if (head >= tail) {
            head = 0;
            tail = stream.Read(buffer, 0, buffer.Length);
            if (tail <= 0) return -1;
        }
        return buffer[head++];
    }
}
