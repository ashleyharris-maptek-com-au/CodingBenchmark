#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>
#include <numeric>

using namespace std;

const long long INF = 10000000000000000LL;

struct FullEdge {
    int u, v;
    long long weight;
    int id;
};

struct EdgeInfo {
    int to;
    long long weight;
    int id;
};

struct TreeResult {
    long long weight;
    vector<int> edges;
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int V, E, K_orig;
    if (!(cin >> V >> E >> K_orig)) return 0;

    vector<FullEdge> edge_list(E);
    vector<vector<EdgeInfo>> adj(V);

    for (int i = 0; i < E; ++i) {
        cin >> edge_list[i].u >> edge_list[i].v >> edge_list[i].weight;
        edge_list[i].id = i;
        adj[edge_list[i].u].push_back({edge_list[i].v, edge_list[i].weight, i});
        adj[edge_list[i].v].push_back({edge_list[i].u, edge_list[i].weight, i});
    }

    vector<int> terminals(K_orig);
    vector<bool> is_terminal(V, false);
    for (int i = 0; i < K_orig; ++i) {
        cin >> terminals[i];
    }
    
    sort(terminals.begin(), terminals.end());
    terminals.erase(unique(terminals.begin(), terminals.end()), terminals.end());
    int K = terminals.size();
    
    for (int t : terminals) {
        is_terminal[t] = true;
    }
    
    if (K <= 1) {
        cout << "0 0\n";
        return 0;
    }

    auto optimize_tree = [&](const vector<int>& edges_in) -> TreeResult {
        if (K <= 1) return {0, {}};
        vector<bool> in_nodes(V, false);
        for (int e : edges_in) {
            in_nodes[edge_list[e].u] = true;
            in_nodes[edge_list[e].v] = true;
        }
        for (int t : terminals) {
            in_nodes[t] = true;
        }
        
        vector<FullEdge> sub_edges;
        for (int e = 0; e < E; ++e) {
            if (in_nodes[edge_list[e].u] && in_nodes[edge_list[e].v]) {
                sub_edges.push_back(edge_list[e]);
            }
        }
        
        sort(sub_edges.begin(), sub_edges.end(), [](const FullEdge& a, const FullEdge& b) {
            return a.weight < b.weight;
        });
        
        vector<int> parent(V);
        iota(parent.begin(), parent.end(), 0);
        auto find_set = [&](int i, auto& find_set_ref) -> int {
            if (parent[i] == i) return i;
            return parent[i] = find_set_ref(parent[i], find_set_ref);
        };
        auto union_set = [&](int i, int j, auto& find_set_ref) {
            int root_i = find_set_ref(i, find_set_ref);
            int root_j = find_set_ref(j, find_set_ref);
            if (root_i != root_j) {
                parent[root_i] = root_j;
                return true;
            }
            return false;
        };
        
        vector<FullEdge> mst_edges;
        vector<vector<pair<int, int>>> mst_adj(V);
        vector<int> degree(V, 0);
        
        long long mst_weight = 0;
        for (const auto& e : sub_edges) {
            if (union_set(e.u, e.v, find_set)) {
                mst_edges.push_back(e);
                mst_weight += e.weight;
                mst_adj[e.u].push_back({e.v, e.id});
                mst_adj[e.v].push_back({e.u, e.id});
                degree[e.u]++;
                degree[e.v]++;
            }
        }
        
        queue<int> leaves;
        for (int u = 0; u < V; ++u) {
            if (in_nodes[u] && degree[u] == 1 && !is_terminal[u]) {
                leaves.push(u);
            }
        }
        
        vector<bool> pruned_edges(E, false);
        
        while (!leaves.empty()) {
            int u = leaves.front();
            leaves.pop();
            
            for (auto& edge : mst_adj[u]) {
                int v = edge.first;
                int id = edge.second;
                if (!pruned_edges[id]) {
                    pruned_edges[id] = true;
                    mst_weight -= edge_list[id].weight;
                    degree[v]--;
                    degree[u]--;
                    if (degree[v] == 1 && !is_terminal[v]) {
                        leaves.push(v);
                    }
                    break;
                }
            }
        }
        
        TreeResult res;
        res.weight = mst_weight;
        for (const auto& e : mst_edges) {
            if (!pruned_edges[e.id]) {
                res.edges.push_back(e.id);
            }
        }
        
        return res;
    };

    bool use_dp = false;
    if (K <= 18) {
        long long ops = 1;
        for (int i = 0; i < K; ++i) ops *= 3;
        long long elements = (1LL << K) * (long long)V;
        if (ops * V / 2 <= 500000000LL && elements <= 5000000LL) {
            use_dp = true;
        }
    }

    TreeResult final_res;
    final_res.weight = INF;

    if (use_dp) {
        struct PrevState {
            uint8_t type;
            int v1;
            int v2;
        };
        vector<vector<long long>> dp(1 << K, vector<long long>(V, INF));
        vector<vector<PrevState>> prev_state(1 << K, vector<PrevState>(V, {0, -1, -1}));

        for (int i = 0; i < K; ++i) {
            dp[1 << i][terminals[i]] = 0;
            prev_state[1 << i][terminals[i]] = {0, -1, -1};
        }

        priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;

        for (int mask = 1; mask < (1 << K); ++mask) {
            if ((mask & (mask - 1)) != 0) {
                for (int sub = (mask - 1) & mask; sub > (mask ^ sub); sub = (sub - 1) & mask) {
                    int comp = mask ^ sub;
                    for (int u = 0; u < V; ++u) {
                        long long val = dp[sub][u] + dp[comp][u];
                        if (val < dp[mask][u]) {
                            dp[mask][u] = val;
                            prev_state[mask][u] = {1, sub, comp};
                        }
                    }
                }
            }
            
            for (int u = 0; u < V; ++u) {
                if (dp[mask][u] < INF) {
                    pq.push({dp[mask][u], u});
                }
            }
            
            while (!pq.empty()) {
                auto [d, u] = pq.top();
                pq.pop();
                if (d > dp[mask][u]) continue;
                
                for (auto& edge : adj[u]) {
                    int v = edge.to;
                    long long w = edge.weight;
                    if (dp[mask][u] + w < dp[mask][v]) {
                        dp[mask][v] = dp[mask][u] + w;
                        prev_state[mask][v] = {2, u, edge.id};
                        pq.push({dp[mask][v], v});
                    }
                }
            }
        }

        vector<int> dp_edges;
        auto traceback = [&](int mask, int u, auto& traceback_ref) -> void {
            if (mask == 0) return;
            auto st = prev_state[mask][u];
            if (st.type == 0) {
                return;
            } else if (st.type == 1) {
                traceback_ref(st.v1, u, traceback_ref);
                traceback_ref(st.v2, u, traceback_ref);
            } else if (st.type == 2) {
                dp_edges.push_back(st.v2);
                traceback_ref(mask, st.v1, traceback_ref);
            }
        };
        
        long long best_dp_weight = INF;
        int best_dp_u = -1;
        for (int u = 0; u < V; ++u) {
            if (dp[(1 << K) - 1][u] < best_dp_weight) {
                best_dp_weight = dp[(1 << K) - 1][u];
                best_dp_u = u;
            }
        }
        
        if (best_dp_weight < INF / 2) {
            traceback((1 << K) - 1, best_dp_u, traceback);
            sort(dp_edges.begin(), dp_edges.end());
            dp_edges.erase(unique(dp_edges.begin(), dp_edges.end()), dp_edges.end());
            final_res = optimize_tree(dp_edges);
        }
    } else {
        long long best_total_weight = INF;
        vector<int> best_tree_edges;

        vector<int> starts;
        int num_starts = min((int)terminals.size(), 100);
        int step = max(1, (int)terminals.size() / num_starts);
        for (int i = 0; i < terminals.size() && starts.size() < num_starts; i += step) {
            starts.push_back(terminals[i]);
        }
        for (int i = 0; i < terminals.size() && starts.size() < num_starts; ++i) {
            if (find(starts.begin(), starts.end(), terminals[i]) == starts.end()) {
                starts.push_back(terminals[i]);
            }
        }

        for (int start_idx = 0; start_idx < starts.size(); ++start_idx) {
            int start_term = starts[start_idx];
            
            vector<long long> dist(V, INF);
            vector<int> parent_edge(V, -1);
            vector<int> parent_node(V, -1);
            vector<bool> in_tree(V, false);
            
            dist[start_term] = 0;
            
            priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;
            pq.push({0, start_term});
            
            in_tree[start_term] = true;
            
            int terminals_found = 1;
            vector<int> tree_edges;
            
            vector<bool> term_in_tree(V, false);
            term_in_tree[start_term] = true;
            
            while (terminals_found < K) {
                int closest_term = -1;
                
                while (!pq.empty()) {
                    auto [d, u] = pq.top();
                    pq.pop();
                    
                    if (d > dist[u]) continue;
                    
                    if (is_terminal[u] && !term_in_tree[u]) {
                        closest_term = u;
                        break;
                    }
                    
                    for (auto& edge : adj[u]) {
                        int v = edge.to;
                        long long w = edge.weight;
                        if (dist[u] + w < dist[v]) {
                            dist[v] = dist[u] + w;
                            parent_edge[v] = edge.id;
                            parent_node[v] = u;
                            pq.push({dist[v], v});
                        }
                    }
                }
                
                if (closest_term == -1) break;
                
                int curr = closest_term;
                vector<int> path_nodes;
                vector<int> path_edges;
                
                while (!in_tree[curr]) {
                    path_nodes.push_back(curr);
                    path_edges.push_back(parent_edge[curr]);
                    curr = parent_node[curr];
                }
                
                for (int i = 0; i < (int)path_nodes.size(); ++i) {
                    int u = path_nodes[i];
                    in_tree[u] = true;
                    tree_edges.push_back(path_edges[i]);
                    
                    if (is_terminal[u] && !term_in_tree[u]) {
                        term_in_tree[u] = true;
                        terminals_found++;
                    }
                    
                    dist[u] = 0;
                    pq.push({0, u});
                }
            }
            
            if (terminals_found == K) {
                TreeResult opt = optimize_tree(tree_edges);
                if (opt.weight < best_total_weight) {
                    best_total_weight = opt.weight;
                    best_tree_edges = opt.edges;
                }
            }
        }
        
        final_res.weight = best_total_weight;
        final_res.edges = best_tree_edges;
    }

    if (final_res.weight >= INF / 2) {
        cout << "0 0\n";
        return 0;
    }

    cout << final_res.weight << " " << final_res.edges.size() << "\n";
    for (int id : final_res.edges) {
        cout << edge_list[id].u << " " << edge_list[id].v << "\n";
    }

    return 0;
}
