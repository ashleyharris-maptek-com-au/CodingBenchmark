using System;
using System.IO;
using System.Collections.Generic;
using System.Diagnostics;
using System.Globalization;
using System.Text;

class Program {
    struct Vec3 {
        public double x, y, z;
        public Vec3(double x, double y, double z) { this.x = x; this.y = y; this.z = z; }
    }
    struct Quat {
        public double w, x, y, z;
        public Quat(double w, double x, double y, double z) { this.w = w; this.x = x; this.y = y; this.z = z; }
        public static Quat Identity = new Quat(1, 0, 0, 0);
        public static Quat Multiply(Quat a, Quat b) {
            return new Quat(
                a.w * b.w - a.x * b.x - a.y * b.y - a.z * b.z,
                a.w * b.x + a.x * b.w + a.y * b.z - a.z * b.y,
                a.w * b.y - a.x * b.z + a.y * b.w + a.z * b.x,
                a.w * b.z + a.x * b.y - a.y * b.x + a.z * b.w
            );
        }
        public static Quat Normalize(Quat q) {
            double len = Math.Sqrt(q.w * q.w + q.x * q.x + q.y * q.y + q.z * q.z);
            if (len < 1e-9) return Identity;
            return new Quat(q.w / len, q.x / len, q.y / len, q.z / len);
        }
        public static double Dot(Quat a, Quat b) {
            return a.w*b.w + a.x*b.x + a.y*b.y + a.z*b.z;
        }
    }
    
    struct TriInfo {
        public double u0, v0, z0;
        public double u1, v1, z1;
        public double u2, v2, z2;
        public double Nu, Nv, Nz;
        public double minU, maxU, minV, maxV;
        public bool isVertical;
    }

    static Vec3 Rotate(Vec3 v, Quat q) {
        Quat vq = new Quat(0, v.x, v.y, v.z);
        Quat qConj = new Quat(q.w, -q.x, -q.y, -q.z);
        Quat res = Quat.Multiply(Quat.Multiply(q, vq), qConj);
        return new Vec3(res.x, res.y, res.z);
    }
    
    static Quat RandomRotation(Random rnd) {
        double u1 = rnd.NextDouble();
        double u2 = rnd.NextDouble();
        double u3 = rnd.NextDouble();
        double w = Math.Sqrt(1 - u1) * Math.Sin(2 * Math.PI * u2);
        double x = Math.Sqrt(1 - u1) * Math.Cos(2 * Math.PI * u2);
        double y = Math.Sqrt(u1) * Math.Sin(2 * Math.PI * u3);
        double z = Math.Sqrt(u1) * Math.Cos(2 * Math.PI * u3);
        return Quat.Normalize(new Quat(w, x, y, z));
    }
    
    static List<Quat> GetOrthogonalRotations() {
        List<Quat> rotations = new List<Quat>();
        double sqrt2 = Math.Sqrt(2) / 2;
        Quat rx = new Quat(sqrt2, sqrt2, 0, 0);
        Quat ry = new Quat(sqrt2, 0, sqrt2, 0);
        Quat rz = new Quat(sqrt2, 0, 0, sqrt2);
        Quat[] bases = { rx, ry, rz };

        Queue<Quat> q = new Queue<Quat>();
        q.Enqueue(Quat.Identity);
        rotations.Add(Quat.Identity);

        while(q.Count > 0) {
            Quat curr = q.Dequeue();
            foreach (var b in bases) {
                Quat next = Quat.Normalize(Quat.Multiply(curr, b));
                bool found = false;
                foreach (var r in rotations) {
                    if (Math.Abs(Quat.Dot(r, next)) > 0.999) {
                        found = true;
                        break;
                    }
                }
                if (!found) {
                    rotations.Add(next);
                    q.Enqueue(next);
                }
            }
        }
        return rotations;
    }
    
    static long CalcCount(double containerSize, double aabbSize, double stepSize) {
        if (containerSize < aabbSize - 1e-10) return 0;
        if (stepSize < 1e-5) stepSize = 1e-5;
        double val = (containerSize - aabbSize) / stepSize;
        if (val < 0) return 1;
        return 1 + (long)(val + 1e-10);
    }
    
    static double Det2D(double u1, double v1, double u2, double v2, double u3, double v3) {
        return (u2 - u1) * (v3 - v1) - (v2 - v1) * (u3 - u1);
    }

    static bool PointInTriangle2D(double pu, double pv, double u1, double v1, double u2, double v2, double u3, double v3) {
        double d1 = Det2D(pu, pv, u1, v1, u2, v2);
        double d2 = Det2D(pu, pv, u2, v2, u3, v3);
        double d3 = Det2D(pu, pv, u3, v3, u1, v1);
        bool has_neg = (d1 < -1e-7) || (d2 < -1e-7) || (d3 < -1e-7);
        bool has_pos = (d1 > 1e-7) || (d2 > 1e-7) || (d3 > 1e-7);
        return !(has_neg && has_pos);
    }
    
    static void CheckAndEval(double u, double v, double zA, ref TriInfo B, ref double maxDiff) {
        if (PointInTriangle2D(u, v, B.u0, B.v0, B.u1, B.v1, B.u2, B.v2)) {
            double zB = B.z0 - (B.Nu * (u - B.u0) + B.Nv * (v - B.v0)) / B.Nz;
            double diff = Math.Abs(zA - zB);
            if (diff > maxDiff) maxDiff = diff;
        }
    }
    
    static void CheckSegIntersect(double p0u, double p0v, double p1u, double p1v, 
                                   double p2u, double p2v, double p3u, double p3v, 
                                   ref TriInfo A, ref TriInfo B, ref double maxDiff) {
        double s1u = p1u - p0u, s1v = p1v - p0v;
        double s2u = p3u - p2u, s2v = p3v - p2v;
        double den = -s2u * s1v + s1u * s2v;
        if (Math.Abs(den) < 1e-9) return;
        double s = (-s1v * (p0u - p2u) + s1u * (p0v - p2v)) / den;
        double t = ( s2u * (p0v - p2v) - s2v * (p0u - p2u)) / den;
        if (s >= -1e-5 && s <= 1.00001 && t >= -1e-5 && t <= 1.00001) {
            double iu = p0u + (t * s1u);
            double iv = p0v + (t * s1v);
            double zA = A.z0 - (A.Nu * (iu - A.u0) + A.Nv * (iv - A.v0)) / A.Nz;
            double zB = B.z0 - (B.Nu * (iu - B.u0) + B.Nv * (iv - B.v0)) / B.Nz;
            double diff = Math.Abs(zA - zB);
            if (diff > maxDiff) maxDiff = diff;
        }
    }
    
    static void CheckEdgeIntersection(double p0u, double p0v, double p1u, double p1v, ref TriInfo A, ref TriInfo B, ref double maxDiff) {
        CheckSegIntersect(p0u, p0v, p1u, p1v, B.u0, B.v0, B.u1, B.v1, ref A, ref B, ref maxDiff);
        CheckSegIntersect(p0u, p0v, p1u, p1v, B.u1, B.v1, B.u2, B.v2, ref A, ref B, ref maxDiff);
        CheckSegIntersect(p0u, p0v, p1u, p1v, B.u2, B.v2, B.u0, B.v0, ref A, ref B, ref maxDiff);
    }
    
    static double TriOverlapMaxZDiff(ref TriInfo A, ref TriInfo B) {
        double maxDiff = 0;
        CheckAndEval(A.u0, A.v0, A.z0, ref B, ref maxDiff);
        CheckAndEval(A.u1, A.v1, A.z1, ref B, ref maxDiff);
        CheckAndEval(A.u2, A.v2, A.z2, ref B, ref maxDiff);
        
        CheckAndEval(B.u0, B.v0, B.z0, ref A, ref maxDiff);
        CheckAndEval(B.u1, B.v1, B.z1, ref A, ref maxDiff);
        CheckAndEval(B.u2, B.v2, B.z2, ref A, ref maxDiff);
        
        CheckEdgeIntersection(A.u0, A.v0, A.u1, A.v1, ref A, ref B, ref maxDiff);
        CheckEdgeIntersection(A.u1, A.v1, A.u2, A.v2, ref A, ref B, ref maxDiff);
        CheckEdgeIntersection(A.u2, A.v2, A.u0, A.v0, ref A, ref B, ref maxDiff);
        
        return maxDiff;
    }
    
    static long testCounter = 1;
    static long[] pairTested = null;
    
    static double ComputeStackDistance(Vec3[] rotVerts, int[] faces, int axis) {
        int F = faces.Length / 3;
        TriInfo[] tris = new TriInfo[F];
        double min_u = double.MaxValue, max_u = double.MinValue;
        double min_v = double.MaxValue, max_v = double.MinValue;
        
        for (int i = 0; i < F; i++) {
            Vec3 v0 = rotVerts[faces[i*3]];
            Vec3 v1 = rotVerts[faces[i*3+1]];
            Vec3 v2 = rotVerts[faces[i*3+2]];
            
            double u0, v0_, z0, u1, v1_, z1, u2, v2_, z2;
            if (axis == 0) { 
                u0 = v0.y; v0_ = v0.z; z0 = v0.x;
                u1 = v1.y; v1_ = v1.z; z1 = v1.x;
                u2 = v2.y; v2_ = v2.z; z2 = v2.x;
            } else if (axis == 1) { 
                u0 = v0.z; v0_ = v0.x; z0 = v0.y;
                u1 = v1.z; v1_ = v1.x; z1 = v1.y;
                u2 = v2.z; v2_ = v2.x; z2 = v2.y;
            } else { 
                u0 = v0.x; v0_ = v0.y; z0 = v0.z;
                u1 = v1.x; v1_ = v1.y; z1 = v1.z;
                u2 = v2.x; v2_ = v2.y; z2 = v2.z;
            }
            
            TriInfo t = new TriInfo();
            t.u0 = u0; t.v0 = v0_; t.z0 = z0;
            t.u1 = u1; t.v1 = v1_; t.z1 = z1;
            t.u2 = u2; t.v2 = v2_; t.z2 = z2;
            
            t.minU = Math.Min(u0, Math.Min(u1, u2));
            t.maxU = Math.Max(u0, Math.Max(u1, u2));
            t.minV = Math.Min(v0_, Math.Min(v1_, v2_));
            t.maxV = Math.Max(v0_, Math.Max(v1_, v2_));
            
            double e1u = u1 - u0, e1v = v1_ - v0_, e1z = z1 - z0;
            double e2u = u2 - u0, e2v = v2_ - v0_, e2z = z2 - z0;
            t.Nu = e1v * e2z - e1z * e2v;
            t.Nv = e1z * e2u - e1u * e2z;
            t.Nz = e1u * e2v - e1v * e2u;
            
            if (Math.Abs(t.Nz) < 1e-7) t.isVertical = true;
            else t.isVertical = false;
            
            tris[i] = t;
            
            if (t.minU < min_u) min_u = t.minU;
            if (t.maxU > max_u) max_u = t.maxU;
            if (t.minV < min_v) min_v = t.minV;
            if (t.maxV > max_v) max_v = t.maxV;
        }
        
        int GRID_SIZE = 20;
        List<int>[,] grid = new List<int>[GRID_SIZE, GRID_SIZE];
        double stepU = (max_u - min_u) / GRID_SIZE;
        double stepV = (max_v - min_v) / GRID_SIZE;
        if (stepU < 1e-9) stepU = 1.0;
        if (stepV < 1e-9) stepV = 1.0;
        
        for (int i = 0; i < F; i++) {
            if (tris[i].isVertical) continue;
            int minGu = (int)((tris[i].minU - min_u) / stepU);
            int maxGu = (int)((tris[i].maxU - min_u) / stepU);
            int minGv = (int)((tris[i].minV - min_v) / stepV);
            int maxGv = (int)((tris[i].maxV - min_v) / stepV);
            minGu = Math.Max(0, Math.Min(GRID_SIZE - 1, minGu));
            maxGu = Math.Max(0, Math.Min(GRID_SIZE - 1, maxGu));
            minGv = Math.Max(0, Math.Min(GRID_SIZE - 1, minGv));
            maxGv = Math.Max(0, Math.Min(GRID_SIZE - 1, maxGv));
            for (int gu = minGu; gu <= maxGu; gu++) {
                for (int gv = minGv; gv <= maxGv; gv++) {
                    if (grid[gu, gv] == null) grid[gu, gv] = new List<int>();
                    grid[gu, gv].Add(i);
                }
            }
        }
        
        double maxDz = 0;
        bool useArray = F <= 5000;
        HashSet<long> testedSet = null;

        if (useArray) {
            if (pairTested == null || pairTested.Length < F * F) {
                pairTested = new long[F * F];
            }
            testCounter++;
        } else {
            testedSet = new HashSet<long>();
        }
        
        for (int gu = 0; gu < GRID_SIZE; gu++) {
            for (int gv = 0; gv < GRID_SIZE; gv++) {
                var list = grid[gu, gv];
                if (list == null) continue;
                for (int i = 0; i < list.Count; i++) {
                    int tA = list[i];
                    for (int j = i + 1; j < list.Count; j++) {
                        int tB = list[j];
                        
                        if (useArray) {
                            long pairId = (long)tA * F + tB;
                            if (pairTested[pairId] == testCounter) continue;
                            pairTested[pairId] = testCounter;
                        } else {
                            long pairId = ((long)tA << 32) | (long)tB;
                            if (!testedSet.Add(pairId)) continue;
                        }
                        
                        if (tris[tA].maxU < tris[tB].minU - 1e-7 || tris[tA].minU > tris[tB].maxU + 1e-7 ||
                            tris[tA].maxV < tris[tB].minV - 1e-7 || tris[tA].minV > tris[tB].maxV + 1e-7)
                            continue;
                            
                        double localMax = TriOverlapMaxZDiff(ref tris[tA], ref tris[tB]);
                        if (localMax > maxDz) maxDz = localMax;
                    }
                }
            }
        }
        
        if (maxDz < 1e-5) return 0.0;
        return maxDz + 1e-5;
    }

    static Queue<string> tokenQueue = new Queue<string>();
    static string GetNextToken() {
        while (tokenQueue.Count == 0) {
            string line = Console.ReadLine();
            if (line == null) return null;
            string[] parts = line.Trim().Split(new char[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
            foreach (var p in parts) tokenQueue.Enqueue(p);
        }
        return tokenQueue.Dequeue();
    }

    static void Main() {
        string tokenV = GetNextToken();
        if (tokenV == null) return;
        int V = int.Parse(tokenV);
        int F = int.Parse(GetNextToken());

        Vec3[] vertices = new Vec3[V];
        for (int i = 0; i < V; i++) {
            vertices[i] = new Vec3(
                double.Parse(GetNextToken(), CultureInfo.InvariantCulture),
                double.Parse(GetNextToken(), CultureInfo.InvariantCulture),
                double.Parse(GetNextToken(), CultureInfo.InvariantCulture)
            );
        }

        int[] faces = new int[F * 3];
        for (int i = 0; i < F; i++) {
            faces[i * 3] = int.Parse(GetNextToken());
            faces[i * 3 + 1] = int.Parse(GetNextToken());
            faces[i * 3 + 2] = int.Parse(GetNextToken());
        }

        double W = double.Parse(GetNextToken(), CultureInfo.InvariantCulture);
        double H = double.Parse(GetNextToken(), CultureInfo.InvariantCulture);
        double D = double.Parse(GetNextToken(), CultureInfo.InvariantCulture);

        var OrthogonalRotations = GetOrthogonalRotations();
        Random rnd = new Random(42);

        int bestMode = 3; 
        Quat bestQ = Quat.Identity;
        double bestDx = 0, bestDy = 0, bestDz = 0;
        double bestMinX = 0, bestMinY = 0, bestMinZ = 0;
        double bestAabbX = 0, bestAabbY = 0, bestAabbZ = 0;
        long bestTotalCount = 0;

        bool enableDense = true;
        Stopwatch swTotal = Stopwatch.StartNew();

        for(int iter = 0; iter < 100000; iter++) {
            if (swTotal.ElapsedMilliseconds > 15000) break; 
            
            Quat q;
            if (iter < OrthogonalRotations.Count) q = OrthogonalRotations[iter];
            else q = RandomRotation(rnd);
            
            Vec3[] rotVerts = new Vec3[V];
            double minX = double.MaxValue, maxX = double.MinValue;
            double minY = double.MaxValue, maxY = double.MinValue;
            double minZ = double.MaxValue, maxZ = double.MinValue;
            
            for(int i = 0; i < V; i++) {
                Vec3 v = Rotate(vertices[i], q);
                rotVerts[i] = v;
                if(v.x < minX) minX = v.x;
                if(v.x > maxX) maxX = v.x;
                if(v.y < minY) minY = v.y;
                if(v.y > maxY) maxY = v.y;
                if(v.z < minZ) minZ = v.z;
                if(v.z > maxZ) maxZ = v.z;
            }
            
            double aabb_dx = maxX - minX;
            double aabb_dy = maxY - minY;
            double aabb_dz = maxZ - minZ;
            if (aabb_dx < 1e-5) aabb_dx = 1e-5;
            if (aabb_dy < 1e-5) aabb_dy = 1e-5;
            if (aabb_dz < 1e-5) aabb_dz = 1e-5;
            
            long count3 = CalcCount(W, aabb_dx, aabb_dx) * CalcCount(H, aabb_dy, aabb_dy) * CalcCount(D, aabb_dz, aabb_dz);
            long maxCount = count3;
            int mode = 3;
            double stepX = aabb_dx, stepY = aabb_dy, stepZ = aabb_dz;
            
            if (enableDense) {
                Stopwatch swDense = Stopwatch.StartNew();
                
                double dz_s = ComputeStackDistance(rotVerts, faces, 2);
                if (dz_s < 1e-5) dz_s = aabb_dz;
                double dz_stack = Math.Min(aabb_dz, dz_s);
                long countZ = CalcCount(W, aabb_dx, aabb_dx) * CalcCount(H, aabb_dy, aabb_dy) * CalcCount(D, aabb_dz, dz_stack);
                if (countZ > maxCount) { maxCount = countZ; mode = 2; stepZ = dz_stack; stepX = aabb_dx; stepY = aabb_dy; }
                
                double dy_s = ComputeStackDistance(rotVerts, faces, 1);
                if (dy_s < 1e-5) dy_s = aabb_dy;
                double dy_stack = Math.Min(aabb_dy, dy_s);
                long countY = CalcCount(W, aabb_dx, aabb_dx) * CalcCount(H, aabb_dy, dy_stack) * CalcCount(D, aabb_dz, aabb_dz);
                if (countY > maxCount) { maxCount = countY; mode = 1; stepY = dy_stack; stepX = aabb_dx; stepZ = aabb_dz; }
                
                double dx_s = ComputeStackDistance(rotVerts, faces, 0);
                if (dx_s < 1e-5) dx_s = aabb_dx;
                double dx_stack = Math.Min(aabb_dx, dx_s);
                long countX = CalcCount(W, aabb_dx, dx_stack) * CalcCount(H, aabb_dy, aabb_dy) * CalcCount(D, aabb_dz, aabb_dz);
                if (countX > maxCount) { maxCount = countX; mode = 0; stepX = dx_stack; stepY = aabb_dy; stepZ = aabb_dz; }
                
                if (swDense.ElapsedMilliseconds > 200 && iter >= OrthogonalRotations.Count) {
                    enableDense = false; 
                }
            }
            
            if (maxCount > bestTotalCount) {
                bestTotalCount = maxCount;
                bestQ = q;
                bestMode = mode;
                bestDx = stepX; bestDy = stepY; bestDz = stepZ;
                bestAabbX = aabb_dx; bestAabbY = aabb_dy; bestAabbZ = aabb_dz;
                bestMinX = minX; bestMinY = minY; bestMinZ = minZ;
            }
        }

        Console.WriteLine(bestTotalCount);
        if (bestTotalCount == 0) return;

        long finalCountX = CalcCount(W, bestAabbX, bestDx);
        long finalCountY = CalcCount(H, bestAabbY, bestDy);
        long finalCountZ = CalcCount(D, bestAabbZ, bestDz);

        string[] txStrings = new string[finalCountX];
        for(int i = 0; i < finalCountX; i++) {
            txStrings[i] = (i * bestDx - bestMinX).ToString("0.00000", CultureInfo.InvariantCulture);
        }

        string[] tyStrings = new string[finalCountY];
        for(int i = 0; i < finalCountY; i++) {
            tyStrings[i] = (i * bestDy - bestMinY).ToString("0.00000", CultureInfo.InvariantCulture);
        }

        string[] tzLines = new string[finalCountZ];
        string qStr = $"{bestQ.w.ToString("0.00000", CultureInfo.InvariantCulture)} {bestQ.x.ToString("0.00000", CultureInfo.InvariantCulture)} {bestQ.y.ToString("0.00000", CultureInfo.InvariantCulture)} {bestQ.z.ToString("0.00000", CultureInfo.InvariantCulture)}";
        for(int i = 0; i < finalCountZ; i++) {
            string tz = (i * bestDz - bestMinZ).ToString("0.00000", CultureInfo.InvariantCulture);
            tzLines[i] = tz + " " + qStr + "\n";
        }

        using (StreamWriter sw = new StreamWriter(Console.OpenStandardOutput(), new UTF8Encoding(false), 1024 * 1024)) {
            for(long ix = 0; ix < finalCountX; ix++) {
                string sx = txStrings[ix];
                for(long iy = 0; iy < finalCountY; iy++) {
                    string sy = tyStrings[iy];
                    for(long iz = 0; iz < finalCountZ; iz++) {
                        sw.Write(sx);
                        sw.Write(" ");
                        sw.Write(sy);
                        sw.Write(" ");
                        sw.Write(tzLines[iz]);
                    }
                }
            }
        }
    }
