use std::io::{Read, Write, BufWriter};

struct Scanner<'a> {
    bytes: &'a [u8],
    pos: usize,
}

impl<'a> Scanner<'a> {
    fn new(bytes: &'a [u8]) -> Self {
        Self { bytes, pos: 0 }
    }

    fn next_u32(&mut self) -> Option<u32> {
        while self.pos < self.bytes.len() && self.bytes[self.pos] <= b' ' {
            self.pos += 1;
        }
        if self.pos >= self.bytes.len() {
            return None;
        }
        let mut res = 0;
        while self.pos < self.bytes.len() && self.bytes[self.pos] > b' ' {
            res = res * 10 + (self.bytes[self.pos] - b'0') as u32;
            self.pos += 1;
        }
        Some(res)
    }
}

#[inline]
fn remove_from_bucket(u: usize, d: usize, head: &mut [i32], prev: &mut [i32], next: &mut [i32]) {
    let p = prev[u];
    let n_ = next[u];
    if p != -1 {
        next[p as usize] = n_;
    } else {
        head[d] = n_;
    }
    if n_ != -1 {
        prev[n_ as usize] = p;
    }
}

#[inline]
fn add_to_bucket(u: usize, d: usize, head: &mut [i32], prev: &mut [i32], next: &mut [i32]) {
    let h = head[d];
    next[u] = h;
    prev[u] = -1;
    if h != -1 {
        prev[h as usize] = u as i32;
    }
    head[d] = u as i32;
}

fn main() {
    let mut input = Vec::new();
    if std::io::stdin().read_to_end(&mut input).is_err() {
        return;
    }
    let mut scanner = Scanner::new(&input);

    let num_vertices = match scanner.next_u32() {
        Some(x) => x as usize,
        None => return,
    };
    let num_edges = match scanner.next_u32() {
        Some(x) => x as usize,
        None => return,
    };

    let mut raw_edges = Vec::with_capacity(num_edges);
    let mut must_include = vec![false; num_vertices];

    for _ in 0..num_edges {
        let u = match scanner.next_u32() {
            Some(x) => x as usize,
            None => break,
        };
        let v = match scanner.next_u32() {
            Some(x) => x as usize,
            None => break,
        };
        if u == v {
            must_include[u] = true;
        } else {
            raw_edges.push((u, v));
        }
    }

    let mut degree_count = vec![0u32; num_vertices];
    for &(u, v) in &raw_edges {
        if !must_include[u] && !must_include[v] {
            degree_count[u] += 1;
            degree_count[v] += 1;
        }
    }

    let mut offsets = vec![0; num_vertices + 1];
    for i in 0..num_vertices {
        offsets[i+1] = offsets[i] + degree_count[i] as usize;
    }

    let mut edges = vec![0u32; offsets[num_vertices]];
    let mut head_pos = offsets.clone();
    for &(u, v) in &raw_edges {
        if !must_include[u] && !must_include[v] {
            edges[head_pos[u]] = v as u32;
            head_pos[u] += 1;
            edges[head_pos[v]] = u as u32;
            head_pos[v] += 1;
        }
    }

    let mut final_edges = Vec::with_capacity(edges.len());
    let mut final_offsets = vec![0; num_vertices + 1];
    for i in 0..num_vertices {
        let start = offsets[i];
        let end = offsets[i+1];
        let slice = &mut edges[start..end];
        slice.sort_unstable();

        if !slice.is_empty() {
            final_edges.push(slice[0]);
            for j in 1..slice.len() {
                if slice[j] != slice[j-1] {
                    final_edges.push(slice[j]);
                }
            }
        }
        final_offsets[i+1] = final_edges.len();
    }

    let edges = final_edges;
    let offsets = final_offsets;

    let n = num_vertices;
    let mut active = vec![false; n];
    let mut degree = vec![0u32; n];
    let mut max_initial_deg = 0;

    for i in 0..n {
        if !must_include[i] {
            active[i] = true;
            let d = (offsets[i+1] - offsets[i]) as u32;
            degree[i] = d;
            if d > max_initial_deg {
                max_initial_deg = d;
            }
        }
    }

    let max_deg_usize = std::cmp::max(max_initial_deg as usize, 2);
    let mut head = vec![-1i32; max_deg_usize + 1];
    let mut prev = vec![-1i32; n];
    let mut next = vec![-1i32; n];

    for i in 0..n {
        if active[i] {
            let d = degree[i] as usize;
            let h = head[d];
            next[i] = h;
            prev[i] = -1;
            if h != -1 {
                prev[h as usize] = i as i32;
            }
            head[d] = i as i32;
        }
    }

    let mut max_deg = max_deg_usize;
    let mut vc = Vec::new();

    loop {
        let mut changed = false;
        while head[0] != -1 {
            let u = head[0] as usize;
            remove_from_bucket(u, 0, &mut head, &mut prev, &mut next);
            active[u] = false;
            changed = true;
        }
        if changed { continue; }

        if head[1] != -1 {
            let u = head[1] as usize;
            remove_from_bucket(u, 1, &mut head, &mut prev, &mut next);
            active[u] = false;

            let mut neighbor = usize::MAX;
            for &w in &edges[offsets[u]..offsets[u+1]] {
                let w = w as usize;
                if active[w] {
                    neighbor = w;
                    break;
                }
            }

            if neighbor != usize::MAX {
                let v = neighbor;
                vc.push(v);
                remove_from_bucket(v, degree[v] as usize, &mut head, &mut prev, &mut next);
                active[v] = false;

                for &w in &edges[offsets[v]..offsets[v+1]] {
                    let w = w as usize;
                    if active[w] {
                        let d = degree[w] as usize;
                        remove_from_bucket(w, d, &mut head, &mut prev, &mut next);
                        degree[w] -= 1;
                        add_to_bucket(w, d - 1, &mut head, &mut prev, &mut next);
                    }
                }
            }
            continue;
        }

        let mut curr_head = head[2];
        let mut found_triangle = false;
        while curr_head != -1 {
            let u = curr_head as usize;
            let p = next[curr_head as usize];

            let mut n1 = usize::MAX;
            let mut n2 = usize::MAX;
            for &w in &edges[offsets[u]..offsets[u+1]] {
                let w = w as usize;
                if active[w] {
                    if n1 == usize::MAX { n1 = w; }
                    else { n2 = w; break; }
                }
            }

            if n1 != usize::MAX && n2 != usize::MAX {
                let start = offsets[n1];
                let end = offsets[n1+1];
                if edges[start..end].binary_search(&(n2 as u32)).is_ok() {
                    vc.push(n1);
                    vc.push(n2);

                    for &node in &[u, n1, n2] {
                        if active[node] {
                            remove_from_bucket(node, degree[node] as usize, &mut head, &mut prev, &mut next);
                            active[node] = false;
                            for &w in &edges[offsets[node]..offsets[node+1]] {
                                let w = w as usize;
                                if active[w] {
                                    let d = degree[w] as usize;
                                    remove_from_bucket(w, d, &mut head, &mut prev, &mut next);
                                    degree[w] -= 1;
                                    add_to_bucket(w, d - 1, &mut head, &mut prev, &mut next);
                                }
                            }
                        }
                    }
                    found_triangle = true;
                    break;
                }
            }
            curr_head = p;
        }
        if found_triangle {
            continue;
        }

        while max_deg > 0 && head[max_deg] == -1 {
            max_deg -= 1;
        }

        if max_deg == 0 {
            break;
        }

        let v = head[max_deg] as usize;
        vc.push(v);
        remove_from_bucket(v, degree[v] as usize, &mut head, &mut prev, &mut next);
        active[v] = false;

        for &w in &edges[offsets[v]..offsets[v+1]] {
            let w = w as usize;
            if active[w] {
                let d = degree[w] as usize;
                remove_from_bucket(w, d, &mut head, &mut prev, &mut next);
                degree[w] -= 1;
                add_to_bucket(w, d - 1, &mut head, &mut prev, &mut next);
            }
        }
    }

    let mut in_vc = vec![false; n];
    for u in 0..n {
        if must_include[u] {
            in_vc[u] = true;
        }
    }

    let mut vc_candidates = Vec::new();
    for &v in &vc {
        in_vc[v] = true;
        vc_candidates.push(v);
    }

    vc_candidates.sort_unstable_by_key(|&v| offsets[v+1] - offsets[v]);

    for v in vc_candidates {
        let mut redundant = true;
        for &w in &edges[offsets[v]..offsets[v+1]] {
            let w = w as usize;
            if !in_vc[w] {
                redundant = false;
                break;
            }
        }
        if redundant {
            in_vc[v] = false;
        }
    }

    let mut final_vc = Vec::new();
    for u in 0..n {
        if in_vc[u] {
            final_vc.push(u);
        }
    }

    let stdout = std::io::stdout();
    let mut out = BufWriter::new(stdout.lock());
    writeln!(out, "{}", final_vc.len()).unwrap();
    for i in 0..final_vc.len() {
        if i > 0 {
            write!(out, " ").unwrap();
        }
        write!(out, "{}", final_vc[i]).unwrap();
    }
    writeln!(out).unwrap();
}
