const double PI = 3.14159265358979323846;

struct State {
    double x, y, z;
    double vx, vy, vz;
    double pitch, yaw, roll;
    double pitch_rate, yaw_rate, roll_rate;
    double fuel;
};

struct Vec3 {
    double x, y, z;
};

long long width, depth, height;
double gravity, max_thrust, fuel;
double start_x, start_y, start_z, target_x, target_y, target_z;
std::string terrain_file_path;
uint16_t* terrain;

std::vector<Vec3> waypoints;
size_t current_wp = 0;

double normalize_angle(double a) {
    while (a > PI) a -= 2 * PI;
    while (a < -PI) a += 2 * PI;
    return a;
}

bool is_vertical_clear(double x, double y, double z1, double z2) {
    long long ix = std::clamp((long long)std::round(x), 0LL, width - 1);
    long long iy = std::clamp((long long)std::round(y), 0LL, depth - 1);
    size_t idx = ((size_t)ix * depth + iy) * 3;
    uint16_t a = terrain[idx + 0];
    uint16_t b = terrain[idx + 1];
    uint16_t c = terrain[idx + 2];
    
    double min_z = std::min(z1, z2);
    double max_z = std::max(z1, z2);
    
    if (min_z < a) return false;
    if (b != c) {
        if (max_z > b && min_z < c) return false;
    }
    if (max_z >= height) return false;
    return true;
}

bool is_horizontal_clear(double x1, double y1, double x2, double y2, double z) {
    int steps = std::max(10, (int)std::hypot(x2 - x1, y2 - y1));
    for (int i = 0; i <= steps; ++i) {
        double t = (double)i / steps;
        double x = x1 + t * (x2 - x1);
        double y = y1 + t * (y2 - y1);
        if (!is_vertical_clear(x, y, z, z)) return false;
    }
    return true;
}

bool find_ascend_point(double sx, double sy, double sz, double cz, double &ax, double &ay) {
    if (is_vertical_clear(sx, sy, sz, cz)) {
        ax = sx; ay = sy; return true;
    }
    for (int r = 1; r < 200; ++r) {
        for (int i = -r; i <= r; ++i) {
            for (int j = -r; j <= r; ++j) {
                if (std::abs(i) != r && std::abs(j) != r) continue;
                double px = sx + i * 5.0;
                double py = sy + j * 5.0;
                if (px < 0 || px >= width || py < 0 || py >= depth) continue;
                
                if (is_vertical_clear(px, py, sz, cz)) {
                    if (is_horizontal_clear(sx, sy, px, py, sz)) {
                        ax = px; ay = py; return true;
                    }
                }
            }
        }
    }
    return false;
}

bool find_drop_point(double tx, double ty, double tz, double cz, double &dx, double &dy) {
    if (is_vertical_clear(tx, ty, cz, tz)) {
        dx = tx; dy = ty; return true;
    }
    for (int r = 1; r < 200; ++r) {
        for (int i = -r; i <= r; ++i) {
            for (int j = -r; j <= r; ++j) {
                if (std::abs(i) != r && std::abs(j) != r) continue;
                double px = tx + i * 5.0;
                double py = ty + j * 5.0;
                if (px < 0 || px >= width || py < 0 || py >= depth) continue;
                
                if (is_vertical_clear(px, py, cz, tz)) {
                    if (is_horizontal_clear(px, py, tx, ty, tz)) {
                        dx = px; dy = py; return true;
                    }
                }
            }
        }
    }
    return false;
}

struct Node {
    int x, y;
    double g, f;
    bool operator>(const Node& other) const { return f > other.f; }
};

std::vector<Vec3> astar_xy(double sx, double sy, double tx, double ty, double z) {
    int cell_size = 20;
    double pad = 10000.0;
    double min_x = std::max(0.0, std::min(sx, tx) - pad);
    double max_x = std::min((double)width, std::max(sx, tx) + pad);
    double min_y = std::max(0.0, std::min(sy, ty) - pad);
    double max_y = std::min((double)depth, std::max(sy, ty) + pad);
    
    int gw = (int)((max_x - min_x) / cell_size) + 1;
    int gh = (int)((max_y - min_y) / cell_size) + 1;
    
    int start_gx = std::clamp((int)((sx - min_x) / cell_size), 0, gw - 1);
    int start_gy = std::clamp((int)((sy - min_y) / cell_size), 0, gh - 1);
    int target_gx = std::clamp((int)((tx - min_x) / cell_size), 0, gw - 1);
    int target_gy = std::clamp((int)((ty - min_y) / cell_size), 0, gh - 1);
    
    std::vector<double> g_score(gw * gh, 1e9);
    std::vector<int> parent(gw * gh, -1);
    
    auto get_idx = [&](int x, int y) { return x * gh + y; };
    
    std::priority_queue<Node, std::vector<Node>, std::greater<Node>> pq;
    
    g_score[get_idx(start_gx, start_gy)] = 0;
    pq.push({start_gx, start_gy, 0, 0});
    
    auto is_cell_safe = [&](int gx, int gy) {
        double px = min_x + gx * cell_size + cell_size / 2.0;
        double py = min_y + gy * cell_size + cell_size / 2.0;
        return is_vertical_clear(px, py, z, z);
    };
    
    bool found = false;
    while (!pq.empty()) {
        Node curr = pq.top();
        pq.pop();
        
        if (curr.g > g_score[get_idx(curr.x, curr.y)]) continue;
        
        if (curr.x == target_gx && curr.y == target_gy) {
            found = true;
            break;
        }
        
        int dx_arr[] = {-1, 1, 0, 0, -1, -1, 1, 1};
        int dy_arr[] = {0, 0, -1, 1, -1, 1, -1, 1};
        
        for (int i = 0; i < 8; ++i) {
            int nx = curr.x + dx_arr[i];
            int ny = curr.y + dy_arr[i];
            
            if (nx < 0 || nx >= gw || ny < 0 || ny >= gh) continue;
            
            if (!is_cell_safe(nx, ny)) continue;
            
            double cost = (i < 4) ? cell_size : cell_size * 1.414;
            double tentative_g = curr.g + cost;
            
            int nidx = get_idx(nx, ny);
            if (tentative_g < g_score[nidx]) {
                g_score[nidx] = tentative_g;
                parent[nidx] = get_idx(curr.x, curr.y);
                double h = std::hypot((nx - target_gx) * cell_size, (ny - target_gy) * cell_size);
                pq.push({nx, ny, tentative_g, tentative_g + h});
            }
        }
    }
    
    std::vector<Vec3> path;
    if (found) {
        int curr_idx = get_idx(target_gx, target_gy);
        while (curr_idx != -1 && curr_idx != get_idx(start_gx, start_gy)) {
            int cx = curr_idx / gh;
            int cy = curr_idx % gh;
            path.push_back({min_x + cx * cell_size + cell_size / 2.0, min_y + cy * cell_size + cell_size / 2.0, z});
            curr_idx = parent[curr_idx];
        }
        std::reverse(path.begin(), path.end());
    } else {
        path.push_back({tx, ty, z});
    }
    
    std::vector<Vec3> simplified;
    if (!path.empty()) {
        simplified.push_back(path.front());
        size_t curr = 0;
        while (curr < path.size() - 1) {
            size_t next = curr + 1;
            size_t limit = std::min(path.size() - 1, curr + 200);
            for (size_t i = limit; i > curr; --i) {
                if (is_horizontal_clear(path[curr].x, path[curr].y, path[i].x, path[i].y, z)) {
                    next = i;
                    break;
                }
            }
            simplified.push_back(path[next]);
            curr = next;
        }
    }
    return simplified;
}

void plan_path() {
    waypoints.clear();
    
    bool direct_clear = true;
    double dist = std::hypot(target_x - start_x, target_y - start_y);
    int steps = std::max(10, (int)dist);
    for (int i = 0; i <= steps; ++i) {
        double t = (double)i / steps;
        double px = start_x + t * (target_x - start_x);
        double py = start_y + t * (target_y - start_y);
        double pz = start_z + t * (target_z - start_z);
        if (!is_vertical_clear(px, py, pz, pz)) {
            direct_clear = false;
            break;
        }
    }
    
    if (direct_clear) {
        waypoints.push_back({target_x, target_y, target_z});
        return;
    }
    
    double max_h = 0;
    for (int i = 0; i <= steps; ++i) {
        double t = (double)i / steps;
        double px = start_x + t * (target_x - start_x);
        double py = start_y + t * (target_y - start_y);
        
        long long ix = std::clamp((long long)std::round(px), 0LL, width - 1);
        long long iy = std::clamp((long long)std::round(py), 0LL, depth - 1);
        size_t idx = ((size_t)ix * depth + iy) * 3;
        uint16_t a = terrain[idx + 0];
        uint16_t b = terrain[idx + 1];
        uint16_t c = terrain[idx + 2];
        
        double h = (b == c) ? a : c;
        if (h > max_h) max_h = h;
    }
    
    double cruise_z = std::max({start_z + 10.0, target_z + 10.0, max_h + 50.0});
    cruise_z = std::min(cruise_z, (double)height - 20.0);
    
    double ax = start_x, ay = start_y;
    bool found_a = find_ascend_point(start_x, start_y, start_z, cruise_z, ax, ay);
    
    double dx = target_x, dy = target_y;
    bool found_d = find_drop_point(target_x, target_y, target_z, cruise_z, dx, dy);
    
    if (!found_a) { ax = start_x; ay = start_y; }
    if (!found_d) { dx = target_x; dy = target_y; }
    
    if (std::hypot(ax - start_x, ay - start_y) > 1.0) {
        waypoints.push_back({ax, ay, start_z});
    }
    waypoints.push_back({ax, ay, cruise_z});
    
    bool cruise_clear = is_horizontal_clear(ax, ay, dx, dy, cruise_z);
    if (cruise_clear) {
        waypoints.push_back({dx, dy, cruise_z});
    } else {
        auto astar_path = astar_xy(ax, ay, dx, dy, cruise_z);
        for (auto& p : astar_path) {
            waypoints.push_back(p);
        }
        if (!waypoints.empty()) {
            waypoints.back() = {dx, dy, cruise_z};
        }
    }
    
    if (std::hypot(dx - target_x, dy - target_y) > 1.0) {
        waypoints.push_back({dx, dy, target_z});
    }
    waypoints.push_back({target_x, target_y, target_z});
}

void compute_control(const State& s, double& thrust, double& pitch_cmd, double& yaw_cmd, double& roll_cmd) {
    if (current_wp < waypoints.size()) {
        Vec3 target = waypoints[current_wp];
        double dist = std::hypot(target.x - s.x, target.y - s.y, target.z - s.z);
        double threshold = (current_wp == waypoints.size() - 1) ? 2.0 : 15.0; 
        if (dist < threshold) {
            current_wp++;
        }
    }
    
    Vec3 target = (current_wp < waypoints.size()) ? waypoints[current_wp] : waypoints.back();
    
    double dist_xy = std::hypot(target.x - s.x, target.y - s.y);
    double dist_z = target.z - s.z;
    
    double a_max_xy = max_thrust * 0.4;
    double allowed_v_xy = std::sqrt(2.0 * a_max_xy * std::max(0.0, dist_xy));
    if (allowed_v_xy < 1.0) allowed_v_xy = 1.0; 
    if (current_wp < waypoints.size() - 1) allowed_v_xy = 40.0;
    
    double v_des_x = target.x - s.x;
    double v_des_y = target.y - s.y;
    double v_des_mag = std::hypot(v_des_x, v_des_y);
    if (v_des_mag > 0.001) {
        v_des_x /= v_des_mag;
        v_des_y /= v_des_mag;
    } else {
        v_des_x = 0; v_des_y = 0;
    }
    
    double Kp_pos_xy = 0.5;
    double v_xy_target = std::min(v_des_mag * Kp_pos_xy, allowed_v_xy);
    v_xy_target = std::min(v_xy_target, 50.0);
    
    v_des_x *= v_xy_target;
    v_des_y *= v_xy_target;
    
    double a_max_z_up = max_thrust - gravity;
    double a_max_z_down = gravity * 0.8;
    
    double allowed_v_z;
    if (dist_z > 0) {
        allowed_v_z = std::sqrt(2.0 * a_max_z_up * std::max(0.0, dist_z));
        if (allowed_v_z < 1.0) allowed_v_z = 1.0;
        if (current_wp < waypoints.size() - 1) allowed_v_z = 20.0;
    } else {
        allowed_v_z = -std::sqrt(2.0 * a_max_z_down * std::max(0.0, -dist_z));
        if (allowed_v_z > -1.0) allowed_v_z = -1.0;
        if (current_wp < waypoints.size() - 1) allowed_v_z = -20.0;
    }
    
    double Kp_pos_z = 0.5;
    double v_des_z = Kp_pos_z * dist_z;
    
    if (dist_z > 0) {
        v_des_z = std::min({v_des_z, allowed_v_z, 20.0});
    } else {
        v_des_z = std::max({v_des_z, allowed_v_z, -20.0});
    }
    
    double Kd_v_xy = 1.0;
    double Kd_v_z = 1.0;
    
    double A_x = Kd_v_xy * (v_des_x - s.vx);
    double A_y = Kd_v_xy * (v_des_y - s.vy);
    double A_z = Kd_v_z * (v_des_z - s.vz) + gravity;
    
    if (A_z < 0.0) A_z = 0.0;
    
    double A_mag = std::sqrt(A_x*A_x + A_y*A_y + A_z*A_z);
    if (A_mag > max_thrust) {
        if (std::abs(A_z) > max_thrust) {
            A_z = (A_z > 0) ? max_thrust : 0.0;
            A_x = 0; A_y = 0;
        } else {
            double rem = std::sqrt(max_thrust * max_thrust - A_z * A_z);
            double a_xy_mag = std::hypot(A_x, A_y);
            if (a_xy_mag > rem) {
                A_x = A_x / a_xy_mag * rem;
                A_y = A_y / a_xy_mag * rem;
            }
        }
    }
    
    A_mag = std::sqrt(A_x*A_x + A_y*A_y + A_z*A_z);
    thrust = A_mag / max_thrust;
    thrust = std::clamp(thrust, 0.0, 1.0);
    
    double p_des = 0, y_des = s.yaw;
    if (A_mag > 0.001) {
        double ux = A_x / A_mag;
        double uy = A_y / A_mag;
        double uz = A_z / A_mag;
        
        p_des = std::acos(std::clamp(uz, -1.0, 1.0));
        if (p_des > 0.001) {
            y_des = std::atan2(-uy, -ux);
        }
    }
    
    double d1 = std::abs(normalize_angle(y_des - s.yaw));
    double p2 = -p_des;
    double y2 = normalize_angle(y_des + PI);
    double d2 = std::abs(normalize_angle(y2 - s.yaw));
    
    double best_p, best_y;
    if (d1 <= d2) {
        best_p = p_des;
        best_y = y_des;
    } else {
        best_p = p2;
        best_y = y2;
    }
    
    best_p = std::clamp(best_p, -PI/2.0, PI/2.0);
    
    double Kp_p = 5.0;
    double Kd_p = 2.0;
    pitch_cmd = std::clamp(Kp_p * (best_p - s.pitch) - Kd_p * s.pitch_rate, -1.0, 1.0);
    yaw_cmd = std::clamp(Kp_p * normalize_angle(best_y - s.yaw) - Kd_p * s.yaw_rate, -1.0, 1.0);
    roll_cmd = std::clamp(-Kp_p * s.roll - Kd_p * s.roll_rate, -1.0, 1.0);
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    if (!(std::cin >> width >> depth >> height >> gravity >> max_thrust >> fuel)) return 0;
    std::cin >> start_x >> start_y >> start_z >> target_x >> target_y >> target_z;
    
    std::ws(std::cin);
    std::getline(std::cin, terrain_file_path);
    while (!terrain_file_path.empty() && std::isspace(terrain_file_path.back())) {
        terrain_file_path.pop_back();
    }
    
    HANDLE hFile = CreateFileA(terrain_file_path.c_str(), GENERIC_READ, FILE_SHARE_READ, NULL, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);
    if (hFile == INVALID_HANDLE_VALUE) {
        return 1;
    }
    HANDLE hMapping = CreateFileMappingA(hFile, NULL, PAGE_READONLY, 0, 0, NULL);
    void* mapped_data = MapViewOfFile(hMapping, FILE_MAP_READ, 0, 0, 0);
    terrain = (uint16_t*)mapped_data;
    
    std::string state_word;
    std::cin >> state_word;
    
    plan_path();
    
    while (true) {
        State s;
        if (!(std::cin >> s.x >> s.y >> s.z >> s.vx >> s.vy >> s.vz >> s.pitch >> s.yaw >> s.roll >> s.pitch_rate >> s.yaw_rate >> s.roll_rate >> s.fuel)) {
            break;
        }
        
        double thrust, pitch_cmd, yaw_cmd, roll_cmd;
        compute_control(s, thrust, pitch_cmd, yaw_cmd, roll_cmd);
        
        std::cout << thrust << " " << pitch_cmd << " " << yaw_cmd << " " << roll_cmd << "\n";
        std::cout.flush();
    }
    
    return 0;
}
