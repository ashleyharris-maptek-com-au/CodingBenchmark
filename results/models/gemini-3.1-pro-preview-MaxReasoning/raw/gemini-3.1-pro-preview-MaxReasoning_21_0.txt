#include <iostream>
#include <vector>
#include <queue>
#include <unordered_set>
#include <unordered_map>
#include <algorithm>
#include <cmath>
#include <tuple>

using namespace std;

const int MAX_D = 25;
int D;

struct Point {
    int coords[MAX_D] = {0};
    bool operator==(const Point& other) const {
        for (int i = 0; i < D; ++i) {
            if (coords[i] != other.coords[i]) return false;
        }
        return true;
    }
};

struct PointHash {
    size_t operator()(const Point& p) const {
        size_t h = 0;
        for (int i = 0; i < D; ++i) {
            h ^= hash<int>{}(p.coords[i]) + 0x9e3779b9 + (h << 6) + (h >> 2);
        }
        return h;
    }
};

struct Node {
    Point p;
    int g;
    int parent_idx;
    int move_axis;
    int move_dir;
};

int max_turns, L;
int bounds[MAX_D];
vector<Point> history;
vector<Point> foods;
unordered_set<Point, PointHash> obstacles;

int get_manhattan(const Point& a, const Point& b) {
    int dist = 0;
    for (int i = 0; i < D; ++i) {
        dist += abs(a.coords[i] - b.coords[i]);
    }
    return dist;
}

vector<Node> nodes;
unordered_map<Point, int, PointHash> visited_g;

bool is_tail_collision(const Point& next_p, int next_g, int node_idx) {
    int L_minus_1 = L - 1;
    if (L_minus_1 == 0) return false;
    
    int curr_idx = node_idx;
    int k = 0;
    while (curr_idx != -1 && k < L_minus_1) {
        if (nodes[curr_idx].p == next_p) return true;
        curr_idx = nodes[curr_idx].parent_idx;
        k++;
    }
    
    int remaining = L_minus_1 - k;
    if (remaining > 0) {
        for (int i = 0; i < remaining; ++i) {
            int hist_idx = (int)history.size() - 2 - i;
            if (hist_idx >= 0) {
                if (history[hist_idx] == next_p) return true;
            } else {
                if (history[0] == next_p) return true;
            }
        }
    }
    return false;
}

struct AStarResult {
    bool found;
    vector<pair<int, int>> moves;
};

AStarResult astar_to_food(const Point& start, const Point& goal) {
    nodes.clear();
    visited_g.clear();
    priority_queue<tuple<int, int, int>> pq;
    
    nodes.push_back({start, 0, -1, -1, -1});
    pq.push({-get_manhattan(start, goal), 0, 0});
    visited_g[start] = 0;
    
    int max_nodes = 100000;
    
    while (!pq.empty() && nodes.size() < max_nodes) {
        auto [minus_f, g, node_idx] = pq.top();
        pq.pop();
        
        Node& curr = nodes[node_idx];
        
        if (curr.p == goal) {
            vector<pair<int, int>> path_moves;
            int c_idx = node_idx;
            while(nodes[c_idx].parent_idx != -1) {
                path_moves.push_back({nodes[c_idx].move_axis, nodes[c_idx].move_dir});
                c_idx = nodes[c_idx].parent_idx;
            }
            reverse(path_moves.begin(), path_moves.end());
            return {true, path_moves};
        }
        
        if (visited_g[curr.p] < g) continue;
        
        for (int axis = 0; axis < D; ++axis) {
            for (int dir : {-1, 1}) {
                Point next_p = curr.p;
                next_p.coords[axis] += dir;
                
                if (next_p.coords[axis] < 0 || next_p.coords[axis] >= bounds[axis]) continue;
                if (obstacles.count(next_p)) continue;
                
                int next_g = g + 1;
                
                if (is_tail_collision(next_p, next_g, node_idx)) continue;
                
                auto it = visited_g.find(next_p);
                if (it != visited_g.end() && it->second <= next_g) continue;
                visited_g[next_p] = next_g;
                
                int next_h = get_manhattan(next_p, goal);
                int next_idx = nodes.size();
                nodes.push_back({next_p, next_g, node_idx, axis, dir});
                pq.push({-(next_g + next_h), next_g, next_idx});
            }
        }
    }
    return {false, {}};
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    if (!(cin >> D >> max_turns)) return 0;
    
    for (int i = 0; i < D; ++i) cin >> bounds[i];
    
    cin >> L;
    Point start_pos;
    for (int i = 0; i < D; ++i) cin >> start_pos.coords[i];
    history.push_back(start_pos);
    
    int F;
    cin >> F;
    foods.resize(F);
    for (int i = 0; i < F; ++i) {
        for (int j = 0; j < D; ++j) {
            cin >> foods[i].coords[j];
        }
    }
    
    int O;
    cin >> O;
    for (int i = 0; i < O; ++i) {
        Point obs;
        for (int j = 0; j < D; ++j) cin >> obs.coords[j];
        obstacles.insert(obs);
    }
    
    nodes.reserve(100000);
    
    int turns = 0;
    while (!foods.empty() && turns < max_turns) {
        sort(foods.begin(), foods.end(), [&](const Point& a, const Point& b) {
            return get_manhattan(history.back(), a) < get_manhattan(history.back(), b);
        });
        
        bool path_found = false;
        for (const auto& goal : foods) {
            AStarResult res = astar_to_food(history.back(), goal);
            if (res.found) {
                path_found = true;
                for (const auto& move : res.moves) {
                    if (turns >= max_turns) break;
                    cout << move.first << " " << move.second << endl;
                    turns++;
                    
                    Point new_head = history.back();
                    new_head.coords[move.first] += move.second;
                    history.push_back(new_head);
                    
                    auto it = find(foods.begin(), foods.end(), new_head);
                    if (it != foods.end()) {
                        foods.erase(it);
                        L++;
                        break; 
                    }
                }
                break;
            }
        }
        
        if (!path_found) {
            bool moved = false;
            for (int axis = 0; axis < D; ++axis) {
                for (int dir : {-1, 1}) {
                    Point next_p = history.back();
                    next_p.coords[axis] += dir;
                    if (next_p.coords[axis] < 0 || next_p.coords[axis] >= bounds[axis]) continue;
                    if (obstacles.count(next_p)) continue;
                    
                    bool tail_coll = false;
                    int L_minus_1 = L - 1;
                    if (L_minus_1 > 0) {
                        for (int i = 0; i < L_minus_1; ++i) {
                            int hist_idx = (int)history.size() - 1 - i;
                            if (hist_idx >= 0) {
                                if (history[hist_idx] == next_p) { tail_coll = true; break; }
                            } else {
                                if (history[0] == next_p) { tail_coll = true; break; }
                            }
                        }
                    }
                    if (tail_coll) continue;
                    
                    cout << axis << " " << dir << endl;
                    turns++;
                    history.push_back(next_p);
                    
                    auto it = find(foods.begin(), foods.end(), next_p);
                    if (it != foods.end()) {
                        foods.erase(it);
                        L++;
                    }
                    moved = true;
                    break;
                }
                if (moved) break;
            }
            if (!moved) {
                break;
            }
        }
    }
    
    return 0;
}
