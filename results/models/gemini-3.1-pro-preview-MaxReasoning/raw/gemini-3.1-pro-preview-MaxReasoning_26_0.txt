use std::io::{self, Read, Write, BufRead, BufReader, BufWriter};

struct SimpleRng { state: u32 }
impl SimpleRng {
    fn new(seed: u32) -> Self { Self { state: seed.max(1) } }
    fn next_u32(&mut self) -> u32 {
        self.state ^= self.state << 13;
        self.state ^= self.state >> 17;
        self.state ^= self.state << 5;
        self.state
    }
    fn next_f32(&mut self) -> f32 {
        (self.next_u32() as f32) / (u32::MAX as f32)
    }
}

#[inline(always)]
fn squared_distance(a: &[f32; 3], b: &[f32; 3]) -> f32 {
    let dx = a[0] - b[0];
    let dy = a[1] - b[1];
    let dz = a[2] - b[2];
    dx * dx + dy * dy + dz * dz
}

struct KdNode {
    point: [f32; 3],
    centroid_idx: u32,
    left: i32,
    right: i32,
    axis: u8,
}

fn build_kd_tree(centroids: &[[f32; 3]]) -> Vec<KdNode> {
    if centroids.is_empty() { return vec![]; }
    let mut items: Vec<([f32; 3], u32)> = centroids.iter().copied().enumerate().map(|(i, p)| (p, i as u32)).collect();
    let mut nodes = Vec::with_capacity(items.len());
    build_kd_tree_recursive(&mut items, 0, &mut nodes);
    nodes
}

fn build_kd_tree_recursive(items: &mut [([f32; 3], u32)], depth: usize, nodes: &mut Vec<KdNode>) -> i32 {
    if items.is_empty() {
        return -1;
    }
    let axis = (depth % 3) as u8;
    items.sort_unstable_by(|a, b| a.0[axis as usize].partial_cmp(&b.0[axis as usize]).unwrap_or(std::cmp::Ordering::Equal));
    
    let mid = items.len() / 2;
    let node_idx = nodes.len() as i32;
    nodes.push(KdNode {
        point: items[mid].0,
        centroid_idx: items[mid].1,
        left: -1,
        right: -1,
        axis,
    });
    
    let (left_part, rest) = items.split_at_mut(mid);
    let right_part = &mut rest[1..];
    
    let left_idx = build_kd_tree_recursive(left_part, depth + 1, nodes);
    let right_idx = build_kd_tree_recursive(right_part, depth + 1, nodes);
    
    nodes[node_idx as usize].left = left_idx;
    nodes[node_idx as usize].right = right_idx;
    
    node_idx
}

fn search_kd_tree(
    node_idx: i32, 
    target: &[f32; 3], 
    nodes: &[KdNode], 
    best_dist: &mut f32, 
    best_idx: &mut u32,
    visits: &mut i32
) {
    if node_idx == -1 || *visits <= 0 {
        return;
    }
    *visits -= 1;
    let node = &nodes[node_idx as usize];
    let dist = squared_distance(&node.point, target);
    if dist < *best_dist {
        *best_dist = dist;
        *best_idx = node.centroid_idx;
    }
    
    let axis = node.axis as usize;
    let diff = target[axis] - node.point[axis];
    
    let (first, second) = if diff < 0.0 {
        (node.left, node.right)
    } else {
        (node.right, node.left)
    };
    
    search_kd_tree(first, target, nodes, best_dist, best_idx, visits);
    
    if diff * diff < *best_dist {
        search_kd_tree(second, target, nodes, best_dist, best_idx, visits);
    }
}

fn initialize_centroids_kmeans_plus_plus(points: &[[f32; 3]], num_clusters: usize) -> Vec<[f32; 3]> {
    let mut centroids = Vec::with_capacity(num_clusters);
    if points.is_empty() || num_clusters == 0 { return centroids; }
    
    let mut rng = SimpleRng::new(42);
    let sample_size = points.len().min(100_000.max(num_clusters * 5));
    let mut sample = Vec::with_capacity(sample_size);
    for _ in 0..sample_size {
        let idx = rng.next_u32() as usize % points.len();
        sample.push(points[idx]);
    }
    
    centroids.push(sample[rng.next_u32() as usize % sample.len()]);
    let mut min_dists = vec![f32::MAX; sample.len()];
    
    for _ in 1..num_clusters {
        let last_centroid = centroids.last().unwrap();
        let mut total_dist = 0.0;
        
        for (i, p) in sample.iter().enumerate() {
            let dist = squared_distance(p, last_centroid);
            if dist < min_dists[i] {
                min_dists[i] = dist;
            }
            total_dist += min_dists[i];
        }
        
        if total_dist == 0.0 {
            centroids.push(sample[rng.next_u32() as usize % sample.len()]);
            continue;
        }
        
        let mut target = rng.next_f32() * total_dist;
        let mut selected = sample.len() - 1;
        for (i, &dist) in min_dists.iter().enumerate() {
            target -= dist;
            if target <= 0.0 {
                selected = i;
                break;
            }
        }
        centroids.push(sample[selected]);
    }
    centroids
}

fn ensure_all_clusters(assignments: &mut [u32], num_clusters: usize) {
    if assignments.is_empty() { return; }
    let mut used = vec![false; num_clusters];
    for &a in assignments.iter() {
        if (a as usize) < num_clusters {
            used[a as usize] = true;
        }
    }
    let mut unused = Vec::new();
    for c in 0..num_clusters {
        if !used[c] {
            unused.push(c as u32);
        }
    }
    
    let to_assign = std::cmp::min(unused.len(), assignments.len());
    let mut rng = SimpleRng::new(12345);
    for i in 0..to_assign {
        let idx = rng.next_u32() as usize % assignments.len();
        assignments[idx] = unused[i];
    }
}

fn main() {
    let stdin = io::stdin();
    let mut reader = BufReader::with_capacity(1024 * 1024, stdin.lock());
    
    let stdout = io::stdout();
    let mut writer = BufWriter::with_capacity(1024 * 1024, stdout.lock());
    
    let mut header = String::new();
    if reader.read_line(&mut header).unwrap() == 0 {
        return;
    }
    
    let parts: Vec<&str> = header.trim().split_whitespace().collect();
    if parts.len() != 2 {
        return;
    }
    let num_points: usize = parts[0].parse().unwrap();
    let num_clusters: usize = parts[1].parse().unwrap();
    
    if num_points == 0 || num_clusters == 0 {
        return;
    }

    let mut points_remaining = num_points;
    
    if num_points <= 20_000_000 {
        // IN-MEMORY BATCH K-MEANS
        let mut points = vec![[0.0; 3]; num_points];
        let mut byte_buffer = vec![0u8; 12 * 1_000_000];
        let mut idx = 0;
        
        while points_remaining > 0 {
            let chunk_points = points_remaining.min(1_000_000);
            reader.read_exact(&mut byte_buffer[..chunk_points * 12]).unwrap();
            for i in 0..chunk_points {
                points[idx][0] = f32::from_le_bytes(byte_buffer[i*12 .. i*12+4].try_into().unwrap());
                points[idx][1] = f32::from_le_bytes(byte_buffer[i*12+4 .. i*12+8].try_into().unwrap());
                points[idx][2] = f32::from_le_bytes(byte_buffer[i*12+8 .. i*12+12].try_into().unwrap());
                idx += 1;
            }
            points_remaining -= chunk_points;
        }
        
        let mut centroids = initialize_centroids_kmeans_plus_plus(&points, num_clusters);
        let mut assignments = vec![0u32; num_points];
        
        let iterations = 15;
        let mut rng = SimpleRng::new(42);
        
        for _ in 0..iterations {
            let mut new_centroids = vec![[0.0; 3]; num_clusters];
            let mut counts = vec![0; num_clusters];
            
            if num_clusters <= 50 {
                for (i, p) in points.iter().enumerate() {
                    let mut best_dist = f32::MAX;
                    let mut best_idx = 0;
                    for c in 0..num_clusters {
                        let d = squared_distance(p, &centroids[c]);
                        if d < best_dist {
                            best_dist = d;
                            best_idx = c as u32;
                        }
                    }
                    assignments[i] = best_idx;
                    new_centroids[best_idx as usize][0] += p[0];
                    new_centroids[best_idx as usize][1] += p[1];
                    new_centroids[best_idx as usize][2] += p[2];
                    counts[best_idx as usize] += 1;
                }
            } else {
                let kd_tree = build_kd_tree(&centroids);
                for (i, p) in points.iter().enumerate() {
                    let mut best_dist = f32::MAX;
                    let mut best_idx = 0;
                    let mut visits = 100;
                    search_kd_tree(0, p, &kd_tree, &mut best_dist, &mut best_idx, &mut visits);
                    assignments[i] = best_idx;
                    new_centroids[best_idx as usize][0] += p[0];
                    new_centroids[best_idx as usize][1] += p[1];
                    new_centroids[best_idx as usize][2] += p[2];
                    counts[best_idx as usize] += 1;
                }
            }
            
            for c in 0..num_clusters {
                if counts[c] > 0 {
                    centroids[c][0] = new_centroids[c][0] / counts[c] as f32;
                    centroids[c][1] = new_centroids[c][1] / counts[c] as f32;
                    centroids[c][2] = new_centroids[c][2] / counts[c] as f32;
                } else {
                    let idx = rng.next_u32() as usize % points.len();
                    centroids[c] = points[idx];
                }
            }
        }
        
        ensure_all_clusters(&mut assignments, num_clusters);
        
        let mut out_buf = vec![0u8; 1_000_000 * 4];
        let mut out_idx = 0;
        for &a in &assignments {
            out_buf[out_idx..out_idx+4].copy_from_slice(&a.to_le_bytes());
            out_idx += 4;
            if out_idx == out_buf.len() {
                writer.write_all(&out_buf).unwrap();
                out_idx = 0;
            }
        }
        if out_idx > 0 {
            writer.write_all(&out_buf[..out_idx]).unwrap();
        }
        
    } else {
        // FAST 1-PASS STREAMING K-MEANS
        let batch_size = 2_000_000;
        let mut buffer = vec![[0.0; 3]; batch_size];
        let mut byte_buffer = vec![0u8; batch_size * 12];
        
        let first_batch_points = points_remaining.min(batch_size);
        reader.read_exact(&mut byte_buffer[..first_batch_points * 12]).unwrap();
        for i in 0..first_batch_points {
            buffer[i][0] = f32::from_le_bytes(byte_buffer[i*12 .. i*12+4].try_into().unwrap());
            buffer[i][1] = f32::from_le_bytes(byte_buffer[i*12+4 .. i*12+8].try_into().unwrap());
            buffer[i][2] = f32::from_le_bytes(byte_buffer[i*12+8 .. i*12+12].try_into().unwrap());
        }
        points_remaining -= first_batch_points;
        
        let mut centroids = initialize_centroids_kmeans_plus_plus(&buffer[..first_batch_points], num_clusters);
        let mut assignments = vec![0u32; first_batch_points];
        
        let mut rng = SimpleRng::new(42);
        for _ in 0..10 {
            let mut new_centroids = vec![[0.0; 3]; num_clusters];
            let mut counts = vec![0; num_clusters];
            
            if num_clusters <= 50 {
                for (i, p) in buffer[..first_batch_points].iter().enumerate() {
                    let mut best_dist = f32::MAX;
                    let mut best_idx = 0;
                    for c in 0..num_clusters {
                        let d = squared_distance(p, &centroids[c]);
                        if d < best_dist {
                            best_dist = d;
                            best_idx = c as u32;
                        }
                    }
                    assignments[i] = best_idx;
                    new_centroids[best_idx as usize][0] += p[0];
                    new_centroids[best_idx as usize][1] += p[1];
                    new_centroids[best_idx as usize][2] += p[2];
                    counts[best_idx as usize] += 1;
                }
            } else {
                let kd_tree = build_kd_tree(&centroids);
                for (i, p) in buffer[..first_batch_points].iter().enumerate() {
                    let mut best_dist = f32::MAX;
                    let mut best_idx = 0;
                    let mut visits = 100;
                    search_kd_tree(0, p, &kd_tree, &mut best_dist, &mut best_idx, &mut visits);
                    assignments[i] = best_idx;
                    new_centroids[best_idx as usize][0] += p[0];
                    new_centroids[best_idx as usize][1] += p[1];
                    new_centroids[best_idx as usize][2] += p[2];
                    counts[best_idx as usize] += 1;
                }
            }
            for c in 0..num_clusters {
                if counts[c] > 0 {
                    centroids[c][0] = new_centroids[c][0] / counts[c] as f32;
                    centroids[c][1] = new_centroids[c][1] / counts[c] as f32;
                    centroids[c][2] = new_centroids[c][2] / counts[c] as f32;
                } else {
                    let idx = rng.next_u32() as usize % first_batch_points;
                    centroids[c] = buffer[idx];
                }
            }
        }
        
        ensure_all_clusters(&mut assignments, num_clusters);
        
        let mut cluster_weights = vec![0usize; num_clusters];
        for &a in &assignments {
            cluster_weights[a as usize] += 1;
        }
        
        let mut out_buf = vec![0u8; batch_size * 4];
        for (i, &a) in assignments.iter().enumerate() {
            out_buf[i*4..i*4+4].copy_from_slice(&a.to_le_bytes());
        }
        writer.write_all(&out_buf[..first_batch_points * 4]).unwrap();
        
        while points_remaining > 0 {
            let chunk_points = points_remaining.min(batch_size);
            reader.read_exact(&mut byte_buffer[..chunk_points * 12]).unwrap();
            for i in 0..chunk_points {
                buffer[i][0] = f32::from_le_bytes(byte_buffer[i*12 .. i*12+4].try_into().unwrap());
                buffer[i][1] = f32::from_le_bytes(byte_buffer[i*12+4 .. i*12+8].try_into().unwrap());
                buffer[i][2] = f32::from_le_bytes(byte_buffer[i*12+8 .. i*12+12].try_into().unwrap());
            }
            points_remaining -= chunk_points;
            
            let mut new_centroids = vec![[0.0; 3]; num_clusters];
            let mut counts = vec![0; num_clusters];
            
            let kd_tree = if num_clusters > 50 { build_kd_tree(&centroids) } else { vec![] };
            
            for i in 0..chunk_points {
                let p = &buffer[i];
                let mut best_dist = f32::MAX;
                let mut best_idx = 0;
                
                if num_clusters <= 50 {
                    for c in 0..num_clusters {
                        let d = squared_distance(p, &centroids[c]);
                        if d < best_dist {
                            best_dist = d;
                            best_idx = c as u32;
                        }
                    }
                } else {
                    let mut visits = 100;
                    search_kd_tree(0, p, &kd_tree, &mut best_dist, &mut best_idx, &mut visits);
                }
                
                out_buf[i*4..i*4+4].copy_from_slice(&best_idx.to_le_bytes());
                
                new_centroids[best_idx as usize][0] += p[0];
                new_centroids[best_idx as usize][1] += p[1];
                new_centroids[best_idx as usize][2] += p[2];
                counts[best_idx as usize] += 1;
            }
            
            writer.write_all(&out_buf[..chunk_points * 4]).unwrap();
            
            for c in 0..num_clusters {
                if counts[c] > 0 {
                    let old_w = cluster_weights[c] as f32;
                    let new_w = counts[c] as f32;
                    let tot_w = old_w + new_w;
                    centroids[c][0] = (centroids[c][0] * old_w + new_centroids[c][0]) / tot_w;
                    centroids[c][1] = (centroids[c][1] * old_w + new_centroids[c][1]) / tot_w;
                    centroids[c][2] = (centroids[c][2] * old_w + new_centroids[c][2]) / tot_w;
                    
                    cluster_weights[c] += counts[c];
                    if cluster_weights[c] > 100_000 {
                        cluster_weights[c] = 100_000;
                    }
                }
            }
        }
    }
    
    writer.flush().unwrap();
}
