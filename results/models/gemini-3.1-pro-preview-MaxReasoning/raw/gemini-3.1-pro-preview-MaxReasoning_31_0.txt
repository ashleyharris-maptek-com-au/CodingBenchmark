use std::io::{BufRead, Read, StdinLock};
use std::sync::atomic::{AtomicBool, Ordering};
use std::sync::{mpsc, Arc};
use std::thread;
use std::time::{Duration, Instant};

struct FastScanner<'a> {
    lock: StdinLock<'a>,
}

impl<'a> FastScanner<'a> {
    fn new(lock: StdinLock<'a>) -> Self {
        FastScanner { lock }
    }

    fn next_i64(&mut self) -> Option<i64> {
        let mut res = 0i64;
        let mut sign = 1i64;
        let mut valid = false;
        let mut in_number = false;

        loop {
            let buf = self.lock.fill_buf().unwrap();
            if buf.is_empty() {
                break;
            }
            let mut consume = 0;
            for &b in buf {
                consume += 1;
                if b <= b' ' {
                    if in_number {
                        self.lock.consume(consume);
                        return Some(res.wrapping_mul(sign));
                    }
                } else if b == b'-' {
                    sign = -1;
                    in_number = true;
                } else if b == b'+' {
                    sign = 1;
                    in_number = true;
                } else if b >= b'0' && b <= b'9' {
                    res = res.wrapping_mul(10).wrapping_add((b - b'0') as i64);
                    in_number = true;
                    valid = true;
                }
            }
            self.lock.consume(consume);
        }
        if valid {
            Some(res.wrapping_mul(sign))
        } else {
            None
        }
    }
}

struct XorShift {
    state: u64,
}

impl XorShift {
    fn new(seed: u64) -> Self {
        XorShift { state: if seed == 0 { 1 } else { seed } }
    }
    fn next(&mut self) -> u64 {
        let mut x = self.state;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.state = x;
        x
    }
    fn next_f64(&mut self) -> f64 {
        (self.next() as f64) / (u64::MAX as f64)
    }
}

fn shuffle<T>(slice: &mut [T], rng: &mut XorShift) {
    let len = slice.len();
    if len <= 1 {
        return;
    }
    for i in (1..len).rev() {
        let j = (rng.next() as usize) % (i + 1);
        slice.swap(i, j);
    }
}

fn print_solution(
    subset: &[usize],
    pos_items: &[(i64, usize)],
    neg_items: &[(i64, usize)],
    zero_indices: &[usize],
) {
    use std::collections::HashSet;
    let mut chosen = HashSet::new();
    for &idx in subset {
        chosen.insert(idx);
    }

    let mut final_ans = Vec::new();
    for &(_, idx) in pos_items {
        if chosen.contains(&idx) {
            final_ans.push(idx);
        }
    }
    for &(_, idx) in neg_items {
        if !chosen.contains(&idx) {
            final_ans.push(idx);
        }
    }
    for &idx in zero_indices {
        final_ans.push(idx);
    }

    final_ans.sort_unstable();

    println!("YES");
    print!("[");
    for i in 0..final_ans.len() {
        if i > 0 {
            print!(" ");
        }
        print!("{}", final_ans[i]);
    }
    println!("]");
}

fn dfs_iterative(
    target: u128,
    items: &[(u64, usize)],
    suffix_sum: &[u128],
    found: Arc<AtomicBool>,
) -> Option<Vec<usize>> {
    let n = items.len();
    let mut stack = Vec::with_capacity(n + 1);
    stack.push((0, 0u128, 0u8));
    let mut subset = Vec::with_capacity(n);
    let mut iters = 0;

    while let Some((idx, current_sum, state)) = stack.pop() {
        iters += 1;
        if iters & 0xFFFFF == 0 {
            if found.load(Ordering::Relaxed) {
                return None;
            }
        }

        if current_sum == target {
            found.store(true, Ordering::Relaxed);
            return Some(subset.clone());
        }

        if idx == n {
            continue;
        }

        if current_sum + suffix_sum[idx] < target {
            continue;
        }

        if current_sum > target {
            continue;
        }

        if state == 0 {
            stack.push((idx, current_sum, 1));
            subset.push(items[idx].1);
            stack.push((idx + 1, current_sum + items[idx].0 as u128, 0));
        } else {
            subset.pop();
            stack.push((idx + 1, current_sum, 0));
        }
    }
    None
}

fn main() {
    let stdin = std::io::stdin();
    let lock = stdin.lock();
    let mut scanner = FastScanner::new(lock);

    let num_numbers = match scanner.next_i64() {
        Some(n) => n,
        None => return,
    };
    let original_target = match scanner.next_i64() {
        Some(t) => t,
        None => return,
    };

    let mut pos_items = Vec::new();
    let mut neg_items = Vec::new();
    let mut zero_indices = Vec::new();

    for i in 0..num_numbers as usize {
        let val = scanner.next_i64().unwrap();
        if val > 0 {
            pos_items.push((val, i));
        } else if val < 0 {
            neg_items.push((val, i));
        } else {
            zero_indices.push(i);
        }
    }

    let mut s_neg: i128 = 0;
    for &(val, _) in &neg_items {
        s_neg += val as i128;
    }

    let target_128 = original_target as i128 - s_neg;
    if target_128 < 0 {
        println!("NO");
        return;
    }
    let target = target_128 as u128;

    let mut positive_array: Vec<(u64, usize)> = Vec::new();
    for &(val, idx) in &pos_items {
        positive_array.push((val as u64, idx));
    }
    for &(val, idx) in &neg_items {
        let abs_val = if val == i64::MIN {
            9223372036854775808u64
        } else {
            (-val) as u64
        };
        positive_array.push((abs_val, idx));
    }

    let mut total_sum: u128 = 0;
    for &(val, _) in &positive_array {
        total_sum += val as u128;
    }

    if target > total_sum {
        println!("NO");
        return;
    }

    if target == 0 {
        print_solution(&[], &pos_items, &neg_items, &zero_indices);
        return;
    }
    if target == total_sum {
        let mut all = Vec::new();
        for i in 0..positive_array.len() {
            all.push(positive_array[i].1);
        }
        print_solution(&all, &pos_items, &neg_items, &zero_indices);
        return;
    }

    let n = positive_array.len();
    if n <= 66 {
        let left_size = n.min(26);
        let right_size = n - left_size;

        let left_items = Arc::new(positive_array[0..left_size].to_vec());
        let right_items = Arc::new(positive_array[left_size..n].to_vec());

        let mut left_sums: Vec<(u128, u32)> = Vec::with_capacity(1 << left_size);
        for mask in 0..(1u32 << left_size) {
            let mut sum = 0u128;
            for i in 0..left_size {
                if (mask & (1 << i)) != 0 {
                    sum += left_items[i].0 as u128;
                }
            }
            left_sums.push((sum, mask));
        }
        left_sums.sort_unstable_by_key(|k| k.0);

        let left_sums = Arc::new(left_sums);
        let (tx, rx) = mpsc::channel();
        let found = Arc::new(AtomicBool::new(false));

        if right_size == 0 {
            if let Ok(idx) = left_sums.binary_search_by_key(&target, |k| k.0) {
                let left_mask = left_sums[idx].1;
                let mut ans = Vec::new();
                for i in 0..left_size {
                    if (left_mask & (1 << i)) != 0 {
                        ans.push(left_items[i].1);
                    }
                }
                print_solution(&ans, &pos_items, &neg_items, &zero_indices);
            } else {
                println!("NO");
            }
            return;
        }

        let k = right_size.min(6);
        let remaining_right_size = right_size - k;
        let num_threads = 1 << k;

        for thread_mask in 0..num_threads {
            let left_sums_clone = Arc::clone(&left_sums);
            let left_items_clone = Arc::clone(&left_items);
            let right_items_clone = Arc::clone(&right_items);
            let tx_clone = tx.clone();
            let found_clone = Arc::clone(&found);

            thread::spawn(move || {
                let mut base_sum = 0u128;
                for i in 0..k {
                    if (thread_mask & (1 << i)) != 0 {
                        base_sum += right_items_clone[remaining_right_size + i].0 as u128;
                    }
                }

                if base_sum > target {
                    return;
                }

                let limit = 1u64 << remaining_right_size;
                let mut rem_mask = 0u64;
                let mut current_sum = base_sum;

                if current_sum <= target {
                    let search_target = target - current_sum;
                    if let Ok(idx) = left_sums_clone.binary_search_by_key(&search_target, |m| m.0) {
                        found_clone.store(true, Ordering::Relaxed);
                        let left_mask = left_sums_clone[idx].1;
                        let mut ans = Vec::new();
                        for j in 0..left_size {
                            if (left_mask & (1 << j)) != 0 {
                                ans.push(left_items_clone[j].1);
                            }
                        }
                        for j in 0..remaining_right_size {
                            if (rem_mask & (1 << j)) != 0 {
                                ans.push(right_items_clone[j].1);
                            }
                        }
                        for j in 0..k {
                            if (thread_mask & (1 << j)) != 0 {
                                ans.push(right_items_clone[remaining_right_size + j].1);
                            }
                        }
                        let _ = tx_clone.send(ans);
                        return;
                    }
                }

                for i in 1..limit {
                    if i & 0xFFFF == 0 {
                        if found_clone.load(Ordering::Relaxed) {
                            return;
                        }
                    }
                    let bit = i.trailing_zeros() as usize;
                    rem_mask ^= 1 << bit;
                    if (rem_mask & (1 << bit)) != 0 {
                        current_sum += right_items_clone[bit].0 as u128;
                    } else {
                        current_sum -= right_items_clone[bit].0 as u128;
                    }

                    if current_sum <= target {
                        let search_target = target - current_sum;
                        if let Ok(idx) = left_sums_clone.binary_search_by_key(&search_target, |m| m.0) {
                            found_clone.store(true, Ordering::Relaxed);
                            let left_mask = left_sums_clone[idx].1;
                            let mut ans = Vec::new();
                            for j in 0..left_size {
                                if (left_mask & (1 << j)) != 0 {
                                    ans.push(left_items_clone[j].1);
                                }
                            }
                            for j in 0..remaining_right_size {
                                if (rem_mask & (1 << j)) != 0 {
                                    ans.push(right_items_clone[j].1);
                                }
                            }
                            for j in 0..k {
                                if (thread_mask & (1 << j)) != 0 {
                                    ans.push(right_items_clone[remaining_right_size + j].1);
                                }
                            }
                            let _ = tx_clone.send(ans);
                            return;
                        }
                    }
                }
            });
        }

        drop(tx);
        match rx.recv_timeout(Duration::from_secs(295)) {
            Ok(subset) => {
                print_solution(&subset, &pos_items, &neg_items, &zero_indices);
            }
            Err(_) => {
                println!("NO");
            }
        }
    } else {
        let num_threads = 60;
        let (tx, rx) = mpsc::channel();
        let found = Arc::new(AtomicBool::new(false));

        for i in 0..num_threads {
            let mut thread_items = positive_array.clone();
            let mut rng = XorShift::new(1337 + i as u64 * 10007);

            if i == 0 {
                thread_items.sort_unstable_by(|a, b| b.0.cmp(&a.0));
            } else if i == 1 {
                thread_items.sort_unstable_by(|a, b| a.0.cmp(&b.0));
            } else if i < 15 {
                thread_items.sort_unstable_by(|a, b| {
                    let noisy_a = a.0 as f64 * (0.8 + 0.4 * rng.next_f64());
                    let noisy_b = b.0 as f64 * (0.8 + 0.4 * rng.next_f64());
                    noisy_b
                        .partial_cmp(&noisy_a)
                        .unwrap_or(std::cmp::Ordering::Equal)
                });
            } else if i < 30 {
                thread_items.sort_unstable_by(|a, b| {
                    let noisy_a = a.0 as f64 * (0.8 + 0.4 * rng.next_f64());
                    let noisy_b = b.0 as f64 * (0.8 + 0.4 * rng.next_f64());
                    noisy_a
                        .partial_cmp(&noisy_b)
                        .unwrap_or(std::cmp::Ordering::Equal)
                });
            } else {
                shuffle(&mut thread_items, &mut rng);
            }

            let mut suffix_sum = vec![0u128; thread_items.len()];
            let mut cur = 0u128;
            for j in (0..thread_items.len()).rev() {
                cur += thread_items[j].0 as u128;
                suffix_sum[j] = cur;
            }

            let found_clone = Arc::clone(&found);
            let tx_clone = tx.clone();
            let target_clone = target;

            thread::spawn(move || {
                if let Some(subset) =
                    dfs_iterative(target_clone, &thread_items, &suffix_sum, found_clone)
                {
                    let _ = tx_clone.send(subset);
                }
            });
        }

        drop(tx);
        match rx.recv_timeout(Duration::from_secs(295)) {
            Ok(subset) => {
                print_solution(&subset, &pos_items, &neg_items, &zero_indices);
            }
            Err(_) => {
                println!("NO");
            }
        }
    }
}
