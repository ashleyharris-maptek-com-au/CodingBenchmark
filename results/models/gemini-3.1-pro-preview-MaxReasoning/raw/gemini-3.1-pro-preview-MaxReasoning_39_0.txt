use std::time::Instant;

#[derive(Clone, Default)]
struct BitSet {
    mask: Vec<u64>,
}
impl BitSet {
    fn new(n: usize) -> Self {
        let len = (n + 63) / 64;
        Self { mask: vec![0; len] }
    }
    fn set(&mut self, i: usize) { self.mask[i / 64] |= 1 << (i % 64); }
    fn clear(&mut self, i: usize) { self.mask[i / 64] &= !(1 << (i % 64)); }
    fn get(&self, i: usize) -> bool { (self.mask[i / 64] & (1 << (i % 64))) != 0 }
    
    fn count_ones(&self) -> u32 { self.mask.iter().map(|x| x.count_ones()).sum() }
    fn and_count_ones(&self, other: &Self) -> u32 {
        self.mask.iter().zip(&other.mask).map(|(a, b)| (a & b).count_ones()).sum()
    }
    
    fn iter_ones(&self) -> Ones {
        Ones { set: self, word_idx: 0, current_word: self.mask.first().copied().unwrap_or(0) }
    }
}

struct Ones<'a> {
    set: &'a BitSet,
    word_idx: usize,
    current_word: u64,
}
impl<'a> Iterator for Ones<'a> {
    type Item = usize;
    fn next(&mut self) -> Option<Self::Item> {
        loop {
            if self.current_word != 0 {
                let bit = self.current_word.trailing_zeros();
                self.current_word &= !(1 << bit);
                return Some(self.word_idx * 64 + bit as usize);
            }
            self.word_idx += 1;
            if self.word_idx >= self.set.mask.len() {
                return None;
            }
            self.current_word = self.set.mask[self.word_idx];
        }
    }
}

struct Solver {
    n: usize,
    adj: Vec<BitSet>,
    best_cut: u32,
    best_A: BitSet,
    nodes_visited: usize,
}

impl Solver {
    fn solve(
        &mut self,
        assigned_A: &mut BitSet,
        assigned_B: &mut BitSet,
        unassigned: &mut BitSet,
        current_cut: u32,
        a_count: usize,
        b_count: usize,
    ) {
        self.nodes_visited += 1;
        if a_count == self.n / 2 && b_count == self.n / 2 {
            if current_cut < self.best_cut {
                self.best_cut = current_cut;
                self.best_A = assigned_A.clone();
            }
            return;
        }

        let k_a = self.n / 2 - a_count;
        let k_b = self.n / 2 - b_count;

        let mut diff = Vec::with_capacity(unassigned.count_ones() as usize);
        let mut sum_A = 0;
        let mut sum_deg_lb = 0;
        let mut E_un = 0;

        for i in unassigned.iter_ones() {
            let c_A = self.adj[i].and_count_ones(assigned_B);
            let c_B = self.adj[i].and_count_ones(assigned_A);
            sum_A += c_A;
            diff.push(c_B as i32 - c_A as i32);

            let d_U = self.adj[i].and_count_ones(unassigned);
            E_un += d_U;
            let lb_A = d_U.saturating_sub(k_a.saturating_sub(1) as u32);
            let lb_B = d_U.saturating_sub(k_b.saturating_sub(1) as u32);
            sum_deg_lb += lb_A.min(lb_B);
        }

        if k_b > 0 {
            if k_b == diff.len() {
                sum_A = (sum_A as i32 + diff.iter().sum::<i32>()) as u32;
            } else {
                let (_, pivot, smaller) = diff.select_nth_unstable(k_b - 1);
                sum_A = (sum_A as i32 + smaller.iter().sum::<i32>() + *pivot) as u32;
            }
        }
        let cost_assigned_unassigned = sum_A;

        E_un /= 2;
        let lb_un_1 = E_un.saturating_sub(
            (k_a as u32 * (k_a as u32).saturating_sub(1)) / 2 + 
            (k_b as u32 * (k_b as u32).saturating_sub(1)) / 2
        );
        let lb_un_2 = (sum_deg_lb + 1) / 2;
        let cost_unassigned_unassigned = lb_un_1.max(lb_un_2);

        let total_lb = current_cut + cost_assigned_unassigned + cost_unassigned_unassigned;
        if total_lb >= self.best_cut {
            return;
        }

        let mut best_v = !0;
        let mut max_conn = -1;
        let mut max_deg = -1;

        for i in unassigned.iter_ones() {
            let conn = self.adj[i].and_count_ones(assigned_A) + self.adj[i].and_count_ones(assigned_B);
            let deg = self.adj[i].count_ones();
            
            if conn as i32 > max_conn || (conn as i32 == max_conn && deg as i32 > max_deg) {
                max_conn = conn as i32;
                max_deg = deg as i32;
                best_v = i;
            }
        }

        if best_v == !0 { return; }

        let c_A = self.adj[best_v].and_count_ones(assigned_B);
        let c_B = self.adj[best_v].and_count_ones(assigned_A);

        let put_in_A_first = c_A < c_B;

        let mut branches = Vec::new();
        if k_a > 0 && k_b > 0 {
            branches.push(put_in_A_first);
            branches.push(!put_in_A_first);
        } else if k_a > 0 {
            branches.push(true);
        } else if k_b > 0 {
            branches.push(false);
        }

        unassigned.clear(best_v);

        for put_in_A in branches {
            if put_in_A {
                assigned_A.set(best_v);
                self.solve(assigned_A, assigned_B, unassigned, current_cut + c_A, a_count + 1, b_count);
                assigned_A.clear(best_v);
            } else {
                assigned_B.set(best_v);
                self.solve(assigned_A, assigned_B, unassigned, current_cut + c_B, a_count, b_count + 1);
                assigned_B.clear(best_v);
            }
        }

        unassigned.set(best_v);
    }
}

fn initial_heuristic(n: usize, adj: &[BitSet]) -> (u32, BitSet) {
    let mut best_cut = u32::MAX;
    let mut best_A = BitSet::new(n);

    for seed in 0..100 {
        let mut A = BitSet::new(n);
        let mut B = BitSet::new(n);
        
        let mut vertices: Vec<usize> = (0..n).collect();
        let mut rng = seed as u32 + 12345;
        for i in 0..n {
            rng ^= rng << 13;
            rng ^= rng >> 17;
            rng ^= rng << 5;
            let j = (rng as usize) % (i + 1);
            vertices.swap(i, j);
        }
        
        for i in 0..n {
            if i < n / 2 {
                A.set(vertices[i]);
            } else {
                B.set(vertices[i]);
            }
        }
        
        let mut cut = 0;
        for i in 0..n {
            if A.get(i) {
                cut += adj[i].and_count_ones(&B);
            }
        }
        
        let mut improved = true;
        while improved {
            improved = false;
            let mut best_swap = None;
            let mut best_delta = 0i32;
            
            for i in 0..n {
                if !A.get(i) { continue; }
                let i_to_B = adj[i].and_count_ones(&B) as i32;
                let i_to_A = adj[i].and_count_ones(&A) as i32;
                
                for j in 0..n {
                    if !B.get(j) { continue; }
                    let j_to_A = adj[j].and_count_ones(&A) as i32;
                    let j_to_B = adj[j].and_count_ones(&B) as i32;
                    
                    let connected = if adj[i].get(j) { 1 } else { 0 };
                    let delta = (i_to_A + j_to_B) - (i_to_B + j_to_A) + 2 * connected;
                    
                    if delta < best_delta {
                        best_delta = delta;
                        best_swap = Some((i, j));
                    }
                }
            }
            if let Some((i, j)) = best_swap {
                A.clear(i); B.set(i);
                B.clear(j); A.set(j);
                cut = (cut as i32 + best_delta) as u32;
                improved = true;
            }
        }
        if cut < best_cut {
            best_cut = cut;
            best_A = A.clone();
        }
    }
    (best_cut, best_A)
}

fn main() {
    let n = 30;
    let p = 0.5; // dense graph
    let mut adj = vec![BitSet::new(n); n];
    let mut rng = 12345u32;
    for i in 0..n {
        for j in i+1..n {
            rng ^= rng << 13;
            rng ^= rng >> 17;
            rng ^= rng << 5;
            if (rng % 100) < (p * 100.0) as u32 {
                adj[i].set(j);
                adj[j].set(i);
            }
        }
    }

    let (heur_cut, heur_A) = initial_heuristic(n, &adj);
    println!("Heuristic cut: {}", heur_cut);

    let mut unassigned = BitSet::new(n);
    for i in 0..n { unassigned.set(i); }

    let mut solver = Solver {
        n,
        adj,
        best_cut: heur_cut,
        best_A: heur_A,
        nodes_visited: 0,
    };

    let start = Instant::now();
    solver.solve(&mut BitSet::new(n), &mut BitSet::new(n), &mut unassigned, 0, 0, 0);
    println!("Optimal cut: {}, Time: {:?}, Nodes: {}", solver.best_cut, start.elapsed(), solver.nodes_visited);
}
