using System;
using System.IO;
using System.Collections.Generic;
using System.Text;
using System.Diagnostics;

class Program {
    static int[] dx = {0, 1, 0, -1};
    static int[] dy = {1, 0, -1, 0};
    
    class Candidate {
        public int X, Y;
        public long Score;
    }

    class FastReader {
        Stream s;
        byte[] buf = new byte[1024 * 1024];
        int head = 0;
        int tail = 0;
        
        public FastReader(Stream s) { this.s = s; }
        
        public void Skip(int count) {
            int remaining = count;
            while (remaining > 0) {
                int avail = tail - head;
                if (avail == 0) {
                    tail = s.Read(buf, 0, buf.Length);
                    head = 0;
                    if (tail == 0) break;
                    avail = tail;
                }
                int toSkip = Math.Min(remaining, avail);
                head += toSkip;
                remaining -= toSkip;
            }
        }
        
        public void Read(byte[] dest, int offset, int count) {
            int remaining = count;
            while (remaining > 0) {
                int avail = tail - head;
                if (avail == 0) {
                    tail = s.Read(buf, 0, buf.Length);
                    head = 0;
                    if (tail == 0) break;
                    avail = tail;
                }
                int toCopy = Math.Min(remaining, avail);
                Array.Copy(buf, head, dest, offset, toCopy);
                head += toCopy;
                offset += toCopy;
                remaining -= toCopy;
            }
        }
    }
    
    static void Main(string[] args) {
        var sw = Stopwatch.StartNew();
        
        var stdin = Console.OpenStandardInput();
        string header = "";
        int b;
        while ((b = stdin.ReadByte()) != -1) {
            char c = (char)b;
            if (c == '\n') break;
            header += c;
        }
        string[] parts = header.Trim().Split(' ', StringSplitOptions.RemoveEmptyEntries);
        int XVoxelCount = int.Parse(parts[0]);
        int YVoxelCount = int.Parse(parts[1]);
        int ZVoxelCount = int.Parse(parts[2]);
        
        int maxX = Math.Min(XVoxelCount, 400);
        int maxY = Math.Min(YVoxelCount, 400);
        int maxZ = Math.Min(ZVoxelCount, 60);

        long totalVolume = (long)maxX * maxY * maxZ;
        if (totalVolume > 15000000) {
            maxX = Math.Min(maxX, 300);
            maxY = Math.Min(maxY, 300);
        }

        byte[][] Map = new byte[maxX][];
        for(int i=0; i<maxX; i++) Map[i] = new byte[maxY * maxZ];

        var reader = new FastReader(stdin);
        for(int x=0; x<XVoxelCount; x++) {
            for(int y=0; y<YVoxelCount; y++) {
                if (x < maxX && y < maxY) {
                    int skip = ZVoxelCount - maxZ;
                    if (skip > 0) reader.Skip(skip);
                    reader.Read(Map[x], y * maxZ, maxZ);
                } else {
                    reader.Skip(ZVoxelCount);
                }
            }
        }
        
        int[,] Z_surface = new int[maxX, maxY];
        int[,] DumpCount = new int[maxX, maxY];
        
        long[,] Dist = new long[maxX, maxY];
        int[,] NextHopX = new int[maxX, maxY];
        int[,] NextHopY = new int[maxX, maxY];
        int[,] TargetZ = new int[maxX, maxY];
        
        int operations = 0;
        var outBuf = new List<int>(20000000);
        
        while(operations < 5000000 && sw.Elapsed.TotalMinutes < 4.5) {
            for(int i=0; i<maxX; i++) {
                for(int j=0; j<maxY; j++) {
                    Dist[i,j] = long.MaxValue;
                    NextHopX[i,j] = -1;
                    NextHopY[i,j] = -1;
                }
            }
            
            Dist[0,0] = 0;
            var pq = new PriorityQueue<(int,int), long>();
            pq.Enqueue((0,0), 0);
            
            while(pq.Count > 0) {
                pq.TryDequeue(out var curr, out long d);
                int x = curr.Item1, y = curr.Item2;
                if (d > Dist[x, y]) continue;
                
                for(int i=0; i<4; i++) {
                    int nx = x + dx[i];
                    int ny = y + dy[i];
                    if (nx>=0 && nx<maxX && ny>=0 && ny<maxY) {
                        int diff = Z_surface[nx, ny] - Z_surface[x, y];
                        if (diff <= 1) {
                            long cost = (diff == 1) ? 5 : ((diff == 0) ? 1 : 0);
                            if (Dist[x,y] + cost < Dist[nx, ny]) {
                                Dist[nx, ny] = Dist[x,y] + cost;
                                NextHopX[nx, ny] = x;
                                NextHopY[nx, ny] = y;
                                pq.Enqueue((nx, ny), Dist[nx, ny]);
                            }
                        }
                    }
                }
            }
            
            for(int x=0; x<maxX; x++) {
                for(int y=0; y<maxY; y++) {
                    TargetZ[x,y] = Z_surface[x,y];
                    long maxProf = 0;
                    int best_d = Z_surface[x,y];
                    long curProf = 0;
                    int curZ = Z_surface[x,y];
                    int dumps = DumpCount[x,y];
                    
                    while(curZ < maxZ) {
                        int val, m;
                        if (dumps > 0) { val=0; m=10; dumps--; }
                        else {
                            byte v_d = Map[x][y*maxZ + curZ];
                            val = v_d/16; m = v_d%16;
                        }
                        long dCost = (Dist[x,y] == long.MaxValue) ? 10000 : Dist[x,y];
                        long np;
                        if (val > 0) np = ((long)val*1000 - dCost) * m;
                        else np = (-1000 - dCost) * m;
                        
                        curProf += Math.Max(np, -1 * m);
                        
                        if (curProf > maxProf) {
                            maxProf = curProf;
                            best_d = curZ + 1;
                        }
                        curZ++;
                    }
                    if (maxProf > 0) TargetZ[x,y] = best_d;
                }
            }
            
            bool tChanged = true;
            int iters = 0;
            while(tChanged && iters < 50) {
                tChanged = false;
                iters++;
                for(int x=0; x<maxX; x++) {
                    for(int y=0; y<maxY; y++) {
                        if (TargetZ[x,y] == 0) continue;
                        for(int d=0; d<4; d++) {
                            int nx = x+dx[d]; int ny = y+dy[d];
                            if (nx>=0 && nx<maxX && ny>=0 && ny<maxY) {
                                if (TargetZ[nx, ny] < TargetZ[x,y] - 1) {
                                    TargetZ[nx, ny] = TargetZ[x,y] - 1;
                                    tChanged = true;
                                }
                            }
                        }
                    }
                }
            }
            
            var candidates = new List<Candidate>();
            for(int x=0; x<maxX; x++) {
                for(int y=0; y<maxY; y++) {
                    if (Dist[x,y] == long.MaxValue) continue;
                    long score = TargetZ[x,y] - Z_surface[x,y];
                    if (score > 0) {
                        int val, m;
                        if (DumpCount[x,y] > 0) { val=0; m=10; }
                        else {
                            byte v_d = Map[x][y*maxZ + Z_surface[x,y]];
                            val = v_d/16; m = v_d%16;
                        }
                        long imm = ((long)val*1000 - Dist[x,y]) * m;
                        if (val == 0) imm = -1 * m;
                        
                        score = imm + 1000000L * score;
                        candidates.Add(new Candidate { X = x, Y = y, Score = score });
                    }
                }
            }
            
            if (candidates.Count == 0) break;
            candidates.Sort((a,b) => b.Score.CompareTo(a.Score));
            
            var changed = new bool[maxX, maxY];
            int minedThisBatch = 0;
            
            foreach(var c in candidates) {
                if (changed[c.X, c.Y]) continue;
                
                int val, m;
                if (DumpCount[c.X, c.Y] > 0) { val=0; m=10; }
                else {
                    byte v_d = Map[c.X][c.Y*maxZ + Z_surface[c.X, c.Y]];
                    val = v_d/16; m = v_d%16;
                }
                
                long netPlant;
                if (val > 0) netPlant = ((long)val*1000 - Dist[c.X, c.Y]) * m;
                else netPlant = (-1000 - Dist[c.X, c.Y]) * m;
                
                long netDump = -1 * m;
                bool toPlant = netPlant >= netDump;
                
                bool conflict = false;
                var path = new List<(int,int)>();
                
                if (toPlant) {
                    int cx = c.X, cy = c.Y;
                    while(cx != 0 || cy != 0) {
                        int nx = NextHopX[cx, cy];
                        int ny = NextHopY[cx, cy];
                        if (nx == -1) { conflict = true; break; }
                        if (changed[nx, ny]) { conflict = true; break; }
                        path.Add((nx, ny));
                        cx = nx; cy = ny;
                    }
                } else {
                    int bestDumpX = -1, bestDumpY = -1;
                    long bestScore = -1;
                    for(int d=0; d<4; d++) {
                        int nx = c.X + dx[d];
                        int ny = c.Y + dy[d];
                        if (nx>=0 && nx<maxX && ny>=0 && ny<maxY) {
                            if (Z_surface[c.X, c.Y] - Z_surface[nx, ny] <= 1) {
                                if (!changed[nx, ny]) {
                                    long dScore = Dist[nx, ny];
                                    if (dScore == long.MaxValue) dScore = long.MaxValue - 1;
                                    if (TargetZ[nx, ny] > Z_surface[nx, ny]) dScore -= 1000000;
                                    
                                    if (dScore > bestScore) {
                                        bestScore = dScore;
                                        bestDumpX = nx; bestDumpY = ny;
                                    }
                                }
                            }
                        }
                    }
                    if (bestDumpX != -1) {
                        path.Add((bestDumpX, bestDumpY));
                    } else conflict = true;
                }
                
                if (!conflict) {
                    changed[c.X, c.Y] = true;
                    if (!toPlant) {
                        changed[path[0].Item1, path[0].Item2] = true;
                        DumpCount[path[0].Item1, path[0].Item2]++;
                        Z_surface[path[0].Item1, path[0].Item2]--;
                    }
                    
                    int digZ = ZVoxelCount - 1 - Z_surface[c.X, c.Y];
                    
                    int N = path.Count - 1;
                    if (N < 0) N = 0;
                    
                    outBuf.Add(N);
                    outBuf.Add(c.X); outBuf.Add(c.Y); outBuf.Add(digZ);
                    if (path.Count > 0) {
                        foreach(var p in path) {
                            outBuf.Add(p.Item1); outBuf.Add(p.Item2);
                        }
                    } else {
                        outBuf.Add(0); outBuf.Add(0);
                    }
                    
                    if (DumpCount[c.X, c.Y] > 0) DumpCount[c.X, c.Y]--;
                    Z_surface[c.X, c.Y]++;
                    
                    operations++;
                    minedThisBatch++;
                }
            }
            if (minedThisBatch == 0) break;
        }
        
        Console.WriteLine(operations);
        using (var stdout = new StreamWriter(Console.OpenStandardOutput(), Encoding.ASCII, 262144)) {
            int i = 0;
            while(i < outBuf.Count) {
                int N = outBuf[i++];
                stdout.Write(N); stdout.Write(' ');
                stdout.Write(outBuf[i++]); stdout.Write(' ');
                stdout.Write(outBuf[i++]); stdout.Write(' ');
                stdout.Write(outBuf[i++]);
                int pairs = N + 1;
                if (pairs == 1 && N == 0) pairs = 1; 
                for(int p=0; p<pairs; p++) {
                    stdout.Write(' '); stdout.Write(outBuf[i++]);
                    stdout.Write(' '); stdout.Write(outBuf[i++]);
                }
                stdout.WriteLine();
            }
        }
    }
}
