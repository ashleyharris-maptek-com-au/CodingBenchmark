import sys
import time
from array import array

def solve():
    start_time = time.time()
    
    iterator = iter(sys.stdin)
    try:
        while True:
            line = next(iterator)
            if line.startswith('c'):
                continue
            first = line.split()
            if not first:
                continue
            if first[0] == 'p':
                n_vars = int(first[2])
                break
            else:
                n_vars = int(first[0])
                break
    except StopIteration:
        return
        
    vars_arr = array('i')
    reqs_arr = bytearray()
    
    pos = array('i', [0] * (n_vars + 1))
    neg = array('i', [0] * (n_vars + 1))
    
    vars_append = vars_arr.append
    reqs_append = reqs_arr.append
    
    for line in iterator:
        if line.startswith('c') or line.startswith('p'):
            continue
            
        tokens = line.split()
        if len(tokens) < 3: 
            continue
            
        l1, l2, l3 = map(int, tokens[:3])
        
        if l1 > 0:
            v1 = l1; r1 = 1; pos[l1] += 1
        else:
            v1 = -l1; r1 = 0; neg[-l1] += 1
            
        if l2 > 0:
            v2 = l2; r2 = 1; pos[l2] += 1
        else:
            v2 = -l2; r2 = 0; neg[-l2] += 1
            
        if l3 > 0:
            v3 = l3; r3 = 1; pos[l3] += 1
        else:
            v3 = -l3; r3 = 0; neg[-l3] += 1
            
        vars_append(v1)
        vars_append(v2)
        vars_append(v3)
        reqs_append(r1)
        reqs_append(r2)
        reqs_append(r3)
        
    assignment = bytearray(n_vars + 1)
    for i in range(1, n_vars + 1):
        assignment[i] = 1 if pos[i] >= neg[i] else 0

    seed = 12345
    
    while True:
        conflicts = 0
        v_iter = iter(vars_arr)
        r_iter = iter(reqs_arr)
        
        for v1, v2, v3, r1, r2, r3 in zip(v_iter, v_iter, v_iter, r_iter, r_iter, r_iter):
            if assignment[v1] == r1: continue
            if assignment[v2] == r2: continue
            if assignment[v3] == r3: continue
            
            conflicts += 1
            seed = (seed * 1103515245 + 12345) & 0x7FFFFFFF
            r = seed % 3
            if r == 0:
                assignment[v1] = r1
            elif r == 1:
                assignment[v2] = r2
            else:
                assignment[v3] = r3
                
        if conflicts == 0:
            break
            
        if time.time() - start_time > 280:
            break

    sys.stdout.write("SAT\n")
    if n_vars > 0:
        out = bytearray(n_vars * 2)
        for i in range(1, n_vars + 1):
            out[(i-1)*2] = 49 if assignment[i] else 48
            out[(i-1)*2+1] = 32
        out[-1] = 10
        sys.stdout.buffer.write(out)

if __name__ == '__main__':
    solve()
