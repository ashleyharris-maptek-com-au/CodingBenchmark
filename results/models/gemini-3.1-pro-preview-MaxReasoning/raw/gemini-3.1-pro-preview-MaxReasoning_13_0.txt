#include <iostream>
#include <vector>
#include <algorithm>
#include <chrono>
#include <random>

using namespace std;

void radix_sort(vector<uint64_t>& a) {
    if (a.empty()) return;
    vector<uint64_t> b(a.size());
    for (int shift = 0; shift < 64; shift += 8) {
        int count[256] = {0};
        for (uint64_t x : a) count[(x >> shift) & 0xFF]++;
        for (int i = 1; i < 256; ++i) count[i] += count[i - 1];
        for (int i = (int)a.size() - 1; i >= 0; --i) {
            b[--count[(a[i] >> shift) & 0xFF]] = a[i];
        }
        a.swap(b);
    }
}

int main() {
    int n = 10000000;
    vector<uint64_t> a(n), b(n), c(n);
    mt19937_64 rng(1337);
    for(int i=0; i<n; ++i) {
        a[i] = rng();
        b[i] = a[i];
        c[i] = a[i];
    }
    
    auto t0 = chrono::high_resolution_clock::now();
    sort(a.begin(), a.end());
    auto t1 = chrono::high_resolution_clock::now();
    cout << "std::sort: " << chrono::duration_cast<chrono::milliseconds>(t1-t0).count() << " ms" << endl;
    
    t0 = chrono::high_resolution_clock::now();
    radix_sort(b);
    t1 = chrono::high_resolution_clock::now();
    cout << "radix_sort: " << chrono::duration_cast<chrono::milliseconds>(t1-t0).count() << " ms" << endl;
    
    return 0;
}
