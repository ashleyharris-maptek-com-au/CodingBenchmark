using System;
using System.Linq;

class TetrisPlayer
{
    static int[][][] GetPiece(string type)
    {
        switch (type)
        {
            case "I": return new int[][][] {
                new int[][] { new int[]{0,0}, new int[]{1,0}, new int[]{2,0}, new int[]{3,0} },
                new int[][] { new int[]{0,0}, new int[]{0,1}, new int[]{0,2}, new int[]{0,3} },
                new int[][] { new int[]{0,0}, new int[]{1,0}, new int[]{2,0}, new int[]{3,0} },
                new int[][] { new int[]{0,0}, new int[]{0,1}, new int[]{0,2}, new int[]{0,3} }
            };
            case "O": return new int[][][] {
                new int[][] { new int[]{0,0}, new int[]{1,0}, new int[]{0,1}, new int[]{1,1} },
                new int[][] { new int[]{0,0}, new int[]{1,0}, new int[]{0,1}, new int[]{1,1} },
                new int[][] { new int[]{0,0}, new int[]{1,0}, new int[]{0,1}, new int[]{1,1} },
                new int[][] { new int[]{0,0}, new int[]{1,0}, new int[]{0,1}, new int[]{1,1} }
            };
            case "T": return new int[][][] {
                new int[][] { new int[]{0,0}, new int[]{1,0}, new int[]{2,0}, new int[]{1,1} },
                new int[][] { new int[]{0,0}, new int[]{0,1}, new int[]{0,2}, new int[]{1,1} },
                new int[][] { new int[]{0,1}, new int[]{1,1}, new int[]{2,1}, new int[]{1,0} },
                new int[][] { new int[]{1,0}, new int[]{0,1}, new int[]{1,1}, new int[]{1,2} }
            };
            case "S": return new int[][][] {
                new int[][] { new int[]{0,0}, new int[]{1,0}, new int[]{1,1}, new int[]{2,1} },
                new int[][] { new int[]{0,2}, new int[]{0,1}, new int[]{1,1}, new int[]{1,0} },
                new int[][] { new int[]{0,0}, new int[]{1,0}, new int[]{1,1}, new int[]{2,1} },
                new int[][] { new int[]{0,2}, new int[]{0,1}, new int[]{1,1}, new int[]{1,0} }
            };
            case "Z": return new int[][][] {
                new int[][] { new int[]{0,1}, new int[]{1,1}, new int[]{1,0}, new int[]{2,0} },
                new int[][] { new int[]{1,2}, new int[]{0,1}, new int[]{1,1}, new int[]{0,0} },
                new int[][] { new int[]{0,1}, new int[]{1,1}, new int[]{1,0}, new int[]{2,0} },
                new int[][] { new int[]{1,2}, new int[]{0,1}, new int[]{1,1}, new int[]{0,0} }
            };
            case "J": return new int[][][] {
                new int[][] { new int[]{0,1}, new int[]{0,0}, new int[]{1,0}, new int[]{2,0} },
                new int[][] { new int[]{0,2}, new int[]{1,2}, new int[]{0,1}, new int[]{0,0} },
                new int[][] { new int[]{0,1}, new int[]{1,1}, new int[]{2,1}, new int[]{2,0} },
                new int[][] { new int[]{1,2}, new int[]{1,1}, new int[]{1,0}, new int[]{0,0} }
            };
            case "L": return new int[][][] {
                new int[][] { new int[]{2,1}, new int[]{0,0}, new int[]{1,0}, new int[]{2,0} },
                new int[][] { new int[]{0,2}, new int[]{0,1}, new int[]{0,0}, new int[]{1,0} },
                new int[][] { new int[]{0,1}, new int[]{1,1}, new int[]{2,1}, new int[]{0,0} },
                new int[][] { new int[]{0,2}, new int[]{1,2}, new int[]{1,1}, new int[]{1,0} }
            };
        }
        return null;
    }

    class EvalResult {
        public int LinesCleared;
        public int ErodedCells;
        public bool GameOver;
        public int LandingHeight;
        public double Score;
    }

    class Board
    {
        public int Width;
        public int Height;
        public bool[,] Grid;
        public int[] ColHeights;

        public Board(int w, int h)
        {
            Width = w;
            Height = h;
            Grid = new bool[h, w];
            ColHeights = new int[w];
        }

        public Board Clone()
        {
            Board b = new Board(Width, Height);
            Array.Copy(this.Grid, b.Grid, this.Grid.Length);
            Array.Copy(this.ColHeights, b.ColHeights, this.ColHeights.Length);
            return b;
        }

        public EvalResult PlacePiece(int[][] piece, int c)
        {
            EvalResult res = new EvalResult();
            int baseY = 0;
            foreach (var pt in piece)
            {
                int x = c + pt[0];
                int y = ColHeights[x] - pt[1];
                if (y > baseY) baseY = y;
            }

            res.LandingHeight = baseY;

            int maxY = 0;
            foreach (var pt in piece)
            {
                int py = pt[1];
                int y = baseY + py;
                if (y >= Height) {
                    res.GameOver = true;
                    return res;
                }
                if (y > maxY) maxY = y;
            }

            // Temporarily mark grid
            foreach (var pt in piece) {
                Grid[baseY + pt[1], c + pt[0]] = true;
            }

            int linesCleared = 0;
            int pieceCellsEliminated = 0;
            bool[] lineCleared = new bool[Height];

            for (int y = baseY; y <= maxY; y++)
            {
                bool full = true;
                for (int x = 0; x < Width; x++)
                {
                    if (!Grid[y, x])
                    {
                        full = false;
                        break;
                    }
                }
                if (full)
                {
                    linesCleared++;
                    lineCleared[y] = true;
                }
            }

            if (linesCleared > 0) {
                foreach (var pt in piece) {
                    if (lineCleared[baseY + pt[1]]) {
                        pieceCellsEliminated++;
                    }
                }
                res.ErodedCells = linesCleared * pieceCellsEliminated;

                int writeY = baseY;
                for (int readY = baseY; readY < Height; readY++) {
                    if (!lineCleared[readY]) {
                        if (writeY != readY) {
                            for (int x = 0; x < Width; x++) Grid[writeY, x] = Grid[readY, x];
                        }
                        writeY++;
                    }
                }
                for (int y = writeY; y < Height; y++) {
                    for (int x = 0; x < Width; x++) Grid[y, x] = false;
                }
            }

            res.LinesCleared = linesCleared;

            int maxH = 0;
            for (int x = 0; x < Width; x++)
            {
                ColHeights[x] = 0;
                for (int y = Height - 1; y >= 0; y--)
                {
                    if (Grid[y, x])
                    {
                        ColHeights[x] = y + 1;
                        break;
                    }
                }
                if (ColHeights[x] > maxH) maxH = ColHeights[x];
            }

            int rowTransitions = 0;
            for (int y = 0; y < maxH; y++)
            {
                bool lastCell = true; 
                for (int x = 0; x < Width; x++)
                {
                    bool cell = Grid[y, x];
                    if (cell != lastCell) rowTransitions++;
                    lastCell = cell;
                }
                if (!lastCell) rowTransitions++; 
            }

            int colTransitions = 0;
            for (int x = 0; x < Width; x++)
            {
                bool lastCell = true; 
                for (int y = 0; y < ColHeights[x]; y++)
                {
                    bool cell = Grid[y, x];
                    if (cell != lastCell) colTransitions++;
                    lastCell = cell;
                }
                if (lastCell) colTransitions++; 
            }

            int holes = 0;
            for (int x = 0; x < Width; x++)
            {
                int colHoles = 0;
                for (int y = 0; y < ColHeights[x]; y++)
                {
                    if (!Grid[y, x]) colHoles++;
                }
                holes += colHoles;
            }

            int wells = 0;
            for (int x = 0; x < Width; x++)
            {
                int depth = 0;
                for (int y = Height - 1; y >= 0; y--)
                {
                    if (!Grid[y, x])
                    {
                        bool leftSolid = (x == 0) || Grid[y, x - 1];
                        bool rightSolid = (x == Width - 1) || Grid[y, x + 1];
                        if (leftSolid && rightSolid)
                        {
                            depth++;
                            wells += depth;
                        }
                        else
                        {
                            depth = 0;
                        }
                    }
                    else
                    {
                        depth = 0;
                    }
                }
            }

            double score = 0;
            score += -4.500158825082766 * res.LandingHeight;
            score +=  3.4181268101392694 * res.LinesCleared;
            score += -3.2178882868487753 * rowTransitions;
            score += -9.348695305445199 * colTransitions;
            score += -7.899265427351652 * holes;
            score += -3.3855972247263626 * wells;
            
            res.Score = score;
            return res;
        }
    }

    public static void Main(string[] args)
    {
        if (args.Length > 0 && args[0] == "test")
        {
            TestSimulator();
            return;
        }

        string initLine = Console.ReadLine();
        if (initLine == null) return;
        string[] parts = initLine.Split(new char[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        int width = int.Parse(parts[0]);
        int height = int.Parse(parts[1]);
        int numPieces = int.Parse(parts[2]);

        Board board = new Board(width, height);

        for (int i = 0; i < numPieces; i++)
        {
            string pieceType = Console.ReadLine()?.Trim();
            if (string.IsNullOrEmpty(pieceType)) break;

            int[][][] pieceRots = GetPiece(pieceType);
            
            double bestScore = double.MinValue;
            int bestR = 0;
            int bestC = 0;
            bool foundMove = false;

            for (int r = 0; r < 4; r++)
            {
                int[][] piece = pieceRots[r];
                int pieceWidth = piece.Max(pt => pt[0]) + 1;
                
                for (int c = 0; c <= width - pieceWidth; c++)
                {
                    Board b = board.Clone();
                    EvalResult res = b.PlacePiece(piece, c);
                    if (!res.GameOver)
                    {
                        if (!foundMove || res.Score > bestScore)
                        {
                            bestScore = res.Score;
                            bestR = r;
                            bestC = c;
                            foundMove = true;
                        }
                    }
                }
            }

            if (!foundMove)
            {
                bestR = 0;
                bestC = 0;
            }

            Console.WriteLine($"{bestR} {bestC}");
            
            string result = Console.ReadLine();
            if (result != null && result.Contains("gameover")) break;

            // Apply best move to real board
            if (foundMove) {
                board.PlacePiece(pieceRots[bestR], bestC);
            }
        }
    }

    static void TestSimulator()
    {
        int width = 10;
        int height = 20;
        Board board = new Board(width, height);
        Random rng = new Random(42);
        string[] types = {"I", "O", "T", "S", "Z", "J", "L"};
        
        int linesTotal = 0;
        int pieces = 0;
        
        while (true)
        {
            string pieceType = types[rng.Next(types.Length)];
            int[][][] pieceRots = GetPiece(pieceType);
            
            double bestScore = double.MinValue;
            int bestR = 0;
            int bestC = 0;
            bool foundMove = false;

            for (int r = 0; r < 4; r++)
            {
                int[][] piece = pieceRots[r];
                int pieceWidth = piece.Max(pt => pt[0]) + 1;
                
                for (int c = 0; c <= width - pieceWidth; c++)
                {
                    Board b = board.Clone();
                    EvalResult res = b.PlacePiece(piece, c);
                    if (!res.GameOver)
                    {
                        if (!foundMove || res.Score > bestScore)
                        {
                            bestScore = res.Score;
                            bestR = r;
                            bestC = c;
                            foundMove = true;
                        }
                    }
                }
            }

            if (!foundMove)
            {
                Console.WriteLine($"Game over after {pieces} pieces. Lines cleared: {linesTotal}");
                break;
            }

            EvalResult finalRes = board.PlacePiece(pieceRots[bestR], bestC);
            linesTotal += finalRes.LinesCleared;
            pieces++;
            
            if (pieces % 1000 == 0) {
                Console.WriteLine($"Pieces: {pieces}, Lines: {linesTotal}");
            }
            if (pieces >= 50000) {
                Console.WriteLine($"Reached {pieces} pieces. Lines: {linesTotal}. Ending test.");
                break;
            }
        }
    }
}
