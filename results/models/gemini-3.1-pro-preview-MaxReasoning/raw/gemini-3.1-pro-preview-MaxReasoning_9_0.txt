using System;
using System.IO;

class Program
{
    static void Main(string[] args)
    {
        using (Stream stream = Console.OpenStandardInput())
        {
            Scanner scanner = new Scanner(stream);
            
            int N = scanner.NextInt();
            if (N <= 0)
            {
                Console.WriteLine(0);
                return;
            }
            
            long stock_length = scanner.NextLong();
            if (stock_length <= 0)
            {
                Console.WriteLine(0);
                return;
            }
            
            long[] lengths = new long[N];
            int[] ids = new int[N];
            
            for (int i = 0; i < N; i++)
            {
                lengths[i] = scanner.NextLong();
                ids[i] = i;
            }
            
            // Sort lengths and ids in ascending order
            Array.Sort(lengths, ids);
            
            long K = 1;
            while (K < N) K *= 2;
            int kInt = (int)K;
            
            // Segment tree for fast capacity queries
            long[] tree = new long[2 * kInt];
            for (int i = 1; i < 2 * kInt; i++)
            {
                tree[i] = stock_length;
            }
            
            int[] head = new int[N];
            int[] tail = new int[N];
            int[] next = new int[N];
            
            for (int i = 0; i < N; i++)
            {
                head[i] = -1;
                tail[i] = -1;
                next[i] = -1;
            }
            
            int max_bin = -1;
            
            // Process items from largest to smallest (First Fit Decreasing)
            for (int i = N - 1; i >= 0; i--)
            {
                long val = lengths[i];
                int id = ids[i];
                
                int node = 1;
                if (tree[1] < val)
                {
                    throw new Exception("Item exceeds stock length or out of bins");
                }
                
                // Find the first bin that can accommodate 'val'
                while (node < kInt)
                {
                    int left = node << 1;
                    if (tree[left] >= val)
                    {
                        node = left;
                    }
                    else
                    {
                        node = left + 1;
                    }
                }
                
                tree[node] -= val;
                int bin_idx = node - kInt;
                if (bin_idx > max_bin) max_bin = bin_idx;
                
                // Append item to the assigned bin
                if (head[bin_idx] == -1)
                {
                    head[bin_idx] = id;
                    tail[bin_idx] = id;
                }
                else
                {
                    next[tail[bin_idx]] = id;
                    tail[bin_idx] = id;
                }
                
                // Update max capacities up to root
                int curr = node >> 1;
                while (curr > 0)
                {
                    int left_idx = curr << 1;
                    long left_val = tree[left_idx];
                    long right_val = tree[left_idx + 1];
                    tree[curr] = left_val > right_val ? left_val : right_val;
                    curr >>= 1;
                }
            }
            
            int num_stocks = max_bin + 1;
            
            using (FastWriter writer = new FastWriter(Console.OpenStandardOutput()))
            {
                writer.WriteInt(num_stocks);
                writer.WriteChar('\n');
                for (int i = 0; i < num_stocks; i++)
                {
                    int curr = head[i];
                    bool first = true;
                    while (curr != -1)
                    {
                        if (!first) writer.WriteChar(' ');
                        writer.WriteInt(curr);
                        first = false;
                        curr = next[curr];
                    }
                    writer.WriteChar('\n');
                }
            }
        }
    }

    class Scanner
    {
        private Stream stream;
        private byte[] buffer = new byte[1024 * 1024];
        private int head = 0;
        private int tail = 0;

        public Scanner(Stream stream)
        {
            this.stream = stream;
        }

        public int NextInt()
        {
            int c = Read();
            while (c <= ' ')
            {
                if (c == -1) return -1;
                c = Read();
            }
            int res = 0;
            while (c > ' ')
            {
                res = res * 10 + (c - '0');
                c = Read();
            }
            return res;
        }
        
        public long NextLong()
        {
            int c = Read();
            while (c <= ' ')
            {
                if (c == -1) return -1;
                c = Read();
            }
            long res = 0;
            while (c > ' ')
            {
                res = res * 10 + (c - '0');
                c = Read();
            }
            return res;
        }

        private int Read()
        {
            if (head >= tail)
            {
                head = 0;
                tail = stream.Read(buffer, 0, buffer.Length);
                if (tail == 0) return -1;
            }
            return buffer[head++];
        }
    }

    class FastWriter : IDisposable
    {
        private Stream stream;
        private byte[] buffer;
        private int pos;

        public FastWriter(Stream stream, int bufferSize = 1024 * 1024)
        {
            this.stream = stream;
            this.buffer = new byte[bufferSize];
            this.pos = 0;
        }

        public void WriteInt(int value)
        {
            if (value == 0)
            {
                WriteChar('0');
                return;
            }
            if (value < 0)
            {
                WriteChar('-');
                value = -value;
            }
            
            int temp = value;
            int numDigits = 0;
            while (temp > 0)
            {
                numDigits++;
                temp /= 10;
            }
            
            if (pos + numDigits > buffer.Length)
            {
                Flush();
            }
            
            int start = pos;
            pos += numDigits;
            int end = pos - 1;
            
            temp = value;
            while (temp > 0)
            {
                buffer[end--] = (byte)('0' + (temp % 10));
                temp /= 10;
            }
        }

        public void WriteChar(char c)
        {
            if (pos == buffer.Length) Flush();
            buffer[pos++] = (byte)c;
        }

        public void Flush()
        {
            if (pos > 0)
            {
                stream.Write(buffer, 0, pos);
                pos = 0;
            }
        }

        public void Dispose()
        {
            Flush();
        }
    }
}
