using System;
using System.Collections.Generic;
using System.Globalization;

namespace AsteroidIntercept
{
    public struct Vector3
    {
        public double X, Y, Z;
        public Vector3(double x, double y, double z) { X = x; Y = y; Z = z; }
        public static Vector3 operator +(Vector3 a, Vector3 b) => new Vector3(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
        public static Vector3 operator -(Vector3 a, Vector3 b) => new Vector3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
        public static Vector3 operator *(Vector3 a, double d) => new Vector3(a.X * d, a.Y * d, a.Z * d);
        public static Vector3 operator /(Vector3 a, double d) => new Vector3(a.X / d, a.Y / d, a.Z / d);
        public static double Dot(Vector3 a, Vector3 b) => a.X * b.X + a.Y * b.Y + a.Z * b.Z;
        public static Vector3 Cross(Vector3 a, Vector3 b) => new Vector3(
            a.Y * b.Z - a.Z * b.Y,
            a.Z * b.X - a.X * b.Z,
            a.X * b.Y - a.Y * b.X
        );
        public double Magnitude() => Math.Sqrt(X * X + Y * Y + Z * Z);
    }

    public class Body
    {
        public string name;
        public double mass;
        public Vector3 pos;
        public Vector3 vel;
        public Vector3 acc;
    }

    public class Program
    {
        const double G = 6.67430e-11;

        public static void Main(string[] args)
        {
            CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;

            string input = Console.In.ReadToEnd();
            string[] tokens = input.Split(new char[] { ' ', '\t', '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries);
            if (tokens.Length < 6) return;

            int num_bodies = int.Parse(tokens[0]);
            double warning_days = double.Parse(tokens[1]);
            double impactor_mass = double.Parse(tokens[2]);
            double delta_v_budget = double.Parse(tokens[3]);
            double asteroid_mass = double.Parse(tokens[4]);
            double t0_unix_utc = double.Parse(tokens[5]);

            List<Body> bodies = new List<Body>();
            int idx = 6;
            for (int i = 0; i < num_bodies; i++)
            {
                Body b = new Body();
                b.name = tokens[idx++];
                b.mass = double.Parse(tokens[idx++]);
                b.pos.X = double.Parse(tokens[idx++]);
                b.pos.Y = double.Parse(tokens[idx++]);
                b.pos.Z = double.Parse(tokens[idx++]);
                b.vel.X = double.Parse(tokens[idx++]);
                b.vel.Y = double.Parse(tokens[idx++]);
                b.vel.Z = double.Parse(tokens[idx++]);
                bodies.Add(b);
            }

            int earth_idx = bodies.FindIndex(b => b.name == "Earth");
            int ast_idx = bodies.FindIndex(b => b.name == "Asteroid");
            int sun_idx = bodies.FindIndex(b => b.name == "Sun");

            double SunMass = sun_idx >= 0 ? bodies[sun_idx].mass : 1.989e30;

            double dt = 3600.0;
            double sim_days = warning_days + 3650.0;
            int max_steps = (int)Math.Ceiling(sim_days * 86400.0 / dt);

            List<Vector3> earth_traj = new List<Vector3>();
            List<Vector3> earth_vel_traj = new List<Vector3>();
            List<Vector3> ast_traj = new List<Vector3>();
            List<Vector3> ast_vel_traj = new List<Vector3>();
            List<Vector3> sun_traj = new List<Vector3>();
            List<Vector3> sun_vel_traj = new List<Vector3>();

            ComputeAccelerations(bodies);

            double t_hit = sim_days;
            double min_dist = double.MaxValue;

            for (int step = 0; step <= max_steps; step++)
            {
                if (step % 24 == 0)
                {
                    earth_traj.Add(bodies[earth_idx].pos);
                    earth_vel_traj.Add(bodies[earth_idx].vel);
                    ast_traj.Add(bodies[ast_idx].pos);
                    ast_vel_traj.Add(bodies[ast_idx].vel);
                    if (sun_idx >= 0) {
                        sun_traj.Add(bodies[sun_idx].pos);
                        sun_vel_traj.Add(bodies[sun_idx].vel);
                    } else {
                        sun_traj.Add(new Vector3());
                        sun_vel_traj.Add(new Vector3());
                    }

                    double dist = (bodies[earth_idx].pos - bodies[ast_idx].pos).Magnitude();
                    if (dist < min_dist)
                    {
                        min_dist = dist;
                        t_hit = step * dt / 86400.0;
                    }
                }

                for (int i = 0; i < num_bodies; i++)
                {
                    bodies[i].vel += bodies[i].acc * (dt / 2.0);
                    bodies[i].pos += bodies[i].vel * dt;
                }

                ComputeAccelerations(bodies);

                for (int i = 0; i < num_bodies; i++)
                {
                    bodies[i].vel += bodies[i].acc * (dt / 2.0);
                }
            }

            int max_iL = (int)Math.Min(warning_days, t_hit - 10);
            if (max_iL < 0) max_iL = 0;

            int stepL = Math.Max(1, max_iL / 200);

            double best_score = double.MinValue;
            double best_dv = 0;
            double best_tL = 0;
            double best_tI = 0;
            Vector3 best_v_depart = new Vector3();

            for (int iL = 0; iL <= max_iL; iL += stepL)
            {
                double tL = iL;
                Vector3 rE = earth_traj[iL];
                Vector3 vE = earth_vel_traj[iL];
                Vector3 rS_L = sun_traj[iL];
                Vector3 vS_L = sun_vel_traj[iL];

                int min_iI = iL + 10;
                int max_iI = (int)t_hit - 1;
                if (max_iI < min_iI) continue;

                int stepI = Math.Max(1, (max_iI - min_iI) / 200);

                for (int iI = min_iI; iI <= max_iI; iI += stepI)
                {
                    double tI = iI;
                    double tof = (tI - tL) * 86400.0;

                    Vector3 rA = ast_traj[iI];
                    Vector3 vA = ast_vel_traj[iI];
                    Vector3 rS_I = sun_traj[iI];
                    Vector3 vS_I = sun_vel_traj[iI];

                    Vector3 rE_sun = rE - rS_L;
                    Vector3 rA_sun = rA - rS_I;

                    if (!SolveLambert(rE_sun, rA_sun, tof, SunMass * G, true, out Vector3 v_depart_sun, out Vector3 v_arrive_sun))
                        continue;

                    Vector3 v_depart = v_depart_sun + vS_L;
                    Vector3 v_arrive = v_arrive_sun + vS_I;

                    Vector3 dv_launch = v_depart - vE;
                    double dv_mag = dv_launch.Magnitude();

                    Vector3 v_rel = v_arrive - vA;
                    Vector3 dv_ast = v_rel * (impactor_mass / (asteroid_mass + impactor_mass));
                    double deflection = dv_ast.Magnitude() * (t_hit - tI) * 86400.0;

                    bool valid = (dv_mag <= delta_v_budget) && (deflection >= 1.0e7);

                    double score = 0;
                    if (valid)
                    {
                        score = 1e9 - dv_mag + deflection * 1e-4;
                    }
                    else
                    {
                        if (dv_mag <= delta_v_budget) score = deflection - 1e9;
                        else score = -1e12 - dv_mag;
                    }

                    if (score > best_score)
                    {
                        best_score = score;
                        best_dv = dv_mag;
                        best_tL = tL;
                        best_tI = tI;
                        best_v_depart = v_depart;
                    }
                }
            }

            if (best_tL >= 0 && best_score > double.MinValue)
            {
                int best_iL_coarse = (int)best_tL;
                int best_iI_coarse = (int)best_tI;

                int refine_rangeL = stepL * 2;
                int refine_rangeI = stepI * 2;

                int ref_minL = Math.Max(0, best_iL_coarse - refine_rangeL);
                int ref_maxL = Math.Min(max_iL, best_iL_coarse + refine_rangeL);

                for (int iL = ref_minL; iL <= ref_maxL; iL++)
                {
                    double tL = iL;
                    Vector3 rE = earth_traj[iL];
                    Vector3 vE = earth_vel_traj[iL];
                    Vector3 rS_L = sun_traj[iL];
                    Vector3 vS_L = sun_vel_traj[iL];

                    int ref_minI = Math.Max(iL + 10, best_iI_coarse - refine_rangeI);
                    int ref_maxI = Math.Min((int)t_hit - 1, best_iI_coarse + refine_rangeI);

                    for (int iI = ref_minI; iI <= ref_maxI; iI++)
                    {
                        double tI = iI;
                        double tof = (tI - tL) * 86400.0;

                        Vector3 rA = ast_traj[iI];
                        Vector3 vA = ast_vel_traj[iI];
                        Vector3 rS_I = sun_traj[iI];
                        Vector3 vS_I = sun_vel_traj[iI];

                        Vector3 rE_sun = rE - rS_L;
                        Vector3 rA_sun = rA - rS_I;

                        if (!SolveLambert(rE_sun, rA_sun, tof, SunMass * G, true, out Vector3 v_depart_sun, out Vector3 v_arrive_sun))
                            continue;

                        Vector3 v_depart = v_depart_sun + vS_L;
                        Vector3 v_arrive = v_arrive_sun + vS_I;

                        Vector3 dv_launch = v_depart - vE;
                        double dv_mag = dv_launch.Magnitude();

                        Vector3 v_rel = v_arrive - vA;
                        Vector3 dv_ast = v_rel * (impactor_mass / (asteroid_mass + impactor_mass));
                        double deflection = dv_ast.Magnitude() * (t_hit - tI) * 86400.0;

                        bool valid = (dv_mag <= delta_v_budget) && (deflection >= 1.0e7);

                        double score = 0;
                        if (valid)
                        {
                            score = 1e9 - dv_mag + deflection * 1e-4;
                        }
                        else
                        {
                            if (dv_mag <= delta_v_budget) score = deflection - 1e9;
                            else score = -1e12 - dv_mag;
                        }

                        if (score > best_score)
                        {
                            best_score = score;
                            best_dv = dv_mag;
                            best_tL = tL;
                            best_tI = tI;
                            best_v_depart = v_depart;
                        }
                    }
                }
            }

            Vector3 final_dv_launch = best_v_depart - earth_vel_traj[(int)best_tL];
            
            Console.WriteLine("2");
            Console.WriteLine($"{best_tL:F6} {final_dv_launch.X:F6} {final_dv_launch.Y:F6} {final_dv_launch.Z:F6}");
            Console.WriteLine($"{best_tI:F6} 0.000000 0.000000 0.000000");
        }

        static void ComputeAccelerations(List<Body> bodies)
        {
            int n = bodies.Count;
            for (int i = 0; i < n; i++) bodies[i].acc = new Vector3(0, 0, 0);

            for (int i = 0; i < n; i++)
            {
                for (int j = i + 1; j < n; j++)
                {
                    Vector3 diff = bodies[j].pos - bodies[i].pos;
                    double distSq = diff.X * diff.X + diff.Y * diff.Y + diff.Z * diff.Z;
                    double dist = Math.Sqrt(distSq);
                    double f = G / (distSq * dist);

                    Vector3 force_i = diff * (f * bodies[j].mass);
                    Vector3 force_j = diff * (-f * bodies[i].mass);

                    bodies[i].acc += force_i;
                    bodies[j].acc += force_j;
                }
            }
        }

        public static bool SolveLambert(Vector3 r1, Vector3 r2, double dt, double mu, bool prograde, out Vector3 v1, out Vector3 v2)
        {
            v1 = new Vector3();
            v2 = new Vector3();
            double r1_mag = r1.Magnitude();
            double r2_mag = r2.Magnitude();
            double cos_dnu = Vector3.Dot(r1, r2) / (r1_mag * r2_mag);
            
            if (cos_dnu > 0.999999) cos_dnu = 0.999999;
            if (cos_dnu < -0.999999) cos_dnu = -0.999999;

            Vector3 cross = Vector3.Cross(r1, r2);
            double sin_dnu = cross.Magnitude() / (r1_mag * r2_mag);
            if (!prograde) sin_dnu = -sin_dnu;

            double A = sin_dnu * Math.Sqrt(r1_mag * r2_mag / (1.0 - cos_dnu));
            if (Math.Abs(A) < 1e-12) return false;

            double z = 0;
            double C, S;
            double y = 0;

            double z_up = 4 * Math.PI * Math.PI;
            double z_low = -4 * Math.PI * Math.PI;

            int iter = 0;
            while (iter < 100)
            {
                Stumpff(z, out C, out S);
                y = r1_mag + r2_mag + A * (z * S - 1.0) / Math.Sqrt(C);

                if (A > 0 && y < 0)
                {
                    z_low = z;
                    z = (z + z_up) / 2.0;
                    iter++;
                    continue;
                }

                double x = Math.Sqrt(Math.Max(0.0, y / C));
                double tof_calc = (x * x * x * S + A * Math.Sqrt(Math.Max(0.0, y))) / Math.Sqrt(mu);

                double error = tof_calc - dt;
                if (Math.Abs(error) < 1e-6)
                {
                    break;
                }

                if (error < 0) z_low = z;
                else z_up = z;

                z = (z_up + z_low) / 2.0;
                iter++;
            }

            if (iter >= 100) return false;

            double f = 1.0 - y / r1_mag;
            double g_dot = 1.0 - y / r2_mag;
            double g = A * Math.Sqrt(Math.Max(0.0, y / mu));

            v1 = (r2 - r1 * f) / g;
            v2 = (r2 * g_dot - r1) / g;

            return true;
        }

        static void Stumpff(double z, out double c, out double s)
        {
            if (z > 1e-6)
            {
                double sqz = Math.Sqrt(z);
                c = (1.0 - Math.Cos(sqz)) / z;
                s = (sqz - Math.Sin(sqz)) / (sqz * sqz * sqz);
            }
            else if (z < -1e-6)
            {
                double sqz = Math.Sqrt(-z);
                c = (Math.Cosh(sqz) - 1.0) / (-z);
                s = (Math.Sinh(sqz) - sqz) / (sqz * sqz * sqz);
            }
            else
            {
                c = 1.0 / 2.0 - z / 24.0 + z * z / 720.0;
                s = 1.0 / 6.0 - z / 120.0 + z * z / 5040.0;
            }
        }
    }
