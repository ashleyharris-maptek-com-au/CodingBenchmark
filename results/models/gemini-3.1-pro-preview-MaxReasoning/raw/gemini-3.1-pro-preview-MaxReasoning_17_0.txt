using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;

class Program {
    class Scanner {
        Stream s;
        byte[] buf = new byte[8192];
        int pos = 0;
        int len = 0;

        public Scanner(Stream stream) { s = stream; }

        public int NextInt() {
            int res = 0;
            int b = Read();
            while (b <= ' ') {
                if (b == -1) return -1;
                b = Read();
            }
            while (b > ' ') {
                res = res * 10 + (b - '0');
                b = Read();
            }
            return res;
        }

        int Read() {
            if (pos >= len) {
                pos = 0;
                len = s.Read(buf, 0, buf.Length);
                if (len == 0) return -1;
            }
            return buf[pos++];
        }
    }

    struct EMS {
        public int x1, y1, z1, x2, y2, z2;
        public int w => x2 - x1;
        public int h => y2 - y1;
        public int d => z2 - z1;
        public long Volume => (long)w * h * d;

        public EMS(int x1, int y1, int z1, int x2, int y2, int z2) {
            this.x1 = x1; this.y1 = y1; this.z1 = z1;
            this.x2 = x2; this.y2 = y2; this.z2 = z2;
        }

        public bool Contains(EMS other) {
            return x1 <= other.x1 && y1 <= other.y1 && z1 <= other.z1 &&
                   x2 >= other.x2 && y2 >= other.y2 && z2 >= other.z2;
        }

        public bool Intersects(int bx1, int by1, int bz1, int bx2, int by2, int bz2) {
            return x1 < bx2 && x2 > bx1 &&
                   y1 < by2 && y2 > by1 &&
                   z1 < bz2 && z2 > bz1;
        }
    }

    struct Vector3 {
        public int x, y, z;
        public Vector3(int x, int y, int z) { this.x = x; this.y = y; this.z = z; }
    }

    class BoxItem {
        public int id;
        public int w, h, d;
        public long vol;
        public int s1, s2, s3;
    }

    class PlacedBox {
        public int id;
        public int x, y, z;
    }

    static Vector3[] _orients = new Vector3[6];
    
    static int FillOrientations(int w, int h, int d) {
        int count = 0;
        void Add(int x, int y, int z) {
            for (int i = 0; i < count; i++) {
                if (_orients[i].x == x && _orients[i].y == y && _orients[i].z == z) return;
            }
            _orients[count++] = new Vector3(x, y, z);
        }
        Add(w, h, d);
        Add(w, d, h);
        Add(h, w, d);
        Add(h, d, w);
        Add(d, w, h);
        Add(d, h, w);
        return count;
    }

    static int CompareEMS(EMS a, EMS b) {
        long posA = (long)a.x1 + a.y1 + a.z1;
        long posB = (long)b.x1 + b.y1 + b.z1;
        if (posA != posB) return posA.CompareTo(posB);
        long sizeA = (long)a.w + a.h + a.d;
        long sizeB = (long)b.w + b.h + b.d;
        return sizeA.CompareTo(sizeB);
    }
    
    static void Main() {
        Stopwatch sw = Stopwatch.StartNew();
        
        Scanner sc = new Scanner(Console.OpenStandardInput());
        int N = sc.NextInt();
        if (N <= 0) return;
        int W = sc.NextInt();
        int H = sc.NextInt();
        int D = sc.NextInt();

        BoxItem[] boxes = new BoxItem[N];
        for (int i = 0; i < N; i++) {
            int w = sc.NextInt();
            int h = sc.NextInt();
            int d = sc.NextInt();
            int s1 = w, s2 = h, s3 = d;
            if (s1 > s2) { int t=s1; s1=s2; s2=t; }
            if (s2 > s3) { int t=s2; s2=s3; s3=t; }
            if (s1 > s2) { int t=s1; s1=s2; s2=t; }
            
            boxes[i] = new BoxItem {
                id = i,
                w = w, h = h, d = d,
                vol = (long)w * h * d,
                s1 = s1, s2 = s2, s3 = s3
            };
        }

        Array.Sort(boxes, (a, b) => {
            int c = a.vol.CompareTo(b.vol);
            if (c != 0) return c;
            return a.s3.CompareTo(b.s3);
        });

        long[] min_vol = new long[N];
        int[] min_s1 = new int[N];
        int[] min_s2 = new int[N];
        int[] min_s3 = new int[N];

        min_vol[N-1] = boxes[N-1].vol;
        min_s1[N-1] = boxes[N-1].s1;
        min_s2[N-1] = boxes[N-1].s2;
        min_s3[N-1] = boxes[N-1].s3;

        for (int i = N - 2; i >= 0; i--) {
            min_vol[i] = Math.Min(boxes[i].vol, min_vol[i+1]);
            min_s1[i] = Math.Min(boxes[i].s1, min_s1[i+1]);
            min_s2[i] = Math.Min(boxes[i].s2, min_s2[i+1]);
            min_s3[i] = Math.Min(boxes[i].s3, min_s3[i+1]);
        }

        List<EMS> EMS_list = new List<EMS>(20000);
        EMS_list.Add(new EMS(0, 0, 0, W, H, D));

        List<PlacedBox> placedBoxes = new List<PlacedBox>(N);

        List<EMS> E_disjoint = new List<EMS>(20000);
        List<EMS> N_prime = new List<EMS>(1000);

        for (int i = 0; i < N; i++) {
            if (sw.ElapsedMilliseconds > 28000) {
                break;
            }

            if (EMS_list.Count > 15000) {
                EMS_list.Sort((a, b) => b.Volume.CompareTo(a.Volume));
                EMS_list.RemoveRange(15000, EMS_list.Count - 15000);
            }
            EMS_list.Sort(CompareEMS);

            bool placed = false;
            EMS best_e = default(EMS);
            Vector3 best_orient = new Vector3();

            for (int j = 0; j < EMS_list.Count; j++) {
                var e = EMS_list[j];
                
                int ew = e.w, eh = e.h, ed = e.d;
                int e1 = ew, e2 = eh, e3 = ed;
                if (e1 > e2) { int t=e1; e1=e2; e2=t; }
                if (e2 > e3) { int t=e2; e2=e3; e3=t; }
                if (e1 > e2) { int t=e1; e1=e2; e2=t; }
                
                if (e1 < boxes[i].s1 || e2 < boxes[i].s2 || e3 < boxes[i].s3) continue;

                int orientsCount = FillOrientations(boxes[i].w, boxes[i].h, boxes[i].d);
                bool orient_found = false;
                int best_res_max = -1;

                for (int o_idx = 0; o_idx < orientsCount; o_idx++) {
                    var o = _orients[o_idx];
                    if (o.x <= e.w && o.y <= e.h && o.z <= e.d) {
                        int res_w = e.w - o.x;
                        int res_h = e.h - o.y;
                        int res_d = e.d - o.z;
                        int max_res = Math.Max(res_w, Math.Max(res_h, res_d));
                        if (max_res > best_res_max) {
                            best_res_max = max_res;
                            best_orient = o;
                            orient_found = true;
                        }
                    }
                }
                
                if (orient_found) {
                    best_e = e;
                    placed = true;
                    break;
                }
            }

            if (placed) {
                placedBoxes.Add(new PlacedBox { id = boxes[i].id, x = best_e.x1, y = best_e.y1, z = best_e.z1 });
                
                if (i + 1 == N) break;

                E_disjoint.Clear();
                N_prime.Clear();

                int B_x1 = best_e.x1;
                int B_y1 = best_e.y1;
                int B_z1 = best_e.z1;
                int B_x2 = best_e.x1 + best_orient.x;
                int B_y2 = best_e.y1 + best_orient.y;
                int B_z2 = best_e.z1 + best_orient.z;

                for (int j = 0; j < EMS_list.Count; j++) {
                    var e = EMS_list[j];
                    if (e.Intersects(B_x1, B_y1, B_z1, B_x2, B_y2, B_z2)) {
                        if (e.x1 < B_x1) N_prime.Add(new EMS(e.x1, e.y1, e.z1, B_x1, e.y2, e.z2));
                        if (B_x2 < e.x2) N_prime.Add(new EMS(B_x2, e.y1, e.z1, e.x2, e.y2, e.z2));
                        if (e.y1 < B_y1) N_prime.Add(new EMS(e.x1, e.y1, e.z1, e.x2, B_y1, e.z2));
                        if (B_y2 < e.y2) N_prime.Add(new EMS(e.x1, B_y2, e.z1, e.x2, e.y2, e.z2));
                        if (e.z1 < B_z1) N_prime.Add(new EMS(e.x1, e.y1, e.z1, e.x2, e.y2, B_z1));
                        if (B_z2 < e.z2) N_prime.Add(new EMS(e.x1, e.y1, B_z2, e.x2, e.y2, e.z2));
                    } else {
                        E_disjoint.Add(e);
                    }
                }

                long next_min_vol = min_vol[i + 1];
                int next_min_s1 = min_s1[i + 1];
                int next_min_s2 = min_s2[i + 1];
                int next_min_s3 = min_s3[i + 1];

                int e_write = 0;
                for (int j = 0; j < E_disjoint.Count; j++) {
                    var e = E_disjoint[j];
                    if (e.Volume < next_min_vol) continue;
                    int e1 = e.w, e2 = e.h, e3 = e.d;
                    if (e1 > e2) { int t=e1; e1=e2; e2=t; }
                    if (e2 > e3) { int t=e2; e2=e3; e3=t; }
                    if (e1 > e2) { int t=e1; e1=e2; e2=t; }
                    if (e1 < next_min_s1 || e2 < next_min_s2 || e3 < next_min_s3) continue;
                    E_disjoint[e_write++] = e;
                }
                E_disjoint.RemoveRange(e_write, E_disjoint.Count - e_write);

                int n_write = 0;
                for (int j = 0; j < N_prime.Count; j++) {
                    var n = N_prime[j];
                    if (n.Volume < next_min_vol) continue;
                    int n1 = n.w, n2 = n.h, n3 = n.d;
                    if (n1 > n2) { int t=n1; n1=n2; n2=t; }
                    if (n2 > n3) { int t=n2; n2=n3; n3=t; }
                    if (n1 > n2) { int t=n1; n1=n2; n2=t; }
                    if (n1 < next_min_s1 || n2 < next_min_s2 || n3 < next_min_s3) continue;
                    N_prime[n_write++] = n;
                }
                N_prime.RemoveRange(n_write, N_prime.Count - n_write);

                EMS_list.Clear();
                EMS_list.AddRange(E_disjoint);

                for (int j = 0; j < N_prime.Count; j++) {
                    var n = N_prime[j];
                    bool contained = false;
                    
                    for (int k = 0; k < E_disjoint.Count; k++) {
                        if (E_disjoint[k].Contains(n)) {
                            contained = true;
                            break;
                        }
                    }
                    if (contained) continue;
                    
                    for (int k = 0; k < N_prime.Count; k++) {
                        if (j == k) continue;
                        var other = N_prime[k];
                        if (other.Contains(n)) {
                            if (other.x1 == n.x1 && other.y1 == n.y1 && other.z1 == n.z1 &&
                                other.x2 == n.x2 && other.y2 == n.y2 && other.z2 == n.z2) {
                                if (k < j) { contained = true; break; }
                            } else {
                                contained = true; break;
                            }
                        }
                    }
                    
                    if (!contained) {
                        EMS_list.Add(n);
                    }
                }
            }
        }

        Console.WriteLine(placedBoxes.Count);
        foreach (var pb in placedBoxes) {
            Console.WriteLine($"{pb.id} {pb.x} {pb.y} {pb.z}");
        }
    }
}
