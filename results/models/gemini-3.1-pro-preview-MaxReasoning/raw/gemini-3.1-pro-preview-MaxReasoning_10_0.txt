#include <iostream>
#include <vector>
#include <algorithm>
#include <chrono>
#include <cmath>

using namespace std;

struct Rect {
    int w, h, id;
};

struct SkylineNode {
    int x, y, w;
};

struct Board {
    vector<SkylineNode> skyline;
    long long free_area;
    int min_y;
    int active_idx;

    Board(int W, int H, int idx) {
        skyline.push_back({0, 0, W});
        free_area = (long long)W * H;
        min_y = 0;
        active_idx = idx;
    }
};

struct Placement {
    int board_idx;
    int x, y;
    int score1, score2;
    int w, h;
    int node_idx;
    bool rotated;
};

void addSkylineLevel(Board& board, int node_idx, int x, int y, int w, int h) {
    SkylineNode newNode;
    newNode.x = x;
    newNode.y = y + h;
    newNode.w = w;
    
    board.skyline.insert(board.skyline.begin() + node_idx, newNode);
    
    for (int i = node_idx + 1; i < (int)board.skyline.size(); ) {
        int prev_x = board.skyline[i-1].x;
        int prev_w = board.skyline[i-1].w;
        
        if (board.skyline[i].x < prev_x + prev_w) {
            int shrink = (prev_x + prev_w) - board.skyline[i].x;
            if (board.skyline[i].w <= shrink) {
                board.skyline.erase(board.skyline.begin() + i);
            } else {
                board.skyline[i].x += shrink;
                board.skyline[i].w -= shrink;
                break;
            }
        } else {
            break;
        }
    }
    
    for (int i = 0; i < (int)board.skyline.size() - 1; ) {
        if (board.skyline[i].y == board.skyline[i+1].y) {
            board.skyline[i].w += board.skyline[i+1].w;
            board.skyline.erase(board.skyline.begin() + i + 1);
        } else {
            i++;
        }
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int N, board_width, board_height;
    if (!(cin >> N >> board_width >> board_height)) return 0;

    vector<Rect> rects(N);
    long long min_rect_area = 2e18;
    int min_rect_dim = 2e9;

    for (int i = 0; i < N; ++i) {
        cin >> rects[i].w >> rects[i].h;
        rects[i].id = i;
        min_rect_area = min(min_rect_area, (long long)rects[i].w * rects[i].h);
        min_rect_dim = min(min_rect_dim, min(rects[i].w, rects[i].h));
    }

    sort(rects.begin(), rects.end(), [](const Rect& a, const Rect& b) {
        int max_a = max(a.w, a.h);
        int max_b = max(b.w, b.h);
        if (max_a != max_b) return max_a > max_b;
        int min_a = min(a.w, a.h);
        int min_b = min(b.w, b.h);
        if (min_a != min_b) return min_a > min_b;
        long long area_a = (long long)a.w * a.h;
        long long area_b = (long long)b.w * b.h;
        if (area_a != area_b) return area_a > area_b;
        return a.id < b.id;
    });

    vector<Board> boards;
    boards.reserve(N);
    vector<int> active_boards;
    active_boards.reserve(N);
    
    struct Result {
        int board_index;
        int x, y;
        int rotated;
    };
    vector<Result> results(N);

    auto start_time = chrono::steady_clock::now();
    double time_limit = 28.0;

    for (int rect_idx = 0; rect_idx < N; ++rect_idx) {
        auto now = chrono::steady_clock::now();
        double elapsed = chrono::duration<double>(now - start_time).count();
        double remaining_time = max(0.1, time_limit - elapsed);
        int remaining_rects = N - rect_idx;
        double time_per_rect = remaining_time / remaining_rects;
        int max_boards = max(50, (int)(time_per_rect / 200e-9));

        const auto& rect = rects[rect_idx];
        long long area = (long long)rect.w * rect.h;
        int min_dim = min(rect.w, rect.h);

        Placement best;
        best.board_idx = -1;
        best.score1 = 2e9;
        best.score2 = 2e9;
        best.x = 2e9;

        int boards_checked = 0;

        auto check_placement = [&](int b, int w, int h, bool rotated) {
            auto& board = boards[b];
            for (int k = 0; k < (int)board.skyline.size(); ++k) {
                int x = board.skyline[k].x;
                if (x + w > board_width) continue;
                
                int y = 0;
                int width_left = w;
                int j = k;
                while (width_left > 0 && j < (int)board.skyline.size()) {
                    y = max(y, board.skyline[j].y);
                    width_left -= board.skyline[j].w;
                    j++;
                }
                if (y + h > board_height) continue;
                
                int score1 = y + h;
                int score2 = abs(board.skyline[k].w - w);
                
                if (score1 < best.score1 || 
                   (score1 == best.score1 && score2 < best.score2) ||
                   (score1 == best.score1 && score2 == best.score2 && x < best.x)) {
                    best.board_idx = b;
                    best.x = x;
                    best.y = y;
                    best.score1 = score1;
                    best.score2 = score2;
                    best.w = w;
                    best.h = h;
                    best.node_idx = k;
                    best.rotated = rotated;
                }
            }
        };

        for (int i = (int)active_boards.size() - 1; i >= 0; --i) {
            int b = active_boards[i];
            if (boards[b].free_area < area) continue;
            if (board_height - boards[b].min_y < min_dim) continue;

            check_placement(b, rect.w, rect.h, false);
            if (rect.w != rect.h) {
                check_placement(b, rect.h, rect.w, true);
            }

            boards_checked++;
            if (boards_checked >= max_boards) break;
        }

        if (best.board_idx == -1) {
            int new_b = boards.size();
            boards.emplace_back(board_width, board_height, active_boards.size());
            active_boards.push_back(new_b);
            
            check_placement(new_b, rect.w, rect.h, false);
            if (rect.w != rect.h) {
                check_placement(new_b, rect.h, rect.w, true);
            }
        }
        
        if (best.board_idx == -1) {
            continue; 
        }

        auto& board = boards[best.board_idx];
        addSkylineLevel(board, best.node_idx, best.x, best.y, best.w, best.h);
        board.free_area -= area;
        
        int current_min_y = board_height;
        for (auto& node : board.skyline) {
            current_min_y = min(current_min_y, node.y);
        }
        board.min_y = current_min_y;

        results[rect.id] = {best.board_idx, best.x, best.y, best.rotated ? 1 : 0};

        if (board.free_area < min_rect_area || board_height - board.min_y < min_rect_dim) {
            if (board.active_idx != -1) {
                int idx = board.active_idx;
                int last_board = active_boards.back();
                active_boards[idx] = last_board;
                boards[last_board].active_idx = idx;
                active_boards.pop_back();
                board.active_idx = -1;
            }
        }
    }

    cout << boards.size() << "\n";
    for (int i = 0; i < N; ++i) {
        cout << results[i].board_index << " " 
             << results[i].x << " " 
             << results[i].y << " " 
             << results[i].rotated << "\n";
    }

    return 0;
}
