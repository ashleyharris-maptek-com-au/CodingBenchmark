#include "board.hpp"

Board::Board() {
    for (auto& colorArray : pieceBitboards) {
        for (auto& bitboard : colorArray) {
            bitboard = 0;
        }
    }
}

void Board::setupDefault() {
    // Pawns
    pieceBitboards[WHITE][PAWN] = 0xFF00;
    pieceBitboards[BLACK][PAWN] = 0xFF000000000000;

    // Knights
    pieceBitboards[WHITE][KNIGHT] = 0x42;
    pieceBitboards[BLACK][KNIGHT] = 0x4200000000000000;

    // Bishops
    pieceBitboards[WHITE][BISHOP] = 0x24;
    pieceBitboards[BLACK][BISHOP] = 0x2400000000000000;

    // Rooks
    pieceBitboards[WHITE][ROOK] = 0x81;
    pieceBitboards[BLACK][ROOK] = 0x8100000000000000;

    // Queens
    pieceBitboards[WHITE][QUEEN] = 0x8;
    pieceBitboards[BLACK][QUEEN] = 0x800000000000000;

    // Kings
    pieceBitboards[WHITE][KING] = 0x10;
    pieceBitboards[BLACK][KING] = 0x1000000000000000;
}

Bitboard Board::getPieces(Color color, Piece piece) const {
    return pieceBitboards[color][piece];
}

Bitboard Board::getOccupancy(Color color) const {
    Bitboard occupancy = 0;
    for (int p = PAWN; p <= KING; ++p) {
        occupancy |= pieceBitboards[color][p];
    }
    return occupancy;
}

Bitboard Board::getAllOccupancy() const {
    return getOccupancy(WHITE) | getOccupancy(BLACK);
}
