const fs = require('fs');

interface Point {
    x: number;
    y: number;
}

interface Package extends Point {
    id: number;
}

interface Input {
    depot: [number, number];
    truckCount: number;
    packages: Package[];
}

function distance(p1: Point, p2: Point): number {
    return Math.sqrt(Math.pow(p1.x - p2.x, 2) + Math.pow(p1.y - p2.y, 2));
}

function solve() {
    const inputStr: string = fs.readFileSync(0, 'utf8');
    const data: Input = JSON.parse(inputStr);

    const depot: Point = { x: data.depot[0], y: data.depot[1] };
    const packages = data.packages;
    
    // Sort packages by polar angle relative to the depot for simple initial clustering
    const packagesWithAngle = packages.map(pkg => ({
        ...pkg,
        angle: Math.atan2(pkg.y - depot.y, pkg.x - depot.x)
    }));
    
    packagesWithAngle.sort((a, b) => a.angle - b.angle);
    
    const routes: number[][] = Array.from({ length: data.truckCount }, () => []);
    
    // Distribute packages into routes
    packagesWithAngle.forEach((pkg, index) => {
        routes[index % data.truckCount].push(pkg.id);
    });
    
    // Simple improvement: within each route, sort by distance to depot to roughly minimize path
    routes.forEach(route => {
        const routePackages = route.map(id => packages.find(p => p.id === id)!);
        routePackages.sort((a, b) => distance(depot, a) - distance(depot, b));
        
        // Update the route order
        for (let i = 0; i < route.length; i++) {
            route[i] = routePackages[i].id;
        }
    });
    
    process.stdout.write(JSON.stringify({ routes }));
}

solve();
