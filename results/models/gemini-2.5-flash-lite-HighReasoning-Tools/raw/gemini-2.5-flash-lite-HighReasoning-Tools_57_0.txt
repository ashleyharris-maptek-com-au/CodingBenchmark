const fs = require('fs');

function solve() {
  const input = fs.readFileSync(0, 'utf8');
  const data = JSON.parse(input);

  const { width, height, grid, ignitions, towns, cutBudget } = data;
  const map = grid.map(row => row.split(''));
  const dirs = [[0, 1], [0, -1], [1, 0], [-1, 0]];

  // BFS from ignitions to find shortest distance to each cell
  const dist = Array.from({ length: height }, () => Array(width).fill(Infinity));
  const queue = [];
  for (const [ix, iy] of ignitions) {
    dist[iy][ix] = 0;
    queue.push([ix, iy]);
  }

  let head = 0;
  while (head < queue.length) {
    const [cx, cy] = queue[head++];
    for (const [dx, dy] of dirs) {
      const nx = cx + dx, ny = cy + dy;
      if (nx >= 0 && nx < width && ny >= 0 && ny < height && map[ny][nx] === '.' && dist[ny][nx] === Infinity) {
        dist[ny][nx] = dist[cy][cx] + 1;
        queue.push([nx, ny]);
      }
    }
  }

  // Identify cells that are on the path between ignitions and high-value towns
  // A simple heuristic: cut cells near high-value towns that are at risk (low distance to fire)
  
  const potentialCuts = [];
  for (const town of towns) {
    // Find burnable cells near the town
    for (const [dx, dy] of dirs) {
      const nx = town.x + dx, ny = town.y + dy;
      if (nx >= 0 && nx < width && ny >= 0 && ny < height && map[ny][nx] === '.') {
        potentialCuts.push({ x: nx, y: ny, dist: dist[ny][nx], value: town.value });
      }
    }
  }

  // Sort potential cuts by value/distance ratio
  potentialCuts.sort((a, b) => (b.value / (b.dist + 1)) - (a.value / (a.dist + 1)));

  const firebreaks = [];
  let cutsUsed = 0;
  for (const cut of potentialCuts) {
    if (cutsUsed < cutBudget) {
      firebreaks.push([cut.x, cut.y]);
      cutsUsed++;
    } else {
      break;
    }
  }

  process.stdout.write(JSON.stringify({ firebreaks }));
}

solve();
