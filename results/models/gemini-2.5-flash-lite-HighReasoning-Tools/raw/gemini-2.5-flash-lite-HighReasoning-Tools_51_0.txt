import time

class PIDController:
    def __init__(self, kp, ki, kd, setpoint=0):
        self.kp = kp
        self.ki = ki
        self.kd = kd
        self.setpoint = setpoint
        self.integral = 0
        self.last_error = 0
        self.last_time = time.time()

    def update(self, current_value):
        now = time.time()
        dt = now - self.last_time
        if dt <= 0: dt = 1e-6
        
        error = self.setpoint - current_value
        self.integral += error * dt
        derivative = (error - self.last_error) / dt
        
        output = (self.kp * error) + (self.ki * self.integral) + (self.kd * derivative)
        
        self.last_error = error
        self.last_time = now
        return output

class Autopilot:
    def __init__(self):
        # Pitch (Elevator) and Roll (Aileron) controllers
        self.pitch_pid = PIDController(kp=0.5, ki=0.01, kd=0.1)
        self.roll_pid = PIDController(kp=0.5, ki=0.01, kd=0.1)

    def set_target(self, pitch, roll):
        self.pitch_pid.setpoint = pitch
        self.roll_pid.setpoint = roll

    def compute_commands(self, current_pitch, current_roll):
        elevator = self.pitch_pid.update(current_pitch)
        aileron = self.roll_pid.update(current_roll)
        return elevator, aileron
