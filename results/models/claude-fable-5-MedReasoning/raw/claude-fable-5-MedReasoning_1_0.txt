// Adaptive Travelling Salesman solver.
//
// Strategy by problem size N:
//   N <= 13          : exact Held-Karp dynamic programming
//   N <= 200,000     : grid-accelerated nearest-neighbour construction,
//                      then candidate-list 2-opt + Or-opt with don't-look bits
//   N <= 20,000,000  : Hilbert-curve construction, then the same local search
//                      (bounded reversal / shift lengths) until the time budget
//   larger           : parallel Hilbert-curve ordering only
//
// Reads:  N, then N lines of "x y".
// Writes: one line, N space-separated city indices starting with 0.

#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <cstdint>
#include <cmath>
#include <vector>
#include <algorithm>
#include <numeric>
#include <chrono>
#include <execution>

#ifdef _WIN32
#include <io.h>
#include <fcntl.h>
#endif

typedef long long ll;
typedef unsigned long long ull;

static std::chrono::steady_clock::time_point T0;
static double elapsedSec()
{
    return std::chrono::duration<double>(std::chrono::steady_clock::now() - T0).count();
}
static const double HARD_LIMIT = 280.0; // stay well under the 300 s cap

static ll N;
static std::vector<float> X, Y;

static inline double dist(ll a, ll b)
{
    double dx = (double)X[a] - (double)X[b];
    double dy = (double)Y[a] - (double)Y[b];
    return std::sqrt(dx * dx + dy * dy);
}

// ---------------------------------------------------------------- input ----

static std::vector<char> readAllStdin()
{
#ifdef _WIN32
    _setmode(_fileno(stdin), _O_BINARY);
#endif
    std::vector<char> buf;
    const size_t CH = size_t(1) << 24;
    size_t sz = 0;
    for (;;)
    {
        buf.resize(sz + CH);
        size_t got = fread(buf.data() + sz, 1, CH, stdin);
        sz += got;
        if (got < CH) break;
    }
    buf.resize(sz);
    return buf;
}

struct Parser
{
    const char* p;
    const char* e;
    void skipWs()
    {
        while (p < e && (*p == ' ' || *p == '\n' || *p == '\r' || *p == '\t' || *p == ','))
            ++p;
    }
    ll readInt()
    {
        skipWs();
        ll sign = 1, v = 0;
        if (p < e && (*p == '-' || *p == '+')) { if (*p == '-') sign = -1; ++p; }
        while (p < e && *p >= '0' && *p <= '9') { v = v * 10 + (*p - '0'); ++p; }
        return v * sign;
    }
    double readFloat()
    {
        skipWs();
        double sign = 1.0;
        if (p < e && (*p == '-' || *p == '+')) { if (*p == '-') sign = -1.0; ++p; }
        double v = 0.0;
        while (p < e && *p >= '0' && *p <= '9') { v = v * 10.0 + (*p - '0'); ++p; }
        if (p < e && *p == '.')
        {
            ++p;
            double f = 0.0, sc = 1.0;
            while (p < e && *p >= '0' && *p <= '9') { f = f * 10.0 + (*p - '0'); sc *= 10.0; ++p; }
            v += f / sc;
        }
        if (p < e && (*p == 'e' || *p == 'E'))
        {
            ++p;
            int es = 1, ev = 0;
            if (p < e && (*p == '-' || *p == '+')) { if (*p == '-') es = -1; ++p; }
            while (p < e && *p >= '0' && *p <= '9') { ev = ev * 10 + (*p - '0'); ++p; }
            v *= std::pow(10.0, (double)(es * ev));
        }
        return sign * v;
    }
};

// --------------------------------------------------------------- output ----

static void writeTourRotated(const int* tour, ll n)
{
#ifdef _WIN32
    _setmode(_fileno(stdout), _O_BINARY);
#endif
    ll p0 = 0;
    for (ll i = 0; i < n; i++)
        if (tour[i] == 0) { p0 = i; break; }

    std::vector<char> out;
    out.reserve(size_t(1) << 22);
    char tmp[16];
    for (ll t = 0; t < n; t++)
    {
        unsigned v = (unsigned)tour[(p0 + t) % n];
        int len = 0;
        do { tmp[len++] = char('0' + v % 10); v /= 10; } while (v);
        while (len) out.push_back(tmp[--len]);
        out.push_back(t + 1 == n ? '\n' : ' ');
        if (out.size() > (size_t(1) << 22) - 32)
        {
            fwrite(out.data(), 1, out.size(), stdout);
            out.clear();
        }
    }
    fwrite(out.data(), 1, out.size(), stdout);
    fflush(stdout);
}

// Same, but the tour lives in the low 32 bits of packed Hilbert keys.
static void writeHilbertRotated(const std::vector<ull>& h)
{
#ifdef _WIN32
    _setmode(_fileno(stdout), _O_BINARY);
#endif
    ll n = (ll)h.size();
    ll p0 = 0;
    for (ll i = 0; i < n; i++)
        if ((uint32_t)h[i] == 0u) { p0 = i; break; }

    std::vector<char> out;
    out.reserve(size_t(1) << 24);
    char tmp[16];
    for (ll t = 0; t < n; t++)
    {
        unsigned v = (uint32_t)h[(p0 + t) % n];
        int len = 0;
        do { tmp[len++] = char('0' + v % 10); v /= 10; } while (v);
        while (len) out.push_back(tmp[--len]);
        out.push_back(t + 1 == n ? '\n' : ' ');
        if (out.size() > (size_t(1) << 24) - 32)
        {
            fwrite(out.data(), 1, out.size(), stdout);
            out.clear();
        }
    }
    fwrite(out.data(), 1, out.size(), stdout);
    fflush(stdout);
}

// ------------------------------------------------------------ Held-Karp ----

static void solveHeldKarp()
{
    int n = (int)N;
    int full = (1 << n) - 1;
    std::vector<float> dp((size_t)(full + 1) * n, 1e30f);
    std::vector<int8_t> par((size_t)(full + 1) * n, -1);
    for (int j = 1; j < n; j++)
        dp[(size_t)((1 | (1 << j))) * n + j] = (float)dist(0, j);
    for (int mask = 1; mask <= full; mask++)
    {
        if (!(mask & 1)) continue;
        for (int j = 1; j < n; j++)
        {
            if (!(mask & (1 << j))) continue;
            float cur = dp[(size_t)mask * n + j];
            if (cur >= 1e29f) continue;
            for (int k = 1; k < n; k++)
            {
                if (mask & (1 << k)) continue;
                int nm = mask | (1 << k);
                float nd = cur + (float)dist(j, k);
                if (nd < dp[(size_t)nm * n + k])
                {
                    dp[(size_t)nm * n + k] = nd;
                    par[(size_t)nm * n + k] = (int8_t)j;
                }
            }
        }
    }
    int bestj = 1;
    float best = 1e30f;
    for (int j = 1; j < n; j++)
    {
        float v = dp[(size_t)full * n + j] + (float)dist(j, 0);
        if (v < best) { best = v; bestj = j; }
    }
    std::vector<int> order;
    int mask = full, j = bestj;
    while (j > 0)
    {
        order.push_back(j);
        int pj = par[(size_t)mask * n + j];
        mask ^= (1 << j);
        j = pj;
    }
    order.push_back(0);
    std::reverse(order.begin(), order.end());
    writeTourRotated(order.data(), n);
}

// ----------------------------------------------------------------- grid ----

struct Grid
{
    int Gx = 1, Gy = 1;
    float minx = 0, miny = 0;
    float sx = 1, sy = 1, invsx = 1, invsy = 1;
    std::vector<int> cellStart; // Gx*Gy + 1
    std::vector<int> items;     // point ids grouped by cell

    void build(ll n, double targetPerCell)
    {
        float mnx = X[0], mxx = X[0], mny = Y[0], mxy = Y[0];
        for (ll i = 1; i < n; i++)
        {
            mnx = std::min(mnx, X[i]); mxx = std::max(mxx, X[i]);
            mny = std::min(mny, Y[i]); mxy = std::max(mxy, Y[i]);
        }
        minx = mnx; miny = mny;
        int G = (int)std::ceil(std::sqrt((double)n / targetPerCell));
        if (G < 1) G = 1;
        Gx = Gy = G;
        float rx = mxx - mnx, ry = mxy - mny;
        sx = (rx > 0) ? rx / Gx : 1.0f;
        sy = (ry > 0) ? ry / Gy : 1.0f;
        invsx = 1.0f / sx;
        invsy = 1.0f / sy;

        size_t cells = (size_t)Gx * Gy;
        cellStart.assign(cells + 1, 0);
        std::vector<int> cid(n);
        for (ll i = 0; i < n; i++)
        {
            int cx = (int)((X[i] - minx) * invsx); if (cx >= Gx) cx = Gx - 1; if (cx < 0) cx = 0;
            int cy = (int)((Y[i] - miny) * invsy); if (cy >= Gy) cy = Gy - 1; if (cy < 0) cy = 0;
            cid[i] = cy * Gx + cx;
            cellStart[cid[i] + 1]++;
        }
        for (size_t c = 0; c < cells; c++) cellStart[c + 1] += cellStart[c];
        items.resize(n);
        std::vector<int> fill(cellStart.begin(), cellStart.end() - 1);
        for (ll i = 0; i < n; i++) items[fill[cid[i]]++] = (int)i;
    }
    inline int cellXOf(ll i) const
    {
        int cx = (int)((X[i] - minx) * invsx);
        if (cx >= Gx) cx = Gx - 1; if (cx < 0) cx = 0;
        return cx;
    }
    inline int cellYOf(ll i) const
    {
        int cy = (int)((Y[i] - miny) * invsy);
        if (cy >= Gy) cy = Gy - 1; if (cy < 0) cy = 0;
        return cy;
    }
};

// ------------------------------------------------- nearest-neighbour NN ----

static std::vector<int> nearestNeighbourTour(const Grid& g)
{
    ll n = N;
    std::vector<int> items = g.items;
    size_t cells = (size_t)g.Gx * g.Gy;
    std::vector<int> cnt(cells);
    for (size_t c = 0; c < cells; c++) cnt[c] = g.cellStart[c + 1] - g.cellStart[c];
    std::vector<int> slotAbs(n), cellOfPt(n);
    for (size_t c = 0; c < cells; c++)
        for (int s = g.cellStart[c]; s < g.cellStart[c + 1]; s++)
        {
            slotAbs[items[s]] = s;
            cellOfPt[items[s]] = (int)c;
        }

    auto removePt = [&](int pnt)
    {
        int c = cellOfPt[pnt];
        int ip = slotAbs[pnt];
        int lastAbs = g.cellStart[c] + cnt[c] - 1;
        int q = items[lastAbs];
        items[ip] = q; slotAbs[q] = ip;
        items[lastAbs] = pnt; slotAbs[pnt] = lastAbs;
        cnt[c]--;
    };

    float minDim = std::min(g.sx, g.sy);
    int maxRing = std::max(g.Gx, g.Gy);

    std::vector<int> tour;
    tour.reserve(n);
    int cur = 0;
    removePt(cur);
    tour.push_back(cur);

    for (ll step = 1; step < n; step++)
    {
        double px = X[cur], py = Y[cur];
        int cx = g.cellXOf(cur), cy = g.cellYOf(cur);
        int best = -1;
        double bestd = 1e300;
        for (int r = 0; r <= maxRing; r++)
        {
            if (best >= 0 && (double)(r - 1) * minDim > bestd) break;
            int x0 = cx - r, x1 = cx + r, y0 = cy - r, y1 = cy + r;
            for (int yy = y0; yy <= y1; yy++)
            {
                if (yy < 0 || yy >= g.Gy) continue;
                bool edgeRow = (yy == y0 || yy == y1);
                int stepx = edgeRow ? 1 : (x1 - x0 == 0 ? 1 : x1 - x0);
                for (int xx = x0; xx <= x1; xx += stepx)
                {
                    if (xx < 0 || xx >= g.Gx) continue;
                    size_t c = (size_t)yy * g.Gx + xx;
                    int k = cnt[c];
                    if (!k) continue;
                    const int* it = items.data() + g.cellStart[c];
                    for (int s = 0; s < k; s++)
                    {
                        int q = it[s];
                        double dx = X[q] - px, dy = Y[q] - py;
                        double d2 = dx * dx + dy * dy;
                        if (d2 < bestd * bestd)
                        {
                            bestd = std::sqrt(d2);
                            best = q;
                        }
                    }
                }
            }
        }
        cur = best;
        removePt(cur);
        tour.push_back(cur);
    }
    return tour;
}

// ------------------------------------------------------ k-nearest lists ----

static int KNN;
static std::vector<int> knnList; // N * KNN, -1 padded, sorted by distance

static void buildKnn(const Grid& g, int K)
{
    KNN = K;
    ll n = N;
    knnList.assign((size_t)n * K, -1);
    int maxRing = std::max(g.Gx, g.Gy);

    ull* dummy = nullptr; (void)dummy;
    std::vector<uint32_t> idx(n);
    for (ll i = 0; i < n; i++) idx[i] = (uint32_t)i;

    std::for_each(std::execution::par, idx.begin(), idx.end(), [&](uint32_t i)
    {
        double px = X[i], py = Y[i];
        int cx = g.cellXOf(i), cy = g.cellYOf(i);
        // gather candidates from expanding rings
        int cand[256];
        int nc = 0;
        int rEnough = -1;
        for (int r = 0; r <= maxRing; r++)
        {
            if (rEnough >= 0 && r > rEnough + 1) break;
            int x0 = cx - r, x1 = cx + r, y0 = cy - r, y1 = cy + r;
            for (int yy = y0; yy <= y1; yy++)
            {
                if (yy < 0 || yy >= g.Gy) continue;
                bool edgeRow = (yy == y0 || yy == y1);
                int stepx = edgeRow ? 1 : (x1 - x0 == 0 ? 1 : x1 - x0);
                for (int xx = x0; xx <= x1; xx += stepx)
                {
                    if (xx < 0 || xx >= g.Gx) continue;
                    size_t c = (size_t)yy * g.Gx + xx;
                    for (int s = g.cellStart[c]; s < g.cellStart[c + 1]; s++)
                    {
                        int q = g.items[s];
                        if (q == (int)i) continue;
                        if (nc < 256) cand[nc++] = q;
                    }
                }
            }
            if (rEnough < 0 && nc >= K) rEnough = r;
            if (nc >= 200) break;
        }
        // pick K nearest among candidates
        std::pair<float, int> bestArr[256];
        for (int t = 0; t < nc; t++)
        {
            double dx = X[cand[t]] - px, dy = Y[cand[t]] - py;
            bestArr[t] = { (float)(dx * dx + dy * dy), cand[t] };
        }
        int kk = std::min(nc, K);
        std::partial_sort(bestArr, bestArr + kk, bestArr + nc);
        int* dst = knnList.data() + (size_t)i * K;
        for (int t = 0; t < kk; t++) dst[t] = bestArr[t].second;
    });
}

// -------------------------------------------------- 2-opt + Or-opt loop ----

static void optimizeTour(std::vector<int>& tour, double deadline)
{
    const ll n = N;
    std::vector<int> pos(n);
    for (ll i = 0; i < n; i++) pos[tour[i]] = (int)i;

    std::vector<uint8_t> dontlook(n, 0);
    const ll revCap = (n <= 300000) ? n : 30000;
    const ll shiftCap = (n <= 300000) ? n : 30000;

    int* tb = tour.data();
    int* pb = pos.data();

    auto succ = [&](int a) { ll p = pb[a] + 1; if (p == n) p = 0; return tb[p]; };
    auto pred = [&](int a) { ll p = pb[a]; p = (p == 0 ? n - 1 : p - 1); return tb[p]; };

    auto reverseSegment = [&](ll i, ll j)
    {
        ll len = ((j - i + n) % n) + 1;
        if (len * 2 > n)
        {
            ll ni = (j + 1) % n, nj = (i - 1 + n) % n;
            i = ni; j = nj;
            len = n - len;
        }
        for (ll k = 0; k < len / 2; k++)
        {
            ll ia = i + k; if (ia >= n) ia -= n;
            ll ib = j - k; if (ib < 0) ib += n;
            std::swap(tb[ia], tb[ib]);
            pb[tb[ia]] = (int)ia;
            pb[tb[ib]] = (int)ib;
        }
    };
    auto segLenOk = [&](ll i, ll j) -> bool
    {
        ll len = ((j - i + n) % n) + 1;
        ll eff = std::min(len, n - len);
        return eff <= revCap;
    };

    auto tryCity = [&](int a) -> bool
    {
        const int* cand = knnList.data() + (size_t)a * KNN;

        // 2-opt on successor edge (a,b): replace (a,b),(c,d1) with (a,c),(b,d1)
        {
            int b = succ(a);
            double dab = dist(a, b);
            for (int t = 0; t < KNN; t++)
            {
                int c = cand[t];
                if (c < 0) break;
                if (c == b || c == a) continue;
                double dac = dist(a, c);
                if (dac >= dab) break;
                int d1 = succ(c);
                if (d1 == a) continue;
                double delta = dab + dist(c, d1) - dac - dist(b, d1);
                if (delta > 1e-9 + 1e-12 * dab)
                {
                    ll i = pb[b], j = pb[c];
                    if (!segLenOk(i, j)) continue;
                    reverseSegment(i, j);
                    dontlook[a] = dontlook[b] = dontlook[c] = dontlook[d1] = 0;
                    return true;
                }
            }
        }
        // 2-opt on predecessor edge (p,a): replace (p,a),(c0,c) with (p,c0),(a,c)
        {
            int p = pred(a);
            double dpa = dist(p, a);
            for (int t = 0; t < KNN; t++)
            {
                int c = cand[t];
                if (c < 0) break;
                if (c == p || c == a) continue;
                double dac = dist(a, c);
                if (dac >= dpa) break;
                int c0 = pred(c);
                if (c0 == a) continue;
                double delta = dpa + dist(c0, c) - dac - dist(p, c0);
                if (delta > 1e-9 + 1e-12 * dpa)
                {
                    ll i = pb[a], j = pb[c0];
                    if (!segLenOk(i, j)) continue;
                    reverseSegment(i, j);
                    dontlook[a] = dontlook[p] = dontlook[c] = dontlook[c0] = 0;
                    return true;
                }
            }
        }
        // Or-opt: move segment [a .. e] (length 1..3) to just after candidate c
        if (n >= 8)
        {
            ll pa = pb[a];
            for (int L = 1; L <= 3; L++)
            {
                ll pe = pa + L - 1;
                if (pe >= n) break; // avoid wrap-around segments
                int e = tb[pe];
                int p = tb[pa == 0 ? n - 1 : pa - 1];
                int nx = tb[(pe + 1) % n];
                if (p == e || nx == a) break;
                double removeGain = dist(p, a) + dist(e, nx) - dist(p, nx);
                if (removeGain <= 1e-9) continue;
                for (int t = 0; t < KNN; t++)
                {
                    int c = cand[t];
                    if (c < 0) break;
                    ll pc = pb[c];
                    if (pc >= pa && pc <= pe) continue; // inside segment
                    if (c == p) continue;
                    double dca = dist(c, a);
                    if (dca >= removeGain) break;
                    int cn = succ(c);
                    if (cn == a) continue;
                    double delta = removeGain - (dca + dist(e, cn) - dist(c, cn));
                    if (delta > 1e-9 + 1e-12 * removeGain)
                    {
                        if (pc > pe)
                        {
                            if (pc - pa > shiftCap) continue;
                            std::rotate(tb + pa, tb + pe + 1, tb + pc + 1);
                            for (ll k = pa; k <= pc; k++) pb[tb[k]] = (int)k;
                        }
                        else // pc < pa - 1
                        {
                            if (pe - pc > shiftCap) continue;
                            std::rotate(tb + pc + 1, tb + pa, tb + pe + 1);
                            for (ll k = pc + 1; k <= pe; k++) pb[tb[k]] = (int)k;
                        }
                        dontlook[a] = dontlook[e] = dontlook[p] = 0;
                        dontlook[nx] = dontlook[c] = dontlook[cn] = 0;
                        return true;
                    }
                }
            }
        }
        return false;
    };

    ll checkCtr = 0;
    bool anyImp = true;
    while (anyImp)
    {
        anyImp = false;
        for (ll idx = 0; idx < n; idx++)
        {
            int a = tb[idx];
            if (dontlook[a]) continue;
            if (((++checkCtr) & 511) == 0 && elapsedSec() > deadline) return;
            bool improved = false;
            while (tryCity(a))
            {
                improved = true;
                if (((++checkCtr) & 511) == 0 && elapsedSec() > deadline) return;
            }
            if (improved) anyImp = true;
            else dontlook[a] = 1;
        }
        if (elapsedSec() > deadline) return;
    }
}

// -------------------------------------------------------- Hilbert order ----

static inline uint32_t hilbertD(uint32_t x, uint32_t y)
{
    // 16-bit order Hilbert curve -> 32-bit index
    uint32_t d = 0;
    for (uint32_t s = 1u << 15; s > 0; s >>= 1)
    {
        uint32_t rx = (x & s) ? 1u : 0u;
        uint32_t ry = (y & s) ? 1u : 0u;
        d += s * s * ((3u * rx) ^ ry);
        if (ry == 0)
        {
            if (rx == 1)
            {
                x = s - 1 - x;
                y = s - 1 - y;
            }
            uint32_t t = x; x = y; y = t;
        }
    }
    return d;
}

// Returns packed (hilbertKey << 32) | index, sorted by key.
static std::vector<ull> hilbertOrder()
{
    ll n = N;
    float mnx = X[0], mxx = X[0], mny = Y[0], mxy = Y[0];
    for (ll i = 1; i < n; i++)
    {
        mnx = std::min(mnx, X[i]); mxx = std::max(mxx, X[i]);
        mny = std::min(mny, Y[i]); mxy = std::max(mxy, Y[i]);
    }
    double sxc = (mxx > mnx) ? 65535.0 / ((double)mxx - mnx) : 0.0;
    double syc = (mxy > mny) ? 65535.0 / ((double)mxy - mny) : 0.0;

    std::vector<ull> h(n);
    ull* base = h.data();
    std::for_each(std::execution::par_unseq, h.begin(), h.end(), [&](ull& v)
    {
        size_t i = (size_t)(&v - base);
        uint32_t xi = (uint32_t)(((double)X[i] - mnx) * sxc);
        uint32_t yi = (uint32_t)(((double)Y[i] - mny) * syc);
        if (xi > 65535u) xi = 65535u;
        if (yi > 65535u) yi = 65535u;
        v = ((ull)hilbertD(xi, yi) << 32) | (uint32_t)i;
    });
    std::sort(std::execution::par_unseq, h.begin(), h.end());
    return h;
}

// ----------------------------------------------------------------- main ----

int main()
{
    T0 = std::chrono::steady_clock::now();

    std::vector<char> data = readAllStdin();
    Parser ps{ data.data(), data.data() + data.size() };
    N = ps.readInt();
    if (N <= 0) return 0;

    X.resize(N);
    Y.resize(N);
    for (ll i = 0; i < N; i++)
    {
        X[i] = (float)ps.readFloat();
        Y[i] = (float)ps.readFloat();
    }
    data.clear();
    data.shrink_to_fit();

    if (N == 1) { fputs("0\n", stdout); return 0; }
    if (N <= 3)
    {
        std::vector<int> t(N);
        for (ll i = 0; i < N; i++) t[i] = (int)i;
        writeTourRotated(t.data(), N);
        return 0;
    }
    if (N <= 13)
    {
        solveHeldKarp();
        return 0;
    }

    const ll OPT_MAX = 20000000; // beyond this: Hilbert order only

    if (N > OPT_MAX)
    {
        std::vector<ull> h = hilbertOrder();
        writeHilbertRotated(h);
        return 0;
    }

    // --- construction ---
    Grid grid;
    grid.build(N, 2.0);

    std::vector<int> tour;
    if (N <= 200000)
    {
        tour = nearestNeighbourTour(grid);
    }
    else
    {
        std::vector<ull> h = hilbertOrder();
        tour.resize(N);
        for (ll i = 0; i < N; i++) tour[i] = (int)(uint32_t)h[i];
    }

    // --- local search ---
    int K = (N <= 2000) ? 16 : (N <= 100000) ? 12 : (N <= 2000000) ? 8 : 6;
    if (K > (int)N - 1) K = (int)N - 1;
    buildKnn(grid, K);

    double outputReserve = 3.0 + (double)N / 2.0e6;
    double deadline = HARD_LIMIT - outputReserve;
    optimizeTour(tour, deadline);

    writeTourRotated(tour.data(), N);
    return 0;
}
