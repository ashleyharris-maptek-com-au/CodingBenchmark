// Graph Layout: Edge Crossing Minimization
// Strategy:
//  - Split into connected components.
//  - Trees  -> layered tidy layout (guaranteed 0 crossings).
//  - Simple cycles -> circle (0 crossings).
//  - General graphs -> force-directed (Fruchterman-Reingold style) with
//    BFS-radial initialisation, exact O(n^2) repulsion for small components
//    and grid-bucketed local repulsion for large ones.
//  - Components are shelf-packed into the plane.
//  - Global time budget keeps everything under the 30s limit.

use std::io::{self, Read, Write, BufWriter};
use std::time::{Duration, Instant};

struct Rng(u64);
impl Rng {
    fn new(seed: u64) -> Self { Rng(seed | 1) }
    fn next(&mut self) -> u64 {
        let mut x = self.0;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.0 = x;
        x
    }
    fn f(&mut self) -> f64 {
        (self.next() >> 11) as f64 / (1u64 << 53) as f64
    }
}

fn read_input() -> (usize, Vec<(u32, u32)>) {
    let mut buf = Vec::new();
    io::stdin().read_to_end(&mut buf).unwrap();
    let mut nums: Vec<u64> = Vec::new();
    let mut cur: u64 = 0;
    let mut in_num = false;
    for &b in &buf {
        if b.is_ascii_digit() {
            cur = cur * 10 + (b - b'0') as u64;
            in_num = true;
        } else if in_num {
            nums.push(cur);
            cur = 0;
            in_num = false;
        }
    }
    if in_num { nums.push(cur); }
    if nums.len() < 2 { return (0, Vec::new()); }
    let n = nums[0] as usize;
    let m = nums[1] as usize;
    let mut edges = Vec::with_capacity(m);
    let mut i = 2;
    while i + 1 < nums.len() && edges.len() < m {
        let u = nums[i] as usize;
        let v = nums[i + 1] as usize;
        i += 2;
        if u == v || u >= n || v >= n { continue; } // drop self-loops / bad edges
        edges.push((u as u32, v as u32));
    }
    (n, edges)
}

// Build CSR adjacency with parallel edges removed.
fn build_csr(n: usize, edges: &[(u32, u32)]) -> (Vec<u32>, Vec<u32>) {
    let mut pairs: Vec<(u32, u32)> = Vec::with_capacity(edges.len() * 2);
    for &(u, v) in edges {
        pairs.push((u, v));
        pairs.push((v, u));
    }
    pairs.sort_unstable();
    pairs.dedup();
    let mut off = vec![0u32; n + 1];
    for &(u, _) in &pairs { off[u as usize + 1] += 1; }
    for i in 0..n { off[i + 1] += off[i]; }
    let adj: Vec<u32> = pairs.iter().map(|&(_, v)| v).collect();
    (off, adj)
}

fn neighbors<'a>(off: &[u32], adj: &'a [u32], u: u32) -> &'a [u32] {
    &adj[off[u as usize] as usize..off[u as usize + 1] as usize]
}

// Layered tidy tree layout: leaves get consecutive x slots, internal nodes
// sit at the mean of their children, y = -depth. Subtrees occupy disjoint
// x-intervals, so no two edges cross.
struct TreeScratch {
    par: Vec<u32>,   // parent, sentinel u32::MAX = unvisited
    depth: Vec<u32>,
    child_sum: Vec<f64>,
    child_cnt: Vec<u32>,
}

fn layout_tree(
    comp: &[u32],
    off: &[u32],
    adj: &[u32],
    px: &mut [f64],
    py: &mut [f64],
    sc: &mut TreeScratch,
) {
    let root = comp[0];
    // BFS order + parents
    let mut order: Vec<u32> = Vec::with_capacity(comp.len());
    sc.par[root as usize] = root; // mark visited; root is its own parent
    sc.depth[root as usize] = 0;
    order.push(root);
    let mut head = 0;
    while head < order.len() {
        let u = order[head];
        head += 1;
        for &v in neighbors(off, adj, u) {
            if sc.par[v as usize] == u32::MAX {
                sc.par[v as usize] = u;
                sc.depth[v as usize] = sc.depth[u as usize] + 1;
                order.push(v);
            }
        }
    }
    // Reverse BFS order = valid post-order for averaging children:
    // leaves take consecutive x slots in an order consistent with subtree
    // grouping (DFS), so do a DFS for leaf numbering.
    let mut next_leaf = 0.0f64;
    let mut stack: Vec<(u32, u32)> = vec![(root, off[root as usize])];
    while let Some(&mut (u, ref mut ci)) = stack.last_mut() {
        let end = off[u as usize + 1];
        let mut advanced = false;
        while *ci < end {
            let v = adj[*ci as usize];
            *ci += 1;
            if v != u && sc.par[v as usize] == u && sc.depth[v as usize] == sc.depth[u as usize] + 1 {
                stack.push((v, off[v as usize]));
                advanced = true;
                break;
            }
        }
        if advanced { continue; }
        stack.pop();
        let cnt = sc.child_cnt[u as usize];
        let xu = if cnt == 0 {
            let v = next_leaf;
            next_leaf += 1.0;
            v
        } else {
            sc.child_sum[u as usize] / cnt as f64
        };
        px[u as usize] = xu; // stash x temporarily
        if u != root {
            let p = sc.par[u as usize] as usize;
            sc.child_sum[p] += xu;
            sc.child_cnt[p] += 1;
        }
    }
    for &u in comp {
        let u = u as usize;
        px[u] *= 1.5;
        py[u] = -(sc.depth[u] as f64) * 1.5;
    }
    // reset scratch for touched nodes
    for &u in comp {
        let u = u as usize;
        sc.par[u] = u32::MAX;
        sc.depth[u] = 0;
        sc.child_sum[u] = 0.0;
        sc.child_cnt[u] = 0;
    }
}

fn layout_cycle(comp: &[u32], off: &[u32], adj: &[u32], px: &mut [f64], py: &mut [f64]) {
    let n = comp.len();
    let r = (n as f64) / (2.0 * std::f64::consts::PI) * 1.5 + 1.0;
    let mut prev = u32::MAX;
    let mut cur = comp[0];
    for i in 0..n {
        let a = 2.0 * std::f64::consts::PI * i as f64 / n as f64;
        px[cur as usize] = r * a.cos();
        py[cur as usize] = r * a.sin();
        let nb = neighbors(off, adj, cur);
        let next = if nb[0] != prev { nb[0] } else { nb[1] };
        prev = cur;
        cur = next;
    }
}

#[allow(clippy::too_many_arguments)]
fn layout_force(
    comp: &[u32],
    cedges: &[(u32, u32)],
    off: &[u32],
    adj: &[u32],
    px: &mut [f64],
    py: &mut [f64],
    deadline: Instant,
    rng: &mut Rng,
) {
    let n = comp.len();
    if n == 1 {
        px[comp[0] as usize] = 0.0;
        py[comp[0] as usize] = 0.0;
        return;
    }
    // --- BFS radial initialisation ---
    let nglob = px.len();
    let mut depth = vec![u32::MAX; nglob];
    let mut order: Vec<u32> = Vec::with_capacity(n);
    depth[comp[0] as usize] = 0;
    order.push(comp[0]);
    let mut head = 0;
    while head < order.len() {
        let u = order[head];
        head += 1;
        for &v in neighbors(off, adj, u) {
            if depth[v as usize] == u32::MAX {
                depth[v as usize] = depth[u as usize] + 1;
                order.push(v);
            }
        }
    }
    let maxd = order.iter().map(|&u| depth[u as usize]).max().unwrap_or(0);
    let mut layer_count = vec![0u32; maxd as usize + 1];
    for &u in &order { layer_count[depth[u as usize] as usize] += 1; }
    let mut layer_idx = vec![0u32; maxd as usize + 1];
    for &u in &order {
        let d = depth[u as usize] as usize;
        let cnt = layer_count[d] as f64;
        let idx = layer_idx[d] as f64;
        layer_idx[d] += 1;
        let a = 2.0 * std::f64::consts::PI * idx / cnt + d as f64 * 0.5;
        let r = d as f64 * 1.5 + 0.3 + rng.f() * 0.3;
        px[u as usize] = r * a.cos() + (rng.f() - 0.5) * 0.1;
        py[u as usize] = r * a.sin() + (rng.f() - 0.5) * 0.1;
    }
    // rescale so average density is ~1 node per unit^2 (keeps the spatial
    // grid's per-cell occupancy low and the layout near its final size)
    {
        let target_r = (n as f64).sqrt() * 0.65 + 1.0;
        let cur_r = maxd as f64 * 1.5 + 1.0;
        let s = target_r / cur_r;
        for &u in comp {
            px[u as usize] *= s;
            py[u as usize] *= s;
        }
    }

    let k: f64 = 1.0; // ideal edge length
    let iters: usize = if n <= 60 { 800 }
        else if n <= 300 { 500 }
        else if n <= 2000 { 350 }
        else if n <= 20_000 { 250 }
        else if n <= 100_000 { 120 }
        else { 60 };
    let t0 = (n as f64).sqrt() * 0.12 + 0.5;
    let tmin = 0.01;
    let cool = (tmin / t0).powf(1.0 / iters as f64);
    let mut t = t0;
    let full_pairs = n <= 1200;

    let mut dx = vec![0.0f64; nglob];
    let mut dy = vec![0.0f64; nglob];
    // grid buckets (reused)
    let rc = 2.0 * k;
    let mut cell_of: Vec<u32> = vec![0; if full_pairs { 0 } else { n }];
    let mut count: Vec<u32> = Vec::new();
    let mut fill: Vec<u32> = Vec::new();
    let mut bucket: Vec<u32> = vec![0; if full_pairs { 0 } else { n }];
    let max_dim = ((4.0 * n as f64).sqrt() as usize + 1).min(2048);
    let check_every = if n > 20_000 { 1 } else { 8 };
    for it in 0..iters {
        if it % check_every == 0 && Instant::now() > deadline { break; }
        for &u in comp { dx[u as usize] = 0.0; dy[u as usize] = 0.0; }

        // repulsion
        if full_pairs {
            for i in 0..n {
                let a = comp[i] as usize;
                let (ax, ay) = (px[a], py[a]);
                for j in (i + 1)..n {
                    let b = comp[j] as usize;
                    let mut ddx = ax - px[b];
                    let mut ddy = ay - py[b];
                    let mut d2 = ddx * ddx + ddy * ddy;
                    if d2 < 1e-6 {
                        ddx = rng.f() - 0.5;
                        ddy = rng.f() - 0.5;
                        d2 = ddx * ddx + ddy * ddy;
                    }
                    let f = k * k / d2;
                    dx[a] += ddx * f; dy[a] += ddy * f;
                    dx[b] -= ddx * f; dy[b] -= ddy * f;
                }
            }
        } else {
            // grid-bucketed local repulsion within radius rc
            let (mut minx, mut miny) = (f64::MAX, f64::MAX);
            let (mut maxx, mut maxy) = (f64::MIN, f64::MIN);
            for &u in comp {
                let u = u as usize;
                if px[u] < minx { minx = px[u]; }
                if py[u] < miny { miny = py[u]; }
                if px[u] > maxx { maxx = px[u]; }
                if py[u] > maxy { maxy = py[u]; }
            }
            let cw = rc.max((maxx - minx).max(maxy - miny) / max_dim as f64);
            let gw = (((maxx - minx) / cw) as usize + 1).max(1);
            let gh = (((maxy - miny) / cw) as usize + 1).max(1);
            let ncells = gw * gh;
            count.clear();
            count.resize(ncells + 1, 0);
            for (li, &u) in comp.iter().enumerate() {
                let u = u as usize;
                let cx = ((px[u] - minx) / cw) as usize;
                let cy = ((py[u] - miny) / cw) as usize;
                let c = (cy.min(gh - 1) * gw + cx.min(gw - 1)) as u32;
                cell_of[li] = c;
                count[c as usize + 1] += 1;
            }
            for c in 0..ncells { count[c + 1] += count[c]; }
            fill.clear();
            fill.extend_from_slice(&count);
            for (li, &c) in cell_of.iter().enumerate() {
                bucket[fill[c as usize] as usize] = li as u32;
                fill[c as usize] += 1;
            }
            let rc2 = rc * rc;
            for li in 0..n {
                let a = comp[li] as usize;
                let (ax, ay) = (px[a], py[a]);
                let c = cell_of[li] as usize;
                let cx = c % gw;
                let cy = c / gw;
                let x0 = cx.saturating_sub(1);
                let x1 = (cx + 1).min(gw - 1);
                let y0 = cy.saturating_sub(1);
                let y1 = (cy + 1).min(gh - 1);
                let mut fx = 0.0;
                let mut fy = 0.0;
                for gy in y0..=y1 {
                    for gx in x0..=x1 {
                        let cc = gy * gw + gx;
                        let s0 = count[cc];
                        let s1 = count[cc + 1];
                        let occ = (s1 - s0) as usize;
                        // cap work per cell: stride-sample crowded cells and
                        // scale the force to keep expected magnitude
                        const CAP: usize = 12;
                        let step = occ.div_ceil(CAP).max(1);
                        let phase = if step > 1 { (li + it) % step } else { 0 };
                        let scale = step as f64;
                        let mut bi = s0 as usize + phase;
                        while bi < s1 as usize {
                            let lj = bucket[bi] as usize;
                            bi += step;
                            if lj == li { continue; }
                            let b = comp[lj] as usize;
                            let mut ddx = ax - px[b];
                            let mut ddy = ay - py[b];
                            let mut d2 = ddx * ddx + ddy * ddy;
                            if d2 > rc2 { continue; }
                            if d2 < 1e-6 {
                                ddx = rng.f() - 0.5;
                                ddy = rng.f() - 0.5;
                                d2 = ddx * ddx + ddy * ddy;
                            }
                            let f = scale * k * k / d2;
                            fx += ddx * f;
                            fy += ddy * f;
                        }
                    }
                }
                dx[a] += fx;
                dy[a] += fy;
            }
        }

        // attraction along edges
        for &(u, v) in cedges {
            let (u, v) = (u as usize, v as usize);
            let ddx = px[u] - px[v];
            let ddy = py[u] - py[v];
            let d = (ddx * ddx + ddy * ddy).sqrt().max(1e-4);
            let f = d / k; // FR: d^2/k divided by d for unit vector
            dx[u] -= ddx * f;
            dy[u] -= ddy * f;
            dx[v] += ddx * f;
            dy[v] += ddy * f;
        }

        // mild gravity toward centroid keeps sparse parts from drifting
        let mut cx = 0.0;
        let mut cy = 0.0;
        for &u in comp { cx += px[u as usize]; cy += py[u as usize]; }
        cx /= n as f64;
        cy /= n as f64;
        let g = if n > 5000 { 0.002 } else { 0.03 };
        for &u in comp {
            let u = u as usize;
            dx[u] += (cx - px[u]) * g;
            dy[u] += (cy - py[u]) * g;
        }

        // apply displacement, capped by temperature
        for &u in comp {
            let u = u as usize;
            let d = (dx[u] * dx[u] + dy[u] * dy[u]).sqrt();
            if d > 1e-9 {
                let s = d.min(t) / d;
                px[u] += dx[u] * s;
                py[u] += dy[u] * s;
            }
        }
        t = (t * cool).max(tmin);
    }
}

fn main() {
    let start = Instant::now();
    let budget = Duration::from_millis(22_000);
    let (n, edges) = read_input();
    let stdout = io::stdout();
    let mut out = BufWriter::new(stdout.lock());
    if n == 0 { return; }

    let (off, adj) = build_csr(n, &edges);

    // connected components
    let mut comp_id = vec![u32::MAX; n];
    let mut comps: Vec<Vec<u32>> = Vec::new();
    for s in 0..n {
        if comp_id[s] != u32::MAX { continue; }
        let id = comps.len() as u32;
        let mut q: Vec<u32> = vec![s as u32];
        comp_id[s] = id;
        let mut head = 0;
        while head < q.len() {
            let u = q[head];
            head += 1;
            for &v in neighbors(&off, &adj, u) {
                if comp_id[v as usize] == u32::MAX {
                    comp_id[v as usize] = id;
                    q.push(v);
                }
            }
        }
        comps.push(q);
    }

    let mut px = vec![0.0f64; n];
    let mut py = vec![0.0f64; n];
    let mut rng = Rng::new(0x9E3779B97F4A7C15);
    let mut tree_scratch = TreeScratch {
        par: vec![u32::MAX; n],
        depth: vec![0; n],
        child_sum: vec![0.0; n],
        child_cnt: vec![0; n],
    };

    // per-component simple edge lists + classification, then layout
    // Time budget split proportional to component work.
    let mut works: Vec<f64> = Vec::with_capacity(comps.len());
    let mut comp_edges: Vec<Vec<(u32, u32)>> = Vec::with_capacity(comps.len());
    for comp in &comps {
        let mut ce = Vec::new();
        for &u in comp {
            for &v in neighbors(&off, &adj, u) {
                if v > u { ce.push((u, v)); }
            }
        }
        works.push((comp.len() + ce.len()) as f64);
        comp_edges.push(ce);
    }
    let total_work: f64 = works.iter().sum::<f64>().max(1.0);
    let mut consumed = 0.0f64;

    for (ci, comp) in comps.iter().enumerate() {
        let cn = comp.len();
        let ce = &comp_edges[ci];
        let cm = ce.len();
        if cn == 1 {
            // point at origin (packing will place it)
        } else if cm == cn - 1 {
            layout_tree(comp, &off, &adj, &mut px, &mut py, &mut tree_scratch);
        } else if cm == cn && comp.iter().all(|&u| neighbors(&off, &adj, u).len() == 2) {
            layout_cycle(comp, &off, &adj, &mut px, &mut py);
        } else {
            consumed += works[ci];
            let frac = consumed / total_work;
            let share = start + Duration::from_millis((budget.as_millis() as f64 * frac) as u64);
            layout_force(comp, ce, &off, &adj, &mut px, &mut py, share, &mut rng);
        }
    }

    // shelf-pack component bounding boxes
    if comps.len() > 1 {
        let pad = 3.0;
        let mut boxes: Vec<(usize, f64, f64, f64, f64)> = Vec::new(); // ci, minx, miny, w, h
        let mut total_area = 0.0;
        for (ci, comp) in comps.iter().enumerate() {
            let (mut mnx, mut mny) = (f64::MAX, f64::MAX);
            let (mut mxx, mut mxy) = (f64::MIN, f64::MIN);
            for &u in comp {
                let u = u as usize;
                mnx = mnx.min(px[u]); mny = mny.min(py[u]);
                mxx = mxx.max(px[u]); mxy = mxy.max(py[u]);
            }
            let w = (mxx - mnx).max(0.5) + pad;
            let h = (mxy - mny).max(0.5) + pad;
            total_area += w * h;
            boxes.push((ci, mnx, mny, w, h));
        }
        boxes.sort_by(|a, b| b.4.partial_cmp(&a.4).unwrap());
        let shelf_w = total_area.sqrt() * 1.3 + 1.0;
        let mut sx = 0.0;
        let mut sy = 0.0;
        let mut row_h = 0.0f64;
        for &(ci, mnx, mny, w, h) in &boxes {
            if sx > 0.0 && sx + w > shelf_w {
                sx = 0.0;
                sy += row_h;
                row_h = 0.0;
            }
            let ox = sx - mnx;
            let oy = sy - mny;
            for &u in &comps[ci] {
                px[u as usize] += ox;
                py[u as usize] += oy;
            }
            sx += w;
            row_h = row_h.max(h);
        }
    }

    // output
    let mut s = String::with_capacity(n * 24);
    for i in 0..n {
        s.push_str(&format!("{:.4} {:.4}\n", px[i], py[i]));
    }
    out.write_all(s.as_bytes()).unwrap();
}
