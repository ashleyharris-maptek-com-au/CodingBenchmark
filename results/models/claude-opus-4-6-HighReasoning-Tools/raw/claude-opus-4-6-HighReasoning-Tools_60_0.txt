const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { machineCount, tasks } = data;
const n = tasks.length;
const taskMap = new Map();
for (const t of tasks) taskMap.set(t.id, t);

const ids = tasks.map(t => t.id);
const idx = new Map();
for (let i = 0; i < n; i++) idx.set(ids[i], i);

const dur = new Float64Array(n);
const dl = new Float64Array(n);
const wt = new Float64Array(n);
const depsIdx = new Array(n);
const revDeps = new Array(n);
for (let i = 0; i < n; i++) revDeps[i] = [];

for (let i = 0; i < n; i++) {
  const t = tasks[i];
  dur[i] = t.duration;
  dl[i] = t.deadline != null ? t.deadline : 1e15;
  wt[i] = t.weight != null ? t.weight : 1;
  const d = t.deps || [];
  depsIdx[i] = d.map(x => idx.get(x));
  for (const di of depsIdx[i]) revDeps[di].push(i);
}

// Critical path length from each task to sinks (longest path including self)
const cpLen = new Float64Array(n);
const topoOrder = [];
{
  const inDeg = new Int32Array(n);
  for (let i = 0; i < n; i++) inDeg[i] = depsIdx[i].length;
  const queue = [];
  for (let i = 0; i < n; i++) if (inDeg[i] === 0) queue.push(i);
  let head = 0;
  while (head < queue.length) {
    const u = queue[head++];
    topoOrder.push(u);
    for (const v of revDeps[u]) {
      if (--inDeg[v] === 0) queue.push(v);
    }
  }
}

// Reverse topo for critical path (bottom-up)
for (let i = topoOrder.length - 1; i >= 0; i--) {
  const u = topoOrder[i];
  let maxChild = 0;
  for (const v of revDeps[u]) {
    if (cpLen[v] > maxChild) maxChild = cpLen[v];
  }
  cpLen[u] = dur[u] + maxChild;
}

// Earliest start times (ASAP based on deps)
const asap = new Float64Array(n);
for (const u of topoOrder) {
  let earliest = 0;
  for (const d of depsIdx[u]) {
    const t = asap[d] + dur[d];
    if (t > earliest) earliest = t;
  }
  asap[u] = earliest;
}

function simulate(order, mc) {
  const startTime = new Float64Array(n);
  const machineAssign = new Int32Array(n);
  const finishTime = new Float64Array(n);
  const machineAvail = new Float64Array(mc);
  const depFinish = new Float64Array(n);

  for (const u of order) {
    let depReady = 0;
    for (const d of depsIdx[u]) {
      if (finishTime[d] > depReady) depReady = finishTime[d];
    }
    depFinish[u] = depReady;

    let bestMachine = 0;
    let bestStart = Math.max(depReady, machineAvail[0]);
    for (let m = 1; m < mc; m++) {
      const s = Math.max(depReady, machineAvail[m]);
      if (s < bestStart) {
        bestStart = s;
        bestMachine = m;
      }
    }
    startTime[u] = bestStart;
    machineAssign[u] = bestMachine;
    finishTime[u] = bestStart + dur[u];
    machineAvail[bestMachine] = finishTime[u];
  }

  let makespan = 0;
  let weightedLateness = 0;
  for (let i = 0; i < n; i++) {
    if (finishTime[i] > makespan) makespan = finishTime[i];
    const late = finishTime[i] - dl[i];
    if (late > 0) weightedLateness += wt[i] * late;
  }

  return { startTime, machineAssign, score: weightedLateness + makespan };
}

function buildScheduleResult(order, mc) {
  const sim = simulate(order, mc);
  const schedule = [];
  for (const u of order) {
    schedule.push({ id: ids[u], start: Math.round(sim.startTime[u]), machine: sim.machineAssign[u] });
  }
  return { schedule, score: sim.score };
}

// Generate multiple priority orderings and pick the best

function sortByKey(keyFn) {
  const arr = topoOrder.slice();
  // Stable topo-aware sort: use a priority queue approach
  const inDeg = new Int32Array(n);
  for (let i = 0; i < n; i++) inDeg[i] = depsIdx[i].length;
  const ready = [];
  for (let i = 0; i < n; i++) if (inDeg[i] === 0) ready.push(i);

  const keys = new Float64Array(n);
  for (let i = 0; i < n; i++) keys[i] = keyFn(i);

  ready.sort((a, b) => keys[a] - keys[b]);

  const result = [];
  while (ready.length > 0) {
    const u = ready.shift();
    result.push(u);
    for (const v of revDeps[u]) {
      if (--inDeg[v] === 0) {
        // Insert sorted
        let lo = 0, hi = ready.length;
        while (lo < hi) {
          const mid = (lo + hi) >> 1;
          if (keys[ready[mid]] <= keys[v]) lo = mid + 1;
          else hi = mid;
        }
        ready.splice(lo, 0, v);
      }
    }
  }
  return result;
}

// Use a heap-based approach for large n
function sortByKeyHeap(keyFn) {
  const inDeg = new Int32Array(n);
  for (let i = 0; i < n; i++) inDeg[i] = depsIdx[i].length;

  const keys = new Float64Array(n);
  for (let i = 0; i < n; i++) keys[i] = keyFn(i);

  // Min-heap
  const heap = [];
  function push(v) {
    heap.push(v);
    let i = heap.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (keys[heap[p]] <= keys[heap[i]]) break;
      [heap[p], heap[i]] = [heap[i], heap[p]];
      i = p;
    }
  }
  function pop() {
    const top = heap[0];
    const last = heap.pop();
    if (heap.length > 0) {
      heap[0] = last;
      let i = 0;
      while (true) {
        let s = i, l = 2*i+1, r = 2*i+2;
        if (l < heap.length && keys[heap[l]] < keys[heap[s]]) s = l;
        if (r < heap.length && keys[heap[r]] < keys[heap[s]]) s = r;
        if (s === i) break;
        [heap[i], heap[s]] = [heap[s], heap[i]];
        i = s;
      }
    }
    return top;
  }

  for (let i = 0; i < n; i++) if (inDeg[i] === 0) push(i);

  const result = [];
  while (heap.length > 0) {
    const u = pop();
    result.push(u);
    for (const v of revDeps[u]) {
      if (--inDeg[v] === 0) push(v);
    }
  }
  return result;
}

const strategies = [
  // Longest critical path first
  i => -cpLen[i],
  // Earliest deadline first
  i => dl[i],
  // Highest weight first
  i => -wt[i],
  // Weighted deadline urgency: deadline - critical_path_remaining
  i => dl[i] - cpLen[i],
  // Weighted urgency: (deadline - asap - duration) / weight  (slack per weight)
  i => (dl[i] - asap[i] - dur[i]) / (wt[i] + 0.001),
  // Combined: critical path + weight
  i => -cpLen[i] * wt[i],
  // Earliest deadline weighted by critical path
  i => dl[i] / (cpLen[i] + 1),
  // WLPT-like: deadline - weight * remaining_path
  i => dl[i] - wt[i] * cpLen[i],
];

let bestSchedule = null;
let bestScore = Infinity;

const sortFn = n > 5000 ? sortByKeyHeap : sortByKey;

for (const strat of strategies) {
  const order = sortFn(strat);
  const result = buildScheduleResult(order, machineCount);
  if (result.score < bestScore) {
    bestScore = result.score;
    bestSchedule = result.schedule;
  }
}

// Also try: full schedule with dynamic priority (simulate with priority queue)
// For each ready task, pick based on urgency considering current time
function dynamicSchedule(mc) {
  const inDeg = new Int32Array(n);
  for (let i = 0; i < n; i++) inDeg[i] = depsIdx[i].length;

  const finishTime = new Float64Array(n);
  const startTimeArr = new Float64Array(n);
  const machineArr = new Int32Array(n);
  const scheduled = new Uint8Array(n);

  const readySet = [];
  for (let i = 0; i < n; i++) if (inDeg[i] === 0) readySet.push(i);

  // Machine available times
  const mAvail = new Float64Array(mc);
  let done = 0;

  // Event-driven simulation
  // Events: machine becomes free, dependency finishes
  while (done < n) {
    // Find earliest time any machine is free
    let earliestMachine = 0;
    for (let m = 1; m < mc; m++) {
      if (mAvail[m] < mAvail[earliestMachine]) earliestMachine = m;
    }

    if (readySet.length === 0) {
      // No ready tasks, advance time to next finish
      break; // shouldn't happen if graph is valid
    }

    // Among ready tasks, pick best candidate
    // For each ready task, compute: urgency = (deadline - currentTime - duration) / weight (lower = more urgent)
    const currentTime = mAvail[earliestMachine];

    let bestIdx = 0;
    let bestKey = Infinity;
    for (let j = 0; j < readySet.length; j++) {
      const u = readySet[j];
      let depReady = 0;
      for (const d of depsIdx[u]) {
        if (finishTime[d] > depReady) depReady = finishTime[d];
      }
      const start = Math.max(currentTime, depReady);
      // If this task can't start now on any machine, skip for now
      // Actually find best machine for this task
      const slack = dl[u] - start - dur[u];
      const key = slack / (wt[u] + 0.001);
      if (key < bestKey) {
        bestKey = key;
        bestIdx = j;
      }
    }

    const u = readySet[bestIdx];
    readySet[bestIdx] = readySet[readySet.length - 1];
    readySet.pop();

    let depReady = 0;
    for (const d of depsIdx[u]) {
      if (finishTime[d] > depReady) depReady = finishTime[d];
    }

    // Find best machine
    let bm = 0;
    let bs = Math.max(depReady, mAvail[0]);
    for (let m = 1; m < mc; m++) {
      const s = Math.max(depReady, mAvail[m]);
      if (s < bs) { bs = s; bm = m; }
    }

    startTimeArr[u] = bs;
    machineArr[u] = bm;
    finishTime[u] = bs + dur[u];
    mAvail[bm] = finishTime[u];
    scheduled[u] = 1;
    done++;

    for (const v of revDeps[u]) {
      if (--inDeg[v] === 0) readySet.push(v);
    }
  }

  let makespan = 0;
  let weightedLateness = 0;
  for (let i = 0; i < n; i++) {
    if (finishTime[i] > makespan) makespan = finishTime[i];
    const late = finishTime[i] - dl[i];
    if (late > 0) weightedLateness += wt[i] * late;
  }

  const score = weightedLateness + makespan;
  if (score < bestScore) {
    bestScore = score;
    bestSchedule = [];
    for (let i = 0; i < n; i++) {
      bestSchedule.push({ id: ids[i], start: Math.round(startTimeArr[i]), machine: machineArr[i] });
    }
  }
}

dynamicSchedule(machineCount);

// Try another dynamic variant: pick task with highest weight * critical_path
function dynamicSchedule2(mc) {
  const inDeg = new Int32Array(n);
  for (let i = 0; i < n; i++) inDeg[i] = depsIdx[i].length;

  const finishTime = new Float64Array(n);
  const startTimeArr = new Float64Array(n);
  const machineArr = new Int32Array(n);

  const readySet = [];
  for (let i = 0; i < n; i++) if (inDeg[i] === 0) readySet.push(i);

  const mAvail = new Float64Array(mc);
  let done = 0;

  while (done < n && readySet.length > 0) {
    // Find task-machine pair that starts earliest with highest priority
    let bestU = -1, bestUIdx = -1, bestM = 0, bestStart = Infinity, bestPri = -Infinity;

    for (let j = 0; j < readySet.length; j++) {
      const u = readySet[j];
      let depReady = 0;
      for (const d of depsIdx[u]) {
        if (finishTime[d] > depReady) depReady = finishTime[d];
      }

      // Find earliest machine
      let bm = 0, bs = Math.max(depReady, mAvail[0]);
      for (let m = 1; m < mc; m++) {
        const s = Math.max(depReady, mAvail[m]);
        if (s < bs) { bs = s; bm = m; }
      }

      const pri = wt[u] * cpLen[u];

      // Prefer earliest start, break ties by priority
      if (bs < bestStart || (bs === bestStart && pri > bestPri)) {
        bestU = u; bestUIdx = j; bestM = bm; bestStart = bs; bestPri = pri;
      }
    }

    readySet[bestUIdx] = readySet[readySet.length - 1];
    readySet.pop();

    startTimeArr[bestU] = bestStart;
    machineArr[bestU] = bestM;
    finishTime[bestU] = bestStart + dur[bestU];
    mAvail[bestM] = finishTime[bestU];
    done++;

    for (const v of revDeps[bestU]) {
      if (--inDeg[v] === 0) readySet.push(v);
    }
  }

  let makespan = 0;
  let weightedLateness = 0;
  for (let i = 0; i < n; i++) {
    if (finishTime[i] > makespan) makespan = finishTime[i];
    const late = finishTime[i] - dl[i];
    if (late > 0) weightedLateness += wt[i] * late;
  }

  const score = weightedLateness + makespan;
  if (score < bestScore) {
    bestScore = score;
    bestSchedule = [];
    for (let i = 0; i < n; i++) {
      bestSchedule.push({ id: ids[i], start: Math.round(startTimeArr[i]), machine: machineArr[i] });
    }
  }
}
