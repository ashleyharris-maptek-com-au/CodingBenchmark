declare const require: any;
const fs = require('fs');
const data = JSON.parse(fs.readFileSync(0, 'utf8'));

const W: number = data.width;
const H: number = data.height;
const grid: string[] = data.grid;
const ignitions: number[][] = data.ignitions;
const towns: { id: string; x: number; y: number; value: number }[] = data.towns;
let budget: number = data.cutBudget;

const N = W * H;
const pass = new Uint8Array(N);
for (let y = 0; y < H; y++) {
  const r = grid[y];
  for (let x = 0; x < W; x++) if (r.charCodeAt(x) === 46) pass[y * W + x] = 1;
}

const isCut = new Uint8Array(N);
const isIgn = new Uint8Array(N);
for (const [x, y] of ignitions) {
  if (x >= 0 && x < W && y >= 0 && y < H) isIgn[y * W + x] = 1;
}

const firebreaks: number[][] = [];

const fd = new Int32Array(N);
const fq = new Int32Array(N);
let fTail = 0;

const tdBuf = new Int32Array(N);
const tq = new Int32Array(N);
let tTail = 0;
let tGen = 0;
const tgb = new Int32Array(N);

function fireB(): void {
  fd.fill(-1);
  let h = 0; fTail = 0;
  for (const [x, y] of ignitions) {
    if (x < 0 || x >= W || y < 0 || y >= H) continue;
    const i = y * W + x;
    if (pass[i] && !isCut[i] && fd[i] < 0) { fd[i] = 0; fq[fTail++] = i; }
  }
  while (h < fTail) {
    const i = fq[h++];
    const iy = (i / W) | 0, ix = i - iy * W, d1 = fd[i] + 1;
    let n: number;
    if (iy > 0)     { n = i - W; if (pass[n] && !isCut[n] && fd[n] < 0) { fd[n] = d1; fq[fTail++] = n; } }
    if (iy < H - 1) { n = i + W; if (pass[n] && !isCut[n] && fd[n] < 0) { fd[n] = d1; fq[fTail++] = n; } }
    if (ix > 0)     { n = i - 1; if (pass[n] && !isCut[n] && fd[n] < 0) { fd[n] = d1; fq[fTail++] = n; } }
    if (ix < W - 1) { n = i + 1; if (pass[n] && !isCut[n] && fd[n] < 0) { fd[n] = d1; fq[fTail++] = n; } }
  }
}

function townB(sx: number, sy: number): void {
  tGen++; tTail = 0; let h = 0;
  const si = sy * W + sx;
  if (pass[si] && !isCut[si] && fd[si] >= 0) {
    tdBuf[si] = 0; tgb[si] = tGen; tq[tTail++] = si;
  }
  while (h < tTail) {
    const i = tq[h++];
    const iy = (i / W) | 0, ix = i - iy * W, d1 = tdBuf[i] + 1;
    let n: number;
    if (iy > 0)     { n = i - W; if (pass[n] && !isCut[n] && fd[n] >= 0 && tgb[n] !== tGen) { tdBuf[n] = d1; tgb[n] = tGen; tq[tTail++] = n; } }
    if (iy < H - 1) { n = i + W; if (pass[n] && !isCut[n] && fd[n] >= 0 && tgb[n] !== tGen) { tdBuf[n] = d1; tgb[n] = tGen; tq[tTail++] = n; } }
    if (ix > 0)     { n = i - 1; if (pass[n] && !isCut[n] && fd[n] >= 0 && tgb[n] !== tGen) { tdBuf[n] = d1; tgb[n] = tGen; tq[tTail++] = n; } }
    if (ix < W - 1) { n = i + 1; if (pass[n] && !isCut[n] && fd[n] >= 0 && tgb[n] !== tGen) { tdBuf[n] = d1; tgb[n] = tGen; tq[tTail++] = n; } }
  }
}

function gtd(i: number): number { return tgb[i] === tGen ? tdBuf[i] : -1; }

function doCut(i: number): void {
  isCut[i] = 1;
  const y = (i / W) | 0;
  firebreaks.push([i - y * W, y]);
  budget--;
}

const T0 = Date.now();

function layerPhase(): void {
  for (let it = 0; it < 500 && budget > 0; it++) {
    if (Date.now() - T0 > 15000) return;
    fireB();
    const thr: any[] = [];
    for (const t of towns) if (fd[t.y * W + t.x] >= 0) thr.push(t);
    if (!thr.length) return;
    thr.sort((a: any, b: any) => b.value - a.value);

    let bR = 0, bTw: any = null, bD = -1;
    const nc = Math.min(thr.length, 40);
    for (let ti = 0; ti < nc; ti++) {
      if (Date.now() - T0 > 15000) break;
      const tw = thr[ti];
      townB(tw.x, tw.y);

      let mID = Infinity;
      for (const [ix, iy] of ignitions) {
        if (ix < 0 || ix >= W || iy < 0 || iy >= H) continue;
        const d = gtd(iy * W + ix);
        if (d >= 0 && d < mID) mID = d;
      }
      if (mID <= 1 || !isFinite(mID)) continue;

      const mL = Math.min(mID, 5000);
      const cnt = new Int32Array(mL);
      for (let qi = 0; qi < tTail; qi++) {
        const d = tdBuf[tq[qi]];
        if (d > 0 && d < mL) cnt[d]++;
      }

      const tvd: { v: number; d: number }[] = [];
      for (const t2 of thr) {
        const d2 = gtd(t2.y * W + t2.x);
        if (d2 >= 0) tvd.push({ v: t2.value, d: d2 });
      }
      tvd.sort((a, b) => a.d - b.d);

      let cv = 0, tvi = 0;
      for (let d = 1; d < mL; d++) {
        while (tvi < tvd.length && tvd[tvi].d < d) { cv += tvd[tvi].v; tvi++; }
        if (!cnt[d] || cnt[d] > budget) continue;
        const r = cv / cnt[d];
        if (r > bR) { bR = r; bTw = tw; bD = d; }
      }
    }

    if (!bTw || bD < 0) return;
    townB(bTw.x, bTw.y);
    for (let qi = 0; qi < tTail; qi++) {
      const idx = tq[qi];
      if (tdBuf[idx] === bD && !isCut[idx]) doCut(idx);
    }
  }
}

function greedyPhase(): void {
  const sc = new Float64Array(N);
  const marked: number[] = [];

  for (let b = 0; b < 200 && budget > 0; b++) {
    if (Date.now() - T0 > 24000) return;
    fireB();
    const thr: any[] = [];
    for (const t of towns) if (fd[t.y * W + t.x] >= 0) thr.push(t);
    if (!thr.length) return;
    thr.sort((a: any, b: any) => b.value - a.value);

    for (const m of marked) sc[m] = 0;
    marked.length = 0;

    const na = Math.min(thr.length, 15);
    for (let ti = 0; ti < na; ti++) {
      const tw = thr[ti];
      townB(tw.x, tw.y);
      const ftd = fd[tw.y * W + tw.x];
      if (ftd <= 0) continue;
      const mL = Math.min(ftd + 1, 10000);
      const lc = new Int32Array(mL);
      for (let qi = 0; qi < tTail; qi++) {
        const idx = tq[qi];
        const d = tdBuf[idx];
        if (d > 0 && d < mL && d + fd[idx] === ftd) lc[d]++;
      }
      for (let qi = 0; qi < tTail; qi++) {
        const idx = tq[qi];
        const d = tdBuf[idx];
        if (d > 0 && d < mL && d + fd[idx] === ftd && lc[d] > 0 && !isIgn[idx] && pass[idx] && !isCut[idx]) {
          if (sc[idx] === 0) marked.push(idx);
          sc[idx] += tw.value / lc[d];
        }
      }
    }

    let bI = -1, bS = 0;
    for (const idx of marked) {
      if (sc[idx] > bS) { bS = sc[idx]; bI = idx; }
    }
    if (bI < 0) return;
    doCut(bI);
  }
}

if (budget > 0 && towns.length > 0 && ignitions.length > 0) {
  layerPhase();
  if (budget > 0) greedyPhase();
}
