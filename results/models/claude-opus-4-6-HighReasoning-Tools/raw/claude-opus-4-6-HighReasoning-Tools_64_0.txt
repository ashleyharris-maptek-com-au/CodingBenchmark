declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const prob: number[][] = data.probability;
const drones: { id: number; x: number; y: number }[] = data.drones;
const maxSteps: number = data.maxStepsPerDrone;
const rows = prob.length;
const cols = prob[0].length;
const N = rows * cols;
const numDrones = drones.length;

const DX = [1, -1, 0, 0];
const DY = [0, 0, 1, -1];

const fp = new Float64Array(N);
for (let y = 0; y < rows; y++)
  for (let x = 0; x < cols; x++)
    fp[y * cols + x] = prob[y][x];

const vis = new Uint8Array(N);
for (const d of drones) vis[d.y * cols + d.x] = 1;

const bg = new Uint32Array(N);
const bp = new Int32Array(N);
const bd = new Int32Array(N);
const bpv = new Float64Array(N);
const bq = new Int32Array(N);
let gen = 0;

function bfsFind(sx: number, sy: number, maxR: number): number[] | null {
  gen++;
  const sk = sy * cols + sx;
  bg[sk] = gen;
  bp[sk] = -1;
  bd[sk] = 0;
  bpv[sk] = 0;
  bq[0] = sk;
  let qh = 0, qt = 1;
  let bestK = -1, bestScore = 0, bestDist = 0;

  while (qh < qt) {
    const ck = bq[qh++];
    const cd = bd[ck];
    if (cd >= maxR) break;
    if (bestK >= 0 && cd > bestDist * 3 + 30) break;
    const cx = ck % cols, cy = (ck - cx) / cols;
    for (let i = 0; i < 4; i++) {
      const nx = cx + DX[i], ny = cy + DY[i];
      if (nx < 0 || nx >= cols || ny < 0 || ny >= rows) continue;
      const nk = ny * cols + nx;
      if (fp[nk] === -1 || bg[nk] === gen) continue;
      bg[nk] = gen;
      bp[nk] = ck;
      const nd = cd + 1;
      bd[nk] = nd;
      const cv = (vis[nk] === 0 && fp[nk] > 0) ? fp[nk] : 0;
      bpv[nk] = bpv[ck] + cv;
      if (bpv[nk] > 0) {
        const sc = bpv[nk] / nd;
        if (sc > bestScore) {
          bestScore = sc;
          bestK = nk;
          bestDist = nd;
        }
      }
      bq[qt++] = nk;
    }
  }
  if (bestK < 0) return null;
  const path: number[] = [];
  let c = bestK;
  while (c !== sk) {
    path.push(c);
    c = bp[c];
  }
  path.reverse();
  return path;
}

function findSeg(sx: number, sy: number, maxD: number): number[] | null {
  let r = Math.min(60, maxD);
  while (true) {
    const res = bfsFind(sx, sy, r);
    if (res) return res;
    if (r >= maxD) return null;
    r = Math.min(r * 3, maxD);
  }
}

interface DS {
  x: number;
  y: number;
  rem: number;
  out: number[][];
  seg: number[];
  si: number;
  tgt: number;
}

const st: DS[] = drones.map(d => ({
  x: d.x,
  y: d.y,
  rem: maxSteps,
  out: [[d.x, d.y]],
  seg: [],
  si: 0,
  tgt: -1,
}));

for (let step = 0; step < maxSteps; step++) {
  for (let j = 0; j < numDrones; j++) {
    const d = st[(step + j) % numDrones];
    if (d.rem <= 0) continue;

    let ak = -1, av = 0;
    for (let i = 0; i < 4; i++) {
      const nx = d.x + DX[i], ny = d.y + DY[i];
      if (nx < 0 || nx >= cols || ny < 0 || ny >= rows) continue;
      const nk = ny * cols + nx;
      if (vis[nk] === 0 && fp[nk] > av) {
        av = fp[nk];
        ak = nk;
      }
    }

    const hasSeg = d.si < d.seg.length && vis[d.tgt] === 0;
    let mk = -1;

    if (ak >= 0 && hasSeg) {
      const snk = d.seg[d.si];
      const sv = (vis[snk] === 0 && fp[snk] > 0) ? fp[snk] : 0;
      if (av > sv) {
        mk = ak;
        d.seg = [];
        d.si = 0;
        d.tgt = -1;
      } else {
        mk = snk;
        d.si++;
      }
    } else if (ak >= 0) {
      mk = ak;
    } else if (hasSeg) {
      mk = d.seg[d.si++];
    } else {
      const s = findSeg(d.x, d.y, d.rem);
      if (s) {
        d.seg = s;
        d.si = 0;
        d.tgt = s[s.length - 1];
        mk = d.seg[d.si++];
      }
    }

    if (mk >= 0) {
      d.x = mk % cols;
      d.y = (mk - d.x) / cols;
      d.out.push([d.x, d.y]);
      vis[mk] = 1;
      d.rem--;
    } else {
      d.out.push([d.x, d.y]);
      d.rem--;
    }
  }
}
