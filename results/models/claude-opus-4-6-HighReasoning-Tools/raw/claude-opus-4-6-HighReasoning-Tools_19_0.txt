using System;
using System.Collections.Generic;
using System.IO;
using System.Text;

class Program
{
    static int XN, YN, ZN;
    static int[,] surface;
    static byte[][][] voxels;
    static int[,] costToPlant;
    static readonly int[] DX = { 1, -1, 0, 0 };
    static readonly int[] DY = { 0, 0, 1, -1 };

    static int Mass(byte d) => d & 0xF;
    static int Value(byte d) => d >> 4;
    static int H(int x, int y) => (x >= 0 && x < XN && y >= 0 && y < YN) ? surface[x, y] + 1 : 0;

    static void Main()
    {
        var stdin = Console.OpenStandardInput(1 << 22);

        var hdr = new List<byte>();
        int b;
        while ((b = stdin.ReadByte()) >= 0 && b != '\n')
            if (b != '\r') hdr.Add((byte)b);
        var p = Encoding.ASCII.GetString(hdr.ToArray()).Split(' ');
        XN = int.Parse(p[0]); YN = int.Parse(p[1]); ZN = int.Parse(p[2]);

        surface = new int[XN, YN];
        voxels = new byte[XN][][];
        long total = (long)XN * YN * ZN;
        bool storeAll = total <= 1_500_000_000L;

        byte[] col = new byte[ZN];
        for (int x = 0; x < XN; x++)
        {
            voxels[x] = new byte[YN][];
            for (int y = 0; y < YN; y++)
            {
                ReadFull(stdin, col, ZN);
                int surf = -1; bool hasVal = false;
                for (int z = ZN - 1; z >= 0; z--)
                {
                    if (col[z] != 0)
                    {
                        if (surf < 0) surf = z;
                        if (!hasVal && Value(col[z]) > 0 && Mass(col[z]) > 0) hasVal = true;
                    }
                }
                surface[x, y] = surf;
                if (storeAll || hasVal)
                {
                    voxels[x][y] = new byte[ZN];
                    Buffer.BlockCopy(col, 0, voxels[x][y], 0, ZN);
                }
            }
        }

        var ops = new List<string>();
        RunDijkstra();
        Mine(ops);

        var sw = new StreamWriter(Console.OpenStandardOutput(), Encoding.ASCII, 1 << 20);
        sw.WriteLine(ops.Count);
        for (int i = 0; i < ops.Count; i++) sw.WriteLine(ops[i]);
        sw.Flush();
    }

    static void ReadFull(Stream s, byte[] buf, int n)
    {
        int off = 0;
        while (off < n) { int r = s.Read(buf, off, n - off); if (r <= 0) break; off += r; }
    }

    static void RunDijkstra()
    {
        costToPlant = new int[XN, YN];
        for (int x = 0; x < XN; x++)
            for (int y = 0; y < YN; y++)
                costToPlant[x, y] = int.MaxValue;
        costToPlant[0, 0] = 0;

        int maxCost = (XN + YN) * 5 + 10;
        var buckets = new List<int>[maxCost + 1];
        for (int i = 0; i <= maxCost; i++) buckets[i] = new List<int>();
        buckets[0].Add(0);

        for (int ptr = 0; ptr <= maxCost; ptr++)
        {
            var bk = buckets[ptr];
            for (int bi = 0; bi < bk.Count; bi++)
            {
                int enc = bk[bi];
                int ux = enc / YN, uy = enc % YN;
                if (costToPlant[ux, uy] != ptr) continue;
                int hU = H(ux, uy);
                for (int d = 0; d < 4; d++)
                {
                    int vx = ux + DX[d], vy = uy + DY[d];
                    if (vx < 0 || vx >= XN || vy < 0 || vy >= YN) continue;
                    int hV = H(vx, vy);
                    int diff = hU - hV;
                    if (diff > 1 || diff < -1) continue;
                    int sc = diff > 0 ? 5 : (diff == 0 ? 1 : 0);
                    int nc = ptr + sc;
                    if (nc < costToPlant[vx, vy] && nc <= maxCost)
                    {
                        costToPlant[vx, vy] = nc;
                        buckets[nc].Add(vx * YN + vy);
                    }
                }
            }
            bk.Clear();
        }
    }

    static int MaxAccessibleDepth(int cx, int cy)
    {
        int sz = surface[cx, cy];
        if (sz < 0) return 0;
        int maxDepth = 0;
        for (int d = 0; d <= sz; d++)
        {
            int truckH_ob = sz - d + 1;
            bool canExitOb = false;
            for (int dir = 0; dir < 4; dir++)
            {
                int nx = cx + DX[dir], ny = cy + DY[dir];
                if (nx >= 0 && nx < XN && ny >= 0 && ny < YN)
                    if (IAbs(H(nx, ny) - truckH_ob) <= 1) { canExitOb = true; break; }
            }
            if (!canExitOb) break;

            int truckH_ore = sz - d;
            bool canExitOre = false;
            for (int dir = 0; dir < 4; dir++)
            {
                int nx = cx + DX[dir], ny = cy + DY[dir];
                if (nx >= 0 && nx < XN && ny >= 0 && ny < YN)
                    if (IAbs(H(nx, ny) - truckH_ore) <= 1) { canExitOre = true; break; }
            }
            if (canExitOre) maxDepth = d;
        }
        return maxDepth;
    }

    static void Mine(List<string> ops)
    {
        var cands = new List<(double profit, int x, int y, int z)>();
        for (int x = 0; x < XN; x++)
            for (int y = 0; y < YN; y++)
            {
                if (voxels[x][y] == null) continue;
                int sz = surface[x, y];
                if (sz < 0) continue;
                int maxD = MaxAccessibleDepth(x, y);
                double obMass = 0;
                for (int z = sz; z >= Math.Max(0, sz - maxD); z--)
                {
                    byte d = voxels[x][y][z];
                    int m = Mass(d), v = Value(d);
                    if (d != 0 && v > 0 && m > 0)
                    {
                        double rev = 1000.0 * m * v;
                        double tc = costToPlant[x, y] < int.MaxValue ? (double)costToPlant[x, y] * m : 1e18;
                        double obc = obMass * 15.0;
                        double pr = rev - tc - obc;
                        if (pr > 0) cands.Add((pr, x, y, z));
                    }
                    if (d != 0) obMass += Mass(d);
                }
            }

        cands.Sort((a, b) => b.profit.CompareTo(a.profit));
        if (cands.Count > 300000) cands.RemoveRange(300000, cands.Count - 300000);

        foreach (var (_, cx, cy, cz) in cands)
        {
            if (voxels[cx][cy] == null) continue;
            if (voxels[cx][cy][cz] == 0 || Value(voxels[cx][cy][cz]) == 0) continue;
            if (surface[cx, cy] < cz) continue;

            int curMaxD = MaxAccessibleDepth(cx, cy);
            int neededD = surface[cx, cy] - cz;
            if (neededD > curMaxD) continue;

            bool ok = true;
            while (surface[cx, cy] > cz && ok)
            {
                int sz = surface[cx, cy];
                byte od = voxels[cx][cy][sz];
                if (od == 0 || Mass(od) == 0)
                {
                    voxels[cx][cy][sz] = 0;
                    DropSurface(cx, cy);
                    continue;
                }

                int om = Mass(od), ov = Value(od);
                int savedSurf = surface[cx, cy];
                voxels[cx][cy][sz] = 0;
                DropSurface(cx, cy);

                bool toPlant = false;
                int destX, destY;
                List<int> path = null;

                if (ov > 0)
                {
                    path = BuildPath(cx, cy, 0, 0);
                    if (path != null)
                    {
                        double rev = 1000.0 * om * ov;
                        double cost = PathCost(cx, cy, path, 0, 0, om);
                        if (rev > cost) { toPlant = true; destX = 0; destY = 0; }
                        else path = null;
                    }
                }

                if (!toPlant)
                {
                    destX = 0; destY = 0;
                    var dumpSites = GetDumpSites(cx, cy);
                    foreach (var ds in dumpSites)
                    {
                        path = BuildPath(cx, cy, ds[0], ds[1]);
                        if (path != null) { destX = ds[0]; destY = ds[1]; break; }
                    }
                }
                else { destX = 0; destY = 0; }

                if (path == null)
                {
                    voxels[cx][cy][sz] = od;
                    surface[cx, cy] = savedSurf;
                    ok = false; break;
                }

                EmitOp(ops, cx, cy, sz, path, destX, destY);

                if (!toPlant)
                {
                    int ns = surface[destX, destY] + 1;
                    surface[destX, destY] = ns;
                    if (voxels[destX][destY] == null)
                        voxels[destX][destY] = new byte[ZN];
                    if (ns < ZN) voxels[destX][destY][ns] = od;
                }
            }

            if (!ok) continue;

            byte tgt = voxels[cx][cy][cz];
            if (tgt == 0 || Value(tgt) == 0 || Mass(tgt) == 0) continue;
            if (surface[cx, cy] != cz) continue;

            int tm = Mass(tgt), tv = Value(tgt);
            int savedS2 = surface[cx, cy];
            voxels[cx][cy][cz] = 0;
            DropSurface(cx, cy);

            var fpath = BuildPath(cx, cy, 0, 0);
            if (fpath == null)
            {
                voxels[cx][cy][cz] = tgt;
                surface[cx, cy] = savedS2;
                continue;
            }

            double fCost = PathCost(cx, cy, fpath, 0, 0, tm);
            double fRev = 1000.0 * tm * tv;
            if (fRev <= fCost)
            {
                voxels[cx][cy][cz] = tgt;
                surface[cx, cy] = savedS2;
                continue;
            }

            EmitOp(ops, cx, cy, cz, fpath, 0, 0);
        }
    }

    static void DropSurface(int x, int y)
    {
        while (surface[x, y] >= 0 && (voxels[x][y] == null || voxels[x][y][surface[x, y]] == 0))
            surface[x, y]--;
    }

    static List<int[]> GetDumpSites(int fx, int fy)
    {
        var sites = new List<(int score, int x, int y)>();
        int[][] raw = {
            new[]{XN-1,YN-1}, new[]{XN-1,fy}, new[]{fx,YN-1},
            new[]{XN-1,0}, new[]{0,YN-1},
            new[]{XN-1,YN/2}, new[]{XN/2,YN-1},
            new[]{XN/2,0}, new[]{0,YN/2}
        };
        foreach (var s in raw)
        {
            int sx = s[0], sy = s[1];
            if (sx == 0 && sy == 0) continue;
            if (sx == fx && sy == fy) continue;
            if (sx < 0 || sx >= XN || sy < 0 || sy >= YN) continue;
            int score = IAbs(fx - sx) + IAbs(fy - sy) + H(sx, sy) / 2;
            sites.Add((score, sx, sy));
        }
        sites.Sort((a, b) => a.score.CompareTo(b.score));
        var result = new List<int[]>();
        foreach (var s in sites) result.Add(new[] { s.x, s.y });
        return result;
    }

    static List<int> BuildPath(int fx, int fy, int tx, int ty)
    {
        if (fx == tx && fy == ty) return new List<int>();
        var r = TryLinear(fx, fy, tx, ty, true);
        if (r != null) return r;
        r = TryLinear(fx, fy, tx, ty, false);
        if (r != null) return r;
        r = TryMixed(fx, fy, tx, ty);
        if (r != null) return r;
        if ((long)XN * YN <= 4_000_000L)
            r = BFSPath(fx, fy, tx, ty);
        return r;
    }

    static List<int> TryLinear(int fx, int fy, int tx, int ty, bool xFirst)
    {
        var steps = new List<int>();
        int cx = fx, cy = fy, ph = H(fx, fy);
        if (xFirst)
        {
            int d = tx > fx ? 1 : -1;
            while (cx != tx)
            {
                cx += d;
                int h = H(cx, cy);
                if (IAbs(h - ph) > 1) return null;
                if (cx != tx || cy != ty) { steps.Add(cx); steps.Add(cy); }
                ph = h;
            }
            d = ty > fy ? 1 : -1;
            while (cy != ty)
            {
                cy += d;
                int h = H(cx, cy);
                if (IAbs(h - ph) > 1) return null;
                if (cx != tx || cy != ty) { steps.Add(cx); steps.Add(cy); }
                ph = h;
            }
        }
        else
        {
            int d = ty > fy ? 1 : -1;
            while (cy != ty)
            {
                cy += d;
                int h = H(cx, cy);
                if (IAbs(h - ph) > 1) return null;
                if (cx != tx || cy != ty) { steps.Add(cx); steps.Add(cy); }
                ph = h;
            }
            d = tx > fx ? 1 : -1;
            while (cx != tx)
            {
                cx += d;
                int h = H(cx, cy);
                if (IAbs(h - ph) > 1) return null;
                if (cx != tx || cy != ty) { steps.Add(cx); steps.Add(cy); }
                ph = h;
            }
        }
        return steps;
    }

    static List<int> TryMixed(int fx, int fy, int tx, int ty)
    {
        var steps = new List<int>();
        int cx = fx, cy = fy, ph = H(fx, fy);
        int ddx = tx > fx ? 1 : (tx < fx ? -1 : 0);
        int ddy = ty > fy ? 1 : (ty < fy ? -1 : 0);
        while (cx != tx || cy != ty)
        {
            bool moved = false;
            if (cx != tx)
            {
                int nx = cx + ddx, h = H(nx, cy);
                if (IAbs(h - ph) <= 1) { cx = nx; if (cx != tx || cy != ty) { steps.Add(cx); steps.Add(cy); } ph = h; moved = true; }
            }
            if (!moved && cy != ty)
            {
                int ny = cy + ddy, h = H(cx, ny);
                if (IAbs(h - ph) <= 1) { cy = ny; if (cx != tx || cy != ty) { steps.Add(cx); steps.Add(cy); } ph = h; moved = true; }
            }
            else if (moved && cy != ty)
            {
                int ny = cy + ddy, h = H(cx, ny);
                if (IAbs(h - ph) <= 1) { cy = ny; if (cx != tx || cy != ty) { steps.Add(cx); steps.Add(cy); } ph = h; }
            }
            if (!moved) return null;
        }
        return steps;
    }

    static List<int> BFSPath(int fx, int fy, int tx, int ty)
    {
        var prev = new int[XN, YN];
        for (int x = 0; x < XN; x++)
            for (int y = 0; y < YN; y++)
                prev[x, y] = -1;
        prev[fx, fy] = fx * YN + fy;
        var queue = new Queue<int>();
        queue.Enqueue(fx * YN + fy);
        while (queue.Count > 0)
        {
            int enc = queue.Dequeue();
            int ux = enc / YN, uy = enc % YN;
            if (ux == tx && uy == ty)
            {
                var path = new List<int>();
                int cx2 = tx, cy2 = ty;
                while (!(cx2 == fx && cy2 == fy))
                {
                    path.Add(cx2); path.Add(cy2);
                    int pe = prev[cx2, cy2];
                    cx2 = pe / YN; cy2 = pe % YN;
                }
                path.Reverse();
                if (path.Count >= 2) path.RemoveRange(path.Count - 2, 2);
                return path;
            }
            int hU = H(ux, uy);
            for (int d = 0; d < 4; d++)
            {
                int vx = ux + DX[d], vy = uy + DY[d];
                if (vx < 0 || vx >= XN || vy < 0 || vy >= YN) continue;
                if (prev[vx, vy] != -1) continue;
                int hV = H(vx, vy);
                if (IAbs(hV - hU) > 1) continue;
                prev[vx, vy] = ux * YN + uy;
                queue.Enqueue(vx * YN + vy);
            }
        }
        return null;
    }

    static double PathCost(int fx, int fy, List<int> steps, int destX, int destY, int mass)
    {
        if (steps.Count == 0 && fx == destX && fy == destY) return 0;
        double cost = 0;
        int ph = H(fx, fy);
        for (int i = 0; i < steps.Count; i += 2)
        {
            int h = H(steps[i], steps[i + 1]);
            if (h > ph) cost += 5.0 * mass;
            else if (h == ph) cost += 1.0 * mass;
            ph = h;
        }
        int dh = H(destX, destY);
        if (dh > ph) cost += 5.0 * mass;
        else if (dh == ph) cost += 1.0 * mass;
        return cost;
    }

    static void EmitOp(List<string> ops, int digX, int digY, int digZ, List<int> steps, int destX, int destY)
    {
        int n = steps.Count / 2;
        var sb = new StringBuilder();
        sb.Append(n); sb.Append(' ');
        sb.Append(digX); sb.Append(' '); sb.Append(digY); sb.Append(' '); sb.Append(digZ);
        for (int i = 0; i < steps.Count; i += 2)
        {
            sb.Append(' '); sb.Append(steps[i]); sb.Append(' '); sb.Append(steps[i + 1]);
        }
        sb.Append(' '); sb.Append(destX); sb.Append(' '); sb.Append(destY);
        ops.Add(sb.ToString());
    }

    static int IAbs(int x) => x < 0 ? -x : x;
}
