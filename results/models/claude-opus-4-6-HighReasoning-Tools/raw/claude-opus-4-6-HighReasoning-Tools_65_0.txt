declare const require: any; const fs = require('fs'); const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

interface Task {
  id: number;
  duration: number;
  deps: number[];
  deadline: number;
  weight: number;
}

const machineCount: number = data.machineCount;
const tasks: Task[] = data.tasks;
const n = tasks.length;

const byId = new Map<number, Task>();
for (const t of tasks) byId.set(t.id, t);

const children = new Map<number, number[]>();
for (const t of tasks) {
  children.set(t.id, []);
}
for (const t of tasks) {
  for (const d of t.deps) {
    children.get(d)!.push(t.id);
  }
}

// Compute longest path from each task to any sink (critical path tail length)
const cpLen = new Map<number, number>();
function getCPLen(id: number): number {
  if (cpLen.has(id)) return cpLen.get(id)!;
  const t = byId.get(id)!;
  let maxChild = 0;
  for (const c of children.get(id)!) {
    const v = getCPLen(c);
    if (v > maxChild) maxChild = v;
  }
  const val = t.duration + maxChild;
  cpLen.set(id, val);
  return val;
}
for (const t of tasks) getCPLen(t.id);

// Compute earliest start time for each task
const est = new Map<number, number>();
function getEST(id: number): number {
  if (est.has(id)) return est.get(id)!;
  const t = byId.get(id)!;
  let maxDepEnd = 0;
  for (const d of t.deps) {
    const v = getEST(d) + byId.get(d)!.duration;
    if (v > maxDepEnd) maxDepEnd = v;
  }
  est.set(id, maxDepEnd);
  return maxDepEnd;
}
for (const t of tasks) getEST(t.id);

// Latest start time (deadline - duration, but also constrained by children)
const lst = new Map<number, number>();
function getLST(id: number): number {
  if (lst.has(id)) return lst.get(id)!;
  const t = byId.get(id)!;
  let minVal = t.deadline - t.duration;
  for (const c of children.get(id)!) {
    const v = getLST(c) - t.duration;
    if (v < minVal) minVal = v;
  }
  lst.set(id, minVal);
  return minVal;
}
for (const t of tasks) getLST(t.id);

// Priority: higher = schedule first
// Use weighted combination: urgency (slack), critical path, weight
function priority(t: Task): number {
  const slack = lst.get(t.id)! - est.get(t.id)!;
  // Primary: weighted urgency (low slack + high weight = high priority)
  // Secondary: critical path length (longer = higher priority)
  // Tertiary: deadline (earlier = higher priority)
  return -(slack) * 0.001 + cpLen.get(t.id)! * 1.0 + t.weight * 100.0 - t.deadline * 0.0001;
}

// List scheduling with priority-based ready queue
const inDegree = new Map<number, number>();
for (const t of tasks) inDegree.set(t.id, t.deps.length);

const ready: number[] = [];
for (const t of tasks) {
  if (t.deps.length === 0) ready.push(t.id);
}

// Sort ready list by priority descending
ready.sort((a, b) => priority(byId.get(b)!) - priority(byId.get(a)!));

// Machine availability times
const machineAvail = new Float64Array(machineCount);

// Track when each task finishes (for dependency resolution)
const finishTime = new Map<number, number>();

const schedule: { id: number; start: number; machine: number }[] = [];

// Use a simulation approach: process ready tasks by priority
while (ready.length > 0) {
  // Pick highest priority ready task
  // But we need to ensure we pick a task whose deps are all finished
  // Since we only add to ready when all deps are done, all ready tasks are schedulable

  // Sort ready by priority (re-sort since new tasks may have been added)
  ready.sort((a, b) => priority(byId.get(b)!) - priority(byId.get(a)!));

  const tid = ready.shift()!;
  const t = byId.get(tid)!;

  // Earliest this task can start (all deps must be finished)
  let earliest = 0;
  for (const d of t.deps) {
    const ft = finishTime.get(d)!;
    if (ft > earliest) earliest = ft;
  }

  // Find machine that gives earliest start
  let bestMachine = 0;
  let bestStart = Math.max(earliest, machineAvail[0]);
  for (let m = 1; m < machineCount; m++) {
    const s = Math.max(earliest, machineAvail[m]);
    if (s < bestStart) {
      bestStart = s;
      bestMachine = m;
    }
  }

  const startTime = bestStart;
  const endTime = startTime + t.duration;
  machineAvail[bestMachine] = endTime;
  finishTime.set(tid, endTime);

  schedule.push({ id: tid, start: Math.round(startTime), machine: bestMachine });

  // Update children
  for (const c of children.get(tid)!) {
    const newDeg = inDegree.get(c)! - 1;
    inDegree.set(c, newDeg);
    if (newDeg === 0) {
      ready.push(c);
    }
  }
}
