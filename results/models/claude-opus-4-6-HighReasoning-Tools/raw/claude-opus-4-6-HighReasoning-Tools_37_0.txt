using System;
using System.Collections.Generic;

class Program
{
    static int _bestK;
    static List<int> _bestFVS;

    static void Main()
    {
        var tokens = Console.In.ReadToEnd().Split(new[] { ' ', '\n', '\r', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        int p = 0;
        int n = int.Parse(tokens[p++]), m = int.Parse(tokens[p++]);

        var fwd = new Dictionary<int, HashSet<int>>(n);
        var rev = new Dictionary<int, HashSet<int>>(n);
        for (int i = 0; i < n; i++)
        {
            fwd[i] = new HashSet<int>();
            rev[i] = new HashSet<int>();
        }

        for (int i = 0; i < m; i++)
        {
            int u = int.Parse(tokens[p++]), v = int.Parse(tokens[p++]);
            fwd[u].Add(v);
            rev[v].Add(u);
        }

        var result = new List<int>();
        Reduce(fwd, rev, result);

        foreach (var scc in GetSCCs(fwd))
        {
            if (scc.Count <= 1) continue;
            var set = new HashSet<int>(scc);
            var sf = new Dictionary<int, HashSet<int>>(scc.Count);
            var sr = new Dictionary<int, HashSet<int>>(scc.Count);
            foreach (int v in scc)
            {
                sf[v] = new HashSet<int>();
                sr[v] = new HashSet<int>();
                foreach (int w in fwd[v]) if (set.Contains(w)) sf[v].Add(w);
                foreach (int w in rev[v]) if (set.Contains(w)) sr[v].Add(w);
            }
            result.AddRange(SolveSCC(sf, sr));
        }

        Console.WriteLine(result.Count);
        Console.WriteLine(result.Count > 0 ? string.Join(" ", result) : "");
    }

    static List<int> SolveSCC(Dictionary<int, HashSet<int>> fwd, Dictionary<int, HashSet<int>> rev)
    {
        var forced = new List<int>();
        Reduce(fwd, rev, forced);
        if (fwd.Count == 0) return forced;

        var sccs = GetSCCs(fwd);
        var parts = new List<KeyValuePair<Dictionary<int, HashSet<int>>, Dictionary<int, HashSet<int>>>>();
        foreach (var scc in sccs)
        {
            if (scc.Count <= 1) continue;
            var set = new HashSet<int>(scc);
            var sf = new Dictionary<int, HashSet<int>>();
            var sr = new Dictionary<int, HashSet<int>>();
            foreach (int v in scc)
            {
                sf[v] = new HashSet<int>();
                sr[v] = new HashSet<int>();
                foreach (int w in fwd[v]) if (set.Contains(w)) sf[v].Add(w);
                foreach (int w in rev[v]) if (set.Contains(w)) sr[v].Add(w);
            }
            parts.Add(new KeyValuePair<Dictionary<int, HashSet<int>>, Dictionary<int, HashSet<int>>>(sf, sr));
        }

        foreach (var part in parts)
        {
            var sf = part.Key;
            var sr = part.Value;
            var greedy = Greedy(sf, sr);
            _bestK = greedy.Count;
            _bestFVS = greedy;
            if (_bestK > 1)
            {
                var clone = Clone(sf, sr);
                Search(clone.Key, clone.Value, new List<int>());
            }
            forced.AddRange(_bestFVS);
        }

        return forced;
    }

    static void Search(Dictionary<int, HashSet<int>> fwd, Dictionary<int, HashSet<int>> rev, List<int> current)
    {
        var f2 = new List<int>();
        Reduce(fwd, rev, f2);
        current.AddRange(f2);

        if (current.Count >= _bestK) return;

        var ntSccs = new List<Dictionary<int, HashSet<int>>>();
        foreach (var scc in GetSCCs(fwd))
        {
            if (scc.Count <= 1) continue;
            var set = new HashSet<int>(scc);
            var sa = new Dictionary<int, HashSet<int>>();
            foreach (int v in scc)
            {
                sa[v] = new HashSet<int>();
                foreach (int w in fwd[v]) if (set.Contains(w)) sa[v].Add(w);
            }
            ntSccs.Add(sa);
        }

        if (ntSccs.Count == 0)
        {
            _bestK = current.Count;
            _bestFVS = new List<int>(current);
            return;
        }

        int lb = 0;
        foreach (var sa in ntSccs) lb += PackCycles(sa);
        if (current.Count + lb >= _bestK) return;

        List<int> shortCycle = null;
        foreach (var sa in ntSccs)
        {
            var c = FindCycle(sa);
            if (c != null && (shortCycle == null || c.Count < shortCycle.Count))
                shortCycle = c;
            if (shortCycle != null && shortCycle.Count == 2) break;
        }
        if (shortCycle == null) return;

        shortCycle.Sort((a, b) =>
        {
            int da = (fwd.ContainsKey(a) ? fwd[a].Count : 0) * (rev.ContainsKey(a) ? rev[a].Count : 0);
            int db = (fwd.ContainsKey(b) ? fwd[b].Count : 0) * (rev.ContainsKey(b) ? rev[b].Count : 0);
            return db.CompareTo(da);
        });

        foreach (int v in shortCycle)
        {
            if (current.Count + 1 >= _bestK) break;
            var cl = Clone(fwd, rev);
            Rem(v, cl.Key, cl.Value);
            var nc = new List<int>(current);
            nc.Add(v);
            Search(cl.Key, cl.Value, nc);
        }
    }

    static List<int> Greedy(Dictionary<int, HashSet<int>> fwd, Dictionary<int, HashSet<int>> rev)
    {
        var cl = Clone(fwd, rev);
        var f = cl.Key;
        var r = cl.Value;
        var fvs = new List<int>();
        Reduce(f, r, fvs);
        while (f.Count > 0)
        {
            var c = FindCycle(f);
            if (c == null) break;
            int best = c[0], bs = 0;
            foreach (int v in c)
            {
                int s = f[v].Count * r[v].Count;
                if (s > bs) { bs = s; best = v; }
            }
            fvs.Add(best);
            Rem(best, f, r);
            Reduce(f, r, fvs);
        }
        return fvs;
    }

    static void Rem(int v, Dictionary<int, HashSet<int>> fwd, Dictionary<int, HashSet<int>> rev)
    {
        if (!fwd.ContainsKey(v)) return;
        foreach (int w in fwd[v]) if (rev.ContainsKey(w)) rev[w].Remove(v);
        foreach (int w in rev[v]) if (fwd.ContainsKey(w)) fwd[w].Remove(v);
        fwd.Remove(v);
        rev.Remove(v);
    }

    static void Reduce(Dictionary<int, HashSet<int>> fwd, Dictionary<int, HashSet<int>> rev, List<int> fvs)
    {
        var q = new Queue<int>(fwd.Keys);
        while (q.Count > 0)
        {
            int v = q.Dequeue();
            if (!fwd.ContainsKey(v)) continue;

            if (fwd[v].Contains(v))
            {
                fvs.Add(v);
                var nb = new HashSet<int>(fwd[v]);
                nb.UnionWith(rev[v]);
                nb.Remove(v);
                Rem(v, fwd, rev);
                foreach (int w in nb) if (fwd.ContainsKey(w)) q.Enqueue(w);
                continue;
            }

            if (fwd[v].Count == 0 || rev[v].Count == 0)
            {
                var nb = new HashSet<int>(fwd[v]);
                nb.UnionWith(rev[v]);
                Rem(v, fwd, rev);
                foreach (int w in nb) if (fwd.ContainsKey(w)) q.Enqueue(w);
                continue;
            }

            if (rev[v].Count == 1)
            {
                int u = First(rev[v]);
                var outV = new List<int>(fwd[v]);
                fwd[u].Remove(v);
                foreach (int w in outV)
                {
                    if (w == v) continue;
                    rev[w].Remove(v);
                    if (w == u) { fwd[u].Add(u); rev[u].Add(u); }
                    else { fwd[u].Add(w); rev[w].Add(u); }
                }
                fwd.Remove(v);
                rev.Remove(v);
                q.Enqueue(u);
                foreach (int w in outV) if (w != v && fwd.ContainsKey(w)) q.Enqueue(w);
                continue;
            }

            if (fwd[v].Count == 1)
            {
                int u = First(fwd[v]);
                var inV = new List<int>(rev[v]);
                rev[u].Remove(v);
                foreach (int w in inV)
                {
                    if (w == v) continue;
                    fwd[w].Remove(v);
                    if (w == u) { fwd[u].Add(u); rev[u].Add(u); }
                    else { rev[u].Add(w); fwd[w].Add(u); }
                }
                fwd.Remove(v);
                rev.Remove(v);
                q.Enqueue(u);
                foreach (int w in inV) if (w != v && fwd.ContainsKey(w)) q.Enqueue(w);
                continue;
            }
        }
    }

    static int First(HashSet<int> s)
    {
        foreach (int v in s) return v;
        return -1;
    }

    static KeyValuePair<Dictionary<int, HashSet<int>>, Dictionary<int, HashSet<int>>> Clone(
        Dictionary<int, HashSet<int>> fwd, Dictionary<int, HashSet<int>> rev)
    {
        var f = new Dictionary<int, HashSet<int>>(fwd.Count);
        var r = new Dictionary<int, HashSet<int>>(rev.Count);
        foreach (var kv in fwd) f[kv.Key] = new HashSet<int>(kv.Value);
        foreach (var kv in rev) r[kv.Key] = new HashSet<int>(kv.Value);
        return new KeyValuePair<Dictionary<int, HashSet<int>>, Dictionary<int, HashSet<int>>>(f, r);
    }

    static List<int> FindCycle(Dictionary<int, HashSet<int>> fwd)
    {
        if (fwd.Count == 0) return null;

        foreach (int u in fwd.Keys)
            foreach (int v in fwd[u])
                if (v != u && fwd.ContainsKey(v) && fwd[v].Contains(u))
                    return new List<int> { u, v };

        var color = new Dictionary<int, int>(fwd.Count);
        foreach (int v in fwd.Keys) color[v] = 0;

        foreach (int s in fwd.Keys)
        {
            if (color[s] != 0) continue;
            var path = new List<int>();
            var onPath = new HashSet<int>();
            var its = new Stack<IEnumerator<int>>();

            color[s] = 1; path.Add(s); onPath.Add(s);
            its.Push(fwd[s].GetEnumerator());

            while (its.Count > 0)
            {
                var it = its.Peek();
                bool adv = false;
                while (it.MoveNext())
                {
                    int nx = it.Current;
                    if (!fwd.ContainsKey(nx)) continue;
                    if (color[nx] == 1 && onPath.Contains(nx))
                    {
                        int i = path.IndexOf(nx);
                        return path.GetRange(i, path.Count - i);
                    }
                    if (color[nx] == 0)
                    {
                        color[nx] = 1; path.Add(nx); onPath.Add(nx);
                        its.Push(fwd[nx].GetEnumerator());
                        adv = true; break;
                    }
                }
                if (!adv)
                {
                    its.Pop();
                    int vv = path[path.Count - 1];
                    path.RemoveAt(path.Count - 1);
                    onPath.Remove(vv);
                    color[vv] = 2;
                }
            }
        }
        return null;
    }

    static List<List<int>> GetSCCs(Dictionary<int, HashSet<int>> fwd)
    {
        var vis = new HashSet<int>();
        var ord = new List<int>();
        foreach (int v in fwd.Keys)
            if (!vis.Contains(v))
                DfsPost(v, fwd, vis, ord);

        var rv = new Dictionary<int, HashSet<int>>();
        foreach (int v in fwd.Keys) rv[v] = new HashSet<int>();
        foreach (int u in fwd.Keys)
            foreach (int v in fwd[u])
                if (rv.ContainsKey(v)) rv[v].Add(u);

        vis.Clear();
        var sccs = new List<List<int>>();
        for (int i = ord.Count - 1; i >= 0; i--)
        {
            int v = ord[i];
            if (!vis.Contains(v) && fwd.ContainsKey(v))
            {
                var scc = new List<int>();
                DfsCollect(v, rv, vis, scc);
                sccs.Add(scc);
            }
        }
        return sccs;
    }

    static void DfsPost(int v, Dictionary<int, HashSet<int>> g, HashSet<int> vis, List<int> ord)
    {
        var stk = new Stack<KeyValuePair<int, bool>>();
        stk.Push(new KeyValuePair<int, bool>(v, false));
        while (stk.Count > 0)
        {
            var top = stk.Pop();
            int n = top.Key;
            bool d = top.Value;
            if (d) { ord.Add(n); continue; }
            if (vis.Contains(n)) continue;
            vis.Add(n);
            stk.Push(new KeyValuePair<int, bool>(n, true));
            if (g.ContainsKey(n))
                foreach (int w in g[n])
                    if (!vis.Contains(w) && g.ContainsKey(w))
                        stk.Push(new KeyValuePair<int, bool>(w, false));
        }
    }

    static void DfsCollect(int v, Dictionary<int, HashSet<int>> g, HashSet<int> vis, List<int> scc)
    {
        var stk = new Stack<int>();
        stk.Push(v);
        while (stk.Count > 0)
        {
            int n = stk.Pop();
            if (vis.Contains(n)) continue;
            vis.Add(n);
            scc.Add(n);
            if (g.ContainsKey(n))
                foreach (int w in g[n])
                    if (!vis.Contains(w)) stk.Push(w);
        }
    }

    static int PackCycles(Dictionary<int, HashSet<int>> adj)
    {
        var a = new Dictionary<int, HashSet<int>>(adj.Count);
        var r = new Dictionary<int, HashSet<int>>(adj.Count);
        foreach (var kv in adj) a[kv.Key] = new HashSet<int>(kv.Value);
        foreach (int v in a.Keys) r[v] = new HashSet<int>();
        foreach (int u in a.Keys) foreach (int v in a[u]) if (r.ContainsKey(v)) r[v].Add(u);

        int cnt = 0;

        var used = new HashSet<int>();
        foreach (int u in a.Keys)
        {
            if (used.Contains(u)) continue;
            foreach (int v in a[u])
            {
                if (v != u && !used.Contains(v) && a.ContainsKey(v) && a[v].Contains(u))
                {
                    cnt++; used.Add(u); used.Add(v); break;
                }
            }
        }
        foreach (int v in used) RemBoth(v, a, r);
        CleanBoth(a, r);

        while (a.Count > 0)
        {
            var c = FindCycle(a);
            if (c == null) break;
            cnt++;
            foreach (int v in c) RemBoth(v, a, r);
            CleanBoth(a, r);
        }
        return cnt;
    }

    static void RemBoth(int v, Dictionary<int, HashSet<int>> a, Dictionary<int, HashSet<int>> r)
    {
        if (!a.ContainsKey(v)) return;
        foreach (int w in a[v]) if (r.ContainsKey(w)) r[w].Remove(v);
        foreach (int w in r[v]) if (a.ContainsKey(w)) a[w].Remove(v);
        a.Remove(v);
        r.Remove(v);
    }

    static void CleanBoth(Dictionary<int, HashSet<int>> a, Dictionary<int, HashSet<int>> r)
    {
        bool ch = true;
        while (ch)
        {
            ch = false;
            foreach (int v in new List<int>(a.Keys))
            {
                if (!a.ContainsKey(v)) continue;
                if (a[v].Count == 0 || r[v].Count == 0)
                {
                    RemBoth(v, a, r);
                    ch = true;
                }
            }
        }
    }
}
