declare const require: any;
const fs = require("fs");
const input: string = fs.readFileSync(0, "utf8");
const data = JSON.parse(input);

const grid: string[] = data.grid;
const robots: { id: number; x: number; y: number }[] = data.robots;
const jobs: { id: number; x: number; y: number; priority: number }[] = data.jobs;

const rows = grid.length;
const cols = grid[0].length;

const R = robots.length;
const J = jobs.length;

const DIRS = [
  [0, 1],
  [0, -1],
  [1, 0],
  [-1, 0],
];

function bfs(sx: number, sy: number): Int32Array {
  const dist = new Int32Array(rows * cols).fill(-1);
  const idx = sy * cols + sx;
  dist[idx] = 0;
  const queue: number[] = [idx];
  let head = 0;
  while (head < queue.length) {
    const cur = queue[head++];
    const cx = cur % cols;
    const cy = (cur - cx) / cols;
    const d = dist[cur] + 1;
    for (let i = 0; i < 4; i++) {
      const nx = cx + DIRS[i][0];
      const ny = cy + DIRS[i][1];
      if (nx >= 0 && nx < cols && ny >= 0 && ny < rows) {
        const ni = ny * cols + nx;
        if (dist[ni] === -1 && grid[ny][nx] === ".") {
          dist[ni] = d;
          queue.push(ni);
        }
      }
    }
  }
  return dist;
}

// Internal indices 0..R-1 for robots, 0..J-1 for jobs
// Map from internal index to external id
const rIds = robots.map((r) => r.id);
const jIds = jobs.map((j) => j.id);

// BFS from each robot position
const rDist: Int32Array[] = new Array(R);
for (let ri = 0; ri < R; ri++) {
  rDist[ri] = bfs(robots[ri].x, robots[ri].y);
}

// BFS from each job position
const jDist: Int32Array[] = new Array(J);
const jCell: number[] = new Array(J);
for (let ji = 0; ji < J; ji++) {
  jCell[ji] = jobs[ji].y * cols + jobs[ji].x;
  jDist[ji] = bfs(jobs[ji].x, jobs[ji].y);
}

const rCell: number[] = new Array(R);
for (let ri = 0; ri < R; ri++) {
  rCell[ri] = robots[ri].y * cols + robots[ri].x;
}

function dRJ(ri: number, ji: number): number {
  const d = rDist[ri][jCell[ji]];
  return d < 0 ? 1e9 : d;
}

function dJJ(ji1: number, ji2: number): number {
  const d = jDist[ji1][jCell[ji2]];
  return d < 0 ? 1e9 : d;
}

function routeCost(ri: number, seq: number[]): number {
  if (seq.length === 0) return 0;
  let cost = dRJ(ri, seq[0]);
  for (let i = 1; i < seq.length; i++) {
    cost += dJJ(seq[i - 1], seq[i]);
  }
  return cost;
}

// Greedy assignment: jobs sorted by priority desc, assign to cheapest robot
const assign: number[][] = new Array(R);
for (let i = 0; i < R; i++) assign[i] = [];

const sortedJobIndices = Array.from({ length: J }, (_, i) => i).sort(
  (a, b) => jobs[b].priority - jobs[a].priority
);

const lastJob: number[] = new Array(R).fill(-1); // last job internal index per robot

for (const ji of sortedJobIndices) {
  let bestR = 0;
  let bestCost = Infinity;
  for (let ri = 0; ri < R; ri++) {
    let cost: number;
    if (lastJob[ri] === -1) {
      cost = dRJ(ri, ji);
    } else {
      cost = dJJ(lastJob[ri], ji);
    }
    if (cost < bestCost) {
      bestCost = cost;
      bestR = ri;
    }
  }
  assign[bestR].push(ji);
  lastJob[bestR] = ji;
}

// --- Local search optimization ---

function segCostRange(ri: number, seq: number[], i: number, j: number): number {
  let cost = 0;
  if (i === 0) {
    cost += dRJ(ri, seq[0]);
  } else {
    cost += dJJ(seq[i - 1], seq[i]);
  }
  for (let k = i; k < j; k++) {
    cost += dJJ(seq[k], seq[k + 1]);
  }
  if (j + 1 < seq.length) {
    cost += dJJ(seq[j], seq[j + 1]);
  }
  return cost;
}

function twoOpt(ri: number): boolean {
  const seq = assign[ri];
  const n = seq.length;
  if (n < 3) return false;
  let improved = false;
  for (let i = 0; i < n - 1; i++) {
    for (let j = i + 1; j < n; j++) {
      const oldC = segCostRange(ri, seq, i, j);
      let l = i, r = j;
      while (l < r) { const t = seq[l]; seq[l] = seq[r]; seq[r] = t; l++; r--; }
      const newC = segCostRange(ri, seq, i, j);
      if (newC < oldC - 0.5) {
        improved = true;
      } else {
        l = i; r = j;
        while (l < r) { const t = seq[l]; seq[l] = seq[r]; seq[r] = t; l++; r--; }
      }
    }
  }
  return improved;
}

function orOpt(ri: number): boolean {
  const seq = assign[ri];
  const n = seq.length;
  if (n < 2) return false;
  let improved = false;
  for (let i = 0; i < seq.length; i++) {
    const jid = seq[i];
    const oldTotal = routeCost(ri, seq);
    seq.splice(i, 1);
    let bestPos = i;
    let bestCost = oldTotal;
    for (let p = 0; p <= seq.length; p++) {
      if (p === i) continue;
      seq.splice(p, 0, jid);
      const c = routeCost(ri, seq);
      if (c < bestCost) {
        bestCost = c;
        bestPos = p;
      }
      seq.splice(p, 1);
    }
    if (bestPos !== i) {
      seq.splice(bestPos, 0, jid);
      improved = true;
    } else {
      seq.splice(i, 0, jid);
    }
  }
  return improved;
}

function tryRelocate(): boolean {
  let improved = false;
  for (let r1 = 0; r1 < R; r1++) {
    const seq1 = assign[r1];
    for (let i = seq1.length - 1; i >= 0; i--) {
      const ji = seq1[i];
      const oldCost1 = routeCost(r1, seq1);
      seq1.splice(i, 1);
      const newCost1 = routeCost(r1, seq1);
      const saving = oldCost1 - newCost1;

      let bestR2 = -1;
      let bestPos = -1;
      let bestInc = Infinity;

      for (let r2 = 0; r2 < R; r2++) {
        if (r2 === r1) continue;
        const seq2 = assign[r2];
        const oldCost2 = routeCost(r2, seq2);
        for (let p = 0; p <= seq2.length; p++) {
          seq2.splice(p, 0, ji);
          const inc = routeCost(r2, seq2) - oldCost2;
          if (inc < bestInc) {
            bestInc = inc;
            bestR2 = r2;
            bestPos = p;
          }
          seq2.splice(p, 1);
        }
      }

      if (bestR2 !== -1 && bestInc < saving - 0.5) {
        assign[bestR2].splice(bestPos, 0, ji);
        improved = true;
      } else {
        seq1.splice(i, 0, ji);
      }
    }
  }
  return improved;
}

// Swap a single job between two robots
function trySwap(): boolean {
  let improved = false;
  for (let r1 = 0; r1 < R - 1; r1++) {
    for (let r2 = r1 + 1; r2 < R; r2++) {
      const s1 = assign[r1];
      const s2 = assign[r2];
      if (s1.length === 0 || s2.length === 0) continue;
      const base = routeCost(r1, s1) + routeCost(r2, s2);
      for (let i = 0; i < s1.length; i++) {
        for (let j = 0; j < s2.length; j++) {
          const t = s1[i]; s1[i] = s2[j]; s2[j] = t;
          const nc = routeCost(r1, s1) + routeCost(r2, s2);
          if (nc < base - 0.5) {
            improved = true;
          } else {
            s2[j] = s1[i]; s1[i] = t;
          }
          if (improved) break;
        }
        if (improved) break;
      }
      if (improved) break;
    }
    if (improved) break;
  }
  return improved;
}

const startTime = Date.now();
const timeLimit = 8000;

let go = true;
while (go && Date.now() - startTime < timeLimit) {
  go = false;
  for (let ri = 0; ri < R; ri++) {
    if (twoOpt(ri)) go = true;
    if (orOpt(ri)) go = true;
  }
  if (J <= 800 && R <= 50) {
    if (tryRelocate()) go = true;
    if (trySwap()) go = true;
  }
}

// Build output indexed by robot external id
const maxRId = Math.max(0, ...rIds);
const result: number[][] = new Array(maxRId + 1);
for (let i = 0; i <= maxRId; i++) result[i] = [];
for (let ri = 0; ri < R; ri++) {
  result[rIds[ri]] = assign[ri].map((ji) => jIds[ji]);
}

process.stdout.write(JSON.stringify({ assignments: result }));
