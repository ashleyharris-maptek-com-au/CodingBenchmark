'use strict';
const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { grid, robots, jobs } = data;
const rows = grid.length;
const cols = grid[0].length;

const DX = [0, 0, 1, -1];
const DY = [1, -1, 0, 0];

function bfs(startX, startY) {
  const dist = new Int32Array(rows * cols).fill(-1);
  const idx = startY * cols + startX;
  dist[idx] = 0;
  const queue = new Int32Array(rows * cols);
  let head = 0, tail = 0;
  queue[tail++] = idx;
  while (head < tail) {
    const cur = queue[head++];
    const cx = cur % cols;
    const cy = (cur - cx) / cols;
    const d = dist[cur] + 1;
    for (let i = 0; i < 4; i++) {
      const nx = cx + DX[i];
      const ny = cy + DY[i];
      if (nx < 0 || nx >= cols || ny < 0 || ny >= rows) continue;
      const ni = ny * cols + nx;
      if (dist[ni] !== -1) continue;
      if (grid[ny][nx] === '#') { dist[ni] = -2; continue; }
      dist[ni] = d;
      queue[tail++] = ni;
    }
  }
  return dist;
}

const R = robots.length;
const J = jobs.length;

if (J === 0) {
  const assignments = [];
  for (let i = 0; i < R; i++) assignments.push([]);
  console.log(JSON.stringify({ assignments }));
  process.exit(0);
}

// Sort jobs by priority descending (higher priority first)
const sortedJobs = jobs.slice().sort((a, b) => b.priority - a.priority);
const jobIdToIdx = new Map();
for (let i = 0; i < J; i++) jobIdToIdx.set(jobs[i].id, i);

// BFS from each robot start position
const robotDists = new Array(R);
for (let r = 0; r < R; r++) {
  robotDists[r] = bfs(robots[r].x, robots[r].y);
}

// BFS from each job position
const jobDists = new Array(J);
for (let j = 0; j < J; j++) {
  jobDists[j] = bfs(jobs[j].x, jobs[j].y);
}

function distFromGrid(distArr, x, y) {
  const d = distArr[y * cols + x];
  return d < 0 ? Infinity : d;
}

// Greedy assignment: for each job (high priority first), assign to robot with least marginal cost
// marginal cost = distance from robot's current position to job
const robotPos = new Array(R);
for (let r = 0; r < R; r++) {
  robotPos[r] = { x: robots[r].x, y: robots[r].y, distArr: robotDists[r] };
}
const assignments = new Array(R);
for (let r = 0; r < R; r++) assignments[r] = [];

const robotTotalDist = new Float64Array(R);

for (let si = 0; si < sortedJobs.length; si++) {
  const job = sortedJobs[si];
  const ji = jobIdToIdx.get(job.id);
  let bestR = -1;
  let bestCost = Infinity;

  for (let r = 0; r < R; r++) {
    let cost;
    if (assignments[r].length === 0) {
      cost = distFromGrid(robotDists[r], job.x, job.y);
    } else {
      const lastJob = assignments[r][assignments[r].length - 1];
      const lastJi = jobIdToIdx.get(lastJob);
      cost = distFromGrid(jobDists[lastJi], job.x, job.y);
    }
    // Tie-break: prefer robot with less total distance so far
    if (cost < bestCost || (cost === bestCost && robotTotalDist[r] < robotTotalDist[bestR])) {
      bestCost = cost;
      bestR = r;
    }
  }

  assignments[bestR].push(job.id);
  robotTotalDist[bestR] += bestCost;
}

// Try to improve: 2-opt style swaps within each robot's job sequence
function computeRouteDistance(robotIdx, jobList) {
  if (jobList.length === 0) return 0;
  let total = 0;
  const ji0 = jobIdToIdx.get(jobList[0]);
  total += distFromGrid(robotDists[robotIdx], jobs[ji0].x, jobs[ji0].y);
  for (let i = 1; i < jobList.length; i++) {
    const prev = jobIdToIdx.get(jobList[i - 1]);
    const cur = jobIdToIdx.get(jobList[i]);
    total += distFromGrid(jobDists[prev], jobs[cur].x, jobs[cur].y);
  }
  return total;
}

// 2-opt within each robot's route
function twoOptRoute(robotIdx, jobList) {
  if (jobList.length <= 2) return;
  let improved = true;
  let maxIter = 50;
  while (improved && maxIter-- > 0) {
    improved = false;
    for (let i = 0; i < jobList.length - 1; i++) {
      for (let k = i + 1; k < jobList.length; k++) {
        // Reverse segment [i, k]
        const newList = jobList.slice();
        let left = i, right = k;
        while (left < right) {
          const tmp = newList[left];
          newList[left] = newList[right];
          newList[right] = tmp;
          left++; right--;
        }
        const oldDist = computeRouteDistance(robotIdx, jobList);
        const newDist = computeRouteDistance(robotIdx, newList);
        if (newDist < oldDist) {
          for (let m = i; m <= k; m++) jobList[m] = newList[m];
          improved = true;
        }
      }
    }
  }
}

// Try inter-robot moves: move a job from one robot to another if it reduces total
function tryInterRobotMoves() {
  let improved = true;
  let iterations = 0;
  while (improved && iterations < 20) {
    improved = false;
    iterations++;
    for (let r1 = 0; r1 < R; r1++) {
      if (assignments[r1].length === 0) continue;
      const d1 = computeRouteDistance(r1, assignments[r1]);
      for (let ji = 0; ji < assignments[r1].length; ji++) {
        const jobId = assignments[r1][ji];
        const jobIdx = jobIdToIdx.get(jobId);
        // Try removing from r1 and inserting into r2
        const r1Without = assignments[r1].slice();
        r1Without.splice(ji, 1);
        const d1Without = computeRouteDistance(r1, r1Without);
        const saving = d1 - d1Without;

        for (let r2 = 0; r2 < R; r2++) {
          if (r2 === r1) continue;
          const d2 = computeRouteDistance(r2, assignments[r2]);
          // Find best insertion position in r2
          let bestPos = 0;
          let bestD2New = Infinity;
          for (let pos = 0; pos <= assignments[r2].length; pos++) {
            const r2With = assignments[r2].slice();
            r2With.splice(pos, 0, jobId);
            const d2New = computeRouteDistance(r2, r2With);
            if (d2New < bestD2New) {
              bestD2New = d2New;
              bestPos = pos;
            }
          }
          const increase = bestD2New - d2;
          if (increase < saving - 0.5) {
            assignments[r1].splice(ji, 1);
            assignments[r2].splice(bestPos, 0, jobId);
            improved = true;
            break;
          }
        }
        if (improved) break;
      }
      if (improved) break;
    }
  }
}

// Limit optimization time for large inputs
const totalItems = R * J;
if (totalItems < 500000) {
  for (let r = 0; r < R; r++) {
    twoOptRoute(r, assignments[r]);
  }
  if (totalItems < 100000) {
    tryInterRobotMoves();
    for (let r = 0; r < R; r++) {
      twoOptRoute(r, assignments[r]);
    }
  }
}

// Build output indexed by robot id
const result = new Array(R);
for (let r = 0; r < R; r++) {
  result[robots[r].id] = assignments[r];
}
// Ensure all slots exist
for (let i = 0; i < R; i++) {
  if (!result[i]) result[i] = [];
}

console.log(JSON.stringify({ assignments: result }));
