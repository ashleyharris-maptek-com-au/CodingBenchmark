'use strict';
const fs = require('fs');
const data = JSON.parse(fs.readFileSync(0, 'utf8'));

const { probability: grid, drones, maxStepsPerDrone: maxSteps } = data;
const H = grid.length;
const W = H ? grid[0].length : 0;
const N = drones.length;

if (!N) { process.stdout.write(JSON.stringify({ paths: [] })); process.exit(); }

const SIZE = H * W;
const DX4 = [1, -1, 0, 0];
const DY4 = [0, 0, 1, -1];
const DX5 = [1, -1, 0, 0, 0];
const DY5 = [0, 0, 1, -1, 0];

const visited = new Uint8Array(SIZE);
const posX = new Int32Array(N);
const posY = new Int32Array(N);
const pathsArr = new Array(N);

for (let i = 0; i < N; i++) {
  posX[i] = drones[i].x;
  posY[i] = drones[i].y;
  pathsArr[i] = [[posX[i], posY[i]]];
  visited[posY[i] * W + posX[i]] = 1;
}

const owner = new Int32Array(SIZE).fill(-1);
{
  const q = [];
  for (let i = 0; i < N; i++) {
    const k = posY[i] * W + posX[i];
    if (owner[k] === -1) { owner[k] = i; q.push(posX[i], posY[i]); }
  }
  for (let qi = 0; qi < q.length; qi += 2) {
    const x = q[qi], y = q[qi + 1], o = owner[y * W + x];
    for (let m = 0; m < 4; m++) {
      const nx = x + DX4[m], ny = y + DY4[m];
      if (nx >= 0 && nx < W && ny >= 0 && ny < H && grid[ny][nx] !== -1 && owner[ny * W + nx] === -1) {
        owner[ny * W + nx] = o;
        q.push(nx, ny);
      }
    }
  }
}

const bfsGen = new Int32Array(SIZE);
const bfsFrom = new Int32Array(SIZE);
let gen = 0;

function findTargetBFS(di, maxDist) {
  const g = ++gen;
  const sx = posX[di], sy = posY[di];
  bfsGen[sy * W + sx] = g;
  const q = [sx, sy, 0];
  let bestScore = 0, bx = -1, by = -1;

  for (let qi = 0; qi < q.length; qi += 3) {
    const x = q[qi], y = q[qi + 1], d = q[qi + 2];
    const k = y * W + x;
    if (!visited[k] && grid[y][x] > 0) {
      const bonus = (owner[k] === di) ? 1.5 : 1.0;
      const score = grid[y][x] * bonus / (d + 1);
      if (score > bestScore) { bestScore = score; bx = x; by = y; }
    }
    if (d < maxDist) {
      for (let m = 0; m < 4; m++) {
        const nx = x + DX4[m], ny = y + DY4[m];
        if (nx >= 0 && nx < W && ny >= 0 && ny < H && grid[ny][nx] !== -1 && bfsGen[ny * W + nx] !== g) {
          bfsGen[ny * W + nx] = g;
          q.push(nx, ny, d + 1);
        }
      }
    }
  }
  return [bx, by];
}

function findTarget(di, rem) {
  for (const r of [25, 60, Math.min(rem, 250)]) {
    const [tx, ty] = findTargetBFS(di, r);
    if (tx >= 0) return [tx, ty];
  }
  return [-1, -1];
}

function getFirstStep(sx, sy, tx, ty) {
  if (sx === tx && sy === ty) return [0, 0];
  const g = ++gen;
  const src = sy * W + sx;
  bfsGen[src] = g;
  bfsFrom[src] = src;
  const q = [sx, sy];

  for (let qi = 0; qi < q.length; qi += 2) {
    const x = q[qi], y = q[qi + 1];
    for (let m = 0; m < 4; m++) {
      const nx = x + DX4[m], ny = y + DY4[m];
      if (nx >= 0 && nx < W && ny >= 0 && ny < H && grid[ny][nx] !== -1) {
        const nk = ny * W + nx;
        if (bfsGen[nk] !== g) {
          bfsGen[nk] = g;
          bfsFrom[nk] = y * W + x;
          if (nx === tx && ny === ty) {
            let k = nk;
            while (bfsFrom[k] !== src) k = bfsFrom[k];
            return [k % W - sx, (k / W | 0) - sy];
          }
          q.push(nx, ny);
        }
      }
    }
  }
  return [0, 0];
}

const _rl = new Int32Array(20);

function rolloutScore(sx, sy, depth) {
  let x = sx, y = sy, total = 0;
  _rl[0] = sy * W + sx;
  let nSeen = 1;

  for (let d = 0; d < depth; d++) {
    let bestV = 0, bx = x, by = y;
    for (let m = 0; m < 4; m++) {
      const nx = x + DX4[m], ny = y + DY4[m];
      if (nx < 0 || nx >= W || ny < 0 || ny >= H || grid[ny][nx] <= 0) continue;
      const nk = ny * W + nx;
      if (visited[nk]) continue;
      let seen = false;
      for (let s = 0; s < nSeen; s++) { if (_rl[s] === nk) { seen = true; break; } }
      if (seen) continue;
      if (grid[ny][nx] > bestV) { bestV = grid[ny][nx]; bx = nx; by = ny; }
    }
    if (bestV <= 0) break;
    total += bestV;
    _rl[nSeen++] = by * W + bx;
    x = bx; y = by;
  }
  return total;
}

const tgtX = new Int32Array(N).fill(-1);
const tgtY = new Int32Array(N).fill(-1);

for (let step = 0; step < maxSteps; step++) {
  const rem = maxSteps - step;
  const off = step % N;

  for (let j = 0; j < N; j++) {
    const i = (j + off) % N;
    const x = posX[i], y = posY[i];

    if (tgtX[i] >= 0 && (visited[tgtY[i] * W + tgtX[i]] || (x === tgtX[i] && y === tgtY[i])))
      tgtX[i] = -1;

    if (tgtX[i] === -1 || (step > 0 && step % 20 === 0))
      [tgtX[i], tgtY[i]] = findTarget(i, rem);

    if (tgtX[i] === -1) {
      pathsArr[i].push([x, y]);
      continue;
    }

    let hasLocalVal = false;
    for (let m = 0; m < 4; m++) {
      const nx = x + DX4[m], ny = y + DY4[m];
      if (nx >= 0 && nx < W && ny >= 0 && ny < H && grid[ny][nx] > 0 && !visited[ny * W + nx]) {
        hasLocalVal = true; break;
      }
    }

    let fx, fy;
    if (hasLocalVal) {
      const td = Math.abs(tgtX[i] - x) + Math.abs(tgtY[i] - y);
      const pw = td > 0 ? grid[tgtY[i]][tgtX[i]] / (td + 1) : 0;
      let bestScore = -Infinity, bestMx = 0, bestMy = 0;

      for (let m = 0; m < 5; m++) {
        const nx = x + DX5[m], ny = y + DY5[m];
        if (nx < 0 || nx >= W || ny < 0 || ny >= H || grid[ny][nx] === -1) continue;

        const nk = ny * W + nx;
        const imm = (!visited[nk] && grid[ny][nx] > 0) ? grid[ny][nx] : 0;
        const rv = rolloutScore(nx, ny, 10);
        const nd = Math.abs(tgtX[i] - nx) + Math.abs(tgtY[i] - ny);
        const prog = td - nd;

        const score = imm + rv * 0.6 + prog * Math.max(pw, 1.0);
        if (score > bestScore) { bestScore = score; bestMx = DX5[m]; bestMy = DY5[m]; }
      }
      fx = bestMx; fy = bestMy;
    } else {
      [fx, fy] = getFirstStep(x, y, tgtX[i], tgtY[i]);
      if (fx === 0 && fy === 0 && (x !== tgtX[i] || y !== tgtY[i])) {
        tgtX[i] = -1;
        pathsArr[i].push([x, y]);
        continue;
      }
    }

    const nx = x + fx, ny = y + fy;
    posX[i] = nx; posY[i] = ny;
    pathsArr[i].push([nx, ny]);
    visited[ny * W + nx] = 1;
  }
}

process.stdout.write(JSON.stringify({ paths: pathsArr }));
