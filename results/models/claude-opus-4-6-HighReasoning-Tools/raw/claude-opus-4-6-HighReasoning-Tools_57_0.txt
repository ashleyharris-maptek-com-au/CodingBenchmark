'use strict';
const fs = require('fs');
const data = JSON.parse(fs.readFileSync(0, 'utf8'));
const W = data.width, H = data.height;
const grid = data.grid, ignitions = data.ignitions || [];
const towns = data.towns || [];
let budget = data.cutBudget || 0;
const N = W * H;
const result = [];

if (!towns.length || !budget || !ignitions.length) {
  process.stdout.write(JSON.stringify({ firebreaks: result }));
  process.exit(0);
}

const burnable = new Uint8Array(N);
for (let y = 0; y < H; y++) {
  const row = grid[y];
  for (let x = 0; x < W; x++) {
    if (row.charCodeAt(x) === 46) burnable[y * W + x] = 1;
  }
}

const isCut = new Uint8Array(N);
const DX = [0, 1, 0, -1], DY = [-1, 0, 1, 0];

const ignPos = [];
const isIgn = new Uint8Array(N);
for (const [x, y] of ignitions) {
  const p = y * W + x;
  if (!isIgn[p]) { ignPos.push(p); isIgn[p] = 1; }
}

const isTownCell = new Uint8Array(N);
for (const t of towns) isTownCell[t.y * W + t.x] = 1;

const noCut = new Uint8Array(N);
for (let i = 0; i < N; i++) noCut[i] = isIgn[i] | isTownCell[i];

const Q = new Int32Array(N);
const startTime = Date.now();
const TIME_LIMIT = 28000;

function simulateFire() {
  const burned = new Uint8Array(N);
  let h = 0, t = 0;
  for (const p of ignPos) {
    if (burnable[p] && !isCut[p]) { burned[p] = 1; Q[t++] = p; }
  }
  while (h < t) {
    const ci = Q[h++], cx = ci % W, cy = (ci - cx) / W;
    for (let d = 0; d < 4; d++) {
      const nx = cx + DX[d], ny = cy + DY[d];
      if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
        const ni = ny * W + nx;
        if (!burned[ni] && burnable[ni] && !isCut[ni]) { burned[ni] = 1; Q[t++] = ni; }
      }
    }
  }
  return burned;
}

function bfsSingle(start, mask) {
  const dist = new Int32Array(N).fill(-1);
  let h = 0, t = 0;
  if (mask[start]) { dist[start] = 0; Q[t++] = start; }
  while (h < t) {
    const ci = Q[h++], cx = ci % W, cy = (ci - cx) / W, cd = dist[ci];
    for (let d = 0; d < 4; d++) {
      const nx = cx + DX[d], ny = cy + DY[d];
      if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
        const ni = ny * W + nx;
        if (dist[ni] === -1 && mask[ni]) { dist[ni] = cd + 1; Q[t++] = ni; }
      }
    }
  }
  return dist;
}

function bfsMulti(sources, mask) {
  const dist = new Int32Array(N).fill(-1);
  let h = 0, t = 0;
  for (const s of sources) {
    if (mask[s] && dist[s] === -1) { dist[s] = 0; Q[t++] = s; }
  }
  while (h < t) {
    const ci = Q[h++], cx = ci % W, cy = (ci - cx) / W, cd = dist[ci];
    for (let d = 0; d < 4; d++) {
      const nx = cx + DX[d], ny = cy + DY[d];
      if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
        const ni = ny * W + nx;
        if (dist[ni] === -1 && mask[ni]) { dist[ni] = cd + 1; Q[t++] = ni; }
      }
    }
  }
  return dist;
}

function getThreatened(burned) {
  const list = [];
  for (const t of towns) if (burned[t.y * W + t.x]) list.push(t);
  return list;
}

function applyCuts(cells) {
  for (const ci of cells) {
    isCut[ci] = 1;
    result.push([ci % W, (ci - ci % W) / W]);
    budget--;
  }
}

// Phase 1: Layer-based cuts
let outerIter = 0;
while (budget > 0 && outerIter++ < 500 && Date.now() - startTime < TIME_LIMIT) {
  const burned = simulateFire();
  const thr = getThreatened(burned);
  if (!thr.length) break;

  let bestDistArr = null, bestD = -1, bestEff = -Infinity;

  // Strategy A: Containment (BFS from ignitions, cut a narrow fire-distance layer)
  const fd = bfsMulti(ignPos, burned);
  let maxD = 0;
  for (let i = 0; i < N; i++) if (fd[i] > maxD) maxD = fd[i];

  if (maxD >= 1) {
    const lCut = new Int32Array(maxD + 1);
    const lBad = new Uint8Array(maxD + 1);
    for (let i = 0; i < N; i++) {
      const d = fd[i];
      if (d < 1) continue;
      if (noCut[i]) lBad[d] = 1;
      else if (burnable[i] && !isCut[i]) lCut[d]++;
    }
    const tvD = new Float64Array(maxD + 2);
    for (const t of thr) { const d = fd[t.y * W + t.x]; if (d >= 0) tvD[d] += t.value; }
    const vs = new Float64Array(maxD + 2);
    for (let d = maxD; d >= 0; d--) vs[d] = vs[d + 1] + tvD[d];

    for (let d = 1; d <= maxD; d++) {
      if (lBad[d] || lCut[d] === 0 || lCut[d] > budget || vs[d] <= 0) continue;
      const eff = vs[d] / lCut[d];
      if (eff > bestEff) { bestEff = eff; bestDistArr = fd; bestD = d; }
    }
  }

  // Strategy B: Per-town protection (BFS from town, cut narrow layer around it)
  thr.sort((a, b) => b.value - a.value);
  const topN = Math.min(thr.length, 20);
  for (let ti = 0; ti < topN && Date.now() - startTime < TIME_LIMIT; ti++) {
    const town = thr[ti];
    const tp = town.y * W + town.x;
    if (isIgn[tp]) continue;

    const td = bfsSingle(tp, burned);
    let minID = Infinity;
    for (const ig of ignPos) { const d = td[ig]; if (d >= 0 && d < minID) minID = d; }
    if (minID <= 1 || minID === Infinity) continue;

    const maxLD = minID - 1;
    const lCut = new Int32Array(maxLD + 1);
    const lBad = new Uint8Array(maxLD + 1);
    for (let i = 0; i < N; i++) {
      const d = td[i];
      if (d < 1 || d > maxLD) continue;
      if (noCut[i]) lBad[d] = 1;
      else if (burnable[i] && !isCut[i]) lCut[d]++;
    }
    const tvD = new Float64Array(maxLD + 1);
    for (const t of thr) { const d = td[t.y * W + t.x]; if (d >= 0 && d <= maxLD) tvD[d] += t.value; }
    let cum = 0;
    const insV = new Float64Array(maxLD + 1);
    for (let d = 0; d <= maxLD; d++) { cum += tvD[d]; insV[d] = cum; }

    for (let d = 1; d <= maxLD; d++) {
      if (lBad[d] || lCut[d] === 0 || lCut[d] > budget) continue;
      const valSaved = insV[d - 1];
      if (valSaved <= 0) continue;
      const eff = valSaved / lCut[d];
      if (eff > bestEff) { bestEff = eff; bestDistArr = td; bestD = d; }
    }
  }

  if (!bestDistArr || bestD < 0) break;

  const cells = [];
  for (let i = 0; i < N; i++) {
    if (bestDistArr[i] === bestD && burnable[i] && !isCut[i] && !noCut[i]) cells.push(i);
  }
  if (!cells.length) break;
  applyCuts(cells);
}

// Phase 2: Greedy fallback using shortest-path importance + corridor tightness
while (budget > 0 && Date.now() - startTime < TIME_LIMIT) {
  const burned = simulateFire();
  const thr = getThreatened(burned);
  if (!thr.length) break;

  thr.sort((a, b) => b.value - a.value);
  const fireDist = bfsMulti(ignPos, burned);
  const importance = new Float64Array(N);
  const topN = Math.min(thr.length, 10);

  for (let ti = 0; ti < topN; ti++) {
    const town = thr[ti];
    const tp = town.y * W + town.x;
    const td = bfsSingle(tp, burned);
    const targetDist = fireDist[tp];
    if (targetDist < 0) continue;
    for (let i = 0; i < N; i++) {
      if (!noCut[i] && burnable[i] && !isCut[i] &&
          fireDist[i] >= 0 && td[i] >= 0 &&
          fireDist[i] + td[i] === targetDist) {
        importance[i] += town.value;
      }
    }
  }

  const candidates = [];
  for (let i = 0; i < N; i++) {
    if (importance[i] <= 0) continue;
    const x = i % W, y = (i - x) / W;
    let nb = 0;
    for (let d = 0; d < 4; d++) {
      const nx = x + DX[d], ny = y + DY[d];
      if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
        const ni = ny * W + nx;
        if (burned[ni] && !isCut[ni]) nb++;
      }
    }
    candidates.push(i | 0, importance[i] / (nb + 0.5));
  }

  if (!candidates.length) break;

  // Sort candidates by score descending (stored as pairs: index, score)
  const pairs = [];
  for (let k = 0; k < candidates.length; k += 2) {
    pairs.push({ i: candidates[k], s: candidates[k + 1] });
  }
  pairs.sort((a, b) => b.s - a.s);

  const batchSize = Math.min(budget, Math.max(1, Math.min(20, Math.ceil(budget / 10))));
  let cutCount = 0;
  for (let k = 0; k < pairs.length && cutCount < batchSize; k++) {
    const ci = pairs[k].i;
    if (!isCut[ci]) {
      isCut[ci] = 1;
      result.push([ci % W, (ci - ci % W) / W]);
      budget--;
      cutCount++;
    }
  }
  if (!cutCount) break;
}

process.stdout.write(JSON.stringify({ firebreaks: result }));
