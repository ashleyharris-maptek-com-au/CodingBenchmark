using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

class MaximumClique
{
    class BitSet
    {
        public ulong[] Bits;
        public int Size;

        public BitSet(int n)
        {
            Size = n;
            Bits = new ulong[(n + 63) / 64];
        }

        public void Set(int i)
        {
            Bits[i / 64] |= (1UL << (i % 64));
        }
        
        public bool Get(int i)
        {
            return (Bits[i / 64] & (1UL << (i % 64))) != 0;
        }

        public BitSet Intersect(BitSet other)
        {
            BitSet result = new BitSet(Size);
            for (int i = 0; i < Bits.Length; i++)
            {
                result.Bits[i] = Bits[i] & other.Bits[i];
            }
            return result;
        }

        public int CountSetBits()
        {
            int count = 0;
            foreach (ulong b in Bits)
            {
                ulong v = b;
                v = v - ((v >> 1) & 0x5555555555555555UL);
                v = (v & 0x3333333333333333UL) + ((v >> 2) & 0x3333333333333333UL);
                count += (int)((((v + (v >> 4)) & 0x0F0F0F0F0F0F0F0FUL) * 0x0101010101010101UL) >> 56);
            }
            return count;
        }
    }

    static void Main(string[] args)
    {
        string line = Console.ReadLine();
        if (line == null) return;
        string[] input = line.Split(' ');
        int numVertices = int.Parse(input[0]);
        int numEdges = int.Parse(input[1]);

        BitSet[] adj = new BitSet[numVertices];
        for (int i = 0; i < numVertices; i++) adj[i] = new BitSet(numVertices);

        for (int i = 0; i < numEdges; i++)
        {
            string edgeLine = Console.ReadLine();
            if (edgeLine == null) break;
            string[] edge = edgeLine.Split(' ');
            int u = int.Parse(edge[0]);
            int v = int.Parse(edge[1]);
            
            if (u < numVertices && v < numVertices)
            {
                adj[u].Set(v);
                adj[v].Set(u);
            }
        }

        List<int> maxClique = FindMaximumClique(numVertices, adj);

        Console.WriteLine(maxClique.Count);
        Console.WriteLine(string.Join(" ", maxClique.Select(v => v.ToString()).ToArray()));
    }

    static List<int> FindMaximumClique(int numVertices, BitSet[] adj)
    {
        List<int> maxClique = new List<int>();
        
        BitSet P = new BitSet(numVertices);
        for(int i = 0; i < numVertices; i++) P.Set(i);
        BitSet R = new BitSet(numVertices);
        BitSet X = new BitSet(numVertices);
        
        BronKerbosch(R, P, X, adj, ref maxClique);
        return maxClique;
    }

    static void BronKerbosch(BitSet R, BitSet P, BitSet X, BitSet[] adj, ref List<int> maxClique)
    {
        if (P.CountSetBits() == 0 && X.CountSetBits() == 0)
        {
            int rCount = R.CountSetBits();
            if (rCount > maxClique.Count)
            {
                maxClique = new List<int>();
                for (int i = 0; i < R.Size; i++)
                {
                    if (R.Get(i)) maxClique.Add(i);
                }
            }
            return;
        }

        // Pivot selection: Vertex in P U X with maximum neighbors in P
        int pivot = -1;
        int maxNeighbors = -1;
        BitSet P_or_X = new BitSet(P.Size);
        for(int i=0; i<P.Bits.Length; i++) P_or_X.Bits[i] = P.Bits[i] | X.Bits[i];
        
        for (int v = 0; v < P.Size; v++)
        {
            if (P_or_X.Get(v))
            {
                int neighbors = P.Intersect(adj[v]).CountSetBits();
                if (neighbors > maxNeighbors)
                {
                    maxNeighbors = neighbors;
                    pivot = v;
                }
            }
        }

        BitSet candidates = new BitSet(P.Size);
        for (int i = 0; i < P.Bits.Length; i++) candidates.Bits[i] = P.Bits[i];
        if (pivot != -1)
        {
             // Candidates = P \ Neighbors(pivot)
             for (int i = 0; i < candidates.Bits.Length; i++)
                 candidates.Bits[i] &= ~adj[pivot].Bits[i];
        }

        for (int v = 0; v < P.Size; v++)
        {
            if (candidates.Get(v))
            {
                BitSet newR = new BitSet(R.Size);
                for(int i=0; i<R.Bits.Length; i++) newR.Bits[i] = R.Bits[i];
                newR.Set(v);
                
                BitSet newP = P.Intersect(adj[v]);
                BitSet newX = X.Intersect(adj[v]);
                
                BronKerbosch(newR, newP, newX, adj, ref maxClique);
                
                P.Bits[v / 64] &= ~(1UL << (v % 64)); // P.Remove(v)
                X.Set(v); // X.Add(v)
            }
        }
    }
