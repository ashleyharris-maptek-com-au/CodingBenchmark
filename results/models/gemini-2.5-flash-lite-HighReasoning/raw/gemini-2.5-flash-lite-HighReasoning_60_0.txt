const fs = require('fs');

function solve() {
  const input = fs.readFileSync(0, 'utf8');
  const { machineCount, tasks } = JSON.parse(input);

  // Build dependency graph and in-degree map
  const adj = new Map();
  const inDegree = new Map();
  const taskMap = new Map();
  
  for (const task of tasks) {
    adj.set(task.id, []);
    inDegree.set(task.id, 0);
    taskMap.set(task.id, task);
  }

  for (const task of tasks) {
    for (const depId of task.deps) {
      adj.get(depId).push(task.id);
      inDegree.set(task.id, inDegree.get(task.id) + 1);
    }
  }

  // Priority queue (simple array for this prototype, sorted by weighted deadline)
  // Heuristic: prioritize tasks with earlier deadlines and higher weights
  const getPriority = (taskId) => {
    const task = taskMap.get(taskId);
    // Lower score is higher priority: weight/deadline ratio or deadline-weight
    return task.deadline - task.weight;
  };

  const readyTasks = [];
  for (const task of tasks) {
    if (inDegree.get(task.id) === 0) {
      readyTasks.push(task.id);
    }
  }
  readyTasks.sort((a, b) => getPriority(a) - getPriority(b));

  const schedule = [];
  const machines = new Array(machineCount).fill(0); // machine free time
  const taskFinishes = new Map(); // when each task finishes

  while (readyTasks.length > 0) {
    const taskId = readyTasks.shift();
    const task = taskMap.get(taskId);

    // Calculate start time: max(dependency finish times, machine free time)
    let startTime = 0;
    for (const depId of task.deps) {
      startTime = Math.max(startTime, taskFinishes.get(depId));
    }
    
    // Find earliest available machine
    let bestMachine = 0;
    let minMachineTime = machines[0];
    for (let i = 1; i < machineCount; i++) {
        if (machines[i] < minMachineTime) {
            minMachineTime = machines[i];
            bestMachine = i;
        }
    }
    
    startTime = Math.max(startTime, minMachineTime);
    
    // Schedule task
    schedule.push({
      id: taskId,
      start: startTime,
      machine: bestMachine
    });
    
    const finishTime = startTime + task.duration;
    taskFinishes.set(taskId, finishTime);
    machines[bestMachine] = finishTime;

    // Update ready tasks
    for (const neighborId of adj.get(taskId)) {
      inDegree.set(neighborId, inDegree.get(neighborId) - 1);
      if (inDegree.get(neighborId) === 0) {
        readyTasks.push(neighborId);
      }
    }
    readyTasks.sort((a, b) => getPriority(a) - getPriority(b));
  }

  process.stdout.write(JSON.stringify({ schedule }));
}

solve();
