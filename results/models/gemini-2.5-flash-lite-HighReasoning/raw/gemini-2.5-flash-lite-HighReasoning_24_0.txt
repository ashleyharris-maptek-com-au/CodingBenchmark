#include <iostream>
#include <vector>
#include <string>
#include <thread>
#include <mutex>
#include <atomic>
#include <fstream>
#include <cmath>
#include <cstdint>

// Data structure for the lander's state
struct LanderState {
    double x, y, z;
    double vx, vy, vz;
    double pitch, yaw, roll;
    double pitch_rate, yaw_rate, roll_rate;
    double fuel;
};

// Data structure for terrain
struct Terrain {
    uint32_t width, depth;
    std::vector<uint16_t> data; // 3 * width * depth

    uint16_t get_ground_top(int x, int y) const { return data[(x * depth + y) * 3]; }
    uint16_t get_rock_bottom(int x, int y) const { return data[(x * depth + y) * 3 + 1]; }
    uint16_t get_rock_top(int x, int y) const { return data[(x * depth + y) * 3 + 2]; }
};

// Global terrain instance
Terrain terrain;
std::atomic<bool> terrain_loaded(false);

// Function to load terrain from file
void load_terrain(std::string path, uint32_t width, uint32_t depth) {
    std::ifstream file(path, std::ios::binary);
    if (!file) return;
    
    terrain.width = width;
    terrain.depth = depth;
    terrain.data.resize(width * depth * 3);
    file.read(reinterpret_cast<char*>(terrain.data.data()), width * depth * 3 * sizeof(uint16_t));
    
    terrain_loaded = true;
}

// Global state variables for communication
std::mutex state_mutex;
LanderState current_state;
std::atomic<bool> simulation_running(true);

// Controller function
void control_loop() {
    // Initial PID parameters (needs tuning)
    double kp_thrust = 0.5, kd_thrust = 0.1;
    double kp_angle = 0.1, kd_angle = 0.05;

    while (simulation_running) {
        // 1. Read latest state
        LanderState local_state;
        {
            std::lock_guard<std::mutex> lock(state_mutex);
            local_state = current_state;
        }

        // 2. Perform basic landing control logic
        // Simple gravity compensation and hover
        double thrust = 1.0; // Assume we need 1g to hover
        if (local_state.z < 100) thrust = 0.5; // Slow down near ground
        
        // Simple tilt to oppose velocity
        double pitch_cmd = local_state.vx * kp_angle + local_state.pitch_rate * kd_angle;
        double yaw_cmd = local_state.vy * kp_angle + local_state.yaw_rate * kd_angle;
        double roll_cmd = 0.0; // Not needed

        // 3. Output commands
        std::cout << thrust << " " << pitch_cmd << " " << yaw_cmd << " " << roll_cmd << std::endl;

        // 4. Sleep to maintain frequency (10hz)
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
    }
}

// Stdin reader
void input_reader() {
    std::string line;
    // Consume the "STATE" header line
    std::getline(std::cin, line);
    
    while (std::cin) {
        LanderState new_state;
        if (!(std::cin >> new_state.x >> new_state.y >> new_state.z 
                     >> new_state.vx >> new_state.vy >> new_state.vz 
                     >> new_state.pitch >> new_state.yaw >> new_state.roll 
                     >> new_state.pitch_rate >> new_state.yaw_rate >> new_state.roll_rate 
                     >> new_state.fuel)) {
            break;
        }

        {
            std::lock_guard<std::mutex> lock(state_mutex);
            current_state = new_state;
        }
    }
    simulation_running = false;
}

int main() {
    // 1. Read initial header
    double width, depth, height, gravity, max_thrust, fuel;
    std::cin >> width >> depth >> height >> gravity >> max_thrust >> fuel;

    double start_x, start_y, start_z, target_x, target_y, target_z;
    std::cin >> start_x >> start_y >> start_z >> target_x >> target_y >> target_z;

    std::string terrain_file_path;
    std::getline(std::cin >> std::ws, terrain_file_path);

    // 2. Start threads
    std::thread terrain_thread(load_terrain, terrain_file_path, (uint32_t)width, (uint32_t)depth);
    std::thread reader_thread(input_reader);
    std::thread controller_thread(control_loop);

    // 3. Wait for threads
    terrain_thread.join();
    reader_thread.join();
    controller_thread.join();

    return 0;
}
