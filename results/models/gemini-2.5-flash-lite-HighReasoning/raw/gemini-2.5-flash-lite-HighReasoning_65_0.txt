import * as fs from 'fs';

interface Task {
  id: string;
  duration: number;
  deps: string[];
  deadline: number;
  weight: number;
}

interface InputData {
  machineCount: number;
  tasks: Task[];
}

interface ScheduleEntry {
  id: string;
  start: number;
  machine: number;
}

function solve() {
  const input: string = fs.readFileSync(0, 'utf8');
  const data: InputData = JSON.parse(input);

  const { machineCount, tasks } = data;

  // Build graph
  const taskMap = new Map<string, Task>();
  const adj = new Map<string, string[]>();
  const inDegree = new Map<string, number>();

  for (const task of tasks) {
    taskMap.set(task.id, task);
    if (!adj.has(task.id)) adj.set(task.id, []);
    if (!inDegree.has(task.id)) inDegree.set(task.id, 0);
  }

  for (const task of tasks) {
    for (const dep of task.deps) {
      if (!adj.has(dep)) adj.set(dep, []);
      adj.get(dep)!.push(task.id);
      inDegree.set(task.id, (inDegree.get(task.id) || 0) + 1);
    }
  }

  // List Scheduling
  const readyQueue: string[] = [];
  for (const [id, degree] of inDegree.entries()) {
    if (degree === 0) readyQueue.push(id);
  }

  const machineAvailableTime: number[] = new Array(machineCount).fill(0);
  const taskFinishTime = new Map<string, number>();
  const schedule: ScheduleEntry[] = [];

  // Simple heuristic: sort ready queue by deadline or weight
  // For now: just process in order
  
  while (readyQueue.length > 0) {
    // Sort ready tasks by a simple heuristic: e.g., earliest deadline first
    readyQueue.sort((a, b) => {
        const taskA = taskMap.get(a)!;
        const taskB = taskMap.get(b)!;
        return taskA.deadline - taskB.deadline;
    });

    const taskId = readyQueue.shift()!;
    const task = taskMap.get(taskId)!;

    // Find earliest start time based on dependencies
    let startTime = 0;
    for (const depId of task.deps) {
      startTime = Math.max(startTime, taskFinishTime.get(depId) || 0);
    }

    // Find earliest available machine
    let bestMachine = 0;
    let minFinish = Infinity;
    
    // We want the machine that can start the task earliest.
    // That's min(max(startTime, machineAvailableTime[m]))
    
    let earliestPossibleStart = Infinity;
    for (let m = 0; m < machineCount; m++) {
        const potentialStart = Math.max(startTime, machineAvailableTime[m]);
        if (potentialStart < earliestPossibleStart) {
            earliestPossibleStart = potentialStart;
            bestMachine = m;
        }
    }

    const start = earliestPossibleStart;
    schedule.push({ id: taskId, start, machine: bestMachine });
    const finish = start + task.duration;
    taskFinishTime.set(taskId, finish);
    machineAvailableTime[bestMachine] = finish;

    // Update dependents
    for (const neighbor of (adj.get(taskId) || [])) {
      inDegree.set(neighbor, inDegree.get(neighbor)! - 1);
      if (inDegree.get(neighbor) === 0) {
        readyQueue.push(neighbor);
      }
    }
  }

  process.stdout.write(JSON.stringify({ schedule }));
}

solve();
