#include <iostream>
#include <vector>
#include <algorithm>

struct Rectangle {
    int id;
    int w, h;
    int x, y;
    bool rotated;
};

struct SkylineSegment {
    int x, y, w;
};

class Bin {
public:
    int width, height;
    std::vector<SkylineSegment> skyline;
    std::vector<Rectangle> packed_items;

    Bin(int w, int h) : width(w), height(h) {
        skyline.push_back({0, 0, w});
    }

    bool add_rect(Rectangle& rect) {
        int best_y = 1e9, best_x = 1e9, best_idx = -1;
        bool best_rotated = false;

        auto try_fit = [&](int w, int h, bool rot) {
            if (w > width || h > height) return;
            for (size_t i = 0; i < skyline.size(); ++i) {
                if (skyline[i].x + w > width) break;
                int max_y = 0, curr_w = 0;
                for (size_t j = i; j < skyline.size(); ++j) {
                    max_y = std::max(max_y, skyline[j].y);
                    curr_w += skyline[j].w;
                    if (curr_w >= w) break;
                }
                if (curr_w >= w && max_y + h <= height) {
                    if (max_y < best_y || (max_y == best_y && skyline[i].x < best_x)) {
                        best_y = max_y;
                        best_x = skyline[i].x;
                        best_idx = i;
                        best_rotated = rot;
                    }
                }
            }
        };

        try_fit(rect.w, rect.h, false);
        try_fit(rect.h, rect.w, true);

        if (best_idx == -1) return false;

        rect.x = best_x;
        rect.y = best_y;
        rect.rotated = best_rotated;
        int rw = best_rotated ? rect.h : rect.w;
        int rh = best_rotated ? rect.w : rect.h;

        std::vector<SkylineSegment> new_skyline;
        for (int i = 0; i < best_idx; ++i) new_skyline.push_back(skyline[i]);
        new_skyline.push_back({best_x, best_y + rh, rw});
        
        int remaining_w = rw;
        size_t j = best_idx;
        while (j < skyline.size() && remaining_w > 0) {
            if (skyline[j].w <= remaining_w) {
                remaining_w -= skyline[j].w;
                j++;
            } else {
                new_skyline.push_back({skyline[j].x + remaining_w, skyline[j].y, skyline[j].w - remaining_w});
                remaining_w = 0;
                j++;
            }
        }
        while (j < skyline.size()) new_skyline.push_back(skyline[j++]);

        skyline.clear();
        for (const auto& s : new_skyline) {
            if (!skyline.empty() && skyline.back().y == s.y) skyline.back().w += s.w;
            else skyline.push_back(s);
        }
        
        packed_items.push_back(rect);
        return true;
    }
};

int main() {
    int N, bin_w, bin_h;
    if (!(std::cin >> N >> bin_w >> bin_h)) return 0;
    std::vector<Rectangle> rects(N);
    for (int i = 0; i < N; ++i) {
        rects[i].id = i;
        std::cin >> rects[i].w >> rects[i].h;
    }

    std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
        return std::max(a.w, a.h) > std::max(b.w, b.h);
    });

    std::vector<Bin> bins;
    for (auto& r : rects) {
        bool packed = false;
        for (int i = std::max(0, (int)bins.size() - 5); i < bins.size(); ++i) {
            if (bins[i].add_rect(r)) {
                packed = true;
                break;
            }
        }
        if (!packed) {
            bins.emplace_back(bin_w, bin_h);
            bins.back().add_rect(r);
        }
    }

    std::cout << bins.size() << "\n";
    std::vector<int> bin_map(N);
    std::vector<int> out_x(N), out_y(N);
    std::vector<bool> out_rot(N);
    for(int i=0; i<bins.size(); ++i) {
        for(const auto& r : bins[i].packed_items) {
            bin_map[r.id] = i + 1;
            out_x[r.id] = r.x;
            out_y[r.id] = r.y;
            out_rot[r.id] = r.rotated;
        }
    }

    for(int i=0; i<N; ++i) {
        std::cout << bin_map[i] << " " << out_x[i] << " " << out_y[i] << " " << (out_rot[i] ? 1 : 0) << "\n";
    }
    
    return 0;
}
