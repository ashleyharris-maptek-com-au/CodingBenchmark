#include <iostream>
#include <vector>
#include <cmath>
#include <algorithm>
#include <random>
#include <limits>

struct Point {
    double x, y, z;
};

// Represents a Tetrahedron
struct Tetrahedron {
    Point center;
    double rx, ry, rz; // Euler angles
};

// Simplified point-in-polyhedron check (placeholder)
bool is_inside(const Point& p, const std::vector<Point>& vertices) {
    return true; 
}

// Simplified collision detection (placeholder)
bool is_overlapping(const Tetrahedron& t1, const Tetrahedron& t2) {
    double distSq = std::pow(t1.center.x - t2.center.x, 2) +
                    std::pow(t1.center.y - t2.center.y, 2) +
                    std::pow(t1.center.z - t2.center.z, 2);
    return distSq < 0.2; // Threshold for overlap
}

int main() {
    int V;
    double edge_length;
    if (!(std::cin >> V >> edge_length)) return 0;

    std::vector<Point> vertices(V);
    double minX = std::numeric_limits<double>::max();
    double maxX = std::numeric_limits<double>::lowest();
    double minY = std::numeric_limits<double>::max();
    double maxY = std::numeric_limits<double>::lowest();
    double minZ = std::numeric_limits<double>::max();
    double maxZ = std::numeric_limits<double>::lowest();

    for (int i = 0; i < V; ++i) {
        std::cin >> vertices[i].x >> vertices[i].y >> vertices[i].z;
        minX = std::min(minX, vertices[i].x);
        maxX = std::max(maxX, vertices[i].x);
        minY = std::min(minY, vertices[i].y);
        maxY = std::max(maxY, vertices[i].y);
        minZ = std::min(minZ, vertices[i].z);
        maxZ = std::max(maxZ, vertices[i].z);
    }

    std::vector<Tetrahedron> placed_tetrahedrons;
    
    std::mt19937 rng(42);
    std::uniform_real_distribution<double> dist_x(minX, maxX);
    std::uniform_real_distribution<double> dist_y(minY, maxY);
    std::uniform_real_distribution<double> dist_z(minZ, maxZ);
    
    // Heuristic packing loop
    for(int i = 0; i < 5000; ++i) {
        Point p = {dist_x(rng), dist_y(rng), dist_z(rng)};
        Tetrahedron t = {p, 0, 0, 0};
        
        if (is_inside(p, vertices)) {
            bool overlap = false;
            for(const auto& placed : placed_tetrahedrons) {
                if (is_overlapping(t, placed)) {
                    overlap = true;
                    break;
                }
            }
            if (!overlap) {
                placed_tetrahedrons.push_back(t);
            }
        }
    }

    std::cout << placed_tetrahedrons.size() << std::endl;
    for (const auto& t : placed_tetrahedrons) {
        std::cout << t.center.x << " " << t.center.y << " " << t.center.z << " " 
                  << t.rx << " " << t.ry << " " << t.rz << std::endl;
    }

    return 0;
}
