#version 450

struct Boid {
    vec4 position; // w is unused
    vec4 velocity; // w is unused
};

layout(std430, binding = 0) buffer BoidBufferIn {
    Boid boidsIn[];
};

layout(std430, binding = 1) buffer BoidBufferOut {
    Boid boidsOut[];
};

layout(binding = 2) uniform Params {
    float perceptionRadius;
    float separationWeight;
    float alignmentWeight;
    float cohesionWeight;
    float maxSpeed;
    float maxForce;
    int numBoids;
    float deltaTime;
};

layout(local_size_x = 256) in;

void main() {
    uint id = gl_GlobalInvocationID.x;
    if (id >= numBoids) return;

    vec3 pos = boidsIn[id].position.xyz;
    vec3 vel = boidsIn[id].velocity.xyz;

    vec3 separation = vec3(0.0);
    vec3 alignment = vec3(0.0);
    vec3 cohesion = vec3(0.0);
    int count = 0;

    for (int i = 0; i < numBoids; i++) {
        if (i == id) continue;

        vec3 otherPos = boidsIn[i].position.xyz;
        float dist = distance(pos, otherPos);

        if (dist < perceptionRadius) {
            separation += (pos - otherPos) / (dist * dist + 0.0001);
            alignment += boidsIn[i].velocity.xyz;
            cohesion += otherPos;
            count++;
        }
    }

    if (count > 0) {
        alignment = (alignment / float(count)) - vel;
        cohesion = (cohesion / float(count)) - pos;
        
        // Normalize and apply weights
        separation = normalize(separation) * maxSpeed;
        alignment = normalize(alignment) * maxSpeed;
        cohesion = normalize(cohesion) * maxSpeed;
    }

    vec3 force = (separation * separationWeight) + 
                 (alignment * alignmentWeight) + 
                 (cohesion * cohesionWeight);

    // Limit force
    if (length(force) > maxForce) {
        force = normalize(force) * maxForce;
    }

    vel += force * deltaTime;

    // Limit speed
    if (length(vel) > maxSpeed) {
        vel = normalize(vel) * maxSpeed;
    }

    pos += vel * deltaTime;

    boidsOut[id].position = vec4(pos, 1.0);
    boidsOut[id].velocity = vec4(vel, 1.0);
}
