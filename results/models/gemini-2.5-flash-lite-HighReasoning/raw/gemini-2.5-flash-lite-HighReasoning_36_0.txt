#include <iostream>
#include <vector>
#include <algorithm>
#include <bitset>

// Using a fixed size for bitset for performance, 
// but we can adjust if N is known to be larger.
// Given the constraints, let's assume N <= 1024 for now.
const int MAXN = 1024;

struct Graph {
    int n;
    std::vector<std::bitset<MAXN>> adj;

    Graph(int n) : n(n), adj(n) {}

    void add_edge(int u, int v) {
        adj[u].set(v);
        adj[v].set(u);
    }
};

std::vector<int> max_independent_set;
std::vector<int> current_set;

void solve(const Graph& g, std::bitset<MAXN> candidates) {
    if (candidates.none()) {
        if (current_set.size() > max_independent_set.size()) {
            max_independent_set = current_set;
        }
        return;
    }

    // Pruning: if current_set + remaining candidates <= max_independent_set, return
    if (current_set.size() + candidates.count() <= max_independent_set.size()) {
        return;
    }

    // Heuristic: pick vertex with highest degree in the subgraph induced by candidates
    int v = -1;
    int max_deg = -1;
    for (int i = 0; i < g.n; ++i) {
        if (candidates.test(i)) {
            int deg = (g.adj[i] & candidates).count();
            if (deg > max_deg) {
                max_deg = deg;
                v = i;
            }
        }
    }

    if (v == -1) return;

    // Branch 1: Include v in the independent set
    current_set.push_back(v);
    std::bitset<MAXN> next_candidates = candidates & ~g.adj[v];
    next_candidates.reset(v);
    solve(g, next_candidates);
    current_set.pop_back();

    // Branch 2: Exclude v from the independent set
    candidates.reset(v);
    solve(g, candidates);
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    int n, m;
    if (!(std::cin >> n >> m)) return 0;

    Graph g(n);
    for (int i = 0; i < m; ++i) {
        int u, v;
        std::cin >> u >> v;
        if (u >= 0 && u < n && v >= 0 && v < n) {
            g.add_edge(u, v);
        }
    }

    std::bitset<MAXN> candidates;
    for (int i = 0; i < n; ++i) candidates.set(i);

    solve(g, candidates);

    std::cout << max_independent_set.size() << "\n";
    for (size_t i = 0; i < max_independent_set.size(); ++i) {
        std::cout << max_independent_set[i] << (i == max_independent_set.size() - 1 ? "" : " ");
    }
    std::cout << std::endl;

    return 0;
}
