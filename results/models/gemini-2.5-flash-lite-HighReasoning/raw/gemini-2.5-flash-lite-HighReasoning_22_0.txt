using System;
using System.Collections.Generic;
using System.Linq;

public class TetrisAI
{
    public static void Main(string[] args)
    {
        string inputLine = Console.ReadLine();
        if (string.IsNullOrEmpty(inputLine)) return;
        string[] input = inputLine.Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
        int width = int.Parse(input[0]);
        int height = int.Parse(input[1]);
        int numPieces = int.Parse(input[2]);

        Board board = new Board(width, height);

        for (int i = 0; i < numPieces; i++)
        {
            string pieceType = Console.ReadLine().Trim();
            
            var bestMove = board.GetBestMove(pieceType);
            Console.WriteLine(string.Format("{0} {1}", bestMove.Rotation, bestMove.Column));
            
            string result = Console.ReadLine().Trim();
            board.ApplyMove(pieceType, bestMove);
        }
    }
}

public class Move
{
    public int Rotation { get; set; }
    public int Column { get; set; }
    public double Score { get; set; }
}

public class Board
{
    private int width;
    private int height;
    private int[,] grid;

    public Board(int width, int height)
    {
        this.width = width;
        this.height = height;
        this.grid = new int[height, width];
    }

    public Move GetBestMove(string pieceType)
    {
        Move bestMove = new Move { Score = double.MinValue };
        var rotations = PieceDefinitions.Pieces[pieceType];

        for (int r = 0; r < rotations.Count; r++)
        {
            var rotation = rotations[r];
            for (int col = 0; col <= width - GetPieceWidth(rotation); col++)
            {
                var simulatedGrid = SimulatePlacement(rotation, col);
                if (simulatedGrid != null)
                {
                    double score = CalculateScore(simulatedGrid);
                    if (score > bestMove.Score)
                    {
                        bestMove = new Move { Rotation = r, Column = col, Score = score };
                    }
                }
            }
        }
        return bestMove;
    }

    private int GetPieceWidth(int[,] rotation)
    {
        int maxCol = 0;
        for (int i = 0; i < rotation.GetLength(0); i++)
            if (rotation[i, 1] > maxCol) maxCol = rotation[i, 1];
        return maxCol + 1;
    }

    private int[,] SimulatePlacement(int[,] rotation, int col)
    {
        int[,] newGrid = (int[,])grid.Clone();
        for (int row = 0; row < height; row++)
        {
            if (CanPlace(newGrid, rotation, row, col))
            {
                if (!CanPlace(newGrid, rotation, row + 1, col))
                {
                    Place(newGrid, rotation, row, col);
                    ClearLines(newGrid);
                    return newGrid;
                }
            }
        }
        return null;
    }

    private bool CanPlace(int[,] grid, int[,] rotation, int row, int col)
    {
        for (int i = 0; i < rotation.GetLength(0); i++)
        {
            int r = row + rotation[i, 0];
            int c = col + rotation[i, 1];
            if (r >= height || c < 0 || c >= width || (r >= 0 && grid[r, c] == 1))
                return false;
        }
        return true;
    }

    private void Place(int[,] grid, int[,] rotation, int row, int col)
    {
        for (int i = 0; i < rotation.GetLength(0); i++)
        {
            int r = row + rotation[i, 0];
            int c = col + rotation[i, 1];
            if (r >= 0) grid[r, c] = 1;
        }
    }

    private int ClearLines(int[,] grid)
    {
        int cleared = 0;
        for (int r = height - 1; r >= 0; r--)
        {
            bool full = true;
            for (int c = 0; c < width; c++)
                if (grid[r, c] == 0) full = false;
            if (full)
            {
                cleared++;
                for (int i = r; i > 0; i--)
                    for (int c = 0; c < width; c++)
                        grid[i, c] = grid[i - 1, c];
                for (int c = 0; c < width; c++) grid[0, c] = 0;
                r++; 
            }
        }
        return cleared;
    }

    private double CalculateScore(int[,] grid)
    {
        int aggregateHeight = 0;
        int holes = 0;
        int bumpiness = 0;

        int[] columnHeights = new int[width];
        for (int c = 0; c < width; c++)
        {
            for (int r = 0; r < height; r++)
            {
                if (grid[r, c] == 1)
                {
                    columnHeights[c] = height - r;
                    break;
                }
            }
            aggregateHeight += columnHeights[c];
        }

        for (int c = 0; c < width; c++)
        {
            if (c < width - 1)
                bumpiness += Math.Abs(columnHeights[c] - columnHeights[c + 1]);

            for (int r = height - columnHeights[c]; r < height; r++)
            {
                if (grid[r, c] == 0) holes++;
            }
        }

        return -(aggregateHeight * 0.51) - (holes * 0.35) - (bumpiness * 0.18);
    }

    public void ApplyMove(string pieceType, Move move)
    {
        var rotation = PieceDefinitions.Pieces[pieceType][move.Rotation];
        // For actual placement, we need to find the correct row
        int row = 0;
        while (CanPlace(grid, rotation, row + 1, move.Column)) row++;
        Place(grid, rotation, row, move.Column);
        ClearLines(grid);
    }
}
