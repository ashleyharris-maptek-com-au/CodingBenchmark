use std::collections::VecDeque;
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = lines.next().unwrap().unwrap();
    let dims: Vec<usize> = first_line
        .split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();
    let h = dims[0];
    let w = dims[1];

    let mut grid = Vec::with_capacity(h * w);
    let mut start = 0;
    let mut end = 0;

    for r in 0..h {
        let line = lines.next().unwrap().unwrap();
        for (c, ch) in line.chars().enumerate() {
            if ch == 'A' {
                start = r * w + c;
            } else if ch == 'B' {
                end = r * w + c;
            }
            grid.push(ch as u8);
        }
    }

    let mut parent: Vec<i64> = vec![-1; h * w];
    let mut queue = VecDeque::new();
    queue.push_back(start);
    parent[start] = start as i64;

    let mut found = false;

    while let Some(curr) = queue.pop_front() {
        if curr == end {
            found = true;
            break;
        }

        let r = curr / w;
        let c = curr % w;

        let neighbors = [
            (r as i32 - 1, c as i32),
            (r as i32 + 1, c as i32),
            (r as i32, c as i32 - 1),
            (r as i32, c as i32 + 1),
        ];

        for (nr, nc) in neighbors {
            if nr >= 0 && nr < h as i32 && nc >= 0 && nc < w as i32 {
                let next = nr as usize * w + nc as usize;
                if grid[next] != '#' as u8 && parent[next] == -1 {
                    parent[next] = curr as i64;
                    queue.push_back(next);
                }
            }
        }
    }

    if found {
        let mut path = Vec::new();
        let mut curr = end;
        while curr != start {
            path.push(curr);
            curr = parent[curr] as usize;
        }
        path.push(start);
        path.reverse();

        println!("{}", path.len());
        for p in path {
            println!("{} {}", p / w, p % w);
        }
    }
}
