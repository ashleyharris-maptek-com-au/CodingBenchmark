import sys
import random

def solve_3sat():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    num_vars = int(input_data[0])
    num_clauses = int(input_data[1])
    
    # Store clauses compactly
    clauses = [int(x) for x in input_data[2:]]
    
    # Map variable -> list of clause indices where it appears
    # Clause indices are 0 to num_clauses-1
    vars_to_clauses = [[] for _ in range(num_vars + 1)]
    for i in range(num_clauses):
        for j in range(3):
            lit = clauses[3*i + j]
            vars_to_clauses[abs(lit)].append(i)
            
    # WalkSAT
    assignment = [random.choice([True, False]) for _ in range(num_vars + 1)]
    
    def is_clause_satisfied(clause_idx, assignment):
        for j in range(3):
            lit = clauses[3*clause_idx + j]
            var = abs(lit)
            val = assignment[var]
            if (lit > 0 and val) or (lit < 0 and not val):
                return True
        return False
        
    unsat_clauses = set()
    for i in range(num_clauses):
        if not is_clause_satisfied(i, assignment):
            unsat_clauses.add(i)
            
    max_tries = 3
    max_flips = num_vars * 5
    
    for _ in range(max_tries):
        if not unsat_clauses:
            break
            
        for _ in range(max_flips):
            if not unsat_clauses:
                break
                
            # Pick a random unsatisfied clause
            clause_idx = random.choice(list(unsat_clauses))
            
            # Flip a random variable in the clause to satisfy it (heuristic)
            lit_to_flip = clauses[3*clause_idx + random.randint(0, 2)]
            var_to_flip = abs(lit_to_flip)
            
            assignment[var_to_flip] = not assignment[var_to_flip]
            
            # Update unsat_clauses by checking affected clauses
            for c_idx in vars_to_clauses[var_to_flip]:
                satisfied = is_clause_satisfied(c_idx, assignment)
                if satisfied:
                    unsat_clauses.discard(c_idx)
                else:
                    unsat_clauses.add(c_idx)
                    
        else:
            # Restart
            assignment = [random.choice([True, False]) for _ in range(num_vars + 1)]
            unsat_clauses = set(i for i in range(num_clauses) if not is_clause_satisfied(i, assignment))
            continue
        break
        
    print("SAT")
    print(" ".join(str(int(assignment[i])) for i in range(1, num_vars + 1)))

if __name__ == '__main__':
    solve_3sat()
