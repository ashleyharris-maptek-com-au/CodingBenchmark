; SPIR-V assembly for a geometry shader that shrinks input triangles
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %gl_out %in_color %out_color

; Execution Modes
OpExecutionMode %main Triangles
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 3
OpExecutionMode %main Invocations 1

; Decorations
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %in_color Location 2
OpDecorate %out_color Location 0

; Types
%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_3 = OpConstant %uint 3
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%float_3 = OpConstant %float 3.0
%float_0_7 = OpConstant %float 0.7

; Input structures
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_3
%ptr_arr_pv_in = OpTypePointer Input %arr_pv_in
%ptr_v4float_in = OpTypePointer Input %v4float

; User Input arrays
%arr_v3_in = OpTypeArray %v3float %uint_3
%ptr_arr_v3_in = OpTypePointer Input %arr_v3_in
%ptr_v3float_in = OpTypePointer Input %v3float

; Output structures
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_v4float_out = OpTypePointer Output %v4float
%ptr_v3float_out = OpTypePointer Output %v3float

; Variables
%gl_in = OpVariable %ptr_arr_pv_in Input
%in_color = OpVariable %ptr_arr_v3_in Input
%gl_out = OpVariable %ptr_out_pv Output
%out_color = OpVariable %ptr_v3float_out Output

%func_type = OpTypeFunction %void
%main = OpFunction %void None %func_type
%label = OpLabel

; 1. Load vertex positions
%p0_ptr = OpAccessChain %ptr_v4float_in %gl_in %int_0 %int_0
%p1_ptr = OpAccessChain %ptr_v4float_in %gl_in %int_1 %int_0
%p2_ptr = OpAccessChain %ptr_v4float_in %gl_in %int_2 %int_0
%p0 = OpLoad %v4float %p0_ptr
%p1 = OpLoad %v4float %p1_ptr
%p2 = OpLoad %v4float %p2_ptr

; 2. Compute Centroid C = (p0 + p1 + p2) / 3
%sum01 = OpFAdd %v4float %p0 %p1
%sum012 = OpFAdd %v4float %sum01 %p2
%centroid = OpFDiv %v4float %sum012 %float_3

; Loop 3 times to shrink and emit
; Since output vertices is 3, let's just unroll for simplicity.

; --- Emit Vertex 0 ---
%v0_raw_ptr = OpAccessChain %ptr_v4float_in %gl_in %int_0 %int_0
%v0_raw = OpLoad %v4float %v0_raw_ptr
%v0_rel = OpFSub %v4float %v0_raw %centroid
%v0_scaled = OpFMul %v4float %v0_rel %float_0_7
%v0_new = OpFAdd %v4float %centroid %v0_scaled

%pos_out_ptr = OpAccessChain %ptr_v4float_out %gl_out %int_0
OpStore %pos_out_ptr %v0_new
%color0_ptr = OpAccessChain %ptr_v3float_in %in_color %int_0
%color0 = OpLoad %v3float %color0_ptr
OpStore %out_color %color0
OpEmitVertex

; --- Emit Vertex 1 ---
%v1_raw_ptr = OpAccessChain %ptr_v4float_in %gl_in %int_1 %int_0
%v1_raw = OpLoad %v4float %v1_raw_ptr
%v1_rel = OpFSub %v4float %v1_raw %centroid
%v1_scaled = OpFMul %v4float %v1_rel %float_0_7
%v1_new = OpFAdd %v4float %centroid %v1_scaled

OpStore %pos_out_ptr %v1_new
%color1_ptr = OpAccessChain %ptr_v3float_in %in_color %int_1
%color1 = OpLoad %v3float %color1_ptr
OpStore %out_color %color1
OpEmitVertex

; --- Emit Vertex 2 ---
%v2_raw_ptr = OpAccessChain %ptr_v4float_in %gl_in %int_2 %int_0
%v2_raw = OpLoad %v4float %v2_raw_ptr
%v2_rel = OpFSub %v4float %v2_raw %centroid
%v2_scaled = OpFMul %v4float %v2_rel %float_0_7
%v2_new = OpFAdd %v4float %centroid %v2_scaled

OpStore %pos_out_ptr %v2_new
%color2_ptr = OpAccessChain %ptr_v3float_in %in_color %int_2
%color2 = OpLoad %v3float %color2_ptr
OpStore %out_color %color2
OpEmitVertex

OpEndPrimitive
OpReturn
OpFunctionEnd
