#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <deque>

// Suffix Array construction using O(N log N) prefix doubling
struct SuffixArray {
    std::vector<int> sa, lcp, rank;
    std::vector<int> s;
    int n;

    SuffixArray(const std::vector<int>& input, int max_val) : s(input), n(input.size()) {
        sa.resize(n);
        rank.resize(n);
        std::vector<int> tmp(n);

        for (int i = 0; i < n; ++i) {
            sa[i] = i;
            rank[i] = s[i];
        }

        for (int k = 1; k < n; k <<= 1) {
            auto cmp = [&](int i, int j) {
                if (rank[i] != rank[j]) return rank[i] < rank[j];
                int ri = (i + k < n) ? rank[i + k] : -1;
                int rj = (j + k < n) ? rank[j + k] : -1;
                return ri < rj;
            };
            std::sort(sa.begin(), sa.end(), cmp);
            tmp[sa[0]] = 0;
            for (int i = 1; i < n; ++i) {
                tmp[sa[i]] = tmp[sa[i - 1]] + (cmp(sa[i - 1], sa[i]) ? 1 : 0);
            }
            rank = tmp;
            if (rank[sa[n - 1]] == n - 1) break;
        }

        // Kasai's algorithm for LCP
        lcp.assign(n, 0);
        int h = 0;
        for (int i = 0; i < n; ++i) {
            if (rank[i] > 0) {
                int j = sa[rank[i] - 1];
                if (h > 0) h--;
                while (i + h < n && j + h < n && s[i + h] == s[j + h]) h++;
                lcp[rank[i]] = h;
            }
        }
    }
};

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int n;
    if (!(std::cin >> n)) return 0;
    if (n == 0) return 0;

    std::vector<std::string> strings(n);
    std::vector<int> combined;
    std::vector<int> owner;
    int sentinel = -1;

    for (int i = 0; i < n; ++i) {
        std::cin >> strings[i];
        for (char c : strings[i]) {
            combined.push_back((unsigned char)c + 256); // Offset to avoid conflict with sentinels
            owner.push_back(i);
        }
        combined.push_back(sentinel--);
        owner.push_back(-1);
    }

    if (n == 1) {
        std::cout << strings[0] << std::endl;
        return 0;
    }

    SuffixArray sa_obj(combined, 256 + 256);
    
    // Sliding window to find LCS
    int left = 0;
    std::vector<int> count(n, 0);
    int distinct = 0;
    int max_len = 0;
    int best_pos = -1;

    // dq stores indices of LCP array such that lcp[dq] is monotonic increasing
    std::deque<int> dq; 

    for (int right = 0; right < combined.size(); ++right) {
        if (owner[sa_obj.sa[right]] != -1) {
            if (count[owner[sa_obj.sa[right]]]++ == 0) distinct++;
        }
        
        // When adding new element to right, LCP[right] is the candidate
        if (right > 0) {
            while (!dq.empty() && sa_obj.lcp[dq.back()] >= sa_obj.lcp[right]) dq.pop_back();
            dq.push_back(right);
        }

        while (distinct == n) {
            // Window is [left, right]
            // Valid LCP range for this window is [left+1, right]
            while (!dq.empty() && dq.front() <= left) dq.pop_front();
            
            if (!dq.empty()) {
                int current_min_lcp = sa_obj.lcp[dq.front()];
                if (current_min_lcp > max_len) {
                    max_len = current_min_lcp;
                    best_pos = sa_obj.sa[right];
                }
            }
            
            if (owner[sa_obj.sa[left]] != -1) {
                if (--count[owner[sa_obj.sa[left]]] == 0) distinct--;
            }
            left++;
        }
    }

    if (best_pos != -1) {
        for (int i = 0; i < max_len; ++i) {
            std::cout << (char)(combined[best_pos + i] - 256);
        }
    }
    std::cout << std::endl;
    return 0;
}
