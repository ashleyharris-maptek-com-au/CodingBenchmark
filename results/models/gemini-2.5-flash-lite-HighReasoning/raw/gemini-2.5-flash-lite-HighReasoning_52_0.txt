from autopilot.perception import PerceptionModule
from autopilot.planner import PlanningModule
from autopilot.controller import ControlModule

def main():
    perception = PerceptionModule()
    planner = PlanningModule()
    controller = ControlModule()

    # Simulation loop
    running = True
    while running:
        # 1. Perception
        camera_frame = None # Get from simulation
        lidar_data = None   # Get from simulation
        lane_info = perception.detect_lanes(camera_frame)
        vehicle_data = perception.detect_vehicles(lidar_data)
        
        # 2. Planning
        vehicle_state = {"speed": 0.0, "position": 0.0} # Get from simulation
        maneuvers = planner.plan_maneuver(lane_info, vehicle_data, vehicle_state)
        
        # 3. Control
        actuator_commands = controller.apply_control(maneuvers)
        
        # Send actuator_commands to simulation
        print(f"Applying commands: {actuator_commands}")
        
        # running = False # Set based on simulation stop condition

if __name__ == "__main__":
    main()
