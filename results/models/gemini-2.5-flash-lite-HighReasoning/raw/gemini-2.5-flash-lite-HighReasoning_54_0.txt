import time

class PIDController:
    def __init__(self, kp, ki, kd):
        self.kp = kp
        self.ki = ki
        self.kd = kd
        self.prev_error = 0
        self.integral = 0

    def update(self, error, dt):
        self.integral += error * dt
        derivative = (error - self.prev_error) / dt
        output = self.kp * error + self.ki * self.integral + self.kd * derivative
        self.prev_error = error
        return output

class AutolandController:
    def __init__(self):
        self.state = "APPROACH"
        self.lat_pid = PIDController(0.5, 0.01, 0.1)
        self.vert_pid = PIDController(0.5, 0.01, 0.1)

    def process(self, loc_dev, gs_dev, alt, dt):
        if alt < 50:
            self.state = "FLARE"
        elif alt < 10:
            self.state = "TOUCHDOWN"

        if self.state == "APPROACH":
            roll = self.lat_pid.update(loc_dev, dt)
            pitch = self.vert_pid.update(gs_dev, dt)
            return roll, pitch, 0.5
        elif self.state == "FLARE":
            return 0, -0.1, 0.2
        return 0, 0, 0

# Mock simulation
controller = AutolandController()
for i in range(100):
    roll, pitch, throttle = controller.process(loc_dev=1.0 - i*0.01, gs_dev=1.0 - i*0.01, alt=100 - i, dt=1.0)
    print(f"State: {controller.state}, Roll: {roll:.2f}, Pitch: {pitch:.2f}, Throttle: {throttle:.2f}")
