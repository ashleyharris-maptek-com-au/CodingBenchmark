; SPIR-V
; Version: 1.0
; Generator: manual

               OpCapability Shader
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Fragment %main "main" %in_normal %in_lightDir_placeholder %outColor
               OpExecutionMode %main OriginUpperLeft
               OpSource GLSL 450

; --- Decorations ---
               OpDecorate %outColor Location 0
               OpDecorate %in_normal Location 1
               OpDecorate %ubo Block
               OpMemberDecorate %ubo 0 Offset 0
               OpMemberDecorate %ubo 1 Offset 64
               OpMemberDecorate %ubo 2 Offset 128
               OpMemberDecorate %ubo 3 Offset 192 ; lightPos
               OpMemberDecorate %ubo 4 Offset 208 ; cameraPos
               OpMemberDecorate %ubo 5 Offset 224 ; params
               OpDecorate %ubo_var DescriptorSet 0
               OpDecorate %ubo_var Binding 0

; --- Types ---
       %void = OpTypeVoid
       %func = OpTypeFunction %void
      %float = OpTypeFloat 32
      %v3float = OpTypeVector %float 3
      %v4float = OpTypeVector %float 4
    %mat4float = OpTypeMatrix %v4float 4
       %uint = OpTypeInt 32 0
        %ubo = OpTypeStruct %mat4float %mat4float %mat4float %v4float %v4float %v4float
   %ptr_uniform_ubo = OpTypePointer Uniform %ubo
   %ptr_input_v3 = OpTypePointer Input %v3float
   %ptr_output_v4 = OpTypePointer Output %v4float
   %ptr_uniform_v4 = OpTypePointer Uniform %v4float

; --- Variables ---
   %ubo_var = OpVariable %ptr_uniform_ubo Uniform
   %in_normal = OpVariable %ptr_input_v3 Input
   %outColor = OpVariable %ptr_output_v4 Output
; Assuming lightDir is computed or passed similarly, for simplicity, I'll use a constant or derive from lightPos
   %in_lightDir_placeholder = OpVariable %ptr_input_v3 Input 

; --- Constants ---
   %float_0 = OpConstant %float 0.0
   %float_0_2 = OpConstant %float 0.2
   %float_0_5 = OpConstant %float 0.5
   %float_0_7 = OpConstant %float 0.7
   %float_0_75 = OpConstant %float 0.75
   %float_1_0 = OpConstant %float 1.0
   %uint_3 = OpConstant %uint 3
   %uint_0 = OpConstant %uint 0

%main = OpFunction %void None %func
%label = OpLabel

; Normalize normal
%normal = OpLoad %v3float %in_normal
%norm_normal = OpExtInst %v3float %1 Normalize %normal

; Get Light Dir (assuming lightPos from UBO - cameraPos is for reflection)
%ptr_lightPos = OpAccessChain %ptr_uniform_v4 %ubo_var %uint_3
%lightPos = OpLoad %v4float %ptr_lightPos
%lightDir = OpVectorShuffle %v3float %lightPos %lightPos 0 1 2
%norm_lightDir = OpExtInst %v3float %1 Normalize %lightDir

; Gooch shading: t = dot(N,L) * 0.5 + 0.5
%dotNL = OpDot %float %norm_normal %norm_lightDir
%t = OpFMul %float %dotNL %float_0_5
%t_final = OpFAdd %float %t %float_0_5

; Colors: Cool(0.2, 0.2, 0.75), Warm(0.7, 0.7, 0.4)
%cool = OpConstantComposite %v3float %float_0_2 %float_0_2 %float_0_75
%warm = OpConstantComposite %v3float %float_0_7 %float_0_7 %float_0_4

; Gooch result = mix(cool, warm, t)
; SPIR-V doesn't have a direct mix for vectors in core without GLSL.std.450
%color_res = OpExtInst %v3float %1 FMix %cool %warm %t_final

; Output alpha 1.0
%out_v4 = OpCompositeConstruct %v4float %color_res %float_1_0
               OpStore %outColor %out_v4
               OpReturn
               OpFunctionEnd
