; SPIR-V Assembly Geometry Shader: Point to Diamond
OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %gl_out %in_color %out_color
OpExecutionMode %main InputPoints
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1

; Decorate Interface
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_in Input
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %gl_out Output
OpDecorate %in_color Location 2
OpDecorate %out_color Location 0

; Types
%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0
%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_v3 = OpTypePointer Output %v3float

%int_0 = OpConstant %int 0
%uint_1 = OpConstant %uint 1
%float_h = OpConstant %float 0.05
%float_neg_h = OpConstant %float -0.05

; Interface Structures
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_1
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

%arr_in_color = OpTypeArray %v3float %uint_1
%ptr_in_arr_v3 = OpTypePointer Input %arr_in_color
%in_color = OpVariable %ptr_in_arr_v3 Input
%out_color = OpVariable %ptr_out_v3 Output

; Main Function
%func_void_v4_v3 = OpTypeFunction %void %v4float %v3float
%func_main = OpTypeFunction %void
%main = OpFunction %void None %func_main
%label_entry = OpLabel

; Load input data
%ptr_in_pos = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%pos = OpLoad %v4float %ptr_in_pos
%x = OpCompositeExtract %float %pos 0
%y = OpCompositeExtract %float %pos 1
%z = OpCompositeExtract %float %pos 2
%w = OpCompositeExtract %float %pos 3
%ptr_color_in = OpAccessChain %ptr_in_v3 %in_color %int_0
%color = OpLoad %v3float %ptr_color_in

; Helper for emitting
%emit_v = OpFunction %void None %func_void_v4_v3
%v_pos = OpFunctionParameter %v4float
%v_color = OpFunctionParameter %v3float
%label_emit = OpLabel
%ptr_out_pos = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out_pos %v_pos
OpStore %out_color %v_color
OpEmitVertex
OpReturn
OpFunctionEnd

; Emit Top: (x, y+h)
%y_top = OpFAdd %float %y %float_h
%pos_top = OpCompositeConstruct %v4float %x %y_top %z %w
OpFunctionCall %void %emit_v %pos_top %color

; Emit Right: (x+h, y)
%x_right = OpFAdd %float %x %float_h
%pos_right = OpCompositeConstruct %v4float %x_right %y %z %w
OpFunctionCall %void %emit_v %pos_right %color

; Emit Left: (x-h, y)
%x_left = OpFAdd %float %x %float_neg_h
%pos_left = OpCompositeConstruct %v4float %x_left %y %z %w
OpFunctionCall %void %emit_v %pos_left %color

; Emit Bottom: (x, y-h)
%y_bot = OpFAdd %float %y %float_neg_h
%pos_bot = OpCompositeConstruct %v4float %x %y_bot %z %w
OpFunctionCall %void %emit_v %pos_bot %color

OpEndPrimitive
OpReturn
OpFunctionEnd
