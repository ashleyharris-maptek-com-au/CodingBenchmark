import sys

def solve():
    try:
        input_data = sys.stdin.read().splitlines()
        if not input_data:
            return
        
        first_line = input_data[0].split()
        if not first_line:
            return
        
        U = int(first_line[0])
        S = int(first_line[1])
        
        sets = []
        for i in range(S):
            sets.append(set(map(int, input_data[i+1].split())))
            
        # matrix[element] = set of row indices that cover this element
        matrix = {e: set() for e in range(U)}
        for i, s in enumerate(sets):
            for e in s:
                if e in matrix:
                    matrix[e].add(i)
        
        solution = []
        
        def backtrack():
            if not matrix:
                return True
            
            # Choose column with fewest rows
            col = min(matrix, key=lambda c: len(matrix[c]))
            if not matrix[col]:
                return False
            
            for row_idx in list(matrix[col]):
                solution.append(row_idx)
                
                # Elements covered by this row
                covered_elements = sets[row_idx]
                
                # Rows to remove
                rows_to_remove = set()
                for e in covered_elements:
                    rows_to_remove.update(matrix[e])
                
                # Backup state
                removed_cols = {}
                for e in covered_elements:
                    removed_cols[e] = matrix.pop(e)
                
                for r in rows_to_remove:
                    if r == row_idx: continue
                    for e in sets[r]:
                        if e in matrix:
                            matrix[e].discard(r)
                
                if backtrack():
                    return True
                
                # Restore state
                for e, rows in removed_cols.items():
                    matrix[e] = rows
                for r in rows_to_remove:
                    if r == row_idx: continue
                    for e in sets[r]:
                        if e in matrix:
                            matrix[e].add(r)
                            
                solution.pop()
            
            return False

        if backtrack():
            print("SOLUTION")
            print(*(sorted(solution)))
        else:
            print("NO SOLUTION")
            
    except Exception:
        print("NO SOLUTION")

if __name__ == "__main__":
    solve()
