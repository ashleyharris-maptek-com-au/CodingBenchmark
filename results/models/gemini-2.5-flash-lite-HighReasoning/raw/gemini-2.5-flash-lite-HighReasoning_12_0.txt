using System;
using System.Collections.Generic;
using System.Numerics;

namespace BinPacking3D
{
    // Basic representation of a 3D vertex
    public record Vertex(double X, double Y, double Z);

    // Basic representation of a 3D face (indices into vertex list)
    public record Face(int A, int B, int C);

    // Representation of a Polyhedron
    public class Polyhedron
    {
        public List<Vertex> Vertices { get; } = new();
        public List<Face> Faces { get; } = new();
        public Vector3 Min { get; private set; }
        public Vector3 Max { get; private set; }

        public Polyhedron(List<Vertex> vertices, List<Face> faces)
        {
            Vertices = vertices;
            Faces = faces;
            CalculateAABB();
        }

        private void CalculateAABB()
        {
            float minX = float.MaxValue, minY = float.MaxValue, minZ = float.MaxValue;
            float maxX = float.MinValue, maxY = float.MinValue, maxZ = float.MinValue;
            foreach (var v in Vertices)
            {
                minX = Math.Min(minX, (float)v.X);
                minY = Math.Min(minY, (float)v.Y);
                minZ = Math.Min(minZ, (float)v.Z);
                maxX = Math.Max(maxX, (float)v.X);
                maxY = Math.Max(maxY, (float)v.Y);
                maxZ = Math.Max(maxZ, (float)v.Z);
            }
            Min = new Vector3(minX, minY, minZ);
            Max = new Vector3(maxX, maxY, maxZ);
        }

        public Vector3 Size => Max - Min;
    }

    public record Placement(Vector3 Translation, Quaternion Rotation);

    // Solver class to hold the logic
    public class Solver
    {
        public void Solve()
        {
            var (polyhedron, containerDims) = ReadInput();
            
            var placements = new List<Placement>();
            var placedItems = new List<(Placement, Polyhedron)>();

            // Greedy grid-based packing
            float stepX = polyhedron.Size.X;
            float stepY = polyhedron.Size.Y;
            float stepZ = polyhedron.Size.Z;

            for (float x = 0; x + polyhedron.Size.X <= containerDims.X; x += stepX)
            {
                for (float y = 0; y + polyhedron.Size.Y <= containerDims.Y; y += stepY)
                {
                    for (float z = 0; z + polyhedron.Size.Z <= containerDims.Z; z += stepZ)
                    {
                        var pos = new Vector3(x, y, z);
                        var p = new Placement(pos, Quaternion.Identity);
                        
                        // Check collision with already placed items
                        bool collision = false;
                        foreach (var (existingP, existingPoly) in placedItems)
                        {
                            // Basic AABB overlap check
                            var min1 = existingP.Translation + existingPoly.Min;
                            var max1 = existingP.Translation + existingPoly.Max;
                            var min2 = p.Translation + polyhedron.Min;
                            var max2 = p.Translation + polyhedron.Max;

                            if (min1.X < max2.X && max1.X > min2.X &&
                                min1.Y < max2.Y && max1.Y > min2.Y &&
                                min1.Z < max2.Z && max1.Z > min2.Z)
                            {
                                collision = true;
                                break;
                            }
                        }

                        if (!collision)
                        {
                            placements.Add(p);
                            placedItems.Add((p, polyhedron));
                        }
                    }
                }
            }

            // Print output
            Console.WriteLine(placements.Count);
            foreach (var p in placements)
            {
                Console.WriteLine($"{p.Translation.X} {p.Translation.Y} {p.Translation.Z} {p.Rotation.W} {p.Rotation.X} {p.Rotation.Y} {p.Rotation.Z}");
            }
        }

        private (Polyhedron, Vector3) ReadInput()
        {
            // Keeping the implementation from previous turn
            string[] line1 = Console.ReadLine()!.Split();
            int V = int.Parse(line1[0]);
            int F = int.Parse(line1[1]);

            var vertices = new List<Vertex>();
            for (int i = 0; i < V; i++)
            {
                string[] v = Console.ReadLine()!.Split();
                vertices.Add(new Vertex(double.Parse(v[0]), double.Parse(v[1]), double.Parse(v[2])));
            }

            var faces = new List<Face>();
            for (int i = 0; i < F; i++)
            {
                string[] f = Console.ReadLine()!.Split();
                faces.Add(new Face(int.Parse(f[0]), int.Parse(f[1]), int.Parse(f[2])));
            }

            string[] container = Console.ReadLine()!.Split();
            var containerDims = new Vector3((float)double.Parse(container[0]), (float)double.Parse(container[1]), (float)double.Parse(container[2]));

            return (new Polyhedron(vertices, faces), containerDims);
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            new Solver().Solve();
        }
    }
}
