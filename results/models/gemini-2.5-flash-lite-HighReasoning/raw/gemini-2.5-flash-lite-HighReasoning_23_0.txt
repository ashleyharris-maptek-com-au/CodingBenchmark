use std::io::{self, BufRead, Write};
use std::sync::atomic::{AtomicBool, Ordering};
use std::sync::Arc;
use std::thread;
use std::time::{Duration, Instant};

// Configuration & State structures
struct Environment {
    width: usize,
    height: usize,
    gravity: f64,
    max_thrust: f64,
    fuel: f64,
    target_x: f64,
    target_y: f64,
}

struct State {
    x: f64,
    y: f64,
    vx: f64,
    vy: f64,
    angle: f64,
    angular_vel: f64,
    fuel_remaining: f64,
}

// ... (rest of imports)

fn parse_initial_setup(handle: &mut io::StdinLock) -> (Environment, (f64, f64), (f64, f64)) {
    let mut line = String::new();
    handle.read_line(&mut line).unwrap();
    let parts: Vec<f64> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
    let env = Environment { width: parts[0] as usize, height: parts[1] as usize, gravity: parts[2], max_thrust: parts[3], fuel: parts[4], target_x: 0.0, target_y: 0.0 };
    
    line.clear();
    handle.read_line(&mut line).unwrap();
    let coords: Vec<f64> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
    
    (env, (coords[0], coords[1]), (coords[2], coords[3]))
}

fn parse_state(line: &str) -> State {
    let parts: Vec<f64> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
    State { x: parts[0], y: parts[1], vx: parts[2], vy: parts[3], angle: parts[4], angular_vel: parts[5], fuel_remaining: parts[6] }
}

fn compute_control(state: &State, env: &Environment, target_x: f64, target_y: f64) -> (f64, f64) {
    // Basic Proportional Control for Y velocity and X position
    let target_vy = if state.y < target_y - 10.0 { 2.0 } else { 0.5 };
    let thrust = if state.vy > target_vy { 1.0 } else { 0.0 };
    let turn = (target_x - state.x) * 0.01; // Very basic turning
    
    (thrust.clamp(0.0, 1.0), turn.clamp(-1.0, 1.0))
}

fn main() {
    let stdin = io::stdin();
    let mut handle = stdin.lock();
    
    let (env, start, target) = parse_initial_setup(&mut handle);
    let mut line = String::new();

    loop {
        line.clear();
        if handle.read_line(&mut line).is_err() || line.is_empty() {
            break;
        }
        
        let state = parse_state(&line);
        let (thrust, turn) = compute_control(&state, &env, target.0, target.1);

        println!("{:.3} {:.3}", thrust, turn);
        io::stdout().flush().unwrap();
    }
}
