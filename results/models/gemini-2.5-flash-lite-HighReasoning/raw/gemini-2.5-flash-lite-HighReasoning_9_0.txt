using System;
using System.Collections.Generic;
using System.Linq;
using System.IO;

public class Solution
{
    struct Cut
    {
        public int Index;
        public int Length;
    }

    class Stock
    {
        public int RemainingCapacity;
        public List<int> CutIndices = new List<int>();

        public Stock(int capacity)
        {
            RemainingCapacity = capacity;
        }
    }

    public static void Main(string[] args)
    {
        string input = Console.In.ReadToEnd();
        string[] tokens = input.Split(new[] { ' ', '\r', '\n', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        if (tokens.Length < 2) return;
        
        int n = int.Parse(tokens[0]);
        int stockLength = int.Parse(tokens[1]);

        Cut[] cuts = new Cut[n];
        for (int i = 0; i < n; i++)
        {
            cuts[i] = new Cut { Index = i, Length = int.Parse(tokens[2 + i]) };
        }

        // FFD: Sort cuts descending
        Array.Sort(cuts, (a, b) => b.Length.CompareTo(a.Length));

        List<Stock> stocks = new List<Stock>();

        foreach (var cut in cuts)
        {
            bool placed = false;
            foreach (var stock in stocks)
            {
                if (stock.RemainingCapacity >= cut.Length)
                {
                    stock.RemainingCapacity -= cut.Length;
                    stock.CutIndices.Add(cut.Index);
                    placed = true;
                    break;
                }
            }

            if (!placed)
            {
                Stock newStock = new Stock(stockLength);
                newStock.RemainingCapacity -= cut.Length;
                newStock.CutIndices.Add(cut.Index);
                stocks.Add(newStock);
            }
        }

        Console.WriteLine(stocks.Count);
        foreach (var stock in stocks)
        {
            stock.CutIndices.Sort();
            Console.WriteLine(string.Join(" ", stock.CutIndices));
        }
    }
}
