use std::collections::{HashSet, VecDeque};

#[derive(Debug, Clone)]
struct Graph {
    vertices: HashSet<usize>,
    edges: HashSet<(usize, usize)>,
}

impl Graph {
    fn new() -> Self {
        Graph {
            vertices: HashSet::new(),
            edges: HashSet::new(),
        }
    }

    fn add_edge(&mut self, u: usize, v: usize) {
        let (min, max) = if u < v { (u, v) } else { (v, u) };
        self.vertices.insert(min);
        self.vertices.insert(max);
        self.edges.insert((min, max));
    }

    fn remove_vertex(&self, v: usize) -> Self {
        let mut new_vertices = self.vertices.clone();
        new_vertices.remove(&v);

        let mut new_edges = HashSet::new();
        for &(u1, u2) in &self.edges {
            if u1 != v && u2 != v {
                new_edges.insert((u1, u2));
            }
        }
        Graph {
            vertices: new_vertices,
            edges: new_edges,
        }
    }

    fn remove_vertices(&self, vertices_to_remove: &[usize]) -> Self {
        let mut new_vertices = self.vertices.clone();
        let mut new_edges = HashSet::new();

        for &v in vertices_to_remove {
            new_vertices.remove(&v);
        }

        for &(u1, u2) in &self.edges {
            if !vertices_to_remove.contains(&u1) && !vertices_to_remove.contains(&u2) {
                new_edges.insert((u1, u2));
            }
        }
        Graph {
            vertices: new_vertices,
            edges: new_edges,
        }
    }

    fn neighbors(&self, v: usize) -> HashSet<usize> {
        let mut neighbors = HashSet::new();
        for &(u1, u2) in &self.edges {
            if u1 == v {
                neighbors.insert(u2);
            } else if u2 == v {
                neighbors.insert(u1);
            }
        }
        neighbors
    }

    fn get_highest_degree_vertex(&self) -> Option<usize> {
        let mut degree_map = std::collections::HashMap::new();
        for &(u, v) in &self.edges {
            *degree_map.entry(u).or_insert(0) += 1;
            *degree_map.entry(v).or_insert(0) += 1;
        }

        degree_map
            .into_iter()
            .max_by_key(|&(_, count)| count)
            .map(|(v, _)| v)
    }
}

fn solve_mvc(graph: &Graph) -> HashSet<usize> {
    if graph.edges.is_empty() {
        return HashSet::new();
    }

    let v = graph.get_highest_degree_vertex().unwrap();

    // Case 1: v is in the cover
    let mut res1 = solve_mvc(&graph.remove_vertex(v));
    res1.insert(v);

    // Case 2: v is NOT in the cover (all neighbors must be)
    let neighbors = graph.neighbors(v);
    let mut vertices_to_remove = vec![v];
    vertices_to_remove.extend(neighbors.iter());

    let mut res2 = solve_mvc(&graph.remove_vertices(&vertices_to_remove));
    res2.extend(neighbors.iter());

    if res1.len() < res2.len() {
        res1
    } else {
        res2
    }
}

fn main() {
    let mut graph = Graph::new();
    graph.add_edge(0, 1);
    graph.add_edge(1, 2);
    graph.add_edge(2, 3);
    graph.add_edge(3, 0);
    graph.add_edge(0, 2);

    let mvc = solve_mvc(&graph);
    println!("Minimum Vertex Cover: {:?}", mvc);
}
