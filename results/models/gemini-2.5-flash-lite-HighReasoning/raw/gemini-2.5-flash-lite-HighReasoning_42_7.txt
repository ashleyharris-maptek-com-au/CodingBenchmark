; SPIR-V Assembly Geometry Shader
; Explodes triangles along face normal

OpCapability Geometry
OpCapability Shader
OpExtInstImport %glsl "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %gl_out %outColor %inColor
OpExecutionMode %main Triangles
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 3
OpExecutionMode %main Invocations 1

; Debug names
OpName %gl_PerVertex_in "gl_PerVertex"
OpMemberName %gl_PerVertex_in 0 "gl_Position"
OpName %gl_in "gl_in"
OpName %gl_PerVertex_out "gl_PerVertex"
OpMemberName %gl_PerVertex_out 0 "gl_Position"
OpName %gl_out "gl_out"
OpName %outColor "outColor"
OpName %inColor "inColor"
OpName %main "main"

; Types and constants
%void = OpTypeVoid
%void_func = OpTypeFunction %void
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%bool = OpTypeBool
%int = OpTypeInt 32 0
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3
%float_offset = OpConstant %float 0.08
%ptr_int_function = OpTypePointer Function %int

; Interface variables
%gl_PerVertex_in = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position

%arr_pv_in = OpTypeArray %gl_PerVertex_in 3
%ptr_arr_pv_in = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_arr_pv_in Input

%gl_PerVertex_out = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

%ptr_pv_out = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_pv_out Output

%arr_inColor = OpTypeArray %v3float 3
%ptr_arr_inColor = OpTypePointer Input %arr_inColor
%inColor = OpVariable %ptr_arr_inColor Input

%ptr_v3_out = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_v3_out Output

%ptr_v4 = OpTypePointer Input %v4float
%ptr_v3 = OpTypePointer Input %v3float
%ptr_out_v4 = OpTypePointer Output %v4float

%main = OpFunction %void None %void_func
%label_entry = OpLabel

; Load positions
%p0_ptr = OpAccessChain %ptr_v4 %gl_in %int_0 %int_0
%p0 = OpLoad %v4float %p0_ptr
%p1_ptr = OpAccessChain %ptr_v4 %gl_in %int_1 %int_0
%p1 = OpLoad %v4float %p1_ptr
%p2_ptr = OpAccessChain %ptr_v4 %gl_in %int_2 %int_0
%p2 = OpLoad %v4float %p2_ptr

; Convert to vec3 for normal calculation
%p0_v3 = OpVectorShuffle %v3float %p0 %p0 0 1 2
%p1_v3 = OpVectorShuffle %v3float %p1 %p1 0 1 2
%p2_v3 = OpVectorShuffle %v3float %p2 %p2 0 1 2

; E1 = p1 - p0
%e1 = OpFSub %v3float %p1_v3 %p0_v3
; E2 = p2 - p0
%e2 = OpFSub %v3float %p2_v3 %p0_v3

; Cross product
%e1x = OpCompositeExtract %float %e1 0
%e1y = OpCompositeExtract %float %e1 1
%e1z = OpCompositeExtract %float %e1 2
%e2x = OpCompositeExtract %float %e2 0
%e2y = OpCompositeExtract %float %e2 1
%e2z = OpCompositeExtract %float %e2 2

%t1 = OpFMul %float %e1y %e2z
%t2 = OpFMul %float %e1z %e2y
%cx = OpFSub %float %t1 %t2

%t3 = OpFMul %float %e1z %e2x
%t4 = OpFMul %float %e1x %e2z
%cy = OpFSub %float %t3 %t4

%t5 = OpFMul %float %e1x %e2y
%t6 = OpFMul %float %e1y %e2x
%cz = OpFSub %float %t5 %t6

%normal = OpCompositeConstruct %v3float %cx %cy %cz

; Normalize normal
%normal_n = OpExtInst %v3float %glsl 69 %normal

; Offset = N * 0.08
%offset = OpVectorTimesScalar %v3float %normal_n %float_offset

; Loop for 3 vertices
%i = OpVariable %ptr_int_function %int_0
OpBranch %loop_start

%loop_start = OpLabel
%i_val = OpLoad %int %i
%cond = OpULessThan %bool %i_val %int_3
OpLoopMerge %loop_end %loop_continue None
OpBranchConditional %cond %loop_body %loop_end

%loop_body = OpLabel
; Get position pI
%ptr_pos = OpAccessChain %ptr_v4 %gl_in %i_val %int_0
%pI = OpLoad %v4float %ptr_pos
%pI_v3 = OpVectorShuffle %v3float %pI %pI 0 1 2
%pI_new_v3 = OpFAdd %v3float %pI_v3 %offset
%pI_new_v4 = OpCompositeInsert %v4float %pI_new_v3 %pI 0 1 2 ; Keep original w
; Store gl_Position
%pos_out_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %pos_out_ptr %pI_new_v4

; Load color
%ptr_col = OpAccessChain %ptr_v3 %inColor %i_val
%col = OpLoad %v3float %ptr_col
OpStore %outColor %col

OpEmitVertex
OpBranch %loop_continue

%loop_continue = OpLabel
%next_i = OpIAdd %int %i_val %int_1
OpStore %i %next_i
OpBranch %loop_start

%loop_end = OpLabel
OpEndPrimitive
OpReturn
OpFunctionEnd
