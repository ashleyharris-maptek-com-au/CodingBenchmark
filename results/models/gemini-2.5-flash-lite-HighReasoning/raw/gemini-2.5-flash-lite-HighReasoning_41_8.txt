; SPIR-V
; Version: 1.0
; Generator: manual
OpCapability Shader
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %worldPos %normal %color %outColor %ubo
OpExecutionMode %main OriginUpperLeft

%GLSL450 = OpExtInstImport "GLSL.std.450"

; Decorations
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 4
OpDecorate %outColor Location 0

OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0
OpDecorate %UBO Block
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224

; Types
%void = OpTypeVoid
%func = OpTypeFunction %void
%float = OpTypeFloat 32
%bool = OpTypeBool
%vec3 = OpTypeVector %float 3
%vec4 = OpTypeVector %float 4
%mat4 = OpTypeMatrix %vec4 4
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0
%UBO = OpTypeStruct %mat4 %mat4 %mat4 %vec4 %vec4 %vec4
%ptr_uniform_UBO = OpTypePointer Uniform %UBO
%ptr_input_vec3 = OpTypePointer Input %vec3
%ptr_input_vec4 = OpTypePointer Input %vec4
%ptr_output_vec4 = OpTypePointer Output %vec4

; Variables
%ubo = OpVariable %ptr_uniform_UBO Uniform
%worldPos = OpVariable %ptr_input_vec3 Input
%normal = OpVariable %ptr_input_vec3 Input
%color = OpVariable %ptr_input_vec3 Input
%outColor = OpVariable %ptr_output_vec4 Output

%main = OpFunction %void None %func
%label = OpLabel

; Load Inputs
%norm_val = OpLoad %vec3 %normal
%pos_val = OpLoad %vec3 %worldPos
%col_val = OpLoad %vec3 %color

; Get LightPos
%ptr_light = OpAccessChain %ptr_input_vec4 %ubo %int 3
%lightPos = OpLoad %vec4 %ptr_light
%lightPos3 = OpVectorShuffle %vec3 %lightPos %lightPos 0 1 2

; Cel shading logic
; d = max(dot(N, L), 0)
%L = OpVectorMinus %vec3 %lightPos3 %pos_val
%L_norm = OpExtInst %vec3 %GLSL450 Normalize %L
%N_norm = OpExtInst %vec3 %GLSL450 Normalize %norm_val
%dot = OpDot %float %N_norm %L_norm
%zero = OpConstant %float 0.0
%d = OpExtInst %float %GLSL450 FMax %dot %zero

; Thresholds/Bands
%t1 = OpConstant %float 0.75
%t2 = OpConstant %float 0.5
%t3 = OpConstant %float 0.25
%v1 = OpConstant %float 1.0
%v2 = OpConstant %float 0.7
%v3 = OpConstant %float 0.4
%v4 = OpConstant %float 0.2

; Band logic
%c1 = OpFOrdGreaterThan %bool %d %t1
%c2 = OpFOrdGreaterThan %bool %d %t2
%c3 = OpFOrdGreaterThan %bool %d %t3

; Selection
%res1 = OpSelect %float %c1 %v1 %v2
%res2 = OpSelect %float %c2 %res1 %v3
%band = OpSelect %float %c3 %res2 %v4

; Final color
%finalCol = OpVectorTimesScalar %vec3 %col_val %band
%alpha = OpConstant %float 1.0
%finalVec4 = OpCompositeConstruct %vec4 %finalCol %alpha

OpStore %outColor %finalVec4
OpReturn
OpFunctionEnd
