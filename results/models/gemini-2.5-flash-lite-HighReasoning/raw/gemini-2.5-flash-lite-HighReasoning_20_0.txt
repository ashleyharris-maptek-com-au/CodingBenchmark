use std::io::{self, BufRead};

struct Sample {
    depth: f64,
    measurements: Vec<f64>,
}

struct Drillhole {
    id: String,
    start: [f64; 3],
    direction: [f64; 3],
    _length: f64,
    samples: Vec<Sample>,
}

struct Suspect {
    hole_id: String,
    sample_idx: usize,
    property_idx: usize,
    confidence: f64,
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = lines.next().unwrap().unwrap();
    let parts: Vec<&str> = first_line.split_whitespace().collect();
    let num_holes: usize = parts[0].parse().unwrap();
    let num_properties: usize = parts[1].parse().unwrap();

    let property_names: Vec<String> = lines.next().unwrap().unwrap()
        .split_whitespace().map(|s| s.to_string()).collect();

    let mut drillholes = Vec::with_capacity(num_holes);

    for _ in 0..num_holes {
        let hole_info = lines.next().unwrap().unwrap();
        let h: Vec<&str> = hole_info.split_whitespace().collect();
        let id = h[0].to_string();
        let start = [h[1].parse().unwrap(), h[2].parse().unwrap(), h[3].parse().unwrap()];
        let direction = [h[4].parse().unwrap(), h[5].parse().unwrap(), h[6].parse().unwrap()];
        let _length = h[7].parse().unwrap();
        let num_samples: usize = h[8].parse().unwrap();

        let mut samples = Vec::with_capacity(num_samples);
        for _ in 0..num_samples {
            let sample_line = lines.next().unwrap().unwrap();
            let s: Vec<&str> = sample_line.split_whitespace().collect();
            let depth = s[0].parse().unwrap();
            let measurements: Vec<f64> = s[1..].iter().map(|&x| x.parse().unwrap()).collect();
            samples.push(Sample { depth, measurements });
        }

        drillholes.push(Drillhole { id, start, direction, _length, samples });
    }

    let suspects = detect_outliers(&drillholes, num_properties);
    
    println!("{}", suspects.len());
    for s in suspects {
        println!("{} {} {} {:.4}", s.hole_id, s.sample_idx, property_names[s.property_idx], s.confidence);
    }
}

fn detect_outliers(drillholes: &[Drillhole], num_properties: usize) -> Vec<Suspect> {
    let mut suspects = Vec::new();
    
    // Calculate global statistics for each property
    let mut sums = vec![0.0; num_properties];
    let mut counts = vec![0.0; num_properties];
    
    for hole in drillholes {
        for sample in &hole.samples {
            for (i, &val) in sample.measurements.iter().enumerate() {
                sums[i] += val;
                counts[i] += 1.0;
            }
        }
    }
    
    let means: Vec<f64> = sums.iter().zip(counts.iter()).map(|(&s, &c)| s / c).collect();
    
    // Identify suspects as those significantly deviating from the mean (e.g., > 3 std devs)
    // Note: A more robust approach would use spatial neighbors, but Z-score is a good start.
    
    for (h_idx, hole) in drillholes.iter().enumerate() {
        for (s_idx, sample) in hole.samples.iter().enumerate() {
            for (p_idx, &val) in sample.measurements.iter().enumerate() {
                let diff = (val - means[p_idx]).abs();
                if diff > 10.0 { // Arbitrary threshold for "obvious" outlier
                    suspects.push(Suspect {
                        hole_id: hole.id.clone(),
                        sample_idx: s_idx,
                        property_idx: p_idx,
                        confidence: diff / 10.0, // Simplistic confidence
                    });
                }
            }
        }
    }
    
    suspects.sort_by(|a, b| b.confidence.partial_cmp(&a.confidence).unwrap());
    suspects
}
