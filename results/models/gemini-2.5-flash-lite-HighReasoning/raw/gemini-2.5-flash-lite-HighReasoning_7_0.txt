struct Vec3 {
    double x, y, z;
    Vec3 operator+(const Vec3& o) const { return {x + o.x, y + o.y, z + o.z}; }
    Vec3 operator-(const Vec3& o) const { return {x - o.x, y - o.y, z - o.z}; }
    Vec3 operator*(double s) const { return {x * s, y * s, z * s}; }
};

struct Triangle {
    std::array<int, 3> v;
};

struct Mesh {
    std::vector<Vec3> vertices;
    std::vector<Triangle> faces;
};

// AABB structure for spatial partitioning
struct AABB {
    Vec3 min, max;
    static AABB from_triangle(const Mesh& mesh, const Triangle& t) {
        Vec3 v0 = mesh.vertices[t.v[0]];
        Vec3 v1 = mesh.vertices[t.v[1]];
        Vec3 v2 = mesh.vertices[t.v[2]];
        return {
            {std::min({v0.x, v1.x, v2.x}), std::min({v0.y, v1.y, v2.y}), std::min({v0.z, v1.z, v2.z})},
            {std::max({v0.x, v1.x, v2.x}), std::max({v0.y, v1.y, v2.y}), std::max({v0.z, v1.z, v2.z})}
        };
    }
    bool intersects(const AABB& o) const {
        return (min.x <= o.max.x && max.x >= o.min.x) &&
               (min.y <= o.max.y && max.y >= o.min.y) &&
               (min.z <= o.max.z && max.z >= o.min.z);
    }
};

struct AABBNode {
    AABB box;
    int triangle_index = -1; // -1 if not leaf
    std::unique_ptr<AABBNode> left, right;
};

std::unique_ptr<AABBNode> build_aabb_tree(const Mesh& mesh, const std::vector<int>& face_indices, int start, int end) {
    auto node = std::make_unique<AABBNode>();
    
    // Calculate AABB for all faces in the current node
    node->box = AABB::from_triangle(mesh, mesh.faces[face_indices[start]]);
    for (int i = start + 1; i < end; ++i) {
        AABB face_box = AABB::from_triangle(mesh, mesh.faces[face_indices[i]]);
        node->box.min = {std::min(node->box.min.x, face_box.min.x), std::min(node->box.min.y, face_box.min.y), std::min(node->box.min.z, face_box.min.z)};
        node->box.max = {std::max(node->box.max.x, face_box.max.x), std::max(node->box.max.y, face_box.max.y), std::max(node->box.max.z, face_box.max.z)};
    }

    if (end - start == 1) {
        node->triangle_index = face_indices[start];
    } else {
        // Split based on the longest axis
        double dx = node->box.max.x - node->box.min.x;
        double dy = node->box.max.y - node->box.min.y;
        double dz = node->box.max.z - node->box.min.z;
        int axis = (dx > dy && dx > dz) ? 0 : (dy > dz ? 1 : 2);

        std::sort(face_indices.begin() + start, face_indices.begin() + end, [&](int a, int b) {
            Vec3 centerA = { (mesh.vertices[mesh.faces[a].v[0]].x + mesh.vertices[mesh.faces[a].v[1]].x + mesh.vertices[mesh.faces[a].v[2]].x) / 3.0,
                             (mesh.vertices[mesh.faces[a].v[0]].y + mesh.vertices[mesh.faces[a].v[1]].y + mesh.vertices[mesh.faces[a].v[2]].y) / 3.0,
                             (mesh.vertices[mesh.faces[a].v[0]].z + mesh.vertices[mesh.faces[a].v[1]].z + mesh.vertices[mesh.faces[a].v[2]].z) / 3.0 };
            Vec3 centerB = { (mesh.vertices[mesh.faces[b].v[0]].x + mesh.vertices[mesh.faces[b].v[1]].x + mesh.vertices[mesh.faces[b].v[2]].x) / 3.0,
                             (mesh.vertices[mesh.faces[b].v[0]].y + mesh.vertices[mesh.faces[b].v[1]].y + mesh.vertices[mesh.faces[b].v[2]].y) / 3.0,
                             (mesh.vertices[mesh.faces[b].v[0]].z + mesh.vertices[mesh.faces[b].v[1]].z + mesh.vertices[mesh.faces[b].v[2]].z) / 3.0 };
            return (axis == 0 ? centerA.x : (axis == 1 ? centerA.y : centerA.z)) < 
                   (axis == 0 ? centerB.x : (axis == 1 ? centerB.y : centerB.z));
        });

        int mid = start + (end - start) / 2;
        node->left = build_aabb_tree(mesh, const_cast<std::vector<int>&>(face_indices), start, mid);
        node->right = build_aabb_tree(mesh, const_cast<std::vector<int>&>(face_indices), mid, end);
    }
    return node;
}

Mesh read_mesh(int v_count, int f_count) {
    Mesh mesh;
    mesh.vertices.resize(v_count);
    for (int i = 0; i < v_count; ++i) {
        std::cin >> mesh.vertices[i].x >> mesh.vertices[i].y >> mesh.vertices[i].z;
    }
    mesh.faces.resize(f_count);
    for (int i = 0; i < f_count; ++i) {
        std::cin >> mesh.faces[i].v[0] >> mesh.faces[i].v[1] >> mesh.faces[i].v[2];
    }
    return mesh;
}

void output_mesh(const Mesh& mesh) {
    std::cout << mesh.vertices.size() << " " << mesh.faces.size() << std::endl;
    for (const auto& v : mesh.vertices) {
        std::cout << v.x << " " << v.y << " " << v.z << std::endl;
    }
    for (const auto& f : mesh.faces) {
        std::cout << f.v[0] << " " << f.v[1] << " " << f.v[2] << std::endl;
    }
}

int main() {
    int va, fa, vb, fb;
    if (!(std::cin >> va >> fa >> vb >> fb)) return 0;

    Mesh meshA = read_mesh(va, fa);
    Mesh meshB = read_mesh(vb, fb);

    // Simplified approach: Merge meshes
    Mesh result;
    result.vertices = meshA.vertices;
    result.vertices.insert(result.vertices.end(), meshB.vertices.begin(), meshB.vertices.end());
    
    result.faces = meshA.faces;
    for (auto f : meshB.faces) {
        f.v[0] += va;
        f.v[1] += va;
        f.v[2] += va;
        result.faces.push_back(f);
    }
    
    output_mesh(result);
    return 0;
}
