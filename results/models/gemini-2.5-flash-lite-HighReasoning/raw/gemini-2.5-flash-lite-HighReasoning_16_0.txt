#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>

struct Task {
    int id;
    int machine;
    int duration;
    int start_time;
    int end_time;
};

struct Job {
    int id;
    std::vector<Task> tasks;
    int next_task_idx = 0;
};

// Simple greedy scheduling logic
void solve() {
    int num_jobs, num_machines;
    if (!(std::cin >> num_jobs >> num_machines)) return;

    std::vector<Job> jobs(num_jobs);
    for (int i = 0; i < num_jobs; ++i) {
        jobs[i].id = i;
        int num_tasks;
        std::cin >> num_tasks;
        for (int j = 0; j < num_tasks; ++j) {
            int m, d;
            std::cin >> m >> d;
            jobs[i].tasks.push_back({j, m, d, -1, -1});
        }
    }

    std::vector<int> machine_free_time(num_machines, 0);
    std::vector<int> job_next_available_time(num_jobs, 0);

    // Using a priority queue or simple loop to schedule tasks.
    // Given the "extreme" scale requirement, a simple greedy approach:
    // Always pick the next available task for each job that can start earliest.
    
    // For simplicity and speed in this version, iterate until all tasks scheduled.
    int tasks_scheduled = 0;
    int total_tasks = 0;
    for(const auto& job : jobs) total_tasks += job.tasks.size();

    while (tasks_scheduled < total_tasks) {
        int best_job_idx = -1;
        int earliest_start = 2e9;

        for (int i = 0; i < num_jobs; ++i) {
            if (jobs[i].next_task_idx < jobs[i].tasks.size()) {
                Task& t = jobs[i].tasks[jobs[i].next_task_idx];
                int machine_ready = machine_free_time[t.machine];
                int job_ready = job_next_available_time[i];
                int start = std::max(machine_ready, job_ready);
                
                if (start < earliest_start) {
                    earliest_start = start;
                    best_job_idx = i;
                }
            }
        }

        if (best_job_idx != -1) {
            Task& t = jobs[best_job_idx].tasks[jobs[best_job_idx].next_task_idx];
            t.start_time = earliest_start;
            t.end_time = t.start_time + t.duration;
            machine_free_time[t.machine] = t.end_time;
            job_next_available_time[best_job_idx] = t.end_time;
            jobs[best_job_idx].next_task_idx++;
            tasks_scheduled++;
        }
    }

    // Output
    for (int i = 0; i < num_jobs; ++i) {
        for (int j = 0; j < jobs[i].tasks.size(); ++j) {
            std::cout << jobs[i].tasks[j].start_time << " " << jobs[i].tasks[j].machine;
            if (j < jobs[i].tasks.size() - 1) std::cout << " ";
        }
        std::cout << std::endl;
    }
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    solve();
    return 0;
}
