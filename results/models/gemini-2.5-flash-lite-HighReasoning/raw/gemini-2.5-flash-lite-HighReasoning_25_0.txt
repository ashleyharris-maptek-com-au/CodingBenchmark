using System;
using System.Collections.Generic;

public class Vector3
{
    public double X, Y, Z;
    public Vector3(double x, double y, double z) { X = x; Y = y; Z = z; }
}

public class Body
{
    public string Name;
    public double Mass;
    public Vector3 Position;
    public Vector3 Velocity;
}

public class Scenario
{
    public int NumBodies;
    public double WarningDays;
    public double ImpactorMass;
    public double DeltaVBudget;
    public double AsteroidMass;
    public double T0UnixUtc;
    public List<Body> Bodies;
    public string De421FilePath;
}

class Program
{
    static void Main(string[] args)
    {
        string[] line1 = Console.ReadLine().Split(' ');
        Scenario scenario = new Scenario
        {
            NumBodies = int.Parse(line1[0]),
            WarningDays = double.Parse(line1[1]),
            ImpactorMass = double.Parse(line1[2]),
            DeltaVBudget = double.Parse(line1[3]),
            AsteroidMass = double.Parse(line1[4]),
            T0UnixUtc = double.Parse(line1[5]),
            Bodies = new List<Body>()
        };

        for (int i = 0; i < scenario.NumBodies; i++)
        {
            string[] line = Console.ReadLine().Split(' ');
            scenario.Bodies.Add(new Body
            {
                Name = line[0],
                Mass = double.Parse(line[1]),
                Position = new Vector3(double.Parse(line[2]), double.Parse(line[3]), double.Parse(line[4])),
                Velocity = new Vector3(double.Parse(line[5]), double.Parse(line[6]), double.Parse(line[7]))
            });
        }
        scenario.De421FilePath = Console.ReadLine();
        
        Console.WriteLine($"Initialized with {scenario.NumBodies} bodies.");
        
        // Find Earth and Asteroid
        Body earth = scenario.Bodies.Find(b => b.Name == "Earth");
        Body asteroid = scenario.Bodies.Find(b => b.Name == "Asteroid");
        
        // Calculate Burn (simplified)
        Vector3 burn = TrajectoryOptimizer.SolveInitialBurn(earth.Position, asteroid.Position, scenario.WarningDays);
        
        // Output format
        Console.WriteLine("1");
        Console.WriteLine($"0 {burn.X} {burn.Y} {burn.Z}");
    }
}
