#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>
#include <limits>
#include <set>

using namespace std;

const long long INF = numeric_limits<long long>::max() / 2;

struct Edge {
    int to;
    long long weight;
};

void get_edges(int mask, int v, const vector<vector<long long>>& dp, const vector<vector<Edge>>& adj, set<pair<int, int>>& result_edges) {
    if (dp[mask][v] == 0) return;

    // Check if it came from a split
    for (int submask = (mask - 1) & mask; submask > 0; submask = (submask - 1) & mask) {
        if (dp[mask][v] == dp[submask][v] + dp[mask ^ submask][v]) {
            get_edges(submask, v, dp, adj, result_edges);
            get_edges(mask ^ submask, v, dp, adj, result_edges);
            return;
        }
    }

    // Otherwise, it came from a neighbor via Dijkstra
    for (const auto& edge : adj[v]) {
        if (dp[mask][v] == dp[mask][edge.to] + edge.weight) {
            int u = edge.to;
            int min_u = min(u, v);
            int max_u = max(u, v);
            result_edges.insert({min_u, max_u});
            get_edges(mask, u, dp, adj, result_edges);
            return;
        }
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int num_vertices, num_edges, num_terminals;
    if (!(cin >> num_vertices >> num_edges >> num_terminals)) return 0;

    vector<vector<Edge>> adj(num_vertices);
    for (int i = 0; i < num_edges; ++i) {
        int u, v;
        long long w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }

    vector<int> terminals(num_terminals);
    for (int i = 0; i < num_terminals; ++i) {
        cin >> terminals[i];
    }

    vector<vector<long long>> dp(1 << num_terminals, vector<long long>(num_vertices, INF));

    for (int i = 0; i < num_terminals; ++i) {
        dp[1 << i][terminals[i]] = 0;
    }

    for (int mask = 1; mask < (1 << num_terminals); ++mask) {
        for (int submask = (mask - 1) & mask; submask > 0; submask = (submask - 1) & mask) {
            for (int v = 0; v < num_vertices; ++v) {
                dp[mask][v] = min(dp[mask][v], dp[submask][v] + dp[mask ^ submask][v]);
            }
        }

        priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;
        for (int v = 0; v < num_vertices; ++v) {
            if (dp[mask][v] < INF) {
                pq.push({dp[mask][v], v});
            }
        }

        while (!pq.empty()) {
            long long d = pq.top().first;
            int u = pq.top().second;
            pq.pop();

            if (d > dp[mask][u]) continue;

            for (auto& edge : adj[u]) {
                if (dp[mask][edge.to] > dp[mask][u] + edge.weight) {
                    dp[mask][edge.to] = dp[mask][u] + edge.weight;
                    pq.push({dp[mask][edge.to], edge.to});
                }
            }
        }
    }

    long long min_weight = INF;
    int root = -1;
    for (int v = 0; v < num_vertices; ++v) {
        if (dp[(1 << num_terminals) - 1][v] < min_weight) {
            min_weight = dp[(1 << num_terminals) - 1][v];
            root = v;
        }
    }

    set<pair<int, int>> result_edges;
    get_edges((1 << num_terminals) - 1, root, dp, adj, result_edges);

    cout << min_weight << " " << result_edges.size() << endl;
    for (auto const& edge : result_edges) {
        cout << edge.first << " " << edge.second << endl;
    }

    return 0;
}
