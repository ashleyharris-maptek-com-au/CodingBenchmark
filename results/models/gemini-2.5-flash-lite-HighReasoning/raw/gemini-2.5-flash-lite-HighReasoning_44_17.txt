#version 450

layout(location = 0) in vec3 worldPos;
layout(location = 1) in vec3 normal;
layout(location = 2) in vec2 uv;
layout(location = 3) in vec3 tangent;
layout(location = 4) in vec3 color;

layout(set = 0, binding = 0) uniform UBO {
    mat4 model;
    mat4 view;
    mat4 projection;
    vec4 lightPos;
    vec4 cameraPos;
    vec4 params;
} ubo;

layout(location = 0) out vec4 outColor;

void main() {
    // 1. Setup
    vec3 N = normalize(normal);
    vec3 L = normalize(ubo.lightPos.xyz - worldPos);
    
    // 2. Procedural Brick Pattern
    // UV scale: 8x16
    vec2 brickUV = uv * vec2(8.0, 16.0);
    
    // Staggered rows: offset every other row
    if (int(brickUV.y) % 2 == 1) {
        brickUV.x += 0.5;
    }
    
    vec2 brickGrid = floor(brickUV);
    vec2 localUV = fract(brickUV);
    
    // Mortar settings
    float mortarX = 0.05;
    float mortarY = 0.1;
    
    vec3 brickColor = vec3(0.7, 0.2, 0.1);
    vec3 mortarColor = vec3(0.7, 0.7, 0.7);
    
    // Determine if in mortar
    bool isMortar = (localUV.x < mortarX) || (localUV.x > 1.0 - mortarX) ||
                    (localUV.y < mortarY) || (localUV.y > 1.0 - mortarY);
                    
    vec3 finalAlbedo = isMortar ? mortarColor : brickColor;
    
    // 3. Lighting
    float diffuse = max(dot(N, L), 0.0);
    float ambient = 0.4;
    float shading = ambient + 0.6 * diffuse;
    
    outColor = vec4(finalAlbedo * shading, 1.0);
}
