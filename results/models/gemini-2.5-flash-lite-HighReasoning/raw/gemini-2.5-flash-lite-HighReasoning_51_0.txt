from autopilot import PIDController

class FMS:
    def __init__(self):
        # Initialize PID controllers for pitch, roll, and yaw
        # Gains would typically be tuned for the specific aircraft dynamics
        self.pitch_pid = PIDController(kp=1.0, ki=0.1, kd=0.05)
        self.roll_pid = PIDController(kp=1.0, ki=0.1, kd=0.05)
        self.yaw_pid = PIDController(kp=1.0, ki=0.1, kd=0.05)
        self.mode = "STANDBY"

    def set_altitude_hold(self, target_altitude):
        self.pitch_pid.setpoint = target_altitude
        self.mode = "ALT_HOLD"

    def set_heading_hold(self, target_heading):
        self.roll_pid.setpoint = target_heading
        self.mode = "HDG_HOLD"

    def get_control_commands(self, current_state, dt):
        pitch_cmd = self.pitch_pid.update(current_state['altitude'], dt)
        roll_cmd = self.roll_pid.update(current_state['heading'], dt)
        yaw_cmd = self.yaw_pid.update(current_state['heading'], dt)
        return {"pitch": pitch_cmd, "roll": roll_cmd, "yaw": yaw_cmd}
