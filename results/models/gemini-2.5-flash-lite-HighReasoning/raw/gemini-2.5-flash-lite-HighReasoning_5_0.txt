use std::io::{self, BufRead};

#[derive(Clone, Copy, PartialEq, Eq, Debug)]
struct Point {
    x: i32,
    y: i32,
}

struct Grid {
    width: i32,
    height: i32,
    obstacles: Vec<Vec<bool>>,
    visited: Vec<Vec<bool>>,
    total_cells: i32,
    non_obstacle_cells: i32,
}

impl Grid {
    fn new(width: i32, height: i32, obstacles: &[(i32, i32)]) -> Self {
        let mut obs = vec![vec![false; height as usize]; width as usize];
        let mut non_obstacle_cells = width * height;
        for &(x, y) in obstacles {
            if x >= 0 && x < width && y >= 0 && y < height {
                if !obs[x as usize][y as usize] {
                    obs[x as usize][y as usize] = true;
                    non_obstacle_cells -= 1;
                }
            }
        }
        Grid {
            width,
            height,
            obstacles: obs,
            visited: vec![vec![false; height as usize]; width as usize],
            total_cells: width * height,
            non_obstacle_cells,
        }
    }

    fn is_valid(&self, x: i32, y: i32) -> bool {
        x >= 0 && x < self.width && y >= 0 && y < self.height && !self.obstacles[x as usize][y as usize] && !self.visited[x as usize][y as usize]
    }

    fn get_neighbors(&self, x: i32, y: i32) -> Vec<Point> {
        let mut neighbors = Vec::new();
        let dirs = [(0, 1), (0, -1), (1, 0), (-1, 0)];
        for (dx, dy) in dirs {
            let nx = x + dx;
            let ny = y + dy;
            if self.is_valid(nx, ny) {
                neighbors.push(Point { x: nx, y: ny });
            }
        }
        neighbors
    }

    // Warnsdorff's heuristic: choose the neighbor with the fewest available neighbors
    fn get_neighbors_with_heuristic(&self, x: i32, y: i32) -> Vec<Point> {
        let mut neighbors = self.get_neighbors(x, y);
        neighbors.sort_by_key(|p| self.get_neighbors(p.x, p.y).len());
        neighbors
    }

    fn solve(&mut self, current: Point, path: &mut Vec<Point>, require_cycle: bool) -> bool {
        path.push(current);
        self.visited[current.x as usize][current.y as usize] = true;

        if path.len() as i32 == self.non_obstacle_cells {
            if require_cycle {
                let start = Point { x: 0, y: 0 };
                let last = *path.last().unwrap();
                let dist = (last.x - start.x).abs() + (last.y - start.y).abs();
                if dist == 1 {
                    return true;
                }
            } else {
                return true;
            }
        }

        for neighbor in self.get_neighbors_with_heuristic(current.x, current.y) {
            if self.solve(neighbor, path, require_cycle) {
                return true;
            }
        }

        self.visited[current.x as usize][current.y as usize] = false;
        path.pop();
        false
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    if let Some(Ok(line)) = lines.next() {
        let parts: Vec<i32> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
        let width = parts[0];
        let height = parts[1];
        let obstacle_count = parts[2];
        let require_cycle = parts[3] == 1;

        let mut obstacles = Vec::new();
        for _ in 0..obstacle_count {
            if let Some(Ok(line)) = lines.next() {
                let parts: Vec<i32> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
                obstacles.push((parts[0], parts[1]));
            }
        }

        let mut grid = Grid::new(width, height, &obstacles);
        let mut path = Vec::new();

        if grid.solve(Point { x: 0, y: 0 }, &mut path, require_cycle) {
            println!("{}", path.len());
            for p in path {
                println!("{} {}", p.x, p.y);
            }
        } else {
            println!("0");
        }
    }
}
