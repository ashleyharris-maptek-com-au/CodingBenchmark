const fs = require('fs');

interface Robot {
  id: number;
  x: number;
  y: number;
}

interface Job {
  id: number;
  x: number;
  y: number;
  priority: number;
}

interface InputData {
  grid: string[];
  robots: Robot[];
  jobs: Job[];
}

function solve() {
  const input: string = fs.readFileSync(0, 'utf8');
  const data: InputData = JSON.parse(input);
  const { grid, robots, jobs } = data;

  const rows = grid.length;
  const cols = grid[0].length;

  // BFS to find shortest path distances
  function getDistances(startX: number, startY: number): number[][] {
    const dists: number[][] = Array.from({ length: rows }, () => Array(cols).fill(-1));
    const queue: [number, number, number][] = [[startX, startY, 0]];
    dists[startY][startX] = 0;

    while (queue.length > 0) {
      const [x, y, d] = queue.shift()!;
      
      const neighbors = [
        [x + 1, y], [x - 1, y], [x, y + 1], [x, y - 1]
      ];

      for (const [nx, ny] of neighbors) {
        if (nx >= 0 && nx < cols && ny >= 0 && ny < rows && grid[ny][nx] === '.' && dists[ny][nx] === -1) {
          dists[ny][nx] = d + 1;
          queue.push([nx, ny, d + 1]);
        }
      }
    }
    return dists;
  }

  // Calculate distances from all robots to all jobs
  const robotDistances = robots.map(r => getDistances(r.x, r.y));

  // Greedily assign jobs based on priority and proximity
  const assignments: number[][] = Array.from({ length: robots.length }, () => []);
  const unassignedJobs = new Set(jobs);

  while (unassignedJobs.size > 0) {
    let bestJob: Job | null = null;
    let bestRobotIdx = -1;
    let minDistance = Infinity;

    for (const job of unassignedJobs) {
      for (let i = 0; i < robots.length; i++) {
        const dist = robotDistances[i][job.y][job.x];
        if (dist !== -1) {
          // Heuristic: prioritize high priority and short distance
          // Simple greedy: pick job that is closest to any robot
          if (dist < minDistance) {
            minDistance = dist;
            bestJob = job;
            bestRobotIdx = i;
          }
        }
      }
    }

    if (bestJob) {
      assignments[bestRobotIdx].push(bestJob.id);
      unassignedJobs.delete(bestJob);
    } else {
      // Should not happen if all jobs are reachable
      break;
    }
  }

  process.stdout.write(JSON.stringify({ assignments }));
}

solve();
