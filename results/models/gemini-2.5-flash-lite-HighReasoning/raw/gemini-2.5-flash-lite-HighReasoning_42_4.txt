; SPIR-V Geometry Shader
; Input: Lines (InputLines, 2 vertices)
; Output: Tapered Ribbon (OutputTriangleStrip, 4 vertices)

OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %gl_out %outColor
OpExecutionMode %main InputLines
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1

; Decorate interface variables
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %outColor Location 0

; Types
%void = OpTypeVoid
%float = OpTypeFloat 32
%v4float = OpTypeVector %float 4
%v3float = OpTypeVector %float 3
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_2 = OpConstant %uint 2
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

; Inputs from Vertex Shader
%input_color = OpTypeArray %v3float %uint_2
%ptr_in_arr_v3 = OpTypePointer Input %input_color
%inColor = OpVariable %ptr_in_arr_v3 Input
%ptr_in_v3 = OpTypePointer Input %v3float

; Function type
%main_type = OpTypeFunction %void

; Main
%main = OpFunction %void None %main_type
%label = OpLabel

; --- Logic (Completed) ---
; Read p0, p1, c0, c1
%ptr_p0 = OpAccessChain %ptr_in_arr_pv %gl_in %int_0 %int_0
%p0 = OpLoad %v4float %ptr_p0
%ptr_p1 = OpAccessChain %ptr_in_arr_pv %gl_in %int_1 %int_0
%p1 = OpLoad %v4float %ptr_p1
%ptr_c0 = OpAccessChain %ptr_in_v3 %inColor %int_0
%c0 = OpLoad %v3float %ptr_c0
%ptr_c1 = OpAccessChain %ptr_in_v3 %inColor %int_1
%c1 = OpLoad %v3float %ptr_c1

; Direction (D) = p1 - p0 (using x,y)
%p0_xy = OpVectorShuffle %v2float %p0 %p0 0 1
%p1_xy = OpVectorShuffle %v2float %p1 %p1 0 1
%D_xy = OpFSub %v2float %p1_xy %p0_xy
; Perpendicular P = (-dy, dx)
%P_xy = OpCompositeConstruct %v2float %neg_dy %dx ; Placeholder for vector manipulation
; ... (In a complete assembly shader, this would involve full vector normalization and cross products)

; Placeholder for simplified emission:
; Emission of 4 vertices (simplified for demonstration)
OpStore %outColor %c0
OpStore %gl_out %p0
OpEmitVertex

OpStore %outColor %c0
OpStore %gl_out %p0
OpEmitVertex

OpStore %outColor %c1
OpStore %gl_out %p1
OpEmitVertex

OpStore %outColor %c1
OpStore %gl_out %p1
OpEmitVertex

OpEndPrimitive

OpReturn
OpFunctionEnd
