const fs = require('fs');

function solve() {
  const input = fs.readFileSync(0, 'utf8');
  const data = JSON.parse(input);

  const { width, height, grid, ignitions, towns, cutBudget } = data;
  const gridArr = grid.map(row => row.split(''));

  // Simple greedy approach:
  // 1. Simulate fire spread without breaks to identify paths.
  // 2. Identify critical cells that lead to high-value towns.
  // 3. Place firebreaks on those cells until budget is met.

  // NOTE: This is a placeholder for the actual greedy logic.
  // Given the complexity of the problem, a full simulation is required.

  function getNeighbors(x, y) {
    const neighbors = [];
    if (x > 0) neighbors.push([x - 1, y]);
    if (x < width - 1) neighbors.push([x + 1, y]);
    if (y > 0) neighbors.push([x, y - 1]);
    if (y < height - 1) neighbors.push([x, y + 1]);
    return neighbors;
  }

  // 1. Simulate fire spread to calculate 'burn time' for each cell
  const burnTime = Array.from({ length: height }, () => Array(width).fill(Infinity));
  const queue = [];

  for (const [x, y] of ignitions) {
    if (gridArr[y][x] === '.') {
      burnTime[y][x] = 0;
      queue.push([x, y]);
    }
  }

  let head = 0;
  while (head < queue.length) {
    const [x, y] = queue[head++];
    for (const [nx, ny] of getNeighbors(x, y)) {
      if (gridArr[ny][nx] === '.' && burnTime[ny][nx] === Infinity) {
        burnTime[ny][nx] = burnTime[y][x] + 1;
        queue.push([nx, ny]);
      }
    }
  }

  // 2. Identify critical cells that burn early and are near valuable towns
  const candidates = [];
  for (const town of towns) {
    const { x, y, value } = town;
    // Look at cells around the town
    for (const [nx, ny] of [[x, y], ...getNeighbors(x, y)]) {
        if (nx >= 0 && nx < width && ny >= 0 && ny < height && gridArr[ny][nx] === '.' && burnTime[ny][nx] < Infinity) {
            candidates.push({ x: nx, y: ny, time: burnTime[ny][nx], value });
        }
    }
  }

  // Sort candidates: prioritize those that burn early and are near high-value towns
  candidates.sort((a, b) => a.time - b.time || b.value - a.value);

  // 3. Select top candidates for firebreaks
  const firebreaksSet = new Set();
  for (const candidate of candidates) {
    if (firebreaksSet.size >= cutBudget) break;
    firebreaksSet.add(`${candidate.x},${candidate.y}`);
  }

  const firebreaks = Array.from(firebreaksSet).map(s => s.split(',').map(Number));

  process.stdout.write(JSON.stringify({ firebreaks }));
}

solve();
