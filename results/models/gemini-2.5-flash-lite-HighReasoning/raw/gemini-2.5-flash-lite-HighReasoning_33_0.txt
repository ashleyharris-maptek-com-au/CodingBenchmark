using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;

namespace QAPSolver
{
    class Program
    {
        static void Main(string[] args)
        {
            var watch = Stopwatch.StartNew();
            
            // 1. Read Input
            int n;
            if (!int.TryParse(Console.ReadLine(), out n)) return;

            int[,] flow = new int[n, n];
            int[,] dist = new int[n, n];

            for (int i = 0; i < n; i++)
            {
                string[] line = Console.ReadLine().Split(new[] { ' ', '	' }, StringSplitOptions.RemoveEmptyEntries);
                for (int j = 0; j < n; j++) flow[i, j] = int.Parse(line[j]);
            }

            for (int i = 0; i < n; i++)
            {
                string[] line = Console.ReadLine().Split(new[] { ' ', '	' }, StringSplitOptions.RemoveEmptyEntries);
                for (int j = 0; j < n; j++) dist[i, j] = int.Parse(line[j]);
            }

            // 2. Initial Solution (Random)
            int[] p = new int[n];
            for (int i = 0; i < n; i++) p[i] = i;
            Random rand = new Random(42);
            for (int i = n - 1; i > 0; i--)
            {
                int j = rand.Next(i + 1);
                int temp = p[i];
                p[i] = p[j];
                p[j] = temp;
            }

            long currentCost = CalculateCost(n, flow, dist, p);

            // 3. Simple Local Search (Hill Climbing / Swap)
            bool improved = true;
            while (improved && watch.ElapsedMilliseconds < 250000) // 250s limit
            {
                improved = false;
                for (int i = 0; i < n; i++)
                {
                    for (int j = i + 1; j < n; j++)
                    {
                        long delta = CalculateDelta(n, flow, dist, p, i, j);
                        if (delta < 0)
                        {
                            // Swap
                            int temp = p[i];
                            p[i] = p[j];
                            p[j] = temp;
                            currentCost += delta;
                            improved = true;
                        }
                    }
                }
            }

            // 4. Output
            Console.WriteLine(currentCost);
            Console.WriteLine(string.Join(" ", p));
        }

        static long CalculateCost(int n, int[,] flow, int[,] dist, int[] p)
        {
            long cost = 0;
            for (int i = 0; i < n; i++)
            {
                for (int j = 0; j < n; j++)
                {
                    cost += (long)flow[i, j] * dist[p[i], p[j]];
                }
            }
            return cost;
        }

        static long CalculateDelta(int n, int[,] flow, int[,] dist, int[] p, int i, int j)
        {
            // Delta cost of swapping p[i] and p[j]
            long delta = 0;
            
            // Pre-calculate terms
            // This is simplified; true delta evaluation is more complex but faster.
            // For N=2000, simple cost re-evaluation might be too slow if done too often.
            // But this is a basic implementation to start.
            
            int pi = p[i];
            int pj = p[j];

            // Recomputing full cost is O(N^2), delta update is O(N).
            // A truly optimized delta update is necessary for N=2000.
            
            // Simple swap delta update:
            // Delta = (flow[i][i] * dist[pj][pj] + flow[i][j] * dist[pj][pi] + flow[j][i] * dist[pi][pj] + flow[j][j] * dist[pi][pi]) 
            //         - (flow[i][i] * dist[pi][pi] + flow[i][j] * dist[pi][pj] + flow[j][i] * dist[pj][pi] + flow[j][j] * dist[pj][pj])
            // Plus interactions with all other k != i, j.
            
            long deltaCost = 0;
            for (int k = 0; k < n; k++)
            {
                if (k == i || k == j) continue;
                deltaCost += (long)flow[i, k] * (dist[pj, p[k]] - dist[pi, p[k]]) +
                             (long)flow[k, i] * (dist[p[k], pj] - dist[p[k], pi]) +
                             (long)flow[j, k] * (dist[pi, p[k]] - dist[pj, p[k]]) +
                             (long)flow[k, j] * (dist[p[k], pi] - dist[p[k], pj]);
            }
            deltaCost += (long)flow[i, j] * (dist[pj, pi] - dist[pi, pj]) +
                         (long)flow[j, i] * (dist[pi, pj] - dist[pj, pi]);
            return deltaCost;
        }
    }
}
