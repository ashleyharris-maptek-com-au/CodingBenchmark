using System;
using System.Collections.Generic;
using System.Linq;

public struct Box
{
    public int Id;
    public int W, H, D;
    public int X, Y, Z;
}

public struct Space
{
    public int X, Y, Z, W, H, D;
}

public class Packer
{
    private int containerW, containerH, containerD;
    private List<Space> freeSpaces = new List<Space>();
    private List<Box> packedBoxes = new List<Box>();

    public Packer(int w, int h, int d)
    {
        containerW = w;
        containerH = h;
        containerD = d;
        freeSpaces.Add(new Space { X = 0, Y = 0, Z = 0, W = w, H = h, D = d });
    }

    public void Pack(List<Box> boxes)
    {
        // Sort by volume descending
        var sortedBoxes = boxes.OrderByDescending(b => (long)b.W * b.H * b.D).ToList();

        foreach (var box in sortedBoxes)
        {
            for (int i = 0; i < freeSpaces.Count; i++)
            {
                var space = freeSpaces[i];
                if (box.W <= space.W && box.H <= space.H && box.D <= space.D)
                {
                    // Place box
                    var placedBox = new Box
                    {
                        Id = box.Id,
                        W = box.W, H = box.H, D = box.D,
                        X = space.X, Y = space.Y, Z = space.Z
                    };
                    packedBoxes.Add(placedBox);

                    // Split space
                    freeSpaces.RemoveAt(i);
                    SplitSpace(space, placedBox);
                    break;
                }
            }
        }
    }

    private void SplitSpace(Space space, Box box)
    {
        // Simple split into 3 new spaces
        if (space.W > box.W)
            freeSpaces.Add(new Space { X = space.X + box.W, Y = space.Y, Z = space.Z, W = space.W - box.W, H = box.H, D = box.D });
        if (space.H > box.H)
            freeSpaces.Add(new Space { X = space.X, Y = space.Y + box.H, Z = space.Z, W = space.W, H = space.H - box.H, D = box.D });
        if (space.D > box.D)
            freeSpaces.Add(new Space { X = space.X, Y = space.Y, Z = space.Z + box.D, W = space.W, H = space.H, D = space.D - box.D });
    }

    public void PrintResult()
    {
        Console.WriteLine(packedBoxes.Count);
        foreach (var b in packedBoxes)
        {
            Console.WriteLine(string.Format("{0} {1} {2} {3}", b.Id, b.X, b.Y, b.Z));
        }
    }
}

class Program
{
    static void Main()
    {
        string line = Console.ReadLine();
        if (string.IsNullOrEmpty(line)) return;
        var parts = line.Split(' ');
        int n = int.Parse(parts[0]);
        int w = int.Parse(parts[1]);
        int h = int.Parse(parts[2]);
        int d = int.Parse(parts[3]);

        var boxes = new List<Box>();
        for (int i = 0; i < n; i++)
        {
            var dims = Console.ReadLine().Split(' ').Select(int.Parse).ToArray();
            boxes.Add(new Box { Id = i, W = dims[0], H = dims[1], D = dims[2] });
        }

        var packer = new Packer(w, h, d);
        packer.Pack(boxes);
        packer.PrintResult();
    }
}
