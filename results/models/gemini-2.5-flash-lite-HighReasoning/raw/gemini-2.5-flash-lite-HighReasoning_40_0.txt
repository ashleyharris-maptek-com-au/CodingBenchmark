#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

// Maximum Weight Independent Set problem:
// Maximize sum(c_i * x_i) subject to x_u + x_v <= 1 for all edges (u, v).
// Since x_u + x_v <= 1 means we cannot select both u and v, this is equivalent
// to finding the Maximum Weight Independent Set in the graph defined by these constraints.

struct Solver {
    int n, m;
    vector<long long> c;
    vector<vector<int>> adj;
    
    long long max_weight = -1;
    vector<int> best_x;
    
    vector<int> current_x;
    long long current_weight = 0;

    void solve(int idx) {
        if (idx == n) {
            if (current_weight > max_weight) {
                max_weight = current_weight;
                best_x = current_x;
            }
            return;
        }

        // Try NOT selecting variable idx
        current_x[idx] = 0;
        solve(idx + 1);

        // Try selecting variable idx (if allowed)
        bool can_select = true;
        for (int neighbor : adj[idx]) {
            if (neighbor < idx && current_x[neighbor] == 1) {
                can_select = false;
                break;
            }
        }

        if (can_select) {
            current_x[idx] = 1;
            current_weight += c[idx];
            solve(idx + 1);
            current_weight -= c[idx];
            current_x[idx] = 0;
        }
    }
};

int main() {
    int n, m;
    if (!(cin >> n >> m)) return 0;

    vector<long long> c(n);
    for (int i = 0; i < n; ++i) cin >> c[i];

    vector<vector<int>> adj(n);
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    Solver solver;
    solver.n = n;
    solver.m = m;
    solver.c = c;
    solver.adj = adj;
    solver.current_x.assign(n, 0);
    solver.best_x.assign(n, 0);

    solver.solve(0);

    cout << solver.max_weight << endl;
    for (int i = 0; i < n; ++i) {
        cout << solver.best_x[i] << (i == n - 1 ? "" : " ");
    }
    cout << endl;

    return 0;
}
