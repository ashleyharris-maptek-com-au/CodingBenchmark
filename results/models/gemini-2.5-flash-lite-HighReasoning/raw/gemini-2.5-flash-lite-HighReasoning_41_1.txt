; SPIR-V
; Version: 1.0
; Generator: None
; Bound: 100
; Schema: 0

OpCapability Shader
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %inNormal %outColor
OpExecutionMode %main OriginUpperLeft

; Decorations
OpDecorate %inNormal Location 1
OpDecorate %outColor Location 0

; Types
%void = OpTypeVoid
%float = OpTypeFloat 32
%vec3 = OpTypeVector %float 3
%vec4 = OpTypeVector %float 4
%func = OpTypeFunction %void

; Inputs/Outputs
%ptrInputVec3 = OpTypePointer Input %vec3
%ptrOutputVec4 = OpTypePointer Output %vec4

%inNormal = OpVariable %ptrInputVec3 Input
%outColor = OpVariable %ptrOutputVec4 Output

; Constants
%float_0_5 = OpConstant %float 0.5
%float_1_0 = OpConstant %float 1.0
%vec3_0_5 = OpConstantComposite %vec3 %float_0_5 %float_0_5 %float_0_5

; Main
%main = OpFunction %void None %func
%label = OpLabel

; Read input normal
%normal = OpLoad %vec3 %inNormal
; Normalize
%normNormal = OpExtInst %vec3 %1 GLSLStd450Normalize %normal

; Convert [-1, 1] to [0, 1]
; (normal * 0.5) + 0.5
%scaled = OpVectorTimesScalar %vec3 %normNormal %float_0_5
%mapped = OpFAdd %vec3 %scaled %vec3_0_5

; Construct vec4 color (r, g, b, 1.0)
%color = OpCompositeConstruct %vec4 %mapped %float_1_0

; Output
OpStore %outColor %color
OpReturn
OpFunctionEnd

; Import instruction set for GLSLstd450
%1 = OpExtInstImport "GLSL.std.450"
