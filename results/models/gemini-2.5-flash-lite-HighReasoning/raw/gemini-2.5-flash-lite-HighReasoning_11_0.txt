use std::io::{self, BufRead};
use std::f64::consts::PI;

#[derive(Debug, Clone, Copy)]
struct Point {
    x: f64,
    y: f64,
}

#[derive(Debug, Clone)]
struct Polygon {
    vertices: Vec<Point>,
}

// Geometric Utilities
fn is_point_in_polygon(point: &Point, polygon: &Polygon) -> bool {
    let mut inside = false;
    let n = polygon.vertices.len();
    for i in 0..n {
        let j = (i + 1) % n;
        let vi = &polygon.vertices[i];
        let vj = &polygon.vertices[j];
        if ((vi.y > point.y) != (vj.y > point.y)) &&
           (point.x < (vj.x - vi.x) * (point.y - vi.y) / (vj.y - vi.y) + vi.x) {
            inside = !inside;
        }
    }
    inside
}

fn is_polygon_in_polygon(inner: &Polygon, outer: &Polygon) -> bool {
    for vertex in &inner.vertices {
        if !is_point_in_polygon(vertex, outer) {
            return false;
        }
    }
    true
}

// Simple SAT-based intersection check (axis-aligned bounding box or simplified separating axis theorem would be better)
// Using a simple point-in-polygon check is insufficient for full intersection, 
// but for a greedy prototype it might suffice if checking all vertices.
fn polygons_overlap(poly1: &Polygon, poly2: &Polygon) -> bool {
    // Check if any point of poly1 is inside poly2
    for p in &poly1.vertices {
        if is_point_in_polygon(p, poly2) { return true; }
    }
    // Check if any point of poly2 is inside poly1
    for p in &poly2.vertices {
        if is_point_in_polygon(p, poly1) { return true; }
    }
    false
}

fn rotate_point(p: Point, angle: f64) -> Point {
    let s = angle.sin();
    let c = angle.cos();
    Point {
        x: p.x * c - p.y * s,
        y: p.x * s + p.y * c,
    }
}

fn transform_polygon(poly: &Polygon, translation: Point, rotation: f64) -> Polygon {
    let new_vertices = poly.vertices.iter()
        .map(|&p| {
            let rotated = rotate_point(p, rotation);
            Point { x: rotated.x + translation.x, y: rotated.y + translation.y }
        })
        .collect();
    Polygon { vertices: new_vertices }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    if let Some(Ok(line)) = lines.next() {
        let parts: Vec<&str> = line.split_whitespace().collect();
        if parts.len() == 2 {
            let sv: usize = parts[0].parse().expect("Invalid SV");
            let np: usize = parts[1].parse().expect("Invalid NP");

            let mut stock_vertices = Vec::with_capacity(sv);
            for _ in 0..sv {
                if let Some(Ok(line)) = lines.next() {
                    let p: Vec<f64> = line.split_whitespace()
                        .map(|s| s.parse().expect("Invalid point coordinate"))
                        .collect();
                    stock_vertices.push(Point { x: p[0], y: p[1] });
                }
            }
            let stock = Polygon { vertices: stock_vertices };
            
            let mut pieces = Vec::with_capacity(np);
            for _ in 0..np {
                if let Some(Ok(line)) = lines.next() {
                    let pv: usize = line.trim().parse().expect("Invalid PV");
                    let mut piece_vertices = Vec::with_capacity(pv);
                    for _ in 0..pv {
                        if let Some(Ok(line)) = lines.next() {
                            let p: Vec<f64> = line.split_whitespace()
                                .map(|s| s.parse().expect("Invalid point coordinate"))
                                .collect();
                            piece_vertices.push(Point { x: p[0], y: p[1] });
                        }
                    }
                    pieces.push(Polygon { vertices: piece_vertices });
                }
            }
            
            // Packing Logic: Greedy placement
            let mut placed_polygons = Vec::new();
            let mut placements = Vec::new();
            
            for piece in pieces {
                let mut found = false;
                // Try basic grid of translations and rotations
                'search: for rot in (0..4).map(|i| i as f64 * PI / 2.0) {
                    for x in (0..10).map(|i| i as f64 * 1.0) {
                        for y in (0..10).map(|i| i as f64 * 1.0) {
                            let candidate = transform_polygon(&piece, Point { x, y }, rot);
                            if is_polygon_in_polygon(&candidate, &stock) {
                                let mut overlap = false;
                                for p in &placed_polygons {
                                    if polygons_overlap(&candidate, p) {
                                        overlap = true;
                                        break;
                                    }
                                }
                                if !overlap {
                                    placements.push((0, x, y, rot));
                                    placed_polygons.push(candidate);
                                    found = true;
                                    break 'search;
                                }
                            }
                        }
                    }
                }
            }
            
            // Output
            println!("1"); 
            for placement in placements {
                println!("{} {} {} {}", placement.0, placement.1, placement.2, placement.3);
            }
        }
    }
}
