using System;
using System.Collections.Generic;
using System.Linq;

class Vector3
{
    public double X, Y, Z;
    public Vector3(double x, double y, double z) { X = x; Y = y; Z = z; }
    public static Vector3 operator -(Vector3 a, Vector3 b) => new Vector3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
    public static Vector3 operator +(Vector3 a, Vector3 b) => new Vector3(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
    public double Length => Math.Sqrt(X * X + Y * Y + Z * Z);
    public double Dot(Vector3 other) => X * other.X + Y * other.Y + Z * other.Z;
}

class Solver
{
    static void Main()
    {
        string[] firstLine = Console.ReadLine().Split(' ', StringSplitOptions.RemoveEmptyEntries);
        int N = int.Parse(firstLine[0]);
        double mu = double.Parse(firstLine[1]);

        string[] startOrbit = Console.ReadLine().Split(' ', StringSplitOptions.RemoveEmptyEntries);
        Vector3 startPos = new Vector3(double.Parse(startOrbit[0]), double.Parse(startOrbit[1]), double.Parse(startOrbit[2]));
        Vector3 startVel = new Vector3(double.Parse(startOrbit[3]), double.Parse(startOrbit[4]), double.Parse(startOrbit[5]));

        List<Tuple<Vector3, Vector3>> stations = new List<Tuple<Vector3, Vector3>>();
        for (int i = 0; i < N; i++)
        {
            string[] stationOrbit = Console.ReadLine().Split(' ', StringSplitOptions.RemoveEmptyEntries);
            stations.Add(new Tuple<Vector3, Vector3>(
                new Vector3(double.Parse(stationOrbit[0]), double.Parse(stationOrbit[1]), double.Parse(stationOrbit[2])),
                new Vector3(double.Parse(stationOrbit[3]), double.Parse(stationOrbit[4]), double.Parse(stationOrbit[5]))
            ));
        }

        var order = SolveTSP(startPos, startVel, stations, mu);

        Console.WriteLine(string.Join(" ", order));
    }

    static double CalculateDeltaV(Vector3 pos1, Vector3 vel1, Vector3 pos2, Vector3 vel2, double mu)
    {
        // Simple approximation: magnitude of difference in velocity vectors + position difference impact
        // In a real scenario, this would be a Lambert solver.
        double dvPos = (pos2 - pos1).Length;
        double dvVel = (vel2 - vel1).Length;
        return dvVel + 0.1 * dvPos; // Heuristic weighting
    }

    static List<int> SolveTSP(Vector3 startPos, Vector3 startVel, List<Tuple<Vector3, Vector3>> stations, double mu)
    {
        int n = stations.Count;
        List<int> order = new List<int>();
        bool[] visited = new bool[n];
        Vector3 currentPos = startPos;
        Vector3 currentVel = startVel;

        for (int i = 0; i < n; i++)
        {
            int bestNext = -1;
            double minCost = double.MaxValue;

            for (int j = 0; j < n; j++)
            {
                if (!visited[j])
                {
                    double cost = CalculateDeltaV(currentPos, currentVel, stations[j].Item1, stations[j].Item2, mu);
                    if (cost < minCost)
                    {
                        minCost = cost;
                        bestNext = j;
                    }
                }
            }

            order.Add(bestNext);
            visited[bestNext] = true;
            currentPos = stations[bestNext].Item1;
            currentVel = stations[bestNext].Item2;
        }

        return order;
    }
}
