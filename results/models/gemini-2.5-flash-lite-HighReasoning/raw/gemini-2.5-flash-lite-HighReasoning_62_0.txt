declare const require: any;
const fs = require('fs');

interface Town {
    id: string;
    x: number;
    y: number;
    value: number;
}

interface InputData {
    width: number;
    height: number;
    grid: string[];
    ignitions: [number, number][];
    towns: Town[];
    cutBudget: number;
}

function solve() {
    const input: string = fs.readFileSync(0, 'utf8');
    const data: InputData = JSON.parse(input);
    const { width, height, grid, ignitions, towns, cutBudget } = data;

    // Use BFS to find the distance from the nearest ignition source to every cell
    const dist = Array.from({ length: height }, () => Array(width).fill(Infinity));
    const queue: [number, number][] = [];

    for (const [ix, iy] of ignitions) {
        dist[iy][ix] = 0;
        queue.push([ix, iy]);
    }

    let head = 0;
    while (head < queue.length) {
        const [x, y] = queue[head++];
        const d = dist[y][x];

        const dirs = [[0, 1], [0, -1], [1, 0], [-1, 0]];
        for (const [dx, dy] of dirs) {
            const nx = x + dx, ny = y + dy;
            if (nx >= 0 && nx < width && ny >= 0 && ny < height && grid[ny][nx] === '.' && dist[ny][nx] === Infinity) {
                dist[ny][nx] = d + 1;
                queue.push([nx, ny]);
            }
        }
    }

    // Sort towns by distance from fire, then by value (descending)
    const sortedTowns = [...towns].sort((a, b) => {
        const distA = dist[a.y][a.x];
        const distB = dist[b.y][b.x];
        if (distA !== distB) return distA - distB;
        return b.value - a.value;
    });

    const firebreaks: [number, number][] = [];
    let cutsRemaining = cutBudget;
    const cutSet = new Set<string>();

    // For each town, try to cut a path
    for (const town of sortedTowns) {
        if (cutsRemaining <= 0) break;
        if (dist[town.y][town.x] === Infinity) continue; // Already unreachable or safe

        // Trace back path from town to ignition
        let curr: [number, number] = [town.x, town.y];
        while (dist[curr[1]][curr[0]] > 0) {
            if (cutsRemaining <= 0) break;

            const [x, y] = curr;
            let bestNext: [number, number] = curr;
            let minD = dist[y][x];

            const dirs = [[0, 1], [0, -1], [1, 0], [-1, 0]];
            for (const [dx, dy] of dirs) {
                const nx = x + dx, ny = y + dy;
                if (nx >= 0 && nx < width && ny >= 0 && ny < height && dist[ny][nx] < minD) {
                    minD = dist[ny][nx];
                    bestNext = [nx, ny];
                }
            }

            // Cut the current cell (if it's not the town itself)
            if (bestNext !== curr) {
                const key = `${curr[0]},${curr[1]}`;
                if (!cutSet.has(key)) {
                    firebreaks.push(curr);
                    cutSet.add(key);
                    cutsRemaining--;
                }
                curr = bestNext;
            } else {
                break; // Should not happen
            }
        }
    }

    process.stdout.write(JSON.stringify({ firebreaks }));
}

solve();
