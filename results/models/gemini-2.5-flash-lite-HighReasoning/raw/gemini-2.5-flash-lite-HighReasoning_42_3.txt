; SPIR-V Geometry Shader: Expand line to ribbon (triangle strip)
; Input: InputLines (2 vertices per invocation)
; Output: OutputTriangleStrip (4 vertices)

OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %gl_out %inColor %outColor
OpExecutionMode %main InputLines
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1

; --- Decorations ---
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %inColor Location 0
OpDecorate %outColor Location 0

; --- Types ---
%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%uint_2 = OpConstant %uint 2
%float_0_03 = OpConstant %float 0.03

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

%arr_v3_2 = OpTypeArray %v3float %uint_2
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_2
%inColor = OpVariable %ptr_in_arr_v3 Input
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Input

%ptr_out_v4 = OpTypePointer Output %v4float

; --- Main ---
%func_void = OpTypeFunction %void
%main = OpFunction %void None %func_void
%label_entry = OpLabel

; Load inputs
%ptr_in_v4_0 = OpAccessChain %ptr_in_arr_pv %gl_in %int_0 %int_0
%pos0 = OpLoad %v4float %ptr_in_v4_0
%ptr_in_v4_1 = OpAccessChain %ptr_in_arr_pv %gl_in %int_1 %int_0
%pos1 = OpLoad %v4float %ptr_in_v4_1

; Load colors
%ptr_in_c0 = OpAccessChain %ptr_out_v3 %inColor %int_0
%c0 = OpLoad %v3float %ptr_in_c0
%ptr_in_c1 = OpAccessChain %ptr_out_v3 %inColor %int_1
%c1 = OpLoad %v3float %ptr_in_c1

; Compute direction and perpendicular
%p0_xyz = OpVectorShuffle %v3float %pos0 %pos0 0 1 2
%p1_xyz = OpVectorShuffle %v3float %pos1 %pos1 0 1 2
%dir = OpFSub %v3float %p1_xyz %p0_xyz
%dir_x = OpCompositeExtract %float %dir 0
%dir_y = OpCompositeExtract %float %dir 1
%perp_x = OpFNegate %float %dir_y
%perp_y = %dir_x
%perp_x_scaled = OpFMul %float %perp_x %float_0_03
%perp_y_scaled = OpFMul %float %perp_y %float_0_03
%perp_v3 = OpCompositeConstruct %v3float %perp_x_scaled %perp_y_scaled %float(0)

; Function to emit a vertex
; Vertex 1: p0+perp, c0
%v1_pos_xyz = OpFAdd %v3float %p0_xyz %perp_v3
%v1_pos = OpVectorShuffle %v4float %v1_pos_xyz %pos0 0 1 2 7
%ptr_pos = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_pos %v1_pos
OpStore %outColor %c0
OpEmitVertex

; Vertex 2: p0-perp, c0
%v2_pos_xyz = OpFSub %v3float %p0_xyz %perp_v3
%v2_pos = OpVectorShuffle %v4float %v2_pos_xyz %pos0 0 1 2 7
OpStore %ptr_pos %v2_pos
OpStore %outColor %c0
OpEmitVertex

; Vertex 3: p1+perp, c1
%v3_pos_xyz = OpFAdd %v3float %p1_xyz %perp_v3
%v3_pos = OpVectorShuffle %v4float %v3_pos_xyz %pos1 0 1 2 7
OpStore %ptr_pos %v3_pos
OpStore %outColor %c1
OpEmitVertex

; Vertex 4: p1-perp, c1
%v4_pos_xyz = OpFSub %v3float %p1_xyz %perp_v3
%v4_pos = OpVectorShuffle %v4float %v4_pos_xyz %pos1 0 1 2 7
OpStore %ptr_pos %v4_pos
OpStore %outColor %c1
OpEmitVertex

OpEndPrimitive
OpReturn
OpFunctionEnd
