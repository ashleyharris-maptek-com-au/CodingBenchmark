#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <set>

// Simple graph representation
struct Graph {
    int num_vertices;
    int num_colors;
    std::vector<std::vector<int>> adj;

    Graph(int v, int k) : num_vertices(v), num_colors(k), adj(v) {}

    void add_edge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
};

// DSATUR Heuristic for better greedy coloring
std::vector<int> dsatur_coloring(const Graph& g) {
    int n = g.num_vertices;
    std::vector<int> colors(n, -1);
    std::vector<int> degree(n);
    std::vector<std::set<int>> adjacent_colors(n);
    std::vector<int> saturation(n, 0);

    for (int i = 0; i < n; ++i) {
        degree[i] = g.adj[i].size();
    }

    for (int i = 0; i < n; ++i) {
        // Choose vertex with max saturation, break ties with degree
        int best_v = -1;
        int max_sat = -1;
        int max_deg = -1;

        for (int v = 0; v < n; ++v) {
            if (colors[v] == -1) {
                if (saturation[v] > max_sat || (saturation[v] == max_sat && degree[v] > max_deg)) {
                    max_sat = saturation[v];
                    max_deg = degree[v];
                    best_v = v;
                }
            }
        }

        if (best_v == -1) break;

        // Color best_v
        std::vector<bool> available(g.num_colors, true);
        for (int c : adjacent_colors[best_v]) {
            if (c < g.num_colors) available[c] = false;
        }

        int chosen_color = -1;
        for (int c = 0; c < g.num_colors; ++c) {
            if (available[c]) {
                chosen_color = c;
                break;
            }
        }

        // If no color available, this greedy heuristic failed.
        // For larger graphs we might need to handle this differently.
        if (chosen_color != -1) {
            colors[best_v] = chosen_color;
            for (int neighbor : g.adj[best_v]) {
                if (colors[neighbor] == -1) {
                    if (adjacent_colors[neighbor].find(chosen_color) == adjacent_colors[neighbor].end()) {
                        adjacent_colors[neighbor].insert(chosen_color);
                        saturation[neighbor]++;
                    }
                }
            }
        }
    }
    return colors;
}

// Simple Backtracking for small graphs
bool is_safe(int v, const Graph& g, const std::vector<int>& colors, int c) {
    for (int neighbor : g.adj[v]) {
        if (colors[neighbor] == c) return false;
    }
    return true;
}

bool backtrack_coloring(int v, const Graph& g, std::vector<int>& colors) {
    if (v == g.num_vertices) return true;

    for (int c = 0; c < g.num_colors; ++c) {
        if (is_safe(v, g, colors, c)) {
            colors[v] = c;
            if (backtrack_coloring(v + 1, g, colors)) return true;
            colors[v] = -1;
        }
    }
    return false;
}

int main() {
    // Optimization for fast I/O
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    int n, m, k;
    if (!(std::cin >> n >> m >> k)) return 0;

    Graph g(n, k);
    for (int i = 0; i < m; ++i) {
        int u, v;
        std::cin >> u >> v;
        g.add_edge(u, v);
    }

    std::vector<int> colors;
    // Adaptive logic: use backtracking for very small graphs, DSATUR otherwise
    if (n < 50) {
        colors.assign(n, -1);
        if (!backtrack_coloring(0, g, colors)) {
            // If backtracking fails, try greedy as a fallback
            colors = dsatur_coloring(g);
        }
    } else {
        colors = dsatur_coloring(g);
    }

    for (int i = 0; i < n; ++i) {
        std::cout << (colors[i] == -1 ? 0 : colors[i]) << (i == n - 1 ? "" : " ");
    }
    std::cout << std::endl;

    return 0;
}
