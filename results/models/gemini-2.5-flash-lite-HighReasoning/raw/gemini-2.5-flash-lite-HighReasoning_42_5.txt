; SPIR-V Geometry Shader
; Input: InputLines
; Output: OutputTriangleStrip
; Vertices emitted: 7 (4 for shaft, 3 for arrowhead)

OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %gl_out %outColor %inColor
OpExecutionMode %main InputLines
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 7
OpExecutionMode %main Invocations 1

; Import GLSL std450
OpExtInstImport %glsl "GLSL.std.450"

; Decorate interface variables
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_in Input
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %gl_out Output
OpDecorate %outColor Location 0
OpDecorate %inColor Location 2

; Types
%void = OpTypeVoid
%float = OpTypeFloat 32
%v4float = OpTypeVector %float 4
%v3float = OpTypeVector %float 3
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_2 = OpConstant %uint 2
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%float_0 = OpConstant %float 0.0
%float_0_015 = OpConstant %float 0.015
%float_0_05 = OpConstant %float 0.05
%float_0_75 = OpConstant %float 0.75
%float_1_0 = OpConstant %float 1.0

; Input structures
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%inColorArr = OpTypeArray %v3float %uint_2
%ptr_in_color = OpTypePointer Input %inColorArr
%inColor = OpVariable %ptr_in_color Input

; Output structures
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v4 = OpTypePointer Output %v4float

%ptr_out_color = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_color Output

%func_void = OpTypeFunction %void

%main = OpFunction %void None %func_void
%entry = OpLabel

; Fetch p0, p1 (as v3)
%ptr_p0 = OpAccessChain %ptr_out_v4 %gl_in %int_0 %int_0
%p0_v4 = OpLoad %v4float %ptr_p0
%p0 = OpVectorShuffle %v3float %p0_v4 %p0_v4 0 1 2

%ptr_p1 = OpAccessChain %ptr_out_v4 %gl_in %int_1 %int_0
%p1_v4 = OpLoad %v4float %ptr_p1
%p1 = OpVectorShuffle %v3float %p1_v4 %p1_v4 0 1 2

; Fetch colors
%ptr_c0 = OpAccessChain %ptr_out_color %inColor %int_0
%c0 = OpLoad %v3float %ptr_c0
%ptr_c1 = OpAccessChain %ptr_out_color %inColor %int_1
%c1 = OpLoad %v3float %ptr_c1

; Compute dir and perp
%dir = OpFSub %v3float %p1 %p0
%up = OpCompositeConstruct %v3float %float_0 %float_0 %float_1_0
%perp = OpExtInst %v3float %glsl Cross %dir %up
%perp_norm = OpExtInst %v3float %glsl Normalize %perp

; Mid = p0 + 0.75 * dir
%dir_scaled = OpVectorTimesScalar %v3float %dir %float_0_75
%mid = OpFAdd %v3float %p0 %dir_scaled

; Shaft offsets
%perp_shaft = OpVectorTimesScalar %v3float %perp_norm %float_0_015
%p0_l = OpFSub %v3float %p0 %perp_shaft
%p0_r = OpFAdd %v3float %p0 %perp_shaft
%mid_l = OpFSub %v3float %mid %perp_shaft
%mid_r = OpFAdd %v3float %mid %perp_shaft

; Helper to emit
%ptr_pos = OpAccessChain %ptr_out_v4 %gl_out %int_0

; Emit Shaft (4 verts)
; 1: p0_l
%pos1 = OpVectorShuffle %v4float %p0_l %p0_l 0 1 2 2 ; dummy w=1
%pos1_w = OpCompositeInsert %v4float %float_1_0 %pos1 3
OpStore %ptr_pos %pos1_w
OpStore %outColor %c0
OpEmitVertex

; 2: p0_r
%pos2 = OpVectorShuffle %v4float %p0_r %p0_r 0 1 2 2
%pos2_w = OpCompositeInsert %v4float %float_1_0 %pos2 3
OpStore %ptr_pos %pos2_w
OpStore %outColor %c0
OpEmitVertex

; 3: mid_l
%pos3 = OpVectorShuffle %v4float %mid_l %mid_l 0 1 2 2
%pos3_w = OpCompositeInsert %v4float %float_1_0 %pos3 3
OpStore %ptr_pos %pos3_w
OpStore %outColor %c0
OpEmitVertex

; 4: mid_r
%pos4 = OpVectorShuffle %v4float %mid_r %mid_r 0 1 2 2
%pos4_w = OpCompositeInsert %v4float %float_1_0 %pos4 3
OpStore %ptr_pos %pos4_w
OpStore %outColor %c0
OpEmitVertex

OpEndPrimitive

; Arrowhead (3 verts: mid+perp*0.05, mid-perp*0.05, p1)
%perp_arr = OpVectorTimesScalar %v3float %perp_norm %float_0_05
%arr_l = OpFSub %v3float %mid %perp_arr
%arr_r = OpFAdd %v3float %mid %perp_arr

; 1: arr_r
%pos5 = OpVectorShuffle %v4float %arr_r %arr_r 0 1 2 2
%pos5_w = OpCompositeInsert %v4float %float_1_0 %pos5 3
OpStore %ptr_pos %pos5_w
OpStore %outColor %c1
OpEmitVertex

; 2: arr_l
%pos6 = OpVectorShuffle %v4float %arr_l %arr_l 0 1 2 2
%pos6_w = OpCompositeInsert %v4float %float_1_0 %pos6 3
OpStore %ptr_pos %pos6_w
OpStore %outColor %c1
OpEmitVertex

; 3: p1
%pos7 = OpVectorShuffle %v4float %p1 %p1 0 1 2 2
%pos7_w = OpCompositeInsert %v4float %float_1_0 %pos7 3
OpStore %ptr_pos %pos7_w
OpStore %outColor %c1
OpEmitVertex

OpEndPrimitive

OpReturn
OpFunctionEnd
