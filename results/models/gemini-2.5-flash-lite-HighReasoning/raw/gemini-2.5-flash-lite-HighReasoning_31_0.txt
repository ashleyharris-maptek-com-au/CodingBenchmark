pub fn subset_sum(nums: &[i32], target: i32) -> bool {
    if target == 0 {
        return true;
    }
    if nums.is_empty() {
        return false;
    }

    let target = target as usize;
    let mut dp = vec![false; target + 1];
    dp[0] = true;

    for &num in nums {
        if num < 0 {
            // Subset sum with negative numbers is more complex.
            // Assuming non-negative for this basic implementation.
            continue;
        }
        let num = num as usize;
        for j in (num..=target).rev() {
            if dp[j - num] {
                dp[j] = true;
            }
        }
    }

    dp[target]
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_subset_sum() {
        assert!(subset_sum(&[3, 34, 4, 12, 5, 2], 9));
        assert!(!subset_sum(&[3, 34, 4, 12, 5, 2], 30));
        assert!(subset_sum(&[], 0));
        assert!(!subset_sum(&[1, 2], 5));
    }
}
