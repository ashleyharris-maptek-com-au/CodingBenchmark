// HLSL Pixel Shader for Vulkan
// Rendering the Mandelbrot set

struct PSInput {
    [[vk::location(0)]] float3 worldPos : TEXCOORD0;   // world-space position
    [[vk::location(1)]] float3 normal   : TEXCOORD1;   // world-space normal
    [[vk::location(2)]] float2 uv       : TEXCOORD2;   // texture coordinates in [0,1]
    [[vk::location(3)]] float3 tangent  : TEXCOORD3;   // tangent vector
    [[vk::location(4)]] float3 color    : TEXCOORD4;   // per-vertex color
};

[[vk::binding(0, 0)]]
cbuffer UBO : register(b0) {
    float4x4 model;
    float4x4 view;
    float4 projection;
    float4 lightPos;
    float4 cameraPos;
    float4 params;
};

float4 main(PSInput input) : SV_Target0 {
    // Map UVs [0,1] to complex plane c = (uv.x*3-2, uv.y*2-1)
    float2 c = float2(input.uv.x * 3.0 - 2.0, input.uv.y * 2.0 - 1.0);
    
    float2 z = float2(0.0, 0.0);
    float iterations = 0.0;
    const float max_iterations = 20.0;
    
    for (float i = 0.0; i < max_iterations; ++i) {
        // z = z^2 + c
        // (x+yi)^2 = x^2 - y^2 + 2xyi
        float x2 = z.x * z.x;
        float y2 = z.y * z.y;
        
        if (x2 + y2 > 4.0) break;
        
        z = float2(x2 - y2, 2.0 * z.x * z.y) + c;
        iterations += 1.0;
    }
    
    float t = iterations / max_iterations;
    
    // Coloring: (t, t*0.5, 1.0-t), alpha 1.0
    return float4(t, t * 0.5, 1.0 - t, 1.0);
}
