; --- SPIR-V Fragment Shader Assembly ---
; This shader outputs a constant red color to the output variable at location 0.

; 1. Capabilities
OpCapability Shader

; 2. Memory Model
OpMemoryModel Logical GLSL450

; 3. Entry Point
; Fragment stage, function %main, name "main", using %outColor
OpEntryPoint Fragment %main "main" %outColor

; 4. Execution Mode
; Fragment shaders must specify the origin
OpExecutionMode %main OriginUpperLeft

; 5. Annotations
; Bind %outColor to location 0
OpDecorate %outColor Location 0

; 6. Types, Constants, and Variables
%void      = OpTypeVoid
%float     = OpTypeFloat 32
%vec4      = OpTypeVector %float 4
%ptr_out   = OpTypePointer Output %vec4
%main_type = OpTypeFunction %void

; Output variable
%outColor  = OpVariable %ptr_out Output

; Constants for color (Red)
%f1        = OpConstant %float 1.0
%f0        = OpConstant %float 0.0
%red       = OpConstantComposite %vec4 %f1 %f0 %f0 %f1

; 7. Function Definitions
%main = OpFunction %void None %main_type
%label = OpLabel
    OpStore %outColor %red
    OpReturn
OpFunctionEnd
