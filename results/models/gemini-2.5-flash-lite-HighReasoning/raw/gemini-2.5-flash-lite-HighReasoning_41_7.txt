; SPIR-V
; Version: 1.0
; Generator: manual
%glsl = OpExtInstImport "GLSL.std.450"

OpCapability Shader
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %worldPos %normal %outColor
OpExecutionMode %main OriginUpperLeft

; Decorations
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %outColor Location 0
OpDecorate %UBO Block
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %ubo_var DescriptorSet 0
OpDecorate %ubo_var Binding 0

; Types
%void = OpTypeVoid
%float = OpTypeFloat 32
%int = OpTypeInt 32 1
%vec3 = OpTypeVector %float 3
%vec4 = OpTypeVector %float 4
%mat4 = OpTypeMatrix %vec4 4
%UBO = OpTypeStruct %mat4 %mat4 %mat4 %vec4 %vec4 %vec4
%ptr_uniform_UBO = OpTypePointer Uniform %UBO
%ptr_input_vec3 = OpTypePointer Input %vec3
%ptr_output_vec4 = OpTypePointer Output %vec4
%ptr_uniform_vec4 = OpTypePointer Uniform %vec4
%func = OpTypeFunction %void

; Constants
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3
%int_4 = OpConstant %int 4
%float_0 = OpConstant %float 0.0
%float_0_3 = OpConstant %float 0.3
%float_0_5 = OpConstant %float 0.5
%float_1 = OpConstant %float 1.0
%float_3 = OpConstant %float 3.0
%float_0_2 = OpConstant %float 0.2
%float_0_8 = OpConstant %float 0.8
%vec3_base = OpConstantComposite %vec3 %float_0_2 %float_0_3 %float_0_8
%vec3_rim = OpConstantComposite %vec3 %float_1 %float_0_8 %float_0_5

; Variables
%ubo_var = OpVariable %ptr_uniform_UBO Uniform
%worldPos = OpVariable %ptr_input_vec3 Input
%normal = OpVariable %ptr_input_vec3 Input
%outColor = OpVariable %ptr_output_vec4 Output

; Main function
%main = OpFunction %void None %func
%label = OpLabel

; Load Inputs
%wpos = OpLoad %vec3 %worldPos
%norm = OpLoad %vec3 %normal
%norm_norm = OpExtInst %vec3 %glsl Normalize %norm

; Access UBO
%ptr_light = OpAccessChain %ptr_uniform_vec4 %ubo_var %int_3
%lightPos = OpLoad %vec4 %ptr_light
%lightPos3 = OpVectorShuffle %vec3 %lightPos %lightPos 0 1 2

%ptr_cam = OpAccessChain %ptr_uniform_vec4 %ubo_var %int_4
%camPos = OpLoad %vec4 %ptr_cam
%camPos3 = OpVectorShuffle %vec3 %camPos %camPos 0 1 2

; Light Calculation
%L = OpVectorSub %vec3 %lightPos3 %wpos
%L_norm = OpExtInst %vec3 %glsl Normalize %L
%diffuse = OpDot %float %norm_norm %L_norm
%diffuse_clamped = OpExtInst %float %glsl FMax %float_0 %diffuse

; Rim Calculation
%V = OpVectorSub %vec3 %camPos3 %wpos
%V_norm = OpExtInst %vec3 %glsl Normalize %V
%NdotV = OpDot %float %norm_norm %V_norm
%rim_factor = OpFSub %float %float_1 %NdotV
%rim_factor_clamped = OpExtInst %float %glsl FMax %float_0 %rim_factor
%rim = OpExtInst %float %glsl Pow %rim_factor_clamped %float_3

; Lighting
%diffuse_mult = OpFMul %float %float_0_5 %diffuse_clamped
%diffuse_mod = OpFAdd %float %float_0_3 %diffuse_mult
%base_color = OpVectorTimesScalar %vec3 %vec3_base %diffuse_mod
%rim_color = OpVectorTimesScalar %vec3 %vec3_rim %rim
%final_rgb = OpFAdd %vec3 %base_color %rim_color
%final_rgba = OpCompositeConstruct %vec4 %final_rgb %float_1

OpStore %outColor %final_rgba
OpReturn
OpFunctionEnd
