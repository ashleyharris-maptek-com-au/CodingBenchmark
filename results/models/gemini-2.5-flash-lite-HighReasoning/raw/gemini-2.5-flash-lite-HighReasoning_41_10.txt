; SPIR-V
; Version: 1.0
; Generator: manual
; Bound: 100

OpCapability Shader
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %in_uv %in_normal %in_worldPos %out_color %in_lightPos_vec %in_cameraPos_vec %ubo
OpExecutionMode %main OriginUpperLeft

; Decorations
OpDecorate %out_color Location 0
OpDecorate %in_worldPos Location 0
OpDecorate %in_normal Location 1
OpDecorate %in_uv Location 2
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0
OpDecorate %ubo_struct Block
OpMemberDecorate %ubo_struct 0 Offset 0
OpMemberDecorate %ubo_struct 1 Offset 64
OpMemberDecorate %ubo_struct 2 Offset 128
OpMemberDecorate %ubo_struct 3 Offset 192
OpMemberDecorate %ubo_struct 4 Offset 208
OpMemberDecorate %ubo_struct 5 Offset 224

; Types
%void = OpTypeVoid
%float = OpTypeFloat 32
%vec2 = OpTypeVector %float 2
%vec3 = OpTypeVector %float 3
%vec4 = OpTypeVector %float 4
%mat4 = OpTypeMatrix %vec4 4
%bool = OpTypeBool
%ptr_input_vec2 = OpTypePointer Input %vec2
%ptr_input_vec3 = OpTypePointer Input %vec3
%ptr_output_vec4 = OpTypePointer Output %vec4
%ptr_uniform_vec4 = OpTypePointer Uniform %vec4
%ptr_uniform_struct = OpTypePointer Uniform %ubo_struct

%ubo_struct = OpTypeStruct %mat4 %mat4 %mat4 %vec4 %vec4 %vec4
%ptr_ubo = OpTypePointer Uniform %ubo_struct

%void_func = OpTypeFunction %void

; Variables
%in_worldPos = OpVariable %ptr_input_vec3 Input
%in_normal = OpVariable %ptr_input_vec3 Input
%in_uv = OpVariable %ptr_input_vec2 Input
%out_color = OpVariable %ptr_output_vec4 Output
%ubo = OpVariable %ptr_ubo Uniform
%in_lightPos_vec = OpVariable %ptr_uniform_vec4 Uniform ; Dummy for entry point
%in_cameraPos_vec = OpVariable %ptr_uniform_vec4 Uniform ; Dummy for entry point

; Constants
%float_0 = OpConstant %float 0.0
%float_1 = OpConstant %float 1.0
%float_0_1 = OpConstant %float 0.1
%float_0_4 = OpConstant %float 0.4
%float_0_6 = OpConstant %float 0.6
%float_0_9 = OpConstant %float 0.9
%float_3_14159 = OpConstant %float 3.14159
%float_16_0 = OpConstant %float 16.0
%int_0 = OpConstant %OpTypeInt 32 0
%int_1 = OpConstant %OpTypeInt 32 1
%int_3 = OpConstant %OpTypeInt 32 3
%int_4 = OpConstant %OpTypeInt 32 4

%main = OpFunction %void None %void_func
%entry = OpLabel

; Texture Coordinate
%uv = OpLoad %vec2 %in_uv
%uv_y = OpCompositeExtract %float %uv 1

; Stripes: sin(uv.y * pi * 16.0)
%factor = OpFMul %float %uv_y %float_3_14159
%factor2 = OpFMul %float %factor %float_16_0
%sin_val = OpExtInst %float %float_1 Sin %factor2

; Determine color: if sin > 0, red, else blue
%is_red = OpFOrdGreaterThan %bool %sin_val %float_0
%color_rgb = OpSelect %vec3 %is_red %vec3 %float_0_9 %float_0_1 %float_0_1 %vec3 %float_0_1 %float_0_1 %float_0_9
%color_rgba = OpCompositeConstruct %vec4 %color_rgb %float_1

; Shading:
%normal = OpLoad %vec3 %in_normal
%normal_n = OpExtInst %vec3 %float_1 Normalize %normal
%lightPos_ptr = OpAccessChain %ptr_uniform_vec4 %ubo %int_3
%lightPos = OpLoad %vec4 %lightPos_ptr
%lightPos_3 = OpVectorShuffle %vec3 %lightPos %lightPos 0 1 2
%worldPos = OpLoad %vec3 %in_worldPos
%light_dir = OpFSub %vec3 %lightPos_3 %worldPos
%light_dir_n = OpExtInst %vec3 %float_1 Normalize %light_dir
%diffuse_raw = OpDot %float %light_dir_n %normal_n
%diffuse = OpExtInst %float %float_1 FClamp %diffuse_raw %float_0 %float_1

; Lambertian: 0.4 + 0.6 * diffuse
%val1 = OpFMul %float %float_0_6 %diffuse
%final_light = OpFAdd %float %float_0_4 %val1
%final_color = OpVectorTimesScalar %vec4 %color_rgba %final_light
%final_color_a = OpCompositeInsert %vec4 %float_1 %final_color 3

OpStore %out_color %final_color_a
OpReturn
OpFunctionEnd
