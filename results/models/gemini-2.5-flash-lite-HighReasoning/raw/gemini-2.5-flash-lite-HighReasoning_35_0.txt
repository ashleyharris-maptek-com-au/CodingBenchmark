use std::io::{self, BufRead};
use std::collections::{HashSet, HashMap};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = lines.next().unwrap().unwrap();
    let parts: Vec<usize> = first_line.split_whitespace().map(|s| s.parse().unwrap()).collect();
    let num_vertices = parts[0];
    let _num_edges = parts[1];

    let mut adj: Vec<Vec<usize>> = vec![vec![]; num_vertices];
    for line in lines {
        let line = line.unwrap();
        let edge: Vec<usize> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
        if edge.len() == 2 {
            let u = edge[0];
            let v = edge[1];
            adj[u].push(v);
            adj[v].push(u);
        }
    }

    // Greedy strategy:
    // 1. Maintain a set of uncovered vertices (initially all).
    // 2. While there are uncovered vertices:
    //    a. Pick the vertex that covers the most currently uncovered vertices (including itself).
    //    b. Add to dominating set.
    //    c. Mark all neighbors and self as covered.

    let mut uncovered: HashSet<usize> = (0..num_vertices).collect();
    let mut dominating_set: Vec<usize> = Vec::new();
    let mut is_covered: Vec<bool> = vec![false; num_vertices];

    while !uncovered.is_empty() {
        let mut best_vertex = 0;
        let mut max_covered = 0;

        for i in 0..num_vertices {
            let mut covered_count = 0;
            if !is_covered[i] {
                covered_count += 1; // Covers itself
            }
            for &neighbor in &adj[i] {
                if !is_covered[neighbor] {
                    covered_count += 1;
                }
            }

            if covered_count > max_covered {
                max_covered = covered_count;
                best_vertex = i;
            }
        }

        // Add to dominating set
        dominating_set.push(best_vertex);
        
        // Update coverage
        if !is_covered[best_vertex] {
            is_covered[best_vertex] = true;
            uncovered.remove(&best_vertex);
        }
        for &neighbor in &adj[best_vertex] {
            if !is_covered[neighbor] {
                is_covered[neighbor] = true;
                uncovered.remove(&neighbor);
            }
        }
    }

    println!("{}", dominating_set.len());
    let output: Vec<String> = dominating_set.iter().map(|v| v.to_string()).collect();
    println!("{}", output.join(" "));
}
