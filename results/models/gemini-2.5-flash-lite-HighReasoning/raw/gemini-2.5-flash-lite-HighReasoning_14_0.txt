using System.Collections.Generic;
using System.Linq;

namespace MinesweeperSolver
{
    public class Solver
    {
        private readonly Board _board;

        public Solver(Board board)
        {
            _board = board;
        }

        /// <summary>
        /// Performs a single pass of the solver, applying basic deduction rules
        /// to identify mines and safe cells.
        /// </summary>
        public void SolveStep()
        {
            // The solver iteratively applies two fundamental Minesweeper rules:
            // 1. If the number of hidden neighbors equals the neighbor mine count, all hidden neighbors are mines.
            // 2. If the number of flagged neighbors equals the neighbor mine count, all other hidden neighbors are safe.

            bool madeProgress = true;
            while (madeProgress)
            {
                madeProgress = false;
                foreach (var cell in GetAllCells())
                {
                    // Only apply logic to revealed cells that have a known neighbor mine count.
                    if (cell.State != CellState.Revealed || !cell.NeighborMineCount.HasValue)
                        continue;

                    var neighbors = _board.GetNeighbors(cell).ToList();
                    var hiddenNeighbors = neighbors.Where(n => n.State == CellState.Hidden).ToList();
                    var flaggedNeighbors = neighbors.Where(n => n.State == CellState.Flagged).ToList();

                    if (hiddenNeighbors.Count == 0) continue;

                    // Rule 1: Identify mines
                    // If the number of hidden neighbors + already flagged neighbors equals the total mine count, 
                    // then all remaining hidden neighbors MUST be mines.
                    if (cell.NeighborMineCount == flaggedNeighbors.Count + hiddenNeighbors.Count)
                    {
                        foreach (var hidden in hiddenNeighbors)
                        {
                            hidden.State = CellState.Flagged;
                            hidden.IsMine = true;
                            madeProgress = true;
                        }
                    }
                    // Rule 2: Identify safe cells
                    // If the number of flagged neighbors equals the total mine count,
                    // then all remaining hidden neighbors MUST be safe.
                    else if (cell.NeighborMineCount == flaggedNeighbors.Count)
                    {
                        foreach (var hidden in hiddenNeighbors)
                        {
                            // This would be the 'Reveal' action in a real game
                            hidden.State = CellState.Revealed;
                            madeProgress = true;
                        }
                    }
                }
            }
        }

        private IEnumerable<Cell> GetAllCells()
        {
            for (int x = 0; x < _board.Width; x++)
            {
                for (int y = 0; y < _board.Height; y++)
                {
                    yield return _board.GetCell(x, y);
                }
            }
        }
    }
}
