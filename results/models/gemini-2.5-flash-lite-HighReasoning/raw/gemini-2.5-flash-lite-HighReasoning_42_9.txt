; SPIR-V Geometry Shader
; Input: InputLinesAdjacency (4 vertices)
; Output: TriangleStrip
; Function: Render thick smooth line segment based on adjacency

               OpCapability Geometry
               OpMemoryModel Logical GLSL450
%glsl_std_450 = OpExtInstImport "GLSL.std.450"
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %in_worldPos %in_normal %in_color %outColor
               OpExecutionMode %main InputLinesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 4
               OpExecutionMode %main Invocations 1

; --- Types ---
%void = OpTypeVoid
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_0 = OpConstant %uint 0
%uint_1 = OpConstant %uint 1
%uint_2 = OpConstant %uint 2
%uint_3 = OpConstant %uint 3
%uint_4 = OpConstant %uint 4
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3

; --- Input/Output Structs ---
%gl_PerVertex_in = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position

%gl_PerVertex_out = OpTypeStruct %v4float
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

; --- Input/Output Variables ---
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

; User inputs (arrays of size 4)
%arr_v3_4 = OpTypeArray %v3float %uint_4
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_4
%in_worldPos = OpVariable %ptr_in_arr_v3 Input
%in_normal = OpVariable %ptr_in_arr_v3 Input
%in_color = OpVariable %ptr_in_arr_v3 Input

%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_in_v3 = OpTypePointer Input %v3float

; --- Constants ---
%float_0 = OpConstant %float 0.0
%float_025 = OpConstant %float 0.025 ; Thickness

; --- Main ---
%main = OpFunction %void None %void_fn
%void_fn = OpTypeFunction %void
               OpLabel

; 1. Read input positions (gl_in[i].gl_Position)
%p0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%p0 = OpLoad %v4float %p0_ptr
%p1_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%p1 = OpLoad %v4float %p1_ptr
%p2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%p2 = OpLoad %v4float %p2_ptr
%p3_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
%p3 = OpLoad %v4float %p3_ptr

; Helper to get vec2 pos
%p0_v2 = OpVectorShuffle %v2float %p0 %p0 0 1
%p1_v2 = OpVectorShuffle %v2float %p1 %p1 0 1
%p2_v2 = OpVectorShuffle %v2float %p2 %p2 0 1
%p3_v2 = OpVectorShuffle %v2float %p3 %p3 0 1

; 2. Compute tangents: t0 = normalize(p2 - p0), t1 = normalize(p3 - p1)
%diff0 = OpFSub %v2float %p2_v2 %p0_v2
%t0 = OpExtInst %v2float %glsl_std_450 Normalize %diff0
%diff1 = OpFSub %v2float %p3_v2 %p1_v2
%t1 = OpExtInst %v2float %glsl_std_450 Normalize %diff1

; 3. Compute perpendiculars: perp = vec2(-t.y, t.x)
%t0_x = OpVectorExtractDynamic %float %t0 %uint_0
%t0_y = OpVectorExtractDynamic %float %t0 %uint_1
%neg_t0_y = OpFNegate %float %t0_y
%perp0 = OpCompositeConstruct %v2float %neg_t0_y %t0_x

%t1_x = OpVectorExtractDynamic %float %t1 %uint_0
%t1_y = OpVectorExtractDynamic %float %t1 %uint_1
%neg_t1_y = OpFNegate %float %t1_y
%perp1 = OpCompositeConstruct %v2float %neg_t1_y %t1_x

; 4. Prepare for emission (thickness 0.025)
%thick_v2 = OpCompositeConstruct %v2float %float_025 %float_025
%off0 = OpFMul %v2float %perp0 %thick_v2
%off1 = OpFMul %v2float %perp1 %thick_v2

; Load colors
%c1_ptr = OpAccessChain %ptr_in_v3 %in_color %int_1
%c1 = OpLoad %v3float %c1_ptr
%c2_ptr = OpAccessChain %ptr_in_v3 %in_color %int_2
%c2 = OpLoad %v3float %c2_ptr

; Emit 4 vertices
; v1 = p1 + perp0*thick
%v1_pos_v2 = OpFAdd %v2float %p1_v2 %off0
%v1_pos = OpCompositeInsert %v4float %v1_pos_v2 %p1 0 1
OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out_v4 %v1_pos
OpStore %outColor %c1
OpEmitVertex

; v2 = p1 - perp0*thick
%v2_pos_v2 = OpFSub %v2float %p1_v2 %off0
%v2_pos = OpCompositeInsert %v4float %v2_pos_v2 %p1 0 1
OpStore %ptr_out_v4 %v2_pos
OpStore %outColor %c1
OpEmitVertex

; v3 = p2 + perp1*thick
%v3_pos_v2 = OpFAdd %v2float %p2_v2 %off1
%v3_pos = OpCompositeInsert %v4float %v3_pos_v2 %p2 0 1
OpStore %ptr_out_v4 %v3_pos
OpStore %outColor %c2
OpEmitVertex

; v4 = p2 - perp1*thick
%v4_pos_v2 = OpFSub %v2float %p2_v2 %off1
%v4_pos = OpCompositeInsert %v4float %v4_pos_v2 %p2 0 1
OpStore %ptr_out_v4 %v4_pos
OpStore %outColor %c2
OpEmitVertex

               OpEndPrimitive
               OpReturn
               OpFunctionEnd
