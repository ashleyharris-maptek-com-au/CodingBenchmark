use std::io::{self, BufRead};
use rand08::{thread_rng, Rng};

#[derive(Debug, Clone)]
struct Node {
    x: f32,
    y: f32,
    vx: f32,
    vy: f32,
    mass: f32,
}

// Barnes-Hut Quadtree
struct QuadNode {
    center_of_mass: (f32, f32),
    total_mass: f32,
    boundary: (f32, f32, f32, f32), // x_min, y_min, x_max, y_max
    children: Option<Box<[QuadNode; 4]>>,
    node_idx: Option<usize>,
}

impl QuadNode {
    fn new(x_min: f32, y_min: f32, x_max: f32, y_max: f32) -> Self {
        QuadNode {
            center_of_mass: (0.0, 0.0),
            total_mass: 0.0,
            boundary: (x_min, y_min, x_max, y_max),
            children: None,
            node_idx: None,
        }
    }

    fn insert(&mut self, node_idx: usize, nodes: &[Node], depth: usize) {
        if depth > 20 { return; } // Max depth limit

        if self.total_mass == 0.0 {
            self.total_mass = nodes[node_idx].mass;
            self.center_of_mass = (nodes[node_idx].x, nodes[node_idx].y);
            self.node_idx = Some(node_idx);
            return;
        }

        if self.children.is_none() {
            let (x_min, y_min, x_max, y_max) = self.boundary;
            let x_mid = (x_min + x_max) / 2.0;
            let y_mid = (y_min + y_max) / 2.0;
            
            let mut children = Box::new([
                QuadNode::new(x_min, y_min, x_mid, y_mid),
                QuadNode::new(x_mid, y_min, x_max, y_mid),
                QuadNode::new(x_min, y_mid, x_mid, y_max),
                QuadNode::new(x_mid, y_mid, x_max, y_max),
            ]);

            let old_node = self.node_idx.unwrap();
            let quadrant = self.get_quadrant(nodes[old_node].x, nodes[old_node].y, x_mid, y_mid);
            children[quadrant].insert(old_node, nodes, depth + 1);

            self.children = Some(children);
            self.node_idx = None;
        }

        let (x_min, y_min, x_max, y_max) = self.boundary;
        let x_mid = (x_min + x_max) / 2.0;
        let y_mid = (y_min + y_max) / 2.0;
        let quadrant = self.get_quadrant(nodes[node_idx].x, nodes[node_idx].y, x_mid, y_mid);
        self.children.as_mut().unwrap()[quadrant].insert(node_idx, nodes, depth + 1);

        let new_mass = self.total_mass + nodes[node_idx].mass;
        self.center_of_mass.0 = (self.center_of_mass.0 * self.total_mass + nodes[node_idx].x * nodes[node_idx].mass) / new_mass;
        self.center_of_mass.1 = (self.center_of_mass.1 * self.total_mass + nodes[node_idx].y * nodes[node_idx].mass) / new_mass;
        self.total_mass = new_mass;
    }

    fn get_quadrant(&self, x: f32, y: f32, x_mid: f32, y_mid: f32) -> usize {
        (if x >= x_mid { 1 } else { 0 }) + (if y >= y_mid { 2 } else { 0 })
    }

    fn calculate_force(&self, node_idx: usize, nodes: &[Node], force: &mut (f32, f32)) {
        if self.node_idx == Some(node_idx) || self.total_mass == 0.0 {
            return;
        }

        let dx = self.center_of_mass.0 - nodes[node_idx].x;
        let dy = self.center_of_mass.1 - nodes[node_idx].y;
        let dist_sq = dx * dx + dy * dy + 1e-4; // Larger epsilon
        let dist = dist_sq.sqrt();

        let (x_min, y_min, x_max, y_max) = self.boundary;
        let width = x_max - x_min;

        // If node is far enough, treat as single particle
        if width / dist < 0.5 || self.children.is_none() {
            let f = 1000.0 * self.total_mass / dist_sq;
            force.0 += f * dx / dist;
            force.1 += f * dy / dist;
        } else {
            for child in self.children.as_ref().unwrap().iter() {
                child.calculate_force(node_idx, nodes, force);
            }
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = lines.next().unwrap().unwrap();
    let parts: Vec<usize> = first_line.split_whitespace().map(|s| s.parse().unwrap()).collect();
    let n = parts[0];
    let m = parts[1];

    let mut adj = vec![Vec::new(); n];
    for _ in 0..m {
        if let Some(Ok(line)) = lines.next() {
            let edge: Vec<usize> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
            adj[edge[0]].push(edge[1]);
            adj[edge[1]].push(edge[0]);
        }
    }

    let mut rng = thread_rng();
    let mut nodes = Vec::with_capacity(n);
    for _ in 0..n {
        nodes.push(Node {
            x: rng.gen_range(-500.0..500.0),
            y: rng.gen_range(-500.0..500.0),
            vx: 0.0,
            vy: 0.0,
            mass: 1.0,
        });
    }

    for iter in 0..50 {
        // Build Quadtree
        let mut root = QuadNode::new(-10000.0, -10000.0, 10000.0, 10000.0);
        for i in 0..n {
            root.insert(i, &nodes, 0);
        }

        let mut forces = vec![(0.0, 0.0); n];

        // Repulsion (Barnes-Hut)
        for i in 0..n {
            root.calculate_force(i, &nodes, &mut forces[i]);
        }

        // Attraction
        for i in 0..n {
            for &neighbor in &adj[i] {
                let dx = nodes[neighbor].x - nodes[i].x;
                let dy = nodes[neighbor].y - nodes[i].y;
                let dist_sq = dx * dx + dy * dy + 1e-4;
                let dist = dist_sq.sqrt();
                let f = dist_sq / 50.0;
                forces[i].0 += f * dx / dist;
                forces[i].1 += f * dy / dist;
            }
        }

        // Integration with cooling
        let temp = 1.0 - (iter as f32 / 50.0);
        for i in 0..n {
            nodes[i].vx = (nodes[i].vx + forces[i].0) * 0.5;
            nodes[i].vy = (nodes[i].vy + forces[i].1) * 0.5;
            nodes[i].x += nodes[i].vx * temp;
            nodes[i].y += nodes[i].vy * temp;
        }
    }

    for node in nodes {
        println!("{:.4} {:.4}", node.x, node.y);
    }
}
