const fs = require('fs');

/**
 * Calculates shortest path distance using BFS on the grid.
 * Grid is '.' for open, '#' for blocked.
 */
function getShortestPath(grid, start, end) {
  const rows = grid.length;
  const cols = grid[0].length;
  const queue = [[start.x, start.y, 0]]; // [x, y, dist]
  const visited = new Set();
  visited.add(`${start.x},${start.y}`);

  while (queue.length > 0) {
    const [x, y, dist] = queue.shift();

    if (x === end.x && y === end.y) {
      return dist;
    }

    const neighbors = [
      [x + 1, y], [x - 1, y], [x, y + 1], [x, y - 1]
    ];

    for (const [nx, ny] of neighbors) {
      if (nx >= 0 && nx < rows && ny >= 0 && ny < cols &&
          grid[nx][ny] === '.' && !visited.has(`${nx},${ny}`)) {
        visited.add(`${nx},${ny}`);
        queue.push([nx, ny, dist + 1]);
      }
    }
  }
  return Infinity; // Should not happen based on problem constraints
}

function solve() {
  const input = fs.readFileSync(0, 'utf8');
  const { grid, robots, jobs } = JSON.parse(input);

  // Precompute distances between every robot and every job
  const distances = robots.map(robot =>
    jobs.map(job => getShortestPath(grid, {x: robot.x, y: robot.y}, {x: job.x, y: job.y}))
  );

  const assignments = robots.map(() => []);
  const remainingJobs = new Set(jobs.map((_, index) => index));

  // Greedy assignment: assign closest available job to robots iteratively
  while (remainingJobs.size > 0) {
    for (let i = 0; i < robots.length; i++) {
      let bestJobIdx = -1;
      let minDistance = Infinity;

      for (const jobIdx of remainingJobs) {
        if (distances[i][jobIdx] < minDistance) {
          minDistance = distances[i][jobIdx];
          bestJobIdx = jobIdx;
        }
      }

      if (bestJobIdx !== -1) {
        assignments[i].push(jobs[bestJobIdx].id);
        remainingJobs.delete(bestJobIdx);
      }
    }
  }

  console.log(JSON.stringify({ assignments: assignments.map(robotJobs => robotJobs.map(j => j)) }));
}

solve();
