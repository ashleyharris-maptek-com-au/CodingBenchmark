; SPIR-V assembly for Geometry Shader
; InputTrianglesAdjacency, OutputTriangleStrip, 3 vertices

               OpCapability Geometry
               OpCapability Shader
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
               OpExecutionMode %main InputTrianglesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 3
               OpExecutionMode %main Invocations 1

; --- Decorators/Types ---
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %worldPos Location 0
               OpDecorate %normal Location 1
               OpDecorate %color Location 2
               OpDecorate %outColor Location 0

%void = OpTypeVoid
%func = OpTypeFunction %void
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_6 = OpConstant %uint 6
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3
%int_4 = OpConstant %int 4
%float_0 = OpConstant %float 0.0
%float_1 = OpConstant %float 1.0
%float_0_2 = OpConstant %float 0.2
%float_0_8 = OpConstant %float 0.8

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_6
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3 = OpTypeArray %v3float %uint_6
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3
%worldPos = OpVariable %ptr_in_arr_v3 Input
%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

; --- Main ---
%main = OpFunction %void None %func
%label = OpLabel

; 1. Main triangle vertices at gl_in indices 0, 2, 4.
; Get positions
%ptr_in_p0 = OpAccessChain %ptr_in_arr_pv %gl_in %int_0 %int_0
%pos0 = OpLoad %v4float %ptr_in_p0
%ptr_in_p2 = OpAccessChain %ptr_in_arr_pv %gl_in %int_2 %int_0
%pos2 = OpLoad %v4float %ptr_in_p2
%ptr_in_p4 = OpAccessChain %ptr_in_arr_pv %gl_in %int_4 %int_0
%pos4 = OpLoad %v4float %ptr_in_p4

; 2. Compute face normal: N = normalize(cross(p2-p0, p4-p0)).
%p0_v3 = OpVectorShuffle %v3float %pos0 %pos0 0 1 2
%p2_v3 = OpVectorShuffle %v3float %pos2 %pos2 0 1 2
%p4_v3 = OpVectorShuffle %v3float %pos4 %pos4 0 1 2
%v2_p0 = OpFSub %v3float %p2_v3 %p0_v3
%v4_p0 = OpFSub %v3float %p4_v3 %p0_v3
%cross = OpExtInst %v3float %1 Cross %v2_p0 %v4_p0
%normal_v = OpExtInst %v3float %1 Normalize %cross

; 3. Light direction: L = normalize(vec3(1, 1, 1)).
%l_vec = OpCompositeConstruct %v3float %float_1 %float_1 %float_1
%l_norm = OpExtInst %v3float %1 Normalize %l_vec

; 4. Diffuse = max(dot(N, L), 0.0).
%dot = OpDot %float %normal_v %l_norm
%diffuse = OpExtInst %float %1 FMax %dot %float_0

; 5. Base color = average of color[0], color[2], color[4].
%ptr_c0 = OpAccessChain %ptr_in_arr_v3 %color %int_0
%c0 = OpLoad %v3float %ptr_c0
%ptr_c2 = OpAccessChain %ptr_in_arr_v3 %color %int_2
%c2 = OpLoad %v3float %ptr_c2
%ptr_c4 = OpAccessChain %ptr_in_arr_v3 %color %int_4
%c4 = OpLoad %v3float %ptr_c4
%c_sum = OpFAdd %v3float %c0 %c2
%c_sum2 = OpFAdd %v3float %c_sum %c4
%float_3 = OpConstant %float 3.0
%div3 = OpCompositeConstruct %v3float %float_3 %float_3 %float_3
%base_color_c = OpFDiv %v3float %c_sum2 %div3

; 6. Final color = base * (0.2 + 0.8 * diffuse).
%d_factor = OpFMul %float %diffuse %float_0_8
%l_factor = OpFAdd %float %d_factor %float_0_2
%l_factor_v = OpCompositeConstruct %v3float %l_factor %l_factor %l_factor
%final_color = OpFMul %v3float %base_color_c %l_factor_v

; 7. Emit the 3 triangle vertices with flat-shaded color.
; Vertices are at indices 0, 2, 4
%emit_v0 = OpLabel
%ptr_idx0 = OpAccessChain %ptr_in_arr_pv %gl_in %int_0 %int_0
%pos0_emit = OpLoad %v4float %ptr_idx0
%ptr_out_pos0 = OpAccessChain %ptr_out_pv %gl_out %int_0
OpStore %ptr_out_pos0 %pos0_emit
OpStore %outColor %final_color
OpEmitVertex

%emit_v2 = OpLabel
%ptr_idx2 = OpAccessChain %ptr_in_arr_pv %gl_in %int_2 %int_0
%pos2_emit = OpLoad %v4float %ptr_idx2
%ptr_out_pos2 = OpAccessChain %ptr_out_pv %gl_out %int_0
OpStore %ptr_out_pos2 %pos2_emit
OpStore %outColor %final_color
OpEmitVertex

%emit_v4 = OpLabel
%ptr_idx4 = OpAccessChain %ptr_in_arr_pv %gl_in %int_4 %int_0
%pos4_emit = OpLoad %v4float %ptr_idx4
%ptr_out_pos4 = OpAccessChain %ptr_out_pv %gl_out %int_0
OpStore %ptr_out_pos4 %pos4_emit
OpStore %outColor %final_color
OpEmitVertex

OpEndPrimitive
OpReturn
OpFunctionEnd
