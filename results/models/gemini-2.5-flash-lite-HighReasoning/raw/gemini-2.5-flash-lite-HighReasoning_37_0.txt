using System;
using System.Collections.Generic;
using System.Linq;

namespace DFVSSolver
{
    public class Graph
    {
        public int VerticesCount { get; }
        public List<int>[] AdjacencyList { get; }

        public Graph(int verticesCount)
        {
            VerticesCount = verticesCount;
            AdjacencyList = new List<int>[verticesCount];
            for (int i = 0; i < verticesCount; i++)
                AdjacencyList[i] = new List<int>();
        }

        public void AddEdge(int u, int v)
        {
            AdjacencyList[u].Add(v);
        }

        public bool IsAcyclic()
        {
            // Simple cycle detection using DFS
            var visited = new int[VerticesCount]; // 0: unvisited, 1: visiting, 2: visited
            for (int i = 0; i < VerticesCount; i++)
            {
                if (visited[i] == 0 && HasCycle(i, visited))
                    return false;
            }
            return true;
        }

        private bool HasCycle(int u, int[] visited)
        {
            visited[u] = 1;
            foreach (var v in AdjacencyList[u])
            {
                if (visited[v] == 1) return true;
                if (visited[v] == 0 && HasCycle(v, visited)) return true;
            }
            visited[u] = 2;
            return false;
        }
    }

    class Solver
    {
        private static HashSet<int> bestFvs;

        public static HashSet<int> FindDFVS(Graph g)
        {
            bestFvs = new HashSet<int>(Enumerable.Range(0, g.VerticesCount)); // Initial upper bound: all vertices
            SolveRecursive(g, new HashSet<int>());
            return bestFvs;
        }

        private static void SolveRecursive(Graph g, HashSet<int> currentFvs)
        {
            if (currentFvs.Count >= bestFvs.Count) return; // Pruning

            // A simplified check for acyclicity without physically removing vertices
            // (Note: This is still inefficient; in a production solver, 
            // you'd maintain a reduced graph or use cycle-based branching.)
            if (IsGraphAcyclicWithRemoval(g, currentFvs))
            {
                bestFvs = new HashSet<int>(currentFvs);
                return;
            }

            for (int i = 0; i < g.VerticesCount; i++)
            {
                if (currentFvs.Contains(i)) continue;
                
                currentFvs.Add(i);
                SolveRecursive(g, currentFvs);
                currentFvs.Remove(i);
            }
        }

        private static bool IsGraphAcyclicWithRemoval(Graph g, HashSet<int> removedVertices)
        {
            var visited = new int[g.VerticesCount];
            for (int i = 0; i < g.VerticesCount; i++)
            {
                if (removedVertices.Contains(i)) continue;
                if (visited[i] == 0 && HasCycle(g, i, visited, removedVertices))
                    return false;
            }
            return true;
        }

        private static bool HasCycle(Graph g, int u, int[] visited, HashSet<int> removedVertices)
        {
            visited[u] = 1;
            foreach (var v in g.AdjacencyList[u])
            {
                if (removedVertices.Contains(v)) continue;
                if (visited[v] == 1) return true;
                if (visited[v] == 0 && HasCycle(g, v, visited, removedVertices)) return true;
            }
            visited[u] = 2;
            return false;
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            var g = new Graph(3);
            g.AddEdge(0, 1);
            g.AddEdge(1, 2);
            g.AddEdge(2, 0); // Cycle: 0->1->2->0

            var fvs = Solver.FindDFVS(g);
            Console.WriteLine("Minimum Feedback Vertex Set: " + string.Join(", ", fvs));
        }
    }
}
