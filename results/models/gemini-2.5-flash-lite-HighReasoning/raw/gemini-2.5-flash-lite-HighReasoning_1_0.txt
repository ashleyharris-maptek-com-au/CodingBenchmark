#include <iostream>
#include <vector>
#include <cmath>
#include <algorithm>
#include <iomanip>

struct City {
    double x, y;
};

// Use a fast squared distance for comparisons
double distSq(const City& a, const City& b) {
    double dx = a.x - b.x;
    double dy = a.y - b.y;
    return dx * dx + dy * dy;
}

// Simple greedy construction
std::vector<int> solve_greedy(int n, const std::vector<City>& cities) {
    std::vector<int> tour;
    tour.reserve(n);
    std::vector<bool> visited(n, false);
    
    tour.push_back(0);
    visited[0] = true;
    
    int current = 0;
    for (int i = 1; i < n; ++i) {
        int next = -1;
        double min_dist = 1e30; // Large enough
        
        // TODO: This is O(N^2), needs optimization for huge N.
        // For N=10^6, this must be improved.
        for (int j = 0; j < n; ++j) {
            if (!visited[j]) {
                double d = distSq(cities[current], cities[j]);
                if (d < min_dist) {
                    min_dist = d;
                    next = j;
                }
            }
        }
        
        tour.push_back(next);
        visited[next] = true;
        current = next;
    }
    return tour;
}

int main() {
    // Speed up I/O
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    int n;
    if (!(std::cin >> n)) return 0;
    
    std::vector<City> cities(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> cities[i].x >> cities[i].y;
    }
    
    std::vector<int> tour = solve_greedy(n, cities);
    
    for (int i = 0; i < n; ++i) {
        std::cout << tour[i] << (i == n - 1 ? "" : " ");
    }
    std::cout << std::endl;
    
    return 0;
}
