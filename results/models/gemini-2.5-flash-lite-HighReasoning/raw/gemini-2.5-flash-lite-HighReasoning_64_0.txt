declare const require: any;
const fs = require('fs');

interface Drone {
    id: number;
    x: number;
    y: number;
}

interface InputData {
    probability: number[][];
    drones: Drone[];
    maxStepsPerDrone: number;
}

interface OutputData {
    paths: [number, number][][];
}

function solve() {
    const input: string = fs.readFileSync(0, 'utf8');
    const data: InputData = JSON.parse(input);

    const { probability, drones, maxStepsPerDrone } = data;
    const height = probability.length;
    const width = probability[0].length;

    // Keep track of visited cells (to avoid double counting in score)
    // Actually, the problem says "score is unique probability covered",
    // implying if two drones cover the same cell, it only counts once.
    // For greedy, we can use a local 'covered' grid.
    const covered = Array.from({ length: height }, () => Array(width).fill(false));

    const paths: [number, number][][] = drones.map(() => []);

    // Current positions of drones
    const currentPositions = drones.map(d => ({ x: d.x, y: d.y }));

    // Initialize paths with starting positions
    for (let i = 0; i < drones.length; i++) {
        paths[i].push([currentPositions[i].x, currentPositions[i].y]);
        covered[currentPositions[i].y][currentPositions[i].x] = true;
    }

    // Directions: Stay, Up, Down, Left, Right
    const moves = [
        [0, 0], [0, -1], [0, 1], [-1, 0], [1, 0]
    ];

    for (let step = 0; step < maxStepsPerDrone; step++) {
        for (let i = 0; i < drones.length; i++) {
            let bestMove = { dx: 0, dy: 0 };
            let maxGain = -1;

            const { x, y } = currentPositions[i];

            for (const [dx, dy] of moves) {
                const nx = x + dx;
                const ny = y + dy;

                if (nx >= 0 && nx < width && ny >= 0 && ny < height && probability[ny][nx] !== -1) {
                    let gain = 0;
                    if (!covered[ny][nx]) {
                        gain = probability[ny][nx];
                    }

                    if (gain > maxGain) {
                        maxGain = gain;
                        bestMove = { dx, dy };
                    }
                }
            }

            // Apply best move
            currentPositions[i].x += bestMove.dx;
            currentPositions[i].y += bestMove.dy;
            
            paths[i].push([currentPositions[i].x, currentPositions[i].y]);
            covered[currentPositions[i].y][currentPositions[i].x] = true;
        }
    }

    const output: OutputData = { paths };
    process.stdout.write(JSON.stringify(output));
}

solve();
