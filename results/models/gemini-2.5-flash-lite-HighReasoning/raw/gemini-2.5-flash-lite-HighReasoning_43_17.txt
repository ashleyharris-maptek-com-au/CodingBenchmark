struct PSInput {
    [[vk::location(0)]] float3 worldPos : TEXCOORD0;
    [[vk::location(1)]] float3 normal   : TEXCOORD1;
    [[vk::location(2)]] float2 uv       : TEXCOORD2;
    [[vk::location(3)]] float3 tangent  : TEXCOORD3;
    [[vk::location(4)]] float3 color    : TEXCOORD4;
};

[[vk::binding(0, 0)]]
cbuffer UBO : register(b0) {
    float4x4 model;
    float4x4 view;
    float4x4 projection;
    float4 lightPos;
    float4 cameraPos;
    float4 params;
};

float4 main(PSInput input) : SV_Target0 {
    float2 scale = float2(8.0, 16.0);
    float2 uv = input.uv * scale;

    // Stagger rows: every other row is shifted by 0.5
    float row = floor(uv.y);
    if (fmod(row, 2.0) >= 1.0) {
        uv.x += 0.5;
    }

    float2 cell = floor(uv);
    float2 localUV = frac(uv);

    // Mortar thickness
    float2 mortar = float2(0.05, 0.1);
    
    // Check if within mortar
    bool isMortar = (localUV.x < mortar.x) || (localUV.x > 1.0 - mortar.x) ||
                    (localUV.y < mortar.y) || (localUV.y > 1.0 - mortar.y);
    
    float3 brickColor = float3(0.7, 0.2, 0.1);
    float3 mortarColor = float3(0.7, 0.7, 0.7);
    
    float3 baseColor = isMortar ? mortarColor : brickColor;
    
    // Lighting
    float3 N = normalize(input.normal);
    float3 L = normalize(lightPos.xyz - input.worldPos);
    
    float diffuse = max(dot(N, L), 0.0);
    float shading = 0.4 + 0.6 * diffuse;
    
    return float4(baseColor * shading, 1.0);
}
