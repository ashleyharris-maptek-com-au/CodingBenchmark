use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = lines.next().unwrap().unwrap();
    let parts: Vec<usize> = first_line
        .split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();
    let n = parts[0];
    let m = parts[1];

    let mut adj = vec![vec![false; n]; n];
    for _ in 0..m {
        let line = lines.next().unwrap().unwrap();
        let edge: Vec<usize> = line
            .split_whitespace()
            .map(|s| s.parse().unwrap())
            .collect();
        let u = edge[0];
        let v = edge[1];
        adj[u][v] = true;
        adj[v][u] = true;
    }

    let mut min_cut = usize::MAX;
    let mut best_partition = Vec::new();

    let mut current_partition = Vec::new();
    let mut used = vec![false; n];

    fn find_min_cut(
        n: usize,
        adj: &Vec<Vec<bool>>,
        k: usize,
        start: usize,
        current_partition: &mut Vec<usize>,
        used: &mut Vec<bool>,
        min_cut: &mut usize,
        best_partition: &mut Vec<usize>,
    ) {
        if current_partition.len() == n / 2 {
            let mut cut = 0;
            let mut in_partition = vec![false; n];
            for &v in current_partition.iter() {
                in_partition[v] = true;
            }

            for &u in current_partition.iter() {
                for v in 0..n {
                    if !in_partition[v] && adj[u][v] {
                        cut += 1;
                    }
                }
            }

            if cut < *min_cut {
                *min_cut = cut;
                *best_partition = current_partition.clone();
            }
            return;
        }

        for i in start..n {
            if !used[i] {
                used[i] = true;
                current_partition.push(i);
                find_min_cut(n, adj, k, i + 1, current_partition, used, min_cut, best_partition);
                current_partition.pop();
                used[i] = false;
            }
        }
    }

    find_min_cut(n, &adj, n / 2, 0, &mut current_partition, &mut used, &mut min_cut, &mut best_partition);

    println!("{}", min_cut);
    let output: Vec<String> = best_partition.iter().map(|v| v.to_string()).collect();
    println!("{}", output.join(" "));
}
