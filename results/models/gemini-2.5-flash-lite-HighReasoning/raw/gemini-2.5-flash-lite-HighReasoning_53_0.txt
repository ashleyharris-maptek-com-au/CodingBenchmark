import numpy as np
from dynamics import SpacecraftDynamics

class DockingAutopilot:
    def __init__(self, n, dt):
        self.dynamics = SpacecraftDynamics(n)
        self.dt = dt
        
    def calculate_thrust(self, state):
        # Proportional feedback control
        # K_pos and K_vel need to be tuned to counteract CW dynamics
        K_pos = 0.05
        K_vel = 0.5
        
        thrust = -K_pos * state[:3] - K_vel * state[3:]
        return thrust

    def run_simulation(self, initial_state, duration):
        state = initial_state
        trajectory = [state]
        
        for _ in range(int(duration / self.dt)):
            thrust = self.calculate_thrust(state)
            # Limit thrust
            thrust = np.clip(thrust, -1.0, 1.0)
            state = self.dynamics.step(state, thrust, self.dt)
            trajectory.append(state)
            
        return np.array(trajectory)
