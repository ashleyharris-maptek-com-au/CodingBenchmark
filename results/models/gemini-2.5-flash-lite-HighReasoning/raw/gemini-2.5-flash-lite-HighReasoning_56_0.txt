const fs = require('fs');

function solve() {
  const input = fs.readFileSync(0, 'utf8');
  if (!input) return;
  const data = JSON.parse(input);
  const { depot, truckCount, packages } = data;

  if (packages.length === 0) {
    console.log(JSON.stringify({ routes: Array(truckCount).fill([]) }));
    return;
  }

  // 1. Group packages by proximity to minimize initial cluster distances
  // Simple approach: Sort by angle relative to depot (polar coordinate clustering)
  const packagesWithAngle = packages.map(pkg => ({
    ...pkg,
    angle: Math.atan2(pkg.y - depot[1], pkg.x - depot[0])
  }));

  packagesWithAngle.sort((a, b) => a.angle - b.angle);

  const routes = Array.from({ length: truckCount }, () => []);

  // Distribute packages among trucks
  for (let i = 0; i < packagesWithAngle.length; i++) {
    routes[i % truckCount].push(packagesWithAngle[i]);
  }

  // 2. Within each route, optimize using nearest neighbor
  const optimizedRoutes = routes.map(routePackages => {
    if (routePackages.length === 0) return [];

    const orderedIds = [];
    let currentPos = { x: depot[0], y: depot[1] };
    const remaining = [...routePackages];

    while (remaining.length > 0) {
      let nearestIdx = 0;
      let minDistance = Infinity;

      for (let i = 0; i < remaining.length; i++) {
        const dist = Math.sqrt(
          Math.pow(remaining[i].x - currentPos.x, 2) +
          Math.pow(remaining[i].y - currentPos.y, 2)
        );
        if (dist < minDistance) {
          minDistance = dist;
          nearestIdx = i;
        }
      }

      const next = remaining.splice(nearestIdx, 1)[0];
      orderedIds.push(next.id);
      currentPos = { x: next.x, y: next.y };
    }

    return orderedIds;
  });

  process.stdout.write(JSON.stringify({ routes: optimizedRoutes }));
}

solve();
