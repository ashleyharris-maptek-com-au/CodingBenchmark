const fs = require('fs');

/**
 * Solves the drone pathfinding problem using a greedy approach.
 * Each drone, at each step, looks for the best immediate local reward
 * while staying within bounds and avoiding obstacles.
 */
function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);

    const { probability, drones, maxStepsPerDrone } = data;
    const height = probability.length;
    const width = probability[0].length;

    const visited = new Set();
    const paths = [];

    // Initialize visited for starting positions
    drones.forEach(d => {
        visited.add(`${d.x},${d.y}`);
    });

    for (const drone of drones) {
        let { x, y } = drone;
        const path = [[x, y]];

        for (let step = 0; step < maxStepsPerDrone; step++) {
            let bestMove = { dx: 0, dy: 0 };
            let maxReward = -Infinity;

            // Possible moves: stay, up, down, left, right
            const moves = [
                { dx: 0, dy: 0 },
                { dx: 0, dy: -1 },
                { dx: 0, dy: 1 },
                { dx: -1, dy: 0 },
                { dx: 1, dy: 0 }
            ];

            for (const move of moves) {
                const nextX = x + move.dx;
                const nextY = y + move.dy;

                // Check bounds and obstacles
                if (nextX >= 0 && nextX < width && nextY >= 0 && nextY < height &&
                    probability[nextY][nextX] !== -1) {
                    
                    // Simple greedy heuristic: probability of next cell
                    // If already visited, reward is 0
                    let reward = 0;
                    if (!visited.has(`${nextX},${nextY}`)) {
                        reward = probability[nextY][nextX];
                    }

                    if (reward > maxReward) {
                        maxReward = reward;
                        bestMove = move;
                    }
                }
            }

            x += bestMove.dx;
            y += bestMove.dy;
            path.push([x, y]);
            visited.add(`${x},${y}`);
        }
        paths.push(path);
    }

    console.log(JSON.stringify({ paths }));
}

solve();
