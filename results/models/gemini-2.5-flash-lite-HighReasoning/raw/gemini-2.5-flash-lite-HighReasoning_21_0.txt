#include <iostream>
#include <vector>
#include <deque>
#include <unordered_set>
#include <cmath>
#include <algorithm>
#include <queue>

struct Point {
    std::vector<int> coords;

    bool operator==(const Point& other) const {
        return coords == other.coords;
    }
};

struct PointHash {
    std::size_t operator()(const Point& p) const {
        std::size_t seed = 0;
        for (int x : p.coords) {
            seed ^= std::hash<int>{}(x) + 0x9e3779b9 + (seed << 6) + (seed >> 2);
        }
        return seed;
    }
};

class SnakeGame {
public:
    int D;
    int max_turns;
    std::vector<int> bounds;
    std::deque<Point> snake;
    std::unordered_set<Point, PointHash> food;
    std::unordered_set<Point, PointHash> obstacles;

    int manhattan_distance(const Point& p1, const Point& p2) {
        int dist = 0;
        for (int i = 0; i < D; ++i) dist += std::abs(p1.coords[i] - p2.coords[i]);
        return dist;
    }

    bool is_valid(const Point& p) {
        for (int i = 0; i < D; ++i) {
            if (p.coords[i] < 0 || p.coords[i] >= bounds[i]) return false;
        }
        if (obstacles.count(p)) return false;
        for (const auto& segment : snake) {
            if (p == segment) return false;
        }
        return true;
    }

    void run() {
        for (int turn = 0; turn < max_turns; ++turn) {
            if (food.empty()) break;

            // Simple greedy approach: find closest food
            Point target = *food.begin();
            int min_dist = 1e9;
            for (const auto& f : food) {
                int d = manhattan_distance(snake.front(), f);
                if (d < min_dist) {
                    min_dist = d;
                    target = f;
                }
            }

            // Find next move
            Point head = snake.front();
            for (int i = 0; i < D; ++i) {
                for (int dir : {-1, 1}) {
                    Point next = head;
                    next.coords[i] += dir;
                    if (is_valid(next)) {
                        std::cout << i << " " << dir << std::endl;
                        // Move snake
                        snake.push_front(next);
                        if (food.count(next)) {
                            food.erase(next);
                        } else {
                            snake.pop_back();
                        }
                        goto next_turn;
                    }
                }
            }
            break; // No move found
            next_turn:;
        }
    }
};

int main() {
    int D, max_turns;
    if (!(std::cin >> D >> max_turns)) return 0;

    SnakeGame game;
    game.D = D;
    game.max_turns = max_turns;
    game.bounds.resize(D);
    for (int i = 0; i < D; ++i) std::cin >> game.bounds[i];

    int snake_length;
    std::cin >> snake_length;
    Point head;
    head.coords.resize(D);
    for (int i = 0; i < D; ++i) std::cin >> head.coords[i];
    game.snake.push_back(head);
    // Note: Assuming initial snake length is 1 for now, or need to handle body construction

    int F;
    std::cin >> F;
    for (int i = 0; i < F; ++i) {
        Point p;
        p.coords.resize(D);
        for (int j = 0; j < D; ++j) std::cin >> p.coords[j];
        game.food.insert(p);
    }

    int O;
    std::cin >> O;
    for (int i = 0; i < O; ++i) {
        Point p;
        p.coords.resize(D);
        for (int j = 0; j < D; ++j) std::cin >> p.coords[j];
        game.obstacles.insert(p);
    }

    game.run();
    return 0;
}
