use std::io::{self, BufRead};

#[derive(Debug, Clone, Copy)]
struct Vec3 {
    x: f64,
    y: f64,
    z: f64,
}

#[derive(Debug, Clone, Copy)]
struct Quat {
    w: f64,
    x: f64,
    y: f64,
    z: f64,
}

struct Tetrahedron {
    pos: Vec3,
    rot: Quat,
    scale: f64,
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let nv: usize = lines.next().unwrap().unwrap().trim().parse().unwrap();
    let mut vertices = Vec::with_capacity(nv);
    for _ in 0..nv {
        let line = lines.next().unwrap().unwrap();
        let coords: Vec<f64> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
        vertices.push(Vec3 { x: coords[0], y: coords[1], z: 0.0 });
    }

    let sun_line = lines.next().unwrap().unwrap();
    let sun: Vec<f64> = sun_line.split_whitespace().map(|s| s.parse().unwrap()).collect();
    let _sun_dir = Vec3 { x: sun[0], y: sun[1], z: sun[2] };

    // Simple strategy: cover the polygon with a grid of tetrahedrons
    // For a real solution, we would need a more sophisticated covering algorithm.
    // This is a placeholder to demonstrate the output format.
    let mut tetrahedrons = Vec::new();
    tetrahedrons.push(Tetrahedron {
        pos: Vec3 { x: 0.0, y: 0.0, z: 0.0 },
        rot: Quat { w: 1.0, x: 0.0, y: 0.0, z: 0.0 },
        scale: 1.0,
    });

    println!("{}", tetrahedrons.len());
    for t in tetrahedrons {
        println!("{:.6} {:.6} {:.6} {:.6} {:.6} {:.6} {:.6} {:.6}", 
            t.pos.x, t.pos.y, t.pos.z, 
            t.rot.w, t.rot.x, t.rot.y, t.rot.z, 
            t.scale);
    }
}
