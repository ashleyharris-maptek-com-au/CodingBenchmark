const fs = require('fs');

function calculateDistance(point1, point2) {
    const dx = point1.x - point2.x;
    const dy = point1.y - point2.y;
    return Math.sqrt(dx * dx + dy * dy);
}

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);

    const depot = { x: data.depot[0], y: data.depot[1] };
    const truckCount = data.truckCount;
    const packages = data.packages;

    if (packages.length === 0) {
        process.stdout.write(JSON.stringify({ routes: [] }));
        return;
    }

    // Initialize routes
    const routes = Array.from({ length: truckCount }, () => []);
    const currentRouteEndPoints = Array.from({ length: truckCount }, () => depot);
    const assignedPackages = new Set();

    // --- NEW: Initial Assignment Phase ---
    // Sort packages by distance to depot
    const packagesSortedByDepotDistance = [...packages].sort((a, b) => {
        return calculateDistance(depot, a) - calculateDistance(depot, b);
    });

    let truckIndex = 0;
    for (const pkg of packagesSortedByDepotDistance) {
        if (assignedPackages.size === packages.length || truckIndex >= truckCount) {
            break; // All packages assigned or all trucks initialized
        }

        if (!assignedPackages.has(pkg.id)) {
            routes[truckIndex].push(pkg.id);
            currentRouteEndPoints[truckIndex] = pkg;
            assignedPackages.add(pkg.id);
            truckIndex++;
        }
    }
    // --- END NEW: Initial Assignment Phase ---


    // --- Existing: Main Assignment Loop ---
    while (assignedPackages.size < packages.length) {
        let bestPackage = null;
        let bestRouteIndex = -1;
        let minIncreasedDistance = Infinity;

        // Iterate through all unassigned packages
        for (const pkg of packages) {
            if (assignedPackages.has(pkg.id)) {
                continue;
            }

            // Try to add this package to each available route
            for (let i = 0; i < truckCount; i++) {
                const currentEndPoint = currentRouteEndPoints[i];
                const distanceToAdd = calculateDistance(currentEndPoint, pkg);

                if (distanceToAdd < minIncreasedDistance) {
                    minIncreasedDistance = distanceToAdd;
                    bestPackage = pkg;
                    bestRouteIndex = i;
                }
            }
        }

        if (bestPackage) {
            routes[bestRouteIndex].push(bestPackage.id);
            currentRouteEndPoints[bestRouteIndex] = bestPackage;
            assignedPackages.add(bestPackage.id);
        } else {
            // This should theoretically not happen if there are unassigned packages
            break;
        }
    }
    // --- END Existing: Main Assignment Loop ---


    // Filter out empty routes if truckCount > actual routes used (e.g. if packages.length < truckCount)
    const finalRoutes = routes.filter(route => route.length > 0);

    process.stdout.write(JSON.stringify({ routes: finalRoutes }));
}

solve();
