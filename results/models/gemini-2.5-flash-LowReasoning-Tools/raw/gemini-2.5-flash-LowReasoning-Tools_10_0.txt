#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>

// Structure to represent a point in 2D space
struct Point {
    int x, y;
};

// Structure to represent a rectangle
struct Rect {
    int id; // Original ID of the rectangle
    int board_id; // ID of the board this rectangle is placed on
    int width, height;
    int x, y; // Position on the board
    bool rotated; // True if rotated 90 degrees
    long long area; // Pre-calculated area

    Rect(int _id, int w, int h) : id(_id), board_id(-1), width(w), height(h), x(0), y(0), rotated(false), area((long long)w * h) {}

    // Method to get current width considering rotation
    int current_width() const {
        return rotated ? height : width;
    }

    // Method to get current height considering rotation
    int current_height() const {
        return rotated ? width : height;
    }

    // Check if this rectangle contains another rectangle 'other'
    bool contains(const Rect& other) const {
        return x <= other.x && y <= other.y &&
               x + current_width() >= other.x + other.current_width() &&
               y + current_height() >= other.y + other.current_height();
    }

    // Check if this rectangle overlaps with another rectangle 'other'
    bool intersects(const Rect& other) const {
        return x < other.x + other.current_width() && x + current_width() > other.x &&
               y < other.y + other.current_height() && y + current_height() > other.y;
    }
};

// Structure to represent a board
struct Board {
    int id;
    int board_width, board_height;
    std::vector<Rect*> placed_rects; // Pointers to rectangles placed on this board

    // Nested struct to manage free spaces
    struct FreeSpaceManager {
        std::vector<Rect> free_spaces;

        FreeSpaceManager(int bw, int bh) {
            free_spaces.emplace_back(-1, bw, bh); // id -1 for free spaces
            free_spaces.back().x = 0;
            free_spaces.back().y = 0;
        }

        // Methods for managing free spaces will go here
        std::vector<Rect> split_rect(const Rect& fs_to_split, const Rect& placed_rect) {
            std::vector<Rect> result;

            // No intersection, the original free space remains intact
            if (!fs_to_split.intersects(placed_rect)) {
                result.push_back(fs_to_split);
                return result;
            }

            // If placed_rect completely contains fs_to_split, then fs_to_split is fully consumed.
            if (placed_rect.x <= fs_to_split.x &&
                placed_rect.y <= fs_to_split.y &&
                placed_rect.x + placed_rect.current_width() >= fs_to_split.x + fs_to_split.width &&
                placed_rect.y + placed_rect.current_height() >= fs_to_split.y + fs_to_split.height) {
                return result; // No free space remains from fs_to_split
            }

            // Calculate coordinates of the intersection
            int overlap_left = std::max(fs_to_split.x, placed_rect.x);
            int overlap_top = std::max(fs_to_split.y, placed_rect.y);
            int overlap_right = std::min(fs_to_split.x + fs_to_split.width, placed_rect.x + placed_rect.current_width());
            int overlap_bottom = std::min(fs_to_split.y + fs_to_split.height, placed_rect.y + placed_rect.current_height());

            // Add the parts of fs_to_split that are to the left of placed_rect
            if (fs_to_split.x < overlap_left) {
                result.emplace_back(-1, overlap_left - fs_to_split.x, fs_to_split.height);
                result.back().x = fs_to_split.x;
                result.back().y = fs_to_split.y;
            }
            // Add the parts of fs_to_split that are to the right of placed_rect
            if (fs_to_split.x + fs_to_split.width > overlap_right) {
                result.emplace_back(-1, fs_to_split.x + fs_to_split.width - overlap_right, fs_to_split.height);
                result.back().x = overlap_right;
                result.back().y = fs_to_split.y;
            }
            // Add the parts of fs_to_split that are above placed_rect (within the overlap x-range)
            if (fs_to_split.y < overlap_top) {
                result.emplace_back(-1, overlap_right - overlap_left, overlap_top - fs_to_split.y);
                result.back().x = overlap_left;
                result.back().y = fs_to_split.y;
            }
            // Add the parts of fs_to_split that are below placed_rect (within the overlap x-range)
            if (fs_to_split.y + fs_to_split.height > overlap_bottom) {
                result.emplace_back(-1, overlap_right - overlap_left, fs_to_split.y + fs_to_split.height - overlap_bottom);
                result.back().x = overlap_left;
                result.back().y = overlap_bottom;
            }
            
            // Filter out zero-area rectangles
            result.erase(std::remove_if(result.begin(), result.end(), [](const Rect& r) {
                return r.width <= 0 || r.height <= 0;
            }), result.end());

            for (auto& r : result) {
                r.area = (long long)r.width * r.height;
            }

            return result;
        }

        void add_placed_rectangle(const Rect& placed_rect) {
            std::vector<Rect> new_current_free_spaces;

            for (const auto& fs : free_spaces) {
                // Get the new free spaces generated by splitting 'fs' with 'placed_rect'
                std::vector<Rect> split_results = split_rect(fs, placed_rect);
                for (const auto& r : split_results) {
                    if (r.width > 0 && r.height > 0) { // Only add valid rectangles
                        new_current_free_spaces.push_back(r);
                    }
                }
            }
            free_spaces = new_current_free_spaces;

            // Pruning: Remove any free space that is contained within another free space
            // This is an O(N^2) operation, which can be slow for many free spaces.
            // For a robust solution, more efficient data structures for overlap detection
            // and pruning might be required (e.g., using a sweep-line algorithm or R-tree).
            std::vector<Rect> pruned_free_spaces;
            for (const auto& fs1 : free_spaces) {
                bool is_contained = false;
                for (const auto& fs2 : free_spaces) {
                    if (&fs1 == &fs2) continue; // Don't compare a rectangle with itself

                    // Check if fs1 is contained within fs2
                    if (fs2.x <= fs1.x && fs2.y <= fs1.y &&
                        fs2.x + fs2.width >= fs1.x + fs1.width &&
                        fs2.y + fs2.height >= fs1.y + fs1.height) {
                        is_contained = true;
                        break;
                    }
                }
                if (!is_contained) {
                    pruned_free_spaces.push_back(fs1);
                }
            }
            free_spaces = pruned_free_spaces;
        }
    };

    FreeSpaceManager fs_manager;

    Board(int _id, int bw, int bh) : id(_id), board_width(bw), board_height(bh), fs_manager(bw, bh) {
        // The FreeSpaceManager constructor now handles initial free space setup.
    }

    // Try to place a rectangle into a free space on this board
    // Returns true if placed, false otherwise
    bool place_rect(Rect* rect_to_place) {
        // Sort free spaces to prioritize smaller ones for a better fit
        std::sort(fs_manager.free_spaces.begin(), fs_manager.free_spaces.end(), [](const Rect& a, const Rect& b) {
            return a.area < b.area; // Best fit: smallest free area first
        });

        for (size_t i = 0; i < fs_manager.free_spaces.size(); ++i) {
            Rect& fs = fs_manager.free_spaces[i]; // Reference to the current free space

            // Try original orientation
            if (rect_to_place->width <= fs.width && rect_to_place->height <= fs.height) {
                rect_to_place->x = fs.x;
                rect_to_place->y = fs.y;
                rect_to_place->rotated = false;
                rect_to_place->board_id = id;
                placed_rects.push_back(rect_to_place);
                fs_manager.add_placed_rectangle(*rect_to_place); // Call the new method in FreeSpaceManager
                return true;
            }

            // Try rotated orientation
            if (rect_to_place->height <= fs.width && rect_to_place->width <= fs.height) {
                rect_to_place->x = fs.x;
                rect_to_place->y = fs.y;
                rect_to_place->rotated = true;
                rect_to_place->board_id = id;
                placed_rects.push_back(rect_to_place);
                fs_manager.add_placed_rectangle(*rect_to_place); // Call the new method in FreeSpaceManager
                return true;
            }
        }
        return false; // Could not place rectangle on this board
    }

private:
    // This method is now moved and renamed to fs_manager.add_placed_rectangle
    // Its contents should be empty or removed.
    void update_free_spaces(const Rect& used_space, const Rect& placed_rect) {}
};

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    int N;
    int board_width, board_height;
    std::cin >> N >> board_width >> board_height;

    std::vector<Rect> all_rects_data;
    all_rects_data.reserve(N);
    std::vector<Rect*> all_rects_ptr;
    all_rects_ptr.reserve(N);

    for (int i = 0; i < N; ++i) {
        int w, h;
        std::cin >> w >> h;
        all_rects_data.emplace_back(i, w, h);
        all_rects_ptr.push_back(&all_rects_data.back());
    }

    // Sort rectangles by decreasing height, then decreasing width (common heuristic)
    std::sort(all_rects_ptr.begin(), all_rects_ptr.end(), [](const Rect* a, const Rect* b) {
        if (a->height != b->height) {
            return a->height > b->height;
        }
        return a->width > b->width;
    });

    std::vector<Board> boards;
    
    for (Rect* rect_to_place : all_rects_ptr) {
        bool placed_on_existing_board = false;
        for (Board& board : boards) {
            if (board.place_rect(rect_to_place)) {
                placed_on_existing_board = true;
                break;
            }
        }

        if (!placed_on_existing_board) {
            // Create a new board and try to place the rectangle
            boards.emplace_back(boards.size(), board_width, board_height);
            boards.back().place_rect(rect_to_place); // This should always succeed on a new empty board
        }
    }

    // Output results
    std::cout << boards.size() << std::endl;
    for (int i = 0; i < N; ++i) {
        const Rect& r = all_rects_data[i];
        std::cout << r.board_id << " " << r.x << " " << r.y << " " << (r.rotated ? 1 : 0) << std::endl;
    }

    return 0;
}
