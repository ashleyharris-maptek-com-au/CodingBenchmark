import matplotlib.pyplot as plt
import numpy as np
import math

from pid_controller import PIDController
from aircraft import Aircraft
from ils import ILS

def run_simulation(aircraft, ils, dt, total_time):
    """
    Runs the ILS autoland simulation.
    """
    print("Entering run_simulation loop...") # Added debug print
    # PID controllers for localizer and glideslope
    # Tuned for a very simplified model; real tuning is complex
    loc_pid = PIDController(kp=0.001, ki=0.00001, kd=0.001, setpoint=0.0, output_limits=(-10.0, 10.0)) # Heading change rate in deg/s
    gs_pid = PIDController(kp=0.1, ki=0.001, kd=0.05, setpoint=0.0, output_limits=(-5.0, 5.0)) # Vertical speed in m/s

    time_points = np.arange(0, total_time, dt)
    history = {
        "x": [], "y": [], "altitude": [], "heading": [],
        "loc_dev": [], "gs_dev": [], "loc_command": [], "gs_command": []
    }

    for t in time_points:
        ac_x, ac_y, ac_alt = aircraft.get_position()

        # Calculate deviations from ILS
        loc_dev = ils.localizer_deviation(ac_x, ac_y)
        gs_dev = ils.glideslope_deviation(ac_x, ac_y, ac_alt)

        # Update PID controllers to get commands
        loc_command = loc_pid.update(loc_dev, dt) # Desired heading change rate
        gs_command = gs_pid.update(gs_dev, dt)   # Desired vertical speed

        # Simple autoland logic: if close to touchdown, initiate flare
        # For simplicity, we'll just level off vertical speed command
        if ac_alt < 50: # Below 50 meters, start to flare
            gs_command = max(0.0, gs_command) # Prevent digging into the ground, target 0 vertical speed
            # Further refinement would involve reducing vertical speed smoothly to 0

        # Update aircraft state
        aircraft.update(dt, heading_command_deg=loc_command, vertical_speed_command_mps=gs_command)

        # Store history
        history["x"].append(ac_x)
        history["y"].append(ac_y)
        history["altitude"].append(ac_alt)
        history["heading"].append(aircraft.get_heading())
        history["loc_dev"].append(loc_dev)
        history["gs_dev"].append(gs_dev)
        history["loc_command"].append(loc_command)
        history["gs_command"].append(gs_command)

        # Add debug prints
        if t % 10 == 0: # Print every 10 seconds
            print(f"Time: {t:.1f}s | Pos: ({ac_x:.1f}, {ac_y:.1f}, {ac_alt:.1f}) | Head: {aircraft.get_heading():.1f} | "
                  f"LocDev: {loc_dev:.1f}m | GSDev: {gs_dev:.1f}m | "
                  f"LocCmd: {loc_command:.1f}deg/s | GSCmd: {gs_command:.1f}m/s")

        # Stop simulation if landed
        if ac_alt <= 0.1 and abs(loc_dev) < 5: # Close to ground and on centerline
            print(f"Aircraft landed at t={t:.2f}s!")
            break
        if ac_y < ils.runway_start_y - 100: # Past the runway start (touchdown point) and gone too far south
            print(f"Aircraft overshot the runway at t={t:.2f}s!")
            break

    print(f"run_simulation finished. History length: {len(history['x'])}") # Debug print
    return history, t # Return last time if broken early

def plot_results(history, ils, final_time):
    """
    Plots the simulation results.
    """
    fig, axs = plt.subplots(3, 1, figsize=(12, 18))
    time_points = np.arange(0, final_time + (final_time/len(history["x"])), final_time/len(history["x"]))

    # Plot 2D Trajectory
    axs[0].plot(history["x"], history["y"], label="Aircraft Path")
    # Draw runway
    axs[0].plot([ils.runway_start_x, ils.runway_end_x],
                [ils.runway_start_y, ils.runway_end_y], 'r-', linewidth=3, label="Runway Centerline")
    axs[0].plot([ils.runway_start_x - ils.runway_width/2, ils.runway_end_x - ils.runway_width/2],
                [ils.runway_start_y, ils.runway_end_y], 'r--', linewidth=1, label="Runway Edge")
    axs[0].plot([ils.runway_start_x + ils.runway_width/2, ils.runway_end_x + ils.runway_width/2],
                [ils.runway_start_y, ils.runway_end_y], 'r--', linewidth=1)
    axs[0].set_xlabel("X (meters)")
    axs[0].set_ylabel("Y (meters)")
    axs[0].set_title("Aircraft 2D Trajectory (Top Down)")
    axs[0].axis('equal')
    axs[0].grid(True)
    axs[0].legend()

    # Plot Altitude Profile
    axs[1].plot(history["y"], history["altitude"], label="Aircraft Altitude")
    # Plot glideslope reference
    # Assuming runway_start_y is the touchdown point for simplicity in this plot
    # The required altitude calculation in ILS is more precise
    # This is an approximation for visual aid
    distance_from_td = np.array(history["y"]) - ils.touchdown_y
    required_alt_plot = np.maximum(0, -distance_from_td * math.tan(ils.glideslope_angle_rad))
    axs[1].plot(history["y"], required_alt_plot, 'g--', label=f"Glideslope ({ils.glideslope_angle_deg} deg)")
    axs[1].set_xlabel("Y (meters)")
    axs[1].set_ylabel("Altitude (meters)")
    axs[1].set_title("Aircraft Altitude Profile")
    axs[1].grid(True)
    axs[1].legend()

    # Plot Deviations
    axs[2].plot(time_points[:len(history["loc_dev"])], history["loc_dev"], label="Localizer Deviation (m)")
    axs[2].plot(time_points[:len(history["gs_dev"])], history["gs_dev"], label="Glideslope Deviation (m)")
    axs[2].set_xlabel("Time (s)")
    axs[2].set_ylabel("Deviation (m)")
    axs[2].set_title("ILS Deviations Over Time")
    axs[2].grid(True)
    axs[2].legend()

    plt.tight_layout()
    plt.savefig("autoland_simulation_results.png")
    plt.show()

if __name__ == "__main__":
    print("Starting Autoland Simulation...")

    # Simulation parameters
    dt = 1.0 # Time step in seconds (increased for faster iteration)
    total_time = 120 # Total simulation time in seconds (2 minutes for debugging)

    # Initialize Aircraft
    # Start ~1000m East, ~10000m North of runway touchdown point, at 500m altitude, flying towards runway
    # Runway assumed to be along Y axis, from (0,0) to (0, 2000)
    # Approach is from positive Y, so initial Y should be large positive, heading towards -Y
    initial_x = 500 # meters, East of centerline
    initial_y = 10000 # meters, North of touchdown point
    initial_altitude = 1000 # meters
    initial_airspeed = 60 # m/s (approx 115 knots)
    initial_heading_deg = 180 # degrees (South)

    aircraft = Aircraft(x=initial_x, y=initial_y, altitude=initial_altitude,
                        airspeed=initial_airspeed, heading_deg=initial_heading_deg)

    # Initialize ILS for a runway from (0,0) to (0, 2000m) along the Y-axis
    # Touchdown point assumed to be at runway_start_y (0,0) for glideslope calculation in ILS class
    ils = ILS(runway_start_x=0, runway_start_y=0, runway_end_x=0, runway_end_y=2000,
              runway_width=45, glideslope_angle_deg=3.0)

    # Run the simulation
    history, final_time = run_simulation(aircraft, ils, dt, total_time)

    # Plot results
    plot_results(history, ils, final_time)

    print("Autoland Simulation Finished. Results saved to autoland_simulation_results.png")
