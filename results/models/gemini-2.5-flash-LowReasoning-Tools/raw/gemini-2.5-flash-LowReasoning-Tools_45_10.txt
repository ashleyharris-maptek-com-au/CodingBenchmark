# Helper to convert float to little-endian hex (IEEE 754 single precision)
def float_to_hex(f):
    return hex(struct.unpack('<I', struct.pack('<f', f))[0])[2:].zfill(8)

# Helper to convert string to SPIR-V string literal (null-terminated, padded to 32-bit words)
def str_to_spirv_str(s):
    bytes_s = s.encode('utf-8') + b'\0'
    padded_len = (len(bytes_s) + 3) // 4 * 4
    padded_bytes = bytes_s.ljust(padded_len, b'\0')
    # Each 32-bit word is little-endian (bytes reversed for hex representation)
    return [padded_bytes[i:i+4][::-1].hex() for i in range(0, padded_len, 4)]

# Helper to make an instruction.
# opcode: decimal opcode
# result_id: integer ID for the result, or None if not an OpType or result-producing instruction
# type_id: integer ID for the result type, or None if not an OpType or result-producing instruction
# operands_hex_list: list of strings, where each string is an 8-char hex representation of a 32-bit word operand.
def make_instruction(opcode, result_id, type_id, operands_hex_list):
    # WordCount = 1 (for instruction word) + (1 if type_id) + (1 if result_id) + len(operands_hex_list)
    word_count = 1
    if type_id is not None:
        word_count += 1
    if result_id is not None:
        word_count += 1
    word_count += len(operands_hex_list)

    instruction_word = (word_count << 16) | opcode
    parts = [f'{instruction_word:08x}']
    if type_id is not None:
        parts.append(f'{type_id:08x}')
    if result_id is not None:
        parts.append(f'{result_id:08x}')
    parts.extend(operands_hex_list)
    return ''.join(parts)
