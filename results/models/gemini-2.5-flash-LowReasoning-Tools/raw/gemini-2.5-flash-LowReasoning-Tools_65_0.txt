// scheduler.ts

import * as fs from 'fs';

interface Task {
    id: string;
    duration: number;
    deps: string[];
    deadline: number;
    weight: number;
}

interface InputData {
    machineCount: number;
    tasks: Task[];
}

// Read input from stdin
const input: string = fs.readFileSync(0, 'utf8');
const data: InputData = JSON.parse(input);

// Build adjacency list and in-degree map for topological sort
const adj: Map<string, string[]> = new Map();
const inDegree: Map<string, number> = new Map();
const taskMap: Map<string, Task> = new Map(); // To quickly access task objects by ID

for (const task of data.tasks) {
    adj.set(task.id, []);
    inDegree.set(task.id, 0);
    taskMap.set(task.id, task);
}

for (const task of data.tasks) {
    for (const depId of task.deps) {
        // depId -> task.id means task.id depends on depId
        const dependents = adj.get(depId);
        if (dependents) {
            dependents.push(task.id);
        } else {
            // This case should ideally not happen if all tasks are pre-initialized in adj and inDegree
            // But good to handle for robustness, though it might indicate an issue with task data.
            adj.set(depId, [task.id]);
        }
        inDegree.set(task.id, (inDegree.get(task.id) || 0) + 1);
    }
}

// Kahn's algorithm for topological sort
const queue: string[] = [];
for (const taskId of inDegree.keys()) {
    if (inDegree.get(taskId) === 0) {
        queue.push(taskId);
    }
}

const topologicalOrder: string[] = [];
let head = 0;
while (head < queue.length) {
    const u = queue[head++];
    topologicalOrder.push(u);

    for (const v of adj.get(u) || []) {
        inDegree.set(v, (inDegree.get(v) || 0) - 1);
        if (inDegree.get(v) === 0) {
            queue.push(v);
        }
    }
}

// Check for cycles
if (topologicalOrder.length !== data.tasks.length) {
    // This indicates a cycle in the dependency graph, which should not happen for a valid schedule.
    // For this benchmark, we might just output the partial order or throw an error.
    // Given the problem description, we'll assume valid input without cycles for now.
    // A robust solution would handle this, e.g., by identifying tasks in the cycle.
    console.error("Error: A cycle was detected in the task dependencies.");
    process.exit(1);
}

// Output the topological order
console.log(JSON.stringify({ order: topologicalOrder }));
