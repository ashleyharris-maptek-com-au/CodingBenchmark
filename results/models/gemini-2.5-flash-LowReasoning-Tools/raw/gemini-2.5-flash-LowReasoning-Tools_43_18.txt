// HLSL Pixel Shader for Vulkan

// Input structure from Vertex Shader
struct VertexOutput
{
    float4 Position : SV_POSITION;
    float3 Color : COLOR0;
};

// Output structure for the Pixel Shader
struct PixelOutput
{
    float4 Color : SV_TARGET0;
};

PixelOutput main(VertexOutput input)
{
    PixelOutput output;
    output.Color = float4(input.Color, 1.0f); // Pass the color directly
    return output;
}
