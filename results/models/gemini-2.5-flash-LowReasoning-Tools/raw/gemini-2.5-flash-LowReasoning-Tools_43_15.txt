// A basic passthrough pixel shader for Vulkan using HLSL

// Define the input structure for the pixel shader
// This should match the output of the vertex shader
struct PsInput
{
    float4 position : SV_Position;
    float4 color : COLOR0; // Assuming the vertex shader passes color in COLOR0 semantic
};

// Main function of the pixel shader
// It takes the interpolated input from the vertex shader and outputs a color
float4 main(PsInput input) : SV_Target
{
    return input.color; // Simply pass through the interpolated color
}
