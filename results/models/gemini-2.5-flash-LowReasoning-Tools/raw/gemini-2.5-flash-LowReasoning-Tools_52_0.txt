import math
from typing import List, Tuple, Dict
from autopilot.utils.data_structures import VehicleState, Obstacle, Lane, Trajectory
from autopilot.utils.geometry import calculate_distance, world_to_vehicle_frame, vehicle_to_world_frame

class Planning:
    """
    The Planning module is responsible for generating a safe, collision-free,
    and efficient trajectory for the ego vehicle.
    It incorporates lane keeping, adaptive cruise control (ACC), and basic
    lane change decision-making.
    """
    def __init__(self, config=None):
        self.config = config if config else self._default_config()

    def _default_config(self):
        """Default configuration parameters for planning."""
        return {
            "target_speed": 10.0,  # m/s
            "min_safe_distance": 10.0, # meters
            "time_gap": 2.0,       # seconds for ACC
            "max_acceleration": 2.0, # m/s^2
            "max_deceleration": -3.0, # m/s^2 (negative value)
            "trajectory_horizon": 3.0, # seconds for planned trajectory
            "trajectory_resolution": 0.1, # seconds per point
            "lane_change_threshold_speed_diff": 2.0, # m/s, to consider changing lane
            "lane_width": 3.7 # meters
        }

    def generate_trajectory(self,
                            ego_state: VehicleState,
                            obstacles: List[Obstacle],
                            predicted_trajectories: Dict[Obstacle, Trajectory],
                            lanes: List[Lane]) -> Trajectory:
        """
        Generates a planned trajectory for the ego vehicle.

        :param ego_state: Current state of the ego vehicle.
        :param obstacles: List of current obstacle objects.
        :param predicted_trajectories: Dictionary of predicted trajectories for obstacles.
        :param lanes: List of available lane objects.
        :return: A Trajectory object.
        """
        target_speed = self.config["target_speed"]
        target_acceleration = 0.0
        
        # 1. Lane Keeping: Determine desired lateral position
        # For simplicity, assume ego is perfectly centered in its lane initially.
        # Find the ego's current lane.
        current_lane = None
        for lane in lanes:
            if lane.id == ego_state.lane_id:
                current_lane = lane
                break
        
        if not current_lane:
            # Fallback if current lane not found (shouldn't happen with good perception/map)
            print("Warning: Ego vehicle's current lane not found in provided lanes.")
            # Assume straight ahead in current orientation for now
            desired_lateral_offset = 0.0
        else:
            # Calculate Frenet coordinates to determine lateral offset
            s, d = current_lane.get_frenet_coordinates(ego_state.x, ego_state.y)
            desired_lateral_offset = -d # Aim for d=0, so if d is positive, move left (negative change)

            # Very simple lateral control target: stay centered
            # For now, we are generating trajectory points in world frame,
            # and the control module will handle lateral error to track it.
            # So, the planning's job is to define the centerline.

        # 2. Adaptive Cruise Control (ACC) & Longitudinal Planning
        leading_obstacle = self._find_leading_obstacle(ego_state, obstacles)

        if leading_obstacle:
            # Calculate desired safe distance
            # Simple time gap model: distance = min_safe_distance + time_gap * leading_obstacle.speed
            safe_distance = self.config["min_safe_distance"] + self.config["time_gap"] * leading_obstacle.speed
            
            distance_to_leading = calculate_distance(
                (ego_state.x, ego_state.y), (leading_obstacle.x, leading_obstacle.y)
            ) - leading_obstacle.length / 2 - ego_state.length / 2 # Approx distance between bumpers

            if distance_to_leading < safe_distance:
                # Too close, decelerate or match speed
                if ego_state.speed > leading_obstacle.speed:
                    target_speed = max(leading_obstacle.speed, 0.0)
                    target_acceleration = self.config["max_deceleration"] # Aggressive deceleration
                else:
                    target_speed = leading_obstacle.speed
                    target_acceleration = 0.0 # Maintain speed
            elif distance_to_leading > safe_distance * 1.2: # A bit further, accelerate back to target
                target_speed = self.config["target_speed"]
                if ego_state.speed < target_speed:
                    target_acceleration = self.config["max_acceleration"]
                else:
                    target_acceleration = 0.0
            else: # Within acceptable range
                target_speed = leading_obstacle.speed
                target_acceleration = 0.0 # Maintain speed
        else:
            # No leading vehicle, accelerate to target speed
            target_acceleration = self.config["max_acceleration"] if ego_state.speed < target_speed else 0.0

        # Clamp target acceleration
        target_acceleration = max(self.config["max_deceleration"], min(self.config["max_acceleration"], target_acceleration))
        
        # 3. Lane Change Decision (Simplified - for future expansion)
        # For this iteration, we'll assume lane keeping. Lane changes are complex
        # and require checking adjacent lanes for safety and speed advantage.
        # This part will mostly ensure we stay in the current lane.
        
        # 4. Trajectory Generation
        trajectory_points = []
        current_x, current_y = ego_state.x, ego_state.y
        current_yaw = ego_state.yaw
        current_speed = ego_state.speed
        
        time_step = self.config["trajectory_resolution"]
        num_trajectory_points = int(self.config["trajectory_horizon"] / time_step)

        for i in range(num_trajectory_points + 1):
            t = i * time_step
            
            # Use simple kinematic model for trajectory generation
            # Assuming constant acceleration for the duration of the trajectory for simplicity
            # And constant yaw (straight line in ego's current direction) for this mock
            
            # Position update
            next_x = current_x + current_speed * t * math.cos(current_yaw) + 0.5 * target_acceleration * t**2 * math.cos(current_yaw)
            next_y = current_y + current_speed * t * math.sin(current_yaw) + 0.5 * target_acceleration * t**2 * math.sin(current_yaw)
            
            # Speed update
            next_speed = current_speed + target_acceleration * t
            next_speed = max(0.0, next_speed) # Speed cannot be negative
            
            # For this simple planning, assume yaw remains constant
            trajectory_points.append((next_x, next_y, current_yaw, next_speed, t))
            
        return Trajectory(trajectory_points)

    def _find_leading_obstacle(self, ego_state: VehicleState, obstacles: List[Obstacle]) -> Obstacle | None:
        """
        Finds the leading obstacle in the ego vehicle's current lane.
        Assumes straight lanes and vehicles roughly aligned with lane direction.
        """
        leading_obstacle = None
        min_distance = float('inf')

        for obs in obstacles:
            # Check if obstacle is in the same lane (approximation for straight road)
            # A more robust check would use Frenet coordinates or detailed lane geometry
            if abs(obs.y - ego_state.y) < self.config["lane_width"] / 2 and obs.x > ego_state.x: # In same y-range and ahead
                distance = calculate_distance((ego_state.x, ego_state.y), (obs.x, obs.y))
                if distance < min_distance:
                    min_distance = distance
                    leading_obstacle = obs
        
        return leading_obstacle
