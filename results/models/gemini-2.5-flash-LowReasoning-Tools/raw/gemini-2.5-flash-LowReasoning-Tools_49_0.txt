#version 450
#extension GL_KHR_shader_subgroup_arithmetic: enable

layout(local_size_x = 128, local_size_y = 1, local_size_z = 1) in;

layout(std430, binding = 0) buffer InputBuffer {
    uint data[];
} inputBuffer;

layout(std430, binding = 1) buffer OutputBuffer {
    uint data[];
} outputBuffer;

shared uint s_data[gl_WorkGroupSize.x];

void main() {
    uint gid = gl_GlobalInvocationID.x;
    uint lid = gl_LocalInvocationID.x;

    // Load data into shared memory
    s_data[lid] = inputBuffer.data[gid];

    // Synchronize all threads in the workgroup
    barrier();

    // Perform inclusive prefix sum within the workgroup
    // This is a simple workgroup scan, not a full parallel prefix sum across the entire buffer
    // For a full parallel prefix sum, multiple passes or a more complex algorithm would be needed.
    for (uint i = 1; i < gl_WorkGroupSize.x; i *= 2) {
        if (lid >= i) {
            s_data[lid] += s_data[lid - i];
        }
        barrier();
    }

    // Write the result to the output buffer
    outputBuffer.data[gid] = s_data[lid];
}
