# autopilot.py

import math
from enum import Enum
from . import vector_math
from .constants import (
    DT,
    DOCKING_POSITION_TOLERANCE, DOCKING_VELOCITY_TOLERANCE, DOCKING_ATTITUDE_TOLERANCE,
    RENDEZVOUS_HOLD_POINT_RELATIVE_POS,
    POS_KP, POS_KI, POS_KD,
    VEL_KP, VEL_KI, VEL_KD,
    ATT_KP, ATT_KI, ATT_KD
)

class PIDController:
    def __init__(self, Kp, Ki, Kd, output_limit=None):
        self.Kp = Kp
        self.Ki = Ki
        self.Kd = Kd
        self.output_limit = output_limit

        self._prev_error = 0.0
        self._integral_error = 0.0

    def calculate(self, current_error, dt):
        # Proportional term
        P_term = self.Kp * current_error

        # Integral term
        self._integral_error += current_error * dt
        I_term = self.Ki * self._integral_error

        # Derivative term
        D_term = self.Kd * (current_error - self._prev_error) / dt
        self._prev_error = current_error

        output = P_term + I_term + D_term

        if self.output_limit is not None:
            output = max(-self.output_limit, min(self.output_limit, output))

        return output

class AutopilotPhase(Enum):
    RENDEZVOUS = 1
    TERMINAL_DOCKING = 2
    DOCKED = 3

class Autopilot:
    def __init__(self, spacecraft, space_station):
        self.spacecraft = spacecraft
        self.space_station = space_station

        self.pid_pos_x = PIDController(POS_KP, POS_KI, POS_KD, output_limit=spacecraft.max_thrust)
        self.pid_pos_y = PIDController(POS_KP, POS_KI, POS_KD, output_limit=spacecraft.max_thrust)
        # For velocity, we'll often use it as part of a cascaded control or directly for braking
        # For simplicity, initially, the position PID will handle velocity implicitly.
        # If explicit velocity PID is needed, it would be another layer.
        self.pid_att = PIDController(ATT_KP, ATT_KI, ATT_KD, output_limit=spacecraft.max_torque)

        self.current_phase = AutopilotPhase.RENDEZVOUS
        self.rendezvous_hold_point_relative_pos = RENDEZVOUS_HOLD_POINT_RELATIVE_POS

    def calculate_thrust_and_torque(self, dt):
        docking_port_pos, docking_port_att = self.space_station.get_docking_port_state()
        rel_pos, rel_vel, att_error = self.spacecraft.get_relative_state(docking_port_pos, docking_port_att)

        thrust_vector = (0.0, 0.0)
        torque_scalar = 0.0

        if self.current_phase == AutopilotPhase.RENDEZVOUS:
            # Target is the rendezvous hold point relative to the docking port
            target_pos = vector_math.add(docking_port_pos, self.rendezvous_hold_point_relative_pos)
            
            # Calculate error to hold point
            error_to_hold_point = vector_math.subtract(target_pos, self.spacecraft.position)
            
            # Simple proportional control to reach the hold point
            # Could use PID for more sophisticated rendezvous
            target_vel_x = self.pid_pos_x.calculate(error_to_hold_point[0], dt)
            target_vel_y = self.pid_pos_y.calculate(error_to_hold_point[1], dt)
            
            # Use velocity difference to generate thrust
            thrust_x = self.spacecraft.mass * (target_vel_x - self.spacecraft.velocity[0])
            thrust_y = self.spacecraft.mass * (target_vel_y - self.spacecraft.velocity[1])
            thrust_vector = (thrust_x, thrust_y)

            # Also try to align attitude during rendezvous
            torque_scalar = self.pid_att.calculate(att_error, dt)

            # Check for transition to TERMINAL_DOCKING
            # If close enough to hold point and relative velocity is low
            if (vector_math.magnitude(error_to_hold_point) < (DOCKING_POSITION_TOLERANCE * 10) and
                vector_math.magnitude(self.spacecraft.velocity) < (DOCKING_VELOCITY_TOLERANCE * 10)):
                print("Transitioning to TERMINAL_DOCKING phase.")
                self.current_phase = AutopilotPhase.TERMINAL_DOCKING
                # Reset PID controllers for the new phase
                self.pid_pos_x = PIDController(POS_KP, POS_KI, POS_KD, output_limit=self.spacecraft.max_thrust)
                self.pid_pos_y = PIDController(POS_KP, POS_KI, POS_KD, output_limit=self.spacecraft.max_thrust)
                self.pid_att = PIDController(ATT_KP, ATT_KI, ATT_KD, output_limit=self.spacecraft.max_torque)


        elif self.current_phase == AutopilotPhase.TERMINAL_DOCKING:
            # Target is the docking port itself
            
            # Position control (P-controller for target velocity, then another P for thrust)
            # Desired relative velocity proportional to relative position error (PD control effectively)
            desired_rel_vel_x = POS_KP * rel_pos[0]
            desired_rel_vel_y = POS_KP * rel_pos[1]
            
            # Thrust to achieve desired relative velocity
            thrust_x = VEL_KP * (desired_rel_vel_x - rel_vel[0]) * self.spacecraft.mass
            thrust_y = VEL_KP * (desired_rel_vel_y - rel_vel[1]) * self.spacecraft.mass
            
            thrust_vector = (thrust_x, thrust_y)
            
            # Attitude control
            torque_scalar = self.pid_att.calculate(att_error, dt)

            if self.is_docked(rel_pos, rel_vel, att_error):
                print("DOCKING SUCCESSFUL!")
                self.current_phase = AutopilotPhase.DOCKED
                thrust_vector = (0.0, 0.0)
                torque_scalar = 0.0

        elif self.current_phase == AutopilotPhase.DOCKED:
            thrust_vector = (0.0, 0.0)
            torque_scalar = 0.0

        # Ensure thrust_vector magnitude does not exceed max_thrust
        current_thrust_magnitude = vector_math.magnitude(thrust_vector)
        if current_thrust_magnitude > self.spacecraft.max_thrust:
            thrust_vector = vector_math.scale(vector_math.normalize(thrust_vector), self.spacecraft.max_thrust)
            
        return thrust_vector, torque_scalar

    def is_docked(self, rel_pos, rel_vel, att_error):
        """
        Checks if the spacecraft is docked based on tolerances.
        :param rel_pos: Relative position vector.
        :param rel_vel: Relative velocity vector.
        :param att_error: Attitude error in radians.
        :return: True if docked, False otherwise.
        """
        pos_mag = vector_math.magnitude(rel_pos)
        vel_mag = vector_math.magnitude(rel_vel)
        
        position_ok = pos_mag < DOCKING_POSITION_TOLERANCE
        velocity_ok = vel_mag < DOCKING_VELOCITY_TOLERANCE
        attitude_ok = abs(att_error) < DOCKING_ATTITUDE_TOLERANCE
        
        return position_ok and velocity_ok and attitude_ok
