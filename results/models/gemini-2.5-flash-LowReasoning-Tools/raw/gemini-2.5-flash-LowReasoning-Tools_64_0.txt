import * as fs from 'fs';

interface Drone {
    id: number;
    x: number;
    y: number;
}

interface InputData {
    probability: number[][];
    drones: Drone[];
    maxStepsPerDrone: number;
}

interface OutputPath {
    paths: [number, number][][];
}

// Helper function to check if a cell is valid
function isValidMove(
    r: number,
    c: number,
    numRows: number,
    numCols: number,
    probabilityGrid: number[][]
): boolean {
    return (
        r >= 0 &&
        r < numRows &&
        c >= 0 &&
        c < numCols &&
        probabilityGrid[r][c] !== -1
    );
}

function solve() {
    const input: string = fs.readFileSync(0, 'utf8');
    const data: InputData = JSON.parse(input);

    const { probability, drones, maxStepsPerDrone } = data;
    const numRows = probability.length;
    const numCols = probability[0].length;

    // To keep track of visited cells globally across all drones
    const visitedCells = new Set<string>(); // Stores "row,col"

    // Initialize paths for each drone
    const dronePaths: [number, number][][] = drones.map(drone => {
        const startCoord: [number, number] = [drone.y, drone.x];
        visitedCells.add(`${drone.y},${drone.x}`);
        return [startCoord];
    });

    // Define possible movements (dr, dc) for: Stay, Up, Down, Left, Right
    const moves = [
        [0, 0], // Stay
        [-1, 0], // Up
        [1, 0], // Down
        [0, -1], // Left
        [0, 1] // Right
    ];

    for (let step = 0; step < maxStepsPerDrone; step++) {
        // Collect all potential moves and their values for each drone
        // Each entry: { droneIndex, newRow, newCol, probabilityValue }
        const allPotentialMoves: { droneIndex: number; newRow: number; newCol: number; probabilityValue: number; }[] = [];

        drones.forEach((drone, droneIndex) => {
            const [currentRow, currentCol] = dronePaths[droneIndex][dronePaths[droneIndex].length - 1];

            for (const [dr, dc] of moves) {
                const newRow = currentRow + dr;
                const newCol = currentCol + dc;
                const moveKey = `${newRow},${newCol}`;

                if (isValidMove(newRow, newCol, numRows, numCols, probability)) {
                    let cellProbability = probability[newRow][newCol];

                    // If the cell is already visited and it's not the current position (meaning a new move to an already covered cell)
                    if (visitedCells.has(moveKey) && !(newRow === currentRow && newCol === currentCol)) {
                        cellProbability = 0; // No unique probability gain
                    }
                    
                    allPotentialMoves.push({ droneIndex, newRow, newCol, probabilityValue: cellProbability });
                }
            }
        });

        // Sort potential moves to prioritize higher probability values.
        // If probabilities are equal, prioritize staying in place, then drone ID (to ensure deterministic behavior if not specified)
        allPotentialMoves.sort((a, b) => {
            if (b.probabilityValue !== a.probabilityValue) {
                return b.probabilityValue - a.probabilityValue; // Higher probability first
            }
            // If probabilities are equal, prefer staying over moving
            const isAStay = (dronePaths[a.droneIndex][dronePaths[a.droneIndex].length - 1][0] === a.newRow && dronePaths[a.droneIndex][dronePaths[a.droneIndex].length - 1][1] === a.newCol);
            const isBStay = (dronePaths[b.droneIndex][dronePaths[b.droneIndex].length - 1][0] === b.newRow && dronePaths[b.droneIndex][dronePaths[b.droneIndex].length - 1][1] === b.newCol);
            if (isAStay && !isBStay) return -1;
            if (!isAStay && isBStay) return 1;
            
            return a.droneIndex - b.droneIndex; // Tie-break by drone ID for deterministic results
        });

        const assignedCellsInThisStep = new Set<string>(); // To track cells chosen in this step
        const nextStepCoordinates: ([number, number] | null)[] = Array(drones.length).fill(null);
        let dronesAssignedCount = 0;

        // Assign moves to drones
        for (const move of allPotentialMoves) {
            if (nextStepCoordinates[move.droneIndex] === null) { // If drone hasn't been assigned a move yet
                const moveKey = `${move.newRow},${move.newCol}`;
                
                // A drone can move to an already visited cell (if it's the only option or the best non-unique option),
                // but its contribution to 'unique probability' was zeroed out.
                // It cannot move to a cell *another drone is claiming in this same step* if that move provides unique value.
                // However, if the current move's probabilityValue is 0 (meaning already visited globally),
                // another drone might still select it if its a better choice than other 0-value moves.
                // To prevent multiple drones from occupying the same *new* cell in the *same step*:
                if (!assignedCellsInThisStep.has(moveKey) || move.probabilityValue === 0) { // Allow multiple drones to "stay" on same cell, or move to an already visited cell if its their best option
                    nextStepCoordinates[move.droneIndex] = [move.newRow, move.newCol];
                    assignedCellsInThisStep.add(moveKey); // Mark the cell as chosen for this step
                    dronesAssignedCount++;

                    if (dronesAssignedCount === drones.length) break; // All drones have been assigned a move
                }
            }
        }

        // Add assigned moves to paths and update global visitedCells
        drones.forEach((drone, droneIndex) => {
            const nextCoord = nextStepCoordinates[droneIndex];
            if (nextCoord) { // If a move was assigned
                 dronePaths[droneIndex].push(nextCoord);
                 // Only mark as visited if it actually contributed unique probability (i.e., its probabilityValue was > 0)
                 // Or if it's the drone's current position and it's the best choice to stay there
                 const [currentRow, currentCol] = dronePaths[droneIndex][dronePaths[droneIndex].length - 2]; // Previous position
                 const currentMoveKey = `${nextCoord[0]},${nextCoord[1]}`;
                 const originalProb = probability[nextCoord[0]][nextCoord[1]];
                 if (!visitedCells.has(currentMoveKey) && originalProb > 0) {
                     visitedCells.add(currentMoveKey);
                 }
            } else { // This case should ideally not happen if all drones are assigned, but as a fallback
                 const [currentRow, currentCol] = dronePaths[droneIndex][dronePaths[droneIndex].length - 1];
                 dronePaths[droneIndex].push([currentRow, currentCol]);
            }
        });
    }

    const output: OutputPath = { paths: dronePaths };
    console.log(JSON.stringify(output));
}

solve();
