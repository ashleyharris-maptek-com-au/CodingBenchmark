#version 450

layout(local_size_x = 256, local_size_y = 1, local_size_z = 1) in;

struct Particle {
    vec4 position; // xyz = position, w = mass
    vec4 velocity; // xyz = velocity, w = padding
};

layout(std430, binding = 0) buffer ParticlesA {
    Particle particlesA[];
};

layout(std430, binding = 1) buffer ParticlesB {
    Particle particlesB[];
};

uniform float deltaTime;
uniform int numParticles;
uniform float gravitationalConstant;
uniform float softeningFactor; // To prevent division by zero and large forces at close distances

void main() {
    uint index = gl_GlobalInvocationID.x;

    if (index >= numParticles) {
        return;
    }

    Particle currentParticle = particlesA[index];
    vec3 force = vec3(0.0);

    for (int i = 0; i < numParticles; ++i) {
        if (i == index) {
            continue;
        }

        Particle otherParticle = particlesA[i];

        vec3 direction = otherParticle.position.xyz - currentParticle.position.xyz;
        float distanceSq = dot(direction, direction) + softeningFactor * softeningFactor;
        float distance = sqrt(distanceSq);

        // Calculate gravitational force
        float magnitude = (gravitationalConstant * currentParticle.position.w * otherParticle.position.w) / (distanceSq * distance);
        force += magnitude * normalize(direction);
    }

    // Update velocity
    vec3 acceleration = force / currentParticle.position.w;
    currentParticle.velocity.xyz += acceleration * deltaTime;

    // Update position
    currentParticle.position.xyz += currentParticle.velocity.xyz * deltaTime;

    particlesB[index] = currentParticle;
}
