// Basic Min-Heap implementation for Dijkstra's
class MinPriorityQueue {
    constructor() {
        this.heap = [];
    }

    push(item) {
        this.heap.push(item);
        this._bubbleUp();
    }

    pop() {
        if (this.isEmpty()) return null;
        const min = this.heap[0];
        const last = this.heap.pop();
        if (!this.isEmpty()) {
            this.heap[0] = last;
            this._sinkDown();
        }
        return min;
    }

    isEmpty() {
        return this.heap.length === 0;
    }

    _bubbleUp() {
        let idx = this.heap.length - 1;
        const element = this.heap[idx];
        while (idx > 0) {
            let parentIdx = Math.floor((idx - 1) / 2);
            let parent = this.heap[parentIdx];
            if (element[0] >= parent[0]) break; // Compare by cost
            this.heap[parentIdx] = element;
            this.heap[idx] = parent;
            idx = parentIdx;
        }
    }

    _sinkDown() {
        let idx = 0;
        const length = this.heap.length;
        const element = this.heap[0];
        while (true) {
            let leftChildIdx = 2 * idx + 1;
            let rightChildIdx = 2 * idx + 2;
            let leftChild, rightChild;
            let swap = null;

            if (leftChildIdx < length) {
                leftChild = this.heap[leftChildIdx];
                if (leftChild[0] < element[0]) { // Compare by cost
                    swap = leftChildIdx;
                }
            }
            if (rightChildIdx < length) {
                rightChild = this.heap[rightChildIdx];
                if (
                    (swap === null && rightChild[0] < element[0]) ||
                    (swap !== null && rightChild[0] < leftChild[0]) // Compare by cost
                ) {
                    swap = rightChildIdx;
                }
            }
            if (swap === null) break;
            this.heap[idx] = this.heap[swap];
            this.heap[swap] = element;
            idx = swap;
        }
    }
}

        pq.push([0, source]); // [cost, node]
        dist[source] = 0;

            const [currentReducedCost, u] = pq.pop();

            if (currentReducedCost > dist[u]) continue;

            for (let i = 0; i < adj[u].length; i++) {
                const edge = adj[u][i];

                    // Reduced cost: c_p(u,v) = c(u,v) + potentials[u] - potentials[v]
                    const reducedCost = edge.cost + potentials[u] - potentials[edge.to];
                    if (dist[u] + reducedCost < dist[edge.to]) {
                        dist[edge.to] = dist[u] + reducedCost;
                        parentNode[edge.to] = u;
                        parentEdgeIdx[edge.to] = i;
                        pq.push([dist[edge.to], edge.to]);
                    }

            break;
        }

        // Update potentials
        for (let i = 0; i < NODE_COUNT; i++) {
            if (dist[i] !== Infinity) {
                potentials[i] += dist[i];
            }
        }

        // Calculate actual path cost
        // The total cost of the path is (potentials[sink] - potentials[source]) because potentials[source] is 0
        const pathCost = potentials[sink]; // Because potentials[source] is always 0 in our setup (S is fixed source)

        // Determine how much flow to send without exceeding budget
        let maxFlowAmount = Infinity;
        let current = sink;
        while (current !== source) {
            const prev = parentNode[current];
            const edgeIdx = parentEdgeIdx[current];
            maxFlowAmount = Math.min(maxFlowAmount, adj[prev][edgeIdx].capacity);
            current = prev;
        }
        
        let flowToSend = maxFlowAmount;
        if (totalCost + pathCost * maxFlowAmount > maxCostBudget) {
            if (pathCost === 0) {
                // If pathCost is 0, we can send as much flow as possible without increasing cost.
                // This means there's no new "cut" being made on this path.
                // We should be able to send all maxFlowAmount
                flowToSend = maxFlowAmount;
            } else {
                flowToSend = Math.floor((maxCostBudget - totalCost) / pathCost);
            }
        }

        if (flowToSend === 0) { // Can't send any more flow within budget
            break;
        }

        // Augment flow
        totalFlow += flowToSend;
        totalCost += pathCost * flowToSend;

        current = sink;

            const edgeIdx = parentEdgeIdx[current];
            const forwardEdge = adj[prev][edgeIdx];
            const reverseEdge = adj[forwardEdge.to][forwardEdge.rev];

            forwardEdge.capacity -= flowToSend;
            reverseEdge.capacity += flowToSend;

            // Identify firebreaks: an edge is a firebreak if it's a 'cut_edge' and its capacity has been reduced to 0
            if (forwardEdge.is_cut_edge && forwardEdge.capacity === 0 && originalCutEdgeCapacities.get(`${prev}-${forwardEdge.to}`) === 1) {
                firebreaks.push([forwardEdge.c_coord, forwardEdge.r_coord]);
            }
            current = prev;
        }
    }
