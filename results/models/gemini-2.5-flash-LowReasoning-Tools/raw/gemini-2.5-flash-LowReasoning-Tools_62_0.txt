import * as fs from 'fs';

interface Town {
    id: number;
    x: number;
    y: number;
    value: number;
}

interface InputData {
    width: number;
    height: number;
    grid: string[]; // '.' for forest, '~' for water/rock
    ignitions: [number, number][]; // [x, y]
    towns: Town[];
    cutBudget: number;
}

interface Point {
    x: number;
    y: number;
}

interface SimulationResult {
    burnedCells: Set<string>; // "x,y" strings of burned cells
    burnedTownValue: number;
    frontierCells: Set<string>; // "x,y" strings of burnable cells adjacent to burned cells, but not burned themselves
}

const dx = [0, 0, 1, -1];
const dy = [1, -1, 0, 0];

/**
 * Simulates fire spread on the grid and identifies burned cells, burned town value, and frontier cells.
 * @param initialIsBurnableGrid A 2D boolean array representing the initial burnable state of the grid.
 * @param ignitions Starting points of the fire.
 * @param proposedFirebreaks Firebreaks already placed.
 * @param width Grid width.
 * @param height Grid height.
 * @param townMap Map for quick town lookup by "x,y" string.
 * @returns SimulationResult containing burned cells, burned town value, and frontier cells.
 */
function simulateFire(
    initialIsBurnableGrid: boolean[][],
    ignitions: Point[],
    proposedFirebreaks: Point[],
    width: number,
    height: number,
    townMap: Map<string, Town>
): SimulationResult {
    // Create a temporary burnable grid applying proposed firebreaks
    const currentIsBurnableGrid: boolean[][] = initialIsBurnableGrid.map(row => [...row]);
    for (const fb of proposedFirebreaks) {
        if (fb.y >= 0 && fb.y < height && fb.x >= 0 && fb.x < width) {
            currentIsBurnableGrid[fb.y][fb.x] = false; // Mark as non-burnable
        }
    }

    const burnedCells = new Set<string>();
    const frontierCells = new Set<string>();
    const queue: Point[] = [];

    // Initialize queue with ignitions
    for (const ignition of ignitions) {
        const key = `${ignition.x},${ignition.y}`;
        if (ignition.y >= 0 && ignition.y < height && ignition.x >= 0 && ignition.x < width && currentIsBurnableGrid[ignition.y][ignition.x] && !burnedCells.has(key)) {
            queue.push(ignition);
            burnedCells.add(key);
        }
    }

    let head = 0;
    while (head < queue.length) {
        const { x, y } = queue[head++];

        for (let i = 0; i < 4; i++) {
            const nx = x + dx[i];
            const ny = y + dy[i];
            const nKey = `${nx},${ny}`;

            if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                if (currentIsBurnableGrid[ny][nx]) { // Is burnable forest
                    if (!burnedCells.has(nKey)) {
                        burnedCells.add(nKey);
                        queue.push({ x: nx, y: ny });
                    }
                } else { // Not burnable, it's a boundary for frontier cells
                    // If the neighbor is not burnable but current cell is burned,
                    // and the neighbor is originally burnable (e.g. not water/rock)
                    // and not already a firebreak, then it's a potential frontier cell if we were looking for those.
                    // For the purpose of finding candidates to cut, we look at cells adjacent to current fire.
                    // The frontierCells should be burnable, not burned, and adjacent to burned.
                    // This logic is handled after initial BFS, by iterating burnedCells' neighbors.
                }
            }
        }
    }

    // After BFS, calculate burned town value and identify true frontier cells
    let burnedTownValue = 0;
    const neighborOffsets = [
        { dx: 0, dy: 1 }, { dx: 0, dy: -1 }, { dx: 1, dy: 0 }, { dx: -1, dy: 0 }
    ];

    for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
            const key = `${x},${y}`;
            if (burnedCells.has(key)) {
                const town = townMap.get(key);
                if (town) {
                    burnedTownValue += town.value;
                }
                // Check neighbors of burned cells to find frontier cells
                for (const offset of neighborOffsets) {
                    const nx = x + offset.dx;
                    const ny = y + offset.dy;
                    const nKey = `${nx},${ny}`;
                    if (nx >= 0 && nx < width && ny >= 0 && ny < height && currentIsBurnableGrid[ny][nx] && !burnedCells.has(nKey)) {
                        frontierCells.add(nKey);
                    }
                }
            }
        }
    }

    return { burnedCells, burnedTownValue, frontierCells };
}

function solve() {
    const input: string = fs.readFileSync(0, 'utf8');
    const data: InputData = JSON.parse(input);

    const { width, height, grid, ignitions: initialIgnitions, towns, cutBudget } = data;

    // Convert ignitions to Point[]
    const ignitions: Point[] = initialIgnitions.map(([x, y]) => ({ x, y }));

    // Pre-process grid to a 2D boolean array for burnable status
    const initialIsBurnableGrid: boolean[][] = Array(height).fill(null).map(() => Array(width).fill(false));
    for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
            initialIsBurnableGrid[y][x] = grid[y][x] === '.';
        }
    }

    // Map towns for quick lookup
    const townMap: Map<string, Town> = new Map();
    for (const town of towns) {
        townMap.set(`${town.x},${town.y}`, town);
    }

    // Step 1: Initial fire simulation without any firebreaks to determine max possible saved value
    const initialSimResult = simulateFire(initialIsBurnableGrid, ignitions, [], width, height, townMap);
    let maxPossibleSavedValue = initialSimResult.burnedTownValue;

    if (maxPossibleSavedValue === 0) {
        // No towns burn initially, so no need for firebreaks
        console.log(JSON.stringify({ firebreaks: [] }));
        return;
    }

    const finalFirebreaks: Point[] = [];
    let currentSavedValue = 0; // Value saved so far
    let currentFirebreaksSet = new Set<string>(); // Keep track of firebreaks by "x,y" strings

    // Step 2: Iteratively place firebreaks
    for (let i = 0; i < cutBudget; i++) {
        // Simulate with current firebreaks to find current fire situation and frontier
        const simResultWithCurrentFBs = simulateFire(initialIsBurnableGrid, ignitions, finalFirebreaks, width, height, townMap);
        const currentlyBurnedTownValue = simResultWithCurrentFBs.burnedTownValue;
        const currentFrontierCells = simResultWithCurrentFBs.frontierCells;

        // If no more towns are burning, we are done
        if (currentlyBurnedTownValue === 0) {
            break;
        }

        let bestCandidate: Point | null = null;
        let maxSavedValueDelta = -1; // Max additional value saved by this potential cut

        // Generate candidate cells for this step
        const candidateCellsForThisStep = new Set<string>();

        // Add cells from the current fire frontier
        for (const cellKey of currentFrontierCells) {
            if (!currentFirebreaksSet.has(cellKey)) { // Only consider cells not already cut
                candidateCellsForThisStep.add(cellKey);
            }
        }
        
        // Add cells adjacent to any town that is currently burning
        for (const town of towns) {
            const townKey = `${town.x},${town.y}`;
            if (simResultWithCurrentFBs.burnedCells.has(townKey)) { // If this town is burning
                const { x, y } = town;
                for (let k = 0; k < 4; k++) {
                    const nx = x + dx[k];
                    const ny = y + dy[k];
                    const nKey = `${nx},${ny}`;

                    if (nx >= 0 && nx < width && ny >= 0 && ny < height &&
                        initialIsBurnableGrid[ny][nx] && !currentFirebreaksSet.has(nKey)) {
                        candidateCellsForThisStep.add(nKey);
                    }
                }
            }
        }

        if (candidateCellsForThisStep.size === 0) {
            break; // No more candidates to evaluate
        }

        // Evaluate each candidate
        for (const candidateKey of candidateCellsForThisStep) {
            const [cx, cy] = candidateKey.split(',').map(Number);
            const candidatePoint: Point = { x: cx, y: cy };

            const testFirebreaks = [...finalFirebreaks, candidatePoint];
            const testSimResult = simulateFire(initialIsBurnableGrid, ignitions, testFirebreaks, width, height, townMap);
            
            // Calculate how much *additional* value this cut saves compared to the current state
            const savedValueDelta = currentlyBurnedTownValue - testSimResult.burnedTownValue;

            if (savedValueDelta > maxSavedValueDelta) {
                maxSavedValueDelta = savedValueDelta;
                bestCandidate = candidatePoint;
            }
        }

        if (bestCandidate && maxSavedValueDelta > 0) {
            finalFirebreaks.push(bestCandidate);
            currentFirebreaksSet.add(`${bestCandidate.x},${bestCandidate.y}`);
        } else {
            // No candidate could save additional value, or budget exhausted
            break;
        }
    }

    // Convert Point[] to [x, y][] for output
    const outputFirebreaks = finalFirebreaks.map(p => [p.x, p.y]);
    console.log(JSON.stringify({ firebreaks: outputFirebreaks }));
}

solve();
