const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);

    const machineCount = data.machineCount;
    const tasks = data.tasks;

    // Create a map for easy access to tasks by ID
    const taskMap = new Map();
    tasks.forEach(task => {
        taskMap.set(task.id, task);
    });

    // Build adjacency list and in-degrees for topological sort
    const adjList = new Map();
    const inDegree = new Map();
    tasks.forEach(task => {
        adjList.set(task.id, []);
        inDegree.set(task.id, 0);
    });

    tasks.forEach(task => {
        task.deps.forEach(depId => {
            if (!adjList.has(depId)) {
                adjList.set(depId, []);
            }
            adjList.get(depId).push(task.id);
            inDegree.set(task.id, inDegree.get(task.id) + 1);
        });
    });

    // Initialize queue for tasks with no dependencies
    const queue = [];
    tasks.forEach(task => {
        if (inDegree.get(task.id) === 0) {
            queue.push(task.id);
        }
    });

    const topologicalOrder = [];
    let head = 0; // Use a head pointer for the queue to avoid performance issues with shift()

    while (head < queue.length) {
        const currentTaskId = queue[head++];
        topologicalOrder.push(currentTaskId);

        adjList.get(currentTaskId).forEach(neighborId => {
            inDegree.set(neighborId, inDegree.get(neighborId) - 1);
            if (inDegree.get(neighborId) === 0) {
                queue.push(neighborId);
            }
        });
    }

    // Check for cycles
    if (topologicalOrder.length !== tasks.length) {
        // This indicates a cycle in the dependencies, or some tasks were unreachable.
        // For this benchmark, we assume valid acyclic graphs.
        // In a real application, this would require error handling.
        console.error("Error: A cycle was detected in the task dependencies or tasks are unreachable.");
        process.exit(1);
    }

    const output = {
        order: topologicalOrder
    };

    console.log(JSON.stringify(output));
}

solve();
