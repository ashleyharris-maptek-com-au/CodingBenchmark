const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);

    const probability = data.probability;
    const drones = data.drones;
    const maxStepsPerDrone = data.maxStepsPerDrone;

    const rows = probability.length;
    const cols = probability[0].length;

    // Initialize paths for each drone
    const paths = drones.map(drone => [[drone.x, drone.y]]);

    // Keep track of visited cells and their step of visit to ensure uniqueness per step
    // visitedCells[step][x][y] = true if cell (x,y) is visited at 'step'
    const visitedCells = Array(maxStepsPerDrone + 1).fill(null).map(() =>
        Array(rows).fill(null).map(() => Array(cols).fill(false))
    );

    // Mark initial drone positions as visited at step 0
    drones.forEach((drone, droneIndex) => {
        const [x, y] = paths[droneIndex][0];
        if (x >= 0 && x < rows && y >= 0 && y < cols && probability[x][y] !== -1) {
            visitedCells[0][x][y] = true;
        }
    });

    // Possible moves: stay, up, down, left, right
    const moves = [[0, 0], [-1, 0], [1, 0], [0, -1], [0, 1]];

    for (let step = 0; step < maxStepsPerDrone; step++) {
        const nextPositions = new Array(drones.length).fill(null);
        const currentDronePositions = paths.map(path => path[path.length - 1]);

        // Determine next best position for each drone
        for (let i = 0; i < drones.length; i++) {
            const [currentX, currentY] = currentDronePositions[i];
            let bestNextMove = [currentX, currentY]; // Default to staying in place
            let maxProb = -Infinity;

            for (const [dx, dy] of moves) {
                const nextX = currentX + dx;
                const nextY = currentY + dy;

                // Check boundaries
                if (nextX < 0 || nextX >= rows || nextY < 0 || nextY >= cols) {
                    continue;
                }

                // Check for obstacle
                if (probability[nextX][nextY] === -1) {
                    continue;
                }

                // Check if already visited by another drone at this *next* step
                // This ensures unique coverage for the current step
                if (visitedCells[step + 1][nextX][nextY]) {
                    continue;
                }

                const currentProb = probability[nextX][nextY];

                // Greedy choice: pick the cell with highest probability
                // If probabilities are equal, prefer staying in place (dx=0, dy=0 is first in moves array)
                // or the first valid move encountered.
                if (currentProb > maxProb) {
                    maxProb = currentProb;
                    bestNextMove = [nextX, nextY];
                }
            }
            nextPositions[i] = bestNextMove;
        }

        // Apply the chosen next positions and mark them as visited for the next step
        for (let i = 0; i < drones.length; i++) {
            const [nextX, nextY] = nextPositions[i];
            paths[i].push([nextX, nextY]);
            visitedCells[step + 1][nextX][nextY] = true;
        }
    }

    const output = { paths: paths };
    console.log(JSON.stringify(output));
}

solve();
