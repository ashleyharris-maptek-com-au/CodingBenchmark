#version 450
      
      // Define work group size
      layout (local_size_x = 256, local_size_y = 1, local_size_z = 1) in;
      
      // Structure for a single boid
      struct Boid {
          vec4 position; // xyz for position, w for padding/future use
          vec4 velocity; // xyz for velocity, w for padding/future use
          // We might add more attributes later, like color or acceleration
      };
      
      // SSBOs for boid data
      layout(std430, binding = 0) buffer BoidData {
          Boid boids[];
      };
      
      // Uniforms for simulation parameters
      layout(location = 0) uniform int numBoids;
      layout(location = 1) uniform float deltaTime;
      layout(location = 2) uniform float separationRadius;
      layout(location = 3) uniform float alignmentRadius;
      layout(location = 4) uniform float cohesionRadius;
      layout(location = 5) uniform float separationWeight;
      layout(location = 6) uniform float alignmentWeight;
      layout(location = 7) uniform float cohesionWeight;
      layout(location = 8) uniform float maxSpeed;
      layout(location = 9) uniform float maxForce;
      layout(location = 10) uniform vec3 worldBoundsMin;
      layout(location = 11) uniform vec3 worldBoundsMax;
      
      void main() {
          uint index = gl_GlobalInvocationID.x;
          if (index >= numBoids) {
              return;
          }
      
          // Get current boid
          Boid currentBoid = boids[index];
      
          vec3 separationForce = vec3(0.0);
          vec3 alignmentForce = vec3(0.0);
          vec3 cohesionForce = vec3(0.0);
      
          int separationCount = 0;
          int alignmentCount = 0;
          int cohesionCount = 0;
      
          vec3 perceivedCenter = vec3(0.0);
          vec3 perceivedVelocity = vec3(0.0);
      
          // Iterate over all other boids to apply flocking rules
          for (int i = 0; i < numBoids; ++i) {
              if (i == index) continue;
      
              Boid otherBoid = boids[i];
              vec3 toOther = otherBoid.position.xyz - currentBoid.position.xyz;
              float distance = length(toOther);
      
              // Separation
              if (distance < separationRadius) {
                  separationForce += normalize(currentBoid.position.xyz - otherBoid.position.xyz) / distance;
                  separationCount++;
              }
      
              // Alignment and Cohesion
              if (distance < alignmentRadius) { // Using alignmentRadius for both for simplicity
                  perceivedVelocity += otherBoid.velocity.xyz;
                  alignmentCount++;
              }
              if (distance < cohesionRadius) {
                  perceivedCenter += otherBoid.position.xyz;
                  cohesionCount++;
              }
          }
      
          // Calculate average forces
          if (separationCount > 0) {
              separationForce /= separationCount;
          }
          if (alignmentCount > 0) {
              perceivedVelocity /= alignmentCount;
              alignmentForce = normalize(perceivedVelocity - currentBoid.velocity.xyz);
          }
          if (cohesionCount > 0) {
              perceivedCenter /= cohesionCount;
              cohesionForce = normalize(perceivedCenter - currentBoid.position.xyz);
          }
      
          // Apply weights and sum forces
          vec3 totalForce = (separationForce * separationWeight) +
                            (alignmentForce * alignmentWeight) +
                            (cohesionForce * cohesionWeight);
      
          // Limit total force
          if (length(totalForce) > maxForce) {
              totalForce = normalize(totalForce) * maxForce;
          }
      
          // Update velocity
          currentBoid.velocity.xyz += totalForce * deltaTime;
      
          // Limit velocity
          if (length(currentBoid.velocity.xyz) > maxSpeed) {
              currentBoid.velocity.xyz = normalize(currentBoid.velocity.xyz) * maxSpeed;
          }
      
          // Update position
          currentBoid.position.xyz += currentBoid.velocity.xyz * deltaTime;
      
          // Apply boundary conditions (wrap around for now)
          if (currentBoid.position.x < worldBoundsMin.x) currentBoid.position.x = worldBoundsMax.x;
          if (currentBoid.position.x > worldBoundsMax.x) currentBoid.position.x = worldBoundsMin.x;
          if (currentBoid.position.y < worldBoundsMin.y) currentBoid.position.y = worldBoundsMax.y;
          if (currentBoid.position.y > worldBoundsMax.y) currentBoid.position.y = worldBoundsMin.y;
          if (currentBoid.position.z < worldBoundsMin.z) currentBoid.position.z = worldBoundsMax.z;
          if (currentBoid.position.z > worldBoundsMax.z) currentBoid.position.z = worldBoundsMin.z;
      
          // Write back updated boid data
          boids[index] = currentBoid;
      }
