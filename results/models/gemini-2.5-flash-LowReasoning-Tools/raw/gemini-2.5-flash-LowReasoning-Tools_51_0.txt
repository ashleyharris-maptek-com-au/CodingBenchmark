from autopilot_system.pid_controller import PIDController

import time

def test_pid_controller():
    print("--- PID Controller Test (from test_pid.py) ---")

    # Scenario 1: Basic step response
    print("Scenario 1: Basic step response to reach target 10 from 0")
    pid = PIDController(kp=0.5, ki=0.01, kd=0.1, setpoint=10, output_limits=(-10, 10))
    current_value = 0
    dt = 0.1 # Simulation time step
    
    print(f"Initial: Current Value={current_value:.2f}, Setpoint={pid.setpoint}")

    for i in range(100): # Simulate for 10 seconds
        output = pid.update(current_value, dt)
        current_value += output * dt # Simple model: output directly affects value
        if (i + 1) % 10 == 0 or i == 0 or abs(pid.setpoint - current_value) < 0.1:
             print(f"Step {i+1}: Current Value={current_value:.2f}, Output={output:.2f}")
        if abs(pid.setpoint - current_value) < 0.1:
            print(f"Target reached at step {i+1} (Current Value={current_value:.2f})")
            break
    
    assert abs(pid.setpoint - current_value) < 1.0, "Scenario 1: PID did not reach setpoint effectively within tolerance."
    print("Scenario 1 Passed.")
    print("-" * 30)

    # Scenario 2: Change setpoint mid-run
    print("Scenario 2: Change setpoint from 10 to 5 mid-run") 
    pid = PIDController(kp=0.5, ki=0.01, kd=0.1, setpoint=10, output_limits=(-10, 10))
    current_value = 0
    dt = 0.1
    
    print(f"Initial: Current Value={current_value:.2f}, Setpoint={pid.setpoint}")

    for i in range(50): # Run for 5 seconds to get close to initial setpoint
        output = pid.update(current_value, dt)
        current_value += output * dt
        
    print(f"Mid-run (after 5s): Current Value={current_value:.2f}, Setpoint={pid.setpoint}")
    new_setpoint = 5
    pid.set_setpoint(new_setpoint) # Change setpoint
    print(f"New Setpoint: {pid.setpoint}")

    for i in range(100): # Simulate for another 10 seconds
        output = pid.update(current_value, dt)
        current_value += output * dt
        if (i + 1) % 10 == 0 or abs(pid.setpoint - current_value) < 0.1:
            print(f"Step {i+51}: Current Value={current_value:.2f}, Output={output:.2f}")
        if abs(pid.setpoint - current_value) < 0.1:
            print(f"New target reached at step {i+51} (Current Value={current_value:.2f})")
            break
            
    assert abs(pid.setpoint - current_value) < 1.0, "Scenario 2: PID did not reach new setpoint effectively within tolerance."
    print("Scenario 2 Passed.")

    print("--- All PID Controller Tests Passed ---") 

def test_aircraft_interface():
    print("--- Aircraft Interface Test ---") 
    aircraft = AircraftInterface(initial_altitude=10000, initial_airspeed=200, initial_heading=90, initial_pitch=0, initial_roll=0, initial_vertical_speed=0, initial_thrust_setting=0.5)
    dt = 1.0 # 1 second time step

    initial_state = aircraft.get_state()
    print(f"Initial State: {initial_state}")
    print() # Empty print for a newline

    # Test pitch command affecting vertical speed and altitude
    print("Applying pitch command +5 degrees for 10 seconds...")
    aircraft.set_pitch_command(5)
    for _ in range(10):
        aircraft.update_dynamics(dt)
        state = aircraft.get_state()
        print(f"Altitude: {state['altitude']:.2f} ft, VS: {state['vertical_speed']:.2f} fpm, Pitch: {state['pitch']:.2f} deg")
    
    assert aircraft.altitude > initial_state['altitude'], "Altitude should increase with positive pitch command."
    print("Positive pitch command test passed.")
    print() # Empty print for a newline

    # Test roll command affecting heading
    initial_heading = aircraft.heading
    print(f"Applying roll command +15 degrees for 20 seconds (initial heading: {initial_heading:.2f})...")
    aircraft.set_roll_command(15)
    for _ in range(20):
        aircraft.update_dynamics(dt)
        state = aircraft.get_state()
        print(f"Heading: {state['heading']:.2f} deg, Roll: {state['roll']:.2f} deg")

    assert aircraft.heading != initial_heading, "Heading should change with roll command."
    print("Roll command test passed.")
    print() # Empty print for a newline

    # Test thrust command affecting airspeed
    initial_airspeed = aircraft.airspeed
    print(f"Applying thrust command 0.9 for 15 seconds (initial airspeed: {initial_airspeed:.2f})...")
    aircraft.set_thrust_command(0.9)
    for _ in range(15):
        aircraft.update_dynamics(dt)
        state = aircraft.get_state()
        print(f"Airspeed: {state['airspeed']:.2f} knots, Thrust: {state['thrust_setting']:.2f}")

    assert aircraft.airspeed > initial_airspeed, "Airspeed should increase with higher thrust command."
    print("Thrust command test passed.")
    print("--- Aircraft Interface Test Passed ---")

def test_mode_control_panel():
    print("--- Mode Control Panel Test ---") 
    mcp = ModeControlPanel()

    # Test initial state
    print(f"Initial AP Master: {mcp.ap_master_on}")
    assert not mcp.ap_master_on, "AP Master should be off initially."

    # Test toggling AP Master
    mcp.toggle_ap_master()
    print(f"AP Master after toggle: {mcp.ap_master_on}")
    assert mcp.ap_master_on, "AP Master should be on after first toggle."

    # Test setting values
    mcp.set_altitude(30000)
    mcp.set_heading(270)
    mcp.set_speed(300)
    mcp.set_vs(-1500)
    print(f"Selected Alt: {mcp.selected_altitude}, HDG: {mcp.selected_heading}, SPD: {mcp.selected_speed}, VS: {mcp.selected_vs}")
    assert mcp.selected_altitude == 30000 and mcp.selected_heading == -90 and mcp.selected_speed == 300 and mcp.selected_vs == -1500, "MCP values not set correctly."

    # Test mode toggling (AP Master ON)
    mcp.toggle_alt_hold()
    print(f"ALT HOLD active: {mcp.alt_hold_active}")
    assert mcp.alt_hold_active, "ALT HOLD should be active."

    mcp.toggle_vs_hold() # Should deactivate ALT HOLD
    print(f"VS HOLD active: {mcp.vs_hold_active}, ALT HOLD active: {mcp.alt_hold_active}")
    assert mcp.vs_hold_active and not mcp.alt_hold_active, "VS HOLD should activate and ALT HOLD deactivate."

    mcp.toggle_hdg_hold()
    print(f"HDG HOLD active: {mcp.hdg_hold_active}")
    assert mcp.hdg_hold_active, "HDG HOLD should be active."

    mcp.toggle_lnav() # Should deactivate HDG HOLD
    print(f"LNAV active: {mcp.lnav_active}, HDG HOLD active: {mcp.hdg_hold_active}")
    assert mcp.lnav_active and not mcp.hdg_hold_active, "LNAV should activate and HDG HOLD deactivate."

    mcp.toggle_ap_master() # Turn off AP, should reset all modes
    print(f"AP Master after second toggle: {mcp.ap_master_on}")
    assert not mcp.ap_master_on, "AP Master should be off after second toggle."
    assert not mcp.alt_hold_active and not mcp.hdg_hold_active and not mcp.vs_hold_active and not mcp.lnav_active, "All modes should be off when AP Master is off."

    print("--- Mode Control Panel Test Passed ---")

def test_autothrottle():
    print("--- Autothrottle Test ---")
    aircraft = AircraftInterface(initial_airspeed=250, initial_thrust_setting=0.5)
    autothrottle = Autothrottle()
    dt = 0.1
    
    initial_airspeed = aircraft.airspeed
    target_airspeed = 300
    autothrottle.set_target_airspeed(target_airspeed)
    
    print(f"Initial Airspeed: {initial_airspeed:.2f} knots, Target Airspeed: {target_airspeed} knots")
    print() # Empty print for a newline
    
    ap_master_on = True
    for i in range(200): # Simulate for 20 seconds
        thrust_command = autothrottle.update(dt, aircraft.airspeed, ap_master_on)
        aircraft.set_thrust_command(thrust_command)
        aircraft.update_dynamics(dt)
        
        if (i + 1) % 20 == 0 or abs(aircraft.airspeed - target_airspeed) < 0.5:
            print(f"Step {i+1}: Airspeed={aircraft.airspeed:.2f}, Thrust={aircraft.thrust_setting:.2f}, Error={target_airspeed - aircraft.airspeed:.2f}")
        
        if abs(aircraft.airspeed - target_airspeed) < 0.5:
            print(f"Target airspeed reached at step {i+1} (Airspeed={aircraft.airspeed:.2f})")
            break
            
    assert abs(aircraft.airspeed - target_airspeed) < 1.0, "Autothrottle did not reach target airspeed effectively."
    print("Autothrottle engaged and speed hold test passed.")

    # Test autothrottle disengagement
    print("
Disengaging Autothrottle...") # Added newline here
    ap_master_on = False
    for i in range(50): # Simulate for 5 seconds after disengagement
        thrust_command = autothrottle.update(dt, aircraft.airspeed, ap_master_on)
        aircraft.set_thrust_command(thrust_command)
        aircraft.update_dynamics(dt)
        if (i + 1) % 10 == 0:
            print(f"Step {i+201}: Airspeed={aircraft.airspeed:.2f}, Thrust={aircraft.thrust_setting:.2f}")

    assert autothrottle.pid._integral == 0 and autothrottle.pid._previous_error == 0, "Autothrottle PID should be reset on disengagement."
    print("Autothrottle disengagement and reset test passed.")
    print("--- Autothrottle Test Passed ---")


if __name__ == "__main__":
    test_pid_controller()
    test_aircraft_interface()
    test_mode_control_panel()
    test_autothrottle()
