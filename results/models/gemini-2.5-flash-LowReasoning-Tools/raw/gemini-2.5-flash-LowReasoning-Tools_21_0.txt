#include "Renderer.h"

#ifdef _WIN32
#include <windows.h>
#else
#include <termios.h>
#include <unistd.h>
#include <sys/ioctl.h> // For ioctl
#endif

Renderer::Renderer(int dimensions, const std::vector<int>& map_size)
    : dimensions(dimensions), map_size(map_size) {}

void Renderer::clear_screen() {
#ifdef _WIN32
    HANDLE hConsole = GetStdHandle(STD_OUTPUT_HANDLE);
    COORD coordScreen = { 0, 0 };
    DWORD cCharsWritten;
    CONSOLE_SCREEN_BUFFER_INFO csbi;
    DWORD dwConSize;

    // Get the number of character cells in the current buffer.
    if (!GetConsoleScreenBufferInfo(hConsole, &csbi)) {
        return;
    }
    dwConSize = csbi.dwSize.X * csbi.dwSize.Y;

    // Fill the entire screen with blanks.
    if (!FillConsoleOutputCharacter(hConsole, (TCHAR) ' ',
        dwConSize, coordScreen, &cCharsWritten)) {
        return;
    }

    // Get the current text attribute.
    if (!GetConsoleScreenBufferInfo(hConsole, &csbi)) {
        return;
    }

    // Set the buffer's attributes accordingly.
    if (!FillConsoleOutputAttribute(hConsole, csbi.wAttributes,
        dwConSize, coordScreen, &cCharsWritten)) {
        return;
    }

    // Put the cursor at its home coordinates.
    SetConsoleCursorPosition(hConsole, coordScreen);
#else
    // ANSI escape codes for clearing screen and moving cursor to home
    std::cout << "\033[2J\033[H";
#endif
}

std::string Renderer::point_to_string(const PointN& p) const {
    std::stringstream ss;
    ss << "(";
    for (size_t i = 0; i < p.coords.size(); ++i) {
        ss << p.coords[i];
        if (i < p.coords.size() - 1) {
            ss << ", ";
        }
    }
    ss << ")";
    return ss.str();
}

void Renderer::print_game_info(GameState game_state, long score) {
    std::cout << "N-Dimensional Snake Game" << std::endl;
    std::cout << "Dimensions: " << dimensions;
    std::cout << " Map Size: ";
    for (int size : map_size) {
        std::cout << size << "x";
    }
    std::cout << "\b " << std::endl; // Remove last 'x'
    std::cout << "Score: " << score << std::endl;
    std::cout << "State: ";
    switch (game_state) {
        case RUNNING: std::cout << "RUNNING"; break;
        case GAME_OVER: std::cout << "GAME OVER"; break;
        case PAUSED: std::cout << "PAUSED"; break;
    }
    std::cout << std::endl;
}

void Renderer::render_3d_coordinates(const Snake& snake, const Food& food) {
    std::cout << "Snake Head: " << point_to_string(snake.get_head()) << std::endl;
    std::cout << "Snake Body (" << snake.body.size() << " segments): ";
    // Print first few segments only
    for(size_t i = 0; i < std::min((size_t)5, snake.body.size()); ++i) {
        std::cout << point_to_string(snake.body[i]) << " ";
    }
    if (snake.body.size() > 5) {
        std::cout << "...";
    }
    std::cout << std::endl;
    std::cout << "Food: " << point_to_string(food.position) << std::endl;
}

void Renderer::render(const Snake& snake, const Food& food, GameState game_state, long score) {
    clear_screen();
    print_game_info(game_state, score);
    if (dimensions == 3) { // Specific rendering for 3D
        render_3d_coordinates(snake, food);
    } else { // Generic rendering for other N
        std::cout << "Snake Head: " << point_to_string(snake.get_head()) << std::endl;
        std::cout << "Food: " << point_to_string(food.position) << std::endl;
    }

    if (game_state == GAME_OVER) {
        std::cout << "
           G A M E   O V E R           " << std::endl;
        std::cout << "         Final Score: " << score << std::endl;
    }
    std::cout.flush();
}
