#define PI 3.14159265359

struct PSInput {
    [[vk::location(0)]] float3 worldPos : TEXCOORD0;
    [[vk::location(1)]] float3 normal   : TEXCOORD1;
    [[vk::location(2)]] float2 uv       : TEXCOORD2;
    [[vk::location(3)]] float3 tangent  : TEXCOORD3;
    [[vk::location(4)]] float3 color    : TEXCOORD4;
};

[[vk::binding(0, 0)]]
cbuffer UBO : register(b0) {
    float4x4 model;
    float4x4 view;
    float4x4 projection;
    float4 lightPos;
    float4 cameraPos;
    float4 params;
};

float4 main(PSInput input) : SV_Target0 {
    // Normalize normal, as it might be interpolated
    float3 N = normalize(input.normal);

    // Calculate light direction
    // lightPos.xyz is world space light position
    // input.worldPos is world space fragment position
    float3 L = normalize(lightPos.xyz - input.worldPos);

    // Calculate diffuse component (Lambertian)
    float diffuse = max(0.0, dot(N, L));

    // Apply Lambertian shading factor
    float shadingFactor = (0.4 + 0.6 * diffuse);

    // Stripe pattern
    float stripe = sin(input.uv.y * PI * 16.0);

    // Define red and blue colors
    float3 red = float3(0.9, 0.1, 0.1);
    float3 blue = float3(0.1, 0.1, 0.9);

    // Alternate between red and blue based on the sine wave
    // If stripe > 0, it leans towards red, if stripe < 0, it leans towards blue
    // We want a clear alternation, so we can use step or a conditional.
    // A simple lerp with a biased stripe value will work to get alternation.
    float3 finalStripeColor;
    if (stripe > 0.0) {
        finalStripeColor = red;
    } else {
        finalStripeColor = blue;
    }

    // Combine stripe color with shading factor
    float3 finalColor = finalStripeColor * shadingFactor;

    // Output with alpha 1.0
    return float4(finalColor, 1.0);
}
