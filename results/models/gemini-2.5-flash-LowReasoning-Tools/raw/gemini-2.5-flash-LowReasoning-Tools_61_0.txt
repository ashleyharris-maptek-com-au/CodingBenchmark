// solution.ts

// Input reading (as provided in question.txt)
declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

interface Package {
    id: string;
    x: number;
    y: number;
}

interface InputData {
    depot: [number, number];
    truckCount: number;
    packages: Package[];
}

interface OutputData {
    routes: string[][];
}

const { depot, truckCount, packages } = data as InputData;

// Helper function for Euclidean distance
function calculateDistance(p1: [number, number], p2: [number, number]): number {
    const dx = p1[0] - p2[0];
    const dy = p1[1] - p2[1];
    return Math.sqrt(dx * dx + dy * dy);
}

// Map package IDs to their coordinates for quick lookup
const packageCoords = new Map<string, [number, number]>();
for (const pkg of packages) {
    packageCoords.set(pkg.id, [pkg.x, pkg.y]);
}

// Route planning
const routes: string[][] = Array.from({ length: truckCount }, () => []);
const currentPositions: [number, number][] = Array.from({ length: truckCount }, () => [...depot]); // Deep copy depot for each truck
const unassignedPackageIds = new Set<string>(packages.map(pkg => pkg.id));

while (unassignedPackageIds.size > 0) {
    let minDistance = Infinity;
    let bestPackageId: string | null = null;
    let bestTruckIndex: number | null = null;

    for (let truckIndex = 0; truckIndex < truckCount; truckIndex++) {
        const currentTruckPosition = currentPositions[truckIndex];
        for (const packageId of unassignedPackageIds) {
            const packagePosition = packageCoords.get(packageId);
            if (packagePosition) {
                const distance = calculateDistance(currentTruckPosition, packagePosition);
                if (distance < minDistance) {
                    minDistance = distance;
                    bestPackageId = packageId;
                    bestTruckIndex = truckIndex;
                }
            }
        }
    }

    if (bestPackageId !== null && bestTruckIndex !== null) {
        routes[bestTruckIndex].push(bestPackageId);
        currentPositions[bestTruckIndex] = packageCoords.get(bestPackageId)!; // Update current position to the assigned package's coordinates
        unassignedPackageIds.delete(bestPackageId);
    } else {
        // This case should theoretically not be reached if unassignedPackageIds.size > 0,
        // as there should always be a closest package.
        break;
    }
}

// Output
const output: OutputData = { routes };
console.log(JSON.stringify(output));
