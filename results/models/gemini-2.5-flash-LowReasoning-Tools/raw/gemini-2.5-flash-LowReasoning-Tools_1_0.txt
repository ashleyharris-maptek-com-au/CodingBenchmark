#include <iostream>
#include <vector>
#include <cmath>
#include <algorithm>
#include <limits>
#include <numeric>

// Structure to represent a city
struct City {
    double x, y;
    int id; // Original index

    City(double x_coord, double y_coord, int city_id) : x(x_coord), y(y_coord), id(city_id) {}
};

// Calculate Euclidean distance between two cities
double calculateDistance(const City& c1, const City& c2) {
    return std::sqrt(std::pow(c1.x - c2.x, 2) + std::pow(c1.y - c2.y, 2));
}

// Nearest Neighbor algorithm
std::vector<int> nearestNeighbor(int N, const std::vector<City>& cities) {
    std::vector<int> path;
    std::vector<bool> visited(N, false);

    // Start at city 0
    int current_city_idx = 0;
    path.push_back(cities[current_city_idx].id);
    visited[current_city_idx] = true;

    for (int i = 1; i < N; ++i) {
        double min_dist = std::numeric_limits<double>::max();
        int next_city_idx = -1;

        for (int j = 0; j < N; ++j) {
            if (!visited[j]) {
                double dist = calculateDistance(cities[current_city_idx], cities[j]);
                if (dist < min_dist) {
                    min_dist = dist;
                    next_city_idx = j;
                }
            }
        }
        current_city_idx = next_city_idx;
        path.push_back(cities[current_city_idx].id);
        visited[current_city_idx] = true;
    }
    return path;
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    int N;
    std::cin >> N;

    std::vector<City> cities;
    cities.reserve(N);
    for (int i = 0; i < N; ++i) {
        double x, y;
        std::cin >> x >> y;
        cities.emplace_back(x, y, i);
    }

    // Currently, only Nearest Neighbor is implemented
    std::vector<int> path = nearestNeighbor(N, cities);

    for (int i = 0; i < path.size(); ++i) {
        std::cout << path[i] << (i == path.size() - 1 ? "" : " ");
    }
    std::cout << std::endl;

    return 0;
}
