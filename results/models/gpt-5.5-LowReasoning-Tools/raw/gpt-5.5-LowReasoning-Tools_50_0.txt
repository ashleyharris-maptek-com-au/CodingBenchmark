#version 450

layout(local_size_x = 256) in;

// Each boid stored as two vec4s: [pos.xyz, 0], [vel.xyz, 0]
// Boid i is at index i*2 and i*2+1
layout(set = 0, binding = 0) readonly buffer InputBoids {
    vec4 inData[];
};

layout(set = 0, binding = 1) buffer OutputBoids {
    vec4 outData[];
};

layout(set = 0, binding = 2) uniform Params {
    uvec4 counts;      // counts.x = N
    vec4 radii;        // x=dt, y=sepRadius, z=alignRadius, w=cohRadius
    vec4 weights;      // x=sepWeight, y=alignWeight, z=cohWeight, w=maxSpeed
    vec4 bounds;       // x=boundSize, y=boundForce
};

void main()
{
    uint i = gl_GlobalInvocationID.x;
    uint n = counts.x;

    if (i >= n) {
        return;
    }

    uint base = i * 2u;
    vec3 pos = inData[base].xyz;
    vec3 vel = inData[base + 1u].xyz;

    float dt = radii.x;
    float sepRadius = radii.y;
    float alignRadius = radii.z;
    float cohRadius = radii.w;

    float sepRadius2 = sepRadius * sepRadius;
    float alignRadius2 = alignRadius * alignRadius;
    float cohRadius2 = cohRadius * cohRadius;

    vec3 sepAccum = vec3(0.0);
    vec3 alignAccum = vec3(0.0);
    vec3 cohAccum = vec3(0.0);

    uint sepCount = 0u;
    uint alignCount = 0u;
    uint cohCount = 0u;

    for (uint j = 0u; j < n; ++j) {
        if (j == i) {
            continue;
        }

        uint otherBase = j * 2u;
        vec3 otherPos = inData[otherBase].xyz;
        vec3 otherVel = inData[otherBase + 1u].xyz;

        vec3 offset = pos - otherPos;
        float dist2 = dot(offset, offset);

        if (dist2 <= 0.0) {
            continue;
        }

        if (dist2 < sepRadius2) {
            float dist = sqrt(dist2);
            sepAccum += offset / dist;
            sepCount++;
        }

        if (dist2 < alignRadius2) {
            alignAccum += otherVel;
            alignCount++;
        }

        if (dist2 < cohRadius2) {
            cohAccum += otherPos;
            cohCount++;
        }
    }

    vec3 accel = vec3(0.0);

    if (sepCount > 0u) {
        accel += (sepAccum / float(sepCount)) * weights.x;
    }

    if (alignCount > 0u) {
        vec3 avgVel = alignAccum / float(alignCount);
        accel += (avgVel - vel) * weights.y;
    }

    if (cohCount > 0u) {
        vec3 center = cohAccum / float(cohCount);
        accel += (center - pos) * weights.z;
    }

    float boundSize = bounds.x;
    float boundForce = bounds.y;
    float limit = boundSize * 0.4;
    float scale = boundSize * 0.5;

    if (pos.x > limit) {
        accel.x -= boundForce * (pos.x - limit) / scale;
    } else if (pos.x < -limit) {
        accel.x += boundForce * (-limit - pos.x) / scale;
    }

    if (pos.y > limit) {
        accel.y -= boundForce * (pos.y - limit) / scale;
    } else if (pos.y < -limit) {
        accel.y += boundForce * (-limit - pos.y) / scale;
    }

    if (pos.z > limit) {
        accel.z -= boundForce * (pos.z - limit) / scale;
    } else if (pos.z < -limit) {
        accel.z += boundForce * (-limit - pos.z) / scale;
    }

    vel += accel * dt;

    float speed2 = dot(vel, vel);
    float maxSpeed = weights.w;
    if (speed2 > maxSpeed * maxSpeed) {
        vel = normalize(vel) * maxSpeed;
    }

    pos += vel * dt;

    outData[base] = vec4(pos, 0.0);
    outData[base + 1u] = vec4(vel, 0.0);
}
