declare const require: any;
declare const process: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input.charCodeAt(0) === 0xFEFF ? input.slice(1) : input);

type Id = string | number;
type Package = { id: Id; x: number; y: number };

const depotX = Number(data.depot[0]);
const depotY = Number(data.depot[1]);
const packages: Package[] = Array.isArray(data.packages) ? data.packages : [];
const n = packages.length;
const truckCount = Math.max(1, Math.floor(Number(data.truckCount) || 1));

if (n === 0) {
  process.stdout.write(JSON.stringify({ routes: [] }));
} else {
  const xs = new Float64Array(n);
  const ys = new Float64Array(n);
  const depotDist = new Float64Array(n);
  for (let i = 0; i < n; i++) {
    xs[i] = Number(packages[i].x);
    ys[i] = Number(packages[i].y);
    depotDist[i] = Math.hypot(xs[i] - depotX, ys[i] - depotY);
  }

  function dist(a: number, b: number): number {
    return Math.hypot(xs[a] - xs[b], ys[a] - ys[b]);
  }

  class MaxHeap {
    private heap: number[] = [];
    constructor(private readonly score: Float64Array) {}
    get size(): number { return this.heap.length; }
    push(v: number): void {
      const h = this.heap;
      let i = h.length;
      h.push(v);
      while (i > 0) {
        const p = (i - 1) >> 1;
        if (this.score[h[p]] >= this.score[v]) break;
        h[i] = h[p];
        i = p;
      }
      h[i] = v;
    }
    pop(): number {
      const h = this.heap;
      const top = h[0];
      const last = h.pop() as number;
      if (h.length > 0) {
        let i = 0;
        while (true) {
          let c = i * 2 + 1;
          if (c >= h.length) break;
          if (c + 1 < h.length && this.score[h[c + 1]] > this.score[h[c]]) c++;
          if (this.score[h[c]] <= this.score[last]) break;
          h[i] = h[c];
          i = c;
        }
        h[i] = last;
      }
      return top;
    }
  }

  type Route = { nodes: number[]; active: boolean };

  function buildSavingsRoutes(): number[][] {
    const routes: Route[] = new Array(n);
    const routeOf = new Int32Array(n);
    const pos = new Int8Array(n);
    for (let i = 0; i < n; i++) {
      routes[i] = { nodes: [i], active: true };
      routeOf[i] = i;
      pos[i] = 3;
    }

    const pairCount = (n * (n - 1)) >> 1;
    const aIdx = new Int32Array(pairCount);
    const bIdx = new Int32Array(pairCount);
    const saving = new Float64Array(pairCount);
    const heap = new MaxHeap(saving);
    let p = 0;
    for (let i = 0; i < n; i++) {
      for (let j = i + 1; j < n; j++) {
        aIdx[p] = i;
        bIdx[p] = j;
        saving[p] = depotDist[i] + depotDist[j] - dist(i, j);
        heap.push(p);
        p++;
      }
    }

    let activeCount = n;
    while (activeCount > truckCount && heap.size > 0) {
      const e = heap.pop();
      const a = aIdx[e], b = bIdx[e];
      const ra = routeOf[a], rb = routeOf[b];
      if (ra === rb || !routes[ra].active || !routes[rb].active) continue;
      const pa = pos[a], pb = pos[b];
      if (pa === 0 || pb === 0) continue;

      let left = routes[ra].nodes;
      let right = routes[rb].nodes;

      if ((pa & 1) === 0) left = left.slice().reverse();
      if ((pb & 2) === 0) right = right.slice().reverse();

      const merged = left.concat(right);
      routes[ra].nodes = merged;
      routes[rb].active = false;
      for (let i = 0; i < merged.length; i++) {
        const node = merged[i];
        routeOf[node] = ra;
        pos[node] = 0;
      }
      pos[merged[0]] |= 1;
      pos[merged[merged.length - 1]] |= 2;
      activeCount--;
    }

    return routes.filter(r => r.active).map(r => r.nodes);
  }

  function angularFallbackRoutes(): number[][] {
    const order = Array.from({ length: n }, (_, i) => i);
    order.sort((a, b) => {
      const aa = Math.atan2(ys[a] - depotY, xs[a] - depotX);
      const bb = Math.atan2(ys[b] - depotY, xs[b] - depotX);
      return aa === bb ? depotDist[b] - depotDist[a] : aa - bb;
    });
    const k = Math.min(truckCount, n);
    const routes: number[][] = Array.from({ length: k }, () => []);
    for (let i = 0; i < order.length; i++) routes[Math.floor(i * k / order.length)].push(order[i]);
    return routes.filter(r => r.length > 0);
  }

  function routeLength(route: number[]): number {
    if (route.length === 0) return 0;
    let total = depotDist[route[0]] + depotDist[route[route.length - 1]];
    for (let i = 1; i < route.length; i++) total += dist(route[i - 1], route[i]);
    return total;
  }

  function nearestNeighborImprove(nodes: number[]): number[] {
    const m = nodes.length;
    if (m <= 2) return nodes.slice();
    const starts = [0];
    let far = 0;
    for (let i = 1; i < m; i++) if (depotDist[nodes[i]] > depotDist[nodes[far]]) far = i;
    if (far !== 0) starts.push(far);

    let best = nodes.slice();
    let bestLen = routeLength(best);
    for (const start of starts) {
      const used = new Uint8Array(m);
      const out: number[] = [];
      let curIdx = start;
      used[curIdx] = 1;
      out.push(nodes[curIdx]);
      for (let step = 1; step < m; step++) {
        let bestIdx = -1;
        let bestD = Infinity;
        const cur = nodes[curIdx];
        for (let i = 0; i < m; i++) {
          if (used[i]) continue;
          const d = dist(cur, nodes[i]);
          if (d < bestD) {
            bestD = d;
            bestIdx = i;
          }
        }
        curIdx = bestIdx;
        used[curIdx] = 1;
        out.push(nodes[curIdx]);
      }
      const len = routeLength(out);
      if (len < bestLen) {
        bestLen = len;
        best = out;
      }
    }
    return best;
  }

  function twoOpt(route: number[]): number[] {
    const m = route.length;
    if (m < 4) return route;
    const limit = m <= 220 ? 18 : m <= 800 ? 7 : 3;
    let improved = true;
    for (let pass = 0; pass < limit && improved; pass++) {
      improved = false;
      for (let i = 0; i < m - 1; i++) {
        const a = i === 0 ? -1 : route[i - 1];
        const b = route[i];
        for (let k = i + 1; k < m; k++) {
          const c = route[k];
          const d = k === m - 1 ? -1 : route[k + 1];
          const oldLen = (a < 0 ? depotDist[b] : dist(a, b)) + (d < 0 ? depotDist[c] : dist(c, d));
          const newLen = (a < 0 ? depotDist[c] : dist(a, c)) + (d < 0 ? depotDist[b] : dist(b, d));
          if (newLen + 1e-9 < oldLen) {
            for (let l = i, r = k; l < r; l++, r--) {
              const t = route[l];
              route[l] = route[r];
              route[r] = t;
            }
            improved = true;
          }
        }
      }
    }
    return route;
  }

  function optimizeRoutes(routes: number[][]): number[][] {
    return routes.map(r => twoOpt(nearestNeighborImprove(r)));
  }

  function totalLength(routes: number[][]): number {
    let t = 0;
    for (const r of routes) t += routeLength(r);
    return t;
  }

  const useSavings = n <= 2500;
  const candidates: number[][][] = [];
  if (useSavings) candidates.push(optimizeRoutes(buildSavingsRoutes()));
  candidates.push(optimizeRoutes(angularFallbackRoutes()));

  let best = candidates[0];
  let bestLen = totalLength(best);
  for (let i = 1; i < candidates.length; i++) {
    const len = totalLength(candidates[i]);
    if (len < bestLen) {
      bestLen = len;
      best = candidates[i];
    }
  }

  const output = {
    routes: best.map(route => route.map(i => packages[i].id))
  };
  process.stdout.write(JSON.stringify(output));
}
