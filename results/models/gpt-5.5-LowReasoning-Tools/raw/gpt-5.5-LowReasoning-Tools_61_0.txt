declare const require: any;
declare const process: any;

const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type Package = { id: any; x: number; y: number };
type Point = {
  id: any;
  x: number;
  y: number;
  angle: number;
  radius: number;
};

const depotX = Number(data.depot[0]);
const depotY = Number(data.depot[1]);
const packages: Package[] = Array.isArray(data.packages) ? data.packages : [];

function distIdx(points: Point[], a: number, b: number): number {
  const dx = points[a].x - points[b].x;
  const dy = points[a].y - points[b].y;
  return Math.sqrt(dx * dx + dy * dy);
}

function distDepot(points: Point[], a: number): number {
  const dx = points[a].x - depotX;
  const dy = points[a].y - depotY;
  return Math.sqrt(dx * dx + dy * dy);
}

function edgeLen(points: Point[], order: number[], posA: number, posB: number): number {
  const n = order.length;
  if (n === 0) return 0;
  if (posA < 0 || posA >= n || posB < 0 || posB >= n) return 0;
  return distIdx(points, order[posA], order[posB]);
}

function tourLength(points: Point[], order: number[]): number {
  const n = order.length;
  if (n === 0) return 0;
  let total = distDepot(points, order[0]) + distDepot(points, order[n - 1]);
  for (let i = 1; i < n; i++) total += distIdx(points, order[i - 1], order[i]);
  return total;
}

function reverseSegment(order: number[], i: number, k: number): void {
  while (i < k) {
    const t = order[i];
    order[i] = order[k];
    order[k] = t;
    i++;
    k--;
  }
}

function twoOpt(points: Point[], order: number[], timeLimitMs: number): void {
  const n = order.length;
  if (n < 4) return;

  const start = Date.now();
  let improved = true;
  let pass = 0;
  const maxPasses = n <= 2500 ? 12 : n <= 10000 ? 5 : 2;
  const window = n <= 2500 ? Math.min(n - 1, 220) : n <= 10000 ? 90 : 36;
  const eps = 1e-9;

  while (improved && pass < maxPasses && Date.now() - start < timeLimitMs) {
    improved = false;
    pass++;

    for (let i = 0; i < n - 1; i++) {
      if ((i & 255) === 0 && Date.now() - start >= timeLimitMs) return;

      const aIsDepot = i === 0;
      const a = aIsDepot ? -1 : order[i - 1];
      const b = order[i];
      const oldLeft = aIsDepot ? distDepot(points, b) : distIdx(points, a, b);
      const maxK = Math.min(n - 1, i + window);

      for (let k = i + 1; k <= maxK; k++) {
        const c = order[k];
        const dIsDepot = k === n - 1;
        const d = dIsDepot ? -1 : order[k + 1];

        const oldRight = dIsDepot ? distDepot(points, c) : distIdx(points, c, d);
        const newLeft = aIsDepot ? distDepot(points, c) : distIdx(points, a, c);
        const newRight = dIsDepot ? distDepot(points, b) : distIdx(points, b, d);

        if (newLeft + newRight + eps < oldLeft + oldRight) {
          reverseSegment(order, i, k);
          improved = true;
          break;
        }
      }
    }
  }
}

function nearestNeighborTour(points: Point[], seed: number): number[] {
  const n = points.length;
  const used = new Uint8Array(n);
  const order: number[] = [];
  let current = seed;
  used[current] = 1;
  order.push(current);

  for (let step = 1; step < n; step++) {
    let best = -1;
    let bestD = Infinity;
    for (let j = 0; j < n; j++) {
      if (used[j]) continue;
      const d = distIdx(points, current, j);
      if (d < bestD) {
        bestD = d;
        best = j;
      }
    }
    current = best;
    used[current] = 1;
    order.push(current);
  }
  return order;
}

function rotateBestDepotBreak(points: Point[], cycle: number[]): number[] {
  const n = cycle.length;
  if (n <= 2) return cycle.slice();

  let bestBreak = 0;
  let bestDelta = -Infinity;
  for (let i = 0; i < n; i++) {
    const a = cycle[i];
    const b = cycle[(i + 1) % n];
    const delta = distIdx(points, a, b) - distDepot(points, a) - distDepot(points, b);
    if (delta > bestDelta) {
      bestDelta = delta;
      bestBreak = i;
    }
  }

  const order: number[] = [];
  for (let t = 1; t <= n; t++) order.push(cycle[(bestBreak + t) % n]);
  return order;
}

function improveTinyByInsertion(points: Point[], order: number[], timeLimitMs: number): void {
  const n = order.length;
  if (n < 5) return;
  const start = Date.now();
  let improved = true;
  const eps = 1e-9;

  while (improved && Date.now() - start < timeLimitMs) {
    improved = false;
    for (let i = 0; i < n; i++) {
      if ((i & 31) === 0 && Date.now() - start >= timeLimitMs) return;
      const node = order[i];
      const prevCost = i === 0 ? distDepot(points, node) : distIdx(points, order[i - 1], node);
      const nextCost = i === n - 1 ? distDepot(points, node) : distIdx(points, node, order[i + 1]);
      const bridgeCost =
        i === 0
          ? i === n - 1
            ? 0
            : distDepot(points, order[i + 1])
          : i === n - 1
            ? distDepot(points, order[i - 1])
            : distIdx(points, order[i - 1], order[i + 1]);
      const removeGain = prevCost + nextCost - bridgeCost;

      let bestJ = -1;
      let bestGain = 0;
      for (let j = 0; j <= n; j++) {
        if (j === i || j === i + 1) continue;
        const leftCost = j === 0 ? distDepot(points, node) : distIdx(points, order[j - 1], node);
        const rightCost = j === n ? distDepot(points, node) : distIdx(points, node, order[j]);
        const splitCost = j === 0 ? distDepot(points, order[0]) : j === n ? distDepot(points, order[n - 1]) : edgeLen(points, order, j - 1, j);
        const gain = removeGain + splitCost - leftCost - rightCost;
        if (gain > bestGain + eps) {
          bestGain = gain;
          bestJ = j;
        }
      }

      if (bestJ >= 0) {
        order.splice(i, 1);
        if (bestJ > i) bestJ--;
        order.splice(bestJ, 0, node);
        improved = true;
      }
    }
  }
}

const n = packages.length;
if (n === 0) {
  process.stdout.write(JSON.stringify({ routes: [] }));
} else {
  const points: Point[] = packages.map((p) => {
    const x = Number(p.x);
    const y = Number(p.y);
    const dx = x - depotX;
    const dy = y - depotY;
    return {
      id: p.id,
      x,
      y,
      angle: Math.atan2(dy, dx),
      radius: Math.sqrt(dx * dx + dy * dy),
    };
  });

  const byAngle = points
    .map((_, i) => i)
    .sort((a, b) => {
      const da = points[a].angle - points[b].angle;
      if (da !== 0) return da;
      const dr = points[a].radius - points[b].radius;
      if (dr !== 0) return dr;
      return String(points[a].id).localeCompare(String(points[b].id));
    });

  let bestOrder = rotateBestDepotBreak(points, byAngle);
  let bestLen = tourLength(points, bestOrder);

  if (n <= 1800) {
    const seeds = [0, Math.floor(n / 3), Math.floor((2 * n) / 3)].map((i) => byAngle[i]);
    for (const seed of seeds) {
      const candidate = nearestNeighborTour(points, seed);
      twoOpt(points, candidate, n <= 600 ? 1800 : 2400);
      improveTinyByInsertion(points, candidate, n <= 600 ? 700 : 250);
      const len = tourLength(points, candidate);
      if (len < bestLen) {
        bestLen = len;
        bestOrder = candidate;
      }
    }
  }

  twoOpt(points, bestOrder, n <= 600 ? 2200 : n <= 2500 ? 3200 : n <= 10000 ? 4200 : 5200);
  if (n <= 900) improveTinyByInsertion(points, bestOrder, 900);

  const route = bestOrder.map((idx) => points[idx].id);
  process.stdout.write(JSON.stringify({ routes: [route] }));
}
