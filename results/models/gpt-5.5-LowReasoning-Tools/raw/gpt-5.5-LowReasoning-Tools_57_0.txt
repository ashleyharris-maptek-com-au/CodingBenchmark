const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const W = data.width | 0;
const H = data.height | 0;
const N = W * H;
const grid = data.grid || [];
const budget = Math.max(0, data.cutBudget | 0);
const ignitions = data.ignitions || [];
const towns = data.towns || [];

function idOf(x, y) { return y * W + x; }
function xOf(id) { return id % W; }
function yOf(id) { return (id / W) | 0; }
function inBounds(x, y) { return x >= 0 && y >= 0 && x < W && y < H; }

const burnable = new Uint8Array(N);
for (let y = 0; y < H; y++) {
  const row = grid[y] || '';
  for (let x = 0; x < W; x++) {
    if (row.charCodeAt(x) === 46) burnable[idOf(x, y)] = 1; // '.'
  }
}

const townValue = new Float64Array(N);
const townCell = new Uint8Array(N);
for (const t of towns) {
  const x = t.x | 0, y = t.y | 0;
  if (inBounds(x, y)) {
    const id = idOf(x, y);
    townCell[id] = 1;
    townValue[id] += Number(t.value) || 0;
  }
}

const ignitionIds = [];
for (const p of ignitions) {
  const x = p[0] | 0, y = p[1] | 0;
  if (inBounds(x, y)) {
    const id = idOf(x, y);
    if (burnable[id]) ignitionIds.push(id);
  }
}

function simulate(cut) {
  const burned = new Uint8Array(N);
  const q = new Int32Array(N);
  let head = 0, tail = 0;
  for (const s of ignitionIds) {
    if (!burnable[s] || cut.has(s) || burned[s]) continue;
    burned[s] = 1;
    q[tail++] = s;
  }
  while (head < tail) {
    const v = q[head++];
    const x = v % W;
    let n;
    if (x > 0) {
      n = v - 1;
      if (burnable[n] && !burned[n] && !cut.has(n)) { burned[n] = 1; q[tail++] = n; }
    }
    if (x + 1 < W) {
      n = v + 1;
      if (burnable[n] && !burned[n] && !cut.has(n)) { burned[n] = 1; q[tail++] = n; }
    }
    if (v >= W) {
      n = v - W;
      if (burnable[n] && !burned[n] && !cut.has(n)) { burned[n] = 1; q[tail++] = n; }
    }
    if (v + W < N) {
      n = v + W;
      if (burnable[n] && !burned[n] && !cut.has(n)) { burned[n] = 1; q[tail++] = n; }
    }
  }
  let lost = 0;
  for (let i = 0; i < N; i++) if (burned[i] && townValue[i] > 0) lost += townValue[i];
  return { burned, lost };
}

function fireBfsWithParents() {
  const dist = new Int32Array(N);
  const parent = new Int32Array(N);
  dist.fill(-1);
  parent.fill(-1);
  const q = new Int32Array(N);
  let head = 0, tail = 0;
  for (const s of ignitionIds) {
    if (!burnable[s] || dist[s] >= 0) continue;
    dist[s] = 0;
    q[tail++] = s;
  }
  while (head < tail) {
    const v = q[head++];
    const x = v % W;
    let n;
    if (x > 0) {
      n = v - 1;
      if (burnable[n] && dist[n] < 0) { dist[n] = dist[v] + 1; parent[n] = v; q[tail++] = n; }
    }
    if (x + 1 < W) {
      n = v + 1;
      if (burnable[n] && dist[n] < 0) { dist[n] = dist[v] + 1; parent[n] = v; q[tail++] = n; }
    }
    if (v >= W) {
      n = v - W;
      if (burnable[n] && dist[n] < 0) { dist[n] = dist[v] + 1; parent[n] = v; q[tail++] = n; }
    }
    if (v + W < N) {
      n = v + W;
      if (burnable[n] && dist[n] < 0) { dist[n] = dist[v] + 1; parent[n] = v; q[tail++] = n; }
    }
  }
  return { dist, parent };
}

const emptyCut = new Set();
const base = simulate(emptyCut);
if (budget === 0 || base.lost <= 0 || ignitionIds.length === 0) {
  console.log(JSON.stringify({ firebreaks: [] }));
  process.exit(0);
}

const { dist, parent } = fireBfsWithParents();
const score = new Float64Array(N);
const candidates = new Set();
const avoid = townCell;
for (const s of ignitionIds) avoid[s] = 1;

function addCandidate(id, val) {
  if (id < 0 || id >= N || !burnable[id] || avoid[id]) return;
  candidates.add(id);
  score[id] += val;
}

const maxPathMarksPerTown = Math.max(20, Math.min(800, 80 + budget * 6));
for (const t of towns) {
  const tx = t.x | 0, ty = t.y | 0;
  if (!inBounds(tx, ty)) continue;
  const tid = idOf(tx, ty);
  const val = Number(t.value) || 0;
  if (val <= 0 || !base.burned[tid] || dist[tid] < 0) continue;

  const nearWeight = val * 1.6;
  const dirs = [-1, 1, -W, W];
  for (const d of dirs) {
    const n = tid + d;
    if (d === -1 && tx === 0) continue;
    if (d === 1 && tx + 1 === W) continue;
    addCandidate(n, nearWeight);
  }

  let cur = tid;
  let steps = 0;
  while (parent[cur] >= 0 && steps < maxPathMarksPerTown) {
    cur = parent[cur];
    steps++;
    const d = Math.max(1, dist[tid] - dist[cur]);
    addCandidate(cur, val / Math.sqrt(d));
  }
}

// Accumulate town value up the shortest-path forest. Cells that dominate many
// valuable towns in the BFS tree are often good approximate firebreaks.
const order = [];
for (let i = 0; i < N; i++) if (dist[i] >= 0) order.push(i);
order.sort((a, b) => dist[b] - dist[a]);
const sub = new Float64Array(N);
for (let i = 0; i < N; i++) if (base.burned[i] && townValue[i] > 0) sub[i] = townValue[i];
for (const v of order) {
  if (sub[v] > 0 && !avoid[v]) addCandidate(v, sub[v] * 0.9);
  const p = parent[v];
  if (p >= 0) sub[p] += sub[v] * 0.985;
}

let ranked = Array.from(candidates).sort((a, b) => score[b] - score[a]);
const staticLimit = Math.max(80, Math.min(1200, budget * 35 + 180));
if (ranked.length > staticLimit) ranked = ranked.slice(0, staticLimit);

const chosen = new Set();
let currentLost = base.lost;

function tryAdd(id) {
  chosen.add(id);
  const r = simulate(chosen);
  chosen.delete(id);
  return currentLost - r.lost;
}

while (chosen.size < budget && ranked.length > 0 && currentLost > 0) {
  let best = -1, bestGain = 0;
  const evalLimit = Math.min(ranked.length, Math.max(40, Math.min(260, 50 + (budget - chosen.size) * 8)));
  for (let i = 0; i < evalLimit; i++) {
    const id = ranked[i];
    if (chosen.has(id)) continue;
    const gain = tryAdd(id);
    const tie = score[id] * 1e-9;
    if (gain + tie > bestGain) {
      bestGain = gain + tie;
      best = id;
    }
  }
  if (best < 0 || bestGain <= 1e-9) break;
  chosen.add(best);
  currentLost = simulate(chosen).lost;
  ranked = ranked.filter(v => v !== best);
}

// If validated greedy stalls, use the remaining budget on high static-score
// cells. These can still improve cases where savings require several cuts.
for (const id of ranked) {
  if (chosen.size >= budget) break;
  if (!chosen.has(id)) chosen.add(id);
}

const firebreaks = Array.from(chosen, id => [xOf(id), yOf(id)]);
console.log(JSON.stringify({ firebreaks }));
