const fs = require('fs');
let input = fs.readFileSync(0, 'utf8');
if (input.charCodeAt(0) === 0xFEFF) input = input.slice(1);
const data = JSON.parse(input);

const W = data.width | 0;
const H = data.height | 0;
const N = W * H;
const grid = data.grid || [];
const BUDGET = Math.max(0, data.cutBudget | 0);

const forest = new Uint8Array(N);
for (let y = 0; y < H; y++) {
  const row = grid[y] || '';
  for (let x = 0; x < W; x++) {
    if (row.charCodeAt(x) === 46) forest[y * W + x] = 1;
  }
}

const isIgnition = new Uint8Array(N);
const ignitions = [];
for (const p of data.ignitions || []) {
  const x = p[0] | 0, y = p[1] | 0;
  if (x >= 0 && x < W && y >= 0 && y < H) {
    const i = y * W + x;
    if (forest[i] && !isIgnition[i]) {
      isIgnition[i] = 1;
      ignitions.push(i);
    }
  }
}

const townValue = new Float64Array(N);
let totalTownValue = 0;
for (const t of data.towns || []) {
  const x = t.x | 0, y = t.y | 0;
  const v = Number(t.value) || 0;
  if (x >= 0 && x < W && y >= 0 && y < H && v > 0) {
    townValue[y * W + x] += v;
    totalTownValue += v;
  }
}

function simulate(cut) {
  const burned = new Uint8Array(N);
  const q = new Int32Array(N);
  let head = 0, tail = 0;
  for (const s of ignitions) {
    if (forest[s] && !cut[s] && !burned[s]) {
      burned[s] = 1;
      q[tail++] = s;
    }
  }
  while (head < tail) {
    const i = q[head++];
    const x = i % W;
    let n = i - W;
    if (i >= W && forest[n] && !cut[n] && !burned[n]) {
      burned[n] = 1; q[tail++] = n;
    }
    n = i + W;
    if (n < N && forest[n] && !cut[n] && !burned[n]) {
      burned[n] = 1; q[tail++] = n;
    }
    n = i - 1;
    if (x > 0 && forest[n] && !cut[n] && !burned[n]) {
      burned[n] = 1; q[tail++] = n;
    }
    n = i + 1;
    if (x + 1 < W && forest[n] && !cut[n] && !burned[n]) {
      burned[n] = 1; q[tail++] = n;
    }
  }
  let burnedValue = 0;
  for (let i = 0; i < N; i++) if (burned[i]) burnedValue += townValue[i];
  return { burned, burnedValue };
}

const emptyCut = new Uint8Array(N);
const base = simulate(emptyCut);
const baselineBurnedValue = base.burnedValue;

if (BUDGET <= 0 || baselineBurnedValue <= 0 || ignitions.length === 0) {
  console.log(JSON.stringify({ firebreaks: [] }));
  process.exit(0);
}

const protectedCell = new Uint8Array(N);
for (let i = 0; i < N; i++) if (townValue[i] > 0 || isIgnition[i]) protectedCell[i] = 1;

function keyOf(cells) {
  cells.sort((a, b) => a - b);
  let out = '';
  let last = -1, w = 0;
  for (const c of cells) {
    if (c !== last) {
      out += c + ',';
      cells[w++] = c;
      last = c;
    }
  }
  cells.length = w;
  return out;
}

const burnedTowns = [];
for (let i = 0; i < N; i++) {
  if (base.burned[i] && townValue[i] > 0 && forest[i] && !isIgnition[i]) {
    burnedTowns.push({ idx: i, value: townValue[i] });
  }
}
burnedTowns.sort((a, b) => b.value - a.value);

const candidates = [];
const seen = new Set();

function addCandidate(cells, hint) {
  if (cells.length === 0 || cells.length > BUDGET) return;
  const k = keyOf(cells);
  if (!k || seen.has(k)) return;
  seen.add(k);
  candidates.push({ cells: cells.slice(), hint });
}

function regionCandidate(center, radius) {
  if (radius === 0) {
    const cells = [];
    const x = center % W;
    const ns = [];
    if (center >= W) ns.push(center - W);
    if (center + W < N) ns.push(center + W);
    if (x > 0) ns.push(center - 1);
    if (x + 1 < W) ns.push(center + 1);
    for (const n of ns) if (forest[n] && !protectedCell[n]) cells.push(n);
    return cells;
  }

  const inRegion = new Uint8Array(N);
  const q = new Int32Array(Math.min(N, 1 + 2 * radius * (radius + 1)));
  const d = new Int16Array(q.length);
  let head = 0, tail = 0;
  inRegion[center] = 1; q[tail] = center; d[tail++] = 0;
  let hasIgnition = isIgnition[center] === 1;
  while (head < tail) {
    const i = q[head], dist = d[head++];
    if (dist >= radius) continue;
    const x = i % W;
    const ns = [];
    if (i >= W) ns.push(i - W);
    if (i + W < N) ns.push(i + W);
    if (x > 0) ns.push(i - 1);
    if (x + 1 < W) ns.push(i + 1);
    for (const n of ns) {
      if (forest[n] && base.burned[n] && !inRegion[n]) {
        inRegion[n] = 1;
        if (isIgnition[n]) hasIgnition = true;
        if (tail < q.length) { q[tail] = n; d[tail++] = dist + 1; }
      }
    }
  }
  if (hasIgnition) return [];

  const cells = [];
  for (let p = 0; p < tail; p++) {
    const i = q[p];
    const x = i % W;
    const ns = [];
    if (i >= W) ns.push(i - W);
    if (i + W < N) ns.push(i + W);
    if (x > 0) ns.push(i - 1);
    if (x + 1 < W) ns.push(i + 1);
    for (const n of ns) {
      if (forest[n] && base.burned[n] && !inRegion[n] && !protectedCell[n]) cells.push(n);
    }
  }
  return cells;
}

const maxTownCandidates = Math.min(burnedTowns.length, N > 500000 ? 80 : N > 150000 ? 160 : 320);
const radii = BUDGET < 8 ? [0, 1] : BUDGET < 24 ? [0, 1, 2, 3] : [0, 1, 2, 3, 5, 8];
for (let t = 0; t < maxTownCandidates; t++) {
  const town = burnedTowns[t];
  for (const r of radii) {
    const cells = regionCandidate(town.idx, r);
    if (cells.length <= BUDGET) addCandidate(cells, town.value / Math.max(1, cells.length));
  }
}

const aroundIgnitions = [];
for (const s of ignitions) {
  const x = s % W;
  const ns = [];
  if (s >= W) ns.push(s - W);
  if (s + W < N) ns.push(s + W);
  if (x > 0) ns.push(s - 1);
  if (x + 1 < W) ns.push(s + 1);
  for (const n of ns) if (forest[n] && !protectedCell[n]) aroundIgnitions.push(n);
}
addCandidate(aroundIgnitions, baselineBurnedValue / Math.max(1, aroundIgnitions.length));

const maxInitialEval = Math.min(candidates.length, N > 500000 ? 120 : N > 150000 ? 220 : 500);
candidates.sort((a, b) => b.hint - a.hint || a.cells.length - b.cells.length);

const scored = [];
for (let ci = 0; ci < maxInitialEval; ci++) {
  const c = candidates[ci];
  const cut = new Uint8Array(N);
  for (const cell of c.cells) cut[cell] = 1;
  const gain = baselineBurnedValue - simulate(cut).burnedValue;
  if (gain > 0) scored.push({ cells: c.cells, gain, density: gain / c.cells.length });
}
scored.sort((a, b) => b.density - a.density || b.gain - a.gain);

let chosen = [];
let chosenCut = new Uint8Array(N);
let used = 0;
let currentBurnedValue = baselineBurnedValue;

for (const c of scored) {
  let extra = 0;
  for (const cell of c.cells) if (!chosenCut[cell]) extra++;
  if (extra === 0 || used + extra > BUDGET) continue;

  const trialCut = chosenCut.slice();
  for (const cell of c.cells) trialCut[cell] = 1;
  const trialBurnedValue = simulate(trialCut).burnedValue;
  if (trialBurnedValue < currentBurnedValue) {
    chosenCut = trialCut;
    currentBurnedValue = trialBurnedValue;
    used += extra;
    for (const cell of c.cells) {
      if (!chosen.includes(cell)) chosen.push(cell);
    }
    if (used >= BUDGET || currentBurnedValue <= 0) break;
  }
}

if (used < BUDGET && currentBurnedValue > 0) {
  for (const t of burnedTowns) {
    if (used >= BUDGET) break;
    const cells = regionCandidate(t.idx, 0);
    let extra = 0;
    for (const cell of cells) if (!chosenCut[cell]) extra++;
    if (extra === 0 || used + extra > BUDGET) continue;
    const trialCut = chosenCut.slice();
    for (const cell of cells) trialCut[cell] = 1;
    const trialBurnedValue = simulate(trialCut).burnedValue;
    if (trialBurnedValue < currentBurnedValue) {
      chosenCut = trialCut;
      currentBurnedValue = trialBurnedValue;
      used += extra;
      for (const cell of cells) if (!chosen.includes(cell)) chosen.push(cell);
    }
  }
}

const firebreaks = [];
const emitted = new Uint8Array(N);
for (const i of chosen) {
  if (firebreaks.length >= BUDGET) break;
  if (forest[i] && !protectedCell[i] && !emitted[i]) {
    emitted[i] = 1;
    firebreaks.push([i % W, Math.floor(i / W)]);
  }
}

console.log(JSON.stringify({ firebreaks }));
