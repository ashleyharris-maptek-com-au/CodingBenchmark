const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input.replace(/^\uFEFF/, ''));

const depot = data.depot || [0, 0];
const packages = Array.isArray(data.packages) ? data.packages : [];
const truckCount = data.truckCount | 0;
const n = packages.length;

if (n === 0 || truckCount <= 0) {
  console.log(JSON.stringify({ routes: [] }));
  process.exit(0);
}

const dx0 = depot[0], dy0 = depot[1];

function distIdx(a, b) {
  const pa = packages[a], pb = packages[b];
  const dx = pa.x - pb.x, dy = pa.y - pb.y;
  return Math.hypot(dx, dy);
}

function distDepotIdx(a) {
  const p = packages[a];
  return Math.hypot(p.x - dx0, p.y - dy0);
}

function routeLength(route) {
  const m = route.length;
  if (m === 0) return 0;
  let total = distDepotIdx(route[0]) + distDepotIdx(route[m - 1]);
  for (let i = 1; i < m; i++) total += distIdx(route[i - 1], route[i]);
  return total;
}

function angleRoute() {
  const ids = new Array(n);
  for (let i = 0; i < n; i++) ids[i] = i;
  ids.sort((a, b) => {
    const pa = packages[a], pb = packages[b];
    const aa = Math.atan2(pa.y - dy0, pa.x - dx0);
    const ab = Math.atan2(pb.y - dy0, pb.x - dx0);
    if (aa !== ab) return aa - ab;
    const ra = (pa.x - dx0) * (pa.x - dx0) + (pa.y - dy0) * (pa.y - dy0);
    const rb = (pb.x - dx0) * (pb.x - dx0) + (pb.y - dy0) * (pb.y - dy0);
    return ra - rb;
  });
  return ids;
}

function nearestNeighborRoute(start) {
  const used = new Uint8Array(n);
  const route = new Array(n);
  let cur = start;
  used[cur] = 1;
  route[0] = cur;
  for (let k = 1; k < n; k++) {
    const pc = packages[cur];
    let best = -1, bestD = Infinity;
    for (let j = 0; j < n; j++) {
      if (used[j]) continue;
      const pj = packages[j];
      const dx = pc.x - pj.x, dy = pc.y - pj.y;
      const d = dx * dx + dy * dy;
      if (d < bestD) {
        bestD = d;
        best = j;
      }
    }
    cur = best;
    used[cur] = 1;
    route[k] = cur;
  }
  return route;
}

function farthestFromDepot() {
  let best = 0, bestD = -1;
  for (let i = 0; i < n; i++) {
    const p = packages[i];
    const dx = p.x - dx0, dy = p.y - dy0;
    const d = dx * dx + dy * dy;
    if (d > bestD) {
      bestD = d;
      best = i;
    }
  }
  return best;
}

function reverseSegment(route, i, k) {
  while (i < k) {
    const t = route[i];
    route[i++] = route[k];
    route[k--] = t;
  }
}

function twoOpt(route, timeLimitMs) {
  if (route.length < 4) return route;
  const deadline = Date.now() + timeLimitMs;
  const m = route.length;
  let improved = true;
  let passes = 0;
  const maxPasses = m <= 1200 ? 40 : m <= 5000 ? 12 : 4;

  while (improved && passes++ < maxPasses && Date.now() < deadline) {
    improved = false;
    for (let i = 0; i < m - 1; i++) {
      const aDepot = i === 0;
      const a = aDepot ? -1 : route[i - 1];
      const b = route[i];
      const oldAB = aDepot ? distDepotIdx(b) : distIdx(a, b);

      for (let k = i + 1; k < m; k++) {
        const c = route[k];
        const dDepot = k === m - 1;
        const d = dDepot ? -1 : route[k + 1];
        const oldCD = dDepot ? distDepotIdx(c) : distIdx(c, d);
        const newAC = aDepot ? distDepotIdx(c) : distIdx(a, c);
        const newBD = dDepot ? distDepotIdx(b) : distIdx(b, d);

        if (newAC + newBD + 1e-12 < oldAB + oldCD) {
          reverseSegment(route, i, k);
          improved = true;
          break;
        }
      }
      if (improved) break;
      if ((i & 31) === 0 && Date.now() >= deadline) return route;
    }
  }
  return route;
}

function rotateBestDepotBreak(cycle) {
  if (cycle.length <= 2) return cycle;
  let bestBreak = 0;
  let bestGain = -Infinity;
  for (let i = 0; i < cycle.length; i++) {
    const a = cycle[i];
    const b = cycle[(i + 1) % cycle.length];
    const gain = distIdx(a, b) - distDepotIdx(a) - distDepotIdx(b);
    if (gain > bestGain) {
      bestGain = gain;
      bestBreak = i;
    }
  }
  return cycle.slice(bestBreak + 1).concat(cycle.slice(0, bestBreak + 1));
}

let route = angleRoute();
let bestLen = routeLength(route);

if (n <= 3500) {
  const starts = [farthestFromDepot(), 0, Math.floor(n / 2)];
  const seen = new Set();
  for (const s of starts) {
    if (seen.has(s)) continue;
    seen.add(s);
    const candidate = nearestNeighborRoute(s);
    const len = routeLength(candidate);
    if (len < bestLen) {
      route = candidate;
      bestLen = len;
    }
  }
}

const limit = n <= 500 ? 1800 : n <= 2000 ? 2500 : n <= 8000 ? 3200 : 1800;
route = twoOpt(route, limit);
route = rotateBestDepotBreak(route);

console.log(JSON.stringify({ routes: [route.map(i => packages[i].id)] }));
