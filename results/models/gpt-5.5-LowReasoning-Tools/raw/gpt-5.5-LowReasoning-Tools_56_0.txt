const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const packages = Array.isArray(data.packages) ? data.packages : [];
const n = packages.length;

if (n === 0) {
  process.stdout.write(JSON.stringify({ routes: [] }));
  process.exit(0);
}

const depot = Array.isArray(data.depot) ? data.depot : [0, 0];
const dx0 = Number(depot[0]) || 0;
const dy0 = Number(depot[1]) || 0;

const pts = packages.map((p, i) => {
  const x = Number(p.x) || 0;
  const y = Number(p.y) || 0;
  const dx = x - dx0;
  const dy = y - dy0;
  return {
    id: p.id,
    idx: i,
    x,
    y,
    ax: dx,
    ay: dy,
    angle: Math.atan2(dy, dx),
    r2: dx * dx + dy * dy
  };
});

function distIdx(a, b) {
  const pa = pts[a], pb = pts[b];
  const dx = pa.x - pb.x, dy = pa.y - pb.y;
  return Math.hypot(dx, dy);
}

function distDepotIdx(a) {
  const p = pts[a];
  return Math.hypot(p.x - dx0, p.y - dy0);
}

function routeLen(route) {
  if (route.length === 0) return 0;
  let s = distDepotIdx(route[0]) + distDepotIdx(route[route.length - 1]);
  for (let i = 1; i < route.length; i++) s += distIdx(route[i - 1], route[i]);
  return s;
}

function betterRoute(a, b) {
  if (!a) return b;
  return routeLen(b) + 1e-9 < routeLen(a) ? b : a;
}

function angleRoute(reverse = false) {
  const arr = pts.slice().sort((a, b) => {
    if (a.angle !== b.angle) return a.angle - b.angle;
    if (a.r2 !== b.r2) return a.r2 - b.r2;
    return a.idx - b.idx;
  }).map(p => p.idx);
  if (reverse) arr.reverse();
  return arr;
}

function mortonRoute() {
  let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
  for (const p of pts) {
    if (p.x < minX) minX = p.x;
    if (p.y < minY) minY = p.y;
    if (p.x > maxX) maxX = p.x;
    if (p.y > maxY) maxY = p.y;
  }
  const sx = maxX > minX ? 65535 / (maxX - minX) : 1;
  const sy = maxY > minY ? 65535 / (maxY - minY) : 1;
  function part(v) {
    v &= 0xffff;
    v = (v | (v << 8)) & 0x00ff00ff;
    v = (v | (v << 4)) & 0x0f0f0f0f;
    v = (v | (v << 2)) & 0x33333333;
    v = (v | (v << 1)) & 0x55555555;
    return v >>> 0;
  }
  return pts.slice().sort((a, b) => {
    const ax = Math.max(0, Math.min(65535, Math.floor((a.x - minX) * sx)));
    const ay = Math.max(0, Math.min(65535, Math.floor((a.y - minY) * sy)));
    const bx = Math.max(0, Math.min(65535, Math.floor((b.x - minX) * sx)));
    const by = Math.max(0, Math.min(65535, Math.floor((b.y - minY) * sy)));
    const ac = (part(ax) | (part(ay) << 1)) >>> 0;
    const bc = (part(bx) | (part(by) << 1)) >>> 0;
    return ac === bc ? a.idx - b.idx : ac - bc;
  }).map(p => p.idx);
}

function nearestNeighborRoute() {
  const unused = new Set();
  for (let i = 0; i < n; i++) unused.add(i);
  let cur = -1;
  const route = [];
  while (unused.size) {
    let best = -1, bestD = Infinity;
    for (const j of unused) {
      const d = cur < 0 ? distDepotIdx(j) : distIdx(cur, j);
      if (d < bestD || (d === bestD && j < best)) {
        bestD = d;
        best = j;
      }
    }
    route.push(best);
    unused.delete(best);
    cur = best;
  }
  return route;
}

function twoOpt(route, timeLimitMs) {
  const start = Date.now();
  const m = route.length;
  if (m < 4) return route;
  const maxPasses = m <= 500 ? 50 : m <= 2000 ? 20 : 8;

  for (let pass = 0; pass < maxPasses; pass++) {
    let improved = false;
    for (let i = 0; i < m - 1; i++) {
      if ((i & 255) === 0 && Date.now() - start > timeLimitMs) return route;
      const a = i === 0 ? -1 : route[i - 1];
      const b = route[i];
      const dab = a < 0 ? distDepotIdx(b) : distIdx(a, b);
      for (let k = i + 1; k < m; k++) {
        if (k === i) continue;
        const c = route[k];
        const d = k === m - 1 ? -1 : route[k + 1];
        const dcd = d < 0 ? distDepotIdx(c) : distIdx(c, d);
        const dac = a < 0 ? distDepotIdx(c) : distIdx(a, c);
        const dbd = d < 0 ? distDepotIdx(b) : distIdx(b, d);
        if (dac + dbd + 1e-10 < dab + dcd) {
          let l = i, r = k;
          while (l < r) {
            const tmp = route[l];
            route[l++] = route[r];
            route[r--] = tmp;
          }
          improved = true;
          break;
        }
      }
      if (improved) break;
    }
    if (!improved) break;
  }
  return route;
}

let best = null;
best = betterRoute(best, angleRoute(false));
best = betterRoute(best, angleRoute(true));
best = betterRoute(best, mortonRoute());
if (n <= 3500) best = betterRoute(best, nearestNeighborRoute());

const limit = n <= 500 ? 1800 : n <= 2000 ? 2600 : n <= 10000 ? 3200 : 4500;
best = twoOpt(best.slice(), limit);

process.stdout.write(JSON.stringify({ routes: [best.map(i => pts[i].id)] }));
