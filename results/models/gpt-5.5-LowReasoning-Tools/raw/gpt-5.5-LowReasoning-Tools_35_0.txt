use std::cmp::Reverse;
use std::collections::BinaryHeap;
use std::io::{self, Read};
use std::time::{Duration, Instant};

struct Rng {
    x: u64,
}

impl Rng {
    fn new(seed: u64) -> Self {
        Self { x: seed | 1 }
    }

    fn next_u64(&mut self) -> u64 {
        let mut z = self.x;
        z ^= z << 7;
        z ^= z >> 9;
        z = z.wrapping_mul(0x9e3779b97f4a7c15);
        self.x = z;
        z
    }

    fn gen_usize(&mut self, n: usize) -> usize {
        if n == 0 {
            0
        } else {
            (self.next_u64() as usize) % n
        }
    }

    fn shuffle<T>(&mut self, a: &mut [T]) {
        for i in (1..a.len()).rev() {
            a.swap(i, self.gen_usize(i + 1));
        }
    }
}

fn compute_cover_count(adj: &[Vec<usize>], in_set: &[bool]) -> Vec<u16> {
    let n = adj.len();
    let mut cover = vec![0u16; n];
    for v in 0..n {
        if in_set[v] {
            cover[v] = cover[v].saturating_add(1);
            for &u in &adj[v] {
                cover[u] = cover[u].saturating_add(1);
            }
        }
    }
    cover
}

fn prune(adj: &[Vec<usize>], in_set: &mut [bool], set: &mut Vec<usize>, rng: &mut Rng) {
    let mut cover = compute_cover_count(adj, in_set);
    let mut order = set.clone();
    order.sort_by_key(|&v| adj[v].len());
    if order.len() > 1 {
        let swaps = (order.len() / 5).max(1).min(1000);
        for _ in 0..swaps {
            let a = rng.gen_usize(order.len());
            let b = rng.gen_usize(order.len());
            order.swap(a, b);
        }
    }

    for v in order {
        if !in_set[v] {
            continue;
        }
        let mut ok = cover[v] > 1;
        if ok {
            for &u in &adj[v] {
                if cover[u] <= 1 {
                    ok = false;
                    break;
                }
            }
        }
        if ok {
            in_set[v] = false;
            cover[v] -= 1;
            for &u in &adj[v] {
                cover[u] -= 1;
            }
        }
    }
    set.clear();
    for v in 0..adj.len() {
        if in_set[v] {
            set.push(v);
        }
    }
}

fn greedy(adj: &[Vec<usize>], rng: &mut Rng, randomized: bool) -> Vec<usize> {
    let n = adj.len();
    let mut dominated = vec![false; n];
    let mut undom = n;
    let mut in_set = vec![false; n];
    let mut set = Vec::new();
    let mut heap = BinaryHeap::new();

    for v in 0..n {
        heap.push((adj[v].len() + 1, Reverse(rng.gen_usize(1 << 20)), v));
    }

    while undom > 0 {
        let mut candidates = Vec::new();
        let mut best_gain = 0usize;

        loop {
            let Some((old_gain, tie, v)) = heap.pop() else {
                break;
            };
            if in_set[v] {
                continue;
            }
            let mut gain = usize::from(!dominated[v]);
            for &u in &adj[v] {
                gain += usize::from(!dominated[u]);
            }
            if gain == old_gain {
                if !randomized {
                    candidates.push((gain, tie, v));
                    break;
                }
                if gain + 2 < best_gain && !candidates.is_empty() {
                    heap.push((gain, tie, v));
                    break;
                }
                best_gain = best_gain.max(gain);
                candidates.push((gain, tie, v));
                if candidates.len() >= 12 {
                    break;
                }
            } else {
                heap.push((gain, tie, v));
            }
        }

        if candidates.is_empty() {
            if let Some(v) = (0..n).find(|&i| !dominated[i]) {
                candidates.push((1, Reverse(0), v));
            }
        }

        let idx = if randomized && candidates.len() > 1 {
            let max_gain = candidates.iter().map(|x| x.0).max().unwrap_or(0);
            let good: Vec<_> = candidates
                .iter()
                .enumerate()
                .filter(|(_, x)| x.0 + 1 >= max_gain)
                .map(|(i, _)| i)
                .collect();
            good[rng.gen_usize(good.len())]
        } else {
            candidates
                .iter()
                .enumerate()
                .max_by_key(|(_, x)| (x.0, Reverse(x.2)))
                .map(|(i, _)| i)
                .unwrap()
        };

        for (i, c) in candidates.into_iter().enumerate() {
            if i != idx {
                heap.push(c);
            } else {
                let v = c.2;
                if in_set[v] {
                    continue;
                }
                in_set[v] = true;
                set.push(v);
                if !dominated[v] {
                    dominated[v] = true;
                    undom -= 1;
                }
                for &u in &adj[v] {
                    if !dominated[u] {
                        dominated[u] = true;
                        undom -= 1;
                    }
                }
            }
        }
    }

    prune(adj, &mut in_set, &mut set, rng);
    set
}

fn repair_after_removals(adj: &[Vec<usize>], base: &[usize], rng: &mut Rng) -> Vec<usize> {
    let n = adj.len();
    let mut in_set = vec![false; n];
    for &v in base {
        in_set[v] = true;
    }

    let mut dominated = vec![false; n];
    let mut undom = n;
    for &v in base {
        if !dominated[v] {
            dominated[v] = true;
            undom -= 1;
        }
        for &u in &adj[v] {
            if !dominated[u] {
                dominated[u] = true;
                undom -= 1;
            }
        }
    }

    let mut set = base.to_vec();
    while undom > 0 {
        let mut best_v = 0usize;
        let mut best_score = 0isize;
        let start = rng.gen_usize(n.max(1));
        for k in 0..n {
            let v = (start + k) % n;
            if in_set[v] {
                continue;
            }
            let mut gain = usize::from(!dominated[v]);
            for &u in &adj[v] {
                gain += usize::from(!dominated[u]);
            }
            let noise = (rng.next_u64() & 7) as isize;
            let score = (gain * 1024 + adj[v].len()) as isize + noise;
            if score > best_score {
                best_score = score;
                best_v = v;
            }
        }
        in_set[best_v] = true;
        set.push(best_v);
        if !dominated[best_v] {
            dominated[best_v] = true;
            undom -= 1;
        }
        for &u in &adj[best_v] {
            if !dominated[u] {
                dominated[u] = true;
                undom -= 1;
            }
        }
    }

    prune(adj, &mut in_set, &mut set, rng);
    set
}

fn verify(adj: &[Vec<usize>], set: &[usize]) -> bool {
    let n = adj.len();
    let mut dominated = vec![false; n];
    for &v in set {
        if v >= n {
            return false;
        }
        dominated[v] = true;
        for &u in &adj[v] {
            dominated[u] = true;
        }
    }
    dominated.into_iter().all(|x| x)
}

fn main() {
    let start_time = Instant::now();
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();
    let n: usize = it.next().unwrap_or("0").parse().unwrap();
    let m: usize = it.next().unwrap_or("0").parse().unwrap();

    let mut adj = vec![Vec::<usize>::new(); n];
    for _ in 0..m {
        let Some(a) = it.next() else { break };
        let Some(b) = it.next() else { break };
        let u: usize = a.parse().unwrap();
        let v: usize = b.parse().unwrap();
        if u < n && v < n && u != v {
            adj[u].push(v);
            adj[v].push(u);
        }
    }
    for v in 0..n {
        adj[v].sort_unstable();
        adj[v].dedup();
    }

    let mut seed = (n as u64).wrapping_mul(1_000_003) ^ (m as u64).wrapping_mul(97_531);
    for (i, ns) in adj.iter().enumerate().take(4096) {
        seed ^= ((i as u64) << 32) ^ ns.len() as u64;
    }
    let mut rng = Rng::new(seed);

    let mut best = greedy(&adj, &mut rng, false);
    let time_limit = if n <= 2_000 {
        Duration::from_millis(250)
    } else if n <= 20_000 {
        Duration::from_secs(2)
    } else if n <= 200_000 {
        Duration::from_secs(8)
    } else {
        Duration::from_secs(25)
    };

    while start_time.elapsed() < time_limit {
        let mut candidate = greedy(&adj, &mut rng, true);
        if candidate.len() >= best.len() && !best.is_empty() {
            let mut keep = best.clone();
            rng.shuffle(&mut keep);
            let remove = (keep.len() / 20).max(1).min(20);
            keep.truncate(keep.len().saturating_sub(remove));
            candidate = repair_after_removals(&adj, &keep, &mut rng);
        }
        if candidate.len() < best.len() && verify(&adj, &candidate) {
            best = candidate;
        }
    }

    if !verify(&adj, &best) {
        best = (0..n).collect();
    }

    best.sort_unstable();
    println!("{}", best.len());
    for (i, v) in best.iter().enumerate() {
        if i > 0 {
            print!(" ");
        }
        print!("{}", v);
    }
    println!();
}
