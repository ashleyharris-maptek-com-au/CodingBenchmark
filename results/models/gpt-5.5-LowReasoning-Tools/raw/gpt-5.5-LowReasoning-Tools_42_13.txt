OpCapability Shader
OpCapability Geometry
%glsl = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main InputTrianglesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 12
OpExecutionMode %main Invocations 1
OpSource GLSL 450
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2
OpDecorate %outColor Location 0
%void = OpTypeVoid
%float = OpTypeFloat 32
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%gl_PerVertex_in = OpTypeStruct %v4float
%gl_PerVertex_out = OpTypeStruct %v4float
%uint_6 = OpConstant %uint 6
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_6
%arr_v3_in = OpTypeArray %v3float %uint_6
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%gl_in = OpVariable %ptr_in_arr_pv Input
%worldPos = OpVariable %ptr_in_arr_v3 Input
%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input
%gl_out = OpVariable %ptr_out_pv Output
%outColor = OpVariable %ptr_out_v3 Output
%float_0 = OpConstant %float 0
%float_1 = OpConstant %float 1
%float_n1 = OpConstant %float -1
%float_half = OpConstant %float 0.5
%float_thickness = OpConstant %float 0.008
%int_0 = OpConstant %int 0
%int_2 = OpConstant %int 2
%int_4 = OpConstant %int 4
%fn_void = OpTypeFunction %void
%fn_emitEdge = OpTypeFunction %void %int %int
%emitEdge = OpFunction %void None %fn_emitEdge
%a = OpFunctionParameter %int
%b = OpFunctionParameter %int
%emit_label = OpLabel
%ptra = OpAccessChain %ptr_in_v4 %gl_in %a %int_0
%ptrb = OpAccessChain %ptr_in_v4 %gl_in %b %int_0
%pa = OpLoad %v4float %ptra
%pb = OpLoad %v4float %ptrb
%xy_a = OpVectorShuffle %v2float %pa %pa 0 1
%xy_b = OpVectorShuffle %v2float %pb %pb 0 1
%wa = OpCompositeExtract %float %pa 3
%wb = OpCompositeExtract %float %pb 3
%inv_wa = OpFDiv %float %float_1 %wa
%inv_wb = OpFDiv %float %float_1 %wb
%ndc_a = OpVectorTimesScalar %v2float %xy_a %inv_wa
%ndc_b = OpVectorTimesScalar %v2float %xy_b %inv_wb
%edge = OpFSub %v2float %ndc_b %ndc_a
%dir = OpExtInst %v2float %glsl Normalize %edge
%dir_y = OpCompositeExtract %float %dir 1
%dir_x = OpCompositeExtract %float %dir 0
%neg_dir_y = OpFNegate %float %dir_y
%perp = OpCompositeConstruct %v2float %neg_dir_y %dir_x
%perp_scaled = OpVectorTimesScalar %v2float %perp %float_thickness
%off_a_xy = OpVectorTimesScalar %v2float %perp_scaled %wa
%off_b_xy = OpVectorTimesScalar %v2float %perp_scaled %wb
%off_a_x = OpCompositeExtract %float %off_a_xy 0
%off_a_y = OpCompositeExtract %float %off_a_xy 1
%off_b_x = OpCompositeExtract %float %off_b_xy 0
%off_b_y = OpCompositeExtract %float %off_b_xy 1
%off_a = OpCompositeConstruct %v4float %off_a_x %off_a_y %float_0 %float_0
%off_b = OpCompositeConstruct %v4float %off_b_x %off_b_y %float_0 %float_0
%pa_plus = OpFAdd %v4float %pa %off_a
%pa_minus = OpFSub %v4float %pa %off_a
%pb_plus = OpFAdd %v4float %pb %off_b
%pb_minus = OpFSub %v4float %pb %off_b
%colpa = OpAccessChain %ptr_in_v3 %color %a
%colpb = OpAccessChain %ptr_in_v3 %color %b
%cola = OpLoad %v3float %colpa
%colb = OpLoad %v3float %colpb
%colsum = OpFAdd %v3float %cola %colb
%edgeColor = OpVectorTimesScalar %v3float %colsum %float_half
%outPos0 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %outPos0 %pa_plus
OpStore %outColor %edgeColor
OpEmitVertex
%outPos1 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %outPos1 %pa_minus
OpStore %outColor %edgeColor
OpEmitVertex
%outPos2 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %outPos2 %pb_plus
OpStore %outColor %edgeColor
OpEmitVertex
%outPos3 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %outPos3 %pb_minus
OpStore %outColor %edgeColor
OpEmitVertex
OpEndPrimitive
OpReturn
OpFunctionEnd
%main = OpFunction %void None %fn_void
%main_label = OpLabel
OpFunctionCall %void %emitEdge %int_0 %int_2
OpFunctionCall %void %emitEdge %int_2 %int_4
OpFunctionCall %void %emitEdge %int_4 %int_0
OpReturn
OpFunctionEnd
