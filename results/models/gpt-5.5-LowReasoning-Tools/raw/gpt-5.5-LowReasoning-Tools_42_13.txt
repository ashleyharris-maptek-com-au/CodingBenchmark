OpCapability Geometry
%GLSL450 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %in_worldPos %in_normal %in_color %gl_out %outColor
OpExecutionMode %main InputTrianglesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 12
OpExecutionMode %main Invocations 1
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %in_worldPos Location 0
OpDecorate %in_normal Location 1
OpDecorate %in_color Location 2
OpDecorate %outColor Location 0
%void = OpTypeVoid
%fn_void = OpTypeFunction %void
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_6 = OpConstant %uint 6
%int_0 = OpConstant %int 0
%int_2 = OpConstant %int 2
%int_4 = OpConstant %int 4
%float_0_004 = OpConstant %float 0.004
%float_0_5 = OpConstant %float 0.5
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_6
%arr_v3_in = OpTypeArray %v3float %uint_6
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%fn_emit = OpTypeFunction %void %v4float %v4float %v3float
%gl_in = OpVariable %ptr_in_arr_pv Input
%in_worldPos = OpVariable %ptr_in_arr_v3 Input
%in_normal = OpVariable %ptr_in_arr_v3 Input
%in_color = OpVariable %ptr_in_arr_v3 Input
%gl_out = OpVariable %ptr_out_pv Output
%outColor = OpVariable %ptr_out_v3 Output
%emitEdge = OpFunction %void None %fn_emit
%p0 = OpFunctionParameter %v4float
%p1 = OpFunctionParameter %v4float
%edgeColor = OpFunctionParameter %v3float
%emit_entry = OpLabel
%x0 = OpCompositeExtract %float %p0 0
%y0 = OpCompositeExtract %float %p0 1
%z0 = OpCompositeExtract %float %p0 2
%w0 = OpCompositeExtract %float %p0 3
%x1 = OpCompositeExtract %float %p1 0
%y1 = OpCompositeExtract %float %p1 1
%z1 = OpCompositeExtract %float %p1 2
%w1 = OpCompositeExtract %float %p1 3
%ndc0x = OpFDiv %float %x0 %w0
%ndc0y = OpFDiv %float %y0 %w0
%ndc1x = OpFDiv %float %x1 %w1
%ndc1y = OpFDiv %float %y1 %w1
%ndc0 = OpCompositeConstruct %v2float %ndc0x %ndc0y
%ndc1 = OpCompositeConstruct %v2float %ndc1x %ndc1y
%edge = OpFSub %v2float %ndc1 %ndc0
%dir = OpExtInst %v2float %GLSL450 Normalize %edge
%dirx = OpCompositeExtract %float %dir 0
%diry = OpCompositeExtract %float %dir 1
%negdiry = OpFNegate %float %diry
%perpUnit = OpCompositeConstruct %v2float %negdiry %dirx
%perp = OpVectorTimesScalar %v2float %perpUnit %float_0_004
%off0 = OpVectorTimesScalar %v2float %perp %w0
%off1 = OpVectorTimesScalar %v2float %perp %w1
%off0x = OpCompositeExtract %float %off0 0
%off0y = OpCompositeExtract %float %off0 1
%off1x = OpCompositeExtract %float %off1 0
%off1y = OpCompositeExtract %float %off1 1
%p0px = OpFAdd %float %x0 %off0x
%p0py = OpFAdd %float %y0 %off0y
%p0nx = OpFSub %float %x0 %off0x
%p0ny = OpFSub %float %y0 %off0y
%p1px = OpFAdd %float %x1 %off1x
%p1py = OpFAdd %float %y1 %off1y
%p1nx = OpFSub %float %x1 %off1x
%p1ny = OpFSub %float %y1 %off1y
%v0p = OpCompositeConstruct %v4float %p0px %p0py %z0 %w0
%v0n = OpCompositeConstruct %v4float %p0nx %p0ny %z0 %w0
%v1p = OpCompositeConstruct %v4float %p1px %p1py %z1 %w1
%v1n = OpCompositeConstruct %v4float %p1nx %p1ny %z1 %w1
%out_pos_0 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_0 %v0p
OpStore %outColor %edgeColor
OpEmitVertex
%out_pos_1 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_1 %v0n
OpStore %outColor %edgeColor
OpEmitVertex
%out_pos_2 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_2 %v1p
OpStore %outColor %edgeColor
OpEmitVertex
%out_pos_3 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_3 %v1n
OpStore %outColor %edgeColor
OpEmitVertex
OpEndPrimitive
OpReturn
OpFunctionEnd
%main = OpFunction %void None %fn_void
%main_entry = OpLabel
%pos0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%pos2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%pos4_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_4 %int_0
%pos0 = OpLoad %v4float %pos0_ptr
%pos2 = OpLoad %v4float %pos2_ptr
%pos4 = OpLoad %v4float %pos4_ptr
%col0_ptr = OpAccessChain %ptr_in_v3 %in_color %int_0
%col2_ptr = OpAccessChain %ptr_in_v3 %in_color %int_2
%col4_ptr = OpAccessChain %ptr_in_v3 %in_color %int_4
%col0 = OpLoad %v3float %col0_ptr
%col2 = OpLoad %v3float %col2_ptr
%col4 = OpLoad %v3float %col4_ptr
%sum02 = OpFAdd %v3float %col0 %col2
%edge_col_02 = OpVectorTimesScalar %v3float %sum02 %float_0_5
%sum24 = OpFAdd %v3float %col2 %col4
%edge_col_24 = OpVectorTimesScalar %v3float %sum24 %float_0_5
%sum40 = OpFAdd %v3float %col4 %col0
%edge_col_40 = OpVectorTimesScalar %v3float %sum40 %float_0_5
%call02 = OpFunctionCall %void %emitEdge %pos0 %pos2 %edge_col_02
%call24 = OpFunctionCall %void %emitEdge %pos2 %pos4 %edge_col_24
%call40 = OpFunctionCall %void %emitEdge %pos4 %pos0 %edge_col_40
OpReturn
OpFunctionEnd
