#include <algorithm>
#include <cstdint>
#include <iostream>
#include <numeric>
#include <queue>
#include <vector>

using namespace std;

vector<int> solveBipartiteComponent(const vector<int>& comp, const vector<vector<int>>& g, const vector<int>& id, const vector<int>& side) {
    int sz = (int)comp.size();
    vector<int> lefts;
    for (int v : comp) {
        if (side[v] == 0) lefts.push_back(id[v]);
    }

    vector<int> leftIndex(sz, -1);
    for (int i = 0; i < (int)lefts.size(); ++i) leftIndex[lefts[i]] = i;

    vector<vector<int>> hk(lefts.size());
    for (int li = 0; li < (int)lefts.size(); ++li) {
        int uLocal = lefts[li];
        int u = comp[uLocal];
        for (int v : g[u]) {
            int vLocal = id[v];
            if (vLocal >= 0 && side[v] == 1) hk[li].push_back(vLocal);
        }
    }

    vector<int> pairL(lefts.size(), -1), pairR(sz, -1), dist(lefts.size());

    auto bfs = [&]() {
        queue<int> q;
        bool found = false;
        for (int i = 0; i < (int)lefts.size(); ++i) {
            if (pairL[i] == -1) {
                dist[i] = 0;
                q.push(i);
            } else {
                dist[i] = -1;
            }
        }
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (int r : hk[u]) {
                int mate = pairR[r];
                if (mate == -1) {
                    found = true;
                } else if (dist[mate] == -1) {
                    dist[mate] = dist[u] + 1;
                    q.push(mate);
                }
            }
        }
        return found;
    };

    auto dfs = [&](auto&& self, int u) -> bool {
        for (int r : hk[u]) {
            int mate = pairR[r];
            if (mate == -1 || (dist[mate] == dist[u] + 1 && self(self, mate))) {
                pairL[u] = r;
                pairR[r] = u;
                return true;
            }
        }
        dist[u] = -1;
        return false;
    };

    while (bfs()) {
        for (int i = 0; i < (int)lefts.size(); ++i) {
            if (pairL[i] == -1) dfs(dfs, i);
        }
    }

    vector<char> visL(lefts.size(), 0), visR(sz, 0);
    queue<int> q;
    for (int i = 0; i < (int)lefts.size(); ++i) {
        if (pairL[i] == -1) {
            visL[i] = 1;
            q.push(i);
        }
    }
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (int r : hk[u]) {
            if (pairL[u] == r || visR[r]) continue;
            visR[r] = 1;
            int mate = pairR[r];
            if (mate != -1 && !visL[mate]) {
                visL[mate] = 1;
                q.push(mate);
            }
        }
    }

    vector<int> result;
    for (int local = 0; local < sz; ++local) {
        bool inMinVertexCover;
        if (side[comp[local]] == 0) {
            inMinVertexCover = !visL[leftIndex[local]];
        } else {
            inMinVertexCover = visR[local];
        }
        if (!inMinVertexCover) result.push_back(comp[local]);
    }
    return result;
}

struct Solver {
    int n = 0;
    int words = 0;
    uint64_t lastMask = ~0ULL;
    vector<vector<uint64_t>> adj;       // original graph adjacency, remapped to this component
    vector<int> cur, best;

    static int popcount64(uint64_t x) {
        int c = 0;
        while (x) {
            x &= x - 1;
            ++c;
        }
        return c;
    }

    static int ctz64(uint64_t x) {
        static const int index64[64] = {
            0, 1, 48, 2, 57, 49, 28, 3,
            61, 58, 50, 42, 38, 29, 17, 4,
            62, 55, 59, 36, 53, 51, 43, 22,
            45, 39, 33, 30, 24, 18, 12, 5,
            63, 47, 56, 27, 60, 41, 37, 16,
            54, 35, 52, 21, 44, 32, 23, 11,
            46, 26, 40, 15, 34, 20, 31, 10,
            25, 14, 19, 9, 13, 8, 7, 6
        };
        return index64[((x & -x) * 0x03f79d71b4cb0a89ULL) >> 58];
    }

    int popcountSet(const vector<uint64_t>& s) const {
        int r = 0;
        for (uint64_t x : s) r += popcount64(x);
        return r;
    }

    bool emptySet(const vector<uint64_t>& s) const {
        for (uint64_t x : s) if (x) return false;
        return true;
    }

    int firstBit(const vector<uint64_t>& s) const {
        for (int i = 0; i < words; ++i) {
            if (s[i]) return i * 64 + ctz64(s[i]);
        }
        return -1;
    }

    void clearBit(vector<uint64_t>& s, int v) const {
        s[v >> 6] &= ~(1ULL << (v & 63));
    }

    bool testBit(const vector<uint64_t>& s, int v) const {
        return (s[v >> 6] >> (v & 63)) & 1ULL;
    }

    vector<uint64_t> complementNeighborsIn(const vector<uint64_t>& p, int v) const {
        vector<uint64_t> r(words);
        for (int i = 0; i < words; ++i) r[i] = p[i] & ~adj[v][i];
        clearBit(r, v);
        r[words - 1] &= lastMask;
        return r;
    }

    // Greedy coloring of the complement graph. Vertices with color <= c cannot
    // extend the current solution by more than c vertices.
    void colorSort(const vector<uint64_t>& p, vector<int>& order, vector<int>& color) const {
        order.clear();
        color.clear();
        vector<uint64_t> uncolored = p;
        uncolored[words - 1] &= lastMask;

        int c = 0;
        while (!emptySet(uncolored)) {
            ++c;
            vector<uint64_t> q = uncolored;
            while (!emptySet(q)) {
                int v = firstBit(q);
                order.push_back(v);
                color.push_back(c);
                clearBit(uncolored, v);
                clearBit(q, v);

                // In complement-graph coloring, one color class is an independent
                // set in the complement, i.e. a clique in the original graph.
                for (int i = 0; i < words; ++i) q[i] &= adj[v][i];
            }
        }
    }

    void expand(vector<uint64_t>& p) {
        if (cur.size() + (size_t)popcountSet(p) <= best.size()) return;

        vector<int> order, colors;
        colorSort(p, order, colors);

        for (int i = (int)order.size() - 1; i >= 0; --i) {
            int v = order[i];
            if (!testBit(p, v)) continue;
            if (cur.size() + (size_t)colors[i] <= best.size()) return;

            cur.push_back(v);
            vector<uint64_t> next = complementNeighborsIn(p, v);

            if (emptySet(next)) {
                if (cur.size() > best.size()) best = cur;
            } else {
                expand(next);
            }

            cur.pop_back();
            clearBit(p, v);
        }
    }

    vector<int> solve(const vector<int>& verts, const vector<pair<int, int>>& edges) {
        n = (int)verts.size();
        words = (n + 63) / 64;
        lastMask = (n & 63) ? ((1ULL << (n & 63)) - 1ULL) : ~0ULL;
        adj.assign(n, vector<uint64_t>(words, 0));
        cur.clear();
        best.clear();

        for (auto [a, b] : edges) {
            adj[a][b >> 6] |= 1ULL << (b & 63);
            adj[b][a >> 6] |= 1ULL << (a & 63);
        }
        for (int i = 0; i < n; ++i) adj[i][i >> 6] |= 1ULL << (i & 63);

        vector<uint64_t> all(words, ~0ULL);
        all[words - 1] &= lastMask;
        expand(all);

        vector<int> result;
        result.reserve(best.size());
        for (int v : best) result.push_back(verts[v]);
        return result;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    if (!(cin >> n >> m)) return 0;

    vector<pair<int, int>> inputEdges;
    inputEdges.reserve(m);
    vector<vector<int>> g(n);

    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        if (u < 0 || v < 0 || u >= n || v >= n || u == v) continue;
        inputEdges.push_back({u, v});
        g[u].push_back(v);
        g[v].push_back(u);
    }

    vector<char> seen(n, 0);
    vector<int> id(n, -1);
    vector<int> side(n, -1);
    vector<int> answer;

    for (int s = 0; s < n; ++s) {
        if (seen[s]) continue;

        vector<int> comp;
        vector<int> stack = {s};
        seen[s] = 1;
        while (!stack.empty()) {
            int v = stack.back();
            stack.pop_back();
            comp.push_back(v);
            for (int to : g[v]) {
                if (!seen[to]) {
                    seen[to] = 1;
                    stack.push_back(to);
                }
            }
        }

        if (comp.size() == 1) {
            answer.push_back(comp[0]);
            continue;
        }

        for (int i = 0; i < (int)comp.size(); ++i) id[comp[i]] = i;

        bool bipartite = true;
        queue<int> colorQ;
        side[comp[0]] = 0;
        colorQ.push(comp[0]);
        while (!colorQ.empty() && bipartite) {
            int u = colorQ.front();
            colorQ.pop();
            for (int v : g[u]) {
                if (id[v] == -1) continue;
                if (side[v] == -1) {
                    side[v] = side[u] ^ 1;
                    colorQ.push(v);
                } else if (side[v] == side[u]) {
                    bipartite = false;
                    break;
                }
            }
        }

        if (bipartite) {
            vector<int> part = solveBipartiteComponent(comp, g, id, side);
            answer.insert(answer.end(), part.begin(), part.end());
            continue;
        }

        vector<pair<int, int>> localEdges;
        for (int u : comp) {
            for (int v : g[u]) {
                if (u < v && id[v] != -1) localEdges.push_back({id[u], id[v]});
            }
        }

        Solver solver;
        vector<int> part = solver.solve(comp, localEdges);
        answer.insert(answer.end(), part.begin(), part.end());
    }

    cout << answer.size() << '\n';
    for (size_t i = 0; i < answer.size(); ++i) {
        if (i) cout << ' ';
        cout << answer[i];
    }
    cout << '\n';
    return 0;
}
