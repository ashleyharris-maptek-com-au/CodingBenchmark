#include <algorithm>
#include <cstdint>
#include <iostream>
#include <numeric>
#include <queue>
#include <unordered_set>
#include <vector>
#ifdef _MSC_VER
#include <intrin.h>
#endif

using namespace std;

struct MaxClique {
    int n = 0, W = 0;
    vector<vector<uint64_t>> adj;
    vector<int> best, cur;

    static int popcount64(uint64_t x) {
#ifdef _MSC_VER
        return (int)__popcnt64(x);
#else
        return __builtin_popcountll(x);
#endif
    }

    static int ctz64(uint64_t x) {
#ifdef _MSC_VER
        unsigned long idx;
        _BitScanForward64(&idx, x);
        return (int)idx;
#else
        return __builtin_ctzll(x);
#endif
    }

    MaxClique(const vector<vector<uint64_t>>& g) : n((int)g.size()), W((n + 63) >> 6), adj(g) {}

    int count_bits(const vector<uint64_t>& s) const {
        int r = 0;
        for (uint64_t x : s) r += popcount64(x);
        return r;
    }

    void color_sort(const vector<uint64_t>& P, vector<int>& verts, vector<int>& colors) const {
        verts.clear();
        colors.clear();
        vector<uint64_t> U = P;
        int left = count_bits(U);
        for (int color = 1; left > 0; ++color) {
            vector<uint64_t> avail = U;
            while (true) {
                int v = -1;
                for (int i = 0; i < W && v < 0; ++i) {
                    uint64_t x = avail[i];
                    if (x) v = (i << 6) + ctz64(x);
                }
                if (v < 0 || v >= n) break;
                verts.push_back(v);
                colors.push_back(color);
                U[v >> 6] &= ~(1ULL << (v & 63));
                --left;
                for (int i = 0; i < W; ++i) avail[i] &= ~adj[v][i];
                avail[v >> 6] &= ~(1ULL << (v & 63));
            }
        }
    }

    void expand(vector<uint64_t>& P) {
        if (cur.size() + count_bits(P) <= best.size()) return;

        vector<int> verts, colors;
        color_sort(P, verts, colors);

        for (int i = (int)verts.size() - 1; i >= 0; --i) {
            if (cur.size() + colors[i] <= best.size()) return;
            int v = verts[i];
            if ((P[v >> 6] & (1ULL << (v & 63))) == 0) continue;

            cur.push_back(v);
            vector<uint64_t> Q(W);
            bool any = false;
            for (int j = 0; j < W; ++j) {
                Q[j] = P[j] & adj[v][j];
                any |= Q[j] != 0;
            }
            if (!any) {
                if (cur.size() > best.size()) best = cur;
            } else {
                expand(Q);
            }
            cur.pop_back();
            P[v >> 6] &= ~(1ULL << (v & 63));
        }
    }

    vector<int> solve() {
        vector<uint64_t> P(W, ~0ULL);
        if (W) {
            int extra = W * 64 - n;
            if (extra) P.back() >>= extra;
        }
        expand(P);
        return best;
    }
};

vector<int> bipartite_mis(const vector<int>& comp, const vector<vector<int>>& g, const vector<int>& side) {
    vector<int> L, R;
    int n = (int)g.size();
    vector<int> li(n, -1), ri(n, -1);
    for (int v : comp) {
        if (side[v] == 0) {
            li[v] = (int)L.size();
            L.push_back(v);
        } else {
            ri[v] = (int)R.size();
            R.push_back(v);
        }
    }

    vector<vector<int>> bg(L.size());
    for (int i = 0; i < (int)L.size(); ++i) {
        int v = L[i];
        for (int to : g[v]) if (ri[to] >= 0) bg[i].push_back(ri[to]);
    }

    vector<int> ml(L.size(), -1), mr(R.size(), -1), dist(L.size());
    auto bfs = [&]() {
        queue<int> q;
        bool found = false;
        for (int i = 0; i < (int)L.size(); ++i) {
            if (ml[i] < 0) dist[i] = 0, q.push(i);
            else dist[i] = -1;
        }
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            for (int r : bg[v]) {
                int u = mr[r];
                if (u < 0) found = true;
                else if (dist[u] < 0) dist[u] = dist[v] + 1, q.push(u);
            }
        }
        return found;
    };
    auto dfs = [&](auto&& self, int v) -> bool {
        for (int r : bg[v]) {
            int u = mr[r];
            if (u < 0 || (dist[u] == dist[v] + 1 && self(self, u))) {
                ml[v] = r;
                mr[r] = v;
                return true;
            }
        }
        dist[v] = -1;
        return false;
    };
    while (bfs()) {
        for (int i = 0; i < (int)L.size(); ++i) {
            if (ml[i] < 0) dfs(dfs, i);
        }
    }

    vector<char> visL(L.size(), 0), visR(R.size(), 0);
    queue<int> q;
    for (int i = 0; i < (int)L.size(); ++i) if (ml[i] < 0) visL[i] = 1, q.push(i);
    while (!q.empty()) {
        int v = q.front();
        q.pop();
        for (int r : bg[v]) {
            if (ml[v] == r || visR[r]) continue;
            visR[r] = 1;
            if (mr[r] >= 0 && !visL[mr[r]]) visL[mr[r]] = 1, q.push(mr[r]);
        }
    }

    vector<int> ans;
    for (int i = 0; i < (int)L.size(); ++i) if (visL[i]) ans.push_back(L[i]);
    for (int i = 0; i < (int)R.size(); ++i) if (!visR[i]) ans.push_back(R[i]);
    return ans;
}

vector<int> exact_component_mis(const vector<int>& comp, const vector<vector<int>>& g) {
    int s = (int)comp.size();
    if (s == 1) return comp;

    vector<int> id(g.size(), -1);
    for (int i = 0; i < s; ++i) id[comp[i]] = i;

    int W = (s + 63) >> 6;
    vector<vector<uint64_t>> cg(s, vector<uint64_t>(W, ~0ULL));
    if (W) {
        int extra = W * 64 - s;
        if (extra) for (auto& row : cg) row.back() >>= extra;
    }
    for (int i = 0; i < s; ++i) cg[i][i >> 6] &= ~(1ULL << (i & 63));
    for (int v : comp) {
        int a = id[v];
        for (int to : g[v]) {
            int b = id[to];
            if (b >= 0) cg[a][b >> 6] &= ~(1ULL << (b & 63));
        }
    }

    MaxClique mc(cg);
    vector<int> local = mc.solve(), ans;
    ans.reserve(local.size());
    for (int x : local) ans.push_back(comp[x]);
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    if (!(cin >> n >> m)) return 0;
    vector<vector<int>> g(n);
    g.reserve(n);
    unordered_set<uint64_t> seen;
    seen.reserve((size_t)m * 2 + 1);
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        if (u < 0 || v < 0 || u >= n || v >= n || u == v) continue;
        int a = min(u, v), b = max(u, v);
        uint64_t key = (uint64_t)(uint32_t)a << 32 | (uint32_t)b;
        if (seen.insert(key).second) {
            g[u].push_back(v);
            g[v].push_back(u);
        }
    }

    vector<char> used(n, 0);
    vector<int> side(n, -1);
    vector<int> answer;
    for (int st = 0; st < n; ++st) {
        if (used[st]) continue;
        vector<int> comp;
        queue<int> q;
        used[st] = 1;
        side[st] = 0;
        q.push(st);
        bool bip = true;
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            comp.push_back(v);
            for (int to : g[v]) {
                if (side[to] < 0) side[to] = side[v] ^ 1;
                else if (side[to] == side[v]) bip = false;
                if (!used[to]) {
                    used[to] = 1;
                    q.push(to);
                }
            }
        }

        vector<int> part = bip ? bipartite_mis(comp, g, side) : exact_component_mis(comp, g);
        answer.insert(answer.end(), part.begin(), part.end());
    }

    cout << answer.size() << '\n';
    for (size_t i = 0; i < answer.size(); ++i) {
        if (i) cout << ' ';
        cout << answer[i];
    }
    cout << '\n';
    return 0;
}
