#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

int n, m;
vector<int64> w;

struct MaskAns {
    int64 val;
    unsigned long long take;
};

vector<unsigned long long> adj64;
unordered_map<unsigned long long, MaskAns> memo64;

static int pop_lsb(unsigned long long &x) {
    int b = __builtin_ctzll(x);
    x &= x - 1;
    return b;
}

MaskAns solve64(unsigned long long mask) {
    mask &= (n == 64 ? ~0ULL : ((1ULL << n) - 1));
    unsigned long long t = mask;
    while (t) {
        int v = __builtin_ctzll(t);
        if (w[v] <= 0) mask &= ~(1ULL << v);
        t &= t - 1;
    }
    if (!mask) return {0, 0};

    auto it = memo64.find(mask);
    if (it != memo64.end()) return it->second;

    unsigned long long rest = mask;
    vector<unsigned long long> comps;
    while (rest) {
        int s = __builtin_ctzll(rest);
        unsigned long long comp = 0, frontier = 1ULL << s;
        rest &= ~frontier;
        while (frontier) {
            unsigned long long f = frontier;
            frontier = 0;
            while (f) {
                int v = pop_lsb(f);
                comp |= 1ULL << v;
                unsigned long long nb = adj64[v] & rest;
                frontier |= nb;
                rest &= ~nb;
            }
        }
        comps.push_back(comp);
    }
    if (comps.size() > 1) {
        MaskAns ans{0, 0};
        for (auto c : comps) {
            MaskAns part = solve64(c);
            ans.val += part.val;
            ans.take |= part.take;
        }
        memo64[mask] = ans;
        return ans;
    }

    bool hasEdge = false;
    for (unsigned long long x = mask; x; ) {
        int v = pop_lsb(x);
        if (adj64[v] & x) {
            hasEdge = true;
            break;
        }
    }
    if (!hasEdge) {
        int64 val = 0;
        for (unsigned long long x = mask; x; ) {
            int v = pop_lsb(x);
            val += w[v];
        }
        MaskAns ans{val, mask};
        memo64[mask] = ans;
        return ans;
    }

    int bestV = -1, bestDeg = -1;
    int64 bestW = LLONG_MIN;
    for (unsigned long long x = mask; x; ) {
        int v = pop_lsb(x);
        int deg = __builtin_popcountll(adj64[v] & mask);
        if (deg > bestDeg || (deg == bestDeg && w[v] > bestW)) {
            bestV = v;
            bestDeg = deg;
            bestW = w[v];
        }
    }

    unsigned long long bit = 1ULL << bestV;
    MaskAns in = solve64(mask & ~adj64[bestV] & ~bit);
    in.val += w[bestV];
    in.take |= bit;

    MaskAns out = solve64(mask & ~bit);
    MaskAns ans = (in.val >= out.val ? in : out);
    memo64[mask] = ans;
    return ans;
}

vector<vector<unsigned long long>> adjBits;

static bool has_edge_big(int a, int b) {
    return (adjBits[a][b >> 6] >> (b & 63)) & 1ULL;
}

struct BigAns {
    int64 val;
    vector<int> take;
};

BigAns solveBig(vector<int> cand);

static vector<vector<int>> componentsBig(const vector<int> &cand) {
    int k = (int)cand.size();
    vector<char> seen(k, 0);
    vector<vector<int>> comps;
    for (int i = 0; i < k; ++i) if (!seen[i]) {
        vector<int> comp;
        queue<int> q;
        q.push(i);
        seen[i] = 1;
        while (!q.empty()) {
            int a = q.front();
            q.pop();
            int va = cand[a];
            comp.push_back(va);
            for (int j = 0; j < k; ++j) {
                if (!seen[j] && has_edge_big(va, cand[j])) {
                    seen[j] = 1;
                    q.push(j);
                }
            }
        }
        comps.push_back(comp);
    }
    return comps;
}

BigAns solveBig(vector<int> cand) {
    vector<int> f;
    f.reserve(cand.size());
    for (int v : cand) if (w[v] > 0) f.push_back(v);
    cand.swap(f);
    if (cand.empty()) return {0, {}};

    auto comps = componentsBig(cand);
    if (comps.size() > 1) {
        BigAns ans{0, {}};
        for (auto &c : comps) {
            BigAns p = solveBig(c);
            ans.val += p.val;
            ans.take.insert(ans.take.end(), p.take.begin(), p.take.end());
        }
        return ans;
    }

    bool hasEdge = false;
    for (int i = 0; i < (int)cand.size() && !hasEdge; ++i)
        for (int j = i + 1; j < (int)cand.size(); ++j)
            if (has_edge_big(cand[i], cand[j])) {
                hasEdge = true;
                break;
            }
    if (!hasEdge) {
        BigAns ans{0, cand};
        for (int v : cand) ans.val += w[v];
        return ans;
    }

    int bestV = cand[0], bestDeg = -1;
    int64 bestW = LLONG_MIN;
    for (int v : cand) {
        int deg = 0;
        for (int u : cand) if (u != v && has_edge_big(v, u)) ++deg;
        if (deg > bestDeg || (deg == bestDeg && w[v] > bestW)) {
            bestV = v;
            bestDeg = deg;
            bestW = w[v];
        }
    }

    vector<int> inCand, outCand;
    for (int u : cand) {
        if (u == bestV) continue;
        outCand.push_back(u);
        if (!has_edge_big(bestV, u)) inCand.push_back(u);
    }
    BigAns in = solveBig(inCand);
    in.val += w[bestV];
    in.take.push_back(bestV);
    BigAns out = solveBig(outCand);
    return in.val >= out.val ? in : out;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    if (!(cin >> n >> m)) return 0;
    w.assign(n, 0);
    for (int i = 0; i < n; ++i) cin >> w[i];

    vector<pair<int,int>> edges;
    edges.reserve(m);
    vector<char> forcedZero(n, 0);
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        if (u >= 0 && u < n && v >= 0 && v < n) {
            if (u == v) forcedZero[u] = 1;
            else edges.push_back({u, v});
        }
    }
    for (int i = 0; i < n; ++i)
        if (forcedZero[i]) w[i] = LLONG_MIN / 4;

    vector<int> x(n, 0);
    int64 opt = 0;

    if (n <= 63) {
        adj64.assign(n, 0);
        for (auto [u, v] : edges) {
            adj64[u] |= 1ULL << v;
            adj64[v] |= 1ULL << u;
        }
        unsigned long long full = (n == 64 ? ~0ULL : ((1ULL << n) - 1));
        MaskAns ans = solve64(full);
        opt = ans.val;
        for (int i = 0; i < n; ++i) x[i] = (ans.take >> i) & 1ULL;
    } else {
        int words = (n + 63) >> 6;
        adjBits.assign(n, vector<unsigned long long>(words, 0));
        for (auto [u, v] : edges) {
            adjBits[u][v >> 6] |= 1ULL << (v & 63);
            adjBits[v][u >> 6] |= 1ULL << (u & 63);
        }
        vector<int> all(n);
        iota(all.begin(), all.end(), 0);
        BigAns ans = solveBig(all);
        opt = ans.val;
        for (int v : ans.take) x[v] = 1;
    }

    cout << opt << '\n';
    for (int i = 0; i < n; ++i) {
        if (i) cout << ' ';
        cout << x[i];
    }
    cout << '\n';
    return 0;
}
