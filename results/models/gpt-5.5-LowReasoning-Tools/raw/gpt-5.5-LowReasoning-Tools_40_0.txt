#include <bits/stdc++.h>
using namespace std;

struct Dinic {
    struct Edge {
        int to, rev;
        long long cap;
    };
    int n;
    vector<vector<Edge>> g;
    vector<int> level, it;

    Dinic(int n = 0) : n(n), g(n), level(n), it(n) {}

    void addEdge(int v, int to, long long cap) {
        Edge a{to, (int)g[to].size(), cap};
        Edge b{v, (int)g[v].size(), 0};
        g[v].push_back(a);
        g[to].push_back(b);
    }

    bool bfs(int s, int t) {
        fill(level.begin(), level.end(), -1);
        queue<int> q;
        level[s] = 0;
        q.push(s);
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            for (const auto &e : g[v]) {
                if (e.cap > 0 && level[e.to] < 0) {
                    level[e.to] = level[v] + 1;
                    q.push(e.to);
                }
            }
        }
        return level[t] >= 0;
    }

    long long dfs(int v, int t, long long f) {
        if (v == t) return f;
        for (int &i = it[v]; i < (int)g[v].size(); ++i) {
            Edge &e = g[v][i];
            if (e.cap <= 0 || level[e.to] != level[v] + 1) continue;
            long long ret = dfs(e.to, t, min(f, e.cap));
            if (ret) {
                e.cap -= ret;
                g[e.to][e.rev].cap += ret;
                return ret;
            }
        }
        return 0;
    }

    long long maxflow(int s, int t) {
        long long flow = 0;
        while (bfs(s, t)) {
            fill(it.begin(), it.end(), 0);
            while (long long f = dfs(s, t, LLONG_MAX / 4)) flow += f;
        }
        return flow;
    }

    vector<int> reachableFrom(int s) const {
        vector<int> seen(n, 0);
        queue<int> q;
        seen[s] = 1;
        q.push(s);
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            for (const auto &e : g[v]) {
                if (e.cap > 0 && !seen[e.to]) {
                    seen[e.to] = 1;
                    q.push(e.to);
                }
            }
        }
        return seen;
    }
};

struct Result {
    long long value = 0;
    vector<int> chosen;
};

int n, m;
vector<long long> w;
vector<vector<int>> adj;
unordered_map<string, Result> memo;

static string keyOf(vector<int> verts) {
    sort(verts.begin(), verts.end());
    string s;
    s.reserve(verts.size() * 4);
    for (int v : verts) {
        s.append(reinterpret_cast<const char*>(&v), sizeof(v));
    }
    return s;
}

static bool hasEdgeInside(const vector<int> &verts) {
    vector<char> in(n, 0);
    for (int v : verts) in[v] = 1;
    for (int v : verts) {
        for (int u : adj[v]) {
            if (in[u]) return true;
        }
    }
    return false;
}

static vector<vector<int>> componentsOf(const vector<int> &verts) {
    vector<char> in(n, 0), seen(n, 0);
    for (int v : verts) in[v] = 1;
    vector<vector<int>> comps;
    for (int s : verts) {
        if (seen[s]) continue;
        comps.push_back({});
        queue<int> q;
        seen[s] = 1;
        q.push(s);
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            comps.back().push_back(v);
            for (int u : adj[v]) {
                if (in[u] && !seen[u]) {
                    seen[u] = 1;
                    q.push(u);
                }
            }
        }
    }
    return comps;
}

static bool bipartition(const vector<int> &verts, vector<int> &color) {
    vector<char> in(n, 0);
    for (int v : verts) in[v] = 1;
    color.assign(n, -1);
    for (int s : verts) {
        if (color[s] != -1) continue;
        color[s] = 0;
        queue<int> q;
        q.push(s);
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            for (int u : adj[v]) {
                if (!in[u]) continue;
                if (color[u] == -1) {
                    color[u] = color[v] ^ 1;
                    q.push(u);
                } else if (color[u] == color[v]) {
                    return false;
                }
            }
        }
    }
    return true;
}

static Result solveSet(vector<int> verts);

static Result solveBipartite(const vector<int> &verts, const vector<int> &color) {
    int S = n, T = n + 1;
    Dinic din(n + 2);
    const long long INF = (1LL << 60);
    long long total = 0;

    vector<char> in(n, 0);
    for (int v : verts) {
        in[v] = 1;
        total += w[v];
        if (color[v] == 0) din.addEdge(S, v, w[v]);
        else din.addEdge(v, T, w[v]);
    }
    for (int v : verts) {
        if (color[v] != 0) continue;
        for (int u : adj[v]) {
            if (in[u] && color[u] == 1) din.addEdge(v, u, INF);
        }
    }

    din.maxflow(S, T);
    vector<int> reach = din.reachableFrom(S);

    Result res;
    res.value = total;
    for (int v : verts) {
        bool inMinCover = (color[v] == 0) ? !reach[v] : reach[v];
        if (inMinCover) res.value -= w[v];
        else res.chosen.push_back(v);
    }
    return res;
}

static Result solveConnectedGeneral(const vector<int> &verts) {
    if (verts.empty()) return {};

    vector<int> positive;
    positive.reserve(verts.size());
    for (int v : verts) {
        if (w[v] > 0) positive.push_back(v);
    }
    if (positive.empty()) return {};

    if (!hasEdgeInside(positive)) {
        Result res;
        for (int v : positive) {
            res.value += w[v];
            res.chosen.push_back(v);
        }
        return res;
    }

    vector<int> color;
    if (bipartition(positive, color)) return solveBipartite(positive, color);

    vector<char> in(n, 0);
    for (int v : positive) in[v] = 1;

    int branch = positive[0];
    long long bestScore = -1;
    for (int v : positive) {
        long long deg = 0, neighWeight = 0;
        for (int u : adj[v]) {
            if (in[u]) {
                ++deg;
                neighWeight += max(0LL, w[u]);
            }
        }
        long long score = deg * 1000000LL + neighWeight;
        if (score > bestScore) {
            bestScore = score;
            branch = v;
        }
    }

    vector<int> excl;
    excl.reserve(positive.size() - 1);
    for (int v : positive) {
        if (v != branch) excl.push_back(v);
    }
    Result best = solveSet(excl);

    vector<char> banned(n, 0);
    banned[branch] = 1;
    for (int u : adj[branch]) banned[u] = 1;
    vector<int> incl;
    incl.reserve(positive.size());
    for (int v : positive) {
        if (!banned[v]) incl.push_back(v);
    }
    Result with = solveSet(incl);
    with.value += w[branch];
    with.chosen.push_back(branch);

    if (with.value > best.value) return with;
    return best;
}

static Result solveSet(vector<int> verts) {
    vector<int> filtered;
    filtered.reserve(verts.size());
    for (int v : verts) {
        if (w[v] > 0) filtered.push_back(v);
    }
    sort(filtered.begin(), filtered.end());
    filtered.erase(unique(filtered.begin(), filtered.end()), filtered.end());
    if (filtered.empty()) return {};

    string key = keyOf(filtered);
    auto it = memo.find(key);
    if (it != memo.end()) return it->second;

    vector<vector<int>> comps = componentsOf(filtered);
    Result total;
    if (comps.size() == 1) {
        total = solveConnectedGeneral(filtered);
    } else {
        for (auto &comp : comps) {
            Result part = solveSet(comp);
            total.value += part.value;
            total.chosen.insert(total.chosen.end(), part.chosen.begin(), part.chosen.end());
        }
    }

    memo.emplace(move(key), total);
    return total;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    if (!(cin >> n >> m)) return 0;
    w.assign(n, 0);
    for (int i = 0; i < n; ++i) cin >> w[i];

    adj.assign(n, {});
    vector<pair<int, int>> edges;
    edges.reserve(m);
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        if (u == v) {
            if (0 <= u && u < n) w[u] = min(w[u], 0LL);
            continue;
        }
        if (0 <= u && u < n && 0 <= v && v < n) {
            adj[u].push_back(v);
            adj[v].push_back(u);
            edges.push_back({u, v});
        }
    }

    for (auto &a : adj) {
        sort(a.begin(), a.end());
        a.erase(unique(a.begin(), a.end()), a.end());
    }

    vector<int> all(n);
    iota(all.begin(), all.end(), 0);
    Result ans = solveSet(all);

    vector<int> x(n, 0);
    for (int v : ans.chosen) x[v] = 1;

    long long objective = 0;
    for (int i = 0; i < n; ++i) {
        if (x[i]) objective += w[i];
    }

    cout << objective << '\n';
    for (int i = 0; i < n; ++i) {
        if (i) cout << ' ';
        cout << x[i];
    }
    cout << '\n';
    return 0;
}
