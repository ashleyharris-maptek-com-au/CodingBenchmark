; SPIR-V
; Version: 1.0
OpCapability Geometry
%glsl = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main InputLines
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 7
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2
OpDecorate %outColor Location 0

%void = OpTypeVoid
%func = OpTypeFunction %void
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%uint_2 = OpConstant %uint 2
%float_0 = OpConstant %float 0
%float_1 = OpConstant %float 1
%float_n1 = OpConstant %float -1
%float_0_015 = OpConstant %float 0.015
%float_0_05 = OpConstant %float 0.05
%float_0_75 = OpConstant %float 0.75

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%ptr_in_v4 = OpTypePointer Input %v4float
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3_in = OpTypeArray %v3float %uint_2
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%ptr_in_v3 = OpTypePointer Input %v3float
%worldPos = OpVariable %ptr_in_arr_v3 Input
%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%gl_out = OpVariable %ptr_out_pv Output
%outColor = OpVariable %ptr_out_v3 Output

%main = OpFunction %void None %func
%entry = OpLabel
%p0_ptr = OpAccessChain %ptr_in_v3 %worldPos %int_0
%p1_ptr = OpAccessChain %ptr_in_v3 %worldPos %int_1
%c0_ptr = OpAccessChain %ptr_in_v3 %color %int_0
%c1_ptr = OpAccessChain %ptr_in_v3 %color %int_1
%p0 = OpLoad %v3float %p0_ptr
%p1 = OpLoad %v3float %p1_ptr
%c0 = OpLoad %v3float %c0_ptr
%c1 = OpLoad %v3float %c1_ptr

%dir = OpFSub %v3float %p1 %p0
%dx = OpCompositeExtract %float %dir 0
%dy = OpCompositeExtract %float %dir 1
%neg_dy = OpFNegate %float %dy
%perp_raw = OpCompositeConstruct %v2float %neg_dy %dx
%perp = OpExtInst %v2float %glsl Normalize %perp_raw
%perp_x = OpCompositeExtract %float %perp 0
%perp_y = OpCompositeExtract %float %perp 1

%mid_delta = OpVectorTimesScalar %v3float %dir %float_0_75
%mid = OpFAdd %v3float %p0 %mid_delta

%shaft_px = OpFMul %float %perp_x %float_0_015
%shaft_py = OpFMul %float %perp_y %float_0_015
%shaft_off = OpCompositeConstruct %v3float %shaft_px %shaft_py %float_0
%arrow_px = OpFMul %float %perp_x %float_0_05
%arrow_py = OpFMul %float %perp_y %float_0_05
%arrow_off = OpCompositeConstruct %v3float %arrow_px %arrow_py %float_0

%p0_plus = OpFAdd %v3float %p0 %shaft_off
%p0_minus = OpFSub %v3float %p0 %shaft_off
%mid_plus = OpFAdd %v3float %mid %shaft_off
%mid_minus = OpFSub %v3float %mid %shaft_off
%head_plus = OpFAdd %v3float %mid %arrow_off
%head_minus = OpFSub %v3float %mid %arrow_off

%out_pos_ptr_0 = OpAccessChain %ptr_out_v4 %gl_out %int_0

%p0_plus_x = OpCompositeExtract %float %p0_plus 0
%p0_plus_y = OpCompositeExtract %float %p0_plus 1
%p0_plus_z = OpCompositeExtract %float %p0_plus 2
%p0_plus_v4 = OpCompositeConstruct %v4float %p0_plus_x %p0_plus_y %p0_plus_z %float_1
OpStore %out_pos_ptr_0 %p0_plus_v4
OpStore %outColor %c0
OpEmitVertex

%p0_minus_x = OpCompositeExtract %float %p0_minus 0
%p0_minus_y = OpCompositeExtract %float %p0_minus 1
%p0_minus_z = OpCompositeExtract %float %p0_minus 2
%p0_minus_v4 = OpCompositeConstruct %v4float %p0_minus_x %p0_minus_y %p0_minus_z %float_1
OpStore %out_pos_ptr_0 %p0_minus_v4
OpStore %outColor %c0
OpEmitVertex

%mid_plus_x = OpCompositeExtract %float %mid_plus 0
%mid_plus_y = OpCompositeExtract %float %mid_plus 1
%mid_plus_z = OpCompositeExtract %float %mid_plus 2
%mid_plus_v4 = OpCompositeConstruct %v4float %mid_plus_x %mid_plus_y %mid_plus_z %float_1
OpStore %out_pos_ptr_0 %mid_plus_v4
OpStore %outColor %c0
OpEmitVertex

%mid_minus_x = OpCompositeExtract %float %mid_minus 0
%mid_minus_y = OpCompositeExtract %float %mid_minus 1
%mid_minus_z = OpCompositeExtract %float %mid_minus 2
%mid_minus_v4 = OpCompositeConstruct %v4float %mid_minus_x %mid_minus_y %mid_minus_z %float_1
OpStore %out_pos_ptr_0 %mid_minus_v4
OpStore %outColor %c0
OpEmitVertex
OpEndPrimitive

%head_plus_x = OpCompositeExtract %float %head_plus 0
%head_plus_y = OpCompositeExtract %float %head_plus 1
%head_plus_z = OpCompositeExtract %float %head_plus 2
%head_plus_v4 = OpCompositeConstruct %v4float %head_plus_x %head_plus_y %head_plus_z %float_1
OpStore %out_pos_ptr_0 %head_plus_v4
OpStore %outColor %c1
OpEmitVertex

%head_minus_x = OpCompositeExtract %float %head_minus 0
%head_minus_y = OpCompositeExtract %float %head_minus 1
%head_minus_z = OpCompositeExtract %float %head_minus 2
%head_minus_v4 = OpCompositeConstruct %v4float %head_minus_x %head_minus_y %head_minus_z %float_1
OpStore %out_pos_ptr_0 %head_minus_v4
OpStore %outColor %c1
OpEmitVertex

%p1_x = OpCompositeExtract %float %p1 0
%p1_y = OpCompositeExtract %float %p1 1
%p1_z = OpCompositeExtract %float %p1 2
%p1_v4 = OpCompositeConstruct %v4float %p1_x %p1_y %p1_z %float_1
OpStore %out_pos_ptr_0 %p1_v4
OpStore %outColor %c1
OpEmitVertex
OpEndPrimitive

OpReturn
OpFunctionEnd
