using System;
using System.Collections.Generic;
using System.Text;

public static class Program
{
    static List<int>[] G = Array.Empty<List<int>>();
    static int N;

    public static void Main()
    {
        var fs = new FastScanner();
        N = fs.NextInt();
        int m = fs.NextInt();
        if (N < 0) return;
        G = new List<int>[N];
        for (int i = 0; i < N; i++) G[i] = new List<int>();
        for (int i = 0; i < m; i++)
        {
            int u = fs.NextInt(), v = fs.NextInt();
            if ((uint)u < (uint)N && (uint)v < (uint)N) G[u].Add(v);
        }

        var alive = new bool[N];
        Fill(alive, true);
        var ans = SolveExact(alive);
        ans.Sort();

        var sb = new StringBuilder();
        sb.AppendLine(ans.Count.ToString());
        for (int i = 0; i < ans.Count; i++)
        {
            if (i > 0) sb.Append(' ');
            sb.Append(ans[i]);
        }
        sb.AppendLine();
        Console.Write(sb.ToString());
    }

    static List<int> SolveExact(bool[] alive)
    {
        TrimAcyclic(alive);
        var comps = Sccs(alive);
        var result = new List<int>();
        foreach (var c in comps)
        {
            bool self = false;
            if (c.Count == 1)
                foreach (int to in G[c[0]]) if (to == c[0]) self = true;
            if (c.Count == 1 && !self) continue;
            var subAlive = new bool[N];
            foreach (int v in c) subAlive[v] = true;
            var solver = new Solver(subAlive);
            result.AddRange(solver.Run());
        }
        return result;
    }

    static void TrimAcyclic(bool[] alive)
    {
        int[] indeg = new int[N], outdeg = new int[N];
        var q = new Queue<int>();
        for (int u = 0; u < N; u++) if (alive[u])
        {
            foreach (int v in G[u]) if (alive[v])
            {
                outdeg[u]++;
                indeg[v]++;
            }
        }
        for (int i = 0; i < N; i++) if (alive[i] && (indeg[i] == 0 || outdeg[i] == 0)) q.Enqueue(i);
        while (q.Count > 0)
        {
            int u = q.Dequeue();
            if (!alive[u] || (indeg[u] != 0 && outdeg[u] != 0)) continue;
            alive[u] = false;
            foreach (int v in G[u]) if (alive[v])
            {
                if (--indeg[v] == 0) q.Enqueue(v);
            }
            for (int p = 0; p < N; p++) if (alive[p])
            {
                foreach (int v in G[p]) if (v == u)
                {
                    if (--outdeg[p] == 0) q.Enqueue(p);
                }
            }
        }
    }

    static List<List<int>> Sccs(bool[] alive)
    {
        int[] idx = new int[N], low = new int[N];
        bool[] on = new bool[N];
        Fill(idx, -1);
        var st = new Stack<int>();
        var comps = new List<List<int>>();
        int time = 0;

        void Dfs(int u)
        {
            idx[u] = low[u] = time++;
            st.Push(u); on[u] = true;
            foreach (int v in G[u]) if (alive[v])
            {
                if (idx[v] < 0) { Dfs(v); low[u] = Math.Min(low[u], low[v]); }
                else if (on[v]) low[u] = Math.Min(low[u], idx[v]);
            }
            if (low[u] == idx[u])
            {
                var c = new List<int>();
                int x;
                do { x = st.Pop(); on[x] = false; c.Add(x); } while (x != u);
                bool useful = c.Count > 1;
                if (!useful)
                    foreach (int v in G[u]) if (v == u) useful = true;
                if (useful) comps.Add(c);
            }
        }

        for (int i = 0; i < N; i++) if (alive[i] && idx[i] < 0) Dfs(i);
        return comps;
    }

    sealed class Solver
    {
        readonly bool[] rootAlive;
        List<int> best = new List<int>();
        readonly List<int> cur = new List<int>();

        public Solver(bool[] alive)
        {
            rootAlive = alive;
            for (int i = 0; i < N; i++) if (alive[i]) best.Add(i);
        }

        public List<int> Run()
        {
            Search((bool[])rootAlive.Clone());
            return best;
        }

        void Search(bool[] alive)
        {
            if (cur.Count >= best.Count) return;
            TrimAcyclic(alive);
            var comps = Sccs(alive);
            if (comps.Count == 0)
            {
                best = new List<int>(cur);
                return;
            }
            if (cur.Count + GreedyDisjointCycleLowerBound(alive) >= best.Count) return;

            List<int> split = null;
            foreach (var c in comps) if (split == null || c.Count < split.Count) split = c;
            var local = new bool[N];
            foreach (int v in split) local[v] = true;
            var cycle = FindCycle(local);
            if (cycle.Count == 0) return;

            cycle.Sort((a, b) => Score(b, alive).CompareTo(Score(a, alive)));
            foreach (int v in cycle)
            {
                if (!alive[v]) continue;
                var next = (bool[])alive.Clone();
                next[v] = false;
                cur.Add(v);
                Search(next);
                cur.RemoveAt(cur.Count - 1);
            }
        }

        static int Score(int u, bool[] alive)
        {
            int a = 0, b = 0;
            foreach (int v in G[u]) if (alive[v]) a++;
            for (int p = 0; p < N; p++) if (alive[p])
                foreach (int v in G[p]) if (v == u) { b++; break; }
            return a * b;
        }

        static int GreedyDisjointCycleLowerBound(bool[] alive)
        {
            var tmp = (bool[])alive.Clone();
            int lb = 0;
            while (true)
            {
                TrimAcyclic(tmp);
                var cyc = FindCycle(tmp);
                if (cyc.Count == 0) return lb;
                lb++;
                foreach (int v in cyc) tmp[v] = false;
            }
        }

        static List<int> FindCycle(bool[] alive)
        {
            int[] color = new int[N], parent = new int[N];
            Fill(parent, -1);
            var ans = new List<int>();

            bool Dfs(int u)
            {
                color[u] = 1;
                foreach (int v in G[u]) if (alive[v])
                {
                    if (u == v) { ans.Add(u); return true; }
                    if (color[v] == 0)
                    {
                        parent[v] = u;
                        if (Dfs(v)) return true;
                    }
                    else if (color[v] == 1)
                    {
                        ans.Add(v);
                        for (int x = u; x != v && x >= 0; x = parent[x]) ans.Add(x);
                        return true;
                    }
                }
                color[u] = 2;
                return false;
            }

            for (int i = 0; i < N; i++) if (alive[i] && color[i] == 0 && Dfs(i)) break;
            return ans;
        }
    }

    sealed class FastScanner
    {
        readonly byte[] data = Console.OpenStandardInput().ReadAllBytes();
        int p;
        public int NextInt()
        {
            while (p < data.Length && data[p] <= 32) p++;
            if (p >= data.Length) return -1;
            int sign = 1;
            if (data[p] == '-') { sign = -1; p++; }
            int x = 0;
            while (p < data.Length && data[p] > 32) x = x * 10 + data[p++] - '0';
            return x * sign;
        }
    }

    static void Fill<T>(T[] a, T value)
    {
        for (int i = 0; i < a.Length; i++) a[i] = value;
    }
}

static class StreamExt
{
    public static byte[] ReadAllBytes(this System.IO.Stream s)
    {
        using var ms = new System.IO.MemoryStream();
        s.CopyTo(ms);
        return ms.ToArray();
    }
}
