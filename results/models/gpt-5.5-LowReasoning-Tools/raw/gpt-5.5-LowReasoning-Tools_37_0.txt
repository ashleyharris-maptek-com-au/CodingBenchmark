using System;
using System.Collections.Generic;
using System.Text;

public class Program
{
    static int N, M;
    static List<int>[] G = null!, RG = null!;
    static bool[] HasSelf = null!;
    static Dictionary<string, int[]> Memo = new Dictionary<string, int[]>();

    public static void Main()
    {
        var fs = new FastScanner();
        N = fs.NextInt();
        M = fs.NextInt();
        G = new List<int>[N];
        RG = new List<int>[N];
        HasSelf = new bool[N];
        for (int i = 0; i < N; i++) { G[i] = new List<int>(); RG[i] = new List<int>(); }
        for (int i = 0; i < M; i++)
        {
            int u = fs.NextInt(), v = fs.NextInt();
            if (u == v) HasSelf[u] = true;
            G[u].Add(v);
            RG[v].Add(u);
        }

        int[] all = new int[N];
        for (int i = 0; i < N; i++) all[i] = i;
        int[] ans = SolveSet(all);
        Array.Sort(ans);
        Console.WriteLine(ans.Length);
        Console.WriteLine(string.Join(" ", ans));
    }

    static int[] SolveSet(int[] input)
    {
        if (input.Length == 0) return Array.Empty<int>();
        Array.Sort(input);
        string? key = input.Length <= 260 ? MakeKey(input) : null;
        if (key != null && Memo.TryGetValue(key, out var cached)) return cached;

        var forced = new List<int>();
        var vertsList = new List<int>(input.Length);
        foreach (int v in input)
        {
            if (HasSelf[v]) forced.Add(v);
            else vertsList.Add(v);
        }

        int[] verts = TrimAcyclicEnds(vertsList.ToArray());
        var comps = StrongComponents(verts);
        var result = new List<int>(forced);
        int cyclicCount = 0;
        int[]? only = null;

        foreach (var c in comps)
        {
            if (c.Length <= 1) continue;
            cyclicCount++;
            only = c;
        }

        if (cyclicCount == 0)
        {
            var arr0 = result.ToArray();
            if (key != null) Memo[key] = arr0;
            return arr0;
        }

        if (cyclicCount > 1)
        {
            foreach (var c in comps)
                if (c.Length > 1)
                    result.AddRange(SolveSet(c));
            var arr = result.ToArray();
            if (key != null) Memo[key] = arr;
            return arr;
        }

        result.AddRange(SolveComponent(only!));
        var final = result.ToArray();
        if (key != null) Memo[key] = final;
        return final;
    }

    static int[] SolveComponent(int[] comp)
    {
        string? key = comp.Length <= 260 ? MakeKey(comp) : null;
        if (key != null && Memo.TryGetValue(key, out var cached)) return cached;

        int[] cycle = FindCycle(comp);
        if (cycle.Length == 0) return Array.Empty<int>();

        int[] best = GreedyUpper(comp);
        foreach (int take in OrderCycleVertices(cycle, comp))
        {
            if (1 >= best.Length) break;
            int[] rest = RemoveOne(comp, take);
            int[] sub = SolveSet(rest);
            if (1 + sub.Length < best.Length)
            {
                var cand = new int[sub.Length + 1];
                cand[0] = take;
                Array.Copy(sub, 0, cand, 1, sub.Length);
                best = cand;
            }
        }

        if (key != null) Memo[key] = best;
        return best;
    }

    static int[] GreedyUpper(int[] verts)
    {
        var alive = new bool[N];
        foreach (int v in verts) alive[v] = true;
        var removed = new List<int>();
        while (true)
        {
            int[] cur = CollectAlive(alive, verts);
            cur = TrimAcyclicEnds(cur);
            int[] cycle = FindCycle(cur);
            if (cycle.Length == 0) break;
            int bestV = cycle[0], bestScore = -1;
            foreach (int v in cycle)
            {
                int outd = 0, ind = 0;
                foreach (int to in G[v]) if (alive[to]) outd++;
                foreach (int u in RG[v]) if (alive[u]) ind++;
                int score = ind * outd + ind + outd;
                if (score > bestScore) { bestScore = score; bestV = v; }
            }
            alive[bestV] = false;
            removed.Add(bestV);
        }
        return removed.ToArray();
    }

    static int[] TrimAcyclicEnds(int[] verts)
    {
        bool[] inSet = new bool[N], alive = new bool[N];
        int[] indeg = new int[N], outdeg = new int[N];
        foreach (int v in verts) { inSet[v] = true; alive[v] = true; }
        foreach (int v in verts)
            foreach (int to in G[v])
                if (inSet[to]) { outdeg[v]++; indeg[to]++; }

        var q = new Queue<int>();
        foreach (int v in verts) if (indeg[v] == 0 || outdeg[v] == 0) q.Enqueue(v);
        while (q.Count > 0)
        {
            int v = q.Dequeue();
            if (!alive[v] || (indeg[v] != 0 && outdeg[v] != 0)) continue;
            alive[v] = false;
            foreach (int to in G[v]) if (alive[to] && --indeg[to] == 0) q.Enqueue(to);
            foreach (int u in RG[v]) if (alive[u] && --outdeg[u] == 0) q.Enqueue(u);
        }
        return CollectAlive(alive, verts);
    }

    static List<int[]> StrongComponents(int[] verts)
    {
        var comps = new List<int[]>();
        if (verts.Length == 0) return comps;
        bool[] inSet = new bool[N];
        foreach (int v in verts) inSet[v] = true;
        int[] idx = new int[N], low = new int[N];
        bool[] on = new bool[N];
        Array.Fill(idx, -1);
        var st = new Stack<int>();
        int time = 0;

        void Dfs(int v)
        {
            idx[v] = low[v] = time++;
            st.Push(v); on[v] = true;
            foreach (int to in G[v]) if (inSet[to])
            {
                if (idx[to] < 0) { Dfs(to); low[v] = Math.Min(low[v], low[to]); }
                else if (on[to]) low[v] = Math.Min(low[v], idx[to]);
            }
            if (low[v] == idx[v])
            {
                var c = new List<int>();
                int x;
                do { x = st.Pop(); on[x] = false; c.Add(x); } while (x != v);
                comps.Add(c.ToArray());
            }
        }
        foreach (int v in verts) if (idx[v] < 0) Dfs(v);
        return comps;
    }

    static int[] FindCycle(int[] verts)
    {
        bool[] inSet = new bool[N];
        foreach (int v in verts) inSet[v] = true;
        int[] color = new int[N], parent = new int[N];
        Array.Fill(parent, -1);
        List<int>? cyc = null;

        bool Dfs(int v)
        {
            color[v] = 1;
            foreach (int to in G[v]) if (inSet[to])
            {
                if (color[to] == 0) { parent[to] = v; if (Dfs(to)) return true; }
                else if (color[to] == 1)
                {
                    cyc = new List<int> { to };
                    for (int x = v; x != to && x >= 0; x = parent[x]) cyc.Add(x);
                    return true;
                }
            }
            color[v] = 2;
            return false;
        }
        foreach (int v in verts) if (color[v] == 0 && Dfs(v)) break;
        return cyc?.ToArray() ?? Array.Empty<int>();
    }

    static int[] OrderCycleVertices(int[] cycle, int[] comp)
    {
        bool[] inSet = new bool[N];
        foreach (int v in comp) inSet[v] = true;
        Array.Sort(cycle, (a, b) => Score(b, inSet).CompareTo(Score(a, inSet)));
        return cycle;
    }

    static int Score(int v, bool[] inSet)
    {
        int o = 0;
        foreach (int to in G[v]) if (inSet[to]) o++;
        return o;
    }

    static int[] RemoveOne(int[] a, int x)
    {
        int[] r = new int[a.Length - 1];
        int p = 0;
        foreach (int v in a) if (v != x) r[p++] = v;
        return r;
    }

    static int[] CollectAlive(bool[] alive, int[] universe)
    {
        var r = new List<int>();
        foreach (int v in universe) if (alive[v]) r.Add(v);
        return r.ToArray();
    }

    static string MakeKey(int[] a) => string.Join(",", a);

    sealed class FastScanner
    {
        readonly System.IO.Stream input = Console.OpenStandardInput();
        readonly byte[] buffer = new byte[1 << 16];
        int len, p;
        int Read()
        {
            if (p >= len)
            {
                len = input.Read(buffer, 0, buffer.Length);
                p = 0;
                if (len == 0) return -1;
            }
            return buffer[p++];
        }
        public int NextInt()
        {
            int c;
            do { c = Read(); } while (c >= 0 && c <= 32);
            int sign = 1;
            if (c == '-') { sign = -1; c = Read(); }
            int v = 0;
            while (c > 32)
            {
                v = v * 10 + c - '0';
                c = Read();
            }
            return v * sign;
        }
    }
}
