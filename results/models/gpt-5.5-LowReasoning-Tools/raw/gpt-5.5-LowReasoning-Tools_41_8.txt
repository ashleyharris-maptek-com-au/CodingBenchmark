; SPIR-V
; Version: 1.0

OpCapability Shader
%glsl = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %inWorldPos %inNormal %inColor %outColor
OpExecutionMode %main OriginUpperLeft

OpName %main "main"
OpName %inWorldPos "worldPos"
OpName %inNormal "normal"
OpName %inColor "color"
OpName %outColor "outColor"
OpName %UBO "UBO"
OpMemberName %UBO 0 "model"
OpMemberName %UBO 1 "view"
OpMemberName %UBO 2 "projection"
OpMemberName %UBO 3 "lightPos"
OpMemberName %UBO 4 "cameraPos"
OpMemberName %UBO 5 "params"
OpName %ubo "ubo"

OpDecorate %inWorldPos Location 0
OpDecorate %inNormal Location 1
OpDecorate %inColor Location 4
OpDecorate %outColor Location 0
OpMemberDecorate %UBO 0 ColMajor
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 0 MatrixStride 16
OpMemberDecorate %UBO 1 ColMajor
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 1 MatrixStride 16
OpMemberDecorate %UBO 2 ColMajor
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 2 MatrixStride 16
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %UBO Block
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0

%void = OpTypeVoid
%fnVoid = OpTypeFunction %void
%float = OpTypeFloat 32
%bool = OpTypeBool
%uint = OpTypeInt 32 0
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%mat4 = OpTypeMatrix %v4float 4
%UBO = OpTypeStruct %mat4 %mat4 %mat4 %v4float %v4float %v4float
%ptrInputV3 = OpTypePointer Input %v3float
%ptrOutputV4 = OpTypePointer Output %v4float
%ptrUniformUBO = OpTypePointer Uniform %UBO
%ptrUniformV4 = OpTypePointer Uniform %v4float

%inWorldPos = OpVariable %ptrInputV3 Input
%inNormal = OpVariable %ptrInputV3 Input
%inColor = OpVariable %ptrInputV3 Input
%outColor = OpVariable %ptrOutputV4 Output
%ubo = OpVariable %ptrUniformUBO Uniform

%uint3 = OpConstant %uint 3
%float0 = OpConstant %float 0.0
%float1 = OpConstant %float 1.0
%float075 = OpConstant %float 0.75
%float05 = OpConstant %float 0.5
%float025 = OpConstant %float 0.25
%float07 = OpConstant %float 0.7
%float04 = OpConstant %float 0.4
%float02 = OpConstant %float 0.2

%main = OpFunction %void None %fnVoid
%entry = OpLabel
%world = OpLoad %v3float %inWorldPos
%normal = OpLoad %v3float %inNormal
%color = OpLoad %v3float %inColor
%lightPtr = OpAccessChain %ptrUniformV4 %ubo %uint3
%light4 = OpLoad %v4float %lightPtr
%light = OpVectorShuffle %v3float %light4 %light4 0 1 2
%n = OpExtInst %v3float %glsl Normalize %normal
%lightVec = OpFSub %v3float %light %world
%l = OpExtInst %v3float %glsl Normalize %lightVec
%dotNL = OpDot %float %n %l
%d = OpExtInst %float %glsl FMax %dotNL %float0
%ge075 = OpFOrdGreaterThanEqual %bool %d %float075
OpSelectionMerge %merge075 None
OpBranchConditional %ge075 %band100 %test050
%band100 = OpLabel
OpBranch %merge075
%test050 = OpLabel
%ge050 = OpFOrdGreaterThanEqual %bool %d %float05
OpSelectionMerge %merge050 None
OpBranchConditional %ge050 %band070 %test025
%band070 = OpLabel
OpBranch %merge050
%test025 = OpLabel
%ge025 = OpFOrdGreaterThanEqual %bool %d %float025
OpSelectionMerge %merge025 None
OpBranchConditional %ge025 %band040 %band020
%band040 = OpLabel
OpBranch %merge025
%band020 = OpLabel
OpBranch %merge025
%merge025 = OpLabel
%bandLow = OpPhi %float %float04 %band040 %float02 %band020
OpBranch %merge050
%merge050 = OpLabel
%bandMid = OpPhi %float %float07 %band070 %bandLow %merge025
OpBranch %merge075
%merge075 = OpLabel
%band = OpPhi %float %float1 %band100 %bandMid %merge050
%rgb = OpVectorTimesScalar %v3float %color %band
%out = OpCompositeConstruct %v4float %rgb %float1
OpStore %outColor %out
OpReturn
OpFunctionEnd
