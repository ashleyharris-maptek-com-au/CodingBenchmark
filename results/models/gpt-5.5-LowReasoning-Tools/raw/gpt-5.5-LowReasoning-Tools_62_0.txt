declare const require: any;
declare const process: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type Town = { id: string; x: number; y: number; value: number };

const W: number = data.width | 0;
const H: number = data.height | 0;
const N = W * H;
const grid: string[] = data.grid || [];
const budget: number = Math.max(0, data.cutBudget | 0);
const ignitions: number[][] = data.ignitions || [];
const towns: Town[] = data.towns || [];

function idx(x: number, y: number): number { return y * W + x; }
function xOf(i: number): number { return i % W; }
function yOf(i: number): number { return (i / W) | 0; }

const burnable = new Uint8Array(N);
for (let y = 0; y < H; y++) {
  const row = grid[y] || "";
  for (let x = 0; x < W; x++) {
    if (row.charCodeAt(x) === 46) burnable[idx(x, y)] = 1;
  }
}

const isIgnition = new Uint8Array(N);
const dist = new Int32Array(N);
dist.fill(-1);
const q = new Int32Array(N);
let head = 0, tail = 0;
for (const p of ignitions) {
  const x = p[0] | 0, y = p[1] | 0;
  if (x >= 0 && x < W && y >= 0 && y < H) {
    const id = idx(x, y);
    if (burnable[id]) {
      isIgnition[id] = 1;
      if (dist[id] < 0) {
        dist[id] = 0;
        q[tail++] = id;
      }
    }
  }
}

while (head < tail) {
  const v = q[head++];
  const x = xOf(v), y = yOf(v), nd = dist[v] + 1;
  let n: number;
  if (x > 0) { n = v - 1; if (burnable[n] && dist[n] < 0) { dist[n] = nd; q[tail++] = n; } }
  if (x + 1 < W) { n = v + 1; if (burnable[n] && dist[n] < 0) { dist[n] = nd; q[tail++] = n; } }
  if (y > 0) { n = v - W; if (burnable[n] && dist[n] < 0) { dist[n] = nd; q[tail++] = n; } }
  if (y + 1 < H) { n = v + W; if (burnable[n] && dist[n] < 0) { dist[n] = nd; q[tail++] = n; } }
}

const townValue = new Float64Array(N);
const isTown = new Uint8Array(N);
const threatened: Town[] = [];
for (const t of towns) {
  const x = t.x | 0, y = t.y | 0;
  if (x < 0 || x >= W || y < 0 || y >= H) continue;
  const id = idx(x, y);
  townValue[id] += Number(t.value) || 0;
  isTown[id] = 1;
  if (dist[id] >= 0) threatened.push(t);
}

const tin = new Int32Array(N);
const low = new Int32Array(N);
const parent = new Int32Array(N);
const iter = new Uint8Array(N);
const subIgn = new Int32Array(N);
const subVal = new Float64Array(N);
parent.fill(-1);
let timer = 0;
const candValue = new Map<number, number>();

function addCand(cell: number, value: number) {
  if (value <= 0 || isIgnition[cell] || isTown[cell] || !burnable[cell]) return;
  const old = candValue.get(cell) || 0;
  if (value > old) candValue.set(cell, value);
}

function neigh(v: number, k: number): number {
  const x = v % W;
  if (k === 0) return x > 0 ? v - 1 : -1;
  if (k === 1) return x + 1 < W ? v + 1 : -1;
  if (k === 2) return v >= W ? v - W : -1;
  return v + W < N ? v + W : -1;
}

const stack: number[] = [];
for (let s = 0; s < N; s++) {
  if (dist[s] < 0 || tin[s] !== 0) continue;
  parent[s] = -1;
  tin[s] = low[s] = ++timer;
  subIgn[s] = isIgnition[s];
  subVal[s] = townValue[s];
  iter[s] = 0;
  stack.push(s);
  while (stack.length) {
    const v = stack[stack.length - 1];
    if (iter[v] < 4) {
      const n = neigh(v, iter[v]++);
      if (n < 0 || dist[n] < 0) continue;
      if (tin[n] === 0) {
        parent[n] = v;
        tin[n] = low[n] = ++timer;
        subIgn[n] = isIgnition[n];
        subVal[n] = townValue[n];
        iter[n] = 0;
        stack.push(n);
      } else if (n !== parent[v] && tin[n] < low[v]) {
        low[v] = tin[n];
      }
    } else {
      stack.pop();
      const p = parent[v];
      if (p >= 0) {
        if (low[v] < low[p]) low[p] = low[v];
        subIgn[p] += subIgn[v];
        subVal[p] += subVal[v];
        if (low[v] >= tin[p] && subIgn[v] === 0) addCand(p, (candValue.get(p) || 0) + subVal[v]);
      }
    }
  }
}

const score = new Map<number, number>();
function bump(cell: number, v: number) {
  if (cell < 0 || cell >= N || !burnable[cell] || dist[cell] < 0 || isIgnition[cell] || isTown[cell]) return;
  score.set(cell, (score.get(cell) || 0) + v);
}

for (const t of threatened) {
  let cur = idx(t.x | 0, t.y | 0);
  const value = Math.max(0, Number(t.value) || 0);
  for (let step = 1; step <= 80 && cur >= 0 && dist[cur] > 0; step++) {
    let best = -1, bestD = dist[cur];
    for (let k = 0; k < 4; k++) {
      const n = neigh(cur, k);
      if (n >= 0 && dist[n] >= 0 && dist[n] < bestD) { bestD = dist[n]; best = n; }
    }
    if (best < 0) break;
    bump(best, value / Math.sqrt(step));
    cur = best;
  }
  const center = idx(t.x | 0, t.y | 0);
  for (let k = 0; k < 4; k++) bump(neigh(center, k), value * 0.75);
}

for (const [cell, v] of score) addCand(cell, v * 0.001);

const candidates = Array.from(candValue.entries())
  .sort((a, b) => b[1] - a[1] || dist[a[0]] - dist[b[0]]);

const chosen: number[] = [];
const used = new Uint8Array(N);
for (const [cell] of candidates) {
  if (chosen.length >= budget) break;
  if (!used[cell]) {
    used[cell] = 1;
    chosen.push(cell);
  }
}

const firebreaks = chosen.map((i) => [xOf(i), yOf(i)]);
process.stdout.write(JSON.stringify({ firebreaks }));
