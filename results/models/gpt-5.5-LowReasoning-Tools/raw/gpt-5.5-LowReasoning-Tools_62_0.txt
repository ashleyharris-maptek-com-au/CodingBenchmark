declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input.charCodeAt(0) === 0xFEFF ? input.slice(1) : input);

type Town = { id: string; x: number; y: number; value: number };
type Plan = { cells: number[]; gain: number; cost: number; score: number };

const width: number = data.width | 0;
const height: number = data.height | 0;
const grid: string[] = data.grid || [];
const ignitions: number[][] = data.ignitions || [];
const towns: Town[] = data.towns || [];
const budget: number = Math.max(0, data.cutBudget | 0);
const n = width * height;

function idx(x: number, y: number): number { return y * width + x; }
function xOf(i: number): number { return i % width; }
function yOf(i: number): number { return (i / width) | 0; }
function inBounds(x: number, y: number): boolean {
  return x >= 0 && y >= 0 && x < width && y < height;
}

const forest = new Uint8Array(n);
for (let y = 0; y < height; y++) {
  const row = grid[y] || "";
  for (let x = 0; x < width; x++) {
    if (row.charCodeAt(x) === 46) forest[idx(x, y)] = 1;
  }
}

const ignitionCell = new Uint8Array(n);
for (const p of ignitions) {
  const x = p[0] | 0, y = p[1] | 0;
  if (inBounds(x, y)) ignitionCell[idx(x, y)] = 1;
}

const townValue = new Float64Array(n);
for (const t of towns) {
  const x = t.x | 0, y = t.y | 0;
  if (inBounds(x, y)) townValue[idx(x, y)] += Number(t.value) || 0;
}

function simulate(cuts: Uint8Array): { burned: Uint8Array; burnedValue: number } {
  const burned = new Uint8Array(n);
  const q = new Int32Array(n);
  let head = 0, tail = 0;
  for (const p of ignitions) {
    const x = p[0] | 0, y = p[1] | 0;
    if (!inBounds(x, y)) continue;
    const s = idx(x, y);
    if (forest[s] && !cuts[s] && !burned[s]) {
      burned[s] = 1;
      q[tail++] = s;
    }
  }
  while (head < tail) {
    const v = q[head++];
    const x = xOf(v);
    let u: number;
    if (x > 0) {
      u = v - 1;
      if (forest[u] && !cuts[u] && !burned[u]) { burned[u] = 1; q[tail++] = u; }
    }
    if (x + 1 < width) {
      u = v + 1;
      if (forest[u] && !cuts[u] && !burned[u]) { burned[u] = 1; q[tail++] = u; }
    }
    u = v - width;
    if (u >= 0 && forest[u] && !cuts[u] && !burned[u]) { burned[u] = 1; q[tail++] = u; }
    u = v + width;
    if (u < n && forest[u] && !cuts[u] && !burned[u]) { burned[u] = 1; q[tail++] = u; }
  }
  let burnedValue = 0;
  for (let i = 0; i < n; i++) if (burned[i] && townValue[i] > 0) burnedValue += townValue[i];
  return { burned, burnedValue };
}

function uniqueCells(cells: number[]): number[] {
  const seen = new Set<number>();
  const out: number[] = [];
  for (const c of cells) {
    if (c >= 0 && c < n && forest[c] && !ignitionCell[c] && !seen.has(c)) {
      seen.add(c);
      out.push(c);
    }
  }
  return out;
}

const emptyCuts = new Uint8Array(n);
const initial = simulate(emptyCuts);
const initialBurnedValue = initial.burnedValue;

if (budget <= 0 || initialBurnedValue <= 0 || n <= 0) {
  console.log(JSON.stringify({ firebreaks: [] }));
} else {
  const threatened = towns
    .map(t => ({ t, cell: inBounds(t.x | 0, t.y | 0) ? idx(t.x | 0, t.y | 0) : -1, value: Number(t.value) || 0 }))
    .filter(o => o.cell >= 0 && initial.burned[o.cell] && forest[o.cell] && !ignitionCell[o.cell] && o.value > 0)
    .sort((a, b) => b.value - a.value);

  const maxTownPlans = Math.min(threatened.length, Math.max(60, Math.min(300, budget * 8 + 40)));
  const rawPlans: number[][] = [];
  const planKeys = new Set<string>();

  function addPlan(cells: number[]) {
    const u = uniqueCells(cells);
    if (u.length === 0 || u.length > budget) return;
    u.sort((a, b) => a - b);
    const key = u.join(",");
    if (!planKeys.has(key)) {
      planKeys.add(key);
      rawPlans.push(u);
    }
  }

  for (let i = 0; i < maxTownPlans; i++) {
    const c = threatened[i].cell;
    const tx = xOf(c), ty = yOf(c);
    addPlan([c]);

    const maxR = Math.min(8, Math.max(1, budget));
    for (let r = 1; r <= maxR; r++) {
      const square: number[] = [];
      for (let x = tx - r; x <= tx + r; x++) {
        if (inBounds(x, ty - r)) square.push(idx(x, ty - r));
        if (inBounds(x, ty + r)) square.push(idx(x, ty + r));
      }
      for (let y = ty - r + 1; y <= ty + r - 1; y++) {
        if (inBounds(tx - r, y)) square.push(idx(tx - r, y));
        if (inBounds(tx + r, y)) square.push(idx(tx + r, y));
      }
      addPlan(square);

      const diamond: number[] = [];
      for (let dx = -r; dx <= r; dx++) {
        const dy = r - Math.abs(dx);
        if (inBounds(tx + dx, ty + dy)) diamond.push(idx(tx + dx, ty + dy));
        if (dy !== 0 && inBounds(tx + dx, ty - dy)) diamond.push(idx(tx + dx, ty - dy));
      }
      addPlan(diamond);
    }
  }

  const scored: Plan[] = [];
  const tempCuts = new Uint8Array(n);
  for (const cells of rawPlans) {
    tempCuts.fill(0);
    for (const c of cells) tempCuts[c] = 1;
    const after = simulate(tempCuts).burnedValue;
    const gain = initialBurnedValue - after;
    if (gain > 0) {
      const cost = cells.length;
      scored.push({ cells, gain, cost, score: gain / Math.pow(cost, 0.82) });
    }
  }
  scored.sort((a, b) => (b.score - a.score) || (b.gain - a.gain) || (a.cost - b.cost));

  const chosen = new Uint8Array(n);
  let used = 0;
  let currentBurnedValue = initialBurnedValue;

  for (const p of scored) {
    if (used >= budget) break;
    let extra = 0;
    for (const c of p.cells) if (!chosen[c]) extra++;
    if (extra === 0 || used + extra > budget) continue;

    const trial = new Uint8Array(chosen);
    for (const c of p.cells) trial[c] = 1;
    const after = simulate(trial).burnedValue;
    if (after < currentBurnedValue) {
      chosen.set(trial);
      used += extra;
      currentBurnedValue = after;
    }
  }

  if (used < budget) {
    for (const o of threatened) {
      if (used >= budget) break;
      if (!chosen[o.cell]) {
        chosen[o.cell] = 1;
        used++;
      }
    }
  }

  const firebreaks: number[][] = [];
  for (let i = 0; i < n; i++) {
    if (chosen[i]) firebreaks.push([xOf(i), yOf(i)]);
  }
  console.log(JSON.stringify({ firebreaks }));
}
