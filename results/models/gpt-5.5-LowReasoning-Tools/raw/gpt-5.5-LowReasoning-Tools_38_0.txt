import sys


def parse_int(token):
    return int(token.lstrip("\ufeff\ufffeï»¿"))


def read_input():
    lines = sys.stdin.read().splitlines()
    if not lines:
        return 0, []

    first = lines[0].lstrip("\ufeff").split()
    if len(first) < 2:
        return 0, []

    universe_size = parse_int(first[0])
    set_count = parse_int(first[1])
    sets = []

    for i in range(set_count):
        line = lines[i + 1] if i + 1 < len(lines) else ""
        elems = frozenset(parse_int(x) for x in line.split())
        sets.append(elems)

    return universe_size, sets


def solve_exact_cover(universe_size, sets):
    if universe_size == 0:
        return []

    full_universe = set(range(universe_size))
    candidates = {e: [] for e in range(universe_size)}
    usable_sets = []

    for idx, elems in enumerate(sets):
        if not elems:
            continue
        if any(e < 0 or e >= universe_size for e in elems):
            continue
        usable_sets.append(idx)
        for e in elems:
            candidates[e].append(idx)

    for e in range(universe_size):
        if not candidates[e]:
            return None
        candidates[e].sort()

    chosen = []

    def search(uncovered, blocked_sets):
        if not uncovered:
            return list(chosen)

        pivot = min(
            uncovered,
            key=lambda e: sum(1 for idx in candidates[e] if idx not in blocked_sets),
        )

        available = [idx for idx in candidates[pivot] if idx not in blocked_sets]
        if not available:
            return None

        for idx in available:
            elems = sets[idx]
            if not elems <= uncovered:
                continue

            newly_blocked = set(blocked_sets)
            for e in elems:
                newly_blocked.update(candidates[e])

            chosen.append(idx)
            result = search(uncovered - elems, newly_blocked)
            if result is not None:
                return result
            chosen.pop()

        return None

    return search(full_universe, set())


def main():
    universe_size, sets = read_input()
    solution = solve_exact_cover(universe_size, sets)

    if solution is None:
        print("NO SOLUTION")
    else:
        print("SOLUTION")
        print(" ".join(map(str, sorted(solution))))


if __name__ == "__main__":
    main()
