import sys


def solve_exact_cover(universe_size, raw_sets):
    if universe_size == 0:
        return []

    masks = []
    contains = [[] for _ in range(universe_size)]

    for idx, elems in enumerate(raw_sets):
        mask = 0
        valid = True
        for e in elems:
            if e < 0 or e >= universe_size:
                valid = False
                break
            bit = 1 << e
            if mask & bit:
                valid = False
                break
            mask |= bit

        masks.append(mask if valid else -1)
        if valid and mask:
            m = mask
            while m:
                bit = m & -m
                contains[bit.bit_length() - 1].append(idx)
                m -= bit

    all_uncovered = (1 << universe_size) - 1
    chosen = []

    sys.setrecursionlimit(max(1000, universe_size + len(raw_sets) + 50))

    def search(uncovered):
        if uncovered == 0:
            return True

        best_element = -1
        best_candidates = None
        scan = uncovered

        while scan:
            bit = scan & -scan
            element = bit.bit_length() - 1
            candidates = []

            for set_idx in contains[element]:
                set_mask = masks[set_idx]
                if set_mask != -1 and (set_mask & uncovered) == set_mask:
                    candidates.append(set_idx)

            if not candidates:
                return False

            if best_candidates is None or len(candidates) < len(best_candidates):
                best_element = element
                best_candidates = candidates
                if len(best_candidates) == 1:
                    break

            scan -= bit

        for set_idx in best_candidates:
            set_mask = masks[set_idx]
            chosen.append(set_idx)
            if search(uncovered ^ set_mask):
                return True
            chosen.pop()

        return False

    if search(all_uncovered):
        return sorted(chosen)
    return None


def main():
    lines = sys.stdin.read().splitlines()
    if not lines:
        return

    first = lines[0].split()
    if len(first) < 2:
        print("NO SOLUTION")
        return

    universe_size = int(first[0])
    set_count = int(first[1])

    raw_sets = []
    for i in range(set_count):
        line = lines[i + 1] if i + 1 < len(lines) else ""
        raw_sets.append([int(x) for x in line.split()])

    solution = solve_exact_cover(universe_size, raw_sets)
    if solution is None:
        print("NO SOLUTION")
    else:
        print("SOLUTION")
        print(" ".join(map(str, solution)))


if __name__ == "__main__":
    main()
