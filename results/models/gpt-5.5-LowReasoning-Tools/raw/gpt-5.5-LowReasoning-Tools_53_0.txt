import math


_memory = {
    "pos": None,
    "vel": None,
    "range": None,
    "t": 0.0,
}


def _clip(x, lo, hi):
    return max(lo, min(hi, x))


def _median(values):
    values = sorted(values)
    n = len(values)
    if n == 0:
        return None
    if n & 1:
        return values[n // 2]
    return 0.5 * (values[n // 2 - 1] + values[n // 2])


def _valid_number(x, limit=1.0e9):
    return isinstance(x, (int, float)) and math.isfinite(x) and abs(x) < limit


def _axis_estimate(state, kind, axis):
    """Robustly combine radar/lidar relative position or velocity for one axis."""
    vals = []
    for sensor in ("lidar", "radar"):
        if kind == "pos":
            key = f"{sensor}_rel_pos_{axis}_m"
            valid = f"{sensor}_rel_{axis}_valid"
            limit = 1.0e6
        else:
            key = f"{sensor}_rel_vel_{axis}_ms"
            valid = f"{sensor}_relv_{axis}_valid"
            limit = 1.0e4
        v = state.get(key)
        if state.get(valid, False) and _valid_number(v, limit):
            vals.append(float(v))
    if not vals:
        return None
    if len(vals) == 2 and abs(vals[0] - vals[1]) > (25.0 if kind == "pos" else 2.0):
        old = _memory.get(kind)
        if old is not None:
            idx = {"x": 0, "y": 1, "z": 2}[axis]
            return min(vals, key=lambda v: abs(v - old[idx]))
        return vals[0]
    return sum(vals) / len(vals)


def _range_estimate(state, pos):
    vals = []
    for i in (1, 2, 3):
        v = state.get(f"range_{i}_m")
        if state.get(f"range_{i}_valid", False) and _valid_number(v, 1.0e6) and v >= 0:
            vals.append(float(v))
    cam_r = state.get("dock_camera_range_m")
    if state.get("dock_camera_active", False) and _valid_number(cam_r, 1.0e6) and cam_r >= 0:
        vals.append(float(cam_r))
    if pos is not None:
        vals.append(math.sqrt(pos[0] * pos[0] + pos[1] * pos[1] + pos[2] * pos[2]))
    if not vals:
        return _memory.get("range") or 1.0e6
    med = _median(vals)
    clean = [v for v in vals if abs(v - med) <= max(8.0, 0.25 * max(1.0, med))]
    return sum(clean) / len(clean) if clean else med


def _estimate_state(state, dt):
    old_p = _memory.get("pos")
    old_v = _memory.get("vel")

    p = [_axis_estimate(state, "pos", a) for a in "xyz"]
    v = [_axis_estimate(state, "vel", a) for a in "xyz"]

    if any(x is None for x in p):
        if old_p is not None and old_v is not None and dt > 0:
            pred = [old_p[i] + old_v[i] * dt for i in range(3)]
            p = [pred[i] if p[i] is None else p[i] for i in range(3)]
        else:
            p = [0.0 if x is None else x for x in p]

    rng_for_cam = math.sqrt(p[0] * p[0] + p[1] * p[1] + p[2] * p[2])
    if state.get("dock_camera_active", False) and rng_for_cam < 600.0:
        cr = state.get("dock_camera_range_m")
        br = []
        ok = state.get("dock_camera_bearing_x_valid", False)
        ok = ok and state.get("dock_camera_bearing_y_valid", False)
        ok = ok and state.get("dock_camera_bearing_z_valid", False)
        if ok and _valid_number(cr, 1.0e6) and cr > 0:
            br = [float(state.get("dock_camera_bearing_x", 0.0)),
                  float(state.get("dock_camera_bearing_y", 0.0)),
                  float(state.get("dock_camera_bearing_z", 0.0))]
            n = math.sqrt(br[0] * br[0] + br[1] * br[1] + br[2] * br[2])
            if n > 1.0e-6:
                cam_p = [cr * br[i] / n for i in range(3)]
                # Camera is precise near the port, but keep it blended to survive bad bearings.
                blend = 0.65 if cr < 80.0 else 0.35
                p = [(1.0 - blend) * p[i] + blend * cam_p[i] for i in range(3)]

    if any(x is None for x in v):
        if old_p is not None and dt > 1.0e-6:
            dv = [(p[i] - old_p[i]) / dt for i in range(3)]
            v = [dv[i] if v[i] is None else v[i] for i in range(3)]
        elif old_v is not None:
            v = [old_v[i] if v[i] is None else v[i] for i in range(3)]
        else:
            v = [0.0 if x is None else x for x in v]

    if old_p is not None:
        jump = math.sqrt(sum((p[i] - old_p[i]) * (p[i] - old_p[i]) for i in range(3)))
        local_range = max(1.0, math.sqrt(p[0] * p[0] + p[1] * p[1] + p[2] * p[2]))
        if jump > max(150.0, 0.75 * local_range):
            old_p = None
            old_v = None

    # Low-pass just enough to reject flicker/noise without hiding real closing speed.
    if old_p is not None:
        alpha = 0.45 if dt <= 1.5 else 0.70
        p = [alpha * p[i] + (1.0 - alpha) * (old_p[i] + (old_v[i] if old_v else 0.0) * dt)
             for i in range(3)]
    if old_v is not None:
        beta = 0.55
        v = [beta * v[i] + (1.0 - beta) * old_v[i] for i in range(3)]

    r = _range_estimate(state, p)
    _memory["pos"], _memory["vel"], _memory["range"] = p[:], v[:], r
    _memory["t"] = _memory.get("t", 0.0) + max(0.0, dt)
    return p, v, r


def _attitude_rates(state):
    rates = {}
    for axis in ("roll", "pitch", "yaw"):
        vals = []
        for i in (1, 2, 3):
            key = f"gyro_{i}_{axis}_dps"
            if state.get(f"gyro_{i}_{axis}_valid", False) and _valid_number(state.get(key), 1000.0):
                vals.append(float(state[key]))
        rate_dps = _median(vals) if vals else 0.0
        rates[axis] = _clip(-math.radians(rate_dps) * 0.45, -0.035, 0.035)
    return rates


def autopilot_step(state: dict, dt: float) -> dict:
    p, v, rng = _estimate_state(state, max(0.0, float(dt or 0.0)))
    x, y, z = p
    vx, vy, vz = v
    speed = math.sqrt(vx * vx + vy * vy + vz * vz)
    fuel_pct = float(state.get("fuel_remaining_pct", 100.0) or 0.0)
    low_fuel = fuel_pct < 15.0 or float(state.get("fuel_remaining_kg", 999.0) or 999.0) < 75.0

    # Desired relative velocity.  Far away, along-track phasing is orbital:
    # behind (negative y) needs retrograde thrust to lower orbit and catch up.
    tx = ty = tz = 0.0
    if rng > 200.0:
        # Do not chase radial error directly at long range; use it mainly as a
        # bounded correction while y phasing does the real rendezvous work.
        desired_vy = _clip(-0.0012 * y, -2.0, 2.0)
        ay_cmd = 0.020 * (desired_vy - vy)
        if abs(y) > 80.0:
            # Orbital phasing: behind means burn retrograde to lower orbit and
            # gain along-track rate; ahead means burn prograde to fall back.
            ay_cmd += -0.060 if y < 0.0 else 0.060
        ax_cmd = _clip(-0.0015 * x - 0.055 * vx, -0.08, 0.08)
        az_cmd = _clip(-0.0025 * z - 0.080 * vz, -0.10, 0.10)
    elif rng > 50.0:
        desired_vy = _clip(-0.0045 * y, -0.8, 0.8)
        ax_cmd = -0.0040 * x - 0.110 * vx
        ay_cmd = 0.080 * (desired_vy - vy)
        az_cmd = -0.0045 * z - 0.120 * vz
    elif rng > 10.0:
        closing = _clip(0.018 * rng, 0.12, 0.42)
        n = max(rng, 1.0e-6)
        desired = [-closing * x / n, -closing * y / n, -closing * z / n]
        ax_cmd = 0.22 * (desired[0] - vx) - 0.010 * x
        ay_cmd = 0.22 * (desired[1] - vy) - 0.010 * y
        az_cmd = 0.22 * (desired[2] - vz) - 0.010 * z
    else:
        closing = min(0.09, max(0.025, 0.010 * rng))
        n = max(rng, 1.0e-6)
        desired = [-closing * x / n, -closing * y / n, -closing * z / n]
        ax_cmd = 0.32 * (desired[0] - vx) - 0.018 * x
        ay_cmd = 0.32 * (desired[1] - vy) - 0.018 * y
        az_cmd = 0.32 * (desired[2] - vz) - 0.018 * z

    # Brake hard if the vehicle is already closing faster than the current
    # range can tolerate.  This protects the 2 m / 0.3 m/s success gate.
    if rng > 1.0e-6:
        radial_v = (x * vx + y * vy + z * vz) / rng
        max_close = 0.09 if rng < 10.0 else (0.35 if rng < 50.0 else (0.9 if rng < 200.0 else 2.2))
        if radial_v < -max_close:
            excess = -radial_v - max_close
            ax_cmd += (x / rng) * excess * 0.45
            ay_cmd += (y / rng) * excess * 0.45
            az_cmd += (z / rng) * excess * 0.45

    # Convert acceleration-like commands to normalized thrust.  The exact mass
    # is unknown, so these gains intentionally stay conservative.
    scale = 7.5 if not low_fuel else 4.2
    tx = _clip(ax_cmd * scale, -1.0, 1.0)
    ty = _clip(ay_cmd * scale, -1.0, 1.0)
    tz = _clip(az_cmd * scale, -1.0, 1.0)

    # Deadband saves propellant after capture and avoids dithering on noisy data.
    dead = 0.015 if not low_fuel else 0.035
    tx = 0.0 if abs(tx) < dead else tx
    ty = 0.0 if abs(ty) < dead else ty
    tz = 0.0 if abs(tz) < dead else tz

    # If inside the success box, stop translating unless residual speed is high.
    if rng < float(state.get("target_dock_range_m", 2.0) or 2.0) and speed < float(state.get("target_dock_speed_ms", 0.3) or 0.3):
        tx = ty = tz = 0.0

    att = _attitude_rates(state)
    return {
        "thrust_x": _clip(tx, -1.0, 1.0),
        "thrust_y": _clip(ty, -1.0, 1.0),
        "thrust_z": _clip(tz, -1.0, 1.0),
        "att_rate_roll": att["roll"],
        "att_rate_pitch": att["pitch"],
        "att_rate_yaw": att["yaw"],
    }
