declare const require: any;

const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type Robot = { id: number; x: number; y: number };
type Job = { id: any; x: number; y: number; priority: number };

const grid: string[] = data.grid || [];
const robots: Robot[] = (data.robots || []).map((r: any) => ({
  id: Number(r.id),
  x: Number(r.x),
  y: Number(r.y),
}));
const jobs: Job[] = (data.jobs || []).map((j: any) => ({
  id: j.id,
  x: Number(j.x),
  y: Number(j.y),
  priority: Number(j.priority || 0),
}));

const height = grid.length;
const width = height === 0 ? 0 : grid[0].length;
const cellCount = width * height;
const INF = 0x3fffffff;

const passable = new Uint8Array(cellCount);
for (let y = 0; y < height; y++) {
  const row = grid[y];
  for (let x = 0; x < width; x++) {
    passable[y * width + x] = row.charCodeAt(x) === 35 ? 0 : 1;
  }
}

function cellOf(x: number, y: number): number {
  return y * width + x;
}

function compareIds(a: any, b: any): number {
  if (typeof a === 'number' && typeof b === 'number') return a - b;
  const sa = String(a);
  const sb = String(b);
  return sa < sb ? -1 : sa > sb ? 1 : 0;
}

jobs.sort((a, b) => {
  const byPriority = b.priority - a.priority;
  return byPriority !== 0 ? byPriority : compareIds(a.id, b.id);
});
robots.sort((a, b) => a.id - b.id);

let maxRobotId = -1;
for (const r of robots) if (r.id > maxRobotId) maxRobotId = r.id;

const assignments: any[][] = Array.from({ length: Math.max(0, maxRobotId + 1) }, () => []);
const currentCell = new Map<number, number>();
const travelSoFar = new Map<number, number>();

for (const r of robots) {
  currentCell.set(r.id, cellOf(r.x, r.y));
  travelSoFar.set(r.id, 0);
}

const seen = new Int32Array(cellCount);
const dist = new Int32Array(cellCount);
const targetStamp = new Int32Array(cellCount);
const queue = new Int32Array(cellCount);
let run = 0;

function computeDistancesFrom(start: number, targets: number[]): void {
  run++;
  if (run === 0x7fffffff) {
    seen.fill(0);
    targetStamp.fill(0);
    run = 1;
  }

  let remaining = 0;
  for (const t of targets) {
    if (t >= 0 && t < cellCount && targetStamp[t] !== run) {
      targetStamp[t] = run;
      remaining++;
    }
  }

  let head = 0;
  let tail = 0;
  if (start < 0 || start >= cellCount || passable[start] === 0) return;

  queue[tail++] = start;
  seen[start] = run;
  dist[start] = 0;

  while (head < tail && remaining > 0) {
    const c = queue[head++];
    if (targetStamp[c] === run) {
      targetStamp[c] = -run;
      remaining--;
      if (remaining === 0) break;
    }

    const d = dist[c] + 1;
    const x = c % width;

    const left = c - 1;
    if (x > 0 && passable[left] && seen[left] !== run) {
      seen[left] = run;
      dist[left] = d;
      queue[tail++] = left;
    }

    const right = c + 1;
    if (x + 1 < width && passable[right] && seen[right] !== run) {
      seen[right] = run;
      dist[right] = d;
      queue[tail++] = right;
    }

    const up = c - width;
    if (up >= 0 && passable[up] && seen[up] !== run) {
      seen[up] = run;
      dist[up] = d;
      queue[tail++] = up;
    }

    const down = c + width;
    if (down < cellCount && passable[down] && seen[down] !== run) {
      seen[down] = run;
      dist[down] = d;
      queue[tail++] = down;
    }
  }
}

for (const job of jobs) {
  if (robots.length === 0) break;

  const jobCell = cellOf(job.x, job.y);
  const robotCells: number[] = [];
  for (const r of robots) robotCells.push(currentCell.get(r.id) as number);

  computeDistancesFrom(jobCell, robotCells);

  let bestRobot = robots[0];
  let bestDistance = INF;
  let bestTravel = INF;

  for (const r of robots) {
    const rc = currentCell.get(r.id) as number;
    const d = seen[rc] === run ? dist[rc] : INF;
    const carried = travelSoFar.get(r.id) as number;

    if (
      d < bestDistance ||
      (d === bestDistance && carried < bestTravel) ||
      (d === bestDistance && carried === bestTravel && r.id < bestRobot.id)
    ) {
      bestRobot = r;
      bestDistance = d;
      bestTravel = carried;
    }
  }

  assignments[bestRobot.id].push(job.id);
  currentCell.set(bestRobot.id, jobCell);
  travelSoFar.set(bestRobot.id, (travelSoFar.get(bestRobot.id) as number) + bestDistance);
}

console.log(JSON.stringify({ assignments }));
