declare const require: any;
declare const process: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input.charCodeAt(0) === 0xFEFF ? input.slice(1) : input);

type Robot = { id: number; x: number; y: number };
type Job = { id: number; x: number; y: number; priority: number };

const grid: string[] = data.grid || [];
const robots: Robot[] = (data.robots || []).slice().sort((a: Robot, b: Robot) => a.id - b.id);
const jobs: Job[] = data.jobs || [];

const height = grid.length;
const width = height ? grid[0].length : 0;
const cellCount = width * height;
const INF = 1 << 28;

const walkable = new Uint8Array(cellCount);
for (let y = 0; y < height; y++) {
  const row = grid[y] || "";
  for (let x = 0; x < width; x++) {
    if (row.charCodeAt(x) === 46) walkable[y * width + x] = 1;
  }
}

function cellOf(x: number, y: number): number {
  if (x < 0 || y < 0 || x >= width || y >= height) return -1;
  const c = y * width + x;
  return walkable[c] ? c : -1;
}

const points: { x: number; y: number; cell: number }[] = [];
for (const r of robots) points.push({ x: r.x, y: r.y, cell: cellOf(r.x, r.y) });
const jobPointOffset = points.length;
for (const j of jobs) points.push({ x: j.x, y: j.y, cell: cellOf(j.x, j.y) });

const distCache: Array<Int32Array | undefined> = new Array(points.length);

function bfsFrom(pointIndex: number): Int32Array {
  const cached = distCache[pointIndex];
  if (cached) return cached;

  const dist = new Int32Array(cellCount);
  dist.fill(INF);
  const start = points[pointIndex].cell;
  if (start < 0 || cellCount === 0) {
    distCache[pointIndex] = dist;
    return dist;
  }

  const q = new Int32Array(cellCount);
  let head = 0, tail = 0;
  q[tail++] = start;
  dist[start] = 0;

  while (head < tail) {
    const c = q[head++];
    const d = dist[c] + 1;
    const x = c % width;

    const left = c - 1;
    if (x > 0 && walkable[left] && dist[left] === INF) {
      dist[left] = d; q[tail++] = left;
    }
    const right = c + 1;
    if (x + 1 < width && walkable[right] && dist[right] === INF) {
      dist[right] = d; q[tail++] = right;
    }
    const up = c - width;
    if (up >= 0 && walkable[up] && dist[up] === INF) {
      dist[up] = d; q[tail++] = up;
    }
    const down = c + width;
    if (down < cellCount && walkable[down] && dist[down] === INF) {
      dist[down] = d; q[tail++] = down;
    }
  }

  distCache[pointIndex] = dist;
  return dist;
}

function distancePointToCell(pointIndex: number, cell: number): number {
  if (cell < 0) return INF;
  return bfsFrom(pointIndex)[cell];
}

function distancePoints(a: number, b: number): number {
  return distancePointToCell(a, points[b].cell);
}

function routeCost(robotIndex: number, route: number[]): number {
  let prevPoint = robotIndex;
  let total = 0;
  for (const jobIndex of route) {
    const d = distancePoints(prevPoint, jobPointOffset + jobIndex);
    total += d >= INF ? 100000000 : d;
    prevPoint = jobPointOffset + jobIndex;
  }
  return total;
}

const robotCount = Math.max(robots.length, 0);
const routes: number[][] = Array.from({ length: robotCount }, () => []);

const jobOrder = jobs.map((_, i) => i).sort((a, b) => {
  const pa = jobs[a].priority || 0, pb = jobs[b].priority || 0;
  if (pa !== pb) return pb - pa;
  if (jobs[a].id !== jobs[b].id) return jobs[a].id - jobs[b].id;
  return a - b;
});

function insertionDelta(route: number[], robotIndex: number, jobIndex: number, pos: number): number {
  const jobPoint = jobPointOffset + jobIndex;
  const prevPoint = pos === 0 ? robotIndex : jobPointOffset + route[pos - 1];
  const nextPoint = pos === route.length ? -1 : jobPointOffset + route[pos];
  const addA = distancePoints(prevPoint, jobPoint);
  if (nextPoint < 0) return addA >= INF ? 100000000 : addA;
  const addB = distancePoints(jobPoint, nextPoint);
  const old = distancePoints(prevPoint, nextPoint);
  const delta = (addA >= INF ? 100000000 : addA) + (addB >= INF ? 100000000 : addB) - (old >= INF ? 100000000 : old);
  return delta;
}

if (robotCount > 0) {
  for (const jobIndex of jobOrder) {
    let bestRobot = 0, bestPos = 0, bestDelta = Infinity;
    for (let r = 0; r < robotCount; r++) {
      const route = routes[r];
      for (let p = 0; p <= route.length; p++) {
        const delta = insertionDelta(route, r, jobIndex, p);
        if (
          delta < bestDelta ||
          (delta === bestDelta && (r < bestRobot || (r === bestRobot && p < bestPos)))
        ) {
          bestDelta = delta;
          bestRobot = r;
          bestPos = p;
        }
      }
    }
    routes[bestRobot].splice(bestPos, 0, jobIndex);
  }
}

function removalSaving(route: number[], robotIndex: number, pos: number): number {
  const prevPoint = pos === 0 ? robotIndex : jobPointOffset + route[pos - 1];
  const jobPoint = jobPointOffset + route[pos];
  const nextPoint = pos + 1 === route.length ? -1 : jobPointOffset + route[pos + 1];
  const before = distancePoints(prevPoint, jobPoint) + (nextPoint < 0 ? 0 : distancePoints(jobPoint, nextPoint));
  const after = nextPoint < 0 ? 0 : distancePoints(prevPoint, nextPoint);
  return (before >= INF ? 100000000 : before) - (after >= INF ? 100000000 : after);
}

const maxImprovePasses = Math.min(8, Math.max(2, Math.floor(20000 / Math.max(1, jobs.length))));
for (let pass = 0; pass < maxImprovePasses; pass++) {
  let changed = false;
  for (let fromR = 0; fromR < robotCount; fromR++) {
    for (let fromPos = 0; fromPos < routes[fromR].length; fromPos++) {
      const jobIndex = routes[fromR][fromPos];
      const saving = removalSaving(routes[fromR], fromR, fromPos);
      let bestGain = 0, bestR = fromR, bestP = fromPos;

      for (let toR = 0; toR < robotCount; toR++) {
        const baseRoute = toR === fromR
          ? routes[toR].slice(0, fromPos).concat(routes[toR].slice(fromPos + 1))
          : routes[toR];
        for (let toPos = 0; toPos <= baseRoute.length; toPos++) {
          if (toR === fromR && (toPos === fromPos || toPos === fromPos + 1)) continue;
          const delta = insertionDelta(baseRoute, toR, jobIndex, toPos);
          const gain = saving - delta;
          if (gain > bestGain + 1e-9) {
            bestGain = gain;
            bestR = toR;
            bestP = toPos;
          }
        }
      }

      if (bestGain > 0) {
        routes[fromR].splice(fromPos, 1);
        routes[bestR].splice(bestP, 0, jobIndex);
        changed = true;
        if (bestR === fromR && bestP <= fromPos) fromPos++;
      }
    }
  }
  if (!changed) break;
}

for (let r = 0; r < robotCount; r++) {
  let improved = true;
  let guard = 0;
  while (improved && guard++ < 3) {
    improved = false;
    const route = routes[r];
    for (let i = 0; i + 1 < route.length; i++) {
      const aPrev = i === 0 ? r : jobPointOffset + route[i - 1];
      const a = jobPointOffset + route[i];
      const b = jobPointOffset + route[i + 1];
      const bNext = i + 2 >= route.length ? -1 : jobPointOffset + route[i + 2];
      const oldCost = distancePoints(aPrev, a) + distancePoints(a, b) + (bNext < 0 ? 0 : distancePoints(b, bNext));
      const newCost = distancePoints(aPrev, b) + distancePoints(b, a) + (bNext < 0 ? 0 : distancePoints(a, bNext));
      if (newCost < oldCost) {
        const tmp = route[i]; route[i] = route[i + 1]; route[i + 1] = tmp;
        improved = true;
      }
    }
  }
}

let maxRobotId = -1;
for (const robot of robots) if (robot.id > maxRobotId) maxRobotId = robot.id;
const assignments: number[][] = Array.from({ length: Math.max(0, maxRobotId + 1) }, () => []);
for (let r = 0; r < robotCount; r++) {
  assignments[robots[r].id] = routes[r].map(i => jobs[i].id);
}
process.stdout.write(JSON.stringify({ assignments }));
