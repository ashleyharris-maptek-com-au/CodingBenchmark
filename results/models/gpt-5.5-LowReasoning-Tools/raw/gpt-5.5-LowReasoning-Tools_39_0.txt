use std::io::{self, Read};

struct Solver {
    n: usize,
    half: usize,
    adj: Vec<Vec<usize>>,
    order: Vec<usize>,
    side: Vec<i8>,
    best_cut: i32,
    best_side: Vec<i8>,
}

impl Solver {
    fn new(n: usize, edges: &[(usize, usize)]) -> Self {
        let mut adj = vec![Vec::new(); n];
        for &(u, v) in edges {
            if u != v {
                adj[u].push(v);
                adj[v].push(u);
            }
        }

        let mut order: Vec<usize> = (0..n).collect();
        order.sort_by_key(|&v| std::cmp::Reverse(adj[v].len()));
        let mut solver = Self {
            n,
            half: n / 2,
            adj,
            order,
            side: vec![-1; n],
            best_cut: i32::MAX,
            best_side: vec![-1; n],
        };
        solver.make_initial_solution();
        solver
    }

    fn cut_of_side(&self, side: &[i8]) -> i32 {
        let mut cut = 0;
        for u in 0..self.n {
            for &v in &self.adj[u] {
                if u < v && side[u] != side[v] {
                    cut += 1;
                }
            }
        }
        cut
    }

    fn make_initial_solution(&mut self) {
        let mut side = vec![1i8; self.n];
        for i in 0..self.half {
            side[self.order[i]] = 0;
        }

        let mut improved = true;
        while improved {
            improved = false;
            let mut best_delta = 0;
            let mut best_pair = (usize::MAX, usize::MAX);

            for a in 0..self.n {
                if side[a] != 0 {
                    continue;
                }
                for b in 0..self.n {
                    if side[b] != 1 {
                        continue;
                    }
                    side[a] = 1;
                    side[b] = 0;
                    let after = self.cut_of_side(&side);
                    side[a] = 0;
                    side[b] = 1;
                    let before = self.cut_of_side(&side);
                    let delta = after - before;
                    if delta < best_delta {
                        best_delta = delta;
                        best_pair = (a, b);
                    }
                }
            }

            if best_pair.0 != usize::MAX {
                side[best_pair.0] = 1;
                side[best_pair.1] = 0;
                improved = true;
            }
        }

        self.best_cut = self.cut_of_side(&side);
        self.best_side = side;
    }

    fn lower_bound(&self, idx: usize, a_count: usize, current_cut: i32) -> i32 {
        let need_a = self.half - a_count;
        let remaining = self.n - idx;
        if need_a > remaining {
            return i32::MAX / 4;
        }

        let mut base = 0i32;
        let mut diffs = Vec::with_capacity(remaining);

        for j in idx..self.n {
            let v = self.order[j];
            let mut cost_if_a = 0i32;
            let mut cost_if_b = 0i32;
            for &u in &self.adj[v] {
                match self.side[u] {
                    0 => cost_if_b += 1,
                    1 => cost_if_a += 1,
                    _ => {}
                }
            }
            base += cost_if_b;
            diffs.push(cost_if_a - cost_if_b);
        }

        diffs.sort_unstable();
        current_cut + base + diffs.iter().take(need_a).sum::<i32>()
    }

    fn added_cut(&self, v: usize, s: i8) -> i32 {
        let mut add = 0;
        for &u in &self.adj[v] {
            if self.side[u] >= 0 && self.side[u] != s {
                add += 1;
            }
        }
        add
    }

    fn search(&mut self, idx: usize, a_count: usize, current_cut: i32) {
        if current_cut >= self.best_cut {
            return;
        }
        if idx == self.n {
            if a_count == self.half {
                self.best_cut = current_cut;
                self.best_side.clone_from(&self.side);
            }
            return;
        }
        if a_count > self.half || a_count + (self.n - idx) < self.half {
            return;
        }
        if self.lower_bound(idx, a_count, current_cut) >= self.best_cut {
            return;
        }

        let v = self.order[idx];

        let try_a_first = a_count < self.half;
        if try_a_first {
            let add = self.added_cut(v, 0);
            self.side[v] = 0;
            self.search(idx + 1, a_count + 1, current_cut + add);
            self.side[v] = -1;
        }

        let b_count = idx - a_count;
        if b_count < self.half {
            let add = self.added_cut(v, 1);
            self.side[v] = 1;
            self.search(idx + 1, a_count, current_cut + add);
            self.side[v] = -1;
        }
    }

    fn solve(&mut self) {
        if self.n == 0 {
            self.best_cut = 0;
            self.best_side.clear();
            return;
        }

        // Symmetry breaking: either side can be printed as partition A, so force
        // the first branching vertex into A.
        let first = self.order[0];
        self.side[first] = 0;
        self.search(1, 1, 0);
        self.side[first] = -1;
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();
    let m: usize = it.next().unwrap().parse().unwrap();
    let mut edges = Vec::with_capacity(m);
    for _ in 0..m {
        let u: usize = it.next().unwrap().parse().unwrap();
        let v: usize = it.next().unwrap().parse().unwrap();
        edges.push((u, v));
    }

    let mut solver = Solver::new(n, &edges);
    solver.solve();

    let mut part = Vec::with_capacity(n / 2);
    for v in 0..n {
        if solver.best_side[v] == 0 {
            part.push(v);
        }
    }

    println!("{}", solver.best_cut);
    for (i, v) in part.iter().enumerate() {
        if i > 0 {
            print!(" ");
        }
        print!("{}", v);
    }
    println!();
}
