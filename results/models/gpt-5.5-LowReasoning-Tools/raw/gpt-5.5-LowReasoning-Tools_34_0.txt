import sys
import math
import random


def euclid(a, b):
    return math.hypot(a[0] - b[0], a[1] - b[1])


def route_distance(route, dist):
    if not route:
        return 0.0
    total = dist[0][route[0] + 1]
    for a, b in zip(route, route[1:]):
        total += dist[a + 1][b + 1]
    total += dist[route[-1] + 1][0]
    return total


def total_distance(routes, dist):
    return sum(route_distance(r, dist) for r in routes)


def route_load(route, demands):
    return sum(demands[i] for i in route)


def two_opt(route, dist, max_passes=50):
    n = len(route)
    if n < 4:
        return route

    route = route[:]
    for _ in range(max_passes):
        improved = False
        for i in range(n - 1):
            a = 0 if i == 0 else route[i - 1] + 1
            b = route[i] + 1
            for k in range(i + 1, n):
                c = route[k] + 1
                d = 0 if k == n - 1 else route[k + 1] + 1
                old = dist[a][b] + dist[c][d]
                new = dist[a][c] + dist[b][d]
                if new + 1e-9 < old:
                    route[i:k + 1] = reversed(route[i:k + 1])
                    improved = True
        if not improved:
            break
    return route


def nearest_order(customers, dist):
    if len(customers) <= 2:
        return customers[:]
    left = set(customers)
    cur = 0
    out = []
    while left:
        nxt = min(left, key=lambda c: dist[cur][c + 1])
        left.remove(nxt)
        out.append(nxt)
        cur = nxt + 1
    return out


def improve_routes(routes, demands, cap, dist, time_limit):
    routes = [two_opt(nearest_order(r, dist), dist) for r in routes if r]
    loads = [route_load(r, demands) for r in routes]

    # Relocate single customers between routes when it reduces distance.
    changed = True
    while changed and now() < time_limit:
        changed = False
        for ri in range(len(routes)):
            if changed:
                break
            r = routes[ri]
            for pos, cust in enumerate(r):
                if changed:
                    break
                old_r = r[:pos] + r[pos + 1:]
                old_cost = route_distance(r, dist) + route_distance(routes[ri], dist)
                for rj in range(len(routes)):
                    if ri == rj or loads[rj] + demands[cust] > cap:
                        continue
                    base_cost = route_distance(routes[ri], dist) + route_distance(routes[rj], dist)
                    target = routes[rj]
                    best_pos = None
                    best_delta = 0.0
                    for ins in range(len(target) + 1):
                        new_t = target[:ins] + [cust] + target[ins:]
                        delta = route_distance(old_r, dist) + route_distance(new_t, dist) - base_cost
                        if delta < best_delta:
                            best_delta = delta
                            best_pos = ins
                    if best_pos is not None:
                        routes[ri] = old_r
                        routes[rj] = target[:best_pos] + [cust] + target[best_pos:]
                        loads[ri] -= demands[cust]
                        loads[rj] += demands[cust]
                        changed = True
                        break

    return [two_opt(r, dist) for r in routes if r]


def clarke_wright(n, k, cap, demands, dist):
    routes = [[i] for i in range(n)]
    loads = [demands[i] for i in range(n)]
    where = list(range(n))

    savings = []
    for i in range(n):
        for j in range(i + 1, n):
            savings.append((dist[0][i + 1] + dist[0][j + 1] - dist[i + 1][j + 1], i, j))
    savings.sort(reverse=True)

    for _, i, j in savings:
        ri, rj = where[i], where[j]
        if ri == rj or not routes[ri] or not routes[rj]:
            continue
        if loads[ri] + loads[rj] > cap:
            continue
        a, b = routes[ri], routes[rj]
        merged = None
        if a[-1] == i and b[0] == j:
            merged = a + b
        elif a[0] == i and b[-1] == j:
            merged = b + a
        elif a[0] == i and b[0] == j:
            merged = list(reversed(a)) + b
        elif a[-1] == i and b[-1] == j:
            merged = a + list(reversed(b))
        if merged is None:
            continue
        routes[ri] = merged
        loads[ri] += loads[rj]
        routes[rj] = []
        loads[rj] = 0
        for c in merged:
            where[c] = ri

    result = [r for r in routes if r]
    if len(result) <= k:
        return result
    return None


def sweep_routes(n, k, cap, coords, demands, dist):
    depot = coords[0]
    angles = [
        (math.atan2(coords[i + 1][1] - depot[1], coords[i + 1][0] - depot[0]), i)
        for i in range(n)
    ]
    angles.sort()
    order = [i for _, i in angles]
    candidates = []
    step = max(1, n // min(n, 24)) if n else 1

    for shift in range(0, n, step):
        seq = order[shift:] + order[:shift]
        routes = []
        cur = []
        load = 0
        for c in seq:
            if cur and load + demands[c] > cap:
                routes.append(cur)
                cur = []
                load = 0
            cur.append(c)
            load += demands[c]
        if cur:
            routes.append(cur)
        if len(routes) <= k:
            routes = [two_opt(nearest_order(r, dist), dist) for r in routes]
            candidates.append(routes)

    if candidates:
        return min(candidates, key=lambda rs: total_distance(rs, dist))
    return None


def best_fit_decreasing(n, k, cap, demands, dist):
    items = sorted(range(n), key=lambda i: (-demands[i], dist[0][i + 1]))
    routes = [[] for _ in range(k)]
    loads = [0] * k

    for c in items:
        choices = [r for r in range(k) if loads[r] + demands[c] <= cap]
        if not choices:
            return None
        # Prefer fuller bins, then close insertion cost.
        def score(r):
            if not routes[r]:
                inc = 2 * dist[0][c + 1]
            else:
                inc = min(
                    dist[(0 if p == 0 else routes[r][p - 1] + 1)][c + 1]
                    + dist[c + 1][(0 if p == len(routes[r]) else routes[r][p] + 1)]
                    - dist[(0 if p == 0 else routes[r][p - 1] + 1)][(0 if p == len(routes[r]) else routes[r][p] + 1)]
                    for p in range(len(routes[r]) + 1)
                )
            return (cap - loads[r] - demands[c], inc)

        r = min(choices, key=score)
        best_p = 0
        best_inc = float("inf")
        for p in range(len(routes[r]) + 1):
            a = 0 if p == 0 else routes[r][p - 1] + 1
            b = 0 if p == len(routes[r]) else routes[r][p] + 1
            inc = dist[a][c + 1] + dist[c + 1][b] - dist[a][b]
            if inc < best_inc:
                best_inc = inc
                best_p = p
        routes[r].insert(best_p, c)
        loads[r] += demands[c]

    return [two_opt(r, dist) for r in routes if r]


def validate(routes, n, k, cap, demands):
    if len(routes) > k:
        return False
    seen = []
    for r in routes:
        if route_load(r, demands) > cap + 1e-9:
            return False
        seen.extend(r)
    return sorted(seen) == list(range(n))


def now():
    try:
        return __import__("time").perf_counter()
    except Exception:
        return 0.0


def solve():
    text = sys.stdin.read().replace(chr(65279), "").replace(chr(239) + chr(187) + chr(191), "")
    data = text.strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    k = int(next(it))
    cap = int(next(it))
    depot = (float(next(it)), float(next(it)))
    coords = [depot]
    demands = []
    for _ in range(n):
        x = float(next(it))
        y = float(next(it))
        d = int(next(it))
        coords.append((x, y))
        demands.append(d)

    m = n + 1
    dist = [[0.0] * m for _ in range(m)]
    for i in range(m):
        for j in range(i + 1, m):
            d = euclid(coords[i], coords[j])
            dist[i][j] = dist[j][i] = d

    time_limit = now() + (1.8 if n <= 400 else 4.5)
    candidates = []

    for builder in (
        lambda: clarke_wright(n, k, cap, demands, dist),
        lambda: sweep_routes(n, k, cap, coords, demands, dist),
        lambda: best_fit_decreasing(n, k, cap, demands, dist),
    ):
        routes = builder()
        if routes is not None and validate(routes, n, k, cap, demands):
            candidates.append(routes)

    if not candidates:
        # Last-resort greedy packing. Feasible benchmark cases should rarely need this.
        unvisited = set(range(n))
        routes = []
        while unvisited and len(routes) < k:
            route = []
            load = 0
            cur = 0
            while True:
                feasible = [c for c in unvisited if load + demands[c] <= cap]
                if not feasible:
                    break
                c = min(feasible, key=lambda x: dist[cur][x + 1])
                route.append(c)
                unvisited.remove(c)
                load += demands[c]
                cur = c + 1
            if not route:
                break
            routes.append(route)
        if unvisited:
            for c in list(unvisited):
                for r in routes:
                    if route_load(r, demands) + demands[c] <= cap:
                        r.append(c)
                        unvisited.remove(c)
                        break
        candidates.append(routes)

    best = min(candidates, key=lambda rs: total_distance(rs, dist))
    if validate(best, n, k, cap, demands):
        best = improve_routes(best, demands, cap, dist, time_limit)

    print(f"{total_distance(best, dist):.6f}")
    print(len(best))
    for r in best:
        print(" ".join(map(str, r)))


if __name__ == "__main__":
    solve()
