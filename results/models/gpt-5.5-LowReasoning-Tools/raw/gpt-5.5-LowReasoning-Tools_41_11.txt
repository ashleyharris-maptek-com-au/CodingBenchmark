; SPIR-V
; Version: 1.0
OpCapability Shader
%glsl = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %worldPos %normal %uv %outColor
OpExecutionMode %main OriginUpperLeft
OpSource GLSL 450
OpName %main "main"
OpName %worldPos "worldPos"
OpName %normal "normal"
OpName %uv "uv"
OpName %outColor "outColor"
OpName %UBO "UBO"
OpMemberName %UBO 0 "model"
OpMemberName %UBO 1 "view"
OpMemberName %UBO 2 "projection"
OpMemberName %UBO 3 "lightPos"
OpMemberName %UBO 4 "cameraPos"
OpMemberName %UBO 5 "params"
OpName %ubo "ubo"
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %uv Location 2
OpDecorate %outColor Location 0
OpDecorate %v4floatarr4 ArrayStride 16
OpMemberDecorate %UBO 0 ColMajor
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 0 MatrixStride 16
OpMemberDecorate %UBO 1 ColMajor
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 1 MatrixStride 16
OpMemberDecorate %UBO 2 ColMajor
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 2 MatrixStride 16
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %UBO Block
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0
%void = OpTypeVoid
%fun = OpTypeFunction %void
%float = OpTypeFloat 32
%bool = OpTypeBool
%v3bool = OpTypeVector %bool 3
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%uint_4 = OpConstant %uint 4
%v4floatarr4 = OpTypeArray %v4float %uint_4
%UBO = OpTypeStruct %v4floatarr4 %v4floatarr4 %v4floatarr4 %v4float %v4float %v4float
%ptr_Input_v3float = OpTypePointer Input %v3float
%ptr_Input_v2float = OpTypePointer Input %v2float
%ptr_Output_v4float = OpTypePointer Output %v4float
%ptr_Uniform_UBO = OpTypePointer Uniform %UBO
%ptr_Uniform_v4float = OpTypePointer Uniform %v4float
%worldPos = OpVariable %ptr_Input_v3float Input
%normal = OpVariable %ptr_Input_v3float Input
%uv = OpVariable %ptr_Input_v2float Input
%outColor = OpVariable %ptr_Output_v4float Output
%ubo = OpVariable %ptr_Uniform_UBO Uniform
%int = OpTypeInt 32 1
%int_3 = OpConstant %int 3
%float_0 = OpConstant %float 0
%float_0_1 = OpConstant %float 0.1
%float_0_3 = OpConstant %float 0.3
%float_0_5 = OpConstant %float 0.5
%float_0_7 = OpConstant %float 0.7
%float_0_8 = OpConstant %float 0.8
%float_1 = OpConstant %float 1
%float_6 = OpConstant %float 6
%v2float_6 = OpConstantComposite %v2float %float_6 %float_6
%v2float_0_5 = OpConstantComposite %v2float %float_0_5 %float_0_5
%dotColor = OpConstantComposite %v3float %float_1 %float_0_8 %float_0
%bgColor = OpConstantComposite %v3float %float_0_1 %float_0_1 %float_0_3
%main = OpFunction %void None %fun
%entry = OpLabel
%uvVal = OpLoad %v2float %uv
%scaledUv = OpFMul %v2float %uvVal %v2float_6
%cellUv = OpExtInst %v2float %glsl Fract %scaledUv
%centered = OpFSub %v2float %cellUv %v2float_0_5
%dist = OpExtInst %float %glsl Length %centered
%inDot = OpFOrdLessThanEqual %bool %dist %float_0_3
%inDot3 = OpCompositeConstruct %v3bool %inDot %inDot %inDot
%baseColor = OpSelect %v3float %inDot3 %dotColor %bgColor
%n0 = OpLoad %v3float %normal
%n = OpExtInst %v3float %glsl Normalize %n0
%lightPtr = OpAccessChain %ptr_Uniform_v4float %ubo %int_3
%light4 = OpLoad %v4float %lightPtr
%light3 = OpVectorShuffle %v3float %light4 %light4 0 1 2
%pos = OpLoad %v3float %worldPos
%lightVec = OpFSub %v3float %light3 %pos
%l = OpExtInst %v3float %glsl Normalize %lightVec
%ndotl = OpDot %float %n %l
%diffuse = OpExtInst %float %glsl FMax %ndotl %float_0
%litPart = OpFMul %float %float_0_7 %diffuse
%shade = OpFAdd %float %float_0_3 %litPart
%shadedRgb = OpVectorTimesScalar %v3float %baseColor %shade
%r = OpCompositeExtract %float %shadedRgb 0
%g = OpCompositeExtract %float %shadedRgb 1
%b = OpCompositeExtract %float %shadedRgb 2
%finalColor = OpCompositeConstruct %v4float %r %g %b %float_1
OpStore %outColor %finalColor
OpReturn
OpFunctionEnd
