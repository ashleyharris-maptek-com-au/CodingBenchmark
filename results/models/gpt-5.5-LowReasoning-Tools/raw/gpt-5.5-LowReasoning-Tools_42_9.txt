OpCapability Geometry
%GLSL450 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main InputLinesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1
OpSource GLSL 450
OpName %main "main"
OpName %gl_PerVertex_in "gl_PerVertex_in"
OpMemberName %gl_PerVertex_in 0 "gl_Position"
OpName %gl_in "gl_in"
OpName %worldPos "worldPos"
OpName %normal "normal"
OpName %color "color"
OpName %gl_PerVertex_out "gl_PerVertex_out"
OpMemberName %gl_PerVertex_out 0 "gl_Position"
OpName %gl_out "gl_out"
OpName %outColor "outColor"
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %outColor Location 0
%void = OpTypeVoid
%fn_void = OpTypeFunction %void
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3
%uint_4 = OpConstant %uint 4
%float_0_025 = OpConstant %float 0.025
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%arr_v3_in = OpTypeArray %v3float %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%gl_in = OpVariable %ptr_in_arr_pv Input
%worldPos = OpVariable %ptr_in_arr_v3 Input
%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%gl_out = OpVariable %ptr_out_pv Output
%outColor = OpVariable %ptr_out_v3 Output
%main = OpFunction %void None %fn_void
%entry = OpLabel
%p0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%p1_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%p2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%p3_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
%p0 = OpLoad %v4float %p0_ptr
%p1 = OpLoad %v4float %p1_ptr
%p2 = OpLoad %v4float %p2_ptr
%p3 = OpLoad %v4float %p3_ptr
%p0_xy = OpVectorShuffle %v2float %p0 %p0 0 1
%p1_xy = OpVectorShuffle %v2float %p1 %p1 0 1
%p2_xy = OpVectorShuffle %v2float %p2 %p2 0 1
%p3_xy = OpVectorShuffle %v2float %p3 %p3 0 1
%d0 = OpFSub %v2float %p2_xy %p0_xy
%d1 = OpFSub %v2float %p3_xy %p1_xy
%t0 = OpExtInst %v2float %GLSL450 Normalize %d0
%t1 = OpExtInst %v2float %GLSL450 Normalize %d1
%t0_x = OpCompositeExtract %float %t0 0
%t0_y = OpCompositeExtract %float %t0 1
%t1_x = OpCompositeExtract %float %t1 0
%t1_y = OpCompositeExtract %float %t1 1
%neg_t0_y = OpFNegate %float %t0_y
%neg_t1_y = OpFNegate %float %t1_y
%perp0 = OpCompositeConstruct %v2float %neg_t0_y %t0_x
%perp1 = OpCompositeConstruct %v2float %neg_t1_y %t1_x
%off0 = OpVectorTimesScalar %v2float %perp0 %float_0_025
%off1 = OpVectorTimesScalar %v2float %perp1 %float_0_025
%start_plus_xy = OpFAdd %v2float %p1_xy %off0
%start_minus_xy = OpFSub %v2float %p1_xy %off0
%end_plus_xy = OpFAdd %v2float %p2_xy %off1
%end_minus_xy = OpFSub %v2float %p2_xy %off1
%p1_z = OpCompositeExtract %float %p1 2
%p1_w = OpCompositeExtract %float %p1 3
%p2_z = OpCompositeExtract %float %p2 2
%p2_w = OpCompositeExtract %float %p2 3
%start_plus_x = OpCompositeExtract %float %start_plus_xy 0
%start_plus_y = OpCompositeExtract %float %start_plus_xy 1
%start_minus_x = OpCompositeExtract %float %start_minus_xy 0
%start_minus_y = OpCompositeExtract %float %start_minus_xy 1
%end_plus_x = OpCompositeExtract %float %end_plus_xy 0
%end_plus_y = OpCompositeExtract %float %end_plus_xy 1
%end_minus_x = OpCompositeExtract %float %end_minus_xy 0
%end_minus_y = OpCompositeExtract %float %end_minus_xy 1
%start_plus_pos = OpCompositeConstruct %v4float %start_plus_x %start_plus_y %p1_z %p1_w
%start_minus_pos = OpCompositeConstruct %v4float %start_minus_x %start_minus_y %p1_z %p1_w
%end_plus_pos = OpCompositeConstruct %v4float %end_plus_x %end_plus_y %p2_z %p2_w
%end_minus_pos = OpCompositeConstruct %v4float %end_minus_x %end_minus_y %p2_z %p2_w
%color1_ptr = OpAccessChain %ptr_in_v3 %color %int_1
%color2_ptr = OpAccessChain %ptr_in_v3 %color %int_2
%color1 = OpLoad %v3float %color1_ptr
%color2 = OpLoad %v3float %color2_ptr
%out_pos_ptr0 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_ptr0 %start_plus_pos
OpStore %outColor %color1
OpEmitVertex
%out_pos_ptr1 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_ptr1 %start_minus_pos
OpStore %outColor %color1
OpEmitVertex
%out_pos_ptr2 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_ptr2 %end_plus_pos
OpStore %outColor %color2
OpEmitVertex
%out_pos_ptr3 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_ptr3 %end_minus_pos
OpStore %outColor %color2
OpEmitVertex
OpEndPrimitive
OpReturn
OpFunctionEnd
