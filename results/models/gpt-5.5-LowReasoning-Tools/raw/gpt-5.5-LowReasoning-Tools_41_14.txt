; SPIR-V
; Version: 1.0
OpCapability Shader
%1 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %inWorldPos %inNormal %outColor
OpExecutionMode %main OriginUpperLeft
OpSource GLSL 450
OpName %main "main"
OpName %inWorldPos "worldPos"
OpName %inNormal "normal"
OpName %outColor "outColor"
OpName %UBO "UBO"
OpMemberName %UBO 0 "model"
OpMemberName %UBO 1 "view"
OpMemberName %UBO 2 "projection"
OpMemberName %UBO 3 "lightPos"
OpMemberName %UBO 4 "cameraPos"
OpMemberName %UBO 5 "params"
OpName %ubo "ubo"
OpDecorate %inWorldPos Location 0
OpDecorate %inNormal Location 1
OpDecorate %outColor Location 0
OpMemberDecorate %UBO 0 ColMajor
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 0 MatrixStride 16
OpMemberDecorate %UBO 1 ColMajor
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 1 MatrixStride 16
OpMemberDecorate %UBO 2 ColMajor
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 2 MatrixStride 16
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %UBO Block
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0
%void = OpTypeVoid
%fun = OpTypeFunction %void
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%uint_4 = OpConstant %uint 4
%mat4 = OpTypeMatrix %v4float 4
%UBO = OpTypeStruct %mat4 %mat4 %mat4 %v4float %v4float %v4float
%ptr_Input_v3float = OpTypePointer Input %v3float
%ptr_Output_v4float = OpTypePointer Output %v4float
%ptr_Uniform_UBO = OpTypePointer Uniform %UBO
%ptr_Uniform_v4float = OpTypePointer Uniform %v4float
%inWorldPos = OpVariable %ptr_Input_v3float Input
%inNormal = OpVariable %ptr_Input_v3float Input
%outColor = OpVariable %ptr_Output_v4float Output
%ubo = OpVariable %ptr_Uniform_UBO Uniform
%int = OpTypeInt 32 1
%int_3 = OpConstant %int 3
%int_4 = OpConstant %int 4
%float_0 = OpConstant %float 0
%float_1 = OpConstant %float 1
%float_0_2 = OpConstant %float 0.2
%float_0_3 = OpConstant %float 0.3
%float_0_7 = OpConstant %float 0.7
%float_0_8 = OpConstant %float 0.8
%float_1_5 = OpConstant %float 1.5
%float_3 = OpConstant %float 3
%baseColor = OpConstantComposite %v3float %float_0_8 %float_0_2 %float_0_2
%fogColor = OpConstantComposite %v3float %float_0_7 %float_0_7 %float_0_8
%main = OpFunction %void None %fun
%entry = OpLabel
%worldPos = OpLoad %v3float %inWorldPos
%normal = OpLoad %v3float %inNormal
%lightPosPtr = OpAccessChain %ptr_Uniform_v4float %ubo %int_3
%lightPos4 = OpLoad %v4float %lightPosPtr
%lightPos = OpVectorShuffle %v3float %lightPos4 %lightPos4 0 1 2
%cameraPosPtr = OpAccessChain %ptr_Uniform_v4float %ubo %int_4
%cameraPos4 = OpLoad %v4float %cameraPosPtr
%cameraPos = OpVectorShuffle %v3float %cameraPos4 %cameraPos4 0 1 2
%n = OpExtInst %v3float %1 Normalize %normal
%toLight = OpFSub %v3float %lightPos %worldPos
%lightDir = OpExtInst %v3float %1 Normalize %toLight
%dotNL = OpDot %float %n %lightDir
%diffuse = OpExtInst %float %1 FMax %dotNL %float_0
%diffuseScaled = OpFMul %float %float_0_7 %diffuse
%lambert = OpFAdd %float %float_0_3 %diffuseScaled
%litColor = OpVectorTimesScalar %v3float %baseColor %lambert
%toCamera = OpFSub %v3float %cameraPos %worldPos
%dist = OpExtInst %float %1 Length %toCamera
%fogNumer = OpFSub %float %dist %float_1_5
%fogRaw = OpFDiv %float %fogNumer %float_3
%fogFactor = OpExtInst %float %1 FClamp %fogRaw %float_0 %float_1
%invFog = OpFSub %float %float_1 %fogFactor
%litPart = OpVectorTimesScalar %v3float %litColor %invFog
%fogPart = OpVectorTimesScalar %v3float %fogColor %fogFactor
%rgb = OpFAdd %v3float %litPart %fogPart
%r = OpCompositeExtract %float %rgb 0
%g = OpCompositeExtract %float %rgb 1
%b = OpCompositeExtract %float %rgb 2
%color = OpCompositeConstruct %v4float %r %g %b %float_1
OpStore %outColor %color
OpReturn
OpFunctionEnd
