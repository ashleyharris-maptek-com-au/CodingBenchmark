const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const grid = Array.isArray(data.probability) ? data.probability : [];
const drones = Array.isArray(data.drones) ? data.drones : [];
const maxSteps = Math.max(0, data.maxStepsPerDrone | 0);
const H = grid.length;
const W = H ? grid[0].length : 0;
const N = W * H;

function idOf(x, y) { return y * W + x; }
function xOf(id) { return id % W; }
function yOf(id) { return (id / W) | 0; }

if (H === 0 || W === 0 || drones.length === 0) {
  process.stdout.write(JSON.stringify({ paths: drones.map(() => []) }));
  process.exit(0);
}

const val = new Float64Array(N);
const pass = new Uint8Array(N);
let totalPositive = 0;
let positiveCount = 0;

for (let y = 0; y < H; y++) {
  for (let x = 0; x < W; x++) {
    const id = idOf(x, y);
    const v = Number(grid[y][x]);
    if (v !== -1) {
      pass[id] = 1;
      if (v > 0) {
        val[id] = v;
        totalPositive += v;
        positiveCount++;
      }
    }
  }
}

const neigh = new Int32Array(N * 4);
const deg = new Uint8Array(N);
for (let y = 0; y < H; y++) {
  for (let x = 0; x < W; x++) {
    const id = idOf(x, y);
    if (!pass[id]) continue;
    let k = id * 4;
    if (y > 0 && pass[id - W]) neigh[k + deg[id]++] = id - W;
    if (y + 1 < H && pass[id + W]) neigh[k + deg[id]++] = id + W;
    if (x > 0 && pass[id - 1]) neigh[k + deg[id]++] = id - 1;
    if (x + 1 < W && pass[id + 1]) neigh[k + deg[id]++] = id + 1;
  }
}

const covered = new Uint8Array(N);
for (const d of drones) {
  const x = d.x | 0, y = d.y | 0;
  if (x >= 0 && x < W && y >= 0 && y < H) covered[idOf(x, y)] = 1;
}

const starts = drones.map(d => {
  const x = Math.max(0, Math.min(W - 1, d.x | 0));
  const y = Math.max(0, Math.min(H - 1, d.y | 0));
  let id = idOf(x, y);
  if (pass[id]) return id;
  for (let r = 1; r < Math.max(W, H); r++) {
    for (let yy = Math.max(0, y - r); yy <= Math.min(H - 1, y + r); yy++) {
      for (let xx = Math.max(0, x - r); xx <= Math.min(W - 1, x + r); xx++) {
        id = idOf(xx, yy);
        if (pass[id]) return id;
      }
    }
  }
  return idOf(x, y);
});

const dist = new Int32Array(N);
const parent = new Int32Array(N);
const seen = new Int32Array(N);
const queue = new Int32Array(N);
let stamp = 1;

function clusterScore(cell) {
  if (covered[cell] || val[cell] <= 0) return 0;
  let s = val[cell] * 1.8;
  const x = xOf(cell), y = yOf(cell);
  for (let dy = -2; dy <= 2; dy++) {
    const yy = y + dy;
    if (yy < 0 || yy >= H) continue;
    for (let dx = -2; dx <= 2; dx++) {
      const xx = x + dx;
      if (xx < 0 || xx >= W) continue;
      const md = Math.abs(dx) + Math.abs(dy);
      if (md === 0 || md > 2) continue;
      const n = idOf(xx, yy);
      if (!covered[n] && pass[n] && val[n] > 0) s += val[n] / (md + 1);
    }
  }
  return s;
}

function chooseTarget(start, remaining) {
  if (!pass[start] || remaining <= 0) return -1;
  if (++stamp === 2147483000) {
    seen.fill(0);
    stamp = 1;
  }
  let head = 0, tail = 0;
  queue[tail++] = start;
  seen[start] = stamp;
  dist[start] = 0;
  parent[start] = -1;
  let best = -1;
  let bestScore = -1;
  const maxSearch = remaining;

  while (head < tail) {
    const cur = queue[head++];
    const cd = dist[cur];
    if (cd > 0 && val[cur] > 0 && !covered[cur]) {
      const c = clusterScore(cur);
      if (c > 0) {
        const score = c / Math.pow(cd + 0.8, 0.72);
        if (score > bestScore) {
          bestScore = score;
          best = cur;
        }
      }
    }
    if (cd >= maxSearch) continue;
    const base = cur * 4;
    for (let i = 0; i < deg[cur]; i++) {
      const nb = neigh[base + i];
      if (seen[nb] === stamp) continue;
      seen[nb] = stamp;
      dist[nb] = cd + 1;
      parent[nb] = cur;
      queue[tail++] = nb;
    }
  }
  return best;
}

function bestOneStep(pos) {
  let best = pos;
  let bestScore = covered[pos] ? 0 : val[pos];
  const base = pos * 4;
  for (let i = 0; i < deg[pos]; i++) {
    const nb = neigh[base + i];
    let s = (!covered[nb] ? val[nb] * 2 : 0);
    const nbBase = nb * 4;
    for (let j = 0; j < deg[nb]; j++) {
      const nn = neigh[nbBase + j];
      if (!covered[nn]) s += val[nn] * 0.35;
    }
    if (s > bestScore) {
      bestScore = s;
      best = nb;
    }
  }
  return best;
}

function pathToTarget(target) {
  const rev = [];
  let p = target;
  while (parent[p] !== -1) {
    rev.push(p);
    p = parent[p];
  }
  rev.reverse();
  return rev;
}

const order = drones.map((_, i) => i).sort((a, b) => {
  const va = val[starts[a]] || 0, vb = val[starts[b]] || 0;
  return vb - va || a - b;
});

const pathsIds = drones.map((_, i) => [starts[i]]);
for (const id of starts) if (val[id] > 0) covered[id] = 1;

for (const di of order) {
  let pos = starts[di];
  let remaining = maxSteps;
  while (remaining > 0) {
    const target = chooseTarget(pos, remaining);
    let segment;
    if (target >= 0) {
      segment = pathToTarget(target);
    } else {
      const next = bestOneStep(pos);
      segment = [next];
    }
    if (segment.length === 0) segment = [pos];
    for (const step of segment) {
      if (remaining <= 0) break;
      pos = step;
      pathsIds[di].push(pos);
      if (val[pos] > 0) covered[pos] = 1;
      remaining--;
    }
  }
}

const paths = pathsIds.map(p => p.map(id => [xOf(id), yOf(id)]));
process.stdout.write(JSON.stringify({ paths }));
