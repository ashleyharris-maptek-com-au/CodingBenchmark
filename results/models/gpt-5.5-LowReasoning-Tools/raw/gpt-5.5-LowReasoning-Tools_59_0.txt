const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input.replace(/^\uFEFF/, ''));

const grid = data.probability || [];
const drones = data.drones || [];
const H = grid.length;
const W = H ? grid[0].length : 0;
const maxSteps = Math.max(0, data.maxStepsPerDrone | 0);
const N = W * H;

function idx(x, y) {
  return y * W + x;
}

function xy(i) {
  return [i % W, Math.floor(i / W)];
}

const value = new Float64Array(N);
const passable = new Uint8Array(N);
let totalPositive = 0;
for (let y = 0; y < H; y++) {
  for (let x = 0; x < W; x++) {
    const i = idx(x, y);
    const v = Number(grid[y][x]);
    if (v !== -1 && Number.isFinite(v)) {
      passable[i] = 1;
      if (v > 0) {
        value[i] = v;
        totalPositive++;
      }
    }
  }
}

const covered = new Uint8Array(N);
const seen = new Int32Array(N);
const dist = new Int32Array(N);
const parent = new Int32Array(N);
const queue = new Int32Array(Math.max(1, N));
let stamp = 1;

const dirs = [
  [1, 0],
  [-1, 0],
  [0, 1],
  [0, -1],
];

function isValidStart(x, y) {
  return x >= 0 && x < W && y >= 0 && y < H && passable[idx(x, y)];
}

function localBonus(cell) {
  const x = cell % W;
  const y = Math.floor(cell / W);
  let bonus = 0;
  for (let dy = -2; dy <= 2; dy++) {
    const yy = y + dy;
    if (yy < 0 || yy >= H) continue;
    const span = 2 - Math.abs(dy);
    for (let dx = -span; dx <= span; dx++) {
      const xx = x + dx;
      if (xx < 0 || xx >= W) continue;
      const j = idx(xx, yy);
      if (!covered[j] && value[j] > 0) {
        bonus += value[j] / (1 + Math.abs(dx) + Math.abs(dy));
      }
    }
  }
  return bonus;
}

function reconstruct(start, target) {
  const rev = [];
  let cur = target;
  while (cur !== start && cur >= 0) {
    rev.push(cur);
    cur = parent[cur];
  }
  rev.reverse();
  return rev;
}

function chooseTarget(start, remaining) {
  stamp++;
  if (stamp === 2147483647) {
    seen.fill(0);
    stamp = 1;
  }

  let head = 0;
  let tail = 0;
  queue[tail++] = start;
  seen[start] = stamp;
  dist[start] = 0;
  parent[start] = -1;

  let best = -1;
  let bestScore = 0;

  while (head < tail) {
    const cur = queue[head++];
    const d = dist[cur];
    const v = !covered[cur] ? value[cur] : 0;
    if (v > 0) {
      const bonus = localBonus(cur) * 0.35;
      const score = (v + bonus) / Math.pow(d + 1, 1.18);
      if (score > bestScore) {
        bestScore = score;
        best = cur;
      }
    }
    if (d >= remaining) continue;

    const x = cur % W;
    const y = Math.floor(cur / W);
    for (let k = 0; k < 4; k++) {
      const nx = x + dirs[k][0];
      const ny = y + dirs[k][1];
      if (nx < 0 || nx >= W || ny < 0 || ny >= H) continue;
      const ni = idx(nx, ny);
      if (!passable[ni] || seen[ni] === stamp) continue;
      seen[ni] = stamp;
      dist[ni] = d + 1;
      parent[ni] = cur;
      queue[tail++] = ni;
    }
  }

  return best;
}

function planDrone(drone) {
  let x = drone.x | 0;
  let y = drone.y | 0;
  if (!isValidStart(x, y)) {
    outer:
    for (let yy = 0; yy < H; yy++) {
      for (let xx = 0; xx < W; xx++) {
        if (passable[idx(xx, yy)]) {
          x = xx;
          y = yy;
          break outer;
        }
      }
    }
  }

  let cur = idx(x, y);
  const path = [xy(cur)];
  if (passable[cur]) covered[cur] = 1;

  let remaining = maxSteps;
  while (remaining > 0) {
    const target = chooseTarget(cur, remaining);
    if (target < 0) break;

    if (target === cur) {
      covered[cur] = 1;
      let moved = false;
      for (let k = 0; k < 4 && !moved; k++) {
        const nx = (cur % W) + dirs[k][0];
        const ny = Math.floor(cur / W) + dirs[k][1];
        if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
          const ni = idx(nx, ny);
          if (passable[ni]) {
            cur = ni;
            path.push(xy(cur));
            if (value[cur] > 0) covered[cur] = 1;
            remaining--;
            moved = true;
          }
        }
      }
      if (!moved) break;
      continue;
    }

    const route = reconstruct(cur, target);
    if (route.length === 0) break;
    for (const step of route) {
      if (remaining <= 0) break;
      cur = step;
      path.push(xy(cur));
      if (value[cur] > 0) covered[cur] = 1;
      remaining--;
    }
  }

  while (remaining > 0) {
    path.push(xy(cur));
    remaining--;
  }
  return path;
}

const paths = [];
for (let i = 0; i < drones.length; i++) {
  paths.push(planDrone(drones[i]));
}
