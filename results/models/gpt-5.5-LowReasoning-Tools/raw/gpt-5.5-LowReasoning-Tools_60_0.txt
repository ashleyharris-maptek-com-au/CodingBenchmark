const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

class MinHeap {
  constructor(cmp) {
    this.a = [];
    this.cmp = cmp;
  }
  get size() {
    return this.a.length;
  }
  push(x) {
    const a = this.a;
    a.push(x);
    let i = a.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (this.cmp(a[p], x) <= 0) break;
      a[i] = a[p];
      i = p;
    }
    a[i] = x;
  }
  pop() {
    const a = this.a;
    const root = a[0];
    const x = a.pop();
    if (a.length) {
      let i = 0;
      while (true) {
        let l = i * 2 + 1;
        if (l >= a.length) break;
        let r = l + 1;
        let c = r < a.length && this.cmp(a[r], a[l]) < 0 ? r : l;
        if (this.cmp(a[c], x) >= 0) break;
        a[i] = a[c];
        i = c;
      }
      a[i] = x;
    }
    return root;
  }
  peek() {
    return this.a[0];
  }
}

const tasksIn = Array.isArray(data.tasks) ? data.tasks : [];
const n = tasksIn.length;
const m = Math.max(1, Math.floor(Number(data.machineCount) || 1));

const idToIdx = new Map();
for (let i = 0; i < n; i++) idToIdx.set(String(tasksIn[i].id), i);

const ids = new Array(n);
const dur = new Array(n);
const deadline = new Array(n);
const weight = new Array(n);
const succ = Array.from({ length: n }, () => []);
const indeg0 = new Int32Array(n);
const depsIdx = Array.from({ length: n }, () => []);

for (let i = 0; i < n; i++) {
  const t = tasksIn[i];
  ids[i] = t.id;
  dur[i] = Math.max(0, Math.floor(Number(t.duration) || 0));
  deadline[i] = Number.isFinite(Number(t.deadline)) ? Number(t.deadline) : 0;
  weight[i] = Math.max(0, Number(t.weight) || 0);
}

for (let i = 0; i < n; i++) {
  const deps = Array.isArray(tasksIn[i].deps) ? tasksIn[i].deps : [];
  for (const depId of deps) {
    const j = idToIdx.get(String(depId));
    if (j === undefined || j === i) continue;
    succ[j].push(i);
    depsIdx[i].push(j);
    indeg0[i]++;
  }
}

const topo = [];
{
  const indeg = new Int32Array(indeg0);
  const q = [];
  for (let i = 0; i < n; i++) if (indeg[i] === 0) q.push(i);
  for (let h = 0; h < q.length; h++) {
    const u = q[h];
    topo.push(u);
    for (const v of succ[u]) if (--indeg[v] === 0) q.push(v);
  }
  for (let i = 0; i < n; i++) {
    if (indeg[i] > 0) topo.push(i);
  }
}

const bottom = new Array(n).fill(0);
const downWeight = new Array(n).fill(0);
for (let ti = topo.length - 1; ti >= 0; ti--) {
  const u = topo[ti];
  let b = dur[u];
  let w = weight[u];
  for (const v of succ[u]) {
    if (dur[u] + bottom[v] > b) b = dur[u] + bottom[v];
    w += downWeight[v] * 0.35;
  }
  bottom[u] = b;
  downWeight[u] = w;
}

const totalDur = dur.reduce((a, b) => a + b, 0);
const horizon = Math.max(1, Math.ceil(totalDur / m));
const avgDur = Math.max(1, totalDur / Math.max(1, n));
const maxWeight = Math.max(1, ...weight);

function priority(i, time, mode) {
  const d = dur[i] || 0.0001;
  const w = weight[i] || 0.0001;
  const dl = deadline[i];
  const finish = time + dur[i];
  const slack = dl - time - bottom[i];
  const tardyNow = Math.max(0, finish - dl);
  const density = w / Math.sqrt(d);
  const critical = bottom[i] / avgDur;
  const downstream = downWeight[i] / maxWeight;

  if (mode === 0) return -slack * 3 + density * 8 + critical * 1.5 + downstream;
  if (mode === 1) return -dl / avgDur + density * 5 + critical * 4;
  if (mode === 2) return tardyNow * w * 4 + density * 10 + critical * 2 - slack;
  if (mode === 3) return critical * 12 + downstream * 2 + density - Math.max(0, slack) * 0.2;
  if (mode === 4) return density * 12 - dl / avgDur + downstream * 1.5;
  return -slack * (1 + Math.log1p(w)) + critical * 4 + density * 4 + downstream * 0.5;
}

function buildSchedule(mode) {
  const indeg = new Int32Array(indeg0);
  const ready = [];
  const done = new Uint8Array(n);
  const finish = new Array(n).fill(0);
  const schedule = [];
  const machines = new MinHeap((a, b) => a.time - b.time || a.machine - b.machine);
  const running = new MinHeap((a, b) => a.finish - b.finish || a.machine - b.machine);

  for (let i = 0; i < m; i++) machines.push({ time: 0, machine: i });
  for (let i = 0; i < n; i++) if (indeg[i] === 0) ready.push(i);

  let completed = 0;
  let time = 0;

  while (completed < n) {
    while (running.size && running.peek().finish <= time) {
      const r = running.pop();
      if (done[r.task]) continue;
      done[r.task] = 1;
      completed++;
      finish[r.task] = r.finish;
      machines.push({ time: r.finish, machine: r.machine });
      for (const v of succ[r.task]) {
        if (--indeg[v] === 0) ready.push(v);
      }
    }

    while (machines.size && machines.peek().time < time) {
      const x = machines.pop();
      x.time = time;
      machines.push(x);
    }

    if (ready.length && machines.size && machines.peek().time <= time) {
      ready.sort((a, b) => {
        const pa = priority(a, time, mode);
        const pb = priority(b, time, mode);
        if (pa !== pb) return pa - pb;
        if (deadline[a] !== deadline[b]) return deadline[b] - deadline[a];
        return String(ids[b]).localeCompare(String(ids[a]));
      });

      while (ready.length && machines.size && machines.peek().time <= time) {
        const task = ready.pop();
        const mach = machines.pop();
        const st = time;
        const ft = st + dur[task];
        schedule.push({ id: ids[task], start: st, machine: mach.machine });
        running.push({ finish: ft, machine: mach.machine, task });
      }
      continue;
    }

    if (running.size) {
      time = Math.max(time, running.peek().finish);
    } else if (ready.length && machines.size) {
      time = Math.max(time, machines.peek().time);
    } else {
      break;
    }
  }

  if (schedule.length < n) {
    const placed = new Uint8Array(n);
    for (const s of schedule) {
      const idx = idToIdx.get(String(s.id));
      if (idx !== undefined) placed[idx] = 1;
    }
    for (const i of topo) {
      if (!placed[i]) {
        schedule.push({ id: ids[i], start: 0, machine: 0 });
      }
    }
  }

  return schedule;
}

function score(schedule) {
  const completion = new Map();
  let makespan = 0;
  let late = 0;
  for (const s of schedule) {
    const i = idToIdx.get(String(s.id));
    if (i === undefined) continue;
    const c = s.start + dur[i];
    completion.set(String(s.id), c);
    if (c > makespan) makespan = c;
    late += Math.max(0, c - deadline[i]) * weight[i];
  }
  return late + makespan;
}

let best = null;
let bestScore = Infinity;
const modes = n > 50000 ? [0, 2, 5] : [0, 1, 2, 3, 4, 5];
for (const mode of modes) {
  const s = buildSchedule(mode);
  const sc = score(s);
  if (sc < bestScore) {
    bestScore = sc;
    best = s;
  }
}

process.stdout.write(JSON.stringify({ schedule: best || [] }));
