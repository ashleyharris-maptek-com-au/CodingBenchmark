const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input.charCodeAt(0) === 0xFEFF ? input.slice(1) : input);

const tasks = Array.isArray(data.tasks) ? data.tasks : [];
const n = tasks.length;
const machineCount = Math.max(1, data.machineCount | 0);

class Heap {
  constructor(cmp) {
    this.a = [];
    this.cmp = cmp;
  }
  get size() {
    return this.a.length;
  }
  push(x) {
    const a = this.a;
    a.push(x);
    let i = a.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (this.cmp(a[p], x) <= 0) break;
      a[i] = a[p];
      i = p;
    }
    a[i] = x;
  }
  pop() {
    const a = this.a;
    const top = a[0];
    const x = a.pop();
    if (a.length > 0) {
      let i = 0;
      while (true) {
        let l = i * 2 + 1;
        if (l >= a.length) break;
        let r = l + 1;
        let c = r < a.length && this.cmp(a[r], a[l]) < 0 ? r : l;
        if (this.cmp(a[c], x) >= 0) break;
        a[i] = a[c];
        i = c;
      }
      a[i] = x;
    }
    return top;
  }
}

function idKey(id) {
  return typeof id + ':' + String(id);
}

const idToIndex = new Map();
for (let i = 0; i < n; i++) idToIndex.set(idKey(tasks[i].id), i);

const duration = new Array(n);
const deadline = new Array(n);
const weight = new Array(n);
const succ = Array.from({ length: n }, () => []);
const indeg0 = new Array(n).fill(0);

for (let i = 0; i < n; i++) {
  const t = tasks[i];
  duration[i] = Math.max(0, Math.floor(Number(t.duration) || 0));
  const d = Number(t.deadline);
  deadline[i] = Number.isFinite(d) ? d : 0;
  const w = Number(t.weight);
  weight[i] = Number.isFinite(w) && w > 0 ? w : 1;
}

for (let i = 0; i < n; i++) {
  const deps = Array.isArray(tasks[i].deps) ? tasks[i].deps : [];
  for (const depId of deps) {
    const j = idToIndex.get(idKey(depId));
    if (j !== undefined) {
      succ[j].push(i);
      indeg0[i]++;
    }
  }
}

// A fast topological pass also supplies upstream longest paths.
const indeg = indeg0.slice();
const topo = [];
const q = [];
const readyFromDeps0 = new Array(n).fill(0);
for (let i = 0; i < n; i++) if (indeg[i] === 0) q.push(i);
for (let qi = 0; qi < q.length; qi++) {
  const u = q[qi];
  topo.push(u);
  const finish = readyFromDeps0[u] + duration[u];
  for (const v of succ[u]) {
    if (finish > readyFromDeps0[v]) readyFromDeps0[v] = finish;
    if (--indeg[v] === 0) q.push(v);
  }
}

// If input is cyclic, return a deterministic partial order followed by leftovers.
if (topo.length !== n) {
  const seen = new Array(n).fill(false);
  for (const i of topo) seen[i] = true;
  const order = topo.map(i => tasks[i].id);
  for (let i = 0; i < n; i++) if (!seen[i]) order.push(tasks[i].id);
  console.log(JSON.stringify({ order }));
  process.exit(0);
}

const tail = duration.slice();
const descWeight = weight.slice();
for (let ti = topo.length - 1; ti >= 0; ti--) {
  const u = topo[ti];
  let bestTail = duration[u];
  let dw = weight[u];
  for (const v of succ[u]) {
    const cand = duration[u] + tail[v];
    if (cand > bestTail) bestTail = cand;
    dw += descWeight[v] * 0.35;
  }
  tail[u] = bestTail;
  descWeight[u] = dw;
}

let totalDur = 0;
for (const d of duration) totalDur += d;
const avgLoad = totalDur / machineCount;

function staticScore(i) {
  const due = deadline[i];
  const urgent = due > 0 ? (avgLoad - due) / Math.max(1, avgLoad) : -1;
  return (
    weight[i] * 9 +
    descWeight[i] * 1.5 +
    tail[i] * 2.2 +
    readyFromDeps0[i] * 0.4 +
    urgent * weight[i] * 10 -
    duration[i] * 0.08
  );
}

const rank = new Array(n);
for (let i = 0; i < n; i++) rank[i] = staticScore(i);

function readyCmp(a, b) {
  if (a.score !== b.score) return b.score - a.score;
  if (deadline[a.i] !== deadline[b.i]) return deadline[a.i] - deadline[b.i];
  return String(tasks[a.i].id).localeCompare(String(tasks[b.i].id));
}

const ready = new Heap(readyCmp);
const machines = new Heap((a, b) => a.time !== b.time ? a.time - b.time : a.machine - b.machine);
const completions = new Heap((a, b) => a.time !== b.time ? a.time - b.time : a.i - b.i);
for (let m = 0; m < machineCount; m++) machines.push({ time: 0, machine: m });

const remaining = indeg0.slice();
const readyFromDeps = new Array(n).fill(0);
const scheduled = new Array(n).fill(false);
const schedule = [];

for (let i = 0; i < n; i++) {
  if (remaining[i] === 0) ready.push({ i, score: rank[i] });
}

function releaseCompleted(upToTime) {
  while (completions.size > 0 && completions.a[0].time <= upToTime) {
    const ev = completions.pop();
    for (const v of succ[ev.i]) {
      if (ev.time > readyFromDeps[v]) readyFromDeps[v] = ev.time;
      if (--remaining[v] === 0) {
        const est = readyFromDeps[v];
        const projectedLate = Math.max(0, est + duration[v] - deadline[v]);
        ready.push({ i: v, score: rank[v] - projectedLate * weight[v] * 3 - est * 0.02 });
      }
    }
  }
}

let done = 0;
while (done < n) {
  const mach = machines.pop();
  releaseCompleted(mach.time);

  while (ready.size === 0 && completions.size > 0) {
    const nextTime = Math.max(mach.time, completions.a[0].time);
    mach.time = nextTime;
    releaseCompleted(nextTime);
  }

  if (ready.size === 0) {
    machines.push(mach);
    break;
  }

  const candidates = [];
  const limit = Math.min(64, ready.size);
  let best = null;
  for (let k = 0; k < limit; k++) {
    const item = ready.pop();
    const i = item.i;
    const start = Math.max(mach.time, readyFromDeps[i]);
    const finish = start + duration[i];
    const late = Math.max(0, finish - deadline[i]);
    const slack = deadline[i] - finish - Math.max(0, tail[i] - duration[i]);
    const pressure = (slack < 0 ? -slack * weight[i] * 4 : -slack * 0.02);
    const score = item.score + pressure - late * weight[i] * 7 - start * 0.015;
    const cand = { item, start, finish, score };
    candidates.push(cand);
    if (!best || cand.score > best.score) best = cand;
  }

  for (const cand of candidates) {
    if (cand !== best) ready.push(cand.item);
  }

  const i = best.item.i;
  scheduled[i] = true;
  schedule.push({ id: tasks[i].id, start: best.start, machine: mach.machine });
  machines.push({ time: best.finish, machine: mach.machine });
  completions.push({ time: best.finish, i });
  done++;
}

if (done !== n) {
  for (let i = 0; i < n; i++) {
    if (!scheduled[i]) schedule.push({ id: tasks[i].id, start: 0, machine: 0 });
  }
}

console.log(JSON.stringify({ schedule }));
