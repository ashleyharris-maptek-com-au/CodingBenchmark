; SPIR-V
; Version: 1.0
OpCapability Shader
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %inWorldPos %inNormal %inUV %inTangent %inColor %outColor
OpExecutionMode %main OriginUpperLeft

OpDecorate %inWorldPos Location 0
OpDecorate %inNormal Location 1
OpDecorate %inUV Location 2
OpDecorate %inTangent Location 3
OpDecorate %inColor Location 4
OpDecorate %outColor Location 0
OpDecorate %UBO Block
OpMemberDecorate %UBO 0 ColMajor
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 0 MatrixStride 16
OpMemberDecorate %UBO 1 ColMajor
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 1 MatrixStride 16
OpMemberDecorate %UBO 2 ColMajor
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 2 MatrixStride 16
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0

%void = OpTypeVoid
%fn = OpTypeFunction %void
%bool = OpTypeBool
%int = OpTypeInt 32 1
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%mat4 = OpTypeMatrix %v4float 4
%UBO = OpTypeStruct %mat4 %mat4 %mat4 %v4float %v4float %v4float
%ptr_Input_v2float = OpTypePointer Input %v2float
%ptr_Input_v3float = OpTypePointer Input %v3float
%ptr_Output_v4float = OpTypePointer Output %v4float
%ptr_Uniform_UBO = OpTypePointer Uniform %UBO
%ptr_Function_float = OpTypePointer Function %float
%ptr_Function_int = OpTypePointer Function %int

%float_0 = OpConstant %float 0.0
%float_0_5 = OpConstant %float 0.5
%float_1 = OpConstant %float 1.0
%float_2 = OpConstant %float 2.0
%float_3 = OpConstant %float 3.0
%float_4 = OpConstant %float 4.0
%float_20 = OpConstant %float 20.0
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_20 = OpConstant %int 20

%inWorldPos = OpVariable %ptr_Input_v3float Input
%inNormal = OpVariable %ptr_Input_v3float Input
%inUV = OpVariable %ptr_Input_v2float Input
%inTangent = OpVariable %ptr_Input_v3float Input
%inColor = OpVariable %ptr_Input_v3float Input
%outColor = OpVariable %ptr_Output_v4float Output
%ubo = OpVariable %ptr_Uniform_UBO Uniform

%main = OpFunction %void None %fn
%entry = OpLabel
%x = OpVariable %ptr_Function_float Function
%y = OpVariable %ptr_Function_float Function
%cx = OpVariable %ptr_Function_float Function
%cy = OpVariable %ptr_Function_float Function
%iter = OpVariable %ptr_Function_int Function
OpStore %x %float_0
OpStore %y %float_0
OpStore %iter %int_0
%uv = OpLoad %v2float %inUV
%uvx = OpCompositeExtract %float %uv 0
%uvy = OpCompositeExtract %float %uv 1
%uvx_scaled = OpFMul %float %uvx %float_3
%mapped_x = OpFSub %float %uvx_scaled %float_2
%uvy_scaled = OpFMul %float %uvy %float_2
%mapped_y = OpFSub %float %uvy_scaled %float_1
OpStore %cx %mapped_x
OpStore %cy %mapped_y
OpBranch %loop

%loop = OpLabel
OpLoopMerge %merge %continue None
OpBranch %check

%check = OpLabel
%i_check = OpLoad %int %iter
%under_limit = OpSLessThan %bool %i_check %int_20
%x_check = OpLoad %float %x
%y_check = OpLoad %float %y
%x_check_sq = OpFMul %float %x_check %x_check
%y_check_sq = OpFMul %float %y_check %y_check
%mag_check = OpFAdd %float %x_check_sq %y_check_sq
%not_escaped = OpFOrdLessThanEqual %bool %mag_check %float_4
%keep_going = OpLogicalAnd %bool %under_limit %not_escaped
OpBranchConditional %keep_going %body %merge

%body = OpLabel
%old_x = OpLoad %float %x
%old_y = OpLoad %float %y
%x_sq = OpFMul %float %old_x %old_x
%y_sq = OpFMul %float %old_y %old_y
%x_minus_y = OpFSub %float %x_sq %y_sq
%cx_val = OpLoad %float %cx
%new_x = OpFAdd %float %x_minus_y %cx_val
%xy = OpFMul %float %old_x %old_y
%two_xy = OpFMul %float %xy %float_2
%cy_val = OpLoad %float %cy
%new_y = OpFAdd %float %two_xy %cy_val
OpStore %x %new_x
OpStore %y %new_y
%i_old = OpLoad %int %iter
%i_new = OpIAdd %int %i_old %int_1
OpStore %iter %i_new
OpBranch %continue

%continue = OpLabel
OpBranch %loop

%merge = OpLabel
%final_i = OpLoad %int %iter
%final_i_f = OpConvertSToF %float %final_i
%t = OpFDiv %float %final_i_f %float_20
%g = OpFMul %float %t %float_0_5
%b = OpFSub %float %float_1 %t
%color = OpCompositeConstruct %v4float %t %g %b %float_1
OpStore %outColor %color
OpReturn
OpFunctionEnd
