; SPIR-V
; Version: 1.0
OpCapability Shader
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %in_uv %outColor
OpExecutionMode %main OriginUpperLeft
OpName %main "main"
OpName %in_uv "uv"
OpName %outColor "outColor"
OpName %UBO "UBO"
OpMemberName %UBO 0 "model"
OpMemberName %UBO 1 "view"
OpMemberName %UBO 2 "projection"
OpMemberName %UBO 3 "lightPos"
OpMemberName %UBO 4 "cameraPos"
OpMemberName %UBO 5 "params"
OpName %ubo "ubo"
OpDecorate %in_uv Location 2
OpDecorate %outColor Location 0
OpMemberDecorate %UBO 0 ColMajor
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 0 MatrixStride 16
OpMemberDecorate %UBO 1 ColMajor
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 1 MatrixStride 16
OpMemberDecorate %UBO 2 ColMajor
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 2 MatrixStride 16
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %UBO Block
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0
%void = OpTypeVoid
%fn_void = OpTypeFunction %void
%float = OpTypeFloat 32
%int = OpTypeInt 32 1
%bool = OpTypeBool
%v2float = OpTypeVector %float 2
%v4float = OpTypeVector %float 4
%mat4 = OpTypeMatrix %v4float 4
%UBO = OpTypeStruct %mat4 %mat4 %mat4 %v4float %v4float %v4float
%ptr_Uniform_UBO = OpTypePointer Uniform %UBO
%ptr_Input_v2float = OpTypePointer Input %v2float
%ptr_Output_v4float = OpTypePointer Output %v4float
%ptr_Function_int = OpTypePointer Function %int
%ptr_Function_v2float = OpTypePointer Function %v2float
%in_uv = OpVariable %ptr_Input_v2float Input
%outColor = OpVariable %ptr_Output_v4float Output
%ubo = OpVariable %ptr_Uniform_UBO Uniform
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_20 = OpConstant %int 20
%float_0 = OpConstant %float 0
%float_0_5 = OpConstant %float 0.5
%float_1 = OpConstant %float 1
%float_2 = OpConstant %float 2
%float_3 = OpConstant %float 3
%float_4 = OpConstant %float 4
%float_20 = OpConstant %float 20
%v2float_0 = OpConstantComposite %v2float %float_0 %float_0
%main = OpFunction %void None %fn_void
%entry = OpLabel
%z_var = OpVariable %ptr_Function_v2float Function
%c_var = OpVariable %ptr_Function_v2float Function
%iter_var = OpVariable %ptr_Function_int Function
%uv = OpLoad %v2float %in_uv
%uv_x = OpCompositeExtract %float %uv 0
%uv_y = OpCompositeExtract %float %uv 1
%cx_mul = OpFMul %float %uv_x %float_3
%cx = OpFSub %float %cx_mul %float_2
%cy_mul = OpFMul %float %uv_y %float_2
%cy = OpFSub %float %cy_mul %float_1
%c = OpCompositeConstruct %v2float %cx %cy
OpStore %c_var %c
OpStore %z_var %v2float_0
OpStore %iter_var %int_0
OpBranch %loop_header
%loop_header = OpLabel
OpLoopMerge %loop_merge %loop_continue None
OpBranch %loop_body
%loop_body = OpLabel
%iter = OpLoad %int %iter_var
%iter_lt_20 = OpSLessThan %bool %iter %int_20
%z = OpLoad %v2float %z_var
%zx = OpCompositeExtract %float %z 0
%zy = OpCompositeExtract %float %z 1
%zx2 = OpFMul %float %zx %zx
%zy2 = OpFMul %float %zy %zy
%mag2 = OpFAdd %float %zx2 %zy2
%inside = OpFOrdLessThanEqual %bool %mag2 %float_4
%keep_going = OpLogicalAnd %bool %iter_lt_20 %inside
OpSelectionMerge %if_merge None
OpBranchConditional %keep_going %if_then %if_merge
%if_then = OpLabel
%two_zx = OpFMul %float %float_2 %zx
%two_zx_zy = OpFMul %float %two_zx %zy
%c_load = OpLoad %v2float %c_var
%c_x = OpCompositeExtract %float %c_load 0
%c_y = OpCompositeExtract %float %c_load 1
%new_x_pre = OpFSub %float %zx2 %zy2
%new_x = OpFAdd %float %new_x_pre %c_x
%new_y = OpFAdd %float %two_zx_zy %c_y
%new_z = OpCompositeConstruct %v2float %new_x %new_y
OpStore %z_var %new_z
%iter_next = OpIAdd %int %iter %int_1
OpStore %iter_var %iter_next
OpBranch %loop_continue
%if_merge = OpLabel
OpBranch %loop_merge
%loop_continue = OpLabel
OpBranch %loop_header
%loop_merge = OpLabel
%final_iter = OpLoad %int %iter_var
%iter_float = OpConvertSToF %float %final_iter
%t = OpFDiv %float %iter_float %float_20
%g = OpFMul %float %t %float_0_5
%b = OpFSub %float %float_1 %t
%color = OpCompositeConstruct %v4float %t %g %b %float_1
OpStore %outColor %color
OpReturn
OpFunctionEnd
