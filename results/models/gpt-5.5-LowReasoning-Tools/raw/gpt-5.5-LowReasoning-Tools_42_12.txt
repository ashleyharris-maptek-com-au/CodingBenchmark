; SPIR-V
; Version: 1.0
OpCapability Geometry
%glsl = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %in_worldPos %in_normal %in_color %gl_out %outColor
OpExecutionMode %main InputTrianglesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 15
OpExecutionMode %main Invocations 1
OpSource GLSL 450
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %in_worldPos Location 0
OpDecorate %in_normal Location 1
OpDecorate %in_color Location 2
OpDecorate %outColor Location 0
%void = OpTypeVoid
%bool = OpTypeBool
%float = OpTypeFloat 32
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint_6 = OpConstant %uint 6
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_4 = OpConstant %int 4
%float_0 = OpConstant %float 0
%float_0_015 = OpConstant %float 0.015
%float_0_3 = OpConstant %float 0.3
%float_1 = OpConstant %float 1
%v3_white = OpConstantComposite %v3float %float_1 %float_1 %float_1
%gl_PerVertex_in = OpTypeStruct %v4float
%gl_PerVertex_out = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_6
%arr_v3_in = OpTypeArray %v3float %uint_6
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%fn_void = OpTypeFunction %void
%fn_emit_v = OpTypeFunction %void %v4float %v3float
%fn_emit_edge = OpTypeFunction %void %v4float %v4float
%gl_in = OpVariable %ptr_in_arr_pv Input
%in_worldPos = OpVariable %ptr_in_arr_v3 Input
%in_normal = OpVariable %ptr_in_arr_v3 Input
%in_color = OpVariable %ptr_in_arr_v3 Input
%gl_out = OpVariable %ptr_out_pv Output
%outColor = OpVariable %ptr_out_v3 Output
%emit_v = OpFunction %void None %fn_emit_v
%emit_pos = OpFunctionParameter %v4float
%emit_col = OpFunctionParameter %v3float
%emit_entry = OpLabel
%emit_pos_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %emit_pos_ptr %emit_pos
OpStore %outColor %emit_col
OpEmitVertex
OpReturn
OpFunctionEnd
%emit_edge = OpFunction %void None %fn_emit_edge
%edge_a = OpFunctionParameter %v4float
%edge_b = OpFunctionParameter %v4float
%edge_entry = OpLabel
%a_x = OpCompositeExtract %float %edge_a 0
%a_y = OpCompositeExtract %float %edge_a 1
%b_x = OpCompositeExtract %float %edge_b 0
%b_y = OpCompositeExtract %float %edge_b 1
%d_x = OpFSub %float %b_x %a_x
%d_y = OpFSub %float %b_y %a_y
%dir = OpCompositeConstruct %v2float %d_x %d_y
%dir_n = OpExtInst %v2float %glsl Normalize %dir
%n_y = OpCompositeExtract %float %dir_n 1
%n_x = OpCompositeExtract %float %dir_n 0
%neg_n_y = OpFNegate %float %n_y
%perp = OpCompositeConstruct %v2float %neg_n_y %n_x
%off2 = OpVectorTimesScalar %v2float %perp %float_0_015
%off_x = OpCompositeExtract %float %off2 0
%off_y = OpCompositeExtract %float %off2 1
%off4 = OpCompositeConstruct %v4float %off_x %off_y %float_0 %float_0
%qa_plus = OpFAdd %v4float %edge_a %off4
%qa_minus = OpFSub %v4float %edge_a %off4
%qb_plus = OpFAdd %v4float %edge_b %off4
%qb_minus = OpFSub %v4float %edge_b %off4
%call_e0 = OpFunctionCall %void %emit_v %qa_plus %v3_white
%call_e1 = OpFunctionCall %void %emit_v %qa_minus %v3_white
%call_e2 = OpFunctionCall %void %emit_v %qb_plus %v3_white
%call_e3 = OpFunctionCall %void %emit_v %qb_minus %v3_white
OpEndPrimitive
OpReturn
OpFunctionEnd
%main = OpFunction %void None %fn_void
%main_entry = OpLabel
%p0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%p2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%p4_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_4 %int_0
%p0 = OpLoad %v4float %p0_ptr
%p2 = OpLoad %v4float %p2_ptr
%p4 = OpLoad %v4float %p4_ptr
%c0_ptr = OpAccessChain %ptr_in_v3 %in_color %int_0
%c2_ptr = OpAccessChain %ptr_in_v3 %in_color %int_2
%c4_ptr = OpAccessChain %ptr_in_v3 %in_color %int_4
%c0 = OpLoad %v3float %c0_ptr
%c2 = OpLoad %v3float %c2_ptr
%c4 = OpLoad %v3float %c4_ptr
%c0_dim = OpVectorTimesScalar %v3float %c0 %float_0_3
%c2_dim = OpVectorTimesScalar %v3float %c2 %float_0_3
%c4_dim = OpVectorTimesScalar %v3float %c4 %float_0_3
%call_t0 = OpFunctionCall %void %emit_v %p0 %c0_dim
%call_t1 = OpFunctionCall %void %emit_v %p2 %c2_dim
%call_t2 = OpFunctionCall %void %emit_v %p4 %c4_dim
OpEndPrimitive
%call_edge_02 = OpFunctionCall %void %emit_edge %p0 %p2
%call_edge_24 = OpFunctionCall %void %emit_edge %p2 %p4
%call_edge_40 = OpFunctionCall %void %emit_edge %p4 %p0
OpReturn
OpFunctionEnd
