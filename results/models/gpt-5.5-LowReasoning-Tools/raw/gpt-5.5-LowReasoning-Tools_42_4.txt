OpCapability Geometry
%1 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main InputLines
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1
OpSource GLSL 450
OpName %main "main"
OpName %gl_PerVertex_in "gl_PerVertex_in"
OpMemberName %gl_PerVertex_in 0 "gl_Position"
OpName %gl_in "gl_in"
OpName %worldPos "worldPos"
OpName %normal "normal"
OpName %color "color"
OpName %gl_PerVertex_out "gl_PerVertex_out"
OpMemberName %gl_PerVertex_out 0 "gl_Position"
OpName %gl_out "gl_out"
OpName %outColor "outColor"
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %outColor Location 0
%void = OpTypeVoid
%fn_void = OpTypeFunction %void
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%uint_2 = OpConstant %uint 2
%int = OpTypeInt 32 1
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%float_0 = OpConstant %float 0
%float_1 = OpConstant %float 1
%float_0_04 = OpConstant %float 0.04
%float_0_005 = OpConstant %float 0.005
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%ptr_in_v4 = OpTypePointer Input %v4float
%arr_v3_in = OpTypeArray %v3float %uint_2
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%ptr_in_v3 = OpTypePointer Input %v3float
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%gl_in = OpVariable %ptr_in_arr_pv Input
%worldPos = OpVariable %ptr_in_arr_v3 Input
%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input
%gl_out = OpVariable %ptr_out_pv Output
%outColor = OpVariable %ptr_out_v3 Output
%main = OpFunction %void None %fn_void
%entry = OpLabel
%p0_ptr = OpAccessChain %ptr_in_v3 %worldPos %int_0
%p0 = OpLoad %v3float %p0_ptr
%p1_ptr = OpAccessChain %ptr_in_v3 %worldPos %int_1
%p1 = OpLoad %v3float %p1_ptr
%c0_ptr = OpAccessChain %ptr_in_v3 %color %int_0
%c0 = OpLoad %v3float %c0_ptr
%c1_ptr = OpAccessChain %ptr_in_v3 %color %int_1
%c1 = OpLoad %v3float %c1_ptr
%p0_x = OpCompositeExtract %float %p0 0
%p0_y = OpCompositeExtract %float %p0 1
%p0_z = OpCompositeExtract %float %p0 2
%p1_x = OpCompositeExtract %float %p1 0
%p1_y = OpCompositeExtract %float %p1 1
%p1_z = OpCompositeExtract %float %p1 2
%dx = OpFSub %float %p1_x %p0_x
%dy = OpFSub %float %p1_y %p0_y
%dx2 = OpFMul %float %dx %dx
%dy2 = OpFMul %float %dy %dy
%len2 = OpFAdd %float %dx2 %dy2
%inv_len = OpExtInst %float %1 InverseSqrt %len2
%neg_dy = OpFNegate %float %dy
%perp_x = OpFMul %float %neg_dy %inv_len
%perp_y = OpFMul %float %dx %inv_len
%off0_x = OpFMul %float %perp_x %float_0_04
%off0_y = OpFMul %float %perp_y %float_0_04
%off1_x = OpFMul %float %perp_x %float_0_005
%off1_y = OpFMul %float %perp_y %float_0_005
%out_pos_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
%v0_x = OpFAdd %float %p0_x %off0_x
%v0_y = OpFAdd %float %p0_y %off0_y
%v0 = OpCompositeConstruct %v4float %v0_x %v0_y %p0_z %float_1
OpStore %out_pos_ptr %v0
OpStore %outColor %c0
OpEmitVertex
%v1_x = OpFSub %float %p0_x %off0_x
%v1_y = OpFSub %float %p0_y %off0_y
%v1 = OpCompositeConstruct %v4float %v1_x %v1_y %p0_z %float_1
OpStore %out_pos_ptr %v1
OpStore %outColor %c0
OpEmitVertex
%v2_x = OpFAdd %float %p1_x %off1_x
%v2_y = OpFAdd %float %p1_y %off1_y
%v2 = OpCompositeConstruct %v4float %v2_x %v2_y %p1_z %float_1
OpStore %out_pos_ptr %v2
OpStore %outColor %c1
OpEmitVertex
%v3_x = OpFSub %float %p1_x %off1_x
%v3_y = OpFSub %float %p1_y %off1_y
%v3 = OpCompositeConstruct %v4float %v3_x %v3_y %p1_z %float_1
OpStore %out_pos_ptr %v3
OpStore %outColor %c1
OpEmitVertex
OpEndPrimitive
OpReturn
OpFunctionEnd
