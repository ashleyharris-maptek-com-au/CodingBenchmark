declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input.charCodeAt(0) === 0xfeff ? input.slice(1) : input);

type Drone = { id?: string; x: number; y: number };

const probability: number[][] = Array.isArray(data.probability) ? data.probability : [];
const drones: Drone[] = Array.isArray(data.drones) ? data.drones : [];
const maxSteps: number = Math.max(0, Math.floor(Number(data.maxStepsPerDrone) || 0));
const height = probability.length;
const width = height > 0 && Array.isArray(probability[0]) ? probability[0].length : 0;
const cellCount = width * height;

function idxOf(x: number, y: number): number {
  return y * width + x;
}

function xOf(idx: number): number {
  return idx % width;
}

function yOf(idx: number): number {
  return Math.floor(idx / width);
}

function isOpen(x: number, y: number): boolean {
  return y >= 0 && y < height && x >= 0 && x < width && probability[y][x] !== -1;
}

if (width <= 0 || height <= 0 || drones.length === 0) {
  console.log(JSON.stringify({ paths: drones.map(() => []) }));
} else {
  const value = new Float64Array(cellCount);
  const open = new Uint8Array(cellCount);
  let totalPositive = 0;

  for (let y = 0; y < height; y++) {
    const row = probability[y] || [];
    for (let x = 0; x < width; x++) {
      const idx = idxOf(x, y);
      const v = Number(row[x]);
      if (v !== -1 && Number.isFinite(v)) {
        open[idx] = 1;
        if (v > 0) {
          value[idx] = v;
          totalPositive++;
        }
      }
    }
  }

  const covered = new Uint8Array(cellCount);
  const paths: number[][][] = drones.map((d) => {
    let sx = Math.floor(Number(d.x));
    let sy = Math.floor(Number(d.y));
    if (!isOpen(sx, sy)) {
      let found = -1;
      for (let i = 0; i < cellCount; i++) {
        if (open[i]) {
          found = i;
          break;
        }
      }
      if (found >= 0) {
        sx = xOf(found);
        sy = yOf(found);
      } else {
        sx = 0;
        sy = 0;
      }
    }
    const startIdx = idxOf(sx, sy);
    if (open[startIdx]) covered[startIdx] = 1;
    return [[sx, sy]];
  });

  const seen = new Int32Array(cellCount);
  const parent = new Int32Array(cellCount);
  const dist = new Int32Array(cellCount);
  const queue = new Int32Array(Math.max(1, cellCount));
  let stamp = 1;

  const dirs = new Int32Array([1, 0, -1, 0, 0, 1, 0, -1]);

  function localBonus(idx: number): number {
    let bonus = 0;
    const x = xOf(idx);
    const y = yOf(idx);
    for (let k = 0; k < 4; k++) {
      const nx = x + dirs[k * 2];
      const ny = y + dirs[k * 2 + 1];
      if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
        const ni = idxOf(nx, ny);
        if (open[ni] && !covered[ni]) bonus += value[ni] * 0.30;
      }
    }
    for (let k = 0; k < 4; k++) {
      const nx = x + dirs[k * 2] * 2;
      const ny = y + dirs[k * 2 + 1] * 2;
      if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
        const ni = idxOf(nx, ny);
        if (open[ni] && !covered[ni]) bonus += value[ni] * 0.10;
      }
    }
    return bonus;
  }

  function chooseTarget(start: number, remaining: number): number[] {
    if (remaining <= 0 || !open[start]) return [];
    if (stamp === 2147483647) {
      seen.fill(0);
      stamp = 1;
    }

    let head = 0;
    let tail = 0;
    queue[tail++] = start;
    seen[start] = stamp;
    parent[start] = -1;
    dist[start] = 0;

    let best = -1;
    let bestScore = covered[start] ? 0 : value[start] * 1.2 + localBonus(start);
    if (bestScore > 0) best = start;

    while (head < tail) {
      const cur = queue[head++];
      const cd = dist[cur];
      if (cd >= remaining) continue;
      const cx = xOf(cur);
      const cy = yOf(cur);

      for (let k = 0; k < 4; k++) {
        const nx = cx + dirs[k * 2];
        const ny = cy + dirs[k * 2 + 1];
        if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;
        const ni = idxOf(nx, ny);
        if (!open[ni] || seen[ni] === stamp) continue;

        seen[ni] = stamp;
        parent[ni] = cur;
        dist[ni] = cd + 1;
        queue[tail++] = ni;

        const gain = (covered[ni] ? 0 : value[ni]) + localBonus(ni);
        if (gain > 0) {
          const d = dist[ni];
          const score = gain / Math.pow(d + 0.75, 0.82);
          if (score > bestScore) {
            bestScore = score;
            best = ni;
          }
        }
      }
    }

    if (best < 0 || best === start) {
      stamp++;
      return [];
    }

    const rev: number[] = [];
    let cur = best;
    while (cur !== start && cur >= 0) {
      rev.push(cur);
      cur = parent[cur];
    }
    rev.reverse();
    stamp++;
    return rev;
  }

  const order = drones.map((_, i) => i);
  order.sort((a, b) => {
    const av = value[idxOf(paths[a][0][0], paths[a][0][1])] || 0;
    const bv = value[idxOf(paths[b][0][0], paths[b][0][1])] || 0;
    return bv - av;
  });

  let remainingPositive = totalPositive;
  for (let i = 0; i < cellCount; i++) {
    if (covered[i] && value[i] > 0) remainingPositive--;
  }

  let madeProgress = true;
  while (madeProgress) {
    madeProgress = false;
    for (const di of order) {
      const path = paths[di];
      if (path.length > maxSteps) continue;
      const last = path[path.length - 1];
      const start = idxOf(last[0], last[1]);
      const remaining = maxSteps - (path.length - 1);
      if (remaining <= 0) continue;

      const segment = chooseTarget(start, remaining);
      if (segment.length === 0) continue;

      for (const idx of segment) {
        path.push([xOf(idx), yOf(idx)]);
        if (!covered[idx]) {
          covered[idx] = 1;
          if (value[idx] > 0) remainingPositive--;
        }
        if (path.length > maxSteps) break;
      }
      madeProgress = true;
    }
    if (remainingPositive <= 0) break;
  }

  for (const path of paths) {
    const last = path[path.length - 1] || [0, 0];
    while (path.length < maxSteps + 1) {
      path.push([last[0], last[1]]);
    }
    if (path.length > maxSteps + 1) path.length = maxSteps + 1;
  }

  console.log(JSON.stringify({ paths }));
}
