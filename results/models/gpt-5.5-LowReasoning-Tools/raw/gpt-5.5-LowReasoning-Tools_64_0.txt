declare const require: any;
declare const process: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type Drone = { id: string | number; x: number; y: number };
type Cell = { idx: number; x: number; y: number; value: number };

const probability: number[][] = Array.isArray(data.probability) ? data.probability : [];
const drones: Drone[] = Array.isArray(data.drones) ? data.drones : [];
const maxSteps: number = Math.max(0, Math.floor(Number(data.maxStepsPerDrone) || 0));

const height = probability.length;
const width = height > 0 && Array.isArray(probability[0]) ? probability[0].length : 0;
const n = width * height;

function indexOf(x: number, y: number): number {
  return y * width + x;
}

function inside(x: number, y: number): boolean {
  return x >= 0 && y >= 0 && x < width && y < height;
}

const values = new Float64Array(n);
const passable = new Uint8Array(n);
const covered = new Uint8Array(n);
const cells: Cell[] = [];

for (let y = 0; y < height; y++) {
  const row = probability[y] || [];
  for (let x = 0; x < width; x++) {
    const idx = indexOf(x, y);
    const v = Number(row[x]);
    values[idx] = Number.isFinite(v) ? v : -1;
    if (values[idx] >= 0) {
      passable[idx] = 1;
      if (values[idx] > 0) cells.push({ idx, x, y, value: values[idx] });
    }
  }
}

cells.sort((a, b) => b.value - a.value);

const stamp = new Int32Array(n);
const dist = new Int32Array(n);
const prev = new Int32Array(n);
const queue = new Int32Array(Math.max(1, n));
let bfsStamp = 1;

const dirs = [
  [0, -1],
  [1, 0],
  [0, 1],
  [-1, 0],
];

function runBfs(start: number, maxDistance: number): void {
  if (n === 0) return;
  bfsStamp++;
  if (bfsStamp > 2000000000) {
    stamp.fill(0);
    bfsStamp = 1;
  }
  let head = 0;
  let tail = 0;
  queue[tail++] = start;
  stamp[start] = bfsStamp;
  dist[start] = 0;
  prev[start] = -1;

  while (head < tail) {
    const cur = queue[head++];
    const d = dist[cur];
    if (d >= maxDistance) continue;
    const x = cur % width;
    const y = (cur / width) | 0;

    for (let k = 0; k < 4; k++) {
      const nx = x + dirs[k][0];
      const ny = y + dirs[k][1];
      if (!inside(nx, ny)) continue;
      const ni = indexOf(nx, ny);
      if (!passable[ni] || stamp[ni] === bfsStamp) continue;
      stamp[ni] = bfsStamp;
      dist[ni] = d + 1;
      prev[ni] = cur;
      queue[tail++] = ni;
    }
  }
}

function localPotential(idx: number): number {
  if (!passable[idx]) return 0;
  let total = covered[idx] ? 0 : values[idx];
  const x = idx % width;
  const y = (idx / width) | 0;
  for (let k = 0; k < 4; k++) {
    const nx = x + dirs[k][0];
    const ny = y + dirs[k][1];
    if (!inside(nx, ny)) continue;
    const ni = indexOf(nx, ny);
    if (passable[ni] && !covered[ni] && values[ni] > 0) total += values[ni] * 0.35;
  }
  return total;
}

function bestAdjacentMove(cur: number): number {
  let best = cur;
  let bestScore = covered[cur] ? 0 : values[cur];
  const x = cur % width;
  const y = (cur / width) | 0;
  for (let k = 0; k < 4; k++) {
    const nx = x + dirs[k][0];
    const ny = y + dirs[k][1];
    if (!inside(nx, ny)) continue;
    const ni = indexOf(nx, ny);
    if (!passable[ni]) continue;
    const score = localPotential(ni);
    if (score > bestScore) {
      bestScore = score;
      best = ni;
    }
  }
  return best;
}

function chooseTarget(start: number, remaining: number): number {
  runBfs(start, remaining);
  let best = -1;
  let bestScore = -1;
  let seen = 0;
  const limit = Math.min(cells.length, Math.max(400, Math.min(3500, 700 + remaining * 18)));

  for (let i = 0; i < cells.length && seen < limit; i++) {
    const c = cells[i];
    if (covered[c.idx]) continue;
    seen++;
    if (stamp[c.idx] !== bfsStamp) continue;
    const d = dist[c.idx];
    if (d > remaining) continue;
    const score = (c.value + localPotential(c.idx) * 0.45) / (d + 1);
    if (score > bestScore) {
      bestScore = score;
      best = c.idx;
    }
  }

  if (best >= 0) return best;

  for (let i = 0; i < cells.length; i++) {
    const c = cells[i];
    if (!covered[c.idx] && stamp[c.idx] === bfsStamp && dist[c.idx] <= remaining) {
      return c.idx;
    }
  }
  return -1;
}

function reconstructNext(start: number, target: number): number[] {
  const rev: number[] = [];
  let cur = target;
  while (cur !== start && cur >= 0) {
    rev.push(cur);
    cur = prev[cur];
  }
  rev.reverse();
  return rev;
}

function validStart(drone: Drone): number {
  const x = Math.floor(Number(drone.x));
  const y = Math.floor(Number(drone.y));
  if (inside(x, y) && passable[indexOf(x, y)]) return indexOf(x, y);
  for (let i = 0; i < n; i++) {
    if (passable[i]) return i;
  }
  return 0;
}

const paths: number[][][] = drones.map((drone) => {
  if (width === 0 || height === 0 || n === 0) return [];

  let cur = validStart(drone);
  const path: number[][] = [[cur % width, (cur / width) | 0]];
  if (passable[cur]) covered[cur] = 1;

  let remaining = maxSteps;
  let localStreak = 0;
  while (remaining > 0) {
    const adjacent = bestAdjacentMove(cur);
    if (adjacent !== cur && !covered[adjacent] && localStreak < 10) {
      cur = adjacent;
      path.push([cur % width, (cur / width) | 0]);
      covered[cur] = 1;
      remaining--;
      localStreak++;
      continue;
    }

    localStreak = 0;
    const target = chooseTarget(cur, remaining);
    if (target < 0 || target === cur) break;
    const route = reconstructNext(cur, target);
    if (route.length === 0) break;

    for (const step of route) {
      if (remaining <= 0) break;
      cur = step;
      path.push([cur % width, (cur / width) | 0]);
      if (passable[cur]) covered[cur] = 1;
      remaining--;
    }
  }

  while (remaining > 0) {
    path.push([cur % width, (cur / width) | 0]);
    remaining--;
  }

  return path;
});

process.stdout.write(JSON.stringify({ paths }));
