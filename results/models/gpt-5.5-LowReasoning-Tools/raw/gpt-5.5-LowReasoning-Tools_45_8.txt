import struct

words = []

def w(x):
    words.append(x & 0xffffffff)

def inst(op, *ops):
    w(((1 + len(ops)) << 16) | op)
    for x in ops:
        w(x)

def s_words(s):
    b = s.encode("ascii") + b"\0"
    while len(b) % 4:
        b += b"\0"
    return [struct.unpack("<I", b[i:i+4])[0] for i in range(0, len(b), 4)]
