using System;
using System.Diagnostics;
using System.IO;
using System.Text;

public class Program
{
    static int N;
    static int[] F = Array.Empty<int>();
    static int[] D = Array.Empty<int>();
    static readonly Random Rng = new Random(1234567);

    public static void Main()
    {
        FastScanner fs = new FastScanner(Console.OpenStandardInput());
        N = fs.NextInt();
        if (N <= 0) return;

        long cells = (long)N * N;
        F = new int[cells];
        D = new int[cells];
        for (int i = 0; i < F.Length; i++) F[i] = fs.NextInt();
        for (int i = 0; i < D.Length; i++) D[i] = fs.NextInt();

        int[] best;
        long bestCost;

        if (N <= 9)
        {
            best = BruteForce(out bestCost);
        }
        else
        {
            best = BuildInitial();
            bestCost = Cost(best);

            int[] identity = new int[N];
            int[] reverse = new int[N];
            for (int i = 0; i < N; i++)
            {
                identity[i] = i;
                reverse[i] = N - 1 - i;
            }
            TryTake(identity, ref best, ref bestCost);
            TryTake(reverse, ref best, ref bestCost);

            Improve(best, ref bestCost);
        }

        StringBuilder sb = new StringBuilder();
        sb.AppendLine(bestCost.ToString());
        for (int i = 0; i < N; i++)
        {
            if (i > 0) sb.Append(' ');
            sb.Append(best[i]);
        }
        Console.WriteLine(sb.ToString());
    }

    static void TryTake(int[] candidate, ref int[] best, ref long bestCost)
    {
        long c = Cost(candidate);
        if (c < bestCost)
        {
            bestCost = c;
            best = (int[])candidate.Clone();
        }
    }

    static int[] BuildInitial()
    {
        long[] flowWeight = new long[N];
        long[] distWeight = new long[N];

        for (int i = 0; i < N; i++)
        {
            long fw = 0, dw = 0;
            int row = i * N;
            for (int j = 0; j < N; j++)
            {
                fw += Math.Abs((long)F[row + j]) + Math.Abs((long)F[j * N + i]);
                dw += Math.Abs((long)D[row + j]) + Math.Abs((long)D[j * N + i]);
            }
            flowWeight[i] = fw;
            distWeight[i] = dw;
        }

        int[] facilities = Range(N);
        int[] locations = Range(N);
        Array.Sort(facilities, (a, b) =>
        {
            int cmp = flowWeight[b].CompareTo(flowWeight[a]);
            return cmp != 0 ? cmp : a.CompareTo(b);
        });
        Array.Sort(locations, (a, b) =>
        {
            int cmp = distWeight[a].CompareTo(distWeight[b]);
            return cmp != 0 ? cmp : a.CompareTo(b);
        });

        int[] p = new int[N];
        for (int k = 0; k < N; k++) p[facilities[k]] = locations[k];
        return p;
    }

    static int[] Range(int n)
    {
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = i;
        return a;
    }

    static long Cost(int[] p)
    {
        long total = 0;
        for (int i = 0; i < N; i++)
        {
            int fi = i * N;
            int dpi = p[i] * N;
            for (int j = 0; j < N; j++)
                total += (long)F[fi + j] * D[dpi + p[j]];
        }
        return total;
    }

    static long SwapDelta(int[] p, int a, int b)
    {
        if (a == b) return 0;
        int la = p[a], lb = p[b];
        if (la == lb) return 0;

        long delta = 0;
        int aN = a * N, bN = b * N;
        int laN = la * N, lbN = lb * N;

        for (int k = 0; k < N; k++)
        {
            if (k == a || k == b) continue;
            int lk = p[k];
            int kN = k * N;
            delta += (long)F[aN + k] * (D[lbN + lk] - D[laN + lk]);
            delta += (long)F[bN + k] * (D[laN + lk] - D[lbN + lk]);
            delta += (long)F[kN + a] * (D[lk * N + lb] - D[lk * N + la]);
            delta += (long)F[kN + b] * (D[lk * N + la] - D[lk * N + lb]);
        }

        delta += (long)F[aN + a] * (D[lbN + lb] - D[laN + la]);
        delta += (long)F[bN + b] * (D[laN + la] - D[lbN + lb]);
        delta += (long)F[aN + b] * (D[lbN + la] - D[laN + lb]);
        delta += (long)F[bN + a] * (D[laN + lb] - D[lbN + la]);
        return delta;
    }

    static void Improve(int[] p, ref long cost)
    {
        Stopwatch sw = Stopwatch.StartNew();
        int timeLimitMs = 280000;

        // Deterministic steepest/first improvement is valuable on small instances.
        if (N <= 180)
        {
            bool changed = true;
            while (changed && sw.ElapsedMilliseconds < timeLimitMs)
            {
                changed = false;
                long bestDelta = 0;
                int bestA = -1, bestB = -1;
                for (int a = 0; a < N - 1; a++)
                {
                    for (int b = a + 1; b < N; b++)
                    {
                        long d = SwapDelta(p, a, b);
                        if (d < bestDelta)
                        {
                            bestDelta = d;
                            bestA = a;
                            bestB = b;
                        }
                    }
                }
                if (bestA >= 0)
                {
                    Swap(p, bestA, bestB);
                    cost += bestDelta;
                    changed = true;
                }
            }
        }

        int[] best = (int[])p.Clone();
        long bestCost = cost;
        double temperature = Math.Max(1.0, Math.Abs(cost) / (double)Math.Max(1, N * N));
        long iter = 0;

        while (sw.ElapsedMilliseconds < timeLimitMs)
        {
            int a = Rng.Next(N);
            int b = Rng.Next(N - 1);
            if (b >= a) b++;

            long d = SwapDelta(p, a, b);
            bool accept = d < 0;
            if (!accept && temperature > 0.01)
            {
                double x = -d / temperature;
                if (x > -40.0 && Rng.NextDouble() < Math.Exp(x)) accept = true;
            }

            if (accept)
            {
                Swap(p, a, b);
                cost += d;
                if (cost < bestCost)
                {
                    bestCost = cost;
                    Array.Copy(p, best, N);
                }
            }

            iter++;
            if ((iter & 4095) == 0) temperature *= 0.995;

            // Very large cases get a good construction quickly; do not spend the full
            // five minutes if the instance size makes each delta expensive.
            if (N >= 1200 && iter >= 200000) break;
            if (N >= 500 && iter >= 600000) break;
        }

        Array.Copy(best, p, N);
        cost = bestCost;
    }

    static void Swap(int[] p, int a, int b)
    {
        int t = p[a];
        p[a] = p[b];
        p[b] = t;
    }

    static int[] BruteForce(out long bestCost)
    {
        int[] p = Range(N);
        int[] best = (int[])p.Clone();
        bestCost = Cost(p);

        while (NextPermutation(p))
        {
            long c = Cost(p);
            if (c < bestCost)
            {
                bestCost = c;
                Array.Copy(p, best, N);
            }
        }
        return best;
    }

    static bool NextPermutation(int[] a)
    {
        int i = a.Length - 2;
        while (i >= 0 && a[i] >= a[i + 1]) i--;
        if (i < 0) return false;
        int j = a.Length - 1;
        while (a[j] <= a[i]) j--;
        Swap(a, i, j);
        Array.Reverse(a, i + 1, a.Length - i - 1);
        return true;
    }

    sealed class FastScanner
    {
        readonly Stream stream;
        readonly byte[] buffer = new byte[1 << 16];
        int len, ptr;

        public FastScanner(Stream stream) { this.stream = stream; }

        int Read()
        {
            if (ptr >= len)
            {
                len = stream.Read(buffer, 0, buffer.Length);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }

        public int NextInt()
        {
            int c;
            do { c = Read(); } while (c <= 32 && c >= 0);

            int sign = 1;
            if (c == '-')
            {
                sign = -1;
                c = Read();
            }

            int val = 0;
            while (c > 32)
            {
                val = val * 10 + (c - '0');
                c = Read();
            }
            return val * sign;
        }
    }
}
