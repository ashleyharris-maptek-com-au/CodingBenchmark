; SPIR-V
; Version: 1.0
OpCapability Shader
%1 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %inWorldPos %inNormal %inUV %outColor
OpExecutionMode %main OriginUpperLeft
OpName %main "main"
OpName %inWorldPos "worldPos"
OpName %inNormal "normal"
OpName %inUV "uv"
OpName %outColor "outColor"
OpName %UBO "UBO"
OpMemberName %UBO 0 "model"
OpMemberName %UBO 1 "view"
OpMemberName %UBO 2 "projection"
OpMemberName %UBO 3 "lightPos"
OpMemberName %UBO 4 "cameraPos"
OpMemberName %UBO 5 "params"
OpName %ubo "ubo"
OpDecorate %inWorldPos Location 0
OpDecorate %inNormal Location 1
OpDecorate %inUV Location 2
OpDecorate %outColor Location 0
OpDecorate %UBO Block
OpMemberDecorate %UBO 0 ColMajor
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 0 MatrixStride 16
OpMemberDecorate %UBO 1 ColMajor
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 1 MatrixStride 16
OpMemberDecorate %UBO 2 ColMajor
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 2 MatrixStride 16
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0
%void = OpTypeVoid
%func = OpTypeFunction %void
%float = OpTypeFloat 32
%bool = OpTypeBool
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%mat4 = OpTypeMatrix %v4float 4
%UBO = OpTypeStruct %mat4 %mat4 %mat4 %v4float %v4float %v4float
%ptr_input_v3 = OpTypePointer Input %v3float
%ptr_input_v2 = OpTypePointer Input %v2float
%ptr_output_v4 = OpTypePointer Output %v4float
%ptr_uniform_UBO = OpTypePointer Uniform %UBO
%ptr_uniform_v4 = OpTypePointer Uniform %v4float
%int = OpTypeInt 32 1
%int_3 = OpConstant %int 3
%float_0 = OpConstant %float 0
%float_0_05 = OpConstant %float 0.05
%float_0_1 = OpConstant %float 0.1
%float_0_2 = OpConstant %float 0.2
%float_0_4 = OpConstant %float 0.4
%float_0_5 = OpConstant %float 0.5
%float_0_6 = OpConstant %float 0.6
%float_0_7 = OpConstant %float 0.7
%float_0_95 = OpConstant %float 0.95
%float_1 = OpConstant %float 1
%float_2 = OpConstant %float 2
%float_8 = OpConstant %float 8
%float_16 = OpConstant %float 16
%scale_uv = OpConstantComposite %v2float %float_8 %float_16
%brick_color = OpConstantComposite %v3float %float_0_7 %float_0_2 %float_0_1
%mortar_color = OpConstantComposite %v3float %float_0_7 %float_0_7 %float_0_7
%inWorldPos = OpVariable %ptr_input_v3 Input
%inNormal = OpVariable %ptr_input_v3 Input
%inUV = OpVariable %ptr_input_v2 Input
%outColor = OpVariable %ptr_output_v4 Output
%ubo = OpVariable %ptr_uniform_UBO Uniform
%main = OpFunction %void None %func
%entry = OpLabel
%uv = OpLoad %v2float %inUV
%scaled_uv = OpFMul %v2float %uv %scale_uv
%sx = OpCompositeExtract %float %scaled_uv 0
%sy = OpCompositeExtract %float %scaled_uv 1
%row = OpExtInst %float %1 Floor %sy
%half_row = OpFDiv %float %row %float_2
%floor_half_row = OpExtInst %float %1 Floor %half_row
%two_floor_half_row = OpFMul %float %floor_half_row %float_2
%odd_row = OpFSub %float %row %two_floor_half_row
%x_offset = OpFMul %float %odd_row %float_0_5
%shifted_x = OpFAdd %float %sx %x_offset
%floor_shifted_x = OpExtInst %float %1 Floor %shifted_x
%brick_x = OpFSub %float %shifted_x %floor_shifted_x
%floor_y = OpExtInst %float %1 Floor %sy
%brick_y = OpFSub %float %sy %floor_y
%mortar_x_low = OpFOrdLessThan %bool %brick_x %float_0_05
%mortar_x_high = OpFOrdGreaterThan %bool %brick_x %float_0_95
%mortar_y = OpFOrdLessThan %bool %brick_y %float_0_1
%mortar_x = OpLogicalOr %bool %mortar_x_low %mortar_x_high
%is_mortar = OpLogicalOr %bool %mortar_x %mortar_y
%base_color = OpSelect %v3float %is_mortar %mortar_color %brick_color
%world_pos = OpLoad %v3float %inWorldPos
%normal = OpLoad %v3float %inNormal
%n = OpExtInst %v3float %1 Normalize %normal
%light_ptr = OpAccessChain %ptr_uniform_v4 %ubo %int_3
%light4 = OpLoad %v4float %light_ptr
%light_pos = OpVectorShuffle %v3float %light4 %light4 0 1 2
%light_vec = OpFSub %v3float %light_pos %world_pos
%l = OpExtInst %v3float %1 Normalize %light_vec
%dot_nl = OpDot %float %n %l
%diffuse = OpExtInst %float %1 FMax %dot_nl %float_0
%diffuse_scale = OpFMul %float %diffuse %float_0_6
%shade = OpFAdd %float %float_0_4 %diffuse_scale
%shaded_rgb = OpVectorTimesScalar %v3float %base_color %shade
%out_r = OpCompositeExtract %float %shaded_rgb 0
%out_g = OpCompositeExtract %float %shaded_rgb 1
%out_b = OpCompositeExtract %float %shaded_rgb 2
%out = OpCompositeConstruct %v4float %out_r %out_g %out_b %float_1
OpStore %outColor %out
OpReturn
OpFunctionEnd
