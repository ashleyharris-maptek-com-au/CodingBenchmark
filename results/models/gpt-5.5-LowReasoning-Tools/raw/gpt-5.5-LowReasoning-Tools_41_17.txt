; SPIR-V
; Version: 1.0
OpCapability Shader
%1 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %worldPos %normal %uv %outColor
OpExecutionMode %main OriginUpperLeft
OpSource GLSL 450
OpName %main "main"
OpName %worldPos "worldPos"
OpName %normal "normal"
OpName %uv "uv"
OpName %outColor "outColor"
OpName %UBO "UBO"
OpMemberName %UBO 0 "model"
OpMemberName %UBO 1 "view"
OpMemberName %UBO 2 "projection"
OpMemberName %UBO 3 "lightPos"
OpMemberName %UBO 4 "cameraPos"
OpMemberName %UBO 5 "params"
OpName %ubo "ubo"
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %uv Location 2
OpDecorate %outColor Location 0
OpMemberDecorate %UBO 0 ColMajor
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 0 MatrixStride 16
OpMemberDecorate %UBO 1 ColMajor
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 1 MatrixStride 16
OpMemberDecorate %UBO 2 ColMajor
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 2 MatrixStride 16
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %UBO Block
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0
%void = OpTypeVoid
%3 = OpTypeFunction %void
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%uint_0 = OpConstant %uint 0
%uint_3 = OpConstant %uint 3
%float_0 = OpConstant %float 0
%float_0_05 = OpConstant %float 0.05
%float_0_1 = OpConstant %float 0.1
%float_0_2 = OpConstant %float 0.2
%float_0_4 = OpConstant %float 0.4
%float_0_5 = OpConstant %float 0.5
%float_0_6 = OpConstant %float 0.6
%float_0_7 = OpConstant %float 0.7
%float_0_9 = OpConstant %float 0.9
%float_0_95 = OpConstant %float 0.95
%float_1 = OpConstant %float 1
%float_2 = OpConstant %float 2
%float_8 = OpConstant %float 8
%float_16 = OpConstant %float 16
%v2_scale = OpConstantComposite %v2float %float_8 %float_16
%brickColor = OpConstantComposite %v3float %float_0_7 %float_0_2 %float_0_1
%mortarColor = OpConstantComposite %v3float %float_0_7 %float_0_7 %float_0_7
%mat4v4float = OpTypeMatrix %v4float 4
%UBO = OpTypeStruct %mat4v4float %mat4v4float %mat4v4float %v4float %v4float %v4float
%ptr_Uniform_UBO = OpTypePointer Uniform %UBO
%ptr_Uniform_v4float = OpTypePointer Uniform %v4float
%ptr_Input_v3float = OpTypePointer Input %v3float
%ptr_Input_v2float = OpTypePointer Input %v2float
%ptr_Output_v4float = OpTypePointer Output %v4float
%bool = OpTypeBool
%v3bool = OpTypeVector %bool 3
%ubo = OpVariable %ptr_Uniform_UBO Uniform
%worldPos = OpVariable %ptr_Input_v3float Input
%normal = OpVariable %ptr_Input_v3float Input
%uv = OpVariable %ptr_Input_v2float Input
%outColor = OpVariable %ptr_Output_v4float Output
%main = OpFunction %void None %3
%5 = OpLabel
%uvv = OpLoad %v2float %uv
%scaledUV = OpFMul %v2float %uvv %v2_scale
%sx = OpCompositeExtract %float %scaledUV 0
%sy = OpCompositeExtract %float %scaledUV 1
%row = OpExtInst %float %1 Floor %sy
%rowHalf = OpFDiv %float %row %float_2
%rowHalfFloor = OpExtInst %float %1 Floor %rowHalf
%evenPart = OpFMul %float %rowHalfFloor %float_2
%odd = OpFSub %float %row %evenPart
%stagger = OpFMul %float %odd %float_0_5
%staggeredX = OpFAdd %float %sx %stagger
%cellXFloor = OpExtInst %float %1 Floor %staggeredX
%cellYFloor = OpExtInst %float %1 Floor %sy
%fx = OpFSub %float %staggeredX %cellXFloor
%fy = OpFSub %float %sy %cellYFloor
%mx0 = OpFOrdLessThan %bool %fx %float_0_05
%mx1 = OpFOrdGreaterThan %bool %fx %float_0_95
%my0 = OpFOrdLessThan %bool %fy %float_0_1
%my1 = OpFOrdGreaterThan %bool %fy %float_0_9
%mx = OpLogicalOr %bool %mx0 %mx1
%my = OpLogicalOr %bool %my0 %my1
%isMortar = OpLogicalOr %bool %mx %my
%isMortar3 = OpCompositeConstruct %v3bool %isMortar %isMortar %isMortar
%baseColor = OpSelect %v3float %isMortar3 %mortarColor %brickColor
%n0 = OpLoad %v3float %normal
%n = OpExtInst %v3float %1 Normalize %n0
%wp = OpLoad %v3float %worldPos
%lightPtr = OpAccessChain %ptr_Uniform_v4float %ubo %uint_3
%light4 = OpLoad %v4float %lightPtr
%light3 = OpVectorShuffle %v3float %light4 %light4 0 1 2
%lightVec = OpFSub %v3float %light3 %wp
%lightDir = OpExtInst %v3float %1 Normalize %lightVec
%dotNL = OpDot %float %n %lightDir
%diffuse = OpExtInst %float %1 FMax %dotNL %float_0
%diffScale = OpFMul %float %diffuse %float_0_6
%shade = OpFAdd %float %float_0_4 %diffScale
%shaded = OpVectorTimesScalar %v3float %baseColor %shade
%r = OpCompositeExtract %float %shaded 0
%g = OpCompositeExtract %float %shaded 1
%b = OpCompositeExtract %float %shaded 2
%finalColor = OpCompositeConstruct %v4float %r %g %b %float_1
OpStore %outColor %finalColor
OpReturn
OpFunctionEnd
