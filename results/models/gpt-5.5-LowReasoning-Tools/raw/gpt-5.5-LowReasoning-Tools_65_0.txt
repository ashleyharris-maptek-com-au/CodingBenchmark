declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type RawTask = {
  id: any;
  duration: number;
  deps?: any[];
  deadline?: number;
  weight?: number;
};

type Task = {
  id: any;
  key: string;
  duration: number;
  deps: string[];
  deadline: number;
  weight: number;
};

class Heap<T> {
  private a: T[] = [];
  constructor(private less: (x: T, y: T) => boolean) {}
  get size(): number { return this.a.length; }
  peek(): T | undefined { return this.a[0]; }
  push(x: T): void {
    const a = this.a;
    a.push(x);
    let i = a.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (!this.less(a[i], a[p])) break;
      const t = a[i]; a[i] = a[p]; a[p] = t;
      i = p;
    }
  }
  pop(): T | undefined {
    const a = this.a;
    if (a.length === 0) return undefined;
    const r = a[0];
    const x = a.pop() as T;
    if (a.length > 0) {
      a[0] = x;
      let i = 0;
      for (;;) {
        let c = i * 2 + 1;
        if (c >= a.length) break;
        if (c + 1 < a.length && this.less(a[c + 1], a[c])) c++;
        if (!this.less(a[c], a[i])) break;
        const t = a[i]; a[i] = a[c]; a[c] = t;
        i = c;
      }
    }
    return r;
  }
}

const rawTasks: RawTask[] = Array.isArray(data.tasks) ? data.tasks : [];
const n = rawTasks.length;
const machineCount = Math.max(1, Math.floor(Number(data.machineCount) || 1));

const tasks: Task[] = rawTasks.map((t, i) => ({
  id: t.id,
  key: String(t.id),
  duration: Math.max(0, Math.floor(Number(t.duration) || 0)),
  deps: Array.isArray(t.deps) ? t.deps.map((d: any) => String(d)) : [],
  deadline: Number.isFinite(Number(t.deadline)) ? Number(t.deadline) : 1e15,
  weight: Math.max(0, Number(t.weight) || 1),
}));

const index = new Map<string, number>();
for (let i = 0; i < n; i++) index.set(tasks[i].key, i);

const succ: number[][] = Array.from({ length: n }, () => []);
const indeg0 = new Int32Array(n);
for (let i = 0; i < n; i++) {
  const seen = new Set<string>();
  for (const dk of tasks[i].deps) {
    if (seen.has(dk)) continue;
    seen.add(dk);
    const j = index.get(dk);
    if (j === undefined || j === i) continue;
    succ[j].push(i);
    indeg0[i]++;
  }
}

// Kahn order is also used to estimate downstream pressure and critical paths.
const indegTopo = new Int32Array(indeg0);
const topo: number[] = [];
const q: number[] = [];
let qh = 0;
for (let i = 0; i < n; i++) if (indegTopo[i] === 0) q.push(i);
while (qh < q.length) {
  const u = q[qh++];
  topo.push(u);
  for (const v of succ[u]) if (--indegTopo[v] === 0) q.push(v);
}
for (let i = 0; i < n; i++) if (indegTopo[i] > 0) topo.push(i);

const bottom = new Float64Array(n);
const downWeight = new Float64Array(n);
for (let ti = topo.length - 1; ti >= 0; ti--) {
  const u = topo[ti];
  let bestChild = 0;
  let w = tasks[u].weight;
  for (const v of succ[u]) {
    const b = bottom[v];
    if (b > bestChild) bestChild = b;
    w += downWeight[v] * 0.35;
  }
  bottom[u] = tasks[u].duration + bestChild;
  downWeight[u] = w;
}

function readyLess(a: number, b: number): boolean {
  const ta = tasks[a], tb = tasks[b];
  const slackA = ta.deadline - bottom[a];
  const slackB = tb.deadline - bottom[b];
  const scoreA = ta.weight * 1000000 + downWeight[a] * 5000 + bottom[a] * 25 - slackA;
  const scoreB = tb.weight * 1000000 + downWeight[b] * 5000 + bottom[b] * 25 - slackB;
  if (scoreA !== scoreB) return scoreA > scoreB;
  if (ta.deadline !== tb.deadline) return ta.deadline < tb.deadline;
  if (bottom[a] !== bottom[b]) return bottom[a] > bottom[b];
  return a < b;
}

const remaining = new Int32Array(indeg0);
const readyAt = new Float64Array(n);
const scheduled = new Uint8Array(n);
const ready = new Heap<number>(readyLess);
const machines = new Heap<[number, number]>((a, b) => a[0] === b[0] ? a[1] < b[1] : a[0] < b[0]);
const doneEvents = new Heap<[number, number]>((a, b) => a[0] === b[0] ? a[1] < b[1] : a[0] < b[0]);

for (let i = 0; i < Math.min(machineCount, Math.max(1, n)); i++) machines.push([0, i]);
for (let i = 0; i < n; i++) if (remaining[i] === 0) ready.push(i);

const schedule: { id: any; start: number; machine: number }[] = [];
let placed = 0;

function releaseThrough(t: number): void {
  while (doneEvents.size > 0 && (doneEvents.peek() as [number, number])[0] <= t) {
    const ev = doneEvents.pop() as [number, number];
    const finish = ev[0], u = ev[1];
    for (const v of succ[u]) {
      if (finish > readyAt[v]) readyAt[v] = finish;
      remaining[v]--;
      if (remaining[v] === 0 && scheduled[v] === 0) ready.push(v);
    }
  }
}

while (placed < n && machines.size > 0) {
  let m = machines.pop() as [number, number];
  let time = m[0];
  const machine = m[1];
  releaseThrough(time);

  if (ready.size === 0) {
    const next = doneEvents.peek();
    if (next !== undefined) {
      time = Math.max(time, next[0]);
      releaseThrough(time);
    }
  }

  let u = ready.pop();
  if (u === undefined) {
    // Degenerate cyclic or invalid input: emit a deterministic fallback for
    // unscheduled tasks rather than hanging.
    for (let i = 0; i < n; i++) {
      if (scheduled[i] === 0) {
        scheduled[i] = 1;
        schedule.push({ id: tasks[i].id, start: Math.floor(time), machine });
        time += tasks[i].duration;
        placed++;
      }
    }
    break;
  }

  const start = Math.max(time, readyAt[u]);
  scheduled[u] = 1;
  schedule.push({ id: tasks[u].id, start: Math.floor(start), machine });
  placed++;
  const finish = start + tasks[u].duration;
  doneEvents.push([finish, u]);
  machines.push([finish, machine]);
}
