declare const require: any;
const fs = require('fs');
let input: string = fs.readFileSync(0, 'utf8');
if (input.charCodeAt(0) === 0xfeff) input = input.slice(1);
const data = JSON.parse(input);

type RawTask = {
  id: any;
  duration: number;
  deps?: any[];
  deadline?: number;
  weight?: number;
};

class Heap<T> {
  private a: T[] = [];
  constructor(private less: (x: T, y: T) => boolean) {}
  get size(): number { return this.a.length; }
  push(v: T): void {
    const a = this.a;
    a.push(v);
    let i = a.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (!this.less(a[i], a[p])) break;
      const t = a[i]; a[i] = a[p]; a[p] = t;
      i = p;
    }
  }
  pop(): T | undefined {
    const a = this.a;
    if (a.length === 0) return undefined;
    const top = a[0];
    const last = a.pop() as T;
    if (a.length > 0) {
      a[0] = last;
      let i = 0;
      for (;;) {
        let best = i;
        const l = i * 2 + 1, r = l + 1;
        if (l < a.length && this.less(a[l], a[best])) best = l;
        if (r < a.length && this.less(a[r], a[best])) best = r;
        if (best === i) break;
        const t = a[i]; a[i] = a[best]; a[best] = t;
        i = best;
      }
    }
    return top;
  }
}

const rawTasks: RawTask[] = Array.isArray(data.tasks) ? data.tasks : [];
const n = rawTasks.length;
const machineCount = Math.max(1, Math.floor(Number(data.machineCount) || 1));

const ids: any[] = new Array(n);
const durations = new Array<number>(n);
const deadlines = new Array<number>(n);
const weights = new Array<number>(n);
const idToIndex = new Map<any, number>();

for (let i = 0; i < n; i++) {
  const t = rawTasks[i];
  ids[i] = t.id;
  durations[i] = Math.max(0, Math.floor(Number(t.duration) || 0));
  deadlines[i] = Number.isFinite(Number(t.deadline)) ? Number(t.deadline) : 1e15;
  weights[i] = Math.max(0, Number(t.weight) || 1);
  idToIndex.set(t.id, i);
}

const succ: number[][] = Array.from({ length: n }, () => []);
const indeg0 = new Array<number>(n).fill(0);
for (let i = 0; i < n; i++) {
  const deps = Array.isArray(rawTasks[i].deps) ? rawTasks[i].deps! : [];
  const seen = new Set<number>();
  for (const d of deps) {
    const j = idToIndex.get(d);
    if (j === undefined || j === i || seen.has(j)) continue;
    seen.add(j);
    succ[j].push(i);
    indeg0[i]++;
  }
}

const indegTopo = indeg0.slice();
const topo: number[] = [];
const q: number[] = [];
for (let i = 0; i < n; i++) if (indegTopo[i] === 0) q.push(i);
for (let head = 0; head < q.length; head++) {
  const u = q[head];
  topo.push(u);
  for (const v of succ[u]) {
    if (--indegTopo[v] === 0) q.push(v);
  }
}

if (topo.length !== n) {
  // Invalid cyclic inputs cannot satisfy all dependencies. Emit a deterministic order.
  console.log(JSON.stringify({ order: rawTasks.map(t => t.id) }));
} else {
  const bottom = durations.slice();
  const downstreamWeight = weights.slice();
  for (let ti = topo.length - 1; ti >= 0; ti--) {
    const u = topo[ti];
    let bestChild = 0;
    let childWeight = 0;
    for (const v of succ[u]) {
      if (bottom[v] > bestChild) bestChild = bottom[v];
      childWeight += downstreamWeight[v] * 0.35;
    }
    bottom[u] = durations[u] + bestChild;
    downstreamWeight[u] = weights[u] + childWeight;
  }

  const indeg = indeg0.slice();
  const readyAt = new Array<number>(n).fill(0);
  let currentTime = 0;

  function urgency(i: number): number {
    const w = Math.log1p(weights[i]) + 0.45 * Math.log1p(downstreamWeight[i]);
    const slack = deadlines[i] - currentTime - bottom[i];
    const pressure = slack / Math.max(1, w);
    const shortBoost = durations[i] <= 0 ? 0 : durations[i] / Math.max(1, weights[i]);
    return pressure + 0.04 * shortBoost - 0.015 * bottom[i];
  }

  const ready = new Heap<number>((a, b) => {
    const sa = urgency(a), sb = urgency(b);
    if (sa !== sb) return sa < sb;
    if (deadlines[a] !== deadlines[b]) return deadlines[a] < deadlines[b];
    if (bottom[a] !== bottom[b]) return bottom[a] > bottom[b];
    return String(ids[a]) < String(ids[b]);
  });

  const freeMachines = new Heap<number>((a, b) => a < b);
  const running = new Heap<{ finish: number; machine: number; task: number }>((a, b) => {
    if (a.finish !== b.finish) return a.finish < b.finish;
    return a.machine < b.machine;
  });

  for (let m = 0; m < machineCount; m++) freeMachines.push(m);
  for (const i of topo) if (indeg[i] === 0) ready.push(i);

  const schedule: { id: any; start: number; machine: number }[] = [];
  let completed = 0;

  function completeUntil(t: number): void {
    while (running.size > 0) {
      const r = running.pop()!;
      if (r.finish > t) {
        running.push(r);
        break;
      }
      currentTime = r.finish;
      freeMachines.push(r.machine);
      completed++;
      for (const v of succ[r.task]) {
        if (readyAt[v] < r.finish) readyAt[v] = r.finish;
        if (--indeg[v] === 0) ready.push(v);
      }
    }
    currentTime = t;
  }

  while (schedule.length < n) {
    while (freeMachines.size > 0 && ready.size > 0) {
      const task = ready.pop()!;
      const machine = freeMachines.pop()!;
      const start = Math.max(currentTime, readyAt[task]);
      schedule.push({ id: ids[task], start, machine });
      running.push({ finish: start + durations[task], machine, task });
    }

    if (schedule.length === n) break;
    const next = running.pop();
    if (!next) break;
    running.push(next);
    completeUntil(next.finish);
  }

  while (completed < n && running.size > 0) {
    const next = running.pop()!;
    running.push(next);
    completeUntil(next.finish);
  }

  console.log(JSON.stringify({ schedule }));
}
