const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const grid = data.grid || [];
const robots = data.robots || [];
const jobs = data.jobs || [];

const h = grid.length;
const w = h ? grid[0].length : 0;
const nCells = w * h;

function cellId(x, y) {
  return y * w + x;
}

function idKey(v) {
  return typeof v === 'number' ? v : String(v);
}

function compareIds(a, b) {
  if (typeof a === 'number' && typeof b === 'number') return a - b;
  const sa = String(a);
  const sb = String(b);
  return sa < sb ? -1 : sa > sb ? 1 : 0;
}

const maxRobotId = robots.reduce((m, r) => Math.max(m, Number(r.id) || 0), -1);
const assignments = Array.from({ length: Math.max(0, maxRobotId + 1) }, () => []);

if (robots.length === 0) {
  console.log(JSON.stringify({ assignments }));
  process.exit(0);
}

const orderedRobots = robots.slice().sort((a, b) => compareIds(a.id, b.id));
const robotStates = orderedRobots.map((r) => ({
  id: r.id,
  pos: cellId(r.x, r.y),
  route: assignments[Number(r.id)]
}));

const open = new Uint8Array(nCells);
for (let y = 0; y < h; y++) {
  const row = grid[y];
  for (let x = 0; x < w; x++) {
    if (row.charCodeAt(x) !== 35) open[cellId(x, y)] = 1;
  }
}

const distCache = new Map();
const cacheLimit = 192;

function bfsFrom(start) {
  let cached = distCache.get(start);
  if (cached) {
    distCache.delete(start);
    distCache.set(start, cached);
    return cached;
  }

  const dist = new Int32Array(nCells);
  dist.fill(-1);

  if (start >= 0 && start < nCells && open[start]) {
    const queue = new Int32Array(nCells);
    let head = 0;
    let tail = 0;
    queue[tail++] = start;
    dist[start] = 0;

    while (head < tail) {
      const c = queue[head++];
      const d = dist[c] + 1;
      const x = c % w;

      let nb = c - w;
      if (nb >= 0 && open[nb] && dist[nb] < 0) {
        dist[nb] = d;
        queue[tail++] = nb;
      }
      nb = c + w;
      if (nb < nCells && open[nb] && dist[nb] < 0) {
        dist[nb] = d;
        queue[tail++] = nb;
      }
      nb = c - 1;
      if (x > 0 && open[nb] && dist[nb] < 0) {
        dist[nb] = d;
        queue[tail++] = nb;
      }
      nb = c + 1;
      if (x + 1 < w && open[nb] && dist[nb] < 0) {
        dist[nb] = d;
        queue[tail++] = nb;
      }
    }
  }

  distCache.set(start, dist);
  if (distCache.size > cacheLimit) {
    const oldest = distCache.keys().next().value;
    distCache.delete(oldest);
  }
  return dist;
}

const orderedJobs = jobs.slice().sort((a, b) => {
  const pa = Number(a.priority) || 0;
  const pb = Number(b.priority) || 0;
  if (pb !== pa) return pb - pa;
  return compareIds(a.id, b.id);
});

for (const job of orderedJobs) {
  const target = cellId(job.x, job.y);
  let best = 0;
  let bestDist = Infinity;

  for (let i = 0; i < robotStates.length; i++) {
    const state = robotStates[i];
    const dist = bfsFrom(state.pos)[target];
    const cost = dist >= 0 ? dist : 1e9 + Math.abs((state.pos % w) - job.x) + Math.abs(Math.floor(state.pos / w) - job.y);

    if (
      cost < bestDist ||
      (cost === bestDist && state.route.length < robotStates[best].route.length) ||
      (cost === bestDist && state.route.length === robotStates[best].route.length && compareIds(state.id, robotStates[best].id) < 0)
    ) {
      best = i;
      bestDist = cost;
    }
  }

  robotStates[best].route.push(job.id);
  robotStates[best].pos = target;
}

console.log(JSON.stringify({ assignments }));
