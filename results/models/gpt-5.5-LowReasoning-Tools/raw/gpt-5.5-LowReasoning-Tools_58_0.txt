const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input.charCodeAt(0) === 0xfeff ? input.slice(1) : input);

const grid = data.grid || [];
const robots = data.robots || [];
const jobs = data.jobs || [];
const h = grid.length;
const w = h ? grid[0].length : 0;
const INF = 0x3fffffff;

function cellIndex(x, y) {
  return y * w + x;
}

function openAt(x, y) {
  return y >= 0 && y < h && x >= 0 && x < w && grid[y][x] !== '#';
}

const allPoints = [];
const robotPointIndex = new Map();
const jobPointIndex = new Map();

for (const r of robots) {
  robotPointIndex.set(r.id, allPoints.length);
  allPoints.push({ x: r.x, y: r.y, key: 'r:' + r.id });
}
for (const j of jobs) {
  jobPointIndex.set(j.id, allPoints.length);
  allPoints.push({ x: j.x, y: j.y, key: 'j:' + j.id });
}

const bfsCache = new Map();

function bfsFrom(pointIdx) {
  let cached = bfsCache.get(pointIdx);
  if (cached) return cached;

  const p = allPoints[pointIdx];
  const n = h * w;
  const dist = new Int32Array(n);
  dist.fill(-1);

  if (!openAt(p.x, p.y)) {
    bfsCache.set(pointIdx, dist);
    return dist;
  }

  const q = new Int32Array(n);
  let head = 0;
  let tail = 0;
  const start = cellIndex(p.x, p.y);
  dist[start] = 0;
  q[tail++] = start;

  while (head < tail) {
    const cur = q[head++];
    const x = cur % w;
    const y = (cur / w) | 0;
    const nd = dist[cur] + 1;

    let ni = cur - 1;
    if (x > 0 && grid[y][x - 1] !== '#' && dist[ni] < 0) {
      dist[ni] = nd;
      q[tail++] = ni;
    }
    ni = cur + 1;
    if (x + 1 < w && grid[y][x + 1] !== '#' && dist[ni] < 0) {
      dist[ni] = nd;
      q[tail++] = ni;
    }
    ni = cur - w;
    if (y > 0 && grid[y - 1][x] !== '#' && dist[ni] < 0) {
      dist[ni] = nd;
      q[tail++] = ni;
    }
    ni = cur + w;
    if (y + 1 < h && grid[y + 1][x] !== '#' && dist[ni] < 0) {
      dist[ni] = nd;
      q[tail++] = ni;
    }
  }

  bfsCache.set(pointIdx, dist);
  return dist;
}

const pairCache = new Map();

function pointDistance(a, b) {
  if (a === b) return 0;
  const lo = a < b ? a : b;
  const hi = a < b ? b : a;
  const key = lo + ',' + hi;
  const cached = pairCache.get(key);
  if (cached !== undefined) return cached;

  const p = allPoints[b];
  const d = bfsFrom(a)[cellIndex(p.x, p.y)];
  const value = d >= 0 ? d : INF;
  pairCache.set(key, value);
  return value;
}

const robotIds = robots.map(r => r.id);
const maxRobotId = robotIds.length ? Math.max(...robotIds) : -1;
const assignments = Array.from({ length: Math.max(maxRobotId + 1, robots.length) }, () => []);
const routes = new Map();
const routePointIds = new Map();
const currentPoint = new Map();

for (const r of robots) {
  routes.set(r.id, []);
  routePointIds.set(r.id, []);
  currentPoint.set(r.id, robotPointIndex.get(r.id));
}

const sortedJobs = jobs.slice().sort((a, b) => {
  const pa = a.priority == null ? 0 : a.priority;
  const pb = b.priority == null ? 0 : b.priority;
  if (pb !== pa) return pb - pa;
  if (a.id !== b.id) return a.id < b.id ? -1 : 1;
  return 0;
});

function routeDistance(robotId, pts) {
  if (pts.length === 0) return 0;
  let total = pointDistance(robotPointIndex.get(robotId), pts[0]);
  for (let i = 1; i < pts.length; i++) total += pointDistance(pts[i - 1], pts[i]);
  return total;
}

function insertionAssign(jobList) {
  for (const job of jobList) {
    const jp = jobPointIndex.get(job.id);
    let bestRobot = robotIds[0];
    let bestPos = 0;
    let bestDelta = INF;

    for (const rid of robotIds) {
      const pts = routePointIds.get(rid);
      for (let pos = 0; pos <= pts.length; pos++) {
        const prev = pos === 0 ? robotPointIndex.get(rid) : pts[pos - 1];
        const next = pos === pts.length ? -1 : pts[pos];
        let delta = pointDistance(prev, jp);
        if (next >= 0) {
          delta += pointDistance(jp, next) - pointDistance(prev, next);
        }
        if (
          delta < bestDelta ||
          (delta === bestDelta && (rid < bestRobot || (rid === bestRobot && pos < bestPos)))
        ) {
          bestDelta = delta;
          bestRobot = rid;
          bestPos = pos;
        }
      }
    }

    routes.get(bestRobot).splice(bestPos, 0, job.id);
    routePointIds.get(bestRobot).splice(bestPos, 0, jp);
  }
}

function appendAssign(jobList) {
  for (const job of jobList) {
    const jp = jobPointIndex.get(job.id);
    let bestRobot = robotIds[0];
    let bestDist = INF;

    for (const rid of robotIds) {
      const d = pointDistance(currentPoint.get(rid), jp);
      if (d < bestDist || (d === bestDist && rid < bestRobot)) {
        bestDist = d;
        bestRobot = rid;
      }
    }

    routes.get(bestRobot).push(job.id);
    routePointIds.get(bestRobot).push(jp);
    currentPoint.set(bestRobot, jp);
  }
}

if (robots.length > 0) {
  const workEstimate = sortedJobs.length * Math.max(1, sortedJobs.length + robots.length);
  if (workEstimate <= 900000) insertionAssign(sortedJobs);
  else appendAssign(sortedJobs);

  if (sortedJobs.length <= 700) {
    let improved = true;
    let passes = 0;
    while (improved && passes < 3) {
      improved = false;
      passes++;

      for (const rid of robotIds) {
        const ids = routes.get(rid);
        const pts = routePointIds.get(rid);
        for (let i = 0; i + 1 < pts.length; i++) {
          const beforePrev = i === 0 ? robotPointIndex.get(rid) : pts[i - 1];
          const a = pts[i];
          const b = pts[i + 1];
          const afterNext = i + 2 < pts.length ? pts[i + 2] : -1;

          let before = pointDistance(beforePrev, a) + pointDistance(a, b);
          let after = pointDistance(beforePrev, b) + pointDistance(b, a);
          if (afterNext >= 0) {
            before += pointDistance(b, afterNext);
            after += pointDistance(a, afterNext);
          }

          if (after < before) {
            [pts[i], pts[i + 1]] = [pts[i + 1], pts[i]];
            [ids[i], ids[i + 1]] = [ids[i + 1], ids[i]];
            improved = true;
          }
        }
      }
    }
  }
}

for (const rid of robotIds) {
  assignments[rid] = routes.get(rid);
}

process.stdout.write(JSON.stringify({ assignments }));
