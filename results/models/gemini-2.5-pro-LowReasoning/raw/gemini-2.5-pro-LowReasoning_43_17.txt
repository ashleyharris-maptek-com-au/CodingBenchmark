// Basic Pixel Shader for Vulkan in HLSL

// This shader outputs a solid white color to the first render target.

// The output color. The [[vk::location(0)]] attribute specifies that this output
// corresponds to the first color attachment (render target) in the framebuffer.
// In Vulkan, this is the 'location' specified in the VkPipelineColorBlendAttachmentState.
struct PixelOutput
{
    float4 Color : SV_TARGET0;
};

// The main entry point for the pixel shader.
// It takes no inputs and returns the final pixel color.
PixelOutput main()
{
    PixelOutput output;

    // Set the output color to solid white (R, G, B, A)
    output.Color = float4(1.0f, 1.0f, 1.0f, 1.0f);

    return output;
}
