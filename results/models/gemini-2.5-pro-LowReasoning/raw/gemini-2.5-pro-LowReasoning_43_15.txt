// A pixel shader demonstrating texture sampling and color modulation.
// This is a practical example for use with Vulkan.

// Define resources: a 2D texture and a sampler for texture lookups.
// register(t0) binds the texture to texture slot 0.
// register(s0) binds the sampler to sampler slot 0.
Texture2D g_texture : register(t0);
SamplerState g_sampler : register(s0);

// Define the input structure for the pixel shader.
// This must match the output structure of the vertex shader.
struct PixelInput
{
    float4 position : SV_POSITION;    // Required for rasterization
    float2 texCoord : TEXCOORD0;      // Interpolated texture coordinates
    float4 color    : TEXCOORD1;      // Interpolated vertex color (e.g., for tinting)
};

// The output of the pixel shader is a 4-component float vector (RGBA).
// The `SV_Target` semantic indicates that this is the render target.
float4 main(PixelInput input) : SV_Target
{
    // 1. Sample the texture using the provided sampler and texture coordinates.
    float4 textureColor = g_texture.Sample(g_sampler, input.texCoord);
    
    // 2. Modulate the texture color with the interpolated vertex color.
    // This allows for effects like tinting the texture.
    return textureColor * input.color;
}
