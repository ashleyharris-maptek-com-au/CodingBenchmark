; SPIR-V
; Version: 1.0
; Generator: Gemini
; Bound: 120

OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main Invocations 1
OpExecutionMode %main InputLinesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4

; Decorations
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position

OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2

OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %outColor Location 0

; Types
%void = OpTypeVoid
%bool = OpTypeBool
%float = OpTypeFloat 32
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1

%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4

%uint_4 = OpConstant %uint 4
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3

%float_0 = OpConstant %float 0
%thickness = OpConstant %float 0.025

; Input per-vertex block for gl_Position
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

; Input user-defined variables
%arr_v3_in = OpTypeArray %v3float %uint_4
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%worldPos = OpVariable %ptr_in_arr_v3 Input
%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input

; Output per-vertex block for gl_Position
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

; Output user-defined variable
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

; Pointer types for access chains
%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_v4 = OpTypePointer Output %v4float

%main_type = OpTypeFunction %void

%main = OpFunction %void None %main_type
%label = OpLabel

; Load input positions
%ptr_pos0 = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%pos0 = OpLoad %v4float %ptr_pos0
%ptr_pos1 = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%pos1 = OpLoad %v4float %ptr_pos1
%ptr_pos2 = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%pos2 = OpLoad %v4float %ptr_pos2
%ptr_pos3 = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
%pos3 = OpLoad %v4float %ptr_pos3

; Load input colors
%ptr_color1 = OpAccessChain %ptr_in_v3 %color %int_1
%color1 = OpLoad %v3float %ptr_color1
%ptr_color2 = OpAccessChain %ptr_in_v3 %color %int_2
%color2 = OpLoad %v3float %ptr_color2

; Tangent t0 = normalize(pos2 - pos0)
%dir0 = OpFSub %v4float %pos2 %pos0
%t0 = OpExtInst %v4float %1 Normalize %dir0

; Tangent t1 = normalize(pos3 - pos1)
%dir1 = OpFSub %v4float %pos3 %pos1
%t1 = OpExtInst %v4float %1 Normalize %dir1

; Perpendiculars
; perp0 = vec2(-t0.y, t0.x)
%t0_y = OpCompositeExtract %float %t0 1
%t0_x = OpCompositeExtract %float %t0 0
%neg_t0_y = OpFNegate %float %t0_y
%perp0 = OpCompositeConstruct %v2float %neg_t0_y %t0_x

; perp1 = vec2(-t1.y, t1.x)
%t1_y = OpCompositeExtract %float %t1 1
%t1_x = OpCompositeExtract %float %t1 0
%neg_t1_y = OpFNegate %float %t1_y
%perp1 = OpCompositeConstruct %v2float %neg_t1_y %t1_x

; Scale perpendiculars by thickness
%perp0_scaled = OpVectorTimesScalar %v2float %perp0 %thickness
%perp1_scaled = OpVectorTimesScalar %v2float %perp1 %thickness

%v4_perp0_scaled = OpCompositeConstruct %v4float %perp0_scaled %float_0 %float_0
%v4_perp1_scaled = OpCompositeConstruct %v4float %perp1_scaled %float_0 %float_0

; --- Emit Vertex 1 ---
%out_pos1 = OpFAdd %v4float %pos1 %v4_perp0_scaled
%ptr_gl_Position_out1 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_gl_Position_out1 %out_pos1
OpStore %outColor %color1
OpEmitVertex

; --- Emit Vertex 2 ---
%out_pos2 = OpFSub %v4float %pos1 %v4_perp0_scaled
%ptr_gl_Position_out2 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_gl_Position_out2 %out_pos2
OpStore %outColor %color1 ; Color is the same
OpEmitVertex

; --- Emit Vertex 3 ---
%out_pos3 = OpFAdd %v4float %pos2 %v4_perp1_scaled
%ptr_gl_Position_out3 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_gl_Position_out3 %out_pos3
OpStore %outColor %color2
OpEmitVertex

; --- Emit Vertex 4 ---
%out_pos4 = OpFSub %v4float %pos2 %v4_perp1_scaled
%ptr_gl_Position_out4 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_gl_Position_out4 %out_pos4
OpStore %outColor %color2 ; Color is the same
OpEmitVertex

OpEndPrimitive
OpReturn
OpFunctionEnd
