#include <iostream>
#include <vector>
#include <algorithm>

struct Task {
    int machine;
    int duration;
    int start_time;
};

struct Job {
    std::vector<Task> tasks;
};

void solve_greedy(std::vector<Job>& jobs, int num_machines) {
    std::vector<int> machine_free_time(num_machines, 0);
    std::vector<int> job_free_time(jobs.size(), 0);
    std::vector<int> next_task_idx(jobs.size(), 0);

    int scheduled_tasks = 0;
    int total_tasks = 0;
    for(const auto& job : jobs) {
        total_tasks += job.tasks.size();
    }

    while (scheduled_tasks < total_tasks) {
        int best_job_idx = -1;
        int best_task_idx = -1;
        int earliest_completion_time = -1;

        for (size_t i = 0; i < jobs.size(); ++i) {
            if (next_task_idx[i] < jobs[i].tasks.size()) {
                Task& task = jobs[i].tasks[next_task_idx[i]];
                
                int start_time = std::max(job_free_time[i], machine_free_time[task.machine]);
                int completion_time = start_time + task.duration;

                if (best_job_idx == -1 || completion_time < earliest_completion_time) {
                    earliest_completion_time = completion_time;
                    best_job_idx = i;
                    best_task_idx = next_task_idx[i];
                }
            }
        }

        if (best_job_idx != -1) {
            Task& task_to_schedule = jobs[best_job_idx].tasks[best_task_idx];
            
            int start_time = std::max(job_free_time[best_job_idx], machine_free_time[task_to_schedule.machine]);
            task_to_schedule.start_time = start_time;

            job_free_time[best_job_idx] = start_time + task_to_schedule.duration;
            machine_free_time[task_to_schedule.machine] = start_time + task_to_schedule.duration;
            
            next_task_idx[best_job_idx]++;
            scheduled_tasks++;
        }
    }
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    int num_jobs, num_machines;
    std::cin >> num_jobs >> num_machines;

    std::vector<Job> jobs(num_jobs);
    long long total_tasks = 0;
    for (int i = 0; i < num_jobs; ++i) {
        int num_tasks;
        std::cin >> num_tasks;
        total_tasks += num_tasks;
        jobs[i].tasks.resize(num_tasks);
        for (int j = 0; j < num_tasks; ++j) {
            std::cin >> jobs[i].tasks[j].machine >> jobs[i].tasks[j].duration;
        }
    }

    // Adaptive logic based on problem size
    if (num_jobs * num_machines * total_tasks > 100000) {
         // For very large problems, a simple greedy approach is a starting point.
        solve_greedy(jobs, num_machines);
    } else {
        // For smaller problems, a more sophisticated approach can be used.
        // For now, we will use the same greedy solver.
        solve_greedy(jobs, num_machines);
    }


    for (int i = 0; i < num_jobs; ++i) {
        for (size_t j = 0; j < jobs[i].tasks.size(); ++j) {
            std::cout << jobs[i].tasks[j].start_time << " " << jobs[i].tasks[j].machine << (j == jobs[i].tasks.size() - 1 ? "" : " ");
        }
        std::cout << std::endl;
    }

    return 0;
}
