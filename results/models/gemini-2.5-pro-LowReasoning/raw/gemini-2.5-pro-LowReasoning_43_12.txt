// Basic HLSL Pixel Shader for Vulkan

// The main entry point for the pixel shader
// It is executed for each pixel on the screen
// It returns a float4, which represents the RGBA color of the pixel
float4 main() : SV_TARGET
{
    // Define a solid color (e.g., red)
    float4 color = float4(1.0f, 0.0f, 0.0f, 1.0f);

    // Return the final color
    return color;
}
