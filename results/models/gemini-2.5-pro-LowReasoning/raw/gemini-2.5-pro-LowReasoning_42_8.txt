; SPIR-V
; Version: 1.0
; Generator: Gemini
; Bound: 100

OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main Triangles
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 3
OpExecutionMode %main Invocations 1

; Debug Information
OpSource GLSL 450

; Names
OpName %main "main"
OpName %gl_PerVertex_in "gl_PerVertex_in"
OpName %gl_in "gl_in"
OpName %worldPos "worldPos"
OpName %normal "normal"
OpName %color "color"
OpName %gl_PerVertex_out "gl_PerVertex_out"
OpName %gl_out "gl_out"
OpName %outColor "outColor"

; Decorations
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %arr_pv_in ArrayStride 16
OpDecorate %gl_in Location 0
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %outColor Location 0

; Types, Variables and Constants
%void = OpTypeVoid
%void_fn = OpTypeFunction %void

%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1

%c_uint_3 = OpConstant %uint 3
%c_int_0 = OpConstant %int 0
%c_int_1 = OpConstant %int 1
%c_int_2 = OpConstant %int 2

%float_1_div_3 = OpConstant %float 0.33333334
%shrink_factor = OpConstant %float 0.7

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %c_uint_3
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3_in = OpTypeArray %v3float %c_uint_3
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in

%worldPos = OpVariable %ptr_in_arr_v3 Input
%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_in_v3 = OpTypePointer Input %v3float

; Main Function
%main = OpFunction %void None %void_fn
%entry = OpLabel

; Load positions
%pos0_ptr = OpAccessChain %ptr_in_v4 %gl_in %c_int_0 %c_int_0
%pos0 = OpLoad %v4float %pos0_ptr
%pos1_ptr = OpAccessChain %ptr_in_v4 %gl_in %c_int_1 %c_int_0
%pos1 = OpLoad %v4float %pos1_ptr
%pos2_ptr = OpAccessChain %ptr_in_v4 %gl_in %c_int_2 %c_int_0
%pos2 = OpLoad %v4float %pos2_ptr

; Compute centroid
%sum_pos = OpFAdd %v4float %pos0 %pos1
%sum_pos = OpFAdd %v4float %sum_pos %pos2
%centroid = OpVectorTimesScalar %v4float %sum_pos %float_1_div_3

; Vertex 0
%vec_to_pos0 = OpFSub %v4float %pos0 %centroid
%shrunk_vec0 = OpVectorTimesScalar %v4float %vec_to_pos0 %shrink_factor
%new_pos0 = OpFAdd %v4float %centroid %shrunk_vec0
%gl_out_pos_ptr = OpAccessChain %ptr_out_v4 %gl_out %c_int_0
OpStore %gl_out_pos_ptr %new_pos0

%color0_ptr = OpAccessChain %ptr_in_v3 %color %c_int_0
%color0 = OpLoad %v3float %color0_ptr
OpStore %outColor %color0
OpEmitVertex

; Vertex 1
%vec_to_pos1 = OpFSub %v4float %pos1 %centroid
%shrunk_vec1 = OpVectorTimesScalar %v4float %vec_to_pos1 %shrink_factor
%new_pos1 = OpFAdd %v4float %centroid %shrunk_vec1
OpStore %gl_out_pos_ptr %new_pos1

%color1_ptr = OpAccessChain %ptr_in_v3 %color %c_int_1
%color1 = OpLoad %v3float %color1_ptr
OpStore %outColor %color1
OpEmitVertex

; Vertex 2
%vec_to_pos2 = OpFSub %v4float %pos2 %centroid
%shrunk_vec2 = OpVectorTimesScalar %v4float %vec_to_pos2 %shrink_factor
%new_pos2 = OpFAdd %v4float %centroid %shrunk_vec2
OpStore %gl_out_pos_ptr %new_pos2

%color2_ptr = OpAccessChain %ptr_in_v3 %color %c_int_2
%color2 = OpLoad %v3float %color2_ptr
OpStore %outColor %color2
OpEmitVertex

OpEndPrimitive
OpReturn
OpFunctionEnd
