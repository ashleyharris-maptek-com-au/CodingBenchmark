// A basic HLSL pixel shader for Vulkan.

struct PixelInput
{
    float4 position : SV_POSITION;
    float2 uv : TEXCOORD0;
};

struct PixelOutput
{
    float4 color : SV_TARGET;
};

PixelOutput main(PixelInput input)
{
    PixelOutput output;
    output.color = float4(1.0, 0.0, 1.0, 1.0); // Output a solid magenta color
    return output;
}
