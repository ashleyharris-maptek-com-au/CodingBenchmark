// A simple pixel shader that outputs a solid color.
// This is a basic example for a Vulkan pipeline.

// The output of the pixel shader is a 4-component float vector (RGBA).
// The semantic SV_Target indicates that this is the render target.
float4 main() : SV_Target
{
    // Return a solid red color
    return float4(1.0f, 0.0f, 0.0f, 1.0f);
}
