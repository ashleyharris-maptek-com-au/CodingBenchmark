use macroquad::prelude::*;

struct Lander {
    pos: Vec2,
    vel: Vec2,
    fuel: f32,
    rotation: f32,
}

#[macroquad::main("Lunar Lander")]
async fn main() {
    let mut lander = Lander {
        pos: vec2(screen_width() / 2.0, 100.0),
        vel: vec2(0.0, 0.0),
        fuel: 1000.0,
        rotation: 0.0,
    };

    loop {
        clear_background(BLACK);

        // Bot logic
        let mut thrust_on = false;

        // Altitude control
        let altitude = screen_height() - 50.0 - lander.pos.y;
        if altitude < 200.0 {
            if lander.vel.y > 1.0 {
                thrust_on = true;
            }
        } else {
            if lander.vel.y > 2.0 {
                thrust_on = true;
            }
        }

        // Attitude control
        if lander.vel.x > 0.5 {
            lander.rotation += 1.0;
        } else if lander.vel.x < -0.5 {
            lander.rotation -= 1.0;
        } else {
            if lander.rotation > 1.0 {
                lander.rotation -= 1.0;
            } else if lander.rotation < -1.0 {
                lander.rotation += 1.0;
            }
        }

        // Physics
        lander.vel.y += 0.05; // Gravity
        if thrust_on && lander.fuel > 0.0 {
            let thrust = vec2(lander.rotation.to_radians().sin(), -lander.rotation.to_radians().cos()) * 0.15;
            lander.vel += thrust;
            lander.fuel -= 1.0;
        }
        
        lander.pos += lander.vel;

        // Draw Lander (Simple Triangle)
        draw_poly(lander.pos.x, lander.pos.y, 3, 15.0, lander.rotation, WHITE);
        
        draw_text(&format!("Fuel: {:.0}", lander.fuel), 20.0, 20.0, 30.0, WHITE);

        // Draw ground
        draw_line(0.0, screen_height() - 50.0, screen_width(), screen_height() - 50.0, 5.0, WHITE);

        // Collision detection
        if lander.pos.y > screen_height() - 50.0 {
            if lander.vel.y < 2.0 && lander.rotation.abs() < 10.0 {
                println!("Safe landing!");
            } else {
                println!("Crashed!");
            }

            // Reset
            lander = Lander {
                pos: vec2(screen_width() / 2.0, 100.0),
                vel: vec2(0.0, 0.0),
                fuel: 1000.0,
                rotation: 0.0,
            };
        }

        next_frame().await
    }
}
