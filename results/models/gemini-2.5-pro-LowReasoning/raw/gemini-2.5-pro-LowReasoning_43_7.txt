// Basic HLSL Pixel Shader for Vulkan
// This shader outputs a solid red color.

// The output of the pixel shader is a 4-component float vector (RGBA).
// [[vk::location(0)]] specifies that this output is for the first render target (attachment 0).
float4 main() : SV_TARGET
{
    // Return a solid red color (R=1, G=0, B=0, A=1)
    return float4(1.0f, 0.0f, 0.0f, 1.0f);
}
