import struct

def make_inst(opcode, *operands):
    word_count = 1 + len(operands)
    inst = [(word_count << 16) | opcode]
    inst.extend(operands)
    return inst

OpExtInstImport = 11
OpMemoryModel = 14
OpEntryPoint = 15
OpExecutionMode = 16
OpCapability = 17
OpTypeVoid = 19
OpTypeFloat = 22
OpTypeVector = 23
OpTypePointer = 32
OpTypeFunction = 33
OpConstant = 43
OpConstantComposite = 44
OpFunction = 54
OpFunctionEnd = 56
OpVariable = 59
OpLoad = 61
OpStore = 62
OpDecorate = 71
OpCompositeConstruct = 80
OpCompositeExtract = 81
OpFAdd = 129
OpFMul = 133
OpLabel = 248
OpReturn = 253
OpExtInst = 12

words = []
words.extend([0x07230203, 0x00010000, 0, 50, 0]) # Header, we'll fix bound later

def add_inst(opcode, *operands):
    words.extend(make_inst(opcode, *operands))

def str_to_words(s):
    b = s.encode('utf-8') + b'\0'
    pad = (4 - len(b) % 4) % 4
    b += b'\0' * pad
    return list(struct.unpack(f'<{len(b)//4}I', b))

add_inst(OpCapability, 1)

glsl_id = 1
words.extend(make_inst(OpExtInstImport, glsl_id, *str_to_words("GLSL.std.450")))

add_inst(OpMemoryModel, 0, 1)

entry_id = 2
in_normal = 3
out_color = 4
words.extend(make_inst(OpEntryPoint, 4, entry_id, *str_to_words("main"), in_normal, out_color))

add_inst(OpExecutionMode, entry_id, 7)

# Location=30
add_inst(OpDecorate, in_normal, 30, 1)
add_inst(OpDecorate, out_color, 30, 0)

void_t = 5
add_inst(OpTypeVoid, void_t)
func_t = 6
add_inst(OpTypeFunction, func_t, void_t)
float_t = 7
add_inst(OpTypeFloat, float_t, 32)
vec3_t = 8
add_inst(OpTypeVector, vec3_t, float_t, 3)
vec4_t = 9
add_inst(OpTypeVector, vec4_t, float_t, 4)

in_ptr_vec3_t = 10
add_inst(OpTypePointer, in_ptr_vec3_t, 1, vec3_t) # Input=1
out_ptr_vec4_t = 11
add_inst(OpTypePointer, out_ptr_vec4_t, 3, vec4_t) # Output=3

c_half = 12
add_inst(OpConstant, float_t, c_half, struct.unpack('<I', struct.pack('<f', 0.5))[0])
c_one = 13
add_inst(OpConstant, float_t, c_one, struct.unpack('<I', struct.pack('<f', 1.0))[0])
vec3_half = 14
add_inst(OpConstantComposite, vec3_t, vec3_half, c_half, c_half, c_half)

add_inst(OpVariable, in_ptr_vec3_t, in_normal, 1)
add_inst(OpVariable, out_ptr_vec4_t, out_color, 3)

add_inst(OpFunction, void_t, entry_id, 0, func_t)
label_main = 15
add_inst(OpLabel, label_main)

v_normal = 16
add_inst(OpLoad, vec3_t, v_normal, in_normal)

v_norm = 17
# Normalize = 69
add_inst(OpExtInst, vec3_t, v_norm, glsl_id, 69, v_normal)

v_mul = 18
add_inst(OpFMul, vec3_t, v_mul, v_norm, vec3_half)

v_add = 19
add_inst(OpFAdd, vec3_t, v_add, v_mul, vec3_half)

v_out_vec4 = 20
# OpCompositeConstruct ResultType ResultId Constituent ...
# Construct vec4 from vec3 and float
# Wait, CompositeConstruct can take a vector and a scalar!
add_inst(OpCompositeConstruct, vec4_t, v_out_vec4, v_add, c_one)

add_inst(OpStore, out_color, v_out_vec4)
add_inst(OpReturn)
add_inst(OpFunctionEnd)

words[3] = 21 # Bound
