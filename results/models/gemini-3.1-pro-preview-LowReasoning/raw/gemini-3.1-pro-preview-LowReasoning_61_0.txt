declare const require: any;
const fs = require('fs');

const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const depot = data.depot;
const truckCount = data.truckCount;
const packages = data.packages;

if (!packages || packages.length === 0) {
    console.log(JSON.stringify({ routes: [] }));
    process.exit(0);
}

packages.forEach((p: any) => {
    p.angle = Math.atan2(p.y - depot[1], p.x - depot[0]);
});

packages.sort((a: any, b: any) => a.angle - b.angle);

const routes: any[][] = [];
const k = Math.min(truckCount || 1, packages.length);
const chunkSize = Math.ceil(packages.length / k);

function dist(p1: any, p2: any) {
    const dx = p1.x - p2.x;
    const dy = p1.y - p2.y;
    return Math.sqrt(dx * dx + dy * dy);
}

for (let i = 0; i < k; i++) {
    const group = packages.slice(i * chunkSize, (i + 1) * chunkSize);
    if (group.length === 0) continue;
    
    let unvisited = [...group];
    let current = { x: depot[0], y: depot[1] };
    let route: any[] = [];
    
    // Nearest neighbor initialization
    while (unvisited.length > 0) {
        let bestIdx = -1;
        let bestDist = Infinity;
        for (let j = 0; j < unvisited.length; j++) {
            const d = dist(current, unvisited[j]);
            if (d < bestDist) {
                bestDist = d;
                bestIdx = j;
            }
        }
        route.push(unvisited[bestIdx]);
        current = unvisited[bestIdx];
        unvisited.splice(bestIdx, 1);
    }
    
    // 2-opt optimization
    let improved = true;
    let passes = 0;
    while (improved && passes < 20) {
        improved = false;
        passes++;
        for (let a = 0; a < route.length - 1; a++) {
            for (let b = a + 1; b < route.length; b++) {
                const pA = a === 0 ? { x: depot[0], y: depot[1] } : route[a - 1];
                const pB = route[a];
                const pC = route[b];
                const pD = b === route.length - 1 ? { x: depot[0], y: depot[1] } : route[b + 1];
                
                const dBefore = dist(pA, pB) + dist(pC, pD);
                const dAfter = dist(pA, pC) + dist(pB, pD);
                
                if (dAfter < dBefore - 1e-6) {
                    let rev = route.slice(a, b + 1).reverse();
                    route.splice(a, b - a + 1, ...rev);
                    improved = true;
                }
            }
        }
    }
    
    routes.push(route.map(p => p.id));
}

console.log(JSON.stringify({ routes }));
