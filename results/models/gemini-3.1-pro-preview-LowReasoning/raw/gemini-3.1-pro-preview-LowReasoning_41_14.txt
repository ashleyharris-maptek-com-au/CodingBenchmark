#version 450

layout(location = 0) in vec3 worldPos;
layout(location = 1) in vec3 normal;
layout(location = 2) in vec2 uv;
layout(location = 3) in vec3 tangent;
layout(location = 4) in vec3 color;

layout(location = 0) out vec4 outColor;

layout(std140, set = 0, binding = 0) uniform UBO {
  mat4 model;
  mat4 view;
  mat4 projection;
  vec4 lightPos;
  vec4 cameraPos;
  vec4 params;
} ubo;

void main() {
    vec3 baseColor = vec3(0.8, 0.2, 0.2);
    vec3 norm = normalize(normal);
    vec3 lightDir = normalize(ubo.lightPos.xyz - worldPos);
    float diffuse = dot(norm, lightDir);
    // Note: glsl usually handles max with 0.0 if not specified, but we'll leave it as dot or we can just use the dot directly since the prompt says "0.3 + 0.7*diffuse". Wait, diffuse is usually max(dot(N, L), 0.0) but I'll add the max just in case. Wait, if diffuse is just the dot product without max... no, standard diffuse is max(dot, 0). Let's use max(dot(norm, lightDir), 0.0)
    diffuse = max(diffuse, 0.0);
    
    float lambert = 0.3 + 0.7 * diffuse;
    
    float dist = distance(worldPos, ubo.cameraPos.xyz);
    float fogFactor = clamp((dist - 1.5) / 3.0, 0.0, 1.0);
    vec3 fogColor = vec3(0.7, 0.7, 0.8);
    
    vec3 finalColor = mix(baseColor * lambert, fogColor, fogFactor);
    outColor = vec4(finalColor, 1.0);
}
