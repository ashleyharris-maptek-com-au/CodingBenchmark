# P/I/D persistent states
_integral_pitch = 0.0
_integral_roll = 0.0
_integral_thrust = 0.0

def autopilot_step(state: dict, dt: float) -> dict:
    global _integral_pitch, _integral_roll, _integral_thrust
    
    # 1. Sensor Fusion
    
    def get_valid_median(v1, val1, v2, val2, v3, val3, default=0.0):
        valid_vals = []
        if val1: valid_vals.append(v1)
        if val2: valid_vals.append(v2)
        if val3: valid_vals.append(v3)
        if len(valid_vals) == 0: return default
        valid_vals.sort()
        if len(valid_vals) == 1: return valid_vals[0]
        if len(valid_vals) == 2: return (valid_vals[0] + valid_vals[1]) / 2.0
        return valid_vals[1]

    # Altitude
    alt1, val1 = state.get('baro_alt_1_ft', 0), state.get('baro_alt_1_valid', True)
    alt2, val2 = state.get('baro_alt_2_ft', 0), state.get('baro_alt_2_valid', True)
    alt3, val3 = state.get('baro_alt_3_ft', 0), state.get('baro_alt_3_valid', True)
    alt = get_valid_median(alt1, val1, alt2, val2, alt3, val3)
    
    # Airspeed
    ias1, val1 = state.get('ias_1_kt', 0), state.get('ias_1_valid', True)
    ias2, val2 = state.get('ias_2_kt', 0), state.get('ias_2_valid', True)
    ias3, val3 = state.get('ias_3_kt', 0), state.get('ias_3_valid', True)
    ias = get_valid_median(ias1, val1, ias2, val2, ias3, val3)
    
    # Heading/Pitch/Roll from IRS
    roll1, val1 = state.get('irs_1_roll_deg', 0), state.get('irs_1_valid', True)
    roll2, val2 = state.get('irs_2_roll_deg', 0), state.get('irs_2_valid', True)
    roll3, val3 = state.get('irs_3_roll_deg', 0), state.get('irs_3_valid', True)
    roll = get_valid_median(roll1, val1, roll2, val2, roll3, val3)
    
    pitch1, val1 = state.get('irs_1_pitch_deg', 0), state.get('irs_1_valid', True)
    pitch2, val2 = state.get('irs_2_pitch_deg', 0), state.get('irs_2_valid', True)
    pitch3, val3 = state.get('irs_3_pitch_deg', 0), state.get('irs_3_valid', True)
    pitch = get_valid_median(pitch1, val1, pitch2, val2, pitch3, val3)
    
    hdg1, val1 = state.get('irs_1_heading_deg', 0), state.get('irs_1_valid', True)
    hdg2, val2 = state.get('irs_2_heading_deg', 0), state.get('irs_2_valid', True)
    hdg3, val3 = state.get('irs_3_heading_deg', 0), state.get('irs_3_valid', True)
    hdg = get_valid_median(hdg1, val1, hdg2, val2, hdg3, val3)

    # Rates
    p_rate = get_valid_median(state.get('gyro_1_pitch_rate_dps',0), state.get('gyro_1_valid',True), state.get('gyro_2_pitch_rate_dps',0), state.get('gyro_2_valid',True), state.get('gyro_3_pitch_rate_dps',0), state.get('gyro_3_valid',True))
    r_rate = get_valid_median(state.get('gyro_1_roll_rate_dps',0), state.get('gyro_1_valid',True), state.get('gyro_2_roll_rate_dps',0), state.get('gyro_2_valid',True), state.get('gyro_3_roll_rate_dps',0), state.get('gyro_3_valid',True))
    y_rate = get_valid_median(state.get('gyro_1_yaw_rate_dps',0), state.get('gyro_1_valid',True), state.get('gyro_2_yaw_rate_dps',0), state.get('gyro_2_valid',True), state.get('gyro_3_yaw_rate_dps',0), state.get('gyro_3_valid',True))

    ap_engaged = state.get('ap_engaged', False)
    at_engaged = state.get('at_engaged', False)
    target_alt = state.get('target_alt_ft', alt)
    target_hdg = state.get('target_hdg_deg', hdg)
    target_ias = state.get('target_ias_kt', ias)
    
    # 2. Control logic
    # Altitude to Pitch
    alt_error = target_alt - alt
    target_pitch = max(-15.0, min(15.0, alt_error * 0.05))
    
    # Heading to Roll
    hdg_error = target_hdg - hdg
    if hdg_error > 180: hdg_error -= 360
    if hdg_error < -180: hdg_error += 360
    target_roll = max(-30.0, min(30.0, hdg_error * 1.5))
    
    # Airspeed to Thrust
    ias_error = target_ias - ias
    
    dt_safe = max(0.001, dt)
    
    # Pitch Control (Elevator)
    pitch_error = target_pitch - pitch
    _integral_pitch += pitch_error * dt_safe
    _integral_pitch = max(-20.0, min(20.0, _integral_pitch))
    elevator = pitch_error * 0.05 + _integral_pitch * 0.01 - p_rate * 0.05
    elevator = max(-1.0, min(1.0, elevator))
    
    # Roll Control (Aileron)
    roll_error = target_roll - roll
    _integral_roll += roll_error * dt_safe
    _integral_roll = max(-20.0, min(20.0, _integral_roll))
    aileron = roll_error * 0.02 + _integral_roll * 0.005 - r_rate * 0.05
    aileron = max(-1.0, min(1.0, aileron))
    
    # Yaw Dampener (Rudder)
    rudder = -y_rate * 0.05
    rudder = max(-1.0, min(1.0, rudder))
    
    # Thrust Control
    _integral_thrust += ias_error * dt_safe
    _integral_thrust = max(-50.0, min(50.0, _integral_thrust))
    throttle = 0.5 + ias_error * 0.01 + _integral_thrust * 0.005
    throttle = max(0.0, min(1.0, throttle))
    
    # Fallback to defaults if AP/AT not engaged
    if not ap_engaged:
        elevator = 0.0
        aileron = 0.0
        rudder = 0.0
    
    out = {}
    if ap_engaged:
        out['elevator'] = elevator
        out['aileron'] = aileron
        out['rudder'] = rudder
    if at_engaged:
        out['throttle_1'] = throttle
        out['throttle_2'] = throttle
        
    return out
