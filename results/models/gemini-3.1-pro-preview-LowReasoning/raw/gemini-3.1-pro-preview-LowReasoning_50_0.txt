```glsl
#version 450

layout(local_size_x = 256) in;

// Each boid stored as two vec4s: [pos.xyz, 0], [vel.xyz, 0]
// Boid i is at index i*2 and i*2+1
layout(set = 0, binding = 0) readonly buffer InputBoids {
    vec4 inData[];   // length = N * 2
};

layout(set = 0, binding = 1) buffer OutputBoids {
    vec4 outData[];  // length = N * 2
};

layout(set = 0, binding = 2) uniform Params {
    uvec4 counts;      // counts.x = N
    vec4 radii;        // radii.x=dt, radii.y=sepRadius, radii.z=alignRadius, radii.w=cohRadius
    vec4 weights;      // weights.x=sepWeight, weights.y=alignWeight, weights.z=cohWeight, weights.w=maxSpeed
    vec4 bounds;       // bounds.x=boundSize, bounds.y=boundForce
};

void main() {
    uint index = gl_GlobalInvocationID.x;
    if (index >= counts.x) return;

    vec3 pos = inData[index * 2].xyz;
    vec3 vel = inData[index * 2 + 1].xyz;

    float dt = radii.x;
    float sepRadius = radii.y;
    float alignRadius = radii.z;
    float cohRadius = radii.w;

    float sepWeight = weights.x;
    float alignWeight = weights.y;
    float cohWeight = weights.z;
    float maxSpeed = weights.w;

    float boundSize = bounds.x;
    float boundForce = bounds.y;

    vec3 sepAcc = vec3(0.0);
    vec3 alignAcc = vec3(0.0);
    vec3 cohAcc = vec3(0.0);

    int sepCount = 0;
    int alignCount = 0;
    int cohCount = 0;

    for (uint j = 0; j < counts.x; j++) {
        if (index == j) continue;

        vec3 otherPos = inData[j * 2].xyz;
        vec3 otherVel = inData[j * 2 + 1].xyz;

        vec3 diff = pos - otherPos;
        float dist = length(diff);

        if (dist > 0.0) {
            if (dist < sepRadius) {
                sepAcc += normalize(diff);
                sepCount++;
            }
            if (dist < alignRadius) {
                alignAcc += otherVel;
                alignCount++;
            }
            if (dist < cohRadius) {
                cohAcc += otherPos;
                cohCount++;
            }
        }
    }

    if (sepCount > 0) {
        sepAcc /= float(sepCount);
    }
    if (alignCount > 0) {
        alignAcc /= float(alignCount);
        alignAcc -= vel;
    }
    if (cohCount > 0) {
        cohAcc /= float(cohCount);
        cohAcc -= pos;
    }

    vec3 boundAcc = vec3(0.0);
    for (int i = 0; i < 3; ++i) {
        if (abs(pos[i]) > boundSize * 0.4) {
            boundAcc[i] = -pos[i] / (boundSize * 0.5) * boundForce;
        }
    }

    vec3 accel = sepAcc * sepWeight + alignAcc * alignWeight + cohAcc * cohWeight + boundAcc;

    vel += accel * dt;

    float speed = length(vel);
    if (speed > 0.0 && speed > maxSpeed) {
        vel = (vel / speed) * maxSpeed;
    }

    pos += vel * dt;

    outData[index * 2]     = vec4(pos, 0.0);
    outData[index * 2 + 1] = vec4(vel, 0.0);
}
```
