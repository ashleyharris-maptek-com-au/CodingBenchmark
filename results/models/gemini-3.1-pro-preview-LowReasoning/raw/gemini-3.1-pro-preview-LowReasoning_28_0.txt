#![allow(warnings)]
use std::io::{self, Read, BufWriter, Write};

struct Scanner<'a> {
    bytes: &'a [u8],
    pos: usize,
}

impl<'a> Scanner<'a> {
    fn new(bytes: &'a [u8]) -> Self {
        Self { bytes, pos: 0 }
    }
    fn next_u32(&mut self) -> Option<u32> {
        while self.pos < self.bytes.len() && self.bytes[self.pos] <= b' ' {
            self.pos += 1;
        }
        if self.pos >= self.bytes.len() {
            return None;
        }
        let mut res = 0u32;
        while self.pos < self.bytes.len() && self.bytes[self.pos] > b' ' {
            res = res * 10 + (self.bytes[self.pos] - b'0') as u32;
            self.pos += 1;
        }
        Some(res)
    }
}

fn main() {
    let mut input = Vec::new();
    io::stdin().read_to_end(&mut input).unwrap();
    let mut scanner = Scanner::new(&input);

    let num_vertices_opt = scanner.next_u32();
    if num_vertices_opt.is_none() {
        return;
    }
    let num_vertices = num_vertices_opt.unwrap() as usize;
    let num_edges = scanner.next_u32().unwrap() as usize;

    let mut edges = Vec::with_capacity(num_edges);
    for _ in 0..num_edges {
        let u = scanner.next_u32().unwrap();
        let v = scanner.next_u32().unwrap();
        edges.push((u, v));
    }

    let mut head = vec![0usize; num_vertices + 1];
    for &(u, v) in &edges {
        head[u as usize + 1] += 1;
        head[v as usize + 1] += 1;
    }
    for i in 1..=num_vertices {
        head[i] += head[i - 1];
    }
    let mut adj = vec![0u32; head[num_vertices]];
    let mut cur_head = head.clone();
    for &(u, v) in &edges {
        adj[cur_head[u as usize]] = v;
        cur_head[u as usize] += 1;
        adj[cur_head[v as usize]] = u;
        cur_head[v as usize] += 1;
    }

    let mut in_cover = vec![false; num_vertices];
    for &(u, v) in &edges {
        if !in_cover[u as usize] && !in_cover[v as usize] {
            in_cover[u as usize] = true;
            in_cover[v as usize] = true;
        }
    }

    let mut order: Vec<usize> = (0..num_vertices).collect();
    order.sort_unstable_by_key(|&u| head[u + 1] - head[u]);

    for u in order {
        if in_cover[u] {
            let mut can_remove = true;
            let start = head[u];
            let end = head[u + 1];
            for i in start..end {
                let v = adj[i] as usize;
                if !in_cover[v] {
                    can_remove = false;
                    break;
                }
            }
            if can_remove {
                in_cover[u] = false;
            }
        }
    }

    let mut cover = Vec::new();
    for i in 0..num_vertices {
        if in_cover[i] {
            cover.push(i as u32);
        }
    }

    let stdout = io::stdout();
    let mut writer = BufWriter::new(stdout.lock());
    writeln!(writer, "{}", cover.len()).unwrap();
    for (i, &v) in cover.iter().enumerate() {
        if i > 0 {
            write!(writer, " ").unwrap();
        }
        write!(writer, "{}", v).unwrap();
    }
    writeln!(writer).unwrap();
}
