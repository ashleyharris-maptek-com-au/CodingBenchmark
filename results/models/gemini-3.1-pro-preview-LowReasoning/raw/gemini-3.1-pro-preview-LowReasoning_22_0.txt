using System;
using System.Collections.Generic;

class Program
{
    static int W, H, N;
    static int[,] board;

    // Piece definitions: [rotation][y, x] relative to top-left
    static Dictionary<string, int[][,]> pieces = new Dictionary<string, int[][,]>
    {
        {"I", new int[][,] { new int[,] {{1,1,1,1}}, new int[,] {{1},{1},{1},{1}}, new int[,] {{1,1,1,1}}, new int[,] {{1},{1},{1},{1}} }},
        {"O", new int[][,] { new int[,] {{1,1},{1,1}}, new int[,] {{1,1},{1,1}}, new int[,] {{1,1},{1,1}}, new int[,] {{1,1},{1,1}} }},
        {"T", new int[][,] { new int[,] {{0,1,0},{1,1,1}}, new int[,] {{1,0},{1,1},{1,0}}, new int[,] {{1,1,1},{0,1,0}}, new int[,] {{0,1},{1,1},{0,1}} }},
        {"S", new int[][,] { new int[,] {{0,1,1},{1,1,0}}, new int[,] {{1,0},{1,1},{0,1}}, new int[,] {{0,1,1},{1,1,0}}, new int[,] {{1,0},{1,1},{0,1}} }},
        {"Z", new int[][,] { new int[,] {{1,1,0},{0,1,1}}, new int[,] {{0,1},{1,1},{1,0}}, new int[,] {{1,1,0},{0,1,1}}, new int[,] {{0,1},{1,1},{1,0}} }},
        {"J", new int[][,] { new int[,] {{1,0,0},{1,1,1}}, new int[,] {{1,1},{1,0},{1,0}}, new int[,] {{1,1,1},{0,0,1}}, new int[,] {{0,1},{0,1},{1,1}} }},
        {"L", new int[][,] { new int[,] {{0,0,1},{1,1,1}}, new int[,] {{1,0},{1,0},{1,1}}, new int[,] {{1,1,1},{1,0,0}}, new int[,] {{1,1},{0,1},{0,1}} }}
    };

    static void Main()
    {
        string[] init = Console.ReadLine().Split(new char[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        if (init.Length < 3) return;
        W = int.Parse(init[0]);
        H = int.Parse(init[1]);
        N = int.Parse(init[2]);
        board = new int[H, W];

        for (int p = 0; p < N; p++)
        {
            string pieceType = Console.ReadLine().Trim();
            
            // Adaptive logic based on game length/complexity
            double wHeight = -0.510066;
            double wLines = 0.760666;
            double wHoles = -0.35663;
            double wBump = -0.184483;

            if (p > 500) { 
                wHeight = -0.6; 
                wHoles = -0.4; 
            }

            int bestRot = 0;
            int bestCol = 0;
            double bestScore = double.MinValue;
            
            var rots = pieces[pieceType];
            
            for (int r = 0; r < 4; r++)
            {
                int[,] shape = rots[r];
                int pw = shape.GetLength(1);

                for (int c = 0; c <= W - pw; c++)
                {
                    int dropRow = GetDropRow(shape, c);
                    if (dropRow < 0) continue; // Doesn't fit

                    Place(shape, dropRow, c, 1);
                    double score = Evaluate(wHeight, wLines, wHoles, wBump);
                    Place(shape, dropRow, c, 0);

                    if (score > bestScore)
                    {
                        bestScore = score;
                        bestRot = r;
                        bestCol = c;
                    }
                }
            }

            // Fallback if no valid move
            if (bestScore == double.MinValue) {
                bestRot = 0;
                bestCol = 0;
            }

            Console.WriteLine($"{bestRot} {bestCol}");
            
            string[] res = Console.ReadLine().Split(new char[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
            if (res[0] == "gameover") break;
            
            int actualDropR = GetDropRow(rots[bestRot], bestCol);
            if (actualDropR >= 0) {
                Place(rots[bestRot], actualDropR, bestCol, 1);
            }
            
            if (res[0].StartsWith("cleared_") || res[0] == "ok")
            {
                ClearLines();
            }
        }
    }

    static int GetDropRow(int[,] shape, int col)
    {
        int ph = shape.GetLength(0);
        for (int r = 0; r <= H - ph; r++)
        {
            if (Collides(shape, r, col)) return r - 1;
        }
        return H - ph;
    }

    static bool Collides(int[,] shape, int row, int col)
    {
        int ph = shape.GetLength(0);
        int pw = shape.GetLength(1);
        if (row < 0) return true; // Above board
        if (row + ph > H || col + pw > W) return true;
        for (int i = 0; i < ph; i++)
            for (int j = 0; j < pw; j++)
                if (shape[i, j] == 1 && board[row + i, col + j] == 1)
                    return true;
        return false;
    }

    static void Place(int[,] shape, int row, int col, int val)
    {
        int ph = shape.GetLength(0);
        int pw = shape.GetLength(1);
        for (int i = 0; i < ph; i++)
            for (int j = 0; j < pw; j++)
                if (shape[i, j] == 1 && row + i >= 0 && row + i < H)
                    board[row + i, col + j] = val;
    }

    static double Evaluate(double wHeight, double wLines, double wHoles, double wBump)
    {
        int[] heights = new int[W];
        int aggregateHeight = 0;
        int holes = 0;
        int completeLines = 0;
        int bumpiness = 0;

        for (int c = 0; c < W; c++)
        {
            bool blockFound = false;
            int colHeight = 0;
            for (int r = 0; r < H; r++)
            {
                if (board[r, c] == 1)
                {
                    if (!blockFound) { colHeight = H - r; blockFound = true; }
                }
                else if (blockFound)
                {
                    holes++;
                }
            }
            heights[c] = colHeight;
            aggregateHeight += colHeight;
        }

        for (int c = 0; c < W - 1; c++)
            bumpiness += Math.Abs(heights[c] - heights[c + 1]);

        for (int r = 0; r < H; r++)
        {
            bool complete = true;
            for (int c = 0; c < W; c++)
            {
                if (board[r, c] == 0) { complete = false; break; }
            }
            if (complete) completeLines++;
        }

        return wHeight * aggregateHeight + wLines * completeLines + wHoles * holes + wBump * bumpiness;
    }

    static void ClearLines()
    {
        for (int r = H - 1; r >= 0; r--)
        {
            bool full = true;
            for (int c = 0; c < W; c++) {
                if (board[r, c] == 0) { full = false; break; }
            }
            
            if (full)
            {
                for (int yr = r; yr > 0; yr--)
                    for (int c = 0; c < W; c++)
                        board[yr, c] = board[yr - 1, c];
                for (int c = 0; c < W; c++) board[0, c] = 0;
                r++; // Check the same row index again because lines shifted down
            }
        }
    }
}
