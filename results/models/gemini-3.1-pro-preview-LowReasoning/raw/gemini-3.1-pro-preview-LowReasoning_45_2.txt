import subprocess
import binascii

glsl_code = '''#version 450
layout(location = 2) in vec2 uv;
layout(location = 0) out vec4 outColor;

void main() {
    outColor = vec4(uv.x, uv.y, 0.0, 1.0);
}
'''

with open('shader.frag', 'w') as f:
    f.write(glsl_code)

subprocess.run(['glslangValidator', '-V', '--target-env', 'vulkan1.0', 'shader.frag', '-o', 'shader.spv'])

with open('shader.spv', 'rb') as f:
    spv_bytes = f.read()

hex_str = binascii.hexlify(spv_bytes).decode('utf-8')

with open('answer.txt', 'w') as f:
    f.write(hex_str.upper())
