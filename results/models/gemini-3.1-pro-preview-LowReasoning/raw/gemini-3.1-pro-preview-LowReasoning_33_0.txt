using System;
using System.Diagnostics;
using System.IO;

class QAP
{
    static int N;
    static int[][] flow;
    static int[][] dist;

    static void Main()
    {
        Stopwatch sw = Stopwatch.StartNew();
        
        // Fast IO
        Stream inStream = Console.OpenStandardInput();
        byte[] buffer = new byte[65536];
        int bytesRead = 0;
        int bufferPos = 0;

        int ReadNextInt()
        {
            int c;
            // Skip non-digit characters
            do
            {
                if (bufferPos >= bytesRead)
                {
                    bytesRead = inStream.Read(buffer, 0, buffer.Length);
                    bufferPos = 0;
                    if (bytesRead == 0) return 0; // EOF
                }
                c = buffer[bufferPos++];
            } while ((c < '0' || c > '9') && c != '-');

            bool neg = false;
            if (c == '-')
            {
                neg = true;
                if (bufferPos >= bytesRead)
                {
                    bytesRead = inStream.Read(buffer, 0, buffer.Length);
                    bufferPos = 0;
                }
                c = buffer[bufferPos++];
            }

            int res = c - '0';
            while (true)
            {
                if (bufferPos >= bytesRead)
                {
                    bytesRead = inStream.Read(buffer, 0, buffer.Length);
                    bufferPos = 0;
                    if (bytesRead == 0) break;
                }
                c = buffer[bufferPos];
                if (c >= '0' && c <= '9')
                {
                    res = res * 10 + (c - '0');
                    bufferPos++;
                }
                else
                {
                    break;
                }
            }
            return neg ? -res : res;
        }

        N = ReadNextInt();
        if (N == 0) return;

        flow = new int[N][];
        for(int i=0; i<N; i++)
        {
            flow[i] = new int[N];
            for(int j=0; j<N; j++)
                flow[i][j] = ReadNextInt();
        }
        
        dist = new int[N][];
        for(int i=0; i<N; i++)
        {
            dist[i] = new int[N];
            for(int j=0; j<N; j++)
                dist[i][j] = ReadNextInt();
        }

        int[] best_p = new int[N];
        for(int i=0; i<N; i++) best_p[i] = i;
        long best_cost = ComputeCost(best_p);

        int[] p = new int[N];
        Random rng = new Random(42);

        // Time limit: 4.5 minutes = 270 seconds
        while (sw.ElapsedMilliseconds < 270000)
        {
            for(int i=0; i<N; i++) p[i] = i;
            for(int i=0; i<N; i++) {
                int r = i + rng.Next(N - i);
                int t = p[i]; p[i] = p[r]; p[r] = t;
            }

            long current_cost = ComputeCost(p);
            if (current_cost < best_cost)
            {
                best_cost = current_cost;
                Array.Copy(p, best_p, N);
            }

            bool improved = true;
            while(improved && sw.ElapsedMilliseconds < 270000)
            {
                improved = false;
                for(int u=0; u<N-1 && sw.ElapsedMilliseconds < 270000; u++)
                {
                    for(int v=u+1; v<N; v++)
                    {
                        long delta = ComputeDelta(p, u, v);
                        if (delta < 0)
                        {
                            current_cost += delta;
                            int t = p[u]; p[u] = p[v]; p[v] = t;
                            improved = true;
                            if (current_cost < best_cost)
                            {
                                best_cost = current_cost;
                                Array.Copy(p, best_p, N);
                            }
                        }
                    }
                }
            }
            
            // If the time limit is almost reached and N is large, break to ensure we output in time
            if (N > 1000 && sw.ElapsedMilliseconds > 250000) break;
        }

        Console.WriteLine(best_cost);
        Console.WriteLine(string.Join(" ", best_p));
    }

    static long ComputeCost(int[] p)
    {
        long c = 0;
        for(int i=0; i<N; i++)
            for(int j=0; j<N; j++)
                c += (long)flow[i][j] * dist[p[i]][p[j]];
        return c;
    }

    static long ComputeDelta(int[] p, int u, int v)
    {
        long delta = 0;
        int pu = p[u];
        int pv = p[v];
        
        for(int k=0; k<N; k++)
        {
            if (k == u || k == v) continue;
            int pk = p[k];
            delta += (long)flow[u][k] * (dist[pv][pk] - dist[pu][pk])
                   + (long)flow[k][u] * (dist[pk][pv] - dist[pk][pu])
                   + (long)flow[v][k] * (dist[pu][pk] - dist[pv][pk])
                   + (long)flow[k][v] * (dist[pk][pu] - dist[pk][pv]);
        }
        
        delta += (long)flow[u][v] * (dist[pv][pu] - dist[pu][pv])
               + (long)flow[v][u] * (dist[pu][pv] - dist[pv][pu]);
               
        return delta;
    }
}
