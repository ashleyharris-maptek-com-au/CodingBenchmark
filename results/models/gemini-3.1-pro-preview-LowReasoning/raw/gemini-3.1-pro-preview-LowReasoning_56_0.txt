const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);
    const depot = data.depot;
    const packages = data.packages;

    if (!packages || packages.length === 0) {
        console.log(JSON.stringify({ routes: [] }));
        return;
    }

    function dist(p1, p2) {
        const dx = p1.x - p2.x;
        const dy = p1.y - p2.y;
        return Math.sqrt(dx * dx + dy * dy);
    }
    
    let unvisited = packages.slice();
    let current = { x: depot[0], y: depot[1] };
    let route = [];
    
    while (unvisited.length > 0) {
        let bestDist = Infinity;
        let bestIdx = -1;
        for (let i = 0; i < unvisited.length; i++) {
            let d = dist(current, unvisited[i]);
            if (d < bestDist) {
                bestDist = d;
                bestIdx = i;
            }
        }
        let nextPkg = unvisited.splice(bestIdx, 1)[0];
        route.push(nextPkg);
        current = nextPkg;
    }
    
    let improved = true;
    while (improved) {
        improved = false;
        for (let i = 0; i < route.length - 1; i++) {
            for (let j = i + 2; j < route.length; j++) {
                if (j === route.length - 1 && i === 0) continue;
                
                let p1 = i === 0 ? {x: depot[0], y: depot[1]} : route[i - 1];
                let p2 = route[i];
                let p3 = route[j];
                let p4 = j === route.length - 1 ? {x: depot[0], y: depot[1]} : route[j + 1];
                
                let d1 = dist(p1, p2) + dist(p3, p4);
                let d2 = dist(p1, p3) + dist(p2, p4);
                
                if (d2 < d1 - 1e-6) {
                    let left = i;
                    let right = j;
                    while (left < right) {
                        let temp = route[left];
                        route[left] = route[right];
                        route[right] = temp;
                        left++;
                        right--;
                    }
                    improved = true;
                }
            }
        }
    }
    
    console.log(JSON.stringify({ routes: [route.map(p => p.id)] }));
}

solve();
