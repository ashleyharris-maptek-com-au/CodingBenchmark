#include <iostream>
#include <vector>
#include <deque>
#include <set>
#include <queue>
#include <algorithm>
#include <map>

using namespace std;

const int MAXD = 20;
int D;

struct Point {
    int x[MAXD];
    bool operator<(const Point& o) const {
        for(int i=0; i<D; ++i) {
            if(x[i] < o.x[i]) return true;
            if(x[i] > o.x[i]) return false;
        }
        return false;
    }
    bool operator==(const Point& o) const {
        for(int i=0; i<D; ++i) if(x[i] != o.x[i]) return false;
        return true;
    }
};

int max_turns;
Point bounds;
int snake_length;
Point start_pos;
int F;
set<Point> foods;
int O_obs;
set<Point> obstacles;

bool is_valid(const Point& p) {
    for(int i=0; i<D; ++i) {
        if(p.x[i] < 0 || p.x[i] >= bounds.x[i]) return false;
    }
    if(obstacles.count(p)) return false;
    return true;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    if (!(cin >> D >> max_turns)) return 0;
    for(int i=0; i<D; ++i) cin >> bounds.x[i];
    cin >> snake_length;
    for(int i=0; i<D; ++i) cin >> start_pos.x[i];
    cin >> F;
    for(int i=0; i<F; ++i) {
        Point p;
        for(int d=0; d<D; ++d) cin >> p.x[d];
        foods.insert(p);
    }
    cin >> O_obs;
    for(int i=0; i<O_obs; ++i) {
        Point p;
        for(int d=0; d<D; ++d) cin >> p.x[d];
        obstacles.insert(p);
    }

    deque<Point> snake;
    for(int i=0; i<snake_length; ++i) snake.push_back(start_pos);

    for(int turn=0; turn<max_turns; ++turn) {
        if(foods.empty()) break; // Win
        
        queue<Point> q;
        map<Point, pair<int, int>> parent_move; // child -> {axis, dir}
        map<Point, Point> parent_point;
        set<Point> visited;

        Point head = snake.front();
        q.push(head);
        visited.insert(head);

        Point target;
        bool found = false;

        set<Point> body;
        for(int i=0; i<(int)snake.size()-1; ++i) {
            body.insert(snake[i]);
        }

        while(!q.empty() && !found) {
            Point curr = q.front();
            q.pop();

            for(int i=0; i<D; ++i) {
                for(int dir : {-1, 1}) {
                    Point next = curr;
                    next.x[i] += dir;
                    
                    if(is_valid(next) && !body.count(next) && !visited.count(next)) {
                        visited.insert(next);
                        parent_move[next] = {i, dir};
                        parent_point[next] = curr;
                        q.push(next);

                        if(foods.count(next)) {
                            target = next;
                            found = true;
                            break;
                        }
                    }
                }
                if(found) break;
            }
        }

        int move_axis = 0, move_dir = 1;
        if(found) {
            Point curr = target;
            while(!(parent_point[curr] == head)) {
                curr = parent_point[curr];
            }
            move_axis = parent_move[curr].first;
            move_dir = parent_move[curr].second;
        } else {
            bool safe_found = false;
            for(int i=0; i<D && !safe_found; ++i) {
                for(int dir : {-1, 1}) {
                    Point next = head;
                    next.x[i] += dir;
                    if(is_valid(next) && !body.count(next)) {
                        move_axis = i;
                        move_dir = dir;
                        safe_found = true;
                        break;
                    }
                }
            }
        }

        cout << move_axis << " " << move_dir << "\n";
        cout.flush();

        Point next_head = head;
        next_head.x[move_axis] += move_dir;
        snake.push_front(next_head);

        if(foods.count(next_head)) {
            foods.erase(next_head);
        } else {
            snake.pop_back();
        }

        if(!is_valid(next_head) || body.count(next_head)) {
            break;
        }
    }
    return 0;
}
