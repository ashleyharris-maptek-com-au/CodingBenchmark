import math

def autopilot_step(state: dict, dt: float) -> dict:
    # 1. Read sensors with Lidar > Radar priority and fallback
    pos = [0.0, 0.0, 0.0]
    vel = [0.0, 0.0, 0.0]
    
    for i, axis in enumerate(['x', 'y', 'z']):
        # Read Position
        if state.get(f'lidar_rel_{axis}_valid', False):
            pos[i] = state.get(f'lidar_rel_pos_{axis}_m', 0.0)
        elif state.get(f'radar_rel_{axis}_valid', False):
            pos[i] = state.get(f'radar_rel_pos_{axis}_m', 0.0)
            
        # Read Velocity
        if state.get(f'lidar_relv_{axis}_valid', False):
            vel[i] = state.get(f'lidar_rel_vel_{axis}_ms', 0.0)
        elif state.get(f'radar_relv_{axis}_valid', False):
            vel[i] = state.get(f'radar_rel_vel_{axis}_ms', 0.0)
            
    dist = math.sqrt(sum(p**2 for p in pos))
    
    # 2. Velocity profile based on distance
    # Very gentle approach
    speed_limit = 0.1
    if dist > 200:
        speed_limit = 2.0
    elif dist > 50:
        speed_limit = 0.5
    elif dist > 10:
        speed_limit = 0.2
    
    # 3. PD Control
    # Target velocity is directly towards the origin (0,0,0)
    target_vel = [0.0, 0.0, 0.0]
    if dist > 0.1:
        # Scale position vector to get a desired velocity vector
        scale = speed_limit / dist
        # For distances less than 2m, slow down proportionally
        if dist < 2.0:
            scale = 0.05 
            
        target_vel = [-p * scale for p in pos]
        
    # Cap target velocity magnitude
    tv_mag = math.sqrt(sum(tv**2 for tv in target_vel))
    if tv_mag > speed_limit:
        target_vel = [tv / tv_mag * speed_limit for tv in target_vel]
        
    # Compute Thrust based on velocity error
    thrust = [0.0, 0.0, 0.0]
    
    # In orbit, if far away, we need to handle orbital mechanics intuitively
    # But a strong PD controller will just brute-force the straight line trajectory
    # which is often fine for < 2km ranges and 500kg fuel.
    
    kp = 2.0  # Velocity gain
    for i in range(3):
        err_v = target_vel[i] - vel[i]
        thrust[i] = err_v * kp
        
    # Add CW gravity gradient cancellation for X (radial) and Y (along-track)
    # n = 0.00113 rad/s (approx LEO)
    n = 0.00113
    ax_cw = 3 * n**2 * pos[0] + 2 * n * vel[1]
    ay_cw = -2 * n * vel[0]
    az_cw = -n**2 * pos[2]
    
    # We add the CW acceleration so our thrusters cancel it out
    # Assuming mass ~ 5000kg, max thrust 220N -> max accel ~ 0.044 m/s^2
    # So we map accel to thrust command [-1, 1] roughly by dividing by max_accel
    mass_estimate = 5000.0
    max_accel = 220.0 / mass_estimate
    
    thrust[0] += (ax_cw / max_accel)
    thrust[1] += (ay_cw / max_accel)
    thrust[2] += (az_cw / max_accel)
        
    # Clip thrust to [-1.0, 1.0]
    thrust = [max(-1.0, min(1.0, t)) for t in thrust]
    
    return {
        "thrust_x": thrust[0],
        "thrust_y": thrust[1],
        "thrust_z": thrust[2],
        "att_rate_roll": 0.0,
        "att_rate_pitch": 0.0,
        "att_rate_yaw": 0.0
    }
