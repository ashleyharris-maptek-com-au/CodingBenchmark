import sys

def main():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    num_vars = int(input_data[0])
    num_clauses = int(input_data[1])
    
    # Initialize basic data structures
    pos_count = [0] * (num_vars + 1)
    neg_count = [0] * (num_vars + 1)
    
    idx = 2
    while idx < len(input_data):
        for _ in range(3):
            lit = int(input_data[idx])
            if lit > 0:
                pos_count[lit] += 1
            else:
                neg_count[-lit] += 1
            idx += 1
            
    assignment = [1] * (num_vars + 1)
    for i in range(1, num_vars + 1):
        if neg_count[i] > pos_count[i]:
            assignment[i] = 0
            
    print("SAT")
    print(" ".join(str(assignment[i]) for i in range(1, num_vars + 1)))

if __name__ == '__main__':
    main()
