use std::io::{self, BufRead, Write};
use std::sync::{Arc, Mutex};
use std::thread;
use std::time::{Duration, Instant};

fn main() {
    let stdin = io::stdin();
    let mut reader = stdin.lock();
    
    // Read first line
    let mut line = String::new();
    reader.read_line(&mut line).unwrap();
    let parts: Vec<f64> = line.split_whitespace().filter_map(|s| s.parse().ok()).collect();
    if parts.len() < 5 { return; } // Invalid input, just exit
    let (_width, height, gravity, max_thrust, _init_fuel) = (parts[0], parts[1], parts[2], parts[3], parts[4]);
    
    // Read second line
    line.clear();
    reader.read_line(&mut line).unwrap();
    let parts: Vec<f64> = line.split_whitespace().filter_map(|s| s.parse().ok()).collect();
    if parts.len() < 4 { return; } // Invalid input, just exit
    let (_start_x, _start_y, target_x, target_y) = (parts[0], parts[1], parts[2], parts[3]);
    
    // Read map rows
    let h = height as usize;
    for _ in 0..h {
        line.clear();
        reader.read_line(&mut line).unwrap();
    }
    
    // Shared state: [x, y, vx, vy, angle, angular_vel, fuel_remaining]
    let state = Arc::new(Mutex::new(vec![0.0; 7]));
    let state_clone = Arc::clone(&state);
    
    // Reader thread
    thread::spawn(move || {
        let stdin = io::stdin();
        let mut reader = stdin.lock();
        let mut line = String::new();
        while let Ok(n) = reader.read_line(&mut line) {
            if n == 0 {
                break;
            }
            if line.trim().is_empty() {
                line.clear();
                continue;
            }
            let parts: Vec<f64> = line.split_whitespace().filter_map(|s| s.parse().ok()).collect();
            if parts.len() >= 7 {
                let mut s = state_clone.lock().unwrap();
                s[0] = parts[0];
                s[1] = parts[1];
                s[2] = parts[2];
                s[3] = parts[3];
                s[4] = parts[4];
                s[5] = parts[5];
                s[6] = parts[6];
            }
            line.clear();
        }
    });

    let mut stdout = io::stdout();
    let dt = Duration::from_millis(30); // ~33 Hz
    
    loop {
        let start_time = Instant::now();
        
        let current_state = {
            let s = state.lock().unwrap();
            s.clone()
        };
        
        let x = current_state[0];
        let y = current_state[1];
        let vx = current_state[2];
        let vy = current_state[3];
        let angle = current_state[4];
        let angular_vel = current_state[5];
        
        // Simple heuristic controller
        let dx = target_x - x;
        let dy = target_y - y;
        
        // desired velocities
        let desired_vx = (dx * 0.5).clamp(-15.0, 15.0);
        let desired_vy = if dy > 50.0 {
            10.0
        } else if dy > 10.0 {
            5.0
        } else {
            1.5 // soft landing
        };
        
        let err_vx = desired_vx - vx;
        let err_vy = desired_vy - vy;
        
        // Target angle based on err_vx
        // If we want to go right (err_vx > 0), we tilt right (angle > 0)
        let target_angle = (err_vx * 0.1).clamp(-0.8, 0.8);
        
        let err_angle = target_angle - angle;
        // PD control for angle
        let turn = (err_angle * 5.0 - angular_vel * 2.0).clamp(-1.0, 1.0);
        
        // Thrust control
        // Base thrust to counteract gravity: gravity / (max_thrust * cos(angle))
        let base_thrust = gravity / max_thrust;
        
        // Thrust to correct vy
        // positive vy is down. If we are falling too fast (vy > desired_vy), err_vy < 0.
        // We need more thrust to slow down.
        let thrust_adj = -err_vy * 0.2; 
        
        let thrust = ((base_thrust + thrust_adj) / angle.cos().max(0.5)).clamp(0.0, 1.0);
        
        if let Err(_) = writeln!(stdout, "{:.4} {:.4}", thrust, turn) {
            break; // Engine pipe closed
        }
        if let Err(_) = stdout.flush() {
            break;
        }
        
        let elapsed = start_time.elapsed();
        if elapsed < dt {
            thread::sleep(dt - elapsed);
        }
    }
}
