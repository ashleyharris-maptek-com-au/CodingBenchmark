import os
import subprocess
import urllib.request
import zipfile

glsl_code = """#version 450
layout(location = 0) in vec3 worldPos;
layout(location = 1) in vec3 normal;
layout(location = 2) in vec2 uv;
layout(location = 3) in vec3 tangent;
layout(location = 4) in vec3 color;

layout(location = 0) out vec4 outColor;

layout(set = 0, binding = 0) uniform UBO {
    mat4 model;
    mat4 view;
    mat4 proj;
    vec4 lightPos;
    vec4 cameraPos;
    vec4 params;
} ubo;

const float PI = 3.14159265359;

void main() {
    float stripe = sin(uv.y * PI * 16.0);
    vec3 red = vec3(0.9, 0.1, 0.1);
    vec3 blue = vec3(0.1, 0.1, 0.9);
    vec3 baseColor = (stripe > 0.0) ? red : blue;
    
    vec3 N = normalize(normal);
    vec3 L = normalize(ubo.lightPos.xyz - worldPos);
    float NdotL = max(dot(N, L), 0.0);
    
    outColor = vec4(baseColor * NdotL, 1.0);
}
"""

with open("shader.frag", "w") as f:
    f.write(glsl_code)

def get_compiler():
    try:
        subprocess.run(["glslc", "--version"], stdout=subprocess.DEVNULL)
        return "glslc"
    except FileNotFoundError:
        pass
    try:
        subprocess.run(["glslangValidator", "--version"], stdout=subprocess.DEVNULL)
        return "glslangValidator"
    except FileNotFoundError:
        pass
    
    print("Downloading glslang...")
    url = "https://github.com/KhronosGroup/glslang/releases/download/main-tot/glslang-master-windows-x64-Release.zip"
    urllib.request.urlretrieve(url, "glslang.zip")
    with zipfile.ZipFile("glslang.zip", 'r') as zip_ref:
        zip_ref.extractall("glslang")
    return r"glslang\bin\glslangValidator.exe"

compiler = get_compiler()
print("Using compiler:", compiler)

if compiler == "glslc":
    subprocess.run([compiler, "shader.frag", "-o", "shader.spv"], check=True)
else:
    subprocess.run([compiler, "-V", "shader.frag", "-o", "shader.spv"], check=True)

with open("shader.spv", "rb") as f:
    spv_data = f.read()

hex_str = spv_data.hex().upper()
with open("answer.txt", "w") as f:
    f.write(hex_str)

print("Done")
