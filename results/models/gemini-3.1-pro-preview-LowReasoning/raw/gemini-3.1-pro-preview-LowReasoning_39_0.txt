use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = if let Some(Ok(line)) = lines.next() { line } else { return; };
    let parts: Vec<usize> = first_line.split_whitespace().filter_map(|s| s.parse().ok()).collect();
    if parts.len() < 2 { return; }
    let n = parts[0];
    let m = parts[1];

    let mut adj = vec![vec![]; n];
    for _ in 0..m {
        if let Some(Ok(line)) = lines.next() {
            let edge: Vec<usize> = line.split_whitespace().filter_map(|s| s.parse().ok()).collect();
            if edge.len() >= 2 {
                adj[edge[0]].push(edge[1]);
                adj[edge[1]].push(edge[0]);
            }
        }
    }

    if n == 0 {
        println!("0\n");
        return;
    }

    let mut best_cut = m + 1;
    let mut best_a = vec![];
    let mut current_a = Vec::with_capacity(n / 2);
    let mut current_b = Vec::with_capacity(n / 2);

    let mut assignment = vec![0u8; n]; // 0: unassigned, 1: A, 2: B

    fn solve(
        u: usize,
        n: usize,
        adj: &[Vec<usize>],
        assignment: &mut Vec<u8>,
        current_a: &mut Vec<usize>,
        current_b: &mut Vec<usize>,
        current_cut: usize,
        best_cut: &mut usize,
        best_a: &mut Vec<usize>,
    ) {
        if current_cut >= *best_cut {
            return;
        }
        if current_a.len() > n / 2 || current_b.len() > n / 2 {
            return;
        }
        if u == n {
            if current_cut < *best_cut {
                *best_cut = current_cut;
                *best_a = current_a.clone();
            }
            return;
        }

        // Lower bound check
        let mut lb = current_cut;
        for i in u..n {
            let mut edges_to_a = 0;
            let mut edges_to_b = 0;
            for &v in &adj[i] {
                if assignment[v] == 1 {
                    edges_to_a += 1;
                } else if assignment[v] == 2 {
                    edges_to_b += 1;
                }
            }
            lb += std::cmp::min(edges_to_a, edges_to_b);
        }
        if lb >= *best_cut {
            return;
        }

        // Try assigning to A
        if current_a.len() < n / 2 {
            let mut added_cut = 0;
            for &v in &adj[u] {
                if assignment[v] == 2 {
                    added_cut += 1;
                }
            }
            assignment[u] = 1;
            current_a.push(u);
            solve(u + 1, n, adj, assignment, current_a, current_b, current_cut + added_cut, best_cut, best_a);
            current_a.pop();
            assignment[u] = 0;
        }

        // Try assigning to B (skip if u == 0 to break symmetry)
        if u > 0 && current_b.len() < n / 2 {
            let mut added_cut = 0;
            for &v in &adj[u] {
                if assignment[v] == 1 {
                    added_cut += 1;
                }
            }
            assignment[u] = 2;
            current_b.push(u);
            solve(u + 1, n, adj, assignment, current_a, current_b, current_cut + added_cut, best_cut, best_a);
            current_b.pop();
            assignment[u] = 0;
        }
    }

    current_a.push(0);
    assignment[0] = 1;
    solve(1, n, &adj, &mut assignment, &mut current_a, &mut current_b, 0, &mut best_cut, &mut best_a);

    println!("{}", best_cut);
    for i in 0..best_a.len() {
        print!("{}{}", best_a[i], if i + 1 == best_a.len() { "" } else { " " });
    }
    println!();
}
