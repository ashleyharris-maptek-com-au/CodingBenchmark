declare const require: any;
const fs = require('fs');

const startTime = Date.now();
const timeLimit = 4500; // 4.5 seconds

const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const width = data.width;
const height = data.height;
const grid = data.grid;
const ignitions = data.ignitions;
const towns = data.towns;
const cutBudget = data.cutBudget;

if (cutBudget === 0) {
    console.log(JSON.stringify({ firebreaks: [] }));
    process.exit(0);
}

const numCells = width * height;
const isBurnable = new Uint8Array(numCells);
const isTown = new Uint8Array(numCells);

for (let y = 0; y < height; y++) {
    const row = grid[y];
    for (let x = 0; x < width; x++) {
        isBurnable[y * width + x] = row[x] === '.' ? 1 : 0;
    }
}

for (const town of towns) {
    isTown[town.y * width + town.x] = 1;
}
for (const [x, y] of ignitions) {
    isTown[y * width + x] = 1;
}

const distArray = new Int32Array(numCells);
const qArray = new Int32Array(numCells);
const visitedArray = new Uint32Array(numCells);
const usageArray = new Float64Array(numCells); // Float64 for large values
let usageCandidates: number[] = [];
let visitedToken = 0;

function getDistances(cuts: Set<number>) {
    distArray.fill(-1);
    let head = 0;
    let tail = 0;
    
    for (const [x, y] of ignitions) {
        const idx = y * width + x;
        distArray[idx] = 0;
        qArray[tail++] = idx;
    }
    
    while (head < tail) {
        const curr = qArray[head++];
        const d = distArray[curr];
        
        const cy = Math.floor(curr / width);
        const cx = curr % width;
        
        for (let i = 0; i < 4; i++) {
            let nx = cx;
            let ny = cy;
            if (i === 0) ny--;
            else if (i === 1) ny++;
            else if (i === 2) nx--;
            else if (i === 3) nx++;
            
            if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                const nidx = ny * width + nx;
                if (isBurnable[nidx] === 1 && !cuts.has(nidx) && distArray[nidx] === -1) {
                    distArray[nidx] = d + 1;
                    qArray[tail++] = nidx;
                }
            }
        }
    }
}

function getCellUsage(cuts: Set<number>) {
    getDistances(cuts);
    usageArray.fill(0);
    usageCandidates = [];
    
    for (const town of towns) {
        const tIdx = town.y * width + town.x;
        if (distArray[tIdx] === -1) continue; 
        
        visitedToken++;
        visitedArray[tIdx] = visitedToken;
        
        let head = 0;
        let tail = 0;
        qArray[tail++] = tIdx;
        
        while(head < tail) {
            const curr = qArray[head++];
            
            if (usageArray[curr] === 0 && isTown[curr] === 0 && !cuts.has(curr) && isBurnable[curr] === 1) {
                usageCandidates.push(curr);
            }
            usageArray[curr] += town.value;
            
            const d = distArray[curr];
            if (d === 0) continue;
            
            const cy = Math.floor(curr / width);
            const cx = curr % width;
            
            for (let i = 0; i < 4; i++) {
                let nx = cx;
                let ny = cy;
                if (i === 0) ny--;
                else if (i === 1) ny++;
                else if (i === 2) nx--;
                else if (i === 3) nx++;
                
                if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                    const nidx = ny * width + nx;
                    if (distArray[nidx] === d - 1 && visitedArray[nidx] !== visitedToken) {
                        visitedArray[nidx] = visitedToken;
                        qArray[tail++] = nidx;
                    }
                }
            }
        }
    }
}

function getSmoothScore(): number {
    let score = 0;
    for (const town of towns) {
        const idx = town.y * width + town.x;
        const d = distArray[idx];
        if (d === -1) {
            score += town.value * 10000;
        } else {
            score += town.value * (1 - 1 / (d + 1));
        }
    }
    return score;
}

function getSavedValue(): number {
    let saved = 0;
    for (const town of towns) {
        const idx = town.y * width + town.x;
        const d = distArray[idx];
        if (d === -1) {
            saved += town.value;
        }
    }
    return saved;
}

function randomizedGreedy(): Set<number> {
    const cuts = new Set<number>();
    while (cuts.size < cutBudget) {
        getCellUsage(cuts);
        
        if (usageCandidates.length === 0) break;
        
        usageCandidates.sort((a, b) => usageArray[b] - usageArray[a]);
        
        const topUsage = usageArray[usageCandidates[0]];
        let topCount = 0;
        while (topCount < usageCandidates.length && usageArray[usageCandidates[topCount]] >= topUsage * 0.8) {
            topCount++;
        }
        
        const topK = Math.min(topCount, 5);
        cuts.add(usageCandidates[Math.floor(Math.random() * topK)]);
    }
    return cuts;
}

function mutate(cuts: Set<number>): Set<number> {
    const newCuts = new Set(cuts);
    if (newCuts.size === 0) return newCuts;
    
    const numToRemove = Math.floor(Math.random() * Math.min(3, newCuts.size)) + 1;
    const cutsArray = Array.from(newCuts);
    for (let i = 0; i < numToRemove; i++) {
        const idx = Math.floor(Math.random() * cutsArray.length);
        newCuts.delete(cutsArray[idx]);
        cutsArray.splice(idx, 1);
    }
    
    while (newCuts.size < cutBudget) {
        getCellUsage(newCuts);
        
        if (usageCandidates.length === 0) break;
        
        usageCandidates.sort((a, b) => usageArray[b] - usageArray[a]);
        
        const topUsage = usageArray[usageCandidates[0]];
        let topCount = 0;
        while (topCount < usageCandidates.length && usageArray[usageCandidates[topCount]] >= topUsage * 0.8) {
            topCount++;
        }
        
        const topK = Math.min(topCount, 5);
        newCuts.add(usageCandidates[Math.floor(Math.random() * topK)]);
    }
    
    return newCuts;
}

getDistances(new Set());
let bestActualValue = getSavedValue();
let bestSmoothScoreOfBest = getSmoothScore();
let bestCuts = new Set<number>();

let currentCuts = randomizedGreedy();
getDistances(currentCuts);
let currentScore = getSmoothScore();
let currentActualValue = getSavedValue();

if (currentActualValue > bestActualValue) {
    bestActualValue = currentActualValue;
    bestSmoothScoreOfBest = currentScore;
    bestCuts = new Set(currentCuts);
} else if (currentActualValue === bestActualValue && currentScore > bestSmoothScoreOfBest) {
    bestSmoothScoreOfBest = currentScore;
    bestCuts = new Set(currentCuts);
}

let temperature = 1000;
const coolingRate = 0.995;

while (Date.now() - startTime < timeLimit) {
    let nextCuts: Set<number>;
    if (Math.random() < 0.2 || currentCuts.size === 0) {
        nextCuts = randomizedGreedy();
    } else {
        nextCuts = mutate(currentCuts);
    }
    
    getDistances(nextCuts);
    const score = getSmoothScore();
    const actualValue = getSavedValue();
    
    if (actualValue > bestActualValue) {
        bestActualValue = actualValue;
        bestSmoothScoreOfBest = score;
        bestCuts = new Set(nextCuts);
    } else if (actualValue === bestActualValue && score > bestSmoothScoreOfBest) {
        bestSmoothScoreOfBest = score;
        bestCuts = new Set(nextCuts);
    }
    
    if (score > currentScore || Math.random() < Math.exp((score - currentScore) / temperature)) {
        currentCuts = nextCuts;
        currentScore = score;
    }
    
    temperature *= coolingRate;
    if (temperature < 1e-3) temperature = 1000;
}

const firebreaks: number[][] = [];
for (const cut of bestCuts) {
    firebreaks.push([cut % width, Math.floor(cut / width)]);
}

console.log(JSON.stringify({ firebreaks }));
