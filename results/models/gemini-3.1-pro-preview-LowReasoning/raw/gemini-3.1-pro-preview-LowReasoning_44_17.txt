#version 450

layout(location = 0) in vec3 worldPos;
layout(location = 1) in vec3 normal;
layout(location = 2) in vec2 uv;
layout(location = 3) in vec3 tangent;
layout(location = 4) in vec3 color;

layout(set = 0, binding = 0) uniform UBO {
    mat4 model;
    mat4 view;
    mat4 projection;
    vec4 lightPos;
    vec4 cameraPos;
    vec4 params;
} ubo;

layout(location = 0) out vec4 outColor;

void main() {
    vec2 scaledUv = uv * vec2(8.0, 16.0);
    float row = floor(scaledUv.y);
    if (mod(row, 2.0) > 0.5) {
        scaledUv.x += 0.5;
    }
    vec2 brickUv = fract(scaledUv);

    float bX = step(0.05, brickUv.x);
    float bY = step(0.1, brickUv.y);
    float isBrick = bX * bY;

    vec3 texColor = mix(vec3(0.7, 0.7, 0.7), vec3(0.7, 0.2, 0.1), isBrick);

    vec3 N = normalize(normal);
    vec3 L = normalize(ubo.lightPos.xyz - worldPos);
    float diffuse = max(dot(N, L), 0.0);
    float shading = 0.4 + 0.6 * diffuse;

    outColor = vec4(texColor * shading, 1.0);
}
