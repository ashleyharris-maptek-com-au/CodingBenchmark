#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <random>
#include <chrono>

using namespace std;

int main() {
    // Fast I/O
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, m, k;
    if (!(cin >> n >> m >> k)) return 0;

    // Efficient adjacency list for large graphs
    vector<vector<int>> adj(n);
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    // Initialize random number generator
    mt19937 rng(42);

    vector<int> color(n, -1);
    vector<int> order(n);
    iota(order.begin(), order.end(), 0);

    // Initial assignment: Largest Degree First (LDF) heuristic
    sort(order.begin(), order.end(), [&](int a, int b) {
        return adj[a].size() > adj[b].size();
    });

    vector<int> used(k, -1);
    for (int u : order) {
        for (int v : adj[u]) {
            if (color[v] != -1) {
                used[color[v]] = u; // mark color as used for node u
            }
        }
        int c = 0;
        while (c < k && used[c] == u) {
            c++;
        }
        if (c < k) {
            color[u] = c;
        } else {
            // If k colors are not enough greedily, assign a random color
            color[u] = uniform_int_distribution<int>(0, k - 1)(rng);
        }
    }

    // Local Search: Min-Conflicts Heuristic
    auto start_time = chrono::steady_clock::now();
    int max_time_seconds = 290; // Time limit is 5 mins, we use 290 seconds to be safe

    // Adaptive logic: loop until no conflicts or time limit exceeded
    while (true) {
        auto now = chrono::steady_clock::now();
        if (chrono::duration_cast<chrono::seconds>(now - start_time).count() > max_time_seconds) {
            break; // Stop if time limit approached
        }

        bool any_conflict = false;
        vector<int> conflict_nodes;
        
        // Find nodes in conflict
        for (int i = 0; i < n; ++i) {
            bool conflict = false;
            for (int v : adj[i]) {
                if (color[i] == color[v]) {
                    conflict = true;
                    break;
                }
            }
            if (conflict) {
                conflict_nodes.push_back(i);
                any_conflict = true;
            }
        }

        if (!any_conflict) {
            break; // Valid coloring found
        }

        // Shuffle conflict nodes for stochastic neighbor exploration
        shuffle(conflict_nodes.begin(), conflict_nodes.end(), rng);

        // Try to resolve conflicts
        for (int u : conflict_nodes) {
            vector<int> counts(k, 0);
            for (int v : adj[u]) {
                counts[color[v]]++;
            }
            
            int min_c = color[u];
            int min_count = counts[color[u]];
            
            // Find color with minimum conflicts
            vector<int> best_colors;
            for (int c = 0; c < k; ++c) {
                if (counts[c] < min_count) {
                    min_count = counts[c];
                    best_colors.clear();
                    best_colors.push_back(c);
                } else if (counts[c] == min_count) {
                    best_colors.push_back(c);
                }
            }
            
            if (!best_colors.empty()) {
                // Break ties randomly
                color[u] = best_colors[uniform_int_distribution<int>(0, best_colors.size() - 1)(rng)];
            }
        }
    }

    // Output formatting
    for (int i = 0; i < n; ++i) {
        cout << color[i] << (i == n - 1 ? "" : " ");
    }
    cout << "\n";

    return 0;
}
