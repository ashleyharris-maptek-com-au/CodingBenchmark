#include <iostream>
#include <string>
#include <vector>
#include <thread>
#include <atomic>
#include <cmath>
#include <fstream>
#include <mutex>

using namespace std;

// Globals
double width, depth, height, gravity, max_thrust, initial_fuel;
double start_x, start_y, start_z, target_x, target_y, target_z;
string terrain_file_path;

struct State {
    double x, y, z;
    double vx, vy, vz;
    double pitch, yaw, roll;
    double pitch_rate, yaw_rate, roll_rate;
    double fuel_remaining;
};

atomic<bool> terrain_loaded{false};

void terrain_loader_thread() {
    // We could load the terrain here, but for a simple agent we just fly high.
    // std::ifstream ifs(terrain_file_path, std::ios::binary);
    // ...
    terrain_loaded = true;
}

double clamp(double v, double min_v, double max_v) {
    if (v < min_v) return min_v;
    if (v > max_v) return max_v;
    return v;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    if (!(cin >> width >> depth >> height >> gravity >> max_thrust >> initial_fuel)) return 0;
    if (!(cin >> start_x >> start_y >> start_z >> target_x >> target_y >> target_z)) return 0;
    
    // Read terrain path (might have spaces, so read rest of line, then next line)
    string dummy;
    getline(cin, dummy);
    getline(cin, terrain_file_path);
    
    string state_header;
    cin >> state_header; // "STATE"
    
    thread loader(terrain_loader_thread);
    loader.detach();

    State s;
    while (cin >> s.x >> s.y >> s.z >> s.vx >> s.vy >> s.vz >> s.pitch >> s.yaw >> s.roll >> s.pitch_rate >> s.yaw_rate >> s.roll_rate >> s.fuel_remaining) {
        
        // Simple PD controller
        double dx = target_x - s.x;
        double dy = target_y - s.y;
        double dz = target_z - s.z;
        
        double dist_xy = sqrt(dx*dx + dy*dy);
        
        // Desired velocities
        double desired_vx = clamp(dx * 0.1, -10.0, 10.0);
        double desired_vy = clamp(dy * 0.1, -10.0, 10.0);
        double desired_vz = clamp(dz * 0.1, -5.0, 5.0);
        
        // Velocity errors
        double err_vx = desired_vx - s.vx;
        double err_vy = desired_vy - s.vy;
        double err_vz = desired_vz - s.vz;
        
        // Required accelerations
        double acc_x = err_vx * 0.5;
        double acc_y = err_vy * 0.5;
        double acc_z = err_vz * 0.5 + gravity; // Counteract gravity
        
        // Determine thrust and desired angles
        // In local coordinates:
        // +Z is thrust direction. 
        // pitch tilts forward (-X in local?)
        
        // For simplicity, just point upright and hover down if close
        double thrust = 0.0;
        double pitch_cmd = 0.0;
        double yaw_cmd = 0.0;
        double roll_cmd = 0.0;
        
        if (acc_z > 0) {
            thrust = clamp(acc_z / max_thrust, 0.0, 1.0);
        } else {
            thrust = 0.0;
        }
        
        // Basic angle stabilization
        pitch_cmd = clamp(-s.pitch - s.pitch_rate * 0.5, -1.0, 1.0);
        yaw_cmd = clamp(-s.yaw - s.yaw_rate * 0.5, -1.0, 1.0);
        roll_cmd = clamp(-s.roll - s.roll_rate * 0.5, -1.0, 1.0);
        
        cout << thrust << " " << pitch_cmd << " " << yaw_cmd << " " << roll_cmd << "\n";
        cout.flush();
    }
    
    return 0;
}
