integral_gs = 0.0
integral_loc = 0.0
integral_speed = 0.0

def autopilot_step(state: dict, dt: float) -> dict:
    global integral_gs, integral_loc, integral_speed
    
    # Sensor fusion / voting
    def get_valid_value(keys, valids, default=0.0):
        for k, v in zip(keys, valids):
            if state.get(v, False):
                return state.get(k, default)
        return state.get(keys[0], default)

    gs = get_valid_value(['glideslope_1_dots', 'glideslope_2_dots', 'glideslope_3_dots'], 
                         ['glideslope_1_valid', 'glideslope_2_valid', 'glideslope_3_valid'])
    loc = get_valid_value(['localizer_1_dots', 'localizer_2_dots', 'localizer_3_dots'],
                          ['localizer_1_valid', 'localizer_2_valid', 'localizer_3_valid'])
    
    alt = state.get('radio_alt_ft', 0.0)
    if state.get('radio_alt_ft', None) is None: # all failed?
        alt = state.get('gps_1_alt_ft', 0.0) - state.get('target_alt_ft', 0.0)

    ias = get_valid_value(['ias_1_kt', 'ias_2_kt', 'ias_3_kt'],
                          ['ias_1_valid', 'ias_2_valid', 'ias_3_valid'], 137.0)
    target_speed = state.get('target_vref_kt', 137.0)

    wow = state.get('weight_on_wheels', False)

    out = {
        'flaps': 40.0,
        'gear': 1.0,
        'speedbrake': 0.0,
        'spoiler': 0.0,
        'autobrake': 0.0,
        'reverse_1': 0.0,
        'reverse_2': 0.0,
        'elevator': 0.0,
        'aileron': 0.0,
        'rudder': 0.0,
        'throttle_1': 0.0,
        'throttle_2': 0.0
    }

    if wow:
        # Ground rollout
        out['spoiler'] = 1.0
        out['autobrake'] = 1.0
        if ias > 60:
            out['reverse_1'] = 1.0
            out['reverse_2'] = 1.0
        out['throttle_1'] = 0.0
        out['throttle_2'] = 0.0
        out['elevator'] = -0.5
        out['rudder'] = loc * 0.5
        out['aileron'] = loc * 0.1
        return out

    # Airborne
    if alt < 50 and alt > 0:
        # Flare
        target_pitch_up = 0.5
        out['elevator'] = target_pitch_up * (50 - alt) / 50.0
        out['throttle_1'] = 0.0
        out['throttle_2'] = 0.0
    else:
        # Approach
        integral_gs += gs * dt
        integral_loc += loc * dt
        
        # Pitch control based on glideslope
        # positive gs means above glideslope, need to pitch down
        pitch_cmd = -0.2 * gs - 0.05 * integral_gs
        out['elevator'] = max(-1.0, min(1.0, pitch_cmd))
        
        # Speed control
        speed_err = target_speed - ias
        integral_speed += speed_err * dt
        thrust_cmd = 0.5 + 0.05 * speed_err + 0.01 * integral_speed
        
        eng1_ok = state.get('engine_1_running', True)
        eng2_ok = state.get('engine_2_running', True)
        
        if eng1_ok and eng2_ok:
            out['throttle_1'] = max(0.0, min(1.0, thrust_cmd))
            out['throttle_2'] = max(0.0, min(1.0, thrust_cmd))
        elif eng1_ok:
            out['throttle_1'] = max(0.0, min(1.0, thrust_cmd * 2))
            out['throttle_2'] = 0.0
            out['rudder'] += 0.2
            out['aileron'] += 0.1
        elif eng2_ok:
            out['throttle_1'] = 0.0
            out['throttle_2'] = max(0.0, min(1.0, thrust_cmd * 2))
            out['rudder'] -= 0.2
            out['aileron'] -= 0.1

    # Lateral control
    # positive loc means right of centerline, need to turn left
    roll_cmd = 0.3 * loc + 0.01 * integral_loc
    out['aileron'] = max(-1.0, min(1.0, out['aileron'] + roll_cmd))
    out['rudder'] = max(-1.0, min(1.0, out['rudder'] + roll_cmd * 0.2))

    return out
