using System;

class Program
{
    static void Main()
    {
        // Read initial parameters
        string[] initialParams = Console.ReadLine()?.Split((char[])null, StringSplitOptions.RemoveEmptyEntries);
        if (initialParams == null || initialParams.Length < 6) return;
        
        if (!int.TryParse(initialParams[0], out int numBodies)) return;
        
        // Read body parameters
        for (int i = 0; i < numBodies; i++)
        {
            Console.ReadLine(); // Read body line
        }
        
        // Read file path
        Console.ReadLine();

        // Output dummy response
        Console.WriteLine("1");
        Console.WriteLine("0 0 0 0");
    }
}
