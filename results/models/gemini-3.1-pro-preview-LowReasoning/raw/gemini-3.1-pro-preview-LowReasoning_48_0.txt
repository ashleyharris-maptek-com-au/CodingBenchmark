#version 450

layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;

layout(std430, set = 0, binding = 0) readonly buffer Buf0 {
    float f_in[];
};

layout(std430, set = 0, binding = 1) buffer Buf1 {
    float f_out[];
};

layout(std140, set = 0, binding = 2) uniform Uniforms {
    uvec4 dims; // W, H, 0, 0
    vec4 params; // omega, 0, 0, 0
};

const int cx[9] = int[9](0, 1, 0, -1, 0, 1, -1, -1, 1);
const int cy[9] = int[9](0, 0, 1, 0, -1, 1, 1, -1, -1);
const float w[9] = float[9](
    4.0/9.0,
    1.0/9.0, 1.0/9.0, 1.0/9.0, 1.0/9.0,
    1.0/36.0, 1.0/36.0, 1.0/36.0, 1.0/36.0
);

void main() {
    uint W = dims.x;
    uint H = dims.y;
    float omega = params.x;

    uint x = gl_GlobalInvocationID.x;
    uint y = gl_GlobalInvocationID.y;

    if (x >= W || y >= H) {
        return;
    }

    uint base_idx = (y * W + x) * 9u;
    
    float f[9];
    float rho = 0.0;
    float ux = 0.0;
    float uy = 0.0;
    
    for (int k = 0; k < 9; ++k) {
        float val = f_in[base_idx + uint(k)];
        f[k] = val;
        rho += val;
        ux += val * float(cx[k]);
        uy += val * float(cy[k]);
    }
    
    ux /= rho;
    uy /= rho;

    float u2 = ux * ux + uy * uy;

    for (int k = 0; k < 9; ++k) {
        float cu = float(cx[k]) * ux + float(cy[k]) * uy;
        float feq = w[k] * rho * (1.0 + 3.0 * cu + 4.5 * cu * cu - 1.5 * u2);
        float fout_val = f[k] + omega * (feq - f[k]);
        
        int nx = (int(x) + cx[k] + int(W)) % int(W);
        int ny = (int(y) + cy[k] + int(H)) % int(H);
        
        uint out_idx = (uint(ny) * W + uint(nx)) * 9u + uint(k);
        f_out[out_idx] = fout_val;
    }
}
