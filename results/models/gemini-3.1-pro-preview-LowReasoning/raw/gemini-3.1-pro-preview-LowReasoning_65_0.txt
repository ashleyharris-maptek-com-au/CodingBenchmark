declare const require: any;
const fs = require('fs');

const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const tasks = data.tasks;
const machineCount = data.machineCount;

const inDegree: Record<string, number> = {};
const graph: Record<string, string[]> = {};
const taskMap: Record<string, any> = {};

for (const task of tasks) {
    taskMap[task.id] = task;
    inDegree[task.id] = 0;
    if (!graph[task.id]) {
        graph[task.id] = [];
    }
}

for (const task of tasks) {
    if (task.deps) {
        for (const dep of task.deps) {
            if (!graph[dep]) graph[dep] = [];
            graph[dep].push(task.id);
            inDegree[task.id] = (inDegree[task.id] || 0) + 1;
        }
    }
}

const available: string[] = [];
for (const task of tasks) {
    if (inDegree[task.id] === 0) {
        available.push(task.id);
    }
}

const order: string[] = [];

while (available.length > 0) {
    available.sort((a, b) => {
        const ta = taskMap[a];
        const tb = taskMap[b];
        
        // Priority 1: Earliest deadline
        const d_a = ta.deadline ?? Infinity;
        const d_b = tb.deadline ?? Infinity;
        if (d_a !== d_b) return d_a - d_b;
        
        // Priority 2: Highest weight
        const w_a = ta.weight ?? 0;
        const w_b = tb.weight ?? 0;
        if (w_a !== w_b) return w_b - w_a;
        
        // Priority 3: Shortest duration
        const dur_a = ta.duration ?? 0;
        const dur_b = tb.duration ?? 0;
        return dur_a - dur_b;
    });

    const current = available.shift()!;
    order.push(current);

    for (const next of graph[current] || []) {
        inDegree[next]--;
        if (inDegree[next] === 0) {
            available.push(next);
        }
    }
}
