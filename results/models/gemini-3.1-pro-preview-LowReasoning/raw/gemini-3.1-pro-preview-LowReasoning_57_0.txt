const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);
    
    const W = data.width;
    const H = data.height;
    const gridStrings = data.grid;
    const ignitions = data.ignitions;
    const towns = data.towns;
    const cutBudget = data.cutBudget;
    
    const grid = new Array(H);
    for(let y=0; y<H; y++) {
        grid[y] = gridStrings[y].split('');
    }
    
    const isIgnition = new Uint8Array(W * H);
    for(let [x, y] of ignitions) {
        isIgnition[y * W + x] = 1;
    }
    
    const isTown = new Uint8Array(W * H);
    for(let t of towns) {
        isTown[t.y * W + t.x] = 1;
    }
    
    const MAX_NODES = 2 * W * H + 2;
    const MAX_EDGES = 10 * W * H + ignitions.length * 2 + 10;
    
    const head = new Int32Array(MAX_NODES).fill(-1);
    const to = new Int32Array(MAX_EDGES);
    const nxt = new Int32Array(MAX_EDGES);
    const cap = new Int32Array(MAX_EDGES);
    const flow = new Int32Array(MAX_EDGES);
    let edgeCount = 0;
    
    function addEdge(u, v, c) {
        to[edgeCount] = v;
        cap[edgeCount] = c;
        flow[edgeCount] = 0;
        nxt[edgeCount] = head[u];
        head[u] = edgeCount++;
        
        to[edgeCount] = u;
        cap[edgeCount] = 0;
        flow[edgeCount] = 0;
        nxt[edgeCount] = head[v];
        head[v] = edgeCount++;
    }
    
    const S = 2 * W * H;
    const T_node = 2 * W * H + 1;
    
    function buildBaseGraph() {
        head.fill(-1, 0, T_node + 1);
        edgeCount = 0;
        for(let y=0; y<H; y++) {
            for(let x=0; x<W; x++) {
                let i = y * W + x;
                if (grid[y][x] === '~') continue;
                
                let u_in = 2 * i;
                let u_out = 2 * i + 1;
                
                let c = 1;
                if (isIgnition[i] || isTown[i]) {
                    c = 1000000;
                }
                addEdge(u_in, u_out, c);
                
                if (isIgnition[i]) {
                    addEdge(S, u_in, 1000000);
                }
                
                if (x > 0 && grid[y][x-1] !== '~') addEdge(u_out, 2 * (i - 1), 1000000);
                if (x < W - 1 && grid[y][x+1] !== '~') addEdge(u_out, 2 * (i + 1), 1000000);
                if (y > 0 && grid[y-1][x] !== '~') addEdge(u_out, 2 * (i - W), 1000000);
                if (y < H - 1 && grid[y+1][x] !== '~') addEdge(u_out, 2 * (i + W), 1000000);
            }
        }
    }
    
    const level = new Int32Array(MAX_NODES);
    const ptr = new Int32Array(MAX_NODES);
    const q = new Int32Array(MAX_NODES);
    
    function bfs() {
        level.fill(-1, 0, T_node + 1);
        level[S] = 0;
        let headQ = 0, tailQ = 0;
        q[tailQ++] = S;
        while(headQ < tailQ) {
            let u = q[headQ++];
            for(let e = head[u]; e !== -1; e = nxt[e]) {
                let v = to[e];
                if (level[v] === -1 && cap[e] - flow[e] > 0) {
                    level[v] = level[u] + 1;
                    q[tailQ++] = v;
                }
            }
        }
        return level[T_node] !== -1;
    }
    
    function dfs(u, pushed) {
        if (pushed === 0) return 0;
        if (u === T_node) return pushed;
        for(let e = ptr[u]; e !== -1; e = nxt[e]) {
            ptr[u] = e;
            let v = to[e];
            let tr = cap[e] - flow[e];
            if (level[u] + 1 !== level[v] || tr === 0) continue;
            let push = dfs(v, Math.min(pushed, tr));
            if (push === 0) continue;
            flow[e] += push;
            flow[e ^ 1] -= push;
            return push;
        }
        return 0;
    }
    
    function dinic() {
        let maxFlow = 0;
        while(bfs()) {
            for(let i=0; i<=T_node; i++) ptr[i] = head[i];
            while(true) {
                let pushed = dfs(S, 1000000);
                if (pushed === 0) break;
                maxFlow += pushed;
            }
        }
        return maxFlow;
    }
    
    let current_budget = cutBudget;
    let final_firebreaks = [];
    const visited = new Uint8Array(W * H);
    
    while(current_budget > 0) {
        let currentBurningValue = 0;
        let currentlyBurningTowns = [];
        
        visited.fill(0);
        let headQ = 0, tailQ = 0;
        for(let [x, y] of ignitions) {
            let i = y * W + x;
            if (grid[y][x] !== '~') {
                visited[i] = 1;
                q[tailQ++] = i;
            }
        }
        
        while(headQ < tailQ) {
            let i = q[headQ++];
            let x = i % W;
            let y = Math.floor(i / W);
            if (x > 0 && grid[y][x-1] !== '~' && !visited[i-1]) { visited[i-1] = 1; q[tailQ++] = i-1; }
            if (x < W-1 && grid[y][x+1] !== '~' && !visited[i+1]) { visited[i+1] = 1; q[tailQ++] = i+1; }
            if (y > 0 && grid[y-1][x] !== '~' && !visited[i-W]) { visited[i-W] = 1; q[tailQ++] = i-W; }
            if (y < H-1 && grid[y+1][x] !== '~' && !visited[i+W]) { visited[i+W] = 1; q[tailQ++] = i+W; }
        }
        
        for(let t of towns) {
            if (visited[t.y * W + t.x]) {
                currentBurningValue += t.value;
                currentlyBurningTowns.push(t);
            }
        }
        
        if (currentlyBurningTowns.length === 0) break;
        
        buildBaseGraph();
        
        let bestCut = null;
        let bestMetric = -1;
        let bestValueSaved = -1;
        let bestCutSize = 1000000;
        
        for(let t of currentlyBurningTowns) {
            let t_u_out = 2 * (t.y * W + t.x) + 1;
            
            let prevHeadOut = head[t_u_out];
            let prevHeadT = head[T_node];
            
            addEdge(t_u_out, T_node, 1000000);
            
            let max_flow = dinic();
            
            if (max_flow > 0 && max_flow <= current_budget) {
                let cutCellsArray = [];
                for(let i=0; i<W*H; i++) {
                    if (grid[Math.floor(i/W)][i%W] === '~') continue;
                    if (level[2*i] !== -1 && level[2*i+1] === -1) {
                        cutCellsArray.push([i%W, Math.floor(i/W)]);
                    }
                }
                
                for(let [cx, cy] of cutCellsArray) {
                    grid[cy][cx] = '~';
                }
                
                let burningValueWithThisCut = 0;
                visited.fill(0);
                headQ = 0; tailQ = 0;
                for(let [x, y] of ignitions) {
                    let i = y * W + x;
                    if (grid[y][x] !== '~') {
                        visited[i] = 1;
                        q[tailQ++] = i;
                    }
                }
                while(headQ < tailQ) {
                    let i = q[headQ++];
                    let x = i % W;
                    let y = Math.floor(i / W);
                    if (x > 0 && grid[y][x-1] !== '~' && !visited[i-1]) { visited[i-1] = 1; q[tailQ++] = i-1; }
                    if (x < W-1 && grid[y][x+1] !== '~' && !visited[i+1]) { visited[i+1] = 1; q[tailQ++] = i+1; }
                    if (y > 0 && grid[y-1][x] !== '~' && !visited[i-W]) { visited[i-W] = 1; q[tailQ++] = i-W; }
                    if (y < H-1 && grid[y+1][x] !== '~' && !visited[i+W]) { visited[i+W] = 1; q[tailQ++] = i+W; }
                }
                for(let town of currentlyBurningTowns) {
                    if (visited[town.y * W + town.x]) {
                        burningValueWithThisCut += town.value;
                    }
                }
                
                for(let [cx, cy] of cutCellsArray) {
                    grid[cy][cx] = '.';
                }
                
                let valueSaved = currentBurningValue - burningValueWithThisCut;
                let metric = valueSaved / cutCellsArray.length;
                
                if (metric > bestMetric || (metric === bestMetric && cutCellsArray.length < bestCutSize)) {
                    bestMetric = metric;
                    bestCut = cutCellsArray;
                    bestValueSaved = valueSaved;
                    bestCutSize = cutCellsArray.length;
                }
            }
            
            head[t_u_out] = prevHeadOut;
            head[T_node] = prevHeadT;
            edgeCount -= 2;
            
            flow.fill(0, 0, edgeCount);
        }
        
        if (bestCut === null || bestValueSaved === 0) break;
        
        for(let [cx, cy] of bestCut) {
            grid[cy][cx] = '~';
            final_firebreaks.push([cx, cy]);
        }
        current_budget -= bestCut.length;
    }
    
    console.log(JSON.stringify({firebreaks: final_firebreaks}));
}

solve();
