```rust
use std::io::{self, BufRead};
use std::collections::HashMap;

fn main() {
    let stdin = io::stdin();
    let mut input = stdin.lock().lines().filter_map(Result::ok).flat_map(|line| {
        let nums: Vec<u64> = line.split_whitespace()
            .filter_map(|s| s.parse::<u64>().ok())
            .collect();
        nums
    });

    let n = match input.next() {
        Some(val) => val as usize,
        None => return,
    };
    let target = match input.next() {
        Some(val) => val,
        None => return,
    };

    let mut nums = Vec::with_capacity(n);
    for _ in 0..n {
        if let Some(val) = input.next() {
            nums.push(val);
        }
    }

    if target == 0 {
        println!("YES\n[]");
        return;
    }

    let total_sum: u64 = nums.iter().sum();
    if total_sum < target {
        println!("NO");
        return;
    }
    if total_sum == target {
        println!("YES");
        let indices: Vec<usize> = (0..n).collect();
        let formatted = indices.iter().map(|x| x.to_string()).collect::<Vec<_>>().join(" ");
        println!("[{}]", formatted);
        return;
    }

    if n <= 40 {
        let mid = n / 2;
        let mut left_sums: HashMap<u64, Vec<usize>> = HashMap::new();
        
        for i in 0..(1u64 << mid) {
            let mut sum = 0;
            let mut indices = Vec::new();
            for j in 0..mid {
                if (i & (1 << j)) != 0 {
                    sum += nums[j];
                    indices.push(j);
                }
            }
            left_sums.insert(sum, indices);
        }

        let right_count = n - mid;
        for i in 0..(1u64 << right_count) {
            let mut sum = 0;
            let mut indices = Vec::new();
            for j in 0..right_count {
                if (i & (1 << j)) != 0 {
                    sum += nums[mid + j];
                    indices.push(mid + j);
                }
            }
            
            if sum <= target {
                let rem = target - sum;
                if let Some(left_indices) = left_sums.get(&rem) {
                    println!("YES");
                    let mut all_indices = left_indices.clone();
                    all_indices.extend(indices);
                    all_indices.sort_unstable();
                    let formatted = all_indices.iter().map(|x| x.to_string()).collect::<Vec<_>>().join(" ");
                    println!("[{}]", formatted);
                    return;
                }
            }
        }
        println!("NO");
        return;
    }

    if target <= 100_000_000 {
        let mut parent = vec![-1i32; (target + 1) as usize];
        parent[0] = -2;
        
        for (i, &num_val) in nums.iter().enumerate() {
            let num = num_val as usize;
            if num > target as usize {
                continue;
            }
            for s in (num..=target as usize).rev() {
                if parent[s - num] != -1 && parent[s] == -1 {
                    parent[s] = i as i32;
                }
            }
            if parent[target as usize] != -1 {
                break;
            }
        }

        if parent[target as usize] != -1 {
            println!("YES");
            let mut indices = Vec::new();
            let mut curr = target as usize;
            while curr > 0 {
                let idx = parent[curr] as usize;
                indices.push(idx);
                curr -= nums[idx] as usize;
            }
            indices.sort_unstable();
            let formatted = indices.iter().map(|x| x.to_string()).collect::<Vec<_>>().join(" ");
            println!("[{}]", formatted);
            return;
        }
    }
    
    // Fallback for very large datasets where exact is difficult
    println!("NO");
}
```
