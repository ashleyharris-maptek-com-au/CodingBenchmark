declare const require: any;
const fs = require('fs');

interface Pos { x: number, y: number }
interface Job { id: string, x: number, y: number, priority: number }
interface Robot { id: string, x: number, y: number }

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);
    
    const grid: string[] = data.grid;
    const robots: Robot[] = data.robots;
    const jobs: Job[] = data.jobs;
    
    const R = grid.length;
    const C = grid[0].length;
    
    const gridFlat = new Uint8Array(R * C);
    for (let r = 0; r < R; r++) {
        for (let c = 0; c < C; c++) {
            if (grid[r][c] === '#') gridFlat[r * C + c] = 1;
        }
    }
    
    const visited = new Uint32Array(R * C);
    const queue = new Int32Array(R * C);
    const dists = new Int32Array(R * C);
    let currentTag = 0;
    
    function getDistances(startPos: Pos): Int32Array {
        currentTag++;
        let head = 0, tail = 0;
        
        const startIdx = startPos.y * C + startPos.x;
        visited[startIdx] = currentTag;
        dists[startIdx] = 0;
        queue[tail++] = startIdx;
        
        const dirs = [-C, C, -1, 1];
        
        while (head < tail) {
            const curr = queue[head++];
            const d = dists[curr];
            
            for (let i = 0; i < 4; i++) {
                const next = curr + dirs[i];
                
                if (i === 2 && curr % C === 0) continue;
                if (i === 3 && curr % C === C - 1) continue;
                
                if (next >= 0 && next < R * C && gridFlat[next] === 0) {
                    if (visited[next] !== currentTag) {
                        visited[next] = currentTag;
                        dists[next] = d + 1;
                        queue[tail++] = next;
                    }
                }
            }
        }
        
        const res = new Int32Array(jobs.length);
        for (let i = 0; i < jobs.length; i++) {
            const tIdx = jobs[i].y * C + jobs[i].x;
            if (visited[tIdx] === currentTag) {
                res[i] = dists[tIdx];
            } else {
                res[i] = 1e9;
            }
        }
        return res;
    }
    
    const robotStartDists = new Map<string, Int32Array>();
    const robotDists = new Map<string, Int32Array>();
    for (const r of robots) {
        const d = getDistances(r);
        robotStartDists.set(r.id.toString(), d);
        robotDists.set(r.id.toString(), d);
    }
    
    const jobDists = new Map<number, Int32Array>();
    
    const unassigned = new Set<number>();
    for (let i = 0; i < jobs.length; i++) unassigned.add(i);
    
    const robotRoutes = new Map<string, number[]>();
    for (const r of robots) robotRoutes.set(r.id.toString(), []);
    
    const robotClosest = new Map<string, {jobIdx: number, dist: number}>();
    
    function updateClosest(rId: string) {
        let minDist = 1e9 + 1;
        let minJobIdx = -1;
        const rd = robotDists.get(rId)!;
        for (const jIdx of unassigned) {
            const d = rd[jIdx];
            if (d < minDist) {
                minDist = d;
                minJobIdx = jIdx;
            }
        }
        robotClosest.set(rId, {jobIdx: minJobIdx, dist: minDist});
    }
    
    for (const r of robots) updateClosest(r.id.toString());
    
    while (unassigned.size > 0) {
        let bestDist = 1e9 + 2;
        let bestJobIdx = -1;
        let bestRobotId = "";
        
        for (const [rId, closest] of robotClosest.entries()) {
            if (closest.dist < bestDist) {
                bestDist = closest.dist;
                bestJobIdx = closest.jobIdx;
                bestRobotId = rId;
            }
        }
        
        unassigned.delete(bestJobIdx);
        robotRoutes.get(bestRobotId)!.push(bestJobIdx);
        
        const newDists = getDistances(jobs[bestJobIdx]);
        jobDists.set(bestJobIdx, newDists);
        robotDists.set(bestRobotId, newDists);
        
        if (unassigned.size === 0) break;
        
        updateClosest(bestRobotId);
        
        for (const [rId, closest] of robotClosest.entries()) {
            if (rId !== bestRobotId && closest.jobIdx === bestJobIdx) {
                updateClosest(rId);
            }
        }
    }
    
    const startTime = Date.now();
    let globalImproved = true;
    let iterations = 0;
    
    while (globalImproved && iterations < 10 && Date.now() - startTime < 5000) {
        globalImproved = false;
        iterations++;
        
        // 2-opt
        for (const [rId, route] of robotRoutes.entries()) {
            let improved = true;
            while (improved && Date.now() - startTime < 5000) {
                improved = false;
                for (let i = 0; i < route.length - 1; i++) {
                    for (let j = i + 1; j < route.length; j++) {
                        const prev = i === 0 ? robotStartDists.get(rId)! : jobDists.get(route[i - 1])!;
                        const nextNode = j === route.length - 1 ? null : route[j + 1];
                        
                        const currentEdge1 = prev[route[i]];
                        const currentEdge2 = nextNode === null ? 0 : jobDists.get(route[j])![nextNode];
                        
                        const newEdge1 = prev[route[j]];
                        const newEdge2 = nextNode === null ? 0 : jobDists.get(route[i])![nextNode];
                        
                        if (newEdge1 + newEdge2 < currentEdge1 + currentEdge2) {
                            const rev = route.slice(i, j + 1).reverse();
                            for (let k = 0; k < rev.length; k++) {
                                route[i + k] = rev[k];
                            }
                            improved = true;
                            globalImproved = true;
                        }
                    }
                }
            }
        }
        
        // Relocate
        let improvedRelocate = true;
        while (improvedRelocate && Date.now() - startTime < 5000) {
            improvedRelocate = false;
            for (const r1 of robots) {
                const route1 = robotRoutes.get(r1.id.toString())!;
                for (let i = 0; i < route1.length; i++) {
                    const jobIdx = route1[i];
                    
                    const prev1 = i === 0 ? robotStartDists.get(r1.id.toString())! : jobDists.get(route1[i - 1])!;
                    const next1 = i === route1.length - 1 ? null : route1[i + 1];
                    const currentCost1 = prev1[jobIdx] + (next1 === null ? 0 : jobDists.get(jobIdx)![next1]);
                    const newCost1 = next1 === null ? 0 : prev1[next1];
                    const removalSavings = currentCost1 - newCost1;
                    
                    let bestMove = null;
                    let bestSavings = 0;
                    
                    for (const r2 of robots) {
                        if (r1.id.toString() === r2.id.toString()) continue;
                        const route2 = robotRoutes.get(r2.id.toString())!;
                        for (let j = 0; j <= route2.length; j++) {
                            const prev2 = j === 0 ? robotStartDists.get(r2.id.toString())! : jobDists.get(route2[j - 1])!;
                            const next2 = j === route2.length ? null : route2[j];
                            
                            const currentCost2 = next2 === null ? 0 : prev2[next2];
                            const newCost2 = prev2[jobIdx] + (next2 === null ? 0 : jobDists.get(jobIdx)![next2]);
                            const insertionCost = newCost2 - currentCost2;
                            
                            const totalSavings = removalSavings - insertionCost;
                            if (totalSavings > bestSavings) {
                                bestSavings = totalSavings;
                                bestMove = { r2: r2.id.toString(), j: j };
                            }
                        }
                    }
                    
                    if (bestMove) {
                        route1.splice(i, 1);
                        robotRoutes.get(bestMove.r2)!.splice(bestMove.j, 0, jobIdx);
                        improvedRelocate = true;
                        globalImproved = true;
                        break;
                    }
                }
                if (improvedRelocate) break;
            }
        }
    }
    
    let maxId = 0;
    for (const r of robots) {
        maxId = Math.max(maxId, Number(r.id));
    }
    
    const assignments: string[][] = [];
    for(let i = 0; i <= maxId; i++) {
        assignments[i] = [];
    }
    
    for (const r of robots) {
        assignments[Number(r.id)] = robotRoutes.get(r.id.toString())!.map(idx => jobs[idx].id);
    }
    
    console.log(JSON.stringify({ assignments }));
}

solve();
