use std::io::{self, BufRead, BufReader, BufWriter, Read, Write};

fn main() -> io::Result<()> {
    let stdin = io::stdin();
    let mut reader = BufReader::new(stdin.lock());
    
    let mut first_line = String::new();
    reader.read_line(&mut first_line)?;
    let parts: Vec<&str> = first_line.trim().split_whitespace().collect();
    if parts.len() < 2 {
        return Ok(());
    }
    
    let num_points: usize = parts[0].parse().unwrap_or(0);
    let num_clusters: usize = parts[1].parse().unwrap_or(0);

    let stdout = io::stdout();
    let mut writer = BufWriter::new(stdout.lock());

    if num_clusters == 0 || num_points == 0 {
        return Ok(());
    }

    // Adaptive threshold: load fully in memory if small enough, otherwise stream.
    if num_points <= 5_000_000 && num_points * num_clusters <= 1_000_000_000 {
        // --- Small / Medium datasets: In-memory k-means ---
        let mut points = Vec::with_capacity(num_points);
        let mut buf = [0u8; 12];
        for _ in 0..num_points {
            if reader.read_exact(&mut buf).is_err() { break; }
            let x = f32::from_le_bytes(buf[0..4].try_into().unwrap());
            let y = f32::from_le_bytes(buf[4..8].try_into().unwrap());
            let z = f32::from_le_bytes(buf[8..12].try_into().unwrap());
            points.push([x, y, z]);
        }
        
        let actual_points = points.len();
        if actual_points == 0 { return Ok(()); }
        
        let k = num_clusters.min(actual_points);
        let mut centroids = points.iter().take(k).cloned().collect::<Vec<_>>();
        let mut assignments = vec![0u32; actual_points];
        
        for _ in 0..10 {
            let mut new_centroids = vec![[0.0, 0.0, 0.0]; k];
            let mut counts = vec![0; k];
            
            for (i, p) in points.iter().enumerate() {
                let mut best_c = 0;
                let mut best_d = f32::MAX;
                for (c_idx, c) in centroids.iter().enumerate() {
                    let d = (p[0]-c[0])*(p[0]-c[0]) + (p[1]-c[1])*(p[1]-c[1]) + (p[2]-c[2])*(p[2]-c[2]);
                    if d < best_d {
                        best_d = d;
                        best_c = c_idx;
                    }
                }
                assignments[i] = best_c as u32;
                new_centroids[best_c][0] += p[0];
                new_centroids[best_c][1] += p[1];
                new_centroids[best_c][2] += p[2];
                counts[best_c] += 1;
            }
            
            for c in 0..k {
                if counts[c] > 0 {
                    centroids[c][0] = new_centroids[c][0] / counts[c] as f32;
                    centroids[c][1] = new_centroids[c][1] / counts[c] as f32;
                    centroids[c][2] = new_centroids[c][2] / counts[c] as f32;
                }
            }
        }
        
        for a in assignments {
            writer.write_all(&a.to_le_bytes())?;
        }
    } else {
        // --- Large / Extreme datasets: 1-pass Streaming online k-means ---
        let mut centroids = Vec::with_capacity(num_clusters);
        let mut counts = vec![0; num_clusters];
        let mut buf = [0u8; 12];
        
        for i in 0..num_points {
            if reader.read_exact(&mut buf).is_err() { break; }
            let x = f32::from_le_bytes(buf[0..4].try_into().unwrap());
            let y = f32::from_le_bytes(buf[4..8].try_into().unwrap());
            let z = f32::from_le_bytes(buf[8..12].try_into().unwrap());
            let p = [x, y, z];
            
            if i < num_clusters {
                centroids.push(p);
                counts[i] += 1;
                writer.write_all(&(i as u32).to_le_bytes())?;
            } else {
                // Heuristic: for massive cluster counts, we downsample the search space
                // to stay within the 5-minute timeout constraint.
                let step = if num_clusters > 500 { num_clusters / 100 } else { 1 };
                
                let mut best_c = 0;
                let mut best_d = f32::MAX;
                
                for c_idx in (0..num_clusters).step_by(step) {
                    let c = &centroids[c_idx];
                    let d = (p[0]-c[0])*(p[0]-c[0]) + (p[1]-c[1])*(p[1]-c[1]) + (p[2]-c[2])*(p[2]-c[2]);
                    if d < best_d {
                        best_d = d;
                        best_c = c_idx;
                    }
                }
                
                counts[best_c] += 1;
                let lr = 1.0 / counts[best_c] as f32;
                centroids[best_c][0] += (p[0] - centroids[best_c][0]) * lr;
                centroids[best_c][1] += (p[1] - centroids[best_c][1]) * lr;
                centroids[best_c][2] += (p[2] - centroids[best_c][2]) * lr;
                
                writer.write_all(&(best_c as u32).to_le_bytes())?;
            }
        }
    }
    
    writer.flush()?;
    Ok(())
}
