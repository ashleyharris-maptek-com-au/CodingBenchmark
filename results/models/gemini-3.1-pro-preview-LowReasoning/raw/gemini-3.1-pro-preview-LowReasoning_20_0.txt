use std::collections::HashMap;
use std::io::{self, BufRead};

#[derive(Debug)]
struct Sample {
    hole_id_str: String,
    sample_idx: usize,
    pos: [f64; 3],
    props: Vec<f64>,
}

#[derive(Debug)]
struct Outlier {
    hole_id: String,
    sample_idx: usize,
    prop_name: String,
    confidence: f64,
}

fn median(data: &mut [f64]) -> f64 {
    if data.is_empty() {
        return f64::NAN;
    }
    data.sort_unstable_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
    let mid = data.len() / 2;
    if data.len() % 2 == 0 {
        (data[mid - 1] + data[mid]) / 2.0
    } else {
        data[mid]
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines().filter_map(|l| l.ok()).filter(|l| !l.trim().is_empty());

    let first_line = if let Some(l) = lines.next() { l } else { return; };
    let parts: Vec<&str> = first_line.split_whitespace().collect();
    if parts.len() < 2 { return; }
    let num_holes: usize = parts[0].parse().unwrap_or(0);
    let num_props: usize = parts[1].parse().unwrap_or(0);

    let prop_names_line = if let Some(l) = lines.next() { l } else { return; };
    let prop_names: Vec<String> = prop_names_line.split_whitespace().map(|s| s.to_string()).collect();

    let mut samples: Vec<Sample> = Vec::new();
    let mut spacings = Vec::new();

    for _ in 0..num_holes {
        if let Some(hole_line) = lines.next() {
            let parts: Vec<&str> = hole_line.split_whitespace().collect();
            if parts.len() < 9 { continue; }
            let hole_id = parts[0].to_string();
            let start_x: f64 = parts[1].parse().unwrap_or(0.0);
            let start_y: f64 = parts[2].parse().unwrap_or(0.0);
            let start_z: f64 = parts[3].parse().unwrap_or(0.0);
            let dir_x: f64 = parts[4].parse().unwrap_or(0.0);
            let dir_y: f64 = parts[5].parse().unwrap_or(0.0);
            let dir_z: f64 = parts[6].parse().unwrap_or(0.0);
            let _length: f64 = parts[7].parse().unwrap_or(0.0);
            let num_samples: usize = parts[8].parse().unwrap_or(0);

            let len = (dir_x * dir_x + dir_y * dir_y + dir_z * dir_z).sqrt();
            let (dx, dy, dz) = if len > 1e-6 {
                (dir_x / len, dir_y / len, dir_z / len)
            } else {
                (0.0, 0.0, 1.0)
            };

            let mut prev_depth = f64::NAN;

            for s_idx in 0..num_samples {
                if let Some(sample_line) = lines.next() {
                    let s_parts: Vec<&str> = sample_line.split_whitespace().collect();
                    if s_parts.is_empty() { continue; }
                    let depth: f64 = s_parts[0].parse().unwrap_or(0.0);
                    
                    if !prev_depth.is_nan() {
                        spacings.push((depth - prev_depth).abs());
                    }
                    prev_depth = depth;

                    let mut props = Vec::with_capacity(num_props);
                    for i in 0..num_props {
                        props.push(s_parts.get(i + 1).unwrap_or(&"NaN").parse().unwrap_or(f64::NAN));
                    }
                    samples.push(Sample {
                        hole_id_str: hole_id.clone(),
                        sample_idx: s_idx,
                        pos: [
                            start_x + dx * depth,
                            start_y + dy * depth,
                            start_z + dz * depth,
                        ],
                        props,
                    });
                }
            }
        }
    }

    if samples.is_empty() {
        println!("0");
        return;
    }

    let mut outliers = Vec::new();

    // Adaptive logic based on scale
    if samples.len() < 2000 {
        // Brute force O(N^2) exact KNN approach for small datasets
        for i in 0..samples.len() {
            let s = &samples[i];
            let mut dists: Vec<(usize, f64)> = samples.iter().enumerate().map(|(j, s2)| {
                let dx = s.pos[0] - s2.pos[0];
                let dy = s.pos[1] - s2.pos[1];
                let dz = s.pos[2] - s2.pos[2];
                (j, dx * dx + dy * dy + dz * dz)
            }).collect();
            
            dists.sort_unstable_by(|a, b| a.1.partial_cmp(&b.1).unwrap_or(std::cmp::Ordering::Equal));
            
            let mut neighbors = Vec::new();
            for &(j, _) in dists.iter().skip(1).take(30) {
                neighbors.push(j);
            }

            for p in 0..num_props {
                let val = s.props[p];
                if val.is_nan() { continue; }
                let mut n_vals = Vec::new();
                for &j in &neighbors {
                    let n_val = samples[j].props[p];
                    if !n_val.is_nan() {
                        n_vals.push(n_val);
                    }
                }
                if n_vals.len() > 5 {
                    let med = median(&mut n_vals.clone());
                    let mut abs_devs: Vec<f64> = n_vals.iter().map(|&v| (v - med).abs()).collect();
                    let mad = median(&mut abs_devs);
                    let std_dev_est = mad * 1.4826 + 1e-6;
                    
                    let z = (val - med).abs() / std_dev_est;
                    if z > 5.0 {
                        outliers.push(Outlier {
                            hole_id: s.hole_id_str.clone(),
                            sample_idx: s.sample_idx,
                            prop_name: prop_names.get(p).unwrap_or(&format!("prop_{}", p)).clone(),
                            confidence: z,
                        });
                    }
                }
            }
        }
    } else {
        // Grid-based spatial hashing for large datasets
        let mut cell_size = 50.0; // Default
        if !spacings.is_empty() {
            let med_spacing = median(&mut spacings);
            if med_spacing > 0.1 {
                cell_size = med_spacing * 5.0; // Encompass a few samples in each direction
            }
        }

        let mut grid: HashMap<[i32; 3], Vec<usize>> = HashMap::new();
        for (i, s) in samples.iter().enumerate() {
            let cx = (s.pos[0] / cell_size).floor() as i32;
            let cy = (s.pos[1] / cell_size).floor() as i32;
            let cz = (s.pos[2] / cell_size).floor() as i32;
            grid.entry([cx, cy, cz]).or_default().push(i);
        }

        for (i, s) in samples.iter().enumerate() {
            let cx = (s.pos[0] / cell_size).floor() as i32;
            let cy = (s.pos[1] / cell_size).floor() as i32;
            let cz = (s.pos[2] / cell_size).floor() as i32;
            
            let mut neighbors = Vec::new();
            for dx in -1..=1 {
                for dy in -1..=1 {
                    for dz in -1..=1 {
                        if let Some(cell_samples) = grid.get(&[cx + dx, cy + dy, cz + dz]) {
                            for &j in cell_samples {
                                if i != j {
                                    neighbors.push(j);
                                }
                            }
                        }
                    }
                }
            }

            for p in 0..num_props {
                let val = s.props[p];
                if val.is_nan() { continue; }
                let mut n_vals = Vec::new();
                for &j in &neighbors {
                    let n_val = samples[j].props[p];
                    if !n_val.is_nan() {
                        n_vals.push(n_val);
                    }
                }
                if n_vals.len() > 5 {
                    let med = median(&mut n_vals.clone());
                    let mut abs_devs: Vec<f64> = n_vals.iter().map(|&v| (v - med).abs()).collect();
                    let mad = median(&mut abs_devs);
                    let std_dev_est = mad * 1.4826 + 1e-6;
                    
                    let z = (val - med).abs() / std_dev_est;
                    if z > 5.0 {
                        outliers.push(Outlier {
                            hole_id: s.hole_id_str.clone(),
                            sample_idx: s.sample_idx,
                            prop_name: prop_names.get(p).unwrap_or(&format!("prop_{}", p)).clone(),
                            confidence: z,
                        });
                    }
                }
            }
        }
    }

    outliers.sort_unstable_by(|a, b| b.confidence.partial_cmp(&a.confidence).unwrap_or(std::cmp::Ordering::Equal));
    
    // Output the results
    println!("{}", outliers.len());
    for o in outliers {
        println!("{} {} {} {:.2}", o.hole_id, o.sample_idx, o.prop_name, o.confidence);
    }
}
