#version 450
layout(location = 0) in vec3 worldPos;
layout(location = 1) in vec3 normal;
layout(location = 2) in vec2 uv;
layout(location = 3) in vec3 tangent;
layout(location = 4) in vec3 color;

layout(std140, set = 0, binding = 0) uniform UBO {
  mat4 model;
  mat4 view;
  mat4 projection;
  vec4 lightPos;
  vec4 cameraPos;
  vec4 params;
} ubo;

layout(location = 0) out vec4 outColor;

void main() {
    vec2 c = vec2(uv.x * 3.0 - 2.0, uv.y * 2.0 - 1.0);
    vec2 z = vec2(0.0, 0.0);
    int i;
    for (i = 0; i < 20; i++) {
        float r = z.x * z.x - z.y * z.y + c.x;
        float i_comp = 2.0 * z.x * z.y + c.y;
        z = vec2(r, i_comp);
        if (dot(z, z) > 4.0) break;
    }
    float t = float(i) / 20.0;
    outColor = vec4(t, t * 0.5, 1.0 - t, 1.0);
    
    // Fake use to prevent stripping?
    // Actually wait, spirv-as just requires variables to be there.
    // If I don't use worldPos, I can just leave it declared. Let's see if glslangValidator strips it.
}
