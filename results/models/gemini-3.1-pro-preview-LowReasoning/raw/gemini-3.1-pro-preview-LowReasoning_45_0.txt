import struct

def w(*args):
    return b''.join(struct.pack('<I', arg) for arg in args)

out = bytearray()

# Header
out += w(0x07230203) # Magic
out += w(0x00010000) # Version
out += w(0)          # Generator
out += w(12)         # Bound
out += w(0)          # Reserved

# OpCapability (17) Shader (1)
out += w(0x00020011, 1)

# OpMemoryModel (14) Logical(0) GLSL450(1)
out += w(0x0003000E, 0, 1)

# OpEntryPoint (15) Fragment(4) main_id(10) "main" outColor_id(9)
out += w(0x0006000F, 4, 10, 0x6E69616D, 0, 9)

# OpExecutionMode (16) main_id(10) OriginUpperLeft(7)
out += w(0x00030010, 10, 7)

# OpDecorate (71) outColor_id(9) Location(30) 0
out += w(0x00040047, 9, 30, 0)

# Types
out += w(0x00020013, 1) # OpTypeVoid void_id(1)
out += w(0x00030021, 2, 1) # OpTypeFunction main_func_id(2) void_id(1)
out += w(0x00030016, 3, 32) # OpTypeFloat float_id(3) 32
out += w(0x00040017, 4, 3, 4) # OpTypeVector vec4_id(4) float_id(3) 4
out += w(0x00040020, 5, 3, 4) # OpTypePointer out_ptr_id(5) Output(3) vec4_id(4)

# Constants
out += w(0x0004002B, 3, 6, 0x3F800000) # OpConstant float_id(3) const_1_id(6) 1.0f
out += w(0x0004002B, 3, 7, 0x00000000) # OpConstant float_id(3) const_0_id(7) 0.0f
out += w(0x0007002C, 4, 8, 6, 7, 7, 6) # OpConstantComposite vec4_id(4) red_id(8) 1.0 0.0 0.0 1.0

# Variables
out += w(0x0004003B, 5, 9, 3) # OpVariable out_ptr_id(5) outColor_id(9) Output(3)

# Function
out += w(0x00050036, 1, 10, 0, 2) # OpFunction void_id(1) main_id(10) None(0) main_func_id(2)
out += w(0x000200F8, 11) # OpLabel label_id(11)
out += w(0x0003003E, 9, 8) # OpStore outColor_id(9) red_id(8)
out += w(0x000100FD) # OpReturn
out += w(0x00010038) # OpFunctionEnd

hex_str = out.hex()

with open('answer.txt', 'w') as f:
    f.write(hex_str)
