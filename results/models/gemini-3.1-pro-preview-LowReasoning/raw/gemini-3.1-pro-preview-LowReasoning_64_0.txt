declare const require: any;
const fs = require('fs');

function solve() {
    const input: string = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);
    const grid: number[][] = data.probability;
    const drones: {id: any, x: number, y: number}[] = data.drones;
    const maxSteps = data.maxStepsPerDrone;

    const width = grid.length;
    const height = grid[0].length;

    const paths: [number, number][][] = [];
    for (const d of drones) {
        paths.push([[d.x, d.y]]);
        if (d.x >= 0 && d.x < width && d.y >= 0 && d.y < height) {
            grid[d.x][d.y] = 0; // mark collected
        }
    }

    const dx = [0, 1, 0, -1, 0];
    const dy = [1, 0, -1, 0, 0];

    for (let step = 0; step < maxSteps; step++) {
        for (let i = 0; i < drones.length; i++) {
            const path = paths[i];
            const curr = path[path.length - 1];
            const cx = curr[0];
            const cy = curr[1];

            // BFS to find the best next step
            let bestNext: [number, number] | null = null;

            // Check immediate neighbors first for max value
            let maxAdjVal = 0;
            for (let dir = 0; dir < 4; dir++) {
                const nx = cx + dx[dir];
                const ny = cy + dy[dir];
                if (nx >= 0 && nx < width && ny >= 0 && ny < height && grid[nx][ny] > 0) {
                    if (grid[nx][ny] > maxAdjVal) {
                        maxAdjVal = grid[nx][ny];
                        bestNext = [nx, ny];
                    }
                }
            }

            if (!bestNext) {
                // BFS
                const queue: {x: number, y: number, firstStep: [number, number]}[] = [];
                const visited = new Set<string>();
                visited.add(`${cx},${cy}`);

                for (let dir = 0; dir < 4; dir++) {
                    const nx = cx + dx[dir];
                    const ny = cy + dy[dir];
                    if (nx >= 0 && nx < width && ny >= 0 && ny < height && grid[nx][ny] !== -1) {
                        queue.push({x: nx, y: ny, firstStep: [nx, ny]});
                        visited.add(`${nx},${ny}`);
                    }
                }

                let head = 0;
                let maxFoundVal = -1;
                let targetFirstStep: [number, number] | null = null;
                let foundDepth = -1;

                // To limit BFS depth and avoid TLE on large grids
                let nodesExplored = 0;
                const MAX_NODES = 4000; 

                while (head < queue.length && nodesExplored < MAX_NODES) {
                    const node = queue[head++];
                    nodesExplored++;

                    if (grid[node.x][node.y] > 0) {
                        if (grid[node.x][node.y] > maxFoundVal) {
                            maxFoundVal = grid[node.x][node.y];
                            targetFirstStep = node.firstStep;
                        }
                    }

                    if (maxFoundVal > 0) {
                        // Found at least one target, maybe keep searching at current depth? 
                        // For simplicity, we just break soon or just continue to find maybe better max?
                        // Let's just pick the first we find to save time.
                        targetFirstStep = node.firstStep;
                        break;
                    }

                    for (let dir = 0; dir < 4; dir++) {
                        const nx = node.x + dx[dir];
                        const ny = node.y + dy[dir];
                        if (nx >= 0 && nx < width && ny >= 0 && ny < height && grid[nx][ny] !== -1) {
                            const key = `${nx},${ny}`;
                            if (!visited.has(key)) {
                                visited.add(key);
                                queue.push({x: nx, y: ny, firstStep: node.firstStep});
                            }
                        }
                    }
                }
                if (targetFirstStep) {
                    bestNext = targetFirstStep;
                } else {
                    bestNext = [cx, cy]; // stay
                }
            }

            path.push(bestNext);
            grid[bestNext[0]][bestNext[1]] = 0; // consume
        }
    }

    console.log(JSON.stringify({ paths }));
}

solve();
