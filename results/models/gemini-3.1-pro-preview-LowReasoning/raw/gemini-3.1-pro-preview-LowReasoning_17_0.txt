using System;
using System.Collections.Generic;

class Program
{
    struct Box
    {
        public int id;
        public long w, h, d;
        public long volume;
    }

    struct Space
    {
        public long x, y, z;
        public long w, h, d;

        public bool Contains(Space other)
        {
            return x <= other.x && y <= other.y && z <= other.z &&
                   x + w >= other.x + other.w &&
                   y + h >= other.y + other.h &&
                   z + d >= other.z + other.d;
        }

        public bool Intersects(long bx, long by, long bz, long bw, long bh, long bd)
        {
            return !(x + w <= bx || x >= bx + bw ||
                     y + h <= by || y >= by + bh ||
                     z + d <= bz || z >= bz + bd);
        }
    }

    struct PlacedBox
    {
        public int id;
        public long x, y, z;
    }

    static void Main(string[] args)
    {
        string[] firstLine = Console.ReadLine().Split(new char[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        if (firstLine.Length < 4) return;
        int n = int.Parse(firstLine[0]);
        long W = long.Parse(firstLine[1]);
        long H = long.Parse(firstLine[2]);
        long D = long.Parse(firstLine[3]);

        List<Box> boxes = new List<Box>(n);
        for (int i = 0; i < n; i++)
        {
            string[] line = Console.ReadLine().Split(new char[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
            long w = long.Parse(line[0]);
            long h = long.Parse(line[1]);
            long d = long.Parse(line[2]);
            boxes.Add(new Box { id = i, w = w, h = h, d = d, volume = w * h * d });
        }

        boxes.Sort((a, b) => a.volume.CompareTo(b.volume));

        List<Space> spaces = new List<Space>();
        spaces.Add(new Space { x = 0, y = 0, z = 0, w = W, h = H, d = D });

        List<PlacedBox> placedBoxes = new List<PlacedBox>();

        foreach (var box in boxes)
        {
            long bestZ = long.MaxValue;
            long bestY = long.MaxValue;
            long bestX = long.MaxValue;
            int bestSpaceIdx = -1;
            long bestW = 0, bestH = 0, bestD = 0;

            long[][] orientations = new long[][]
            {
                new long[] { box.w, box.h, box.d },
                new long[] { box.w, box.d, box.h },
                new long[] { box.h, box.w, box.d },
                new long[] { box.h, box.d, box.w },
                new long[] { box.d, box.w, box.h },
                new long[] { box.d, box.h, box.w }
            };

            for (int i = 0; i < spaces.Count; i++)
            {
                var space = spaces[i];
                if (space.w < Math.Min(box.w, Math.Min(box.h, box.d))) continue;
                if (space.h < Math.Min(box.w, Math.Min(box.h, box.d))) continue;
                if (space.d < Math.Min(box.w, Math.Min(box.h, box.d))) continue;

                foreach (var ori in orientations)
                {
                    if (space.w >= ori[0] && space.h >= ori[1] && space.d >= ori[2])
                    {
                        if (space.z < bestZ ||
                            (space.z == bestZ && space.y < bestY) ||
                            (space.z == bestZ && space.y == bestY && space.x < bestX))
                        {
                            bestZ = space.z;
                            bestY = space.y;
                            bestX = space.x;
                            bestSpaceIdx = i;
                            bestW = ori[0];
                            bestH = ori[1];
                            bestD = ori[2];
                        }
                    }
                }
            }

            if (bestSpaceIdx != -1)
            {
                var space = spaces[bestSpaceIdx];
                long bx = space.x;
                long by = space.y;
                long bz = space.z;

                placedBoxes.Add(new PlacedBox { id = box.id, x = bx, y = by, z = bz });

                List<Space> newSpaces = new List<Space>();
                for (int i = spaces.Count - 1; i >= 0; i--)
                {
                    var s = spaces[i];
                    if (s.Intersects(bx, by, bz, bestW, bestH, bestD))
                    {
                        spaces.RemoveAt(i);
                        if (bx > s.x) newSpaces.Add(new Space { x = s.x, y = s.y, z = s.z, w = bx - s.x, h = s.h, d = s.d });
                        if (bx + bestW < s.x + s.w) newSpaces.Add(new Space { x = bx + bestW, y = s.y, z = s.z, w = s.x + s.w - (bx + bestW), h = s.h, d = s.d });
                        if (by > s.y) newSpaces.Add(new Space { x = s.x, y = s.y, z = s.z, w = s.w, h = by - s.y, d = s.d });
                        if (by + bestH < s.y + s.h) newSpaces.Add(new Space { x = s.x, y = by + bestH, z = s.z, w = s.w, h = s.y + s.h - (by + bestH), d = s.d });
                        if (bz > s.z) newSpaces.Add(new Space { x = s.x, y = s.y, z = s.z, w = s.w, h = s.h, d = bz - s.z });
                        if (bz + bestD < s.z + s.d) newSpaces.Add(new Space { x = s.x, y = s.y, z = bz + bestD, w = s.w, h = s.h, d = s.z + s.d - (bz + bestD) });
                    }
                }

                foreach (var ns in newSpaces)
                {
                    bool isContained = false;
                    foreach (var s in spaces)
                    {
                        if (s.Contains(ns))
                        {
                            isContained = true;
                            break;
                        }
                    }
                    if (!isContained)
                    {
                        for (int i = spaces.Count - 1; i >= 0; i--)
                        {
                            if (ns.Contains(spaces[i]))
                            {
                                spaces.RemoveAt(i);
                            }
                        }
                        spaces.Add(ns);
                    }
                }

                if (spaces.Count > 4000)
                {
                    spaces.Sort((a, b) => (b.w * b.h * b.d).CompareTo(a.w * a.h * a.d));
                    spaces.RemoveRange(4000, spaces.Count - 4000);
                }
            }
        }

        Console.WriteLine(placedBoxes.Count);
        foreach (var pb in placedBoxes)
        {
            Console.WriteLine($"{pb.id} {pb.x} {pb.y} {pb.z}");
        }
    }
}
