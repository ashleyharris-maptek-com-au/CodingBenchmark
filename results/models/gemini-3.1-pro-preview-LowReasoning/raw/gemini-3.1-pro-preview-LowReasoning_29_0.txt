using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Linq;
using System.Threading.Tasks;

class Program
{
    static int numVertices;
    static int numEdges;
    static int[][] adj;
    
    static int[] maxClique = new int[0];
    static object lockObj = new object();
    static Stopwatch stopwatch = new Stopwatch();
    static long TIME_LIMIT_MS = 290000; // 4 minutes 50 seconds to allow safe shutdown
    
    static void Main()
    {
        stopwatch.Start();
        
        // Fast reading
        using (StreamReader reader = new StreamReader(Console.OpenStandardInput(), bufferSize: 1048576 * 16))
        {
            string line = reader.ReadLine();
            if (line == null) return;
            string[] parts = line.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
            if (parts.Length < 2) return;
            numVertices = int.Parse(parts[0]);
            numEdges = int.Parse(parts[1]);
            
            List<int>[] tempAdj = new List<int>[numVertices];
            for (int i = 0; i < numVertices; i++)
                tempAdj[i] = new List<int>();
            
            for (int i = 0; i < numEdges; i++)
            {
                line = reader.ReadLine();
                if (line == null) break;
                parts = line.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
                if (parts.Length < 2) continue;
                int u = int.Parse(parts[0]);
                int v = int.Parse(parts[1]);
                if (u != v) {
                    tempAdj[u].Add(v);
                    tempAdj[v].Add(u);
                }
            }

            adj = new int[numVertices][];
            for (int i = 0; i < numVertices; i++)
            {
                tempAdj[i].Sort();
                adj[i] = tempAdj[i].Distinct().ToArray();
            }
        }
        
        // 1. Initial Greedy Solution
        var degrees = adj.Select((neighbors, index) => new { Index = index, Degree = neighbors.Length })
                         .OrderByDescending(x => x.Degree).ToList();
                         
        List<int> initialClique = new List<int>();
        foreach (var v in degrees)
        {
            if (stopwatch.ElapsedMilliseconds > TIME_LIMIT_MS) break;
            bool canAdd = true;
            foreach (var u in initialClique)
            {
                if (Array.BinarySearch(adj[v.Index], u) < 0)
                {
                    canAdd = false;
                    break;
                }
            }
            if (canAdd) initialClique.Add(v.Index);
        }
        
        maxClique = initialClique.ToArray();
        
        // 2. Parallel Randomized Heuristic for local search
        long heuristicEndTime = stopwatch.ElapsedMilliseconds + 5000;
        if (heuristicEndTime > TIME_LIMIT_MS) heuristicEndTime = TIME_LIMIT_MS;
        
        Parallel.For(0, Environment.ProcessorCount, (threadId, state) =>
        {
            Random rnd = new Random(42 + threadId);
            while (stopwatch.ElapsedMilliseconds < heuristicEndTime)
            {
                List<int> currentClique = new List<int>();
                int startNode = rnd.Next(numVertices);
                currentClique.Add(startNode);
                
                List<int> candidates = new List<int>(adj[startNode]);
                while (candidates.Count > 0)
                {
                    if (stopwatch.ElapsedMilliseconds > heuristicEndTime) break;
                    int pick = candidates[rnd.Next(candidates.Count)];
                    currentClique.Add(pick);
                    
                    List<int> nextCandidates = new List<int>(candidates.Count);
                    foreach (int c in candidates)
                    {
                        if (Array.BinarySearch(adj[pick], c) >= 0)
                            nextCandidates.Add(c);
                    }
                    candidates = nextCandidates;
                }
                
                lock (lockObj)
                {
                    if (currentClique.Count > maxClique.Length)
                    {
                        maxClique = currentClique.ToArray();
                    }
                }
            }
        });

        // 3. Exact Branch and Bound for small graphs / Continued Heuristic for massive ones
        if (numVertices <= 10000 && stopwatch.ElapsedMilliseconds < TIME_LIMIT_MS - 5000)
        {
            int[] order = degrees.Select(x => x.Index).ToArray();
            
            Parallel.ForEach(order, new ParallelOptions { MaxDegreeOfParallelism = Environment.ProcessorCount }, (v, state) =>
            {
                if (stopwatch.ElapsedMilliseconds > TIME_LIMIT_MS) state.Break();
                
                List<int> currentClique = new List<int> { v };
                List<int> candidates = new List<int>();
                
                bool found = false;
                foreach (int u in order)
                {
                    if (found && Array.BinarySearch(adj[v], u) >= 0)
                    {
                        candidates.Add(u);
                    }
                    if (u == v) found = true;
                }
                
                Expand(currentClique, candidates, state);
            });
        }
        else if (stopwatch.ElapsedMilliseconds < TIME_LIMIT_MS - 5000)
        {
            // Fallback: Continue running parallel randomized heuristic
            Parallel.For(0, Environment.ProcessorCount, (threadId, state) =>
            {
                Random rnd = new Random(100 + threadId);
                while (stopwatch.ElapsedMilliseconds < TIME_LIMIT_MS)
                {
                    List<int> currentClique = new List<int>();
                    int startNode = rnd.Next(numVertices);
                    currentClique.Add(startNode);
                    
                    List<int> candidates = new List<int>(adj[startNode]);
                    while (candidates.Count > 0)
                    {
                        if (stopwatch.ElapsedMilliseconds > TIME_LIMIT_MS) break;
                        int pick = candidates[rnd.Next(candidates.Count)];
                        currentClique.Add(pick);
                        
                        List<int> nextCandidates = new List<int>(candidates.Count);
                        foreach (int c in candidates)
                        {
                            if (Array.BinarySearch(adj[pick], c) >= 0)
                                nextCandidates.Add(c);
                        }
                        candidates = nextCandidates;
                    }
                    
                    lock (lockObj)
                    {
                        if (currentClique.Count > maxClique.Length)
                        {
                            maxClique = currentClique.ToArray();
                        }
                    }
                }
            });
        }
        
        // Output result
        Console.WriteLine(maxClique.Length);
        Console.WriteLine(string.Join(" ", maxClique));
    }

    static void Expand(List<int> currentClique, List<int> candidates, ParallelLoopState state)
    {
        if (stopwatch.ElapsedMilliseconds > TIME_LIMIT_MS)
        {
            state.Break();
            return;
        }

        if (candidates.Count == 0)
        {
            lock (lockObj)
            {
                if (currentClique.Count > maxClique.Length)
                {
                    maxClique = currentClique.ToArray();
                }
            }
            return;
        }

        if (currentClique.Count + candidates.Count <= maxClique.Length)
        {
            return;
        }

        // Pivot choice (Tomita-like strategy)
        int pivot = candidates[0];
        int maxDeg = -1;
        foreach (int c in candidates)
        {
            int deg = 0;
            foreach (int u in candidates)
            {
                if (Array.BinarySearch(adj[c], u) >= 0) deg++;
            }
            if (deg > maxDeg)
            {
                maxDeg = deg;
                pivot = c;
            }
        }

        List<int> toBranch = new List<int>();
        foreach (int c in candidates)
        {
            if (c == pivot || Array.BinarySearch(adj[pivot], c) < 0)
            {
                toBranch.Add(c);
            }
        }

        foreach (int v in toBranch)
        {
            if (currentClique.Count + candidates.Count <= maxClique.Length) return;
            if (stopwatch.ElapsedMilliseconds > TIME_LIMIT_MS) return;
            
            currentClique.Add(v);
            List<int> newCandidates = new List<int>(candidates.Count);
            foreach (int c in candidates)
            {
                if (c != v && Array.BinarySearch(adj[v], c) >= 0)
                {
                    newCandidates.Add(c);
                }
            }
            
            Expand(currentClique, newCandidates, state);
            
            currentClique.RemoveAt(currentClique.Count - 1);
            candidates.Remove(v);
        }
    }
}
