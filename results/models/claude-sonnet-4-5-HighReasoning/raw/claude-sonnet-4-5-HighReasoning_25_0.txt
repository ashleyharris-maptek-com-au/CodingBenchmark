using System;
using System.Collections.Generic;
using System.Globalization;
using System.Linq;

namespace AsteroidInterception
{
    class Program
    {
        static void Main(string[] args)
        {
            var solver = new AsteroidInterceptionSolver();
            solver.Run();
        }
    }

    public struct Vector3D
    {
        public double X, Y, Z;

        public Vector3D(double x, double y, double z)
        {
            X = x; Y = y; Z = z;
        }

        public static Vector3D operator +(Vector3D a, Vector3D b) => new Vector3D(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
        public static Vector3D operator -(Vector3D a, Vector3D b) => new Vector3D(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
        public static Vector3D operator *(Vector3D a, double s) => new Vector3D(a.X * s, a.Y * s, a.Z * s);
        public static Vector3D operator *(double s, Vector3D a) => new Vector3D(a.X * s, a.Y * s, a.Z * s);
        public static Vector3D operator /(Vector3D a, double s) => new Vector3D(a.X / s, a.Y / s, a.Z / s);

        public double Magnitude() => Math.Sqrt(X * X + Y * Y + Z * Z);
        public double MagnitudeSquared() => X * X + Y * Y + Z * Z;
        public Vector3D Normalized()
        {
            double mag = Magnitude();
            return mag > 0 ? this / mag : new Vector3D(0, 0, 0);
        }

        public static double Dot(Vector3D a, Vector3D b) => a.X * b.X + a.Y * b.Y + a.Z * b.Z;

        public static Vector3D Cross(Vector3D a, Vector3D b)
        {
            return new Vector3D(
                a.Y * b.Z - a.Z * b.Y,
                a.Z * b.X - a.X * b.Z,
                a.X * b.Y - a.Y * b.X
            );
        }
    }

    public class Body
    {
        public string Name;
        public double Mass;
        public Vector3D Position;
        public Vector3D Velocity;

        public Body Clone()
        {
            return new Body
            {
                Name = this.Name,
                Mass = this.Mass,
                Position = this.Position,
                Velocity = this.Velocity
            };
        }
    }

    public class Burn
    {
        public double Time;
        public Vector3D DeltaV;
    }

    public class AsteroidInterceptionSolver
    {
        private const double G = 6.67430e-11; // Gravitational constant
        private const double AU = 1.495978707e11; // Astronomical unit in meters
        private const double DAY_SECONDS = 86400.0;

        private List<Body> bodies;
        private Body earth;
        private Body asteroid;
        private double warningDays;
        private double impactorMass;
        private double deltaVBudget;
        private double asteroidMass;
        private double t0UnixUtc;
        private string de421Path;

        public void Run()
        {
            ParseInput();
            var solution = Solve();
            OutputSolution(solution);
        }

        private void ParseInput()
        {
            string line = Console.ReadLine();
            var parts = line.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);

            int numBodies = int.Parse(parts[0]);
            warningDays = double.Parse(parts[1], CultureInfo.InvariantCulture);
            impactorMass = double.Parse(parts[2], CultureInfo.InvariantCulture);
            deltaVBudget = double.Parse(parts[3], CultureInfo.InvariantCulture);
            asteroidMass = double.Parse(parts[4], CultureInfo.InvariantCulture);
            t0UnixUtc = double.Parse(parts[5], CultureInfo.InvariantCulture);

            bodies = new List<Body>();

            for (int i = 0; i < numBodies; i++)
            {
                line = Console.ReadLine();
                parts = line.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);

                var body = new Body
                {
                    Name = parts[0],
                    Mass = double.Parse(parts[1], CultureInfo.InvariantCulture),
                    Position = new Vector3D(
                        double.Parse(parts[2], CultureInfo.InvariantCulture),
                        double.Parse(parts[3], CultureInfo.InvariantCulture),
                        double.Parse(parts[4], CultureInfo.InvariantCulture)
                    ),
                    Velocity = new Vector3D(
                        double.Parse(parts[5], CultureInfo.InvariantCulture),
                        double.Parse(parts[6], CultureInfo.InvariantCulture),
                        double.Parse(parts[7], CultureInfo.InvariantCulture)
                    )
                };

                bodies.Add(body);

                if (body.Name == "Earth") earth = body;
                if (body.Name == "Asteroid") asteroid = body;
            }

            de421Path = Console.ReadLine()?.Trim();
        }

        private List<Burn> Solve()
        {
            // Find approximate Earth-asteroid close approach time
            double impactTime = FindCloseApproachTime(asteroid, earth, 0, warningDays * 2);

            // Try different launch times
            double bestScore = double.MaxValue;
            List<Burn> bestSolution = null;

            int numLaunchTries = Math.Min(50, (int)(warningDays / 10) + 5);
            double launchStep = warningDays / numLaunchTries;

            for (int i = 0; i < numLaunchTries; i++)
            {
                double launchTime = i * launchStep;
                if (launchTime > warningDays * 0.95) continue;

                var solution = FindTrajectory(launchTime, impactTime);
                if (solution != null)
                {
                    double totalDv = solution.Sum(b => b.DeltaV.Magnitude());
                    if (totalDv < deltaVBudget && totalDv < bestScore)
                    {
                        bestScore = totalDv;
                        bestSolution = solution;
                    }
                }
            }

            // If no solution found, try more refined search
            if (bestSolution == null)
            {
                for (int i = 0; i < numLaunchTries * 2; i++)
                {
                    double launchTime = i * launchStep / 2.0;
                    if (launchTime > warningDays * 0.98) continue;

                    var solution = FindTrajectory(launchTime, impactTime);
                    if (solution != null)
                    {
                        double totalDv = solution.Sum(b => b.DeltaV.Magnitude());
                        if (totalDv < deltaVBudget)
                        {
                            bestSolution = solution;
                            break;
                        }
                    }
                }
            }

            // Optimize the best solution found
            if (bestSolution != null)
            {
                bestSolution = OptimizeSolution(bestSolution);
            }
            else
            {
                // Emergency fallback - simple direct trajectory
                bestSolution = CreateFallbackSolution(impactTime);
            }

            return bestSolution;
        }

        private double FindCloseApproachTime(Body ast, Body ert, double tStart, double tEnd)
        {
            double minDist = double.MaxValue;
            double minTime = tStart;
            int steps = 200;
            double dt = (tEnd - tStart) / steps;

            for (int i = 0; i < steps; i++)
            {
                double t = tStart + i * dt;
                var astPos = PropagateBody(ast, t);
                var ertPos = PropagateBody(ert, t);
                double dist = (astPos.Position - ertPos.Position).Magnitude();

                if (dist < minDist)
                {
                    minDist = dist;
                    minTime = t;
                }
            }

            // Refine around minimum
            double refineDt = dt / 10;
            for (int i = -10; i <= 10; i++)
            {
                double t = minTime + i * refineDt;
                if (t < tStart || t > tEnd) continue;

                var astPos = PropagateBody(ast, t);
                var ertPos = PropagateBody(ert, t);
                double dist = (astPos.Position - ertPos.Position).Magnitude();

                if (dist < minDist)
                {
                    minDist = dist;
                    minTime = t;
                }
            }

            return minTime;
        }

        private List<Burn> FindTrajectory(double launchTime, double targetTime)
        {
            double interceptTime = targetTime - 5.0; // Intercept before Earth impact
            if (interceptTime <= launchTime + 1.0) return null;

            var earthAtLaunch = PropagateBody(earth, launchTime);
            var asteroidAtIntercept = PropagateBody(asteroid, interceptTime);

            // Solve Lambert's problem
            double tof = (interceptTime - launchTime) * DAY_SECONDS;
            var lambert = SolveLambert(earthAtLaunch.Position, asteroidAtIntercept.Position, tof);

            if (lambert == null) return null;

            var burns = new List<Burn>();

            // Launch burn
            Vector3D launchDv = lambert.Item1 - earthAtLaunch.Velocity;
            burns.Add(new Burn { Time = launchTime, DeltaV = launchDv });

            // Mid-course correction (optional, helps with accuracy)
            double midTime = (launchTime + interceptTime) / 2.0;
            var scAtMid = PropagateSpacecraft(earthAtLaunch.Position, lambert.Item1, launchTime, midTime);
            var astAtMid = PropagateBody(asteroid, midTime);

            Vector3D targetVelAtMid = (astAtMid.Position - scAtMid.Position).Normalized() * lambert.Item1.Magnitude();
            Vector3D midDv = targetVelAtMid - scAtMid.Velocity;

            if (midDv.Magnitude() > 100 && midDv.Magnitude() < deltaVBudget * 0.3)
            {
                burns.Add(new Burn { Time = midTime, DeltaV = midDv * 0.5 });
            }

            // Impact burn
            var scAtIntercept = PropagateSpacecraft(earthAtLaunch.Position, lambert.Item1, launchTime, interceptTime);
            Vector3D impactDv = asteroidAtIntercept.Velocity - scAtIntercept.Velocity;

            // Scale impact velocity for effectiveness
            double impactSpeed = Math.Max(3000, Math.Min(15000, impactDv.Magnitude()));
            impactDv = impactDv.Normalized() * impactSpeed;

            burns.Add(new Burn { Time = interceptTime, DeltaV = impactDv });

            return burns;
        }

        private Tuple<Vector3D, Vector3D> SolveLambert(Vector3D r1, Vector3D r2, double tof)
        {
            // Simplified Lambert's problem solver using universal variable formulation
            double r1Mag = r1.Magnitude();
            double r2Mag = r2.Magnitude();
            double cosDeltaNu = Vector3D.Dot(r1, r2) / (r1Mag * r2Mag);

            // Choose short way
            Vector3D c = r2 - r1;
            double cMag = c.Magnitude();
            double s = (r1Mag + r2Mag + cMag) / 2.0;

            // Find gravitational parameter (use Sun for most trajectories)
            double mu = G * GetDominantMass(r1);

            // Initial guess for semi-major axis
            double a = s / 2.0;

            // Iterate to find correct semi-major axis
            for (int iter = 0; iter < 30; iter++)
            {
                double alpha = 2 * Math.Asin(Math.Sqrt(s / (2 * a)));
                double beta = 2 * Math.Asin(Math.Sqrt((s - cMag) / (2 * a)));

                if (double.IsNaN(alpha) || double.IsNaN(beta)) break;

                double tofCalc = Math.Sqrt(a * a * a / mu) * (alpha - Math.Sin(alpha) - (beta - Math.Sin(beta)));
                double error = tof - tofCalc;

                if (Math.Abs(error) < 1.0) break;

                // Newton-Raphson update
                double dtda = Math.Sqrt(a * a * a / mu) * 1.5 / a * (alpha - Math.Sin(alpha) - (beta - Math.Sin(beta)));
                if (Math.Abs(dtda) > 1e-10)
                {
                    a += error / dtda * 0.5;
                }

                if (a < 0) a = s / 2.0;
            }

            // Compute velocity vectors
            double f = 1 - r2Mag / a * (1 - Math.Cos(Math.Sqrt(mu / (a * a * a)) * tof));
            double g = r1Mag * r2Mag * Math.Sin(Math.Sqrt(mu / (a * a * a)) * tof) / Math.Sqrt(mu * a);

            if (Math.Abs(g) < 1e-10) return null;

            Vector3D v1 = (r2 - r1 * f) / g;

            return Tuple.Create(v1, new Vector3D(0, 0, 0));
        }

        private double GetDominantMass(Vector3D pos)
        {
            // Return mass of dominant gravitational body at position
            Body sun = bodies.FirstOrDefault(b => b.Name == "Sun");
            if (sun != null) return sun.Mass;

            // Find closest massive body
            double minDist = double.MaxValue;
            Body closest = null;

            foreach (var body in bodies)
            {
                if (body.Mass > 1e20)
                {
                    double dist = (body.Position - pos).Magnitude();
                    if (dist < minDist)
                    {
                        minDist = dist;
                        closest = body;
                    }
                }
            }

            return closest?.Mass ?? 1.989e30; // Default to Sun mass
        }

        private Body PropagateBody(Body body, double time)
        {
            // Simple Keplerian propagation with multi-body perturbations
            double dt = time * DAY_SECONDS;

            var result = body.Clone();
            int steps = Math.Max(1, (int)(Math.Abs(dt) / (3600 * 24))); // Daily steps
            double stepDt = dt / steps;

            for (int i = 0; i < steps; i++)
            {
                // Calculate acceleration from all massive bodies
                Vector3D acc = new Vector3D(0, 0, 0);

                foreach (var otherBody in bodies)
                {
                    if (otherBody.Name == body.Name) continue;
                    if (otherBody.Mass < 1e20) continue; // Skip small bodies

                    Vector3D otherPos = otherBody.Position + otherBody.Velocity * (stepDt * i);
                    Vector3D r = otherPos - result.Position;
                    double rMag = r.Magnitude();

                    if (rMag > 1e6)
                    {
                        acc = acc + r * (G * otherBody.Mass / (rMag * rMag * rMag));
                    }
                }

                // Leapfrog integration
                result.Velocity = result.Velocity + acc * stepDt;
                result.Position = result.Position + result.Velocity * stepDt;
            }

            return result;
        }

        private Body PropagateSpacecraft(Vector3D pos0, Vector3D vel0, double t0, double t1)
        {
            double dt = (t1 - t0) * DAY_SECONDS;

            var sc = new Body
            {
                Name = "Spacecraft",
                Mass = impactorMass,
                Position = pos0,
                Velocity = vel0
            };

            int steps = Math.Max(1, (int)(Math.Abs(dt) / (3600 * 12))); // Half-day steps
            double stepDt = dt / steps;

            for (int i = 0; i < steps; i++)
            {
                Vector3D acc = new Vector3D(0, 0, 0);

                foreach (var body in bodies)
                {
                    if (body.Mass < 1e20) continue;

                    double bodyTime = (t0 + i * stepDt / DAY_SECONDS);
                    Vector3D bodyPos = body.Position + body.Velocity * (bodyTime * DAY_SECONDS);

                    Vector3D r = bodyPos - sc.Position;
                    double rMag = r.Magnitude();

                    if (rMag > 1e6)
                    {
                        acc = acc + r * (G * body.Mass / (rMag * rMag * rMag));
                    }
                }

                sc.Velocity = sc.Velocity + acc * stepDt;
                sc.Position = sc.Position + sc.Velocity * stepDt;
            }

            return sc;
        }

        private List<Burn> OptimizeSolution(List<Burn> solution)
        {
            // Simple optimization: adjust burn magnitudes to minimize total dv
            var optimized = new List<Burn>();

            foreach (var burn in solution)
            {
                optimized.Add(new Burn { Time = burn.Time, DeltaV = burn.DeltaV });
            }

            // Try small adjustments
            for (int iter = 0; iter < 10; iter++)
            {
                for (int i = 0; i < optimized.Count - 1; i++)
                {
                    double currentDv = optimized.Sum(b => b.DeltaV.Magnitude());

                    // Try reducing this burn slightly
                    var testBurn = new Vector3D(
                        optimized[i].DeltaV.X * 0.95,
                        optimized[i].DeltaV.Y * 0.95,
                        optimized[i].DeltaV.Z * 0.95
                    );

                    var testSolution = new List<Burn>(optimized);
                    testSolution[i].DeltaV = testBurn;

                    double testDv = testSolution.Sum(b => b.DeltaV.Magnitude());
                    if (testDv < currentDv)
                    {
                        optimized[i].DeltaV = testBurn;
                    }
                }
            }

            return optimized;
        }

        private List<Burn> CreateFallbackSolution(double impactTime)
        {
            // Create a simple direct trajectory as fallback
            double launchTime = Math.Min(10.0, warningDays * 0.1);
            double interceptTime = Math.Max(launchTime + 10.0, impactTime - 10.0);

            var earthAtLaunch = PropagateBody(earth, launchTime);
            var asteroidAtIntercept = PropagateBody(asteroid, interceptTime);

            double tof = (interceptTime - launchTime) * DAY_SECONDS;
            Vector3D direction = (asteroidAtIntercept.Position - earthAtLaunch.Position).Normalized();
            double distance = (asteroidAtIntercept.Position - earthAtLaunch.Position).Magnitude();

            Vector3D velocity = direction * (distance / tof);
            Vector3D launchDv = velocity - earthAtLaunch.Velocity;

            // Scale to use available delta-v efficiently
            double dvMag = launchDv.Magnitude();
            if (dvMag > deltaVBudget * 0.7)
            {
                launchDv = launchDv * (deltaVBudget * 0.7 / dvMag);
            }

            var burns = new List<Burn>
            {
                new Burn { Time = launchTime, DeltaV = launchDv },
                new Burn { Time = interceptTime, DeltaV = asteroidAtIntercept.Velocity * 0.1 }
            };

            return burns;
        }

        private void OutputSolution(List<Burn> solution)
        {
            Console.WriteLine(solution.Count);

            foreach (var burn in solution)
            {
                Console.WriteLine($"{burn.Time.ToString("F6", CultureInfo.InvariantCulture)} {burn.DeltaV.X.ToString("E", CultureInfo.InvariantCulture)} {burn.DeltaV.Y.ToString("E", CultureInfo.InvariantCulture)} {burn.DeltaV.Z.ToString("E", CultureInfo.InvariantCulture)}");
            }
        }
    }
