using System;
using System.Collections.Generic;
using System.Linq;

class Program
{
    static void Main()
    {
        var input = Console.ReadLine().Split();
        int n = int.Parse(input[0]);
        int W = int.Parse(input[1]);
        int H = int.Parse(input[2]);
        int D = int.Parse(input[3]);

        var boxes = new List<Box>();
        for (int i = 0; i < n; i++)
        {
            var dims = Console.ReadLine().Split().Select(int.Parse).ToArray();
            boxes.Add(new Box(i, dims[0], dims[1], dims[2]));
        }

        var packer = new Packer(W, H, D);
        var result = packer.Pack(boxes);

        Console.WriteLine(result.Count);
        foreach (var r in result.OrderBy(r => r.Item1))
        {
            Console.WriteLine($"{r.Item1} {r.Item2} {r.Item3} {r.Item4}");
        }
    }
}

class Box
{
    public int Id;
    public int W, H, D;
    public long Vol;

    public Box(int id, int w, int h, int d)
    {
        Id = id;
        W = w; H = h; D = d;
        Vol = (long)w * h * d;
    }
}

class Space
{
    public int X, Y, Z, W, H, D;
}

class Packer
{
    int W, H, D;
    List<(int id, int x, int y, int z, int w, int h, int d)> placed = new();
    List<Space> spaces = new();

    public Packer(int w, int h, int d)
    {
        W = w; H = h; D = d;
        spaces.Add(new Space { X = 0, Y = 0, Z = 0, W = w, H = h, D = d });
    }

    public List<(int, int, int, int)> Pack(List<Box> boxes)
    {
        // Sort by volume descending for better packing
        boxes = boxes.OrderByDescending(b => b.Vol).ThenBy(b => b.Id).ToList();

        foreach (var box in boxes)
        {
            TryPlace(box);
        }

        return placed.Select(p => (p.id, p.x, p.y, p.z)).ToList();
    }

    void TryPlace(Box box)
    {
        var rots = GetRotations(box.W, box.H, box.D);

        int bestIdx = -1;
        var bestRot = (0, 0, 0);
        int bestScore = int.MaxValue;

        for (int i = 0; i < spaces.Count; i++)
        {
            var s = spaces[i];
            foreach (var (w, h, d) in rots)
            {
                if (w <= s.W && h <= s.H && d <= s.D)
                {
                    // Best-fit: minimize leftover space (prefer tight fits)
                    int waste = (s.W - w) + (s.H - h) + (s.D - d);
                    if (waste < bestScore)
                    {
                        bestScore = waste;
                        bestIdx = i;
                        bestRot = (w, h, d);
                    }
                }
            }
        }

        if (bestIdx >= 0)
        {
            var s = spaces[bestIdx];
            var (bw, bh, bd) = bestRot;

            placed.Add((box.Id, s.X, s.Y, s.Z, bw, bh, bd));
            spaces.RemoveAt(bestIdx);

            // Guillotine split: create up to 3 new spaces from the remainder
            if (s.W > bw)
                spaces.Add(new Space { X = s.X + bw, Y = s.Y, Z = s.Z, W = s.W - bw, H = s.H, D = s.D });
            if (s.H > bh)
                spaces.Add(new Space { X = s.X, Y = s.Y + bh, Z = s.Z, W = bw, H = s.H - bh, D = s.D });
            if (s.D > bd)
                spaces.Add(new Space { X = s.X, Y = s.Y, Z = s.Z + bd, W = bw, H = bh, D = s.D - bd });

            // Periodically prune to keep space list manageable
            if (spaces.Count > 100)
                Prune();
        }
    }

    void Prune()
    {
        // Remove spaces that are completely contained within other spaces
        for (int i = 0; i < spaces.Count; i++)
        {
            for (int j = i + 1; j < spaces.Count; j++)
            {
                if (Contains(spaces[i], spaces[j]))
                {
                    spaces.RemoveAt(j--);
                }
                else if (Contains(spaces[j], spaces[i]))
                {
                    spaces.RemoveAt(i--);
                    break;
                }
            }
        }
    }

    bool Contains(Space a, Space b)
    {
        return a.X <= b.X && a.Y <= b.Y && a.Z <= b.Z &&
               a.X + a.W >= b.X + b.W &&
               a.Y + a.H >= b.Y + b.H &&
               a.Z + a.D >= b.Z + b.D;
    }

    List<(int, int, int)> GetRotations(int w, int h, int d)
    {
        // Generate all 6 possible orientations of the box
        return new HashSet<(int, int, int)>
        {
            (w, h, d), (w, d, h),
            (h, w, d), (h, d, w),
            (d, w, h), (d, h, w)
        }.ToList();
    }
}
