import sys

def solve_3sat():
    # Read input
    first_line = sys.stdin.readline().split()
    n_vars = int(first_line[0])
    n_clauses = int(first_line[1])

    clauses = []
    for _ in range(n_clauses):
        clause = list(map(int, sys.stdin.readline().split()))
        clauses.append(clause)

    # Assignment: None = unassigned, True/False = assigned value
    assignment = [None] * (n_vars + 1)  # 1-indexed

    def is_satisfied(lit):
        """Check if a literal is satisfied"""
        var = abs(lit)
        if assignment[var] is None:
            return None  # Unassigned
        return assignment[var] == (lit > 0)

    def is_clause_satisfied(clause):
        """Check if a clause is satisfied"""
        for lit in clause:
            if is_satisfied(lit) == True:
                return True
        return False

    def has_conflict():
        """Check if there's a conflict (unsatisfiable clause with all literals false)"""
        for clause in clauses:
            if not is_clause_satisfied(clause):
                all_false = all(is_satisfied(lit) == False for lit in clause)
                if all_false:
                    return True
        return False

    def unit_propagate():
        """Perform unit propagation"""
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                if is_clause_satisfied(clause):
                    continue

                # Find unassigned literals
                unassigned = []
                for lit in clause:
                    val = is_satisfied(lit)
                    if val is None:
                        unassigned.append(lit)
                    elif val == True:
                        # Clause already satisfied
                        break
                else:
                    # Clause not satisfied
                    if len(unassigned) == 0:
                        # All literals false - conflict
                        return False
                    if len(unassigned) == 1:
                        # Unit clause - must assign this literal to true
                        lit = unassigned[0]
                        var = abs(lit)
                        assignment[var] = lit > 0
                        changed = True
        return True

    def dpll_search():
        """DPLL search algorithm"""
        # Check for immediate conflict
        if has_conflict():
            return False

        # Unit propagation
        if not unit_propagate():
            return False

        # Check if all clauses are satisfied
        if all(is_clause_satisfied(clause) for clause in clauses):
            return True

        # Find an unassigned variable
        chosen_var = None
        for i in range(1, n_vars + 1):
            if assignment[i] is None:
                chosen_var = i
                break

        if chosen_var is None:
            # All variables assigned
            return all(is_clause_satisfied(clause) for clause in clauses)

        # Try assigning True
        saved_assignment = assignment[:]
        assignment[chosen_var] = True
        if dpll_search():
            return True

        # Backtrack and try False
        assignment[:] = saved_assignment
        assignment[chosen_var] = False
        if dpll_search():
            return True

        # Backtrack completely
        assignment[:] = saved_assignment
        return False

    # Solve
    dpll_search()

    # Output result
    print("SAT")
    result = []
    for i in range(1, n_vars + 1):
        if assignment[i] is None or assignment[i]:
            result.append('1')
        else:
            result.append('0')
    print(' '.join(result))

if __name__ == "__main__":
    solve_3sat()
