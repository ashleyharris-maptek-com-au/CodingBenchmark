using System;
using System.Collections.Generic;

class Program
{
    static void Main()
    {
        var input = ParseInput();
        var solver = new BinPackingSolver(input);
        var result = solver.Solve();
        PrintResult(result);
    }

    static Input ParseInput()
    {
        var line1 = Console.ReadLine().Split();
        int V = int.Parse(line1[0]);
        int F = int.Parse(line1[1]);

        var vertices = new List<Vector3>();
        for (int i = 0; i < V; i++)
        {
            var parts = Console.ReadLine().Split();
            vertices.Add(new Vector3(
                double.Parse(parts[0]),
                double.Parse(parts[1]),
                double.Parse(parts[2])
            ));
        }

        var faces = new List<Face>();
        for (int i = 0; i < F; i++)
        {
            var parts = Console.ReadLine().Split();
            faces.Add(new Face(
                int.Parse(parts[0]),
                int.Parse(parts[1]),
                int.Parse(parts[2])
            ));
        }

        var lastLine = Console.ReadLine().Split();
        var container = new Vector3(
            double.Parse(lastLine[0]),
            double.Parse(lastLine[1]),
            double.Parse(lastLine[2])
        );

        return new Input { Vertices = vertices, Faces = faces, Container = container };
    }

    static void PrintResult(List<Placement> placements)
    {
        Console.WriteLine(placements.Count);
        foreach (var p in placements)
        {
            Console.WriteLine($"{p.Translation.X} {p.Translation.Y} {p.Translation.Z} {p.Rotation.W} {p.Rotation.X} {p.Rotation.Y} {p.Rotation.Z}");
        }
    }
}

struct Vector3
{
    public double X, Y, Z;
    public Vector3(double x, double y, double z) { X = x; Y = y; Z = z; }
    public static Vector3 operator +(Vector3 a, Vector3 b) => new Vector3(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
    public static Vector3 operator -(Vector3 a, Vector3 b) => new Vector3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
}

struct Quaternion
{
    public double W, X, Y, Z;
    public Quaternion(double w, double x, double y, double z) { W = w; X = x; Y = y; Z = z; }

    public static Quaternion Identity => new Quaternion(1, 0, 0, 0);

    public Vector3 Rotate(Vector3 v)
    {
        // Optimized quaternion rotation: v' = v + w*t + q.xyz x t, where t = 2*(q.xyz x v)
        double tx = 2 * (Y * v.Z - Z * v.Y);
        double ty = 2 * (Z * v.X - X * v.Z);
        double tz = 2 * (X * v.Y - Y * v.X);

        return new Vector3(
            v.X + W * tx + (Y * tz - Z * ty),
            v.Y + W * ty + (Z * tx - X * tz),
            v.Z + W * tz + (X * ty - Y * tx)
        );
    }
}

struct Face
{
    public int A, B, C;
    public Face(int a, int b, int c) { A = a; B = b; C = c; }
}

struct AABB
{
    public Vector3 Min, Max;

    public bool Overlaps(AABB other)
    {
        return Min.X < other.Max.X && Max.X > other.Min.X &&
               Min.Y < other.Max.Y && Max.Y > other.Min.Y &&
               Min.Z < other.Max.Z && Max.Z > other.Min.Z;
    }

    public bool IsInsideContainer(Vector3 container)
    {
        const double eps = 1e-9;
        return Min.X >= -eps && Min.Y >= -eps && Min.Z >= -eps &&
               Max.X <= container.X + eps && Max.Y <= container.Y + eps && Max.Z <= container.Z + eps;
    }
}

class Input
{
    public List<Vector3> Vertices;
    public List<Face> Faces;
    public Vector3 Container;
}

class Placement
{
    public Vector3 Translation;
    public Quaternion Rotation;
}

class BinPackingSolver
{
    private Input input;
    private List<Quaternion> rotations;
    private double gridStep;

    public BinPackingSolver(Input input)
    {
        this.input = input;

        // Adaptive strategy based on container volume
        double volume = input.Container.X * input.Container.Y * input.Container.Z;
        int maxRotations;
        double stepMultiplier;

        if (volume < 1000) // Trivial: dense packing, many rotations
        {
            maxRotations = 24;
            stepMultiplier = 0.7;
        }
        else if (volume < 8000) // Medium: balanced approach
        {
            maxRotations = 12;
            stepMultiplier = 0.85;
        }
        else if (volume < 1000000) // Large: fewer rotations, standard grid
        {
            maxRotations = 6;
            stepMultiplier = 1.0;
        }
        else // Extreme: minimal rotations, coarse grid for speed
        {
            maxRotations = 4;
            stepMultiplier = 1.3;
        }

        GenerateRotations(maxRotations);

        // Calculate grid step based on polyhedron size
        var baseAABB = CalculateAABB(input.Vertices, new Vector3(0, 0, 0), Quaternion.Identity);
        var dims = new Vector3(
            baseAABB.Max.X - baseAABB.Min.X,
            baseAABB.Max.Y - baseAABB.Min.Y,
            baseAABB.Max.Z - baseAABB.Min.Z
        );
        gridStep = Math.Max(dims.X, Math.Max(dims.Y, dims.Z)) * stepMultiplier;
    }

    private void GenerateRotations(int maxCount)
    {
        rotations = new List<Quaternion> { Quaternion.Identity };
        if (maxCount <= 1) return;

        double s = Math.Sqrt(0.5); // sin/cos of 45 degrees

        // 90-degree rotations around principal axes
        rotations.Add(new Quaternion(s, s, 0, 0)); // 90° around X
        rotations.Add(new Quaternion(s, 0, s, 0)); // 90° around Y
        rotations.Add(new Quaternion(s, 0, 0, s)); // 90° around Z
        if (maxCount <= 4) return;

        // 180-degree rotations
        rotations.Add(new Quaternion(0, 1, 0, 0)); // 180° around X
        rotations.Add(new Quaternion(0, 0, 1, 0)); // 180° around Y
        if (maxCount <= 6) return;

        // 270-degree and diagonal rotations
        rotations.Add(new Quaternion(s, -s, 0, 0)); // 270° around X
        rotations.Add(new Quaternion(s, 0, -s, 0)); // 270° around Y
        rotations.Add(new Quaternion(s, 0, 0, -s)); // 270° around Z
        rotations.Add(new Quaternion(0, 0, 0, 1)); // 180° around Z
        rotations.Add(new Quaternion(0.5, 0.5, 0.5, 0.5)); // Diagonal
        rotations.Add(new Quaternion(0.5, 0.5, -0.5, 0.5));
        if (maxCount <= 12) return;

        // Additional rotations for fine-grained packing
        for (int i = 12; i < maxCount; i++)
        {
            double angle = Math.PI * i / 12;
            double c = Math.Cos(angle / 2);
            double ss = Math.Sin(angle / 2);
            rotations.Add(new Quaternion(c, ss, 0, 0));
            if (rotations.Count >= maxCount) break;
        }
    }

    public List<Placement> Solve()
    {
        var placements = new List<Placement>();
        var occupied = new List<AABB>();

        // Grid-based greedy packing
        for (double x = 0; x < input.Container.X; x += gridStep)
        {
            for (double y = 0; y < input.Container.Y; y += gridStep)
            {
                for (double z = 0; z < input.Container.Z; z += gridStep)
                {
                    // Try each rotation at this grid position
                    foreach (var rotation in rotations)
                    {
                        // Calculate AABB at origin, then adjust translation
                        var aabbAtOrigin = CalculateAABB(input.Vertices, new Vector3(0, 0, 0), rotation);

                        // Translate so that AABB min corner aligns with grid position
                        var translation = new Vector3(
                            x - aabbAtOrigin.Min.X,
                            y - aabbAtOrigin.Min.Y,
                            z - aabbAtOrigin.Min.Z
                        );

                        var aabb = CalculateAABB(input.Vertices, translation, rotation);

                        // Validate placement
                        if (!aabb.IsInsideContainer(input.Container))
                            continue;

                        // Check for overlaps with already placed polyhedra
                        bool overlaps = false;
                        foreach (var other in occupied)
                        {
                            if (aabb.Overlaps(other))
                            {
                                overlaps = true;
                                break;
                            }
                        }

                        if (!overlaps)
                        {
                            placements.Add(new Placement { Translation = translation, Rotation = rotation });
                            occupied.Add(aabb);
                            break; // Found valid rotation, move to next grid position
                        }
                    }
                }
            }
        }

        return placements;
    }

    private AABB CalculateAABB(List<Vector3> vertices, Vector3 translation, Quaternion rotation)
    {
        double minX = double.MaxValue, minY = double.MaxValue, minZ = double.MaxValue;
        double maxX = double.MinValue, maxY = double.MinValue, maxZ = double.MinValue;

        foreach (var v in vertices)
        {
            var transformed = rotation.Rotate(v) + translation;

            minX = Math.Min(minX, transformed.X);
            minY = Math.Min(minY, transformed.Y);
            minZ = Math.Min(minZ, transformed.Z);
            maxX = Math.Max(maxX, transformed.X);
            maxY = Math.Max(maxY, transformed.Y);
            maxZ = Math.Max(maxZ, transformed.Z);
        }

        return new AABB { Min = new Vector3(minX, minY, minZ), Max = new Vector3(maxX, maxY, maxZ) };
    }
}
