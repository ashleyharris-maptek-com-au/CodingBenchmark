declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

interface Drone {
    id: number;
    x: number;
    y: number;
}

interface Input {
    probability: number[][];
    drones: Drone[];
    maxStepsPerDrone: number;
}

const { probability, drones, maxStepsPerDrone }: Input = data;

const height = probability.length;
const width = probability[0]?.length || 0;

function cellKey(x: number, y: number): string {
    return `${x},${y}`;
}

function isValid(x: number, y: number): boolean {
    return x >= 0 && x < width && y >= 0 && y < height && probability[y][x] !== -1;
}

const globalCovered = new Set<string>();

function planDronePath(drone: Drone, maxSteps: number): [number, number][] {
    const path: [number, number][] = [];
    let x = drone.x;
    let y = drone.y;

    path.push([x, y]);
    globalCovered.add(cellKey(x, y));

    for (let step = 1; step < maxSteps; step++) {
        let bestMove = { x, y, score: -Infinity };

        // Consider all possible moves including staying in place
        const moves = [
            { dx: 0, dy: 1 },
            { dx: 0, dy: -1 },
            { dx: 1, dy: 0 },
            { dx: -1, dy: 0 },
            { dx: 0, dy: 0 }
        ];

        for (const { dx, dy } of moves) {
            const nx = x + dx;
            const ny = y + dy;

            if (!isValid(nx, ny)) continue;

            const key = cellKey(nx, ny);
            let score = 0;

            // Immediate value of this cell if not yet covered
            if (!globalCovered.has(key)) {
                score += probability[ny][nx] * 1000;
            }

            // Lookahead: evaluate potential future moves
            const remaining = maxSteps - step - 1;
            if (remaining > 0) {
                const searchRadius = Math.min(remaining, 25);

                for (let dy2 = -searchRadius; dy2 <= searchRadius; dy2++) {
                    for (let dx2 = -searchRadius; dx2 <= searchRadius; dx2++) {
                        const lx = nx + dx2;
                        const ly = ny + dy2;

                        if (!isValid(lx, ly)) continue;

                        const dist = Math.abs(dx2) + Math.abs(dy2);
                        if (dist === 0 || dist > remaining) continue;

                        const lkey = cellKey(lx, ly);
                        if (globalCovered.has(lkey)) continue;

                        const value = probability[ly][lx];
                        if (value > 0) {
                            // Weight by inverse distance
                            score += value / (1 + dist * 0.3);
                        }
                    }
                }
            }

            if (score > bestMove.score) {
                bestMove = { x: nx, y: ny, score };
            }
        }

        x = bestMove.x;
        y = bestMove.y;
        path.push([x, y]);
        globalCovered.add(cellKey(x, y));
    }

    return path;
}

// Plan paths for all drones sequentially
const paths: [number, number][][] = [];
for (const drone of drones) {
    const path = planDronePath(drone, maxStepsPerDrone);
    paths.push(path);
}
