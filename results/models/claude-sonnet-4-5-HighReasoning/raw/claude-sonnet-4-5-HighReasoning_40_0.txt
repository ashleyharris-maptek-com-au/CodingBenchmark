#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int n, m;
vector<int> c;
vector<vector<int>> adj;
vector<int> best_solution;
int best_value = 0;

// Compute greedy upper bound for remaining unassigned variables
int compute_upper_bound(const vector<int>& assignment, int current_value) {
    vector<bool> available(n);
    for (int i = 0; i < n; i++) {
        available[i] = (assignment[i] == -1);
        if (assignment[i] == 1) {
            // Mark neighbors as unavailable
            for (int j : adj[i]) {
                available[j] = false;
            }
        }
    }

    // Sort remaining variables by coefficient (greedy)
    vector<pair<int, int>> candidates;
    for (int i = 0; i < n; i++) {
        if (available[i]) {
            candidates.push_back({c[i], i});
        }
    }
    sort(candidates.rbegin(), candidates.rend());

    int upper = current_value;
    for (auto [val, i] : candidates) {
        if (available[i]) {
            upper += val;
            // Mark neighbors as unavailable for this greedy bound
            for (int j : adj[i]) {
                available[j] = false;
            }
        }
    }

    return upper;
}

void solve(vector<int>& assignment, int current_value) {
    // Prune if upper bound can't beat current best
    if (compute_upper_bound(assignment, current_value) <= best_value) {
        return;
    }

    // Find next unassigned variable with highest coefficient
    int next = -1;
    int max_c = -1;
    for (int i = 0; i < n; i++) {
        if (assignment[i] == -1 && c[i] > max_c) {
            max_c = c[i];
            next = i;
        }
    }

    // All variables assigned - update best if better
    if (next == -1) {
        if (current_value > best_value) {
            best_value = current_value;
            best_solution = assignment;
        }
        return;
    }

    // Check if we can set next to 1 (no neighbor is 1)
    bool can_set_1 = true;
    for (int j : adj[next]) {
        if (assignment[j] == 1) {
            can_set_1 = false;
            break;
        }
    }

    // Try setting to 1 first (if feasible)
    if (can_set_1) {
        assignment[next] = 1;
        solve(assignment, current_value + c[next]);
        assignment[next] = -1;
    }

    // Try setting to 0
    assignment[next] = 0;
    solve(assignment, current_value);
    assignment[next] = -1;
}

int main() {
    cin >> n >> m;

    c.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> c[i];
    }

    adj.resize(n);
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    vector<int> assignment(n, -1);  // -1 = unassigned, 0/1 = assigned value
    best_solution.resize(n, 0);

    solve(assignment, 0);

    cout << best_value << endl;
    for (int i = 0; i < n; i++) {
        cout << best_solution[i];
        if (i < n - 1) cout << " ";
    }
    cout << endl;

    return 0;
}
