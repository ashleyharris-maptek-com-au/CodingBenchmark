use std::collections::{BinaryHeap, HashSet};
use std::io::{self, BufRead};

#[derive(Eq, PartialEq)]
struct VertexDegree {
    vertex: usize,
    degree: usize,
}

impl Ord for VertexDegree {
    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
        self.degree.cmp(&other.degree)
            .then_with(|| other.vertex.cmp(&self.vertex))
    }
}

impl PartialOrd for VertexDegree {
    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
        Some(self.cmp(other))
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = lines.next().unwrap().unwrap();
    let parts: Vec<usize> = first_line.split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();
    let n = parts[0];
    let m = parts[1];

    let mut edges = Vec::with_capacity(m);

    for _ in 0..m {
        let line = lines.next().unwrap().unwrap();
        let parts: Vec<usize> = line.split_whitespace()
            .map(|s| s.parse().unwrap())
            .collect();
        let u = parts[0];
        let v = parts[1];
        edges.push((u, v));
    }

    // Choose strategy based on graph size
    let cover = if n <= 10000 && m <= 50000 {
        greedy_degree_cover_heap(&edges, n)
    } else {
        maximal_matching_cover(&edges, n)
    };

    // Try to improve with local search
    let cover = local_search_improvement(&edges, cover);

    // Output
    let mut cover_vec: Vec<_> = cover.into_iter().collect();
    cover_vec.sort_unstable();
    println!("{}", cover_vec.len());
    for (i, v) in cover_vec.iter().enumerate() {
        if i > 0 {
            print!(" ");
        }
        print!("{}", v);
    }
    println!();
}

fn greedy_degree_cover_heap(edges: &[(usize, usize)], n: usize) -> HashSet<usize> {
    let mut cover = HashSet::new();
    let mut edge_covered = vec![false; edges.len()];
    let mut uncovered_count = edges.len();

    // Build edge index for each vertex
    let mut vertex_edges: Vec<Vec<usize>> = vec![Vec::new(); n];
    for (i, &(u, v)) in edges.iter().enumerate() {
        vertex_edges[u].push(i);
        vertex_edges[v].push(i);
    }

    // Build initial heap
    let mut heap = BinaryHeap::new();
    let mut current_degrees = vec![0; n];
    for v in 0..n {
        let degree = vertex_edges[v].len();
        current_degrees[v] = degree;
        if degree > 0 {
            heap.push(VertexDegree { vertex: v, degree });
        }
    }

    while uncovered_count > 0 {
        // Find vertex with highest degree
        let mut best = None;
        while let Some(vd) = heap.pop() {
            if current_degrees[vd.vertex] == vd.degree && !cover.contains(&vd.vertex) {
                best = Some(vd.vertex);
                break;
            }
        }

        let best_vertex = match best {
            Some(v) => v,
            None => break,
        };

        cover.insert(best_vertex);

        // Mark edges as covered and update degrees
        let mut affected_vertices = HashSet::new();
        for &e in &vertex_edges[best_vertex] {
            if !edge_covered[e] {
                edge_covered[e] = true;
                uncovered_count -= 1;

                let (u, v) = edges[e];
                if u != best_vertex {
                    affected_vertices.insert(u);
                }
                if v != best_vertex {
                    affected_vertices.insert(v);
                }
            }
        }

        // Recompute degrees for affected vertices and re-add to heap
        for v in affected_vertices {
            let new_degree = vertex_edges[v].iter()
                .filter(|&&e| !edge_covered[e])
                .count();
            current_degrees[v] = new_degree;
            if new_degree > 0 {
                heap.push(VertexDegree { vertex: v, degree: new_degree });
            }
        }
    }

    cover
}

fn maximal_matching_cover(edges: &[(usize, usize)], n: usize) -> HashSet<usize> {
    let mut matched = vec![false; n];
    let mut cover = HashSet::new();

    for &(u, v) in edges {
        if !matched[u] && !matched[v] {
            matched[u] = true;
            matched[v] = true;
            cover.insert(u);
            cover.insert(v);
        }
    }

    // Handle remaining unmatched edges
    for &(u, v) in edges {
        if !cover.contains(&u) && !cover.contains(&v) {
            cover.insert(u);
        }
    }

    cover
}

fn local_search_improvement(edges: &[(usize, usize)], mut cover: HashSet<usize>) -> HashSet<usize> {
    if edges.is_empty() {
        return cover;
    }

    let max_iterations = if edges.len() > 1000000 { 10 } else if edges.len() > 100000 { 50 } else { 100 };

    // Build edge list per vertex
    let n = edges.iter().flat_map(|&(u, v)| vec![u, v]).max().unwrap() + 1;
    let mut vertex_edges: Vec<Vec<usize>> = vec![Vec::new(); n];
    for (i, &(u, v)) in edges.iter().enumerate() {
        vertex_edges[u].push(i);
        vertex_edges[v].push(i);
    }

    let mut improved = true;
    let mut iteration = 0;

    while improved && iteration < max_iterations {
        improved = false;
        iteration += 1;

        let vertices: Vec<_> = cover.iter().copied().collect();
        for v in vertices {
            // Check if removing v would leave any edge uncovered
            let mut can_remove = true;
            for &e in &vertex_edges[v] {
                let (u, w) = edges[e];
                let other = if u == v { w } else { u };
                if !cover.contains(&other) {
                    can_remove = false;
                    break;
                }
            }

            if can_remove {
                cover.remove(&v);
                improved = true;
                break;
            }
        }
    }

    cover
}
