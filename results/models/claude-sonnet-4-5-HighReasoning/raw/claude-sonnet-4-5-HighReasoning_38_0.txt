def solve_exact_cover():
    # Read input
    line = input().split()
    U = int(line[0])
    S = int(line[1])

    sets = []
    for _ in range(S):
        line = input().strip()
        if line:
            sets.append(set(int(x) for x in line.split()))
        else:
            sets.append(set())

    # Backtracking to find exact cover
    def backtrack(covered, chosen, start_idx):
        # If all elements are covered, we found a solution
        if len(covered) == U:
            return chosen[:]

        # Try each remaining set
        for i in range(start_idx, S):
            # Check if this set can be added (no overlap with covered)
            if sets[i].isdisjoint(covered):
                # Add this set
                new_covered = covered | sets[i]
                chosen.append(i)

                result = backtrack(new_covered, chosen, i + 1)
                if result is not None:
                    return result

                # Backtrack
                chosen.pop()

        return None

    solution = backtrack(set(), [], 0)

    if solution is not None:
        print("SOLUTION")
        print(' '.join(map(str, solution)))
    else:
        print("NO SOLUTION")

solve_exact_cover()
