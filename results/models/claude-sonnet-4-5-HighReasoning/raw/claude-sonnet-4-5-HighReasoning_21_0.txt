#include <iostream>
#include <vector>
#include <queue>
#include <set>
#include <unordered_set>
#include <unordered_map>
#include <algorithm>
#include <deque>
#include <limits>
#include <cmath>

using namespace std;

struct Pos {
    vector<int> coords;

    Pos() {}
    explicit Pos(int d) : coords(d, 0) {}
    explicit Pos(const vector<int>& c) : coords(c) {}

    bool operator==(const Pos& other) const {
        return coords == other.coords;
    }

    bool operator<(const Pos& other) const {
        return coords < other.coords;
    }

    int manhattanDist(const Pos& other) const {
        int d = 0;
        for (size_t i = 0; i < coords.size(); i++) {
            d += abs(coords[i] - other.coords[i]);
        }
        return d;
    }

    Pos move(int axis, int dir) const {
        Pos result = *this;
        result.coords[axis] += dir;
        return result;
    }
};

struct PosHash {
    size_t operator()(const Pos& p) const {
        size_t h = 0;
        for (size_t i = 0; i < p.coords.size(); i++) {
            h ^= hash<int>()(p.coords[i]) + 0x9e3779b9 + (h << 6) + (h >> 2);
        }
        return h;
    }
};

class SnakeGame {
private:
    int D;
    int maxTurns;
    vector<int> bounds;
    deque<Pos> snake;
    vector<Pos> food;
    set<Pos> foodSet;
    set<Pos> obstacles;
    int turnCount;

    bool isValid(const Pos& pos) const {
        for (int i = 0; i < D; i++) {
            if (pos.coords[i] < 0 || pos.coords[i] >= bounds[i]) {
                return false;
            }
        }
        return true;
    }

    bool isOccupiedBySnake(const Pos& pos) const {
        for (const auto& segment : snake) {
            if (segment == pos) return true;
        }
        return false;
    }

    // BFS pathfinding to target
    pair<bool, pair<int, int>> findPathBFS(const Pos& target) {
        unordered_map<Pos, pair<int, int>, PosHash> parent;
        queue<Pos> q;

        q.push(snake.front());
        parent[snake.front()] = {-1, 0};

        int searchLimit = min(50000, 5000 * max(1, 8 - D));
        int visited = 0;

        while (!q.empty() && visited < searchLimit) {
            Pos current = q.front();
            q.pop();
            visited++;

            if (current == target) {
                // Backtrack to find first move
                Pos pos = current;
                while (parent[pos].first != -1) {
                    Pos prev = pos;
                    for (int axis = 0; axis < D; axis++) {
                        for (int dir = -1; dir <= 1; dir += 2) {
                            if (prev.move(axis, -dir) == parent[pos] ||
                                (parent[pos].first == -2 && prev.move(axis, -dir) == snake.front())) {
                                pos = prev.move(axis, -dir);
                                if (pos == snake.front()) {
                                    return {true, {axis, dir}};
                                }
                                goto next_iter;
                            }
                        }
                    }
                    next_iter:;
                }
            }

            for (int axis = 0; axis < D; axis++) {
                for (int dir = -1; dir <= 1; dir += 2) {
                    Pos next = current.move(axis, dir);

                    if (!isValid(next) || parent.count(next)) {
                        continue;
                    }

                    // Check if blocked
                    if (obstacles.count(next)) {
                        continue;
                    }

                    // Allow tail position (it will move away)
                    bool blocked = false;
                    for (size_t i = 0; i < snake.size() - 1; i++) {
                        if (snake[i] == next) {
                            blocked = true;
                            break;
                        }
                    }
                    if (blocked) continue;

                    parent[next] = current;
                    q.push(next);
                }
            }
        }

        return {false, {-1, 0}};
    }

    // Greedy move towards target
    pair<int, int> greedyMove(const Pos& target) {
        Pos head = snake.front();
        int bestAxis = 0;
        int bestDir = 1;
        int bestScore = numeric_limits<int>::min();
        bool foundValid = false;

        for (int axis = 0; axis < D; axis++) {
            for (int dir = -1; dir <= 1; dir += 2) {
                Pos next = head.move(axis, dir);

                if (!isValid(next)) continue;
                if (obstacles.count(next)) continue;

                // Check snake collision (allow tail)
                bool blocked = false;
                for (size_t i = 0; i < snake.size() - 1; i++) {
                    if (snake[i] == next) {
                        blocked = true;
                        break;
                    }
                }
                if (blocked) continue;

                foundValid = true;

                // Score based on distance to target
                int score = -next.manhattanDist(target) * 1000;

                // Bonus for having free neighbors (safety)
                int freeNeighbors = 0;
                for (int a = 0; a < D; a++) {
                    for (int d = -1; d <= 1; d += 2) {
                        Pos neighbor = next.move(a, d);
                        if (isValid(neighbor) && !obstacles.count(neighbor)) {
                            bool snakeBlock = false;
                            for (size_t i = 0; i < snake.size() - 1; i++) {
                                if (snake[i] == neighbor) {
                                    snakeBlock = true;
                                    break;
                                }
                            }
                            if (!snakeBlock) freeNeighbors++;
                        }
                    }
                }
                score += freeNeighbors * 10;

                if (score > bestScore) {
                    bestScore = score;
                    bestAxis = axis;
                    bestDir = dir;
                }
            }
        }

        if (!foundValid) {
            // Desperate: try any move
            for (int axis = 0; axis < D; axis++) {
                for (int dir = -1; dir <= 1; dir += 2) {
                    Pos next = head.move(axis, dir);
                    if (isValid(next)) {
                        return {axis, dir};
                    }
                }
            }
        }

        return {bestAxis, bestDir};
    }

    pair<int, int> findBestMove() {
        if (food.empty()) {
            return {0, 1};
        }

        // Find nearest food
        int minDist = numeric_limits<int>::max();
        Pos nearestFood = food[0];
        for (const auto& f : food) {
            int d = snake.front().manhattanDist(f);
            if (d < minDist) {
                minDist = d;
                nearestFood = f;
            }
        }

        // Use BFS for lower dimensions
        if (D <= 5) {
            auto [found, move] = findPathBFS(nearestFood);
            if (found) {
                return move;
            }
        }

        // Fallback to greedy
        return greedyMove(nearestFood);
    }

public:
    void readInput() {
        cin >> D >> maxTurns;

        bounds.resize(D);
        for (int i = 0; i < D; i++) {
            cin >> bounds[i];
        }

        int snakeLength;
        cin >> snakeLength;

        Pos head(D);
        for (int i = 0; i < D; i++) {
            cin >> head.coords[i];
        }

        // Initialize snake - all segments start at head position
        for (int i = 0; i < snakeLength; i++) {
            snake.push_back(head);
        }

        int F;
        cin >> F;
        for (int i = 0; i < F; i++) {
            Pos f(D);
            for (int j = 0; j < D; j++) {
                cin >> f.coords[j];
            }
            food.push_back(f);
            foodSet.insert(f);
        }

        int O;
        cin >> O;
        for (int i = 0; i < O; i++) {
            Pos o(D);
            for (int j = 0; j < D; j++) {
                cin >> o.coords[j];
            }
            obstacles.insert(o);
        }

        turnCount = 0;
    }

    void play() {
        while (turnCount < maxTurns && !food.empty()) {
            auto [axis, dir] = findBestMove();

            Pos newHead = snake.front().move(axis, dir);

            // Validate move
            if (!isValid(newHead)) {
                break;
            }

            if (obstacles.count(newHead)) {
                break;
            }

            // Check self-collision (excluding tail if not eating)
            bool ateFood = foodSet.count(newHead) > 0;
            if (!ateFood) {
                for (size_t i = 0; i < snake.size() - 1; i++) {
                    if (snake[i] == newHead) {
                        // Game over
                        return;
                    }
                }
            } else {
                for (size_t i = 0; i < snake.size(); i++) {
                    if (snake[i] == newHead) {
                        // Game over
                        return;
                    }
                }
            }

            // Output move
            cout << axis << " " << dir << "\n";

            // Update snake
            snake.push_front(newHead);

            if (ateFood) {
                // Grow snake
                foodSet.erase(newHead);
                food.erase(remove(food.begin(), food.end(), newHead), food.end());
            } else {
                // Remove tail
                snake.pop_back();
            }

            turnCount++;
        }
    }
};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    SnakeGame game;
    game.readInput();
    game.play();

    return 0;
}
