#include <iostream>
#include <vector>
#include <algorithm>
#include <set>

using namespace std;

class MaxIndependentSet {
private:
    int n;
    vector<vector<int>> adj;
    vector<int> best_set;
    int best_size;

    // Compute upper bound using greedy coloring (chromatic number)
    int compute_upper_bound(const vector<int>& candidates, const vector<bool>& excluded) {
        if (candidates.empty()) return 0;

        vector<int> color(n, -1);
        int max_color = -1;

        for (int v : candidates) {
            if (excluded[v]) continue;

            set<int> used_colors;
            for (int u : adj[v]) {
                if (color[u] >= 0) {
                    used_colors.insert(color[u]);
                }
            }

            int c = 0;
            while (used_colors.count(c)) c++;
            color[v] = c;
            max_color = max(max_color, c);
        }

        return max_color + 1;
    }

    void branch_and_bound(vector<int>& current, vector<int>& candidates, vector<bool>& excluded) {
        // Update best solution if current is better
        if ((int)current.size() > best_size) {
            best_size = current.size();
            best_set = current;
        }

        // Pruning: count remaining vertices
        int remaining = 0;
        for (int v : candidates) {
            if (!excluded[v]) remaining++;
        }

        if ((int)current.size() + remaining <= best_size) {
            return;
        }

        // Pruning: upper bound check
        int ub = compute_upper_bound(candidates, excluded);
        if ((int)current.size() + ub <= best_size) {
            return;
        }

        // Find vertex with maximum degree among non-excluded candidates
        int best_v = -1;
        int max_degree = -1;

        for (int v : candidates) {
            if (excluded[v]) continue;

            int degree = 0;
            for (int u : adj[v]) {
                if (!excluded[u]) {
                    degree++;
                }
            }

            if (degree > max_degree) {
                max_degree = degree;
                best_v = v;
            }
        }

        if (best_v == -1) return;

        // Branch 1: Include best_v in the independent set
        current.push_back(best_v);
        vector<bool> new_excluded = excluded;
        new_excluded[best_v] = true;

        // Exclude all neighbors of best_v
        for (int u : adj[best_v]) {
            new_excluded[u] = true;
        }

        branch_and_bound(current, candidates, new_excluded);
        current.pop_back();

        // Branch 2: Exclude best_v from the independent set
        excluded[best_v] = true;
        branch_and_bound(current, candidates, excluded);
    }

public:
    MaxIndependentSet(int n) : n(n), adj(n), best_size(0) {}

    void add_edge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    vector<int> solve() {
        vector<int> current;
        vector<int> candidates;

        for (int i = 0; i < n; i++) {
            candidates.push_back(i);
        }

        // Sort candidates by degree (descending) for better pruning
        sort(candidates.begin(), candidates.end(), [&](int a, int b) {
            return adj[a].size() > adj[b].size();
        });

        vector<bool> excluded(n, false);
        branch_and_bound(current, candidates, excluded);

        return best_set;
    }
};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    MaxIndependentSet solver(n);

    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        solver.add_edge(u, v);
    }

    vector<int> result = solver.solve();

    cout << result.size() << "\n";
    for (size_t i = 0; i < result.size(); i++) {
        if (i > 0) cout << " ";
        cout << result[i];
    }
    if (!result.empty()) {
        cout << "\n";
    }

    return 0;
}
