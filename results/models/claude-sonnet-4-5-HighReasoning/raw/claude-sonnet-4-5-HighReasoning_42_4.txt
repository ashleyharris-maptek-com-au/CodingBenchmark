OpCapability Geometry
%1 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %color_in %gl_out %outColor
OpExecutionMode %main InputLines
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1

; Decorations
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %color_in Location 2
OpDecorate %outColor Location 0

; Types
%void = OpTypeVoid
%void_func = OpTypeFunction %void
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0

; Constants
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%uint_2 = OpConstant %uint 2
%float_0_04 = OpConstant %float 0.04
%float_0_005 = OpConstant %float 0.005

; Input types
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in_2 = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in_2
%ptr_in_v4 = OpTypePointer Input %v4float
%arr_v3_2 = OpTypeArray %v3float %uint_2
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_2
%ptr_in_v3 = OpTypePointer Input %v3float

; Output types
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float

; Input variables
%gl_in = OpVariable %ptr_in_arr_pv Input
%color_in = OpVariable %ptr_in_arr_v3 Input

; Output variables
%gl_out = OpVariable %ptr_out_pv Output
%outColor = OpVariable %ptr_out_v3 Output

; Main function
%main = OpFunction %void None %void_func
%entry = OpLabel

; Read input positions
%50 = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%p0 = OpLoad %v4float %50
%51 = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%p1 = OpLoad %v4float %51

; Read input colors
%52 = OpAccessChain %ptr_in_v3 %color_in %int_0
%c0 = OpLoad %v3float %52
%53 = OpAccessChain %ptr_in_v3 %color_in %int_1
%c1 = OpLoad %v3float %53

; Extract xy components for 2D calculations
%p0_xy = OpVectorShuffle %v2float %p0 %p0 0 1
%p1_xy = OpVectorShuffle %v2float %p1 %p1 0 1

; Compute direction: dir = p1_xy - p0_xy
%dir = OpFSub %v2float %p1_xy %p0_xy

; Extract direction components
%dir_x = OpCompositeExtract %float %dir 0
%dir_y = OpCompositeExtract %float %dir 1

; Compute perpendicular: perp = (-dir.y, dir.x)
%neg_dir_y = OpFNegate %float %dir_y
%perp_unnorm = OpCompositeConstruct %v2float %neg_dir_y %dir_x

; Normalize perpendicular
%perp_norm = OpExtInst %v2float %1 Normalize %perp_unnorm

; Scale by thickness at start and end
%perp_start = OpVectorTimesScalar %v2float %perp_norm %float_0_04
%perp_end = OpVectorTimesScalar %v2float %perp_norm %float_0_005

; Compute 4 corner positions in xy plane
%p0_plus_xy = OpFAdd %v2float %p0_xy %perp_start
%p0_minus_xy = OpFSub %v2float %p0_xy %perp_start
%p1_plus_xy = OpFAdd %v2float %p1_xy %perp_end
%p1_minus_xy = OpFSub %v2float %p1_xy %perp_end

; Extract z and w components to preserve them
%p0_zw = OpVectorShuffle %v2float %p0 %p0 2 3
%p1_zw = OpVectorShuffle %v2float %p1 %p1 2 3

; Reconstruct vec4 positions
%v0 = OpVectorShuffle %v4float %p0_plus_xy %p0_zw 0 1 2 3
%v1 = OpVectorShuffle %v4float %p0_minus_xy %p0_zw 0 1 2 3
%v2 = OpVectorShuffle %v4float %p1_plus_xy %p1_zw 0 1 2 3
%v3 = OpVectorShuffle %v4float %p1_minus_xy %p1_zw 0 1 2 3

; Get pointer to output position
%gl_pos_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0

; Emit vertex 0: p0 + perp*thick_start, color c0
OpStore %gl_pos_ptr %v0
OpStore %outColor %c0
OpEmitVertex

; Emit vertex 1: p0 - perp*thick_start, color c0
OpStore %gl_pos_ptr %v1
OpStore %outColor %c0
OpEmitVertex

; Emit vertex 2: p1 + perp*thick_end, color c1
OpStore %gl_pos_ptr %v2
OpStore %outColor %c1
OpEmitVertex

; Emit vertex 3: p1 - perp*thick_end, color c1
OpStore %gl_pos_ptr %v3
OpStore %outColor %c1
OpEmitVertex

OpEndPrimitive

OpReturn
OpFunctionEnd
