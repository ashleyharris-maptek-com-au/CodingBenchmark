use std::io::{self, BufRead};
use std::collections::HashMap;

#[derive(Debug, Clone)]
struct Sample {
    hole_id: String,
    sample_idx: usize,
    position: [f64; 3],
    properties: Vec<f64>,
}

#[derive(Debug)]
struct Outlier {
    hole_id: String,
    sample_idx: usize,
    property_name: String,
    confidence: f64,
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Read header
    let first_line = lines.next().unwrap().unwrap();
    let parts: Vec<&str> = first_line.split_whitespace().collect();
    let num_holes: usize = parts[0].parse().unwrap();
    let num_properties: usize = parts[1].parse().unwrap();

    // Read property names
    let property_line = lines.next().unwrap().unwrap();
    let property_names: Vec<String> = property_line.split_whitespace()
        .map(|s| s.to_string()).collect();

    // Read all holes and samples
    let mut samples = Vec::new();

    for _ in 0..num_holes {
        let hole_line = lines.next().unwrap().unwrap();
        let parts: Vec<&str> = hole_line.split_whitespace().collect();

        let hole_id = parts[0].to_string();
        let start_x: f64 = parts[1].parse().unwrap_or(0.0);
        let start_y: f64 = parts[2].parse().unwrap_or(0.0);
        let start_z: f64 = parts[3].parse().unwrap_or(0.0);
        let dir_x: f64 = parts[4].parse().unwrap_or(0.0);
        let dir_y: f64 = parts[5].parse().unwrap_or(0.0);
        let dir_z: f64 = parts[6].parse().unwrap_or(0.0);
        let _length: f64 = parts[7].parse().unwrap_or(0.0);
        let num_samples: usize = parts[8].parse().unwrap_or(0);

        for sample_idx in 0..num_samples {
            if let Some(Ok(sample_line)) = lines.next() {
                let parts: Vec<&str> = sample_line.split_whitespace().collect();

                if parts.is_empty() {
                    continue;
                }

                let depth: f64 = parts[0].parse().unwrap_or(0.0);
                let properties: Vec<f64> = parts[1..].iter()
                    .take(num_properties)
                    .map(|s| s.parse().unwrap_or(0.0)).collect();

                // Ensure we have the right number of properties
                if properties.len() != num_properties {
                    continue;
                }

                let position = [
                    start_x + dir_x * depth,
                    start_y + dir_y * depth,
                    start_z + dir_z * depth,
                ];

                samples.push(Sample {
                    hole_id: hole_id.clone(),
                    sample_idx,
                    position,
                    properties,
                });
            }
        }
    }

    if samples.is_empty() {
        println!("0");
        return;
    }

    // Choose strategy based on dataset size
    let total_samples = samples.len();
    let outliers = if total_samples < 5000 {
        // Small dataset: brute force with more neighbors
        detect_outliers_bruteforce(&samples, &property_names, 20)
    } else if total_samples < 50000 {
        // Medium dataset: grid-based spatial index
        detect_outliers_grid(&samples, &property_names, 20, 50)
    } else {
        // Large dataset: coarser grid, fewer neighbors
        detect_outliers_grid(&samples, &property_names, 15, 100)
    };

    // Output results
    println!("{}", outliers.len());
    for outlier in outliers {
        println!("{} {} {} {:.6}",
            outlier.hole_id,
            outlier.sample_idx,
            outlier.property_name,
            outlier.confidence);
    }
}

fn detect_outliers_bruteforce(
    samples: &[Sample],
    property_names: &[String],
    k: usize
) -> Vec<Outlier> {
    let mut outliers = Vec::new();

    for (i, sample) in samples.iter().enumerate() {
        // Find k nearest neighbors
        let mut distances: Vec<(usize, f64)> = samples.iter().enumerate()
            .filter(|(j, _)| *j != i)
            .map(|(j, other)| {
                let dist = euclidean_distance(&sample.position, &other.position);
                (j, dist)
            })
            .collect();

        if distances.is_empty() {
            continue;
        }

        distances.sort_by(|a, b| a.1.partial_cmp(&b.1).unwrap_or(std::cmp::Ordering::Equal));
        let neighbors: Vec<usize> = distances.iter()
            .take(k.min(distances.len()))
            .map(|(j, _)| *j)
            .collect();

        // Check each property
        for (prop_idx, prop_name) in property_names.iter().enumerate() {
            if prop_idx >= sample.properties.len() {
                continue;
            }

            let value = sample.properties[prop_idx];

            // Collect neighbor values
            let neighbor_values: Vec<f64> = neighbors.iter()
                .filter_map(|&j| {
                    if prop_idx < samples[j].properties.len() {
                        Some(samples[j].properties[prop_idx])
                    } else {
                        None
                    }
                })
                .collect();

            if neighbor_values.len() < 3 {
                continue;
            }

            // Calculate statistics
            let mean = neighbor_values.iter().sum::<f64>() / neighbor_values.len() as f64;
            let variance = neighbor_values.iter()
                .map(|v| (v - mean).powi(2))
                .sum::<f64>() / neighbor_values.len() as f64;
            let std_dev = variance.sqrt();

            if std_dev > 1e-10 {
                let z_score = (value - mean).abs() / std_dev;

                // Flag as outlier if z-score > 3
                if z_score > 3.0 {
                    outliers.push(Outlier {
                        hole_id: sample.hole_id.clone(),
                        sample_idx: sample.sample_idx,
                        property_name: prop_name.clone(),
                        confidence: z_score,
                    });
                }
            }
        }
    }

    // Sort by confidence descending
    outliers.sort_by(|a, b| b.confidence.partial_cmp(&a.confidence).unwrap_or(std::cmp::Ordering::Equal));
    outliers
}

fn detect_outliers_grid(
    samples: &[Sample],
    property_names: &[String],
    k: usize,
    grid_size: usize
) -> Vec<Outlier> {
    // Build spatial grid - find bounds
    let mut min_bounds = [f64::INFINITY; 3];
    let mut max_bounds = [f64::NEG_INFINITY; 3];

    for sample in samples {
        for i in 0..3 {
            min_bounds[i] = min_bounds[i].min(sample.position[i]);
            max_bounds[i] = max_bounds[i].max(sample.position[i]);
        }
    }

    // Calculate cell size with safeguards
    let mut cell_size = [0.0; 3];
    for i in 0..3 {
        let span = max_bounds[i] - min_bounds[i];
        cell_size[i] = if span > 1e-10 {
            span / grid_size as f64
        } else {
            1.0 // Default cell size for degenerate dimensions
        };
    }

    // Assign samples to grid cells
    let mut grid: HashMap<(i32, i32, i32), Vec<usize>> = HashMap::new();

    for (i, sample) in samples.iter().enumerate() {
        let cell = get_cell(&sample.position, &min_bounds, &cell_size);
        grid.entry(cell).or_insert_with(Vec::new).push(i);
    }

    let mut outliers = Vec::new();

    for (i, sample) in samples.iter().enumerate() {
        let cell = get_cell(&sample.position, &min_bounds, &cell_size);

        // Search neighboring cells
        let mut candidates = Vec::new();
        for dx in -1..=1 {
            for dy in -1..=1 {
                for dz in -1..=1 {
                    let neighbor_cell = (cell.0 + dx, cell.1 + dy, cell.2 + dz);
                    if let Some(indices) = grid.get(&neighbor_cell) {
                        for &j in indices {
                            if i != j {
                                let dist = euclidean_distance(&sample.position, &samples[j].position);
                                candidates.push((j, dist));
                            }
                        }
                    }
                }
            }
        }

        if candidates.is_empty() {
            continue;
        }

        candidates.sort_by(|a, b| a.1.partial_cmp(&b.1).unwrap_or(std::cmp::Ordering::Equal));
        let neighbors: Vec<usize> = candidates.iter()
            .take(k.min(candidates.len()))
            .map(|(j, _)| *j)
            .collect();

        // Check each property
        for (prop_idx, prop_name) in property_names.iter().enumerate() {
            if prop_idx >= sample.properties.len() {
                continue;
            }

            let value = sample.properties[prop_idx];

            let neighbor_values: Vec<f64> = neighbors.iter()
                .filter_map(|&j| {
                    if prop_idx < samples[j].properties.len() {
                        Some(samples[j].properties[prop_idx])
                    } else {
                        None
                    }
                })
                .collect();

            if neighbor_values.len() < 3 {
                continue;
            }

            let mean = neighbor_values.iter().sum::<f64>() / neighbor_values.len() as f64;
            let variance = neighbor_values.iter()
                .map(|v| (v - mean).powi(2))
                .sum::<f64>() / neighbor_values.len() as f64;
            let std_dev = variance.sqrt();

            if std_dev > 1e-10 {
                let z_score = (value - mean).abs() / std_dev;

                if z_score > 3.0 {
                    outliers.push(Outlier {
                        hole_id: sample.hole_id.clone(),
                        sample_idx: sample.sample_idx,
                        property_name: prop_name.clone(),
                        confidence: z_score,
                    });
                }
            }
        }
    }

    outliers.sort_by(|a, b| b.confidence.partial_cmp(&a.confidence).unwrap_or(std::cmp::Ordering::Equal));
    outliers
}

fn euclidean_distance(a: &[f64; 3], b: &[f64; 3]) -> f64 {
    ((a[0] - b[0]).powi(2) + (a[1] - b[1]).powi(2) + (a[2] - b[2]).powi(2)).sqrt()
}

fn get_cell(position: &[f64; 3], min_bounds: &[f64; 3], cell_size: &[f64; 3]) -> (i32, i32, i32) {
    (
        if cell_size[0] > 1e-10 {
            ((position[0] - min_bounds[0]) / cell_size[0]).floor() as i32
        } else {
            0
        },
        if cell_size[1] > 1e-10 {
            ((position[1] - min_bounds[1]) / cell_size[1]).floor() as i32
        } else {
            0
        },
        if cell_size[2] > 1e-10 {
            ((position[2] - min_bounds[2]) / cell_size[2]).floor() as i32
        } else {
            0
        },
    )
}
