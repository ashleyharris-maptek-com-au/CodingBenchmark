use std::collections::VecDeque;
use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut lines = input.lines();
    let first_line = lines.next().unwrap();
    let dims: Vec<usize> = first_line.split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();
    let height = dims[0];
    let width = dims[1];

    let mut maze = Vec::with_capacity(height * width);
    let mut start = 0;
    let mut end = 0;

    for (y, line) in lines.enumerate() {
        let bytes = line.as_bytes();
        for x in 0..width {
            let ch = if x < bytes.len() { bytes[x] } else { b' ' };
            let idx = y * width + x;
            if ch == b'A' {
                start = idx;
            } else if ch == b'B' {
                end = idx;
            }
            maze.push(ch);
        }
    }

    if let Some(path) = bfs(&maze, width, height, start, end) {
        println!("{}", path.len());
        for idx in path {
            let x = idx % width;
            let y = idx / width;
            println!("{} {}", x, y);
        }
    } else {
        println!("0");
    }
}

fn bfs(maze: &[u8], width: usize, height: usize, start: usize, end: usize) -> Option<Vec<usize>> {
    let size = width * height;
    let mut queue = VecDeque::new();
    let mut visited = vec![false; size];
    let mut parent = vec![u32::MAX; size];

    queue.push_back(start);
    visited[start] = true;

    while let Some(idx) = queue.pop_front() {
        if idx == end {
            // Reconstruct path
            let mut path = Vec::new();
            let mut current = end;
            path.push(current);

            while current != start {
                current = parent[current] as usize;
                path.push(current);
            }

            path.reverse();
            return Some(path);
        }

        let x = idx % width;
        let y = idx / width;

        // Try all 4 directions
        // Up
        if y > 0 {
            let next = idx - width;
            if !visited[next] && maze[next] != b'#' {
                visited[next] = true;
                parent[next] = idx as u32;
                queue.push_back(next);
            }
        }
        // Down
        if y + 1 < height {
            let next = idx + width;
            if !visited[next] && maze[next] != b'#' {
                visited[next] = true;
                parent[next] = idx as u32;
                queue.push_back(next);
            }
        }
        // Left
        if x > 0 {
            let next = idx - 1;
            if !visited[next] && maze[next] != b'#' {
                visited[next] = true;
                parent[next] = idx as u32;
                queue.push_back(next);
            }
        }
        // Right
        if x + 1 < width {
            let next = idx + 1;
            if !visited[next] && maze[next] != b'#' {
                visited[next] = true;
                parent[next] = idx as u32;
                queue.push_back(next);
            }
        }
    }

    None
}
