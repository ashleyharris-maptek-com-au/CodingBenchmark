OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %color_in %gl_out %outColor
               OpExecutionMode %main InputLinesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 8
               OpExecutionMode %main Invocations 1

               ; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %color_in Location 2
               OpDecorate %outColor Location 0

               ; Types
       %void = OpTypeVoid
          %3 = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v2float = OpTypeVector %float 2
    %v3float = OpTypeVector %float 3
    %v4float = OpTypeVector %float 4
        %int = OpTypeInt 32 1
       %uint = OpTypeInt 32 0
     %uint_4 = OpConstant %uint 4

               ; Integer constants
      %int_0 = OpConstant %int 0
      %int_1 = OpConstant %int 1
      %int_2 = OpConstant %int 2
      %int_3 = OpConstant %int 3

               ; Float constants
  %float_0_0 = OpConstant %float 0.0
  %float_1_0 = OpConstant %float 1.0
  %float_0_5 = OpConstant %float 0.5
 %float_0_02 = OpConstant %float 0.02
%float_1_3 = OpConstant %float 0.333333
%float_2_3 = OpConstant %float 0.666667

               ; Input structure types
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%ptr_in_v4 = OpTypePointer Input %v4float

%arr_color_in = OpTypeArray %v3float %uint_4
%ptr_in_arr_color = OpTypePointer Input %arr_color_in
%ptr_in_v3 = OpTypePointer Input %v3float

               ; Output structure types
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float

               ; Variables
     %gl_in = OpVariable %ptr_in_arr_pv Input
   %color_in = OpVariable %ptr_in_arr_color Input
    %gl_out = OpVariable %ptr_out_pv Output
   %outColor = OpVariable %ptr_out_v3 Output

               ; Main function
       %main = OpFunction %void None %3
      %label = OpLabel

               ; Load positions from gl_in[0..3].gl_Position
    %p0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
        %p0 = OpLoad %v4float %p0_ptr
    %p1_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
        %p1 = OpLoad %v4float %p1_ptr
    %p2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
        %p2 = OpLoad %v4float %p2_ptr
    %p3_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
        %p3 = OpLoad %v4float %p3_ptr

               ; Load colors from color_in[1] and color_in[2]
    %c1_ptr = OpAccessChain %ptr_in_v3 %color_in %int_1
        %c1 = OpLoad %v3float %c1_ptr
    %c2_ptr = OpAccessChain %ptr_in_v3 %color_in %int_2
        %c2 = OpLoad %v3float %c2_ptr

               ; Extract xy coordinates for tangent computation
     %p0_xy = OpVectorShuffle %v2float %p0 %p0 0 1
     %p1_xy = OpVectorShuffle %v2float %p1 %p1 0 1
     %p2_xy = OpVectorShuffle %v2float %p2 %p2 0 1
     %p3_xy = OpVectorShuffle %v2float %p3 %p3 0 1

               ; Tangent at start (p1): direction from p0 to p2
   %tang_s0 = OpFSub %v2float %p2_xy %p0_xy
    %tang_s = OpExtInst %v2float %1 Normalize %tang_s0

               ; Tangent at end (p2): direction from p1 to p3
   %tang_e0 = OpFSub %v2float %p3_xy %p1_xy
    %tang_e = OpExtInst %v2float %1 Normalize %tang_e0

               ; Perpendiculars (rotate 90 degrees: (x,y) -> (-y,x))
   %tang_sx = OpCompositeExtract %float %tang_s 0
   %tang_sy = OpCompositeExtract %float %tang_s 1
  %neg_sy = OpFNegate %float %tang_sy
    %perp_s = OpCompositeConstruct %v2float %neg_sy %tang_sx

   %tang_ex = OpCompositeExtract %float %tang_e 0
   %tang_ey = OpCompositeExtract %float %tang_e 1
  %neg_ey = OpFNegate %float %tang_ey
    %perp_e = OpCompositeConstruct %v2float %neg_ey %tang_ex

               ; Scale perpendiculars by thickness/2
 %half_thick = OpFMul %float %float_0_02 %float_0_5
  %perp_s_scaled = OpVectorTimesScalar %v2float %perp_s %half_thick
  %perp_e_scaled = OpVectorTimesScalar %v2float %perp_e %half_thick

               ; Segment direction
    %seg_dir = OpFSub %v2float %p2_xy %p1_xy

               ; --- FIRST DASH: t = 0 to 1/3 ---
               ; Position at t=0
     %pos_0 = %p1_xy
               ; Position at t=1/3
  %dir_1_3 = OpVectorTimesScalar %v2float %seg_dir %float_1_3
  %pos_1_3 = OpFAdd %v2float %p1_xy %dir_1_3

               ; Perpendicular at t=1/3 (lerp between perp_s and perp_e)
 %perp_diff = OpFSub %v2float %perp_e_scaled %perp_s_scaled
%perp_1_3_offset = OpVectorTimesScalar %v2float %perp_diff %float_1_3
 %perp_1_3 = OpFAdd %v2float %perp_s_scaled %perp_1_3_offset

               ; Color at t=0 and t=1/3
   %color_0 = %c1
%color_diff = OpFSub %v3float %c2 %c1
%color_1_3_offset = OpVectorTimesScalar %v3float %color_diff %float_1_3
%color_1_3 = OpFAdd %v3float %c1 %color_1_3_offset

               ; Emit quad for first dash (4 vertices in triangle strip)
               ; Vertex 1: pos_0 - perp_s_scaled
   %v1_pos2 = OpFSub %v2float %pos_0 %perp_s_scaled
    %v1_x = OpCompositeExtract %float %v1_pos2 0
    %v1_y = OpCompositeExtract %float %v1_pos2 1
  %v1_pos4 = OpCompositeConstruct %v4float %v1_x %v1_y %float_0_0 %float_1_0
%out_pos_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %out_pos_ptr %v1_pos4
               OpStore %outColor %color_0
               OpEmitVertex

               ; Vertex 2: pos_0 + perp_s_scaled
   %v2_pos2 = OpFAdd %v2float %pos_0 %perp_s_scaled
    %v2_x = OpCompositeExtract %float %v2_pos2 0
    %v2_y = OpCompositeExtract %float %v2_pos2 1
  %v2_pos4 = OpCompositeConstruct %v4float %v2_x %v2_y %float_0_0 %float_1_0
               OpStore %out_pos_ptr %v2_pos4
               OpStore %outColor %color_0
               OpEmitVertex

               ; Vertex 3: pos_1_3 - perp_1_3
   %v3_pos2 = OpFSub %v2float %pos_1_3 %perp_1_3
    %v3_x = OpCompositeExtract %float %v3_pos2 0
    %v3_y = OpCompositeExtract %float %v3_pos2 1
  %v3_pos4 = OpCompositeConstruct %v4float %v3_x %v3_y %float_0_0 %float_1_0
               OpStore %out_pos_ptr %v3_pos4
               OpStore %outColor %color_1_3
               OpEmitVertex

               ; Vertex 4: pos_1_3 + perp_1_3
   %v4_pos2 = OpFAdd %v2float %pos_1_3 %perp_1_3
    %v4_x = OpCompositeExtract %float %v4_pos2 0
    %v4_y = OpCompositeExtract %float %v4_pos2 1
  %v4_pos4 = OpCompositeConstruct %v4float %v4_x %v4_y %float_0_0 %float_1_0
               OpStore %out_pos_ptr %v4_pos4
               OpStore %outColor %color_1_3
               OpEmitVertex

               OpEndPrimitive

               ; --- SECOND DASH: t = 2/3 to 1 ---
               ; Position at t=2/3
  %dir_2_3 = OpVectorTimesScalar %v2float %seg_dir %float_2_3
  %pos_2_3 = OpFAdd %v2float %p1_xy %dir_2_3
               ; Position at t=1
     %pos_1 = %p2_xy

               ; Perpendicular at t=2/3 (lerp between perp_s and perp_e)
%perp_2_3_offset = OpVectorTimesScalar %v2float %perp_diff %float_2_3
 %perp_2_3 = OpFAdd %v2float %perp_s_scaled %perp_2_3_offset

               ; Color at t=2/3 and t=1
%color_2_3_offset = OpVectorTimesScalar %v3float %color_diff %float_2_3
%color_2_3 = OpFAdd %v3float %c1 %color_2_3_offset
   %color_1 = %c2

               ; Emit quad for second dash (4 vertices in triangle strip)
               ; Vertex 5: pos_2_3 - perp_2_3
   %v5_pos2 = OpFSub %v2float %pos_2_3 %perp_2_3
    %v5_x = OpCompositeExtract %float %v5_pos2 0
    %v5_y = OpCompositeExtract %float %v5_pos2 1
  %v5_pos4 = OpCompositeConstruct %v4float %v5_x %v5_y %float_0_0 %float_1_0
               OpStore %out_pos_ptr %v5_pos4
               OpStore %outColor %color_2_3
               OpEmitVertex

               ; Vertex 6: pos_2_3 + perp_2_3
   %v6_pos2 = OpFAdd %v2float %pos_2_3 %perp_2_3
    %v6_x = OpCompositeExtract %float %v6_pos2 0
    %v6_y = OpCompositeExtract %float %v6_pos2 1
  %v6_pos4 = OpCompositeConstruct %v4float %v6_x %v6_y %float_0_0 %float_1_0
               OpStore %out_pos_ptr %v6_pos4
               OpStore %outColor %color_2_3
               OpEmitVertex

               ; Vertex 7: pos_1 - perp_e_scaled
   %v7_pos2 = OpFSub %v2float %pos_1 %perp_e_scaled
    %v7_x = OpCompositeExtract %float %v7_pos2 0
    %v7_y = OpCompositeExtract %float %v7_pos2 1
  %v7_pos4 = OpCompositeConstruct %v4float %v7_x %v7_y %float_0_0 %float_1_0
               OpStore %out_pos_ptr %v7_pos4
               OpStore %outColor %color_1
               OpEmitVertex

               ; Vertex 8: pos_1 + perp_e_scaled
   %v8_pos2 = OpFAdd %v2float %pos_1 %perp_e_scaled
    %v8_x = OpCompositeExtract %float %v8_pos2 0
    %v8_y = OpCompositeExtract %float %v8_pos2 1
  %v8_pos4 = OpCompositeConstruct %v4float %v8_x %v8_y %float_0_0 %float_1_0
               OpStore %out_pos_ptr %v8_pos4
               OpStore %outColor %color_1
               OpEmitVertex

               OpEndPrimitive

               OpReturn
               OpFunctionEnd
