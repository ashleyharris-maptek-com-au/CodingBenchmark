OpCapability Geometry
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %color_in %gl_out %outColor
               OpExecutionMode %main InputLinesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 10
               OpExecutionMode %main Invocations 1

               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %color_in Location 2
               OpDecorate %outColor Location 0

       %void = OpTypeVoid
          %3 = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v4float = OpTypeVector %float 4
    %v3float = OpTypeVector %float 3
    %v2float = OpTypeVector %float 2
        %int = OpTypeInt 32 1
       %uint = OpTypeInt 32 0
     %uint_4 = OpConstant %uint 4
      %int_0 = OpConstant %int 0
      %int_1 = OpConstant %int 1
      %int_2 = OpConstant %int 2
      %int_3 = OpConstant %int 3
      %int_4 = OpConstant %int 4
    %float_0 = OpConstant %float 0
  %float_0_2 = OpConstant %float 0.2
  %float_0_4 = OpConstant %float 0.4
  %float_0_6 = OpConstant %float 0.6
  %float_0_8 = OpConstant %float 0.8
    %float_1 = OpConstant %float 1
 %float_0_02 = OpConstant %float 0.02
 %float_0_25 = OpConstant %float 0.25
  %float_0_5 = OpConstant %float 0.5
 %float_0_75 = OpConstant %float 0.75

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
      %gl_in = OpVariable %ptr_in_arr_pv Input
%ptr_in_v4 = OpTypePointer Input %v4float

%arr_color_in = OpTypeArray %v3float %uint_4
%ptr_in_arr_color = OpTypePointer Input %arr_color_in
   %color_in = OpVariable %ptr_in_arr_color Input
%ptr_in_v3 = OpTypePointer Input %v3float

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
     %gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v4 = OpTypePointer Output %v4float

%ptr_out_v3 = OpTypePointer Output %v3float
   %outColor = OpVariable %ptr_out_v3 Output

       %main = OpFunction %void None %3
          %5 = OpLabel

; Load 4 input positions P0, P1, P2, P3
        %p0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
        %P0 = OpLoad %v4float %p0_ptr
        %p1_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
        %P1 = OpLoad %v4float %p1_ptr
        %p2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
        %P2 = OpLoad %v4float %p2_ptr
        %p3_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
        %P3 = OpLoad %v4float %p3_ptr

; Load colors for interpolation
        %c1_ptr = OpAccessChain %ptr_in_v3 %color_in %int_1
        %color1 = OpLoad %v3float %c1_ptr
        %c2_ptr = OpAccessChain %ptr_in_v3 %color_in %int_2
        %color2 = OpLoad %v3float %c2_ptr

; Compute tangent directions for smooth perpendiculars
; tangent_start ~ (P2 - P0), tangent_end ~ (P3 - P1)
        %P2_P0 = OpFSub %v4float %P2 %P0
        %P3_P1 = OpFSub %v4float %P3 %P1

; ===== Point 0: t=0.0 =====
        %pos0 = OpVectorShuffle %v4float %P1 %P1 0 1 2 3
        %tang0 = OpVectorShuffle %v2float %P2_P0 %P2_P0 0 1
        %perp0_raw = OpCompositeConstruct %v2float %float_0 %float_0
        %perp0_x_tmp = OpCompositeExtract %float %tang0 1
        %perp0_x = OpFNegate %float %perp0_x_tmp
        %perp0_y = OpCompositeExtract %float %tang0 0
        %perp0_unnorm = OpCompositeConstruct %v2float %perp0_x %perp0_y
        %perp0_len_sq = OpDot %float %perp0_unnorm %perp0_unnorm
        %perp0_len = OpExtInst %float %1 Sqrt %perp0_len_sq
        %perp0_scale = OpFDiv %float %float_0_02 %perp0_len
        %perp0_scaled = OpVectorTimesScalar %v2float %perp0_unnorm %perp0_scale
        %color0 = OpVectorShuffle %v3float %color1 %color1 0 1 2

        %pos0_xy = OpVectorShuffle %v2float %pos0 %pos0 0 1
        %pos0_plus_xy = OpFAdd %v2float %pos0_xy %perp0_scaled
        %pos0_minus_xy = OpFSub %v2float %pos0_xy %perp0_scaled
        %pos0_plus = OpVectorShuffle %v4float %pos0 %pos0_plus_xy 4 5 2 3
        %pos0_minus = OpVectorShuffle %v4float %pos0 %pos0_minus_xy 4 5 2 3

        %gl_pos_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
        OpStore %gl_pos_ptr %pos0_plus
        OpStore %outColor %color0
        OpEmitVertex
        OpStore %gl_pos_ptr %pos0_minus
        OpStore %outColor %color0
        OpEmitVertex

; ===== Point 1: t=0.25 =====
        %diff_1 = OpFSub %v4float %P2 %P1
        %interp1 = OpVectorTimesScalar %v4float %diff_1 %float_0_25
        %pos1 = OpFAdd %v4float %P1 %interp1
        %tang1_raw = OpVectorShuffle %v2float %P2_P0 %P2_P0 0 1
        %perp1_x_tmp = OpCompositeExtract %float %tang1_raw 1
        %perp1_x = OpFNegate %float %perp1_x_tmp
        %perp1_y = OpCompositeExtract %float %tang1_raw 0
        %perp1_unnorm = OpCompositeConstruct %v2float %perp1_x %perp1_y
        %perp1_len_sq = OpDot %float %perp1_unnorm %perp1_unnorm
        %perp1_len = OpExtInst %float %1 Sqrt %perp1_len_sq
        %perp1_scale = OpFDiv %float %float_0_02 %perp1_len
        %perp1_scaled = OpVectorTimesScalar %v2float %perp1_unnorm %perp1_scale
        %color1_diff = OpFSub %v3float %color2 %color1
        %color1_interp_v = OpVectorTimesScalar %v3float %color1_diff %float_0_25
        %color1_val = OpFAdd %v3float %color1 %color1_interp_v

        %pos1_xy = OpVectorShuffle %v2float %pos1 %pos1 0 1
        %pos1_plus_xy = OpFAdd %v2float %pos1_xy %perp1_scaled
        %pos1_minus_xy = OpFSub %v2float %pos1_xy %perp1_scaled
        %pos1_plus = OpVectorShuffle %v4float %pos1 %pos1_plus_xy 4 5 2 3
        %pos1_minus = OpVectorShuffle %v4float %pos1 %pos1_minus_xy 4 5 2 3

        OpStore %gl_pos_ptr %pos1_plus
        OpStore %outColor %color1_val
        OpEmitVertex
        OpStore %gl_pos_ptr %pos1_minus
        OpStore %outColor %color1_val
        OpEmitVertex

; ===== Point 2: t=0.5 =====
        %interp2 = OpVectorTimesScalar %v4float %diff_1 %float_0_5
        %pos2 = OpFAdd %v4float %P1 %interp2
        %tang2_base = OpFAdd %v2float %tang0 %tang1_raw
        %tang2_avg = OpVectorTimesScalar %v2float %tang2_base %float_0_5
        %perp2_x_tmp = OpCompositeExtract %float %tang2_avg 1
        %perp2_x = OpFNegate %float %perp2_x_tmp
        %perp2_y = OpCompositeExtract %float %tang2_avg 0
        %perp2_unnorm = OpCompositeConstruct %v2float %perp2_x %perp2_y
        %perp2_len_sq = OpDot %float %perp2_unnorm %perp2_unnorm
        %perp2_len = OpExtInst %float %1 Sqrt %perp2_len_sq
        %perp2_scale = OpFDiv %float %float_0_02 %perp2_len
        %perp2_scaled = OpVectorTimesScalar %v2float %perp2_unnorm %perp2_scale
        %color2_interp_v = OpVectorTimesScalar %v3float %color1_diff %float_0_5
        %color2_val = OpFAdd %v3float %color1 %color2_interp_v

        %pos2_xy = OpVectorShuffle %v2float %pos2 %pos2 0 1
        %pos2_plus_xy = OpFAdd %v2float %pos2_xy %perp2_scaled
        %pos2_minus_xy = OpFSub %v2float %pos2_xy %perp2_scaled
        %pos2_plus = OpVectorShuffle %v4float %pos2 %pos2_plus_xy 4 5 2 3
        %pos2_minus = OpVectorShuffle %v4float %pos2 %pos2_minus_xy 4 5 2 3

        OpStore %gl_pos_ptr %pos2_plus
        OpStore %outColor %color2_val
        OpEmitVertex
        OpStore %gl_pos_ptr %pos2_minus
        OpStore %outColor %color2_val
        OpEmitVertex

; ===== Point 3: t=0.75 =====
        %interp3 = OpVectorTimesScalar %v4float %diff_1 %float_0_75
        %pos3 = OpFAdd %v4float %P1 %interp3
        %tang3_raw = OpVectorShuffle %v2float %P3_P1 %P3_P1 0 1
        %perp3_x_tmp = OpCompositeExtract %float %tang3_raw 1
        %perp3_x = OpFNegate %float %perp3_x_tmp
        %perp3_y = OpCompositeExtract %float %tang3_raw 0
        %perp3_unnorm = OpCompositeConstruct %v2float %perp3_x %perp3_y
        %perp3_len_sq = OpDot %float %perp3_unnorm %perp3_unnorm
        %perp3_len = OpExtInst %float %1 Sqrt %perp3_len_sq
        %perp3_scale = OpFDiv %float %float_0_02 %perp3_len
        %perp3_scaled = OpVectorTimesScalar %v2float %perp3_unnorm %perp3_scale
        %color3_interp_v = OpVectorTimesScalar %v3float %color1_diff %float_0_75
        %color3_val = OpFAdd %v3float %color1 %color3_interp_v

        %pos3_xy = OpVectorShuffle %v2float %pos3 %pos3 0 1
        %pos3_plus_xy = OpFAdd %v2float %pos3_xy %perp3_scaled
        %pos3_minus_xy = OpFSub %v2float %pos3_xy %perp3_scaled
        %pos3_plus = OpVectorShuffle %v4float %pos3 %pos3_plus_xy 4 5 2 3
        %pos3_minus = OpVectorShuffle %v4float %pos3 %pos3_minus_xy 4 5 2 3

        OpStore %gl_pos_ptr %pos3_plus
        OpStore %outColor %color3_val
        OpEmitVertex
        OpStore %gl_pos_ptr %pos3_minus
        OpStore %outColor %color3_val
        OpEmitVertex

; ===== Point 4: t=1.0 =====
        %pos4 = OpVectorShuffle %v4float %P2 %P2 0 1 2 3
        %perp4_x_tmp = OpCompositeExtract %float %tang3_raw 1
        %perp4_x = OpFNegate %float %perp4_x_tmp
        %perp4_y = OpCompositeExtract %float %tang3_raw 0
        %perp4_unnorm = OpCompositeConstruct %v2float %perp4_x %perp4_y
        %perp4_len_sq = OpDot %float %perp4_unnorm %perp4_unnorm
        %perp4_len = OpExtInst %float %1 Sqrt %perp4_len_sq
        %perp4_scale = OpFDiv %float %float_0_02 %perp4_len
        %perp4_scaled = OpVectorTimesScalar %v2float %perp4_unnorm %perp4_scale
        %color4 = OpVectorShuffle %v3float %color2 %color2 0 1 2

        %pos4_xy = OpVectorShuffle %v2float %pos4 %pos4 0 1
        %pos4_plus_xy = OpFAdd %v2float %pos4_xy %perp4_scaled
        %pos4_minus_xy = OpFSub %v2float %pos4_xy %perp4_scaled
        %pos4_plus = OpVectorShuffle %v4float %pos4 %pos4_plus_xy 4 5 2 3
        %pos4_minus = OpVectorShuffle %v4float %pos4 %pos4_minus_xy 4 5 2 3

        OpStore %gl_pos_ptr %pos4_plus
        OpStore %outColor %color4
        OpEmitVertex
        OpStore %gl_pos_ptr %pos4_minus
        OpStore %outColor %color4
        OpEmitVertex

        OpEndPrimitive
        OpReturn
        OpFunctionEnd
