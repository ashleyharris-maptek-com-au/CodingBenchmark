const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { width, height, grid, ignitions, towns, cutBudget } = data;

const dirs = [[0, -1], [0, 1], [-1, 0], [1, 0]];

function simulateFire(cutsSet) {
    const burned = new Set();
    const queue = [];

    // Initialize with ignition points
    for (const [x, y] of ignitions) {
        const key = `${x},${y}`;
        if (x >= 0 && x < width && y >= 0 && y < height &&
            grid[y][x] !== '~' && !cutsSet.has(key)) {
            burned.add(key);
            queue.push([x, y]);
        }
    }

    // BFS to spread fire
    let head = 0;
    while (head < queue.length) {
        const [x, y] = queue[head++];

        for (const [dx, dy] of dirs) {
            const nx = x + dx;
            const ny = y + dy;

            if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;

            const key = `${nx},${ny}`;
            if (burned.has(key)) continue;
            if (grid[ny][nx] === '~') continue;
            if (cutsSet.has(key)) continue;

            burned.add(key);
            queue.push([nx, ny]);
        }
    }

    return burned;
}

function getSavedValue(burned) {
    let saved = 0;
    for (const town of towns) {
        if (!burned.has(`${town.x},${town.y}`)) {
            saved += town.value;
        }
    }
    return saved;
}

function getCandidates(burned, cutsSet) {
    const candidates = new Set();

    // Find cells adjacent to fire that could be cut
    for (const key of burned) {
        const [x, y] = key.split(',').map(Number);

        for (const [dx, dy] of dirs) {
            const nx = x + dx;
            const ny = y + dy;

            if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;

            const nkey = `${nx},${ny}`;
            if (grid[ny][nx] === '.' && !cutsSet.has(nkey) && !burned.has(nkey)) {
                candidates.add(nkey);
            }
        }
    }

    return Array.from(candidates).map(s => s.split(',').map(Number));
}

// Greedy algorithm: iteratively add best cut
let cuts = [];
let cutsSet = new Set();

for (let i = 0; i < cutBudget; i++) {
    const burned = simulateFire(cutsSet);
    const currentValue = getSavedValue(burned);

    // Check if all towns saved
    const totalValue = towns.reduce((sum, t) => sum + t.value, 0);
    if (currentValue === totalValue) break;

    // Get candidate cells for cutting
    const candidates = getCandidates(burned, cutsSet);
    if (candidates.length === 0) break;

    // Find best cut
    let bestCut = null;
    let bestValue = currentValue;

    for (const [x, y] of candidates) {
        const testCutsSet = new Set(cutsSet);
        testCutsSet.add(`${x},${y}`);

        const testBurned = simulateFire(testCutsSet);
        const testValue = getSavedValue(testBurned);

        if (testValue > bestValue) {
            bestValue = testValue;
            bestCut = [x, y];
        }
    }

    if (bestCut) {
        cuts.push(bestCut);
        cutsSet.add(`${bestCut[0]},${bestCut[1]}`);
    } else {
        break;
    }
}

console.log(JSON.stringify({ firebreaks: cuts }));
