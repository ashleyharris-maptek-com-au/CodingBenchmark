declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { width, height, grid, ignitions, towns, cutBudget } = data;

// Simulate fire spread using BFS and return set of burned cells
function simulateFire(cuts: Set<string>): Set<string> {
  const burned = new Set<string>();
  const queue: [number, number][] = [];

  // Start fire at all ignition points
  for (const [x, y] of ignitions) {
    const key = `${x},${y}`;
    burned.add(key);
    queue.push([x, y]);
  }

  // BFS spread
  let head = 0;
  while (head < queue.length) {
    const [x, y] = queue[head++];

    // Check 4 cardinal directions
    const neighbors = [[x + 1, y], [x - 1, y], [x, y + 1], [x, y - 1]];

    for (const [nx, ny] of neighbors) {
      // Bounds check
      if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;

      const key = `${nx},${ny}`;

      // Already burned
      if (burned.has(key)) continue;

      // Check if burnable (not water/rock)
      if (grid[ny][nx] !== '.') continue;

      // Check if cut
      if (cuts.has(key)) continue;

      // Spread fire to this cell
      burned.add(key);
      queue.push([nx, ny]);
    }
  }

  return burned;
}

// Calculate total value of towns saved (not burned)
function calculateSavedValue(burned: Set<string>): number {
  let saved = 0;
  for (const town of towns) {
    if (!burned.has(`${town.x},${town.y}`)) {
      saved += town.value;
    }
  }
  return saved;
}

// Find cells that would burn without cuts (candidate locations for firebreaks)
const initialBurn = simulateFire(new Set());
const candidates: string[] = Array.from(initialBurn);

// Greedy algorithm: iteratively select the cut that provides maximum value improvement
const cuts = new Set<string>();
let currentValue = calculateSavedValue(initialBurn);

for (let i = 0; i < cutBudget; i++) {
  let bestCut: string | null = null;
  let bestValue = currentValue;

  // Try each candidate cell
  for (const candidate of candidates) {
    if (cuts.has(candidate)) continue;

    // Try adding this cut
    cuts.add(candidate);
    const newBurned = simulateFire(cuts);
    const newValue = calculateSavedValue(newBurned);
    cuts.delete(candidate);

    // Track best improvement
    if (newValue > bestValue) {
      bestValue = newValue;
      bestCut = candidate;
    }
  }

  // No improvement possible, stop early
  if (bestCut === null || bestValue === currentValue) break;

  // Apply best cut
  cuts.add(bestCut);
  currentValue = bestValue;
}

// Convert to output format
const firebreaks: [number, number][] = [];
for (const cut of cuts) {
  const [x, y] = cut.split(',').map(Number);
  firebreaks.push([x, y]);
}
