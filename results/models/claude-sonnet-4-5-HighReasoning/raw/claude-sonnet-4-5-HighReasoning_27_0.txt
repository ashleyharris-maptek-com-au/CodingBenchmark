#include <iostream>
#include <vector>
#include <algorithm>
#include <set>

using namespace std;

class GraphColoring {
private:
    int n, m, k;
    vector<vector<int>> adj;
    vector<int> colors;

public:
    GraphColoring(int vertices, int edges, int num_colors)
        : n(vertices), m(edges), k(num_colors) {
        adj.resize(n);
        colors.resize(n, -1);
    }

    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    // Greedy coloring with largest-degree-first ordering
    // Time: O(n log n + m) - very efficient for large graphs
    bool greedyLDF() {
        // Create vertex ordering by degree (highest first)
        vector<pair<int, int>> vertices;
        vertices.reserve(n);
        for (int i = 0; i < n; i++) {
            vertices.push_back({static_cast<int>(adj[i].size()), i});
        }
        sort(vertices.rbegin(), vertices.rend());

        // Color vertices in degree order
        for (const auto& p : vertices) {
            int v = p.second;
            vector<bool> used_colors(k, false);

            // Mark colors used by neighbors
            for (int u : adj[v]) {
                if (colors[u] != -1) {
                    used_colors[colors[u]] = true;
                }
            }

            // Find smallest available color
            int color = 0;
            while (color < k && used_colors[color]) {
                color++;
            }

            if (color >= k) {
                return false;
            }

            colors[v] = color;
        }

        return true;
    }

    // DSATUR (Degree of Saturation) heuristic
    // Time: O(n² + m) - better quality but slower
    // One of the best heuristics for graph coloring
    bool dsatur() {
        vector<bool> colored(n, false);
        vector<set<int>> neighbor_colors(n);

        // Start with highest degree vertex
        int first = 0;
        for (int i = 1; i < n; i++) {
            if (adj[i].size() > adj[first].size()) {
                first = i;
            }
        }

        // Color first vertex
        colors[first] = 0;
        colored[first] = true;
        for (int v : adj[first]) {
            neighbor_colors[v].insert(0);
        }

        // Color remaining vertices
        for (int iter = 1; iter < n; iter++) {
            // Find uncolored vertex with maximum saturation degree
            // (number of different colors used by neighbors)
            // Break ties by maximum degree
            int best = -1;
            int max_sat = -1;
            int max_deg = -1;

            for (int i = 0; i < n; i++) {
                if (!colored[i]) {
                    int sat = static_cast<int>(neighbor_colors[i].size());
                    int deg = static_cast<int>(adj[i].size());

                    if (best == -1 || sat > max_sat ||
                        (sat == max_sat && deg > max_deg)) {
                        best = i;
                        max_sat = sat;
                        max_deg = deg;
                    }
                }
            }

            if (best == -1) break;

            // Find smallest available color
            int color = 0;
            while (color < k && neighbor_colors[best].count(color)) {
                color++;
            }

            if (color >= k) {
                return false;
            }

            colors[best] = color;
            colored[best] = true;

            // Update saturation of neighbors
            for (int v : adj[best]) {
                if (!colored[v]) {
                    neighbor_colors[v].insert(color);
                }
            }
        }

        return true;
    }

    bool solve() {
        // Adaptive strategy based on graph size
        // DSATUR for smaller graphs (better quality)
        // Greedy LDF for larger graphs (faster)

        if (n <= 10000) {
            // Try DSATUR first for better coloring quality
            if (dsatur()) {
                return true;
            }
            // If DSATUR fails, reset and try greedy
            fill(colors.begin(), colors.end(), -1);
        }

        // Use greedy LDF for large graphs or as fallback
        return greedyLDF();
    }

    void printSolution() {
        for (int i = 0; i < n; i++) {
            if (i > 0) cout << " ";
            cout << colors[i];
        }
        cout << endl;
    }
};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m, k;
    cin >> n >> m >> k;

    GraphColoring gc(n, m, k);

    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        gc.addEdge(u, v);
    }

    gc.solve();
    gc.printSolution();

    return 0;
}
