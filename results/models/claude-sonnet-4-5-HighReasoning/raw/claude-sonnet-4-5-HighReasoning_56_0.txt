const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { depot, truckCount, packages } = data;

function distance(p1, p2) {
  const dx = p1.x - p2.x;
  const dy = p1.y - p2.y;
  return Math.sqrt(dx * dx + dy * dy);
}

function calculateRouteDistance(route, depot, packageMap) {
  if (route.length === 0) return 0;
  let dist = 0;
  let prev = depot;
  for (const pkgId of route) {
    const pkg = packageMap.get(pkgId);
    dist += distance(prev, pkg);
    prev = pkg;
  }
  dist += distance(prev, depot);
  return dist;
}

// Clarke-Wright Savings Algorithm for initial route construction
function clarkWrightSavings(packages, depot, truckCount) {
  const packageMap = new Map(packages.map(p => [p.id, p]));

  // Calculate savings for all package pairs
  const savings = [];
  for (let i = 0; i < packages.length; i++) {
    for (let j = i + 1; j < packages.length; j++) {
      const pi = packages[i];
      const pj = packages[j];
      const saving = distance(depot, pi) + distance(depot, pj) - distance(pi, pj);
      savings.push({ i: pi.id, j: pj.id, saving });
    }
  }

  // Sort by savings (descending)
  savings.sort((a, b) => b.saving - a.saving);

  // Initialize routes - each package starts in its own route
  const routes = new Map();
  const packageToRoute = new Map();

  packages.forEach(pkg => {
    routes.set(pkg.id, [pkg.id]);
    packageToRoute.set(pkg.id, pkg.id);
  });

  // Merge routes based on savings
  for (const { i, j, saving } of savings) {
    if (saving <= 0) break;

    const routeI = packageToRoute.get(i);
    const routeJ = packageToRoute.get(j);

    if (routeI === routeJ) continue; // Already in same route
    if (routes.size <= truckCount && routes.size <= 1) continue; // Keep minimum routes

    const route1 = routes.get(routeI);
    const route2 = routes.get(routeJ);

    // Check if packages are at ends of their respective routes
    const iAtEnd = route1[0] === i || route1[route1.length - 1] === i;
    const jAtEnd = route2[0] === j || route2[route2.length - 1] === j;

    if (!iAtEnd || !jAtEnd) continue;

    // Merge routes
    let mergedRoute;
    if (route1[route1.length - 1] === i && route2[0] === j) {
      mergedRoute = [...route1, ...route2];
    } else if (route1[0] === i && route2[route2.length - 1] === j) {
      mergedRoute = [...route2, ...route1];
    } else if (route1[route1.length - 1] === i && route2[route2.length - 1] === j) {
      mergedRoute = [...route1, ...route2.reverse()];
    } else if (route1[0] === i && route2[0] === j) {
      mergedRoute = [...route1.reverse(), ...route2];
    } else {
      continue;
    }

    // Update routes
    routes.delete(routeI);
    routes.delete(routeJ);
    routes.set(mergedRoute[0], mergedRoute);

    mergedRoute.forEach(pkgId => packageToRoute.set(pkgId, mergedRoute[0]));

    if (routes.size <= truckCount) break;
  }

  return Array.from(routes.values());
}

// 2-opt improvement for TSP within each route
function improve2Opt(route, depot, packageMap) {
  if (route.length < 2) return route;

  let improved = true;
  let currentRoute = [...route];

  while (improved) {
    improved = false;

    for (let i = 0; i < currentRoute.length - 1; i++) {
      for (let j = i + 2; j < currentRoute.length; j++) {
        // Reverse segment between i and j
        const p1 = i === 0 ? depot : packageMap.get(currentRoute[i - 1]);
        const p2 = packageMap.get(currentRoute[i]);
        const p3 = packageMap.get(currentRoute[j]);
        const p4 = j === currentRoute.length - 1 ? depot : packageMap.get(currentRoute[j + 1]);

        const currentDist = distance(p1, p2) + distance(p3, p4);
        const newDist = distance(p1, p3) + distance(p2, p4);

        if (newDist < currentDist - 1e-10) {
          // Perform 2-opt swap
          const newRoute = [
            ...currentRoute.slice(0, i),
            ...currentRoute.slice(i, j + 1).reverse(),
            ...currentRoute.slice(j + 1)
          ];
          currentRoute = newRoute;
          improved = true;
          break;
        }
      }
      if (improved) break;
    }
  }

  return currentRoute;
}

// Build initial solution using Clarke-Wright
let routes = clarkWrightSavings(packages, depot, truckCount);

// If we have fewer routes than trucks, we might want to split large routes
if (routes.length < truckCount && routes.length > 0) {
  // Split the longest route if beneficial
  while (routes.length < truckCount) {
    let longestIdx = 0;
    let longestLen = routes[0].length;

    for (let i = 1; i < routes.length; i++) {
      if (routes[i].length > longestLen) {
        longestLen = routes[i].length;
        longestIdx = i;
      }
    }

    if (longestLen < 2) break;

    const route = routes[longestIdx];
    const mid = Math.floor(route.length / 2);
    const route1 = route.slice(0, mid);
    const route2 = route.slice(mid);

    routes.splice(longestIdx, 1, route1, route2);
  }
}

// Improve each route with 2-opt
const packageMap = new Map(packages.map(p => [p.id, p]));
routes = routes.map(route => improve2Opt(route, depot, packageMap));

// Remove empty routes
routes = routes.filter(r => r.length > 0);
