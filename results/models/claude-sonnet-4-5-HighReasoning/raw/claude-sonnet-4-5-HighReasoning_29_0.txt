using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading;
using System.Diagnostics;

class MaximumClique
{
    static int numVertices;
    static HashSet<int>[] adj;
    static int[] degree;
    static int[] bestClique;
    static int bestSize;
    static object lockObj = new object();
    static Stopwatch timer;
    static long timeLimit = 270000; // 4.5 minutes
    static volatile bool timeExpired = false;

    static void Main()
    {
        timer = Stopwatch.StartNew();
        ReadInput();
        FindMaximumClique();
        OutputResult();
    }

    static void ReadInput()
    {
        string[] firstLine = Console.ReadLine().Split();
        numVertices = int.Parse(firstLine[0]);
        int numEdges = int.Parse(firstLine[1]);

        adj = new HashSet<int>[numVertices];
        degree = new int[numVertices];

        for (int i = 0; i < numVertices; i++)
        {
            adj[i] = new HashSet<int>();
        }

        for (int i = 0; i < numEdges; i++)
        {
            string[] edge = Console.ReadLine().Split();
            int u = int.Parse(edge[0]);
            int v = int.Parse(edge[1]);
            adj[u].Add(v);
            adj[v].Add(u);
        }

        for (int i = 0; i < numVertices; i++)
        {
            degree[i] = adj[i].Count;
        }
    }

    static void FindMaximumClique()
    {
        // Always start with greedy to get a good baseline
        bestClique = GreedyClique();
        bestSize = bestClique.Length;

        // For smaller graphs, try exact algorithm
        if (numVertices <= 5000 && timer.ElapsedMilliseconds < timeLimit - 10000)
        {
            // Only consider vertices that could be in a larger clique
            var vertices = Enumerable.Range(0, numVertices)
                .Where(v => degree[v] >= bestSize - 1)
                .OrderByDescending(v => degree[v])
                .ToList();

            if (vertices.Count > 0)
            {
                int numThreads = Math.Min(Environment.ProcessorCount, Math.Min(vertices.Count, 12));
                var threads = new List<Thread>();

                for (int i = 0; i < numThreads && i < vertices.Count; i++)
                {
                    int idx = i;
                    var thread = new Thread(() => SearchFromVertex(vertices[idx]));
                    thread.Start();
                    threads.Add(thread);
                }

                foreach (var thread in threads)
                {
                    thread.Join();
                }
            }
        }
    }

    static void SearchFromVertex(int startVertex)
    {
        var current = new List<int> { startVertex };
        var candidates = adj[startVertex]
            .Where(v => degree[v] >= bestSize - 1)
            .ToList();
        BranchAndBound(current, candidates);
    }

    static void BranchAndBound(List<int> current, List<int> candidates)
    {
        if (timeExpired || timer.ElapsedMilliseconds > timeLimit)
        {
            timeExpired = true;
            return;
        }

        if (candidates.Count == 0)
        {
            lock(lockObj)
            {
                if (current.Count > bestSize)
                {
                    bestSize = current.Count;
                    bestClique = current.ToArray();
                }
            }
            return;
        }

        // Pruning: can we beat current best?
        int currentBest = bestSize;
        if (current.Count + candidates.Count <= currentBest)
        {
            return;
        }

        // Order candidates by degree in subgraph (descending)
        var ordered = candidates
            .Select(v => new {
                Vertex = v,
                SubDegree = candidates.Count(u => adj[v].Contains(u))
            })
            .OrderByDescending(x => x.SubDegree)
            .Select(x => x.Vertex)
            .ToList();

        for (int i = 0; i < ordered.Count; i++)
        {
            if (timeExpired) return;

            // Pruning check
            if (current.Count + (ordered.Count - i) <= bestSize)
            {
                break;
            }

            int v = ordered[i];
            current.Add(v);

            // Find candidates adjacent to v
            var newCandidates = new List<int>();
            for (int j = i + 1; j < ordered.Count; j++)
            {
                if (adj[v].Contains(ordered[j]))
                {
                    newCandidates.Add(ordered[j]);
                }
            }

            BranchAndBound(current, newCandidates);
            current.RemoveAt(current.Count - 1);
        }
    }

    static int[] GreedyClique()
    {
        if (numVertices == 0) return new int[0];

        var clique = new List<int>();

        // Start with highest degree vertex
        int start = 0;
        for (int i = 1; i < numVertices; i++)
        {
            if (degree[i] > degree[start])
            {
                start = i;
            }
        }

        clique.Add(start);
        var candidates = new List<int>(adj[start]);

        while (candidates.Count > 0)
        {
            // Pick vertex with most connections to candidates (highest degree in subgraph)
            int best = candidates[0];
            int bestDegree = 0;

            foreach (var v in candidates)
            {
                int deg = 0;
                foreach (var u in candidates)
                {
                    if (adj[v].Contains(u))
                    {
                        deg++;
                    }
                }
                if (deg > bestDegree)
                {
                    bestDegree = deg;
                    best = v;
                }
            }

            clique.Add(best);

            // Keep only vertices connected to best
            var newCandidates = new List<int>();
            foreach (var v in candidates)
            {
                if (v != best && adj[best].Contains(v))
                {
                    newCandidates.Add(v);
                }
            }
            candidates = newCandidates;
        }

        return clique.ToArray();
    }

    static void OutputResult()
    {
        Console.WriteLine(bestSize);
        Console.WriteLine(string.Join(" ", bestClique.OrderBy(x => x)));
    }
}
