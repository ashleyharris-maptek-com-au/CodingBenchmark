using System;
using System.Collections.Generic;

class Program
{
    static void Main()
    {
        // Read input
        var line1 = Console.ReadLine().Split();
        int n = int.Parse(line1[0]);
        int stockLength = int.Parse(line1[1]);

        var cuts = new (int length, int index)[n];
        if (n > 0)
        {
            var line2 = Console.ReadLine().Split();
            for (int i = 0; i < n; i++)
            {
                cuts[i] = (int.Parse(line2[i]), i);
            }
        }

        // Sort by length descending (First Fit Decreasing)
        Array.Sort(cuts, (a, b) => b.length.CompareTo(a.length));

        // Bin packing using First Fit Decreasing algorithm
        var bins = new List<List<int>>();
        var binCaps = new List<int>();

        foreach (var (length, index) in cuts)
        {
            // Find first bin that fits
            int binIdx = -1;
            for (int i = 0; i < bins.Count; i++)
            {
                if (binCaps[i] >= length)
                {
                    binIdx = i;
                    break;
                }
            }

            if (binIdx == -1)
            {
                // Create new bin
                bins.Add(new List<int>());
                binCaps.Add(stockLength);
                binIdx = bins.Count - 1;
            }

            bins[binIdx].Add(index);
            binCaps[binIdx] -= length;
        }

        // Output
        Console.WriteLine(bins.Count);
        foreach (var bin in bins)
        {
            Console.WriteLine(string.Join(" ", bin));
        }
    }
}
