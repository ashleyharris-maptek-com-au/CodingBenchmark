OpCapability Geometry
               OpCapability Shader
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %color_in %gl_out %outColor
               OpExecutionMode %main InputPoints
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 4
               OpExecutionMode %main Invocations 1

               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %color_in Location 2
               OpDecorate %outColor Location 0

       %void = OpTypeVoid
          %3 = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v4float = OpTypeVector %float 4
    %v3float = OpTypeVector %float 3
        %int = OpTypeInt 32 1
       %uint = OpTypeInt 32 0
      %int_0 = OpConstant %int 0
      %uint_1 = OpConstant %uint 1

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_1
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
      %gl_in = OpVariable %ptr_in_arr_pv Input
%ptr_in_v4float = OpTypePointer Input %v4float

%arr_color_in = OpTypeArray %v3float %uint_1
%ptr_in_arr_color = OpTypePointer Input %arr_color_in
   %color_in = OpVariable %ptr_in_arr_color Input
%ptr_in_v3float = OpTypePointer Input %v3float

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
     %gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v4float = OpTypePointer Output %v4float

%ptr_out_v3float = OpTypePointer Output %v3float
   %outColor = OpVariable %ptr_out_v3float Output

   %float_h = OpConstant %float 0.05
  %float_1 = OpConstant %float 1.0

       %main = OpFunction %void None %3
          %5 = OpLabel

               ; Load input position gl_in[0].gl_Position
         %20 = OpAccessChain %ptr_in_v4float %gl_in %int_0 %int_0
         %21 = OpLoad %v4float %20

               ; Load input color color_in[0]
         %25 = OpAccessChain %ptr_in_v3float %color_in %int_0
         %26 = OpLoad %v3float %25

               ; Extract x, y, z, w from position
         %30 = OpCompositeExtract %float %21 0
         %31 = OpCompositeExtract %float %21 1
         %32 = OpCompositeExtract %float %21 2
         %33 = OpCompositeExtract %float %21 3

               ; Vertex 1: Top (x, y+h, z, w)
         %40 = OpFAdd %float %31 %float_h
         %41 = OpCompositeConstruct %v4float %30 %40 %32 %33
         %42 = OpAccessChain %ptr_out_v4float %gl_out %int_0
               OpStore %42 %41
               OpStore %outColor %26
               OpEmitVertex

               ; Vertex 2: Right (x+h, y, z, w)
         %50 = OpFAdd %float %30 %float_h
         %51 = OpCompositeConstruct %v4float %50 %31 %32 %33
         %52 = OpAccessChain %ptr_out_v4float %gl_out %int_0
               OpStore %52 %51
               OpStore %outColor %26
               OpEmitVertex

               ; Vertex 3: Left (x-h, y, z, w)
         %60 = OpFSub %float %30 %float_h
         %61 = OpCompositeConstruct %v4float %60 %31 %32 %33
         %62 = OpAccessChain %ptr_out_v4float %gl_out %int_0
               OpStore %62 %61
               OpStore %outColor %26
               OpEmitVertex

               ; Vertex 4: Bottom (x, y-h, z, w)
         %70 = OpFSub %float %31 %float_h
         %71 = OpCompositeConstruct %v4float %30 %70 %32 %33
         %72 = OpAccessChain %ptr_out_v4float %gl_out %int_0
               OpStore %72 %71
               OpStore %outColor %26
               OpEmitVertex

               OpEndPrimitive
               OpReturn
               OpFunctionEnd
