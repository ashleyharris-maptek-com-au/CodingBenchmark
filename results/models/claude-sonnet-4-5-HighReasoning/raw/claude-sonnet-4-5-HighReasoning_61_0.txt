declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const depot = data.depot;
const truckCount = data.truckCount;
const packages = data.packages;

function dist(x1: number, y1: number, x2: number, y2: number): number {
  const dx = x1 - x2;
  const dy = y1 - y2;
  return Math.sqrt(dx * dx + dy * dy);
}

// Build package coordinate lookup
const pkgMap = new Map<number, [number, number]>();
for (const pkg of packages) {
  pkgMap.set(pkg.id, [pkg.x, pkg.y]);
}

function getCoords(id: number | null): [number, number] {
  return id === null ? depot : pkgMap.get(id)!;
}

// 2-opt improvement
function twoOpt(route: number[]): number[] {
  if (route.length <= 1) return route;

  let best = [...route];
  let improved = true;

  while (improved) {
    improved = false;

    for (let i = 0; i < best.length - 1; i++) {
      for (let k = i + 1; k < best.length; k++) {
        const prevI = i === 0 ? null : best[i - 1];
        const nextK = k === best.length - 1 ? null : best[k + 1];

        const [px, py] = getCoords(prevI);
        const [ix, iy] = getCoords(best[i]);
        const [kx, ky] = getCoords(best[k]);
        const [nx, ny] = getCoords(nextK);

        const oldDist = dist(px, py, ix, iy) + dist(kx, ky, nx, ny);
        const newDist = dist(px, py, kx, ky) + dist(ix, iy, nx, ny);

        if (newDist < oldDist - 1e-9) {
          const newRoute = [...best];
          let left = i;
          let right = k;
          while (left < right) {
            [newRoute[left], newRoute[right]] = [newRoute[right], newRoute[left]];
            left++;
            right--;
          }
          best = newRoute;
          improved = true;
          break;
        }
      }
      if (improved) break;
    }
  }

  return best;
}

// Sweep algorithm: partition by polar angle
const sorted = packages.map(p => ({
  id: p.id,
  angle: Math.atan2(p.y - depot[1], p.x - depot[0])
})).sort((a, b) => a.angle - b.angle);

const routes: number[][] = [];
const perRoute = Math.ceil(packages.length / truckCount);

for (let i = 0; i < truckCount; i++) {
  const start = i * perRoute;
  const end = Math.min(start + perRoute, packages.length);
  if (start < packages.length) {
    routes.push(sorted.slice(start, end).map(p => p.id));
  }
}

// Optimize each route
const optimized = routes.map(twoOpt);

console.log(JSON.stringify({ routes: optimized }));
