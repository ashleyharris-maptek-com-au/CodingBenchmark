declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

interface Task {
  id: string;
  duration: number;
  deps: string[];
  deadline: number;
  weight: number;
}

interface ScheduleEntry {
  id: string;
  start: number;
  machine: number;
}

const machineCount: number = data.machineCount;
const tasks: Task[] = data.tasks;

// Build task map
const taskMap = new Map<string, Task>();
for (const task of tasks) {
  taskMap.set(task.id, task);
}

// Build reverse dependency graph (successors)
const successors = new Map<string, Set<string>>();
for (const task of tasks) {
  if (!successors.has(task.id)) {
    successors.set(task.id, new Set());
  }
  for (const dep of task.deps) {
    if (!successors.has(dep)) {
      successors.set(dep, new Set());
    }
    successors.get(dep)!.add(task.id);
  }
}

// Compute critical path (longest path from this task forward)
const criticalPath = new Map<string, number>();

function computeCriticalPath(taskId: string): number {
  if (criticalPath.has(taskId)) {
    return criticalPath.get(taskId)!;
  }

  const task = taskMap.get(taskId)!;
  let maxPath = task.duration;

  const succ = successors.get(taskId);
  if (succ && succ.size > 0) {
    let maxSuccPath = 0;
    for (const succId of succ) {
      maxSuccPath = Math.max(maxSuccPath, computeCriticalPath(succId));
    }
    maxPath += maxSuccPath;
  }

  criticalPath.set(taskId, maxPath);
  return maxPath;
}

for (const task of tasks) {
  computeCriticalPath(task.id);
}

// Initialize scheduling data structures
const schedule: ScheduleEntry[] = [];
const finishTimes = new Map<string, number>();
const machineAvailable = new Array(machineCount).fill(0);

const inDegree = new Map<string, number>();
for (const task of tasks) {
  inDegree.set(task.id, task.deps.length);
}

// Ready queue
const ready: Task[] = [];
for (const task of tasks) {
  if (task.deps.length === 0) {
    ready.push(task);
  }
}

// Priority function - higher value = higher priority
function getPriority(task: Task, currentTime: number): number {
  const cp = criticalPath.get(task.id) || 0;
  const estimatedFinish = currentTime + cp;
  const slack = task.deadline - estimatedFinish;
  const weightRatio = task.weight / Math.max(1, task.duration);

  let priority = 0;

  // Slack urgency - tasks that will miss deadline get highest priority
  if (slack < 0) {
    priority += -slack * 10000;
  } else {
    priority += -slack * 100;
  }

  // Weight-to-duration ratio
  priority += weightRatio * 1000;

  // Critical path length
  priority += cp;

  return priority;
}

// Main scheduling loop
while (ready.length > 0) {
  // Find current minimum time across all machines
  let minTime = 0;
  for (let m = 0; m < machineCount; m++) {
    if (m === 0 || machineAvailable[m] < minTime) {
      minTime = machineAvailable[m];
    }
  }

  // Sort ready tasks by priority (highest first)
  ready.sort((a, b) => getPriority(b, minTime) - getPriority(a, minTime));

  // Take highest priority task
  const task = ready.shift()!;

  // Find earliest start time based on dependencies
  let earliestStart = 0;
  for (const depId of task.deps) {
    const depFinish = finishTimes.get(depId);
    if (depFinish !== undefined) {
      earliestStart = Math.max(earliestStart, depFinish);
    }
  }

  // Find best machine (earliest available at or after earliestStart)
  let bestMachine = 0;
  let bestStart = Math.max(machineAvailable[0], earliestStart);
  for (let m = 1; m < machineCount; m++) {
    const start = Math.max(machineAvailable[m], earliestStart);
    if (start < bestStart) {
      bestStart = start;
      bestMachine = m;
    }
  }

  // Schedule the task
  schedule.push({
    id: task.id,
    start: bestStart,
    machine: bestMachine
  });

  const finishTime = bestStart + task.duration;
  finishTimes.set(task.id, finishTime);
  machineAvailable[bestMachine] = finishTime;

  // Update successors - add newly ready tasks
  const succ = successors.get(task.id);
  if (succ) {
    for (const succId of succ) {
      const newDegree = inDegree.get(succId)! - 1;
      inDegree.set(succId, newDegree);
      if (newDegree === 0) {
        ready.push(taskMap.get(succId)!);
      }
    }
  }
}

console.log(JSON.stringify({ schedule }));
