use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Read first line: num_vertices num_edges
    let first_line = lines.next().unwrap().unwrap();
    let parts: Vec<usize> = first_line.split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();
    let n = parts[0];
    let m = parts[1];

    // Build adjacency list
    let mut adj: Vec<Vec<usize>> = vec![Vec::new(); n];
    for _ in 0..m {
        let line = lines.next().unwrap().unwrap();
        let edge: Vec<usize> = line.split_whitespace()
            .map(|s| s.parse().unwrap())
            .collect();
        let u = edge[0];
        let v = edge[1];
        adj[u].push(v);
        adj[v].push(u);
    }

    // Find minimum dominating set using greedy algorithm
    let solution = greedy_dominating_set(n, &adj);

    // Output
    println!("{}", solution.len());
    for (i, &v) in solution.iter().enumerate() {
        if i > 0 {
            print!(" ");
        }
        print!("{}", v);
    }
    println!();
}

fn greedy_dominating_set(n: usize, adj: &[Vec<usize>]) -> Vec<usize> {
    let mut dominating_set = Vec::new();
    let mut in_dominating_set = vec![false; n];
    let mut covered = vec![false; n];
    let mut num_covered = 0;

    while num_covered < n {
        // Find the vertex that covers the most uncovered vertices
        let mut best_vertex = None;
        let mut best_coverage_count = 0;
        let mut best_degree = 0;

        for v in 0..n {
            if in_dominating_set[v] {
                continue;
            }

            let mut coverage_count = 0;

            // v itself
            if !covered[v] {
                coverage_count += 1;
            }

            // Neighbors of v
            for &neighbor in &adj[v] {
                if !covered[neighbor] {
                    coverage_count += 1;
                }
            }

            let degree = adj[v].len();

            // Choose vertex with best coverage, breaking ties by degree
            if coverage_count > best_coverage_count ||
               (coverage_count == best_coverage_count && degree > best_degree) {
                best_coverage_count = coverage_count;
                best_degree = degree;
                best_vertex = Some(v);
            }
        }

        if let Some(v) = best_vertex {
            dominating_set.push(v);
            in_dominating_set[v] = true;

            // Update covered vertices
            if !covered[v] {
                covered[v] = true;
                num_covered += 1;
            }

            for &neighbor in &adj[v] {
                if !covered[neighbor] {
                    covered[neighbor] = true;
                    num_covered += 1;
                }
            }
        } else {
            // Fallback: add any uncovered vertex
            for v in 0..n {
                if !covered[v] {
                    dominating_set.push(v);
                    in_dominating_set[v] = true;
                    covered[v] = true;
                    num_covered += 1;

                    for &neighbor in &adj[v] {
                        if !covered[neighbor] {
                            covered[neighbor] = true;
                            num_covered += 1;
                        }
                    }
                    break;
                }
            }
        }
    }

    dominating_set
}
