#version 450

layout(local_size_x = 256) in;

// Each body is stored as two consecutive vec4s:
//   vec4(position.xyz, mass)
//   vec4(velocity.xyz, 0.0)
// So body i starts at index i*2 in the buffer.

layout(set = 0, binding = 0) readonly buffer InputBodies {
    vec4 inData[];   // length = N * 2
};

layout(set = 0, binding = 1) buffer OutputBodies {
    vec4 outData[];  // length = N * 2
};

layout(set = 0, binding = 2) uniform Params {
    uvec4 counts;    // counts.x = N (number of bodies)
    vec4 physics;    // physics.x = dt, physics.y = softening, physics.z = G
};

void main() {
    uint i = gl_GlobalInvocationID.x;
    uint N = counts.x;

    // Guard against out of bounds
    if (i >= N) {
        return;
    }

    // Extract parameters
    float dt = physics.x;
    float softening = physics.y;
    float G = physics.z;

    // Read this body's data
    uint idx = i * 2u;
    vec4 positionMass = inData[idx];
    vec4 velocityData = inData[idx + 1u];

    vec3 pos = positionMass.xyz;
    float mass = positionMass.w;
    vec3 vel = velocityData.xyz;

    // Compute acceleration from all other bodies
    vec3 acc = vec3(0.0);

    for (uint j = 0u; j < N; j++) {
        // Skip self-interaction
        if (i == j) {
            continue;
        }

        // Read body j
        uint jIdx = j * 2u;
        vec4 posJ_massJ = inData[jIdx];
        vec3 posJ = posJ_massJ.xyz;
        float massJ = posJ_massJ.w;

        // Compute displacement vector
        vec3 r = posJ - pos;

        // Compute distance squared with softening
        float distSq = dot(r, r) + softening * softening;

        // Compute (distSq)^(3/2) = distSq * sqrt(distSq)
        float distCubed = distSq * sqrt(distSq);

        // Add contribution to acceleration
        acc += G * massJ * r / distCubed;
    }

    // Update velocity and position using Euler integration
    vel += acc * dt;
    pos += vel * dt;

    // Write output
    outData[idx] = vec4(pos, mass);
    outData[idx + 1u] = vec4(vel, 0.0);
}
