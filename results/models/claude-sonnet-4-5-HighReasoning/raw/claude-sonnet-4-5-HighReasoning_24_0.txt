#include <iostream>
#include <fstream>
#include <vector>
#include <thread>
#include <atomic>
#include <mutex>
#include <cmath>
#include <string>
#include <sstream>
#include <algorithm>
#include <chrono>

constexpr double PI = 3.14159265358979323846;
constexpr double GRAVITY = 1.62;
constexpr double CONTROL_HZ = 12.0;

struct State {
    double x, y, z;
    double vx, vy, vz;
    double pitch, yaw, roll;
    double pitch_rate, yaw_rate, roll_rate;
    double fuel;
    std::chrono::steady_clock::time_point timestamp;
};

struct MissionParams {
    int width, depth, height;
    double gravity, max_thrust, fuel;
    double start_x, start_y, start_z;
    double target_x, target_y, target_z;
    std::string terrain_file;
};

// Global state
std::atomic<bool> running{true};
std::atomic<bool> terrain_ready{false};
std::mutex state_mutex;
State current_state{};
MissionParams mission;

std::vector<std::vector<uint16_t>> terrain_ground;
std::vector<std::vector<uint16_t>> terrain_float_bottom;
std::vector<std::vector<uint16_t>> terrain_float_top;

uint16_t read_uint16_le(std::ifstream& file) {
    uint8_t bytes[2];
    file.read(reinterpret_cast<char*>(bytes), 2);
    return bytes[0] | (static_cast<uint16_t>(bytes[1]) << 8);
}

void load_terrain() {
    std::ifstream file(mission.terrain_file, std::ios::binary);
    if (!file.is_open()) {
        std::cerr << "Failed to open terrain file\n";
        return;
    }

    terrain_ground.resize(mission.width, std::vector<uint16_t>(mission.depth));
    terrain_float_bottom.resize(mission.width, std::vector<uint16_t>(mission.depth));
    terrain_float_top.resize(mission.width, std::vector<uint16_t>(mission.depth));

    for (int x = 0; x < mission.width; x++) {
        for (int y = 0; y < mission.depth; y++) {
            terrain_ground[x][y] = read_uint16_le(file);
            terrain_float_bottom[x][y] = read_uint16_le(file);
            terrain_float_top[x][y] = read_uint16_le(file);
        }
    }

    terrain_ready = true;
}

double get_ground_height(double x, double y) {
    if (!terrain_ready) return 0.0;

    int ix = static_cast<int>(std::floor(x));
    int iy = static_cast<int>(std::floor(y));

    if (ix < 0 || ix >= mission.width || iy < 0 || iy >= mission.depth) {
        return 0.0;
    }

    return static_cast<double>(terrain_ground[ix][iy]);
}

template<typename T>
T clamp(T val, T min_val, T max_val) {
    return std::max(min_val, std::min(max_val, val));
}

double normalize_angle(double angle) {
    while (angle > PI) angle -= 2.0 * PI;
    while (angle < -PI) angle += 2.0 * PI;
    return angle;
}

void control_loop() {
    auto next_output = std::chrono::steady_clock::now();
    const auto output_interval = std::chrono::duration<double>(1.0 / CONTROL_HZ);

    while (running) {
        auto now = std::chrono::steady_clock::now();

        if (now < next_output) {
            std::this_thread::sleep_for(std::chrono::milliseconds(5));
            continue;
        }

        next_output += std::chrono::duration_cast<std::chrono::steady_clock::duration>(output_interval);

        State state;
        {
            std::lock_guard<std::mutex> lock(state_mutex);
            state = current_state;
        }

        // Compute errors
        double dx = mission.target_x - state.x;
        double dy = mission.target_y - state.y;
        double dz = mission.target_z - state.z;

        double dist_horizontal = std::sqrt(dx * dx + dy * dy);
        double dist_total = std::sqrt(dx * dx + dy * dy + dz * dz);

        double speed_horizontal = std::sqrt(state.vx * state.vx + state.vy * state.vy);
        double speed_vertical = state.vz;
        double speed_total = std::sqrt(state.vx * state.vx + state.vy * state.vy + state.vz * state.vz);

        // Determine landing zone
        double landing_radius = std::max(5.0, std::min(mission.width, mission.depth) * 0.02);
        bool near_target = dist_horizontal < landing_radius * 3.0 && std::abs(dz) < 30.0;
        bool very_close = dist_horizontal < landing_radius && std::abs(dz) < 10.0;

        // Control gains (adaptive to scale)
        double scale = std::max(mission.width, mission.depth);
        double position_gain = std::min(1.0, 100.0 / scale);

        double thrust = 0.0;
        double pitch_cmd = 0.0;
        double yaw_cmd = 0.0;
        double roll_cmd = 0.0;

        if (very_close) {
            // Final landing: gentle touchdown
            double target_vx = -dx * 0.8;
            double target_vy = -dy * 0.8;
            double target_vz = -dz * 0.5;

            target_vx = clamp(target_vx, -1.0, 1.0);
            target_vy = clamp(target_vy, -1.0, 1.0);
            target_vz = clamp(target_vz, -1.5, 0.5);

            double vx_error = target_vx - state.vx;
            double vy_error = target_vy - state.vy;
            double vz_error = target_vz - state.vz;

            double ax = vx_error * 3.0;
            double ay = vy_error * 3.0;
            double az = vz_error * 4.0 + GRAVITY;

            double a_total = std::sqrt(ax * ax + ay * ay + az * az);
            thrust = clamp(a_total / mission.max_thrust, 0.0, 1.0);

            // Ensure minimum thrust for stability
            if (thrust < 0.3) thrust = 0.3;

            // Desired thrust direction
            double tx = ax / (a_total + 1e-9);
            double ty = ay / (a_total + 1e-9);
            double tz = az / (a_total + 1e-9);

            double desired_pitch = -std::asin(clamp(tx, -0.8, 0.8));
            double desired_yaw = std::atan2(ty, tz);

            double pitch_error = normalize_angle(desired_pitch - state.pitch);
            double yaw_error = normalize_angle(desired_yaw - state.yaw);

            pitch_cmd = clamp(pitch_error * 5.0 - state.pitch_rate * 0.8, -1.0, 1.0);
            yaw_cmd = clamp(yaw_error * 5.0 - state.yaw_rate * 0.8, -1.0, 1.0);
            roll_cmd = clamp(-state.roll * 3.0 - state.roll_rate * 0.7, -1.0, 1.0);

        } else if (near_target) {
            // Approach phase: careful positioning
            double target_vx = -dx * 0.5;
            double target_vy = -dy * 0.5;

            // Limit approach speed
            double max_approach_speed = std::min(5.0, dist_horizontal * 0.3);
            double approach_speed = std::sqrt(target_vx * target_vx + target_vy * target_vy);
            if (approach_speed > max_approach_speed) {
                target_vx *= max_approach_speed / (approach_speed + 1e-9);
                target_vy *= max_approach_speed / (approach_speed + 1e-9);
            }

            // Vertical control
            double target_vz = -dz * 0.4;
            target_vz = clamp(target_vz, -3.0, 2.0);

            double vx_error = target_vx - state.vx;
            double vy_error = target_vy - state.vy;
            double vz_error = target_vz - state.vz;

            double ax = vx_error * 2.0;
            double ay = vy_error * 2.0;
            double az = vz_error * 3.0 + GRAVITY;

            double a_total = std::sqrt(ax * ax + ay * ay + az * az);
            thrust = clamp(a_total / mission.max_thrust, 0.0, 1.0);

            double tx = ax / (a_total + 1e-9);
            double ty = ay / (a_total + 1e-9);
            double tz = az / (a_total + 1e-9);

            double desired_pitch = -std::asin(clamp(tx, -0.9, 0.9));
            double desired_yaw = std::atan2(ty, tz);

            double pitch_error = normalize_angle(desired_pitch - state.pitch);
            double yaw_error = normalize_angle(desired_yaw - state.yaw);

            pitch_cmd = clamp(pitch_error * 4.0 - state.pitch_rate * 0.6, -1.0, 1.0);
            yaw_cmd = clamp(yaw_error * 4.0 - state.yaw_rate * 0.6, -1.0, 1.0);
            roll_cmd = clamp(-state.roll * 2.5 - state.roll_rate * 0.6, -1.0, 1.0);

        } else {
            // Cruise phase: navigate to target area
            double max_cruise_speed = std::min(30.0, dist_horizontal * 0.2);
            max_cruise_speed = std::max(max_cruise_speed, 5.0);

            double target_vx = (dx / (dist_horizontal + 1e-9)) * max_cruise_speed;
            double target_vy = (dy / (dist_horizontal + 1e-9)) * max_cruise_speed;

            // Vertical strategy
            double ground_height = get_ground_height(state.x, state.y);
            double height_above_ground = state.z - ground_height;
            double safe_altitude = std::max(50.0, scale * 0.05);

            double target_vz = 0.0;
            if (dz > safe_altitude) {
                target_vz = -std::min(10.0, std::abs(dz) * 0.1);
            } else if (dz < 10.0 && height_above_ground > safe_altitude) {
                target_vz = 0.0;
            } else if (height_above_ground < safe_altitude * 0.5) {
                target_vz = 5.0;
            }

            double vx_error = target_vx - state.vx;
            double vy_error = target_vy - state.vy;
            double vz_error = target_vz - state.vz;

            double ax = vx_error * 1.0;
            double ay = vy_error * 1.0;
            double az = vz_error * 2.0 + GRAVITY;

            double a_total = std::sqrt(ax * ax + ay * ay + az * az);
            thrust = clamp(a_total / mission.max_thrust, 0.0, 1.0);

            double tx = ax / (a_total + 1e-9);
            double ty = ay / (a_total + 1e-9);
            double tz = az / (a_total + 1e-9);

            double desired_pitch = -std::asin(clamp(tx, -1.0, 1.0));
            double desired_yaw = std::atan2(ty, tz);

            double pitch_error = normalize_angle(desired_pitch - state.pitch);
            double yaw_error = normalize_angle(desired_yaw - state.yaw);

            pitch_cmd = clamp(pitch_error * 3.0 - state.pitch_rate * 0.5, -1.0, 1.0);
            yaw_cmd = clamp(yaw_error * 3.0 - state.yaw_rate * 0.5, -1.0, 1.0);
            roll_cmd = clamp(-state.roll * 2.0 - state.roll_rate * 0.5, -1.0, 1.0);
        }

        // Safety: prevent dangerous descent
        if (state.vz < -5.0 && !very_close) {
            thrust = std::max(thrust, 0.7);
        }

        // Output control command
        std::cout << thrust << " " << pitch_cmd << " " << yaw_cmd << " " << roll_cmd << "\n";
        std::cout.flush();
    }
}

void input_loop() {
    std::string line;

    while (running && std::getline(std::cin, line)) {
        std::istringstream iss(line);
        State new_state;

        if (iss >> new_state.x >> new_state.y >> new_state.z
               >> new_state.vx >> new_state.vy >> new_state.vz
               >> new_state.pitch >> new_state.yaw >> new_state.roll
               >> new_state.pitch_rate >> new_state.yaw_rate >> new_state.roll_rate
               >> new_state.fuel) {

            new_state.timestamp = std::chrono::steady_clock::now();

            std::lock_guard<std::mutex> lock(state_mutex);
            current_state = new_state;
        }
    }

    running = false;
}

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    // Read mission parameters
    std::cin >> mission.width >> mission.depth >> mission.height
             >> mission.gravity >> mission.max_thrust >> mission.fuel;
    std::cin >> mission.start_x >> mission.start_y >> mission.start_z
             >> mission.target_x >> mission.target_y >> mission.target_z;

    std::cin.ignore();
    std::getline(std::cin, mission.terrain_file);

    std::string state_marker;
    std::getline(std::cin, state_marker);

    // Initialize current state
    current_state.x = mission.start_x;
    current_state.y = mission.start_y;
    current_state.z = mission.start_z;
    current_state.vx = current_state.vy = current_state.vz = 0.0;
    current_state.pitch = current_state.yaw = current_state.roll = 0.0;
    current_state.pitch_rate = current_state.yaw_rate = current_state.roll_rate = 0.0;
    current_state.fuel = mission.fuel;

    // Start threads
    std::thread terrain_thread(load_terrain);
    std::thread control_thread(control_loop);

    // Main thread handles input
    input_loop();

    // Cleanup
    running = false;
    if (control_thread.joinable()) control_thread.join();
    if (terrain_thread.joinable()) terrain_thread.join();

    return 0;
}
