const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

class PriorityQueue {
    constructor(compareFn) {
        this.heap = [];
        this.compare = compareFn;
    }

    push(item) {
        this.heap.push(item);
        this._bubbleUp(this.heap.length - 1);
    }

    pop() {
        if (this.heap.length === 0) return undefined;
        if (this.heap.length === 1) return this.heap.pop();

        const result = this.heap[0];
        this.heap[0] = this.heap.pop();
        this._bubbleDown(0);
        return result;
    }

    get length() {
        return this.heap.length;
    }

    _bubbleUp(idx) {
        while (idx > 0) {
            const parent = Math.floor((idx - 1) / 2);
            if (this.compare(this.heap[idx], this.heap[parent]) >= 0) break;
            [this.heap[idx], this.heap[parent]] = [this.heap[parent], this.heap[idx]];
            idx = parent;
        }
    }

    _bubbleDown(idx) {
        while (true) {
            const left = 2 * idx + 1;
            const right = 2 * idx + 2;
            let smallest = idx;

            if (left < this.heap.length && this.compare(this.heap[left], this.heap[smallest]) < 0) {
                smallest = left;
            }
            if (right < this.heap.length && this.compare(this.heap[right], this.heap[smallest]) < 0) {
                smallest = right;
            }

            if (smallest === idx) break;
            [this.heap[idx], this.heap[smallest]] = [this.heap[smallest], this.heap[idx]];
            idx = smallest;
        }
    }
}

const { machineCount, tasks } = data;

// Build task structures
const taskMap = new Map(tasks.map(t => [t.id, t]));
const children = new Map(tasks.map(t => [t.id, []]));
const parentCount = new Map(tasks.map(t => [t.id, 0]));

for (const task of tasks) {
    for (const depId of (task.deps || [])) {
        children.get(depId).push(task.id);
        parentCount.set(task.id, parentCount.get(task.id) + 1);
    }
}

// Calculate critical path length from each task (longest path to any leaf)
const criticalPath = new Map();

function calcCriticalPath(id) {
    if (criticalPath.has(id)) return criticalPath.get(id);

    let maxChildPath = 0;
    for (const childId of children.get(id)) {
        maxChildPath = Math.max(maxChildPath, calcCriticalPath(childId));
    }

    const result = taskMap.get(id).duration + maxChildPath;
    criticalPath.set(id, result);
    return result;
}

tasks.forEach(t => calcCriticalPath(t.id));

// Priority comparison: deadline, then critical path, then weight
function compareTasks(a, b) {
    const ta = taskMap.get(a);
    const tb = taskMap.get(b);

    const cpA = criticalPath.get(a);
    const cpB = criticalPath.get(b);
    const wA = ta.weight || 1;
    const wB = tb.weight || 1;
    const dA = ta.deadline ?? Infinity;
    const dB = tb.deadline ?? Infinity;

    // Sort by deadline (earlier first)
    if (dA !== Infinity || dB !== Infinity) {
        if (dA !== dB) return dA - dB;
    }

    // Then by critical path (longer first)
    if (cpA !== cpB) return cpB - cpA;

    // Then by weight (higher first)
    if (wA !== wB) return wB - wA;

    return 0;
}

// Schedule tasks using priority queue
const schedule = [];
const machineNext = new Array(machineCount).fill(0);
const finishTime = new Map();
const ready = new PriorityQueue(compareTasks);
const remaining = new Map(parentCount);

// Initialize with tasks that have no dependencies
for (const [id, count] of parentCount) {
    if (count === 0) ready.push(id);
}

while (ready.length > 0) {
    const taskId = ready.pop();
    const task = taskMap.get(taskId);

    // Find earliest start time based on dependencies
    const deps = task.deps || [];
    const depFinish = deps.length > 0 ? Math.max(...deps.map(d => finishTime.get(d))) : 0;

    // Find earliest available machine
    let bestMachine = 0;
    for (let i = 1; i < machineCount; i++) {
        if (machineNext[i] < machineNext[bestMachine]) {
            bestMachine = i;
        }
    }

    const start = Math.max(depFinish, machineNext[bestMachine]);
    const finish = start + task.duration;

    schedule.push({ id: taskId, start, machine: bestMachine });
    machineNext[bestMachine] = finish;
    finishTime.set(taskId, finish);

    // Add newly ready tasks
    for (const childId of children.get(taskId)) {
        const newCount = remaining.get(childId) - 1;
        remaining.set(childId, newCount);
        if (newCount === 0) ready.push(childId);
    }
}
