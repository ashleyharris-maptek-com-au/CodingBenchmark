OpCapability Shader
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %worldPos %normal %outColor
OpExecutionMode %main OriginUpperLeft

OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %outColor Location 0

OpDecorate %UBO Block
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0

%void = OpTypeVoid
%void_fn = OpTypeFunction %void
%float = OpTypeFloat 32
%vec3 = OpTypeVector %float 3
%vec4 = OpTypeVector %float 4
%mat4 = OpTypeMatrix %vec4 4
%UBO = OpTypeStruct %mat4 %mat4 %mat4 %vec4 %vec4 %vec4
%int = OpTypeInt 32 1

%ptr_input_vec3 = OpTypePointer Input %vec3
%ptr_output_vec4 = OpTypePointer Output %vec4
%ptr_uniform_ubo = OpTypePointer Uniform %UBO
%ptr_uniform_vec4 = OpTypePointer Uniform %vec4

%worldPos = OpVariable %ptr_input_vec3 Input
%normal = OpVariable %ptr_input_vec3 Input
%outColor = OpVariable %ptr_output_vec4 Output
%ubo = OpVariable %ptr_uniform_ubo Uniform

%float_0 = OpConstant %float 0.0
%float_1 = OpConstant %float 1.0
%float_3 = OpConstant %float 3.0
%float_0_2 = OpConstant %float 0.2
%float_0_3 = OpConstant %float 0.3
%float_0_5 = OpConstant %float 0.5
%float_0_8 = OpConstant %float 0.8
%int_3 = OpConstant %int 3
%int_4 = OpConstant %int 4

%glsl_ext = OpExtInstImport "GLSL.std.450"

%main = OpFunction %void None %void_fn
%entry = OpLabel

%norm_val = OpLoad %vec3 %normal
%N = OpExtInst %vec3 %glsl_ext Normalize %norm_val

%pos_val = OpLoad %vec3 %worldPos

%cam_ptr = OpAccessChain %ptr_uniform_vec4 %ubo %int_4
%cam_vec4 = OpLoad %vec4 %cam_ptr
%cam_pos = OpVectorShuffle %vec3 %cam_vec4 %cam_vec4 0 1 2

%view_unnorm = OpFSub %vec3 %cam_pos %pos_val
%V = OpExtInst %vec3 %glsl_ext Normalize %view_unnorm

%nv_dot = OpDot %float %N %V
%nv_clamped = OpExtInst %float %glsl_ext FMax %nv_dot %float_0
%one_minus_nv = OpFSub %float %float_1 %nv_clamped
%rim = OpExtInst %float %glsl_ext Pow %one_minus_nv %float_3

%light_ptr = OpAccessChain %ptr_uniform_vec4 %ubo %int_3
%light_vec4 = OpLoad %vec4 %light_ptr
%light_pos = OpVectorShuffle %vec3 %light_vec4 %light_vec4 0 1 2

%light_unnorm = OpFSub %vec3 %light_pos %pos_val
%L = OpExtInst %vec3 %glsl_ext Normalize %light_unnorm

%nl_dot = OpDot %float %N %L
%diffuse = OpExtInst %float %glsl_ext FMax %nl_dot %float_0

%diff_scaled = OpFMul %float %float_0_5 %diffuse
%modulator = OpFAdd %float %float_0_3 %diff_scaled

%base = OpCompositeConstruct %vec3 %float_0_2 %float_0_3 %float_0_8
%base_mod = OpVectorTimesScalar %vec3 %base %modulator

%rim_tint = OpCompositeConstruct %vec3 %float_1 %float_0_8 %float_0_5
%rim_contrib = OpVectorTimesScalar %vec3 %rim_tint %rim

%final_rgb = OpFAdd %vec3 %base_mod %rim_contrib

%r = OpCompositeExtract %float %final_rgb 0
%g = OpCompositeExtract %float %final_rgb 1
%b = OpCompositeExtract %float %final_rgb 2

%final = OpCompositeConstruct %vec4 %r %g %b %float_1

OpStore %outColor %final

OpReturn
OpFunctionEnd
