; SPIR-V
; Version: 1.0
; Generator: Khronos; 0
; Bound: 80
; Schema: 0
OpCapability Shader
%1 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %worldPos %normal %color %outColor
OpExecutionMode %main OriginUpperLeft

; Decorations
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 4
OpDecorate %outColor Location 0
OpDecorate %UBO Block
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0

; Types
%void = OpTypeVoid
%3 = OpTypeFunction %void
%float = OpTypeFloat 32
%vec3 = OpTypeVector %float 3
%vec4 = OpTypeVector %float 4
%mat4 = OpTypeMatrix %vec4 4
%UBO = OpTypeStruct %mat4 %mat4 %mat4 %vec4 %vec4 %vec4
%ptr_Uniform_UBO = OpTypePointer Uniform %UBO
%ptr_Uniform_vec4 = OpTypePointer Uniform %vec4
%ptr_Input_vec3 = OpTypePointer Input %vec3
%ptr_Output_vec4 = OpTypePointer Output %vec4
%int = OpTypeInt 32 1
%bool = OpTypeBool

; Constants
%int_3 = OpConstant %int 3
%float_1 = OpConstant %float 1
%float_0 = OpConstant %float 0
%float_0_75 = OpConstant %float 0.75
%float_0_5 = OpConstant %float 0.5
%float_0_25 = OpConstant %float 0.25
%float_0_7 = OpConstant %float 0.7
%float_0_4 = OpConstant %float 0.4
%float_0_2 = OpConstant %float 0.2

; Variables
%ubo = OpVariable %ptr_Uniform_UBO Uniform
%worldPos = OpVariable %ptr_Input_vec3 Input
%normal = OpVariable %ptr_Input_vec3 Input
%color = OpVariable %ptr_Input_vec3 Input
%outColor = OpVariable %ptr_Output_vec4 Output

; Main function
%main = OpFunction %void None %3
%label = OpLabel

; Load inputs
%worldPos_val = OpLoad %vec3 %worldPos
%normal_val = OpLoad %vec3 %normal
%color_val = OpLoad %vec3 %color

; Normalize the normal
%N = OpExtInst %vec3 %1 Normalize %normal_val

; Load lightPos from UBO (member 3)
%lightPos_ptr = OpAccessChain %ptr_Uniform_vec4 %ubo %int_3
%lightPos_vec4 = OpLoad %vec4 %lightPos_ptr
%lightPos_xyz = OpVectorShuffle %vec3 %lightPos_vec4 %lightPos_vec4 0 1 2

; Compute light direction L = normalize(lightPos - worldPos)
%lightDir = OpFSub %vec3 %lightPos_xyz %worldPos_val
%L = OpExtInst %vec3 %1 Normalize %lightDir

; Compute dot(N, L)
%dot_NL = OpDot %float %N %L

; d = max(dot(N, L), 0)
%d = OpExtInst %float %1 FMax %dot_NL %float_0

; Quantize into bands
%cmp_0_75 = OpFOrdGreaterThanEqual %bool %d %float_0_75
%cmp_0_5 = OpFOrdGreaterThanEqual %bool %d %float_0_5
%cmp_0_25 = OpFOrdGreaterThanEqual %bool %d %float_0_25

; Select band value (using nested selects)
%band_0 = OpSelect %float %cmp_0_25 %float_0_4 %float_0_2
%band_1 = OpSelect %float %cmp_0_5 %float_0_7 %band_0
%band = OpSelect %float %cmp_0_75 %float_1 %band_1

; Multiply color by band (broadcast band to vec3)
%band_vec3 = OpCompositeConstruct %vec3 %band %band %band
%final_color = OpFMul %vec3 %color_val %band_vec3

; Extract components and construct vec4 with alpha 1.0
%r = OpCompositeExtract %float %final_color 0
%g = OpCompositeExtract %float %final_color 1
%b = OpCompositeExtract %float %final_color 2
%output = OpCompositeConstruct %vec4 %r %g %b %float_1

; Store output
OpStore %outColor %output
OpReturn
OpFunctionEnd
