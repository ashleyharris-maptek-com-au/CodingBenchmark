using System;
using System.Linq;

class QAP
{
    static int N;
    static int[][] flow;
    static int[][] dist;
    static Random rand = new Random(42);

    static void Main()
    {
        ReadInput();

        int[] solution = Solve();

        long cost = CalculateCost(solution);
        Console.WriteLine(cost);
        Console.WriteLine(string.Join(" ", solution));
    }

    static void ReadInput()
    {
        N = int.Parse(Console.ReadLine());
        flow = new int[N][];
        dist = new int[N][];

        for (int i = 0; i < N; i++)
        {
            flow[i] = Console.ReadLine().Split().Select(int.Parse).ToArray();
        }

        for (int i = 0; i < N; i++)
        {
            dist[i] = Console.ReadLine().Split().Select(int.Parse).ToArray();
        }
    }

    static long CalculateCost(int[] perm)
    {
        long cost = 0;
        for (int i = 0; i < N; i++)
        {
            for (int j = 0; j < N; j++)
            {
                cost += (long)flow[i][j] * dist[perm[i]][perm[j]];
            }
        }
        return cost;
    }

    static long CalculateDelta(int[] perm, int i, int j)
    {
        if (i == j) return 0;

        long delta = 0;
        int pi = perm[i];
        int pj = perm[j];

        for (int k = 0; k < N; k++)
        {
            if (k == i || k == j) continue;
            int pk = perm[k];

            // Remove old contributions
            delta -= (long)flow[i][k] * dist[pi][pk];
            delta -= (long)flow[k][i] * dist[pk][pi];
            delta -= (long)flow[j][k] * dist[pj][pk];
            delta -= (long)flow[k][j] * dist[pk][pj];

            // Add new contributions after swap
            delta += (long)flow[i][k] * dist[pj][pk];
            delta += (long)flow[k][i] * dist[pk][pj];
            delta += (long)flow[j][k] * dist[pi][pk];
            delta += (long)flow[k][j] * dist[pk][pi];
        }

        // Handle i-j interaction
        delta -= (long)flow[i][j] * dist[pi][pj];
        delta -= (long)flow[j][i] * dist[pj][pi];
        delta += (long)flow[i][j] * dist[pj][pi];
        delta += (long)flow[j][i] * dist[pi][pj];

        return delta;
    }

    static int[] Solve()
    {
        int[] best = null;
        long bestCost = long.MaxValue;

        // Determine number of restarts based on problem size
        int restarts;
        if (N <= 10) restarts = 1000;
        else if (N <= 20) restarts = 500;
        else if (N <= 50) restarts = 100;
        else if (N <= 100) restarts = 20;
        else if (N <= 500) restarts = 5;
        else restarts = 3;

        for (int restart = 0; restart < restarts; restart++)
        {
            int[] current = Enumerable.Range(0, N).ToArray();
            Shuffle(current);

            // Initial local search to find local optimum
            LocalSearch(current);

            // Simulated annealing to escape local optima
            long currentCost = CalculateCost(current);
            double temp = currentCost * 0.1;
            int iterations = N * N * Math.Min(100, 10000 / N);
            double alpha = Math.Pow(0.001, 1.0 / iterations);

            for (int iter = 0; iter < iterations; iter++)
            {
                int i = rand.Next(N);
                int j = rand.Next(N);
                if (i == j) continue;

                long delta = CalculateDelta(current, i, j);

                if (delta < 0 || rand.NextDouble() < Math.Exp(-delta / temp))
                {
                    Swap(current, i, j);
                    currentCost += delta;
                }

                temp *= alpha;
            }

            // Final local search refinement
            LocalSearch(current);

            long cost = CalculateCost(current);
            if (cost < bestCost)
            {
                bestCost = cost;
                best = (int[])current.Clone();
            }
        }

        return best;
    }

    static void LocalSearch(int[] perm)
    {
        bool improved = true;
        while (improved)
        {
            improved = false;
            for (int i = 0; i < N - 1; i++)
            {
                for (int j = i + 1; j < N; j++)
                {
                    long delta = CalculateDelta(perm, i, j);
                    if (delta < 0)
                    {
                        Swap(perm, i, j);
                        improved = true;
                        break;
                    }
                }
                if (improved) break;
            }
        }
    }

    static void Swap(int[] arr, int i, int j)
    {
        int temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }

    static void Shuffle(int[] arr)
    {
        for (int i = N - 1; i > 0; i--)
        {
            int j = rand.Next(i + 1);
            Swap(arr, i, j);
        }
    }
}
