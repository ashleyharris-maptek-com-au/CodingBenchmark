using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

class Program
{
    static void Main()
    {
        var solver = new MiningSolver();
        solver.Solve();
    }
}

class MiningSolver
{
    private int xCount, yCount, zCount;
    private double xSize, ySize, zSize;
    private byte[,,] voxelData;
    private int[,] heightMap;
    private List<MiningOp> operations = new List<MiningOp>();
    private HashSet<(int, int, int)> removed = new HashSet<(int, int, int)>();

    public void Solve()
    {
        ReadInput();
        BuildHeightMap();

        bool isLarge = (long)xCount * yCount * zCount > 100000;

        if (isLarge)
        {
            SolveLargeScale();
        }
        else
        {
            SolveSmallScale();
        }

        WriteOutput();
    }

    private void ReadInput()
    {
        var header = Console.ReadLine().Split(' ', StringSplitOptions.RemoveEmptyEntries);
        xCount = int.Parse(header[0]);
        yCount = int.Parse(header[1]);
        zCount = int.Parse(header[2]);
        xSize = double.Parse(header[3]);
        ySize = double.Parse(header[4]);
        zSize = double.Parse(header[5]);

        voxelData = new byte[xCount, yCount, zCount];

        using (var stdin = Console.OpenStandardInput())
        {
            for (int x = 0; x < xCount; x++)
            {
                for (int y = 0; y < yCount; y++)
                {
                    for (int z = 0; z < zCount; z++)
                    {
                        int b = stdin.ReadByte();
                        voxelData[x, y, z] = (byte)b;
                    }
                }
            }
        }
    }

    private void BuildHeightMap()
    {
        heightMap = new int[xCount, yCount];

        for (int x = 0; x < xCount; x++)
        {
            for (int y = 0; y < yCount; y++)
            {
                heightMap[x, y] = -1;
                for (int z = zCount - 1; z >= 0; z--)
                {
                    if (voxelData[x, y, z] != 0)
                    {
                        heightMap[x, y] = z;
                        break;
                    }
                }
            }
        }
    }

    private void SolveSmallScale()
    {
        // For small problems, be thorough
        var candidates = new List<VoxelCandidate>();

        for (int x = 0; x < xCount; x++)
        {
            for (int y = 0; y < yCount; y++)
            {
                for (int z = 0; z <= Math.Max(0, heightMap[x, y]); z++)
                {
                    if (removed.Contains((x, y, z))) continue;

                    byte data = voxelData[x, y, z];
                    if (data == 0) continue;

                    int mass = data & 0xF;
                    int value = data >> 4;

                    if (value > 0)
                    {
                        double profit = CalculateProfit(x, y, z, mass, value);
                        if (profit > 0)
                        {
                            candidates.Add(new VoxelCandidate { x = x, y = y, z = z, profit = profit });
                        }
                    }
                }
            }
        }

        candidates.Sort((a, b) => b.profit.CompareTo(a.profit));

        foreach (var candidate in candidates)
        {
            TryMineVoxel(candidate.x, candidate.y, candidate.z, true);
        }
    }

    private void SolveLargeScale()
    {
        // For large problems, use spatial sampling and focus on high-value areas
        int sampleStride = Math.Max(1, Math.Min(xCount, yCount) / 100);
        var candidates = new List<VoxelCandidate>();

        for (int x = 0; x < xCount; x += sampleStride)
        {
            for (int y = 0; y < yCount; y += sampleStride)
            {
                ProcessColumn(x, y, candidates);
            }
        }

        // Also process area near processing plant more densely
        int denseRadius = Math.Min(50, Math.Min(xCount, yCount) / 4);
        for (int x = Math.Max(0, -denseRadius); x < Math.Min(xCount, denseRadius); x++)
        {
            for (int y = Math.Max(0, -denseRadius); y < Math.Min(yCount, denseRadius); y++)
            {
                if ((x % sampleStride != 0) || (y % sampleStride != 0))
                {
                    ProcessColumn(x, y, candidates);
                }
            }
        }

        candidates.Sort((a, b) => b.profit.CompareTo(a.profit));

        // Process top candidates
        int limit = Math.Min(candidates.Count, 10000);
        for (int i = 0; i < limit; i++)
        {
            var c = candidates[i];
            TryMineVoxel(c.x, c.y, c.z, true);
        }
    }

    private void ProcessColumn(int x, int y, List<VoxelCandidate> candidates)
    {
        int h = heightMap[x, y];
        if (h < 0) return;

        // Process from top down
        for (int z = h; z >= 0; z--)
        {
            if (removed.Contains((x, y, z))) continue;

            byte data = voxelData[x, y, z];
            if (data == 0) continue;

            int mass = data & 0xF;
            int value = data >> 4;

            if (value > 0)
            {
                double profit = CalculateProfit(x, y, z, mass, value);
                if (profit > 0)
                {
                    candidates.Add(new VoxelCandidate { x = x, y = y, z = z, profit = profit });
                }
            }
        }
    }

    private double CalculateProfit(int x, int y, int z, int mass, int value)
    {
        if (mass == 0) return 0;

        int manDist = Math.Abs(x) + Math.Abs(y);
        double revenue = mass * value * 1000.0;

        // Estimate transport cost
        int startHeight = GetHeight(x, y);
        int endHeight = GetHeight(0, 0);
        int heightChange = startHeight - endHeight;

        double cost = mass * manDist; // Flat travel

        // Add uphill penalty (simplified)
        if (heightChange < 0)
        {
            cost += mass * Math.Abs(heightChange) * 5;
        }

        return revenue - cost;
    }

    private bool TryMineVoxel(int x, int y, int z, bool toPlant)
    {
        if (removed.Contains((x, y, z))) return false;
        if (voxelData[x, y, z] == 0) return false;

        // Check if exposed
        if (!IsExposed(x, y, z))
        {
            // Try to clear blocking voxels
            if (!ClearPath(x, y, z)) return false;
        }

        byte data = voxelData[x, y, z];
        int mass = data & 0xF;
        int value = data >> 4;

        if (mass == 0) return false;

        // Find path to destination
        int destX = toPlant ? 0 : FindDumpX(x);
        int destY = toPlant ? 0 : FindDumpY(y);

        var path = FindValidPath(x, y, destX, destY);
        if (path == null || path.Count == 0) return false;

        // Create operation
        var op = new MiningOp
        {
            digX = x,
            digY = y,
            digZ = z,
            steps = path,
            destX = destX,
            destY = destY
        };

        operations.Add(op);
        removed.Add((x, y, z));

        // Update height map
        if (z == heightMap[x, y])
        {
            UpdateHeightAt(x, y);
        }

        // If dumping elsewhere, update that location's height
        if (!toPlant)
        {
            heightMap[destX, destY]++;
        }

        return true;
    }

    private bool IsExposed(int x, int y, int z)
    {
        // Check from above
        if (z >= GetHeight(x, y)) return true;

        // Check from sides
        int[] dx = { -1, 1, 0, 0 };
        int[] dy = { 0, 0, -1, 1 };

        for (int i = 0; i < 4; i++)
        {
            int nx = x + dx[i];
            int ny = y + dy[i];

            if (nx < 0 || nx >= xCount || ny < 0 || ny >= yCount)
            {
                return true;
            }

            if (GetHeight(nx, ny) < z)
            {
                return true;
            }
        }

        return false;
    }

    private bool ClearPath(int x, int y, int z)
    {
        // Try to remove blocking voxels above
        for (int zz = GetHeight(x, y); zz > z; zz--)
        {
            if (!removed.Contains((x, y, zz)) && voxelData[x, y, zz] != 0)
            {
                // Mine waste to a dump location
                TryMineVoxel(x, y, zz, false);
            }
        }

        return IsExposed(x, y, z);
    }

    private List<(int, int)> FindValidPath(int x1, int y1, int x2, int y2)
    {
        var path = new List<(int, int)>();

        int x = x1, y = y1;
        int currentZ = GetHeight(x, y);

        while (x != x2 || y != y2)
        {
            int nextX = x, nextY = y;

            // Choose next step (prefer towards destination)
            if (x != x2)
            {
                nextX = x + Math.Sign(x2 - x);
            }
            else if (y != y2)
            {
                nextY = y + Math.Sign(y2 - y);
            }

            int nextZ = GetHeight(nextX, nextY);

            // Check slope constraint: max 1 Z per 1 XY step
            // Since Z is 10x precision, this means max 10 Z units per XY step
            if (Math.Abs(nextZ - currentZ) > 10)
            {
                // Path too steep, try alternative
                if (y != y2 && nextY == y)
                {
                    nextX = x;
                    nextY = y + Math.Sign(y2 - y);
                    nextZ = GetHeight(nextX, nextY);

                    if (Math.Abs(nextZ - currentZ) > 10)
                    {
                        return null; // Cannot find valid path
                    }
                }
                else if (x != x2 && nextX == x)
                {
                    nextX = x + Math.Sign(x2 - x);
                    nextY = y;
                    nextZ = GetHeight(nextX, nextY);

                    if (Math.Abs(nextZ - currentZ) > 10)
                    {
                        return null;
                    }
                }
                else
                {
                    return null;
                }
            }

            x = nextX;
            y = nextY;
            currentZ = nextZ;
            path.Add((x, y));
        }

        return path;
    }

    private int GetHeight(int x, int y)
    {
        if (x < 0 || x >= xCount || y < 0 || y >= yCount) return 0;
        return Math.Max(0, heightMap[x, y]);
    }

    private void UpdateHeightAt(int x, int y)
    {
        heightMap[x, y] = -1;
        for (int z = zCount - 1; z >= 0; z--)
        {
            if (!removed.Contains((x, y, z)) && voxelData[x, y, z] != 0)
            {
                heightMap[x, y] = z;
                return;
            }
        }
    }

    private int FindDumpX(int x)
    {
        // Dump waste away from origin
        if (x > xCount / 2) return Math.Min(xCount - 1, x + 10);
        if (x < -xCount / 2) return Math.Max(0, x - 10);
        return x > 0 ? Math.Min(xCount - 1, xCount - 10) : 0;
    }

    private int FindDumpY(int y)
    {
        if (y > yCount / 2) return Math.Min(yCount - 1, y + 10);
        if (y < -yCount / 2) return Math.Max(0, y - 10);
        return y > 0 ? Math.Min(yCount - 1, yCount - 10) : 0;
    }

    private void WriteOutput()
    {
        Console.WriteLine(operations.Count);

        foreach (var op in operations)
        {
            Console.Write(op.steps.Count);
            Console.Write($" {op.digX} {op.digY} {op.digZ}");

            foreach (var step in op.steps)
            {
                Console.Write($" {step.Item1} {step.Item2}");
            }

            Console.WriteLine($" {op.destX} {op.destY}");
        }
    }
}

class VoxelCandidate
{
    public int x, y, z;
    public double profit;
}

class MiningOp
{
    public int digX, digY, digZ;
    public List<(int, int)> steps;
    public int destX, destY;
}
