const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { probability, drones, maxStepsPerDrone } = data;
const rows = probability.length;
const cols = probability[0].length;

// Track which cells have been covered
const covered = Array(rows).fill(null).map(() => Array(cols).fill(false));

// Initialize paths with starting positions
const paths = drones.map(drone => [[drone.x, drone.y]]);

// Mark starting positions as covered
drones.forEach(drone => {
  if (probability[drone.y][drone.x] > 0) {
    covered[drone.y][drone.x] = true;
  }
});

// Manhattan distance heuristic
function manhattanDist(x1, y1, x2, y2) {
  return Math.abs(x1 - x2) + Math.abs(y1 - y2);
}

// Get next step towards target using greedy direction
function getNextStep(fromX, fromY, toX, toY) {
  if (fromX === toX && fromY === toY) return [fromX, fromY];

  const dx = toX - fromX;
  const dy = toY - fromY;

  // Try moving in the direction that reduces distance most
  const moves = [];

  if (dx !== 0) {
    const nx = fromX + Math.sign(dx);
    if (nx >= 0 && nx < cols && probability[fromY][nx] !== -1) {
      moves.push([nx, fromY, Math.abs(dx - Math.sign(dx)) + Math.abs(dy)]);
    }
  }

  if (dy !== 0) {
    const ny = fromY + Math.sign(dy);
    if (ny >= 0 && ny < rows && probability[ny][fromX] !== -1) {
      moves.push([fromX, ny, Math.abs(dx) + Math.abs(dy - Math.sign(dy))]);
    }
  }

  // If we can't move toward target, try any valid move
  if (moves.length === 0) {
    const dirs = [[0, 1], [0, -1], [1, 0], [-1, 0]];
    for (const [ddx, ddy] of dirs) {
      const nx = fromX + ddx;
      const ny = fromY + ddy;
      if (nx >= 0 && nx < cols && ny >= 0 && ny < rows && probability[ny][nx] !== -1) {
        moves.push([nx, ny, manhattanDist(nx, ny, toX, toY)]);
      }
    }
  }

  if (moves.length === 0) {
    return [fromX, fromY]; // Stay in place if stuck
  }

  // Choose move that gets closest to target
  moves.sort((a, b) => a[2] - b[2]);
  return [moves[0][0], moves[0][1]];
}

// Build paths step by step
for (let step = 1; step < maxStepsPerDrone; step++) {
  const remainingSteps = maxStepsPerDrone - step;

  // Collect all uncovered valuable cells
  const uncoveredCells = [];
  for (let y = 0; y < rows; y++) {
    for (let x = 0; x < cols; x++) {
      if (probability[y][x] > 0 && !covered[y][x]) {
        uncoveredCells.push({ x, y, value: probability[y][x] });
      }
    }
  }

  // For each drone, decide next move
  const nextMoves = [];
  const targetedCells = new Set(); // Track which cells drones are targeting this step

  for (let i = 0; i < drones.length; i++) {
    const [currentX, currentY] = paths[i][paths[i].length - 1];

    let bestTarget = null;
    let bestScore = -Infinity;

    // Find best target among uncovered cells
    for (const cell of uncoveredCells) {
      const cellKey = `${cell.x},${cell.y}`;

      // Prefer cells not already targeted by other drones this step
      const targetPenalty = targetedCells.has(cellKey) ? 0.5 : 1.0;

      const dist = manhattanDist(currentX, currentY, cell.x, cell.y);
      if (dist <= remainingSteps) {
        // Score: value / (distance + 1), with penalty for already targeted cells
        const score = (cell.value * targetPenalty) / (dist + 1);
        if (score > bestScore) {
          bestScore = score;
          bestTarget = cell;
        }
      }
    }

    if (bestTarget) {
      targetedCells.add(`${bestTarget.x},${bestTarget.y}`);
      nextMoves.push(getNextStep(currentX, currentY, bestTarget.x, bestTarget.y));
    } else {
      // No valuable cells in range, stay in place
      nextMoves.push([currentX, currentY]);
    }
  }

  // Apply moves and update coverage
  for (let i = 0; i < drones.length; i++) {
    paths[i].push(nextMoves[i]);
    const [x, y] = nextMoves[i];
    if (probability[y][x] > 0) {
      covered[y][x] = true;
    }
  }
}
