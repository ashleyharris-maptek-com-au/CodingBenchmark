#include <iostream>
#include <vector>
#include <algorithm>
#include <limits>

struct Rectangle {
    int w, h;
    int index;
};

struct Placement {
    int board;
    int x, y;
    int rotated;
};

struct SkylineNode {
    int x, y, width;
};

class SkylineBoard {
private:
    int boardWidth, boardHeight;
    std::vector<SkylineNode> skyline;

public:
    SkylineBoard(int w, int h) : boardWidth(w), boardHeight(h) {
        skyline.push_back({0, 0, w});
    }

    bool canFit(int nodeIdx, int rectW, int rectH, int& outY) const {
        int x = skyline[nodeIdx].x;
        if (x + rectW > boardWidth) return false;

        int y = skyline[nodeIdx].y;
        int widthLeft = rectW;
        size_t i = nodeIdx;

        while (widthLeft > 0 && i < skyline.size()) {
            y = std::max(y, skyline[i].y);
            if (y + rectH > boardHeight) return false;
            widthLeft -= skyline[i].width;
            i++;
        }

        if (widthLeft > 0) return false;

        outY = y;
        return true;
    }

    bool tryPlace(int rectW, int rectH, int& outX, int& outY) {
        int bestIdx = -1;
        int bestY = boardHeight + 1;
        int bestX = boardWidth + 1;
        int bestFitY = 0;

        // Find best position using bottom-left heuristic
        for (size_t i = 0; i < skyline.size(); i++) {
            int fitY;
            if (canFit(i, rectW, rectH, fitY)) {
                if (fitY < bestY || (fitY == bestY && skyline[i].x < bestX)) {
                    bestIdx = static_cast<int>(i);
                    bestY = fitY;
                    bestX = skyline[i].x;
                    bestFitY = fitY;
                }
            }
        }

        if (bestIdx == -1) return false;

        outX = skyline[bestIdx].x;
        outY = bestFitY;

        addSkylineLevel(bestIdx, rectW, rectH, bestFitY);

        return true;
    }

private:
    void addSkylineLevel(int nodeIdx, int rectW, int rectH, int y) {
        SkylineNode newNode = {skyline[nodeIdx].x, y + rectH, rectW};

        skyline.insert(skyline.begin() + nodeIdx, newNode);
        nodeIdx++;

        // Consume width from affected nodes
        int widthLeft = rectW;
        while (nodeIdx < static_cast<int>(skyline.size()) && widthLeft > 0) {
            widthLeft -= skyline[nodeIdx].width;
            if (widthLeft >= 0) {
                skyline.erase(skyline.begin() + nodeIdx);
            } else {
                skyline[nodeIdx].x += skyline[nodeIdx].width + widthLeft;
                skyline[nodeIdx].width = -widthLeft;
                break;
            }
        }

        // Merge adjacent nodes at same height
        for (size_t i = 0; i < skyline.size() - 1; ) {
            if (skyline[i].y == skyline[i + 1].y) {
                skyline[i].width += skyline[i + 1].width;
                skyline.erase(skyline.begin() + i + 1);
            } else {
                i++;
            }
        }
    }
};

class MultiboardPacker {
private:
    int boardWidth, boardHeight;
    std::vector<SkylineBoard> boards;

public:
    MultiboardPacker(int w, int h) : boardWidth(w), boardHeight(h) {
        boards.emplace_back(w, h);
    }

    Placement place(int rectW, int rectH, int originalIndex) {
        int x, y;

        // Try current board with original orientation
        if (boards.back().tryPlace(rectW, rectH, x, y)) {
            return {static_cast<int>(boards.size()) - 1, x, y, 0};
        }

        // Try current board with rotated orientation
        if (rectW != rectH && boards.back().tryPlace(rectH, rectW, x, y)) {
            return {static_cast<int>(boards.size()) - 1, x, y, 1};
        }

        // Need new board - try original orientation
        boards.emplace_back(boardWidth, boardHeight);

        if (boards.back().tryPlace(rectW, rectH, x, y)) {
            return {static_cast<int>(boards.size()) - 1, x, y, 0};
        }

        // Try rotated orientation on new board
        if (rectW != rectH && boards.back().tryPlace(rectH, rectW, x, y)) {
            return {static_cast<int>(boards.size()) - 1, x, y, 1};
        }

        // Fallback (should not happen with valid input)
        return {static_cast<int>(boards.size()) - 1, 0, 0, 0};
    }

    int getBoardCount() const {
        return static_cast<int>(boards.size());
    }
};

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int N, boardWidth, boardHeight;
    std::cin >> N >> boardWidth >> boardHeight;

    std::vector<Rectangle> rects(N);
    for (int i = 0; i < N; i++) {
        std::cin >> rects[i].w >> rects[i].h;
        rects[i].index = i;
    }

    // Sort by area (decreasing) for better packing density
    std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
        long long areaA = static_cast<long long>(a.w) * a.h;
        long long areaB = static_cast<long long>(b.w) * b.h;
        if (areaA != areaB) return areaA > areaB;
        // Tie-breaker: prefer rectangles with larger max dimension
        return std::max(a.w, a.h) > std::max(b.w, b.h);
    });

    MultiboardPacker packer(boardWidth, boardHeight);
    std::vector<Placement> placements(N);

    for (const auto& rect : rects) {
        placements[rect.index] = packer.place(rect.w, rect.h, rect.index);
    }

    // Output results
    std::cout << packer.getBoardCount() << "\n";
    for (const auto& p : placements) {
        std::cout << p.board << " " << p.x << " " << p.y << " " << p.rotated << "\n";
    }

    return 0;
}
