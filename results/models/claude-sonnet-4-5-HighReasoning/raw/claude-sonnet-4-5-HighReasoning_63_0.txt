declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

interface Robot {
  id: number;
  x: number;
  y: number;
}

interface Job {
  id: number;
  x: number;
  y: number;
  priority: number;
}

interface Input {
  grid: string[];
  robots: Robot[];
  jobs: Job[];
}

const { grid, robots, jobs } = data as Input;

const rows = grid.length;
const cols = rows > 0 ? grid[0].length : 0;

// BFS to find shortest distance through open floor cells
function bfs(startX: number, startY: number, endX: number, endY: number): number {
  if (startX === endX && startY === endY) return 0;

  const queue: number[] = [];
  let head = 0;

  queue.push(startX, startY, 0);

  const visited = new Set<number>();
  visited.add(startY * cols + startX);

  const directions = [[0, 1], [1, 0], [0, -1], [-1, 0]];

  while (head < queue.length) {
    const x = queue[head++];
    const y = queue[head++];
    const dist = queue[head++];

    for (const [dx, dy] of directions) {
      const nx = x + dx;
      const ny = y + dy;

      if (ny < 0 || ny >= rows || nx < 0 || nx >= cols) continue;

      const key = ny * cols + nx;
      if (visited.has(key)) continue;

      if (nx === endX && ny === endY) {
        return dist + 1;
      }

      if (grid[ny][nx] === '.') {
        visited.add(key);
        queue.push(nx, ny, dist + 1);
      }
    }
  }

  return Infinity;
}

// Greedy assignment: process jobs by priority, assign to nearest robot
const assignments: number[][] = robots.map(() => []);
const robotPositions = robots.map(r => ({ x: r.x, y: r.y }));

// Sort jobs by priority (descending), then by id for determinism
const sortedJobs = [...jobs].sort((a, b) => {
  if (b.priority !== a.priority) return b.priority - a.priority;
  return a.id - b.id;
});

for (const job of sortedJobs) {
  let bestRobot = 0;
  let minDistance = Infinity;

  for (let i = 0; i < robots.length; i++) {
    const pos = robotPositions[i];
    const dist = bfs(pos.x, pos.y, job.x, job.y);

    if (dist < minDistance) {
      minDistance = dist;
      bestRobot = i;
    }
  }

  assignments[bestRobot].push(job.id);
  robotPositions[bestRobot] = { x: job.x, y: job.y };
}
