using System;
using System.Collections.Generic;
using System.Linq;

class Program
{
    static void Main()
    {
        var input = Console.ReadLine().Split();
        int n = int.Parse(input[0]);
        int m = int.Parse(input[1]);

        var graph = new List<int>[n];
        for (int i = 0; i < n; i++)
            graph[i] = new List<int>();

        for (int i = 0; i < m; i++)
        {
            var edge = Console.ReadLine().Split();
            int u = int.Parse(edge[0]);
            int v = int.Parse(edge[1]);
            graph[u].Add(v);
        }

        var result = SolveFVS(graph, n);

        Console.WriteLine(result.Count);
        Console.WriteLine(string.Join(" ", result.OrderBy(x => x)));
    }

    static HashSet<int> SolveFVS(List<int>[] graph, int n)
    {
        var result = new HashSet<int>();
        var sccs = FindSCCs(graph, n);

        foreach (var scc in sccs)
        {
            if (scc.Count > 1 || (scc.Count == 1 && graph[scc[0]].Contains(scc[0])))
            {
                var fvs = FindMinFVSForSCC(scc, graph);
                foreach (var v in fvs)
                    result.Add(v);
            }
        }

        return result;
    }

    static List<int> FindMinFVSForSCC(List<int> scc, List<int>[] graph)
    {
        var sccSet = new HashSet<int>(scc);

        var subgraph = new Dictionary<int, List<int>>();
        foreach (var v in scc)
        {
            subgraph[v] = new List<int>();
            foreach (var u in graph[v])
            {
                if (sccSet.Contains(u))
                    subgraph[v].Add(u);
            }
        }

        for (int k = 1; k <= scc.Count; k++)
        {
            var fvs = FindFVSOfSize(scc, subgraph, k);
            if (fvs != null)
                return fvs;
        }

        return scc;
    }

    static List<int> FindFVSOfSize(List<int> vertices, Dictionary<int, List<int>> graph, int k)
    {
        var ordered = vertices.OrderByDescending(v => graph[v].Count +
            vertices.Count(u => graph[u].Contains(v))).ToList();

        var result = new List<int>();
        if (SearchFVS(ordered, graph, k, 0, result, new HashSet<int>()))
            return result;

        return null;
    }

    static bool SearchFVS(List<int> vertices, Dictionary<int, List<int>> graph,
        int k, int start, List<int> current, HashSet<int> removed)
    {
        if (current.Count == k)
        {
            return IsDAGAfterRemoval(vertices, graph, removed);
        }

        if (start + (k - current.Count) > vertices.Count)
            return false;

        for (int i = start; i < vertices.Count; i++)
        {
            var v = vertices[i];
            current.Add(v);
            removed.Add(v);

            if (SearchFVS(vertices, graph, k, i + 1, current, removed))
                return true;

            current.RemoveAt(current.Count - 1);
            removed.Remove(v);
        }

        return false;
    }

    static bool IsDAGAfterRemoval(List<int> vertices, Dictionary<int, List<int>> graph, HashSet<int> removed)
    {
        var color = new Dictionary<int, int>();

        foreach (var v in vertices)
        {
            if (!removed.Contains(v))
                color[v] = 0;
        }

        foreach (var v in vertices)
        {
            if (!removed.Contains(v) && color[v] == 0)
            {
                if (HasCycle(v, graph, color, removed))
                    return false;
            }
        }

        return true;
    }

    static bool HasCycle(int v, Dictionary<int, List<int>> graph, Dictionary<int, int> color, HashSet<int> removed)
    {
        color[v] = 1;

        foreach (var u in graph[v])
        {
            if (removed.Contains(u))
                continue;

            if (color[u] == 1)
                return true;

            if (color[u] == 0 && HasCycle(u, graph, color, removed))
                return true;
        }

        color[v] = 2;
        return false;
    }

    static List<List<int>> FindSCCs(List<int>[] graph, int n)
    {
        var index = new int[n];
        var lowlink = new int[n];
        var onStack = new bool[n];
        var stack = new Stack<int>();
        var sccs = new List<List<int>>();
        int idx = 0;

        Array.Fill(index, -1);

        for (int i = 0; i < n; i++)
        {
            if (index[i] == -1)
                Tarjan(i, graph, index, lowlink, onStack, stack, sccs, ref idx);
        }

        return sccs;
    }

    static void Tarjan(int v, List<int>[] graph, int[] index, int[] lowlink,
        bool[] onStack, Stack<int> stack, List<List<int>> sccs, ref int idx)
    {
        index[v] = idx;
        lowlink[v] = idx;
        idx++;
        stack.Push(v);
        onStack[v] = true;

        foreach (var w in graph[v])
        {
            if (index[w] == -1)
            {
                Tarjan(w, graph, index, lowlink, onStack, stack, sccs, ref idx);
                lowlink[v] = Math.Min(lowlink[v], lowlink[w]);
            }
            else if (onStack[w])
            {
                lowlink[v] = Math.Min(lowlink[v], index[w]);
            }
        }

        if (lowlink[v] == index[v])
        {
            var scc = new List<int>();
            int w;
            do
            {
                w = stack.Pop();
                onStack[w] = false;
                scc.Add(w);
            } while (w != v);

            sccs.Add(scc);
        }
    }
}
