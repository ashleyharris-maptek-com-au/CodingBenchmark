OpCapability Geometry
%ext_glsl = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %color_in %gl_out %color_out
OpExecutionMode %main InputLinesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1

; Decorations
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %color_in Location 2
OpDecorate %color_out Location 0

; Types
%void = OpTypeVoid
%void_func = OpTypeFunction %void
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0

; Constants
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3
%uint_4 = OpConstant %uint 4
%float_0 = OpConstant %float 0
%float_1 = OpConstant %float 1
%float_thickness = OpConstant %float 0.025

; Composite constants
%vec2_0 = OpConstantComposite %v2float %float_0 %float_0

; Input/Output structures
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%arr_color_in = OpTypeArray %v3float %uint_4
%gl_PerVertex_out = OpTypeStruct %v4float

; Pointer types
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_arr_color = OpTypePointer Input %arr_color_in
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float

; Interface variables
%gl_in = OpVariable %ptr_in_arr_pv Input
%color_in = OpVariable %ptr_in_arr_color Input
%gl_out = OpVariable %ptr_out_pv Output
%color_out = OpVariable %ptr_out_v3 Output

; Main function
%main = OpFunction %void None %void_func
%entry = OpLabel

; Load all 4 positions
%pos0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%pos0 = OpLoad %v4float %pos0_ptr
%pos1_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%pos1 = OpLoad %v4float %pos1_ptr
%pos2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%pos2 = OpLoad %v4float %pos2_ptr
%pos3_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
%pos3 = OpLoad %v4float %pos3_ptr

; Load colors from vertices 1 and 2
%color1_ptr = OpAccessChain %ptr_in_v3 %color_in %int_1
%color1 = OpLoad %v3float %color1_ptr
%color2_ptr = OpAccessChain %ptr_in_v3 %color_in %int_2
%color2 = OpLoad %v3float %color2_ptr

; Compute tangent at start: t0 = normalize(pos2.xy - pos0.xy)
%diff0 = OpFSub %v4float %pos2 %pos0
%diff0_xy = OpVectorShuffle %v2float %diff0 %diff0 0 1
%t0 = OpExtInst %v2float %ext_glsl Normalize %diff0_xy

; Compute tangent at end: t1 = normalize(pos3.xy - pos1.xy)
%diff1 = OpFSub %v4float %pos3 %pos1
%diff1_xy = OpVectorShuffle %v2float %diff1 %diff1 0 1
%t1 = OpExtInst %v2float %ext_glsl Normalize %diff1_xy

; Extract tangent components for perpendicular computation
%t0_x = OpCompositeExtract %float %t0 0
%t0_y = OpCompositeExtract %float %t0 1
%t1_x = OpCompositeExtract %float %t1 0
%t1_y = OpCompositeExtract %float %t1 1

; Compute perpendiculars: perp = vec2(-t.y, t.x)
%t0_y_neg = OpFNegate %float %t0_y
%perp0 = OpCompositeConstruct %v2float %t0_y_neg %t0_x
%t1_y_neg = OpFNegate %float %t1_y
%perp1 = OpCompositeConstruct %v2float %t1_y_neg %t1_x

; Scale perpendiculars by thickness
%thickness_vec = OpCompositeConstruct %v2float %float_thickness %float_thickness
%offset0 = OpFMul %v2float %perp0 %thickness_vec
%offset1 = OpFMul %v2float %perp1 %thickness_vec

; Extract start and end positions (xy only)
%start_xy = OpVectorShuffle %v2float %pos1 %pos1 0 1
%end_xy = OpVectorShuffle %v2float %pos2 %pos2 0 1

; Compute vertex positions for triangle strip
%v0_xy = OpFAdd %v2float %start_xy %offset0
%v1_xy = OpFSub %v2float %start_xy %offset0
%v2_xy = OpFAdd %v2float %end_xy %offset1
%v3_xy = OpFSub %v2float %end_xy %offset1

; Reconstruct v4 positions (preserve z and w from original positions)
%pos1_zw = OpVectorShuffle %v2float %pos1 %pos1 2 3
%pos2_zw = OpVectorShuffle %v2float %pos2 %pos2 2 3
%vertex0 = OpVectorShuffle %v4float %v0_xy %pos1_zw 0 1 2 3
%vertex1 = OpVectorShuffle %v4float %v1_xy %pos1_zw 0 1 2 3
%vertex2 = OpVectorShuffle %v4float %v2_xy %pos2_zw 0 1 2 3
%vertex3 = OpVectorShuffle %v4float %v3_xy %pos2_zw 0 1 2 3

; Emit vertex 0 (start + offset)
%out_pos_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_ptr %vertex0
OpStore %color_out %color1
OpEmitVertex

; Emit vertex 1 (start - offset)
OpStore %out_pos_ptr %vertex1
OpStore %color_out %color1
OpEmitVertex

; Emit vertex 2 (end + offset)
OpStore %out_pos_ptr %vertex2
OpStore %color_out %color2
OpEmitVertex

; Emit vertex 3 (end - offset)
OpStore %out_pos_ptr %vertex3
OpStore %color_out %color2
OpEmitVertex

OpEndPrimitive

OpReturn
OpFunctionEnd
