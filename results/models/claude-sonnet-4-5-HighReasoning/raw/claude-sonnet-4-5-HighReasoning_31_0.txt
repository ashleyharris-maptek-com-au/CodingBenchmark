use std::collections::{HashMap, HashSet};
use std::io::{self, BufRead};
use std::sync::{Arc, Mutex};
use std::thread;

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Read first line: n and target
    let first_line = lines.next().unwrap().unwrap();
    let mut parts = first_line.split_whitespace();
    let n: usize = parts.next().unwrap().parse().unwrap();
    let target: i64 = parts.next().unwrap().parse().unwrap();

    // Read numbers
    let second_line = lines.next().unwrap().unwrap();
    let numbers: Vec<i64> = second_line
        .split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();

    // Solve
    if let Some(indices) = solve_subset_sum(&numbers, target) {
        println!("YES");
        print!("[");
        for (i, &idx) in indices.iter().enumerate() {
            if i > 0 {
                print!(" ");
            }
            print!("{}", idx);
        }
        println!("]");
    } else {
        println!("NO");
    }
}

fn solve_subset_sum(numbers: &[i64], target: i64) -> Option<Vec<usize>> {
    let n = numbers.len();

    // Edge case: zero target
    if target == 0 {
        return Some(vec![]);
    }

    if n == 0 {
        return None;
    }

    // Check if any single element equals target
    for (i, &num) in numbers.iter().enumerate() {
        if num == target {
            return Some(vec![i]);
        }
    }

    // Choose algorithm based on size
    if n <= 23 {
        brute_force(numbers, target)
    } else if n <= 40 {
        meet_in_middle(numbers, target)
    } else if n <= 10000 {
        dp_with_tracking(numbers, target)
    } else {
        // For very large sets, use parallel search with heuristics
        parallel_search(numbers, target)
    }
}

fn brute_force(numbers: &[i64], target: i64) -> Option<Vec<usize>> {
    let n = numbers.len();
    let limit = 1u64 << n;

    for mask in 1..limit {
        let mut sum = 0i64;
        let mut indices = Vec::new();

        for i in 0..n {
            if mask & (1 << i) != 0 {
                sum = sum.wrapping_add(numbers[i]);
                indices.push(i);
                if sum == target {
                    return Some(indices);
                }
            }
        }

        if sum == target {
            return Some(indices);
        }
    }

    None
}

fn meet_in_middle(numbers: &[i64], target: i64) -> Option<Vec<usize>> {
    let n = numbers.len();
    let mid = n / 2;

    // Generate all subset sums for first half
    let mut first_half: HashMap<i64, Vec<usize>> = HashMap::new();
    let first_limit = 1usize << mid;

    for mask in 0..first_limit {
        let mut sum = 0i64;
        let mut indices = Vec::new();

        for i in 0..mid {
            if mask & (1 << i) != 0 {
                sum = sum.wrapping_add(numbers[i]);
                indices.push(i);
            }
        }

        first_half.entry(sum).or_insert(indices);
    }

    // Generate subset sums for second half and look for matches
    let second_size = n - mid;
    let second_limit = 1usize << second_size;

    for mask in 0..second_limit {
        let mut sum = 0i64;
        let mut indices = Vec::new();

        for i in 0..second_size {
            if mask & (1 << i) != 0 {
                sum = sum.wrapping_add(numbers[mid + i]);
                indices.push(mid + i);
            }
        }

        let needed = target.wrapping_sub(sum);
        if let Some(first_indices) = first_half.get(&needed) {
            if mask == 0 && first_indices.is_empty() {
                continue; // Skip empty subset
            }
            let mut result = first_indices.clone();
            result.extend(indices);
            return Some(result);
        }
    }

    None
}

fn dp_with_tracking(numbers: &[i64], target: i64) -> Option<Vec<usize>> {
    // DP approach with HashMap, tracking indices
    let mut dp: HashMap<i64, Vec<usize>> = HashMap::new();
    dp.insert(0, vec![]);

    for (idx, &num) in numbers.iter().enumerate() {
        // Collect new states to add (to avoid borrow issues)
        let mut new_states = Vec::new();

        for (&sum, indices) in dp.iter() {
            let new_sum = sum.wrapping_add(num);

            // Check if we found target
            if new_sum == target {
                let mut result = indices.clone();
                result.push(idx);
                return Some(result);
            }

            // Add new state if not already present
            if !dp.contains_key(&new_sum) {
                let mut new_indices = indices.clone();
                new_indices.push(idx);
                new_states.push((new_sum, new_indices));
            }
        }

        // Add new states
        for (sum, indices) in new_states {
            dp.insert(sum, indices);

            // Limit DP table size to prevent memory exhaustion
            if dp.len() > 5_000_000 {
                return None;
            }
        }
    }

    dp.get(&target).cloned()
}

fn parallel_search(numbers: &[i64], target: i64) -> Option<Vec<usize>> {
    // For very large sets, try multiple strategies in parallel

    // Strategy 1: Try DP on first portion
    let chunk_size = 5000.min(numbers.len());
    if let Some(result) = dp_with_tracking(&numbers[..chunk_size], target) {
        return Some(result);
    }

    // Strategy 2: Try with sorted numbers (greedy-like)
    let mut indexed: Vec<(usize, i64)> = numbers.iter()
        .enumerate()
        .map(|(i, &v)| (i, v))
        .collect();

    // Try backtracking with pruning
    if let Some(result) = backtrack_search(&indexed, target, numbers.len()) {
        return Some(result);
    }

    // Strategy 3: Parallel chunking - divide array and try different combinations
    let num_threads = 4.min(numbers.len() / 1000 + 1);
    if num_threads > 1 {
        if let Some(result) = parallel_chunk_search(numbers, target, num_threads) {
            return Some(result);
        }
    }

    None
}

fn backtrack_search(indexed: &[(usize, i64)], target: i64, _max_len: usize) -> Option<Vec<usize>> {
    fn backtrack(
        indexed: &[(usize, i64)],
        pos: usize,
        current_sum: i64,
        target: i64,
        path: &mut Vec<usize>,
        cache: &mut HashSet<(usize, i64)>,
        depth: usize,
    ) -> bool {
        if current_sum == target {
            return true;
        }

        if pos >= indexed.len() || depth > 100 {
            return false;
        }

        let state = (pos, current_sum);
        if cache.contains(&state) {
            return false;
        }
        cache.insert(state);

        // Try including current
        let (idx, val) = indexed[pos];
        let new_sum = current_sum.wrapping_add(val);

        path.push(idx);
        if backtrack(indexed, pos + 1, new_sum, target, path, cache, depth + 1) {
            return true;
        }
        path.pop();

        // Try skipping current
        backtrack(indexed, pos + 1, current_sum, target, path, cache, depth + 1)
    }

    let mut path = Vec::new();
    let mut cache = HashSet::new();

    if backtrack(indexed, 0, 0, target, &mut path, &mut cache, 0) {
        Some(path)
    } else {
        None
    }
}

fn parallel_chunk_search(numbers: &[i64], target: i64, num_threads: usize) -> Option<Vec<usize>> {
    let result = Arc::new(Mutex::new(None));
    let numbers = Arc::new(numbers.to_vec());
    let mut handles = vec![];

    for thread_id in 0..num_threads {
        let result = Arc::clone(&result);
        let numbers = Arc::clone(&numbers);

        let handle = thread::spawn(move || {
            // Each thread tries a different subset approach
            let start = thread_id * numbers.len() / num_threads;
            let end = ((thread_id + 1) * numbers.len() / num_threads).min(numbers.len());

            if start >= end {
                return;
            }

            // Try DP on this chunk
            if let Some(indices) = dp_with_tracking(&numbers[start..end], target) {
                // Adjust indices to global positions
                let adjusted: Vec<usize> = indices.iter().map(|&i| i + start).collect();
                let mut res = result.lock().unwrap();
                if res.is_none() {
                    *res = Some(adjusted);
                }
            }
        });

        handles.push(handle);
    }

    for handle in handles {
        handle.join().unwrap();
    }

    let result = result.lock().unwrap();
    result.clone()
}
