#include <iostream>
#include <vector>
#include <cmath>
#include <algorithm>
#include <limits>

using namespace std;

struct Point {
    float x, y;
};

inline float distSq(const Point& a, const Point& b) {
    float dx = a.x - b.x;
    float dy = a.y - b.y;
    return dx * dx + dy * dy;
}

vector<int> nearestNeighbor(const vector<Point>& cities) {
    int n = cities.size();
    vector<int> tour;
    tour.reserve(n);
    vector<bool> visited(n, false);

    tour.push_back(0);
    visited[0] = true;

    for (int i = 1; i < n; i++) {
        int current = tour.back();
        float minDistSq = numeric_limits<float>::max();
        int nearest = -1;

        for (int j = 0; j < n; j++) {
            if (!visited[j]) {
                float dSq = distSq(cities[current], cities[j]);
                if (dSq < minDistSq) {
                    minDistSq = dSq;
                    nearest = j;
                }
            }
        }

        tour.push_back(nearest);
        visited[nearest] = true;
    }

    return tour;
}

void twoOpt(vector<int>& tour, const vector<Point>& cities, int maxIter) {
    int n = tour.size();

    for (int iter = 0; iter < maxIter; iter++) {
        bool improved = false;

        for (int i = 1; i < n - 2; i++) {
            for (int j = i + 1; j < n; j++) {
                float d1 = distSq(cities[tour[i-1]], cities[tour[i]]) +
                          distSq(cities[tour[j]], cities[tour[(j+1)%n]]);
                float d2 = distSq(cities[tour[i-1]], cities[tour[j]]) +
                          distSq(cities[tour[i]], cities[tour[(j+1)%n]]);

                if (d2 < d1) {
                    reverse(tour.begin() + i, tour.begin() + j + 1);
                    improved = true;
                }
            }
        }

        if (!improved) break;
    }
}

vector<int> angleSortTour(const vector<Point>& cities) {
    int n = cities.size();

    // Calculate centroid
    double cx = 0, cy = 0;
    for (const auto& p : cities) {
        cx += p.x;
        cy += p.y;
    }
    cx /= n;
    cy /= n;

    // Sort by angle from centroid
    vector<pair<float, int>> angles;
    angles.reserve(n);
    for (int i = 0; i < n; i++) {
        float angle = atan2f(cities[i].y - cy, cities[i].x - cx);
        angles.emplace_back(angle, i);
    }
    sort(angles.begin(), angles.end());

    // Find city 0 and build tour starting from it
    vector<int> tour;
    tour.reserve(n);
    int start = 0;
    for (int i = 0; i < n; i++) {
        if (angles[i].second == 0) {
            start = i;
            break;
        }
    }

    for (int i = 0; i < n; i++) {
        tour.push_back(angles[(start + i) % n].second);
    }

    return tour;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<Point> cities(n);
    for (int i = 0; i < n; i++) {
        cin >> cities[i].x >> cities[i].y;
    }

    vector<int> tour;

    if (n <= 500) {
        tour = nearestNeighbor(cities);
        twoOpt(tour, cities, 50);
    } else if (n <= 2000) {
        tour = nearestNeighbor(cities);
        twoOpt(tour, cities, 5);
    } else if (n <= 10000) {
        tour = nearestNeighbor(cities);
    } else {
        tour = angleSortTour(cities);
    }

    // Output
    for (int i = 0; i < n; i++) {
        if (i > 0) cout << " ";
        cout << tour[i];
    }
    cout << "\n";

    return 0;
}
