; SPIR-V Geometry Shader: Wireframe with quads for each edge
               OpCapability Geometry
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %color_in %gl_out %outColor
               OpExecutionMode %main InputTrianglesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 12
               OpExecutionMode %main Invocations 1

; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %color_in Location 0
               OpDecorate %outColor Location 0

; Types
       %void = OpTypeVoid
          %3 = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v2float = OpTypeVector %float 2
    %v3float = OpTypeVector %float 3
    %v4float = OpTypeVector %float 4
        %int = OpTypeInt 32 1
       %uint = OpTypeInt 32 0
     %uint_6 = OpConstant %uint 6
      %int_0 = OpConstant %int 0
      %int_1 = OpConstant %int 1
      %int_2 = OpConstant %int 2
      %int_4 = OpConstant %int 4
    %float_0 = OpConstant %float 0
  %float_0_5 = OpConstant %float 0.5
%float_0_008 = OpConstant %float 0.008
    %float_1 = OpConstant %float 1

; Input blocks
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_6
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
      %gl_in = OpVariable %ptr_in_arr_pv Input
%ptr_in_v4 = OpTypePointer Input %v4float

; Color input
%arr_color_in = OpTypeArray %v3float %uint_6
%ptr_in_arr_color = OpTypePointer Input %arr_color_in
   %color_in = OpVariable %ptr_in_arr_color Input
%ptr_in_v3 = OpTypePointer Input %v3float

; Output blocks
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
     %gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
   %outColor = OpVariable %ptr_out_v3 Output

; Main function
       %main = OpFunction %void None %3
      %label = OpLabel

; Load positions for main triangle vertices (0, 2, 4)
       %p0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
       %pos0 = OpLoad %v4float %p0_ptr
       %p2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
       %pos2 = OpLoad %v4float %p2_ptr
       %p4_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_4 %int_0
       %pos4 = OpLoad %v4float %p4_ptr

; Load colors for main triangle vertices
       %c0_ptr = OpAccessChain %ptr_in_v3 %color_in %int_0
       %col0 = OpLoad %v3float %c0_ptr
       %c2_ptr = OpAccessChain %ptr_in_v3 %color_in %int_2
       %col2 = OpLoad %v3float %c2_ptr
       %c4_ptr = OpAccessChain %ptr_in_v3 %color_in %int_4
       %col4 = OpLoad %v3float %c4_ptr

; ===== Edge 0→2 =====
; Average color for edge 0→2
    %col_02_sum = OpFAdd %v3float %col0 %col2
    %col_02 = OpVectorTimesScalar %v3float %col_02_sum %float_0_5

; Compute edge direction in screen space
    %pos0_xy = OpVectorShuffle %v2float %pos0 %pos0 0 1
    %pos2_xy = OpVectorShuffle %v2float %pos2 %pos2 0 1
    %dir_02 = OpFSub %v2float %pos2_xy %pos0_xy

; Compute perpendicular: perp = vec2(-dir.y, dir.x)
    %dir_02_y = OpCompositeExtract %float %dir_02 1
    %dir_02_x = OpCompositeExtract %float %dir_02 0
    %neg_dir_y = OpFNegate %float %dir_02_y
    %perp_02_raw = OpCompositeConstruct %v2float %neg_dir_y %dir_02_x

; Normalize perpendicular
    %perp_02_dot = OpDot %float %perp_02_raw %perp_02_raw
    %perp_02_len = OpExtInst %float %1 Sqrt %perp_02_dot
    %perp_02_norm = OpFDiv %v2float %perp_02_raw %perp_02_len

; Scale by thickness
    %perp_02 = OpVectorTimesScalar %v2float %perp_02_norm %float_0_008

; Emit 4 vertices for edge quad (triangle strip)
; Vertex 0: pos0 - perp
    %offset_0_neg = OpFNegate %v2float %perp_02
    %v0_xy = OpFAdd %v2float %pos0_xy %offset_0_neg
    %v0_pos = OpCompositeConstruct %v4float %v0_xy %float_0 %float_1
    %out_pos_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %out_pos_ptr %v0_pos
               OpStore %outColor %col_02
               OpEmitVertex

; Vertex 1: pos0 + perp
    %v1_xy = OpFAdd %v2float %pos0_xy %perp_02
    %v1_pos = OpCompositeConstruct %v4float %v1_xy %float_0 %float_1
               OpStore %out_pos_ptr %v1_pos
               OpStore %outColor %col_02
               OpEmitVertex

; Vertex 2: pos2 - perp
    %v2_xy = OpFAdd %v2float %pos2_xy %offset_0_neg
    %v2_pos = OpCompositeConstruct %v4float %v2_xy %float_0 %float_1
               OpStore %out_pos_ptr %v2_pos
               OpStore %outColor %col_02
               OpEmitVertex

; Vertex 3: pos2 + perp
    %v3_xy = OpFAdd %v2float %pos2_xy %perp_02
    %v3_pos = OpCompositeConstruct %v4float %v3_xy %float_0 %float_1
               OpStore %out_pos_ptr %v3_pos
               OpStore %outColor %col_02
               OpEmitVertex

               OpEndPrimitive

; ===== Edge 2→4 =====
; Average color for edge 2→4
    %col_24_sum = OpFAdd %v3float %col2 %col4
    %col_24 = OpVectorTimesScalar %v3float %col_24_sum %float_0_5

; Compute edge direction
    %pos4_xy = OpVectorShuffle %v2float %pos4 %pos4 0 1
    %dir_24 = OpFSub %v2float %pos4_xy %pos2_xy

; Compute perpendicular
    %dir_24_y = OpCompositeExtract %float %dir_24 1
    %dir_24_x = OpCompositeExtract %float %dir_24 0
    %neg_dir_24_y = OpFNegate %float %dir_24_y
    %perp_24_raw = OpCompositeConstruct %v2float %neg_dir_24_y %dir_24_x

; Normalize perpendicular
    %perp_24_dot = OpDot %float %perp_24_raw %perp_24_raw
    %perp_24_len = OpExtInst %float %1 Sqrt %perp_24_dot
    %perp_24_norm = OpFDiv %v2float %perp_24_raw %perp_24_len

; Scale by thickness
    %perp_24 = OpVectorTimesScalar %v2float %perp_24_norm %float_0_008

; Emit 4 vertices
    %offset_24_neg = OpFNegate %v2float %perp_24
    %v4_xy = OpFAdd %v2float %pos2_xy %offset_24_neg
    %v4_pos = OpCompositeConstruct %v4float %v4_xy %float_0 %float_1
               OpStore %out_pos_ptr %v4_pos
               OpStore %outColor %col_24
               OpEmitVertex

    %v5_xy = OpFAdd %v2float %pos2_xy %perp_24
    %v5_pos = OpCompositeConstruct %v4float %v5_xy %float_0 %float_1
               OpStore %out_pos_ptr %v5_pos
               OpStore %outColor %col_24
               OpEmitVertex

    %v6_xy = OpFAdd %v2float %pos4_xy %offset_24_neg
    %v6_pos = OpCompositeConstruct %v4float %v6_xy %float_0 %float_1
               OpStore %out_pos_ptr %v6_pos
               OpStore %outColor %col_24
               OpEmitVertex

    %v7_xy = OpFAdd %v2float %pos4_xy %perp_24
    %v7_pos = OpCompositeConstruct %v4float %v7_xy %float_0 %float_1
               OpStore %out_pos_ptr %v7_pos
               OpStore %outColor %col_24
               OpEmitVertex

               OpEndPrimitive

; ===== Edge 4→0 =====
; Average color for edge 4→0
    %col_40_sum = OpFAdd %v3float %col4 %col0
    %col_40 = OpVectorTimesScalar %v3float %col_40_sum %float_0_5

; Compute edge direction
    %dir_40 = OpFSub %v2float %pos0_xy %pos4_xy

; Compute perpendicular
    %dir_40_y = OpCompositeExtract %float %dir_40 1
    %dir_40_x = OpCompositeExtract %float %dir_40 0
    %neg_dir_40_y = OpFNegate %float %dir_40_y
    %perp_40_raw = OpCompositeConstruct %v2float %neg_dir_40_y %dir_40_x

; Normalize perpendicular
    %perp_40_dot = OpDot %float %perp_40_raw %perp_40_raw
    %perp_40_len = OpExtInst %float %1 Sqrt %perp_40_dot
    %perp_40_norm = OpFDiv %v2float %perp_40_raw %perp_40_len

; Scale by thickness
    %perp_40 = OpVectorTimesScalar %v2float %perp_40_norm %float_0_008

; Emit 4 vertices
    %offset_40_neg = OpFNegate %v2float %perp_40
    %v8_xy = OpFAdd %v2float %pos4_xy %offset_40_neg
    %v8_pos = OpCompositeConstruct %v4float %v8_xy %float_0 %float_1
               OpStore %out_pos_ptr %v8_pos
               OpStore %outColor %col_40
               OpEmitVertex

    %v9_xy = OpFAdd %v2float %pos4_xy %perp_40
    %v9_pos = OpCompositeConstruct %v4float %v9_xy %float_0 %float_1
               OpStore %out_pos_ptr %v9_pos
               OpStore %outColor %col_40
               OpEmitVertex

    %v10_xy = OpFAdd %v2float %pos0_xy %offset_40_neg
    %v10_pos = OpCompositeConstruct %v4float %v10_xy %float_0 %float_1
               OpStore %out_pos_ptr %v10_pos
               OpStore %outColor %col_40
               OpEmitVertex

    %v11_xy = OpFAdd %v2float %pos0_xy %perp_40
    %v11_pos = OpCompositeConstruct %v4float %v11_xy %float_0 %float_1
               OpStore %out_pos_ptr %v11_pos
               OpStore %outColor %col_40
               OpEmitVertex

               OpEndPrimitive

               OpReturn
               OpFunctionEnd
