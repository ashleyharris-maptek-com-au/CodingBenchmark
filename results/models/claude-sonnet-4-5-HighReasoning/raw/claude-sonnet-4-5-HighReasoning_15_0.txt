use std::io::{self, BufRead};

#[derive(Clone, Copy, Debug)]
struct Vec3 {
    x: f64,
    y: f64,
    z: f64,
}

impl Vec3 {
    fn new(x: f64, y: f64, z: f64) -> Self {
        Vec3 { x, y, z }
    }

    fn dot(&self, other: &Vec3) -> f64 {
        self.x * other.x + self.y * other.y + self.z * other.z
    }

    fn sub(&self, other: &Vec3) -> Vec3 {
        Vec3 {
            x: self.x - other.x,
            y: self.y - other.y,
            z: self.z - other.z,
        }
    }

    fn add(&self, other: &Vec3) -> Vec3 {
        Vec3 {
            x: self.x + other.x,
            y: self.y + other.y,
            z: self.z + other.z,
        }
    }
}

#[derive(Clone, Copy, Debug)]
struct Quaternion {
    w: f64,
    x: f64,
    y: f64,
    z: f64,
}

impl Quaternion {
    fn identity() -> Self {
        Quaternion { w: 1.0, x: 0.0, y: 0.0, z: 0.0 }
    }
}

struct Tetrahedron {
    position: Vec3,
    rotation: Quaternion,
    scale: f64,
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let nv: usize = lines.next().unwrap().unwrap().trim().parse().unwrap();

    let mut polygon: Vec<Vec3> = Vec::new();
    for _ in 0..nv {
        let line = lines.next().unwrap().unwrap();
        let parts: Vec<f64> = line.split_whitespace()
            .map(|s| s.parse().unwrap())
            .collect();
        polygon.push(Vec3::new(parts[0], parts[1], 0.0));
    }

    let sun_line = lines.next().unwrap().unwrap();
    let sun_parts: Vec<f64> = sun_line.split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();
    let sun_dir = Vec3::new(sun_parts[0], sun_parts[1], sun_parts[2]);

    let tetrahedrons = generate_solution(&polygon, &sun_dir, nv);

    println!("{}", tetrahedrons.len());
    for tet in tetrahedrons {
        println!("{} {} {} {} {} {} {} {}",
            tet.position.x, tet.position.y, tet.position.z,
            tet.rotation.w, tet.rotation.x, tet.rotation.y, tet.rotation.z,
            tet.scale);
    }
}

fn generate_solution(polygon: &[Vec3], sun_dir: &Vec3, nv: usize) -> Vec<Tetrahedron> {
    if nv <= 4 {
        generate_simple_solution(polygon, sun_dir)
    } else if nv <= 50 {
        generate_medium_solution(polygon, sun_dir)
    } else {
        generate_complex_solution(polygon, sun_dir)
    }
}

fn generate_simple_solution(polygon: &[Vec3], _sun_dir: &Vec3) -> Vec<Tetrahedron> {
    let centroid = compute_centroid(polygon);
    let max_dist = compute_max_distance(polygon, &centroid);

    let scale = (max_dist / 1.2).clamp(1.0, 10.0);
    let height = scale * 2.0;

    vec![Tetrahedron {
        position: Vec3::new(centroid.x, centroid.y, height),
        rotation: Quaternion::identity(),
        scale,
    }]
}

fn generate_medium_solution(polygon: &[Vec3], _sun_dir: &Vec3) -> Vec<Tetrahedron> {
    let triangles = triangulate(polygon);
    let mut tetrahedrons = Vec::new();

    for (i, tri) in triangles.iter().enumerate() {
        let centroid = Vec3::new(
            (tri[0].x + tri[1].x + tri[2].x) / 3.0,
            (tri[0].y + tri[1].y + tri[2].y) / 3.0,
            0.0,
        );

        let max_dist = tri.iter()
            .map(|v| ((v.x - centroid.x).powi(2) + (v.y - centroid.y).powi(2)).sqrt())
            .fold(0.0f64, f64::max);

        let scale = (max_dist / 0.8).clamp(0.5, 10.0);
        let height = 2.0 + (i as f64 * 0.8) % 4.0;

        tetrahedrons.push(Tetrahedron {
            position: Vec3::new(centroid.x, centroid.y, height),
            rotation: Quaternion::identity(),
            scale,
        });
    }

    tetrahedrons
}

fn generate_complex_solution(polygon: &[Vec3], _sun_dir: &Vec3) -> Vec<Tetrahedron> {
    let triangles = triangulate(polygon);
    let mut tetrahedrons = Vec::new();
    let grid_size = (triangles.len() as f64).sqrt().ceil() as usize;

    for (i, tri) in triangles.iter().enumerate() {
        let centroid = Vec3::new(
            (tri[0].x + tri[1].x + tri[2].x) / 3.0,
            (tri[0].y + tri[1].y + tri[2].y) / 3.0,
            0.0,
        );

        let max_dist = tri.iter()
            .map(|v| ((v.x - centroid.x).powi(2) + (v.y - centroid.y).powi(2)).sqrt())
            .fold(0.0f64, f64::max);

        let scale = (max_dist / 0.8).clamp(0.5, 10.0);
        let layer = (i / grid_size) % 5;
        let height = 1.5 + layer as f64 * 0.5;

        tetrahedrons.push(Tetrahedron {
            position: Vec3::new(centroid.x, centroid.y, height),
            rotation: Quaternion::identity(),
            scale,
        });
    }

    tetrahedrons
}

fn compute_centroid(polygon: &[Vec3]) -> Vec3 {
    let mut sum = Vec3::new(0.0, 0.0, 0.0);
    for v in polygon {
        sum = sum.add(v);
    }
    Vec3::new(sum.x / polygon.len() as f64, sum.y / polygon.len() as f64, 0.0)
}

fn compute_max_distance(polygon: &[Vec3], center: &Vec3) -> f64 {
    polygon.iter()
        .map(|v| ((v.x - center.x).powi(2) + (v.y - center.y).powi(2)).sqrt())
        .fold(0.0f64, f64::max)
}

fn triangulate(polygon: &[Vec3]) -> Vec<[Vec3; 3]> {
    let mut triangles = Vec::new();
    let mut remaining: Vec<Vec3> = polygon.to_vec();

    while remaining.len() > 3 {
        let n = remaining.len();
        let mut found_ear = false;

        for i in 0..n {
            let prev = remaining[(i + n - 1) % n];
            let curr = remaining[i];
            let next = remaining[(i + 1) % n];

            if is_ear(&prev, &curr, &next, &remaining) {
                triangles.push([prev, curr, next]);
                remaining.remove(i);
                found_ear = true;
                break;
            }
        }

        if !found_ear {
            if remaining.len() >= 3 {
                triangles.push([remaining[0], remaining[1], remaining[2]]);
                remaining.remove(1);
            } else {
                break;
            }
        }
    }

    if remaining.len() == 3 {
        triangles.push([remaining[0], remaining[1], remaining[2]]);
    }

    triangles
}

fn is_ear(prev: &Vec3, curr: &Vec3, next: &Vec3, polygon: &[Vec3]) -> bool {
    let v1 = curr.sub(prev);
    let v2 = next.sub(curr);
    let cross_z = v1.x * v2.y - v1.y * v2.x;

    if cross_z <= 0.0 {
        return false;
    }

    for p in polygon {
        if (p.x - prev.x).abs() < 1e-9 && (p.y - prev.y).abs() < 1e-9 {
            continue;
        }
        if (p.x - curr.x).abs() < 1e-9 && (p.y - curr.y).abs() < 1e-9 {
            continue;
        }
        if (p.x - next.x).abs() < 1e-9 && (p.y - next.y).abs() < 1e-9 {
            continue;
        }

        if point_in_triangle(p, prev, curr, next) {
            return false;
        }
    }

    true
}

fn point_in_triangle(p: &Vec3, a: &Vec3, b: &Vec3, c: &Vec3) -> bool {
    let v0 = Vec3::new(c.x - a.x, c.y - a.y, 0.0);
    let v1 = Vec3::new(b.x - a.x, b.y - a.y, 0.0);
    let v2 = Vec3::new(p.x - a.x, p.y - a.y, 0.0);

    let dot00 = v0.dot(&v0);
    let dot01 = v0.dot(&v1);
    let dot02 = v0.dot(&v2);
    let dot11 = v1.dot(&v1);
    let dot12 = v1.dot(&v2);

    let denom = dot00 * dot11 - dot01 * dot01;
    if denom.abs() < 1e-9 {
        return false;
    }

    let inv_denom = 1.0 / denom;
    let u = (dot11 * dot02 - dot01 * dot12) * inv_denom;
    let v = (dot00 * dot12 - dot01 * dot02) * inv_denom;

    u >= -1e-9 && v >= -1e-9 && u + v <= 1.0 + 1e-9
}
