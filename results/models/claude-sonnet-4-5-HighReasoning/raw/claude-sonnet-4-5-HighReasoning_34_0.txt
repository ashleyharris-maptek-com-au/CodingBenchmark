import sys
import math

def euclidean_distance(p1, p2):
    return math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

def calculate_route_distance(route, depot, customers):
    if not route:
        return 0
    distance = euclidean_distance(depot, customers[route[0]][:2])
    for i in range(len(route) - 1):
        distance += euclidean_distance(customers[route[i]][:2], customers[route[i+1]][:2])
    distance += euclidean_distance(customers[route[-1]][:2], depot)
    return distance

def clarke_wright_savings(depot, customers, capacity, max_vehicles):
    n = len(customers)

    # Calculate all distances from depot
    depot_distances = [euclidean_distance(depot, customers[i][:2]) for i in range(n)]

    # Calculate savings for all pairs
    savings = []
    for i in range(n):
        for j in range(i + 1, n):
            dist_ij = euclidean_distance(customers[i][:2], customers[j][:2])
            saving = depot_distances[i] + depot_distances[j] - dist_ij
            savings.append((saving, i, j))

    # Sort by savings in descending order
    savings.sort(reverse=True, key=lambda x: x[0])

    # Initialize routes - each customer in its own route
    routes = [[i] for i in range(n)]
    route_demands = [customers[i][2] for i in range(n)]
    route_map = {i: i for i in range(n)}  # maps customer to route index

    # Merge routes based on savings
    for saving, i, j in savings:
        route_i = route_map[i]
        route_j = route_map[j]

        if route_i == route_j:
            continue

        # Check if i is at the end of its route and j is at the start of its route (or vice versa)
        route_i_list = routes[route_i]
        route_j_list = routes[route_j]

        if route_i_list is None or route_j_list is None:
            continue

        # Check capacity constraint
        if route_demands[route_i] + route_demands[route_j] > capacity:
            continue

        # Check if we can merge (i and j must be at the ends of their routes)
        can_merge = False
        reverse_i = False
        reverse_j = False

        if route_i_list[-1] == i and route_j_list[0] == j:
            can_merge = True
        elif route_i_list[0] == i and route_j_list[-1] == j:
            can_merge = True
            reverse_i = True
            reverse_j = True
        elif route_i_list[-1] == i and route_j_list[-1] == j:
            can_merge = True
            reverse_j = True
        elif route_i_list[0] == i and route_j_list[0] == j:
            can_merge = True
            reverse_i = True

        if not can_merge:
            continue

        # Merge routes
        if reverse_i:
            route_i_list.reverse()
        if reverse_j:
            route_j_list.reverse()

        new_route = route_i_list + route_j_list
        new_demand = route_demands[route_i] + route_demands[route_j]

        routes[route_i] = new_route
        route_demands[route_i] = new_demand
        routes[route_j] = None
        route_demands[route_j] = 0

        # Update route_map
        for customer in route_j_list:
            route_map[customer] = route_i

    # Collect valid routes
    final_routes = [route for route in routes if route is not None]

    return final_routes

def solve_cvrp():
    lines = sys.stdin.read().strip().split('\n')

    # Parse first line
    first_line = lines[0].split()
    n = int(first_line[0])
    k = int(first_line[1])
    capacity = int(first_line[2])

    # Parse depot
    depot_line = lines[1].split()
    depot = (float(depot_line[0]), float(depot_line[1]))

    # Parse customers
    customers = []
    for i in range(2, n + 2):
        parts = lines[i].split()
        x = float(parts[0])
        y = float(parts[1])
        demand = int(parts[2])
        customers.append((x, y, demand))

    # Solve using Clarke-Wright
    routes = clarke_wright_savings(depot, customers, capacity, k)

    # Calculate total distance
    total_distance = sum(calculate_route_distance(route, depot, customers) for route in routes)

    # Output results
    print(f"{total_distance:.6f}")
    print(len(routes))
    for route in routes:
        print(' '.join(map(str, route)))

if __name__ == "__main__":
    solve_cvrp()
