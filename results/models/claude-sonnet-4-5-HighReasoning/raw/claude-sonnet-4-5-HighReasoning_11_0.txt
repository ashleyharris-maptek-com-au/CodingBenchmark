use std::io::{self, BufRead};
use std::f64::consts::PI;

#[derive(Clone, Copy, Debug)]
struct Point {
    x: f64,
    y: f64,
}

#[derive(Clone, Debug)]
struct Polygon {
    vertices: Vec<Point>,
}

#[derive(Clone, Copy, Debug)]
struct BBox {
    min_x: f64,
    max_x: f64,
    min_y: f64,
    max_y: f64,
}

#[derive(Clone, Debug)]
struct Placement {
    stock_index: usize,
    x: f64,
    y: f64,
    rotation: f64,
}

#[derive(Clone, Debug)]
struct PlacedPiece {
    polygon: Polygon,
    bbox: BBox,
}

impl Point {
    fn new(x: f64, y: f64) -> Self {
        Point { x, y }
    }

    fn rotate(&self, angle: f64) -> Point {
        let cos_a = angle.cos();
        let sin_a = angle.sin();
        Point {
            x: self.x * cos_a - self.y * sin_a,
            y: self.x * sin_a + self.y * cos_a,
        }
    }

    fn translate(&self, dx: f64, dy: f64) -> Point {
        Point {
            x: self.x + dx,
            y: self.y + dy,
        }
    }
}

impl Polygon {
    fn new(vertices: Vec<Point>) -> Self {
        Polygon { vertices }
    }

    fn bbox(&self) -> BBox {
        let mut min_x = f64::INFINITY;
        let mut max_x = f64::NEG_INFINITY;
        let mut min_y = f64::INFINITY;
        let mut max_y = f64::NEG_INFINITY;

        for p in &self.vertices {
            min_x = min_x.min(p.x);
            max_x = max_x.max(p.x);
            min_y = min_y.min(p.y);
            max_y = max_y.max(p.y);
        }

        BBox { min_x, max_x, min_y, max_y }
    }

    fn transform(&self, rotation: f64, tx: f64, ty: f64) -> Polygon {
        let vertices = self.vertices.iter()
            .map(|p| p.rotate(rotation).translate(tx, ty))
            .collect();
        Polygon { vertices }
    }

    fn contains_point(&self, point: &Point) -> bool {
        let mut inside = false;
        let n = self.vertices.len();

        for i in 0..n {
            let j = (i + 1) % n;
            let vi = &self.vertices[i];
            let vj = &self.vertices[j];

            if ((vi.y > point.y) != (vj.y > point.y)) &&
               (point.x < (vj.x - vi.x) * (point.y - vi.y) / (vj.y - vi.y) + vi.x) {
                inside = !inside;
            }
        }

        inside
    }

    fn fits_inside(&self, container: &Polygon) -> bool {
        self.vertices.iter().all(|p| container.contains_point(p))
    }

    fn intersects(&self, other: &Polygon) -> bool {
        // Check if any vertex of one is inside the other
        for v in &self.vertices {
            if other.contains_point(v) {
                return true;
            }
        }

        for v in &other.vertices {
            if self.contains_point(v) {
                return true;
            }
        }

        // Check edge intersections
        for i in 0..self.vertices.len() {
            let j = (i + 1) % self.vertices.len();
            for k in 0..other.vertices.len() {
                let l = (k + 1) % other.vertices.len();
                if segments_intersect(
                    &self.vertices[i], &self.vertices[j],
                    &other.vertices[k], &other.vertices[l]
                ) {
                    return true;
                }
            }
        }

        false
    }

    fn area(&self) -> f64 {
        let mut area = 0.0;
        let n = self.vertices.len();
        for i in 0..n {
            let j = (i + 1) % n;
            area += self.vertices[i].x * self.vertices[j].y;
            area -= self.vertices[j].x * self.vertices[i].y;
        }
        area.abs() / 2.0
    }

    fn centroid(&self) -> Point {
        let mut cx = 0.0;
        let mut cy = 0.0;
        for v in &self.vertices {
            cx += v.x;
            cy += v.y;
        }
        let n = self.vertices.len() as f64;
        Point::new(cx / n, cy / n)
    }
}

impl BBox {
    fn intersects(&self, other: &BBox) -> bool {
        !(self.max_x < other.min_x || other.max_x < self.min_x ||
          self.max_y < other.min_y || other.max_y < self.min_y)
    }

    fn width(&self) -> f64 {
        self.max_x - self.min_x
    }

    fn height(&self) -> f64 {
        self.max_y - self.min_y
    }
}

fn segments_intersect(p1: &Point, p2: &Point, p3: &Point, p4: &Point) -> bool {
    let d1 = direction(p3, p4, p1);
    let d2 = direction(p3, p4, p2);
    let d3 = direction(p1, p2, p3);
    let d4 = direction(p1, p2, p4);

    if ((d1 > 0.0 && d2 < 0.0) || (d1 < 0.0 && d2 > 0.0)) &&
       ((d3 > 0.0 && d4 < 0.0) || (d3 < 0.0 && d4 > 0.0)) {
        return true;
    }

    if d1.abs() < 1e-10 && on_segment(p3, p1, p4) { return true; }
    if d2.abs() < 1e-10 && on_segment(p3, p2, p4) { return true; }
    if d3.abs() < 1e-10 && on_segment(p1, p3, p2) { return true; }
    if d4.abs() < 1e-10 && on_segment(p1, p4, p2) { return true; }

    false
}

fn direction(p1: &Point, p2: &Point, p3: &Point) -> f64 {
    (p3.x - p1.x) * (p2.y - p1.y) - (p2.x - p1.x) * (p3.y - p1.y)
}

fn on_segment(p1: &Point, p2: &Point, p3: &Point) -> bool {
    p2.x >= p1.x.min(p3.x) && p2.x <= p1.x.max(p3.x) &&
    p2.y >= p1.y.min(p3.y) && p2.y <= p1.y.max(p3.y)
}

struct Stock {
    polygon: Polygon,
    placed_pieces: Vec<PlacedPiece>,
}

impl Stock {
    fn new(polygon: Polygon) -> Self {
        Stock {
            polygon,
            placed_pieces: Vec::new(),
        }
    }

    fn try_place(&mut self, piece: &Polygon, rotation: f64, x: f64, y: f64) -> bool {
        let transformed = piece.transform(rotation, x, y);

        // Quick bbox check
        let bbox = transformed.bbox();
        let stock_bbox = self.polygon.bbox();
        if bbox.min_x < stock_bbox.min_x || bbox.max_x > stock_bbox.max_x ||
           bbox.min_y < stock_bbox.min_y || bbox.max_y > stock_bbox.max_y {
            return false;
        }

        // Check if fits in stock
        if !transformed.fits_inside(&self.polygon) {
            return false;
        }

        // Check for overlaps with existing pieces (use bbox first)
        for placed in &self.placed_pieces {
            if bbox.intersects(&placed.bbox) {
                if transformed.intersects(&placed.polygon) {
                    return false;
                }
            }
        }

        // Place the piece
        self.placed_pieces.push(PlacedPiece {
            polygon: transformed,
            bbox,
        });

        true
    }
}

fn solve(stock_polygon: Polygon, pieces: Vec<Polygon>) -> Vec<Placement> {
    let num_pieces = pieces.len();

    // Adaptive parameters based on problem size
    let (rotation_angles, grid_step, max_tries) = if num_pieces < 50 {
        // Small problems: high quality
        (vec![0.0, PI/6.0, PI/4.0, PI/3.0, PI/2.0, 2.0*PI/3.0, 3.0*PI/4.0, 5.0*PI/6.0,
              PI, 7.0*PI/6.0, 5.0*PI/4.0, 4.0*PI/3.0, 3.0*PI/2.0, 5.0*PI/3.0, 7.0*PI/4.0, 11.0*PI/6.0],
         2.0, 200)
    } else if num_pieces < 500 {
        // Medium problems: balanced
        (vec![0.0, PI/4.0, PI/2.0, 3.0*PI/4.0, PI, 5.0*PI/4.0, 3.0*PI/2.0, 7.0*PI/4.0],
         5.0, 100)
    } else if num_pieces < 2000 {
        // Large problems: fast
        (vec![0.0, PI/2.0, PI, 3.0*PI/2.0],
         10.0, 50)
    } else {
        // Very large problems: very fast
        (vec![0.0, PI/2.0],
         20.0, 25)
    };

    // Sort pieces by area (largest first)
    let mut indexed_pieces: Vec<(usize, &Polygon, f64)> = pieces.iter()
        .enumerate()
        .map(|(i, p)| (i, p, p.area()))
        .collect();
    indexed_pieces.sort_by(|a, b| b.2.partial_cmp(&a.2).unwrap());

    let mut stocks = vec![Stock::new(stock_polygon.clone())];
    let mut piece_placements = vec![None; num_pieces];
    let stock_bbox = stock_polygon.bbox();

    for (original_idx, piece, _) in indexed_pieces {
        let mut placed = false;

        // Try to place in existing stocks first
        'stock_loop: for (stock_idx, stock) in stocks.iter_mut().enumerate() {
            for &rotation in &rotation_angles {
                // Get centroid offset for this rotation
                let rotated_piece = piece.transform(rotation, 0.0, 0.0);
                let centroid = rotated_piece.centroid();
                let bbox = rotated_piece.bbox();

                // Calculate adjustment to align piece origin
                let offset_x = -bbox.min_x;
                let offset_y = -bbox.min_y;

                // Try positions on a grid
                let mut attempts = 0;
                let mut y = stock_bbox.min_y;
                while y <= stock_bbox.max_y && attempts < max_tries {
                    let mut x = stock_bbox.min_x;
                    while x <= stock_bbox.max_x && attempts < max_tries {
                        attempts += 1;

                        let try_x = x + offset_x;
                        let try_y = y + offset_y;

                        if stock.try_place(piece, rotation, try_x, try_y) {
                            piece_placements[original_idx] = Some(Placement {
                                stock_index: stock_idx,
                                x: try_x,
                                y: try_y,
                                rotation,
                            });
                            placed = true;
                            break 'stock_loop;
                        }

                        x += grid_step;
                    }
                    y += grid_step;
                }
            }
        }

        // If not placed in existing stocks, create new stock
        if !placed {
            let mut new_stock = Stock::new(stock_polygon.clone());

            for &rotation in &rotation_angles {
                let rotated_piece = piece.transform(rotation, 0.0, 0.0);
                let bbox = rotated_piece.bbox();
                let offset_x = -bbox.min_x;
                let offset_y = -bbox.min_y;

                // Try bottom-left corner first
                let positions = vec![
                    (stock_bbox.min_x + offset_x, stock_bbox.min_y + offset_y),
                    (stock_bbox.min_x + offset_x, stock_bbox.min_y + offset_y + 10.0),
                    (stock_bbox.min_x + offset_x + 10.0, stock_bbox.min_y + offset_y),
                ];

                for (try_x, try_y) in positions {
                    if new_stock.try_place(piece, rotation, try_x, try_y) {
                        piece_placements[original_idx] = Some(Placement {
                            stock_index: stocks.len(),
                            x: try_x,
                            y: try_y,
                            rotation,
                        });
                        placed = true;
                        break;
                    }
                }

                if placed {
                    break;
                }
            }

            stocks.push(new_stock);
        }

        // Fallback: place at origin with no rotation
        if !placed {
            eprintln!("Warning: Fallback placement for piece {}", original_idx);
            piece_placements[original_idx] = Some(Placement {
                stock_index: stocks.len(),
                x: stock_bbox.min_x,
                y: stock_bbox.min_y,
                rotation: 0.0,
            });
        }
    }

    piece_placements.into_iter().map(|p| p.unwrap()).collect()
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Read first line
    let first_line = lines.next().unwrap().unwrap();
    let parts: Vec<usize> = first_line.split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();
    let (sv, np) = (parts[0], parts[1]);

    // Read stock polygon
    let mut stock_vertices = Vec::new();
    for _ in 0..sv {
        let line = lines.next().unwrap().unwrap();
        let coords: Vec<f64> = line.split_whitespace()
            .map(|s| s.parse().unwrap())
            .collect();
        stock_vertices.push(Point::new(coords[0], coords[1]));
    }
    let stock_polygon = Polygon::new(stock_vertices);

    // Read pieces
    let mut pieces = Vec::new();
    for _ in 0..np {
        let pv_line = lines.next().unwrap().unwrap();
        let pv: usize = pv_line.trim().parse().unwrap();

        let mut piece_vertices = Vec::new();
        for _ in 0..pv {
            let line = lines.next().unwrap().unwrap();
            let coords: Vec<f64> = line.split_whitespace()
                .map(|s| s.parse().unwrap())
                .collect();
            piece_vertices.push(Point::new(coords[0], coords[1]));
        }
        pieces.push(Polygon::new(piece_vertices));
    }

    // Solve
    let placements = solve(stock_polygon, pieces);

    // Find max stock index used
    let num_stocks = placements.iter().map(|p| p.stock_index).max().unwrap_or(0) + 1;

    // Output
    println!("{}", num_stocks);
    for placement in placements {
        println!("{} {} {} {}",
            placement.stock_index,
            placement.x,
            placement.y,
            placement.rotation);
    }
}
