use std::io::{self, BufRead};
use std::f64::consts::PI;

struct Graph {
    n: usize,
    edges: Vec<(usize, usize)>,
    adj: Vec<Vec<usize>>,
}

impl Graph {
    fn new(n: usize, edges: Vec<(usize, usize)>) -> Self {
        let mut adj = vec![Vec::new(); n];
        for &(u, v) in &edges {
            if u < n && v < n && u != v {
                adj[u].push(v);
                adj[v].push(u);
            }
        }
        Graph { n, edges, adj }
    }

    fn find_components(&self) -> Vec<Vec<usize>> {
        let mut visited = vec![false; self.n];
        let mut components = Vec::new();

        for start in 0..self.n {
            if !visited[start] {
                let mut component = Vec::new();
                let mut stack = vec![start];
                visited[start] = true;

                while let Some(node) = stack.pop() {
                    component.push(node);
                    for &neighbor in &self.adj[node] {
                        if !visited[neighbor] {
                            visited[neighbor] = true;
                            stack.push(neighbor);
                        }
                    }
                }
                components.push(component);
            }
        }
        components
    }
}

struct Layout {
    positions: Vec<(f64, f64)>,
}

impl Layout {
    fn new(n: usize) -> Self {
        Layout {
            positions: vec![(0.0, 0.0); n],
        }
    }

    fn initialize_circular(&mut self, nodes: &[usize], center: (f64, f64), radius: f64) {
        let n = nodes.len();
        if n == 1 {
            self.positions[nodes[0]] = center;
        } else {
            for (i, &node) in nodes.iter().enumerate() {
                let angle = 2.0 * PI * i as f64 / n as f64;
                self.positions[node] = (
                    center.0 + radius * angle.cos(),
                    center.1 + radius * angle.sin(),
                );
            }
        }
    }

    fn fruchterman_reingold(
        &mut self,
        graph: &Graph,
        nodes: &[usize],
        iterations: usize,
        area: f64,
    ) {
        let k = (area / nodes.len() as f64).sqrt();
        let k2 = k * k;
        let mut temp = k * 2.0;
        let cooling = (0.01_f64 / temp).powf(1.0 / iterations as f64);

        for iter in 0..iterations {
            let mut forces = vec![(0.0, 0.0); graph.n];

            // Repulsive forces between all pairs (or sampled pairs for large graphs)
            if nodes.len() < 500 {
                // Full O(n²) for small graphs
                for i in 0..nodes.len() {
                    for j in (i + 1)..nodes.len() {
                        let u = nodes[i];
                        let v = nodes[j];
                        let dx = self.positions[u].0 - self.positions[v].0;
                        let dy = self.positions[u].1 - self.positions[v].1;
                        let dist2 = dx * dx + dy * dy;
                        let dist = dist2.sqrt().max(0.01);
                        let force = k2 / dist;
                        let fx = (dx / dist) * force;
                        let fy = (dy / dist) * force;
                        forces[u].0 += fx;
                        forces[u].1 += fy;
                        forces[v].0 -= fx;
                        forces[v].1 -= fy;
                    }
                }
            } else {
                // Limited repulsion for larger graphs
                let sample_size = (500.0 * (1.0 - iter as f64 / iterations as f64)).max(50.0) as usize;
                for &u in nodes {
                    let mut count = 0;
                    for &v in nodes {
                        if u == v || count >= sample_size {
                            continue;
                        }
                        count += 1;
                        let dx = self.positions[u].0 - self.positions[v].0;
                        let dy = self.positions[u].1 - self.positions[v].1;
                        let dist2 = dx * dx + dy * dy;
                        if dist2 < k * k * 16.0 {
                            let dist = dist2.sqrt().max(0.01);
                            let force = k2 / dist;
                            let fx = (dx / dist) * force;
                            let fy = (dy / dist) * force;
                            forces[u].0 += fx;
                            forces[u].1 += fy;
                        }
                    }
                }
            }

            // Attractive forces along edges
            for &(u, v) in &graph.edges {
                if u == v || u >= graph.n || v >= graph.n {
                    continue;
                }
                let dx = self.positions[u].0 - self.positions[v].0;
                let dy = self.positions[u].1 - self.positions[v].1;
                let dist = (dx * dx + dy * dy).sqrt().max(0.01);
                let force = dist * dist / k;
                let fx = (dx / dist) * force;
                let fy = (dy / dist) * force;
                forces[u].0 -= fx;
                forces[u].1 -= fy;
                forces[v].0 += fx;
                forces[v].1 += fy;
            }

            // Apply forces with temperature-based displacement limit
            for &node in nodes {
                let fx = forces[node].0;
                let fy = forces[node].1;
                let force_mag = (fx * fx + fy * fy).sqrt();
                if force_mag > 0.001 {
                    let displacement = force_mag.min(temp);
                    self.positions[node].0 += (fx / force_mag) * displacement;
                    self.positions[node].1 += (fy / force_mag) * displacement;
                }
            }

            temp *= cooling;
        }
    }

    fn layout_small(&mut self, graph: &Graph) {
        // For small to medium graphs, use proper force-directed layout per component
        let components = graph.find_components();
        let mut offset_x = 0.0;
        let area_per_node = 10000.0;

        for component in components {
            let comp_size = component.len();
            let area = area_per_node * comp_size as f64;
            let width = area.sqrt();

            // Initialize circular layout
            self.initialize_circular(&component, (offset_x + width / 2.0, width / 2.0), width / 3.0);

            // Adaptive iterations based on component size
            let iterations = if comp_size < 20 {
                800
            } else if comp_size < 50 {
                500
            } else if comp_size < 200 {
                300
            } else if comp_size < 1000 {
                150
            } else {
                50
            };

            // Run force-directed layout
            self.fruchterman_reingold(graph, &component, iterations, area);

            offset_x += width * 1.2;
        }
    }

    fn layout_large(&mut self, graph: &Graph) {
        // For very large graphs (10k+ nodes), use fast grid initialization + minimal force
        let grid_size = (graph.n as f64).sqrt().ceil() as usize;

        // Grid initialization with jitter
        for i in 0..graph.n {
            let row = i / grid_size;
            let col = i % grid_size;
            let jitter_x = ((i * 7919) % 1000) as f64 / 1000.0 - 0.5;
            let jitter_y = ((i * 7823) % 1000) as f64 / 1000.0 - 0.5;
            self.positions[i] = (
                col as f64 * 50.0 + jitter_x * 20.0,
                row as f64 * 50.0 + jitter_y * 20.0,
            );
        }

        // Quick attractive-only iterations
        let k = 50.0;
        for _ in 0..3 {
            let mut forces = vec![(0.0, 0.0); graph.n];

            for &(u, v) in &graph.edges {
                if u == v || u >= graph.n || v >= graph.n {
                    continue;
                }
                let dx = self.positions[u].0 - self.positions[v].0;
                let dy = self.positions[u].1 - self.positions[v].1;
                let dist = (dx * dx + dy * dy).sqrt().max(0.01);
                let force = (dist - k).min(k * 0.5);
                let fx = (dx / dist) * force * 0.05;
                let fy = (dy / dist) * force * 0.05;
                forces[u].0 -= fx;
                forces[u].1 -= fy;
                forces[v].0 += fx;
                forces[v].1 += fy;
            }

            for i in 0..graph.n {
                self.positions[i].0 += forces[i].0;
                self.positions[i].1 += forces[i].1;
            }
        }
    }

    fn layout_huge(&mut self, graph: &Graph) {
        // For extremely large graphs (50k+ nodes), minimal computation
        let grid_size = (graph.n as f64).sqrt().ceil() as usize;

        for i in 0..graph.n {
            let row = i / grid_size;
            let col = i % grid_size;
            self.positions[i] = (col as f64 * 10.0, row as f64 * 10.0);
        }

        // Single spring-only pass
        let mut forces = vec![(0.0, 0.0); graph.n];
        for &(u, v) in &graph.edges {
            if u == v || u >= graph.n || v >= graph.n {
                continue;
            }
            let dx = self.positions[u].0 - self.positions[v].0;
            let dy = self.positions[u].1 - self.positions[v].1;
            let fx = dx * 0.01;
            let fy = dy * 0.01;
            forces[u].0 -= fx;
            forces[u].1 -= fy;
            forces[v].0 += fx;
            forces[v].1 += fy;
        }

        for i in 0..graph.n {
            self.positions[i].0 += forces[i].0;
            self.positions[i].1 += forces[i].1;
        }
    }

    fn compute_layout(&mut self, graph: &Graph) {
        if graph.n < 5000 {
            self.layout_small(graph);
        } else if graph.n < 50000 {
            self.layout_large(graph);
        } else {
            self.layout_huge(graph);
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Read first line: N M
    let first_line = lines.next().unwrap().unwrap();
    let parts: Vec<usize> = first_line
        .split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();
    let n = parts[0];
    let m = parts[1];

    // Read edges
    let mut edges = Vec::with_capacity(m);
    for _ in 0..m {
        if let Some(Ok(line)) = lines.next() {
            let parts: Vec<usize> = line
                .split_whitespace()
                .map(|s| s.parse().unwrap())
                .collect();
            if parts.len() >= 2 {
                edges.push((parts[0], parts[1]));
            }
        }
    }

    // Build graph
    let graph = Graph::new(n, edges);

    // Compute layout
    let mut layout = Layout::new(n);
    layout.compute_layout(&graph);

    // Output results
    println!("{}", n);
    for i in 0..n {
        println!("{} {}", layout.positions[i].0, layout.positions[i].1);
    }
}
