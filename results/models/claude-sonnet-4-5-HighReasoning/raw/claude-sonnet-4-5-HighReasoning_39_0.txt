use std::io::{self, BufRead};
use std::collections::HashSet;

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = lines.next().unwrap().unwrap();
    let parts: Vec<usize> = first_line.split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();
    let n = parts[0];
    let m = parts[1];

    let mut edges = Vec::new();

    for _ in 0..m {
        let line = lines.next().unwrap().unwrap();
        let edge: Vec<usize> = line.split_whitespace()
            .map(|s| s.parse().unwrap())
            .collect();
        edges.push((edge[0], edge[1]));
    }

    let (cut_size, mut partition) = solve(n, &edges);
    partition.sort();

    println!("{}", cut_size);
    for (i, &v) in partition.iter().enumerate() {
        if i > 0 {
            print!(" ");
        }
        print!("{}", v);
    }
    println!();
}

fn solve(n: usize, edges: &[(usize, usize)]) -> (usize, Vec<usize>) {
    let half = n / 2;

    if n <= 20 {
        brute_force(n, half, edges)
    } else {
        heuristic_search(n, half, edges)
    }
}

fn brute_force(n: usize, half: usize, edges: &[(usize, usize)]) -> (usize, Vec<usize>) {
    let mut best_cut = usize::MAX;
    let mut best_partition = Vec::new();

    fn generate(
        n: usize,
        half: usize,
        edges: &[(usize, usize)],
        current: &mut Vec<usize>,
        start: usize,
        best_cut: &mut usize,
        best_partition: &mut Vec<usize>,
    ) {
        if current.len() == half {
            let cut = count_cut(current, edges);
            if cut < *best_cut {
                *best_cut = cut;
                *best_partition = current.clone();
            }
            return;
        }

        for v in start..n {
            current.push(v);
            generate(n, half, edges, current, v + 1, best_cut, best_partition);
            current.pop();
        }
    }

    let mut current = Vec::new();
    generate(n, half, edges, &mut current, 0, &mut best_cut, &mut best_partition);

    (best_cut, best_partition)
}

fn count_cut(partition: &[usize], edges: &[(usize, usize)]) -> usize {
    let set: HashSet<usize> = partition.iter().copied().collect();
    edges.iter()
        .filter(|(u, v)| set.contains(u) != set.contains(v))
        .count()
}

fn heuristic_search(n: usize, half: usize, edges: &[(usize, usize)]) -> (usize, Vec<usize>) {
    let mut best_cut = usize::MAX;
    let mut best_partition = Vec::new();

    // Try many random starting points
    for seed in 0..10000 {
        let mut partition = initial_partition(n, half, seed);

        // Local optimization using hill climbing
        loop {
            if !local_improve(&mut partition, n, half, edges) {
                break;
            }
        }

        let cut = count_cut(&partition, edges);
        if cut < best_cut {
            best_cut = cut;
            best_partition = partition;
        }
    }

    (best_cut, best_partition)
}

fn initial_partition(n: usize, half: usize, seed: usize) -> Vec<usize> {
    let mut vertices: Vec<usize> = (0..n).collect();

    // Simple pseudo-random shuffle using LCG
    let mut state = seed.wrapping_mul(1103515245).wrapping_add(12345);
    for i in 0..n-1 {
        state = state.wrapping_mul(1664525).wrapping_add(1013904223);
        let j = i + (state % (n - i));
        vertices.swap(i, j);
    }

    vertices[..half].to_vec()
}

fn local_improve(partition: &mut Vec<usize>, n: usize, half: usize, edges: &[(usize, usize)]) -> bool {
    let current_cut = count_cut(partition, edges);
    let partition_set: HashSet<usize> = partition.iter().copied().collect();
    let other: Vec<usize> = (0..n).filter(|v| !partition_set.contains(v)).collect();

    // Try all pairwise swaps
    for i in 0..half {
        for j in 0..half {
            let old_v = partition[i];
            partition[i] = other[j];

            let new_cut = count_cut(partition, edges);
            if new_cut < current_cut {
                return true; // Found improvement
            }

            partition[i] = old_v; // Revert
        }
    }

    false
}
