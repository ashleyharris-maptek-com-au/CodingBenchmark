using namespace std;

const double EPSILON = 1e-9;

struct Vec3 {
    double x, y, z;

    Vec3() : x(0), y(0), z(0) {}
    Vec3(double x, double y, double z) : x(x), y(y), z(z) {}

    Vec3 operator+(const Vec3& v) const { return Vec3(x + v.x, y + v.y, z + v.z); }
    Vec3 operator-(const Vec3& v) const { return Vec3(x - v.x, y - v.y, z - v.z); }
    Vec3 operator*(double s) const { return Vec3(x * s, y * s, z * s); }
    Vec3 operator/(double s) const { return Vec3(x / s, y / s, z / s); }

    double dot(const Vec3& v) const { return x * v.x + y * v.y + z * v.z; }
    Vec3 cross(const Vec3& v) const {
        return Vec3(y * v.z - z * v.y, z * v.x - x * v.z, x * v.y - y * v.x);
    }

    double length() const { return sqrt(x * x + y * y + z * z); }
    double lengthSq() const { return x * x + y * y + z * z; }
    Vec3 normalize() const {
        double len = length();
        return len > EPSILON ? *this / len : Vec3(0, 0, 0);
    }
};

struct Face {
    int a, b, c;
    Face(int a = 0, int b = 0, int c = 0) : a(a), b(b), c(c) {}
};

struct AABB {
    Vec3 min, max;

    AABB() : min(Vec3(1e30, 1e30, 1e30)), max(Vec3(-1e30, -1e30, -1e30)) {}

    void expand(const Vec3& p) {
        min.x = std::min(min.x, p.x);
        min.y = std::min(min.y, p.y);
        min.z = std::min(min.z, p.z);
        max.x = std::max(max.x, p.x);
        max.y = std::max(max.y, p.y);
        max.z = std::max(max.z, p.z);
    }

    bool intersects(const AABB& other) const {
        return !(max.x < other.min.x || min.x > other.max.x ||
                 max.y < other.min.y || min.y > other.max.y ||
                 max.z < other.min.z || min.z > other.max.z);
    }

    Vec3 center() const { return (min + max) * 0.5; }
};

struct Mesh {
    vector<Vec3> vertices;
    vector<Face> faces;

    AABB getBounds() const {
        AABB box;
        for (const auto& v : vertices) {
            box.expand(v);
        }
        return box;
    }

    AABB getFaceBounds(int faceIdx) const {
        const Face& f = faces[faceIdx];
        AABB box;
        box.expand(vertices[f.a]);
        box.expand(vertices[f.b]);
        box.expand(vertices[f.c]);
        return box;
    }

    Vec3 getFaceNormal(int faceIdx) const {
        const Face& f = faces[faceIdx];
        Vec3 v0 = vertices[f.a];
        Vec3 v1 = vertices[f.b];
        Vec3 v2 = vertices[f.c];
        Vec3 n = (v1 - v0).cross(v2 - v0);
        return n.normalize();
    }

    Vec3 getFaceCenter(int faceIdx) const {
        const Face& f = faces[faceIdx];
        return (vertices[f.a] + vertices[f.b] + vertices[f.c]) / 3.0;
    }
};

// BVH Node for spatial acceleration
struct BVHNode {
    AABB bounds;
    int left, right;
    int faceIdx;
    bool isLeaf;

    BVHNode() : left(-1), right(-1), faceIdx(-1), isLeaf(false) {}
};

class BVH {
public:
    vector<BVHNode> nodes;
    const Mesh* mesh;

    BVH(const Mesh& m) : mesh(&m) {
        if (mesh->faces.empty()) return;

        vector<int> faceIndices(mesh->faces.size());
        for (size_t i = 0; i < faceIndices.size(); i++) {
            faceIndices[i] = i;
        }

        nodes.reserve(mesh->faces.size() * 2);
        buildRecursive(faceIndices, 0, faceIndices.size());
    }

    int buildRecursive(vector<int>& indices, int start, int end) {
        int nodeIdx = nodes.size();
        nodes.emplace_back();
        BVHNode& node = nodes[nodeIdx];

        // Compute bounds
        for (int i = start; i < end; i++) {
            AABB faceBounds = mesh->getFaceBounds(indices[i]);
            node.bounds.expand(faceBounds.min);
            node.bounds.expand(faceBounds.max);
        }

        int count = end - start;
        if (count <= 1) {
            node.isLeaf = true;
            node.faceIdx = indices[start];
            return nodeIdx;
        }

        // Find longest axis
        Vec3 extent = node.bounds.max - node.bounds.min;
        int axis = 0;
        if (extent.y > extent.x) axis = 1;
        if (extent.z > extent.x && extent.z > extent.y) axis = 2;

        // Sort by centroid
        sort(indices.begin() + start, indices.begin() + end, [&](int a, int b) {
            Vec3 ca = mesh->getFaceCenter(a);
            Vec3 cb = mesh->getFaceCenter(b);
            return (axis == 0 ? ca.x : (axis == 1 ? ca.y : ca.z)) <
                   (axis == 0 ? cb.x : (axis == 1 ? cb.y : cb.z));
        });

        int mid = start + count / 2;
        node.left = buildRecursive(indices, start, mid);
        node.right = buildRecursive(indices, mid, end);

        return nodeIdx;
    }
};

// Moller-Trumbore ray-triangle intersection
bool rayTriangleIntersect(const Vec3& orig, const Vec3& dir,
                          const Vec3& v0, const Vec3& v1, const Vec3& v2,
                          double& t) {
    Vec3 e1 = v1 - v0;
    Vec3 e2 = v2 - v0;
    Vec3 h = dir.cross(e2);
    double a = e1.dot(h);

    if (abs(a) < EPSILON) return false;

    double f = 1.0 / a;
    Vec3 s = orig - v0;
    double u = f * s.dot(h);

    if (u < 0.0 || u > 1.0) return false;

    Vec3 q = s.cross(e1);
    double v = f * dir.dot(q);

    if (v < 0.0 || u + v > 1.0) return false;

    t = f * e2.dot(q);
    return t > EPSILON;
}

// Check if point is inside mesh using ray casting with BVH
bool isPointInsideMesh(const Vec3& point, const Mesh& mesh, const BVH& bvh) {
    if (mesh.faces.empty()) return false;

    // Cast multiple rays to handle edge cases
    Vec3 rayDirs[] = {
        Vec3(1, 0, 0),
        Vec3(0.7071, 0.7071, 0),
        Vec3(0.5773, 0.5773, 0.5773)
    };

    int votes = 0;
    for (const Vec3& rayDir : rayDirs) {
        int intersectionCount = 0;

        // Traverse BVH
        vector<int> stack;
        stack.push_back(0);

        while (!stack.empty()) {
            int nodeIdx = stack.back();
            stack.pop_back();

            if (nodeIdx >= (int)bvh.nodes.size()) continue;

            const BVHNode& node = bvh.nodes[nodeIdx];

            // Check if ray intersects node bounds
            Vec3 invDir(1.0 / (rayDir.x + EPSILON),
                       1.0 / (rayDir.y + EPSILON),
                       1.0 / (rayDir.z + EPSILON));

            double t1 = (node.bounds.min.x - point.x) * invDir.x;
            double t2 = (node.bounds.max.x - point.x) * invDir.x;
            double t3 = (node.bounds.min.y - point.y) * invDir.y;
            double t4 = (node.bounds.max.y - point.y) * invDir.y;
            double t5 = (node.bounds.min.z - point.z) * invDir.z;
            double t6 = (node.bounds.max.z - point.z) * invDir.z;

            double tmin = max(max(min(t1, t2), min(t3, t4)), min(t5, t6));
            double tmax = min(min(max(t1, t2), max(t3, t4)), max(t5, t6));

            if (tmax < 0 || tmin > tmax) continue;

            if (node.isLeaf) {
                const Face& f = mesh.faces[node.faceIdx];
                double t;
                if (rayTriangleIntersect(point, rayDir,
                                        mesh.vertices[f.a],
                                        mesh.vertices[f.b],
                                        mesh.vertices[f.c], t)) {
                    intersectionCount++;
                }
            } else {
                if (node.left >= 0) stack.push_back(node.left);
                if (node.right >= 0) stack.push_back(node.right);
            }
        }

        if (intersectionCount % 2 == 1) {
            votes++;
        }
    }

    return votes >= 2;
}

// CSG Union implementation
Mesh csgUnion(const Mesh& meshA, const Mesh& meshB) {
    Mesh result;

    // Build BVH for both meshes
    BVH bvhA(meshA);
    BVH bvhB(meshB);

    // Check if meshes intersect at all
    AABB boundsA = meshA.getBounds();
    AABB boundsB = meshB.getBounds();

    bool mayIntersect = boundsA.intersects(boundsB);

    // Combine all vertices
    result.vertices = meshA.vertices;
    int offsetB = result.vertices.size();
    result.vertices.insert(result.vertices.end(), meshB.vertices.begin(), meshB.vertices.end());

    if (!mayIntersect) {
        // No intersection - just combine all faces
        result.faces = meshA.faces;
        for (const Face& f : meshB.faces) {
            result.faces.push_back(Face(f.a + offsetB, f.b + offsetB, f.c + offsetB));
        }
    } else {
        // Potential intersection - filter faces

        // Keep faces from A that are outside B
        for (size_t i = 0; i < meshA.faces.size(); i++) {
            Vec3 center = meshA.getFaceCenter(i);
            Vec3 normal = meshA.getFaceNormal(i);

            // Test point slightly outside the face
            Vec3 testPoint = center + normal * 1e-6;

            if (!isPointInsideMesh(testPoint, meshB, bvhB)) {
                result.faces.push_back(meshA.faces[i]);
            }
        }

        // Keep faces from B that are outside A
        for (size_t i = 0; i < meshB.faces.size(); i++) {
            Vec3 center = meshB.getFaceCenter(i);
            Vec3 normal = meshB.getFaceNormal(i);

            Vec3 testPoint = center + normal * 1e-6;

            if (!isPointInsideMesh(testPoint, meshA, bvhA)) {
                Face f = meshB.faces[i];
                result.faces.push_back(Face(f.a + offsetB, f.b + offsetB, f.c + offsetB));
            }
        }
    }

    return result;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int VA, FA, VB, FB;
    cin >> VA >> FA >> VB >> FB;

    Mesh meshA, meshB;

    // Read mesh A
    meshA.vertices.resize(VA);
    for (int i = 0; i < VA; i++) {
        cin >> meshA.vertices[i].x >> meshA.vertices[i].y >> meshA.vertices[i].z;
    }

    meshA.faces.resize(FA);
    for (int i = 0; i < FA; i++) {
        cin >> meshA.faces[i].a >> meshA.faces[i].b >> meshA.faces[i].c;
    }

    // Read mesh B
    meshB.vertices.resize(VB);
    for (int i = 0; i < VB; i++) {
        cin >> meshB.vertices[i].x >> meshB.vertices[i].y >> meshB.vertices[i].z;
    }

    meshB.faces.resize(FB);
    for (int i = 0; i < FB; i++) {
        cin >> meshB.faces[i].a >> meshB.faces[i].b >> meshB.faces[i].c;
    }

    // Compute union
    Mesh result = csgUnion(meshA, meshB);

    // Output result
    cout << result.vertices.size() << " " << result.faces.size() << "\n";

    cout.precision(17);
    for (const auto& v : result.vertices) {
        cout << v.x << " " << v.y << " " << v.z << "\n";
    }

    for (const auto& f : result.faces) {
        cout << f.a << " " << f.b << " " << f.c << "\n";
    }

    return 0;
}
