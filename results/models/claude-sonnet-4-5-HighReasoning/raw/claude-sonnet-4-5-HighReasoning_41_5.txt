; Main function
%main = OpFunction %void None %void_func
%entry = OpLabel

; Load inputs
%worldPos_val = OpLoad %vec3 %worldPos
%normal_val = OpLoad %vec3 %normal

; Load lightPos (member 3) and cameraPos (member 4)
%lightPos_ptr = OpAccessChain %ptr_uniform_vec4 %ubo %int_3
%lightPos_val = OpLoad %vec4 %lightPos_ptr
%cameraPos_ptr = OpAccessChain %ptr_uniform_vec4 %ubo %int_4
%cameraPos_val = OpLoad %vec4 %cameraPos_ptr

; Extract xyz from vec4
%lightPos_xyz = OpVectorShuffle %vec3 %lightPos_val %lightPos_val 0 1 2
%cameraPos_xyz = OpVectorShuffle %vec3 %cameraPos_val %cameraPos_val 0 1 2

; N = normalize(normal)
%N = OpExtInst %vec3 %glsl450 Normalize %normal_val

; L = normalize(lightPos - worldPos)

; V = normalize(cameraPos - worldPos)
%V_unnorm = OpFSub %vec3 %cameraPos_xyz %worldPos_val
%V = OpExtInst %vec3 %glsl450 Normalize %V_unnorm

; -L
%neg_L = OpFNegate %vec3 %L

; R = reflect(-L, N)
%R = OpExtInst %vec3 %glsl450 Reflect %neg_L %N

; diffuse = max(dot(N, L), 0)

%diffuse = OpExtInst %float %glsl450 FMax %NdotL %float_0

; specular = pow(max(dot(R, V), 0), 32)

%RdotV_clamped = OpExtInst %float %glsl450 FMax %RdotV %float_0
%specular = OpExtInst %float %glsl450 Pow %RdotV_clamped %float_32

%specular_scaled = OpFMul %float %float_0_8 %specular
%temp1 = OpFAdd %float %float_0_2 %diffuse_scaled
%intensity = OpFAdd %float %temp1 %specular_scaled

; outColor = vec4(intensity, intensity, intensity, 1.0)
%result = OpCompositeConstruct %vec4 %intensity %intensity %intensity %float_1
