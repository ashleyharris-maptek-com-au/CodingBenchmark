use std::io::{self, BufReader, BufWriter, Read, Write};

#[derive(Clone, Copy, Debug)]
struct Point3D {
    x: f32,
    y: f32,
    z: f32,
}

impl Point3D {
    fn new(x: f32, y: f32, z: f32) -> Self {
        Point3D { x, y, z }
    }

    fn distance_squared(&self, other: &Point3D) -> f32 {
        let dx = self.x - other.x;
        let dy = self.y - other.y;
        let dz = self.z - other.z;
        dx * dx + dy * dy + dz * dz
    }

    fn zero() -> Self {
        Point3D { x: 0.0, y: 0.0, z: 0.0 }
    }
}

fn read_header<R: Read>(reader: &mut R) -> io::Result<(usize, usize)> {
    let mut buffer = String::new();
    let mut byte_buf = [0u8; 1];

    loop {
        reader.read_exact(&mut byte_buf)?;
        if byte_buf[0] == b'\n' {
            break;
        }
        buffer.push(byte_buf[0] as char);
    }

    let parts: Vec<&str> = buffer.trim().split_whitespace().collect();
    if parts.len() != 2 {
        return Err(io::Error::new(io::ErrorKind::InvalidData, "Invalid header"));
    }

    let num_points: usize = parts[0].parse()
        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "Invalid num_points"))?;
    let num_clusters: usize = parts[1].parse()
        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "Invalid num_clusters"))?;

    Ok((num_points, num_clusters))
}

fn read_point<R: Read>(reader: &mut R) -> io::Result<Point3D> {
    let mut buffer = [0u8; 12];
    reader.read_exact(&mut buffer)?;

    let x = f32::from_le_bytes([buffer[0], buffer[1], buffer[2], buffer[3]]);
    let y = f32::from_le_bytes([buffer[4], buffer[5], buffer[6], buffer[7]]);
    let z = f32::from_le_bytes([buffer[8], buffer[9], buffer[10], buffer[11]]);

    Ok(Point3D::new(x, y, z))
}

fn find_nearest_cluster(point: &Point3D, centroids: &[Point3D]) -> usize {
    let mut min_dist = f32::INFINITY;
    let mut nearest = 0;

    for (i, centroid) in centroids.iter().enumerate() {
        let dist = point.distance_squared(centroid);
        if dist < min_dist {
            min_dist = dist;
            nearest = i;
        }
    }

    nearest
}

// Simple LCG for random number generation
struct SimpleRng {
    state: u64,
}

impl SimpleRng {
    fn new(seed: u64) -> Self {
        SimpleRng { state: seed }
    }

    fn next_u64(&mut self) -> u64 {
        self.state = self.state.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
        self.state
    }

    fn next_usize(&mut self, max: usize) -> usize {
        (self.next_u64() as usize) % max
    }

    fn next_f32(&mut self) -> f32 {
        (self.next_u64() as f32) / (u64::MAX as f32)
    }
}

fn cluster_standard<R: Read>(
    reader: &mut R,
    num_points: usize,
    num_clusters: usize,
) -> io::Result<()> {
    let mut rng = SimpleRng::new(42);

    // Read all points into memory
    let mut points = Vec::with_capacity(num_points);
    for _ in 0..num_points {
        points.push(read_point(reader)?);
    }

    // Initialize centroids using k-means++
    let mut centroids = vec![Point3D::zero(); num_clusters];
    centroids[0] = points[rng.next_usize(points.len())];

    for i in 1..num_clusters {
        let mut distances: Vec<f32> = points.iter()
            .map(|p| {
                centroids[..i].iter()
                    .map(|c| p.distance_squared(c))
                    .fold(f32::INFINITY, f32::min)
            })
            .collect();

        let sum: f32 = distances.iter().sum();
        if sum > 0.0 {
            let mut threshold = rng.next_f32() * sum;
            let mut selected = 0;
            for (idx, &dist) in distances.iter().enumerate() {
                threshold -= dist;
                if threshold <= 0.0 {
                    selected = idx;
                    break;
                }
            }
            centroids[i] = points[selected];
        } else {
            centroids[i] = points[i % points.len()];
        }
    }

    // Lloyd's algorithm
    let mut assignments = vec![0usize; num_points];
    let max_iterations = 50;

    for iter in 0..max_iterations {
        // Assign points to clusters
        let mut changed = false;
        for (idx, point) in points.iter().enumerate() {
            let nearest = find_nearest_cluster(point, &centroids);
            if assignments[idx] != nearest {
                assignments[idx] = nearest;
                changed = true;
            }
        }

        if !changed && iter > 5 {
            break;
        }

        // Update centroids
        let mut sums = vec![Point3D::zero(); num_clusters];
        let mut counts = vec![0usize; num_clusters];

        for (idx, point) in points.iter().enumerate() {
            let cluster = assignments[idx];
            sums[cluster].x += point.x;
            sums[cluster].y += point.y;
            sums[cluster].z += point.z;
            counts[cluster] += 1;
        }

        for i in 0..num_clusters {
            if counts[i] > 0 {
                let count = counts[i] as f32;
                centroids[i].x = sums[i].x / count;
                centroids[i].y = sums[i].y / count;
                centroids[i].z = sums[i].z / count;
            } else {
                // Reinitialize empty cluster with farthest point
                let mut max_dist = 0.0f32;
                let mut farthest_idx = 0;
                for (idx, point) in points.iter().enumerate() {
                    let dist = point.distance_squared(&centroids[assignments[idx]]);
                    if dist > max_dist {
                        max_dist = dist;
                        farthest_idx = idx;
                    }
                }
                centroids[i] = points[farthest_idx];
            }
        }
    }

    // Write output
    let stdout = io::stdout();
    let mut writer = BufWriter::with_capacity(8 * 1024 * 1024, stdout.lock());

    for &assignment in &assignments {
        writer.write_all(&(assignment as u32).to_le_bytes())?;
    }
    writer.flush()?;

    Ok(())
}

fn cluster_streaming<R: Read>(
    reader: &mut R,
    num_points: usize,
    num_clusters: usize,
) -> io::Result<()> {
    let mut rng = SimpleRng::new(42);

    // Initialize centroids by reading first batch for sampling
    let init_sample_size = (num_clusters * 100).min(100_000).min(num_points);
    let mut init_points = Vec::with_capacity(init_sample_size);

    for _ in 0..init_sample_size {
        init_points.push(read_point(reader)?);
    }

    // Initialize centroids with k-means++ on sample
    let mut centroids = vec![Point3D::zero(); num_clusters];

    if !init_points.is_empty() {
        centroids[0] = init_points[rng.next_usize(init_points.len())];

        for i in 1..num_clusters.min(init_points.len()) {
            let mut best_idx = 0;
            let mut best_dist = 0.0f32;

            for (idx, point) in init_points.iter().enumerate() {
                let min_dist = centroids[..i].iter()
                    .map(|c| point.distance_squared(c))
                    .fold(f32::INFINITY, f32::min);
                if min_dist > best_dist {
                    best_dist = min_dist;
                    best_idx = idx;
                }
            }

            centroids[i] = init_points[best_idx];
        }

        // Fill remaining centroids if needed
        for i in init_points.len()..num_clusters {
            centroids[i] = init_points[i % init_points.len()];
        }
    } else {
        // Fallback: space-filling initialization
        for i in 0..num_clusters {
            centroids[i] = Point3D::new(
                (i as f32) * 10.0,
                (i as f32 / num_clusters as f32) * 10.0,
                0.0,
            );
        }
    }

    // Process points in streaming fashion with online updates
    let mut assignments = Vec::with_capacity(num_points);
    let mut centroid_counts = vec![0usize; num_clusters];
    let mut centroid_sums = vec![Point3D::zero(); num_clusters];

    // Process initial sample
    for point in init_points {
        let nearest = find_nearest_cluster(&point, &centroids);
        assignments.push(nearest as u32);
        centroid_sums[nearest].x += point.x;
        centroid_sums[nearest].y += point.y;
        centroid_sums[nearest].z += point.z;
        centroid_counts[nearest] += 1;
    }

    // Process remaining points
    let mut points_processed = assignments.len();

    // Adaptive update frequency based on dataset size
    let update_frequency = if num_points > 1_000_000_000 {
        50_000  // Very large: update less often
    } else if num_points > 100_000_000 {
        10_000  // Large: moderate updates
    } else {
        1_000   // Medium: frequent updates
    };

    while points_processed < num_points {
        match read_point(reader) {
            Ok(point) => {
                let nearest = find_nearest_cluster(&point, &centroids);
                assignments.push(nearest as u32);

                // Accumulate for batch update
                centroid_sums[nearest].x += point.x;
                centroid_sums[nearest].y += point.y;
                centroid_sums[nearest].z += point.z;
                centroid_counts[nearest] += 1;

                points_processed += 1;

                // Periodically update centroids for better convergence
                if points_processed % update_frequency == 0 {
                    for i in 0..num_clusters {
                        if centroid_counts[i] > 0 {
                            let count = centroid_counts[i] as f32;
                            centroids[i].x = centroid_sums[i].x / count;
                            centroids[i].y = centroid_sums[i].y / count;
                            centroids[i].z = centroid_sums[i].z / count;
                        }
                    }
                }
            }
            Err(_) => break,
        }
    }

    // Final centroid update
    for i in 0..num_clusters {
        if centroid_counts[i] > 0 {
            let count = centroid_counts[i] as f32;
            centroids[i].x = centroid_sums[i].x / count;
            centroids[i].y = centroid_sums[i].y / count;
            centroids[i].z = centroid_sums[i].z / count;
        }
    }

    // Write output
    let stdout = io::stdout();
    let mut writer = BufWriter::with_capacity(8 * 1024 * 1024, stdout.lock());

    for &assignment in &assignments {
        writer.write_all(&assignment.to_le_bytes())?;
    }
    writer.flush()?;

    Ok(())
}

fn main() -> io::Result<()> {
    let stdin = io::stdin();
    let mut reader = BufReader::with_capacity(8 * 1024 * 1024, stdin.lock());

    let (num_points, num_clusters) = read_header(&mut reader)?;

    // Adaptive strategy selection based on dataset size
    // Small datasets: Full k-means with multiple iterations
    // Large datasets: Streaming k-means with single-pass online updates
    if num_points <= 10_000_000 {
        // Small: load all into memory, run standard Lloyd's k-means
        cluster_standard(&mut reader, num_points, num_clusters)?;
    } else {
        // Medium/Large/Extreme: streaming mini-batch k-means
        cluster_streaming(&mut reader, num_points, num_clusters)?;
    }

    Ok(())
}
