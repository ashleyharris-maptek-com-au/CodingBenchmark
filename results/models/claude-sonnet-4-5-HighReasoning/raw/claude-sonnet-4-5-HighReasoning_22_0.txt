using System;
using System.Collections.Generic;
using System.Linq;

class TetrisPlayer
{
    static void Main()
    {
        var input = Console.ReadLine().Split();
        int width = int.Parse(input[0]);
        int height = int.Parse(input[1]);
        int numPieces = int.Parse(input[2]);

        var game = new Game(width, height);

        for (int i = 0; i < numPieces; i++)
        {
            string pieceType = Console.ReadLine().Trim();
            var (rotation, column) = game.FindBestMove(pieceType);
            Console.WriteLine($"{rotation} {column}");

            string result = Console.ReadLine().Trim();
            game.ApplyMove(pieceType, rotation, column);

            if (result.StartsWith("gameover"))
                break;
        }
    }
}

class Game
{
    private bool[,] board;
    private int width;
    private int height;
    private int moveCount;

    public Game(int width, int height)
    {
        this.width = width;
        this.height = height;
        this.board = new bool[height, width];
        this.moveCount = 0;
    }

    public (int rotation, int column) FindBestMove(string pieceType)
    {
        moveCount++;

        double bestScore = double.MinValue;
        int bestRotation = 0;
        int bestColumn = 0;

        for (int rotation = 0; rotation < 4; rotation++)
        {
            var shape = GetRotatedShape(pieceType, rotation);
            int pieceWidth = shape.Max(p => p.col) + 1;

            for (int col = 0; col <= width - pieceWidth; col++)
            {
                var testBoard = (bool[,])board.Clone();

                if (SimulatePlacement(testBoard, shape, col, out int linesCleared))
                {
                    double score = EvaluateBoard(testBoard, linesCleared);

                    if (score > bestScore)
                    {
                        bestScore = score;
                        bestRotation = rotation;
                        bestColumn = col;
                    }
                }
            }
        }

        return (bestRotation, bestColumn);
    }

    public void ApplyMove(string pieceType, int rotation, int column)
    {
        var shape = GetRotatedShape(pieceType, rotation);
        SimulatePlacement(board, shape, column, out _);
    }

    private bool SimulatePlacement(bool[,] testBoard, List<(int row, int col)> shape, int startCol, out int linesCleared)
    {
        linesCleared = 0;

        int dropRow = FindDropRow(testBoard, shape, startCol);
        if (dropRow < 0)
            return false;

        foreach (var (r, c) in shape)
        {
            int row = dropRow + r;
            int col = startCol + c;

            if (row < 0 || row >= height || col < 0 || col >= width || testBoard[row, col])
                return false;

            testBoard[row, col] = true;
        }

        linesCleared = ClearFullLines(testBoard);
        return true;
    }

    private int FindDropRow(bool[,] testBoard, List<(int row, int col)> shape, int startCol)
    {
        for (int dropRow = 0; dropRow < height; dropRow++)
        {
            bool collision = false;

            foreach (var (r, c) in shape)
            {
                int row = dropRow + r;
                int col = startCol + c;

                if (row >= height || (row >= 0 && col >= 0 && col < width && testBoard[row, col]))
                {
                    collision = true;
                    break;
                }
            }

            if (collision)
                return dropRow - 1;
        }

        return height - 1;
    }

    private int ClearFullLines(bool[,] testBoard)
    {
        int cleared = 0;

        for (int row = height - 1; row >= 0; row--)
        {
            bool isFull = true;
            for (int col = 0; col < width; col++)
            {
                if (!testBoard[row, col])
                {
                    isFull = false;
                    break;
                }
            }

            if (isFull)
            {
                cleared++;

                for (int r = row; r > 0; r--)
                {
                    for (int c = 0; c < width; c++)
                    {
                        testBoard[r, c] = testBoard[r - 1, c];
                    }
                }

                for (int c = 0; c < width; c++)
                {
                    testBoard[0, c] = false;
                }

                row++;
            }
        }

        return cleared;
    }

    private double EvaluateBoard(bool[,] testBoard, int linesCleared)
    {
        double score = 0;

        score += linesCleared * linesCleared * 150;

        int aggregateHeight = GetAggregateHeight(testBoard);
        int holes = CountHoles(testBoard);
        int bumpiness = GetBumpiness(testBoard);
        int wells = CountDeepWells(testBoard);

        bool isComplex = width > 12 || moveCount > 50;

        if (isComplex)
        {
            score -= aggregateHeight * 0.8;
            score -= holes * 80;
            score -= bumpiness * 4;
            score -= wells * 15;
        }
        else
        {
            score -= aggregateHeight * 0.5;
            score -= holes * 50;
            score -= bumpiness * 2;
            score -= wells * 8;
        }

        int maxHeight = GetMaxHeight(testBoard);
        if (maxHeight > height * 0.7)
        {
            score -= (maxHeight - height * 0.7) * 100;
        }

        return score;
    }

    private int[] GetColumnHeights(bool[,] testBoard)
    {
        int[] heights = new int[width];

        for (int col = 0; col < width; col++)
        {
            for (int row = 0; row < height; row++)
            {
                if (testBoard[row, col])
                {
                    heights[col] = height - row;
                    break;
                }
            }
        }

        return heights;
    }

    private int GetAggregateHeight(bool[,] testBoard)
    {
        return GetColumnHeights(testBoard).Sum();
    }

    private int GetMaxHeight(bool[,] testBoard)
    {
        return GetColumnHeights(testBoard).Max();
    }

    private int CountHoles(bool[,] testBoard)
    {
        int holes = 0;

        for (int col = 0; col < width; col++)
        {
            bool foundBlock = false;
            for (int row = 0; row < height; row++)
            {
                if (testBoard[row, col])
                {
                    foundBlock = true;
                }
                else if (foundBlock)
                {
                    holes++;
                }
            }
        }

        return holes;
    }

    private int GetBumpiness(bool[,] testBoard)
    {
        int[] heights = GetColumnHeights(testBoard);
        int bumpiness = 0;

        for (int i = 0; i < width - 1; i++)
        {
            bumpiness += Math.Abs(heights[i] - heights[i + 1]);
        }

        return bumpiness;
    }

    private int CountDeepWells(bool[,] testBoard)
    {
        int[] heights = GetColumnHeights(testBoard);
        int wells = 0;

        for (int col = 0; col < width; col++)
        {
            int leftHeight = col > 0 ? heights[col - 1] : 0;
            int rightHeight = col < width - 1 ? heights[col + 1] : 0;
            int currentHeight = heights[col];

            if (col == 0 && rightHeight > currentHeight)
            {
                wells += (rightHeight - currentHeight);
            }
            else if (col == width - 1 && leftHeight > currentHeight)
            {
                wells += (leftHeight - currentHeight);
            }
            else if (col > 0 && col < width - 1 && leftHeight > currentHeight && rightHeight > currentHeight)
            {
                wells += (Math.Min(leftHeight, rightHeight) - currentHeight);
            }
        }

        return wells;
    }

    private List<(int row, int col)> GetRotatedShape(string pieceType, int rotation)
    {
        var baseShape = GetBaseShape(pieceType);
        var rotated = baseShape;

        for (int i = 0; i < rotation; i++)
        {
            rotated = RotateClockwise(rotated);
        }

        return Normalize(rotated);
    }

    private List<(int row, int col)> GetBaseShape(string type)
    {
        return type switch
        {
            "I" => new List<(int, int)> { (0, 0), (0, 1), (0, 2), (0, 3) },
            "O" => new List<(int, int)> { (0, 0), (0, 1), (1, 0), (1, 1) },
            "T" => new List<(int, int)> { (0, 1), (1, 0), (1, 1), (1, 2) },
            "S" => new List<(int, int)> { (0, 1), (0, 2), (1, 0), (1, 1) },
            "Z" => new List<(int, int)> { (0, 0), (0, 1), (1, 1), (1, 2) },
            "J" => new List<(int, int)> { (0, 0), (1, 0), (1, 1), (1, 2) },
            "L" => new List<(int, int)> { (0, 2), (1, 0), (1, 1), (1, 2) },
            _ => new List<(int, int)>()
        };
    }

    private List<(int row, int col)> RotateClockwise(List<(int row, int col)> shape)
    {
        return shape.Select(p => (-p.col, p.row)).ToList();
    }

    private List<(int row, int col)> Normalize(List<(int row, int col)> shape)
    {
        int minRow = shape.Min(p => p.row);
        int minCol = shape.Min(p => p.col);

        return shape.Select(p => (p.row - minRow, p.col - minCol)).ToList();
    }
}
