use std::io::{self, BufRead};

struct Grid {
    width: usize,
    height: usize,
    obstacles: Vec<Vec<bool>>,
    require_cycle: bool,
    total_cells: usize,
}

impl Grid {
    fn new(width: usize, height: usize, require_cycle: bool) -> Self {
        let obstacles = vec![vec![false; height]; width];
        let total_cells = width * height;
        Grid { width, height, obstacles, require_cycle, total_cells }
    }

    fn add_obstacle(&mut self, x: usize, y: usize) {
        self.obstacles[x][y] = true;
        self.total_cells -= 1;
    }

    fn is_valid(&self, x: i32, y: i32) -> bool {
        x >= 0 && y >= 0 &&
        (x as usize) < self.width &&
        (y as usize) < self.height &&
        !self.obstacles[x as usize][y as usize]
    }

    fn get_neighbors(&self, x: usize, y: usize) -> Vec<(usize, usize)> {
        let mut neighbors = Vec::new();
        let dirs = [(0, 1), (1, 0), (0, -1), (-1, 0)];

        for (dx, dy) in &dirs {
            let nx = x as i32 + dx;
            let ny = y as i32 + dy;
            if self.is_valid(nx, ny) {
                neighbors.push((nx as usize, ny as usize));
            }
        }
        neighbors
    }

    fn count_unvisited_neighbors(&self, x: usize, y: usize, visited: &Vec<Vec<bool>>) -> usize {
        let dirs = [(0i32, 1i32), (1, 0), (0, -1), (-1, 0)];
        let mut count = 0;

        for (dx, dy) in &dirs {
            let nx = x as i32 + dx;
            let ny = y as i32 + dy;
            if self.is_valid(nx, ny) && !visited[nx as usize][ny as usize] {
                count += 1;
            }
        }
        count
    }
}

struct Solver {
    grid: Grid,
    path: Vec<(usize, usize)>,
    visited: Vec<Vec<bool>>,
    nodes_explored: u64,
}

impl Solver {
    fn new(grid: Grid) -> Self {
        let width = grid.width;
        let height = grid.height;
        let visited = vec![vec![false; height]; width];

        Solver {
            grid,
            path: Vec::with_capacity(width * height),
            visited,
            nodes_explored: 0,
        }
    }

    fn solve(&mut self) -> bool {
        // Check if start is blocked
        if self.grid.obstacles[0][0] {
            return false;
        }

        // Quick connectivity check from start
        if !self.is_reachable_from_start() {
            return false;
        }

        self.path.push((0, 0));
        self.visited[0][0] = true;

        self.backtrack()
    }

    fn is_reachable_from_start(&self) -> bool {
        // BFS to verify all non-obstacle cells are reachable from (0,0)
        let mut queue = std::collections::VecDeque::new();
        let mut temp_visited = vec![vec![false; self.grid.height]; self.grid.width];

        queue.push_back((0, 0));
        temp_visited[0][0] = true;
        let mut reachable = 1;

        while let Some((x, y)) = queue.pop_front() {
            for (nx, ny) in self.grid.get_neighbors(x, y) {
                if !temp_visited[nx][ny] {
                    temp_visited[nx][ny] = true;
                    queue.push_back((nx, ny));
                    reachable += 1;
                }
            }
        }

        reachable == self.grid.total_cells
    }

    fn check_articulation_point(&self, skip_x: usize, skip_y: usize) -> bool {
        // Check if remaining unvisited cells (excluding skip) form a connected component
        // Find first unvisited cell that isn't skip
        let mut start = None;
        for x in 0..self.grid.width {
            for y in 0..self.grid.height {
                if !self.visited[x][y] && !self.grid.obstacles[x][y] && (x != skip_x || y != skip_y) {
                    start = Some((x, y));
                    break;
                }
            }
            if start.is_some() { break; }
        }

        if start.is_none() {
            return true; // No other unvisited cells, connected
        }

        let (sx, sy) = start.unwrap();
        let mut queue = std::collections::VecDeque::new();
        let mut temp_visited = vec![vec![false; self.grid.height]; self.grid.width];

        queue.push_back((sx, sy));
        temp_visited[sx][sy] = true;
        let mut count = 1;

        while let Some((x, y)) = queue.pop_front() {
            for (nx, ny) in self.grid.get_neighbors(x, y) {
                if !self.visited[nx][ny] && !temp_visited[nx][ny] && (nx != skip_x || ny != skip_y) {
                    temp_visited[nx][ny] = true;
                    queue.push_back((nx, ny));
                    count += 1;
                }
            }
        }

        // Count total unvisited cells excluding skip
        let mut total_unvisited = 0;
        for x in 0..self.grid.width {
            for y in 0..self.grid.height {
                if !self.visited[x][y] && !self.grid.obstacles[x][y] && (x != skip_x || y != skip_y) {
                    total_unvisited += 1;
                }
            }
        }

        count == total_unvisited
    }

    fn backtrack(&mut self) -> bool {
        self.nodes_explored += 1;

        // Check if we've visited all cells
        if self.path.len() == self.grid.total_cells {
            if self.grid.require_cycle {
                let (last_x, last_y) = *self.path.last().unwrap();
                // Check if last cell is adjacent to (0, 0)
                for (nx, ny) in self.grid.get_neighbors(last_x, last_y) {
                    if nx == 0 && ny == 0 {
                        return true;
                    }
                }
                return false;
            }
            return true;
        }

        let (curr_x, curr_y) = *self.path.last().unwrap();

        // Get unvisited neighbors and compute their degrees
        let mut neighbors: Vec<(usize, usize, usize)> = Vec::new();

        for (nx, ny) in self.grid.get_neighbors(curr_x, curr_y) {
            if !self.visited[nx][ny] {
                let degree = self.grid.count_unvisited_neighbors(nx, ny, &self.visited);
                neighbors.push((nx, ny, degree));
            }
        }

        if neighbors.is_empty() {
            return false; // Dead end
        }

        // Sort by degree (ascending) - Warnsdorff's heuristic
        // Visit cells with fewer unvisited neighbors first
        neighbors.sort_by_key(|&(_, _, degree)| degree);

        // Check for forced moves (cells with degree 1)
        let min_degree = neighbors[0].2;
        if min_degree == 1 {
            // Prioritize forced moves
            let forced: Vec<_> = neighbors.iter()
                .filter(|&&(_, _, deg)| deg == 1)
                .copied()
                .collect();

            // If there are multiple forced moves, we might be creating isolation
            // But we should still try them
            for (nx, ny, _) in forced {
                self.path.push((nx, ny));
                self.visited[nx][ny] = true;

                if self.backtrack() {
                    return true;
                }

                self.path.pop();
                self.visited[nx][ny] = false;
            }
            return false;
        }

        // Try each neighbor
        for (nx, ny, _) in neighbors {
            // Pruning: check if this move would disconnect the graph
            // Skip this check for small grids or when close to completion
            let remaining = self.grid.total_cells - self.path.len();
            let should_check_connectivity = remaining > 5 && self.grid.total_cells > 50;

            if should_check_connectivity {
                // Quick check: would visiting this cell create an isolated region?
                self.visited[nx][ny] = true;
                let connected = self.check_articulation_point(nx, ny);
                self.visited[nx][ny] = false;

                if !connected {
                    continue; // This move disconnects the graph
                }
            }

            self.path.push((nx, ny));
            self.visited[nx][ny] = true;

            if self.backtrack() {
                return true;
            }

            self.path.pop();
            self.visited[nx][ny] = false;
        }

        false
    }

    fn print_solution(&self) {
        if self.path.len() == self.grid.total_cells {
            println!("{}", self.path.len());
            for (x, y) in &self.path {
                println!("{} {}", x, y);
            }
        } else {
            println!("0");
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Read first line
    let first_line = lines.next().unwrap().unwrap();
    let parts: Vec<&str> = first_line.split_whitespace().collect();

    let width: usize = parts[0].parse().unwrap();
    let height: usize = parts[1].parse().unwrap();
    let obstacle_count: usize = parts[2].parse().unwrap();
    let require_cycle: bool = parts[3].parse::<u8>().unwrap() == 1;

    let mut grid = Grid::new(width, height, require_cycle);

    // Read obstacles
    for _ in 0..obstacle_count {
        let line = lines.next().unwrap().unwrap();
        let parts: Vec<&str> = line.split_whitespace().collect();
        let x: usize = parts[0].parse().unwrap();
        let y: usize = parts[1].parse().unwrap();
        grid.add_obstacle(x, y);
    }

    let mut solver = Solver::new(grid);
    solver.solve();
    solver.print_solution();
}
