using System;
using System.Collections.Generic;
using System.Linq;

class Program
{
    static void Main()
    {
        var firstLine = Console.ReadLine().Split();
        int H = int.Parse(firstLine[0]);
        int W = int.Parse(firstLine[1]);

        char[][] board = new char[H][];
        for (int i = 0; i < H; i++)
        {
            board[i] = Console.ReadLine().ToCharArray();
        }

        var solver = new MinesweeperSolver(board, H, W);
        var safeCells = solver.Solve();

        Console.WriteLine(safeCells.Count);
        foreach (var (x, y) in safeCells)
        {
            Console.WriteLine($"{x} {y}");
        }
    }
}

class MinesweeperSolver
{
    private char[][] board;
    private int H, W;
    private HashSet<(int, int)> safe;
    private HashSet<(int, int)> mines;

    public MinesweeperSolver(char[][] board, int H, int W)
    {
        this.board = board;
        this.H = H;
        this.W = W;
        safe = new HashSet<(int, int)>();
        mines = new HashSet<(int, int)>();

        for (int y = 0; y < H; y++)
        {
            for (int x = 0; x < W; x++)
            {
                if (board[y][x] == '*')
                {
                    mines.Add((x, y));
                }
            }
        }
    }

    public List<(int, int)> Solve()
    {
        long size = (long)H * W;

        if (size > 10000)
        {
            SolveOptimized();
        }
        else
        {
            SolveSimple();
        }

        TryAdvancedDeductions();

        return safe.Where(pos => board[pos.y][pos.x] == ' ')
                   .OrderBy(p => p.y)
                   .ThenBy(p => p.x)
                   .ToList();
    }

    private void SolveSimple()
    {
        bool changed = true;
        while (changed)
        {
            changed = false;
            for (int y = 0; y < H; y++)
            {
                for (int x = 0; x < W; x++)
                {
                    if (char.IsDigit(board[y][x]) && ProcessCell(x, y))
                    {
                        changed = true;
                    }
                }
            }
        }
    }

    private void SolveOptimized()
    {
        var frontier = new Queue<(int, int)>();
        var inQueue = new HashSet<(int, int)>();

        for (int y = 0; y < H; y++)
        {
            for (int x = 0; x < W; x++)
            {
                if (char.IsDigit(board[y][x]))
                {
                    frontier.Enqueue((x, y));
                    inQueue.Add((x, y));
                }
            }
        }

        while (frontier.Count > 0)
        {
            var (x, y) = frontier.Dequeue();
            inQueue.Remove((x, y));

            if (ProcessCell(x, y))
            {
                foreach (var (nx, ny) in GetNeighbors(x, y))
                {
                    if (char.IsDigit(board[ny][nx]) && !inQueue.Contains((nx, ny)))
                    {
                        frontier.Enqueue((nx, ny));
                        inQueue.Add((nx, ny));
                    }
                }
            }
        }
    }

    private bool ProcessCell(int x, int y)
    {
        if (!char.IsDigit(board[y][x])) return false;

        int mineCount = board[y][x] - '0';
        var neighbors = GetNeighbors(x, y);

        var unknown = new List<(int, int)>();
        int knownMines = 0;

        foreach (var (nx, ny) in neighbors)
        {
            if (board[ny][nx] == ' ')
            {
                if (mines.Contains((nx, ny)))
                {
                    knownMines++;
                }
                else if (!safe.Contains((nx, ny)))
                {
                    unknown.Add((nx, ny));
                }
            }
            else if (board[ny][nx] == '*')
            {
                knownMines++;
            }
        }

        int remainingMines = mineCount - knownMines;
        bool changed = false;

        if (remainingMines == 0 && unknown.Count > 0)
        {
            foreach (var cell in unknown)
            {
                if (safe.Add(cell))
                {
                    changed = true;
                }
            }
        }
        else if (remainingMines == unknown.Count && unknown.Count > 0)
        {
            foreach (var cell in unknown)
            {
                if (mines.Add(cell))
                {
                    changed = true;
                }
            }
        }

        return changed;
    }

    private void TryAdvancedDeductions()
    {
        var constraints = new List<Constraint>();

        for (int y = 0; y < H; y++)
        {
            for (int x = 0; x < W; x++)
            {
                if (char.IsDigit(board[y][x]))
                {
                    var c = BuildConstraint(x, y);
                    if (c != null && c.unknowns.Count > 0)
                    {
                        constraints.Add(c);
                    }
                }
            }
        }

        bool changed = true;
        int iterations = 0;
        int maxIterations = Math.Min(10, constraints.Count);

        while (changed && iterations < maxIterations)
        {
            changed = false;
            iterations++;

            for (int i = 0; i < constraints.Count && i < 10000; i++)
            {
                for (int j = 0; j < constraints.Count && j < 10000; j++)
                {
                    if (i == j) continue;

                    var c1 = constraints[i];
                    var c2 = constraints[j];

                    if (c1.unknowns.IsSubsetOf(c2.unknowns) && c1.unknowns.Count < c2.unknowns.Count)
                    {
                        var diff = new HashSet<(int, int)>(c2.unknowns);
                        diff.ExceptWith(c1.unknowns);

                        int diffMines = c2.mines - c1.mines;

                        if (diffMines == 0)
                        {
                            foreach (var cell in diff)
                            {
                                if (safe.Add(cell))
                                {
                                    changed = true;
                                }
                            }
                        }
                        else if (diffMines == diff.Count)
                        {
                            foreach (var cell in diff)
                            {
                                if (mines.Add(cell))
                                {
                                    changed = true;
                                }
                            }
                        }
                    }
                }
            }

            if (changed)
            {
                constraints.Clear();
                for (int y = 0; y < H; y++)
                {
                    for (int x = 0; x < W; x++)
                    {
                        if (char.IsDigit(board[y][x]))
                        {
                            var c = BuildConstraint(x, y);
                            if (c != null && c.unknowns.Count > 0)
                            {
                                constraints.Add(c);
                            }
                        }
                    }
                }
            }
        }
    }

    private Constraint BuildConstraint(int x, int y)
    {
        if (!char.IsDigit(board[y][x])) return null;

        int mineCount = board[y][x] - '0';
        var neighbors = GetNeighbors(x, y);

        var unknown = new HashSet<(int, int)>();
        int knownMines = 0;

        foreach (var (nx, ny) in neighbors)
        {
            if (board[ny][nx] == ' ')
            {
                if (mines.Contains((nx, ny)))
                {
                    knownMines++;
                }
                else if (!safe.Contains((nx, ny)))
                {
                    unknown.Add((nx, ny));
                }
            }
            else if (board[ny][nx] == '*')
            {
                knownMines++;
            }
        }

        return new Constraint
        {
            unknowns = unknown,
            mines = mineCount - knownMines
        };
    }

    private List<(int, int)> GetNeighbors(int x, int y)
    {
        var result = new List<(int, int)>();
        for (int dy = -1; dy <= 1; dy++)
        {
            for (int dx = -1; dx <= 1; dx++)
            {
                if (dx == 0 && dy == 0) continue;
                int nx = x + dx;
                int ny = y + dy;
                if (nx >= 0 && nx < W && ny >= 0 && ny < H)
                {
                    result.Add((nx, ny));
                }
            }
        }
        return result;
    }

    private class Constraint
    {
        public HashSet<(int, int)> unknowns;
        public int mines;
    }
}
