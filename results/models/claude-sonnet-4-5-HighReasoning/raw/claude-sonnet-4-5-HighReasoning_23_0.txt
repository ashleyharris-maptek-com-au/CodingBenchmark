use std::io::{self, BufRead, Write};
use std::sync::{Arc, Mutex};
use std::thread;
use std::time::{Duration, Instant};

#[derive(Clone, Copy)]
struct GameState {
    x: f64,
    y: f64,
    vx: f64,
    vy: f64,
    angle: f64,
    angular_vel: f64,
    fuel: f64,
}

struct Environment {
    gravity: f64,
    max_thrust: f64,
    target_x: f64,
    target_y: f64,
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Read initial configuration
    let config_line = lines.next().unwrap().unwrap();
    let config: Vec<f64> = config_line
        .split_whitespace()
        .filter_map(|s| s.parse().ok())
        .collect();

    let height = config[1];
    let gravity = config[2];
    let max_thrust = config[3];

    // Read start and target positions
    let target_line = lines.next().unwrap().unwrap();
    let positions: Vec<f64> = target_line
        .split_whitespace()
        .filter_map(|s| s.parse().ok())
        .collect();

    let target_x = positions[2];
    let target_y = positions[3];

    // Read and discard map to avoid memory issues with huge maps
    let map_height = height as usize;
    for _ in 0..map_height {
        lines.next();
    }

    let env = Environment {
        gravity,
        max_thrust,
        target_x,
        target_y,
    };

    // Shared state between threads
    let state = Arc::new(Mutex::new(None::<GameState>));
    let state_reader = Arc::clone(&state);

    // Spawn input reader thread to continuously read state updates
    thread::spawn(move || {
        for line in lines {
            if let Ok(line) = line {
                let parts: Vec<f64> = line
                    .split_whitespace()
                    .filter_map(|s| s.parse().ok())
                    .collect();

                if parts.len() >= 7 {
                    let new_state = GameState {
                        x: parts[0],
                        y: parts[1],
                        vx: parts[2],
                        vy: parts[3],
                        angle: parts[4],
                        angular_vel: parts[5],
                        fuel: parts[6],
                    };
                    if let Ok(mut state_lock) = state_reader.lock() {
                        *state_lock = Some(new_state);
                    }
                }
            }
        }
    });

    // Main control loop at 50Hz (exceeds 20Hz requirement)
    let output_interval = Duration::from_millis(20);
    let mut last_output = Instant::now();

    let stdout = io::stdout();
    let mut stdout_lock = stdout.lock();

    loop {
        if last_output.elapsed() >= output_interval {
            let current_state = state.lock().ok().and_then(|s| *s);

            if let Some(s) = current_state {
                let (thrust, turn) = compute_control(&s, &env);
                writeln!(stdout_lock, "{} {}", thrust, turn).ok();
            } else {
                // No state yet, output safe defaults
                writeln!(stdout_lock, "0.0 0.0").ok();
            }
            stdout_lock.flush().ok();

            last_output = Instant::now();
        }

        thread::sleep(Duration::from_millis(1));
    }
}

fn compute_control(state: &GameState, env: &Environment) -> (f64, f64) {
    let dx = env.target_x - state.x;
    let dy = env.target_y - state.y;

    // Calculate distance metrics
    let altitude_above_target = dy;
    let horizontal_offset = dx.abs();

    // Phase-based target velocities for smooth landing
    let (target_vx, target_vy) = if altitude_above_target < 2.0 && horizontal_offset < 1.5 {
        // Final landing phase: minimal velocity for soft touchdown
        (0.0, 0.3)
    } else if altitude_above_target < 5.0 && horizontal_offset < 5.0 {
        // Approach phase: slow down significantly
        (dx.signum() * (horizontal_offset * 0.5).min(1.0), 0.8)
    } else if altitude_above_target < 10.0 {
        // Descent phase: moderate velocities
        (dx.signum() * (horizontal_offset * 0.4).min(2.0), 1.5)
    } else {
        // High altitude: navigate toward target while descending
        (dx.signum() * (horizontal_offset * 0.3).min(3.0), 2.5)
    };

    // Calculate velocity errors
    let vx_error = state.vx - target_vx;
    let vy_error = state.vy - target_vy;

    // Proportional control for desired accelerations
    let kp_x = 2.0;
    let kp_y = 2.5;
    let desired_ax = -vx_error * kp_x;
    let desired_ay = -vy_error * kp_y;

    // Required acceleration accounting for gravity
    // Physics: ax = thrust * max_thrust * sin(angle)
    //          ay = -thrust * max_thrust * cos(angle) + gravity
    let required_ax = desired_ax;
    let required_ay = desired_ay - env.gravity;

    // Calculate desired angle to achieve required acceleration vector
    let desired_angle = required_ax.atan2(-required_ay);
    let desired_angle = desired_angle.clamp(-1.0, 1.0);

    // Calculate required thrust magnitude
    let required_accel_mag = (required_ax * required_ax + required_ay * required_ay).sqrt();
    let mut thrust = (required_accel_mag / env.max_thrust).clamp(0.0, 1.0);

    // Ensure minimum thrust when airborne to maintain control authority
    if altitude_above_target > 0.5 {
        thrust = thrust.max(0.15);
    }

    // Be conservative with fuel when far from target
    if state.fuel < 2.0 && altitude_above_target > 5.0 {
        thrust = thrust.min(0.5);
    }

    // PD control for turning (proportional-derivative)
    let angle_error = desired_angle - state.angle;
    let kp_angle = 4.0;
    let kd_angle = 0.5;
    let turn = (angle_error * kp_angle - state.angular_vel * kd_angle).clamp(-1.0, 1.0);

    (thrust, turn)
}
