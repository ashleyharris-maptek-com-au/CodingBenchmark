using System;
using System.Collections.Generic;
using System.Linq;

class OrbitalTSP
{
    struct OrbitalState
    {
        public double X, Y, Z;
        public double VX, VY, VZ;

        public OrbitalState(double x, double y, double z, double vx, double vy, double vz)
        {
            X = x; Y = y; Z = z;
            VX = vx; VY = vy; VZ = vz;
        }
    }

    // Estimate delta-V cost between two orbital states
    static double CalculateDeltaV(OrbitalState from, OrbitalState to, double mu)
    {
        // Position difference
        double dx = to.X - from.X;
        double dy = to.Y - from.Y;
        double dz = to.Z - from.Z;
        double distance = Math.Sqrt(dx*dx + dy*dy + dz*dz);

        // Velocity difference (direct proxy for delta-V)
        double dvx = to.VX - from.VX;
        double dvy = to.VY - from.VY;
        double dvz = to.VZ - from.VZ;
        double velocityDiff = Math.Sqrt(dvx*dvx + dvy*dvy + dvz*dvz);

        // Orbital radii
        double r1 = Math.Sqrt(from.X*from.X + from.Y*from.Y + from.Z*from.Z);
        double r2 = Math.Sqrt(to.X*to.X + to.Y*to.Y + to.Z*to.Z);

        // Heuristic: velocity difference + normalized distance penalty
        // This approximates transfer cost without full Lambert solver
        double cost = velocityDiff + distance / 20000.0;

        return cost;
    }

    // Build cost matrix for all station pairs plus start position
    static double[,] BuildCostMatrix(OrbitalState start, List<OrbitalState> stations, double mu)
    {
        int n = stations.Count;
        double[,] cost = new double[n + 1, n + 1];

        // Cost from start (index n) to each station
        for (int i = 0; i < n; i++)
        {
            cost[n, i] = CalculateDeltaV(start, stations[i], mu);
            cost[i, n] = double.MaxValue / 2; // No return to start
        }

        // Cost between stations
        for (int i = 0; i < n; i++)
        {
            for (int j = 0; j < n; j++)
            {
                if (i == j)
                    cost[i, j] = double.MaxValue / 2;
                else
                    cost[i, j] = CalculateDeltaV(stations[i], stations[j], mu);
            }
        }

        cost[n, n] = 0;
        return cost;
    }

    // Nearest Neighbor heuristic
    static List<int> SolveTSP_NearestNeighbor(double[,] cost, int n, int startIdx)
    {
        List<int> path = new List<int>();
        bool[] visited = new bool[n];
        int current = startIdx;

        for (int i = 0; i < n; i++)
        {
            int nearest = -1;
            double minCost = double.MaxValue;

            for (int j = 0; j < n; j++)
            {
                if (!visited[j] && cost[current, j] < minCost)
                {
                    minCost = cost[current, j];
                    nearest = j;
                }
            }

            if (nearest != -1)
            {
                visited[nearest] = true;
                path.Add(nearest);
                current = nearest;
            }
        }

        return path;
    }

    // Calculate total path cost
    static double CalculatePathCost(List<int> path, double[,] cost, int startIdx)
    {
        if (path.Count == 0) return 0;

        double total = cost[startIdx, path[0]];
        for (int i = 0; i < path.Count - 1; i++)
            total += cost[path[i], path[i + 1]];

        return total;
    }

    // 2-opt local search improvement
    static void Improve2Opt(List<int> path, double[,] cost, int startIdx, int maxPasses = 10)
    {
        int n = path.Count;
        if (n < 2) return;

        for (int pass = 0; pass < maxPasses; pass++)
        {
            bool improved = false;

            for (int i = 0; i < n - 1; i++)
            {
                for (int j = i + 1; j < n; j++)
                {
                    // Calculate cost change if we reverse segment [i, j]
                    int before_i = (i == 0) ? startIdx : path[i - 1];
                    int after_j = (j == n - 1) ? -1 : path[j + 1];

                    double oldCost = cost[before_i, path[i]];
                    double newCost = cost[before_i, path[j]];

                    if (after_j != -1)
                    {
                        oldCost += cost[path[j], after_j];
                        newCost += cost[path[i], after_j];
                    }

                    if (newCost < oldCost - 1e-9)
                    {
                        // Reverse segment [i, j]
                        int left = i, right = j;
                        while (left < right)
                        {
                            int temp = path[left];
                            path[left] = path[right];
                            path[right] = temp;
                            left++;
                            right--;
                        }
                        improved = true;
                    }
                }
            }

            if (!improved) break;
        }
    }

    // Simulated Annealing for large problems
    static List<int> SolveTSP_SimulatedAnnealing(double[,] cost, int n, int startIdx, int maxIterations)
    {
        Random rand = new Random(42);

        // Start with nearest neighbor
        List<int> current = SolveTSP_NearestNeighbor(cost, n, startIdx);
        List<int> best = new List<int>(current);

        double currentCost = CalculatePathCost(current, cost, startIdx);
        double bestCost = currentCost;

        double temperature = 10.0;
        double coolingRate = 0.99995;
        double minTemp = 0.001;

        for (int iter = 0; iter < maxIterations && temperature > minTemp; iter++)
        {
            // Random 2-opt move: pick two positions and reverse segment
            int i = rand.Next(n);
            int j = rand.Next(n);

            if (i > j)
            {
                int temp = i;
                i = j;
                j = temp;
            }

            if (i == j) continue;

            // Calculate delta for this 2-opt move
            int before_i = (i == 0) ? startIdx : current[i - 1];
            int after_j = (j == n - 1) ? -1 : current[j + 1];

            double oldCost = cost[before_i, current[i]];
            double newCost = cost[before_i, current[j]];

            if (after_j != -1)
            {
                oldCost += cost[current[j], after_j];
                newCost += cost[current[i], after_j];
            }

            double delta = newCost - oldCost;

            // Accept move based on simulated annealing criterion
            if (delta < 0 || rand.NextDouble() < Math.Exp(-delta / temperature))
            {
                // Apply 2-opt: reverse segment [i, j]
                int left = i, right = j;
                while (left < right)
                {
                    int tmp = current[left];
                    current[left] = current[right];
                    current[right] = tmp;
                    left++;
                    right--;
                }

                currentCost += delta;

                if (currentCost < bestCost)
                {
                    bestCost = currentCost;
                    best = new List<int>(current);
                }
            }

            temperature *= coolingRate;
        }

        return best;
    }

    // Brute force with branch-and-bound for small N
    static List<int> SolveTSP_BranchAndBound(double[,] cost, int n, int startIdx)
    {
        List<int> bestPath = SolveTSP_NearestNeighbor(cost, n, startIdx);
        double bestCost = CalculatePathCost(bestPath, cost, startIdx);

        List<int> currentPath = new List<int>();
        bool[] visited = new bool[n];

        BranchAndBound(cost, n, startIdx, currentPath, visited, 0, startIdx, ref bestPath, ref bestCost);

        return bestPath;
    }

    static void BranchAndBound(double[,] cost, int n, int startIdx, List<int> currentPath,
        bool[] visited, double currentCost, int lastNode, ref List<int> bestPath, ref double bestCost)
    {
        if (currentPath.Count == n)
        {
            if (currentCost < bestCost)
            {
                bestCost = currentCost;
                bestPath = new List<int>(currentPath);
            }
            return;
        }

        // Try each unvisited node
        for (int i = 0; i < n; i++)
        {
            if (visited[i]) continue;

            double newCost = currentCost + cost[lastNode, i];

            // Prune if already worse than best
            if (newCost >= bestCost) continue;

            visited[i] = true;
            currentPath.Add(i);

            BranchAndBound(cost, n, startIdx, currentPath, visited, newCost, i, ref bestPath, ref bestCost);

            currentPath.RemoveAt(currentPath.Count - 1);
            visited[i] = false;
        }
    }

    static void Main()
    {
        // Parse input
        string[] firstLine = Console.ReadLine().Split();
        int n = int.Parse(firstLine[0]);
        double mu = double.Parse(firstLine[1]);

        string[] startLine = Console.ReadLine().Split();
        OrbitalState start = new OrbitalState(
            double.Parse(startLine[0]), double.Parse(startLine[1]), double.Parse(startLine[2]),
            double.Parse(startLine[3]), double.Parse(startLine[4]), double.Parse(startLine[5])
        );

        List<OrbitalState> stations = new List<OrbitalState>();
        for (int i = 0; i < n; i++)
        {
            string[] stationLine = Console.ReadLine().Split();
            stations.Add(new OrbitalState(
                double.Parse(stationLine[0]), double.Parse(stationLine[1]), double.Parse(stationLine[2]),
                double.Parse(stationLine[3]), double.Parse(stationLine[4]), double.Parse(stationLine[5])
            ));
        }

        // Build cost matrix (index n = start position)
        double[,] cost = BuildCostMatrix(start, stations, mu);

        // Adaptive TSP solver based on problem size
        List<int> solution;

        if (n <= 8)
        {
            // Small: Branch-and-bound
            solution = SolveTSP_BranchAndBound(cost, n, n);
        }
        else if (n <= 15)
        {
            // Medium: Nearest neighbor + 2-opt
            solution = SolveTSP_NearestNeighbor(cost, n, n);
            Improve2Opt(solution, cost, n, 20);
        }
        else if (n <= 30)
        {
            // Large: Multiple NN starts + 2-opt
            solution = SolveTSP_NearestNeighbor(cost, n, n);
            Improve2Opt(solution, cost, n, 50);

            // Try a few random start points
            Random rand = new Random(42);
            for (int trial = 0; trial < Math.Min(5, n); trial++)
            {
                List<int> candidate = Enumerable.Range(0, n).OrderBy(x => rand.Next()).ToList();
                Improve2Opt(candidate, cost, n, 50);

                double candidateCost = CalculatePathCost(candidate, cost, n);
                double solutionCost = CalculatePathCost(solution, cost, n);

                if (candidateCost < solutionCost)
                    solution = candidate;
            }
        }
        else
        {
            // Very large: Simulated annealing
            int iterations = Math.Min(200000, n * 2000);
            solution = SolveTSP_SimulatedAnnealing(cost, n, n, iterations);

            // Final 2-opt polish
            Improve2Opt(solution, cost, n, 10);
        }

        // Output solution
        Console.WriteLine(string.Join(" ", solution));
    }
}
