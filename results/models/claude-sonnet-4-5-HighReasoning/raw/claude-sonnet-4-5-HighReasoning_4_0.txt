#include <cmath>

using namespace std;

const double PI = 3.14159265358979323846;
const double TIME_LIMIT = 29.0; // seconds

struct Vec3 {
    double x, y, z;

    Vec3(double x = 0, double y = 0, double z = 0) : x(x), y(y), z(z) {}

    Vec3 operator+(const Vec3& v) const { return Vec3(x + v.x, y + v.y, z + v.z); }
    Vec3 operator-(const Vec3& v) const { return Vec3(x - v.x, y - v.y, z - v.z); }
    Vec3 operator*(double s) const { return Vec3(x * s, y * s, z * s); }
    Vec3 operator/(double s) const { return Vec3(x / s, y / s, z / s); }

    double dot(const Vec3& v) const { return x * v.x + y * v.y + z * v.z; }

    Vec3 cross(const Vec3& v) const {
        return Vec3(y * v.z - z * v.y, z * v.x - x * v.z, x * v.y - y * v.x);
    }

    double length() const { return sqrt(x * x + y * y + z * z); }
    double lengthSquared() const { return x * x + y * y + z * z; }
    Vec3 normalized() const {
        double len = length();
        return len > 1e-10 ? *this / len : Vec3(0, 0, 0);
    }
};

struct Tetrahedron {
    Vec3 center;
    double rx, ry, rz; // Euler angles
    vector<Vec3> vertices;
    double boundingRadius;

    Tetrahedron(const Vec3& c, double rx, double ry, double rz, double edge_length)
        : center(c), rx(rx), ry(ry), rz(rz) {
        computeVertices(edge_length);
        boundingRadius = edge_length * sqrt(3.0 / 8.0); // circumradius
    }

    void computeVertices(double e) {
        // Create regular tetrahedron with edge length e, centered at origin
        // Using standard coordinates for regular tetrahedron
        double a = e / sqrt(2.0);
        vector<Vec3> localVerts = {
            Vec3(a, 0, -e / (2.0 * sqrt(2.0))),
            Vec3(-a / 2.0, a * sqrt(3.0) / 2.0, -e / (2.0 * sqrt(2.0))),
            Vec3(-a / 2.0, -a * sqrt(3.0) / 2.0, -e / (2.0 * sqrt(2.0))),
            Vec3(0, 0, e * sqrt(3.0) / (2.0 * sqrt(2.0)))
        };

        vertices.clear();
        for (const auto& v : localVerts) {
            Vec3 rotated = rotatePoint(v, rx, ry, rz);
            vertices.push_back(rotated + center);
        }
    }

    static Vec3 rotatePoint(const Vec3& v, double rx, double ry, double rz) {
        Vec3 result = v;

        // Rotate around X axis
        double y1 = result.y * cos(rx) - result.z * sin(rx);
        double z1 = result.y * sin(rx) + result.z * cos(rx);
        result.y = y1;
        result.z = z1;

        // Rotate around Y axis
        double x2 = result.x * cos(ry) + result.z * sin(ry);
        double z2 = -result.x * sin(ry) + result.z * cos(ry);
        result.x = x2;
        result.z = z2;

        // Rotate around Z axis
        double x3 = result.x * cos(rz) - result.y * sin(rz);
        double y3 = result.x * sin(rz) + result.y * cos(rz);
        result.x = x3;
        result.y = y3;

        return result;
    }
};

struct BoundingBox {
    Vec3 min, max;

    BoundingBox() : min(1e9, 1e9, 1e9), max(-1e9, -1e9, -1e9) {}

    void expand(const Vec3& v) {
        min.x = std::min(min.x, v.x);
        min.y = std::min(min.y, v.y);
        min.z = std::min(min.z, v.z);
        max.x = std::max(max.x, v.x);
        max.y = std::max(max.y, v.y);
        max.z = std::max(max.z, v.z);
    }

    bool contains(const Vec3& v, double margin = 0) const {
        return v.x >= min.x - margin && v.x <= max.x + margin &&
               v.y >= min.y - margin && v.y <= max.y + margin &&
               v.z >= min.z - margin && v.z <= max.z + margin;
    }

    double volume() const {
        return (max.x - min.x) * (max.y - min.y) * (max.z - min.z);
    }

    Vec3 size() const {
        return max - min;
    }
};

struct SpatialHash {
    double cellSize;
    unordered_map<long long, vector<int>> grid;

    SpatialHash(double cellSize) : cellSize(cellSize) {}

    long long hash(int x, int y, int z) const {
        return ((long long)x * 73856093) ^ ((long long)y * 19349663) ^ ((long long)z * 83492791);
    }

    long long getKey(const Vec3& pos) const {
        int x = (int)floor(pos.x / cellSize);
        int y = (int)floor(pos.y / cellSize);
        int z = (int)floor(pos.z / cellSize);
        return hash(x, y, z);
    }

    void insert(const Vec3& pos, int id) {
        grid[getKey(pos)].push_back(id);
    }

    vector<int> query(const Vec3& pos, double radius) const {
        vector<int> result;
        int r = (int)ceil(radius / cellSize);
        int cx = (int)floor(pos.x / cellSize);
        int cy = (int)floor(pos.y / cellSize);
        int cz = (int)floor(pos.z / cellSize);

        for (int dx = -r; dx <= r; dx++) {
            for (int dy = -r; dy <= r; dy++) {
                for (int dz = -r; dz <= r; dz++) {
                    long long key = hash(cx + dx, cy + dy, cz + dz);
                    auto it = grid.find(key);
                    if (it != grid.end()) {
                        result.insert(result.end(), it->second.begin(), it->second.end());
                    }
                }
            }
        }
        return result;
    }
};

bool checkTetrahedronCollision(const Tetrahedron& t1, const Tetrahedron& t2) {
    // Quick bounding sphere check
    double dist = (t1.center - t2.center).length();
    double minDist = t1.boundingRadius + t2.boundingRadius;

    if (dist > minDist) return false;

    // More detailed check: vertex-to-vertex distance
    for (const auto& v1 : t1.vertices) {
        for (const auto& v2 : t2.vertices) {
            if ((v1 - v2).lengthSquared() < 0.01) { // vertices too close
                return true;
            }
        }
    }

    return dist < minDist * 0.8; // conservative overlap check
}

bool isTetrahedronInside(const Tetrahedron& tet, const BoundingBox& bounds, double edge_length) {
    double margin = edge_length * 0.1;
    for (const auto& v : tet.vertices) {
        if (!bounds.contains(v, -margin)) {
            return false;
        }
    }
    return true;
}

class TetrahedronPacker {
private:
    vector<Vec3> containerVertices;
    BoundingBox bounds;
    double edgeLength;
    vector<Tetrahedron> placed;
    SpatialHash* spatialHash;
    chrono::steady_clock::time_point startTime;

public:
    TetrahedronPacker(const vector<Vec3>& vertices, double edge)
        : containerVertices(vertices), edgeLength(edge), spatialHash(nullptr) {

        // Compute bounding box
        for (const auto& v : vertices) {
            bounds.expand(v);
        }

        startTime = chrono::steady_clock::now();
    }

    ~TetrahedronPacker() {
        delete spatialHash;
    }

    bool timeExpired() const {
        auto now = chrono::steady_clock::now();
        double elapsed = chrono::duration<double>(now - startTime).count();
        return elapsed > TIME_LIMIT;
    }

    void pack() {
        double volume = bounds.volume();
        double tetVolume = edgeLength * edgeLength * edgeLength * sqrt(2.0) / 12.0;
        double estimatedCount = volume / tetVolume * 0.3; // rough packing efficiency

        // Adapt resolution based on container size
        double gridStep;
        vector<double> rotations;

        if (estimatedCount < 100) {
            // Small container: fine grid, many rotations
            gridStep = edgeLength * 0.4;
            rotations = {0, PI/6, PI/4, PI/3, PI/2, 2*PI/3, 3*PI/4, 5*PI/6};
        } else if (estimatedCount < 10000) {
            // Medium container: medium grid
            gridStep = edgeLength * 0.5;
            rotations = {0, PI/4, PI/2, 3*PI/4};
        } else {
            // Large container: coarse grid, fewer rotations
            gridStep = edgeLength * 0.7;
            rotations = {0, PI/3, 2*PI/3};
        }

        // Initialize spatial hash
        spatialHash = new SpatialHash(edgeLength * 2.0);

        Vec3 size = bounds.size();
        int nx = (int)(size.x / gridStep) + 1;
        int ny = (int)(size.y / gridStep) + 1;
        int nz = (int)(size.z / gridStep) + 1;

        // Try to place tetrahedrons on grid
        for (int ix = 0; ix < nx && !timeExpired(); ix++) {
            double x = bounds.min.x + ix * gridStep;

            for (int iy = 0; iy < ny && !timeExpired(); iy++) {
                double y = bounds.min.y + iy * gridStep;

                for (int iz = 0; iz < nz && !timeExpired(); iz++) {
                    double z = bounds.min.z + iz * gridStep;
                    Vec3 center(x, y, z);

                    // Try different orientations
                    bool placed_here = false;
                    for (double rx : rotations) {
                        if (placed_here) break;
                        for (double ry : rotations) {
                            if (placed_here) break;
                            for (double rz : rotations) {
                                if (tryPlaceTetrahedron(center, rx, ry, rz)) {
                                    placed_here = true;
                                    break;
                                }
                            }
                        }
                    }
                }
            }
        }
    }

    bool tryPlaceTetrahedron(const Vec3& center, double rx, double ry, double rz) {
        Tetrahedron tet(center, rx, ry, rz, edgeLength);

        // Check if inside container bounds
        if (!isTetrahedronInside(tet, bounds, edgeLength)) {
            return false;
        }

        // Check collision with nearby tetrahedrons
        vector<int> nearby = spatialHash->query(center, edgeLength * 3.0);
        for (int idx : nearby) {
            if (checkTetrahedronCollision(tet, placed[idx])) {
                return false;
            }
        }

        // Place tetrahedron
        int id = placed.size();
        placed.push_back(tet);
        spatialHash->insert(center, id);

        return true;
    }

    const vector<Tetrahedron>& getPlaced() const {
        return placed;
    }
};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int V;
    double edgeLength;
    cin >> V >> edgeLength;

    vector<Vec3> containerVertices(V);
    for (int i = 0; i < V; i++) {
        cin >> containerVertices[i].x >> containerVertices[i].y >> containerVertices[i].z;
    }

    TetrahedronPacker packer(containerVertices, edgeLength);
    packer.pack();

    const auto& placed = packer.getPlaced();

    cout << placed.size() << "\n";
    cout << fixed << setprecision(6);
    for (const auto& tet : placed) {
        cout << tet.center.x << " " << tet.center.y << " " << tet.center.z << " "
             << tet.rx << " " << tet.ry << " " << tet.rz << "\n";
    }

    return 0;
}
