const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { grid, robots, jobs } = data;
const height = grid.length;
const width = grid[0].length;

// BFS to find shortest distances from a start position to all reachable cells
function bfs(startX, startY) {
  const distances = Array(height).fill(null).map(() => Array(width).fill(Infinity));
  distances[startY][startX] = 0;
  const queue = [[startX, startY]];
  const directions = [[0, 1], [1, 0], [0, -1], [-1, 0]];

  for (let i = 0; i < queue.length; i++) {
    const [x, y] = queue[i];
    const dist = distances[y][x];

    for (const [dx, dy] of directions) {
      const nx = x + dx;
      const ny = y + dy;

      if (nx >= 0 && nx < width && ny >= 0 && ny < height &&
          grid[ny][nx] === '.' && distances[ny][nx] === Infinity) {
        distances[ny][nx] = dist + 1;
        queue.push([nx, ny]);
      }
    }
  }

  return distances;
}

// Cache distances from all relevant positions (robot starts and job locations)
const distanceCache = new Map();

function getDistances(x, y) {
  const key = `${x},${y}`;
  if (!distanceCache.has(key)) {
    distanceCache.set(key, bfs(x, y));
  }
  return distanceCache.get(key);
}

// Pre-compute distances from robot and job positions
robots.forEach(r => getDistances(r.x, r.y));
jobs.forEach(j => getDistances(j.x, j.y));

// Greedy assignment: prioritize high-priority jobs, assign to nearest robot
const assignments = robots.map(() => []);
const robotPositions = robots.map(r => ({ x: r.x, y: r.y }));
const robotTotalDistance = robots.map(() => 0);

// Sort jobs by priority (descending), then by id for determinism
const sortedJobs = [...jobs].sort((a, b) => {
  if (b.priority !== a.priority) return b.priority - a.priority;
  return a.id - b.id;
});

// Assign each job to the robot with minimum distance from its current position
for (const job of sortedJobs) {
  let bestRobot = 0;
  let bestDistance = Infinity;
  let bestTotalDistance = Infinity;

  for (let i = 0; i < robots.length; i++) {
    const distances = getDistances(robotPositions[i].x, robotPositions[i].y);
    const dist = distances[job.y][job.x];

    // Prefer shorter distance; break ties by choosing robot with less total travel
    if (dist < bestDistance || (dist === bestDistance && robotTotalDistance[i] < bestTotalDistance)) {
      bestDistance = dist;
      bestTotalDistance = robotTotalDistance[i];
      bestRobot = i;
    }
  }

  assignments[bestRobot].push(job.id);
  robotTotalDistance[bestRobot] += bestDistance;
  robotPositions[bestRobot] = { x: job.x, y: job.y };
}
