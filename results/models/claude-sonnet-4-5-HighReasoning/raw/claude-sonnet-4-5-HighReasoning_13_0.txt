#include <iostream>
#include <vector>
#include <string>
#include <unordered_set>
#include <unordered_map>
#include <algorithm>

using namespace std;

// Rolling hash for efficient substring matching
class RollingHash {
public:
    static const long long BASE = 31;
    static const long long MOD = 1e9 + 7;

    // Get all unique hashes for substrings of given length
    static unordered_set<long long> getHashes(const string& s, int len) {
        unordered_set<long long> hashes;
        if (s.length() < len) return hashes;

        // Precompute BASE^(len-1)
        long long pow = 1;
        for (int i = 0; i < len - 1; i++) {
            pow = (pow * BASE) % MOD;
        }

        // Compute initial hash
        long long hash = 0;
        for (int i = 0; i < len; i++) {
            hash = (hash * BASE + s[i]) % MOD;
        }
        hashes.insert(hash);

        // Rolling hash for remaining positions
        for (size_t i = len; i < s.length(); i++) {
            hash = ((hash - s[i - len] * pow % MOD + MOD) % MOD * BASE + s[i]) % MOD;
            hashes.insert(hash);
        }

        return hashes;
    }

    // Get hashes with one position per hash
    static unordered_map<long long, int> getHashesWithPositions(const string& s, int len) {
        unordered_map<long long, int> hashToPos;
        if (s.length() < len) return hashToPos;

        long long pow = 1;
        for (int i = 0; i < len - 1; i++) {
            pow = (pow * BASE) % MOD;
        }

        long long hash = 0;
        for (int i = 0; i < len; i++) {
            hash = (hash * BASE + s[i]) % MOD;
        }
        hashToPos[hash] = 0;

        for (size_t i = len; i < s.length(); i++) {
            hash = ((hash - s[i - len] * pow % MOD + MOD) % MOD * BASE + s[i]) % MOD;
            if (hashToPos.find(hash) == hashToPos.end()) {
                hashToPos[hash] = i - len + 1;
            }
        }

        return hashToPos;
    }
};

// Check if there exists a common substring of given length
bool hasCommonSubstring(const vector<string>& strings, int len) {
    if (len == 0) return true;
    if (strings.empty()) return false;

    // Start with hashes from first string
    auto commonHashes = RollingHash::getHashes(strings[0], len);
    if (commonHashes.empty()) return false;

    // Intersect with hashes from each subsequent string
    for (size_t i = 1; i < strings.size(); i++) {
        auto hashes = RollingHash::getHashes(strings[i], len);

        unordered_set<long long> intersection;
        for (long long h : commonHashes) {
            if (hashes.count(h)) {
                intersection.insert(h);
            }
        }

        commonHashes = move(intersection);
        if (commonHashes.empty()) return false;
    }

    return !commonHashes.empty();
}

// Extract the actual substring once we know the length
string findActualSubstring(const vector<string>& strings, int len) {
    if (len == 0) return "";
    if (strings.empty()) return "";

    // Use shortest string as reference to minimize candidates
    int shortestIdx = 0;
    for (size_t i = 1; i < strings.size(); i++) {
        if (strings[i].length() < strings[shortestIdx].length()) {
            shortestIdx = i;
        }
    }

    const string& shortest = strings[shortestIdx];
    if (shortest.length() < len) return "";

    // Get hashes from shortest string with positions
    auto shortestHashes = RollingHash::getHashesWithPositions(shortest, len);
    unordered_set<long long> commonHashes;
    for (const auto& [h, pos] : shortestHashes) {
        commonHashes.insert(h);
    }

    // Intersect with all other strings
    for (size_t i = 0; i < strings.size(); i++) {
        if (i == shortestIdx) continue;

        auto hashes = RollingHash::getHashes(strings[i], len);
        unordered_set<long long> intersection;
        for (long long h : commonHashes) {
            if (hashes.count(h)) {
                intersection.insert(h);
            }
        }
        commonHashes = move(intersection);
        if (commonHashes.empty()) return "";
    }

    // Find and verify a matching candidate (to handle hash collisions)
    for (const auto& [hash, pos] : shortestHashes) {
        if (commonHashes.count(hash)) {
            string candidate = shortest.substr(pos, len);

            // Verify against all strings
            bool verified = true;
            for (size_t i = 0; i < strings.size(); i++) {
                if (i == shortestIdx) continue;
                if (strings[i].find(candidate) == string::npos) {
                    verified = false;
                    break;
                }
            }

            if (verified) return candidate;
        }
    }

    return "";
}

string solve(const vector<string>& strings) {
    if (strings.empty()) return "";

    // Find minimum string length
    size_t minLen = strings[0].length();
    for (const auto& s : strings) {
        minLen = min(minLen, s.length());
    }

    if (minLen == 0) return "";

    // Binary search on substring length
    int left = 0, right = minLen;
    int maxLen = 0;

    while (left <= right) {
        int mid = (left + right) / 2;
        if (hasCommonSubstring(strings, mid)) {
            maxLen = mid;
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }

    if (maxLen == 0) return "";
    return findActualSubstring(strings, maxLen);
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    cin.ignore();

    vector<string> strings(n);
    for (int i = 0; i < n; i++) {
        getline(cin, strings[i]);
    }

    string result = solve(strings);
    cout << result << endl;

    return 0;
}
