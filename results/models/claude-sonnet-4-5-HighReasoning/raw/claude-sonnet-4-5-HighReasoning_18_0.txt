#include <iostream>
#include <vector>
#include <algorithm>
#include <limits>
#include <set>
#include <queue>
#include <cstring>

using namespace std;

class MinCutSolver {
private:
    int n, m;
    vector<pair<int, int>> originalEdges;

    // Stoer-Wagner algorithm for global minimum cut
    int stoerWagner(vector<vector<int>>& capacity, set<int>& bestCutSet) {
        int vertices = n;
        vector<int> merged(n);
        for (int i = 0; i < n; i++) merged[i] = i;

        vector<set<int>> components(n);
        for (int i = 0; i < n; i++) components[i].insert(i);

        int minCut = INT_MAX;

        while (vertices > 1) {
            vector<int> weight(n, 0);
            vector<bool> added(n, false);

            int prev = -1, last = -1;

            // Maximum adjacency search
            for (int i = 0; i < vertices; i++) {
                int maxWeight = -1;
                int maxVertex = -1;

                for (int v = 0; v < n; v++) {
                    if (merged[v] == v && !added[v] && weight[v] > maxWeight) {
                        maxWeight = weight[v];
                        maxVertex = v;
                    }
                }

                if (maxVertex == -1) break;

                added[maxVertex] = true;
                prev = last;
                last = maxVertex;

                // Update weights
                for (int v = 0; v < n; v++) {
                    if (merged[v] == v && !added[v]) {
                        weight[v] += capacity[maxVertex][v];
                    }
                }
            }

            if (prev == -1 || last == -1) break;

            // Cut-of-the-phase: weight[last] is the cut value
            if (weight[last] < minCut) {
                minCut = weight[last];
                bestCutSet = components[last];
            }

            // Merge last into prev
            for (int v = 0; v < n; v++) {
                if (merged[v] == v && v != last) {
                    capacity[prev][v] += capacity[last][v];
                    capacity[v][prev] += capacity[v][last];
                }
            }
            capacity[prev][last] = 0;
            capacity[last][prev] = 0;

            // Update component tracking
            for (int v : components[last]) {
                components[prev].insert(v);
            }
            components[last].clear();

            merged[last] = prev;
            vertices--;
        }

        return minCut;
    }

    // For very large sparse graphs, use a more memory-efficient approach
    int sparseMinCut(set<int>& bestCutSet) {
        // Build adjacency list
        vector<vector<pair<int, int>>> adj(n);
        for (int i = 0; i < m; i++) {
            int u = originalEdges[i].first;
            int v = originalEdges[i].second;
            adj[u].push_back({v, i});
            adj[v].push_back({u, i});
        }

        int minCut = m;

        // Try a limited number of source-sink pairs using max-flow
        for (int t = 1; t < min(n, 50); t++) {
            // Build flow network
            vector<vector<int>> capacity(n, vector<int>(n, 0));
            for (auto& e : originalEdges) {
                capacity[e.first][e.second]++;
                capacity[e.second][e.first]++;
            }

            // Dinic's algorithm for max-flow
            auto bfs = [&](int s, int t, vector<int>& level) -> bool {
                fill(level.begin(), level.end(), -1);
                queue<int> q;
                level[s] = 0;
                q.push(s);

                while (!q.empty()) {
                    int v = q.front();
                    q.pop();
                    for (int u = 0; u < n; u++) {
                        if (capacity[v][u] > 0 && level[u] < 0) {
                            level[u] = level[v] + 1;
                            q.push(u);
                        }
                    }
                }
                return level[t] >= 0;
            };

            function<int(int, int, int, vector<int>&, vector<int>&)> dfs;
            dfs = [&](int v, int t, int pushed, vector<int>& level, vector<int>& iter) -> int {
                if (v == t) return pushed;

                for (int& i = iter[v]; i < n; i++) {
                    if (level[v] + 1 == level[i] && capacity[v][i] > 0) {
                        int tr = dfs(i, t, min(pushed, capacity[v][i]), level, iter);
                        if (tr > 0) {
                            capacity[v][i] -= tr;
                            capacity[i][v] += tr;
                            return tr;
                        }
                    }
                }
                return 0;
            };

            vector<int> level(n), iter(n);
            int flow = 0;
            while (bfs(0, t, level)) {
                fill(iter.begin(), iter.end(), 0);
                int pushed;
                while ((pushed = dfs(0, t, INT_MAX, level, iter)) > 0) {
                    flow += pushed;
                }
            }

            if (flow < minCut) {
                minCut = flow;
                // Extract reachable set from source
                bestCutSet.clear();
                vector<bool> visited(n, false);
                queue<int> q;
                q.push(0);
                visited[0] = true;

                while (!q.empty()) {
                    int v = q.front();
                    q.pop();
                    bestCutSet.insert(v);

                    for (int u = 0; u < n; u++) {
                        if (capacity[v][u] > 0 && !visited[u]) {
                            visited[u] = true;
                            q.push(u);
                        }
                    }
                }
            }
        }

        return minCut;
    }

public:
    void solve() {
        cin >> n >> m;

        originalEdges.resize(m);
        for (int i = 0; i < m; i++) {
            cin >> originalEdges[i].first >> originalEdges[i].second;
        }

        set<int> bestCutSet;
        int minCutValue;

        // Adaptive algorithm selection based on graph size
        if (n <= 500 || (long long)n * n * 4 <= 1000000000LL) {
            // Small to medium graphs: use Stoer-Wagner (exact)
            vector<vector<int>> capacity(n, vector<int>(n, 0));
            for (auto& e : originalEdges) {
                capacity[e.first][e.second]++;
                capacity[e.second][e.first]++;
            }
            minCutValue = stoerWagner(capacity, bestCutSet);
        } else {
            // Large graphs: use max-flow with limited iterations
            minCutValue = sparseMinCut(bestCutSet);
        }

        // Extract cut edges
        vector<pair<int, int>> cutEdges;
        for (auto& e : originalEdges) {
            bool u_in = (bestCutSet.count(e.first) > 0);
            bool v_in = (bestCutSet.count(e.second) > 0);
            if (u_in != v_in) {
                cutEdges.push_back(e);
            }
        }

        // Output
        cout << cutEdges.size() << "\n";
        for (auto& e : cutEdges) {
            cout << e.first << " " << e.second << "\n";
        }
    }
};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    MinCutSolver solver;
    solver.solve();

    return 0;
}
