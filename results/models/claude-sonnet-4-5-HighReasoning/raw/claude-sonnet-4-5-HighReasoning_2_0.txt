using System;
using System.Collections.Generic;
using System.Linq;

class Solution
{
    class Edge
    {
        public int To, Weight, Id;
        public Edge(int to, int weight, int id) { To = to; Weight = weight; Id = id; }
    }

    static void Main()
    {
        var input = Console.ReadLine().Split();
        int n = int.Parse(input[0]);
        int m = int.Parse(input[1]);

        var graph = new List<Edge>[n];
        for (int i = 0; i < n; i++)
            graph[i] = new List<Edge>();

        int nextEdgeId = 0;
        for (int i = 0; i < m; i++)
        {
            var parts = Console.ReadLine().Split();
            int u = int.Parse(parts[0]);
            int v = int.Parse(parts[1]);
            int w = int.Parse(parts[2]);

            graph[u].Add(new Edge(v, w, nextEdgeId));
            graph[v].Add(new Edge(u, w, nextEdgeId));
            nextEdgeId++;
        }

        var oddVertices = new List<int>();
        for (int i = 0; i < n; i++)
            if (graph[i].Count % 2 == 1)
                oddVertices.Add(i);

        if (oddVertices.Count > 0)
        {
            var distCache = new Dictionary<int, (int[] dist, int[] prev)>();

            var pairs = new List<(int u, int v, int dist)>();
            for (int i = 0; i < oddVertices.Count; i++)
            {
                int u = oddVertices[i];
                if (!distCache.ContainsKey(u))
                    distCache[u] = Dijkstra(graph, n, u);

                var (dist, _) = distCache[u];
                for (int j = i + 1; j < oddVertices.Count; j++)
                {
                    int v = oddVertices[j];
                    pairs.Add((u, v, dist[v]));
                }
            }

            pairs.Sort((a, b) => a.dist.CompareTo(b.dist));

            var used = new HashSet<int>();
            foreach (var (u, v, _) in pairs)
            {
                if (used.Contains(u) || used.Contains(v))
                    continue;

                used.Add(u);
                used.Add(v);

                var (_, prev) = distCache[u];
                var path = new List<int>();
                for (int at = v; at != -1; at = prev[at])
                    path.Add(at);
                path.Reverse();

                for (int i = 0; i < path.Count - 1; i++)
                {
                    int from = path[i], to = path[i + 1];
                    var origEdge = graph[from].First(e => e.To == to);

                    graph[from].Add(new Edge(to, origEdge.Weight, nextEdgeId));
                    graph[to].Add(new Edge(from, origEdge.Weight, nextEdgeId));
                    nextEdgeId++;
                }
            }
        }

        var circuit = FindEulerianCircuit(graph, n, 0);
        Console.WriteLine(string.Join(" ", circuit));
    }

    static (int[] dist, int[] prev) Dijkstra(List<Edge>[] graph, int n, int start)
    {
        var dist = new int[n];
        var prev = new int[n];
        Array.Fill(dist, int.MaxValue);
        Array.Fill(prev, -1);
        dist[start] = 0;

        var pq = new PriorityQueue<int, int>();
        pq.Enqueue(start, 0);
        var visited = new bool[n];

        while (pq.Count > 0)
        {
            int u = pq.Dequeue();
            if (visited[u]) continue;
            visited[u] = true;

            foreach (var edge in graph[u])
            {
                int v = edge.To;
                int newDist = dist[u] + edge.Weight;
                if (newDist < dist[v])
                {
                    dist[v] = newDist;
                    prev[v] = u;
                    pq.Enqueue(v, newDist);
                }
            }
        }

        return (dist, prev);
    }

    static List<int> FindEulerianCircuit(List<Edge>[] graph, int n, int start)
    {
        var adj = new List<Edge>[n];
        for (int i = 0; i < n; i++)
            adj[i] = new List<Edge>(graph[i]);

        var circuit = new List<int>();
        var stack = new Stack<int>();
        stack.Push(start);
        var usedEdges = new HashSet<int>();

        while (stack.Count > 0)
        {
            int u = stack.Peek();
            Edge nextEdge = null;

            for (int i = adj[u].Count - 1; i >= 0; i--)
            {
                var edge = adj[u][i];
                if (!usedEdges.Contains(edge.Id))
                {
                    nextEdge = edge;
                    adj[u].RemoveAt(i);
                    break;
                }
            }

            if (nextEdge != null)
            {
                usedEdges.Add(nextEdge.Id);
                stack.Push(nextEdge.To);
            }
            else
            {
                circuit.Add(stack.Pop());
            }
        }

        circuit.Reverse();
        return circuit;
    }
}
