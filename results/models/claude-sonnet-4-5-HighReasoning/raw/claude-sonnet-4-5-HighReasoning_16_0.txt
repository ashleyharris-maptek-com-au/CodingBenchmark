#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>

using namespace std;

struct Task {
    int machine;
    int duration;
    int start_time = -1;
};

class JobShopScheduler {
private:
    int num_jobs;
    int num_machines;
    vector<vector<Task>> jobs;
    vector<int> machine_available;
    vector<int> job_available;

    struct ReadyTask {
        int job_id;
        int task_id;
        int duration;

        bool operator<(const ReadyTask& other) const {
            // SPT (Shortest Processing Time) priority
            if (duration != other.duration) return duration > other.duration;
            if (job_id != other.job_id) return job_id > other.job_id;
            return task_id > other.task_id;
        }
    };

public:
    void read_input() {
        cin >> num_jobs >> num_machines;
        jobs.resize(num_jobs);
        machine_available.resize(num_machines, 0);
        job_available.resize(num_jobs, 0);

        for (int j = 0; j < num_jobs; j++) {
            int num_tasks;
            cin >> num_tasks;
            jobs[j].resize(num_tasks);

            for (int t = 0; t < num_tasks; t++) {
                cin >> jobs[j][t].machine >> jobs[j][t].duration;
            }
        }
    }

    void solve() {
        size_t total_tasks = 0;
        for (const auto& job : jobs) {
            total_tasks += job.size();
        }

        // Adaptive strategy: small problems try multiple heuristics
        if (num_jobs <= 50 && num_machines <= 20 && total_tasks <= 1000) {
            solve_with_multiple_heuristics();
        } else {
            // Large problems: use fast greedy SPT
            solve_greedy_spt();
        }
    }

    void solve_greedy_spt() {
        priority_queue<ReadyTask> ready;
        vector<int> next_task_idx(num_jobs, 0);

        // Initialize with first task of each job
        for (int j = 0; j < num_jobs; j++) {
            if (!jobs[j].empty()) {
                ready.push({j, 0, jobs[j][0].duration});
            }
        }

        while (!ready.empty()) {
            ReadyTask current = ready.top();
            ready.pop();

            int j = current.job_id;
            int t = current.task_id;

            // Skip outdated entries
            if (t != next_task_idx[j]) continue;

            Task& task = jobs[j][t];
            int m = task.machine;

            // Schedule at earliest time when both job and machine are ready
            int start = max(job_available[j], machine_available[m]);
            task.start_time = start;

            int finish = start + task.duration;
            machine_available[m] = finish;
            job_available[j] = finish;
            next_task_idx[j]++;

            // Add next task from this job to ready queue
            if (t + 1 < (int)jobs[j].size()) {
                ready.push({j, t + 1, jobs[j][t + 1].duration});
            }
        }
    }

    void solve_with_multiple_heuristics() {
        vector<vector<vector<Task>>> solutions;
        vector<int> makespans;

        // Try SPT (Shortest Processing Time)
        solve_greedy_spt();
        solutions.push_back(jobs);
        makespans.push_back(calculate_makespan());

        // Try LPT (Longest Processing Time)
        reset();
        solve_greedy_lpt();
        solutions.push_back(jobs);
        makespans.push_back(calculate_makespan());

        // Try FCFS (First Come First Served)
        reset();
        solve_fcfs();
        solutions.push_back(jobs);
        makespans.push_back(calculate_makespan());

        // Select best solution (minimum makespan)
        int best = 0;
        for (int i = 1; i < (int)solutions.size(); i++) {
            if (makespans[i] < makespans[best]) {
                best = i;
            }
        }
        jobs = solutions[best];
    }

    void solve_greedy_lpt() {
        struct LPTTask {
            int job_id, task_id, duration;
            bool operator<(const LPTTask& o) const {
                // LPT: Longest first
                if (duration != o.duration) return duration < o.duration;
                if (job_id != o.job_id) return job_id > o.job_id;
                return task_id > o.task_id;
            }
        };

        priority_queue<LPTTask> ready;
        vector<int> next_task_idx(num_jobs, 0);

        for (int j = 0; j < num_jobs; j++) {
            if (!jobs[j].empty()) {
                ready.push({j, 0, jobs[j][0].duration});
            }
        }

        while (!ready.empty()) {
            auto current = ready.top();
            ready.pop();

            int j = current.job_id;
            int t = current.task_id;
            if (t != next_task_idx[j]) continue;

            Task& task = jobs[j][t];
            int start = max(job_available[j], machine_available[task.machine]);
            task.start_time = start;

            int finish = start + task.duration;
            machine_available[task.machine] = finish;
            job_available[j] = finish;
            next_task_idx[j]++;

            if (t + 1 < (int)jobs[j].size()) {
                ready.push({j, t + 1, jobs[j][t + 1].duration});
            }
        }
    }

    void solve_fcfs() {
        queue<pair<int, int>> ready;
        vector<int> next_task_idx(num_jobs, 0);

        for (int j = 0; j < num_jobs; j++) {
            if (!jobs[j].empty()) {
                ready.push({j, 0});
            }
        }

        while (!ready.empty()) {
            auto [j, t] = ready.front();
            ready.pop();

            if (t != next_task_idx[j]) continue;

            Task& task = jobs[j][t];
            int start = max(job_available[j], machine_available[task.machine]);
            task.start_time = start;

            int finish = start + task.duration;
            machine_available[task.machine] = finish;
            job_available[j] = finish;
            next_task_idx[j]++;

            if (t + 1 < (int)jobs[j].size()) {
                ready.push({j, t + 1});
            }
        }
    }

    void reset() {
        fill(machine_available.begin(), machine_available.end(), 0);
        fill(job_available.begin(), job_available.end(), 0);
        for (auto& job : jobs) {
            for (auto& task : job) {
                task.start_time = -1;
            }
        }
    }

    int calculate_makespan() {
        int makespan = 0;
        for (const auto& job : jobs) {
            for (const auto& task : job) {
                if (task.start_time >= 0) {
                    makespan = max(makespan, task.start_time + task.duration);
                }
            }
        }
        return makespan;
    }

    void output() {
        for (const auto& job : jobs) {
            bool first = true;
            for (const auto& task : job) {
                if (!first) cout << " ";
                cout << task.start_time << " " << task.machine;
                first = false;
            }
            cout << "\n";
        }
    }
};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    JobShopScheduler scheduler;
    scheduler.read_input();
    scheduler.solve();
    scheduler.output();

    return 0;
}
