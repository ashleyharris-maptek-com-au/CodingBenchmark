#!/usr/bin/env python3
"""
Generate SPIR-V binary for inclusive prefix sum compute shader.
Outputs hex-encoded SPIR-V binary.
"""

import struct

class SPIRVBuilder:
    def __init__(self):
        self.words = []
        self.id_counter = 1

    def add_word(self, word):
        self.words.append(word)

    def add_words(self, *words):
        self.words.extend(words)

    def add_instruction(self, opcode, *operands):
        word_count = 1 + len(operands)
        self.words.append((word_count << 16) | opcode)
        self.words.extend(operands)

    def add_string(self, s):
        """Add string as sequence of words (null-terminated, padded)"""
        bytes_data = s.encode('utf-8') + b'\x00'
        # Pad to 4-byte boundary
        while len(bytes_data) % 4 != 0:
            bytes_data += b'\x00'
        for i in range(0, len(bytes_data), 4):
            word = struct.unpack('<I', bytes_data[i:i+4])[0]
            self.words.append(word)

    def new_id(self):
        result = self.id_counter
        self.id_counter += 1
        return result

    def to_hex(self):
        """Convert to hex string"""
        result = ""
        for word in self.words:
            result += f"{word:08x}"
        return result

def build_shader():
    b = SPIRVBuilder()

    # Header
    b.add_word(0x07230203)  # Magic number
    b.add_word(0x00010000)  # Version 1.0
    b.add_word(0x00000000)  # Generator (0 = unknown)
    bound_placeholder_idx = len(b.words)
    b.add_word(0x00000000)  # Bound (will update later)
    b.add_word(0x00000000)  # Schema

    # Allocate all IDs upfront
    id_void = b.new_id()
    id_func_type = b.new_id()
    id_uint = b.new_id()
    id_uint_0 = b.new_id()
    id_uint_1 = b.new_id()
    id_uint_2 = b.new_id()
    id_uint_256 = b.new_id()
    id_bool = b.new_id()
    id_uvec3 = b.new_id()
    id_ptr_input_uvec3 = b.new_id()
    id_gl_GlobalInvocationID = b.new_id()
    id_gl_LocalInvocationID = b.new_id()
    id_gl_WorkGroupID = b.new_id()
    id_ptr_input_uint = b.new_id()
    id_runtime_array_uint = b.new_id()
    id_struct_input = b.new_id()
    id_ptr_storage_input = b.new_id()
    id_inputBuf = b.new_id()
    id_struct_output = b.new_id()
    id_ptr_storage_output = b.new_id()
    id_outputBuf = b.new_id()
    id_uint4 = b.new_id()
    id_struct_params = b.new_id()
    id_ptr_uniform_params = b.new_id()
    id_params = b.new_id()
    id_ptr_uniform_uint = b.new_id()
    id_array_shared_256 = b.new_id()
    id_ptr_workgroup_array = b.new_id()
    id_sharedData = b.new_id()
    id_ptr_workgroup_uint = b.new_id()
    id_prevWorkgroupsSum = b.new_id()
    id_main = b.new_id()
    id_label_entry = b.new_id()
    id_ptr_storage_uint = b.new_id()

    # Many more IDs for the actual algorithm
    ids_needed = 150  # Allocate enough for all operations
    extra_ids = [b.new_id() for _ in range(ids_needed)]

    # Capabilities
    b.add_instruction(17, 1)  # OpCapability Shader

    # Memory model
    b.add_instruction(14, 0, 1)  # OpMemoryModel Logical GLSL450

    # Entry point
    b.add_instruction(15, 5, id_main)  # OpEntryPoint GLCompute %main
    b.add_string("main")
    b.add_words(id_gl_GlobalInvocationID, id_gl_LocalInvocationID, id_gl_WorkGroupID)

    # Execution mode
    b.add_instruction(16, id_main, 17, 256, 1, 1)  # OpExecutionMode %main LocalSize 256 1 1

    # Decorations
    b.add_instruction(71, id_gl_GlobalInvocationID, 11, 28)  # BuiltIn GlobalInvocationId
    b.add_instruction(71, id_gl_LocalInvocationID, 11, 27)  # BuiltIn LocalInvocationId
    b.add_instruction(71, id_gl_WorkGroupID, 11, 26)  # BuiltIn WorkgroupId

    b.add_instruction(72, id_runtime_array_uint, 6, 4)  # ArrayStride 4
    b.add_instruction(72, id_struct_input, 3)  # Block
    b.add_instruction(72, id_struct_output, 3)  # Block
    b.add_instruction(72, id_struct_params, 3)  # Block

    b.add_instruction(72, id_struct_input, 0, 0, 6, 0)  # MemberDecorate offset 0
    b.add_instruction(72, id_struct_output, 0, 0, 6, 0)  # MemberDecorate offset 0
    b.add_instruction(72, id_struct_params, 0, 0, 6, 0)  # MemberDecorate offset 0

    b.add_instruction(71, id_inputBuf, 33, 0)  # Binding 0
    b.add_instruction(71, id_inputBuf, 34, 0)  # DescriptorSet 0
    b.add_instruction(71, id_outputBuf, 33, 1)  # Binding 1
    b.add_instruction(71, id_outputBuf, 34, 0)  # DescriptorSet 0
    b.add_instruction(71, id_params, 33, 2)  # Binding 2
    b.add_instruction(71, id_params, 34, 0)  # DescriptorSet 0

    # For readonly buffer
    b.add_instruction(71, id_inputBuf, 26)  # NonWritable decoration

    # Type declarations
    b.add_instruction(19, id_void)  # OpTypeVoid
    b.add_instruction(33, id_func_type, id_void)  # OpTypeFunction %void
    b.add_instruction(21, id_uint, 32, 0)  # OpTypeInt 32 0
    b.add_instruction(43, id_uint, id_uint_0, 0, 0)  # OpConstant %uint 0
    b.add_instruction(43, id_uint, id_uint_1, 1, 0)  # OpConstant %uint 1
    b.add_instruction(43, id_uint, id_uint_2, 2, 0)  # OpConstant %uint 2
    b.add_instruction(43, id_uint, id_uint_256, 0, 1)  # OpConstant %uint 256
    b.add_instruction(20, id_bool, 1)  # OpTypeBool
    b.add_instruction(23, id_uvec3, id_uint, 3)  # OpTypeVector %uint 3
    b.add_instruction(32, id_ptr_input_uvec3, 1, id_uvec3)  # OpTypePointer Input %uvec3
    b.add_instruction(32, id_ptr_input_uint, 1, id_uint)  # OpTypePointer Input %uint
    b.add_instruction(28, id_runtime_array_uint, id_uint)  # OpTypeRuntimeArray %uint
    b.add_instruction(30, id_struct_input, id_runtime_array_uint)  # OpTypeStruct
    b.add_instruction(32, id_ptr_storage_input, 12, id_struct_input)  # OpTypePointer StorageBuffer
    b.add_instruction(30, id_struct_output, id_runtime_array_uint)  # OpTypeStruct
    b.add_instruction(32, id_ptr_storage_output, 12, id_struct_output)  # OpTypePointer StorageBuffer
    b.add_instruction(23, id_uint4, id_uint, 4)  # OpTypeVector %uint 4
    b.add_instruction(30, id_struct_params, id_uint4)  # OpTypeStruct
    b.add_instruction(32, id_ptr_uniform_params, 2, id_struct_params)  # OpTypePointer Uniform
    b.add_instruction(32, id_ptr_uniform_uint, 2, id_uint)  # OpTypePointer Uniform %uint
    b.add_instruction(28, id_array_shared_256, id_uint, id_uint_256)  # OpTypeArray %uint 256
    b.add_instruction(32, id_ptr_workgroup_array, 4, id_array_shared_256)  # OpTypePointer Workgroup
    b.add_instruction(32, id_ptr_workgroup_uint, 4, id_uint)  # OpTypePointer Workgroup %uint
    b.add_instruction(32, id_ptr_storage_uint, 12, id_uint)  # OpTypePointer StorageBuffer %uint

    # Variables
    b.add_instruction(59, id_ptr_input_uvec3, id_gl_GlobalInvocationID, 1)  # OpVariable Input
    b.add_instruction(59, id_ptr_input_uvec3, id_gl_LocalInvocationID, 1)  # OpVariable Input
    b.add_instruction(59, id_ptr_input_uvec3, id_gl_WorkGroupID, 1)  # OpVariable Input
    b.add_instruction(59, id_ptr_storage_input, id_inputBuf, 12)  # OpVariable StorageBuffer
    b.add_instruction(59, id_ptr_storage_output, id_outputBuf, 12)  # OpVariable StorageBuffer
    b.add_instruction(59, id_ptr_uniform_params, id_params, 2)  # OpVariable Uniform
    b.add_instruction(59, id_ptr_workgroup_array, id_sharedData, 4)  # OpVariable Workgroup
    b.add_instruction(59, id_ptr_workgroup_uint, id_prevWorkgroupsSum, 4)  # OpVariable Workgroup

    # Function
    b.add_instruction(54, id_void, id_main, 0, id_func_type)  # OpFunction
    b.add_instruction(248, id_label_entry)  # OpLabel

    # Load global/local/workgroup IDs
    id_gid_vec = extra_ids[0]
    id_gid = extra_ids[1]
    id_lid_vec = extra_ids[2]
    id_lid = extra_ids[3]
    id_wgid_vec = extra_ids[4]
    id_wgid = extra_ids[5]

    b.add_instruction(61, id_uvec3, id_gid_vec, id_gl_GlobalInvocationID)  # OpLoad
    b.add_instruction(79, id_uint, id_gid, id_gid_vec, 0)  # OpCompositeExtract
    b.add_instruction(61, id_uvec3, id_lid_vec, id_gl_LocalInvocationID)  # OpLoad
    b.add_instruction(79, id_uint, id_lid, id_lid_vec, 0)  # OpCompositeExtract
    b.add_instruction(61, id_uvec3, id_wgid_vec, id_gl_WorkGroupID)  # OpLoad
    b.add_instruction(79, id_uint, id_wgid, id_wgid_vec, 0)  # OpCompositeExtract

    # Load N
    id_n_ptr = extra_ids[6]
    id_n = extra_ids[7]
    b.add_instruction(65, id_ptr_uniform_uint, id_n_ptr, id_params, id_uint_0, id_uint_0)  # AccessChain
    b.add_instruction(61, id_uint, id_n, id_n_ptr)  # OpLoad

    # Check if gid < N
    id_cond1 = extra_ids[8]
    b.add_instruction(177, id_bool, id_cond1, id_gid, id_n)  # OpULessThan

    id_label_load = extra_ids[9]
    id_label_load_merge = extra_ids[10]
    b.add_instruction(247, id_label_load, id_label_load_merge)  # OpSelectionMerge
    b.add_instruction(249, id_cond1, id_label_load, id_label_load_merge)  # OpBranchConditional

    b.add_instruction(248, id_label_load)  # OpLabel
    # Load inputBuf.data[gid]
    id_input_ptr = extra_ids[11]
    id_input_val = extra_ids[12]
    b.add_instruction(65, id_ptr_storage_uint, id_input_ptr, id_inputBuf, id_uint_0, id_gid)
    b.add_instruction(61, id_uint, id_input_val, id_input_ptr)
    # Store to sharedData[lid]
    id_shared_ptr = extra_ids[13]
    b.add_instruction(65, id_ptr_workgroup_uint, id_shared_ptr, id_sharedData, id_lid)
    b.add_instruction(62, id_shared_ptr, id_input_val)  # OpStore
    b.add_instruction(253, id_label_load_merge)  # OpBranch

    b.add_instruction(248, id_label_load_merge)  # OpLabel

    # Check if lid == 0
    id_cond_lid0 = extra_ids[14]
    b.add_instruction(166, id_bool, id_cond_lid0, id_lid, id_uint_0)  # OpIEqual

    id_label_init_prev = extra_ids[15]
    id_label_init_merge = extra_ids[16]
    b.add_instruction(247, id_label_init_prev, id_label_init_merge)
    b.add_instruction(249, id_cond_lid0, id_label_init_prev, id_label_init_merge)

    b.add_instruction(248, id_label_init_prev)  # OpLabel
    # prevWorkgroupsSum = 0
    b.add_instruction(62, id_prevWorkgroupsSum, id_uint_0)
    # prevEnd = wgid * 256
    id_prev_end = extra_ids[17]
    b.add_instruction(133, id_uint, id_prev_end, id_wgid, id_uint_256)  # OpIMul

    # Loop to sum all previous workgroups
    id_loop_i_var = extra_ids[18]
    id_loop_header = extra_ids[19]
    id_loop_body = extra_ids[20]
    id_loop_continue = extra_ids[21]
    id_loop_merge = extra_ids[22]

    # i = 0
    id_ptr_func_uint = extra_ids[23]
    b.add_instruction(32, id_ptr_func_uint, 7, id_uint)  # OpTypePointer Function
    b.add_instruction(59, id_ptr_func_uint, id_loop_i_var, 7)  # OpVariable Function
    b.add_instruction(62, id_loop_i_var, id_uint_0)

    b.add_instruction(253, id_loop_header)  # Branch to loop header
    b.add_instruction(248, id_loop_header)  # OpLabel
    b.add_instruction(246, id_loop_merge, id_loop_continue)  # OpLoopMerge
    b.add_instruction(253, id_loop_body)

    b.add_instruction(248, id_loop_body)  # OpLabel
    id_i_val = extra_ids[24]
    b.add_instruction(61, id_uint, id_i_val, id_loop_i_var)
    # Check i < prevEnd && i < N
    id_cond_lt_prev = extra_ids[25]
    id_cond_lt_n = extra_ids[26]
    id_cond_and = extra_ids[27]
    b.add_instruction(177, id_bool, id_cond_lt_prev, id_i_val, id_prev_end)
    b.add_instruction(177, id_bool, id_cond_lt_n, id_i_val, id_n)
    b.add_instruction(166, id_bool, id_cond_and, id_cond_lt_prev, id_cond_lt_n)  # OpLogicalAnd
    b.add_instruction(249, id_cond_and, id_loop_continue, id_loop_merge)

    b.add_instruction(248, id_loop_continue)  # OpLabel
    # Load inputBuf.data[i]
    id_loop_input_ptr = extra_ids[28]
    id_loop_input_val = extra_ids[29]
    b.add_instruction(65, id_ptr_storage_uint, id_loop_input_ptr, id_inputBuf, id_uint_0, id_i_val)
    b.add_instruction(61, id_uint, id_loop_input_val, id_loop_input_ptr)
    # prevWorkgroupsSum += val
    id_prev_sum_old = extra_ids[30]
    id_prev_sum_new = extra_ids[31]
    b.add_instruction(61, id_uint, id_prev_sum_old, id_prevWorkgroupsSum)
    b.add_instruction(128, id_uint, id_prev_sum_new, id_prev_sum_old, id_loop_input_val)  # OpIAdd
    b.add_instruction(62, id_prevWorkgroupsSum, id_prev_sum_new)
    # i++
    id_i_inc = extra_ids[32]
    b.add_instruction(128, id_uint, id_i_inc, id_i_val, id_uint_1)
    b.add_instruction(62, id_loop_i_var, id_i_inc)
    b.add_instruction(253, id_loop_header)

    b.add_instruction(248, id_loop_merge)  # OpLabel
    b.add_instruction(253, id_label_init_merge)

    b.add_instruction(248, id_label_init_merge)  # OpLabel

    # Barrier
    b.add_instruction(224, id_uint_2, id_uint_256)  # OpControlBarrier Workgroup Workgroup AcquireRelease

    # Parallel scan loop: for stride in [1, 2, 4, ..., 128]
    # stride = 1
    id_stride_var = extra_ids[33]
    b.add_instruction(59, id_ptr_func_uint, id_stride_var, 7)
    b.add_instruction(62, id_stride_var, id_uint_1)

    id_scan_loop_header = extra_ids[34]
    id_scan_loop_body = extra_ids[35]
    id_scan_loop_continue = extra_ids[36]
    id_scan_loop_merge = extra_ids[37]

    b.add_instruction(253, id_scan_loop_header)
    b.add_instruction(248, id_scan_loop_header)
    b.add_instruction(246, id_scan_loop_merge, id_scan_loop_continue)
    b.add_instruction(253, id_scan_loop_body)

    b.add_instruction(248, id_scan_loop_body)
    id_stride_val = extra_ids[38]
    b.add_instruction(61, id_uint, id_stride_val, id_stride_var)
    # Check stride < 256
    id_cond_stride = extra_ids[39]
    b.add_instruction(177, id_bool, id_cond_stride, id_stride_val, id_uint_256)

    id_scan_body_inner = extra_ids[40]
    b.add_instruction(249, id_cond_stride, id_scan_body_inner, id_scan_loop_merge)

    b.add_instruction(248, id_scan_body_inner)
    # temp = 0
    id_temp_var = extra_ids[41]
    b.add_instruction(59, id_ptr_func_uint, id_temp_var, 7)
    b.add_instruction(62, id_temp_var, id_uint_0)

    # if lid >= stride
    id_cond_gte = extra_ids[42]
    b.add_instruction(175, id_bool, id_cond_gte, id_lid, id_stride_val)  # OpUGreaterThanEqual

    id_label_read_temp = extra_ids[43]
    id_label_read_merge = extra_ids[44]
    b.add_instruction(247, id_label_read_temp, id_label_read_merge)
    b.add_instruction(249, id_cond_gte, id_label_read_temp, id_label_read_merge)

    b.add_instruction(248, id_label_read_temp)
    # temp = sharedData[lid - stride]
    id_lid_minus_stride = extra_ids[45]
    b.add_instruction(130, id_uint, id_lid_minus_stride, id_lid, id_stride_val)  # OpISub
    id_temp_ptr = extra_ids[46]
    id_temp_val = extra_ids[47]
    b.add_instruction(65, id_ptr_workgroup_uint, id_temp_ptr, id_sharedData, id_lid_minus_stride)
    b.add_instruction(61, id_uint, id_temp_val, id_temp_ptr)
    b.add_instruction(62, id_temp_var, id_temp_val)
    b.add_instruction(253, id_label_read_merge)

    b.add_instruction(248, id_label_read_merge)
    # Barrier
    b.add_instruction(224, id_uint_2, id_uint_256)

    # if lid >= stride, sharedData[lid] += temp
    id_label_write_add = extra_ids[48]
    id_label_write_merge = extra_ids[49]
    b.add_instruction(247, id_label_write_add, id_label_write_merge)
    b.add_instruction(249, id_cond_gte, id_label_write_add, id_label_write_merge)

    b.add_instruction(248, id_label_write_add)
    id_shared_lid_ptr = extra_ids[50]
    id_shared_old = extra_ids[51]
    id_temp_loaded = extra_ids[52]
    id_shared_new = extra_ids[53]
    b.add_instruction(65, id_ptr_workgroup_uint, id_shared_lid_ptr, id_sharedData, id_lid)
    b.add_instruction(61, id_uint, id_shared_old, id_shared_lid_ptr)
    b.add_instruction(61, id_uint, id_temp_loaded, id_temp_var)
    b.add_instruction(128, id_uint, id_shared_new, id_shared_old, id_temp_loaded)
    b.add_instruction(62, id_shared_lid_ptr, id_shared_new)
    b.add_instruction(253, id_label_write_merge)

    b.add_instruction(248, id_label_write_merge)
    # Barrier
    b.add_instruction(224, id_uint_2, id_uint_256)

    b.add_instruction(253, id_scan_loop_continue)

    b.add_instruction(248, id_scan_loop_continue)
    # stride *= 2
    id_stride_new = extra_ids[54]
    b.add_instruction(133, id_uint, id_stride_new, id_stride_val, id_uint_2)
    b.add_instruction(62, id_stride_var, id_stride_new)
    b.add_instruction(253, id_scan_loop_header)

    b.add_instruction(248, id_scan_loop_merge)

    # Write result: if gid < N
    id_cond_write = extra_ids[55]
    b.add_instruction(177, id_bool, id_cond_write, id_gid, id_n)

    id_label_write = extra_ids[56]
    id_label_write_merge2 = extra_ids[57]
    b.add_instruction(247, id_label_write, id_label_write_merge2)
    b.add_instruction(249, id_cond_write, id_label_write, id_label_write_merge2)

    b.add_instruction(248, id_label_write)
    id_shared_final_ptr = extra_ids[58]
    id_shared_final = extra_ids[59]
    id_prev_sum_final = extra_ids[60]
    id_output_val = extra_ids[61]
    id_output_ptr = extra_ids[62]
    b.add_instruction(65, id_ptr_workgroup_uint, id_shared_final_ptr, id_sharedData, id_lid)
    b.add_instruction(61, id_uint, id_shared_final, id_shared_final_ptr)
    b.add_instruction(61, id_uint, id_prev_sum_final, id_prevWorkgroupsSum)
    b.add_instruction(128, id_uint, id_output_val, id_shared_final, id_prev_sum_final)
    b.add_instruction(65, id_ptr_storage_uint, id_output_ptr, id_outputBuf, id_uint_0, id_gid)
    b.add_instruction(62, id_output_ptr, id_output_val)
    b.add_instruction(253, id_label_write_merge2)

    b.add_instruction(248, id_label_write_merge2)

    # Return
    b.add_instruction(253, 0)  # OpReturn
    b.add_instruction(56)  # OpFunctionEnd

    # Update bound
    b.words[bound_placeholder_idx] = b.id_counter

    return b.to_hex()

if __name__ == "__main__":
    hex_string = build_shader()
    with open("answer.txt", "w") as f:
        f.write(hex_string)
    print(f"Generated SPIR-V hex ({len(hex_string)} chars, {len(hex_string)//8} words)")
