public struct Vector3
{
    public double X, Y, Z;
    public Vector3(double x, double y, double z) { X = x; Y = y; Z = z; }
    public static Vector3 operator +(Vector3 a, Vector3 b) => new Vector3(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
    public static Vector3 operator -(Vector3 a, Vector3 b) => new Vector3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
    public static Vector3 operator *(Vector3 a, double s) => new Vector3(a.X * s, a.Y * s, a.Z * s);
    public static Vector3 operator *(double s, Vector3 a) => new Vector3(a.X * s, a.Y * s, a.Z * s);
    public static Vector3 operator /(Vector3 a, double s) => new Vector3(a.X / s, a.Y / s, a.Z / s);
    public double Length() => Math.Sqrt(X * X + Y * Y + Z * Z);
    public double LengthSquared() => X * X + Y * Y + Z * Z;
    public static double Dot(Vector3 a, Vector3 b) => a.X * b.X + a.Y * b.Y + a.Z * b.Z;
    public static Vector3 Cross(Vector3 a, Vector3 b) => new Vector3(
        a.Y * b.Z - a.Z * b.Y,
        a.Z * b.X - a.X * b.Z,
        a.X * b.Y - a.Y * b.X
    );
    public Vector3 Normalize() {
        double len = Length();
        return len > 0 ? this / len : new Vector3(0, 0, 0);
    }
}

public class NBodyIntegrator
{
    public double G = 6.67430e-11;
    public double[] PlanetMasses;
    public Func<int, double, Vector3> GetPlanetPosition;
    
    public void Step(ref Vector3 astPos, ref Vector3 astVel, double astMass,
                     ref Vector3 scPos, ref Vector3 scVel, double scMass, bool scActive,
                     double timeSec, double dt, int stepIdx)
    {
        double half_dt = dt * 0.5;
        
        Vector3 accAst1, accSc1;
        ComputeAccel(astPos, astMass, scPos, scMass, scActive, timeSec, out accAst1, out accSc1);
        
        if (scActive) scVel += accSc1 * half_dt;
        astVel += accAst1 * half_dt;
        
        if (scActive) scPos += scVel * dt;
        astPos += astVel * dt;
        
        double nextTimeSec = timeSec + dt;
        
        Vector3 accAst2, accSc2;
        ComputeAccel(astPos, astMass, scPos, scMass, scActive, nextTimeSec, out accAst2, out accSc2);
        
        if (scActive) scVel += accSc2 * half_dt;
        astVel += accAst2 * half_dt;
    }
    
    private void ComputeAccel(Vector3 astPos, double astMass, Vector3 scPos, double scMass, bool scActive,
                              double t, out Vector3 accAst, out Vector3 accSc)
    {
        accAst = new Vector3(0, 0, 0);
        accSc = new Vector3(0, 0, 0);
        
        for (int j = 0; j < 10; j++)
        {
            Vector3 planetPos = GetPlanetPosition(j, t);
            double pjMass = PlanetMasses[j];
            
            Vector3 dAst = planetPos - astPos;
            double r2Ast = dAst.LengthSquared();
            if (r2Ast >= 1e6)
            {
                double rAst = Math.Sqrt(r2Ast);
                accAst += (G * pjMass / (rAst * r2Ast)) * dAst;
            }
            
            if (scActive)
            {
                Vector3 dSc = planetPos - scPos;
                double r2Sc = dSc.LengthSquared();
                if (r2Sc >= 1e6)
                {
                    double rSc = Math.Sqrt(r2Sc);
                    accSc += (G * pjMass / (rSc * r2Sc)) * dSc;
                }
            }
        }
        
        if (scActive)
        {
            Vector3 dAstSc = scPos - astPos;
            double r2AstSc = dAstSc.LengthSquared();
            if (r2AstSc >= 1e6)
            {
                double rAstSc = Math.Sqrt(r2AstSc);
                accAst += (G * scMass / (rAstSc * r2AstSc)) * dAstSc;
                accSc += (G * astMass / (rAstSc * r2AstSc)) * (-1.0 * dAstSc);
            }
        }
    }
}

public class Program

    public static double C(double z)
    {
        if (z > 1e-5)
        {
            double sz = Math.Sqrt(z);
            return (1 - Math.Cos(sz)) / z;
        }
        else if (z < -1e-5)
        {
            double sz = Math.Sqrt(-z);
            return (Math.Cosh(sz) - 1) / (-z);
        }
        else
        {
            return 0.5 - z / 24.0 + z * z / 720.0;
        }
    }

    public static double S(double z)
    {
        if (z > 1e-5)
        {
            double sz = Math.Sqrt(z);
            return (sz - Math.Sin(sz)) / (sz * z);
        }
        else if (z < -1e-5)
        {
            double sz = Math.Sqrt(-z);
            return (Math.Sinh(sz) - sz) / (sz * (-z));
        }
        else
        {
            return 1.0 / 6.0 - z / 120.0 + z * z / 5040.0;
        }
    }

    public static bool SolveLambert(Vector3 r1_vec, Vector3 r2_vec, double tof, double mu, out Vector3 v1, out Vector3 v2)
    {
        v1 = new Vector3(0, 0, 0);
        v2 = new Vector3(0, 0, 0);
        
        double r1 = r1_vec.Length();
        double r2 = r2_vec.Length();
        
        double cos_dnu = Vector3.Dot(r1_vec, r2_vec) / (r1 * r2);
        if (cos_dnu > 1.0) cos_dnu = 1.0;
        if (cos_dnu < -1.0) cos_dnu = -1.0;
        
        double dnu = Math.Acos(cos_dnu);
        Vector3 cross = Vector3.Cross(r1_vec, r2_vec);
        if (cross.Z < 0)
        {
            dnu = 2.0 * Math.PI - dnu;
        }
        
        if (dnu < 1e-5 || Math.Abs(dnu - Math.PI) < 1e-5)
        {
            return false;
        }
        
        double A = Math.Sin(dnu) * Math.Sqrt(r1 * r2 / (1.0 - cos_dnu));
        if (A == 0) return false;
        
        double z_low = -39.4;
        double z_high = 1e6;
        double z = 0.0;
        double y = 0.0;
        
        bool found = false;
        for (int iter = 0; iter < 100; iter++)
        {
            z = (z_low + z_high) * 0.5;
            double c_z = C(z);
            double s_z = S(z);
            
            y = r1 + r2 + A * (z * s_z - 1.0) / Math.Sqrt(c_z);
            if (y < 0)
            {
                if (A > 0) z_low = z;
                else z_high = z;
                continue;
            }
            
            double t = (Math.Pow(y / c_z, 1.5) * s_z + A * Math.Sqrt(y)) / Math.Sqrt(mu);
            if (t < tof)
            {
                z_low = z;
            }
            else
            {
                z_high = z;
            }
            
            if (Math.Abs(t - tof) < 1e-5)
            {
                found = true;
                break;
            }
        }
        
        double c_val = C(z);
        double s_val = S(z);
        y = r1 + r2 + A * (z * s_val - 1.0) / Math.Sqrt(c_val);
        if (y < 0) return false;
        
        double f = 1.0 - y / r1;
        double g = A * Math.Sqrt(y / mu);
        double g_dot = 1.0 - y / r2;
        
        v1 = (r2_vec - f * r1_vec) / g;
        v2 = (g_dot * r2_vec - r1_vec) / g;
        return true;
    }

    static Vector3 GetPlanetPosition(int bodyIdx, double timeSec)
    {
        double day = timeSec / 86400.0;
        double pt = day / h_step_days;
        int idx = (int)Math.Floor(pt);
        if (idx < 0) idx = 0;
        if (idx >= nPoints - 1) idx = nPoints - 2;
        
        double s = pt - idx;
        double h = h_step_sec;
        
        Vector3 r0 = ephPos[bodyIdx][idx];
        Vector3 v0 = ephVel[bodyIdx][idx];
        Vector3 r1 = ephPos[bodyIdx][idx + 1];
        Vector3 v1 = ephVel[bodyIdx][idx + 1];
        
        double s2 = s * s;
        double s3 = s2 * s;
        
        double h00 = 2 * s3 - 3 * s2 + 1;
        double h10 = s3 - 2 * s2 + s;
        double h01 = -2 * s3 + 3 * s2;
        double h11 = s3 - s2;
        
        return h00 * r0 + h10 * h * v0 + h01 * r1 + h11 * h * v1;
    }

    static Vector3 GetPlanetVelocity(int bodyIdx, double timeSec)
    {
        double day = timeSec / 86400.0;
        double pt = day / h_step_days;
        int idx = (int)Math.Floor(pt);
        if (idx < 0) idx = 0;
        if (idx >= nPoints - 1) idx = nPoints - 2;
        
        double s = pt - idx;
        double h = h_step_sec;
        
        Vector3 r0 = ephPos[bodyIdx][idx];
        Vector3 v0 = ephVel[bodyIdx][idx];
        Vector3 r1 = ephPos[bodyIdx][idx + 1];
        Vector3 v1 = ephVel[bodyIdx][idx + 1];
        
        double s2 = s * s;
        
        double dh00 = 6 * s2 - 6 * s;
        double dh10 = 3 * s2 - 4 * s + 1;
        double dh01 = -6 * s2 + 6 * s;
        double dh11 = 3 * s2 - 2 * s;
        
        return (dh00 * r0 + dh10 * h * v0 + dh01 * r1 + dh11 * h * v1) / h;
    }

    static void IntegrateAsteroidAlone(Vector3 astPos0, Vector3 astVel0, double astMass, int numDays, out Vector3 astPos, out Vector3 astVel)
    {
        astPos = astPos0;
        astVel = astVel0;
        Vector3 dummy1 = new Vector3(0, 0, 0);
        Vector3 dummy2 = new Vector3(0, 0, 0);
        
        double dt = 3600.0;
        int stepsPerDay = 24;
        int totalSteps = numDays * stepsPerDay;
        
        var integrator = new NBodyIntegrator
        {
            PlanetMasses = masses,
            GetPlanetPosition = GetPlanetPosition
        };
        
        for (int step = 0; step < totalSteps; step++)
        {
            double timeSec = step * dt;
            integrator.Step(ref astPos, ref astVel, astMass,
                            ref dummy1, ref dummy2, 0.0, false,
                            timeSec, dt, step);
        }
    }

    static void SimulateWithImpactBurn(Vector3 astPos0, Vector3 astVel0, double astMass,
                                       Vector3 scPos0, Vector3 scVel0, double scMass,
                                       int interceptDay, Vector3 impact_dV,
                                       out Vector3 astPos, out Vector3 astVel,
                                       out Vector3 scPos, out Vector3 scVel)
    {
        astPos = astPos0;
        astVel = astVel0;
        scPos = scPos0;
        scVel = scVel0;
        
        double dt = 3600.0;
        int stepsPerDay = 24;
        
        var integrator = new NBodyIntegrator
        {
            PlanetMasses = masses,
            GetPlanetPosition = GetPlanetPosition
        };
        
        int step = 0;
        int stepsBeforeBurn = (interceptDay - 1) * stepsPerDay;
        for (; step < stepsBeforeBurn; step++)
        {
            double timeSec = step * dt;
            integrator.Step(ref astPos, ref astVel, astMass,
                            ref scPos, ref scVel, scMass, true,
                            timeSec, dt, step);
        }
        
        scVel += impact_dV;
        
        int totalSteps = interceptDay * stepsPerDay;
        for (; step < totalSteps; step++)
        {
            double timeSec = step * dt;
            integrator.Step(ref astPos, ref astVel, astMass,
                            ref scPos, ref scVel, scMass, true,
                            timeSec, dt, step);
        }
    }

    static bool SolveInterception(Vector3 astPos0, Vector3 astVel0, double astMass,
                                  Vector3 earthPos0, Vector3 earthVel0, double scMass,
                                  int interceptDay, Vector3 initialGuess_dV, double v_leftover,
                                  out Vector3 final_dV, out Vector3 finalScVel, out Vector3 finalAstVel, out Vector3 finalImpact_dV)
    {
        Vector3 dv = initialGuess_dV;
        double epsilon = 1e-3;
        
        final_dV = dv;
        finalScVel = new Vector3(0, 0, 0);
        finalAstVel = new Vector3(0, 0, 0);
        finalImpact_dV = new Vector3(0, 0, 0);
        
        for (int iter = 0; iter < 12; iter++)
        {
            Vector3 astPosPre, astVelPre, scPosPre, scVelPre;
            SimulateWithImpactBurn(astPos0, astVel0, astMass, earthPos0, earthVel0 + dv, scMass, interceptDay, new Vector3(0, 0, 0),
                                   out astPosPre, out astVelPre, out scPosPre, out scVelPre);
            
            Vector3 rel_v_dir = (scVelPre - astVelPre).Normalize();
            Vector3 impact_dV = v_leftover * rel_v_dir;
            
            Vector3 astPos, astVel, scPos, scVel;
            SimulateWithImpactBurn(astPos0, astVel0, astMass, earthPos0, earthVel0 + dv, scMass, interceptDay, impact_dV,
                                   out astPos, out astVel, out scPos, out scVel);
            
            Vector3 err = scPos - astPos;
            if (err.Length() < 1e-1)
            {
                final_dV = dv;
                finalScVel = scVel;
                finalAstVel = astVel;
                finalImpact_dV = impact_dV;
                return true;
            }
            
            Vector3 scPos_x, scPos_y, scPos_z;
            Vector3 dummy1, dummy2, dummy3;
            
            SimulateWithImpactBurn(astPos0, astVel0, astMass, earthPos0, earthVel0 + dv + new Vector3(epsilon, 0, 0), scMass, interceptDay, impact_dV,
                                   out dummy1, out dummy2, out scPos_x, out dummy3);
            SimulateWithImpactBurn(astPos0, astVel0, astMass, earthPos0, earthVel0 + dv + new Vector3(0, epsilon, 0), scMass, interceptDay, impact_dV,
                                   out dummy1, out dummy2, out scPos_y, out dummy3);
            SimulateWithImpactBurn(astPos0, astVel0, astMass, earthPos0, earthVel0 + dv + new Vector3(0, 0, epsilon), scMass, interceptDay, impact_dV,
                                   out dummy1, out dummy2, out scPos_z, out dummy3);
            
            double J00 = (scPos_x.X - scPos.X) / epsilon;
            double J10 = (scPos_x.Y - scPos.Y) / epsilon;
            double J20 = (scPos_x.Z - scPos.Z) / epsilon;
            
            double J01 = (scPos_y.X - scPos.X) / epsilon;
            double J11 = (scPos_y.Y - scPos.Y) / epsilon;
            double J21 = (scPos_y.Z - scPos.Z) / epsilon;
            
            double J02 = (scPos_z.X - scPos.X) / epsilon;
            double J12 = (scPos_z.Y - scPos.Y) / epsilon;
            double J22 = (scPos_z.Z - scPos.Z) / epsilon;
            
            double det = J00 * (J11 * J22 - J12 * J21) - J01 * (J10 * J22 - J12 * J20) + J02 * (J10 * J21 - J11 * J20);
            if (Math.Abs(det) < 1e-15)
            {
                break;
            }
            
            double invDet = 1.0 / det;
            double b0 = -err.X;
            double b1 = -err.Y;
            double b2 = -err.Z;
            
            double delta_dv_x = invDet * (b0 * (J11 * J22 - J12 * J21) - J01 * (b1 * J22 - J12 * b2) + J02 * (b1 * J21 - J11 * b2));
            double delta_dv_y = invDet * (J00 * (b1 * J22 - J12 * b2) - b0 * (J10 * J22 - J12 * J20) + J02 * (J10 * b2 - b1 * J20));
            double delta_dv_z = invDet * (J00 * (J11 * b2 - b1 * J21) - J01 * (J10 * b2 - b1 * J20) + b0 * (J10 * J21 - J11 * J20));
            
            dv.X += delta_dv_x;
            dv.Y += delta_dv_y;
            dv.Z += delta_dv_z;
        }
        
        return false;
    }

        for (int I = searchStart; I <= searchEnd; I += coarseStep)
        {
            Vector3 astPosI = asteroidPos[I];
            double TOF = I * 86400.0;
            
            Vector3 v1_Lambert, v2_Lambert;
            bool lambertOk = SolveLambert(earthPos0, astPosI, TOF, mu, out v1_Lambert, out v2_Lambert);
            
            var guesses = new List<Vector3>();
            if (lambertOk) guesses.Add(v1_Lambert - earthVel0);
            guesses.Add(new Vector3(0, 0, 0));
            
            foreach (var guess in guesses)
            {
                if (guess.Length() > deltaVBudget * 1.5) continue;
                
                double v_leftover = Math.Max(0.0, deltaVBudget - guess.Length());
                Vector3 dv_launch, scVel_I, astVel_I, impact_dV;
                bool success = SolveInterception(astPos0, astVel0, astMass, earthPos0, earthVel0, impactorMass, I, guess, v_leftover,
                                                 out dv_launch, out scVel_I, out astVel_I, out impact_dV);
                
                if (success)
                {
                    double dv_mag = dv_launch.Length();
                    if (dv_mag <= deltaVBudget)
                    {
                        double v_left = deltaVBudget - dv_mag;
                        Vector3 rel_v = (scVel_I - astVel_I);
                        double rel_v_mag = rel_v.Length();
                        double dv_ast = (impactorMass / asteroidMass) * rel_v_mag;
                        double timeRemaining = (warningDays - I) * 86400.0;
                        double deflectionScore = dv_ast * timeRemaining;
                        
                        if (deflectionScore > maxDeflection)
                        {
                            maxDeflection = deflectionScore;
                            bestInterceptDay = I;
                            best_dV_launch = dv_launch;
                            bestImpact_dV = impact_dV;
                        }
                    }
                }
            }
        }
        
        if (bestInterceptDay != -1)
        {
            int fineStart = Math.Max(searchStart, bestInterceptDay - coarseStep);
            int fineEnd = Math.Min(searchEnd, bestInterceptDay + coarseStep);
            
            for (int I = fineStart; I <= fineEnd; I++)
            {
                Vector3 astPosI = asteroidPos[I];
                double TOF = I * 86400.0;
                
                Vector3 v1_Lambert, v2_Lambert;
                bool lambertOk = SolveLambert(earthPos0, astPosI, TOF, mu, out v1_Lambert, out v2_Lambert);
                
                var guesses = new List<Vector3>();
                if (lambertOk) guesses.Add(v1_Lambert - earthVel0);
                guesses.Add(new Vector3(0, 0, 0));
                
                foreach (var guess in guesses)
                {
                    if (guess.Length() > deltaVBudget * 1.5) continue;
                    
                    double v_leftover = Math.Max(0.0, deltaVBudget - guess.Length());
                    Vector3 dv_launch, scVel_I, astVel_I, impact_dV;
                    bool success = SolveInterception(astPos0, astVel0, astMass, earthPos0, earthVel0, impactorMass, I, guess, v_leftover,
                                                     out dv_launch, out scVel_I, out astVel_I, out impact_dV);
                    
                    if (success)
                    {
                        double dv_mag = dv_launch.Length();
                        if (dv_mag <= deltaVBudget)
                        {
                            double v_left = deltaVBudget - dv_mag;
                            Vector3 rel_v = (scVel_I - astVel_I);
                            double rel_v_mag = rel_v.Length();
                            double dv_ast = (impactorMass / asteroidMass) * rel_v_mag;
                            double timeRemaining = (warningDays - I) * 86400.0;
                            double deflectionScore = dv_ast * timeRemaining;
                            
                            if (deflectionScore > maxDeflection)
                            {
                                maxDeflection = deflectionScore;
                                bestInterceptDay = I;
                                best_dV_launch = dv_launch;
                                bestImpact_dV = impact_dV;
                            }
                        }
                    }
                }
            }
        }
        
        if (bestInterceptDay != -1)
        {
            Console.WriteLine("2");
            Console.WriteLine($"0.0 {best_dV_launch.X:e12} {best_dV_launch.Y:e12} {best_dV_launch.Z:e12}");
            Console.WriteLine($"{(double)(bestInterceptDay - 1):F1} {bestImpact_dV.X:e12} {bestImpact_dV.Y:e12} {bestImpact_dV.Z:e12}");
        }
        else
        {
            Console.WriteLine("1");
            Console.WriteLine($"0.0 {deltaVBudget*0.9:e12} 0.0 0.0");
        }
    }
