using namespace std;

using Weight = long long;
const Weight INF = 1e16;

struct AdjEdge {
    int to;
    Weight weight;
};

struct EdgeInput {
    int u, v;
    Weight w;
    bool operator<(const EdgeInput& other) const {
        if (u != other.u) return u < other.u;
        if (v != other.v) return v < other.v;
        return w < other.w;
    }
};

struct DSU {
    vector<int> parent;
    DSU(int n) {
        parent.resize(n);
        iota(parent.begin(), parent.end(), 0);
    }
    int find(int i) {
        if (parent[i] == i)
            return i;
        return parent[i] = find(parent[i]);
    }
    bool unite(int i, int j) {
        int root_i = find(i);
        int root_j = find(j);
        if (root_i != root_j) {
            parent[root_i] = root_j;
            return true;
        }
        return false;
    }
};

struct Edge {
    int u, v;
    Weight w;
    bool operator<(const Edge& other) const {
        return w < other.w;
    }
};

int V, E, K;
vector<vector<AdjEdge>> adj;
vector<int> term;
vector<bool> is_terminal;

Weight get_edge_weight(int u, int v) {
    auto it = lower_bound(adj[u].begin(), adj[u].end(), AdjEdge{v, 0}, 
                          [](const AdjEdge& a, const AdjEdge& b) {
                              return a.to < b.to;
                          });
    if (it != adj[u].end() && it->to == v) {
        return it->weight;
    }
    return INF;
}

Weight get_edges_weight(const vector<pair<int, int>>& tree_edges) {
    Weight total = 0;
    for (auto& p : tree_edges) {
        total += get_edge_weight(p.first, p.second);
    }
    return total;
}

vector<pair<int, int>> optimize_candidate_set(const vector<int>& vertices) {
    vector<bool> in_set(V, false);
    for (int u : vertices) in_set[u] = true;
    
    vector<Edge> sub_edges;
    for (int u : vertices) {
        for (auto& edge : adj[u]) {
            if (edge.to > u && in_set[edge.to]) {
                sub_edges.push_back({u, edge.to, edge.weight});
            }
        }
    }
    
    sort(sub_edges.begin(), sub_edges.end());
    DSU dsu(V);
    vector<pair<int, int>> mst_edges;
    for (auto& edge : sub_edges) {
        if (dsu.unite(edge.u, edge.v)) {
            mst_edges.push_back({edge.u, edge.v});
        }
    }
    
    vector<vector<int>> mst_adj(V);
    vector<int> degree(V, 0);
    for (auto& p : mst_edges) {
        mst_adj[p.first].push_back(p.second);
        mst_adj[p.second].push_back(p.first);
        degree[p.first]++;
        degree[p.second]++;
    }
    
    queue<int> q;
    for (int u : vertices) {
        if (degree[u] == 1 && !is_terminal[u]) {
            q.push(u);
        }
    }
    
    vector<bool> removed(V, false);
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        if (degree[u] == 1 && !is_terminal[u]) {
            removed[u] = true;
            degree[u] = 0;
            for (int neighbor : mst_adj[u]) {
                if (degree[neighbor] > 0) {
                    degree[neighbor]--;
                    if (degree[neighbor] == 1 && !is_terminal[neighbor]) {
                        q.push(neighbor);
                    }
                }
            }
        }
    }
    
    vector<pair<int, int>> final_edges;
    for (auto& p : mst_edges) {
        if (!removed[p.first] && !removed[p.second]) {
            final_edges.push_back(p);
        }
    }
    return final_edges;
}

vector<pair<int, int>> local_search(const vector<pair<int, int>>& initial_edges) {
    vector<bool> in_tree(V, false);
    for (auto& p : initial_edges) {
        in_tree[p.first] = true;
        in_tree[p.second] = true;
    }
    
    vector<int> tree_vertices;
    for (int v = 0; v < V; ++v) {
        if (in_tree[v]) tree_vertices.push_back(v);
    }
    
    auto current_edges = optimize_candidate_set(tree_vertices);
    Weight best_weight = get_edges_weight(current_edges);
    
    bool improved = true;
    int iterations = 0;
    while (improved && iterations < 5) {
        improved = false;
        iterations++;
        
        fill(in_tree.begin(), in_tree.end(), false);
        for (auto& p : current_edges) {
            in_tree[p.first] = true;
            in_tree[p.second] = true;
        }
        
        vector<int> candidates;
        for (int u = 0; u < V; ++u) {
            if (is_terminal[u] || in_tree[u]) continue;
            int count = 0;
            for (auto& edge : adj[u]) {
                if (in_tree[edge.to]) {
                    count++;
                    if (count >= 2) {
                        candidates.push_back(u);
                        break;
                    }
                }
            }
        }
        
        for (int c : candidates) {
            vector<int> test_vertices;
            for (int v = 0; v < V; ++v) {
                if (in_tree[v]) test_vertices.push_back(v);
            }
            test_vertices.push_back(c);
            
            auto test_edges = optimize_candidate_set(test_vertices);
            Weight test_weight = get_edges_weight(test_edges);
            if (test_weight < best_weight) {
                current_edges = test_edges;
                best_weight = test_weight;
                improved = true;
                break;
            }
        }
    }
    return current_edges;
}

// Dreyfus-Wagner Exact DP Solver
void reconstruct_dw(int v, int S, const vector<int>& parent_vertex, const vector<int>& split_mask, vector<pair<int, int>>& edges) {
    if (S == 0) return;
    if (std::popcount(static_cast<unsigned int>(S)) == 1) {
        int idx = std::countr_zero(static_cast<unsigned int>(S));
        int p = parent_vertex[S * V + v];
        if (p != -1) {
            edges.push_back({v, p});
            reconstruct_dw(p, S, parent_vertex, split_mask, edges);
        }
        return;
    }
    
    int sp = split_mask[S * V + v];
    if (sp != -1) {
        int S1 = sp;
        int S2 = S ^ S1;
        reconstruct_dw(v, S1, parent_vertex, split_mask, edges);
        reconstruct_dw(v, S2, parent_vertex, split_mask, edges);
    } else {
        int p = parent_vertex[S * V + v];
        if (p != -1) {
            edges.push_back({v, p});
            reconstruct_dw(p, S, parent_vertex, split_mask, edges);
        }
    }
}

vector<pair<int, int>> run_dreyfus_wagner() {
    int num_masks = 1 << K;
    vector<Weight> DP(num_masks * V, INF);
    vector<int> split_mask(num_masks * V, -1);
    vector<int> parent_vertex(num_masks * V, -1);
    
    for (int i = 0; i < K; ++i) {
        int mask = 1 << i;
        int s = term[i];
        
        vector<Weight> dist(V, INF);
        vector<int> pred(V, -1);
        priority_queue<pair<Weight, int>, vector<pair<Weight, int>>, greater<>> pq;
        
        dist[s] = 0;
        pq.push({0, s});
        while (!pq.empty()) {
            auto [d, u] = pq.top();
            pq.pop();
            if (d > dist[u]) continue;
            for (auto& edge : adj[u]) {
                int v = edge.to;
                Weight w = edge.weight;
                if (dist[v] > d + w) {
                    dist[v] = d + w;
                    pred[v] = u;
                    pq.push({dist[v], v});
                }
            }
        }
        
        for (int v = 0; v < V; ++v) {
            int index = mask * V + v;
            DP[index] = dist[v];
            parent_vertex[index] = pred[v];
            split_mask[index] = -1;
        }
    }
    
    vector<vector<int>> masks_by_popcount(K + 1);
    for (int mask = 1; mask < num_masks; ++mask) {
        int pop = std::popcount(static_cast<unsigned int>(mask));
        masks_by_popcount[pop].push_back(mask);
    }
    
    for (int pop = 2; pop <= K; ++pop) {
        for (int S : masks_by_popcount[pop]) {
            int first_bit = S & -S;
            for (int S1 = (S - 1) & S; S1 > 0; S1 = (S1 - 1) & S) {
                if (S1 & first_bit) {
                    int S2 = S ^ S1;
                    int idx_S = S * V;
                    int idx_S1 = S1 * V;
                    int idx_S2 = S2 * V;
                    
                    for (int v = 0; v < V; ++v) {
                        Weight val = DP[idx_S1 + v] + DP[idx_S2 + v];
                        if (val < DP[idx_S + v]) {
                            DP[idx_S + v] = val;
                            split_mask[idx_S + v] = S1;
                            parent_vertex[idx_S + v] = -1;
                        }
                    }
                }
            }
            
            priority_queue<pair<Weight, int>, vector<pair<Weight, int>>, greater<>> pq;
            int idx_S = S * V;
            for (int v = 0; v < V; ++v) {
                if (DP[idx_S + v] < INF) {
                    pq.push({DP[idx_S + v], v});
                }
            }
            
            while (!pq.empty()) {
                auto [d, u] = pq.top();
                pq.pop();
                if (d > DP[idx_S + u]) continue;
                for (auto& edge : adj[u]) {
                    int v = edge.to;
                    Weight w = edge.weight;
                    if (DP[idx_S + v] > d + w) {
                        DP[idx_S + v] = d + w;
                        parent_vertex[idx_S + v] = u;
                        split_mask[idx_S + v] = -1;
                        pq.push({DP[idx_S + v], v});
                    }
                }
            }
        }
    }
    
    int root = term[0];
    vector<pair<int, int>> edges;
    reconstruct_dw(root, num_masks - 1, parent_vertex, split_mask, edges);
    return edges;
}

// Shortest Path Heuristic (SPH)
vector<pair<int, int>> run_sph(int t_start) {
    vector<bool> in_tree(V, false);
    vector<int> parent(V, -1);
    vector<Weight> dist(V, INF);
    priority_queue<pair<Weight, int>, vector<pair<Weight, int>>, greater<>> pq;
    
    auto add_to_tree = [&](int u) {
        in_tree[u] = true;
        dist[u] = 0;
        pq.push({0, u});
    };
    
    add_to_tree(t_start);
    int terminals_connected = 1;
    vector<pair<int, int>> tree_edges;
    
    while (terminals_connected < K) {
        int found_terminal = -1;
        while (!pq.empty()) {
            auto [d, u] = pq.top();
            pq.pop();
            if (d > dist[u]) continue;
            if (is_terminal[u] && !in_tree[u]) {
                found_terminal = u;
                break;
            }
            for (auto& edge : adj[u]) {
                int v = edge.to;
                Weight w = edge.weight;
                if (dist[v] > d + w) {
                    dist[v] = d + w;
                    parent[v] = u;
                    pq.push({dist[v], v});
                }
            }
        }
        
        if (found_terminal == -1) {
            break;
        }
        
        int curr = found_terminal;
        vector<int> path;
        while (!in_tree[curr] && curr != -1) {
            path.push_back(curr);
            curr = parent[curr];
        }
        
        int prev = curr;
        for (int i = (int)path.size() - 1; i >= 0; --i) {
            int u = path[i];
            tree_edges.push_back({prev, u});
            add_to_tree(u);
            prev = u;
        }
        terminals_connected++;
    }
    
    return tree_edges;
}

// Kou-Markowsky-Berman (KMB) Heuristic
vector<pair<int, int>> run_kmb() {
    vector<vector<Weight>> dist(K, vector<Weight>(V, INF));
    vector<vector<int>> pred(K, vector<int>(V, -1));
    for (int i = 0; i < K; ++i) {
        priority_queue<pair<Weight, int>, vector<pair<Weight, int>>, greater<>> pq;
        dist[i][term[i]] = 0;
        pq.push({0, term[i]});
        while (!pq.empty()) {
            auto [d, u] = pq.top();
            pq.pop();
            if (d > dist[i][u]) continue;
            for (auto& edge : adj[u]) {
                int v = edge.to;
                Weight w = edge.weight;
                if (dist[i][v] > d + w) {
                    dist[i][v] = d + w;
                    pred[i][v] = u;
                    pq.push({dist[i][v], v});
                }
            }
        }
    }
    
    struct MetricEdge {
        int u, v;
        Weight w;
        bool operator<(const MetricEdge& other) const {
            return w < other.w;
        }
    };
    vector<MetricEdge> metric_edges;
    for (int i = 0; i < K; ++i) {
        for (int j = i + 1; j < K; ++j) {
            if (dist[i][term[j]] < INF) {
                metric_edges.push_back({i, j, dist[i][term[j]]});
            }
        }
    }
    
    sort(metric_edges.begin(), metric_edges.end());
    DSU dsu(K);
    vector<MetricEdge> metric_mst;
    for (auto& edge : metric_edges) {
        if (dsu.unite(edge.u, edge.v)) {
            metric_mst.push_back(edge);
        }
    }
    
    vector<bool> in_candidate_set(V, false);
    for (int t : term) {
        in_candidate_set[t] = true;
    }
    for (auto& edge : metric_mst) {
        int curr = term[edge.v];
        int start = term[edge.u];
        while (curr != start && curr != -1) {
            in_candidate_set[curr] = true;
            curr = pred[edge.u][curr];
        }
    }
    
    vector<int> candidate_vertices;
    for (int v = 0; v < V; ++v) {
        if (in_candidate_set[v]) {
            candidate_vertices.push_back(v);
        }
    }
    
    return optimize_candidate_set(candidate_vertices);
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    if (!(cin >> V >> E >> K)) return 0;
    
    vector<EdgeInput> raw_edges(E);
    for (int i = 0; i < E; ++i) {
        cin >> raw_edges[i].u >> raw_edges[i].v >> raw_edges[i].w;
        if (raw_edges[i].u > raw_edges[i].v) {
            swap(raw_edges[i].u, raw_edges[i].v);
        }
    }
    
    term.resize(K);
    is_terminal.assign(V, false);
    for (int i = 0; i < K; ++i) {
        cin >> term[i];
        is_terminal[term[i]] = true;
    }
    
    if (K <= 1) {
        cout << "0 0\n";
        return 0;
    }
    
    sort(raw_edges.begin(), raw_edges.end());
    vector<EdgeInput> unique_edges;
    for (auto& edge : raw_edges) {
        if (unique_edges.empty() || unique_edges.back().u != edge.u || unique_edges.back().v != edge.v) {
            unique_edges.push_back(edge);
        }
    }
    
    adj.resize(V);
    for (auto& edge : unique_edges) {
        adj[edge.u].push_back({edge.v, edge.w});
        adj[edge.v].push_back({edge.u, edge.w});
    }
    
    for (int u = 0; u < V; ++u) {
        sort(adj[u].begin(), adj[u].end(), [](const AdjEdge& a, const AdjEdge& b) {
            return a.to < b.to;
        });
    }
    
    bool run_exact = false;
    if (K <= 15) {
        double ops = 1.0;
        for (int i = 0; i < K; ++i) ops *= 3.0;
        ops *= V;
        if (ops <= 2.5e8) {
            run_exact = true;
        }
    }
    
    vector<pair<int, int>> best_edges;
    Weight best_weight = INF;
    
    if (run_exact) {
        auto dw_edges = run_dreyfus_wagner();
        vector<int> dw_vertices;
        vector<bool> in_dw(V, false);
        for (auto& p : dw_edges) {
            in_dw[p.first] = true;
            in_dw[p.second] = true;
        }
        for (int v = 0; v < V; ++v) {
            if (in_dw[v]) dw_vertices.push_back(v);
        }
        best_edges = optimize_candidate_set(dw_vertices);
        best_weight = get_edges_weight(best_edges);
    } else {
        // Run heuristics
        // 1. KMB Heuristic
        if (K <= 500) {
            auto kmb_edges = run_kmb();
            Weight kmb_w = get_edges_weight(kmb_edges);
            if (kmb_w < best_weight) {
                best_weight = kmb_w;
                best_edges = kmb_edges;
            }
        }
        
        // 2. SPH Heuristic starting from various terminals
        int num_starts = min(K, 100);
        vector<int> start_terminals = term;
        mt19937 rng(42);
        shuffle(start_terminals.begin(), start_terminals.end(), rng);
        
        for (int i = 0; i < num_starts; ++i) {
            auto sph_edges = run_sph(start_terminals[i]);
            vector<bool> in_sph(V, false);
            for (auto& p : sph_edges) {
                in_sph[p.first] = true;
                in_sph[p.second] = true;
            }
            vector<int> sph_vertices;
            for (int v = 0; v < V; ++v) {
                if (in_sph[v]) sph_vertices.push_back(v);
            }
            auto opt_edges = optimize_candidate_set(sph_vertices);
            Weight opt_w = get_edges_weight(opt_edges);
            if (opt_w < best_weight) {
                best_weight = opt_w;
                best_edges = opt_edges;
            }
        }
        
        // 3. Local Search to improve the best tree
        best_edges = local_search(best_edges);
        best_weight = get_edges_weight(best_edges);
    }
    
    cout << best_weight << " " << best_edges.size() << "\n";
    for (auto& p : best_edges) {
        cout << p.first << " " << p.second << "\n";
    }
    
    return 0;
}
