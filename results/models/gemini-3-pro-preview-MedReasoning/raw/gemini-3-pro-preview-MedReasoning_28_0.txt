use std::io::{self, Read, Write};
use std::time::Instant;

struct FastParser<'a> {
    bytes: &'a [u8],
    pos: usize,
}

impl<'a> FastParser<'a> {
    fn new(bytes: &'a [u8]) -> Self {
        Self { bytes, pos: 0 }
    }

    #[inline(always)]
    fn next_u32(&mut self) -> Option<u32> {
        let bytes = self.bytes;
        let mut pos = self.pos;
        let len = bytes.len();
        
        while pos < len && unsafe { *bytes.get_unchecked(pos) } <= b' ' {
            pos += 1;
        }
        if pos >= len {
            self.pos = pos;
            return None;
        }
        let mut val = 0u32;
        while pos < len {
            let b = unsafe { *bytes.get_unchecked(pos) };
            if b > b' ' {
                val = val * 10 + (b - b'0') as u32;
                pos += 1;
            } else {
                break;
            }
        }
        self.pos = pos;
        Some(val)
    }
}

struct SimpleRng {
    state: u64,
}

impl SimpleRng {
    fn new(seed: u64) -> Self {
        Self { state: seed }
    }

    fn next_u64(&mut self) -> u64 {
        let mut x = self.state;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.state = x;
        x
    }

    fn next_usize(&mut self, limit: usize) -> usize {
        if limit == 0 {
            return 0;
        }
        (self.next_u64() % limit as u64) as usize
    }

    fn shuffle<T>(&mut self, slice: &mut [T]) {
        for i in (1..slice.len()).rev() {
            let j = self.next_usize(i + 1);
            slice.swap(i, j);
        }
    }
}

#[derive(Clone, Copy)]
struct VertexState {
    degree: i32,
    prev: i32,
    next: i32,
    removed: bool,
    in_cover: bool,
}

fn main() {
    let start_time = Instant::now();

    let mut stdin_bytes = Vec::new();
    if io::stdin().read_to_end(&mut stdin_bytes).is_err() {
        return;
    }

    let mut parser = FastParser::new(&stdin_bytes);
    let num_vertices = match parser.next_u32() {
        Some(val) => val as usize,
        None => return,
    };
    let num_edges = match parser.next_u32() {
        Some(val) => val as usize,
        None => return,
    };

    let mut edges_flat = vec![0u32; num_edges * 2];
    let mut initial_degrees = vec![0u32; num_vertices];
    for i in 0..num_edges {
        let u = match parser.next_u32() {
            Some(val) => val,
            None => break,
        };
        let v = match parser.next_u32() {
            Some(val) => val,
            None => break,
        };
        edges_flat[i * 2] = u;
        edges_flat[i * 2 + 1] = v;
        initial_degrees[u as usize] += 1;
        initial_degrees[v as usize] += 1;
    }

    let mut offsets = vec![0usize; num_vertices + 1];
    for i in 0..num_vertices {
        offsets[i + 1] = offsets[i] + initial_degrees[i] as usize;
    }

    let mut edges = vec![0u32; offsets[num_vertices]];
    let mut current_offsets = offsets.clone();
    for i in 0..num_edges {
        let u = edges_flat[i * 2];
        let v = edges_flat[i * 2 + 1];
        edges[current_offsets[u as usize]] = v;
        current_offsets[u as usize] += 1;
        edges[current_offsets[v as usize]] = u;
        current_offsets[v as usize] += 1;
    }

    let mut max_degree = 0;
    for &d in &initial_degrees {
        if d as i32 > max_degree {
            max_degree = d as i32;
        }
    }

    let mut order: Vec<u32> = (0..num_vertices as u32).collect();
    let mut rng = SimpleRng::new(123456789u64);

    let mut best_cover_size = usize::MAX;
    let mut best_in_cover = vec![false; num_vertices];

    // Allocate reusable structures for the solver runs
    let mut state = vec![
        VertexState {
            degree: 0,
            prev: -1,
            next: -1,
            removed: false,
            in_cover: false,
        };
        num_vertices
    ];
    let mut head = vec![-1i32; (max_degree + 1) as usize];

    let mut run_count = 0;
    while start_time.elapsed().as_secs_f64() < 4.0 && run_count < 15 {
        run_count += 1;

        // Shuffle the order in which vertices are added to the buckets
        rng.shuffle(&mut order);

        // Reset structures
        head.fill(-1);
        for i in 0..num_vertices {
            let d = initial_degrees[i] as i32;
            state[i] = VertexState {
                degree: d,
                prev: -1,
                next: -1,
                removed: d == 0,
                in_cover: false,
            };
        }

        // Populate buckets
        for &u in &order {
            let u_idx = u as usize;
            let d = state[u_idx].degree;
            if d > 0 {
                let h = head[d as usize];
                state[u_idx].next = h;
                state[u_idx].prev = -1;
                if h != -1 {
                    state[h as usize].prev = u_idx as i32;
                }
                head[d as usize] = u_idx as i32;
            }
        }

        let mut run_max_degree = max_degree;

        while head[1] != -1 || run_max_degree > 1 {
            if head[1] != -1 {
                let u = head[1] as usize;
                state[u].removed = true;
                let n = state[u].next;
                head[1] = n;
                if n != -1 {
                    state[n as usize].prev = -1;
                }

                let start = offsets[u];
                let end = offsets[u + 1];
                let mut neighbor = None;
                for idx in start..end {
                    let v = edges[idx] as usize;
                    if !state[v].removed {
                        neighbor = Some(v);
                        break;
                    }
                }

                if let Some(v) = neighbor {
                    state[v].in_cover = true;
                    state[v].removed = true;
                    let d_v = state[v].degree as usize;
                    let p_v = state[v].prev;
                    let n_v = state[v].next;
                    if p_v != -1 {
                        state[p_v as usize].next = n_v;
                    } else {
                        head[d_v] = n_v;
                    }
                    if n_v != -1 {
                        state[n_v as usize].prev = p_v;
                    }

                    let start_v = offsets[v];
                    let end_v = offsets[v + 1];
                    for idx in start_v..end_v {
                        let w = edges[idx] as usize;
                        if !state[w].removed {
                            let d_w = state[w].degree as usize;
                            let p_w = state[w].prev;
                            let n_w = state[w].next;
                            if p_w != -1 {
                                state[p_w as usize].next = n_w;
                            } else {
                                head[d_w] = n_w;
                            }
                            if n_w != -1 {
                                state[n_w as usize].prev = p_w;
                            }

                            state[w].degree -= 1;
                            let new_d = state[w].degree;
                            if new_d == 0 {
                                state[w].removed = true;
                            } else {
                                let h = head[new_d as usize];
                                state[w].next = h;
                                state[w].prev = -1;
                                if h != -1 {
                                    state[h as usize].prev = w as i32;
                                }
                                head[new_d as usize] = w as i32;
                            }
                        }
                    }
                }
            } else {
                while run_max_degree > 1 && head[run_max_degree as usize] == -1 {
                    run_max_degree -= 1;
                }
                if run_max_degree <= 1 {
                    continue;
                }
                
                let mut choices = [0usize; 8];
                let mut count = 0;
                let mut curr = head[run_max_degree as usize];
                while curr != -1 && count < 8 {
                    choices[count] = curr as usize;
                    count += 1;
                    curr = state[curr as usize].next;
                }
                
                let u = if count > 1 {
                    choices[rng.next_usize(count)]
                } else {
                    choices[0]
                };
                
                state[u].in_cover = true;
                state[u].removed = true;
                
                let p = state[u].prev;
                let n = state[u].next;
                if p != -1 {
                    state[p as usize].next = n;
                } else {
                    head[run_max_degree as usize] = n;
                }
                if n != -1 {
                    state[n as usize].prev = p;
                }
                
                let start_u = offsets[u];
                let end_u = offsets[u + 1];
                for idx in start_u..end_u {
                    let w = edges[idx] as usize;
                    if !state[w].removed {
                        let d_w = state[w].degree as usize;
                        let p_w = state[w].prev;
                        let n_w = state[w].next;
                        if p_w != -1 {
                            state[p_w as usize].next = n_w;
                        } else {
                            head[d_w] = n_w;
                        }
                        if n_w != -1 {
                            state[n_w as usize].prev = p_w;
                        }

                        state[w].degree -= 1;
                        let new_d = state[w].degree;
                        if new_d == 0 {
                            state[w].removed = true;
                        } else {
                            let h = head[new_d as usize];
                            state[w].next = h;
                            state[w].prev = -1;
                            if h != -1 {
                                state[h as usize].prev = w as i32;
                            }
                            head[new_d as usize] = w as i32;
                        }
                    }
                }
            }
        }

        // Post-processing prune step
        let mut cover_vertices = Vec::new();
        for u in 0..num_vertices {
            if state[u].in_cover {
                cover_vertices.push(u);
            }
        }

        cover_vertices.sort_by_key(|&u| initial_degrees[u]);

        for &u in &cover_vertices {
            let mut redundant = true;
            let start = offsets[u];
            let end = offsets[u + 1];
            for idx in start..end {
                let v = edges[idx] as usize;
                if !state[v].in_cover {
                    redundant = false;
                    break;
                }
            }
            if redundant {
                state[u].in_cover = false;
            }
        }

        let mut current_cover_size = 0;
        for u in 0..num_vertices {
            if state[u].in_cover {
                current_cover_size += 1;
            }
        }

        if current_cover_size < best_cover_size {
            best_cover_size = current_cover_size;
            for u in 0..num_vertices {
                best_in_cover[u] = state[u].in_cover;
            }
        }
    }

    // Output formatting with BufWriter
    let mut final_vertices = Vec::new();
    for u in 0..num_vertices {
        if best_in_cover[u] {
            final_vertices.push(u);
        }
    }

    let stdout = io::stdout();
    let mut writer = io::BufWriter::new(stdout.lock());

    writeln!(writer, "{}", final_vertices.len()).unwrap();
    for i in 0..final_vertices.len() {
        if i > 0 {
            write!(writer, " ").unwrap();
        }
        write!(writer, "{}", final_vertices[i]).unwrap();
    }
    writeln!(writer).unwrap();
    writer.flush().unwrap();
}
