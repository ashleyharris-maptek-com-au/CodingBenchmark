use std::io::{self, BufRead};

struct SimpleRng {
    state: u64,
}

impl SimpleRng {
    fn new(seed: u64) -> Self {
        SimpleRng { state: seed }
    }
    fn next_u64(&mut self) -> u64 {
        self.state = self.state.wrapping_mul(6364136223846793005).wrapping_add(1);
        self.state
    }
    fn next_usize(&mut self, limit: usize) -> usize {
        (self.next_u64() as usize) % limit
    }
    fn shuffle<T>(&mut self, slice: &mut [T]) {
        for i in (1..slice.len()).rev() {
            let j = self.next_usize(i + 1);
            slice.swap(i, j);
        }
    }
}

fn get_heuristic_bisection(n: usize, adj: &[Vec<usize>], matrix: &[bool]) -> (usize, Vec<bool>) {
    let mut best_cut = usize::MAX;
    let mut best_partition = vec![false; n];
    let mut rng = SimpleRng::new(12345);

    // Run 100 iterations of local search
    for _ in 0..100 {
        let mut partition = vec![false; n];
        let mut indices: Vec<usize> = (0..n).collect();
        rng.shuffle(&mut indices);
        for &i in &indices[..n / 2] {
            partition[i] = true;
        }

        let mut d = vec![0i32; n];
        let mut current_cut = 0;
        for u in 0..n {
            let in_a = partition[u];
            let mut neighbors_a = 0;
            let mut neighbors_b = 0;
            for &v in &adj[u] {
                if partition[v] {
                    neighbors_a += 1;
                } else {
                    neighbors_b += 1;
                }
            }
            if in_a {
                d[u] = neighbors_b - neighbors_a;
                current_cut += neighbors_b as usize;
            } else {
                d[u] = neighbors_a - neighbors_b;
            }
        }

        let mut improved = true;
        while improved {
            improved = false;
            let mut best_swap = None;
            let mut best_delta = 0i32;

            for u in 0..n {
                if partition[u] {
                    for v in 0..n {
                        if !partition[v] {
                            let is_edge = if matrix[u * n + v] { 1 } else { 0 };
                            let delta = -d[u] - d[v] + 2 * is_edge;
                            if delta < best_delta {
                                best_delta = delta;
                                best_swap = Some((u, v));
                            }
                        }
                    }
                }
            }

            if let Some((u, v)) = best_swap {
                partition[u] = false;
                partition[v] = true;
                current_cut = (current_cut as i32 + best_delta) as usize;

                for i in 0..n {
                    let is_u_nbr = matrix[i * n + u];
                    let is_v_nbr = matrix[i * n + v];
                    if i != u && i != v {
                        let mut change = 0;
                        if is_u_nbr {
                            change += 2;
                        }
                        if is_v_nbr {
                            change -= 2;
                        }
                        if partition[i] {
                            d[i] += change;
                        } else {
                            d[i] -= change;
                        }
                    }
                }
                
                let mut u_a = 0;
                let mut u_b = 0;
                let mut v_a = 0;
                let mut v_b = 0;
                for &i in &adj[u] {
                    if partition[i] { u_a += 1; } else { u_b += 1; }
                }
                for &i in &adj[v] {
                    if partition[i] { v_a += 1; } else { v_b += 1; }
                }
                d[u] = u_a - u_b;
                d[v] = v_b - v_a;
                
                improved = true;
            }
        }

        if current_cut < best_cut {
            best_cut = current_cut;
            best_partition = partition;
        }
    }

    (best_cut, best_partition)
}

struct Solver {
    n: usize,
    m: usize,
    adj: Vec<Vec<usize>>,
    matrix: Vec<bool>,
    degrees: Vec<usize>,
    best_cut: usize,
    best_partition: Vec<bool>,
}

impl Solver {
    fn new(n: usize, m: usize, adj: Vec<Vec<usize>>, matrix: Vec<bool>, degrees: Vec<usize>) -> Self {
        Solver {
            n,
            m,
            adj,
            matrix,
            degrees,
            best_cut: usize::MAX,
            best_partition: vec![false; n],
        }
    }

    fn solve(&mut self) {
        // Run heuristic to initialize best_cut and best_partition
        let (h_cut, h_part) = get_heuristic_bisection(self.n, &self.adj, &self.matrix);
        self.best_cut = h_cut;
        self.best_partition = h_part;

        let mut assigned_a = vec![false; self.n];
        assigned_a[0] = true;

        let mut c_a = vec![0u32; self.n];
        let mut c_b = vec![0u32; self.n];

        // Since 0 is assigned to A, update c_a for neighbors of 0
        for &v in &self.adj[0] {
            c_a[v] += 1;
        }

        let m_u = self.m - self.degrees[0];

        let mut unassigned_list: Vec<usize> = (0..self.n).collect();
        let mut diffs_buf = vec![0i32; self.n];
        
        self.dfs(
            1, // size of A
            0, // size of B
            0, // current cut size
            m_u,
            &mut assigned_a,
            1, // assigned_count
            &mut unassigned_list,
            &mut c_a,
            &mut c_b,
            &mut diffs_buf,
        );
    }

    fn dfs(
        &mut self,
        size_a: usize,
        size_b: usize,
        current_cut: usize,
        m_u: usize,
        assigned_a: &mut Vec<bool>,
        assigned_count: usize,
        unassigned_list: &mut [usize],
        c_a: &mut [u32],
        c_b: &mut [u32],
        diffs_buf: &mut [i32],
    ) {
        let r_a = self.n / 2 - size_a;
        let r_b = self.n / 2 - size_b;

        // Base case: all vertices assigned
        if r_a == 0 && r_b == 0 {
            if current_cut < self.best_cut {
                self.best_cut = current_cut;
                self.best_partition = assigned_a.clone();
            }
            return;
        }

        // Calculate lower bound
        let lb = self.get_lower_bound(
            current_cut,
            r_a,
            r_b,
            m_u,
            assigned_count,
            unassigned_list,
            c_a,
            c_b,
            diffs_buf,
        );
        if lb >= self.best_cut {
            return;
        }

        // Choose the next vertex to assign
        let mut best_idx = 0;
        let mut max_val = (-1, -1);
        let unassigned_slice = &unassigned_list[assigned_count..];
        for (idx, &v) in unassigned_slice.iter().enumerate() {
            let val = ((c_a[v] + c_b[v]) as i32, self.degrees[v] as i32);
            if val > max_val {
                max_val = val;
                best_idx = idx;
            }
        }

        let u = unassigned_slice[best_idx];

        // Swap u to the front of the unassigned region
        unassigned_list.swap(assigned_count, assigned_count + best_idx);

        let u_unassigned_deg = self.degrees[u] - (c_a[u] + c_b[u]) as usize;
        let next_m_u = m_u - u_unassigned_deg;

        // Try assigning u to A
        if r_a > 0 {
            assigned_a[u] = true;
            let inc = c_b[u] as usize;
            
            for &v in &self.adj[u] {
                c_a[v] += 1;
            }

            self.dfs(
                size_a + 1,
                size_b,
                current_cut + inc,
                next_m_u,
                assigned_a,
                assigned_count + 1,
                unassigned_list,
                c_a,
                c_b,
                diffs_buf,
            );

            for &v in &self.adj[u] {
                c_a[v] -= 1;
            }
            assigned_a[u] = false;
        }

        // Try assigning u to B
        if r_b > 0 {
            let inc = c_a[u] as usize;
            
            for &v in &self.adj[u] {
                c_b[v] += 1;
            }

            self.dfs(
                size_a,
                size_b + 1,
                current_cut + inc,
                next_m_u,
                assigned_a,
                assigned_count + 1,
                unassigned_list,
                c_a,
                c_b,
                diffs_buf,
            );

            for &v in &self.adj[u] {
                c_b[v] -= 1;
            }
        }

        // Swap back to restore order
        unassigned_list.swap(assigned_count, assigned_count + best_idx);
    }

    fn get_lower_bound(
        &self,
        current_cut: usize,
        r_a: usize,
        r_b: usize,
        m_u: usize,
        assigned_count: usize,
        unassigned_list: &[usize],
        c_a: &[u32],
        c_b: &[u32],
        diffs_buf: &mut [i32],
    ) -> usize {
        let lb_internal = if m_u > (r_a * (r_a - 1)) / 2 + (r_b * (r_b - 1)) / 2 {
            m_u - (r_a * (r_a - 1)) / 2 - (r_b * (r_b - 1)) / 2
        } else {
            0
        };

        // 1. Fast simple lower bound
        let mut simple_sum = 0;
        let unassigned_slice = &unassigned_list[assigned_count..];
        for &v in unassigned_slice {
            simple_sum += std::cmp::min(c_a[v], c_b[v]) as usize;
        }

        let simple_lb = current_cut + simple_sum + lb_internal;
        if simple_lb >= self.best_cut {
            return simple_lb;
        }

        // 2. Tighter sorted-diff lower bound
        let count = unassigned_slice.len();
        let mut sum_c_b = 0;

        for (i, &v) in unassigned_slice.iter().enumerate() {
            sum_c_b += c_b[v] as usize;
            diffs_buf[i] = c_a[v] as i32 - c_b[v] as i32;
        }

        let diffs_slice = &mut diffs_buf[..count];
        if r_b > 0 && r_b < count {
            diffs_slice.select_nth_unstable(r_b);
        }
        let sum_diffs: i32 = diffs_slice[..r_b].iter().sum();

        current_cut + sum_c_b + sum_diffs as usize + lb_internal
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = match lines.next() {
        Some(Ok(l)) => l,
        _ => return,
    };
    let parts: Vec<&str> = first_line.split_whitespace().collect();
    if parts.len() < 2 {
        return;
    }
    let n: usize = parts[0].parse().unwrap();
    let m_input: usize = parts[1].parse().unwrap();

    let mut adj = vec![Vec::new(); n];
    let mut matrix = vec![false; n * n];
    let mut degrees = vec![0; n];
    let mut m = 0;

    for _ in 0..m_input {
        if let Some(Ok(line)) = lines.next() {
            let edge: Vec<&str> = line.split_whitespace().collect();
            if edge.len() >= 2 {
                let u: usize = edge[0].parse().unwrap();
                let v: usize = edge[1].parse().unwrap();
                if u < n && v < n && u != v {
                    if !matrix[u * n + v] {
                        matrix[u * n + v] = true;
                        matrix[v * n + u] = true;
                        adj[u].push(v);
                        adj[v].push(u);
                        degrees[u] += 1;
                        degrees[v] += 1;
                        m += 1;
                    }
                }
            }
        }
    }

    let mut solver = Solver::new(n, m, adj, matrix, degrees);
    solver.solve();

    println!("{}", solver.best_cut);
    let mut first = true;
    for i in 0..n {
        if solver.best_partition[i] {
            if !first {
                print!(" ");
            }
            print!("{}", i);
            first = false;
        }
    }
    println!();
}
