// Longest Common Substring of N strings — adaptive, parallel solver.
//
// Strategy overview (adaptive by problem scale):
//   * Tiny inputs (total <= 4096 bytes): direct brute force using std::string_view::find.
//   * Everything else: binary search on the answer length L. Feasibility of L is tested
//     with 122-bit (double 61-bit prime) rolling hashes: build a hash set of all length-L
//     windows of the SHORTEST string, then stream every other string through it, marking
//     per-string presence bits. A window that is present in all N strings proves L feasible.
//     Hashing only produces false POSITIVES (never false negatives), so the final candidate
//     is verified deterministically with parallel KMP; on the (astronomically unlikely)
//     failure the whole search is re-run with fresh random hash bases.
//   * Parallelism adapts: many strings -> one thread per string; few huge strings ->
//     each string is scanned in parallel chunks (chunk size chosen so the O(L) window
//     re-initialisation per chunk stays negligible).
//   * A one-bit Bloom-style prefilter (4 bits per set entry) rejects most misses with a
//     single probe into a small table before touching the main open-addressing arrays.
//
// MSVC x64 / C++20. Uses _umul128/__popcnt64 on MSVC, __int128 elsewhere.

#include <cstdio>
#include <cstdint>
#include <cstring>
#include <cstdlib>
#include <string>

#include <vector>
#include <thread>
#include <atomic>
#include <random>
#include <algorithm>
#include <memory>
#include <bit>

  #include <intrin.h>

#endif

using u8  = uint8_t;
using u32 = uint32_t;
using u64 = uint64_t;

static const u64 MOD61 = (1ull << 61) - 1;

static inline u64 mulmod(u64 a, u64 b) {
#if defined(_MSC_VER) && defined(_M_X64)
    u64 hi;
    u64 lo = _umul128(a, b, &hi);
#else
    unsigned __int128 p = (unsigned __int128)a * b;
    u64 hi = (u64)(p >> 64), lo = (u64)p;
#endif
    u64 r = (lo & MOD61) + ((hi << 3) | (lo >> 61));
    r = (r & MOD61) + (r >> 61);
    if (r >= MOD61) r -= MOD61;
    return r;
}
static inline u64 addmod(u64 a, u64 b) { u64 r = a + b; if (r >= MOD61) r -= MOD61; return r; }
static inline u64 submod(u64 a, u64 b) { return a >= b ? a - b : a + MOD61 - b; }
static u64 powmod(u64 a, u64 e) {
    u64 r = 1;
    while (e) { if (e & 1) r = mulmod(r, a); a = mulmod(a, a); e >>= 1; }
    return r;
}
static inline int pop64(u64 x) {
#if defined(_MSC_VER) && defined(_M_X64)
    return (int)__popcnt64(x);
#else
    return std::popcount(x);
#endif
}
static inline u64 mixh(u64 h1, u64 h2) {
    u64 x = h1 * 0x9E3779B97F4A7C15ull ^ (h2 + 0x9E3779B97F4A7C15ull + (h1 << 6) + (h1 >> 2));
    x ^= x >> 33; x *= 0xFF51AFD7ED558CCDull; x ^= x >> 29;
    return x;
}

template <class F>
static void parallel_for(size_t n, unsigned threads, F&& f) {
    if (n == 0) return;
    size_t t = std::min<size_t>(threads, n);
    if (t <= 1) { for (size_t i = 0; i < n; i++) f(i); return; }
    std::atomic<size_t> next{0};
    std::vector<std::thread> pool;
    pool.reserve(t);
    for (size_t k = 0; k < t; k++)
        pool.emplace_back([&] {
            size_t i;
            while ((i = next.fetch_add(1, std::memory_order_relaxed)) < n) f(i);
        });
    for (auto& th : pool) th.join();
}

struct Solver {
    std::vector<std::string_view> S;
    size_t N = 0, refIdx = 0;
    unsigned threads = 1;

    // Open-addressing set over the reference string's windows, reused every iteration.
    size_t maxSlots = 0, W = 0;              // W = 64-bit mask words per entry
    std::unique_ptr<u8[]>  used;
    std::unique_ptr<u64[]> H1, H2, MASK, BLOOM;
    std::unique_ptr<u32[]> POS;
    size_t slots = 0, slotMask = 0, bloomWords = 0;
    int bloomShift = 64;

    u64 B1 = 0, B2 = 0;

    void alloc() {
        size_t maxM = S[refIdx].size();
        maxSlots = std::bit_ceil(std::max<size_t>(16, 2 * maxM));
        W = (N + 63) / 64;
        used.reset(new u8[maxSlots]);
        H1.reset(new u64[maxSlots]);
        H2.reset(new u64[maxSlots]);
        POS.reset(new u32[maxSlots]);
        MASK.reset(new u64[maxSlots * W]);
        BLOOM.reset(new u64[std::max<size_t>(1, maxSlots / 16)]);
    }

    void insertRef(u64 h1, u64 h2, u32 pos, size_t rw, u64 rb) {
        u64 x = mixh(h1, h2);
        u64 bidx = x >> bloomShift;
        BLOOM[bidx >> 6] |= 1ull << (bidx & 63);
        size_t j = (size_t)(x & slotMask);
        while (used[j]) {
            if (H1[j] == h1 && H2[j] == h2) return;   // duplicate window, keep first pos
            j = (j + 1) & slotMask;
        }
        used[j] = 1; H1[j] = h1; H2[j] = h2; POS[j] = pos;
        MASK[j * W + rw] |= rb;
    }

    inline int64_t findSlot(u64 h1, u64 h2) const {
        u64 x = mixh(h1, h2);
        u64 bidx = x >> bloomShift;
        if (!((BLOOM[bidx >> 6] >> (bidx & 63)) & 1)) return -1;
        size_t j = (size_t)(x & slotMask);
        while (used[j]) {
            if (H1[j] == h1 && H2[j] == h2) return (int64_t)j;
            j = (j + 1) & slotMask;
        }
        return -1;
    }

    // Scan windows [a, b) of string i, marking presence bits. Returns #hits.
    size_t scanRange(size_t i, size_t L, size_t a, size_t b,
                     u64 BL1, u64 BL2, std::atomic<bool>* abortF) {
        const u8* s = (const u8*)S[i].data();
        u64 x1 = 0, x2 = 0;
        for (size_t k = 0; k < L; k++) {
            x1 = addmod(mulmod(x1, B1), (u64)s[a + k] + 1);
            x2 = addmod(mulmod(x2, B2), (u64)s[a + k] + 1);
        }
        size_t found = 0;
        size_t w = i >> 6;
        u64 bit = 1ull << (i & 63);
        for (size_t p = a;;) {
            int64_t j = findSlot(x1, x2);
            if (j >= 0) {
                found++;
                std::atomic_ref<u64> mr(MASK[(size_t)j * W + w]);
                // Cheap read first: avoids contended RMW storms on repetitive inputs.
                if (!(mr.load(std::memory_order_relaxed) & bit))
                    mr.fetch_or(bit, std::memory_order_relaxed);
            }
            if (++p >= b) break;
            if (abortF && (p & 8191) == 0 && abortF->load(std::memory_order_relaxed)) break;
            x1 = addmod(mulmod(submod(x1, mulmod((u64)s[p - 1] + 1, BL1)), B1), (u64)s[p - 1 + L] + 1);
            x2 = addmod(mulmod(submod(x2, mulmod((u64)s[p - 1] + 1, BL2)), B2), (u64)s[p - 1 + L] + 1);
        }
        return found;
    }

    bool feasible(size_t L, u32& outPos) {
        const std::string_view ref = S[refIdx];
        size_t m = ref.size() - L + 1;
        slots = std::bit_ceil(std::max<size_t>(16, 2 * m));
        slotMask = slots - 1;
        bloomWords = std::max<size_t>(1, slots / 16);
        bloomShift = 64 - (std::countr_zero(slots) + 2);

        size_t clr = std::min<size_t>((size_t)threads * 2, 256);
        parallel_for(clr, threads, [&](size_t c) {
            auto rng = [&](size_t tot) {
                return std::pair<size_t, size_t>(tot * c / clr, tot * (c + 1) / clr);
            };
            { auto [a, b] = rng(slots);      memset(used.get() + a, 0, b - a); }
            { auto [a, b] = rng(slots * W);  memset(MASK.get() + a, 0, (b - a) * 8); }
            { auto [a, b] = rng(bloomWords); memset(BLOOM.get() + a, 0, (b - a) * 8); }
        });

        u64 BL1 = powmod(B1, (u64)(L - 1));
        u64 BL2 = powmod(B2, (u64)(L - 1));

        {   // Build the reference window set.
            const u8* s = (const u8*)ref.data();
            u64 x1 = 0, x2 = 0;
            for (size_t k = 0; k < L; k++) {
                x1 = addmod(mulmod(x1, B1), (u64)s[k] + 1);
                x2 = addmod(mulmod(x2, B2), (u64)s[k] + 1);
            }
            size_t rw = refIdx >> 6;
            u64 rb = 1ull << (refIdx & 63);
            for (size_t p = 0;;) {
                insertRef(x1, x2, (u32)p, rw, rb);
                if (++p >= m) break;
                x1 = addmod(mulmod(submod(x1, mulmod((u64)s[p - 1] + 1, BL1)), B1), (u64)s[p - 1 + L] + 1);
                x2 = addmod(mulmod(submod(x2, mulmod((u64)s[p - 1] + 1, BL2)), B2), (u64)s[p - 1 + L] + 1);
            }
        }

        std::atomic<bool> infeasible{false};
        size_t others = N - 1;
        if (others >= std::max<size_t>(1, threads / 2)) {
            // Many strings: one thread per string.
            parallel_for(N, threads, [&](size_t i) {
                if (i == refIdx || infeasible.load(std::memory_order_relaxed)) return;
                size_t n = S[i].size();
                size_t f = scanRange(i, L, 0, n - L + 1, BL1, BL2, &infeasible);
                if (f == 0) infeasible.store(true, std::memory_order_relaxed);
            });
            if (infeasible.load()) return false;
        } else {
            // Few (possibly huge) strings: chunk each string across all cores.
            for (size_t i = 0; i < N; i++) {
                if (i == refIdx) continue;
                size_t mm = S[i].size() - L + 1;
                size_t chunkLen = std::max<size_t>(4 * L, (size_t)1 << 16);
                size_t chunks = std::min<size_t>(threads, std::max<size_t>(1, mm / chunkLen));
                std::atomic<size_t> tot{0};
                parallel_for(chunks, threads, [&](size_t c) {
                    size_t a = mm * c / chunks, b = mm * (c + 1) / chunks;
                    if (a < b) tot.fetch_add(scanRange(i, L, a, b, BL1, BL2, nullptr),
                                             std::memory_order_relaxed);
                });
                if (tot.load() == 0) return false;
            }
        }

        // Any window present in all N strings?
        std::atomic<int64_t> win{-1};
        size_t sc = std::min<size_t>((size_t)threads * 4, 1024);
        parallel_for(sc, threads, [&](size_t c) {
            size_t a = slots * c / sc, b = slots * (c + 1) / sc;
            for (size_t j = a; j < b; j++) {
                if (!used[j]) continue;
                size_t cnt = 0;
                const u64* mw = &MASK[j * W];
                for (size_t w2 = 0; w2 < W; w2++) cnt += (size_t)pop64(mw[w2]);
                if (cnt == N) { win.store((int64_t)j, std::memory_order_relaxed); return; }
            }
        });
        int64_t j = win.load();
        if (j < 0) return false;
        outPos = POS[j];
        return true;
    }

    // Deterministic check: does pat occur in every string? (parallel KMP, worst-case linear)
    bool verify(std::string_view pat) {
        size_t L = pat.size();
        std::vector<u32> pf(L);
        pf[0] = 0;
        for (size_t q = 1, k = 0; q < L; q++) {
            while (k && pat[q] != pat[k]) k = pf[k - 1];
            if (pat[q] == pat[k]) k++;
            pf[q] = (u32)k;
        }
        std::atomic<bool> ok{true};
        parallel_for(N, threads, [&](size_t i) {
            if (!ok.load(std::memory_order_relaxed)) return;
            std::string_view t = S[i];
            size_t k = 0;
            bool f = false;
            for (size_t x = 0; x < t.size(); x++) {
                while (k && t[x] != pat[k]) k = pf[k - 1];
                if (t[x] == pat[k]) k++;
                if (k == L) { f = true; break; }
            }
            if (!f) ok.store(false, std::memory_order_relaxed);
        });
        return ok.load();
    }
};

static std::string readAllStdin() {
#if defined(_MSC_VER)
    _setmode(_fileno(stdin), _O_BINARY);
#endif
    std::string buf;
    buf.resize(1 << 22);
    size_t len = 0;
    for (;;) {
        if (len == buf.size()) buf.resize(buf.size() * 2);
        size_t r = fread(&buf[len], 1, buf.size() - len, stdin);
        len += r;
        if (r == 0) break;
    }
    buf.resize(len);
    return buf;
}

static void writeAnswer(std::string_view s) {
    if (!s.empty()) fwrite(s.data(), 1, s.size(), stdout);
    fputc('\n', stdout);
}

int main() {
#if defined(_MSC_VER)
    _setmode(_fileno(stdout), _O_BINARY);
#endif
    std::string buf = readAllStdin();

    // Parse: first integer N, then N raw lines (CR stripped).
    const char* p = buf.data();
    const char* end = p + buf.size();
    while (p < end && (*p == ' ' || *p == '\t' || *p == '\r' || *p == '\n')) p++;
    long long n = 0;
    while (p < end && *p >= '0' && *p <= '9') n = n * 10 + (*p++ - '0');
    while (p < end && *p != '\n') p++;
    if (p < end) p++;

    if (n <= 0) { writeAnswer({}); return 0; }

    std::vector<std::string_view> S;
    S.reserve((size_t)n);
    for (long long i = 0; i < n; i++) {
        const char* q = p;
        while (q < end && *q != '\n') q++;
        const char* e = q;
        if (e > p && e[-1] == '\r') e--;
        S.emplace_back(p, (size_t)(e - p));
        p = (q < end) ? q + 1 : q;
    }

    size_t N = S.size();
    size_t refIdx = 0;
    for (size_t i = 1; i < N; i++)
        if (S[i].size() < S[refIdx].size()) refIdx = i;
    size_t minLen = S[refIdx].size();

    if (minLen == 0) { writeAnswer({}); return 0; }
    if (N == 1)      { writeAnswer(S[0]); return 0; }

    size_t total = 0;
    for (auto& s : S) total += s.size();

    // --- Tiny problems: exact brute force, no hashing needed. ---
    if (total <= 4096) {
        std::string_view ref = S[refIdx];
        for (size_t len = minLen; len >= 1; len--) {
            for (size_t st = 0; st + len <= ref.size(); st++) {
                std::string_view cand = ref.substr(st, len);
                bool all = true;
                for (size_t i = 0; i < N && all; i++)
                    if (i != refIdx && S[i].find(cand) == std::string_view::npos) all = false;
                if (all) { writeAnswer(cand); return 0; }
            }
        }
        writeAnswer({});
        return 0;
    }

    // --- General path: parallel binary search on answer length via double rolling hash. ---
    Solver sv;
    sv.S = S;
    sv.N = N;
    sv.refIdx = refIdx;
    unsigned hw = std::thread::hardware_concurrency();
    sv.threads = hw ? hw : 8;
    sv.alloc();

    std::mt19937_64 rng((u64)std::chrono::steady_clock::now().time_since_epoch().count()
                        ^ 0x0123456789ABCDEFull);
    for (int attempt = 0; attempt < 8; attempt++) {
        sv.B1 = 260 + rng() % (MOD61 - 300);
        sv.B2 = 260 + rng() % (MOD61 - 300);
        if (sv.B1 == sv.B2) sv.B2 = addmod(sv.B2, 1);

        size_t lo = 1, hi = minLen, bestL = 0;
        u32 bestPos = 0;
        while (lo <= hi) {
            size_t mid = lo + (hi - lo) / 2;
            u32 pos;
            if (sv.feasible(mid, pos)) { bestL = mid; bestPos = pos; lo = mid + 1; }
            else { if (mid == 1) break; hi = mid - 1; }
        }

        // Infeasibility is exact (hashing never yields false negatives): no common substring.
        if (bestL == 0) { writeAnswer({}); return 0; }

        std::string_view ans(S[refIdx].data() + bestPos, bestL);
        if (sv.verify(ans)) { writeAnswer(ans); return 0; }
        // Hash collision hit the pivotal decision (probability ~2^-40 per run): reseed & retry.
    }

    // Unreachable in practice; emit a guaranteed-correct (possibly suboptimal) answer.
    bool present[256];
    for (int c = 0; c < 256; c++) present[c] = true;
    for (size_t i = 0; i < N; i++) {
        bool has[256] = {};
        for (unsigned char c : S[i]) has[c] = true;
        for (int c = 0; c < 256; c++) present[c] = present[c] && has[c];
    }
    for (int c = 0; c < 256; c++)
        if (present[c]) { char ch = (char)c; writeAnswer({&ch, 1}); return 0; }
    writeAnswer({});
    return 0;
}
