// Tetrahedron packing into convex polyhedra.
//
// Strategy:
//  - Tetrahedra are emitted at Euler rotation (0, 0, pi/4). For the standard
//    tetrahedron (a,0,-a/sqrt2),(-a,0,-a/sqrt2),(0,a,a/sqrt2),(0,-a,a/sqrt2)
//    with a = e/sqrt2, this rotation yields the cube-inscribed orientation
//    with vertices at (+-e/2, +-e/2, +-e/2) (odd parity), i.e. the minimal
//    possible axis-aligned half-extent e/2 in every axis.
//  - Centers are placed on an HCP lattice with minimum pairwise distance d,
//    clipped so that every tetra vertex stays inside the container's bounding
//    box and inside the circumscribing sphere of the container's vertex set.
//  - The placement count adapts to container volume: we estimate the target
//    from the container size and stop emitting once it is reached, with a
//    hard cap so even "ludicrous" containers finish well inside 30 seconds.
//
#define _CRT_SECURE_NO_WARNINGS
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <cstdint>
#include <cmath>
#include <vector>
#include <algorithm>
#ifdef _WIN32
#include <io.h>
#include <fcntl.h>
#endif

static char obuf[1 << 22];
static size_t opos = 0;

static void flushOut() {
  if (opos) {
    fwrite(obuf, 1, opos, stdout);
    opos = 0;
  }
}

static inline void putRaw(const char* s, size_t n) {
  if (opos + n > sizeof(obuf)) flushOut();
  memcpy(obuf + opos, s, n);
  opos += n;
}

// Print a coordinate with 5 decimal places.
static inline void putCoord(double v) {
  char tmp[32];
  int n = 0;
  long long t = llround(v * 100000.0);
  if (t < 0) {
    tmp[n++] = '-';
    t = -t;
  }
  long long ip = t / 100000;
  int fp = (int)(t % 100000);
  char rev[24];
  int rn = 0;
  do {
    rev[rn++] = (char)('0' + (int)(ip % 10));
    ip /= 10;
  } while (ip);
  while (rn) tmp[n++] = rev[--rn];
  tmp[n++] = '.';
  tmp[n++] = (char)('0' + fp / 10000); fp %= 10000;
  tmp[n++] = (char)('0' + fp / 1000);  fp %= 1000;
  tmp[n++] = (char)('0' + fp / 100);   fp %= 100;
  tmp[n++] = (char)('0' + fp / 10);
  tmp[n++] = (char)('0' + fp % 10);
  putRaw(tmp, n);
}

struct V3 { double x, y, z; };

// Geometry shared by the counting and emission passes.
struct Packer {
  double lox, hix, loy, hiy, loz, hiz;   // allowed range for tetra centers
  double d, dy, dz, byoff;               // HCP lattice spacings
  double Gx, Gy, Gz, S2;                 // containment sphere (center, r^2)
  bool sphereActive;
  V3 off[4];                             // tetra vertex offsets from center

  // Allowed x interval for centers in the row at (y, z). Returns false if empty.
  bool rowRange(double y, double z, double& lo, double& hi) const {
    lo = lox;
    hi = hix;
    if (sphereActive) {
      for (int i = 0; i < 4; i++) {
        double ry = y + off[i].y - Gy;
        double rz = z + off[i].z - Gz;
        double rhs = S2 - ry * ry - rz * rz;
        if (rhs < 0.0) return false;
        double w = sqrt(rhs);
        lo = std::max(lo, Gx - off[i].x - w);
        hi = std::min(hi, Gx - off[i].x + w);
      }
    }
    return hi >= lo - 1e-12;
  }

  // emit=false: count lattice points (up to limit). emit=true: write them out.
  long long run(bool emit, long long limit) const {
    long long produced = 0;
    long long kmax = (long long)floor((hiz - loz) / dz + 1e-9);
    for (long long k = 0; k <= kmax; k++) {
      double z = loz + (double)k * dz;
      double ybase = loy + ((k & 1) ? byoff : 0.0);
      if (ybase > hiy + 1e-12) continue;
      long long jmax = (long long)floor((hiy - ybase) / dy + 1e-9);
      for (long long j = 0; j <= jmax; j++) {
        double y = ybase + (double)j * dy;
        double lo, hi;
        if (!rowRange(y, z, lo, hi)) continue;
        double xoff = lox + (((j & 1) ? 0.5 : 0.0) + ((k & 1) ? 0.5 : 0.0)) * d;
        long long i0 = (long long)ceil((lo - xoff) / d - 1e-9);
        long long i1 = (long long)floor((hi - xoff) / d + 1e-9);
        if (i1 < i0) continue;
        long long cnt = i1 - i0 + 1;
        if (!emit) {
          produced += cnt;
          if (produced >= limit) return produced;
        } else {
          for (long long i = i0; i <= i1; i++) {
            double x = xoff + (double)i * d;
            putCoord(x);
            putRaw(" ", 1);
            putCoord(y);
            putRaw(" ", 1);
            putCoord(z);
            putRaw(" 0 0 0.78540\n", 13);
            if (++produced >= limit) return produced;
          }
        }
      }
    }
    return produced;
  }
};

int main() {
#ifdef _WIN32
  _setmode(_fileno(stdout), _O_BINARY);
#endif
  int V;
  double E;
  if (scanf("%d %lf", &V, &E) != 2 || V < 4 || E <= 0.0) {
    printf("0\n");
    return 0;
  }
  std::vector<V3> P((size_t)V);
  for (int i = 0; i < V; i++) {
    if (scanf("%lf %lf %lf", &P[i].x, &P[i].y, &P[i].z) != 3) {
      printf("0\n");
      return 0;
    }
  }

  double mnx = 1e300, mny = 1e300, mnz = 1e300;
  double mxx = -1e300, mxy = -1e300, mxz = -1e300;
  double Gx = 0, Gy = 0, Gz = 0;
  for (const V3& p : P) {
    mnx = std::min(mnx, p.x); mxx = std::max(mxx, p.x);
    mny = std::min(mny, p.y); mxy = std::max(mxy, p.y);
    mnz = std::min(mnz, p.z); mxz = std::max(mxz, p.z);
    Gx += p.x; Gy += p.y; Gz += p.z;
  }
  Gx /= V; Gy /= V; Gz /= V;
  double maxDist = 0.0;
  for (const V3& p : P) {
    double dx = p.x - Gx, dy = p.y - Gy, dz = p.z - Gz;
    maxDist = std::max(maxDist, sqrt(dx * dx + dy * dy + dz * dz));
  }

  Packer pk;
  double b = 0.5 * E;                    // tetra half-extent at rz = pi/4
  double m = b + 3e-4;                   // wall margin (with safety slack)
  pk.d = (1.8 / sqrt(24.0)) * E * 1.005; // min center separation
  pk.dy = pk.d * sqrt(3.0) / 2.0;
  pk.dz = pk.d * sqrt(2.0 / 3.0);
  pk.byoff = pk.d * sqrt(3.0) / 6.0;
  pk.lox = mnx + m; pk.hix = mxx - m;
  pk.loy = mny + m; pk.hiy = mxy - m;
  pk.loz = mnz + m; pk.hiz = mxz - m;
  pk.Gx = Gx; pk.Gy = Gy; pk.Gz = Gz;
  double S = 1.1 * maxDist - 1e-3;
  pk.S2 = (S > 0.0) ? S * S : -1.0;
  pk.off[0] = { b,  b, -b};
  pk.off[1] = {-b, -b, -b};
  pk.off[2] = {-b,  b,  b};
  pk.off[3] = { b, -b,  b};

  if (pk.hix < pk.lox || pk.hiy < pk.loy || pk.hiz < pk.loz || S <= 0.0) {
    printf("0\n");
    return 0;
  }

  // Sphere clipping can be skipped if every tetra vertex of every corner of
  // the center box already lies inside the sphere.
  pk.sphereActive = true;
  {
    bool allIn = true;
    for (int c = 0; c < 8 && allIn; c++) {
      double cx = (c & 1) ? pk.hix : pk.lox;
      double cy = (c & 2) ? pk.hiy : pk.loy;
      double cz = (c & 4) ? pk.hiz : pk.loz;
      for (int i = 0; i < 4; i++) {
        double dx = cx + pk.off[i].x - Gx;
        double dy = cy + pk.off[i].y - Gy;
        double dz = cz + pk.off[i].z - Gz;
        if (dx * dx + dy * dy + dz * dz > pk.S2) { allIn = false; break; }
      }
    }
    if (allIn) pk.sphereActive = false;
  }

  // Adapt the emission target to the container size.
  double tvol = E * E * E / (6.0 * sqrt(2.0));
  double cvol = (mxx - mnx) * (mxy - mny) * (mxz - mnz) * 0.6;
  double eff = (cvol / tvol < 10.0) ? 0.4 : 0.856;
  long long base = std::max(1LL, (long long)(cvol / tvol * eff));
  const long long CAP = 22000000LL;      // I/O budget for ludicrous containers
  long long target = base + base / 2 + base / 16 + 8;
  if (target > CAP) target = CAP;

  long long avail = pk.run(false, target);
  long long K = std::min(avail, target);

  char head[32];
  int hn = snprintf(head, sizeof(head), "%lld\n", K);
  putRaw(head, (size_t)hn);
  if (K > 0) pk.run(true, K);
  flushOut();
  return 0;
}
