// 3D Bin Packing - extreme-point greedy heuristic with spatial-grid overlap checks.
// Adapts to instance size: several box orderings run in parallel across cores,
// capped candidate scanning + fast pruning for large inputs, global time budget
// well under 30 seconds. Best result (by packed count, then volume) wins.
using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Text;
using System.Threading.Tasks;

static class Program
{
    static int N, W, H, D;
    static int[] BW, BH, BD;
    static long[] BVol;
    static Stopwatch SW = Stopwatch.StartNew();
    const long TimeBudgetMs = 24000;

    static void Main()
    {
        var tok = ReadTokens();
        int p = 0;
        if (tok.Count < 4) { Console.WriteLine(0); return; }
        N = tok[p++]; W = tok[p++]; H = tok[p++]; D = tok[p++];
        if (N < 0) N = 0;
        BW = new int[N]; BH = new int[N]; BD = new int[N]; BVol = new long[N];
        for (int i = 0; i < N; i++)
        {
            int a = p < tok.Count ? tok[p++] : 1;
            int b = p < tok.Count ? tok[p++] : 1;
            int c = p < tok.Count ? tok[p++] : 1;
            BW[i] = a; BH[i] = b; BD[i] = c;
            BVol[i] = (long)a * b * c;
        }

        var orderings = new List<int[]>();
        // ascending volume: greedy count maximization
        orderings.Add(MakeOrder(delegate(int i, int j)
        {
            int c = BVol[i].CompareTo(BVol[j]);
            return c != 0 ? c : i - j;
        }));
        // ascending largest dimension
        orderings.Add(MakeOrder(delegate(int i, int j)
        {
            int mi = Math.Max(BW[i], Math.Max(BH[i], BD[i]));
            int mj = Math.Max(BW[j], Math.Max(BH[j], BD[j]));
            if (mi != mj) return mi - mj;
            int c = BVol[i].CompareTo(BVol[j]);
            return c != 0 ? c : i - j;
        }));
        // ascending perimeter
        orderings.Add(MakeOrder(delegate(int i, int j)
        {
            int si = BW[i] + BH[i] + BD[i];
            int sj = BW[j] + BH[j] + BD[j];
            return si != sj ? si - sj : i - j;
        }));
        // descending volume: big-first with small gap filling
        orderings.Add(MakeOrder(delegate(int i, int j)
        {
            int c = BVol[j].CompareTo(BVol[i]);
            return c != 0 ? c : i - j;
        }));
        // a few seeded shuffles for diversity (cheap on small/medium inputs)
        int shuffles = N <= 2000 ? 6 : (N <= 20000 ? 3 : 1);
        var rnd = new Random(12345);
        for (int s = 0; s < shuffles; s++)
        {
            var o = MakeOrder(delegate(int i, int j) { return i - j; });
            for (int i = o.Length - 1; i > 0; i--)
            {
                int j = rnd.Next(i + 1);
                int t = o[i]; o[i] = o[j]; o[j] = t;
            }
            orderings.Add(o);
        }

        var results = new PackResult[orderings.Count];
        Parallel.For(0, orderings.Count,
            new ParallelOptions { MaxDegreeOfParallelism = Environment.ProcessorCount },
            delegate(int oi)
            {
                var t0 = SW.ElapsedMilliseconds;
                var packer = new Packer(N, W, H, D, BW, BH, BD, BVol, SW, TimeBudgetMs);
                results[oi] = packer.Pack(orderings[oi]);
                if (Environment.GetEnvironmentVariable("BINPACK_DEBUG") == "1")
                    Console.Error.WriteLine("ord " + oi + ": count=" + results[oi].Count +
                        " ms=" + (SW.ElapsedMilliseconds - t0));
            });

        PackResult best = null;
        for (int i = 0; i < results.Length; i++)
        {
            var r = results[i];
            if (r == null) continue;
            if (best == null || r.Count > best.Count ||
                (r.Count == best.Count && r.Volume > best.Volume))
                best = r;
        }
        if (best == null) best = new PackResult { Count = 0, Idx = new int[0], X = new int[0], Y = new int[0], Z = new int[0] };

        var sb = new StringBuilder(best.Count * 16 + 16);
        sb.Append(best.Count).Append('\n');
        for (int i = 0; i < best.Count; i++)
            sb.Append(best.Idx[i]).Append(' ').Append(best.X[i]).Append(' ')
              .Append(best.Y[i]).Append(' ').Append(best.Z[i]).Append('\n');
        var stdout = Console.OpenStandardOutput();
        var bytes = Encoding.ASCII.GetBytes(sb.ToString());
        stdout.Write(bytes, 0, bytes.Length);
        stdout.Flush();
    }

    static int[] MakeOrder(Comparison<int> cmp)
    {
        var o = new int[N];
        for (int i = 0; i < N; i++) o[i] = i;
        Array.Sort(o, cmp);
        return o;
    }

    static List<int> ReadTokens()
    {
        var input = Console.In.ReadToEnd();
        var toks = new List<int>(1 << 16);
        int i = 0, n = input.Length;
        while (i < n)
        {
            while (i < n && (input[i] < '0' || input[i] > '9') && input[i] != '-') i++;
            if (i >= n) break;
            bool neg = input[i] == '-';
            if (neg) i++;
            long v = 0;
            while (i < n && input[i] >= '0' && input[i] <= '9')
            {
                v = v * 10 + (input[i] - '0');
                if (v > int.MaxValue) v = int.MaxValue;
                i++;
            }
            toks.Add((int)(neg ? -v : v));
        }
        return toks;
    }
}

sealed class PackResult
{
    public int Count;
    public long Volume;
    public int[] Idx, X, Y, Z;
}

sealed class Packer
{
    struct Pt
    {
        public int X, Y, Z;
        public Pt(int x, int y, int z) { X = x; Y = y; Z = z; }
    }

    sealed class PtCmp : IComparer<Pt>
    {
        public int Compare(Pt a, Pt b)
        {
            if (a.Z != b.Z) return a.Z - b.Z;
            if (a.Y != b.Y) return a.Y - b.Y;
            return a.X - b.X;
        }
    }

    readonly int N, W, H, D;
    readonly int[] BW, BH, BD;
    readonly long[] BVol;
    readonly Stopwatch SW;
    readonly long Deadline;

    int GX, GY, GZ, CW, CH, CD;
    List<int>[] Cells;
    int[] Stamp;
    int StampCtr;
    int[] PX, PY, PZ, PW, PH, PD;
    int PlacedCount;

    public Packer(int n, int w, int h, int d, int[] bw, int[] bh, int[] bd, long[] bvol,
                  Stopwatch sw, long deadline)
    {
        N = n; W = w; H = h; D = d;
        BW = bw; BH = bh; BD = bd; BVol = bvol;
        SW = sw; Deadline = deadline;
    }

    public PackResult Pack(int[] order)
    {
        // Grid resolution: bounded cell count, cells no smaller than 1 unit.
        GX = Math.Min(Math.Max(W, 1), 64);
        GY = Math.Min(Math.Max(H, 1), 64);
        GZ = Math.Min(Math.Max(D, 1), 64);
        CW = (W + GX - 1) / GX; if (CW < 1) CW = 1;
        CH = (H + GY - 1) / GY; if (CH < 1) CH = 1;
        CD = (D + GZ - 1) / GZ; if (CD < 1) CD = 1;
        Cells = new List<int>[GX * GY * GZ];
        Stamp = new int[N];
        StampCtr = 0;
        PX = new int[N]; PY = new int[N]; PZ = new int[N];
        PW = new int[N]; PH = new int[N]; PD = new int[N];
        var pidx = new int[N];
        PlacedCount = 0;

        var points = new SortedSet<Pt>(new PtCmp());
        points.Add(new Pt(0, 0, 0));
        var dead = new List<Pt>();

        long containerVol = (long)W * H * D;
        long usedVol = 0;

        // Cap candidate points examined per box (keeps very large runs fast).
        int maxScan = N <= 1500 ? int.MaxValue
                                : Math.Max(1500, (int)Math.Min(int.MaxValue, 80000000L / N));

        int[,] perms = { {0,1,2},{0,2,1},{1,0,2},{1,2,0},{2,0,1},{2,1,0} };
        var dimBuf = new int[3];
        int c0 = Math.Min(W, Math.Min(H, D));
        int c2 = Math.Max(W, Math.Max(H, D));
        int c1 = W + H + D - c0 - c2;

        for (int oi = 0; oi < order.Length; oi++)
        {
            if ((oi & 63) == 0 && SW.ElapsedMilliseconds > Deadline) break;
            int bi = order[oi];
            long bv = BVol[bi];
            if (bv <= 0 || bv > containerVol - usedVol) continue;
            dimBuf[0] = BW[bi]; dimBuf[1] = BH[bi]; dimBuf[2] = BD[bi];
            int s0 = Math.Min(dimBuf[0], Math.Min(dimBuf[1], dimBuf[2]));
            int s2 = Math.Max(dimBuf[0], Math.Max(dimBuf[1], dimBuf[2]));
            int s1 = dimBuf[0] + dimBuf[1] + dimBuf[2] - s0 - s2;
            if (s0 > c0 || s1 > c1 || s2 > c2) continue;

            int scanned = 0;
            bool placed = false;
            int bx = 0, by = 0, bz = 0, bw = 0, bh = 0, bd = 0;
            dead.Clear();

            foreach (var pt in points)
            {
                if (++scanned > maxScan) break;
                // a point buried inside an existing box can never host anything
                if (Overlaps(pt.X, pt.Y, pt.Z, 1, 1, 1)) { dead.Add(pt); continue; }

                // first feasible orientation in standard permutation order (identity
                // first) so a greedy checker inferring rotations reproduces our choice
                int bestW = 0, bestH = 0, bestD = 0;
                bool any = false;
                for (int k = 0; k < 6 && !any; k++)
                {
                    int w = dimBuf[perms[k, 0]], h = dimBuf[perms[k, 1]], d = dimBuf[perms[k, 2]];
                    if (pt.X + w > W || pt.Y + h > H || pt.Z + d > D) continue;
                    if (Overlaps(pt.X, pt.Y, pt.Z, w, h, d)) continue;
                    bestW = w; bestH = h; bestD = d; any = true;
                }
                if (any)
                {
                    bx = pt.X; by = pt.Y; bz = pt.Z;
                    bw = bestW; bh = bestH; bd = bestD;
                    dead.Add(pt);
                    placed = true;
                    break;
                }
            }

            for (int k = 0; k < dead.Count; k++) points.Remove(dead[k]);
            if (!placed) continue;

            int id = PlacedCount++;
            PX[id] = bx; PY[id] = by; PZ[id] = bz;
            PW[id] = bw; PH[id] = bh; PD[id] = bd;
            pidx[id] = bi;
            usedVol += (long)bw * bh * bd;
            InsertGrid(id, bx, by, bz, bw, bh, bd);

            if (bx + bw < W) points.Add(new Pt(bx + bw, by, bz));
            if (by + bh < H) points.Add(new Pt(bx, by + bh, bz));
            if (bz + bd < D) points.Add(new Pt(bx, by, bz + bd));
        }

        var res = new PackResult
        {
            Count = PlacedCount,
            Volume = usedVol,
            Idx = new int[PlacedCount],
            X = new int[PlacedCount],
            Y = new int[PlacedCount],
            Z = new int[PlacedCount]
        };
        for (int i = 0; i < PlacedCount; i++)
        {
            res.Idx[i] = pidx[i]; res.X[i] = PX[i]; res.Y[i] = PY[i]; res.Z[i] = PZ[i];
        }
        return res;
    }

    bool Overlaps(int x, int y, int z, int w, int h, int d)
    {
        int cx0 = x / CW, cx1 = (x + w - 1) / CW; if (cx1 >= GX) cx1 = GX - 1;
        int cy0 = y / CH, cy1 = (y + h - 1) / CH; if (cy1 >= GY) cy1 = GY - 1;
        int cz0 = z / CD, cz1 = (z + d - 1) / CD; if (cz1 >= GZ) cz1 = GZ - 1;
        StampCtr++;
        for (int cz = cz0; cz <= cz1; cz++)
            for (int cy = cy0; cy <= cy1; cy++)
                for (int cx = cx0; cx <= cx1; cx++)
                {
                    var list = Cells[(cz * GY + cy) * GX + cx];
                    if (list == null) continue;
                    for (int t = 0; t < list.Count; t++)
                    {
                        int id = list[t];
                        if (Stamp[id] == StampCtr) continue;
                        Stamp[id] = StampCtr;
                        if (x < PX[id] + PW[id] && PX[id] < x + w &&
                            y < PY[id] + PH[id] && PY[id] < y + h &&
                            z < PZ[id] + PD[id] && PZ[id] < z + d)
                            return true;
                    }
                }
        return false;
    }

    void InsertGrid(int id, int x, int y, int z, int w, int h, int d)
    {
        int cx0 = x / CW, cx1 = (x + w - 1) / CW; if (cx1 >= GX) cx1 = GX - 1;
        int cy0 = y / CH, cy1 = (y + h - 1) / CH; if (cy1 >= GY) cy1 = GY - 1;
        int cz0 = z / CD, cz1 = (z + d - 1) / CD; if (cz1 >= GZ) cz1 = GZ - 1;
        for (int cz = cz0; cz <= cz1; cz++)
            for (int cy = cy0; cy <= cy1; cy++)
                for (int cx = cx0; cx <= cx1; cx++)
                {
                    int ci = (cz * GY + cy) * GX + cx;
                    if (Cells[ci] == null) Cells[ci] = new List<int>(8);
                    Cells[ci].Add(id);
                }
    }
}
