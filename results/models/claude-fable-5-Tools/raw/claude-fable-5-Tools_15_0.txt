// Shadow covering of a polygon with non-intersecting tetrahedra.
//
// Key geometric fact used
// -----------------------
// The standard tetrahedron, orthogonally projected along its local +z axis,
// covers the diamond |x| + |y| <= 1 (times scale s), which contains a disk of
// radius s/sqrt(2) around the centre.  If the tetrahedron is rotated so that
// its local +z axis points toward the sun (i.e. along -sun_dir), its shadow on
// the ground is the image of that diamond under a linear map with singular
// values {1, 1/cos(theta)} (theta = sun tilt), i.e. the shadow only stretches.
// Hence for ANY sun direction the shadow contains a disk of radius s/sqrt(2)
// around the shadow of the tet centre - and therefore an axis-aligned square
// of side s centred there.
//
// Covering strategy (adaptive)
// ----------------------------
// * Tiny polygons (bounding box fits in one shadow square): a single
//   tetrahedron with an adapted scale.
// * Everything else: tile the plane with grid cells of side L slightly less
//   than the max shadow square (s = ~10, L = 0.95 s) and drop one tetrahedron
//   per cell touching the polygon.  Cells touching the polygon are found by
//   exact segment rasterisation of the boundary (Amanatides-Woo traversal)
//   plus an active-edge scanline sweep for interior cells, so the cost is
//   O(NV + perimeter/L + cells) and scales from tiny triangles to huge,
//   complex geometries.
//
// Non-intersection guarantee
// --------------------------
// Each grid tetrahedron is assigned one of 9 height layers by
// (i mod 3, j mod 3).  Tets in the same layer sit at the same height and are
// at least 3L apart in the plane, which exceeds the bounding-sphere diameter
// (2 * 1.2248 * s).  Tets in different layers occupy disjoint z-slabs
// (layer spacing > bounding-sphere diameter).  So no two tetrahedra intersect.

use std::collections::HashSet;
use std::io::{self, Read, Write};

// Max vertex norm of the standard tetrahedron, sqrt(1 + 0.707^2), padded up.
const TET_RADIUS: f64 = 1.2248;

// Quaternion (w,x,y,z) rotating (0,0,1) onto unit vector u.
fn quat_z_to(u: [f64; 3]) -> [f64; 4] {
    let w = 1.0 + u[2];
    if w < 1e-12 {
        return [0.0, 1.0, 0.0, 0.0]; // 180 degrees about x axis
    }
    let x = -u[1];
    let y = u[0];
    let n = (w * w + x * x + y * y).sqrt();
    [w / n, x / n, y / n, 0.0]
}

// Mark every grid cell crossed by segment a-b (Amanatides & Woo traversal).
fn raster_segment(
    a: (f64, f64),
    b: (f64, f64),
    ox: f64,
    oy: f64,
    l: f64,
    cells: &mut HashSet<(i64, i64)>,
) {
    let cell = |x: f64, y: f64| -> (i64, i64) {
        (((x - ox) / l).floor() as i64, ((y - oy) / l).floor() as i64)
    };
    let (mut i, mut j) = cell(a.0, a.1);
    let (ie, je) = cell(b.0, b.1);
    cells.insert((i, j));
    cells.insert((ie, je));
    let dx = b.0 - a.0;
    let dy = b.1 - a.1;
    let step_i: i64 = if dx > 0.0 { 1 } else { -1 };
    let step_j: i64 = if dy > 0.0 { 1 } else { -1 };
    let mut tmax_x = if dx != 0.0 {
        let next = ox + ((i + if dx > 0.0 { 1 } else { 0 }) as f64) * l;
        (next - a.0) / dx
    } else {
        f64::INFINITY
    };
    let mut tmax_y = if dy != 0.0 {
        let next = oy + ((j + if dy > 0.0 { 1 } else { 0 }) as f64) * l;
        (next - a.1) / dy
    } else {
        f64::INFINITY
    };
    let tdx = if dx != 0.0 { l / dx.abs() } else { f64::INFINITY };
    let tdy = if dy != 0.0 { l / dy.abs() } else { f64::INFINITY };
    let max_steps = (((ie - i).abs() + (je - j).abs()) as usize) + 2;
    let mut steps = 0usize;
    while (i != ie || j != je) && steps < max_steps {
        if tmax_x < tmax_y {
            i += step_i;
            tmax_x += tdx;
        } else {
            j += step_j;
            tmax_y += tdy;
        }
        cells.insert((i, j));
        steps += 1;
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).expect("read stdin");
    let mut nums = input
        .split_ascii_whitespace()
        .filter_map(|t| t.parse::<f64>().ok());
    let nv = nums.next().unwrap_or(0.0).max(0.0) as usize;
    let mut pts: Vec<(f64, f64)> = Vec::with_capacity(nv);
    for _ in 0..nv {
        let x = nums.next().unwrap_or(0.0);
        let y = nums.next().unwrap_or(0.0);
        pts.push((x, y));
    }
    let mut d = [
        nums.next().unwrap_or(0.0),
        nums.next().unwrap_or(0.0),
        nums.next().unwrap_or(-1.0),
    ];
    let dlen = (d[0] * d[0] + d[1] * d[1] + d[2] * d[2]).sqrt();
    if dlen < 1e-12 {
        d = [0.0, 0.0, -1.0];
    } else {
        for v in d.iter_mut() {
            *v /= dlen;
        }
    }
    if d[2].abs() < 1e-9 {
        // Exactly grazing sun cannot cast finite shadows; nudge to stay finite.
        d[2] = -1e-9;
    }

    let stdout = io::stdout();
    let mut out = io::BufWriter::new(stdout.lock());

    if pts.is_empty() {
        writeln!(out, "0").unwrap();
        return;
    }

    let (mut minx, mut miny) = pts[0];
    let (mut maxx, mut maxy) = pts[0];
    for &(x, y) in &pts {
        minx = minx.min(x);
        maxx = maxx.max(x);
        miny = miny.min(y);
        maxy = maxy.max(y);
    }
    let maxdim = (maxx - minx).max(maxy - miny);

    // Shadow position of a point at height h is shifted by -h * (dx/dz, dy/dz).
    let ax = d[0] / d[2];
    let ay = d[1] / d[2];
    // Keep the tetrahedra on the sun side of the ground plane.
    let sign = if d[2] < 0.0 { 1.0 } else { -1.0 };
    let q = quat_z_to([-d[0], -d[1], -d[2]]);

    let mut body = String::new();
    let mut count: usize = 0;
    let emit = |body: &mut String, cx: f64, cy: f64, h: f64, s: f64, count: &mut usize| {
        use std::fmt::Write as _;
        let px = cx + h * ax;
        let py = cy + h * ay;
        writeln!(
            body,
            "{:.6} {:.6} {:.6} {:.9} {:.9} {:.9} {:.9} {:.6}",
            px, py, h, q[0], q[1], q[2], q[3], s
        )
        .unwrap();
        *count += 1;
    };

    let s_max = 9.99_f64;
    if maxdim <= 0.9 * s_max {
        // Adaptive strategy 1: the whole polygon fits under one shadow square.
        let s = (maxdim * 1.1).max(0.05).min(s_max);
        let h = sign * TET_RADIUS * s * 1.05;
        emit(
            &mut body,
            0.5 * (minx + maxx),
            0.5 * (miny + maxy),
            h,
            s,
            &mut count,
        );
        writeln!(out, "{}", count).unwrap();
        out.write_all(body.as_bytes()).unwrap();
        return;
    }

    // Adaptive strategy 2: grid tiling with maximal-scale tetrahedra.
    let s = s_max;
    let l = 0.95 * s; // grid cell side; each shadow square (side s) overlaps neighbours
    let ox = minx;
    let oy = miny;
    let mut cells: HashSet<(i64, i64)> = HashSet::new();

    // Boundary cells: every cell an edge passes through.
    for k in 0..pts.len() {
        let a = pts[k];
        let b = pts[(k + 1) % pts.len()];
        raster_segment(a, b, ox, oy, l, &mut cells);
    }

    // Interior cells: scanline over row centres with an active edge list.
    #[derive(Clone, Copy)]
    struct Edge {
        x0: f64,
        y0: f64,
        x1: f64,
        y1: f64,
        ymin: f64,
        ymax: f64,
    }
    let mut edges: Vec<Edge> = Vec::with_capacity(pts.len());
    for k in 0..pts.len() {
        let (x0, y0) = pts[k];
        let (x1, y1) = pts[(k + 1) % pts.len()];
        if y0 != y1 {
            edges.push(Edge {
                x0,
                y0,
                x1,
                y1,
                ymin: y0.min(y1),
                ymax: y0.max(y1),
            });
        }
    }
    edges.sort_by(|a, b| a.ymin.partial_cmp(&b.ymin).unwrap());
    let jmax = ((maxy - oy) / l).floor() as i64;
    let mut next_edge = 0usize;
    let mut active: Vec<Edge> = Vec::new();
    let mut xs: Vec<f64> = Vec::new();
    for j in 0..=jmax {
        let yc = oy + (j as f64 + 0.5) * l;
        while next_edge < edges.len() && edges[next_edge].ymin <= yc {
            active.push(edges[next_edge]);
            next_edge += 1;
        }
        active.retain(|e| e.ymax >= yc);
        if active.is_empty() {
            continue;
        }
        xs.clear();
        for e in &active {
            if (e.y0 <= yc) != (e.y1 <= yc) {
                xs.push(e.x0 + (yc - e.y0) * (e.x1 - e.x0) / (e.y1 - e.y0));
            }
        }
        xs.sort_by(|a, b| a.partial_cmp(b).unwrap());
        let mut k = 0;
        while k + 1 < xs.len() {
            let (xa, xb) = (xs[k], xs[k + 1]);
            let i0 = ((xa - ox) / l - 0.5).ceil() as i64;
            let i1 = ((xb - ox) / l - 0.5).floor() as i64;
            for i in i0..=i1 {
                cells.insert((i, j));
            }
            k += 2;
        }
    }

    // Emit one tetrahedron per cell, layered so none can intersect.
    let brad = TET_RADIUS * s;
    let gap = 2.0 * brad * 1.02; // > bounding-sphere diameter => disjoint z-slabs
    let h0 = brad * 1.05; // keep the lowest layer above the ground plane
    let mut cell_list: Vec<(i64, i64)> = cells.into_iter().collect();
    cell_list.sort_unstable();
    for &(i, j) in &cell_list {
        let cx = ox + (i as f64 + 0.5) * l;
        let cy = oy + (j as f64 + 0.5) * l;
        let layer = (i.rem_euclid(3) + 3 * j.rem_euclid(3)) as f64;
        let h = sign * (h0 + layer * gap);
        emit(&mut body, cx, cy, h, s, &mut count);
    }
    writeln!(out, "{}", count).unwrap();
    out.write_all(body.as_bytes()).unwrap();
}
