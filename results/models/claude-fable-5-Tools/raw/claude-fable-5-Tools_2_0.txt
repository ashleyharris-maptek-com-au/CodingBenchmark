// Chinese Postman Problem (Route Inspection) solver.
//
// Strategy:
//   1. Read the (undirected, connected) graph.
//   2. Find vertices of odd degree.
//   3. Pair the odd vertices and duplicate the shortest paths between the
//      pairs so every vertex gets even degree:
//        - K <= 12 odd vertices: exact minimum-weight perfect matching via
//          bitmask DP over all-pairs shortest path distances.
//        - larger K: greedy "nearest unmatched odd vertex" matching using
//          early-exit Dijkstra searches (very fast even with tens of
//          thousands of odd vertices, since each search stops as soon as it
//          settles any unmatched odd vertex).
//   4. Run an iterative Hierholzer algorithm on the augmented multigraph to
//      produce an Euler circuit starting and ending at node 0.
//
// All hot paths use flat arrays (CSR adjacency, array-based binary heap,
// generation-stamped Dijkstra scratch) so the solver scales to 100k nodes /
// 1M edges well inside the 30 second budget.

using System;
using System.Collections.Generic;
using System.IO;

static class Program
{
    static int N, M;
    static int[] EU, EV, EW;            // original edges

    // CSR adjacency over original edges (used by Dijkstra)
    static int[] aStart, aTo, aEdge;

    // Dijkstra scratch (generation stamped so we never re-clear arrays)
    static long[] dist;
    static int[] stamp;
    static int[] parentEdge;
    static int curStamp;

    // binary heap (parallel arrays)
    static long[] heapKey;
    static int[] heapNode;
    static int heapCnt;

    // duplicated edge ids (each entry is an original edge id to traverse again)
    static List<int> dups = new List<int>();

    static void Main()
    {
        byte[] inp = ReadAllStdin();
        int p = 0;
        N = NextInt(inp, ref p);
        M = NextInt(inp, ref p);

        var stdout = Console.OpenStandardOutput();
        if (M == 0)
        {
            var b = System.Text.Encoding.ASCII.GetBytes("0\n");
            stdout.Write(b, 0, b.Length);
            stdout.Flush();
            return;
        }

        EU = new int[M]; EV = new int[M]; EW = new int[M];
        var deg = new int[N];
        for (int i = 0; i < M; i++)
        {
            int u = NextInt(inp, ref p);
            int v = NextInt(inp, ref p);
            int w = NextInt(inp, ref p);
            EU[i] = u; EV[i] = v; EW[i] = w;
            deg[u]++; deg[v]++;            // self-loop adds 2 (parity unchanged)
        }

        // ---- build CSR adjacency over original edges ----
        aStart = new int[N + 1];
        for (int i = 0; i < M; i++)
        {
            aStart[EU[i] + 1]++;
            aStart[EV[i] + 1]++;
        }
        for (int i = 0; i < N; i++) aStart[i + 1] += aStart[i];
        aTo = new int[2 * M];
        aEdge = new int[2 * M];
        var fill = new int[N];
        Array.Copy(aStart, fill, N);
        for (int i = 0; i < M; i++)
        {
            int u = EU[i], v = EV[i];
            aTo[fill[u]] = v; aEdge[fill[u]] = i; fill[u]++;
            aTo[fill[v]] = u; aEdge[fill[v]] = i; fill[v]++;
        }

        // ---- collect odd-degree vertices ----
        var oddList = new List<int>();
        for (int v = 0; v < N; v++)
            if ((deg[v] & 1) != 0) oddList.Add(v);
        int K = oddList.Count;

        if (K > 0)
        {
            dist = new long[N];
            stamp = new int[N];
            parentEdge = new int[N];
            heapKey = new long[1024];
            heapNode = new int[1024];

            if (K <= 12)
                MatchExact(oddList);
            else
                MatchGreedy(oddList);
        }

        // ---- build augmented multigraph and find Euler circuit ----
        int D = dups.Count;
        int T = M + D;                      // total edge instances
        // endpoints of instance t: t<M -> edge t, else edge dups[t-M]
        var gStart = new int[N + 1];
        for (int t = 0; t < T; t++)
        {
            int e = t < M ? t : dups[t - M];
            gStart[EU[e] + 1]++;
            gStart[EV[e] + 1]++;
        }
        for (int i = 0; i < N; i++) gStart[i + 1] += gStart[i];
        var gTo = new int[2 * T];
        var gInst = new int[2 * T];
        var gFill = new int[N];
        Array.Copy(gStart, gFill, N);
        for (int t = 0; t < T; t++)
        {
            int e = t < M ? t : dups[t - M];
            int u = EU[e], v = EV[e];
            gTo[gFill[u]] = v; gInst[gFill[u]] = t; gFill[u]++;
            gTo[gFill[v]] = u; gInst[gFill[v]] = t; gFill[v]++;
        }

        // ---- iterative Hierholzer starting at node 0 ----
        var used = new bool[T];
        var it = new int[N];
        Array.Copy(gStart, it, N);
        var stack = new int[T + 1];
        int sp = 0;
        var circuit = new int[T + 1];
        int cn = 0;
        stack[sp++] = 0;
        while (sp > 0)
        {
            int v = stack[sp - 1];
            int i = it[v], end = gStart[v + 1];
            while (i < end && used[gInst[i]]) i++;
            it[v] = i;
            if (i < end)
            {
                used[gInst[i]] = true;
                it[v] = i + 1;
                stack[sp++] = gTo[i];
            }
            else
            {
                circuit[cn++] = v;
                sp--;
            }
        }

        // ---- write route (circuit is a closed walk through 0; order or
        //      reverse order are both valid Euler circuits at 0) ----
        var sb = new byte[cn * 8 + 16];
        int op = 0;
        for (int i = 0; i < cn; i++)
        {
            op = WriteInt(sb, op, circuit[i]);
            sb[op++] = (byte)(i + 1 == cn ? '\n' : ' ');
        }
        stdout.Write(sb, 0, op);
        stdout.Flush();
    }

    // ---------------- matching: exact (K <= 12) ----------------

    static void MatchExact(List<int> odd)
    {
        int K = odd.Count;
        var d = new long[K, K];
        for (int i = 0; i < K; i++)
        {
            DijkstraFull(odd[i]);
            for (int j = 0; j < K; j++)
                d[i, j] = stamp[odd[j]] == curStamp ? dist[odd[j]] : long.MaxValue / 4;
        }
        int full = 1 << K;
        var dp = new long[full];
        var choice = new int[full];
        for (int m = 1; m < full; m++) dp[m] = long.MaxValue / 4;
        dp[0] = 0;
        for (int m = 0; m < full; m++)
        {
            if (dp[m] >= long.MaxValue / 4) continue;
            int i = 0;
            while (i < K && (m & (1 << i)) != 0) i++;
            if (i >= K) continue;
            for (int j = i + 1; j < K; j++)
            {
                if ((m & (1 << j)) != 0) continue;
                int nm = m | (1 << i) | (1 << j);
                long c = dp[m] + d[i, j];
                if (c < dp[nm]) { dp[nm] = c; choice[nm] = (i << 8) | j; }
            }
        }
        // reconstruct pairs and duplicate the connecting paths
        int mask = full - 1;
        while (mask != 0)
        {
            int ch = choice[mask];
            int i = ch >> 8, j = ch & 0xFF;
            DuplicatePath(odd[i], odd[j]);
            mask &= ~((1 << i) | (1 << j));
        }
    }

    // ---------------- matching: greedy (large K) ----------------

    static void MatchGreedy(List<int> odd)
    {
        var unmatched = new bool[N];
        foreach (int v in odd) unmatched[v] = true;
        foreach (int s in odd)
        {
            if (!unmatched[s]) continue;
            unmatched[s] = false;
            int t = DijkstraToNearest(s, unmatched);
            unmatched[t] = false;
            // duplicate path s..t using parent edges from this search
            int v = t;
            while (v != s)
            {
                int e = parentEdge[v];
                dups.Add(e);
                v = (EU[e] == v) ? EV[e] : EU[e];
            }
        }
    }

    // ---------------- Dijkstra variants ----------------

    static void HeapPush(long key, int node)
    {
        if (heapCnt == heapKey.Length)
        {
            Array.Resize(ref heapKey, heapCnt * 2);
            Array.Resize(ref heapNode, heapCnt * 2);
        }
        int i = heapCnt++;
        while (i > 0)
        {
            int par = (i - 1) >> 1;
            if (heapKey[par] <= key) break;
            heapKey[i] = heapKey[par]; heapNode[i] = heapNode[par];
            i = par;
        }
        heapKey[i] = key; heapNode[i] = node;
    }

    static int HeapPop(out long key)
    {
        key = heapKey[0];
        int node = heapNode[0];
        heapCnt--;
        long lk = heapKey[heapCnt]; int ln = heapNode[heapCnt];
        int i = 0;
        while (true)
        {
            int c = 2 * i + 1;
            if (c >= heapCnt) break;
            if (c + 1 < heapCnt && heapKey[c + 1] < heapKey[c]) c++;
            if (heapKey[c] >= lk) break;
            heapKey[i] = heapKey[c]; heapNode[i] = heapNode[c];
            i = c;
        }
        heapKey[i] = lk; heapNode[i] = ln;
        return node;
    }

    static void DijkstraInit(int s)
    {
        curStamp++;
        heapCnt = 0;
        dist[s] = 0; stamp[s] = curStamp; parentEdge[s] = -1;
        HeapPush(0, s);
    }

    static void Relax(int v)
    {
        long dv = dist[v];
        for (int i = aStart[v], end = aStart[v + 1]; i < end; i++)
        {
            int to = aTo[i];
            if (to == v) continue;
            long nd = dv + EW[aEdge[i]];
            if (stamp[to] != curStamp || nd < dist[to])
            {
                stamp[to] = curStamp;
                dist[to] = nd;
                parentEdge[to] = aEdge[i];
                HeapPush(nd, to);
            }
        }
    }

    // full Dijkstra from s (distances for all reachable nodes)
    static void DijkstraFull(int s)
    {
        DijkstraInit(s);
        // settled tracking via a second use of dist: standard lazy deletion
        var done = curStamp; // (we just skip stale heap entries by key check)
        while (heapCnt > 0)
        {
            long k;
            int v = HeapPop(out k);
            if (k != dist[v] || stamp[v] != curStamp) continue;
            Relax(v);
        }
    }

    // Dijkstra from s, stop at first settled node with flag[node] true
    static int DijkstraToNearest(int s, bool[] flag)
    {
        DijkstraInit(s);
        while (heapCnt > 0)
        {
            long k;
            int v = HeapPop(out k);
            if (k != dist[v] || stamp[v] != curStamp) continue;
            if (flag[v]) return v;
            Relax(v);
        }
        throw new InvalidOperationException("graph not connected over odd vertices");
    }

    // Dijkstra from s with early exit at t, then duplicate the s..t path
    static void DuplicatePath(int s, int t)
    {
        DijkstraInit(s);
        while (heapCnt > 0)
        {
            long k;
            int v = HeapPop(out k);
            if (k != dist[v] || stamp[v] != curStamp) continue;
            if (v == t) break;
            Relax(v);
        }
        int x = t;
        while (x != s)
        {
            int e = parentEdge[x];
            dups.Add(e);
            x = (EU[e] == x) ? EV[e] : EU[e];
        }
    }

    // ---------------- fast IO ----------------

    static byte[] ReadAllStdin()
    {
        using (var s = Console.OpenStandardInput())
        using (var ms = new MemoryStream())
        {
            var buf = new byte[1 << 16];
            int n;
            while ((n = s.Read(buf, 0, buf.Length)) > 0) ms.Write(buf, 0, n);
            return ms.ToArray();
        }
    }

    static int NextInt(byte[] b, ref int p)
    {
        while (p < b.Length && (b[p] < (byte)'0' || b[p] > (byte)'9') && b[p] != (byte)'-') p++;
        bool neg = false;
        if (p < b.Length && b[p] == (byte)'-') { neg = true; p++; }
        int x = 0;
        while (p < b.Length && b[p] >= (byte)'0' && b[p] <= (byte)'9')
        {
            x = x * 10 + (b[p] - (byte)'0');
            p++;
        }
        return neg ? -x : x;
    }

    static int WriteInt(byte[] b, int p, int v)
    {
        if (v == 0) { b[p++] = (byte)'0'; return p; }
        int start = p;
        while (v > 0) { b[p++] = (byte)('0' + v % 10); v /= 10; }
        for (int i = start, j = p - 1; i < j; i++, j--)
        {
            byte tmp = b[i]; b[i] = b[j]; b[j] = tmp;
        }
        return p;
    }
}
