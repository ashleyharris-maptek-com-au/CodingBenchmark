// 2D Cutting Stock / Rectangle Bin Packing solver.
//
// Strategy (adaptive by problem size):
//  - Small instances (N <= ~1200): MaxRects packing tried with several sort
//    orders and two placement heuristics (Best Short Side Fit, Best Area Fit),
//    placing each rectangle on the best position across all open boards.
//    The run using the fewest boards wins.
//  - Large instances: Skyline bottom-left packing, first-fit-decreasing over
//    open boards (with a free-area filter so full boards are skipped cheaply).
//    Multiple sort orders are tried while the time budget allows.
//
// Rotation by 90 degrees is considered everywhere. A global ~25 second time
// guard stops extra heuristic passes; a single pass always completes.

#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <cstdint>
#include <vector>
#include <algorithm>
#include <chrono>
#include <string>

using namespace std;
using Clock = chrono::steady_clock;

static Clock::time_point g_start;
static double elapsedSec() {
    return chrono::duration<double>(Clock::now() - g_start).count();
}
static const double TIME_LIMIT = 25.0; // leave margin inside 30s

struct RectIn { int w, h, idx; };
struct Placement { int board, x, y, rot; };

static int W, H, N;

// ---------------------------------------------------------------------------
// MaxRects packer
// ---------------------------------------------------------------------------
struct FR { int x, y, w, h; };

struct MRBoard {
    vector<FR> freeRects;
    long long freeArea;
    MRBoard() { freeRects.push_back({0, 0, W, H}); freeArea = (long long)W * H; }
};

// heuristic: 0 = best short side fit, 1 = best area fit
struct MRPos { int x, y; bool rot; long long s1, s2; bool found; };

static MRPos mrFind(const MRBoard& b, int w, int h, int heuristic) {
    MRPos best; best.found = false; best.s1 = best.s2 = INT64_MAX;
    for (const FR& fr : b.freeRects) {
        for (int r = 0; r < 2; ++r) {
            int rw = r ? h : w, rh = r ? w : h;
            if (r && w == h) break;
            if (rw > fr.w || rh > fr.h) continue;
            long long s1, s2;
            long long dx = fr.w - rw, dy = fr.h - rh;
            if (heuristic == 0) { s1 = min(dx, dy); s2 = max(dx, dy); }
            else { s1 = (long long)fr.w * fr.h - (long long)rw * rh; s2 = min(dx, dy); }
            if (s1 < best.s1 || (s1 == best.s1 && s2 < best.s2)) {
                best = {fr.x, fr.y, r != 0, s1, s2, true};
            }
        }
    }
    return best;
}

static void mrPlace(MRBoard& b, int x, int y, int w, int h) {
    // Split all free rectangles that intersect the placed node.
    vector<FR>& fl = b.freeRects;
    size_t n = fl.size();
    vector<FR> add;
    for (size_t i = 0; i < n;) {
        FR fr = fl[i];
        if (x < fr.x + fr.w && x + w > fr.x && y < fr.y + fr.h && y + h > fr.y) {
            // split into up to 4 pieces
            if (y > fr.y)                 add.push_back({fr.x, fr.y, fr.w, y - fr.y});
            if (y + h < fr.y + fr.h)      add.push_back({fr.x, y + h, fr.w, fr.y + fr.h - (y + h)});
            if (x > fr.x)                 add.push_back({fr.x, fr.y, x - fr.x, fr.h});
            if (x + w < fr.x + fr.w)      add.push_back({x + w, fr.y, fr.x + fr.w - (x + w), fr.h});
            fl[i] = fl[--n];
            fl.pop_back();
        } else ++i;
    }
    for (const FR& a : add) fl.push_back(a);
    // Prune contained rectangles.
    for (size_t i = 0; i < fl.size(); ++i) {
        for (size_t j = i + 1; j < fl.size(); ++j) {
            const FR& a = fl[i]; const FR& c = fl[j];
            if (a.x >= c.x && a.y >= c.y && a.x + a.w <= c.x + c.w && a.y + a.h <= c.y + c.h) {
                fl[i] = fl.back(); fl.pop_back(); --i; break;
            }
            if (c.x >= a.x && c.y >= a.y && c.x + c.w <= a.x + a.w && c.y + c.h <= a.y + a.h) {
                fl[j] = fl.back(); fl.pop_back(); --j;
            }
        }
    }
    b.freeArea -= (long long)w * h;
}

// Pack all rects (in the given order) with MaxRects, best placement across
// all open boards. Returns board count; fills out placements.
static int solveMaxRects(const vector<RectIn>& rects, int heuristic,
                         vector<Placement>& out) {
    vector<MRBoard> boards;
    out.assign(N, {});
    for (const RectIn& r : rects) {
        long long area = (long long)r.w * r.h;
        int bestB = -1; MRPos best; best.found = false; best.s1 = best.s2 = INT64_MAX;
        for (int bi = 0; bi < (int)boards.size(); ++bi) {
            if (boards[bi].freeArea < area) continue;
            MRPos p = mrFind(boards[bi], r.w, r.h, heuristic);
            if (p.found && (p.s1 < best.s1 || (p.s1 == best.s1 && p.s2 < best.s2))) {
                best = p; bestB = bi;
            }
        }
        if (bestB < 0) {
            boards.emplace_back();
            bestB = (int)boards.size() - 1;
            best = mrFind(boards[bestB], r.w, r.h, heuristic);
            if (!best.found) { // cannot fit even on empty board; force place
                best = {0, 0, false, 0, 0, true};
            }
        }
        int pw = best.rot ? r.h : r.w, ph = best.rot ? r.w : r.h;
        mrPlace(boards[bestB], best.x, best.y, pw, ph);
        out[r.idx] = {bestB, best.x, best.y, best.rot ? 1 : 0};
    }
    return (int)boards.size();
}

// ---------------------------------------------------------------------------
// Skyline packer (bottom-left, with wasted-area accounting)
// ---------------------------------------------------------------------------
struct Seg { int x, y, w; };

struct SkyBoard {
    vector<Seg> segs;
    long long freeArea; // usable area at or above the skyline
    SkyBoard() { segs.push_back({0, 0, W}); freeArea = (long long)W * H; }
};

struct SkyPos { int x, y; bool rot; bool found; long long cost1, cost2; };

// Find bottom-left position for w x h (single orientation).
// cost1 = resulting top edge (y+h), cost2 = wasted area beneath.
static void skyFindOne(const SkyBoard& b, int w, int h, bool rot, SkyPos& best) {
    const vector<Seg>& s = b.segs;
    int ns = (int)s.size();
    for (int i = 0; i < ns; ++i) {
        int x = s[i].x;
        if (x + w > W) break;
        int y = 0;
        long long waste = 0;
        int covered = 0;
        for (int j = i; j < ns && covered < w; ++j) {
            int segRight = s[j].x + s[j].w;
            int use = min(segRight - (x + covered), w - covered);
            if (s[j].y > y) {
                // area previously counted as waste-free under the raised part
                waste += (long long)(s[j].y - y) * covered;
                y = s[j].y;
            } else {
                waste += (long long)(y - s[j].y) * use;
            }
            covered += use;
        }
        if (y + h > H) continue;
        long long c1 = y + h, c2 = waste;
        if (!best.found || c1 < best.cost1 ||
            (c1 == best.cost1 && (c2 < best.cost2 ||
             (c2 == best.cost2 && x < best.x)))) {
            best.found = true; best.x = x; best.y = y; best.rot = rot;
            best.cost1 = c1; best.cost2 = c2;
        }
    }
}

static SkyPos skyFind(const SkyBoard& b, int w, int h) {
    SkyPos best; best.found = false; best.cost1 = best.cost2 = INT64_MAX; best.x = 0;
    if (w <= W && h <= H) skyFindOne(b, w, h, false, best);
    if (w != h && h <= W && w <= H) skyFindOne(b, h, w, true, best);
    return best;
}

static void skyPlace(SkyBoard& b, int x, int y, int w, int h) {
    vector<Seg> ns;
    ns.reserve(b.segs.size() + 2);
    long long waste = 0;
    for (const Seg& s : b.segs) {
        int sl = s.x, sr = s.x + s.w;
        if (sr <= x || sl >= x + w) { ns.push_back(s); continue; }
        if (sl < x) ns.push_back({sl, s.y, x - sl});
        // overlapped part: area between old skyline and new rect bottom is waste
        int ol = max(sl, x), orr = min(sr, x + w);
        waste += (long long)(y - s.y) * (orr - ol);
        if (sr > x + w) ns.push_back({x + w, s.y, sr - (x + w)});
    }
    // insert the new top segment, keep sorted by x
    Seg add{x, y + h, w};
    vector<Seg> fin;
    fin.reserve(ns.size() + 1);
    bool ins = false;
    for (const Seg& s : ns) {
        if (!ins && s.x > add.x) { fin.push_back(add); ins = true; }
        fin.push_back(s);
    }
    if (!ins) fin.push_back(add);
    // merge adjacent equal-height segments
    vector<Seg>& out = b.segs;
    out.clear();
    for (const Seg& s : fin) {
        if (!out.empty() && out.back().y == s.y && out.back().x + out.back().w == s.x)
            out.back().w += s.w;
        else out.push_back(s);
    }
    b.freeArea -= (long long)w * h + waste;
}

static int solveSkyline(const vector<RectIn>& rects, vector<Placement>& out) {
    vector<SkyBoard> boards;
    out.assign(N, {});
    for (const RectIn& r : rects) {
        long long area = (long long)r.w * r.h;
        int placedB = -1; SkyPos pos;
        for (int bi = 0; bi < (int)boards.size(); ++bi) {
            if (boards[bi].freeArea < area) continue;
            pos = skyFind(boards[bi], r.w, r.h);
            if (pos.found) { placedB = bi; break; }
        }
        if (placedB < 0) {
            boards.emplace_back();
            placedB = (int)boards.size() - 1;
            pos = skyFind(boards[placedB], r.w, r.h);
            if (!pos.found) { pos.found = true; pos.x = 0; pos.y = 0; pos.rot = false; }
        }
        int pw = pos.rot ? r.h : r.w, ph = pos.rot ? r.w : r.h;
        skyPlace(boards[placedB], pos.x, pos.y, pw, ph);
        out[r.idx] = {placedB, pos.x, pos.y, pos.rot ? 1 : 0};
    }
    return (int)boards.size();
}

// ---------------------------------------------------------------------------

int main() {
    g_start = Clock::now();
    if (scanf("%d %d %d", &N, &W, &H) != 3) return 0;
    vector<RectIn> rects(N);
    for (int i = 0; i < N; ++i) {
        scanf("%d %d", &rects[i].w, &rects[i].h);
        rects[i].idx = i;
    }

    // Sort comparators (all decreasing).
    auto byArea = [](const RectIn& a, const RectIn& b) {
        long long aa = (long long)a.w * a.h, bb = (long long)b.w * b.h;
        if (aa != bb) return aa > bb;
        return max(a.w, a.h) > max(b.w, b.h);
    };
    auto byMaxSide = [](const RectIn& a, const RectIn& b) {
        int am = max(a.w, a.h), bm = max(b.w, b.h);
        if (am != bm) return am > bm;
        return min(a.w, a.h) > min(b.w, b.h);
    };
    auto byHeight = [](const RectIn& a, const RectIn& b) {
        int ah = max(a.w, a.h), bh = max(b.w, b.h); // normalized tall side
        int aw = min(a.w, a.h), bw = min(b.w, b.h);
        if (ah != bh) return ah > bh;
        return aw > bw;
    };
    auto byPerim = [](const RectIn& a, const RectIn& b) {
        long long ap = a.w + a.h, bp = b.w + b.h;
        if (ap != bp) return ap > bp;
        return (long long)a.w * a.h > (long long)b.w * b.h;
    };

    vector<Placement> best, cur;
    int bestBoards = -1;

    auto consider = [&](int nb, vector<Placement>& p) {
        if (bestBoards < 0 || nb < bestBoards) { bestBoards = nb; best.swap(p); }
    };

    typedef bool (*CmpFn)(const RectIn&, const RectIn&);

    vector<RectIn> order = rects;

    if (N <= 1200) {
        // MaxRects with multiple orders and heuristics; skyline as extra try.
        struct Try { int cmp; int heur; };
        int cmpCount = 4;
        auto applySort = [&](int c) {
            order = rects;
            switch (c) {
                case 0: stable_sort(order.begin(), order.end(), byArea); break;
                case 1: stable_sort(order.begin(), order.end(), byMaxSide); break;
                case 2: stable_sort(order.begin(), order.end(), byHeight); break;
                case 3: stable_sort(order.begin(), order.end(), byPerim); break;
            }
        };
        for (int c = 0; c < cmpCount; ++c) {
            for (int hu = 0; hu < 2; ++hu) {
                if (bestBoards > 0 && elapsedSec() > TIME_LIMIT) break;
                applySort(c);
                int nb = solveMaxRects(order, hu, cur);
                consider(nb, cur);
            }
        }
        // Skyline passes too — occasionally beats MaxRects.
        for (int c = 0; c < cmpCount; ++c) {
            if (bestBoards > 0 && elapsedSec() > TIME_LIMIT) break;
            applySort(c);
            int nb = solveSkyline(order, cur);
            consider(nb, cur);
        }
    } else {
        // Large instance: skyline first-fit decreasing, several orders while
        // time allows.
        int cmps[4] = {0, 1, 2, 3};
        for (int t = 0; t < 4; ++t) {
            if (bestBoards > 0 && elapsedSec() > TIME_LIMIT * 0.7) break;
            order = rects;
            switch (cmps[t]) {
                case 0: stable_sort(order.begin(), order.end(), byArea); break;
                case 1: stable_sort(order.begin(), order.end(), byMaxSide); break;
                case 2: stable_sort(order.begin(), order.end(), byHeight); break;
                case 3: stable_sort(order.begin(), order.end(), byPerim); break;
            }
            int nb = solveSkyline(order, cur);
            consider(nb, cur);
        }
        // If moderate size and time remains, a MaxRects pass can improve.
        if (N <= 6000 && elapsedSec() < TIME_LIMIT * 0.4) {
            order = rects;
            stable_sort(order.begin(), order.end(), byArea);
            int nb = solveMaxRects(order, 0, cur);
            consider(nb, cur);
        }
    }

    // Output
    {
        string buf;
        buf.reserve((size_t)N * 20 + 16);
        char line[64];
        int len = snprintf(line, sizeof(line), "%d\n", bestBoards);
        buf.append(line, len);
        for (int i = 0; i < N; ++i) {
            const Placement& p = best[i];
            len = snprintf(line, sizeof(line), "%d %d %d %d\n", p.board, p.x, p.y, p.rot);
            buf.append(line, len);
        }
        fwrite(buf.data(), 1, buf.size(), stdout);
    }
    return 0;
}
