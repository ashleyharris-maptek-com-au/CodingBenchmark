// Minimum Edge Cut solver — adaptive across graph scales.
//
// Strategy:
//   * Streamed, buffered stdin parsing into compact CSR (uint32 endpoints).
//   * Trivial cases: N<2, no edges, already-disconnected -> cut of 0.
//   * Degree-1 vertex -> cut of 1 immediately.
//   * Small graphs (N <= 900): exact Stoer-Wagner global min cut (parallel
//     edges handled as integer weights).
//   * Larger graphs:
//       - Bridge search (iterative Tarjan) -> exact answer 1 if any bridge.
//       - No bridge & min degree 2 -> exact answer 2 (isolate that vertex).
//       - Otherwise start from the min-degree candidate cut (always valid)
//         and improve it with unit-capacity max-flow (BFS augmenting, early
//         abort at current best) between a min-degree source and sampled
//         targets (farthest vertex, max-degree vertex, random), under a
//         time budget so multi-GB graphs still finish promptly.
//   * The emitted cut is always recounted from the edge list, so the output
//     is a valid disconnecting cut whose size K matches the edge lines.
//
// Standard C++20 only; compiles with MSVC cl /std:c++20 (and g++/clang).

#include <cstdio>
#include <cstdint>
#include <cstring>
#include <cstdlib>
#include <vector>
#include <algorithm>
#include <numeric>
#include <chrono>
#include <random>

#ifdef _WIN32
#include <io.h>
#include <fcntl.h>
#endif

using u8  = uint8_t;
using u32 = uint32_t;
using u64 = uint64_t;
using i64 = int64_t;

static std::chrono::steady_clock::time_point T0;
static double elapsedSec() {
    return std::chrono::duration<double>(std::chrono::steady_clock::now() - T0).count();
}

// ---------------------------------------------------------------- fast I/O
struct Reader {
    static const size_t BUFSZ = 1u << 22;
    std::vector<char> buf;
    size_t pos = 0, len = 0;
    Reader() : buf(BUFSZ) {}
    inline int gc() {
        if (pos >= len) {
            len = fread(buf.data(), 1, BUFSZ, stdin);
            pos = 0;
            if (len == 0) return -1;
        }
        return (unsigned char)buf[pos++];
    }
    // Reads the next unsigned integer; returns false at EOF.
    inline bool readU64(u64& out) {
        int c = gc();
        while (c != -1 && (c < '0' || c > '9')) c = gc();
        if (c == -1) return false;
        u64 x = 0;
        while (c >= '0' && c <= '9') { x = x * 10 + (u64)(c - '0'); c = gc(); }
        out = x;
        return true;
    }
};

struct Writer {
    std::vector<char> b;
    Writer() { b.reserve(1u << 20); }
    inline void num(u64 x) {
        char t[20]; int i = 0;
        do { t[i++] = (char)('0' + x % 10); x /= 10; } while (x);
        while (i) b.push_back(t[--i]);
    }
    inline void ch(char c) { b.push_back(c); }
    void flush() { fwrite(b.data(), 1, b.size(), stdout); }
};

// Emit the cut induced by a 0/1 side marking: recount crossing edges from the
// original edge list so K always matches the listed edges.
static void emitCut(const std::vector<u8>& side, const std::vector<u32>& E) {
    u64 K = 0;
    for (size_t i = 0; i < E.size(); i += 2)
        if (side[E[i]] != side[E[i + 1]]) ++K;
    Writer w;
    w.num(K); w.ch('\n');
    for (size_t i = 0; i < E.size(); i += 2) {
        if (side[E[i]] != side[E[i + 1]]) {
            w.num(E[i]); w.ch(' '); w.num(E[i + 1]); w.ch('\n');
            if (w.b.size() > (1u << 22)) { w.flush(); w.b.clear(); }
        }
    }
    w.flush();
}

// ------------------------------------------------- exact Stoer-Wagner (small N)
static void solveStoerWagner(u32 n, const std::vector<u32>& E) {
    std::vector<i64> W((size_t)n * n, 0);
    for (size_t i = 0; i < E.size(); i += 2) {
        u32 u = E[i], v = E[i + 1];
        W[(size_t)u * n + v] += 1;
        W[(size_t)v * n + u] += 1;
    }
    std::vector<int> act(n);
    std::iota(act.begin(), act.end(), 0);
    std::vector<std::vector<int>> grp(n);
    for (u32 i = 0; i < n; ++i) grp[i] = { (int)i };
    std::vector<i64> wsum(n);
    std::vector<char> added(n);
    i64 best = INT64_MAX;
    std::vector<int> bestGrp = { 0 };

    while (act.size() > 1) {
        int m = (int)act.size();
        for (int v : act) { wsum[v] = 0; added[v] = 0; }
        int prev = -1, last = -1;
        i64 cutw = 0;
        for (int step = 0; step < m; ++step) {
            int sel = -1;
            for (int v : act)
                if (!added[v] && (sel < 0 || wsum[v] > wsum[sel])) sel = v;
            added[sel] = 1;
            prev = last; last = sel;
            cutw = wsum[sel];
            const i64* row = &W[(size_t)sel * n];
            for (int v : act)
                if (!added[v]) wsum[v] += row[v];
        }
        if (cutw < best) { best = cutw; bestGrp = grp[last]; }
        for (int v : act)
            if (v != last && v != prev) {
                W[(size_t)prev * n + v] += W[(size_t)last * n + v];
                W[(size_t)v * n + prev] = W[(size_t)prev * n + v];
            }
        grp[prev].insert(grp[prev].end(), grp[last].begin(), grp[last].end());
        act.erase(std::find(act.begin(), act.end(), last));
    }

    std::vector<u8> side(n, 0);
    for (int v : bestGrp) side[v] = 1;
    emitCut(side, E);
}

// -------------------------------------------------------------------- main
int main() {
    T0 = std::chrono::steady_clock::now();
#ifdef _WIN32
    _setmode(_fileno(stdin), _O_BINARY);
    _setmode(_fileno(stdout), _O_BINARY);
#endif

    Reader rd;
    u64 N64 = 0, M64 = 0;
    if (!rd.readU64(N64)) { fputs("0\n", stdout); return 0; }
    rd.readU64(M64);

    if (N64 < 2) { fputs("0\n", stdout); return 0; }
    u64 N = N64;

    // Parse edges; drop self-loops (never part of a minimum cut).
    std::vector<u32> E;
    {
        u64 want = M64 * 2;
        if (want > (u64)3e9) want = (u64)3e9; // cap the speculative reserve
        E.reserve((size_t)want);
    }
    for (u64 m = 0; m < M64; ++m) {
        u64 a, b;
        if (!rd.readU64(a) || !rd.readU64(b)) break;
        if (a == b || a >= N || b >= N) continue;
        E.push_back((u32)a);
        E.push_back((u32)b);
    }
    const u64 M = E.size() / 2;

    if (M == 0) { fputs("0\n", stdout); return 0; } // N>=2, no usable edges

    // Degrees + CSR adjacency.
    std::vector<u32> deg(N, 0);
    for (size_t i = 0; i < E.size(); ++i) deg[E[i]]++;

    std::vector<u64> off(N + 1, 0);
    for (u64 v = 0; v < N; ++v) off[v + 1] = off[v] + deg[v];
    const u64 A = 2 * M; // arc count
    std::vector<u32> adjTo(A);
    {
        std::vector<u64> cur(off.begin(), off.end() - 1);
        for (size_t i = 0; i < E.size(); i += 2) {
            u32 u = E[i], v = E[i + 1];
            adjTo[cur[u]++] = v;
            adjTo[cur[v]++] = u;
        }
    }

    // Connectivity check (BFS from 0) — also records the farthest vertex.
    std::vector<u32> q;
    q.reserve((size_t)N);
    std::vector<u8> vis(N, 0);
    u32 farthest = 0;
    {
        q.push_back(0); vis[0] = 1;
        size_t qh = 0;
        while (qh < q.size()) {
            u32 u = q[qh++];
            farthest = u;
            for (u64 i = off[u]; i < off[u + 1]; ++i) {
                u32 v = adjTo[i];
                if (!vis[v]) { vis[v] = 1; q.push_back(v); }
            }
        }
        if (q.size() < N) { fputs("0\n", stdout); return 0; } // already disconnected
    }

    // Min-degree vertex — isolating it is always a valid cut of size deg.
    u32 vmin = 0, vmax = 0;
    for (u64 v = 1; v < N; ++v) {
        if (deg[v] < deg[vmin]) vmin = (u32)v;
        if (deg[v] > deg[vmax]) vmax = (u32)v;
    }

    if (deg[vmin] == 1) {
        Writer w;
        w.num(1); w.ch('\n');
        w.num(vmin); w.ch(' '); w.num(adjTo[off[vmin]]); w.ch('\n');
        w.flush();
        return 0;
    }

    // Small graphs: exact.
    if (N <= 900) { solveStoerWagner((u32)N, E); return 0; }

    // Bridge search (iterative Tarjan, parallel-edge safe). Any bridge => 1.
    {
        std::vector<u32> disc(N, 0), low(N, 0), pred(N, 0);
        std::vector<u64> it(N);
        std::vector<u8> skippedPar(N, 0);
        std::vector<u32> stk;
        stk.reserve((size_t)N);
        u32 timer = 1;
        const u32 NOPRED = 0xFFFFFFFFu;

        disc[0] = low[0] = timer++;
        it[0] = off[0];
        pred[0] = NOPRED;
        stk.push_back(0);
        u32 bu = NOPRED, bv = NOPRED; // found bridge

        while (!stk.empty() && bu == NOPRED) {
            u32 u = stk.back();
            if (it[u] < off[u + 1]) {
                u32 v = adjTo[it[u]++];
                if (v == pred[u] && !skippedPar[u]) { skippedPar[u] = 1; continue; }
                if (!disc[v]) {
                    disc[v] = low[v] = timer++;
                    pred[v] = u;
                    skippedPar[v] = 0;
                    it[v] = off[v];
                    stk.push_back(v);
                } else if (disc[v] < low[u]) {
                    low[u] = disc[v];
                }
            } else {
                stk.pop_back();
                u32 p = pred[u];
                if (p != NOPRED) {
                    if (low[u] < low[p]) low[p] = low[u];
                    if (low[u] > disc[p]) { bu = p; bv = u; }
                }
            }
        }
        if (bu != NOPRED) {
            Writer w;
            w.num(1); w.ch('\n');
            w.num(bu); w.ch(' '); w.num(bv); w.ch('\n');
            w.flush();
            return 0;
        }
    }

    // No bridge => min cut >= 2. Min degree 2 => exactly 2.
    if (deg[vmin] == 2) {
        Writer w;
        w.num(2); w.ch('\n');
        w.num(vmin); w.ch(' '); w.num(adjTo[off[vmin]]); w.ch('\n');
        w.num(vmin); w.ch(' '); w.num(adjTo[off[vmin] + 1]); w.ch('\n');
        w.flush();
        return 0;
    }

    // Baseline: isolate the min-degree vertex.
    u64 best = deg[vmin];
    std::vector<u8> bestSide(N, 0);
    bestSide[vmin] = 1;

    // Max-flow improvement phase (unit capacities, BFS augmenting paths with
    // early abort at the current best). Skipped when arc indices would not
    // fit u32 or the candidate cut is too large for byte residuals.
    if (A < 0xFFFFFF00ull && best <= 250) {
        std::vector<u32> rev(A);
        {
            std::vector<u64> cur(off.begin(), off.end() - 1);
            for (size_t i = 0; i < E.size(); i += 2) {
                u32 u = E[i], v = E[i + 1];
                u64 p1 = cur[u]++, p2 = cur[v]++;
                rev[p1] = (u32)p2;
                rev[p2] = (u32)p1;
            }
        }
        std::vector<u8> cap(A, 1);
        std::vector<u32> visStamp(N, 0), pred(N), parArc(N);
        u32 curStamp = 0;
        std::vector<u32> undoArcs;
        std::vector<u8> sideBuf;

        // Returns achieved flow; if < limit, a strictly better cut was found
        // and sideBuf holds the source-side marking. Residuals are restored.
        auto maxflow = [&](u32 s, u32 t, u64 limit, double deadline) -> u64 {
            u64 flow = 0;
            undoArcs.clear();
            bool cutFound = false;
            while (true) {
                if (elapsedSec() > deadline) break; // give up: no improvement claimed
                ++curStamp;
                q.clear();
                q.push_back(s);
                visStamp[s] = curStamp;
                bool reached = false;
                size_t qh = 0;
                while (qh < q.size() && !reached) {
                    u32 u = q[qh++];
                    for (u64 i = off[u]; i < off[u + 1]; ++i) {
                        if (!cap[i]) continue;
                        u32 v = adjTo[i];
                        if (visStamp[v] == curStamp) continue;
                        visStamp[v] = curStamp;
                        pred[v] = u;
                        parArc[v] = (u32)i;
                        if (v == t) { reached = true; break; }
                        q.push_back(v);
                    }
                }
                if (!reached) {
                    // Full residual BFS done: visited set = source side of a
                    // cut of size `flow` (< limit).
                    sideBuf.assign((size_t)N, 0);
                    sideBuf[s] = 1;
                    for (u32 v : q) sideBuf[v] = 1;
                    cutFound = true;
                    break;
                }
                // Augment along t -> s.
                ++flow;
                u32 v = t;
                while (v != s) {
                    u32 a = parArc[v];
                    cap[a]--;
                    cap[rev[a]]++;
                    undoArcs.push_back(a);
                    undoArcs.push_back(rev[a]);
                    v = pred[v];
                }
                if (flow >= limit) break; // cannot improve via this pair
            }
            for (u32 a : undoArcs) cap[a] = 1;
            return cutFound ? flow : limit;
        };

        const double deadline = elapsedSec() +
            (A < 2'000'000ull ? 3.0 : A < 40'000'000ull ? 5.0 : 12.0);
        size_t maxTargets =
            A < 2'000'000ull ? 256 : A < 20'000'000ull ? 96 :
            A < 200'000'000ull ? 24 : 8;
        // A non-improving attempt costs ~best full-graph BFS passes, so the
        // affordable streak shrinks as the graph grows.
        const int stopAfter = (int)(
            A < 2'000'000ull ? 32 : A < 20'000'000ull ? 14 :
            A < 200'000'000ull ? 7 : 4);

        std::mt19937_64 rng(0xC0FFEE1234ull);
        std::vector<std::pair<u32, u32>> pairs;
        pairs.reserve(maxTargets);
        auto addPair = [&](u32 s, u32 t) {
            if (s != t && pairs.size() < maxTargets) pairs.push_back({ s, t });
        };
        addPair(vmin, farthest);
        addPair(vmin, vmax);
        while (pairs.size() < maxTargets) {
            u32 t = (u32)(rng() % N);
            u32 s = (pairs.size() % 3 == 2) ? (u32)(rng() % N) : vmin;
            addPair(s, t);
            if (N <= pairs.size()) break;
        }

        // Stop early once a streak of attempts fails to improve the cut.
        int noImprove = 0;
        for (auto& [s, t] : pairs) {
            if (best <= 2 || noImprove >= stopAfter || elapsedSec() > deadline) break;
            u64 f = maxflow(s, t, best, deadline);
            if (f < best) {
                best = f;
                bestSide = sideBuf;
                noImprove = 0;
            } else {
                ++noImprove;
            }
        }
    }

    emitCut(bestSide, E);
    return 0;
}
