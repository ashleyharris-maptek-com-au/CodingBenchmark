use std::io::{self, Read};
use std::collections::HashSet;

#[derive(Clone, Copy, Debug)]
struct Point2D {
    x: f64,
    y: f64,
}

enum ActiveGrid {
    Flat {
        data: Vec<bool>,
        cols: usize,
        rows: usize,
    },
    Sparse {
        set: HashSet<(usize, usize)>,
    },
}

impl ActiveGrid {
    fn new(cols: usize, rows: usize) -> Self {
        if cols * rows <= 5_000_000 {
            ActiveGrid::Flat {
                data: vec![false; cols * rows],
                cols,
                rows,
            }
        } else {
            ActiveGrid::Sparse {
                set: HashSet::new(),
            }
        }
    }

    fn set(&mut self, col: usize, row: usize) {
        match self {
            ActiveGrid::Flat { data, cols, .. } => {
                data[row * *cols + col] = true;
            }
            ActiveGrid::Sparse { set, .. } => {
                set.insert((col, row));
            }
        }
    }

    fn count(&self) -> usize {
        match self {
            ActiveGrid::Flat { data, .. } => {
                data.iter().filter(|&&b| b).count()
            }
            ActiveGrid::Sparse { set, .. } => {
                set.len()
            }
        }
    }

    fn get_active_cells(&self) -> Vec<(usize, usize)> {
        match self {
            ActiveGrid::Flat { data, cols, rows } => {
                let mut cells = Vec::new();
                for r in 0..*rows {
                    for c in 0..*cols {
                        if data[r * *cols + c] {
                            cells.push((c, r));
                        }
                    }
                }
                cells
            }
            ActiveGrid::Sparse { set, .. } => {
                let mut cells: Vec<(usize, usize)> = set.iter().cloned().collect();
                cells.sort_by(|a, b| a.0.cmp(&b.0).then(a.1.cmp(&b.1)));
                cells
            }
        }
    }
}

fn matrix_to_quaternion(r: [[f64; 3]; 3]) -> (f64, f64, f64, f64) {
    let tr = r[0][0] + r[1][1] + r[2][2];
    let (qw, qx, qy, qz) = if tr > 0.0 {
        let s = (tr + 1.0).sqrt() * 2.0;
        let qw = 0.25 * s;
        let qx = (r[2][1] - r[1][2]) / s;
        let qy = (r[0][2] - r[2][0]) / s;
        let qz = (r[1][0] - r[0][1]) / s;
        (qw, qx, qy, qz)
    } else if (r[0][0] > r[1][1]) && (r[0][0] > r[2][2]) {
        let s = (1.0 + r[0][0] - r[1][1] - r[2][2]).sqrt() * 2.0;
        let qw = (r[2][1] - r[1][2]) / s;
        let qx = 0.25 * s;
        let qy = (r[0][1] + r[1][0]) / s;
        let qz = (r[0][2] + r[2][0]) / s;
        (qw, qx, qy, qz)
    } else if r[1][1] > r[2][2] {
        let s = (1.0 + r[1][1] - r[0][0] - r[2][2]).sqrt() * 2.0;
        let qw = (r[0][2] - r[2][0]) / s;
        let qx = (r[0][1] + r[1][0]) / s;
        let qy = 0.25 * s;
        let qz = (r[1][2] + r[2][1]) / s;
        (qw, qx, qy, qz)
    } else {
        let s = (1.0 + r[2][2] - r[0][0] - r[1][1]).sqrt() * 2.0;
        let qw = (r[1][0] - r[0][1]) / s;
        let qx = (r[0][2] + r[2][0]) / s;
        let qy = (r[1][2] + r[2][1]) / s;
        let qz = 0.25 * s;
        (qw, qx, qy, qz)
    };
    
    let len = (qw*qw + qx*qx + qy*qy + qz*qz).sqrt();
    if len > 1e-9 {
        (qw / len, qx / len, qy / len, qz / len)
    } else {
        (1.0, 0.0, 0.0, 0.0)
    }
}

fn solve_for_offset(
    rotated_pts: &[Point2D],
    min_x: f64,
    min_y: f64,
    max_x: f64,
    max_y: f64,
    dx: f64,
    dy: f64,
    ox: f64,
    oy: f64,
) -> ActiveGrid {
    let nv = rotated_pts.len();
    
    // Shift points
    let mut pv = Vec::with_capacity(nv);
    let shift_x = min_x - ox;
    let shift_y = min_y - oy;
    for p in rotated_pts {
        pv.push(Point2D {
            x: p.x - shift_x,
            y: p.y - shift_y,
        });
    }
    
    let cols = ((max_x - min_x + ox) / dx).ceil() as usize;
    let rows = ((max_y - min_y + oy) / dy).ceil() as usize;
    let cols = cols.max(1);
    let rows = rows.max(1);
    
    let mut grid = ActiveGrid::new(cols, rows);
    
    // Bucket edges for scanline
    let mut buckets = vec![Vec::new(); rows];
    for i in 0..nv {
        let p1 = pv[i];
        let p2 = pv[(i + 1) % nv];
        let y_min = p1.y.min(p2.y);
        let y_max = p1.y.max(p2.y);
        let r_min = (y_min / dy).floor() as i32;
        let r_max = (y_max / dy).floor() as i32;
        let r_start = r_min.max(0).min(rows as i32 - 1) as usize;
        let r_end = r_max.max(0).min(rows as i32 - 1) as usize;
        for r in r_start..=r_end {
            buckets[r].push((p1, p2));
        }
    }
    
    // Scanline fill
    for r in 0..rows {
        let mut intersections = Vec::new();
        let y_center = (r as f64 + 0.5) * dy;
        for &(p1, p2) in &buckets[r] {
            let y_min = p1.y.min(p2.y);
            let y_max = p1.y.max(p2.y);
            if y_center >= y_min && y_center < y_max {
                if (p2.y - p1.y).abs() > 1e-9 {
                    let x_int = p1.x + (y_center - p1.y) * (p2.x - p1.x) / (p2.y - p1.y);
                    intersections.push(x_int);
                }
            }
        }
        intersections.sort_by(|a, b| a.partial_cmp(b).unwrap());
        
        let n = intersections.len();
        for i in (0..n).step_by(2) {
            if i + 1 < n {
                let x_start = intersections[i].max(0.0);
                let x_end = intersections[i+1].min(cols as f64 * dx);
                if x_start < x_end {
                    let col_start = (x_start / dx).floor() as i32;
                    let col_end = (x_end / dx).floor() as i32;
                    let c_start = col_start.max(0).min(cols as i32 - 1) as usize;
                    let c_end = col_end.max(0).min(cols as i32 - 1) as usize;
                    for c in c_start..=c_end {
                        grid.set(c, r);
                    }
                }
            }
        }
    }
    
    // Boundary cell marking
    for i in 0..nv {
        let p1 = pv[i];
        let p2 = pv[(i + 1) % nv];
        
        let col_start = (p1.x / dx).floor() as i32;
        let col_end = (p2.x / dx).floor() as i32;
        let row_start = (p1.y / dy).floor() as i32;
        let row_end = (p2.y / dy).floor() as i32;
        
        let mut t_values = Vec::new();
        if (p2.x - p1.x).abs() > 1e-9 {
            let c_min = col_start.min(col_end) + 1;
            let c_max = col_start.max(col_end);
            for c in c_min..=c_max {
                let t = (c as f64 * dx - p1.x) / (p2.x - p1.x);
                if t >= 0.0 && t <= 1.0 {
                    t_values.push(t);
                }
            }
        }
        if (p2.y - p1.y).abs() > 1e-9 {
            let r_min = row_start.min(row_end) + 1;
            let r_max = row_start.max(row_end);
            for r in r_min..=r_max {
                let t = (r as f64 * dy - p1.y) / (p2.y - p1.y);
                if t >= 0.0 && t <= 1.0 {
                    t_values.push(t);
                }
            }
        }
        t_values.push(0.0);
        t_values.push(1.0);
        t_values.sort_by(|a, b| a.partial_cmp(b).unwrap());
        t_values.dedup_by(|a, b| (*a - *b).abs() < 1e-9);
        
        for w in t_values.windows(2) {
            let t_mid = 0.5 * (w[0] + w[1]);
            let xm = p1.x + t_mid * (p2.x - p1.x);
            let ym = p1.y + t_mid * (p2.y - p1.y);
            let col = (xm / dx).floor() as i32;
            let row = (ym / dy).floor() as i32;
            let c = col.max(0).min(cols as i32 - 1) as usize;
            let r = row.max(0).min(rows as i32 - 1) as usize;
            grid.set(c, r);
        }
    }
    
    grid
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut tokens = input.split_whitespace();
    
    let nv_str = match tokens.next() {
        Some(t) => t,
        None => return,
    };
    let nv: usize = nv_str.parse().unwrap();
    
    let mut orig_pts = Vec::with_capacity(nv);
    for _ in 0..nv {
        let x: f64 = tokens.next().unwrap().parse().unwrap();
        let y: f64 = tokens.next().unwrap().parse().unwrap();
        orig_pts.push(Point2D { x, y });
    }
    
    let sx: f64 = tokens.next().unwrap().parse().unwrap();
    let sy: f64 = tokens.next().unwrap().parse().unwrap();
    let sz: f64 = tokens.next().unwrap().parse().unwrap();
    
    // Basis vectors U and V_unit on z=0
    let h_len = (sx*sx + sy*sy).sqrt();
    let (ux, uy, vx, vy, l_v) = if h_len < 1e-9 {
        (1.0, 0.0, 0.0, 1.0, 1.0)
    } else {
        let ux = sy / h_len;
        let uy = -sx / h_len;
        let vx = sx / h_len;
        let vy = sy / h_len;
        let s_len = (sx*sx + sy*sy + sz*sz).sqrt();
        let l_v = s_len / sz.abs();
        (ux, uy, vx, vy, l_v)
    };
    
    // Rotate polygon vertices to U, V_unit basis
    let mut rotated_pts = Vec::with_capacity(nv);
    for p in &orig_pts {
        let rx = p.x * ux + p.y * uy;
        let ry = p.x * vx + p.y * vy;
        rotated_pts.push(Point2D { x: rx, y: ry });
    }
    
    let mut min_x = rotated_pts[0].x;
    let mut max_x = rotated_pts[0].x;
    let mut min_y = rotated_pts[0].y;
    let mut max_y = rotated_pts[0].y;
    for p in &rotated_pts {
        if p.x < min_x { min_x = p.x; }
        if p.x > max_x { max_x = p.x; }
        if p.y < min_y { min_y = p.y; }
        if p.y > max_y { max_y = p.y; }
    }
    
    // Max scale and grid dimensions with safety margin
    let scale_factor = 10.0;
    let scale_raster = 9.9999;
    let dx = scale_raster * 2.0f64.sqrt();
    let dy = scale_raster * 2.0f64.sqrt() * l_v;
    
    // Choose number of offsets adaptively
    let n_offsets = if nv >= 2000 {
        1
    } else {
        // Estimate rough grid size
        let cols_est = ((max_x - min_x) / dx).ceil() as usize;
        let rows_est = ((max_y - min_y) / dy).ceil() as usize;
        if nv >= 500 || cols_est * rows_est >= 500 {
            2
        } else {
            4
        }
    };
    
    let mut best_grid = ActiveGrid::new(1, 1);
    let mut best_count = usize::MAX;
    let mut best_ox = 0.0;
    let mut best_oy = 0.0;
    
    for i in 0..n_offsets {
        for j in 0..n_offsets {
            let ox = (i as f64 / n_offsets as f64) * dx;
            let oy = (j as f64 / n_offsets as f64) * dy;
            let grid = solve_for_offset(
                &rotated_pts,
                min_x,
                min_y,
                max_x,
                max_y,
                dx,
                dy,
                ox,
                oy,
            );
            let count = grid.count();
            if count < best_count {
                best_count = count;
                best_grid = grid;
                best_ox = ox;
                best_oy = oy;
            }
        }
    }
    
    // Reconstruction of the 3D rotation matrix R
    let s_len = (sx*sx + sy*sy + sz*sz).sqrt();
    let s_hat = [sx / s_len, sy / s_len, sz / s_len];
    
    let (r0, r1, r2) = if h_len < 1e-9 {
        let sz_sign = if sz < 0.0 { -1.0 } else { 1.0 };
        let u = [1.0, 0.0, 0.0];
        let v = [0.0, sz_sign, 0.0];
        let r0 = [(u[0] + v[0]) / 2.0f64.sqrt(), (u[1] + v[1]) / 2.0f64.sqrt(), (u[2] + v[2]) / 2.0f64.sqrt()];
        let r1 = [(-u[0] + v[0]) / 2.0f64.sqrt(), (-u[1] + v[1]) / 2.0f64.sqrt(), (-u[2] + v[2]) / 2.0f64.sqrt()];
        let r2 = s_hat;
        (r0, r1, r2)
    } else {
        let u = [sy / h_len, -sx / h_len, 0.0];
        let v = [sz * sx / (s_len * h_len), sz * sy / (s_len * h_len), -h_len / s_len];
        let r0 = [(u[0] + v[0]) / 2.0f64.sqrt(), (u[1] + v[1]) / 2.0f64.sqrt(), (u[2] + v[2]) / 2.0f64.sqrt()];
        let r1 = [(-u[0] + v[0]) / 2.0f64.sqrt(), (-u[1] + v[1]) / 2.0f64.sqrt(), (-u[2] + v[2]) / 2.0f64.sqrt()];
        let r2 = s_hat;
        (r0, r1, r2)
    };
    
    let r_mat = [
        [r0[0], r1[0], r2[0]],
        [r0[1], r1[1], r2[1]],
        [r0[2], r1[2], r2[2]]
    ];
    
    let (qw, qx, qy, qz) = matrix_to_quaternion(r_mat);
    
    let active_cells = best_grid.get_active_cells();
    println!("{}", active_cells.len());
    
    let shift_x = min_x - best_ox;
    let shift_y = min_y - best_oy;
    
    for (k, &(col, row)) in active_cells.iter().enumerate() {
        let cx_rot = (col as f64 + 0.5) * dx + shift_x;
        let cy_rot = (row as f64 + 0.5) * dy + shift_y;
        
        // Convert back to original frame
        let cx = cx_rot * ux + cy_rot * vx;
        let cy = cx_rot * uy + cy_rot * vy;
        
        let pz = 100.0 * (k as f64);
        let px = cx + pz * sx / sz;
        let py = cy + pz * sy / sz;
        
        println!(
            "{:.9} {:.9} {:.9} {:.9} {:.9} {:.9} {:.9} {:.9}",
            px, py, pz, qw, qx, qy, qz, scale_factor
        );
    }
}
