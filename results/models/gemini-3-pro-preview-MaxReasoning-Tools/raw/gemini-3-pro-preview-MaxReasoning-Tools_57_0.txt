const fs = require('fs');

function solve(inputData) {
  const { width, height, grid, ignitions, towns, cutBudget } = inputData;

  // Helper to get 1D index
  const getIdx = (x, y) => y * width + x;

  // 1. BFS from ignitions to find all reachable cells
  const visited_ign = new Uint8Array(width * height);
  const queue_ign = new Int32Array(width * height);
  let q_ign_head = 0;
  let q_ign_tail = 0;
  const ignitionSet = new Set();
  
  for (const [ix, iy] of ignitions) {
    if (ix >= 0 && ix < width && iy >= 0 && iy < height) {
      const idx = getIdx(ix, iy);
      visited_ign[idx] = 1;
      queue_ign[q_ign_tail++] = idx;
      ignitionSet.add(idx);
    }
  }

  const dirs = [
    [1, 0], [-1, 0], [0, 1], [0, -1]
  ];

  while (q_ign_head < q_ign_tail) {
    const idx = queue_ign[q_ign_head++];
    const x = idx % width;
    const y = Math.floor(idx / width);
    for (const [dx, dy] of dirs) {
      const nx = x + dx;
      const ny = y + dy;
      if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
        if (grid[ny][nx] !== '~') {
          const nidx = getIdx(nx, ny);
          if (!visited_ign[nidx]) {
            visited_ign[nidx] = 1;
            queue_ign[q_ign_tail++] = nidx;
          }
        }
      }
    }
  }

  // 2. BFS from towns to find all cells that can reach some town
  const visited_town = new Uint8Array(width * height);
  const queue_town = new Int32Array(width * height);
  let q_town_head = 0;
  let q_town_tail = 0;
  const townMap = new Map();
  
  for (const t of towns) {
    if (t.x >= 0 && t.x < width && t.y >= 0 && t.y < height) {
      const idx = getIdx(t.x, t.y);
      visited_town[idx] = 1;
      queue_town[q_town_tail++] = idx;
      townMap.set(idx, t);
    }
  }

  while (q_town_head < q_town_tail) {
    const idx = queue_town[q_town_head++];
    const x = idx % width;
    const y = Math.floor(idx / width);
    for (const [dx, dy] of dirs) {
      const nx = x + dx;
      const ny = y + dy;
      if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
        if (grid[ny][nx] !== '~') {
          const nidx = getIdx(nx, ny);
          if (!visited_town[nidx]) {
            visited_town[nidx] = 1;
            queue_town[q_town_tail++] = nidx;
          }
        }
      }
    }
  }

  // Find active cells (intersection of reachable from ignitions and can reach towns)
  const activeCells = [];
  const cellToIndex = new Int32Array(width * height).fill(-1);
  for (let y = 0; y < height; y++) {
    for (let x = 0; x < width; x++) {
      const idx = getIdx(x, y);
      if (visited_ign[idx] && visited_town[idx]) {
        cellToIndex[idx] = activeCells.length;
        activeCells.push({ x, y, idx });
      }
    }
  }

  // If no town is reachable, we don't need any firebreaks
  if (activeCells.length === 0) {
    return { firebreaks: [] };
  }

  const M = activeCells.length;
  const S = 2 * M;
  const T = 2 * M + 1;

  // Identify reachable towns
  let maxTownVal = 0;
  const activeTowns = [];
  for (const t of towns) {
    if (t.x >= 0 && t.x < width && t.y >= 0 && t.y < height) {
      const idx = getIdx(t.x, t.y);
      if (cellToIndex[idx] !== -1) {
        activeTowns.push(t);
        if (t.value > maxTownVal) {
          maxTownVal = t.value;
        }
      }
    }
  }

  // Normalize town values relative to maxTownVal
  const townNormVals = new Map();
  for (const t of activeTowns) {
    townNormVals.set(getIdx(t.x, t.y), maxTownVal > 0 ? t.value / maxTownVal : 0);
  }

  // Dinic Max-Flow Solver
  const EPS = 1e-9;
  class Dinic {
    constructor(V, S, T) {
      this.V = V;
      this.S = S;
      this.T = T;
      this.graph = Array.from({ length: V }, () => []);
      this.level = new Int32Array(V);
      this.ptr = new Int32Array(V);
      this.queue = new Int32Array(V);
    }

    addEdge(u, v, cap) {
      const uLen = this.graph[u].length;
      const vLen = this.graph[v].length;
      this.graph[u].push({ to: v, cap: cap, origCap: cap, revIdx: vLen });
      this.graph[v].push({ to: u, cap: 0, origCap: 0, revIdx: uLen });
    }

    bfs() {
      this.level.fill(-1);
      this.level[this.S] = 0;
      this.queue[0] = this.S;
      let head = 0;
      let tail = 1;
      while (head < tail) {
        const u = this.queue[head++];
        const edges = this.graph[u];
        for (let i = 0; i < edges.length; i++) {
          const edge = edges[i];
          if (edge.cap > EPS && this.level[edge.to] === -1) {
            this.level[edge.to] = this.level[u] + 1;
            this.queue[tail++] = edge.to;
          }
        }
      }
      return this.level[this.T] !== -1;
    }

    dfs(u, pushed) {
      if (pushed < EPS) return 0;
      if (u === this.T) return pushed;
      const edges = this.graph[u];
      for (let i = this.ptr[u]; i < edges.length; i++) {
        this.ptr[u] = i;
        const edge = edges[i];
        const v = edge.to;
        if (this.level[u] + 1 !== this.level[v] || edge.cap < EPS) continue;
        const tr = this.dfs(v, Math.min(pushed, edge.cap));
        if (tr > EPS) {
          edge.cap -= tr;
          this.graph[v][edge.revIdx].cap += tr;
          return tr;
        }
      }
      return 0;
    }

    maxFlow() {
      let flow = 0;
      while (this.bfs()) {
        this.ptr.fill(0);
        while (true) {
          const pushed = this.dfs(this.S, Infinity);
          if (pushed < EPS) break;
          flow += pushed;
        }
      }
      return flow;
    }

    reset(W) {
      for (let u = 0; u < this.V; u++) {
        for (const edge of this.graph[u]) {
          edge.cap = edge.origCap;
        }
      }
      // Update town capacities to T
      for (const t of activeTowns) {
        const idx = getIdx(t.x, t.y);
        const k = cellToIndex[idx];
        const townCap = townNormVals.get(idx) * W;
        const u = 2 * k + 1;
        for (const edge of this.graph[u]) {
          if (edge.to === this.T) {
            edge.cap = townCap;
            this.graph[this.T][edge.revIdx].cap = 0;
          }
        }
      }
    }
  }

  const dinic = new Dinic(2 * M + 2, S, T);

  // Add edges between node input and output splits
  const FOREST_CAP = 1.0;
  for (let k = 0; k < M; k++) {
    const cell = activeCells[k];
    const idx = cell.idx;
    const uIn = 2 * k;
    const uOut = 2 * k + 1;

    if (ignitionSet.has(idx)) {
      dinic.addEdge(uIn, uOut, Infinity);
      dinic.addEdge(S, uIn, Infinity);
    } else if (townMap.has(idx)) {
      dinic.addEdge(uIn, uOut, Infinity);
      dinic.addEdge(uOut, T, 0); // initial dummy capacity
    } else {
      dinic.addEdge(uIn, uOut, FOREST_CAP);
    }
  }

  // Add edges between adjacent active cells
  for (let k = 0; k < M; k++) {
    const cell = activeCells[k];
    const uOut = 2 * k + 1;
    for (const [dx, dy] of dirs) {
      const nx = cell.x + dx;
      const ny = cell.y + dy;
      if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
        const nidx = getIdx(nx, ny);
        const nk = cellToIndex[nidx];
        if (nk !== -1) {
          const vIn = 2 * nk;
          dinic.addEdge(uOut, vIn, Infinity);
        }
      }
    }
  }

  // 3. Lagrangian Multiplier Search
  let L = 0;
  let R = M * FOREST_CAP;
  let bestCuts = [];
  let bestValSaved = -1;

  // Check if we can save all towns with the budget
  dinic.reset(M * FOREST_CAP * 10);
  dinic.maxFlow();
  
  let visited_res = new Uint8Array(2 * M + 2);
  let queue_res = new Int32Array(2 * M + 2);
  let q_res_head = 0;
  let q_res_tail = 0;
  visited_res[S] = 1;
  queue_res[q_res_tail++] = S;

  while (q_res_head < q_res_tail) {
    const u = queue_res[q_res_head++];
    for (const edge of dinic.graph[u]) {
      if (edge.cap > EPS && !visited_res[edge.to]) {
        visited_res[edge.to] = 1;
        queue_res[q_res_tail++] = edge.to;
      }
    }
  }

  let initialForestCuts = [];
  for (let k = 0; k < M; k++) {
    const cell = activeCells[k];
    if (!ignitionSet.has(cell.idx) && !townMap.has(cell.idx)) {
      if (visited_res[2 * k] && !visited_res[2 * k + 1]) {
        initialForestCuts.push([cell.x, cell.y]);
      }
    }
  }

  if (initialForestCuts.length <= cutBudget) {
    return { firebreaks: initialForestCuts };
  }

  if (cutBudget === 0) {
    return { firebreaks: [] };
  }

  // Adaptive binary search iterations depending on problem size
  const maxIter = M < 5000 ? 25 : (M < 30000 ? 20 : 12);

  for (let iter = 0; iter < maxIter; iter++) {
    const mid = (L + R) / 2;
    dinic.reset(mid);
    dinic.maxFlow();

    // Recompute reachable set in residual graph
    visited_res.fill(0);
    q_res_head = 0;
    q_res_tail = 0;
    visited_res[S] = 1;
    queue_res[q_res_tail++] = S;
    
    while (q_res_head < q_res_tail) {
      const u = queue_res[q_res_head++];
      for (const edge of dinic.graph[u]) {
        if (edge.cap > EPS && !visited_res[edge.to]) {
          visited_res[edge.to] = 1;
          queue_res[q_res_tail++] = edge.to;
        }
      }
    }

    const currentCuts = [];
    for (let k = 0; k < M; k++) {
      const cell = activeCells[k];
      if (!ignitionSet.has(cell.idx) && !townMap.has(cell.idx)) {
        if (visited_res[2 * k] && !visited_res[2 * k + 1]) {
          currentCuts.push([cell.x, cell.y]);
        }
      }
    }

    let valSaved = 0;
    for (const t of activeTowns) {
      const idx = getIdx(t.x, t.y);
      const k = cellToIndex[idx];
      if (!visited_res[2 * k + 1]) {
        valSaved += t.value;
      }
    }

    if (currentCuts.length <= cutBudget) {
      if (valSaved > bestValSaved) {
        bestValSaved = valSaved;
        bestCuts = currentCuts;
      }
      L = mid;
    } else {
      R = mid;
    }
  }

  // 4. Greedy Post-processing
  // If we have unused budget, try to save the most valuable unsaved towns by isolating them locally.
  if (bestCuts.length < cutBudget) {
    const cutSet = new Set(bestCuts.map(([cx, cy]) => getIdx(cx, cy)));

    // BFS to find the burned zone under the current bestCuts
    const burned = new Uint8Array(width * height);
    const queue_b = new Int32Array(width * height);
    let q_b_head = 0;
    let q_b_tail = 0;
    
    for (const [ix, iy] of ignitions) {
      if (ix >= 0 && ix < width && iy >= 0 && iy < height) {
        const idx = getIdx(ix, iy);
        burned[idx] = 1;
        queue_b[q_b_tail++] = idx;
      }
    }
    
    while (q_b_head < q_b_tail) {
      const idx = queue_b[q_b_head++];
      const x = idx % width;
      const y = Math.floor(idx / width);
      for (const [dx, dy] of dirs) {
        const nx = x + dx;
        const ny = y + dy;
        if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
          const nidx = getIdx(nx, ny);
          if (grid[ny][nx] !== '~' && !cutSet.has(nidx) && !burned[nidx]) {
            burned[nidx] = 1;
            queue_b[q_b_tail++] = nidx;
          }
        }
      }
    }

    // Sort currently burned towns by value descending
    const unsavedTowns = activeTowns
      .filter(t => burned[getIdx(t.x, t.y)] === 1)
      .sort((a, b) => b.value - a.value);

    let bfsCount = 0;
    for (const t of unsavedTowns) {
      if (bestCuts.length >= cutBudget) break;

      const candidates = [];
      for (const [dx, dy] of dirs) {
        const nx = t.x + dx;
        const ny = t.y + dy;
        if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
          const nidx = getIdx(nx, ny);
          if (burned[nidx] && grid[ny][nx] === '.' && !ignitionSet.has(nidx) && !townMap.has(nidx) && !cutSet.has(nidx)) {
            candidates.push(nidx);
          }
        }
      }

      if (candidates.length > 0 && bestCuts.length + candidates.length <= cutBudget) {
        for (const c of candidates) {
          cutSet.add(c);
          bestCuts.push([c % width, Math.floor(c / width)]);
        }

        bfsCount++;
        // Limit BFS updates to 50 times to prevent performance bottlenecks on massive budgets
        if (bfsCount <= 50) {
          burned.fill(0);
          q_b_head = 0;
          q_b_tail = 0;
          for (const [ix, iy] of ignitions) {
            if (ix >= 0 && ix < width && iy >= 0 && iy < height) {
              const idx = getIdx(ix, iy);
              burned[idx] = 1;
              queue_b[q_b_tail++] = idx;
            }
          }
          while (q_b_head < q_b_tail) {
            const idx = queue_b[q_b_head++];
            const x = idx % width;
            const y = Math.floor(idx / width);
            for (const [dx, dy] of dirs) {
              const nx = x + dx;
              const ny = y + dy;
              if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                const nidx = getIdx(nx, ny);
                if (grid[ny][nx] !== '~' && !cutSet.has(nidx) && !burned[nidx]) {
                  burned[nidx] = 1;
                  queue_b[q_b_tail++] = nidx;
                }
              }
            }
          }
        }
      }
    }
  }

  return { firebreaks: bestCuts };
}

// Read JSON input from stdin and print JSON output to stdout
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);
const result = solve(data);
console.log(JSON.stringify(result));
