// Fast I/O
struct FastReader {
    vector<char> buf;
    int pos = 0;
    int len = 0;

    FastReader() {
        buf.resize(1 << 20);
    }

    char readChar() {
        if (pos >= len) {
            pos = 0;
            len = fread(buf.data(), 1, buf.size(), stdin);
            if (len == 0) return EOF;
        }
        return buf[pos++];
    }

    int readInt() {
        char c = readChar();
        while (c <= 32 && c != EOF) {
            c = readChar();
        }
        if (c == EOF) return -1;
        int res = 0;
        while (c > 32) {
            res = res * 10 + (c - '0');
            c = readChar();
        }
        return res;
    }
};

struct Graph {
    int num_vertices;
    int num_edges;
    vector<int> edge_to;
    vector<int> vertex_offset;
    vector<int> degrees;
};

// Global variables for thread synchronization
atomic<bool> found_solution(false);
atomic<int> global_best_colors_count(1e9);
atomic<long long> global_best_conflicts(1e18);
vector<int> global_best_coloring;
mutex global_best_mutex;

auto start_time = chrono::high_resolution_clock::now();

double elapsed_seconds() {
    auto current_time = chrono::high_resolution_clock::now();
    return chrono::duration<double>(current_time - start_time).count();
}

void update_best_coloring_conflict_free(const vector<int>& coloring, int color_count, int target_k) {
    lock_guard<mutex> lock(global_best_mutex);
    if (color_count < global_best_colors_count) {
        global_best_colors_count = color_count;
        if (color_count <= target_k) {
            global_best_coloring = coloring;
            global_best_conflicts = 0;
            found_solution = true;
        } else if (global_best_conflicts > 0) {
            global_best_coloring = coloring;
        }
    }
}

void update_best_coloring_conflicts(const vector<int>& coloring, long long conflicts, int target_k) {
    lock_guard<mutex> lock(global_best_mutex);
    if (conflicts < global_best_conflicts) {
        global_best_conflicts = conflicts;
        global_best_coloring = coloring;
        if (conflicts == 0) {
            global_best_colors_count = target_k;
            found_solution = true;
        }
    }
}

int get_num_colors_used(const vector<int>& colors) {
    int max_c = -1;
    for (int c : colors) {
        if (c > max_c) max_c = c;
    }
    return max_c + 1;
}

vector<int> get_sdl_order(const Graph& g) {
    int n = g.num_vertices;
    vector<int> deg = g.degrees;
    vector<int> order;
    order.reserve(n);

    vector<int> head(n, -1);
    vector<int> prev_node(n, -1);
    vector<int> next_node(n, -1);
    
    auto add_to_bucket = [&](int v, int d) {
        next_node[v] = head[d];
        prev_node[v] = -1;
        if (head[d] != -1) {
            prev_node[head[d]] = v;
        }
        head[d] = v;
    };
    
    auto remove_from_bucket = [&](int v, int d) {
        if (prev_node[v] != -1) {
            next_node[prev_node[v]] = next_node[v];
        } else {
            head[d] = next_node[v];
        }
        if (next_node[v] != -1) {
            prev_node[next_node[v]] = prev_node[v];
        }
    };

    for (int i = 0; i < n; ++i) {
        add_to_bucket(i, deg[i]);
    }

    vector<bool> removed(n, false);
    int min_deg = 0;

    for (int step = 0; step < n; ++step) {
        while (min_deg < n && head[min_deg] == -1) {
            min_deg++;
        }
        if (min_deg == n) break;

        int u = head[min_deg];
        remove_from_bucket(u, min_deg);
        removed[u] = true;
        order.push_back(u);

        int start = g.vertex_offset[u];
        int end = g.vertex_offset[u + 1];
        for (int idx = start; idx < end; ++idx) {
            int v = g.edge_to[idx];
            if (!removed[v]) {
                int d = deg[v];
                remove_from_bucket(v, d);
                deg[v]--;
                add_to_bucket(v, d - 1);
                if (d - 1 < min_deg) {
                    min_deg = d - 1;
                }
            }
        }
    }

    reverse(order.begin(), order.end());
    return order;
}

vector<int> get_ldf_order(const Graph& g) {
    int n = g.num_vertices;
    vector<int> order(n);
    for (int i = 0; i < n; ++i) order[i] = i;
    sort(order.begin(), order.end(), [&](int a, int b) {
        return g.degrees[a] > g.degrees[b];
    });
    return order;
}

vector<int> get_random_order(const Graph& g, uint32_t seed) {
    int n = g.num_vertices;
    vector<int> order(n);
    for (int i = 0; i < n; ++i) order[i] = i;
    mt19937 rng(seed);
    shuffle(order.begin(), order.end(), rng);
    return order;
}

vector<int> color_greedy(const Graph& g, const vector<int>& order) {
    int n = g.num_vertices;
    vector<int> colors(n, -1);
    vector<int> last_seen(n, -1);

    for (int i = 0; i < n; ++i) {
        int u = order[i];
        int start = g.vertex_offset[u];
        int end = g.vertex_offset[u + 1];
        for (int idx = start; idx < end; ++idx) {
            int v = g.edge_to[idx];
            if (colors[v] != -1) {
                last_seen[colors[v]] = i;
            }
        }
        int c = 0;
        while (last_seen[c] == i) {
            c++;
        }
        colors[u] = c;
    }
    return colors;
}

vector<int> iterative_greedy_step_flat(const Graph& g, const vector<int>& colors, int num_colors, uint32_t seed) {
    int n = g.num_vertices;
    
    vector<int> class_size(num_colors, 0);
    for (int c : colors) {
        class_size[c]++;
    }
    
    vector<int> class_offset(num_colors, 0);
    vector<int> class_order(num_colors);
    for (int i = 0; i < num_colors; ++i) class_order[i] = i;
    mt19937 rng(seed);
    shuffle(class_order.begin(), class_order.end(), rng);
    
    int current_offset = 0;
    for (int c : class_order) {
        class_offset[c] = current_offset;
        current_offset += class_size[c];
    }
    
    vector<int> order(n);
    vector<int> temp_offset = class_offset;
    for (int i = 0; i < n; ++i) {
        int c = colors[i];
        order[temp_offset[c]++] = i;
    }
    
    for (int c : class_order) {
        int start = class_offset[c];
        int end = start + class_size[c];
        shuffle(order.begin() + start, order.begin() + end, rng);
    }
    
    return color_greedy(g, order);
}

void ig_worker(const Graph& g, int target_k, uint32_t seed, int max_iterations) {
    vector<int> order = get_random_order(g, seed);
    vector<int> current_coloring = color_greedy(g, order);
    int current_count = get_num_colors_used(current_coloring);
    update_best_coloring_conflict_free(current_coloring, current_count, target_k);
    
    if (found_solution) return;
    
    mt19937 rng(seed);
    for (int iter = 0; iter < max_iterations; ++iter) {
        if (found_solution || elapsed_seconds() > 280.0) return;
        
        current_coloring = iterative_greedy_step_flat(g, current_coloring, current_count, rng());
        current_count = get_num_colors_used(current_coloring);
        update_best_coloring_conflict_free(current_coloring, current_count, target_k);
        
        if (found_solution) return;
    }
}

void tabucol_worker(const Graph& g, int k, const vector<int>& initial_coloring, uint32_t seed, int max_iterations, int thread_id) {
    int n = g.num_vertices;
    mt19937 rng(seed);
    
    vector<int> color(n);
    for (int i = 0; i < n; ++i) {
        if (initial_coloring[i] < k) {
            color[i] = initial_coloring[i];
        } else {
            int start = g.vertex_offset[i];
            int end = g.vertex_offset[i + 1];
            vector<int> freq(k, 0);
            for (int idx = start; idx < end; ++idx) {
                int v = g.edge_to[idx];
                if (initial_coloring[v] < k) {
                    freq[initial_coloring[v]]++;
                }
            }
            int best_c = 0;
            int min_f = freq[0];
            for (int c = 1; c < k; ++c) {
                if (freq[c] < min_f) {
                    min_f = freq[c];
                    best_c = c;
                }
            }
            color[i] = best_c;
        }
    }
    
    vector<int> adj_color_count((size_t)n * k, 0);
    for (int u = 0; u < n; ++u) {
        int start = g.vertex_offset[u];
        int end = g.vertex_offset[u + 1];
        for (int idx = start; idx < end; ++idx) {
            int v = g.edge_to[idx];
            adj_color_count[(size_t)v * k + color[u]]++;
        }
    }
    
    vector<int> conflict_vertices;
    vector<int> position_in_conflict_vertices(n, -1);
    
    auto add_conflict = [&](int u) {
        if (position_in_conflict_vertices[u] == -1) {
            position_in_conflict_vertices[u] = conflict_vertices.size();
            conflict_vertices.push_back(u);
        }
    };
    
    auto remove_conflict = [&](int u) {
        int pos = position_in_conflict_vertices[u];
        if (pos != -1) {
            int last_v = conflict_vertices.back();
            conflict_vertices[pos] = last_v;
            position_in_conflict_vertices[last_v] = pos;
            conflict_vertices.pop_back();
            position_in_conflict_vertices[u] = -1;
        }
    };
    
    long long total_conflicting_edges = 0;
    for (int u = 0; u < n; ++u) {
        int u_conf = adj_color_count[(size_t)u * k + color[u]];
        total_conflicting_edges += u_conf;
        if (u_conf > 0) {
            add_conflict(u);
        }
    }
    total_conflicting_edges /= 2;
    
    long long best_conflicts = total_conflicting_edges;
    vector<int> best_coloring_local = color;
    
    if (total_conflicting_edges == 0) {
        update_best_coloring_conflicts(color, 0, k);
        return;
    }
    
    vector<int> tabu((size_t)n * k, 0);
    double lambda = 0.4 + (rng() % 100) * 0.016;
    
    int max_sample = 50;
    if (thread_id % 3 == 0) max_sample = 20;
    else if (thread_id % 3 == 1) max_sample = 100;
    else max_sample = 500;
    
    for (int iter = 1; iter <= max_iterations; ++iter) {
        if (found_solution || elapsed_seconds() > 280.0) return;
        
        if (conflict_vertices.empty()) {
            update_best_coloring_conflicts(color, 0, k);
            return;
        }
        
        int best_move_u = -1;
        int best_move_new_c = -1;
        int best_move_delta = 1e9;
        bool best_move_tabu = true;
        
        int sample_size = min((int)conflict_vertices.size(), max_sample);
        for (int s = 0; s < sample_size; ++s) {
            int idx = rng() % conflict_vertices.size();
            int u = conflict_vertices[idx];
            int current_c = color[u];
            int current_conf = adj_color_count[(size_t)u * k + current_c];
            
            for (int new_c = 0; new_c < k; ++new_c) {
                if (new_c == current_c) continue;
                
                int delta = adj_color_count[(size_t)u * k + new_c] - current_conf;
                bool is_tabu = (tabu[(size_t)u * k + new_c] > iter);
                
                if (is_tabu && total_conflicting_edges + delta < best_conflicts) {
                    is_tabu = false;
                }
                
                if (best_move_u == -1) {
                    best_move_u = u;
                    best_move_new_c = new_c;
                    best_move_delta = delta;
                    best_move_tabu = is_tabu;
                } else {
                    if (is_tabu && !best_move_tabu) {
                        continue;
                    } else if (!is_tabu && best_move_tabu) {
                        best_move_u = u;
                        best_move_new_c = new_c;
                        best_move_delta = delta;
                        best_move_tabu = false;
                    } else {
                        if (delta < best_move_delta || (delta == best_move_delta && rng() % 2 == 0)) {
                            best_move_u = u;
                            best_move_new_c = new_c;
                            best_move_delta = delta;
                            best_move_tabu = is_tabu;
                        }
                    }
                }
            }
        }
        
        if (best_move_u == -1) continue;
        
        int u = best_move_u;
        int old_c = color[u];
        int new_c = best_move_new_c;
        
        int tabu_tenure = (rng() % 10) + (int)(lambda * conflict_vertices.size());
        tabu[(size_t)u * k + old_c] = iter + tabu_tenure;
        
        color[u] = new_c;
        total_conflicting_edges += best_move_delta;
        
        if (total_conflicting_edges < best_conflicts) {
            best_conflicts = total_conflicting_edges;
            best_coloring_local = color;
            update_best_coloring_conflicts(best_coloring_local, best_conflicts, k);
        }
        
        int start = g.vertex_offset[u];
        int end = g.vertex_offset[u + 1];
        for (int idx = start; idx < end; ++idx) {
            int v = g.edge_to[idx];
            
            int v_old_c_count = --adj_color_count[(size_t)v * k + old_c];
            int v_new_c_count = ++adj_color_count[(size_t)v * k + new_c];
            
            int v_c = color[v];
            if (v_c == old_c) {
                if (v_old_c_count == 0) remove_conflict(v);
            } else if (v_c == new_c) {
                if (v_new_c_count == 1) add_conflict(v);
            }
        }
        
        if (adj_color_count[(size_t)u * k + new_c] > 0) {
            add_conflict(u);
        } else {
            remove_conflict(u);
        }
        
        if (total_conflicting_edges == 0) {
            update_best_coloring_conflicts(color, 0, k);
            return;
        }
    }
}

int get_num_threads_for_tabu(int n, int k) {
    long long mem_per_thread = (long long)n * k * 8 + (long long)n * 8;
    long long max_allowed_mem = 64LL * 1024 * 1024 * 1024;
    int threads = thread::hardware_concurrency();
    if (threads == 0) threads = 4;
    
    if (mem_per_thread > 0) {
        int cap = max_allowed_mem / mem_per_thread;
        if (cap < threads) {
            threads = max(1, cap);
        }
    }
    threads = min(threads, 64);
    return threads;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    FastReader reader;
    int n = reader.readInt();
    if (n == -1) return 0;
    int m = reader.readInt();
    int k = reader.readInt();

    // Corner cases
    if (m == 0) {
        for (int i = 0; i < n; ++i) {
            cout << 0 << (i == n - 1 ? "" : " ");
        }
        cout << "\n";
        return 0;
    }

    if (k >= n) {
        for (int i = 0; i < n; ++i) {
            cout << i << (i == n - 1 ? "" : " ");
        }
        cout << "\n";
        return 0;
    }

    // Read edges
    vector<int> edges_u(m);
    vector<int> edges_v(m);
    vector<int> raw_degrees(n, 0);
    for (int i = 0; i < m; ++i) {
        int u = reader.readInt();
        if (u == -1) {
            m = i;
            edges_u.resize(i);
            edges_v.resize(i);
            break;
        }
        int v = reader.readInt();
        edges_u[i] = u;
        edges_v[i] = v;
        raw_degrees[u]++;
        raw_degrees[v]++;
    }

    // Build initial flat graph representation
    Graph g;
    g.num_vertices = n;
    g.num_edges = m;
    g.degrees = raw_degrees;
    g.vertex_offset.assign(n + 1, 0);
    for (int i = 0; i < n; ++i) {
        g.vertex_offset[i + 1] = g.vertex_offset[i] + raw_degrees[i];
    }
    g.edge_to.resize(2 * m);
    vector<int> temp_offset = g.vertex_offset;
    for (int i = 0; i < m; ++i) {
        int u = edges_u[i];
        int v = edges_v[i];
        g.edge_to[temp_offset[u]++] = v;
        g.edge_to[temp_offset[v]++] = u;
    }

    // Free memory of edge vectors
    vector<int>().swap(edges_u);
    vector<int>().swap(edges_v);
    vector<int>().swap(temp_offset);
    vector<int>().swap(raw_degrees);

    // Sort and deduplicate neighbors to remove self-loops and multi-edges
    vector<int> new_edge_to;
    new_edge_to.reserve(g.edge_to.size());
    vector<int> new_offsets(n + 1, 0);
    g.degrees.assign(n, 0);
    for (int u = 0; u < n; ++u) {
        int start = g.vertex_offset[u];
        int end = g.vertex_offset[u + 1];
        vector<int> neighbors;
        for (int idx = start; idx < end; ++idx) {
            int v = g.edge_to[idx];
            if (v != u) {
                neighbors.push_back(v);
            }
        }
        sort(neighbors.begin(), neighbors.end());
        neighbors.erase(unique(neighbors.begin(), neighbors.end()), neighbors.end());
        new_offsets[u] = new_edge_to.size();
        for (int v : neighbors) {
            new_edge_to.push_back(v);
        }
        g.degrees[u] = neighbors.size();
    }
    new_offsets[n] = new_edge_to.size();
    g.edge_to = move(new_edge_to);
    g.vertex_offset = move(new_offsets);
    g.num_edges = g.edge_to.size() / 2;

    // Phase 1: Fast Greedy Heuristics
    vector<int> sdl_order = get_sdl_order(g);
    vector<int> sdl_coloring = color_greedy(g, sdl_order);
    int sdl_colors = get_num_colors_used(sdl_coloring);
    update_best_coloring_conflict_free(sdl_coloring, sdl_colors, k);

    if (found_solution) {
        for (int i = 0; i < n; ++i) {
            cout << global_best_coloring[i] << (i == n - 1 ? "" : " ");
        }
        cout << "\n";
        return 0;
    }

    vector<int> ldf_order = get_ldf_order(g);
    vector<int> ldf_coloring = color_greedy(g, ldf_order);
    int ldf_colors = get_num_colors_used(ldf_coloring);
    update_best_coloring_conflict_free(ldf_coloring, ldf_colors, k);

    if (found_solution) {
        for (int i = 0; i < n; ++i) {
            cout << global_best_coloring[i] << (i == n - 1 ? "" : " ");
        }
        cout << "\n";
        return 0;
    }

    // Phase 2: Parallel Iterative Greedy
    unsigned int hardware_threads = thread::hardware_concurrency();
    if (hardware_threads == 0) hardware_threads = 4;
    unsigned int ig_threads_count = min(hardware_threads, 64u);
    
    vector<uint32_t> seeds(ig_threads_count);
    mt19937 seed_rng(1337);
    for (unsigned int i = 0; i < ig_threads_count; ++i) seeds[i] = seed_rng();

    vector<thread> threads;
    threads.reserve(ig_threads_count);

    int max_ig_iterations = 500;
    if (n > 100000) max_ig_iterations = 100;
    if (n > 500000) max_ig_iterations = 30;

    for (unsigned int i = 0; i < ig_threads_count; ++i) {
        threads.emplace_back(ig_worker, ref(g), k, seeds[i], max_ig_iterations);
    }
    for (auto& t : threads) {
        if (t.joinable()) t.join();
    }
    threads.clear();

    if (found_solution) {
        for (int i = 0; i < n; ++i) {
            cout << global_best_coloring[i] << (i == n - 1 ? "" : " ");
        }
        cout << "\n";
        return 0;
    }

    // Phase 3: Parallel Tabu Search (TABUCOL)
    int tabu_threads_count = get_num_threads_for_tabu(n, k);
    seeds.resize(tabu_threads_count);
    for (int i = 0; i < tabu_threads_count; ++i) seeds[i] = seed_rng();

    vector<int> initial_coloring_for_tabu;
    {
        lock_guard<mutex> lock(global_best_mutex);
        initial_coloring_for_tabu = global_best_coloring;
    }

    int max_tabu_iterations = 1000000;
    if (n > 100000) max_tabu_iterations = 100000;
    if (n > 500000) max_tabu_iterations = 20000;

    for (int i = 0; i < tabu_threads_count; ++i) {
        threads.emplace_back(tabucol_worker, ref(g), k, ref(initial_coloring_for_tabu), seeds[i], max_tabu_iterations, i);
    }
    for (auto& t : threads) {
        if (t.joinable()) t.join();
    }

    for (int i = 0; i < n; ++i) {
        cout << global_best_coloring[i] << (i == n - 1 ? "" : " ");
    }
    cout << "\n";

    return 0;
}
