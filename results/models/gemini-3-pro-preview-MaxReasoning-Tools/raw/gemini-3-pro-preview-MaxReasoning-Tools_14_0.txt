using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Linq;
using System.Threading.Tasks;

class Program
{
    static int[] varMap;
    static int[] eqMap;
    static List<int> modifiedVars = new List<int>();
    static List<int> modifiedEqs = new List<int>();

    // Preallocated static arrays for sequential methods to avoid GC allocations
    static int[] unknownNeighbors = new int[8];
    static int[] checkedB = new int[24];
    static int[] SA = new int[8];
    static int[] SB = new int[8];
    static int[] diff = new int[8];

    static void Fill<T>(T[] array, T value)
    {
        for (int i = 0; i < array.Length; i++)
        {
            array[i] = value;
        }
    }

    static void Main(string[] args)
    {
        string firstLine = Console.ReadLine();
        if (string.IsNullOrEmpty(firstLine)) return;
        
        string[] parts = firstLine.Trim().Split(new char[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        if (parts.Length < 2) return;
        
        int H = int.Parse(parts[0]);
        int W = int.Parse(parts[1]);
        int N = H * W;
        
        varMap = new int[N];
        eqMap = new int[N];
        Fill(varMap, -1);
        Fill(eqMap, -1);
        
        byte[] board = new byte[N];
        
        for (int r = 0; r < H; r++)
        {
            string line = Console.ReadLine();
            if (line == null) break;
            int offset = r * W;
            for (int c = 0; c < W; c++)
            {
                char ch = line[c];
                if (ch == ' ') board[offset + c] = 9;
                else if (ch == '*') board[offset + c] = 10;
                else board[offset + c] = (byte)(ch - '0');
            }
        }
        
        Stopwatch sw = Stopwatch.StartNew();
        
        int[] queue = new int[N];
        bool[] inQueue = new bool[N];
        int head = 0;
        int tail = 0;
        
        for (int i = 0; i < N; i++)
        {
            if (board[i] <= 8 && HasUnknownNeighbors(board, i, H, W))
            {
                queue[tail++] = i;
                inQueue[i] = true;
            }
        }
        
        bool globalProgress = true;
        
        while (globalProgress)
        {
            globalProgress = false;
            
            if (sw.ElapsedMilliseconds > 4500)
            {
                break;
            }
            
            bool singleCellProgress = RunSingleCellPropagation(board, H, W, queue, inQueue, ref head, ref tail);
            if (singleCellProgress)
            {
                globalProgress = true;
            }
            
            if (head == tail)
            {
                bool subsetProgress = RunSubsetSolver(board, H, W, queue, inQueue, ref tail);
                if (subsetProgress)
                {
                    globalProgress = true;
                    continue;
                }
            }
            
            if (head == tail)
            {
                int maxGaussianSize = 800;
                if (N > 1000000)
                {
                    maxGaussianSize = 300;
                }
                if (sw.ElapsedMilliseconds > 3000)
                {
                    maxGaussianSize = 100;
                }
                if (sw.ElapsedMilliseconds > 4000)
                {
                    maxGaussianSize = 0;
                }
                
                if (maxGaussianSize > 0)
                {
                    bool cspProgress = RunCspSolver(board, H, W, queue, inQueue, ref tail, maxGaussianSize);
                    if (cspProgress)
                    {
                        globalProgress = true;
                    }
                }
            }
        }
        
        List<int> safeCells = new List<int>();
        for (int i = 0; i < N; i++)
        {
            if (board[i] == 11)
            {
                safeCells.Add(i);
            }
        }
        
        Console.WriteLine(safeCells.Count);
        foreach (int idx in safeCells)
        {
            int r = idx / W;
            int c = idx % W;
            Console.WriteLine($"{c} {r}");
        }
    }

    static bool HasUnknownNeighbors(byte[] board, int idx, int H, int W)
    {
        int r = idx / W;
        int c = idx % W;
        for (int dr = -1; dr <= 1; dr++)
        {
            int nr = r + dr;
            if (nr < 0 || nr >= H) continue;
            int rowOffset = nr * W;
            for (int dc = -1; dc <= 1; dc++)
            {
                if (dr == 0 && dc == 0) continue;
                int nc = c + dc;
                if (nc < 0 || nc >= W) continue;
                if (board[rowOffset + nc] == 9) return true;
            }
        }
        return false;
    }

    static void QueueAdjacentNumbered(
        byte[] board,
        int idx,
        int H,
        int W,
        int[] queue,
        bool[] inQueue,
        ref int tail)
    {
        int r = idx / W;
        int c = idx % W;
        for (int dr = -1; dr <= 1; dr++)
        {
            int nr = r + dr;
            if (nr < 0 || nr >= H) continue;
            int rowOffset = nr * W;
            for (int dc = -1; dc <= 1; dc++)
            {
                if (dr == 0 && dc == 0) continue;
                int nc = c + dc;
                if (nc < 0 || nc >= W) continue;
                
                int nidx = rowOffset + nc;
                if (board[nidx] <= 8 && !inQueue[nidx])
                {
                    queue[tail++] = nidx;
                    inQueue[nidx] = true;
                }
            }
        }
    }

    static bool RunSingleCellPropagation(
        byte[] board,
        int H,
        int W,
        int[] queue,
        bool[] inQueue,
        ref int head,
        ref int tail)
    {
        bool progress = false;
        
        while (head < tail)
        {
            int idx = queue[head++];
            inQueue[idx] = false;
            
            byte val = board[idx];
            if (val > 8) continue;
            
            int r = idx / W;
            int c = idx % W;
            
            int mines = 0;
            int unknowns = 0;
            
            for (int dr = -1; dr <= 1; dr++)
            {
                int nr = r + dr;
                if (nr < 0 || nr >= H) continue;
                int rowOffset = nr * W;
                for (int dc = -1; dc <= 1; dc++)
                {
                    if (dr == 0 && dc == 0) continue;
                    int nc = c + dc;
                    if (nc < 0 || nc >= W) continue;
                    
                    int nidx = rowOffset + nc;
                    byte nval = board[nidx];
                    if (nval == 10)
                    {
                        mines++;
                    }
                    else if (nval == 9)
                    {
                        unknownNeighbors[unknowns++] = nidx;
                    }
                }
            }
            
            if (unknowns == 0) continue;
            
            if (mines == val)
            {
                for (int i = 0; i < unknowns; i++)
                {
                    int nidx = unknownNeighbors[i];
                    board[nidx] = 11;
                    QueueAdjacentNumbered(board, nidx, H, W, queue, inQueue, ref tail);
                }
                progress = true;
            }
            else if (mines + unknowns == val)
            {
                for (int i = 0; i < unknowns; i++)
                {
                    int nidx = unknownNeighbors[i];
                    board[nidx] = 10;
                    QueueAdjacentNumbered(board, nidx, H, W, queue, inQueue, ref tail);
                }
                progress = true;
            }
        }
        
        return progress;
    }

    static bool RunSubsetSolver(
        byte[] board,
        int H,
        int W,
        int[] queue,
        bool[] inQueue,
        ref int tail)
    {
        List<int> activeNumbered = new List<int>();
        for (int i = 0; i < H * W; i++)
        {
            if (board[i] <= 8 && HasUnknownNeighbors(board, i, H, W))
            {
                activeNumbered.Add(i);
            }
        }
        
        bool progress = false;
        
        foreach (int idxA in activeNumbered)
        {
            int rA = idxA / W;
            int cA = idxA % W;
            
            int countA = 0;
            int minesA = 0;
            
            for (int dr = -1; dr <= 1; dr++)
            {
                int nr = rA + dr;
                if (nr < 0 || nr >= H) continue;
                int rowOffset = nr * W;
                for (int dc = -1; dc <= 1; dc++)
                {
                    if (dr == 0 && dc == 0) continue;
                    int nc = cA + dc;
                    if (nc < 0 || nc >= W) continue;
                    
                    int nidx = rowOffset + nc;
                    byte nval = board[nidx];
                    if (nval == 9)
                    {
                        SA[countA++] = nidx;
                    }
                    else if (nval == 10)
                    {
                        minesA++;
                    }
                }
            }
            
            if (countA == 0) continue;
            int remA = board[idxA] - minesA;
            
            int checkedCount = 0;
            
            for (int uIdx = 0; uIdx < countA; uIdx++)
            {
                int u = SA[uIdx];
                int ru = u / W;
                int cu = u % W;
                
                for (int dr = -1; dr <= 1; dr++)
                {
                    int nru = ru + dr;
                    if (nru < 0 || nru >= H) continue;
                    int rowOffset = nru * W;
                    for (int dc = -1; dc <= 1; dc++)
                    {
                        if (dr == 0 && dc == 0) continue;
                        int ncu = cu + dc;
                        if (ncu < 0 || ncu >= W) continue;
                        
                        int idxB = rowOffset + ncu;
                        if (idxB == idxA) continue;
                        if (board[idxB] > 8) continue;
                        
                        bool alreadyChecked = false;
                        for (int k = 0; k < checkedCount; k++)
                        {
                            if (checkedB[k] == idxB)
                            {
                                alreadyChecked = true;
                                break;
                            }
                        }
                        if (alreadyChecked) continue;
                        checkedB[checkedCount++] = idxB;
                        
                        int rB = idxB / W;
                        int cB = idxB % W;
                        int countB = 0;
                        int minesB = 0;
                        
                        for (int dr2 = -1; dr2 <= 1; dr2++)
                        {
                            int nr2 = rB + dr2;
                            if (nr2 < 0 || nr2 >= H) continue;
                            int rowOffset2 = nr2 * W;
                            for (int dc2 = -1; dc2 <= 1; dc2++)
                            {
                                if (dr2 == 0 && dc2 == 0) continue;
                                int nc2 = cB + dc2;
                                if (nc2 < 0 || nc2 >= W) continue;
                                
                                int nidx2 = rowOffset2 + nc2;
                                byte nval2 = board[nidx2];
                                if (nval2 == 9)
                                {
                                    SB[countB++] = nidx2;
                                }
                                else if (nval2 == 10)
                                {
                                    minesB++;
                                }
                            }
                        }
                        
                        if (countB == 0) continue;
                        int remB = board[idxB] - minesB;
                        
                        if (IsSubset(SA, countA, SB, countB))
                        {
                            int diffCount = GetDifference(SA, countA, SB, countB, diff);
                            if (diffCount > 0)
                            {
                                if (remB - remA == 0)
                                {
                                    for (int d = 0; d < diffCount; d++)
                                    {
                                        int nidx = diff[d];
                                        board[nidx] = 11;
                                        QueueAdjacentNumbered(board, nidx, H, W, queue, inQueue, ref tail);
                                    }
                                    progress = true;
                                }
                                else if (remB - remA == diffCount)
                                {
                                    for (int d = 0; d < diffCount; d++)
                                    {
                                        int nidx = diff[d];
                                        board[nidx] = 10;
                                        QueueAdjacentNumbered(board, nidx, H, W, queue, inQueue, ref tail);
                                    }
                                    progress = true;
                                }
                            }
                        }
                    }
                }
            }
            
            if (progress)
            {
                break;
            }
        }
        
        return progress;
    }

    static bool IsSubset(int[] A, int countA, int[] B, int countB)
    {
        if (countA > countB) return false;
        int i = 0, j = 0;
        while (i < countA && j < countB)
        {
            if (A[i] == B[j])
            {
                i++;
                j++;
            }
            else if (A[i] > B[j])
            {
                j++;
            }
            else
            {
                return false;
            }
        }
        return i == countA;
    }

    static int GetDifference(int[] A, int countA, int[] B, int countB, int[] diffOut)
    {
        int i = 0, j = 0;
        int diffCount = 0;
        while (j < countB)
        {
            if (i < countA && A[i] == B[j])
            {
                i++;
                j++;
            }
            else
            {
                diffOut[diffCount++] = B[j++];
            }
        }
        return diffCount;
    }

    class Component
    {
        public List<int> Equations;
        public List<int> Variables;
    }

    class LocalEquation
    {
        public int[] VarIndices;
        public int Target;
    }

    struct Deduction
    {
        public int FrontierVarId;
        public byte Value;
    }

    static bool RunCspSolver(
        byte[] board,
        int H,
        int W,
        int[] queue,
        bool[] inQueue,
        ref int tail,
        int maxGaussianSize)
    {
        foreach (int idx in modifiedVars) varMap[idx] = -1;
        modifiedVars.Clear();
        foreach (int idx in modifiedEqs) eqMap[idx] = -1;
        modifiedEqs.Clear();
        
        List<int> activeNumbered = new List<int>();
        List<int> frontier = new List<int>();
        
        for (int i = 0; i < H * W; i++)
        {
            if (board[i] <= 8 && HasUnknownNeighbors(board, i, H, W))
            {
                eqMap[i] = activeNumbered.Count;
                modifiedEqs.Add(i);
                activeNumbered.Add(i);
            }
        }
        
        if (activeNumbered.Count == 0) return false;
        
        foreach (int idx in activeNumbered)
        {
            int r = idx / W;
            int c = idx % W;
            for (int dr = -1; dr <= 1; dr++)
            {
                int nr = r + dr;
                if (nr < 0 || nr >= H) continue;
                int rowOffset = nr * W;
                for (int dc = -1; dc <= 1; dc++)
                {
                    if (dr == 0 && dc == 0) continue;
                    int nc = c + dc;
                    if (nc < 0 || nc >= W) continue;
                    
                    int nidx = rowOffset + nc;
                    if (board[nidx] == 9 && varMap[nidx] == -1)
                    {
                        varMap[nidx] = frontier.Count;
                        modifiedVars.Add(nidx);
                        frontier.Add(nidx);
                    }
                }
            }
        }
        
        if (frontier.Count == 0) return false;
        
        bool[] visitedEq = new bool[activeNumbered.Count];
        bool[] visitedVar = new bool[frontier.Count];
        
        List<Component> components = new List<Component>();
        
        for (int startEq = 0; startEq < activeNumbered.Count; startEq++)
        {
            if (visitedEq[startEq]) continue;
            
            List<int> compEqs = new List<int>();
            List<int> compVars = new List<int>();
            
            Queue<int> qEq = new Queue<int>();
            qEq.Enqueue(startEq);
            visitedEq[startEq] = true;
            
            while (qEq.Count > 0)
            {
                int eqId = qEq.Dequeue();
                compEqs.Add(eqId);
                
                int idx = activeNumbered[eqId];
                int r = idx / W;
                int c = idx % W;
                for (int dr = -1; dr <= 1; dr++)
                {
                    int nr = r + dr;
                    if (nr < 0 || nr >= H) continue;
                    int rowOffset = nr * W;
                    for (int dc = -1; dc <= 1; dc++)
                    {
                        if (dr == 0 && dc == 0) continue;
                        int nc = c + dc;
                        if (nc < 0 || nc >= W) continue;
                        
                        int nidx = rowOffset + nc;
                        if (board[nidx] == 9)
                        {
                            int varId = varMap[nidx];
                            if (varId != -1 && !visitedVar[varId])
                            {
                                visitedVar[varId] = true;
                                compVars.Add(varId);
                                QueueEquationsForVariable(board, nidx, H, W, eqMap, visitedEq, qEq);
                            }
                        }
                    }
                }
            }
            
            components.Add(new Component { Equations = compEqs, Variables = compVars });
        }
        
        var deductions = new System.Collections.Concurrent.ConcurrentBag<Deduction>();
        
        Parallel.ForEach(components, comp =>
        {
            int V = comp.Variables.Count;
            if (V == 0) return;
            if (V > maxGaussianSize) return;
            
            LocalEquation[] localEqs = new LocalEquation[comp.Equations.Count];
            for (int i = 0; i < comp.Equations.Count; i++)
            {
                int eqId = comp.Equations[i];
                int idx = activeNumbered[eqId];
                
                int r = idx / W;
                int c = idx % W;
                int mines = 0;
                List<int> localVars = new List<int>();
                
                for (int dr = -1; dr <= 1; dr++)
                {
                    int nr = r + dr;
                    if (nr < 0 || nr >= H) continue;
                    int rowOffset = nr * W;
                    for (int dc = -1; dc <= 1; dc++)
                    {
                        if (dr == 0 && dc == 0) continue;
                        int nc = c + dc;
                        if (nc < 0 || nc >= W) continue;
                        
                        int nidx = rowOffset + nc;
                        byte nval = board[nidx];
                        if (nval == 10)
                        {
                            mines++;
                        }
                        else if (nval == 9)
                        {
                            int varId = varMap[nidx];
                            if (varId != -1)
                            {
                                int localIdx = -1;
                                for (int k = 0; k < V; k++)
                                {
                                    if (comp.Variables[k] == varId)
                                    {
                                        localIdx = k;
                                        break;
                                    }
                                }
                                if (localIdx != -1)
                                {
                                    localVars.Add(localIdx);
                                }
                            }
                        }
                    }
                }
                
                localEqs[i] = new LocalEquation
                {
                    VarIndices = localVars.ToArray(),
                    Target = board[idx] - mines
                };
            }
            
            if (V <= 20)
            {
                List<int>[] varToEqs = new List<int>[V];
                for (int i = 0; i < V; i++) varToEqs[i] = new List<int>();
                for (int i = 0; i < localEqs.Length; i++)
                {
                    foreach (int v in localEqs[i].VarIndices)
                    {
                        varToEqs[v].Add(i);
                    }
                }
                
                int[] varOrder = Enumerable.Range(0, V).OrderByDescending(i => varToEqs[i].Count).ToArray();
                int[] assignment = new int[V];
                Fill(assignment, -1);
                int[] eqSums = new int[localEqs.Length];
                int[] eqUnassigned = new int[localEqs.Length];
                for (int i = 0; i < localEqs.Length; i++)
                {
                    eqUnassigned[i] = localEqs[i].VarIndices.Length;
                }
                
                int solutionCount = 0;
                int[] varOneCount = new int[V];
                
                SolveBacktrack(0, V, varOrder, localEqs, varToEqs, assignment, eqSums, eqUnassigned, ref solutionCount, varOneCount);
                
                if (solutionCount > 0)
                {
                    for (int i = 0; i < V; i++)
                    {
                        if (varOneCount[i] == 0)
                        {
                            deductions.Add(new Deduction { FrontierVarId = comp.Variables[i], Value = 11 });
                        }
                        else if (varOneCount[i] == solutionCount)
                        {
                            deductions.Add(new Deduction { FrontierVarId = comp.Variables[i], Value = 10 });
                        }
                    }
                }
            }
            else
            {
                int[] solution = new int[V];
                Fill(solution, -1);
                SolveGaussian(V, localEqs, solution);
                
                for (int i = 0; i < V; i++)
                {
                    if (solution[i] == 0)
                    {
                        deductions.Add(new Deduction { FrontierVarId = comp.Variables[i], Value = 11 });
                    }
                    else if (solution[i] == 1)
                    {
                        deductions.Add(new Deduction { FrontierVarId = comp.Variables[i], Value = 10 });
                    }
                }
            }
        });
        
        bool progress = false;
        foreach (var ded in deductions)
        {
            int gIdx = frontier[ded.FrontierVarId];
            if (board[gIdx] == 9)
            {
                board[gIdx] = ded.Value;
                QueueAdjacentNumbered(board, gIdx, H, W, queue, inQueue, ref tail);
                progress = true;
            }
        }
        
        return progress;
    }

    static void QueueEquationsForVariable(
        byte[] board,
        int idx,
        int H,
        int W,
        int[] eqMap,
        bool[] visitedEq,
        Queue<int> qEq)
    {
        int r = idx / W;
        int c = idx % W;
        for (int dr = -1; dr <= 1; dr++)
        {
            int nr = r + dr;
            if (nr < 0 || nr >= H) continue;
            int rowOffset = nr * W;
            for (int dc = -1; dc <= 1; dc++)
            {
                if (dr == 0 && dc == 0) continue;
                int nc = c + dc;
                if (nc < 0 || nc >= W) continue;
                
                int nidx = rowOffset + nc;
                if (board[nidx] <= 8)
                {
                    int eqId = eqMap[nidx];
                    if (eqId != -1 && !visitedEq[eqId])
                    {
                        visitedEq[eqId] = true;
                        qEq.Enqueue(eqId);
                    }
                }
            }
        }
    }

    static void SolveBacktrack(
        int step,
        int V,
        int[] varOrder,
        LocalEquation[] eqs,
        List<int>[] varToEqs,
        int[] assignment,
        int[] eqSums,
        int[] eqUnassigned,
        ref int solutionCount,
        int[] varOneCount)
    {
        if (solutionCount > 10000) return;

        if (step == V)
        {
            solutionCount++;
            for (int i = 0; i < V; i++)
            {
                if (assignment[i] == 1)
                {
                    varOneCount[i]++;
                }
            }
            return;
        }
        
        int varIdx = varOrder[step];
        
        for (int val = 0; val <= 1; val++)
        {
            assignment[varIdx] = val;
            bool ok = true;
            
            var list = varToEqs[varIdx];
            for (int i = 0; i < list.Count; i++)
            {
                int eqId = list[i];
                eqSums[eqId] += val;
                eqUnassigned[eqId]--;
                
                int sum = eqSums[eqId];
                int target = eqs[eqId].Target;
                int unassigned = eqUnassigned[eqId];
                
                if (sum > target || sum + unassigned < target)
                {
                    ok = false;
                }
            }
            
            if (ok)
            {
                SolveBacktrack(step + 1, V, varOrder, eqs, varToEqs, assignment, eqSums, eqUnassigned, ref solutionCount, varOneCount);
            }
            
            for (int i = 0; i < list.Count; i++)
            {
                int eqId = list[i];
                eqSums[eqId] -= val;
                eqUnassigned[eqId]++;
            }
        }
        assignment[varIdx] = -1;
    }

    static void SolveGaussian(int V, LocalEquation[] eqs, int[] solution)
    {
        int rows = eqs.Length;
        int cols = V + 1;
        double[,] matrix = new double[rows, cols];
        
        bool progress = true;
        while (progress)
        {
            progress = false;
            
            for (int r = 0; r < rows; r++)
            {
                for (int c = 0; c < cols; c++)
                {
                    matrix[r, c] = 0.0;
                }
            }
            
            for (int r = 0; r < rows; r++)
            {
                double rhs = eqs[r].Target;
                foreach (int v in eqs[r].VarIndices)
                {
                    if (solution[v] == 1)
                    {
                        rhs -= 1.0;
                    }
                    else if (solution[v] == -1)
                    {
                        matrix[r, v] = 1.0;
                    }
                }
                matrix[r, V] = rhs;
            }
            
            ComputeRREF(matrix, rows, cols);
            
            for (int r = 0; r < rows; r++)
            {
                double P = 0.0;
                double N = 0.0;
                double rhs = matrix[r, V];
                
                bool allZeros = true;
                for (int c = 0; c < V; c++)
                {
                    double val = matrix[r, c];
                    if (Math.Abs(val) > 1e-9)
                    {
                        allZeros = false;
                        if (val > 0) P += val;
                        else N += val;
                    }
                }
                
                if (allZeros) continue;
                
                if (Math.Abs(P - rhs) < 1e-9)
                {
                    for (int c = 0; c < V; c++)
                    {
                        double val = matrix[r, c];
                        if (Math.Abs(val) > 1e-9)
                        {
                            int ded = val > 0 ? 1 : 0;
                            if (solution[c] == -1)
                            {
                                solution[c] = ded;
                                progress = true;
                            }
                        }
                    }
                }
                else if (Math.Abs(N - rhs) < 1e-9)
                {
                    for (int c = 0; c < V; c++)
                    {
                        double val = matrix[r, c];
                        if (Math.Abs(val) > 1e-9)
                        {
                            int ded = val > 0 ? 0 : 1;
                            if (solution[c] == -1)
                            {
                                solution[c] = ded;
                                progress = true;
                            }
                        }
                    }
                }
            }
        }
    }

    static void ComputeRREF(double[,] matrix, int rows, int cols)
    {
        int lead = 0;
        for (int r = 0; r < rows; r++)
        {
            if (lead >= cols - 1)
                break;
                
            int i = r;
            while (Math.Abs(matrix[i, lead]) < 1e-9)
            {
                i++;
                if (i == rows)
                {
                    i = r;
                    lead++;
                    if (lead == cols - 1)
                        return;
                }
            }
            
            if (i != r)
            {
                for (int c = 0; c < cols; c++)
                {
                    double temp = matrix[r, c];
                    matrix[r, c] = matrix[i, c];
                    matrix[i, c] = temp;
                }
            }
            
            double val = matrix[r, lead];
            if (Math.Abs(val) > 1e-9)
            {
                for (int c = 0; c < cols; c++)
                {
                    matrix[r, c] /= val;
                }
            }
            
            for (int r2 = 0; r2 < rows; r2++)
            {
                if (r2 != r)
                {
                    double factor = matrix[r2, lead];
                    for (int c = 0; c < cols; c++)
                    {
                        matrix[r2, c] -= factor * matrix[r, c];
                    }
                }
            }
            
            lead++;
        }
    }
