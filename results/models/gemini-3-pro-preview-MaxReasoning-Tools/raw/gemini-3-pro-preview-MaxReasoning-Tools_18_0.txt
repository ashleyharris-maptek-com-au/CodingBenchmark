#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <random>
#include <chrono>
#include <cassert>
#include <queue>

// Fast I/O
struct FastReader {
    inline int read_int() {
        int res;
        if (std::cin >> res) return res;
        return -1;
    }
};

struct DSU {
    std::vector<int> parent;
    DSU(int n) {
        parent.resize(n);
        std::iota(parent.begin(), parent.end(), 0);
    }
    int find(int i) {
        if (parent[i] == i)
            return i;
        return parent[i] = find(parent[i]);
    }
    bool unite(int i, int j) {
        int root_i = find(i);
        int root_j = find(j);
        if (root_i != root_j) {
            parent[root_j] = root_i;
            return true;
        }
        return false;
    }
};

struct Edge {
    int to;
    int weight; // For DFS, weight can be the edge ID
    
    bool operator<(const Edge& other) const {
        return to < other.to;
    }
};

struct SW_Edge {
    int to;
    int weight;
};

// Function to clean adjacency list of node i
void clean(int i, DSU& dsu, std::vector<std::vector<Edge>>& adj, std::vector<int>& deg) {
    auto& list = adj[i];
    int new_deg = 0;
    std::vector<Edge> cleaned;
    cleaned.reserve(list.size());
    for (auto& edge : list) {
        int r = dsu.find(edge.to);
        if (r != i) {
            cleaned.push_back({r, edge.weight});
        }
    }
    std::sort(cleaned.begin(), cleaned.end());
    list.clear();
    for (auto& edge : cleaned) {
        if (!list.empty() && list.back().to == edge.to) {
            list.back().weight += edge.weight;
        } else {
            list.push_back(edge);
        }
        new_deg += edge.weight;
    }
    deg[i] = new_deg;
}

void contract(int u, int v, DSU& dsu, std::vector<std::vector<Edge>>& adj, std::vector<int>& deg, std::vector<bool>& active) {
    int r_u = dsu.find(u);
    int r_v = dsu.find(v);
    if (r_u == r_v) return;
    
    dsu.parent[r_v] = r_u;
    active[r_v] = false;
    
    adj[r_u].insert(adj[r_u].end(), adj[r_v].begin(), adj[r_v].end());
    adj[r_v].clear();
    adj[r_v].shrink_to_fit();
    
    clean(r_u, dsu, adj, deg);
}

void run_stoer_wagner(
    int nh,
    const std::vector<int>& h_nodes,
    std::vector<std::vector<SW_Edge>>& h_adj,
    int& min_cut_val,
    std::vector<bool>& in_S
) {
    struct LocalDSU {
        std::vector<int> parent;
        LocalDSU(int n) {
            parent.resize(n);
            std::iota(parent.begin(), parent.end(), 0);
        }
        int find(int i) {
            if (parent[i] == i) return i;
            return parent[i] = find(parent[i]);
        }
        void unite(int i, int j) {
            int root_i = find(i);
            int root_j = find(j);
            if (root_i != root_j) {
                parent[root_j] = root_i;
            }
        }
    };
    
    LocalDSU sw_dsu(nh);
    std::vector<int> sw_dsu_best_parent;
    int best_t = -1;
    
    std::vector<bool> active(nh, true);
    int active_count = nh;
    
    std::vector<std::vector<SW_Edge>> cur_adj = h_adj;
    
    for (int phase = 0; phase < nh - 1; ++phase) {
        std::vector<int> weight(nh, 0);
        std::vector<bool> in_A(nh, false);
        
        int last = -1;
        int prev_last = -1;
        
        int start_node = -1;
        for (int i = 0; i < nh; ++i) {
            if (active[i]) {
                start_node = i;
                break;
            }
        }
        
        in_A[start_node] = true;
        last = start_node;
        for (auto& edge : cur_adj[start_node]) {
            if (active[edge.to]) {
                weight[edge.to] += edge.weight;
            }
        }
        
        for (int step = 1; step < active_count; ++step) {
            int max_w = -1;
            int next_node = -1;
            for (int i = 0; i < nh; ++i) {
                if (active[i] && !in_A[i]) {
                    if (weight[i] > max_w) {
                        max_w = weight[i];
                        next_node = i;
                    }
                }
            }
            
            in_A[next_node] = true;
            prev_last = last;
            last = next_node;
            
            for (auto& edge : cur_adj[next_node]) {
                if (active[edge.to] && !in_A[edge.to]) {
                    weight[edge.to] += edge.weight;
                }
            }
        }
        
        int cut_val = 0;
        for (auto& edge : cur_adj[last]) {
            if (active[edge.to]) {
                cut_val += edge.weight;
            }
        }
        
        if (cut_val < min_cut_val) {
            min_cut_val = cut_val;
            best_t = last;
            sw_dsu_best_parent = sw_dsu.parent;
        }
        
        sw_dsu.unite(prev_last, last);
        active[last] = false;
        active_count--;
        
        cur_adj[prev_last].insert(cur_adj[prev_last].end(), cur_adj[last].begin(), cur_adj[last].end());
        cur_adj[last].clear();
        
        auto& list = cur_adj[prev_last];
        std::vector<SW_Edge> cleaned;
        cleaned.reserve(list.size());
        for (auto& edge : list) {
            int r = sw_dsu.find(edge.to);
            if (r != prev_last) {
                cleaned.push_back({r, edge.weight});
            }
        }
        std::sort(cleaned.begin(), cleaned.end(), [](const SW_Edge& a, const SW_Edge& b){
            return a.to < b.to;
        });
        list.clear();
        for (auto& edge : cleaned) {
            if (!list.empty() && list.back().to == edge.to) {
                list.back().weight += edge.weight;
            } else {
                list.push_back(edge);
            }
        }
    }
    
    in_S.assign(nh, false);
    if (best_t != -1) {
        LocalDSU temp_dsu(nh);
        temp_dsu.parent = sw_dsu_best_parent;
        int root_t = temp_dsu.find(best_t);
        for (int i = 0; i < nh; ++i) {
            if (temp_dsu.find(i) == root_t) {
                in_S[i] = true;
            }
        }
    } else {
        in_S.clear();
    }
}

int main() {
    // Basic standard setup
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);
    
    FastReader reader;
    int n = reader.read_int();
    int m = reader.read_int();
    if (n <= 0) return 0;
    
    std::vector<int> edge_u(m);
    std::vector<int> edge_v(m);
    std::vector<int> original_degrees(n, 0);
    
    // Forest packing DSUs
    int K = (int)(2LL * m / n);
    if (K < 3) K = 3;
    if (K > 30) K = 30;
    
    std::vector<DSU> forest_dsu;
    forest_dsu.reserve(K);
    for (int i = 0; i < K; ++i) {
        forest_dsu.emplace_back(n);
    }
    
    struct PackedEdge {
        int u, v, id;
    };
    std::vector<PackedEdge> packed_edges;
    packed_edges.reserve(K * n);
    
    for (int i = 0; i < m; ++i) {
        int u = reader.read_int();
        int v = reader.read_int();
        edge_u[i] = u;
        edge_v[i] = v;
        original_degrees[u]++;
        original_degrees[v]++;
        
        for (int j = 0; j < K; ++j) {
            if (forest_dsu[j].find(u) != forest_dsu[j].find(v)) {
                forest_dsu[j].unite(u, v);
                packed_edges.push_back({u, v, i});
                break;
            }
        }
    }
    
    // Clear forest_dsu as it is no longer needed
    forest_dsu.clear();
    forest_dsu.shrink_to_fit();
    
    // Initial best cut is the minimum degree vertex
    int min_cut_val = m + 1;
    int best_cut_node = -1;
    for (int i = 0; i < n; ++i) {
        if (original_degrees[i] < min_cut_val) {
            min_cut_val = original_degrees[i];
            best_cut_node = i;
        }
    }
    
    std::vector<bool> in_S_global(n, false);
    if (best_cut_node != -1) {
        in_S_global[best_cut_node] = true;
    }
    
    // Build adjacency list for G_K
    std::vector<std::vector<Edge>> adj_gk(n);
    for (auto& pe : packed_edges) {
        adj_gk[pe.u].push_back({pe.v, pe.id});
        adj_gk[pe.v].push_back({pe.u, pe.id});
    }
    
    // Clear packed_edges as it is no longer needed
    packed_edges.clear();
    packed_edges.shrink_to_fit();
    
    // ----------------------------------------------------
    // Run DFS on G_K to find 1-cut (bridge) and 2-cuts
    // ----------------------------------------------------
    std::vector<int> disc(n, -1), low(n, -1), parent(n, -1), parent_edge(n, -1);
    std::vector<int> edge_idx(n, 0);
    std::vector<int> dfs_stack;
    dfs_stack.reserve(n);
    
    std::vector<uint64_t> xor_val(n, 0);
    std::mt19937_64 rng(1337);
    
    std::vector<uint64_t> back_edge_hash(m, 0);
    
    int timer = 0;
    bool found_cut = false;
    
    for (int start = 0; start < n; ++start) {
        if (disc[start] != -1) continue;
        
        dfs_stack.push_back(start);
        disc[start] = low[start] = ++timer;
        
        while (!dfs_stack.empty()) {
            int u = dfs_stack.back();
            if (edge_idx[u] < (int)adj_gk[u].size()) {
                auto& edge = adj_gk[u][edge_idx[u]];
                int v = edge.to;
                int edge_id = edge.weight; // stores original edge ID
                edge_idx[u]++;
                
                if (edge_id == parent_edge[u]) continue;
                
                if (disc[v] != -1) {
                    // Back-edge
                    low[u] = std::min(low[u], disc[v]);
                    if (disc[v] < disc[u]) {
                        uint64_t h = rng();
                        while (h == 0) h = rng();
                        xor_val[u] ^= h;
                        xor_val[v] ^= h;
                        back_edge_hash[edge_id] = h;
                    }
                } else {
                    // Tree-edge
                    parent[v] = u;
                    parent_edge[v] = edge_id;
                    disc[v] = low[v] = ++timer;
                    dfs_stack.push_back(v);
                }
            } else {
                dfs_stack.pop_back();
                if (!dfs_stack.empty()) {
                    int p = dfs_stack.back();
                    low[p] = std::min(low[p], low[u]);
                    
                    // Propagate XOR value up the DFS tree
                    uint64_t val = xor_val[u];
                    xor_val[p] ^= val;
                    
                    if (low[u] > disc[p]) {
                        // Bridge found!
                        // The cut partitions the graph into the subtree of u and the rest
                        min_cut_val = 1;
                        std::vector<bool> visited_sub(n, false);
                        std::vector<int> q;
                        q.push_back(u);
                        visited_sub[u] = true;
                        size_t head = 0;
                        while (head < q.size()) {
                            int curr = q[head++];
                            for (auto& edge : adj_gk[curr]) {
                                int next = edge.to;
                                if (edge.weight != parent_edge[u] && !visited_sub[next]) {
                                    visited_sub[next] = true;
                                    q.push_back(next);
                                }
                            }
                        }
                        in_S_global = visited_sub;
                        found_cut = true;
                        break;
                    }
                }
            }
        }
        if (found_cut) break;
    }
    
    if (min_cut_val == 1) {
        // Output bridge and terminate
        int count = 0;
        for (int i = 0; i < m; ++i) {
            if (in_S_global[edge_u[i]] != in_S_global[edge_v[i]]) {
                count++;
            }
        }
        std::cout << count << "\n";
        for (int i = 0; i < m; ++i) {
            if (in_S_global[edge_u[i]] != in_S_global[edge_v[i]]) {
                std::cout << edge_u[i] << " " << edge_v[i] << "\n";
            }
        }
        return 0;
    }
    
    // Check for 2-cuts
    // A 2-cut can consist of:
    // 1. A tree edge with XOR value matching a back-edge's hash.
    // 2. Two tree edges with the same non-zero XOR value.
    if (!found_cut) {
        struct ValueNode {
            uint64_t val;
            int node;
        };
        std::vector<ValueNode> tree_vals;
        tree_vals.reserve(n);
        for (int i = 0; i < n; ++i) {
            if (parent[i] != -1 && xor_val[i] != 0) {
                tree_vals.push_back({xor_val[i], i});
            }
        }
        
        std::sort(tree_vals.begin(), tree_vals.end(), [](const ValueNode& a, const ValueNode& b) {
            return a.val < b.val;
        });
        
        // Check for duplicate tree edge values (Case 2)
        int best_u1 = -1, best_u2 = -1;
        for (size_t i = 0; i + 1 < tree_vals.size(); ++i) {
            if (tree_vals[i].val == tree_vals[i+1].val) {
                best_u1 = tree_vals[i].node;
                best_u2 = tree_vals[i+1].node;
                break;
            }
        }
        
        if (best_u1 != -1) {
            // Found two tree edges with the same XOR val.
            // Removing them splits the tree.
            // S can be found by removing these two edges and running BFS from one node.
            int e1 = parent_edge[best_u1];
            int e2 = parent_edge[best_u2];
            min_cut_val = 2;
            std::vector<bool> visited_sub(n, false);
            std::vector<int> q;
            q.push_back(best_u1);
            visited_sub[best_u1] = true;
            size_t head = 0;
            while (head < q.size()) {
                int curr = q[head++];
                for (auto& edge : adj_gk[curr]) {
                    int next = edge.to;
                    if (edge.weight != e1 && edge.weight != e2 && !visited_sub[next]) {
                        visited_sub[next] = true;
                        q.push_back(next);
                    }
                }
            }
            in_S_global = visited_sub;
            found_cut = true;
        } else {
            // Check for tree edge value matching back-edge hash (Case 1)
            struct HashEdge {
                uint64_t hash;
                int edge_id;
            };
            std::vector<HashEdge> back_hashes;
            back_hashes.reserve(m);
            for (int i = 0; i < m; ++i) {
                if (back_edge_hash[i] != 0) {
                    back_hashes.push_back({back_edge_hash[i], i});
                }
            }
            std::sort(back_hashes.begin(), back_hashes.end(), [](const HashEdge& a, const HashEdge& b) {
                return a.hash < b.hash;
            });
            
            size_t i = 0, j = 0;
            while (i < tree_vals.size() && j < back_hashes.size()) {
                if (tree_vals[i].val == back_hashes[j].hash) {
                    // Match!
                    int tu = tree_vals[i].node;
                    int te = parent_edge[tu];
                    int be = back_hashes[j].edge_id;
                    min_cut_val = 2;
                    std::vector<bool> visited_sub(n, false);
                    std::vector<int> q;
                    q.push_back(tu);
                    visited_sub[tu] = true;
                    size_t head = 0;
                    while (head < q.size()) {
                        int curr = q[head++];
                        for (auto& edge : adj_gk[curr]) {
                            int next = edge.to;
                            if (edge.weight != te && edge.weight != be && !visited_sub[next]) {
                                visited_sub[next] = true;
                                q.push_back(next);
                            }
                        }
                    }
                    in_S_global = visited_sub;
                    found_cut = true;
                    break;
                } else if (tree_vals[i].val < back_hashes[j].hash) {
                    ++i;
                } else {
                    ++j;
                }
            }
        }
    }
    
    if (min_cut_val == 2) {
        // Output 2-cut and terminate
        int count = 0;
        for (int i = 0; i < m; ++i) {
            if (in_S_global[edge_u[i]] != in_S_global[edge_v[i]]) {
                count++;
            }
        }
        std::cout << count << "\n";
        for (int i = 0; i < m; ++i) {
            if (in_S_global[edge_u[i]] != in_S_global[edge_v[i]]) {
                std::cout << edge_u[i] << " " << edge_v[i] << "\n";
            }
        }
        return 0;
    }
    
    // ----------------------------------------------------
    // If min cut >= 3, proceed to PR contraction and SW
    // ----------------------------------------------------
    // We convert G_K to a weighted graph representation for contractions
    std::vector<std::vector<Edge>> adj(n);
    std::vector<int> deg(n, 0);
    for (auto& pe : packed_edges) {
        adj[pe.u].push_back({pe.v, 1}); // initial weight 1
        adj[pe.v].push_back({pe.u, 1});
    }
    for (int i = 0; i < n; ++i) {
        deg[i] = adj[i].size();
    }
    
    DSU contract_dsu(n);
    std::vector<bool> active(n, true);
    int active_count = n;
    
    bool contracted_any = true;
    while (contracted_any) {
        contracted_any = false;
        for (int i = 0; i < n; ++i) {
            if (!active[i] || contract_dsu.find(i) != i) continue;
            
            clean(i, contract_dsu, adj, deg);
            if (deg[i] < min_cut_val) {
                min_cut_val = deg[i];
                // Update partition
                for (int v = 0; v < n; ++v) {
                    in_S_global[v] = (contract_dsu.find(v) == i);
                }
            }
            
            bool local_contracted = false;
            for (size_t j = 0; j < adj[i].size(); ++j) {
                int v = contract_dsu.find(adj[i][j].to);
                int w = adj[i][j].weight;
                if (v == i) continue;
                
                if (w >= min_cut_val || w >= (deg[i] + 1) / 2 || w >= (deg[v] + 1) / 2) {
                    contract(i, v, contract_dsu, adj, deg, active);
                    active_count--;
                    local_contracted = true;
                    contracted_any = true;
                    break;
                }
            }
            if (local_contracted) {
                --i;
            }
        }
    }
    
    // Run Stoer-Wagner on remaining active vertices if more than 1 active
    if (active_count > 1) {
        std::vector<int> h_nodes;
        h_nodes.reserve(active_count);
        for (int i = 0; i < n; ++i) {
            if (active[i] && contract_dsu.find(i) == i) {
                h_nodes.push_back(i);
            }
        }
        
        int nh = h_nodes.size();
        std::vector<int> h_node_index(n, -1);
        for (int j = 0; j < nh; ++j) {
            h_node_index[h_nodes[j]] = j;
        }
        
        std::vector<std::vector<SW_Edge>> h_adj(nh);
        for (int j = 0; j < nh; ++j) {
            int u = h_nodes[j];
            clean(u, contract_dsu, adj, deg);
            for (auto& edge : adj[u]) {
                int v = contract_dsu.find(edge.to);
                if (v != u) {
                    h_adj[j].push_back({h_node_index[v], edge.weight});
                }
            }
        }
        
        std::vector<bool> in_S_sw;
        run_stoer_wagner(nh, h_nodes, h_adj, min_cut_val, in_S_sw);
        
        if (!in_S_sw.empty()) {
            for (int i = 0; i < n; ++i) {
                int r = contract_dsu.find(i);
                int idx = h_node_index[r];
                if (idx != -1) {
                    in_S_global[i] = in_S_sw[idx];
                } else {
                    in_S_global[i] = false;
                }
            }
        }
    }
    
    // Output final min cut
    int count = 0;
    for (int i = 0; i < m; ++i) {
        if (in_S_global[edge_u[i]] != in_S_global[edge_v[i]]) {
            count++;
        }
    }
    std::cout << count << "\n";
    for (int i = 0; i < m; ++i) {
        if (in_S_global[edge_u[i]] != in_S_global[edge_v[i]]) {
            std::cout << edge_u[i] << " " << edge_v[i] << "\n";
        }
    }
    
    return 0;
}
