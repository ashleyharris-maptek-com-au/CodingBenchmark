using namespace std;

const double INF = 1e18;

struct Edge {
    int to;
    double weight;
    double orig_weight;
};

struct InputEdge {
    int u, v;
    double w;
    bool operator<(const InputEdge& other) const {
        if (u != other.u) return u < other.u;
        if (v != other.v) return v < other.v;
        return w < other.w;
    }
};

struct KruskalEdge {
    int u, v;
    double weight;
    bool operator<(const KruskalEdge& other) const {
        return weight < other.weight;
    }
};

struct DSU {
    vector<int> parent;
    DSU(int n) {
        parent.resize(n);
        iota(parent.begin(), parent.end(), 0);
    }
    int find(int i) {
        if (parent[i] == i)
            return i;
        return parent[i] = find(parent[i]);
    }
    bool unite(int i, int j) {
        int root_i = find(i);
        int root_j = find(j);
        if (root_i != root_j) {
            parent[root_i] = root_j;
            return true;
        }
        return false;
    }
};

double get_edge_weight(int u, int v, const vector<vector<Edge>>& adj) {
    double min_w = INF;
    for (const auto& edge : adj[u]) {
        if (edge.to == v) {
            min_w = min(min_w, edge.weight);
        }
    }
    return min_w;
}

vector<pair<int, int>> get_clean_tree(const vector<pair<int, int>>& dirty_edges,
                                      const vector<vector<Edge>>& adj,
                                      const vector<int>& T,
                                      int V,
                                      double& out_weight) {
    vector<int> V_sub;
    vector<bool> in_sub(V, false);
    for (const auto& edge : dirty_edges) {
        if (!in_sub[edge.first]) {
            in_sub[edge.first] = true;
            V_sub.push_back(edge.first);
        }
        if (!in_sub[edge.second]) {
            in_sub[edge.second] = true;
            V_sub.push_back(edge.second);
        }
    }
    for (int t : T) {
        if (!in_sub[t]) {
            in_sub[t] = true;
            V_sub.push_back(t);
        }
    }
    
    vector<KruskalEdge> sub_edges;
    for (int u : V_sub) {
        for (const auto& edge : adj[u]) {
            int v = edge.to;
            if (v > u && in_sub[v]) {
                sub_edges.push_back({u, v, edge.weight});
            }
        }
    }
    
    sort(sub_edges.begin(), sub_edges.end());
    
    DSU dsu(V);
    vector<pair<int, int>> mst_edges;
    for (const auto& ke : sub_edges) {
        if (dsu.unite(ke.u, ke.v)) {
            mst_edges.push_back({ke.u, ke.v});
        }
    }
    
    vector<int> degree(V, 0);
    struct TreeNeighbor {
        int to;
        int edge_idx;
    };
    vector<vector<TreeNeighbor>> t_adj(V);
    for (size_t i = 0; i < mst_edges.size(); ++i) {
        int u = mst_edges[i].first;
        int v = mst_edges[i].second;
        t_adj[u].push_back({v, (int)i});
        t_adj[v].push_back({u, (int)i});
        degree[u]++;
        degree[v]++;
    }
    
    vector<bool> is_terminal(V, false);
    for (int t : T) is_terminal[t] = true;
    
    queue<int> q;
    for (int u : V_sub) {
        if (degree[u] == 1 && !is_terminal[u]) {
            q.push(u);
        }
    }
    
    vector<bool> edge_exists(mst_edges.size(), true);
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        if (degree[u] != 1) continue;
        
        for (const auto& neighbor : t_adj[u]) {
            int v = neighbor.to;
            int idx = neighbor.edge_idx;
            if (edge_exists[idx]) {
                edge_exists[idx] = false;
                degree[u]--;
                degree[v]--;
                if (degree[v] == 1 && !is_terminal[v]) {
                    q.push(v);
                }
                break;
            }
        }
    }
    
    vector<pair<int, int>> final_edges;
    out_weight = 0;
    for (size_t i = 0; i < mst_edges.size(); ++i) {
        if (edge_exists[i]) {
            final_edges.push_back(mst_edges[i]);
            out_weight += get_edge_weight(mst_edges[i].first, mst_edges[i].second, adj);
        }
    }
    return final_edges;
}

vector<pair<int, int>> dw_edges;
vector<bool> dw_visited;

void reconstruct_dw(int u, int mask, int num_states,
                    const vector<double>& dp,
                    const vector<int>& parent_vertex,
                    const vector<int>& split_mask) {
    int idx = u * num_states + mask;
    if (dw_visited[idx]) return;
    dw_visited[idx] = true;
    
    int pop = std::popcount((unsigned int)mask);
    if (pop == 1) {
        int p = parent_vertex[idx];
        if (p != -1) {
            dw_edges.push_back({u, p});
            reconstruct_dw(p, mask, num_states, dp, parent_vertex, split_mask);
        }
        return;
    }
    
    int p = parent_vertex[idx];
    if (p == -1) return;
    if (p != u) {
        dw_edges.push_back({u, p});
        reconstruct_dw(p, mask, num_states, dp, parent_vertex, split_mask);
    } else {
        int sub = split_mask[idx];
        int sib = mask ^ sub;
        reconstruct_dw(u, sub, num_states, dp, parent_vertex, split_mask);
        reconstruct_dw(u, sib, num_states, dp, parent_vertex, split_mask);
    }
}

vector<pair<int, int>> run_sph(int v_start, const vector<int>& T, const vector<vector<Edge>>& adj, int V) {
    vector<bool> in_tree(V, false);
    in_tree[v_start] = true;
    
    vector<int> tree_vertices;
    tree_vertices.push_back(v_start);
    
    vector<bool> is_terminal(V, false);
    for (int t : T) is_terminal[t] = true;
    
    int terminals_connected = is_terminal[v_start] ? 1 : 0;
    int total_terminals = T.size();
    
    vector<pair<int, int>> tree_edges;
    
    vector<double> dist(V, INF);
    vector<int> parent(V, -1);
    vector<int> touched;
    
    priority_queue<pair<double, int>, vector<pair<double, int>>, greater<>> pq;
    
    while (terminals_connected < total_terminals) {
        for (int u : tree_vertices) {
            dist[u] = 0;
            pq.push({0, u});
        }
        
        int target = -1;
        while (!pq.empty()) {
            auto [d, u] = pq.top();
            pq.pop();
            
            if (d > dist[u]) continue;
            
            if (!in_tree[u] && is_terminal[u]) {
                target = u;
                break;
            }
            
            for (const auto& edge : adj[u]) {
                int v = edge.to;
                double w = edge.weight;
                if (dist[v] > d + w) {
                    if (dist[v] == INF) {
                        touched.push_back(v);
                    }
                    dist[v] = d + w;
                    parent[v] = u;
                    pq.push({dist[v], v});
                }
            }
        }
        
        if (target == -1) {
            break;
        }
        
        int curr = target;
        while (!in_tree[curr]) {
            int p = parent[curr];
            tree_edges.push_back({curr, p});
            in_tree[curr] = true;
            tree_vertices.push_back(curr);
            if (is_terminal[curr]) {
                terminals_connected++;
            }
            curr = p;
        }
        
        for (int u : touched) {
            dist[u] = INF;
            parent[u] = -1;
        }
        touched.clear();
        for (int u : tree_vertices) {
            dist[u] = INF;
            parent[u] = -1;
        }
        while (!pq.empty()) pq.pop();
    }
    
    return tree_edges;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    int V, E, K;
    if (!(cin >> V >> E >> K)) return 0;
    
    vector<InputEdge> input_edges;
    for (int i = 0; i < E; ++i) {
        int u, v;
        double w;
        if (!(cin >> u >> v >> w)) break;
        if (u > v) swap(u, v);
        input_edges.push_back({u, v, w});
    }
    
    sort(input_edges.begin(), input_edges.end());
    
    vector<InputEdge> clean_input_edges;
    for (const auto& edge : input_edges) {
        if (clean_input_edges.empty() || 
            clean_input_edges.back().u != edge.u || 
            clean_input_edges.back().v != edge.v) {
            clean_input_edges.push_back(edge);
        }
    }
    
    vector<vector<Edge>> adj(V);
    for (const auto& edge : clean_input_edges) {
        adj[edge.u].push_back({edge.v, edge.w, edge.w});
        adj[edge.v].push_back({edge.u, edge.w, edge.w});
    }
    
    vector<int> T(K);
    for (int i = 0; i < K; ++i) {
        cin >> T[i];
    }
    
    sort(T.begin(), T.end());
    T.erase(unique(T.begin(), T.end()), T.end());
    K = T.size();
    
    if (K == 1) {
        cout << "0 0\n";
        return 0;
    }
    
    bool all_integers = true;
    for (const auto& edge : clean_input_edges) {
        if (edge.w != (long long)edge.w) {
            all_integers = false;
            break;
        }
    }
    
    auto start_time = chrono::high_resolution_clock::now();
    
    bool use_dw = false;
    
    double best_weight = INF;
    vector<pair<int, int>> best_edges;
    
    if (use_dw) {
        int num_states = 1 << K;
        vector<double> dp(V * num_states, INF);
        vector<int> parent_vertex(V * num_states, -1);
        vector<int> split_mask(V * num_states, 0);
        
        for (int i = 0; i < K; ++i) {
            int t = T[i];
            dp[t * num_states + (1 << i)] = 0;
        }
        
        for (int mask = 1; mask < num_states; ++mask) {
            int pop = std::popcount((unsigned int)mask);
            if (pop >= 2) {
                int first_bit = std::countr_zero((unsigned int)mask);
                for (int u = 0; u < V; ++u) {
                    double min_val = INF;
                    int best_submask = 0;
                    for (int submask = (mask - 1) & mask; submask > 0; submask = (submask - 1) & mask) {
                        if (submask & (1 << first_bit)) {
                            int sibling = mask ^ submask;
                            double val = dp[u * num_states + submask] + dp[u * num_states + sibling];
                            if (val < min_val) {
                                min_val = val;
                                best_submask = submask;
                            }
                        }
                    }
                    if (min_val < INF) {
                        dp[u * num_states + mask] = min_val;
                        split_mask[u * num_states + mask] = best_submask;
                        parent_vertex[u * num_states + mask] = u;
                    }
                }
            }
            
            priority_queue<pair<double, int>, vector<pair<double, int>>, greater<>> pq;
            for (int u = 0; u < V; ++u) {
                if (dp[u * num_states + mask] < INF) {
                    pq.push({dp[u * num_states + mask], u});
                }
            }
            while (!pq.empty()) {
                auto [d, u] = pq.top();
                pq.pop();
                
                if (d > dp[u * num_states + mask]) continue;
                
                for (const auto& edge : adj[u]) {
                    int v = edge.to;
                    double w = edge.weight;
                    if (dp[v * num_states + mask] > d + w) {
                        dp[v * num_states + mask] = d + w;
                        parent_vertex[v * num_states + mask] = u;
                        pq.push({dp[v * num_states + mask], v});
                    }
                }
            }
        }
        
        int t0 = T[0];
        if (dp[t0 * num_states + num_states - 1] < INF) {
            dw_edges.clear();
            dw_visited.assign(V * num_states, false);
            reconstruct_dw(t0, num_states - 1, num_states, dp, parent_vertex, split_mask);
            best_edges = get_clean_tree(dw_edges, adj, T, V, best_weight);
        }
    }
    
    if (best_edges.empty()) {
        int M = min((int)T.size(), 50);
        for (int i = 0; i < M; ++i) {
            auto tree = run_sph(T[i], T, adj, V);
            double w;
            auto clean_tree = get_clean_tree(tree, adj, T, V, w);
            if (!clean_tree.empty() && w < best_weight) {
                best_weight = w;
                best_edges = clean_tree;
            }
        }
        
        vector<pair<int, int>> degs;
        vector<bool> is_terminal(V, false);
        for (int t : T) is_terminal[t] = true;
        for (int i = 0; i < V; ++i) {
            if (!is_terminal[i] && !adj[i].empty()) {
                degs.push_back({(int)adj[i].size(), i});
            }
        }
        sort(degs.rbegin(), degs.rend());
        int non_term_runs = min((int)degs.size(), 15);
        for (int i = 0; i < non_term_runs; ++i) {
            int start_v = degs[i].second;
            auto tree = run_sph(start_v, T, adj, V);
            double w;
            auto clean_tree = get_clean_tree(tree, adj, T, V, w);
            if (!clean_tree.empty() && w < best_weight) {
                best_weight = w;
                best_edges = clean_tree;
            }
        }
        
        mt19937 rng(1337);
        while (true) {
            auto current_time = chrono::high_resolution_clock::now();
            double elapsed = chrono::duration<double>(current_time - start_time).count();
            if (elapsed > 1.6) break;
            
            for (int u = 0; u < V; ++u) {
                for (auto& edge : adj[u]) {
                    double r = (rng() % 100000) / 100000.0;
                    edge.weight = edge.orig_weight * (1.0 + r * 0.15);
                }
            }
            
            int start_v = T[rng() % T.size()];
            if (rng() % 10 < 3 && !degs.empty()) {
                start_v = degs[rng() % degs.size()].second;
            }
            
            auto tree = run_sph(start_v, T, adj, V);
            
            for (int u = 0; u < V; ++u) {
                for (auto& edge : adj[u]) {
                    edge.weight = edge.orig_weight;
                }
            }
            
            double w;
            auto clean_tree = get_clean_tree(tree, adj, T, V, w);
            if (!clean_tree.empty() && w < best_weight) {
                best_weight = w;
                best_edges = clean_tree;
            }
        }
    }
    
    if (all_integers) {
        cout << (long long)round(best_weight) << " " << best_edges.size() << "\n";
    } else {
        cout << fixed << setprecision(10) << best_weight << " " << best_edges.size() << "\n";
    }
    for (const auto& edge : best_edges) {
        cout << edge.first << " " << edge.second << "\n";
    }
    
    return 0;
}
