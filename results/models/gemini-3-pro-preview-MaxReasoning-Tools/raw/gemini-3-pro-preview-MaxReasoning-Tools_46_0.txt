workspace_dir = r"C:\AshDev\CodingBenchmark"

sys.path.insert(0, os.path.join(workspace_dir, "LLMBenchCore"))

import importlib
# Import 46.py
test_module = importlib.import_module("46")

shader_code = """#version 450

layout(local_size_x = 256) in;

// Each body is stored as two consecutive vec4s:
//   vec4(position.xyz, mass)
//   vec4(velocity.xyz, 0.0)
// So body i starts at index i*2 in the buffer.

layout(set = 0, binding = 0) readonly buffer InputBodies {
    vec4 inData[];   // length = N * 2
};

layout(set = 0, binding = 1) buffer OutputBodies {
    vec4 outData[];  // length = N * 2
};

layout(set = 0, binding = 2) uniform Params {
    uvec4 counts;    // counts.x = N (number of bodies)
    vec4 physics;    // physics.x = dt, physics.y = softening, physics.z = G
};

void main() {
    uint i = gl_GlobalInvocationID.x;
    uint N = counts.x;
    if (i >= N) {
        return;
    }

    float dt = physics.x;
    float softening = physics.y;
    float G = physics.z;

    vec4 body_pos_mass = inData[i * 2 + 0];
    vec4 body_vel = inData[i * 2 + 1];

    vec3 pos_i = body_pos_mass.xyz;
    float mass_i = body_pos_mass.w;
    vec3 vel_i = body_vel.xyz;

    vec3 acc = vec3(0.0);

    for (uint j = 0; j < N; ++j) {
        if (i == j) {
            continue;
        }
        vec4 other_pos_mass = inData[j * 2 + 0];
        vec3 pos_j = other_pos_mass.xyz;
        float mass_j = other_pos_mass.w;

        vec3 diff = pos_j - pos_i;
        float dist_sq = dot(diff, diff) + softening * softening;
        float dist = sqrt(dist_sq);
        
        acc += (G * mass_j * diff) / (dist_sq * dist);
    }

    // Semi-implicit Euler integration:
    // Update velocity first
    vec3 next_vel = vel_i + acc * dt;
    // Then update position using the updated velocity
    vec3 next_pos = pos_i + next_vel * dt;

    // Write updated values back
    outData[i * 2 + 0] = vec4(next_pos, mass_i);
    outData[i * 2 + 1] = vec4(next_vel, 0.0);
}
"""

for sp in range(6):
    print(f"Running test subpass {sp}...")
    res = test_module.gradeAnswer({"shader_code": shader_code}, sp, "test_engine")
    print(f"Subpass {sp} result:", res)
