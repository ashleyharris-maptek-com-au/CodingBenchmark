@group(0) @binding(0) var<storage, read>       baseData : array<u32, 16>;
@group(0) @binding(1) var<storage, read_write>  result   : array<u32, 4>;
@group(0) @binding(2) var<uniform>              params   : vec4<u32>;
// params.x = difficulty, params.y = range_start, params.z = range_size

@compute @workgroup_size(256)
fn main(@builtin(global_invocation_id) gid : vec3<u32>) {
    if (gid.x >= params.z) {
        return;
    }

    let nonce = params.y + gid.x;
    var state : array<u32, 16>;

    state[0u] = baseData[0u] ^ (nonce + 0u);
    state[1u] = baseData[1u] ^ (nonce + 1u);
    state[2u] = baseData[2u] ^ (nonce + 2u);
    state[3u] = baseData[3u] ^ (nonce + 3u);
    state[4u] = baseData[4u] ^ (nonce + 4u);
    state[5u] = baseData[5u] ^ (nonce + 5u);
    state[6u] = baseData[6u] ^ (nonce + 6u);
    state[7u] = baseData[7u] ^ (nonce + 7u);
    state[8u] = baseData[8u] ^ (nonce + 8u);
    state[9u] = baseData[9u] ^ (nonce + 9u);
    state[10u] = baseData[10u] ^ (nonce + 10u);
    state[11u] = baseData[11u] ^ (nonce + 11u);
    state[12u] = baseData[12u] ^ (nonce + 12u);
    state[13u] = baseData[13u] ^ (nonce + 13u);
    state[14u] = baseData[14u] ^ (nonce + 14u);
    state[15u] = baseData[15u] ^ (nonce + 15u);

    for (var round = 0u; round < 8u; round = round + 1u) {
        // Group 1: state[0..3]
        state[0u] = state[0u] + state[1u];
        state[3u] = state[3u] ^ state[0u];
        state[3u] = (state[3u] << 16u) | (state[3u] >> 16u);
        state[2u] = state[2u] + state[3u];
        state[1u] = state[1u] ^ state[2u];
        state[1u] = (state[1u] << 12u) | (state[1u] >> 20u);
        state[0u] = state[0u] + state[1u];
        state[3u] = state[3u] ^ state[0u];
        state[3u] = (state[3u] << 8u) | (state[3u] >> 24u);
        state[2u] = state[2u] + state[3u];
        state[1u] = state[1u] ^ state[2u];
        state[1u] = (state[1u] << 7u) | (state[1u] >> 25u);

        // Group 2: state[4..7]
        state[4u] = state[4u] + state[5u];
        state[7u] = state[7u] ^ state[4u];
        state[7u] = (state[7u] << 16u) | (state[7u] >> 16u);
        state[6u] = state[6u] + state[7u];
        state[5u] = state[5u] ^ state[6u];
        state[5u] = (state[5u] << 12u) | (state[5u] >> 20u);
        state[4u] = state[4u] + state[5u];
        state[7u] = state[7u] ^ state[4u];
        state[7u] = (state[7u] << 8u) | (state[7u] >> 24u);
        state[6u] = state[6u] + state[7u];
        state[5u] = state[5u] ^ state[6u];
        state[5u] = (state[5u] << 7u) | (state[5u] >> 25u);

        // Group 3: state[8..11]
        state[8u] = state[8u] + state[9u];
        state[11u] = state[11u] ^ state[8u];
        state[11u] = (state[11u] << 16u) | (state[11u] >> 16u);
        state[10u] = state[10u] + state[11u];
        state[9u] = state[9u] ^ state[10u];
        state[9u] = (state[9u] << 12u) | (state[9u] >> 20u);
        state[8u] = state[8u] + state[9u];
        state[11u] = state[11u] ^ state[8u];
        state[11u] = (state[11u] << 8u) | (state[11u] >> 24u);
        state[10u] = state[10u] + state[11u];
        state[9u] = state[9u] ^ state[10u];
        state[9u] = (state[9u] << 7u) | (state[9u] >> 25u);

        // Group 4: state[12..15]
        state[12u] = state[12u] + state[13u];
        state[15u] = state[15u] ^ state[12u];
        state[15u] = (state[15u] << 16u) | (state[15u] >> 16u);
        state[14u] = state[14u] + state[15u];
        state[13u] = state[13u] ^ state[14u];
        state[13u] = (state[13u] << 12u) | (state[13u] >> 20u);
        state[12u] = state[12u] + state[13u];
        state[15u] = state[15u] ^ state[12u];
        state[15u] = (state[15u] << 8u) | (state[15u] >> 24u);
        state[14u] = state[14u] + state[15u];
        state[13u] = state[13u] ^ state[14u];
        state[13u] = (state[13u] << 7u) | (state[13u] >> 25u);
    }

    let h0 = state[0u] ^ state[2u] ^ state[4u] ^ state[6u] ^ state[8u] ^ state[10u] ^ state[12u] ^ state[14u];
    let h1 = state[1u] ^ state[3u] ^ state[5u] ^ state[7u] ^ state[9u] ^ state[11u] ^ state[13u] ^ state[15u];

    var trailing_zeros = 0u;
    if (h0 != 0u) {
        trailing_zeros = 31u - countLeadingZeros(h0 & (~h0 + 1u));
    } else {
        if (h1 != 0u) {
            trailing_zeros = 32u + (31u - countLeadingZeros(h1 & (~h1 + 1u)));
        } else {
            trailing_zeros = 64u;
        }
    }

    if (trailing_zeros >= params.x) {
        result[0u] = 1u;
        result[1u] = nonce;
        result[2u] = h0;
        result[3u] = h1;
    }
}
