const fs = require('fs');

function solve() {
    let input;
    try {
        input = fs.readFileSync(0, 'utf8');
    } catch (e) {
        return;
    }
    if (!input.trim()) return;
    const data = JSON.parse(input);
    
    const w = data.width || (data.grid && data.grid[0] ? data.grid[0].length : 0);
    const h = data.height || (data.grid ? data.grid.length : 0);
    
    data.robots.sort((a, b) => a.id - b.id);
    data.jobs.sort((a, b) => a.id - b.id);
    
    const robotStarts = data.robots.map(r => ({ x: r.x, y: r.y }));
    const jobsMap = new Map();
    for (const j of data.jobs) {
        jobsMap.set(j.id, { x: j.x, y: j.y });
    }
    
    const exactJobLimit = process.env.WAREHOUSE_EXACT_JOB_LIMIT ? parseInt(process.env.WAREHOUSE_EXACT_JOB_LIMIT, 10) : 150;
    const useManhattan = data.jobs.length > exactJobLimit;
    
    const distanceCache = new Map();
    function getDistanceGrid(x, y) {
        const key = y * w + x;
        if (distanceCache.has(key)) {
            return distanceCache.get(key);
        }
        const dist = Array.from({ length: h }, () => new Int32Array(w).fill(1e9));
        dist[y][x] = 0;
        const queue = new Int32Array(w * h);
        let head = 0;
        let tail = 0;
        queue[tail++] = (y << 16) | x;
        
        const dx = [1, -1, 0, 0];
        const dy = [0, 0, 1, -1];
        
        while (head < tail) {
            const val = queue[head++];
            const cx = val & 0xFFFF;
            const cy = val >> 16;
            const d = dist[cy][cx];
            const nd = d + 1;
            
            for (let i = 0; i < 4; i++) {
                const nx = cx + dx[i];
                const ny = cy + dy[i];
                
                if (nx >= 0 && nx < w && ny >= 0 && ny < h) {
                    if (data.grid[ny][nx] !== '#' && dist[ny][nx] === 1e9) {
                        dist[ny][nx] = nd;
                        queue[tail++] = (ny << 16) | nx;
                    }
                }
            }
        }
        distanceCache.set(key, dist);
        return dist;
    }
    
    function getDistance(p1, p2) {
        if (useManhattan) {
            return Math.abs(p1.x - p2.x) + Math.abs(p1.y - p2.y);
        } else {
            const distGrid = getDistanceGrid(p1.x, p1.y);
            return distGrid[p2.y][p2.x];
        }
    }
    
    if (!useManhattan) {
        for (const r of data.robots) {
            getDistanceGrid(r.x, r.y);
        }
        for (const j of data.jobs) {
            getDistanceGrid(j.x, j.y);
        }
    }
    
    const remaining = new Set(data.jobs.map(j => j.id));
    const currentPos = data.robots.map(r => ({ x: r.x, y: r.y }));
    const routes = Array.from({ length: data.robots.length }, () => []);
    
    while (remaining.size > 0) {
        let bestDist = 2e9;
        let bestRobotIdx = -1;
        let bestJobId = -1;
        
        for (let ri = 0; ri < data.robots.length; ri++) {
            const pos = currentPos[ri];
            for (const job of data.jobs) {
                if (remaining.has(job.id)) {
                    const d = getDistance(pos, job);
                    if (bestRobotIdx === -1 || d < bestDist) {
                        bestDist = d;
                        bestRobotIdx = ri;
                        bestJobId = job.id;
                    }
                }
            }
        }
        
        routes[bestRobotIdx].push(bestJobId);
        const assignedJob = jobsMap.get(bestJobId);
        currentPos[bestRobotIdx] = { x: assignedJob.x, y: assignedJob.y };
        remaining.delete(bestJobId);
    }
    
    function getSingleRouteCost(route, startPos) {
        let total = 0;
        let cur = startPos;
        for (const jid of route) {
            const jobPos = jobsMap.get(jid);
            const d = getDistance(cur, jobPos);
            if (d >= 1e9) return 1e12;
            total += d;
            cur = jobPos;
        }
        return total;
    }
    
    function reverseSegment(arr, i, j) {
        while (i < j) {
            const temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
            i++;
            j--;
        }
    }
    
    function optimize2Opt(route, startPos) {
        let improved = true;
        while (improved) {
            improved = false;
            for (let i = 0; i < route.length - 1; i++) {
                const prevNode = i === 0 ? startPos : jobsMap.get(route[i-1]);
                const nodeI = jobsMap.get(route[i]);
                for (let j = i + 1; j < route.length; j++) {
                    const nodeJ = jobsMap.get(route[j]);
                    const nextNode = j + 1 < route.length ? jobsMap.get(route[j+1]) : null;
                    
                    const oldDist1 = getDistance(prevNode, nodeI);
                    const oldDist2 = nextNode ? getDistance(nodeJ, nextNode) : 0;
                    
                    const newDist1 = getDistance(prevNode, nodeJ);
                    const newDist2 = nextNode ? getDistance(nodeI, nextNode) : 0;
                    
                    const delta = (newDist1 + newDist2) - (oldDist1 + oldDist2);
                    if (delta < -1e-5) {
                        reverseSegment(route, i, j);
                        improved = true;
                        break;
                    }
                }
                if (improved) break;
            }
        }
    }
    
    function tryInterRouteRelocate() {
        const numRobots = routes.length;
        for (let ri = 0; ri < numRobots; ri++) {
            const routeI = routes[ri];
            if (routeI.length === 0) continue;
            for (let i = 0; i < routeI.length; i++) {
                const jobI = routeI[i];
                const nodeI = jobsMap.get(jobI);
                const prevI = i === 0 ? robotStarts[ri] : jobsMap.get(routeI[i-1]);
                const nextI = i + 1 < routeI.length ? jobsMap.get(routeI[i+1]) : null;
                
                const oldDistI1 = getDistance(prevI, nodeI);
                const oldDistI2 = nextI ? getDistance(nodeI, nextI) : 0;
                const newDistI = nextI ? getDistance(prevI, nextI) : 0;
                const deltaI = newDistI - (oldDistI1 + oldDistI2);
                
                for (let rj = 0; rj < numRobots; rj++) {
                    if (ri === rj) continue;
                    const routeJ = routes[rj];
                    
                    for (let j = 0; j <= routeJ.length; j++) {
                        const prevJ = j === 0 ? robotStarts[rj] : jobsMap.get(routeJ[j-1]);
                        const nextJ = j < routeJ.length ? jobsMap.get(routeJ[j]) : null;
                        
                        const oldDistJ = nextJ ? getDistance(prevJ, nextJ) : 0;
                        const newDistJ1 = getDistance(prevJ, nodeI);
                        const newDistJ2 = nextJ ? getDistance(nodeI, nextJ) : 0;
                        const deltaJ = (newDistJ1 + newDistJ2) - oldDistJ;
                        
                        if (deltaI + deltaJ < -1e-5) {
                            routeI.splice(i, 1);
                            routeJ.splice(j, 0, jobI);
                            return true;
                        }
                    }
                }
            }
        }
        return false;
    }
    
    function tryInterRouteSwap() {
        const numRobots = routes.length;
        for (let ri = 0; ri < numRobots; ri++) {
            const routeI = routes[ri];
            for (let i = 0; i < routeI.length; i++) {
                const jobI = routeI[i];
                const nodeI = jobsMap.get(jobI);
                const prevI = i === 0 ? robotStarts[ri] : jobsMap.get(routeI[i-1]);
                const nextI = i + 1 < routeI.length ? jobsMap.get(routeI[i+1]) : null;
                
                const oldDistI1 = getDistance(prevI, nodeI);
                const oldDistI2 = nextI ? getDistance(nodeI, nextI) : 0;
                
                for (let rj = ri + 1; rj < numRobots; rj++) {
                    const routeJ = routes[rj];
                    for (let j = 0; j < routeJ.length; j++) {
                        const jobJ = routeJ[j];
                        const nodeJ = jobsMap.get(jobJ);
                        const prevJ = j === 0 ? robotStarts[rj] : jobsMap.get(routeJ[j-1]);
                        const nextJ = j + 1 < routeJ.length ? jobsMap.get(routeJ[j+1]) : null;
                        
                        const oldDistJ1 = getDistance(prevJ, nodeJ);
                        const oldDistJ2 = nextJ ? getDistance(nodeJ, nextJ) : 0;
                        
                        const newDistI1 = getDistance(prevI, nodeJ);
                        const newDistI2 = nextI ? getDistance(nodeJ, nextI) : 0;
                        
                        const newDistJ1 = getDistance(prevJ, nodeI);
                        const newDistJ2 = nextJ ? getDistance(nodeI, nextJ) : 0;
                        
                        const delta = (newDistI1 + newDistI2 + newDistJ1 + newDistJ2) - (oldDistI1 + oldDistI2 + oldDistJ1 + oldDistJ2);
                        if (delta < -1e-5) {
                            routeI[i] = jobJ;
                            routeJ[j] = jobI;
                            return true;
                        }
                    }
                }
            }
        }
        return false;
    }
    
    function tryIntraRouteRelocate() {
        for (let ri = 0; ri < routes.length; ri++) {
            const route = routes[ri];
            if (route.length < 2) continue;
            for (let i = 0; i < route.length; i++) {
                const jobI = route[i];
                const nodeI = jobsMap.get(jobI);
                const prevI = i === 0 ? robotStarts[ri] : jobsMap.get(route[i-1]);
                const nextI = i + 1 < route.length ? jobsMap.get(route[i+1]) : null;
                
                const oldDistI1 = getDistance(prevI, nodeI);
                const oldDistI2 = nextI ? getDistance(nodeI, nextI) : 0;
                const newDistI = nextI ? getDistance(prevI, nextI) : 0;
                const deltaRemove = newDistI - (oldDistI1 + oldDistI2);
                
                const tempRoute = [...route];
                tempRoute.splice(i, 1);
                
                for (let j = 0; j <= tempRoute.length; j++) {
                    if (j === i) continue;
                    
                    const prevJ = j === 0 ? robotStarts[ri] : jobsMap.get(tempRoute[j-1]);
                    const nextJ = j < tempRoute.length ? jobsMap.get(tempRoute[j]) : null;
                    
                    const oldDistJ = nextJ ? getDistance(prevJ, nextJ) : 0;
                    const newDistJ1 = getDistance(prevJ, nodeI);
                    const newDistJ2 = nextJ ? getDistance(nodeI, nextJ) : 0;
                    const deltaInsert = (newDistJ1 + newDistJ2) - oldDistJ;
                    
                    if (deltaRemove + deltaInsert < -1e-5) {
                        route.splice(i, 1);
                        route.splice(j, 0, jobI);
                        return true;
                    }
                }
            }
        }
        return false;
    }
    
    let improved = true;
    const startTime = Date.now();
    const maxDuration = 4000;
    
    while (improved && (Date.now() - startTime < maxDuration)) {
        improved = false;
        
        for (let ri = 0; ri < routes.length; ri++) {
            optimize2Opt(routes[ri], robotStarts[ri]);
        }
        
        if (tryInterRouteRelocate()) {
            improved = true;
            continue;
        }
        
        if (tryInterRouteSwap()) {
            improved = true;
            continue;
        }
        
        if (tryIntraRouteRelocate()) {
            improved = true;
            continue;
        }
    }
    
    const assignments = routes;
    console.log(JSON.stringify({ assignments }));
}

solve();
