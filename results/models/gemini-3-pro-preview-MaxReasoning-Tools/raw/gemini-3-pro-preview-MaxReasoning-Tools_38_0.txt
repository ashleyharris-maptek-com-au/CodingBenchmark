import sys

# Increase recursion depth to handle deep search trees
sys.setrecursionlimit(200000)

def solve(X, Y, solution=None):
    if solution is None:
        solution = []
    if not X:
        yield list(solution)
        return

    # Find the column with the minimum number of rows to minimize branching factor.
    # Early backtrack if any column has 0 rows covering it.
    c = None
    min_len = len(X) + 1
    for col in X:
        l = len(X[col])
        if l < min_len:
            min_len = l
            c = col
            if min_len == 0:
                return

    for r in list(X[c]):
        solution.append(r)
        cols = select(X, Y, r)
        yield from solve(X, Y, solution)
        deselect(X, Y, r, cols)
        solution.pop()

def select(X, Y, r):
    cols = []
    for j in Y[r]:
        for i in X[j]:
            for k in Y[i]:
                if k != j:
                    X[k].remove(i)
        cols.append(X.pop(j))
    return cols

def deselect(X, Y, r, cols):
    for j in reversed(Y[r]):
        X[j] = cols.pop()
        for i in X[j]:
            for k in Y[i]:
                if k != j:
                    X[k].add(i)

def main():
    # Read all lines from standard input
    input_lines = sys.stdin.read().splitlines()
    
    # Find the line containing U and S (ignoring leading empty lines)
    idx = 0
    while idx < len(input_lines) and not input_lines[idx].strip():
        idx += 1
        
    if idx >= len(input_lines):
        print("NO SOLUTION")
        return
        
    first_line = input_lines[idx].strip().split()
    if len(first_line) < 2:
        print("NO SOLUTION")
        return
        
    try:
        U = int(first_line[0])
        S = int(first_line[1])
    except ValueError:
        print("NO SOLUTION")
        return
        
    idx += 1
    
    # Read exactly S set lines.
    sets = []
    for _ in range(S):
        if idx < len(input_lines):
            line = input_lines[idx].strip()
            if line:
                sets.append([int(x) for x in line.split()])
            else:
                sets.append([])
            idx += 1
        else:
            sets.append([])
            
    # Set up columns (X) and rows (Y) for Algorithm X
    # X maps each element (0..U-1) to the set of row indices covering it.
    X = {j: set() for j in range(U)}
    # Y maps each row index to the list of elements it covers.
    Y = {}
    
    for i, s in enumerate(sets):
        # Keep only elements within the valid range [0, U-1] and deduplicate
        unique_s = [x for x in set(s) if 0 <= x < U]
        Y[i] = unique_s
        for j in unique_s:
            X[j].add(i)
            
    # Solve
    try:
        sol = next(solve(X, Y))
        print("SOLUTION")
        print(" ".join(map(str, sorted(sol))))
    except StopIteration:
        print("NO SOLUTION")

if __name__ == "__main__":
    main()
