#include <iostream>
#include <vector>
#include <deque>
#include <queue>
#include <unordered_set>
#include <unordered_map>
#include <algorithm>
#include <cmath>

constexpr int MAX_D = 128;
int G_D = 0;

struct Coord {
    int val[MAX_D] = {0};

    bool operator==(const Coord& other) const {
        for (int i = 0; i < G_D; ++i) {
            if (val[i] != other.val[i]) return false;
        }
        return true;
    }
};

struct CoordHash {
    std::size_t operator()(const Coord& c) const {
        std::size_t h = 0;
        for (int i = 0; i < G_D; ++i) {
            h ^= std::hash<int>{}(c.val[i]) + 0x9e3779b9 + (h << 6) + (h >> 2);
        }
        return h;
    }
};

inline int get_manhattan_dist(const Coord& a, const Coord& b) {
    int dist = 0;
    for (int i = 0; i < G_D; ++i) {
        dist += std::abs(a.val[i] - b.val[i]);
    }
    return dist;
}

struct AStarNode {
    Coord coord;
    int g;
    int h;
    int f;
    AStarNode* parent;
};

struct AStarPool {
    std::vector<AStarNode> nodes;
    int pool_ptr = 0;

    void reset() {
        pool_ptr = 0;
    }

    AStarNode* allocate(const Coord& coord, int g, int h, AStarNode* parent) {
        if (pool_ptr >= (int)nodes.size()) {
            nodes.resize(std::max((size_t)1024, nodes.size() * 2));
        }
        AStarNode& n = nodes[pool_ptr++];
        n.coord = coord;
        n.g = g;
        n.h = h;
        n.f = g + h;
        n.parent = parent;
        return &n;
    }
};

struct CompareNode {
    bool operator()(const AStarNode* a, const AStarNode* b) const {
        if (a->f != b->f) {
            return a->f > b->f; // smaller f has higher priority
        }
        return a->g < b->g; // larger g has higher priority
    }
};

bool is_valid(const Coord& neighbor, AStarNode* current_node, int next_g, 
              const Coord& bounds, 
              const std::unordered_set<Coord, CoordHash>& obstacles,
              const std::unordered_map<Coord, int, CoordHash>& body_min_index_ge_1,
              int snake_length) {
    
    // 1. Check bounds
    for (int i = 0; i < G_D; ++i) {
        if (neighbor.val[i] < 0 || neighbor.val[i] >= bounds.val[i]) {
            return false;
        }
    }

    // 2. Check obstacles
    if (obstacles.count(neighbor)) {
        return false;
    }

    // 3. Check snake body collision
    AStarNode* curr = current_node;
    int steps_back = 0;
    int k = snake_length - 1;
    while (curr != nullptr && steps_back < k) {
        if (curr->coord == neighbor) {
            return false;
        }
        curr = curr->parent;
        steps_back++;
    }

    if (steps_back < k) {
        auto it = body_min_index_ge_1.find(neighbor);
        if (it != body_min_index_ge_1.end()) {
            if (it->second <= snake_length - steps_back) {
                return false;
            }
        }
    }

    return true;
}

bool is_immediately_valid(
    const Coord& neighbor,
    const std::deque<Coord>& snake_body,
    const Coord& bounds,
    const std::unordered_set<Coord, CoordHash>& obstacles
) {
    // 1. Check bounds
    for (int i = 0; i < G_D; ++i) {
        if (neighbor.val[i] < 0 || neighbor.val[i] >= bounds.val[i]) {
            return false;
        }
    }
    // 2. Check obstacles
    if (obstacles.count(neighbor)) {
        return false;
    }
    // 3. Body collision (the tail moves out, so check index 0 to L-2)
    if (snake_body.size() > 1) {
        for (size_t i = 0; i < snake_body.size() - 1; ++i) {
            if (snake_body[i] == neighbor) {
                return false;
            }
        }
    }
    return true;
}

std::vector<Coord> find_path_multi(
    const Coord& start,
    const std::unordered_set<Coord, CoordHash>& food_set,
    const Coord& bounds,
    const std::unordered_set<Coord, CoordHash>& obstacles,
    const std::deque<Coord>& snake_body,
    AStarPool& pool,
    int max_nodes = 30000
) {
    if (food_set.empty()) return {};
    pool.reset();

    std::unordered_map<Coord, int, CoordHash> body_min_index_ge_1;
    for (int i = 1; i < (int)snake_body.size(); ++i) {
        const auto& c = snake_body[i];
        auto it = body_min_index_ge_1.find(c);
        if (it == body_min_index_ge_1.end()) {
            body_min_index_ge_1[c] = i;
        } else {
            body_min_index_ge_1[c] = std::min(it->second, i);
        }
    }

    std::vector<Coord> S;
    {
        std::vector<std::pair<int, Coord>> food_dists;
        food_dists.reserve(food_set.size());
        for (const auto& f : food_set) {
            int d = get_manhattan_dist(start, f);
            food_dists.push_back({d, f});
        }
        int num_targets = std::min(5, (int)food_dists.size());
        std::nth_element(food_dists.begin(), food_dists.begin() + num_targets - 1, food_dists.end(), 
            [](const std::pair<int, Coord>& a, const std::pair<int, Coord>& b) {
                return a.first < b.first;
            });
        for (int i = 0; i < num_targets; ++i) {
            S.push_back(food_dists[i].second);
        }
    }

    std::unordered_set<Coord, CoordHash> S_set(S.begin(), S.end());

    auto get_h = [&](const Coord& c) {
        int min_d = 1e9;
        for (const auto& target : S) {
            int d = get_manhattan_dist(c, target);
            if (d < min_d) min_d = d;
        }
        return min_d;
    };

    std::priority_queue<AStarNode*, std::vector<AStarNode*>, CompareNode> open_set;
    std::unordered_map<Coord, int, CoordHash> visited;

    AStarNode* root = pool.allocate(start, 0, get_h(start), nullptr);
    open_set.push(root);
    visited[start] = 0;

    int nodes_expanded = 0;

    while (!open_set.empty()) {
        AStarNode* curr = open_set.top();
        open_set.pop();

        if (S_set.count(curr->coord)) {
            std::vector<Coord> path;
            AStarNode* p = curr;
            while (p != nullptr) {
                path.push_back(p->coord);
                p = p->parent;
            }
            std::reverse(path.begin(), path.end());
            return path;
        }

        if (curr->g > visited[curr->coord]) {
            continue;
        }

        nodes_expanded++;
        if (nodes_expanded > max_nodes) {
            break;
        }

        for (int i = 0; i < G_D; ++i) {
            for (int dir : {-1, 1}) {
                Coord neighbor = curr->coord;
                neighbor.val[i] += dir;

                if (!is_valid(neighbor, curr, curr->g + 1, bounds, obstacles, body_min_index_ge_1, snake_body.size())) {
                    continue;
                }

                int next_g = curr->g + 1;
                auto it = visited.find(neighbor);
                if (it == visited.end() || next_g < it->second) {
                    visited[neighbor] = next_g;
                    AStarNode* next_node = pool.allocate(neighbor, next_g, get_h(neighbor), curr);
                    open_set.push(next_node);
                }
            }
        }
    }

    return {};
}

std::vector<Coord> find_any_reachable_path(
    const Coord& start,
    const std::unordered_set<Coord, CoordHash>& food_set,
    const Coord& bounds,
    const std::unordered_set<Coord, CoordHash>& obstacles,
    const std::deque<Coord>& snake_body,
    AStarPool& pool,
    int max_nodes = 10000
) {
    if (food_set.empty()) return {};
    pool.reset();

    std::unordered_map<Coord, int, CoordHash> body_min_index_ge_1;
    for (int i = 1; i < (int)snake_body.size(); ++i) {
        const auto& c = snake_body[i];
        auto it = body_min_index_ge_1.find(c);
        if (it == body_min_index_ge_1.end()) {
            body_min_index_ge_1[c] = i;
        } else {
            body_min_index_ge_1[c] = std::min(it->second, i);
        }
    }

    std::priority_queue<AStarNode*, std::vector<AStarNode*>, CompareNode> open_set;
    std::unordered_map<Coord, int, CoordHash> visited;

    AStarNode* root = pool.allocate(start, 0, 0, nullptr);
    open_set.push(root);
    visited[start] = 0;

    int nodes_expanded = 0;

    while (!open_set.empty()) {
        AStarNode* curr = open_set.top();
        open_set.pop();

        if (food_set.count(curr->coord)) {
            std::vector<Coord> path;
            AStarNode* p = curr;
            while (p != nullptr) {
                path.push_back(p->coord);
                p = p->parent;
            }
            std::reverse(path.begin(), path.end());
            return path;
        }

        if (curr->g > visited[curr->coord]) {
            continue;
        }

        nodes_expanded++;
        if (nodes_expanded > max_nodes) {
            break;
        }

        for (int i = 0; i < G_D; ++i) {
            for (int dir : {-1, 1}) {
                Coord neighbor = curr->coord;
                neighbor.val[i] += dir;

                if (!is_valid(neighbor, curr, curr->g + 1, bounds, obstacles, body_min_index_ge_1, snake_body.size())) {
                    continue;
                }

                int next_g = curr->g + 1;
                auto it = visited.find(neighbor);
                if (it == visited.end() || next_g < it->second) {
                    visited[neighbor] = next_g;
                    AStarNode* next_node = pool.allocate(neighbor, next_g, 0, curr);
                    open_set.push(next_node);
                }
            }
        }
    }

    return {};
}

int compute_reachable_area(
    const Coord& start,
    const Coord& bounds,
    const std::unordered_set<Coord, CoordHash>& obstacles,
    const std::deque<Coord>& snake_body,
    int max_visited = 150
) {
    std::unordered_set<Coord, CoordHash> visited;
    std::queue<std::pair<Coord, int>> q;

    q.push({start, 0});
    visited.insert(start);

    std::unordered_map<Coord, int, CoordHash> body_min_index_ge_1;
    for (int i = 1; i < (int)snake_body.size(); ++i) {
        const auto& c = snake_body[i];
        auto it = body_min_index_ge_1.find(c);
        if (it == body_min_index_ge_1.end()) {
            body_min_index_ge_1[c] = i;
        } else {
            body_min_index_ge_1[c] = std::min(it->second, i);
        }
    }

    int count = 0;
    while (!q.empty() && count < max_visited) {
        auto [curr_coord, g] = q.front();
        q.pop();
        count++;

        for (int i = 0; i < G_D; ++i) {
            for (int dir : {-1, 1}) {
                Coord neighbor = curr_coord;
                neighbor.val[i] += dir;

                if (visited.count(neighbor)) continue;

                // Bounds check
                bool out_of_bounds = false;
                for (int j = 0; j < G_D; ++j) {
                    if (neighbor.val[j] < 0 || neighbor.val[j] >= bounds.val[j]) {
                        out_of_bounds = true;
                        break;
                    }
                }
                if (out_of_bounds) continue;

                // Obstacles
                if (obstacles.count(neighbor)) continue;

                // Body suffix
                auto it = body_min_index_ge_1.find(neighbor);
                if (it != body_min_index_ge_1.end() && it->second >= g + 1) {
                    continue;
                }

                visited.insert(neighbor);
                q.push({neighbor, g + 1});
            }
        }
    }
    return count;
}

Coord get_survival_move(
    const Coord& head,
    const Coord& bounds,
    const std::unordered_set<Coord, CoordHash>& obstacles,
    const std::deque<Coord>& snake_body
) {
    int best_area = -1;
    Coord best_neighbor;
    bool found_any = false;

    std::unordered_map<Coord, int, CoordHash> body_min_index_ge_1;
    for (int i = 1; i < (int)snake_body.size(); ++i) {
        const auto& c = snake_body[i];
        auto it = body_min_index_ge_1.find(c);
        if (it == body_min_index_ge_1.end()) {
            body_min_index_ge_1[c] = i;
        } else {
            body_min_index_ge_1[c] = std::min(it->second, i);
        }
    }

    for (int i = 0; i < G_D; ++i) {
        for (int dir : {-1, 1}) {
            Coord neighbor = head;
            neighbor.val[i] += dir;

            // Bounds check
            bool out_of_bounds = false;
            for (int j = 0; j < G_D; ++j) {
                if (neighbor.val[j] < 0 || neighbor.val[j] >= bounds.val[j]) {
                    out_of_bounds = true;
                    break;
                }
            }
            if (out_of_bounds) continue;

            // Obstacles
            if (obstacles.count(neighbor)) continue;

            // Body suffix at index >= 1
            auto it = body_min_index_ge_1.find(neighbor);
            if (it != body_min_index_ge_1.end() && it->second >= 1) {
                continue;
            }

            // Simulate next body
            std::deque<Coord> next_body = snake_body;
            next_body.push_front(neighbor);
            next_body.pop_back();

            int area = compute_reachable_area(neighbor, bounds, obstacles, next_body, 150);
            if (area > best_area) {
                best_area = area;
                best_neighbor = neighbor;
                found_any = true;
            }
        }
    }

    if (found_any) {
        return best_neighbor;
    }
    return {};
}

void output_move(const Coord& from, const Coord& to) {
    for (int i = 0; i < G_D; ++i) {
        if (to.val[i] != from.val[i]) {
            int dir = to.val[i] - from.val[i];
            std::cout << i << " " << dir << "\n";
            std::cout.flush();
            return;
        }
    }
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    int max_turns;
    if (!(std::cin >> G_D >> max_turns)) return 0;

    Coord bounds;
    for (int i = 0; i < G_D; ++i) {
        std::cin >> bounds.val[i];
    }

    int snake_length;
    std::cin >> snake_length;

    Coord start_pos;
    for (int i = 0; i < G_D; ++i) {
        std::cin >> start_pos.val[i];
    }

    int F;
    std::cin >> F;
    std::unordered_set<Coord, CoordHash> food_set;
    for (int i = 0; i < F; ++i) {
        Coord food;
        for (int j = 0; j < G_D; ++j) {
            std::cin >> food.val[j];
        }
        food_set.insert(food);
    }

    int O;
    std::cin >> O;
    std::unordered_set<Coord, CoordHash> obstacles;
    for (int i = 0; i < O; ++i) {
        Coord obs;
        for (int j = 0; j < G_D; ++j) {
            std::cin >> obs.val[j];
        }
        obstacles.insert(obs);
    }

    std::deque<Coord> snake_body(snake_length, start_pos);

    // Handle initial food under snake head
    if (food_set.count(start_pos)) {
        food_set.erase(start_pos);
        snake_body.push_back(start_pos);
    }

    AStarPool pool;

    std::vector<Coord> current_path;
    int path_index = 0;
    int turns_played = 0;

    while (turns_played < max_turns && !food_set.empty()) {
        Coord next_coord;
        bool found_move = false;

        // 1. Follow current path if valid
        if (path_index < (int)current_path.size()) {
            next_coord = current_path[path_index];
            if (is_immediately_valid(next_coord, snake_body, bounds, obstacles)) {
                path_index++;
                found_move = true;
            } else {
                current_path.clear();
                path_index = 0;
            }
        }

        // 2. Plan a path to the nearest food using multi-target A*
        if (!found_move) {
            std::vector<Coord> path = find_path_multi(snake_body.front(), food_set, bounds, obstacles, snake_body, pool);
            if (!path.empty()) {
                current_path = path;
                path_index = 1;
                next_coord = current_path[path_index++];
                found_move = true;
            }
        }

        // 3. Fallback to BFS for any reachable food
        if (!found_move) {
            std::vector<Coord> path = find_any_reachable_path(snake_body.front(), food_set, bounds, obstacles, snake_body, pool);
            if (!path.empty()) {
                current_path = path;
                path_index = 1;
                next_coord = current_path[path_index++];
                found_move = true;
            }
        }

        // 4. Survival fallback
        if (!found_move) {
            next_coord = get_survival_move(snake_body.front(), bounds, obstacles, snake_body);
            if (next_coord == Coord{}) {
                // Completely trapped
                break;
            }
            found_move = true;
        }

        // Output the move
        output_move(snake_body.front(), next_coord);

        // Update state
        bool ate_food = food_set.count(next_coord);
        if (ate_food) {
            food_set.erase(next_coord);
            current_path.clear();
            path_index = 0;
        }

        snake_body.push_front(next_coord);
        if (!ate_food) {
            snake_body.pop_back();
        }

        turns_played++;
    }

    return 0;
}
