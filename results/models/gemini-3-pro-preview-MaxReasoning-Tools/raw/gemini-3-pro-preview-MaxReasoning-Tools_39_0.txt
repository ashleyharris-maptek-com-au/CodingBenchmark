use std::io::{self, BufRead};
use std::time::Instant;

struct SimpleRng {
    state: u64,
}

impl SimpleRng {
    fn new(seed: u64) -> Self {
        SimpleRng { state: seed }
    }

    fn next_u64(&mut self) -> u64 {
        let mut x = self.state;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.state = x;
        x
    }

    fn next_usize(&mut self, limit: usize) -> usize {
        (self.next_u64() as usize) % limit
    }

    fn shuffle<T>(&mut self, slice: &mut [T]) {
        for i in (1..slice.len()).rev() {
            let j = self.next_usize(i + 1);
            slice.swap(i, j);
        }
    }
}

#[derive(Clone, Debug)]
struct Edge {
    to: usize,
    rev: usize,
    cap: i32,
    flow: i32,
}

struct Dinic {
    n: usize,
    graph: Vec<Vec<Edge>>,
    level: Vec<i32>,
    ptr: Vec<usize>,
}

impl Dinic {
    fn new(n: usize) -> Self {
        Dinic {
            n,
            graph: vec![vec![]; n],
            level: vec![0; n],
            ptr: vec![0; n],
        }
    }

    fn add_edge(&mut self, from: usize, to: usize, cap: i32) {
        let from_len = self.graph[from].len();
        let to_len = self.graph[to].len();
        self.graph[from].push(Edge {
            to,
            rev: to_len,
            cap,
            flow: 0,
        });
        self.graph[to].push(Edge {
            to: from,
            rev: from_len,
            cap: 0,
            flow: 0,
        });
    }

    fn add_undirected_edge(&mut self, u: usize, v: usize, cap: i32) {
        let u_len = self.graph[u].len();
        let v_len = self.graph[v].len();
        self.graph[u].push(Edge {
            to: v,
            rev: v_len,
            cap,
            flow: 0,
        });
        self.graph[v].push(Edge {
            to: u,
            rev: u_len,
            cap,
            flow: 0,
        });
    }

    fn bfs(&mut self, s: usize, t: usize) -> bool {
        self.level.fill(-1);
        self.level[s] = 0;
        let mut queue = std::collections::VecDeque::new();
        queue.push_back(s);
        while let Some(v) = queue.pop_front() {
            for edge in &self.graph[v] {
                if edge.cap - edge.flow > 0 && self.level[edge.to] == -1 {
                    self.level[edge.to] = self.level[v] + 1;
                    queue.push_back(edge.to);
                }
            }
        }
        self.level[t] != -1
    }

    fn dfs(&mut self, v: usize, t: usize, pushed: i32) -> i32 {
        if pushed == 0 {
            return 0;
        }
        if v == t {
            return pushed;
        }
        for cid in self.ptr[v]..self.graph[v].len() {
            self.ptr[v] = cid;
            let edge = &self.graph[v][cid];
            let tr = edge.to;
            if self.level[v] + 1 != self.level[tr] || edge.cap - edge.flow == 0 {
                continue;
            }
            let tr_flow = edge.cap - edge.flow;
            let push = self.dfs(tr, t, pushed.min(tr_flow));
            if push == 0 {
                continue;
            }
            self.graph[v][cid].flow += push;
            let rev = self.graph[v][cid].rev;
            self.graph[tr][rev].flow -= push;
            return push;
        }
        0
    }

    fn max_flow(&mut self, s: usize, t: usize) -> i32 {
        let mut flow = 0;
        while self.bfs(s, t) {
            self.ptr.fill(0);
            loop {
                let pushed = self.dfs(s, t, i32::MAX);
                if pushed == 0 {
                    break;
                }
                flow += pushed;
            }
        }
        flow
    }
}

fn run_kernighan_lin(
    n: usize,
    adj: &Vec<Vec<usize>>,
    has_edge: &Vec<Vec<bool>>,
    best_cut: &mut i32,
    best_partition: &mut Vec<bool>,
) {
    let mut rng = SimpleRng::new(42);
    let num_restarts = 100;
    let mut vertices: Vec<usize> = (0..n).collect();

    for _ in 0..num_restarts {
        rng.shuffle(&mut vertices);
        let mut current_partition = vec![false; n];
        for i in 0..n / 2 {
            current_partition[vertices[i]] = true;
        }

        loop {
            let mut d = vec![0; n];
            for u in 0..n {
                let mut ext = 0;
                let mut int = 0;
                let u_part = current_partition[u];
                for &v in &adj[u] {
                    if current_partition[v] == u_part {
                        int += 1;
                    } else {
                        ext += 1;
                    }
                }
                d[u] = ext - int;
            }

            let mut swapped = vec![false; n];
            let mut swaps = Vec::with_capacity(n / 2);
            let mut temp_partition = current_partition.clone();

            for _ in 0..n / 2 {
                let mut max_gain = i32::MIN;
                let mut best_a = usize::MAX;
                let mut best_b = usize::MAX;

                for a in 0..n {
                    if temp_partition[a] && !swapped[a] {
                        for b in 0..n {
                            if !temp_partition[b] && !swapped[b] {
                                let c_ab = if has_edge[a][b] { 1 } else { 0 };
                                let gain = d[a] + d[b] - 2 * c_ab;
                                if gain > max_gain {
                                    max_gain = gain;
                                    best_a = a;
                                    best_b = b;
                                }
                            }
                        }
                    }
                }

                if best_a == usize::MAX || best_b == usize::MAX {
                    break;
                }

                temp_partition[best_a] = false;
                temp_partition[best_b] = true;
                swapped[best_a] = true;
                swapped[best_b] = true;
                swaps.push((best_a, best_b, max_gain));

                for x in 0..n {
                    if !swapped[x] {
                        let c_xa = if has_edge[x][best_a] { 1 } else { 0 };
                        let c_xb = if has_edge[x][best_b] { 1 } else { 0 };
                        if temp_partition[x] {
                            d[x] += 2 * c_xa - 2 * c_xb;
                        } else {
                            d[x] += 2 * c_xb - 2 * c_xa;
                        }
                    }
                }
            }

            let mut cumulative_gain = 0;
            let mut max_cum_gain = i32::MIN;
            let mut best_k = 0;

            for k in 0..swaps.len() {
                cumulative_gain += swaps[k].2;
                if cumulative_gain > max_cum_gain {
                    max_cum_gain = cumulative_gain;
                    best_k = k + 1;
                }
            }

            if max_cum_gain > 0 {
                for k in 0..best_k {
                    let (a, b, _) = swaps[k];
                    current_partition[a] = false;
                    current_partition[b] = true;
                }
            } else {
                break;
            }
        }

        let mut cut = 0;
        for u in 0..n {
            if current_partition[u] {
                for &v in &adj[u] {
                    if !current_partition[v] {
                        cut += 1;
                    }
                }
            }
        }


        if cut < *best_cut {
            *best_cut = cut;
            *best_partition = current_partition;
        }
    }
}

fn get_static_order(n: usize, adj: &Vec<Vec<usize>>) -> Vec<usize> {
    let mut order = Vec::with_capacity(n);
    let mut visited = vec![false; n];

    let mut start_v = 0;
    let mut max_deg = 0;
    for i in 0..n {
        if adj[i].len() > max_deg {
            max_deg = adj[i].len();
            start_v = i;
        }
    }

    order.push(start_v);
    visited[start_v] = true;

    while order.len() < n {
        let mut best_v = usize::MAX;
        let mut max_visited_neigh = -1;
        let mut max_total_deg = -1;

        for u in 0..n {
            if !visited[u] {
                let mut visited_neigh = 0;
                for &v in &adj[u] {
                    if visited[v] {
                        visited_neigh += 1;
                    }
                }
                let total_deg = adj[u].len() as i32;
                if visited_neigh > max_visited_neigh
                    || (visited_neigh == max_visited_neigh && total_deg > max_total_deg)
                {
                    max_visited_neigh = visited_neigh;
                    max_total_deg = total_deg;
                    best_v = u;
                }
            }
        }

        order.push(best_v);
        visited[best_v] = true;
    }
    order
}

fn check_lower_bound(
    depth: usize,
    count_A: usize,
    count_B: usize,
    current_cut: i32,
    pos: &Vec<usize>,
    deg_A: &Vec<i32>,
    deg_B: &Vec<i32>,
    order: &Vec<usize>,
    adj: &Vec<Vec<usize>>,
    best_cut: i32,
    n: usize,
) -> bool {
    let a_rem = n / 2 - count_A;
    let b_rem = n / 2 - count_B;

    let mut sum_deg_A = 0;
    let mut diffs = Vec::with_capacity(n - depth);

    for i in depth..n {
        let v = order[i];
        sum_deg_A += deg_A[v];
        diffs.push(deg_B[v] - deg_A[v]);
    }

    diffs.sort_unstable();

    let mut sum_diffs = 0;
    for i in 0..a_rem {
        sum_diffs += diffs[i];
    }

    let lb_deg = current_cut + sum_deg_A + sum_diffs;
    if lb_deg >= best_cut {
        return false;
    }

    if count_A > 0 && count_B > 0 {
        let mut dinic = Dinic::new(n + 2);
        let s = n;
        let t = n + 1;

        for i in 0..n {
            if pos[i] == 0 {
                dinic.add_edge(s, i, 1_000_000);
            } else if pos[i] == 1 {
                dinic.add_edge(i, t, 1_000_000);
            }

            for &v in &adj[i] {
                if i < v {
                    dinic.add_undirected_edge(i, v, 1);
                }
            }
        }

        let lb_mincut = dinic.max_flow(s, t);
        if lb_mincut >= best_cut {
            return false;
        }
    }

    true
}

fn dfs(
    depth: usize,
    count_A: usize,
    count_B: usize,
    current_cut: i32,
    pos: &mut Vec<usize>,
    deg_A: &mut Vec<i32>,
    deg_B: &mut Vec<i32>,
    order: &Vec<usize>,
    adj: &Vec<Vec<usize>>,
    best_cut: &mut i32,
    best_partition: &mut Vec<bool>,
    n: usize,
) {
    if depth == n {
        if current_cut < *best_cut {
            *best_cut = current_cut;
            for i in 0..n {
                best_partition[i] = pos[i] == 0;
            }
        }
        return;
    }

    let u = order[depth];

    let try_A_first = deg_A[u] >= deg_B[u];
    let branches = if try_A_first { [0, 1] } else { [1, 0] };

    for &p in &branches {
        if p == 0 {
            if count_A < n / 2 {
                let new_cut = current_cut + deg_B[u];
                
                pos[u] = 0;
                for &v in &adj[u] {
                    deg_A[v] += 1;
                }

                if check_lower_bound(
                    depth + 1,
                    count_A + 1,
                    count_B,
                    new_cut,
                    pos,
                    deg_A,
                    deg_B,
                    order,
                    adj,
                    *best_cut,
                    n,
                ) {
                    dfs(
                        depth + 1,
                        count_A + 1,
                        count_B,
                        new_cut,
                        pos,
                        deg_A,
                        deg_B,
                        order,
                        adj,
                        best_cut,
                        best_partition,
                        n,
                    );
                }

                for &v in &adj[u] {
                    deg_A[v] -= 1;
                }
                pos[u] = 2;
            }
        } else {
            if count_B < n / 2 {
                let new_cut = current_cut + deg_A[u];

                pos[u] = 1;
                for &v in &adj[u] {
                    deg_B[v] += 1;
                }

                if check_lower_bound(
                    depth + 1,
                    count_A,
                    count_B + 1,
                    new_cut,
                    pos,
                    deg_A,
                    deg_B,
                    order,
                    adj,
                    *best_cut,
                    n,
                ) {
                    dfs(
                        depth + 1,
                        count_A,
                        count_B + 1,
                        new_cut,
                        pos,
                        deg_A,
                        deg_B,
                        order,
                        adj,
                        best_cut,
                        best_partition,
                        n,
                    );
                }

                for &v in &adj[u] {
                    deg_B[v] -= 1;
                }
                pos[u] = 2;
            }
        }
    }
}

fn main() {
    // Generate a random graph to test correctness and performance
    let n = 50; // even number
    let mut rng = SimpleRng::new(12345);
    let mut adj = vec![vec![]; n];
    let mut has_edge = vec![vec![false; n]; n];

    // Erdos-Renyi graph G(n, p) with p = 0.2
    for i in 0..n {
        for j in i + 1..n {
            if rng.next_usize(100) < 15 {
                adj[i].push(j);
                adj[j].push(i);
                has_edge[i][j] = true;
                has_edge[j][i] = true;
            }
        }
    }

    println!("Graph generated with {} vertices.", n);
    let mut m = 0;
    for i in 0..n {
        m += adj[i].len();
    }
    println!("Edges count: {}", m / 2);

    let start_time = Instant::now();

    let mut best_cut = i32::MAX;
    let mut best_partition = vec![false; n];

    // Run Kernighan-Lin
    run_kernighan_lin(n, &adj, &has_edge, &mut best_cut, &mut best_partition);
    println!("KL found cut: {}", best_cut);

    // Prepare branch-and-bound
    let order = get_static_order(n, &adj);
    println!("Static order: {:?}", order);

    let mut pos = vec![2; n];
    let mut deg_A = vec![0; n];
    let mut deg_B = vec![0; n];

    // Fix first vertex in order to A (0)
    let first_v = order[0];
    pos[first_v] = 0;
    for &v in &adj[first_v] {
        deg_A[v] += 1;
    }

    dfs(
        1,
        1,
        0,
        0,
        &mut pos,
        &mut deg_A,
        &mut deg_B,
        &order,
        &adj,
        &mut best_cut,
        &mut best_partition,
        n,
    );

    println!("Exact min cut: {}", best_cut);
    let mut part_a = Vec::new();
    let mut part_b = Vec::new();
    for i in 0..n {
        if best_partition[i] {
            part_a.push(i);
        } else {
            part_b.push(i);
        }
    }
    println!("Part A size: {}, Part B size: {}", part_a.len(), part_b.len());
    
    let mut actual_cut = 0;
    for u in 0..n {
        for &v in &adj[u] {
            if u < v && best_partition[u] != best_partition[v] {
                actual_cut += 1;
            }
        }
    }
    println!("Directly computed cut: {}", actual_cut);
    println!("Time elapsed: {:?}", start_time.elapsed());
}
