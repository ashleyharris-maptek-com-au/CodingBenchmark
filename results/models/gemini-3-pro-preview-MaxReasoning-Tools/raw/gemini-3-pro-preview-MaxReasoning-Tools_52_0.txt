import math

def autopilot_step(state: dict, dt: float) -> dict:
    # --------------------------------------------------------------------------
    # 1. Helper function for median
    # --------------------------------------------------------------------------
    def median(lst):
        if not lst:
            return None
        n = len(lst)
        s = sorted(lst)
        if n % 2 == 1:
            return s[n // 2]
        else:
            return 0.5 * (s[n // 2 - 1] + s[n // 2])

    # --------------------------------------------------------------------------
    # 2. Speed Fusion
    # --------------------------------------------------------------------------
    speeds = []
    
    # Wheel speeds
    for wheel in ['fl', 'fr', 'rl', 'rr']:
        val = state.get(f'wheel_speed_{wheel}_kmh')
        valid = state.get(f'wheel_speed_{wheel}_valid')
        if valid and val is not None:
            speeds.append(val)
            
    # GPS speeds
    for gps in [1, 2]:
        val = state.get(f'gps_{gps}_speed_kmh')
        valid = state.get(f'gps_{gps}_valid')
        if valid and val is not None:
            speeds.append(val)
            
    # Vehicle speeds
    for veh in [1, 2]:
        val = state.get(f'vehicle_speed_{veh}_kmh')
        if val is not None:
            speeds.append(val)
            
    if speeds:
        nonzero_speeds = [s for s in speeds if s > 1.0]
        if len(nonzero_speeds) >= len(speeds) // 2:
            estimated_speed_kmh = median(nonzero_speeds)
        else:
            estimated_speed_kmh = median(speeds)
    else:
        estimated_speed_kmh = 110.0
        
    speed_m_s = estimated_speed_kmh / 3.6

    # --------------------------------------------------------------------------
    # 3. Accelerometer Fusion
    # --------------------------------------------------------------------------
    lat_accels = []
    long_accels = []
    for i in range(1, 4):
        valid = state.get(f'accel_{i}_valid')
        lat_val = state.get(f'accel_{i}_lateral_ms2')
        long_val = state.get(f'accel_{i}_longitudinal_ms2')
        if valid:
            if lat_val is not None:
                lat_accels.append(lat_val)
            if long_val is not None:
                long_accels.append(long_val)
                
    fused_lateral_accel = median(lat_accels) if lat_accels else 0.0
    fused_longitudinal_accel = median(long_accels) if long_accels else 0.0

    # --------------------------------------------------------------------------
    # 4. Gyro and Steering Sensor Fusion
    # --------------------------------------------------------------------------
    yaw_rate_accel_dps = (fused_lateral_accel / max(speed_m_s, 1.0)) * (180.0 / math.pi)
    
    gyro_candidates = []
    for i in range(1, 4):
        valid = state.get(f'gyro_{i}_valid')
        val = state.get(f'gyro_{i}_yaw_rate_dps')
        if valid and val is not None:
            gyro_candidates.append(val)
            
    if not gyro_candidates:
        fused_gyro_yaw_rate_dps = yaw_rate_accel_dps
    else:
        best_gyro = gyro_candidates[0]
        min_diff = abs(best_gyro - yaw_rate_accel_dps)
        for val in gyro_candidates[1:]:
            diff = abs(val - yaw_rate_accel_dps)
            if diff < min_diff:
                min_diff = diff
                best_gyro = val
        fused_gyro_yaw_rate_dps = best_gyro

    yaw_rate_rad_s = fused_gyro_yaw_rate_dps * (math.pi / 180.0)
    expected_steer_deg = math.atan2(yaw_rate_rad_s * 2.7, speed_m_s) * (180.0 / math.pi)
    
    steer1 = state.get('steering_angle_1_deg')
    steer2 = state.get('steering_angle_2_deg')
    steer1_valid = state.get('steering_angle_1_valid')
    steer2_valid = state.get('steering_angle_2_valid')
    
    if steer1_valid and steer2_valid and steer1 is not None and steer2 is not None:
        if abs(steer1 - steer2) > 2.0:
            if abs(steer1 - expected_steer_deg) < abs(steer2 - expected_steer_deg):
                steer_deg = steer1
            else:
                steer_deg = steer2
        else:
            steer_deg = 0.5 * (steer1 + steer2)
    elif steer1_valid and steer1 is not None:
        steer_deg = steer1
    elif steer2_valid and steer2 is not None:
        steer_deg = steer2
    else:
        steer_deg = expected_steer_deg

    # --------------------------------------------------------------------------
    # 5. Yaw Angle Fusion (for Lidar projection)
    # --------------------------------------------------------------------------
    yaws = []
    for i in [1, 2]:
        if state.get(f'gps_{i}_valid') and state.get(f'gps_{i}_heading_deg') is not None:
            yaws.append(state[f'gps_{i}_heading_deg'])
            
    if yaws:
        fused_yaw_rad = median(yaws) * (math.pi / 180.0)
    else:
        if not hasattr(autopilot_step, 'fused_yaw_rad'):
            autopilot_step.fused_yaw_rad = 0.0
        autopilot_step.fused_yaw_rad += fused_gyro_yaw_rate_dps * (math.pi / 180.0) * dt
        autopilot_step.fused_yaw_rad = max(-0.5, min(autopilot_step.fused_yaw_rad, 0.5))
        fused_yaw_rad = autopilot_step.fused_yaw_rad

    if yaws:
        autopilot_step.fused_yaw_rad = fused_yaw_rad

    # --------------------------------------------------------------------------
    # 6. Vision Fusion (with persistence)
    # --------------------------------------------------------------------------
    if not hasattr(autopilot_step, 'fused_lane_offset'):
        autopilot_step.fused_lane_offset = 0.0
        autopilot_step.fused_curvature = 0.0
        autopilot_step.fused_speed_limit = 110.0
        
    offsets = []
    curvatures = []
    speed_limits = []
    
    for i in range(1, 4):
        offset = state.get(f'vision_{i}_lane_offset_m')
        curvature = state.get(f'vision_{i}_road_curvature_1pm')
        limit = state.get(f'vision_{i}_speed_limit_kmh')
        limit_valid = state.get(f'vision_{i}_speed_limit_valid')
        
        if offset is not None and abs(offset) < 5.0:
            offsets.append(offset)
        if curvature is not None:
            curvatures.append(curvature)
        if limit_valid and limit is not None:
            speed_limits.append(limit)
            
    if offsets:
        autopilot_step.fused_lane_offset = median(offsets)
    if curvatures:
        autopilot_step.fused_curvature = median(curvatures)
    if speed_limits:
        autopilot_step.fused_speed_limit = median(speed_limits)
        
    fused_lane_offset = autopilot_step.fused_lane_offset
    fused_curvature = autopilot_step.fused_curvature
    fused_speed_limit = autopilot_step.fused_speed_limit

    # --------------------------------------------------------------------------
    # 7. Obstacle Map Construction (Lidar Occupancy Grid)
    # --------------------------------------------------------------------------
    # Initialize lane obstacles to max range
    lane_obstacles = {0: 150.0, 1: 150.0, 2: 150.0}
    y_est = (state.get('current_lane', 1) + 0.5) * 3.7 + fused_lane_offset
    
    for angle in [-90, -60, -45, -30, -15, -5, 5, 15, 30, 45, 60, 90]:
        val = state.get(f'lidar_{angle:+04d}_deg_range_m')
        valid = state.get(f'lidar_{angle:+04d}_deg_valid')
        if valid and val is not None and val < 145.0:
            angle_rad = angle * (math.pi / 180.0)
            total_angle = fused_yaw_rad + angle_rad
            dy = val * math.sin(total_angle)
            y_hit = y_est + dy
            # Ignore hits that are outside the road boundaries (guardrails / boundaries)
            if 0.5 < y_hit < 10.6:
                lane_hit = int(y_hit / 3.7)
                lane_hit = max(0, min(2, lane_hit))
                dx = val * math.cos(total_angle)
                if dx > 0.0:
                    if dx < lane_obstacles[lane_hit]:
                        lane_obstacles[lane_hit] = dx

    # --------------------------------------------------------------------------
    # 8. Vision Obstacle Fusion
    # --------------------------------------------------------------------------
    vision_dists = []
    vision_rel_speeds = []
    for i in range(1, 4):
        detected = state.get(f'vision_{i}_obstacle_detected')
        valid = state.get(f'vision_{i}_obstacle_valid')
        dist = state.get(f'vision_{i}_obstacle_dist_m')
        rel_speed = state.get(f'vision_{i}_obstacle_rel_speed_kmh')
        if detected and valid:
            if dist is not None and dist < 149.0:
                vision_dists.append(dist)
            if rel_speed is not None:
                vision_rel_speeds.append(rel_speed)
                
    fused_vision_dist = median(vision_dists) if vision_dists else 150.0
    fused_vision_rel_speed = median(vision_rel_speeds) if vision_rel_speeds else 0.0

    current_lane = state.get('current_lane', 1)
    obstacle_dist = min(fused_vision_dist, lane_obstacles[current_lane])
    obstacle_detected = (obstacle_dist < 130.0)

    # --------------------------------------------------------------------------
    # 9. Lane Selector & Active Avoidance
    # --------------------------------------------------------------------------
    if not hasattr(autopilot_step, 'my_target_lane'):
        autopilot_step.my_target_lane = state.get('current_lane', 1)
        autopilot_step.lane_change_timer = 0.0
        
    autopilot_step.lane_change_timer += dt
    
    # We want to choose the best lane if our lane is blocked
    if obstacle_detected and obstacle_dist < 85.0 and autopilot_step.lane_change_timer > 3.0:
        best_lane = current_lane
        max_dist = obstacle_dist
        
        for L in [current_lane - 1, current_lane + 1]:
            if 0 <= L <= 2:
                # Adjacent lane must have enough clear space to be safe
                if lane_obstacles[L] > 50.0:
                    if lane_obstacles[L] > max_dist + 15.0:
                        max_dist = lane_obstacles[L]
                        best_lane = L
                        
        if best_lane != current_lane:
            autopilot_step.my_target_lane = best_lane
            autopilot_step.lane_change_timer = 0.0

    # --------------------------------------------------------------------------
    # 10. Steering Jam Detection
    # --------------------------------------------------------------------------
    if not hasattr(autopilot_step, 'prev_steering_cmd'):
        autopilot_step.prev_steering_cmd = 0.0
    if not hasattr(autopilot_step, 'steer_command_history'):
        autopilot_step.steer_command_history = []
        autopilot_step.steer_actual_history = []
        autopilot_step.is_steer_jammed = False
        
    autopilot_step.steer_command_history.append(autopilot_step.prev_steering_cmd * 35.0)
    autopilot_step.steer_actual_history.append(steer_deg)
    if len(autopilot_step.steer_command_history) > 50:
        autopilot_step.steer_command_history.pop(0)
        autopilot_step.steer_actual_history.pop(0)
        
    if len(autopilot_step.steer_command_history) == 50:
        jammed = True
        for cmd, act in zip(autopilot_step.steer_command_history, autopilot_step.steer_actual_history):
            if abs(cmd - act) < 4.5:
                jammed = False
                break
        if jammed:
            autopilot_step.is_steer_jammed = True

    # --------------------------------------------------------------------------
    # 11. Lateral Control (Steering PD/PID)
    # --------------------------------------------------------------------------
    target_lane = autopilot_step.my_target_lane
    lateral_error = fused_lane_offset + (current_lane - target_lane) * 3.7
    
    if not hasattr(autopilot_step, 'prev_lateral_error'):
        autopilot_step.prev_lateral_error = 0.0
        autopilot_step.error_deriv_filtered = 0.0
        autopilot_step.lateral_error_integral = 0.0
        
    lateral_error_deriv = (lateral_error - autopilot_step.prev_lateral_error) / dt if dt > 0 else 0.0
    autopilot_step.prev_lateral_error = lateral_error
    
    autopilot_step.error_deriv_filtered = 0.85 * autopilot_step.error_deriv_filtered + 0.15 * lateral_error_deriv
    
    autopilot_step.lateral_error_integral += lateral_error * dt
    autopilot_step.lateral_error_integral = max(-1.5, min(autopilot_step.lateral_error_integral, 1.5))
    
    steer_ff = -fused_curvature * 2.7 / 0.61
    
    Kp_steer = 0.02
    Ki_steer = 0.005
    Kd_steer = 0.10
    
    steering_cmd = -Kp_steer * lateral_error - Ki_steer * autopilot_step.lateral_error_integral - Kd_steer * autopilot_step.error_deriv_filtered + steer_ff
    
    estimated_mu = state.get('estimated_mu', 1.0)
    steering_scale = max(0.2, estimated_mu)
    steering_cmd *= steering_scale
    
    steering_cmd = max(-1.0, min(steering_cmd, 1.0))
    autopilot_step.prev_steering_cmd = steering_cmd

    # --------------------------------------------------------------------------
    # 12. Speed Control
    # --------------------------------------------------------------------------
    target_speed_kmh = state.get('target_speed_kmh', 110)
    if fused_speed_limit is not None:
        target_speed_kmh = min(target_speed_kmh, fused_speed_limit)
        
    if estimated_mu < 0.25:
        target_speed_kmh = min(target_speed_kmh, 45.0)
    elif estimated_mu < 0.55:
        target_speed_kmh = min(target_speed_kmh, 70.0)
        
    v_target = target_speed_kmh / 3.6
    
    if obstacle_detected:
        rel_speed_m_s = fused_vision_rel_speed / 3.6
        obstacle_speed_m_s = speed_m_s - rel_speed_m_s
        desired_gap = max(15.0, speed_m_s * 1.8 + 6.0)
        
        if obstacle_dist < 8.0:
            v_desired = 0.0
        else:
            v_desired = obstacle_speed_m_s + 0.35 * (obstacle_dist - desired_gap)
            v_desired = max(0.0, min(v_desired, v_target))
    else:
        v_desired = v_target
        
    if not hasattr(autopilot_step, 'speed_integral'):
        autopilot_step.speed_integral = 0.0
        autopilot_step.prev_speed = 0.0
        
    speed_error = v_desired - speed_m_s
    autopilot_step.speed_integral += speed_error * dt
    autopilot_step.speed_integral = max(-15.0, min(autopilot_step.speed_integral, 15.0))
    
    f_resist = 0.4312 * (v_desired ** 2) + 235.0
    throttle_ff = f_resist / (7000.0 * max(estimated_mu, 0.15))
    throttle_ff = max(0.0, min(throttle_ff, 0.45))
    
    Kp_speed = 0.16
    Ki_speed = 0.04
    Kd_speed = 0.04
    
    speed_deriv = (speed_m_s - autopilot_step.prev_speed) / dt if dt > 0 else 0.0
    autopilot_step.prev_speed = speed_m_s
    
    speed_out = Kp_speed * speed_error + Ki_speed * autopilot_step.speed_integral - Kd_speed * speed_deriv + throttle_ff
    
    if speed_out > 0:
        throttle_cmd = max(0.0, min(speed_out, 1.0))
        brake_cmd = 0.0
    else:
        throttle_cmd = 0.0
        brake_cmd = max(0.0, min(-speed_out * 0.45, 1.0))
        
    ttc = obstacle_dist / max(fused_vision_rel_speed / 3.6, 0.1) if (fused_vision_rel_speed > 0) else 999.0
    if obstacle_detected and (obstacle_dist < 15.0 or (obstacle_dist < 55.0 and ttc < 2.0)):
        throttle_cmd = 0.0
        brake_cmd = 1.0

    max_brake = max(0.12, estimated_mu * 0.5)
    brake_cmd = min(brake_cmd, max_brake)
    
    max_throttle = max(0.15, estimated_mu)
    throttle_cmd = min(throttle_cmd, max_throttle)

    # --------------------------------------------------------------------------
    # 13. Command Overrides
    # --------------------------------------------------------------------------
    if autopilot_step.is_steer_jammed:
        throttle_cmd = 0.0
        brake_cmd = 1.0
        
    return {
        'steering': steering_cmd,
        'throttle': throttle_cmd,
        'brake': brake_cmd
    }
