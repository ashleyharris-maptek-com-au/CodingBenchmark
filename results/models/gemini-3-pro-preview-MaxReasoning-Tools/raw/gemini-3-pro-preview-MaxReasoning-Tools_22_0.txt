using System;
using System.IO;
using System.Text;

public struct Point {
    public int X;
    public int Y;
    public Point(int x, int y) { X = x; Y = y; }
}

public class Piece {
    public Point[] Blocks;
    public int Width;
    public int Height;
    public ulong[] RowMasks;

    public Piece(Point[] blocks) {
        Blocks = blocks;
        int minX = int.MaxValue, maxX = int.MinValue;
        int minY = int.MaxValue, maxY = int.MinValue;
        foreach (var b in blocks) {
            if (b.X < minX) minX = b.X;
            if (b.X > maxX) maxX = b.X;
            if (b.Y < minY) minY = b.Y;
            if (b.Y > maxY) maxY = b.Y;
        }
        for (int i = 0; i < blocks.Length; i++) {
            blocks[i] = new Point(blocks[i].X - minX, blocks[i].Y - minY);
        }
        Width = maxX - minX + 1;
        Height = maxY - minY + 1;
        RowMasks = new ulong[Height];
        foreach (var b in blocks) {
            RowMasks[b.Y] |= (1UL << b.X);
        }
    }
}

public class Board {
    public ulong[] Rows;
    public int[] Heights;
    public int Width;
    public int Height;

    public Board(int width, int height) {
        Width = width;
        Height = height;
        Rows = new ulong[height];
        Heights = new int[width];
    }

    public void CopyFrom(Board other) {
        Array.Copy(other.Rows, Rows, Height);
        Array.Copy(other.Heights, Heights, Width);
    }

    public int GetLandingRow(Piece piece, int col) {
        int yLand = 0;
        for (int i = 0; i < piece.Blocks.Length; i++) {
            int c = col + piece.Blocks[i].X;
            int h = Heights[c] - piece.Blocks[i].Y;
            if (h > yLand) {
                yLand = h;
            }
        }
        return yLand;
    }

    public (int linesCleared, int cellsEliminated) PlacePiece(Piece piece, int col, int yLand) {
        int cellsEliminated = 0;
        int linesCleared = 0;

        for (int i = 0; i < piece.Blocks.Length; i++) {
            int r = yLand + piece.Blocks[i].Y;
            int c = col + piece.Blocks[i].X;
            if (r < Height) {
                Rows[r] |= (1UL << c);
            }
        }

        // Recompute heights
        for (int c = 0; c < Width; c++) {
            int h = 0;
            for (int r = Height - 1; r >= 0; r--) {
                if ((Rows[r] & (1UL << c)) != 0) {
                    h = r + 1;
                    break;
                }
            }
            Heights[c] = h;
        }

        ulong fullMask = (1UL << Width) - 1;
        int writeIdx = 0;
        for (int r = 0; r < Height; r++) {
            if (Rows[r] == fullMask) {
                linesCleared++;
                for (int i = 0; i < piece.Blocks.Length; i++) {
                    if (yLand + piece.Blocks[i].Y == r) {
                        cellsEliminated++;
                    }
                }
            } else {
                Rows[writeIdx] = Rows[r];
                writeIdx++;
            }
        }
        for (int r = writeIdx; r < Height; r++) {
            Rows[r] = 0;
        }

        for (int c = 0; c < Width; c++) {
            int h = 0;
            for (int r = Height - 1; r >= 0; r--) {
                if ((Rows[r] & (1UL << c)) != 0) {
                    h = r + 1;
                    break;
                }
            }
            Heights[c] = h;
        }

        return (linesCleared, cellsEliminated);
    }
}

public static class Heuristic {
    // Normal / Marathon Weights (Dellacherie / El-Tetris)
    public static double LandingHeightWeight = -4.500158825082766;
    public static double ErodedPiecesWeight = 3.4181268101392694;
    public static double RowTransitionsWeight = -3.2178882868487753;
    public static double ColumnTransitionsWeight = -9.348695305445199;
    public static double HolesWeight = -7.899265427351652;
    public static double WellSumsWeight = -3.3855972247263626;

    // Survival Mode Weights (Aggressive stack height reduction)
    public static double LandingHeightWeightSurvival = -9.5;
    public static double ErodedPiecesWeightSurvival = 5.0;
    public static double RowTransitionsWeightSurvival = -2.5;
    public static double ColumnTransitionsWeightSurvival = -7.5;
    public static double HolesWeightSurvival = -15.0;
    public static double WellSumsWeightSurvival = -2.0;

    // Narrow Board Weights (Optimized for Width < 8)
    public static double LandingHeightWeightNarrow = -3.5;
    public static double ErodedPiecesWeightNarrow = 4.0;
    public static double RowTransitionsWeightNarrow = -4.5;
    public static double ColumnTransitionsWeightNarrow = -12.0;
    public static double HolesWeightNarrow = -18.0;
    public static double WellSumsWeightNarrow = -4.0;

    private static readonly int[] WellDepthBuffer = new int[100];

    static int PopCount(ulong x) {
        x = x - ((x >> 1) & 0x5555555555555555UL);
        x = (x & 0x3333333333333333UL) + ((x >> 2) & 0x3333333333333333UL);
        x = (x + (x >> 4)) & 0x0F0F0F0F0F0F0F0FUL;
        return (int)((x * 0x0101010101010101UL) >> 56);
    }

    public static double Evaluate(Board board, double landingHeight, int linesCleared, int cellsEliminated, bool survival, bool narrow) {
        double wLandingHeight = LandingHeightWeight;
        double wErodedPieces = ErodedPiecesWeight;
        double wRowTransitions = RowTransitionsWeight;
        double wColumnTransitions = ColumnTransitionsWeight;
        double wHoles = HolesWeight;
        double wWellSums = WellSumsWeight;

        if (survival) {
            wLandingHeight = LandingHeightWeightSurvival;
            wErodedPieces = ErodedPiecesWeightSurvival;
            wRowTransitions = RowTransitionsWeightSurvival;
            wColumnTransitions = ColumnTransitionsWeightSurvival;
            wHoles = HolesWeightSurvival;
            wWellSums = WellSumsWeightSurvival;
        } else if (narrow) {
            wLandingHeight = LandingHeightWeightNarrow;
            wErodedPieces = ErodedPiecesWeightNarrow;
            wRowTransitions = RowTransitionsWeightNarrow;
            wColumnTransitions = ColumnTransitionsWeightNarrow;
            wHoles = HolesWeightNarrow;
            wWellSums = WellSumsWeightNarrow;
        }

        double erodedPieces = linesCleared * cellsEliminated;

        int rowTransitions = 0;
        ulong fullMask = (1UL << board.Width) - 1;
        for (int r = 0; r < board.Height; r++) {
            ulong row = board.Rows[r];
            ulong extRow = 1UL | ((row & fullMask) << 1) | (1UL << (board.Width + 1));
            ulong diff = extRow ^ (extRow >> 1);
            rowTransitions += PopCount(diff & ((1UL << (board.Width + 1)) - 1));
        }

        int colTransitions = 0;
        ulong lastRow = board.Rows[0];
        colTransitions += PopCount(lastRow ^ fullMask);
        for (int r = 1; r < board.Height; r++) {
            colTransitions += PopCount(board.Rows[r] ^ board.Rows[r - 1]);
        }
        colTransitions += PopCount(board.Rows[board.Height - 1]);

        int holes = 0;
        ulong covered = 0;
        for (int r = board.Height - 1; r >= 0; r--) {
            covered |= board.Rows[r];
            holes += PopCount(covered & ~board.Rows[r]);
        }

        int wellSum = 0;
        int[] depth = WellDepthBuffer;
        Array.Clear(depth, 0, board.Width);
        for (int r = board.Height - 1; r >= 0; r--) {
            ulong row = board.Rows[r];
            ulong leftNeigh = (row << 1) | 1UL;
            ulong rightNeigh = (row >> 1) | (1UL << (board.Width - 1));
            ulong rowWells = ~row & leftNeigh & rightNeigh & fullMask;

            for (int c = 0; c < board.Width; c++) {
                if ((rowWells & (1UL << c)) != 0) {
                    depth[c]++;
                    wellSum += depth[c];
                } else {
                    depth[c] = 0;
                }
            }
        }

        return landingHeight * wLandingHeight +
               erodedPieces * wErodedPieces +
               rowTransitions * wRowTransitions +
               colTransitions * wColumnTransitions +
               holes * wHoles +
               wellSum * wWellSums;
    }
}

class Program {
    static readonly Point[][] BasePieces = new Point[][] {
        new Point[] { new Point(0,0), new Point(1,0), new Point(2,0), new Point(3,0) }, // I
        new Point[] { new Point(0,0), new Point(1,0), new Point(0,1), new Point(1,1) }, // O
        new Point[] { new Point(0,0), new Point(1,0), new Point(2,0), new Point(1,1) }, // T
        new Point[] { new Point(0,0), new Point(1,0), new Point(1,1), new Point(2,1) }, // S
        new Point[] { new Point(1,0), new Point(2,0), new Point(0,1), new Point(1,1) }, // Z
        new Point[] { new Point(0,0), new Point(1,0), new Point(2,0), new Point(0,1) }, // J
        new Point[] { new Point(0,0), new Point(1,0), new Point(2,0), new Point(2,1) }  // L
    };

    static Piece[][] AllPieces = new Piece[7][];
    static Board[] TempBoards;

    static void InitializePieces() {
        for (int i = 0; i < 7; i++) {
            AllPieces[i] = GenerateRotations(BasePieces[i]);
        }
    }

    static Piece[] GenerateRotations(Point[] baseBlocks) {
        Piece[] rotations = new Piece[4];
        Point[] current = (Point[])baseBlocks.Clone();
        for (int r = 0; r < 4; r++) {
            rotations[r] = new Piece(current);
            Point[] next = new Point[current.Length];
            for (int i = 0; i < current.Length; i++) {
                next[i] = new Point(current[i].Y, -current[i].X);
            }
            current = next;
        }
        return rotations;
    }

    static void InitializeTempBoards(int width, int height) {
        TempBoards = new Board[4];
        for (int i = 0; i < TempBoards.Length; i++) {
            TempBoards[i] = new Board(width, height);
        }
    }

    static int GetPieceIndex(string type) {
        switch (type.ToUpper()) {
            case "I": return 0;
            case "O": return 1;
            case "T": return 2;
            case "S": return 3;
            case "Z": return 4;
            case "J": return 5;
            case "L": return 6;
            default: return 0;
        }
    }

    public static void FindBestMove(int pieceIdx, Board board, out int bestRot, out int bestCol) {
        Piece[] rotations = AllPieces[pieceIdx];
        double bestScore = double.NegativeInfinity;
        bestRot = 0;
        bestCol = 0;

        int maxColHeight = 0;
        for (int c = 0; c < board.Width; c++) {
            if (board.Heights[c] > maxColHeight) {
                maxColHeight = board.Heights[c];
            }
        }

        bool narrow = board.Width < 8;
        bool survival = maxColHeight >= (int)(board.Height * 0.70);
        int searchDepth = narrow ? (survival ? 2 : 3) : 2;

        Board tempBoard1 = TempBoards[0];

        for (int r = 0; r < rotations.Length; r++) {
            Piece piece = rotations[r];
            int maxCol = board.Width - piece.Width;
            for (int col = 0; col <= maxCol; col++) {
                int yLand = board.GetLandingRow(piece, col);
                if (yLand + piece.Height > board.Height) {
                    continue;
                }

                tempBoard1.CopyFrom(board);
                var (linesCleared, cellsEliminated) = tempBoard1.PlacePiece(piece, col, yLand);

                double landingHeight = yLand + (piece.Height - 1) / 2.0;
                double score = Heuristic.Evaluate(tempBoard1, landingHeight, linesCleared, cellsEliminated, survival, narrow);

                if (searchDepth > 1) {
                    double lookaheadScore = EvaluateLookahead(tempBoard1, searchDepth - 1, survival, narrow);
                    score += 0.9 * lookaheadScore;
                }

                if (score > bestScore) {
                    bestScore = score;
                    bestRot = r;
                    bestCol = col;
                }
            }
        }

        if (bestScore == double.NegativeInfinity) {
            bestRot = 0;
            bestCol = 0;
        }
    }

    private static double EvaluateLookahead(Board board, int depth, bool survival, bool narrow) {
        int boardIdx = 3 - depth;
        Board tempBoard = TempBoards[boardIdx];

        double sumMaxScores = 0;

        for (int nextPieceIdx = 0; nextPieceIdx < 7; nextPieceIdx++) {
            Piece[] rotations = AllPieces[nextPieceIdx];
            double maxPieceScore = double.NegativeInfinity;

            for (int r = 0; r < rotations.Length; r++) {
                Piece piece = rotations[r];
                int maxCol = board.Width - piece.Width;
                for (int col = 0; col <= maxCol; col++) {
                    int yLand = board.GetLandingRow(piece, col);
                    if (yLand + piece.Height > board.Height) {
                        continue;
                    }

                    tempBoard.CopyFrom(board);
                    var (linesCleared, cellsEliminated) = tempBoard.PlacePiece(piece, col, yLand);

                    double landingHeight = yLand + (piece.Height - 1) / 2.0;
                    double score = Heuristic.Evaluate(tempBoard, landingHeight, linesCleared, cellsEliminated, survival, narrow);

                    if (depth > 1) {
                        double subLookahead = EvaluateLookahead(tempBoard, depth - 1, survival, narrow);
                        score += 0.9 * subLookahead;
                    }

                    if (score > maxPieceScore) {
                        maxPieceScore = score;
                    }
                }
            }

            if (maxPieceScore == double.NegativeInfinity) {
                maxPieceScore = -999999;
            }

            sumMaxScores += maxPieceScore;
        }

        return sumMaxScores / 7.0;
    }

    static string ReadNonEmptyLine() {
        while (true) {
            string line = Console.ReadLine();
            if (line == null) return null;
            line = line.Trim();
            if (line.Length > 0) return line;
        }
    }

    static void Main() {
        InitializePieces();

        string firstLine = ReadNonEmptyLine();
        if (firstLine == null) return;

        string[] tokens = firstLine.Split(new char[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        if (tokens.Length < 3) return;

        int width = int.Parse(tokens[0]);
        int height = int.Parse(tokens[1]);
        int numPieces = int.Parse(tokens[2]);

        InitializeTempBoards(width, height);
        Board gameBoard = new Board(width, height);

        for (int step = 0; step < numPieces; step++) {
            string pieceType = ReadNonEmptyLine();
            if (pieceType == null) break;

            int pieceIdx = GetPieceIndex(pieceType);

            int bestRot, bestCol;
            FindBestMove(pieceIdx, gameBoard, out bestRot, out bestCol);

            // Output rotation and column
            Console.WriteLine($"{bestRot} {bestCol}");
            Console.Out.Flush();

            // Read game status result
            string result = ReadNonEmptyLine();
            if (result == null || result.Contains("gameover")) {
                break;
            }

            // Apply move to our board state to keep in sync
            Piece piece = AllPieces[pieceIdx][bestRot];
            int yLand = gameBoard.GetLandingRow(piece, bestCol);
            gameBoard.PlacePiece(piece, bestCol, yLand);
        }
    }
}
