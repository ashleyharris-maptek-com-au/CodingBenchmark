const fs = require('fs');

function solve() {
    let input;
    try {
        input = fs.readFileSync(0, 'utf8');
    } catch (e) {
        return;
    }
    
    let data;
    try {
        data = JSON.parse(input);
    } catch (e) {
        return;
    }

    const probability = data.probability;
    const drones = data.drones;
    const maxStepsPerDrone = data.maxStepsPerDrone;

    if (!probability || probability.length === 0 || probability[0].length === 0 || !drones || drones.length === 0) {
        console.log(JSON.stringify({ paths: [] }));
        return;
    }

    const H = probability.length;
    const W = probability[0].length;

    // Ensure all drones have valid IDs
    for (let i = 0; i < drones.length; i++) {
        if (drones[i].id === undefined) {
            drones[i].id = i;
        }
    }

    // Pre-allocate flat arrays for optimal BFS performance
    const visited = new Int32Array(W * H);
    const parent = new Int32Array(W * H);
    const dist = new Int32Array(W * H);
    const queue = new Int32Array(W * H);
    const accumProb = new Float64Array(W * H);
    let globalVisitedToken = 0;

    const orderings = [
        // 0: original order
        (a, b) => a.id - b.id,
        // 1: reverse order
        (a, b) => b.id - a.id,
        // 2: sort by x ascending (left to right)
        (a, b) => a.x - b.x,
        // 3: sort by y ascending (top to bottom)
        (a, b) => a.y - b.y,
        // 4: sort by x+y ascending (diagonal top-left to bottom-right)
        (a, b) => (a.x + a.y) - (b.x + b.y),
        // 5: sort by x-y ascending (diagonal bottom-left to top-right)
        (a, b) => (a.x - a.y) - (b.x - b.y)
    ];

    function runPlanning(p, delta, orderIdx) {
        const globalCovered = new Uint8Array(W * H);
        const paths = new Array(drones.length);

        // Sort drones based on chosen spatial ordering
        const dronesWithOriginalIndex = drones.map((d, index) => ({ drone: d, index }));
        dronesWithOriginalIndex.sort((a, b) => orderings[orderIdx](a.drone, b.drone));

        // Mark starting positions as covered
        for (let i = 0; i < drones.length; i++) {
            const drone = drones[i];
            if (drone.y >= 0 && drone.y < H && drone.x >= 0 && drone.x < W) {
                const startIdx = drone.y * W + drone.x;
                globalCovered[startIdx] = 1;
            }
        }

        for (const item of dronesWithOriginalIndex) {
            const drone = item.drone;
            const idx = item.index;
            const startX = drone.x;
            const startY = drone.y;

            let currX = startX;
            let currY = startY;

            const path = [[currX, currY]];
            let remSteps = maxStepsPerDrone;
            let targetPath = [];

            while (remSteps > 0) {
                if (targetPath.length === 0) {
                    let visitedToken = ++globalVisitedToken;
                    const startIdx = currY * W + currX;
                    visited[startIdx] = visitedToken;
                    dist[startIdx] = 0;
                    parent[startIdx] = -1;
                    accumProb[startIdx] = 0;

                    let head = 0;
                    let tail = 0;
                    queue[tail++] = startIdx;

                    let bestTargetIdx = -1;
                    let bestUtility = -1;
                    let firstTargetDist = -1;
                    let maxDepth = remSteps;

                    while (head < tail) {
                        const currIdx = queue[head++];
                        const d = dist[currIdx];

                        if (d > maxDepth) {
                            break;
                        }

                        const cy = Math.floor(currIdx / W);
                        const cx = currIdx % W;

                        if (d > 0) {
                            const prob = probability[cy][cx];
                            if (prob > 0 && globalCovered[currIdx] === 0) {
                                if (firstTargetDist === -1) {
                                    firstTargetDist = d;
                                    maxDepth = Math.min(remSteps, d + delta);
                                }
                                const utility = accumProb[currIdx] / Math.pow(d, p);
                                if (utility > bestUtility) {
                                    bestUtility = utility;
                                    bestTargetIdx = currIdx;
                                } else if (Math.abs(utility - bestUtility) < 1e-9) {
                                    // Tie-break: prefer closer distance, then higher probability
                                    const bestD = dist[bestTargetIdx];
                                    if (d < bestD) {
                                        bestTargetIdx = currIdx;
                                    } else if (d === bestD) {
                                        const bestProb = probability[Math.floor(bestTargetIdx / W)][bestTargetIdx % W];
                                        if (prob > bestProb) {
                                            bestTargetIdx = currIdx;
                                        }
                                    }
                                }
                            }
                        }

                        if (d < maxDepth) {
                            // Expand neighbors
                            // Up
                            if (cy > 0) {
                                const nIdx = currIdx - W;
                                if (probability[cy - 1][cx] !== -1 && visited[nIdx] !== visitedToken) {
                                    visited[nIdx] = visitedToken;
                                    dist[nIdx] = d + 1;
                                    parent[nIdx] = currIdx;
                                    const nProb = (globalCovered[nIdx] === 0) ? probability[cy - 1][cx] : 0;
                                    accumProb[nIdx] = accumProb[currIdx] + nProb;
                                    queue[tail++] = nIdx;
                                }
                            }
                            // Down
                            if (cy < H - 1) {
                                const nIdx = currIdx + W;
                                if (probability[cy + 1][cx] !== -1 && visited[nIdx] !== visitedToken) {
                                    visited[nIdx] = visitedToken;
                                    dist[nIdx] = d + 1;
                                    parent[nIdx] = currIdx;
                                    const nProb = (globalCovered[nIdx] === 0) ? probability[cy + 1][cx] : 0;
                                    accumProb[nIdx] = accumProb[currIdx] + nProb;
                                    queue[tail++] = nIdx;
                                }
                            }
                            // Left
                            if (cx > 0) {
                                const nIdx = currIdx - 1;
                                if (probability[cy][cx - 1] !== -1 && visited[nIdx] !== visitedToken) {
                                    visited[nIdx] = visitedToken;
                                    dist[nIdx] = d + 1;
                                    parent[nIdx] = currIdx;
                                    const nProb = (globalCovered[nIdx] === 0) ? probability[cy][cx - 1] : 0;
                                    accumProb[nIdx] = accumProb[currIdx] + nProb;
                                    queue[tail++] = nIdx;
                                }
                            }
                            // Right
                            if (cx < W - 1) {
                                const nIdx = currIdx + 1;
                                if (probability[cy][cx + 1] !== -1 && visited[nIdx] !== visitedToken) {
                                    visited[nIdx] = visitedToken;
                                    dist[nIdx] = d + 1;
                                    parent[nIdx] = currIdx;
                                    const nProb = (globalCovered[nIdx] === 0) ? probability[cy][cx + 1] : 0;
                                    accumProb[nIdx] = accumProb[currIdx] + nProb;
                                    queue[tail++] = nIdx;
                                }
                            }
                        }
                    }

                    if (bestTargetIdx !== -1) {
                        const tPath = [];
                        let temp = bestTargetIdx;
                        while (temp !== startIdx) {
                            tPath.push(temp);
                            temp = parent[temp];
                        }
                        tPath.reverse();
                        targetPath = tPath;
                    } else {
                        while (remSteps > 0) {
                            path.push([currX, currY]);
                            remSteps--;
                        }
                        break;
                    }
                }

                if (targetPath.length > 0) {
                    const nextIdx = targetPath.shift();
                    const nextY = Math.floor(nextIdx / W);
                    const nextX = nextIdx % W;
                    path.push([nextX, nextY]);
                    globalCovered[nextIdx] = 1;
                    currX = nextX;
                    currY = nextY;
                    remSteps--;
                }
            }

            paths[idx] = path;
        }

        return paths;
    }

    function evaluateScore(paths) {
        const visitedSet = new Uint8Array(W * H);
        let totalScore = 0;
        for (const path of paths) {
            for (const coord of path) {
                const x = coord[0];
                const y = coord[1];
                const idx = y * W + x;
                if (visitedSet[idx] === 0) {
                    visitedSet[idx] = 1;
                    const val = probability[y][x];
                    if (val > 0) {
                        totalScore += val;
                    }
                }
            }
        }
        return totalScore;
    }

    const startTime = Date.now();
    let bestScore = -1;
    let bestPaths = null;

    const configs = [
        { p: 1.2, delta: 15, orderIdx: 0 },
        { p: 1.0, delta: 25, orderIdx: 0 },
        { p: 1.5, delta: 15, orderIdx: 0 },
        { p: 1.2, delta: 15, orderIdx: 1 },
        { p: 1.0, delta: 25, orderIdx: 1 },
        { p: 1.2, delta: 15, orderIdx: 2 },
        { p: 1.2, delta: 15, orderIdx: 3 },
        { p: 1.5, delta: 15, orderIdx: 4 },
        { p: 1.0, delta: 20, orderIdx: 5 },
        { p: 0.8, delta: 30, orderIdx: 0 },
        { p: 2.0, delta: 10, orderIdx: 0 },
        { p: 0.5, delta: 35, orderIdx: 0 }
    ];

    for (let i = 0; i < configs.length; i++) {
        const elapsed = Date.now() - startTime;
        if (i > 0 && elapsed > 1800) {
            break;
        }
        const paths = runPlanning(configs[i].p, configs[i].delta, configs[i].orderIdx);
        const score = evaluateScore(paths);
        if (score > bestScore) {
            bestScore = score;
            bestPaths = paths;
        }
    }

    console.log(JSON.stringify({ paths: bestPaths }));
}

solve();
