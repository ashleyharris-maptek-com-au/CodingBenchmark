use std::io::{self, BufRead};
use std::time::Instant;

// --- GEOMETRY TYPES & OPERATIONS ---

#[derive(Debug, Clone, Copy)]
struct Point {
    x: f64,
    y: f64,
}

impl Point {
    fn new(x: f64, y: f64) -> Self {
        Point { x, y }
    }
}

#[derive(Debug, Clone, Copy)]
struct BBox {
    min_x: f64,
    max_x: f64,
    min_y: f64,
    max_y: f64,
}

impl BBox {
    fn from_points(pts: &[Point]) -> Self {
        let mut min_x = f64::INFINITY;
        let mut max_x = f64::NEG_INFINITY;
        let mut min_y = f64::INFINITY;
        let mut max_y = f64::NEG_INFINITY;
        for p in pts {
            if p.x < min_x { min_x = p.x; }
            if p.x > max_x { max_x = p.x; }
            if p.y < min_y { min_y = p.y; }
            if p.y > max_y { max_y = p.y; }
        }
        BBox { min_x, max_x, min_y, max_y }
    }

    fn intersects(&self, other: &Self) -> bool {
        self.min_x <= other.max_x + 1e-9 && self.max_x >= other.min_x - 1e-9 &&
        self.min_y <= other.max_y + 1e-9 && self.max_y >= other.min_y - 1e-9
    }

    fn contains(&self, other: &Self) -> bool {
        self.min_x <= other.min_x + 1e-9 && self.max_x >= other.max_x - 1e-9 &&
        self.min_y <= other.min_y + 1e-9 && self.max_y >= other.min_y - 1e-9
    }

    fn translate(&self, tx: f64, ty: f64) -> Self {
        BBox {
            min_x: self.min_x + tx,
            max_x: self.max_x + tx,
            min_y: self.min_y + ty,
            max_y: self.max_y + ty,
        }
    }
}

#[derive(Debug, Clone, Copy)]
struct Triangle {
    a: Point,
    b: Point,
    c: Point,
}

impl Triangle {
    fn translate(&self, tx: f64, ty: f64) -> Self {
        Triangle {
            a: Point::new(self.a.x + tx, self.a.y + ty),
            b: Point::new(self.b.x + tx, self.b.y + ty),
            c: Point::new(self.c.x + tx, self.c.y + ty),
        }
    }
}

struct Edge {
    a: Point,
    b: Point,
    bbox: BBox,
}

impl Edge {
    fn new(a: Point, b: Point) -> Self {
        let bbox = BBox::from_points(&[a, b]);
        Edge { a, b, bbox }
    }
}

// --- GEOMETRY UTILITIES ---

fn signed_area(poly: &[Point]) -> f64 {
    let mut area = 0.0;
    let n = poly.len();
    for i in 0..n {
        let p1 = poly[i];
        let p2 = poly[(i + 1) % n];
        area += p1.x * p2.y - p2.x * p1.y;
    }
    area / 2.0
}

fn point_in_polygon(p: Point, poly: &[Point]) -> bool {
    let mut inside = false;
    let n = poly.len();
    if n < 3 { return false; }
    let mut j = n - 1;
    for i in 0..n {
        let pi = poly[i];
        let pj = poly[j];
        if ((pi.y > p.y) != (pj.y > p.y))
            && (p.x < (pj.x - pi.x) * (p.y - pi.y) / (pj.y - pi.y + 1e-12) + pi.x)
        {
            inside = !inside;
        }
        j = i;
    }
    inside
}

fn point_on_segment(p: Point, a: Point, b: Point) -> bool {
    let cross = (p.y - a.y) * (b.x - a.x) - (p.x - a.x) * (b.y - a.y);
    if cross.abs() > 1e-9 {
        return false;
    }
    let dot = (p.x - a.x) * (b.x - a.x) + (p.y - a.y) * (b.y - a.y);
    if dot < -1e-9 {
        return false;
    }
    let len_sq = (b.x - a.x).powi(2) + (b.y - a.y).powi(2);
    if dot > len_sq + 1e-9 {
        return false;
    }
    true
}

fn point_in_or_on_polygon(p: Point, poly: &[Point]) -> bool {
    if point_in_polygon(p, poly) {
        return true;
    }
    let n = poly.len();
    for i in 0..n {
        let a = poly[i];
        let b = poly[(i + 1) % n];
        if point_on_segment(p, a, b) {
            return true;
        }
    }
    false
}

fn cross_product(p1: Point, p2: Point, p3: Point) -> f64 {
    (p2.x - p1.x) * (p3.y - p1.y) - (p2.y - p1.y) * (p3.x - p1.x)
}

fn segments_intersect_interior(a: Point, b: Point, c: Point, d: Point) -> bool {
    let cp1 = cross_product(a, b, c);
    let cp2 = cross_product(a, b, d);
    let cp3 = cross_product(c, d, a);
    let cp4 = cross_product(c, d, b);

    ((cp1 > 1e-9 && cp2 < -1e-9) || (cp1 < -1e-9 && cp2 > 1e-9)) &&
    ((cp3 > 1e-9 && cp4 < -1e-9) || (cp3 < -1e-9 && cp4 > 1e-9))
}

// --- TRIANGULATION ---

fn point_in_triangle(p: Point, a: Point, b: Point, c: Point) -> bool {
    let sign1 = (p.x - a.x) * (b.y - a.y) - (p.y - a.y) * (b.x - a.x);
    let sign2 = (p.x - b.x) * (c.y - b.y) - (p.y - b.y) * (c.x - b.x);
    let sign3 = (p.x - c.x) * (a.y - c.y) - (p.y - c.y) * (a.x - c.x);

    let has_neg = (sign1 < -1e-9) || (sign2 < -1e-9) || (sign3 < -1e-9);
    let has_pos = (sign1 > 1e-9) || (sign2 > 1e-9) || (sign3 > 1e-9);

    !(has_neg && has_pos)
}

fn is_ear(a: Point, b: Point, c: Point, vertices: &[Point], indices: &[usize], prev_idx: usize, curr_idx: usize, next_idx: usize) -> bool {
    let cross = (b.x - a.x) * (c.y - b.y) - (b.y - a.y) * (c.x - b.x);
    if cross <= 1e-9 {
        return false;
    }

    for &idx in indices {
        if idx == prev_idx || idx == curr_idx || idx == next_idx {
            continue;
        }
        let p = vertices[idx];
        if point_in_triangle(p, a, b, c) {
            return false;
        }
    }
    true
}

fn triangulate(poly: &[Point]) -> Vec<Triangle> {
    let mut vertices = poly.to_vec();
    if signed_area(&vertices) < 0.0 {
        vertices.reverse();
    }

    let mut triangles = Vec::new();
    let mut indices: Vec<usize> = (0..vertices.len()).collect();

    let mut count = 2 * indices.len();
    while indices.len() > 3 && count > 0 {
        count -= 1;
        let mut ear_found = false;
        let n = indices.len();
        for i in 0..n {
            let prev = indices[(i + n - 1) % n];
            let curr = indices[i];
            let next = indices[(i + 1) % n];

            let a = vertices[prev];
            let b = vertices[curr];
            let c = vertices[next];

            if is_ear(a, b, c, &vertices, &indices, prev, curr, next) {
                triangles.push(Triangle { a, b, c });
                indices.remove(i);
                ear_found = true;
                break;
            }
        }
        if !ear_found {
            break;
        }
    }

    if indices.len() == 3 {
        triangles.push(Triangle {
            a: vertices[indices[0]],
            b: vertices[indices[1]],
            c: vertices[indices[2]],
        });
    }

    triangles
}

// --- SEPARATING AXIS THEOREM (SAT) ---

fn triangles_overlap(t1: &Triangle, t2: &Triangle) -> bool {
    let axes = [
        Point::new(-(t1.b.y - t1.a.y), t1.b.x - t1.a.x),
        Point::new(-(t1.c.y - t1.b.y), t1.c.x - t1.b.x),
        Point::new(-(t1.a.y - t1.c.y), t1.a.x - t1.c.x),
        Point::new(-(t2.b.y - t2.a.y), t2.b.x - t2.a.x),
        Point::new(-(t2.c.y - t2.b.y), t2.c.x - t2.b.x),
        Point::new(-(t2.a.y - t2.c.y), t2.a.x - t2.c.x),
    ];

    for axis in &axes {
        let p1_a = t1.a.x * axis.x + t1.a.y * axis.y;
        let p1_b = t1.b.x * axis.x + t1.b.y * axis.y;
        let p1_c = t1.c.x * axis.x + t1.c.y * axis.y;
        let min1 = p1_a.min(p1_b).min(p1_c);
        let max1 = p1_a.max(p1_b).max(p1_c);

        let p2_a = t2.a.x * axis.x + t2.a.y * axis.y;
        let p2_b = t2.b.x * axis.x + t2.b.y * axis.y;
        let p2_c = t2.c.x * axis.x + t2.c.y * axis.y;
        let min2 = p2_a.min(p2_b).min(p2_c);
        let max2 = p2_a.max(p2_b).max(p2_c);

        if max1 < min2 - 1e-8 || max2 < min1 - 1e-8 {
            return false;
        }
    }
    true
}

// --- CONVEXITY & RECTANGLE CHECKS ---

fn is_convex(poly: &[Point]) -> bool {
    let n = poly.len();
    if n < 3 { return false; }
    let mut sign = 0.0;
    for i in 0..n {
        let a = poly[i];
        let b = poly[(i + 1) % n];
        let c = poly[(i + 2) % n];
        let cross = (b.x - a.x) * (c.y - b.y) - (b.y - a.y) * (c.x - b.x);
        if cross.abs() > 1e-9 {
            if sign == 0.0 {
                sign = cross.signum();
            } else if sign != cross.signum() {
                return false;
            }
        }
    }
    true
}

fn is_axis_aligned_rectangle(poly: &[Point]) -> bool {
    if poly.len() != 4 { return false; }
    let mut xs = [poly[0].x, poly[1].x, poly[2].x, poly[3].x];
    let mut ys = [poly[0].y, poly[1].y, poly[2].y, poly[3].y];
    xs.sort_by(|a, b| a.partial_cmp(b).unwrap());
    ys.sort_by(|a, b| a.partial_cmp(b).unwrap());
    
    let w = xs[3] - xs[0];
    let h = ys[3] - ys[0];
    
    if (xs[1] - xs[0]).abs() > 1e-9 && (xs[3] - xs[2]).abs() > 1e-9 { return false; }
    if (ys[1] - ys[0]).abs() > 1e-9 && (ys[3] - ys[2]).abs() > 1e-9 { return false; }
    
    let area = signed_area(poly).abs();
    (area - w * h).abs() < 1e-9
}

// --- SPATIAL PARTITIONING GRID ---

struct SpatialGrid {
    min_x: f64,
    min_y: f64,
    cell_w: f64,
    cell_h: f64,
    cols: usize,
    rows: usize,
    cells: Vec<Vec<usize>>,
}

impl SpatialGrid {
    fn new(bbox: BBox, cols: usize, rows: usize) -> Self {
        let cell_w = (bbox.max_x - bbox.min_x) / (cols as f64).max(1.0);
        let cell_h = (bbox.max_y - bbox.min_y) / (rows as f64).max(1.0);
        SpatialGrid {
            min_x: bbox.min_x,
            min_y: bbox.min_y,
            cell_w,
            cell_h,
            cols,
            rows,
            cells: vec![Vec::new(); cols * rows],
        }
    }

    fn get_cell_range(&self, bbox: &BBox) -> (usize, usize, usize, usize) {
        let col_start = (((bbox.min_x - self.min_x) / self.cell_w).floor() as isize).max(0) as usize;
        let col_start = col_start.min(self.cols - 1);
        let col_end = (((bbox.max_x - self.min_x) / self.cell_w).floor() as isize).max(0) as usize;
        let col_end = col_end.min(self.cols - 1);
        
        let row_start = (((bbox.min_y - self.min_y) / self.cell_h).floor() as isize).max(0) as usize;
        let row_start = row_start.min(self.rows - 1);
        let row_end = (((bbox.max_y - self.min_y) / self.cell_h).floor() as isize).max(0) as usize;
        let row_end = row_end.min(self.rows - 1);

        (col_start, col_end, row_start, row_end)
    }

    fn insert(&mut self, idx: usize, bbox: &BBox) {
        let (cs, ce, rs, re) = self.get_cell_range(bbox);
        for r in rs..=re {
            for c in cs..=ce {
                let cell_idx = r * self.cols + c;
                self.cells[cell_idx].push(idx);
            }
        }
    }

    fn query(&self, bbox: &BBox, candidates: &mut Vec<usize>, visited: &mut [bool]) {
        candidates.clear();
        let (cs, ce, rs, re) = self.get_cell_range(bbox);
        for r in rs..=re {
            for c in cs..=ce {
                let cell_idx = r * self.cols + c;
                for &idx in &self.cells[cell_idx] {
                    if !visited[idx] {
                        visited[idx] = true;
                        candidates.push(idx);
                    }
                }
            }
        }
        for &idx in candidates.iter() {
            visited[idx] = false;
        }
    }
}

// --- REPRESENTATION OF PIECES & STOCKS ---

struct Piece {
    id: usize,
    vertices: Vec<Point>,
    triangles: Vec<Triangle>,
    area: f64,
}

struct PlacedPiece {
    bbox: BBox,
    triangles: Vec<Triangle>,
    triangle_bboxes: Vec<BBox>,
}

struct Stock {
    index: usize,
    placed: Vec<PlacedPiece>,
    grid: SpatialGrid,
    anchors: Vec<Point>,
}

fn main() {
    let start_time = Instant::now();
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = match lines.next() {
        Some(Ok(line)) => line,
        _ => return,
    };
    let mut parts = first_line.split_whitespace();
    let sv: usize = parts.next().unwrap().parse().unwrap();
    let np: usize = parts.next().unwrap().parse().unwrap();

    let mut stock_vertices = Vec::with_capacity(sv);
    for _ in 0..sv {
        let line = lines.next().unwrap().unwrap();
        let mut pts = line.split_whitespace();
        let x: f64 = pts.next().unwrap().parse().unwrap();
        let y: f64 = pts.next().unwrap().parse().unwrap();
        stock_vertices.push(Point::new(x, y));
    }
    let stock_bbox = BBox::from_points(&stock_vertices);
    
    // Multi-tiered stock classification
    let is_stock_rect = is_axis_aligned_rectangle(&stock_vertices);
    let is_stock_conv = if is_stock_rect { true } else { is_convex(&stock_vertices) };

    let mut stock_edges = Vec::with_capacity(sv);
    for i in 0..sv {
        let a = stock_vertices[i];
        let b = stock_vertices[(i + 1) % sv];
        stock_edges.push(Edge::new(a, b));
    }

    let mut pieces = Vec::with_capacity(np);
    let mut total_piece_w = 0.0;
    let mut total_piece_h = 0.0;
    for i in 0..np {
        let pv_line = lines.next().unwrap().unwrap();
        let pv: usize = pv_line.trim().parse().unwrap();
        let mut piece_vertices = Vec::with_capacity(pv);
        for _ in 0..pv {
            let line = lines.next().unwrap().unwrap();
            let mut pts = line.split_whitespace();
            let x: f64 = pts.next().unwrap().parse().unwrap();
            let y: f64 = pts.next().unwrap().parse().unwrap();
            piece_vertices.push(Point::new(x, y));
        }
        let area = signed_area(&piece_vertices).abs();
        let triangles = triangulate(&piece_vertices);
        let bbox = BBox::from_points(&piece_vertices);
        total_piece_w += bbox.max_x - bbox.min_x;
        total_piece_h += bbox.max_y - bbox.min_y;
        pieces.push(Piece {
            id: i,
            vertices: piece_vertices,
            triangles,
            area,
        });
    }

    let avg_w = total_piece_w / np as f64;
    let avg_h = total_piece_h / np as f64;
    let stock_w = stock_bbox.max_x - stock_bbox.min_x;
    let stock_h = stock_bbox.max_y - stock_bbox.min_y;
    let grid_cols = ((stock_w / avg_w).round() as usize).clamp(10, 100);
    let grid_rows = ((stock_h / avg_h).round() as usize).clamp(10, 100);

    let mut sorted_pieces: Vec<&Piece> = pieces.iter().collect();
    sorted_pieces.sort_by(|a, b| b.area.partial_cmp(&a.area).unwrap());

    let mut placement_results = vec![(0, 0.0, 0.0, 0.0); np];
    let mut stocks: Vec<Stock> = Vec::new();

    let mut query_candidates = Vec::new();
    let mut visited_buffer = vec![false; np + 1];

    let rotations = if np < 50 {
        16
    } else if np < 500 {
        8
    } else {
        4
    };
    let mut candidate_angles = Vec::with_capacity(rotations);
    for r in 0..rotations {
        candidate_angles.push(r as f64 * (2.0 * std::f64::consts::PI / rotations as f64));
    }

    for piece in sorted_pieces {
        let mut placed = false;

        for stock in &mut stocks {
            if visited_buffer.len() < stock.placed.len() + 1 {
                visited_buffer.resize(stock.placed.len() * 2 + 10, false);
            }

            if try_place_piece(
                piece,
                stock,
                &stock_vertices,
                &stock_bbox,
                &stock_edges,
                is_stock_rect,
                is_stock_conv,
                &candidate_angles,
                &mut query_candidates,
                &mut visited_buffer,
                &mut placement_results,
            ) {
                placed = true;
                break;
            }
        }

        if !placed {
            let mut new_stock = Stock {
                index: stocks.len(),
                placed: Vec::new(),
                grid: SpatialGrid::new(stock_bbox, grid_cols, grid_rows),
                anchors: stock_vertices.clone(),
            };
            
            if visited_buffer.len() < new_stock.placed.len() + 1 {
                visited_buffer.resize(10, false);
            }

            let success = try_place_piece(
                piece,
                &mut new_stock,
                &stock_vertices,
                &stock_bbox,
                &stock_edges,
                is_stock_rect,
                is_stock_conv,
                &candidate_angles,
                &mut query_candidates,
                &mut visited_buffer,
                &mut placement_results,
            );

            assert!(success, "A piece must fit in an empty stock!");
            stocks.push(new_stock);
        }
    }

    println!("{}", stocks.len());
    for res in &placement_results {
        println!("{} {} {} {}", res.0, res.1, res.2, res.3);
    }

    eprintln!("Completed in {:?}", start_time.elapsed());
}

fn try_place_piece(
    piece: &Piece,
    stock: &mut Stock,
    stock_vertices: &[Point],
    stock_bbox: &BBox,
    stock_edges: &[Edge],
    is_stock_rect: bool,
    is_stock_conv: bool,
    candidate_angles: &[f64],
    query_candidates: &mut Vec<usize>,
    visited_buffer: &mut [bool],
    placement_results: &mut [(usize, f64, f64, f64)],
) -> bool {
    let mut best_placement: Option<(f64, f64, f64, BBox, Vec<Point>, Vec<Triangle>, Vec<BBox>)> = None;
    let mut best_score = f64::INFINITY;

    let anchor_limit = if stock.anchors.len() > 100 { 100 } else { stock.anchors.len() };
    
    let mut active_anchors = stock.anchors.clone();
    active_anchors.sort_by(|a, b| {
        if (a.y - b.y).abs() > 1e-9 {
            a.y.partial_cmp(&b.y).unwrap()
        } else {
            a.x.partial_cmp(&b.x).unwrap()
        }
    });

    for &angle in candidate_angles {
        let cos_t = angle.cos();
        let sin_t = angle.sin();

        let rot_vertices: Vec<Point> = piece.vertices.iter().map(|p| {
            Point::new(p.x * cos_t - p.y * sin_t, p.x * sin_t + p.y * cos_t)
        }).collect();

        let rot_triangles: Vec<Triangle> = piece.triangles.iter().map(|t| {
            Triangle {
                a: Point::new(t.a.x * cos_t - t.a.y * sin_t, t.a.x * sin_t + t.a.y * cos_t),
                b: Point::new(t.b.x * cos_t - t.b.y * sin_t, t.b.x * sin_t + t.b.y * cos_t),
                c: Point::new(t.c.x * cos_t - t.c.y * sin_t, t.c.x * sin_t + t.c.y * cos_t),
            }
        }).collect();

        let rot_triangle_bboxes: Vec<BBox> = rot_triangles.iter().map(|t| {
            BBox::from_points(&[t.a, t.b, t.c])
        }).collect();

        let rot_bbox = BBox::from_points(&rot_vertices);

        let pv = rot_vertices.len();
        let rot_edges: Vec<Edge> = (0..pv).map(|i| {
            Edge::new(rot_vertices[i], rot_vertices[(i + 1) % pv])
        }).collect();

        for &anchor in active_anchors.iter().take(anchor_limit) {
            let step = if rot_vertices.len() > 16 { 2 } else { 1 };
            for v_idx in (0..rot_vertices.len()).step_by(step) {
                let v = rot_vertices[v_idx];
                let tx = anchor.x - v.x;
                let ty = anchor.y - v.y;

                let score = ty * 10.0 + tx;
                if score >= best_score {
                    continue;
                }

                let trans_bbox = rot_bbox.translate(tx, ty);
                if !stock_bbox.contains(&trans_bbox) {
                    continue;
                }

                // --- OVERLAP CHECK WITH PLACED PIECES ---
                let mut overlaps = false;
                stock.grid.query(&trans_bbox, query_candidates, visited_buffer);
                
                for &placed_idx in query_candidates.iter() {
                    let placed = &stock.placed[placed_idx];
                    if !trans_bbox.intersects(&placed.bbox) {
                        continue;
                    }

                    // Check triangle-triangle overlaps with bbox pre-filtering
                    for (t1, b1) in rot_triangles.iter().zip(&rot_triangle_bboxes) {
                        let t1_trans = t1.translate(tx, ty);
                        let b1_trans = b1.translate(tx, ty);
                        for (t2, b2) in placed.triangles.iter().zip(&placed.triangle_bboxes) {
                            if !b1_trans.intersects(b2) {
                                continue;
                            }
                            if triangles_overlap(&t1_trans, t2) {
                                overlaps = true;
                                break;
                            }
                        }
                        if overlaps { break; }
                    }
                    if overlaps { break; }
                }
                if overlaps { continue; }

                // --- CONTAINMENT CHECK (THREE-TIERED) ---
                let mut is_inside = true;

                if is_stock_rect {
                    // Containment is fully guaranteed by trans_bbox being within stock_bbox
                } else if is_stock_conv {
                    // Convex stock containment only requires checking vertices
                    for rv in &rot_vertices {
                        let p_trans = Point::new(rv.x + tx, rv.y + ty);
                        if !point_in_or_on_polygon(p_trans, stock_vertices) {
                            is_inside = false;
                            break;
                        }
                    }
                } else {
                    // General simple polygon containment requires checking vertices, midpoints, and crossings
                    for rv in &rot_vertices {
                        let p_trans = Point::new(rv.x + tx, rv.y + ty);
                        if !point_in_or_on_polygon(p_trans, stock_vertices) {
                            is_inside = false;
                            break;
                        }
                    }
                    if is_inside {
                        for edge in &rot_edges {
                            let a_trans = Point::new(edge.a.x + tx, edge.a.y + ty);
                            let b_trans = Point::new(edge.b.x + tx, edge.b.y + ty);
                            let m_trans = Point::new((a_trans.x + b_trans.x) / 2.0, (a_trans.y + b_trans.y) / 2.0);

                            if !point_in_or_on_polygon(m_trans, stock_vertices) {
                                is_inside = false;
                                break;
                            }

                            let e_trans_bbox = edge.bbox.translate(tx, ty);
                            for se in stock_edges {
                                if se.bbox.intersects(&e_trans_bbox) {
                                    if segments_intersect_interior(a_trans, b_trans, se.a, se.b) {
                                        is_inside = false;
                                        break;
                                    }
                                }
                            }
                            if !is_inside { break; }
                        }
                    }
                }

                if !is_inside { continue; }

                // Valid placement found! Update best
                best_score = score;
                
                let trans_vertices = rot_vertices.iter().map(|p| {
                    Point::new(p.x + tx, p.y + ty)
                }).collect();
                
                let trans_triangles = rot_triangles.iter().map(|t| {
                    t.translate(tx, ty)
                }).collect();

                let trans_triangle_bboxes = rot_triangle_bboxes.iter().map(|b| {
                    b.translate(tx, ty)
                }).collect();

                best_placement = Some((tx, ty, angle, trans_bbox, trans_vertices, trans_triangles, trans_triangle_bboxes));
            }
        }
    }

    if let Some((tx, ty, angle, trans_bbox, trans_vertices, trans_triangles, trans_triangle_bboxes)) = best_placement {
        placement_results[piece.id] = (stock.index, tx, ty, angle);

        let idx = stock.placed.len();
        stock.placed.push(PlacedPiece {
            bbox: trans_bbox,
            triangles: trans_triangles,
            triangle_bboxes: trans_triangle_bboxes,
        });

        stock.grid.insert(idx, &trans_bbox);
        stock.anchors.extend(trans_vertices);
        true
    } else {
        false
    }
}
