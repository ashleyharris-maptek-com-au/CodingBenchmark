using System;

class TestDelta
{
    static int N = 10;
    static int[,] F = new int[N, N];
    static int[,] D = new int[N, N];
    static int[] p = new int[N];
    static int[,] Delta = new int[N, N];

    static void Main()
    {
        Random rand = new Random(42);
        for (int i = 0; i < N; i++)
        {
            p[i] = i;
            for (int j = 0; j < N; j++)
            {
                F[i, j] = rand.Next(-100, 100);
                D[i, j] = rand.Next(-100, 100);
            }
        }

        // Shuffle p
        for (int i = N - 1; i > 0; i--)
        {
            int j = rand.Next(i + 1);
            int temp = p[i];
            p[i] = p[j];
            p[j] = temp;
        }

        // Compute initial Delta matrix from scratch
        for (int i = 0; i < N; i++)
        {
            for (int j = 0; j < N; j++)
            {
                if (i == j) Delta[i, j] = 0;
                else Delta[i, j] = ComputeDeltaFromScratch(i, j);
            }
        }

        // Pick a random swap
        int r = rand.Next(N);
        int s = rand.Next(N);
        while (s == r) s = rand.Next(N);

        Console.WriteLine("Swapping {0} and {1}", r, s);
        int actualDelta = ComputeDeltaFromScratch(r, s);

        // Apply swap to p
        int oldPr = p[r];
        int oldPs = p[s];
        
        // Update Delta matrix using our formula
        int[,] nextDelta = new int[N, N];
        for (int i = 0; i < N; i++)
        {
            for (int j = 0; j < N; j++)
            {
                nextDelta[i, j] = Delta[i, j];
            }
        }

        // Update entries for disjoint u, v
        for (int u = 0; u < N; u++)
        {
            if (u == r || u == s) continue;
            for (int v = 0; v < N; v++)
            {
                if (v == r || v == s || v == u) continue;

                int A = (D[oldPs, p[v]] - D[oldPr, p[v]]) - (D[oldPs, p[u]] - D[oldPr, p[u]]);
                int B = (D[p[v], oldPs] - D[p[v], oldPr]) - (D[p[u], oldPs] - D[p[u], oldPr]);

                nextDelta[u, v] = Delta[u, v] + 
                    (F[u, r] - F[u, s] - F[v, r] + F[v, s]) * B + 
                    (F[r, u] - F[s, u] - F[r, v] + F[s, v]) * A;
            }
        }

        // Swap in permutation p
        p[r] = oldPs;
        p[s] = oldPr;

        // Recompute row/column r and s for nextDelta
        for (int i = 0; i < N; i++)
        {
            if (i != r) nextDelta[r, i] = nextDelta[i, r] = ComputeDeltaFromScratch(r, i);
            if (i != s) nextDelta[s, i] = nextDelta[i, s] = ComputeDeltaFromScratch(s, i);
        }
        nextDelta[r, r] = 0;
        nextDelta[s, s] = 0;

        // Check correctness against scratch calculation for the new p
        bool match = true;
        for (int i = 0; i < N; i++)
        {
            for (int j = 0; j < N; j++)
            {
                int expected = (i == j) ? 0 : ComputeDeltaFromScratch(i, j);
                if (nextDelta[i, j] != expected)
                {
                    Console.WriteLine("Mismatch at ({0}, {1}): Expected {2}, Got {3}", i, j, expected, nextDelta[i, j]);
                    match = false;
                }
            }
        }

        if (match)
        {
            Console.WriteLine("SUCCESS! All Delta matrix elements match.");
        }
        else
        {
            Console.WriteLine("FAILURE! Mismatch detected.");
        }
    }

    static int ComputeDeltaFromScratch(int r, int s)
    {
        int sum = 0;
        for (int k = 0; k < N; k++)
        {
            if (k == r || k == s) continue;
            sum += (F[r, k] - F[s, k]) * (D[p[s], p[k]] - D[p[r], p[k]]) + 
                   (F[k, r] - F[k, s]) * (D[p[k], p[s]] - D[p[k], p[r]]);
        }
        sum += (F[r, r] - F[s, s]) * (D[p[s], p[s]] - D[p[r], p[r]]) + 
               F[r, s] * (D[p[s], p[r]] - D[p[r], p[s]]) + 
               F[s, r] * (D[p[r], p[s]] - D[p[s], p[r]]);
        return sum;
    }
}
