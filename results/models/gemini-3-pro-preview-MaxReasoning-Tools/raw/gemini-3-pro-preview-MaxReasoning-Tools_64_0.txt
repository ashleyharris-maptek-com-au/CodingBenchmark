declare const require: any;
const fs = require('fs');
const perf = require('perf_hooks').performance;

interface Drone {
    id: string;
    x: number;
    y: number;
}

interface InputData {
    probability: number[][];
    drones: Drone[];
    maxStepsPerDrone: number;
}

function main() {
    const input: string = fs.readFileSync(0, 'utf8');
    if (!input.trim()) {
        console.log(JSON.stringify({ paths: [] }));
        return;
    }
    const data: InputData = JSON.parse(input);

    const probability = data.probability;
    const drones = data.drones;
    const S = data.maxStepsPerDrone;

    if (!drones || drones.length === 0) {
        console.log(JSON.stringify({ paths: [] }));
        return;
    }

    if (S <= 0) {
        console.log(JSON.stringify({ paths: drones.map(d => [[d.x, d.y]]) }));
        return;
    }

    const H = probability.length;
    const W = probability[0].length;
    const D = drones.length;

    const dirsX = new Int32Array([0, 0, 1, -1]);
    const dirsY = new Int32Array([1, -1, 0, 0]);

    // Precompute static potential using diffusion
    const staticPotential = Array.from({ length: H }, () => new Float32Array(W));
    for (let y = 0; y < H; y++) {
        for (let x = 0; x < W; x++) {
            if (probability[y][x] > 0) {
                staticPotential[y][x] = probability[y][x];
            }
        }
    }

    const tempPot = Array.from({ length: H }, () => new Float32Array(W));
    const K = 4;
    const gamma = 0.15;
    for (let iter = 0; iter < K; iter++) {
        for (let y = 0; y < H; y++) {
            for (let x = 0; x < W; x++) {
                if (probability[y][x] === -1) {
                    tempPot[y][x] = 0;
                    continue;
                }
                let sum = 0;
                for (let i = 0; i < 4; i++) {
                    const nx = x + dirsX[i];
                    const ny = y + dirsY[i];
                    if (nx >= 0 && nx < W && ny >= 0 && ny < H && probability[ny][nx] !== -1) {
                        sum += staticPotential[ny][nx];
                    }
                }
                tempPot[y][x] = staticPotential[y][x] + gamma * sum;
            }
        }
        for (let y = 0; y < H; y++) {
            staticPotential[y].set(tempPot[y]);
        }
    }

    // Global arrays for BFS to avoid allocations
    const MAX_QUEUE_SIZE = 10000;
    const queueX = new Int32Array(MAX_QUEUE_SIZE);
    const queueY = new Int32Array(MAX_QUEUE_SIZE);
    const queueDist = new Int16Array(MAX_QUEUE_SIZE);
    const queueParent = new Int32Array(MAX_QUEUE_SIZE);
    
    let bfsToken = 0;
    const bfsVisited = Array.from({ length: H }, () => new Int32Array(W));

    let pathVisitedToken = 0;
    const pathVisited = Array.from({ length: H }, () => new Int32Array(W));

    let rolloutVisitedToken = 0;
    const rolloutVisited = Array.from({ length: H }, () => new Int32Array(W));

    let tempSeenToken = 0;
    const tempSeen = Array.from({ length: H }, () => new Int32Array(W));

    const globalVisitedCount = Array.from({ length: H }, () => new Int32Array(W));

    // Pre-allocated arrays for neighbor selection during rollouts and step choices
    const nextMovesX = new Int32Array(4);
    const nextMovesY = new Int32Array(4);
    const unvisitedMovesX = new Int32Array(4);
    const unvisitedMovesY = new Int32Array(4);

    const validMovesX = new Int32Array(5);
    const validMovesY = new Int32Array(5);

    function findNearestReward(
        sx: number,
        sy: number,
        maxDist: number
    ): { tx: number; ty: number; nextX: number; nextY: number } | null {
        bfsToken++;
        bfsVisited[sy][sx] = bfsToken;
        
        let head = 0;
        let tail = 0;
        
        queueX[tail] = sx;
        queueY[tail] = sy;
        queueDist[tail] = 0;
        queueParent[tail] = -1;
        tail++;
        
        while (head < tail) {
            const cx = queueX[head];
            const cy = queueY[head];
            const cd = queueDist[head];
            
            if (cd > 0 && probability[cy][cx] > 0 && globalVisitedCount[cy][cx] === 0 && pathVisited[cy][cx] !== pathVisitedToken) {
                let curr = head;
                while (queueParent[curr] !== -1 && queueParent[queueParent[curr]] !== -1) {
                    curr = queueParent[curr];
                }
                return {
                    tx: cx,
                    ty: cy,
                    nextX: queueX[curr],
                    nextY: queueY[curr]
                };
            }
            
            if (cd < maxDist) {
                for (let i = 0; i < 4; i++) {
                    const nx = cx + dirsX[i];
                    const ny = cy + dirsY[i];
                    if (nx >= 0 && nx < W && ny >= 0 && ny < H && probability[ny][nx] !== -1) {
                        if (bfsVisited[ny][nx] !== bfsToken) {
                            bfsVisited[ny][nx] = bfsToken;
                            if (tail < MAX_QUEUE_SIZE) {
                                queueX[tail] = nx;
                                queueY[tail] = ny;
                                queueDist[tail] = cd + 1;
                                queueParent[tail] = head;
                                tail++;
                            }
                        }
                    }
                }
            }
            head++;
        }
        return null;
    }

    function scoreMoveWithRollouts(
        sx: number,
        sy: number,
        numRollouts: number,
        rolloutLen: number
    ): number {
        let totalReward = 0;
        
        for (let r = 0; r < numRollouts; r++) {
            rolloutVisitedToken++;
            let cx = sx;
            let cy = sy;
            let rewardSum = 0;
            
            if (globalVisitedCount[cy][cx] === 0 && pathVisited[cy][cx] !== pathVisitedToken && rolloutVisited[cy][cx] !== rolloutVisitedToken) {
                rolloutVisited[cy][cx] = rolloutVisitedToken;
                if (probability[cy][cx] > 0) {
                    rewardSum += probability[cy][cx];
                }
            }
            
            for (let step = 0; step < rolloutLen; step++) {
                let nextMovesCount = 0;
                let unvisitedMovesCount = 0;
                
                for (let i = 0; i < 4; i++) {
                    const nx = cx + dirsX[i];
                    const ny = cy + dirsY[i];
                    if (nx >= 0 && nx < W && ny >= 0 && ny < H && probability[ny][nx] !== -1) {
                        nextMovesX[nextMovesCount] = nx;
                        nextMovesY[nextMovesCount] = ny;
                        nextMovesCount++;
                        
                        const isUnvisited = globalVisitedCount[ny][nx] === 0 && 
                                            pathVisited[ny][nx] !== pathVisitedToken && 
                                            rolloutVisited[ny][nx] !== rolloutVisitedToken;
                        if (isUnvisited) {
                            unvisitedMovesX[unvisitedMovesCount] = nx;
                            unvisitedMovesY[unvisitedMovesCount] = ny;
                            unvisitedMovesCount++;
                        }
                    }
                }
                
                let rx = 0;
                let ry = 0;
                if (unvisitedMovesCount > 0) {
                    const idx = Math.floor(Math.random() * unvisitedMovesCount);
                    rx = unvisitedMovesX[idx];
                    ry = unvisitedMovesY[idx];
                } else if (nextMovesCount > 0) {
                    const idx = Math.floor(Math.random() * nextMovesCount);
                    rx = nextMovesX[idx];
                    ry = nextMovesY[idx];
                } else {
                    break;
                }
                
                cx = rx;
                cy = ry;
                
                if (globalVisitedCount[cy][cx] === 0 && pathVisited[cy][cx] !== pathVisitedToken && rolloutVisited[cy][cx] !== rolloutVisitedToken) {
                    rolloutVisited[cy][cx] = rolloutVisitedToken;
                    if (probability[cy][cx] > 0) {
                        rewardSum += probability[cy][cx];
                    }
                }
            }
            totalReward += rewardSum;
        }
        
        return totalReward / numRollouts;
    }

    function generatePath(
        startX: number,
        startY: number,
        prefix: [number, number][] | null
    ): [number, number][] {
        const path: [number, number][] = Array.from({ length: S + 1 });
        pathVisitedToken++;

        let startStep = 0;
        if (prefix !== null && prefix.length > 0) {
            startStep = prefix.length - 1;
            for (let j = 0; j <= startStep; j++) {
                path[j] = prefix[j];
                const [px, py] = prefix[j];
                pathVisited[py][px] = pathVisitedToken;
            }
        } else {
            path[0] = [startX, startY];
            pathVisited[startY][startX] = pathVisitedToken;
        }

        let cx = path[startStep][0];
        let cy = path[startStep][1];
        let target: { tx: number; ty: number } | null = null;
        let bfsCooldown = 0;

        for (let t = startStep; t < S; t++) {
            const remainingSteps = S - t;
            let nextX = cx;
            let nextY = cy;

            let validMovesCount = 0;
            validMovesX[0] = cx;
            validMovesY[0] = cy;
            validMovesCount = 1;

            for (let i = 0; i < 4; i++) {
                const nx = cx + dirsX[i];
                const ny = cy + dirsY[i];
                if (nx >= 0 && nx < W && ny >= 0 && ny < H && probability[ny][nx] !== -1) {
                    validMovesX[validMovesCount] = nx;
                    validMovesY[validMovesCount] = ny;
                    validMovesCount++;
                }
            }

            let bestMoveX = validMovesX[0];
            let bestMoveY = validMovesY[0];
            let bestRolloutScore = -1;
            
            const rolloutLen = Math.min(6, remainingSteps);
            const numRollouts = 5;

            for (let i = 0; i < validMovesCount; i++) {
                const mx = validMovesX[i];
                const my = validMovesY[i];
                let score = 0;
                if (mx !== cx || my !== cy) {
                    score = scoreMoveWithRollouts(mx, my, numRollouts, rolloutLen);
                }
                if (score > bestRolloutScore) {
                    bestRolloutScore = score;
                    bestMoveX = mx;
                    bestMoveY = my;
                }
            }

            if (bestRolloutScore > 0) {
                nextX = bestMoveX;
                nextY = bestMoveY;
                target = null;
            } else {
                let targetValid = false;
                if (target !== null) {
                    const { tx, ty } = target;
                    const targetStillUnvisited = globalVisitedCount[ty][tx] === 0 && pathVisited[ty][tx] !== pathVisitedToken && probability[ty][tx] > 0;
                    if (targetStillUnvisited) {
                        const currDist = Math.abs(cx - tx) + Math.abs(cy - ty);
                        let bestDist = currDist;
                        let bestMoveForTargetX = cx;
                        let bestMoveForTargetY = cy;
                        let moved = false;

                        for (let i = 0; i < 4; i++) {
                            const nx = cx + dirsX[i];
                            const ny = cy + dirsY[i];
                            if (nx >= 0 && nx < W && ny >= 0 && ny < H && probability[ny][nx] !== -1) {
                                const dist = Math.abs(nx - tx) + Math.abs(ny - ty);
                                if (dist < bestDist) {
                                    bestDist = dist;
                                    bestMoveForTargetX = nx;
                                    bestMoveForTargetY = ny;
                                    moved = true;
                                }
                            }
                        }
                        if (moved) {
                            nextX = bestMoveForTargetX;
                            nextY = bestMoveForTargetY;
                            targetValid = true;
                        }
                    }
                }

                if (!targetValid) {
                    if (bfsCooldown > 0) {
                        bfsCooldown--;
                        target = null;
                    } else {
                        const bfsRes = findNearestReward(cx, cy, 45);
                        if (bfsRes !== null) {
                            target = { tx: bfsRes.tx, ty: bfsRes.ty };
                            nextX = bfsRes.nextX;
                            nextY = bfsRes.nextY;
                        } else {
                            target = null;
                            bfsCooldown = 15;
                        }
                    }

                    if (target === null) {
                        let bestScore = -Infinity;
                        let bestMoveIndex = 0;

                        for (let i = 0; i < validMovesCount; i++) {
                            const nx = validMovesX[i];
                            const ny = validMovesY[i];
                            const visitedPenalty = globalVisitedCount[ny][nx] * 10;
                            const stayPenalty = (nx === cx && ny === cy) ? 1.0 : 0.0;
                            const pot = staticPotential[ny][nx];
                            const score = pot - visitedPenalty - stayPenalty;
                            
                            const jitterScore = score + Math.random() * 1e-5;
                            if (jitterScore > bestScore) {
                                bestScore = jitterScore;
                                bestMoveIndex = i;
                            }
                        }
                        nextX = validMovesX[bestMoveIndex];
                        nextY = validMovesY[bestMoveIndex];
                    }
                }
            }

            path[t + 1] = [nextX, nextY];
            pathVisited[nextY][nextX] = pathVisitedToken;
            cx = nextX;
            cy = nextY;
        }

        return path;
    }

    function addPathToGlobal(path: [number, number][]) {
        tempSeenToken++;
        for (let i = 0; i < path.length; i++) {
            const [x, y] = path[i];
            if (tempSeen[y][x] !== tempSeenToken) {
                tempSeen[y][x] = tempSeenToken;
                globalVisitedCount[y][x]++;
            }
        }
    }

    function removePathFromGlobal(path: [number, number][]) {
        tempSeenToken++;
        for (let i = 0; i < path.length; i++) {
            const [x, y] = path[i];
            if (tempSeen[y][x] !== tempSeenToken) {
                tempSeen[y][x] = tempSeenToken;
                globalVisitedCount[y][x]--;
            }
        }
    }

    const paths: [number, number][][] = [];
    for (let d = 0; d < D; d++) {
        const drone = drones[d];
        const p = generatePath(drone.x, drone.y, null);
        paths.push(p);
        addPathToGlobal(p);
    }

    const startTime = perf.now();
    const timeLimit = 850; // MS
    let iter = 0;

    while (true) {
        if (iter % 100 === 0) {
            if (perf.now() - startTime > timeLimit) {
                break;
            }
        }
        iter++;

        const d = Math.floor(Math.random() * D);
        const drone = drones[d];
        const oldPath = paths[d];

        tempSeenToken++;
        let loss = 0;
        for (let i = 0; i < oldPath.length; i++) {
            const [x, y] = oldPath[i];
            if (tempSeen[y][x] !== tempSeenToken) {
                tempSeen[y][x] = tempSeenToken;
                if (globalVisitedCount[y][x] === 1) {
                    if (probability[y][x] > 0) {
                        loss += probability[y][x];
                    }
                }
            }
        }

        removePathFromGlobal(oldPath);

        const cutStep = Math.floor(Math.random() * S);
        const prefix = oldPath.slice(0, cutStep + 1);

        const newPath = generatePath(drone.x, drone.y, prefix);

        tempSeenToken++;
        let gain = 0;
        for (let i = 0; i < newPath.length; i++) {
            const [x, y] = newPath[i];
            if (tempSeen[y][x] !== tempSeenToken) {
                tempSeen[y][x] = tempSeenToken;
                if (globalVisitedCount[y][x] === 0) {
                    if (probability[y][x] > 0) {
                        gain += probability[y][x];
                    }
                }
            }
        }

        const scoreDiff = gain - loss;
        let accept = false;
        if (scoreDiff > 0) {
            accept = true;
        } else if (scoreDiff === 0) {
            accept = Math.random() < 0.2;
        }

        if (accept) {
            paths[d] = newPath;
            addPathToGlobal(newPath);
        } else {
            addPathToGlobal(oldPath);
        }
    }

    console.log(JSON.stringify({ paths }));
}

main();
