use std::collections::VecDeque;
use std::io::{self, BufRead, Write};

struct Grid {
    width: usize,
    data: Vec<u8>,
}

impl Grid {
    fn new(width: usize, height: usize) -> Self {
        let size = (width * height + 1) / 2;
        Self {
            width,
            data: vec![0; size],
        }
    }

    #[inline(always)]
    fn get(&self, r: usize, c: usize) -> u8 {
        let idx = r * self.width + c;
        let byte_idx = idx >> 1;
        let shift = (idx & 1) << 2;
        (self.data[byte_idx] >> shift) & 0x0F
    }

    #[inline(always)]
    fn set(&mut self, r: usize, c: usize, val: u8) {
        let idx = r * self.width + c;
        let byte_idx = idx >> 1;
        let shift = (idx & 1) << 2;
        let mask = !(0x0F << shift);
        self.data[byte_idx] = (self.data[byte_idx] & mask) | ((val & 0x0F) << shift);
    }
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let stdin = io::stdin();
    let mut reader = io::BufReader::with_capacity(16 * 1024 * 1024, stdin.lock());
    let mut line = String::new();

    // Read H and W
    while reader.read_line(&mut line)? > 0 {
        let trimmed = line.trim();
        if !trimmed.is_empty() {
            break;
        }
        line.clear();
    }
    
    let mut parts = line.split_whitespace();
    let h: usize = match parts.next() {
        Some(s) => s.parse()?,
        None => {
            println!("0");
            return Ok(());
        }
    };
    let w: usize = match parts.next() {
        Some(s) => s.parse()?,
        None => {
            println!("0");
            return Ok(());
        }
    };
    line.clear();

    // Allocate grid
    let mut grid = Grid::new(w, h);
    let mut start_pos = None;
    let mut goal_pos = None;

    let mut r = 0;
    while r < h {
        line.clear();
        if reader.read_line(&mut line)? == 0 {
            break;
        }
        let trimmed = line.trim_end_matches(|c| c == '\r' || c == '\n');
        if trimmed.is_empty() && r == 0 {
            continue;
        }
        let bytes = trimmed.as_bytes();
        let limit = std::cmp::min(w, bytes.len());
        for c in 0..limit {
            let ch = bytes[c];
            match ch {
                b'#' => grid.set(r, c, 1),
                b'A' => {
                    grid.set(r, c, 2);
                    start_pos = Some((r as u16, c as u16));
                }
                b'B' => {
                    grid.set(r, c, 3);
                    goal_pos = Some((r as u16, c as u16));
                }
                _ => {}
            }
        }
        r += 1;
    }

    let start_pos = match start_pos {
        Some(p) => p,
        None => {
            println!("0");
            return Ok(());
        }
    };
    let goal_pos = match goal_pos {
        Some(p) => p,
        None => {
            println!("0");
            return Ok(());
        }
    };

    if start_pos == goal_pos {
        println!("1");
        println!("{} {}", start_pos.1, start_pos.0);
        return Ok(());
    }

    let mut q_start = VecDeque::with_capacity(100_000);
    let mut q_goal = VecDeque::with_capacity(100_000);

    q_start.push_back(start_pos);
    q_goal.push_back(goal_pos);

    let mut path = None;

    while !q_start.is_empty() && !q_goal.is_empty() {
        if q_start.len() <= q_goal.len() {
            let curr = q_start.pop_front().unwrap();
            let r = curr.0;
            let c = curr.1;

            // Neighbors
            // Up
            if r > 0 {
                let nr = r - 1;
                let nc = c;
                let n_val = grid.get(nr as usize, nc as usize);
                if n_val == 0 {
                    grid.set(nr as usize, nc as usize, 5); // Parent is Down
                    q_start.push_back((nr, nc));
                } else if (8..=11).contains(&n_val) || n_val == 3 {
                    path = Some(reconstruct_path(
                        &grid,
                        start_pos,
                        goal_pos,
                        curr,
                        (nr, nc),
                    ));
                    break;
                }
            }
            // Down
            if (r as usize) + 1 < h {
                let nr = r + 1;
                let nc = c;
                let n_val = grid.get(nr as usize, nc as usize);
                if n_val == 0 {
                    grid.set(nr as usize, nc as usize, 4); // Parent is Up
                    q_start.push_back((nr, nc));
                } else if (8..=11).contains(&n_val) || n_val == 3 {
                    path = Some(reconstruct_path(
                        &grid,
                        start_pos,
                        goal_pos,
                        curr,
                        (nr, nc),
                    ));
                    break;
                }
            }
            // Left
            if c > 0 {
                let nr = r;
                let nc = c - 1;
                let n_val = grid.get(nr as usize, nc as usize);
                if n_val == 0 {
                    grid.set(nr as usize, nc as usize, 7); // Parent is Right
                    q_start.push_back((nr, nc));
                } else if (8..=11).contains(&n_val) || n_val == 3 {
                    path = Some(reconstruct_path(
                        &grid,
                        start_pos,
                        goal_pos,
                        curr,
                        (nr, nc),
                    ));
                    break;
                }
            }
            // Right
            if (c as usize) + 1 < w {
                let nr = r;
                let nc = c + 1;
                let n_val = grid.get(nr as usize, nc as usize);
                if n_val == 0 {
                    grid.set(nr as usize, nc as usize, 6); // Parent is Left
                    q_start.push_back((nr, nc));
                } else if (8..=11).contains(&n_val) || n_val == 3 {
                    path = Some(reconstruct_path(
                        &grid,
                        start_pos,
                        goal_pos,
                        curr,
                        (nr, nc),
                    ));
                    break;
                }
            }
        } else {
            let curr = q_goal.pop_front().unwrap();
            let r = curr.0;
            let c = curr.1;

            // Neighbors
            // Up
            if r > 0 {
                let nr = r - 1;
                let nc = c;
                let n_val = grid.get(nr as usize, nc as usize);
                if n_val == 0 {
                    grid.set(nr as usize, nc as usize, 9); // Parent is Down
                    q_goal.push_back((nr, nc));
                } else if (4..=7).contains(&n_val) || n_val == 2 {
                    path = Some(reconstruct_path(
                        &grid,
                        start_pos,
                        goal_pos,
                        (nr, nc),
                        curr,
                    ));
                    break;
                }
            }
            // Down
            if (r as usize) + 1 < h {
                let nr = r + 1;
                let nc = c;
                let n_val = grid.get(nr as usize, nc as usize);
                if n_val == 0 {
                    grid.set(nr as usize, nc as usize, 8); // Parent is Up
                    q_goal.push_back((nr, nc));
                } else if (4..=7).contains(&n_val) || n_val == 2 {
                    path = Some(reconstruct_path(
                        &grid,
                        start_pos,
                        goal_pos,
                        (nr, nc),
                        curr,
                    ));
                    break;
                }
            }
            // Left
            if c > 0 {
                let nr = r;
                let nc = c - 1;
                let n_val = grid.get(nr as usize, nc as usize);
                if n_val == 0 {
                    grid.set(nr as usize, nc as usize, 11); // Parent is Right
                    q_goal.push_back((nr, nc));
                } else if (4..=7).contains(&n_val) || n_val == 2 {
                    path = Some(reconstruct_path(
                        &grid,
                        start_pos,
                        goal_pos,
                        (nr, nc),
                        curr,
                    ));
                    break;
                }
            }
            // Right
            if (c as usize) + 1 < w {
                let nr = r;
                let nc = c + 1;
                let n_val = grid.get(nr as usize, nc as usize);
                if n_val == 0 {
                    grid.set(nr as usize, nc as usize, 10); // Parent is Left
                    q_goal.push_back((nr, nc));
                } else if (4..=7).contains(&n_val) || n_val == 2 {
                    path = Some(reconstruct_path(
                        &grid,
                        start_pos,
                        goal_pos,
                        (nr, nc),
                        curr,
                    ));
                    break;
                }
            }
        }
    }

    match path {
        Some(p) => {
            print_path(&p)?;
        }
        None => {
            println!("0");
        }
    }

    Ok(())
}

fn reconstruct_path(
    grid: &Grid,
    start_pos: (u16, u16),
    goal_pos: (u16, u16),
    forward_meet: (u16, u16),
    backward_meet: (u16, u16),
) -> Vec<(u16, u16)> {
    let mut f_path = Vec::new();
    let mut r = forward_meet.0;
    let mut c = forward_meet.1;
    f_path.push((r, c));
    while (r, c) != start_pos {
        let val = grid.get(r as usize, c as usize);
        match val {
            4 => r -= 1,
            5 => r += 1,
            6 => c -= 1,
            7 => c += 1,
            _ => break,
        }
        f_path.push((r, c));
    }
    f_path.reverse();

    let mut b_path = Vec::new();
    r = backward_meet.0;
    c = backward_meet.1;
    b_path.push((r, c));
    while (r, c) != goal_pos {
        let val = grid.get(r as usize, c as usize);
        match val {
            8 => r -= 1,
            9 => r += 1,
            10 => c -= 1,
            11 => c += 1,
            _ => break,
        }
        b_path.push((r, c));
    }

    f_path.extend(b_path);
    f_path
}

fn print_path(path: &[(u16, u16)]) -> io::Result<()> {
    let stdout = io::stdout();
    let mut writer = io::BufWriter::with_capacity(16 * 1024 * 1024, stdout.lock());
    
    let mut buf = [0u8; 24];
    let len_str = format_u64(&mut buf, path.len() as u64);
    writer.write_all(len_str)?;
    writer.write_all(b"\n")?;
    
    for &(r, c) in path {
        let c_str = format_u64(&mut buf, c as u64);
        writer.write_all(c_str)?;
        writer.write_all(b" ")?;
        let r_str = format_u64(&mut buf, r as u64);
        writer.write_all(r_str)?;
        writer.write_all(b"\n")?;
    }
    writer.flush()?;
    Ok(())
}

fn format_u64(buf: &mut [u8; 24], mut val: u64) -> &[u8] {
    if val == 0 {
        buf[23] = b'0';
        return &buf[23..24];
    }
    let mut idx = 24;
    while val > 0 {
        idx -= 1;
        buf[idx] = b'0' + (val % 10) as u8;
        val /= 10;
    }
    &buf[idx..24]
}
