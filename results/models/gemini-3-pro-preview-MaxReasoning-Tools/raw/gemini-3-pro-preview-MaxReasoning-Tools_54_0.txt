import importlib.util
import sys
import os

def main():
    # Read the autopilot code from answer.txt
    if not os.path.exists("answer.txt"):
        print("answer.txt does not exist yet. Please create it.")
        return

    with open("answer.txt", "r", encoding="utf-8") as f:
        code = f.read()

    # Load 54.py from parent directory
    spec = importlib.util.spec_from_file_location("problem_54", "../54.py")
    problem_54 = importlib.util.module_from_spec(spec)
    sys.path.insert(0, "..")
    try:
        spec.loader.exec_module(problem_54)
    except Exception as e:
        print(f"Failed to load 54.py: {e}")
        return

    print("Successfully loaded 54.py and simulator.")
    print("Running 25 scenarios...")
    print("-" * 80)

    total_score = 0.0
    passed_scenarios = 0

    for i in range(25):
        sc = problem_54.SCENARIOS[i]
        # We wrap the code in the structure expected by gradeAnswer
        result_dict = {'python_code': code}
        try:
            score, msg = problem_54.gradeAnswer(result_dict, i, "local_test")
            print(f"Scenario {i:2d} [{sc['name']}]: Score = {score:.2f}")
            print(f"    Detail: {msg}")
            total_score += score
            if score > 0.0:
                passed_scenarios += 1
        except Exception as e:
            print(f"Scenario {i:2d} [{sc['name']}]: ERROR: {e}")
            import traceback
            traceback.print_exc()
        print("-" * 80)

    print(f"Summary: Passed {passed_scenarios}/25 scenarios. Total score: {total_score:.2f} (Max: 25.00)")

if __name__ == "__main__":
    main()
