#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <cstring>
#include <thread>
#include <mutex>
#include <atomic>

using namespace std;

// Suffix Automaton (SAM) implementation optimized for memory and cache locality
struct SuffixAutomaton {
    struct State {
        int len;
        int link;
        int first_pos; // End position of the first occurrence of this state in the string
        int head_trans; // Index of the first transition in the global pool (if num_trans > 3)
        int num_trans;  // Total number of transitions from this state
        char inline_chars[3]; // Inline transitions (characters) to avoid heap allocation for small sets
        int inline_tos[3];    // Inline transitions (target states)
    };

    struct Transition {
        char c;
        int to;
        int next;
    };

    vector<State> states;
    vector<Transition> trans;
    vector<int> root_trans; // Flat table for root state (state 0) to guarantee O(1) lookups
    int last;

    SuffixAutomaton(const string& s) : root_trans(256, -1) {
        // Pre-allocate memory to avoid reallocations during construction
        states.reserve(2 * s.length() + 2);
        trans.reserve(3 * s.length() + 2);

        // Root state
        states.push_back({0, -1, -1, -1, 0, {}, {}});
        last = 0;

        for (int i = 0; i < (int)s.length(); i++) {
            extend(s[i], i);
        }
    }

    void add_transition(int from, int to, char c) {
        if (from == 0) {
            root_trans[(unsigned char)c] = to;
        }
        auto& st = states[from];
        if (st.num_trans < 3) {
            st.inline_chars[st.num_trans] = c;
            st.inline_tos[st.num_trans] = to;
            st.num_trans++;
        } else {
            if (st.num_trans == 3) {
                // Spill inline transitions to the pool
                for (int i = 0; i < 3; i++) {
                    trans.push_back({st.inline_chars[i], st.inline_tos[i], st.head_trans});
                    st.head_trans = (int)trans.size() - 1;
                }
            }
            trans.push_back({c, to, st.head_trans});
            st.head_trans = (int)trans.size() - 1;
            st.num_trans++;
        }
    }

    inline int get_transition(int u, char c) const {
        if (u == 0) {
            return root_trans[(unsigned char)c];
        }
        const auto& st = states[u];
        if (st.num_trans <= 3) {
            for (int i = 0; i < st.num_trans; i++) {
                if (st.inline_chars[i] == c) return st.inline_tos[i];
            }
            return -1;
        } else {
            for (int t = st.head_trans; t != -1; t = trans[t].next) {
                if (trans[t].c == c) return trans[t].to;
            }
            return -1;
        }
    }

    void extend(char c, int idx) {
        int cur = states.size();
        states.push_back({states[last].len + 1, -1, idx, -1, 0, {}, {}});
        
        int p = last;
        while (p != -1 && get_transition(p, c) == -1) {
            add_transition(p, cur, c);
            p = states[p].link;
        }

        if (p == -1) {
            states[cur].link = 0;
        } else {
            int q = get_transition(p, c);
            if (states[p].len + 1 == states[q].len) {
                states[cur].link = q;
            } else {
                int clone = states.size();
                states.push_back({states[p].len + 1, states[q].link, states[q].first_pos, -1, 0, {}, {}});
                
                // Copy transitions of q to clone
                const auto& sq = states[q];
                if (sq.num_trans <= 3) {
                    for (int i = 0; i < sq.num_trans; i++) {
                        add_transition(clone, sq.inline_tos[i], sq.inline_chars[i]);
                    }
                } else {
                    for (int t = sq.head_trans; t != -1; t = trans[t].next) {
                        add_transition(clone, trans[t].to, trans[t].c);
                    }
                }

                // Redirect transitions going to q to clone instead
                while (p != -1 && get_transition(p, c) == q) {
                    if (p == 0) {
                        root_trans[(unsigned char)c] = clone;
                    }
                    auto& st = states[p];
                    if (st.num_trans <= 3) {
                        for (int i = 0; i < st.num_trans; i++) {
                            if (st.inline_chars[i] == c) {
                                st.inline_tos[i] = clone;
                                break;
                            }
                        }
                    } else {
                        for (int t = st.head_trans; t != -1; t = trans[t].next) {
                            if (trans[t].c == c) {
                                trans[t].to = clone;
                                break;
                            }
                        }
                    }
                    p = states[p].link;
                }
                states[q].link = states[cur].link = clone;
            }
        }
        last = cur;
    }
};

string solve_lcs(const vector<string>& strings) {
    if (strings.empty()) return "";
    if (strings.size() == 1) return strings[0];

    // Find the shortest string to build the SAM on (minimizes memory and states)
    int min_idx = 0;
    for (size_t i = 1; i < strings.size(); i++) {
        if (strings[i].length() < strings[min_idx].length()) {
            min_idx = i;
        }
    }

    const string& s_min = strings[min_idx];
    if (s_min.empty()) return "";

    // Build SAM for the shortest string
    SuffixAutomaton sam(s_min);
    int num_states = sam.states.size();

    // min_of_max[u] tracks the minimum (across all strings) of the maximum match length at state u
    vector<int> min_of_max(num_states);
    for (int i = 0; i < num_states; i++) {
        min_of_max[i] = sam.states[i].len;
    }

    // Topological sort of states by length (using fast bucket sort)
    vector<int> cnt(s_min.length() + 1, 0);
    for (int i = 0; i < num_states; i++) cnt[sam.states[i].len]++;
    for (size_t i = 1; i <= s_min.length(); i++) cnt[i] += cnt[i-1];
    vector<int> sorted_states(num_states);
    for (int i = 0; i < num_states; i++) {
        sorted_states[--cnt[sam.states[i].len]] = i;
    }

    // Identify indices of strings to match
    vector<size_t> work_indices;
    long long total_length = 0;
    for (size_t i = 0; i < strings.size(); i++) {
        if (i != (size_t)min_idx) {
            work_indices.push_back(i);
            total_length += strings[i].length();
        }
    }

    // Adaptive scale parallelization logic:
    // Determine the optimal number of threads based on CPU core count,
    // problem size, and memory constraints to avoid OOM for ludicrous scales.
    int max_threads = (int)thread::hardware_concurrency();
    if (max_threads <= 0) max_threads = 1;
    
    int num_threads = 1;
    if (total_length > 200000 && work_indices.size() > 1) {
        // Keep combined memory for all thread-local match vectors under ~2 GB
        int max_mem_threads = 2000000000LL / (1LL * num_states * sizeof(int));
        if (max_mem_threads < 1) max_mem_threads = 1;
        
        num_threads = min({(int)work_indices.size(), max_threads, max_mem_threads});
    }

    mutex mtx;
    atomic<size_t> next_string_idx(0);
    vector<thread> workers;

    for (int t = 0; t < num_threads; t++) {
        workers.emplace_back([&]() {
            // Reusable thread-local arrays to eliminate allocation overhead
            vector<int> temp_max(num_states, 0);
            vector<int> visited_states;
            visited_states.reserve(num_states);

            while (true) {
                size_t idx = next_string_idx.fetch_add(1);
                if (idx >= work_indices.size()) break;

                size_t str_idx = work_indices[idx];
                const string& t_str = strings[str_idx];

                // 1. Tracing Phase
                int curr = 0;
                int curr_len = 0;
                for (char c : t_str) {
                    int next_state = -1;
                    while (curr != -1) {
                        next_state = sam.get_transition(curr, c);
                        if (next_state != -1) break;
                        curr = sam.states[curr].link;
                        if (curr != -1) curr_len = sam.states[curr].len;
                    }
                    if (curr == -1) {
                        curr = 0;
                        curr_len = 0;
                    } else {
                        curr = next_state;
                        curr_len++;
                    }
                    if (temp_max[curr] == 0) {
                        visited_states.push_back(curr);
                    }
                    temp_max[curr] = max(temp_max[curr], curr_len);
                }

                // 2. Propagation Phase: propagate match lengths down suffix link tree
                for (int j = num_states - 1; j > 0; j--) {
                    int u = sorted_states[j];
                    if (temp_max[u] == 0) continue; // Skip non-visited branches for O(active_states) speed
                    int p = sam.states[u].link;
                    if (p != -1) {
                        if (temp_max[p] == 0) {
                            visited_states.push_back(p);
                        }
                        temp_max[p] = max(temp_max[p], min(sam.states[p].len, temp_max[u]));
                    }
                }

                // 3. Merging Phase: update global min_of_max
                {
                    lock_guard<mutex> lock(mtx);
                    for (int j = 0; j < num_states; j++) {
                        min_of_max[j] = min(min_of_max[j], temp_max[j]);
                    }
                }

                // Reset temp_max for only active states (much faster than full vector fill)
                for (int u : visited_states) {
                    temp_max[u] = 0;
                }
                visited_states.clear();
            }
        });
    }

    // Join all workers
    for (auto& worker : workers) {
        worker.join();
    }

    // Find the state with the maximum value in min_of_max
    int best_len = 0;
    int best_state = -1;
    for (int i = 1; i < num_states; i++) {
        if (min_of_max[i] > best_len) {
            best_len = min_of_max[i];
            best_state = i;
        }
    }

    if (best_state == -1 || best_len == 0) return "";

    // Recover the substring from first_pos
    int end_pos = sam.states[best_state].first_pos;
    return s_min.substr(end_pos - best_len + 1, best_len);
}

int main() {
    // Optimize standard C++ streams for fast I/O
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int N;
    if (!(cin >> N)) return 0;

    // Consume the trailing newline character
    string dummy;
    getline(cin, dummy);

    vector<string> strings(N);
    bool has_empty = false;
    for (int i = 0; i < N; i++) {
        if (!getline(cin, strings[i])) {
            strings.resize(i);
            break;
        }
        // Remove trailing carriage returns to support Windows line endings correctly
        if (!strings[i].empty() && strings[i].back() == '\r') {
            strings[i].pop_back();
        }
        if (strings[i].empty()) {
            has_empty = true;
        }
    }

    if (has_empty) {
        cout << "\n";
        return 0;
    }

    string ans = solve_lcs(strings);
    cout << ans << "\n";

    return 0;
}
