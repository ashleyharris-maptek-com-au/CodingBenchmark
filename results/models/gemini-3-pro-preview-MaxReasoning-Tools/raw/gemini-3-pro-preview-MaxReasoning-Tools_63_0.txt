declare const require: any;
const fs = require('fs');

interface Robot {
    id: number;
    x: number;
    y: number;
}

interface Job {
    id: number;
    x: number;
    y: number;
    priority: number;
}

interface InputData {
    grid: string[];
    robots: Robot[];
    jobs: Job[];
}

interface Point {
    x: number;
    y: number;
}

function main() {
    const input: string = fs.readFileSync(0, 'utf8');
    const data: InputData = JSON.parse(input);

    const grid = data.grid;
    const robots = data.robots;
    const jobs = data.jobs;

    const h = grid.length;
    const w = h > 0 ? grid[0].length : 0;

    // Determine if we should use Manhattan distance (large cases) or grid walk distance (small cases)
    // The grader uses Manhattan distance if jobs.length > 150.
    const isLarge = jobs.length > 150;

    // Cache for BFS grid walk distances
    const bfsCache = new Map<number, Map<number, number>>();

    function computeBfsDistances(sx: number, sy: number): Map<number, number> {
        const dist = new Map<number, number>();
        const q: number[] = [];
        let head = 0;

        const startKey = sy * w + sx;
        dist.set(startKey, 0);
        q.push(startKey);

        const dx = [1, -1, 0, 0];
        const dy = [0, 0, 1, -1];

        while (head < q.length) {
            const key = q[head++];
            const cx = key % w;
            const cy = Math.floor(key / w);
            const d = dist.get(key)!;

            for (let i = 0; i < 4; i++) {
                const nx = cx + dx[i];
                const ny = cy + dy[i];
                if (nx >= 0 && nx < w && ny >= 0 && ny < h) {
                    if (grid[ny][nx] !== '#') {
                        const nkey = ny * w + nx;
                        if (!dist.has(nkey)) {
                            dist.set(nkey, d + 1);
                            q.push(nkey);
                        }
                    }
                }
            }
        }
        return dist;
    }

    function getGridDistance(x1: number, y1: number, x2: number, y2: number): number {
        const key1 = y1 * w + x1;
        let distMap = bfsCache.get(key1);
        if (!distMap) {
            distMap = computeBfsDistances(x1, y1);
            bfsCache.set(key1, distMap);
        }
        const key2 = y2 * w + x2;
        return distMap.get(key2) ?? 1000000000;
    }

    const getDistance = isLarge
        ? (x1: number, y1: number, x2: number, y2: number) => Math.abs(x1 - x2) + Math.abs(y1 - y2)
        : (x1: number, y1: number, x2: number, y2: number) => getGridDistance(x1, y1, x2, y2);

    // Map job ID to job point
    const jobsMap = new Map<number, Point>();
    for (const j of jobs) {
        jobsMap.set(j.id, { x: j.x, y: j.y });
    }

    // Robot starts
    const robotStarts: Point[] = robots.map(r => ({ x: r.x, y: r.y }));

    // 1. Initial Greedy Assignment (matching reference structure)
    const routes: number[][] = robots.map(() => []);
    const currentPos = robots.map(r => ({ x: r.x, y: r.y }));
    const remainingJobs = new Set(jobs.map(j => j.id));

    while (remainingJobs.size > 0) {
        let bestDist = Infinity;
        let bestRobotIdx = -1;
        let bestJobId = -1;

        for (let ri = 0; ri < robots.length; ri++) {
            const rPos = currentPos[ri];
            for (const j of jobs) {
                if (!remainingJobs.has(j.id)) continue;
                const d = getDistance(rPos.x, rPos.y, j.x, j.y);
                if (d < bestDist) {
                    bestDist = d;
                    bestRobotIdx = ri;
                    bestJobId = j.id;
                }
            }
        }

        routes[bestRobotIdx].push(bestJobId);
        const assignedJob = jobsMap.get(bestJobId)!;
        currentPos[bestRobotIdx] = { x: assignedJob.x, y: assignedJob.y };
        remainingJobs.delete(bestJobId);
    }

    // Helpers for local search cost calculations
    function removeCostChange(ri: number, idx: number): number {
        const route = routes[ri];
        const start = robotStarts[ri];
        const job = jobsMap.get(route[idx])!;
        const prev = idx === 0 ? start : jobsMap.get(route[idx - 1])!;
        const next = idx === route.length - 1 ? null : jobsMap.get(route[idx + 1])!;

        const oldCost = getDistance(prev.x, prev.y, job.x, job.y) + (next ? getDistance(job.x, job.y, next.x, next.y) : 0);
        const newCost = next ? getDistance(prev.x, prev.y, next.x, next.y) : 0;
        return newCost - oldCost;
    }

    // idx is the position where the job jid will be inserted.
    // The element previously at idx (if any) will become the succeeding element.
    function insertCostChange(ri: number, idx: number, jid: number): number {
        const route = routes[ri];
        const start = robotStarts[ri];
        const job = jobsMap.get(jid)!;
        const prev = idx === 0 ? start : jobsMap.get(route[idx - 1])!;
        const next = idx === route.length ? null : jobsMap.get(route[idx])!;

        const oldCost = next ? getDistance(prev.x, prev.y, next.x, next.y) : 0;
        const newCost = getDistance(prev.x, prev.y, job.x, job.y) + (next ? getDistance(job.x, job.y, next.x, next.y) : 0);
        return newCost - oldCost;
    }

    function replaceCostChange(ri: number, idx: number, newJid: number): number {
        const route = routes[ri];
        const start = robotStarts[ri];
        const oldJob = jobsMap.get(route[idx])!;
        const newJob = jobsMap.get(newJid)!;
        const prev = idx === 0 ? start : jobsMap.get(route[idx - 1])!;
        const next = idx === route.length - 1 ? null : jobsMap.get(route[idx + 1])!;

        const oldCost = getDistance(prev.x, prev.y, oldJob.x, oldJob.y) + (next ? getDistance(oldJob.x, oldJob.y, next.x, next.y) : 0);
        const newCost = getDistance(prev.x, prev.y, newJob.x, newJob.y) + (next ? getDistance(newJob.x, newJob.y, next.x, next.y) : 0);
        return newCost - oldCost;
    }

    function optimizeRoute2Opt(ri: number) {
        const start = robotStarts[ri];
        const route = routes[ri];
        if (route.length <= 1) return;

        let improved = true;
        const path = [start, ...route.map(id => jobsMap.get(id)!)];
        const n = path.length;

        while (improved) {
            improved = false;
            for (let i = 1; i < n - 1; i++) {
                for (let j = i + 1; j < n; j++) {
                    const d_old = getDistance(path[i - 1].x, path[i - 1].y, path[i].x, path[i].y) +
                        (j + 1 < n ? getDistance(path[j].x, path[j].y, path[j + 1].x, path[j + 1].y) : 0);
                    const d_new = getDistance(path[i - 1].x, path[i - 1].y, path[j].x, path[j].y) +
                        (j + 1 < n ? getDistance(path[i].x, path[i].y, path[j + 1].x, path[j + 1].y) : 0);

                    if (d_new < d_old - 1e-9) {
                        // Reverse path[i...j]
                        let l = i, r = j;
                        while (l < r) {
                            const temp = path[l];
                            path[l] = path[r];
                            path[r] = temp;

                            const tempId = route[l - 1];
                            route[l - 1] = route[r - 1];
                            route[r - 1] = tempId;

                            l++;
                            r--;
                        }
                        improved = true;
                    }
                }
            }
        }
    }

    // 2. Local Search refinement loop
    const startTime = Date.now();
    let globalImproved = true;

    while (globalImproved) {
        globalImproved = false;

        // Check time limit (4 seconds)
        if (Date.now() - startTime > 4000) {
            break;
        }

        // A. Optimize each robot's route locally (2-opt)
        for (let ri = 0; ri < robots.length; ri++) {
            const oldRouteStr = JSON.stringify(routes[ri]);
            optimizeRoute2Opt(ri);
            if (JSON.stringify(routes[ri]) !== oldRouteStr) {
                globalImproved = true;
            }
        }

        // B. Try Relocate move between different robots
        let bestRelocate = {
            riA: -1,
            idxA: -1,
            riB: -1,
            idxB: -1,
            delta: 0
        };

        for (let riA = 0; riA < robots.length; riA++) {
            for (let idxA = 0; idxA < routes[riA].length; idxA++) {
                const jid = routes[riA][idxA];
                const delA = removeCostChange(riA, idxA);

                for (let riB = 0; riB < robots.length; riB++) {
                    if (riA === riB) continue;
                    for (let idxB = 0; idxB <= routes[riB].length; idxB++) {
                        const delB = insertCostChange(riB, idxB, jid);
                        const delta = delA + delB;
                        if (delta < bestRelocate.delta - 1e-9) {
                            bestRelocate = { riA, idxA, riB, idxB, delta };
                        }
                    }
                }
            }
        }

        if (bestRelocate.delta < -1e-9) {
            const { riA, idxA, riB, idxB } = bestRelocate;
            const jid = routes[riA][idxA];
            routes[riA].splice(idxA, 1);
            routes[riB].splice(idxB, 0, jid);
            globalImproved = true;
            continue;
        }

        // C. Try Swap move between different robots
        let bestSwap = {
            riA: -1,
            idxA: -1,
            riB: -1,
            idxB: -1,
            delta: 0
        };

        for (let riA = 0; riA < robots.length; riA++) {
            for (let idxA = 0; idxA < routes[riA].length; idxA++) {
                const jidA = routes[riA][idxA];
                for (let riB = riA + 1; riB < robots.length; riB++) {
                    for (let idxB = 0; idxB < routes[riB].length; idxB++) {
                        const jidB = routes[riB][idxB];

                        const delA = replaceCostChange(riA, idxA, jidB);
                        const delB = replaceCostChange(riB, idxB, jidA);
                        const delta = delA + delB;
                        if (delta < bestSwap.delta - 1e-9) {
                            bestSwap = { riA, idxA, riB, idxB, delta };
                        }
                    }
                }
            }
        }

        if (bestSwap.delta < -1e-9) {
            const { riA, idxA, riB, idxB } = bestSwap;
            const jidA = routes[riA][idxA];
            const jidB = routes[riB][idxB];
            routes[riA][idxA] = jidB;
            routes[riB][idxB] = jidA;
            globalImproved = true;
            continue;
        }
    }

    // Format output
    // The grader expects assignments array indices to map to robot ids.
    // Ensure all robots have an assignment entry, matching their id index.
    const maxRobotId = Math.max(...robots.map(r => r.id), robots.length - 1);
    const finalAssignments: number[][] = Array.from({ length: maxRobotId + 1 }, () => []);
    for (let ri = 0; ri < robots.length; ri++) {
        finalAssignments[robots[ri].id] = routes[ri];
    }

    console.log(JSON.stringify({ assignments: finalAssignments }));
}

main();
