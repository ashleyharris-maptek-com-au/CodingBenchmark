const fs = require('fs');

class Xorshift {
  constructor(seed) {
    this.state = seed || 1;
  }
  next() {
    let x = this.state;
    x ^= x << 13;
    x ^= x >> 17;
    x ^= x << 5;
    this.state = x;
    return x >>> 0;
  }
  nextRange(limit) {
    if (limit <= 0) return 0;
    return this.next() % limit;
  }
}

function distance(p1, p2) {
  const dx = p1.x - p2.x;
  const dy = p1.y - p2.y;
  return Math.sqrt(dx * dx + dy * dy);
}

function getTourDistance(tour, depot) {
  if (tour.length === 0) return 0;
  let dist = distance(depot, tour[0]);
  for (let i = 0; i < tour.length - 1; i++) {
    dist += distance(tour[i], tour[i + 1]);
  }
  dist += distance(tour[tour.length - 1], depot);
  return dist;
}

function reverseSubsegment(arr, i, j) {
  while (i < j) {
    const temp = arr[i];
    arr[i] = arr[j];
    arr[j] = temp;
    i++;
    j--;
  }
}

function optimizeTour2Opt(tour, depot, maxTimeMs) {
  const M = tour.length;
  if (M <= 1) return tour;
  
  const fullTour = [depot, ...tour, depot];
  let improved = true;
  const startTime = Date.now();
  
  // Choose window size
  const W = M <= 500 ? M : 100;
  
  const dontLook = new Uint8Array(M + 2);
  
  while (improved) {
    improved = false;
    if (Date.now() - startTime > maxTimeMs) break;
    
    for (let i = 1; i <= M; i++) {
      if (dontLook[i]) continue;
      let localImprovement = false;
      
      const limit = Math.min(M, i + W);
      for (let j = i + 1; j <= limit; j++) {
        const p_i_prev = fullTour[i - 1];
        const p_i = fullTour[i];
        const p_j = fullTour[j];
        const p_j_next = fullTour[j + 1];
        
        const d_old = distance(p_i_prev, p_i) + distance(p_j, p_j_next);
        const d_new = distance(p_i_prev, p_j) + distance(p_i, p_j_next);
        
        if (d_new < d_old - 1e-6) {
          reverseSubsegment(fullTour, i, j);
          improved = true;
          localImprovement = true;
          dontLook[i - 1] = 0;
          dontLook[i] = 0;
          dontLook[j] = 0;
          dontLook[j + 1] = 0;
        }
      }
      
      if (!localImprovement) {
        dontLook[i] = 1;
      }
    }
  }
  
  return fullTour.slice(1, M + 1);
}

function solveTSP_Basic(packages, depot, maxTimeMs) {
  if (packages.length <= 1) return packages;
  
  if (packages.length > 2000) {
    const sorted = [...packages].sort((a, b) => {
      const angleA = Math.atan2(a.y - depot.y, a.x - depot.x);
      const angleB = Math.atan2(b.y - depot.y, b.x - depot.x);
      return angleA - angleB;
    });
    return optimizeTour2Opt(sorted, depot, maxTimeMs);
  }
  
  // Nearest Neighbor starting from depot
  const tour = [];
  const visited = new Set();
  let current = depot;
  
  for (let step = 0; step < packages.length; step++) {
    let nearest = null;
    let minDist = Infinity;
    for (let i = 0; i < packages.length; i++) {
      const p = packages[i];
      if (!visited.has(p.id)) {
        const d = distance(current, p);
        if (d < minDist) {
          minDist = d;
          nearest = p;
        }
      }
    }
    tour.push(nearest);
    visited.add(nearest.id);
    current = nearest;
  }
  
  return optimizeTour2Opt(tour, depot, maxTimeMs);
}

function doubleBridgePerturbation(tour, rng) {
  const N = tour.length;
  if (N < 8) return [...tour];
  
  const i = 2 + rng.nextRange(Math.floor(N / 4) - 1);
  const j = i + 2 + rng.nextRange(Math.floor(N / 4) - 1);
  const k = j + 2 + rng.nextRange(Math.floor(N / 4) - 1);
  
  const part1 = tour.slice(0, i);
  const part2 = tour.slice(i, j);
  const part3 = tour.slice(j, k);
  const part4 = tour.slice(k);
  
  return [...part1, ...part4, ...part3, ...part2];
}

function solveTSP(packages, depot, maxTimeMs) {
  const bestTour = solveTSP_Basic(packages, depot, maxTimeMs * 0.4);
  const M = packages.length;
  if (M < 8) return bestTour;
  
  let iterations = 0;
  if (M <= 50) iterations = 30;
  else if (M <= 200) iterations = 15;
  else if (M <= 500) iterations = 5;
  
  if (iterations === 0) return bestTour;
  
  let bestDist = getTourDistance(bestTour, depot);
  let tour = bestTour;
  const rng = new Xorshift(12345);
  const startTime = Date.now();
  
  for (let iter = 0; iter < iterations; iter++) {
    if (Date.now() - startTime > maxTimeMs * 0.6) break;
    
    const perturbed = doubleBridgePerturbation(tour, rng);
    const opt = optimizeTour2Opt(perturbed, depot, maxTimeMs * 0.2);
    const dist = getTourDistance(opt, depot);
    
    if (dist < bestDist - 1e-6) {
      tour = opt;
      bestDist = dist;
    }
  }
  
  return tour;
}

function refineBoundaries(sorted, boundaries, depot, maxTimeMs) {
  const R = boundaries.length - 1;
  let improved = true;
  const startTime = Date.now();
  
  const tours = [];
  const distances = [];
  for (let k = 0; k < R; k++) {
    const slice = sorted.slice(boundaries[k], boundaries[k+1]);
    const tour = solveTSP(slice, depot, 100);
    tours.push(tour);
    distances.push(getTourDistance(tour, depot));
  }
  
  while (improved) {
    improved = false;
    if (Date.now() - startTime > maxTimeMs) break;
    
    for (let k = 1; k < R; k++) {
      // Try boundaries[k] - 1
      if (boundaries[k] > boundaries[k-1] + 1) {
        const slicePrev = sorted.slice(boundaries[k-1], boundaries[k] - 1);
        const sliceCurr = sorted.slice(boundaries[k] - 1, boundaries[k+1]);
        
        const tourPrev = solveTSP(slicePrev, depot, 50);
        const tourCurr = solveTSP(sliceCurr, depot, 50);
        
        const distPrev = getTourDistance(tourPrev, depot);
        const distCurr = getTourDistance(tourCurr, depot);
        
        if (distPrev + distCurr < distances[k-1] + distances[k] - 1e-6) {
          boundaries[k]--;
          tours[k-1] = tourPrev;
          tours[k] = tourCurr;
          distances[k-1] = distPrev;
          distances[k] = distCurr;
          improved = true;
          continue;
        }
      }
      
      // Try boundaries[k] + 1
      if (boundaries[k] < boundaries[k+1] - 1) {
        const slicePrev = sorted.slice(boundaries[k-1], boundaries[k] + 1);
        const sliceCurr = sorted.slice(boundaries[k] + 1, boundaries[k+1]);
        
        const tourPrev = solveTSP(slicePrev, depot, 50);
        const tourCurr = solveTSP(sliceCurr, depot, 50);
        
        const distPrev = getTourDistance(tourPrev, depot);
        const distCurr = getTourDistance(tourCurr, depot);
        
        if (distPrev + distCurr < distances[k-1] + distances[k] - 1e-6) {
          boundaries[k]++;
          tours[k-1] = tourPrev;
          tours[k] = tourCurr;
          distances[k-1] = distPrev;
          distances[k] = distCurr;
          improved = true;
          continue;
        }
      }
    }
  }
  
  return { tours, distances, boundaries };
}

function solveVRP(data) {
  const depot = { id: -1, x: data.depot[0], y: data.depot[1] };
  const packages = data.packages;
  const truckCount = data.truckCount;
  
  const N = packages.length;
  const R = Math.min(truckCount, N);
  
  if (N === 0) {
    return { routes: [] };
  }
  if (R === 1) {
    const tour = solveTSP(packages, depot, 2000);
    return { routes: [tour.map(p => p.id)] };
  }
  
  let S = 16;
  if (N > 1000) S = 8;
  if (N > 5000) S = 4;
  
  let bestSorted = null;
  let bestBoundaries = null;
  let bestTotalDist = Infinity;
  
  const startTime = Date.now();
  const timeLimit = 4000;
  
  for (let s = 0; s < S; s++) {
    if (Date.now() - startTime > timeLimit) break;
    
    const theta = (2 * Math.PI * s) / S;
    
    const sorted = [...packages].sort((a, b) => {
      const angleA = (Math.atan2(a.y - depot.y, a.x - depot.x) - theta + 3 * Math.PI) % (2 * Math.PI);
      const angleB = (Math.atan2(b.y - depot.y, b.x - depot.x) - theta + 3 * Math.PI) % (2 * Math.PI);
      return angleA - angleB;
    });
    
    const boundaries = [];
    for (let k = 0; k <= R; k++) {
      boundaries.push(Math.floor((k * N) / R));
    }
    
    let totalDist = 0;
    const currentTours = [];
    for (let k = 0; k < R; k++) {
      const slice = sorted.slice(boundaries[k], boundaries[k+1]);
      const tour = solveTSP(slice, depot, 100);
      currentTours.push(tour);
      totalDist += getTourDistance(tour, depot);
    }
    
    if (totalDist < bestTotalDist) {
      bestTotalDist = totalDist;
      bestSorted = sorted;
      bestBoundaries = boundaries;
    }
  }
  
  console.log("Distance after Sweep starts:", bestTotalDist);

  // Refine partition boundaries
  const refined = refineBoundaries(bestSorted, bestBoundaries, depot, 2000);
  const refinedDist = refined.distances.reduce((sum, d) => sum + d, 0);
  console.log("Distance after boundary refinement:", refinedDist);
  
  // Final thorough optimization of each route
  const finalRoutes = refined.tours.map(r => solveTSP(r, depot, 1000));
  const finalDist = finalRoutes.reduce((sum, r) => sum + getTourDistance(r, depot), 0);
  console.log("Distance after final optimization:", finalDist);
  
  return {
    routes: finalRoutes.map(r => r.map(p => p.id))
  };
}

// Generate a random VRP input
function generateTestInput(numPackages, truckCount) {
  const depot = [500, 500];
  const packages = [];
  for (let i = 0; i < numPackages; i++) {
    packages.push({
      id: i,
      x: Math.random() * 1000,
      y: Math.random() * 1000
    });
  }
  return {
    depot,
    truckCount,
    packages
  };
}
