#include <iostream>
#include <vector>
#include <algorithm>
#include <chrono>
#include <random>
#include <numeric>

#include <mutex>
#include <future>
#include <string>
#include <cassert>

struct Rectangle {
    int id;
    int w, h;
};

struct Placement {
    int board_idx;
    int x, y;
    int rotated;
};

struct Segment {
    int x, y, w;
};

struct FreeRect {
    int x, y, w, h;
};

struct Score {
    int val1;
    int val2;
    int val3;
    bool operator<(const Score& o) const {
        if (val1 != o.val1) return val1 < o.val1;
        if (val2 != o.val2) return val2 < o.val2;
        return val3 < o.val3;
    }
};

enum class SortCriteria {
    AreaDesc,
    PerimeterDesc,
    MaxDimDesc,
    MinDimDesc,
    WidthDesc,
    HeightDesc,
    RatioDesc,
    RatioAsc,
    SquarishDesc,
    SkinnyDesc
};

enum class Orientation {
    None,
    Horizontal, // w >= h
    Vertical    // h >= w
};

struct Config {
    SortCriteria sort_crit;
    Orientation orient;
    int scoring_mode;
    bool use_waste;
    int priority;
};

void merge_segments(std::vector<Segment>& segments) {
    if (segments.empty()) return;
    size_t write_idx = 0;
    for (size_t i = 1; i < segments.size(); ++i) {
        if (segments[write_idx].y == segments[i].y) {
            segments[write_idx].w += segments[i].w;
        } else {
            write_idx++;
            segments[write_idx] = segments[i];
        }
    }
    segments.resize(write_idx + 1);
}

void prune_waste_map(std::vector<FreeRect>& waste_map, size_t max_size) {
    if (waste_map.size() <= max_size) return;
    std::nth_element(waste_map.begin(), waste_map.begin() + max_size, waste_map.end(),
                     [](const FreeRect& a, const FreeRect& b) {
                         return (long long)a.w * a.h > (long long)b.w * b.h;
                     });
    waste_map.resize(max_size);
}

void place_in_waste_map(size_t best_idx, int w, int h, std::vector<FreeRect>& waste_map) {
    FreeRect F = waste_map[best_idx];
    if (best_idx != waste_map.size() - 1) {
        waste_map[best_idx] = waste_map.back();
    }
    waste_map.pop_back();
    
    int dx = F.w - w;
    int dy = F.h - h;
    if (dx < dy) {
        if (dx > 0) waste_map.push_back({F.x + w, F.y, dx, h});
        if (dy > 0) waste_map.push_back({F.x, F.y + h, F.w, dy});
    } else {
        if (dx > 0) waste_map.push_back({F.x + w, F.y, dx, F.h});
        if (dy > 0) waste_map.push_back({F.x, F.y + h, w, dy});
    }
}

int pack_skyline(const std::vector<Rectangle>& base_rects,
                 const std::vector<Rectangle>& rects,
                 int board_width,
                 int board_height,
                 int scoring_mode,
                 bool use_waste,
                 std::vector<Placement>& placements) {
    placements.resize(rects.size());
    
    std::vector<bool> packed(rects.size(), false);
    
    std::vector<int> unpacked;
    unpacked.reserve(rects.size());
    for (int i = 0; i < (int)rects.size(); ++i) {
        unpacked.push_back(i);
    }
    
    int board_idx = 0;
    
    const size_t MAX_WASTE_MAP_SIZE = (rects.size() < 1000) ? 5000 : 2000;
    
    std::vector<Segment> segments;
    std::vector<FreeRect> waste_map;
    
    while (!unpacked.empty()) {
        segments.clear();
        segments.push_back({0, 0, board_width});
        waste_map.clear();
        
        bool packed_any = false;
        
        for (int idx : unpacked) {
            const auto& r = rects[idx];
            int w = r.w;
            int h = r.h;
            
            // 1. Try to fit in waste map
            int best_waste_idx = -1;
            long long best_leftover = -1;
            bool waste_rotated = false;
            
            if (use_waste) {
                for (size_t j = 0; j < waste_map.size(); ++j) {
                    const auto& F = waste_map[j];
                    if (w <= F.w && h <= F.h) {
                        long long leftover = (long long)F.w * F.h - (long long)w * h;
                        if (best_waste_idx == -1 || leftover < best_leftover) {
                            best_waste_idx = j;
                            best_leftover = leftover;
                            waste_rotated = false;
                        }
                    }
                    if (w != h && h <= F.w && w <= F.h) {
                        long long leftover = (long long)F.w * F.h - (long long)w * h;
                        if (best_waste_idx == -1 || leftover < best_leftover) {
                            best_waste_idx = j;
                            best_leftover = leftover;
                            waste_rotated = true;
                        }
                    }
                }
            }
            
            if (best_waste_idx != -1) {
                int final_w = waste_rotated ? h : w;
                int final_h = waste_rotated ? w : h;
                const auto& F = waste_map[best_waste_idx];
                
                int orig_w = base_rects[r.id].w;
                int rotated_val = (final_w == orig_w) ? 0 : 1;
                
                placements[r.id] = {board_idx, F.x, F.y, rotated_val};
                place_in_waste_map(best_waste_idx, final_w, final_h, waste_map);
                packed[idx] = true;
                packed_any = true;
                continue;
            }
            
            // 2. Try to fit on the skyline
            int best_seg_idx = -1;
            Score best_score = {1000000000, 1000000000, 1000000000};
            bool skyline_rotated = false;
            int best_Y = -1;
            int best_last_idx = -1;
            
            auto check_fit = [&](int cur_w, int cur_h, bool rot) {
                for (size_t i = 0; i < segments.size(); ++i) {
                    int x = segments[i].x;
                    if (x + cur_w > board_width) break;
                    
                    int Y = 0;
                    int last_j = i;
                    for (size_t j = i; j < segments.size(); ++j) {
                        if (segments[j].x >= x + cur_w) break;
                        Y = std::max(Y, segments[j].y);
                        last_j = j;
                    }
                    
                    if (Y + cur_h <= board_height) {
                        Score score;
                        if (scoring_mode == 0) {
                            score = {Y + cur_h, Y, x};
                        } else if (scoring_mode == 1) {
                            score = {Y, Y + cur_h, x};
                        } else if (scoring_mode == 2) {
                            score = {x, Y + cur_h, Y};
                        } else {
                            score = {Y, x, Y + cur_h};
                        }
                        if (best_seg_idx == -1 || score < best_score) {
                            best_score = score;
                            best_seg_idx = i;
                            skyline_rotated = rot;
                            best_Y = Y;
                            best_last_idx = last_j;
                        }
                    }
                }
            };
            
            check_fit(w, h, false);
            if (w != h) {
                check_fit(h, w, true);
            }
            
            if (best_seg_idx != -1) {
                int final_w = skyline_rotated ? h : w;
                int final_h = skyline_rotated ? w : h;
                int place_x = segments[best_seg_idx].x;
                int place_y = best_Y;
                
                int orig_w = base_rects[r.id].w;
                int rotated_val = (final_w == orig_w) ? 0 : 1;
                
                placements[r.id] = {board_idx, place_x, place_y, rotated_val};
                
                // Collect gaps
                if (use_waste) {
                    for (int j = best_seg_idx; j <= best_last_idx; ++j) {
                        int seg_left = segments[j].x;
                        int seg_right = segments[j].x + segments[j].w;
                        int overlap_left = std::max(seg_left, place_x);
                        int overlap_right = std::min(seg_right, place_x + final_w);
                        int overlap_w = overlap_right - overlap_left;
                        int gap_h = best_Y - segments[j].y;
                        if (gap_h > 0 && overlap_w > 0) {
                            waste_map.push_back({overlap_left, segments[j].y, overlap_w, gap_h});
                        }
                    }
                }
                
                // Update segments
                std::vector<Segment> next_segments;
                next_segments.reserve(segments.size() + 2);
                for (int j = 0; j < best_seg_idx; ++j) {
                    next_segments.push_back(segments[j]);
                }
                if (segments[best_seg_idx].x < place_x) {
                    next_segments.push_back({segments[best_seg_idx].x, segments[best_seg_idx].y, place_x - segments[best_seg_idx].x});
                }
                next_segments.push_back({place_x, best_Y + final_h, final_w});
                
                int right_x = place_x + final_w;
                int right_end = segments[best_last_idx].x + segments[best_last_idx].w;
                if (right_end > right_x) {
                    next_segments.push_back({right_x, segments[best_last_idx].y, right_end - right_x});
                }
                for (size_t j = best_last_idx + 1; j < segments.size(); ++j) {
                    next_segments.push_back(segments[j]);
                }
                
                segments = std::move(next_segments);
                merge_segments(segments);
                if (use_waste) {
                    prune_waste_map(waste_map, MAX_WASTE_MAP_SIZE);
                }
                
                packed[idx] = true;
                packed_any = true;
            }
        }
        
        if (!packed_any) {
            return -1;
        }
        
        // Filter unpacked
        std::vector<int> next_unpacked;
        next_unpacked.reserve(unpacked.size());
        for (int idx : unpacked) {
            if (!packed[idx]) {
                next_unpacked.push_back(idx);
            }
        }
        unpacked = std::move(next_unpacked);
        
        board_idx++;
    }
    
    return board_idx;
}

bool validate_placements(const std::vector<Rectangle>& rects, int board_width, int board_height,
                         const std::vector<Placement>& placements) {
    if (placements.empty()) return false;
    std::vector<std::vector<int>> board_rects;
    for (int i = 0; i < (int)rects.size(); ++i) {
        int b = placements[i].board_idx;
        if (b >= (int)board_rects.size()) {
            board_rects.resize(b + 1);
        }
        board_rects[b].push_back(i);
    }
    
    for (int b = 0; b < (int)board_rects.size(); ++b) {
        const auto& ids = board_rects[b];
        for (size_t i = 0; i < ids.size(); ++i) {
            int id1 = ids[i];
            const auto& p1 = placements[id1];
            int w1 = p1.rotated ? rects[id1].h : rects[id1].w;
            int h1 = p1.rotated ? rects[id1].w : rects[id1].h;
            
            if (p1.x < 0 || p1.y < 0 || p1.x + w1 > board_width || p1.y + h1 > board_height) {
                return false;
            }
            
            for (size_t j = i + 1; j < ids.size(); ++j) {
                int id2 = ids[j];
                const auto& p2 = placements[id2];
                int w2 = p2.rotated ? rects[id2].h : rects[id2].w;
                int h2 = p2.rotated ? rects[id2].w : rects[id2].h;
                
                if (p1.x < p2.x + w2 && p1.x + w1 > p2.x &&
                    p1.y < p2.y + h2 && p1.y + h1 > p2.y) {
                    return false;
                }
            }
        }
    }
    return true;
}

int get_priority(const Config& conf) {
    int score = 0;
    if (conf.sort_crit == SortCriteria::AreaDesc) score += 10;
    else if (conf.sort_crit == SortCriteria::MaxDimDesc) score += 9;
    else if (conf.sort_crit == SortCriteria::PerimeterDesc) score += 8;
    else if (conf.sort_crit == SortCriteria::HeightDesc) score += 7;
    else if (conf.sort_crit == SortCriteria::WidthDesc) score += 7;
    else score += 3;
    
    if (conf.orient == Orientation::Horizontal) score += 5;
    else if (conf.orient == Orientation::Vertical) score += 5;
    else score += 2;
    
    if (conf.scoring_mode == 0) score += 4;
    else score += 2;
    
    score += 3;
    return score;
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    int N, W, H;
    if (!(std::cin >> N >> W >> H)) return 0;
    
    std::vector<Rectangle> base_rects(N);
    for (int i = 0; i < N; ++i) {
        base_rects[i].id = i;
        std::cin >> base_rects[i].w >> base_rects[i].h;
    }
    
    // Generate all configs
    std::vector<SortCriteria> criteria = {
        SortCriteria::AreaDesc,
        SortCriteria::PerimeterDesc,
        SortCriteria::MaxDimDesc,
        SortCriteria::MinDimDesc,
        SortCriteria::WidthDesc,
        SortCriteria::HeightDesc,
        SortCriteria::RatioDesc,
        SortCriteria::RatioAsc,
        SortCriteria::SquarishDesc,
        SortCriteria::SkinnyDesc
    };
    std::vector<Orientation> orients = {
        Orientation::None,
        Orientation::Horizontal,
        Orientation::Vertical
    };
    std::vector<int> scores = {0, 1, 2, 3};
    std::vector<bool> wastes = {true, false};
    
    std::vector<Config> configs;
    for (auto c : criteria) {
        for (auto o : orients) {
            for (auto s : scores) {
                for (auto w : wastes) {
                    Config conf = {c, o, s, w, 0};
                    conf.priority = get_priority(conf);
                    configs.push_back(conf);
                }
            }
        }
    }
    
    // Sort configurations by priority descending
    std::sort(configs.begin(), configs.end(), [](const Config& a, const Config& b) {
        return a.priority > b.priority;
    });
    
    // Select configurations to run based on N
    size_t num_configs_to_run = 48;
    if (N <= 1000) {
        num_configs_to_run = 120;
    } else if (N <= 5000) {
        num_configs_to_run = 96;
    } else if (N <= 20000) {
        num_configs_to_run = 64;
    } else {
        num_configs_to_run = 48;
    }
    
    if (num_configs_to_run > configs.size()) {
        num_configs_to_run = configs.size();
    }
    configs.resize(num_configs_to_run);
    
    std::mutex best_mutex;
    int best_boards = 1e9;
    std::vector<Placement> best_placements;
    
    std::vector<std::future<void>> futures;
    for (const auto& conf : configs) {
        futures.push_back(std::async(std::launch::async, [&, conf]() {
            auto rects = base_rects;
            
            // 1. Orient
            for (auto& r : rects) {
                if (conf.orient == Orientation::Horizontal) {
                    if (r.w < r.h) std::swap(r.w, r.h);
                } else if (conf.orient == Orientation::Vertical) {
                    if (r.w > r.h) std::swap(r.w, r.h);
                }
            }
            
            // 2. Sort
            std::sort(rects.begin(), rects.end(), [conf](const Rectangle& a, const Rectangle& b) {
                switch (conf.sort_crit) {
                    case SortCriteria::AreaDesc: {
                        long long area_a = (long long)a.w * a.h;
                        long long area_b = (long long)b.w * b.h;
                        if (area_a != area_b) return area_a > area_b;
                        return a.id < b.id;
                    }
                    case SortCriteria::PerimeterDesc: {
                        int p_a = a.w + a.h;
                        int p_b = b.w + b.h;
                        if (p_a != p_b) return p_a > p_b;
                        return a.id < b.id;
                    }
                    case SortCriteria::MaxDimDesc: {
                        int m_a = std::max(a.w, a.h);
                        int m_b = std::max(b.w, b.h);
                        if (m_a != m_b) return m_a > m_b;
                        return a.id < b.id;
                    }
                    case SortCriteria::MinDimDesc: {
                        int m_a = std::min(a.w, a.h);
                        int m_b = std::min(b.w, b.h);
                        if (m_a != m_b) return m_a > m_b;
                        return a.id < b.id;
                    }
                    case SortCriteria::WidthDesc: {
                        if (a.w != b.w) return a.w > b.w;
                        return a.id < b.id;
                    }
                    case SortCriteria::HeightDesc: {
                        if (a.h != b.h) return a.h > b.h;
                        return a.id < b.id;
                    }
                    case SortCriteria::RatioDesc: {
                        double r_a = (double)a.w / a.h;
                        double r_b = (double)b.w / b.h;
                        if (r_a != r_b) return r_a > r_b;
                        return a.id < b.id;
                    }
                    case SortCriteria::RatioAsc: {
                        double r_a = (double)a.w / a.h;
                        double r_b = (double)b.w / b.h;
                        if (r_a != r_b) return r_a < r_b;
                        return a.id < b.id;
                    }
                    case SortCriteria::SquarishDesc: {
                        double r_a = (double)std::max(a.w, a.h) / std::min(a.w, a.h);
                        double r_b = (double)std::max(b.w, b.h) / std::min(b.w, b.h);
                        if (r_a != r_b) return r_a < r_b; // squares first
                        return a.id < b.id;
                    }
                    case SortCriteria::SkinnyDesc: {
                        double r_a = (double)std::max(a.w, a.h) / std::min(a.w, a.h);
                        double r_b = (double)std::max(b.w, b.h) / std::min(b.w, b.h);
                        if (r_a != r_b) return r_a > r_b; // skinny first
                        return a.id < b.id;
                    }
                }
                return a.id < b.id;
            });
            
            // 3. Pack
            std::vector<Placement> placements;
            int boards = pack_skyline(base_rects, rects, W, H, conf.scoring_mode, conf.use_waste, placements);
            
            if (boards != -1) {
                std::lock_guard<std::mutex> lock(best_mutex);
                if (boards < best_boards) {
                    best_boards = boards;
                    best_placements = std::move(placements);
                }
            }
        }));
    }
    
    for (auto& f : futures) {
        f.get();
    }
    
    // Output solution with robust fallback
    if (best_boards == 1e9) {
        best_placements.resize(N);
        for (int i = 0; i < N; ++i) {
            int rotated = 0;
            if (base_rects[i].w > W || base_rects[i].h > H) {
                rotated = 1;
            }
            best_placements[i] = {i, 0, 0, rotated};
        }
        best_boards = N;
    }
    
    // Safety check
    bool valid = validate_placements(base_rects, W, H, best_placements);
    assert(valid);
    
    std::cout << best_boards << "\n";
    for (int i = 0; i < N; ++i) {
        std::cout << best_placements[i].board_idx << " "
                  << best_placements[i].x << " "
                  << best_placements[i].y << " "
                  << best_placements[i].rotated << "\n";
    }
    
    return 0;
}
