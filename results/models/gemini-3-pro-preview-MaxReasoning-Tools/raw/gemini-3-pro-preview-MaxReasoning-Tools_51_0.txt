import sys
import os
import math

# Add parent directory to path to resolve local imports
sys.path.append(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

import importlib
test51 = importlib.import_module("51")
autopilot_sim = importlib.import_module("autopilot_sim")

def main():
    # Setup Scenario 20
    runner, scenario = test51._setup_scenario(22)
    
    # Compile our autopilot
    ns = {}
    exec(open("answer.txt").read(), ns)
    autopilot_fn = ns["autopilot_step"]
    
    print(f"Step | Time (s) | Alt (ft) | VS (fpm) | Pitch (deg) | Roll (deg) | IAS (kt) | Elevator | Trim | Throttle1 | N1_1 | N1_2 | Theta_Cmd | VS_Cmd")
    print("-" * 140)
    
    dt = 0.05
    for step in range(1200):
        state = runner.sensors.build_state(runner.truth)
        state['target_alt_ft'] = runner.target_alt_ft
        state['target_hdg_deg'] = runner.target_hdg_deg
        state['target_ias_kt'] = runner.target_ias_kt
        
        controls = autopilot_fn(state, dt)
        
        # Print every 50 steps
        if step % 50 == 0 or step == 1199:
            V = math.sqrt(runner.truth['u']**2 + runner.truth['v']**2 + runner.truth['w']**2)
            ias = autopilot_sim.ias_from_tas(V, runner.truth['z']) * autopilot_sim.MS2KT
            
            hdg_int = ns.get('hdg_integral', 0.0)
            hdg_err = (runner.target_hdg_deg - runner.truth['psi']*autopilot_sim.RAD2DEG + 180) % 360 - 180
            ail = controls.get('aileron', 0.0)
            
            debug_dict = ns.get('debug_vars', {})
            t_cmd = debug_dict.get('theta_cmd', 0.0)
            v_cmd = debug_dict.get('vs_cmd', 0.0)
            r_cmd = debug_dict.get('roll_cmd', 0.0)
            n1_2 = state.get('engine_2_n1_pct', 0.0)
            
            print(f"{step:4d} | {runner.truth['time']:7.2f} | {runner.truth['z']*autopilot_sim.M2FT:8.1f} | {state.get('vertical_speed_fpm', 0):8.1f} | {runner.truth['theta']*autopilot_sim.RAD2DEG:11.2f} | {runner.truth['phi']*autopilot_sim.RAD2DEG:10.2f} | {ias:8.1f} | {ail:8.3f} | {r_cmd:8.3f} | {hdg_err:8.1f} | {hdg_int:8.1f} | {n1_2:5.1f} | {t_cmd:9.2f} | {v_cmd:6.1f}")
            
        runner.truth = autopilot_sim.physics_step(runner.truth, controls, dt)
        
        # Record history every 20 steps (~1s)
        if step % 20 == 0:
            V = math.sqrt(runner.truth['u']**2 + runner.truth['v']**2 + runner.truth['w']**2)
            runner.history.append({
                'time': runner.truth['time'],
                'alt_ft': runner.truth['z'] * autopilot_sim.M2FT,
                'ias_kt': autopilot_sim.ias_from_tas(V, runner.truth['z']) * autopilot_sim.MS2KT,
                'hdg_deg': runner.truth['psi'] * autopilot_sim.RAD2DEG,
                'pitch_deg': runner.truth['theta'] * autopilot_sim.RAD2DEG,
                'roll_deg': runner.truth['phi'] * autopilot_sim.RAD2DEG,
                'vs_fpm': state.get('vertical_speed_fpm', 0),
            })
        
        if runner.truth['z'] <= 0 and step > 10:
            print("Crashed!")
            break

    # Score scenario
    score, msg = runner.score_recovery()
    print(f"Final Score: {score:.2f} | {msg}")

if __name__ == "__main__":
    main()
