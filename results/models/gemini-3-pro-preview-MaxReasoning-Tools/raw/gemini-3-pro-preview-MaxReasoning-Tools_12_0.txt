public struct Vector3
{
    public double X, Y, Z;
    public Vector3(double x, double y, double z) { X = x; Y = y; Z = z; }
    
    public static Vector3 Cross(Vector3 a, Vector3 b)
    {
        return new Vector3(
            a.Y * b.Z - a.Z * b.Y,
            a.Z * b.X - a.X * b.Z,
            a.X * b.Y - a.Y * b.X
        );
    }
    
    public static double Dot(Vector3 a, Vector3 b)
    {
        return a.X * b.X + a.Y * b.Y + a.Z * b.Z;
    }
    
    public static Vector3 operator +(Vector3 a, Vector3 b) => new Vector3(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
    public static Vector3 operator -(Vector3 a, Vector3 b) => new Vector3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
    public static Vector3 operator *(Vector3 a, double s) => new Vector3(a.X * s, a.Y * s, a.Z * s);
    public static Vector3 operator /(Vector3 a, double s) => new Vector3(a.X / s, a.Y / s, a.Z / s);
    
    public double Length() => Math.Sqrt(X * X + Y * Y + Z * Z);
    public Vector3 Normalize()
    {
        double len = Length();
        return len > 1e-9 ? this / len : new Vector3(0, 0, 0);
    }
    public static double Distance(Vector3 a, Vector3 b) => (a - b).Length();
}

public struct Quaternion
{
    public double W, X, Y, Z;
    public Quaternion(double w, double x, double y, double z) { W = w; X = x; Y = y; Z = z; }
    
    public Quaternion Normalize()
    {
        double len = Math.Sqrt(W * W + X * X + Y * Y + Z * Z);
        if (len > 1e-9)
        {
            return new Quaternion(W / len, X / len, Y / len, Z / len);
        }
        return new Quaternion(1, 0, 0, 0);
    }
}

public struct Face
{
    public int A, B, C;
    public Face(int a, int b, int c) { A = a; B = b; C = c; }
}

public class Collider
{
    public Quaternion Rotation;
    public Vector3[] Vertices;
    public Vector3[] FaceNormals;
    public Vector3[] EdgeDirections;
    public Vector3 AABBMin;
    public Vector3 AABBMax;
}

public class PlacedItem
{
    public int Id;
    public Quaternion Rotation;
    public Vector3 Translation;
    public Collider Collider;
    public Vector3 AABBMin => Collider.AABBMin + Translation;
    public Vector3 AABBMax => Collider.AABBMax + Translation;
}

public class SpatialGrid
{
    private double bucketSizeX, bucketSizeY, bucketSizeZ;
    private int numX, numY, numZ;
    private List<PlacedItem>[,,] grid;
    
    public SpatialGrid(double W, double H, double D, double itemW, double itemH, double itemD)
    {
        bucketSizeX = Math.Max(itemW, 1.0);
        bucketSizeY = Math.Max(itemH, 1.0);
        bucketSizeZ = Math.Max(itemD, 1.0);
        
        numX = (int)Math.Ceiling(W / bucketSizeX);
        numY = (int)Math.Ceiling(H / bucketSizeY);
        numZ = (int)Math.Ceiling(D / bucketSizeZ);
        
        if (numX < 1) numX = 1; if (numX > 50) numX = 50;
        if (numY < 1) numY = 1; if (numY > 50) numY = 50;
        if (numZ < 1) numZ = 1; if (numZ > 50) numZ = 50;
        
        bucketSizeX = W / numX;
        bucketSizeY = H / numY;
        bucketSizeZ = D / numZ;
        
        grid = new List<PlacedItem>[numX, numY, numZ];
        for (int i = 0; i < numX; i++)
            for (int j = 0; j < numY; j++)
                for (int k = 0; k < numZ; k++)
                    grid[i, j, k] = new List<PlacedItem>();
    }
    
    public void Insert(PlacedItem item)
    {
        int minX = (int)Math.Floor(item.AABBMin.X / bucketSizeX);
        int maxX = (int)Math.Floor(item.AABBMax.X / bucketSizeX);
        int minY = (int)Math.Floor(item.AABBMin.Y / bucketSizeY);
        int maxY = (int)Math.Floor(item.AABBMax.Y / bucketSizeY);
        int minZ = (int)Math.Floor(item.AABBMin.Z / bucketSizeZ);
        int maxZ = (int)Math.Floor(item.AABBMax.Z / bucketSizeZ);
        
        minX = Math.Max(0, Math.Min(numX - 1, minX));
        maxX = Math.Max(0, Math.Min(numX - 1, maxX));
        minY = Math.Max(0, Math.Min(numY - 1, minY));
        maxY = Math.Max(0, Math.Min(numY - 1, maxY));
        minZ = Math.Max(0, Math.Min(numZ - 1, minZ));
        maxZ = Math.Max(0, Math.Min(numZ - 1, maxZ));
        
        for (int i = minX; i <= maxX; i++)
            for (int j = minY; j <= maxY; j++)
                for (int k = minZ; k <= maxZ; k++)
                    grid[i, j, k].Add(item);
    }
    
    public bool CheckCollision(Collider cand, Vector3 tCand, bool preciseSAT)
    {
        Vector3 cMin = cand.AABBMin + tCand;
        Vector3 cMax = cand.AABBMax + tCand;
        
        int minX = (int)Math.Floor(cMin.X / bucketSizeX);
        int maxX = (int)Math.Floor(cMax.X / bucketSizeX);
        int minY = (int)Math.Floor(cMin.Y / bucketSizeY);
        int maxY = (int)Math.Floor(cMax.Y / bucketSizeY);
        int minZ = (int)Math.Floor(cMin.Z / bucketSizeZ);
        int maxZ = (int)Math.Floor(cMax.Z / bucketSizeZ);
        
        minX = Math.Max(0, Math.Min(numX - 1, minX));
        maxX = Math.Max(0, Math.Min(numX - 1, maxX));
        minY = Math.Max(0, Math.Min(numY - 1, minY));
        maxY = Math.Max(0, Math.Min(numY - 1, maxY));
        minZ = Math.Max(0, Math.Min(numZ - 1, minZ));
        maxZ = Math.Max(0, Math.Min(numZ - 1, maxZ));
        
        HashSet<int> checkedIds = new HashSet<int>();
        
        for (int i = minX; i <= maxX; i++)
        {
            for (int j = minY; j <= maxY; j++)
            {
                for (int k = minZ; k <= maxZ; k++)
                {
                    var list = grid[i, j, k];
                    for (int idx = 0; idx < list.Count; idx++)
                    {
                        var placed = list[idx];
                        if (checkedIds.Add(placed.Id))
                        {
                            if (preciseSAT)
                            {
                                if (Solver.Overlaps(cand, tCand, placed.Collider, placed.Translation))
                                {
                                    return true;
                                }
                            }
                            else
                            {
                                // AABB-only check
                                double minAx = cand.AABBMin.X + tCand.X, maxAx = cand.AABBMax.X + tCand.X;
                                double minAy = cand.AABBMin.Y + tCand.Y, maxAy = cand.AABBMax.Y + tCand.Y;
                                double minAz = cand.AABBMin.Z + tCand.Z, maxAz = cand.AABBMax.Z + tCand.Z;

                                double minBx = placed.AABBMin.X, maxBx = placed.AABBMax.X;
                                double minBy = placed.AABBMin.Y, maxBy = placed.AABBMax.Y;
                                double minBz = placed.AABBMin.Z, maxBz = placed.AABBMax.Z;

                                if (!(maxAx < minBx - 1e-5 || minAx > maxBx + 1e-5 ||
                                      maxAy < minBy - 1e-5 || minAy > maxBy + 1e-5 ||
                                      maxAz < minBz - 1e-5 || minAz > maxBz + 1e-5))
                                {
                                    return true;
                                }
                            }
                        }
                    }
                }
            }
        }
        return false;
    }
}

public class Solver
{
    public static Vector3 Rotate(Vector3 p, Quaternion q)
    {
        double qw = q.W, qx = q.X, qy = q.Y, qz = q.Z;
        double tw = -qx * p.X - qy * p.Y - qz * p.Z;
        double tx =  qw * p.X + qy * p.Z - qz * p.Y;
        double ty =  qw * p.Y - qx * p.Z + qz * p.X;
        double tz =  qw * p.Z + qx * p.Y - qy * p.X;
        
        double rx = tw * (-qx) + tx * qw + ty * (-qz) - tz * (-qy);
        double ry = tw * (-qy) - tx * (-qz) + ty * qw + tz * (-qx);
        double rz = tw * (-qz) + tx * (-qy) - ty * (-qx) + tz * qw;
        
        return new Vector3(rx, ry, rz);
    }
    
    public static Quaternion Multiply(Quaternion q1, Quaternion q2)
    {
        double w = q1.W * q2.W - q1.X * q2.X - q1.Y * q2.Y - q1.Z * q2.Z;
        double x = q1.W * q2.X + q1.X * q2.W + q1.Y * q2.Z - q1.Z * q2.Y;
        double y = q1.W * q2.Y - q1.X * q2.Z + q1.Y * q2.W + q1.Z * q2.X;
        double z = q1.W * q2.Z + q1.X * q2.Y - q1.Y * q2.X + q1.Z * q2.W;
        return new Quaternion(w, x, y, z);
    }
    
    public static Quaternion FromAxisAngle(Vector3 axis, double angleRadians)
    {
        double halfAngle = angleRadians / 2.0;
        double sin = Math.Sin(halfAngle);
        double cos = Math.Cos(halfAngle);
        Vector3 normAxis = axis.Normalize();
        return new Quaternion(cos, normAxis.X * sin, normAxis.Y * sin, normAxis.Z * sin);
    }
    
    public static List<Vector3> GetUniqueDirections(IEnumerable<Vector3> dirs)
    {
        List<Vector3> unique = new List<Vector3>();
        foreach (var d in dirs)
        {
            Vector3 normalized = d.Normalize();
            if (double.IsNaN(normalized.X) || double.IsNaN(normalized.Y) || double.IsNaN(normalized.Z))
                continue;
            if (normalized.X < -1e-7 || 
                (Math.Abs(normalized.X) <= 1e-7 && normalized.Y < -1e-7) ||
                (Math.Abs(normalized.X) <= 1e-7 && Math.Abs(normalized.Y) <= 1e-7 && normalized.Z < -1e-7))
            {
                normalized = new Vector3(-normalized.X, -normalized.Y, -normalized.Z);
            }
            
            bool found = false;
            foreach (var u in unique)
            {
                if (Vector3.Distance(u, normalized) < 1e-5)
                {
                    found = true;
                    break;
                }
            }
            if (!found)
            {
                unique.Add(normalized);
            }
        }
        return unique;
    }
    
    public static Collider CreateCollider(Vector3[] originalVertices, Face[] faces, Quaternion rotation)
    {
        int V = originalVertices.Length;
        Vector3[] rotatedVertices = new Vector3[V];
        
        double minX = double.MaxValue, maxX = double.MinValue;
        double minY = double.MaxValue, maxY = double.MinValue;
        double minZ = double.MaxValue, maxZ = double.MinValue;
        
        for (int i = 0; i < V; i++)
        {
            Vector3 rotated = Rotate(originalVertices[i], rotation);
            rotatedVertices[i] = rotated;
            
            if (rotated.X < minX) minX = rotated.X;
            if (rotated.X > maxX) maxX = rotated.X;
            if (rotated.Y < minY) minY = rotated.Y;
            if (rotated.Y > maxY) maxY = rotated.Y;
            if (rotated.Z < minZ) minZ = rotated.Z;
            if (rotated.Z > maxZ) maxZ = rotated.Z;
        }
        
        int F = faces.Length;
        Vector3[] faceNormals = new Vector3[F];
        List<Vector3> allEdges = new List<Vector3>();
        
        for (int i = 0; i < F; i++)
        {
            Vector3 va = rotatedVertices[faces[i].A];
            Vector3 vb = rotatedVertices[faces[i].B];
            Vector3 vc = rotatedVertices[faces[i].C];
            
            Vector3 normal = Vector3.Cross(vb - va, vc - va);
            faceNormals[i] = normal.Normalize();
            
            allEdges.Add(vb - va);
            allEdges.Add(vc - vb);
            allEdges.Add(va - vc);
        }
        
        Vector3[] uniqueEdges = GetUniqueDirections(allEdges).ToArray();
        
        return new Collider
        {
            Rotation = rotation,
            Vertices = rotatedVertices,
            FaceNormals = faceNormals,
            EdgeDirections = uniqueEdges,
            AABBMin = new Vector3(minX, minY, minZ),
            AABBMax = new Vector3(maxX, maxY, maxZ)
        };
    }
    
    private static bool IsSeparatingAxis(Vector3 D, Collider a, Vector3 tA, Collider b, Vector3 tB)
    {
        double dotD = D.X * D.X + D.Y * D.Y + D.Z * D.Z;
        if (dotD < 1e-9) return false;
        
        D = D / Math.Sqrt(dotD);
        
        double minA = double.MaxValue;
        double maxA = double.MinValue;
        for (int i = 0; i < a.Vertices.Length; i++)
        {
            double p = a.Vertices[i].X * D.X + a.Vertices[i].Y * D.Y + a.Vertices[i].Z * D.Z;
            if (p < minA) minA = p;
            if (p > maxA) maxA = p;
        }
        double dotTA = tA.X * D.X + tA.Y * D.Y + tA.Z * D.Z;
        minA += dotTA;
        maxA += dotTA;
        
        double minB = double.MaxValue;
        double maxB = double.MinValue;
        for (int i = 0; i < b.Vertices.Length; i++)
        {
            double p = b.Vertices[i].X * D.X + b.Vertices[i].Y * D.Y + b.Vertices[i].Z * D.Z;
            if (p < minB) minB = p;
            if (p > maxB) maxB = p;
        }
        double dotTB = tB.X * D.X + tB.Y * D.Y + tB.Z * D.Z;
        minB += dotTB;
        maxB += dotTB;
        
        if (maxA < minB - 1e-5 || minA > maxB + 1e-5)
        {
            return true;
        }
        return false;
    }
    
    public static bool Overlaps(Collider a, Vector3 tA, Collider b, Vector3 tB)
    {
        double minAx = a.AABBMin.X + tA.X, maxAx = a.AABBMax.X + tA.X;
        double minAy = a.AABBMin.Y + tA.Y, maxAy = a.AABBMax.Y + tA.Y;
        double minAz = a.AABBMin.Z + tA.Z, maxAz = a.AABBMax.Z + tA.Z;

        double minBx = b.AABBMin.X + tB.X, maxBx = b.AABBMax.X + tB.X;
        double minBy = b.AABBMin.Y + tB.Y, maxBy = b.AABBMax.Y + tB.Y;
        double minBz = b.AABBMin.Z + tB.Z, maxBz = b.AABBMax.Z + tB.Z;

        if (maxAx < minBx - 1e-5 || minAx > maxBx + 1e-5 ||
            maxAy < minBy - 1e-5 || minAy > maxBy + 1e-5 ||
            maxAz < minBz - 1e-5 || minAz > maxBz + 1e-5)
        {
            return false;
        }

        for (int i = 0; i < a.FaceNormals.Length; i++)
        {
            if (IsSeparatingAxis(a.FaceNormals[i], a, tA, b, tB))
                return false;
        }

        for (int i = 0; i < b.FaceNormals.Length; i++)
        {
            if (IsSeparatingAxis(b.FaceNormals[i], a, tA, b, tB))
                return false;
        }

        for (int i = 0; i < a.EdgeDirections.Length; i++)
        {
            for (int j = 0; j < b.EdgeDirections.Length; j++)
            {
                Vector3 axis = Vector3.Cross(a.EdgeDirections[i], b.EdgeDirections[j]);
                if (IsSeparatingAxis(axis, a, tA, b, tB))
                    return false;
            }
        }

        return true;
    }
    
    public static List<Quaternion> GetAxisAlignedRotations()
    {
        List<Quaternion> list = new List<Quaternion>();
        double[] angles = { 0, Math.PI / 2, Math.PI, 3 * Math.PI / 2 };
        Vector3 xAxis = new Vector3(1, 0, 0);
        Vector3 yAxis = new Vector3(0, 1, 0);
        Vector3 zAxis = new Vector3(0, 0, 1);
        
        foreach (double ax in angles)
        {
            Quaternion qx = FromAxisAngle(xAxis, ax);
            foreach (double ay in angles)
            {
                Quaternion qy = FromAxisAngle(yAxis, ay);
                foreach (double az in angles)
                {
                    Quaternion qz = FromAxisAngle(zAxis, az);
                    Quaternion q = Multiply(qz, Multiply(qy, qx)).Normalize();
                    
                    bool found = false;
                    foreach (var existing in list)
                    {
                        double dot = q.W * existing.W + q.X * existing.X + q.Y * existing.Y + q.Z * existing.Z;
                        if (Math.Abs(dot) > 0.999)
                        {
                            found = true;
                            break;
                        }
                    }
                    if (!found)
                    {
                        list.Add(q);
                    }
                }
            }
        }
        return list;
    }
    
    public static List<Quaternion> GetUniformRotations(int count, int seed = 42)
    {
        List<Quaternion> list = new List<Quaternion>();
        Random rand = new Random(seed);
        for (int i = 0; i < count; i++)
        {
            double u1 = rand.NextDouble();
            double u2 = rand.NextDouble();
            double u3 = rand.NextDouble();
            
            double w = Math.Sqrt(1 - u1) * Math.Sin(2 * Math.PI * u2);
            double x = Math.Sqrt(1 - u1) * Math.Cos(2 * Math.PI * u2);
            double y = Math.Sqrt(u1) * Math.Sin(2 * Math.PI * u3);
            double z = Math.Sqrt(u1) * Math.Cos(2 * Math.PI * u3);
            
            list.Add(new Quaternion(w, x, y, z).Normalize());
        }
        return list;
    }
    
    public struct GridResult
    {
        public int Count;
        public Quaternion Rotation;
        public Vector3 TranslationOffset;
        public double Wb, Hb, Db;
    }
    
    public static GridResult SolveSingleGrid(double W, double H, double D, List<Collider> colliders)
    {
        GridResult best = new GridResult { Count = 0 };
        foreach (var c in colliders)
        {
            double wb = c.AABBMax.X - c.AABBMin.X;
            double hb = c.AABBMax.Y - c.AABBMin.Y;
            double db = c.AABBMax.Z - c.AABBMin.Z;
            
            double eps = 1e-4;
            int nx = (int)((W + eps) / (wb + eps));
            int ny = (int)((H + eps) / (hb + eps));
            int nz = (int)((D + eps) / (db + eps));
            int count = nx * ny * nz;
            
            if (count > best.Count)
            {
                best.Count = count;
                best.Rotation = c.Rotation;
                best.TranslationOffset = c.AABBMin * -1.0;
                best.Wb = wb;
                best.Hb = hb;
                best.Db = db;
            }
        }
        return best;
    }
    
    public struct DimerResult
    {
        public int Count;
        public Quaternion Q1, Q2;
        public Vector3 T1, T2;
        public double Wc, Hc, Dc;
    }
    
    public static DimerResult SearchBestDimer(
        double W, double H, double D,
        List<Collider> singleBestColliders,
        List<Collider> candRotColliders,
        int k = 6)
    {
        DimerResult bestDimer = new DimerResult { Count = 0 };
        
        foreach (var A in singleBestColliders)
        {
            Vector3 t1 = A.AABBMin * -1.0;
            double wA = A.AABBMax.X - A.AABBMin.X;
            double hA = A.AABBMax.Y - A.AABBMin.Y;
            double dA = A.AABBMax.Z - A.AABBMin.Z;
            
            foreach (var B in candRotColliders)
            {
                Vector3 tBase2 = B.AABBMin * -1.0;
                double wB = B.AABBMax.X - B.AABBMin.X;
                double hB = B.AABBMax.Y - B.AABBMin.Y;
                double dB = B.AABBMax.Z - B.AABBMin.Z;
                
                List<Vector3> candidateTranslations = new List<Vector3>();
                for (int i = 0; i <= k; i++)
                {
                    double tx = -wB + i * (wA + wB) / k;
                    for (int j = 0; j <= k; j++)
                    {
                        double ty = -hB + j * (hA + hB) / k;
                        for (int m = 0; m <= k; m++)
                        {
                            double tz = -dB + m * (dA + dB) / k;
                            candidateTranslations.Add(new Vector3(tx, ty, tz));
                        }
                    }
                }
                candidateTranslations.Add(new Vector3(wA, 0, 0));
                candidateTranslations.Add(new Vector3(-wB, 0, 0));
                candidateTranslations.Add(new Vector3(0, hA, 0));
                candidateTranslations.Add(new Vector3(0, -hB, 0));
                candidateTranslations.Add(new Vector3(0, 0, dA));
                candidateTranslations.Add(new Vector3(0, 0, -dB));
                
                foreach (var txOffset in candidateTranslations)
                {
                    Vector3 t2 = tBase2 + txOffset;
                    
                    if (Overlaps(A, t1, B, t2))
                    {
                        continue;
                    }
                    
                    double minX = Math.Min(0, txOffset.X);
                    double maxX = Math.Max(wA, txOffset.X + wB);
                    double minY = Math.Min(0, txOffset.Y);
                    double maxY = Math.Max(hA, txOffset.Y + hB);
                    double minZ = Math.Min(0, txOffset.Z);
                    double maxZ = Math.Max(dA, txOffset.Z + dB);
                    
                    double w_c = maxX - minX;
                    double h_c = maxY - minY;
                    double d_c = maxZ - minZ;
                    
                    double eps = 1e-4;
                    int nx = (int)((W + eps) / (w_c + eps));
                    int ny = (int)((H + eps) / (h_c + eps));
                    int nz = (int)((D + eps) / (d_c + eps));
                    int count = 2 * nx * ny * nz;
                    
                    if (count > bestDimer.Count)
                    {
                        Vector3 shift = new Vector3(minX, minY, minZ);
                        bestDimer.Count = count;
                        bestDimer.Q1 = A.Rotation;
                        bestDimer.Q2 = B.Rotation;
                        bestDimer.T1 = t1 - shift;
                        bestDimer.T2 = t2 - shift;
                        bestDimer.Wc = w_c;
                        bestDimer.Hc = h_c;
                        bestDimer.Dc = d_c;
                    }
                }
            }
        }
        
        return bestDimer;
    }
    
    static bool IsInsideAnyAABB(Vector3 ep, List<PlacedItem> placed)
    {
        for (int i = 0; i < placed.Count; i++)
        {
            var item = placed[i];
            Vector3 min = item.AABBMin;
            Vector3 max = item.AABBMax;
            if (ep.X > min.X + 1e-5 && ep.X < max.X - 1e-5 &&
                ep.Y > min.Y + 1e-5 && ep.Y < max.Y - 1e-5 &&
                ep.Z > min.Z + 1e-5 && ep.Z < max.Z - 1e-5)
            {
                return true;
            }
        }
        return false;
    }
    
    public static List<PlacedItem> RunBLF(
        double W, double H, double D,
        List<Collider> rotColliders,
        int maxItems,
        double limitSeconds,
        bool preciseSAT)
    {
        List<PlacedItem> placed = new List<PlacedItem>();
        List<Vector3> extremePoints = new List<Vector3> { new Vector3(0, 0, 0) };
        
        double itemW = rotColliders[0].AABBMax.X - rotColliders[0].AABBMin.X;
        double itemH = rotColliders[0].AABBMax.Y - rotColliders[0].AABBMin.Y;
        double itemD = rotColliders[0].AABBMax.Z - rotColliders[0].AABBMin.Z;
        SpatialGrid grid = new SpatialGrid(W, H, D, itemW, itemH, itemD);
        
        var stopwatch = Stopwatch.StartNew();
        
        int nextId = 0;
        while (placed.Count < maxItems)
        {
            if (stopwatch.Elapsed.TotalSeconds > limitSeconds)
            {
                break;
            }
            
            extremePoints.Sort((a, b) => {
                int cmp = a.Z.CompareTo(b.Z);
                if (cmp != 0) return cmp;
                cmp = a.Y.CompareTo(b.Y);
                if (cmp != 0) return cmp;
                return a.X.CompareTo(b.X);
            });
            
            bool foundPlacement = false;
            
            for (int epIdx = 0; epIdx < extremePoints.Count; epIdx++)
            {
                Vector3 ep = extremePoints[epIdx];
                
                for (int rotIdx = 0; rotIdx < rotColliders.Count; rotIdx++)
                {
                    Collider cand = rotColliders[rotIdx];
                    Vector3 t = ep - cand.AABBMin;
                    
                    Vector3 cMin = cand.AABBMin + t;
                    Vector3 cMax = cand.AABBMax + t;
                    if (cMin.X < 1e-5 || cMax.X > W - 1e-5 ||
                        cMin.Y < 1e-5 || cMax.Y > H - 1e-5 ||
                        cMin.Z < 1e-5 || cMax.Z > D - 1e-5)
                    {
                        continue;
                    }
                    
                    if (grid.CheckCollision(cand, t, preciseSAT))
                    {
                        continue;
                    }
                    
                    PlacedItem newItem = new PlacedItem
                    {
                        Id = nextId++,
                        Rotation = cand.Rotation,
                        Translation = t,
                        Collider = cand
                    };
                    
                    placed.Add(newItem);
                    grid.Insert(newItem);
                    
                    extremePoints.RemoveAt(epIdx);
                    
                    Vector3 min = newItem.AABBMin;
                    Vector3 max = newItem.AABBMax;
                    extremePoints.Add(new Vector3(max.X, min.Y, min.Z));
                    extremePoints.Add(new Vector3(min.X, max.Y, min.Z));
                    extremePoints.Add(new Vector3(min.X, min.Y, max.Z));
                    
                    List<Vector3> cleaned = new List<Vector3>();
                    foreach (var p in extremePoints)
                    {
                        if (p.X < 0 || p.X > W - 1e-5 ||
                            p.Y < 0 || p.Y > H - 1e-5 ||
                            p.Z < 0 || p.Z > D - 1e-5)
                        {
                            continue;
                        }
                        
                        if (!IsInsideAnyAABB(p, placed))
                        {
                            bool isDuplicate = false;
                            foreach (var u in cleaned)
                            {
                                if (Vector3.Distance(u, p) < 1e-5)
                                {
                                    isDuplicate = true;
                                    break;
                                }
                            }
                            if (!isDuplicate)
                            {
                                cleaned.Add(p);
                            }
                        }
                    }
                    extremePoints = cleaned;
                    
                    foundPlacement = true;
                    break;
                }
                
                if (foundPlacement)
                {
                    break;
                }
            }
            
            if (!foundPlacement)
            {
                break;
            }
        }
        
        return placed;
    }
    
    public static void Main(string[] args)
    {
        Thread.CurrentThread.CurrentCulture = CultureInfo.InvariantCulture;
        
        List<string> tokens = new List<string>();
        string line;
        while ((line = Console.ReadLine()) != null)
        {
            string[] parts = line.Split((char[])null, StringSplitOptions.RemoveEmptyEntries);
            foreach (var p in parts)
            {
                tokens.Add(p);
            }
        }
        
        if (tokens.Count == 0) return;
        
        int tokenIdx = 0;
        int V = int.Parse(tokens[tokenIdx++]);
        int F = int.Parse(tokens[tokenIdx++]);
        
        Vector3[] vertices = new Vector3[V];
        for (int i = 0; i < V; i++)
        {
            double x = double.Parse(tokens[tokenIdx++]);
            double y = double.Parse(tokens[tokenIdx++]);
            double z = double.Parse(tokens[tokenIdx++]);
            vertices[i] = new Vector3(x, y, z);
        }
        
        Face[] faces = new Face[F];
        for (int i = 0; i < F; i++)
        {
            int a = int.Parse(tokens[tokenIdx++]);
            int b = int.Parse(tokens[tokenIdx++]);
            int c = int.Parse(tokens[tokenIdx++]);
            faces[i] = new Face(a, b, c);
        }
        
        double W = double.Parse(tokens[tokenIdx++]);
        double H = double.Parse(tokens[tokenIdx++]);
        double D = double.Parse(tokens[tokenIdx++]);
        
        // 1. Generate candidate rotations
        List<Quaternion> axisAligned = GetAxisAlignedRotations();
        List<Quaternion> uniform = GetUniformRotations(300);
        
        List<Quaternion> allRotations = new List<Quaternion>();
        allRotations.AddRange(axisAligned);
        allRotations.AddRange(uniform);
        
        // Precompute colliders
        List<Collider> colliders = new List<Collider>();
        foreach (var q in allRotations)
        {
            colliders.Add(CreateCollider(vertices, faces, q));
        }
        
        // 2. Solve Single Grid
        GridResult singleGrid = SolveSingleGrid(W, H, D, colliders);
        
        // Find top single-grid colliders
        List<Collider> topColliders = new List<Collider>();
        var sortedColliders = colliders
            .Select(c => {
                double wb = c.AABBMax.X - c.AABBMin.X;
                double hb = c.AABBMax.Y - c.AABBMin.Y;
                double db = c.AABBMax.Z - c.AABBMin.Z;
                double eps = 1e-4;
                int count = (int)((W + eps) / (wb + eps)) * (int)((H + eps) / (hb + eps)) * (int)((D + eps) / (db + eps));
                return new { Collider = c, Count = count };
            })
            .OrderByDescending(x => x.Count)
            .ToList();
            
        for (int i = 0; i < Math.Min(3, sortedColliders.Count); i++)
        {
            topColliders.Add(sortedColliders[i].Collider);
        }
        
        // 3. Solve Dimer Grid
        List<Collider> axisAlignedColliders = new List<Collider>();
        foreach (var q in axisAligned)
        {
            axisAlignedColliders.Add(CreateCollider(vertices, faces, q));
        }
        
        List<Collider> dimerCandidates = new List<Collider>();
        dimerCandidates.AddRange(axisAlignedColliders);
        for (int i = 0; i < Math.Min(5, sortedColliders.Count); i++)
        {
            var tc = sortedColliders[i].Collider;
            if (!dimerCandidates.Any(c => Math.Abs(c.Rotation.W - tc.Rotation.W) < 1e-5 &&
                                           Math.Abs(c.Rotation.X - tc.Rotation.X) < 1e-5 &&
                                           Math.Abs(c.Rotation.Y - tc.Rotation.Y) < 1e-5 &&
                                           Math.Abs(c.Rotation.Z - tc.Rotation.Z) < 1e-5))
            {
                dimerCandidates.Add(tc);
            }
        }
        DimerResult dimerGrid = SearchBestDimer(W, H, D, topColliders, dimerCandidates, k: 6);
        
        // Determine if we should run BLF
        int bestCount = Math.Max(singleGrid.Count, dimerGrid.Count);
        
        List<PlacedItem> blfResult = null;
        
        // Only run BLF if expected grid packing is not too massive
        if (bestCount <= 2500)
        {
            // Run BLF
            bool preciseSAT = (F <= 30);
            
            // For BLF, use axis-aligned rotations, but prepend the best single-grid rotation
            List<Collider> blfColliders = new List<Collider>();
            if (sortedColliders.Count > 0)
            {
                blfColliders.Add(sortedColliders[0].Collider);
            }
            foreach (var c in axisAlignedColliders)
            {
                // Avoid duplicating the best one
                if (sortedColliders.Count > 0 && Math.Abs(c.Rotation.W - sortedColliders[0].Collider.Rotation.W) < 1e-5 &&
                    Math.Abs(c.Rotation.X - sortedColliders[0].Collider.Rotation.X) < 1e-5 &&
                    Math.Abs(c.Rotation.Y - sortedColliders[0].Collider.Rotation.Y) < 1e-5 &&
                    Math.Abs(c.Rotation.Z - sortedColliders[0].Collider.Rotation.Z) < 1e-5)
                {
                    continue;
                }
                blfColliders.Add(c);
            }
            
            blfResult = RunBLF(W, H, D, blfColliders, maxItems: 3000, limitSeconds: 8.0, preciseSAT: preciseSAT);
        }
        
        // Compare and output the best result
        int blfCount = blfResult != null ? blfResult.Count : 0;
        
        using (var stdout = new StreamWriter(Console.OpenStandardOutput(), System.Text.Encoding.ASCII, 65536))
        {
            if (blfCount > bestCount)
            {
                // Output BLF
                stdout.WriteLine(blfResult.Count);
                foreach (var item in blfResult)
                {
                    stdout.WriteLine($"{item.Translation.X:F8} {item.Translation.Y:F8} {item.Translation.Z:F8} {item.Rotation.W:F8} {item.Rotation.X:F8} {item.Rotation.Y:F8} {item.Rotation.Z:F8}");
                }
            }
            else if (dimerGrid.Count > singleGrid.Count)
            {
                // Output Dimer Grid
                double eps = 1e-4;
                int nx = (int)((W + eps) / (dimerGrid.Wc + eps));
                int ny = (int)((H + eps) / (dimerGrid.Hc + eps));
                int nz = (int)((D + eps) / (dimerGrid.Dc + eps));
                
                stdout.WriteLine(dimerGrid.Count);
                for (int i = 0; i < nx; i++)
                {
                    for (int j = 0; j < ny; j++)
                    {
                        for (int k = 0; k < nz; k++)
                        {
                            double tx1 = i * (dimerGrid.Wc + eps) + dimerGrid.T1.X;
                            double ty1 = j * (dimerGrid.Hc + eps) + dimerGrid.T1.Y;
                            double tz1 = k * (dimerGrid.Dc + eps) + dimerGrid.T1.Z;
                            stdout.WriteLine($"{tx1:F8} {ty1:F8} {tz1:F8} {dimerGrid.Q1.W:F8} {dimerGrid.Q1.X:F8} {dimerGrid.Q1.Y:F8} {dimerGrid.Q1.Z:F8}");
                            
                            double tx2 = i * (dimerGrid.Wc + eps) + dimerGrid.T2.X;
                            double ty2 = j * (dimerGrid.Hc + eps) + dimerGrid.T2.Y;
                            double tz2 = k * (dimerGrid.Dc + eps) + dimerGrid.T2.Z;
                            stdout.WriteLine($"{tx2:F8} {ty2:F8} {tz2:F8} {dimerGrid.Q2.W:F8} {dimerGrid.Q2.X:F8} {dimerGrid.Q2.Y:F8} {dimerGrid.Q2.Z:F8}");
                        }
                    }
                }
            }
            else
            {
                // Output Single Grid
                double eps = 1e-4;
                int nx = (int)((W + eps) / (singleGrid.Wb + eps));
                int ny = (int)((H + eps) / (singleGrid.Hb + eps));
                int nz = (int)((D + eps) / (singleGrid.Db + eps));
                
                stdout.WriteLine(singleGrid.Count);
                for (int i = 0; i < nx; i++)
                {
                    for (int j = 0; j < ny; j++)
                    {
                        for (int k = 0; k < nz; k++)
                        {
                            double tx = i * (singleGrid.Wb + eps) + singleGrid.TranslationOffset.X;
                            double ty = j * (singleGrid.Hb + eps) + singleGrid.TranslationOffset.Y;
                            double tz = k * (singleGrid.Db + eps) + singleGrid.TranslationOffset.Z;
                            stdout.WriteLine($"{tx:F8} {ty:F8} {tz:F8} {singleGrid.Rotation.W:F8} {singleGrid.Rotation.X:F8} {singleGrid.Rotation.Y:F8} {singleGrid.Rotation.Z:F8}");
                        }
                    }
                }
            }
        }
    }
