#version 450

layout(location = 0) in vec3 worldPos;
layout(location = 1) in vec3 normal;
layout(location = 2) in vec2 uv;
layout(location = 3) in vec3 tangent;
layout(location = 4) in vec3 color;

layout(set = 0, binding = 0) uniform UniformBufferObject {
    mat4 model;
    mat4 view;
    mat4 projection;
    vec4 lightPos;
    vec4 cameraPos;
    vec4 params;
} ubo;

layout(location = 0) out vec4 outColor;

void main() {
    vec3 N = normalize(normal);
    vec3 L = normalize(ubo.lightPos.xyz - worldPos);
    vec3 V = normalize(ubo.cameraPos.xyz - worldPos);
    
    // R = reflect(-L, N)
    vec3 R = reflect(-L, N);
    
    // Ambient
    vec3 ambient = vec3(0.1, 0.05, 0.05);
    
    // Diffuse
    float dotNL = dot(N, L);
    vec3 diffuse = color * max(dotNL, 0.0);
    
    // Specular
    float dotRV = dot(R, V);
    vec3 specular = vec3(1.0, 1.0, 1.0) * pow(max(dotRV, 0.0), 32.0);
    
    // Clamp the sum to [0,1], output alpha 1.0
    vec3 rgb = clamp(ambient + diffuse + specular, 0.0, 1.0);
    outColor = vec4(rgb, 1.0);
}
