use std::collections::HashMap;
use std::io::{self, Read};
use std::sync::{Arc, Mutex};
use std::sync::atomic::{AtomicBool, Ordering};
use std::thread;

// Simple LCG PRNG for shuffling and randomized search
struct Lcg {
    state: u64,
}

impl Lcg {
    fn new(seed: u64) -> Self {
        Self { state: seed }
    }

    fn next_u64(&mut self) -> u64 {
        let mut x = self.state;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.state = x;
        x
    }

    fn next_usize(&mut self, limit: usize) -> usize {
        if limit == 0 {
            return 0;
        }
        (self.next_u64() as usize) % limit
    }

    fn shuffle<T>(&mut self, slice: &mut [T]) {
        let n = slice.len();
        for i in (1..n).rev() {
            let j = self.next_usize(i + 1);
            slice.swap(i, j);
        }
    }
}

#[derive(Clone, Debug)]
struct DecomposedItem {
    value: u128,
    is_positive: bool,
    indices: Vec<usize>,
}

#[derive(Clone, Copy)]
struct SumMask {
    sum: u128,
    mask: u64,
}

// Parallel Meet-in-the-Middle solver
fn search_right_rec(
    left_sums: &[SumMask],
    right_items: &[DecomposedItem],
    idx: usize,
    current_sum: u128,
    current_mask: u64,
    target: u128,
    found: &AtomicBool,
    result: &Mutex<Option<(u64, u64)>>,
    K: usize,
) {
    if found.load(Ordering::Relaxed) {
        return;
    }
    if idx == right_items.len() {
        if current_sum <= target {
            let target_L = target - current_sum;
            if let Ok(pos) = left_sums.binary_search_by_key(&target_L, |x| x.sum) {
                let mut res = result.lock().unwrap();
                if res.is_none() {
                    *res = Some((left_sums[pos].mask, current_mask));
                    found.store(true, Ordering::Relaxed);
                }
            }
        }
        return;
    }
    search_right_rec(
        left_sums,
        right_items,
        idx + 1,
        current_sum,
        current_mask,
        target,
        found,
        result,
        K,
    );
    search_right_rec(
        left_sums,
        right_items,
        idx + 1,
        current_sum + right_items[idx].value,
        current_mask | (1 << (idx - K)),
        target,
        found,
        result,
        K,
    );
}

fn parallel_meet_in_the_middle(items: &[DecomposedItem], target: u128) -> Option<Vec<usize>> {
    let M = items.len();
    if M == 0 {
        if target == 0 {
            return Some(Vec::new());
        }
        return None;
    }

    // Split into left and right halves
    let L = M / 2;
    let R = M - L;

    let left_items = &items[0..L];
    let right_items = &items[L..M];

    // Generate left sums
    let mut left_sums = vec![SumMask { sum: 0, mask: 0 }];
    for i in 0..L {
        let val = left_items[i].value;
        let len = left_sums.len();
        for j in 0..len {
            left_sums.push(SumMask {
                sum: left_sums[j].sum + val,
                mask: left_sums[j].mask | (1 << i),
            });
        }
    }
    left_sums.sort_unstable_by_key(|x| x.sum);

    let found = Arc::new(AtomicBool::new(false));
    let result = Arc::new(Mutex::new(None));
    let left_sums = Arc::new(left_sums);

    // Branch on the first K elements of the right side to run in parallel
    let K = std::cmp::min(R, 6);
    let num_threads = 1 << K;
    let mut handles = Vec::new();

    for thread_id in 0..num_threads {
        let left_sums = Arc::clone(&left_sums);
        let found = Arc::clone(&found);
        let result = Arc::clone(&result);

        let mut base_sum = 0;
        let mut base_mask = 0u64;
        for i in 0..K {
            if (thread_id & (1 << i)) != 0 {
                base_sum += right_items[i].value;
                base_mask |= 1 << i;
            }
        }

        let right_items_clone = right_items.to_vec();

        let handle = thread::spawn(move || {
            search_right_rec(
                &left_sums,
                &right_items_clone,
                K,
                base_sum,
                base_mask,
                target,
                &found,
                &result,
                K,
            );
        });
        handles.push(handle);
    }

    for handle in handles {
        let _ = handle.join();
    }

    let res = result.lock().unwrap().clone();
    if let Some((left_mask, right_mask)) = res {
        let mut chosen = vec![false; M];
        for i in 0..L {
            if (left_mask & (1 << i)) != 0 {
                chosen[i] = true;
            }
        }
        for i in 0..R {
            if (right_mask & (1 << i)) != 0 {
                chosen[L + i] = true;
            }
        }

        // Reconstruct indices based on positivity
        let mut final_indices = Vec::new();
        for i in 0..M {
            let item = &items[i];
            if chosen[i] {
                if item.is_positive {
                    final_indices.extend(&item.indices);
                }
            } else {
                if !item.is_positive {
                    final_indices.extend(&item.indices);
                }
            }
        }
        return Some(final_indices);
    }

    None
}

// Bitset Dynamic Programming solver
fn bitset_dp(items: &[DecomposedItem], target: u128) -> Option<Vec<usize>> {
    let target_usize = target as usize;
    let n = (target_usize + 63) / 64;
    let mut words = vec![0u64; n];
    words[0] = 1; // 0 is reachable

    // first_reached[s] stores the index of the item that first reached sum s.
    // Initialize with u32::MAX
    let mut first_reached = vec![u32::MAX; target_usize + 1];

    let M = items.len();
    for idx in 0..M {
        let v = items[idx].value;
        if v > target {
            continue;
        }
        let v_usize = v as usize;
        let w = v_usize / 64;
        let b = v_usize % 64;

        if b == 0 {
            for j in (w..n).rev() {
                let new_bits = words[j - w] & !words[j];
                if new_bits != 0 {
                    let mut temp = new_bits;
                    while temp > 0 {
                        let bit_idx = temp.trailing_zeros() as usize;
                        let sum = j * 64 + bit_idx;
                        if sum <= target_usize && first_reached[sum] == u32::MAX {
                            first_reached[sum] = idx as u32;
                        }
                        temp &= temp - 1;
                    }
                    words[j] |= new_bits;
                }
            }
        } else {
            for j in ((w + 1)..n).rev() {
                let shifted = (words[j - w] << b) | (words[j - w - 1] >> (64 - b));
                let new_bits = shifted & !words[j];
                if new_bits != 0 {
                    let mut temp = new_bits;
                    while temp > 0 {
                        let bit_idx = temp.trailing_zeros() as usize;
                        let sum = j * 64 + bit_idx;
                        if sum <= target_usize && first_reached[sum] == u32::MAX {
                            first_reached[sum] = idx as u32;
                        }
                        temp &= temp - 1;
                    }
                    words[j] |= new_bits;
                }
            }
            if w < n {
                let shifted = words[0] << b;
                let new_bits = shifted & !words[w];
                if new_bits != 0 {
                    let mut temp = new_bits;
                    while temp > 0 {
                        let bit_idx = temp.trailing_zeros() as usize;
                        let sum = w * 64 + bit_idx;
                        if sum <= target_usize && first_reached[sum] == u32::MAX {
                            first_reached[sum] = idx as u32;
                        }
                        temp &= temp - 1;
                    }
                    words[w] |= new_bits;
                }
            }
        }

        // Early exit check
        if first_reached[target_usize] != u32::MAX {
            break;
        }
    }

    if first_reached[target_usize] != u32::MAX {
        let mut chosen = vec![false; M];
        let mut curr = target_usize;
        while curr > 0 {
            let idx = first_reached[curr];
            if idx == u32::MAX {
                break;
            }
            chosen[idx as usize] = true;
            curr -= items[idx as usize].value as usize;
        }

        // Reconstruct original indices based on positivity
        let mut final_indices = Vec::new();
        for i in 0..M {
            let item = &items[i];
            if chosen[i] {
                if item.is_positive {
                    final_indices.extend(&item.indices);
                }
            } else {
                if !item.is_positive {
                    final_indices.extend(&item.indices);
                }
            }
        }
        return Some(final_indices);
    }

    None
}

// Randomized/Greedy Local Search solver
fn heuristic_search(items: &[DecomposedItem], target: u128, time_limit_ms: u64) -> Option<Vec<usize>> {
    let start_time = std::time::Instant::now();
    let mut rng = Lcg::new(987654321);

    let n = items.len();
    if n == 0 {
        if target == 0 {
            return Some(Vec::new());
        }
        return None;
    }

    // Try shuffle + greedy first
    let mut indices: Vec<usize> = (0..n).collect();
    for _ in 0..2000 {
        if (start_time.elapsed().as_millis() as u64) > time_limit_ms {
            break;
        }
        rng.shuffle(&mut indices);
        let mut current_sum = 0;
        let mut selected = vec![false; n];
        for &idx in &indices {
            let val = items[idx].value;
            if current_sum + val <= target {
                current_sum += val;
                selected[idx] = true;
                if current_sum == target {
                    let mut res = Vec::new();
                    for i in 0..n {
                        if selected[i] {
                            if items[i].is_positive {
                                res.extend(&items[i].indices);
                            }
                        } else {
                            if !items[i].is_positive {
                                res.extend(&items[i].indices);
                            }
                        }
                    }
                    return Some(res);
                }
            }
        }
    }

    // Try Hill Climbing / Local Search
    let mut current_sum = 0;
    let mut selected = vec![false; n];
    let mut idxs: Vec<usize> = (0..n).collect();
    idxs.sort_by(|&a, &b| items[b].value.cmp(&items[a].value));

    for &idx in &idxs {
        let val = items[idx].value;
        if current_sum + val <= target {
            current_sum += val;
            selected[idx] = true;
        }
    }

    if current_sum == target {
        let mut res = Vec::new();
        for i in 0..n {
            if selected[i] {
                if items[i].is_positive {
                    res.extend(&items[i].indices);
                }
            } else {
                if !items[i].is_positive {
                    res.extend(&items[i].indices);
                }
            }
        }
        return Some(res);
    }

    let mut selected_indices = Vec::new();
    let mut unselected_indices = Vec::new();
    for i in 0..n {
        if selected[i] {
            selected_indices.push(i);
        } else {
            unselected_indices.push(i);
        }
    }

    while (start_time.elapsed().as_millis() as u64) < time_limit_ms {
        let r = rng.next_u64() % 100;
        if r < 35 && !unselected_indices.is_empty() {
            let u_idx = rng.next_usize(unselected_indices.len());
            let item_idx = unselected_indices[u_idx];
            let val = items[item_idx].value;
            if current_sum + val <= target {
                current_sum += val;
                selected[item_idx] = true;
                selected_indices.push(item_idx);
                unselected_indices.swap_remove(u_idx);

                if current_sum == target {
                    let mut res = Vec::new();
                    for i in 0..n {
                        if selected[i] {
                            if items[i].is_positive {
                                res.extend(&items[i].indices);
                            }
                        } else {
                            if !items[i].is_positive {
                                res.extend(&items[i].indices);
                            }
                        }
                    }
                    return Some(res);
                }
            }
        } else if r < 70 && !selected_indices.is_empty() {
            let s_idx = rng.next_usize(selected_indices.len());
            let item_idx = selected_indices[s_idx];
            let val = items[item_idx].value;
            current_sum -= val;
            selected[item_idx] = false;
            unselected_indices.push(item_idx);
            selected_indices.swap_remove(s_idx);
        } else if !selected_indices.is_empty() && !unselected_indices.is_empty() {
            let s_idx = rng.next_usize(selected_indices.len());
            let u_idx = rng.next_usize(unselected_indices.len());
            let s_item = selected_indices[s_idx];
            let u_item = unselected_indices[u_idx];

            let val_s = items[s_item].value;
            let val_u = items[u_item].value;

            let new_sum = current_sum - val_s + val_u;
            if new_sum <= target {
                current_sum = new_sum;
                selected[s_item] = false;
                selected[u_item] = true;

                selected_indices[s_idx] = u_item;
                unselected_indices[u_idx] = s_item;

                if current_sum == target {
                    let mut res = Vec::new();
                    for i in 0..n {
                        if selected[i] {
                            if items[i].is_positive {
                                res.extend(&items[i].indices);
                            }
                        } else {
                            if !items[i].is_positive {
                                res.extend(&items[i].indices);
                            }
                        }
                    }
                    return Some(res);
                }
            }
        }
    }

    None
}

// Fast decimal integer parser
fn parse_integers(input: &[u8]) -> Vec<i64> {
    let mut result = Vec::new();
    let mut current = 0i64;
    let mut sign = 1i64;
    let mut in_number = false;
    for &b in input {
        match b {
            b'-' => {
                sign = -1;
                current = 0;
                in_number = true;
            }
            b'0'..=b'9' => {
                if !in_number {
                    current = 0;
                    in_number = true;
                }
                current = current * 10 + (b - b'0') as i64;
            }
            _ => {
                if in_number {
                    result.push(current * sign);
                    sign = 1;
                    in_number = false;
                }
            }
        }
    }
    if in_number {
        result.push(current * sign);
    }
    result
}

fn main() {
    // Read all input from stdin
    let mut input_bytes = Vec::new();
    if io::stdin().read_to_end(&mut input_bytes).is_err() {
        println!("NO");
        return;
    }

    let parsed = parse_integers(&input_bytes);
    if parsed.len() < 2 {
        println!("NO");
        return;
    }

    let num_elements = parsed[0] as usize;
    let target = parsed[1] as i128;

    let elements = &parsed[2..];

    // Filter out 0s first, but keep track of indices
    let mut original_elements = Vec::new();
    let mut zero_indices = Vec::new();
    for (i, &val) in elements.iter().enumerate() {
        if val == 0 {
            zero_indices.push(i);
        } else {
            original_elements.push((val, i));
        }
    }

    // Trivial check: target = 0
    if target == 0 {
        println!("YES");
        println!("[]");
        return;
    }

    // Separate positive and negative numbers
    let mut pos_nums = Vec::new();
    let mut neg_nums = Vec::new();
    for &(val, idx) in &original_elements {
        if val > 0 {
            pos_nums.push((val as u128, idx));
        } else {
            neg_nums.push(((-val) as u128, idx));
        }
    }

    let sum_neg: u128 = neg_nums.iter().map(|&(val, _)| val).sum();
    let new_target_i128 = target + sum_neg as i128;
    if new_target_i128 < 0 {
        println!("NO");
        return;
    }
    let new_target = new_target_i128 as u128;

    // Filter out elements > new_target (since all are positive, they can't contribute)
    // Wait, originally negative elements are represented as positive value = -val.
    // If -val > new_target:
    // Can we filter them out?
    // Let's check: in the transformed set B, any element > new_target cannot be chosen.
    // If an originally negative element is not chosen, it means it IS included in the final subset.
    // If it is chosen, it is NOT included in the final subset.
    // So yes, we can filter them out!
    // But wait: if we filter out an element, it means it is NEVER chosen in B.
    // So if it was originally positive and filtered out, it is never chosen (OUT).
    // If it was originally negative and filtered out, it is never chosen (IN).
    // This is correct! So we can filter out any element from the transformed set if its value > new_target!
    // Let's define the transformed positive elements B:
    let mut B = Vec::new();
    for &(val, idx) in &pos_nums {
        if val <= new_target {
            B.push((val, idx, true));
        }
    }
    for &(val, idx) in &neg_nums {
        if val <= new_target {
            B.push((val, idx, false));
        } else {
            // Originally negative element with val > new_target.
            // Since it is > new_target, it can never be chosen in B.
            // So it must be NOT chosen in B, which means it IS included in the final subset.
            // But wait! If it's included, it adds a negative number to the sum.
            // Since it is NOT chosen, the minimum sum we can get from B is 0,
            // and the final subset sum would include this negative number.
            // Wait, is this handled correctly?
            // Yes! Because we subtracted sum_neg from target to get new_target.
            // So we assumed all negative numbers are included in the subset, unless their corresponding positive representation is chosen in B.
            // If some negative number's positive representation has val > new_target, then it is impossible to choose it anyway (as it would exceed new_target).
            // So it will never be chosen, which means it will remain in the subset.
            // This is perfectly correct!
        }
    }

    // Now, group by value and positivity
    let mut groups: HashMap<(u128, bool), Vec<usize>> = HashMap::new();
    for &(val, idx, is_pos) in &B {
        groups.entry((val, is_pos)).or_default().push(idx);
    }

    // Binary decomposition of groups
    let mut decomposed = Vec::new();
    for ((val, is_pos), idxs) in groups {
        let mut c = idxs.len();
        let mut i = 0;
        let mut power = 1;
        while c >= power {
            let chunk = &idxs[i..i + power];
            decomposed.push(DecomposedItem {
                value: val * power as u128,
                is_positive: is_pos,
                indices: chunk.to_vec(),
            });
            i += power;
            c -= power;
            power *= 2;
        }
        if c > 0 {
            let chunk = &idxs[i..i + c];
            decomposed.push(DecomposedItem {
                value: val * c as u128,
                is_positive: is_pos,
                indices: chunk.to_vec(),
            });
        }
    }

    // Calculate sum of all decomposed values
    let total_decomposed_sum: u128 = decomposed.iter().map(|item| item.value).sum();
    if total_decomposed_sum < new_target {
        println!("NO");
        return;
    }
    if total_decomposed_sum == new_target {
        // All items must be chosen
        let mut final_indices = Vec::new();
        for item in &decomposed {
            if item.is_positive {
                final_indices.extend(&item.indices);
            }
        }
        // Originally negative elements that were completely filtered out (val > new_target)
        // are also not chosen, so their indices are included
        for &(val, idx) in &neg_nums {
            if val > new_target {
                final_indices.push(idx);
            }
        }
        final_indices.sort_unstable();
        println!("YES");
        print_indices(&final_indices);
        return;
    }

    // Determine strategy based on problem complexity
    let num_items = decomposed.len();

    let result = if num_items <= 48 {
        parallel_meet_in_the_middle(&decomposed, new_target)
    } else if new_target <= 50_000_000 && (num_items as u128 * new_target / 64) < 1_500_000_000 {
        bitset_dp(&decomposed, new_target)
    } else {
        // Fallback to randomized heuristic search
        heuristic_search(&decomposed, new_target, 25000)
    };


    if let Some(mut final_indices) = result {
        // Also add the originally negative elements that were filtered out (val > new_target)
        for &(val, idx) in &neg_nums {
            if val > new_target {
                final_indices.push(idx);
            }
        }
        final_indices.sort_unstable();
        println!("YES");
        print_indices(&final_indices);
    } else {
        println!("NO");
    }
}

fn print_indices(indices: &[usize]) {
    print!("[");
    for (i, &idx) in indices.iter().enumerate() {
        if i > 0 {
            print!(" ");
        }
        print!("{}", idx);
    }
    println!("]");
}
