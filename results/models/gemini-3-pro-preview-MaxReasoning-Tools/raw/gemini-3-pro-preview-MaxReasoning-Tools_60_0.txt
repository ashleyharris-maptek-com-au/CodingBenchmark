const fs = require('fs');

class MaxHeap {
    constructor(compare) {
        this.heap = [];
        this.compare = compare;
    }
    push(val) {
        this.heap.push(val);
        this.up(this.heap.length - 1);
    }
    pop() {
        if (this.heap.length === 0) return undefined;
        const top = this.heap[0];
        const bottom = this.heap.pop();
        if (this.heap.length > 0) {
            this.heap[0] = bottom;
            this.down(0);
        }
        return top;
    }
    size() {
        return this.heap.length;
    }
    up(i) {
        const h = this.heap;
        const comp = this.compare;
        const val = h[i];
        while (i > 0) {
            const p = (i - 1) >> 1;
            if (comp(val, h[p]) <= 0) break;
            h[i] = h[p];
            i = p;
        }
        h[i] = val;
    }
    down(i) {
        const h = this.heap;
        const comp = this.compare;
        const len = h.length;
        const val = h[i];
        const half = len >> 1;
        while (i < half) {
            let left = (i << 1) + 1;
            const right = left + 1;
            let best = left;
            if (right < len && comp(h[right], h[left]) > 0) {
                best = right;
            }
            if (comp(h[best], val) <= 0) break;
            h[i] = h[best];
            i = best;
        }
        h[i] = val;
    }
}

function solve() {
    const startTime = Date.now();

    // Read input from stdin
    const input = fs.readFileSync(0, 'utf8');
    if (!input || input.trim() === '') return;
    const data = JSON.parse(input);

    const machineCount = data.machineCount;
    const tasks = data.tasks;
    const N = tasks.length;

    if (N === 0) {
        console.log(JSON.stringify({ schedule: [] }));
        return;
    }

    const idToIndex = new Map();
    const indexToId = new Array(N);
    for (let i = 0; i < N; i++) {
        idToIndex.set(tasks[i].id, i);
        indexToId[i] = tasks[i].id;
    }

    const durations = new Int32Array(N);
    const deadlines = new Float64Array(N);
    const weights = new Float64Array(N);

    const tempDeps = Array.from({length: N}, () => []);
    const tempChildren = Array.from({length: N}, () => []);

    for (let i = 0; i < N; i++) {
        const t = tasks[i];
        durations[i] = Math.ceil(t.duration || 0);
        deadlines[i] = t.deadline || 0;
        weights[i] = t.weight || 0;
        const deps = t.deps || [];
        for (const depId of deps) {
            const depIdx = idToIndex.get(depId);
            if (depIdx !== undefined) {
                tempDeps[i].push(depIdx);
                tempChildren[depIdx].push(i);
            }
        }
    }

    const depStarts = new Int32Array(N);
    const depLens = new Int32Array(N);
    let totalDeps = 0;
    for (let i = 0; i < N; i++) {
        depStarts[i] = totalDeps;
        depLens[i] = tempDeps[i].length;
        totalDeps += depLens[i];
    }
    const depList = new Int32Array(totalDeps);
    let depPtr = 0;
    for (let i = 0; i < N; i++) {
        for (const d of tempDeps[i]) {
            depList[depPtr++] = d;
        }
    }

    const childStarts = new Int32Array(N);
    const childLens = new Int32Array(N);
    let totalChildren = 0;
    for (let i = 0; i < N; i++) {
        childStarts[i] = totalChildren;
        childLens[i] = tempChildren[i].length;
        totalChildren += tempChildren[i].length;
    }
    const childList = new Int32Array(totalChildren);
    let childPtr = 0;
    for (let i = 0; i < N; i++) {
        for (const c of tempChildren[i]) {
            childList[childPtr++] = c;
        }
    }

    function getAnyTopologicalOrder() {
        const inDegree = new Int32Array(N);
        for (let i = 0; i < N; i++) {
            inDegree[i] = depLens[i];
        }
        const queue = [];
        for (let i = 0; i < N; i++) {
            if (inDegree[i] === 0) queue.push(i);
        }
        const order = new Int32Array(N);
        let ptr = 0;
        while (queue.length > 0) {
            const u = queue.pop();
            order[ptr++] = u;
            const cStart = childStarts[u];
            const cLen = childLens[u];
            for (let j = 0; j < cLen; j++) {
                const v = childList[cStart + j];
                inDegree[v]--;
                if (inDegree[v] === 0) {
                    queue.push(v);
                }
            }
        }
        if (ptr < N) {
            const visited = new Uint8Array(N);
            for (let i = 0; i < ptr; i++) visited[order[i]] = 1;
            for (let i = 0; i < N; i++) {
                if (!visited[i]) {
                    order[ptr++] = i;
                }
            }
        }
        return order;
    }

    const topoOrder = getAnyTopologicalOrder();

    const cp_length = new Float64Array(N);
    for (let i = N - 1; i >= 0; i--) {
        const u = topoOrder[i];
        let maxChildCp = 0;
        const cStart = childStarts[u];
        const cLen = childLens[u];
        for (let j = 0; j < cLen; j++) {
            const v = childList[cStart + j];
            if (cp_length[v] > maxChildCp) {
                maxChildCp = cp_length[v];
            }
        }
        cp_length[u] = durations[u] + maxChildCp;
    }

    const weight_sum = new Float64Array(N);
    for (let i = N - 1; i >= 0; i--) {
        const u = topoOrder[i];
        let sum = weights[u];
        const cStart = childStarts[u];
        const cLen = childLens[u];
        for (let j = 0; j < cLen; j++) {
            const v = childList[cStart + j];
            sum += weight_sum[v];
        }
        weight_sum[u] = sum;
    }

    function generateOrderWithPriority(priorities) {
        const inDegree = new Int32Array(N);
        for (let i = 0; i < N; i++) {
            inDegree[i] = depLens[i];
        }
        
        const heap = new MaxHeap((a, b) => {
            const diff = priorities[a] - priorities[b];
            if (diff !== 0) return diff;
            return b - a;
        });
        
        for (let i = 0; i < N; i++) {
            if (inDegree[i] === 0) {
                heap.push(i);
            }
        }
        
        const order = new Int32Array(N);
        let ptr = 0;
        while (heap.size() > 0) {
            const u = heap.pop();
            order[ptr++] = u;
            const cStart = childStarts[u];
            const cLen = childLens[u];
            for (let j = 0; j < cLen; j++) {
                const v = childList[cStart + j];
                inDegree[v]--;
                if (inDegree[v] === 0) {
                    heap.push(v);
                }
            }
        }
        return order;
    }

    const mFree = new Float64Array(machineCount);
    const tFinish = new Float64Array(N);
    const currentStartTimes = new Int32Array(N);
    const currentMachines = new Int32Array(N);

    function simulate(order, record = false) {
        mFree.fill(0);
        tFinish.fill(0);
        let makespan = 0;
        let totalWeightedLateness = 0;
        
        for (let i = 0; i < N; i++) {
            const u = order[i];
            
            let depReadyTime = 0;
            const depStart = depStarts[u];
            const depLen = depLens[u];
            for (let j = 0; j < depLen; j++) {
                const dep = depList[depStart + j];
                const ft = tFinish[dep];
                if (ft > depReadyTime) {
                    depReadyTime = ft;
                }
            }
            
            let bestMachine = 0;
            let minFreeTime = mFree[0];
            for (let m = 1; m < machineCount; m++) {
                const t = mFree[m];
                if (t < minFreeTime) {
                    minFreeTime = t;
                    bestMachine = m;
                }
            }
            
            const startTime = depReadyTime > minFreeTime ? depReadyTime : minFreeTime;
            const finishTime = startTime + durations[u];
            
            mFree[bestMachine] = finishTime;
            tFinish[u] = finishTime;
            
            if (finishTime > makespan) {
                makespan = finishTime;
            }
            
            const lateness = finishTime - deadlines[u];
            if (lateness > 0) {
                totalWeightedLateness += lateness * weights[u];
            }
            
            if (record) {
                currentStartTimes[u] = startTime;
                currentMachines[u] = bestMachine;
            }
        }
        
        return makespan + totalWeightedLateness;
    }

    function shiftInPlace(order, pos, fromIdx, toIdx) {
        const u = order[fromIdx];
        if (fromIdx < toIdx) {
            for (let i = fromIdx; i < toIdx; i++) {
                const val = order[i + 1];
                order[i] = val;
                pos[val] = i;
            }
        } else {
            for (let i = fromIdx; i > toIdx; i--) {
                const val = order[i - 1];
                order[i] = val;
                pos[val] = i;
            }
        }
        order[toIdx] = u;
        pos[u] = toIdx;
    }

    // Evaluate multiple heuristic priorities
    const priorityHeuristics = [];

    // CP
    const cpPr = new Float64Array(N);
    for (let i = 0; i < N; i++) cpPr[i] = cp_length[i];
    priorityHeuristics.push(cpPr);

    // WCP
    const wcpPr = new Float64Array(N);
    for (let i = 0; i < N; i++) wcpPr[i] = cp_length[i] * weight_sum[i];
    priorityHeuristics.push(wcpPr);

    // EDF
    const edfPr = new Float64Array(N);
    for (let i = 0; i < N; i++) edfPr[i] = -deadlines[i];
    priorityHeuristics.push(edfPr);

    // MST
    const mstPr = new Float64Array(N);
    for (let i = 0; i < N; i++) mstPr[i] = -(deadlines[i] - durations[i]);
    priorityHeuristics.push(mstPr);

    // WSPT
    const wsptPr = new Float64Array(N);
    for (let i = 0; i < N; i++) wsptPr[i] = weights[i] / (durations[i] || 1);
    priorityHeuristics.push(wsptPr);

    // Weight Sum
    const wsPr = new Float64Array(N);
    for (let i = 0; i < N; i++) wsPr[i] = weight_sum[i];
    priorityHeuristics.push(wsPr);

    // Combined Heuristics
    const c1Pr = new Float64Array(N);
    for (let i = 0; i < N; i++) c1Pr[i] = cp_length[i] * 10 - deadlines[i];
    priorityHeuristics.push(c1Pr);

    const c2Pr = new Float64Array(N);
    for (let i = 0; i < N; i++) c2Pr[i] = cp_length[i] + weight_sum[i] * 2;
    priorityHeuristics.push(c2Pr);

    let bestOrder = null;
    let bestScore = Infinity;

    for (let h = 0; h < priorityHeuristics.length; h++) {
        const ord = generateOrderWithPriority(priorityHeuristics[h]);
        const sc = simulate(ord);
        if (sc < bestScore) {
            bestScore = sc;
            bestOrder = ord;
        }
    }

    const rawTopoScore = simulate(topoOrder);
    if (rawTopoScore < bestScore) {
        bestScore = rawTopoScore;
        bestOrder = topoOrder;
    }

    // Initialize best recorded schedule
    const bestStartTimes = new Int32Array(N);
    const bestMachines = new Int32Array(N);
    simulate(bestOrder, true);
    bestStartTimes.set(currentStartTimes);
    bestMachines.set(currentMachines);

    // Adapt SA time limit to graph size
    let TIME_LIMIT = 1500;
    if (N > 20000) TIME_LIMIT = 800;
    if (N > 80000) TIME_LIMIT = 400;

    // Local Search via Simulated Annealing
    const order = new Int32Array(bestOrder);
    const pos = new Int32Array(N);
    for (let i = 0; i < N; i++) {
        pos[order[i]] = i;
    }

    let currentScore = bestScore;
    const currentTFinish = new Float64Array(N);
    currentTFinish.set(tFinish);

    const T0 = currentScore * 0.02;
    const Tmin = T0 * 1e-4;

    const lateTasks = [];
    const criticalTasks = [];

    while (true) {
        const elapsed = Date.now() - startTime;
        if (elapsed >= TIME_LIMIT) break;

        const progress = Math.min(1.0, elapsed / TIME_LIMIT);
        const T = T0 * Math.pow(Tmin / T0, progress);

        let u;
        if (N <= 2000) {
            lateTasks.length = 0;
            criticalTasks.length = 0;
            let makespan = 0;
            for (let i = 0; i < N; i++) {
                if (currentTFinish[i] > makespan) {
                    makespan = currentTFinish[i];
                }
            }
            for (let i = 0; i < N; i++) {
                if (currentTFinish[i] > deadlines[i]) {
                    lateTasks.push(i);
                }
                if (currentTFinish[i] === makespan) {
                    criticalTasks.push(i);
                }
            }

            const r = Math.random();
            if (r < 0.5 && lateTasks.length > 0) {
                u = lateTasks[Math.floor(Math.random() * lateTasks.length)];
            } else if (r < 0.8 && criticalTasks.length > 0) {
                u = criticalTasks[Math.floor(Math.random() * criticalTasks.length)];
            } else {
                u = Math.floor(Math.random() * N);
            }
        } else {
            u = Math.floor(Math.random() * N);
        }

        const currentPos = pos[u];

        // Find valid range
        let min_pos = 0;
        const dStart = depStarts[u];
        const dLen = depLens[u];
        for (let j = 0; j < dLen; j++) {
            const p = pos[depList[dStart + j]];
            if (p > min_pos) min_pos = p;
        }
        if (dLen > 0) min_pos++;

        let max_pos = N - 1;
        const cStart = childStarts[u];
        const cLen = childLens[u];
        for (let j = 0; j < cLen; j++) {
            const p = pos[childList[cStart + j]];
            if (p < max_pos) max_pos = p;
        }
        if (cLen > 0) max_pos--;

        if (min_pos >= max_pos) continue;

        let k = -1;
        const r = Math.random();
        if (currentPos > min_pos && (r < 0.5 || Math.random() < 0.5)) {
            k = min_pos + Math.floor(Math.random() * (currentPos - min_pos));
        } else if (currentPos < max_pos) {
            k = (currentPos + 1) + Math.floor(Math.random() * (max_pos - currentPos));
        }

        if (k !== -1 && k !== currentPos) {
            shiftInPlace(order, pos, currentPos, k);
            const newScore = simulate(order);
            const delta = newScore - currentScore;
            const accept = delta <= 0 || Math.random() < Math.exp(-delta / T);

            if (accept) {
                currentScore = newScore;
                currentTFinish.set(tFinish);
                if (newScore < bestScore) {
                    bestScore = newScore;
                    bestOrder.set(order);
                    // Record start times and machine assignments
                    simulate(order, true);
                    bestStartTimes.set(currentStartTimes);
                    bestMachines.set(currentMachines);
                }
            } else {
                shiftInPlace(order, pos, k, currentPos);
            }
        }
    }

    const scheduleOut = [];
    for (let i = 0; i < N; i++) {
        scheduleOut.push({
            id: indexToId[i],
            start: bestStartTimes[i],
            machine: bestMachines[i]
        });
    }

    console.log(JSON.stringify({ schedule: scheduleOut }));
}

solve();
