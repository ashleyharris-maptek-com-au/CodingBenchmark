const double PI = 3.14159265358979323846;

struct Vector3 {
    double x, y, z;
};

struct PathPoint {
    double x, y, z;
    double s;
};

struct LanderState {
    double x, y, z;
    double vx, vy, vz;
    double pitch, yaw, roll;
    double pitch_rate, yaw_rate, roll_rate;
    double fuel;
    bool valid = false;
};

struct ControlCommand {
    double thrust;
    double pitch_cmd;
    double yaw_cmd;
    double roll_cmd;
};

#pragma pack(push, 1)
struct Cell {
    uint16_t a;
    uint16_t b;
    uint16_t c;
};
#pragma pack(pop)

class TerrainMap {
public:
    int width = 0;
    int depth = 0;
    int height = 0;
    
    HANDLE hFile = INVALID_HANDLE_VALUE;
    HANDLE hMapping = NULL;
    void* m_data = nullptr;
    const Cell* cells = nullptr;
    std::vector<Cell> fallback_cells;

    ~TerrainMap() {
        if (m_data) UnmapViewOfFile(m_data);
        if (hMapping) CloseHandle(hMapping);
        if (hFile != INVALID_HANDLE_VALUE) CloseHandle(hFile);
    }

    bool load(const std::string& path, int w, int d, int h) {
        width = w;
        depth = d;
        height = h;
        
        hFile = CreateFileA(path.c_str(), GENERIC_READ, FILE_SHARE_READ, NULL, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);
        if (hFile == INVALID_HANDLE_VALUE) {
            return load_fallback(path);
        }
        
        LARGE_INTEGER size;
        if (!GetFileSizeEx(hFile, &size)) {
            CloseHandle(hFile);
            hFile = INVALID_HANDLE_VALUE;
            return load_fallback(path);
        }

        hMapping = CreateFileMappingA(hFile, NULL, PAGE_READONLY, 0, 0, NULL);
        if (hMapping == NULL) {
            CloseHandle(hFile);
            hFile = INVALID_HANDLE_VALUE;
            return load_fallback(path);
        }

        m_data = MapViewOfFile(hMapping, FILE_MAP_READ, 0, 0, 0);
        if (m_data == nullptr) {
            CloseHandle(hMapping);
            hMapping = NULL;
            CloseHandle(hFile);
            hFile = INVALID_HANDLE_VALUE;
            return load_fallback(path);
        }

        cells = reinterpret_cast<const Cell*>(m_data);
        return true;
    }

    bool load_fallback(const std::string& path) {
        std::ifstream fs(path, std::ios::binary);
        if (!fs) return false;
        size_t num_cells = (size_t)width * depth;
        fallback_cells.resize(num_cells);
        fs.read(reinterpret_cast<char*>(fallback_cells.data()), num_cells * sizeof(Cell));
        cells = fallback_cells.data();
        return true;
    }

    Cell get_cell(int x, int y) const {
        if (x < 0 || x >= width || y < 0 || y >= depth || cells == nullptr) {
            return {0, 0, 0};
        }
        return cells[x * depth + y];
    }
};

struct AStarNode {
    int idx;
    double f_score;
    bool operator>(const AStarNode& other) const {
        return f_score > other.f_score;
    }
};

std::vector<std::pair<int, int>> plan_coarse_path(
    int sx_c, int sy_c, int tx_c, int ty_c,
    int W_c, int D_c, double K,
    const std::vector<uint16_t>& H_coarse,
    double height, double safety_margin
) {
    int start_idx = sx_c * D_c + sy_c;
    int target_idx = tx_c * D_c + ty_c;

    std::vector<double> g_score(W_c * D_c, std::numeric_limits<double>::infinity());
    std::vector<int> came_from(W_c * D_c, -1);
    std::vector<bool> closed(W_c * D_c, false);

    std::priority_queue<AStarNode, std::vector<AStarNode>, std::greater<AStarNode>> open_set;

    g_score[start_idx] = 0.0;
    open_set.push({start_idx, 0.0});

    auto heuristic = [&](int idx) {
        int cx = idx / D_c;
        int cy = idx % D_c;
        double dx = (cx - tx_c) * K;
        double dy = (cy - ty_c) * K;
        return std::sqrt(dx * dx + dy * dy);
    };

    while (!open_set.empty()) {
        AStarNode curr = open_set.top();
        open_set.pop();

        int curr_idx = curr.idx;
        if (closed[curr_idx]) continue;
        closed[curr_idx] = true;

        if (curr_idx == target_idx) {
            break;
        }

        int cx = curr_idx / D_c;
        int cy = curr_idx % D_c;

        for (int dx = -1; dx <= 1; ++dx) {
            for (int dy = -1; dy <= 1; ++dy) {
                if (dx == 0 && dy == 0) continue;
                int nx = cx + dx;
                int ny = cy + dy;
                if (nx < 0 || nx >= W_c || ny < 0 || ny >= D_c) continue;

                int neighbor_idx = nx * D_c + ny;
                if (closed[neighbor_idx]) continue;

                double h_limit = height - safety_margin;
                double h_neighbor = H_coarse[neighbor_idx];
                if (h_neighbor >= h_limit) continue;

                double dist = std::sqrt(dx * dx + dy * dy) * K;
                double cost = dist * (1.0 + 0.01 * h_neighbor);

                double tentative_g = g_score[curr_idx] + cost;
                if (tentative_g < g_score[neighbor_idx]) {
                    g_score[neighbor_idx] = tentative_g;
                    came_from[neighbor_idx] = curr_idx;
                    open_set.push({neighbor_idx, tentative_g + heuristic(neighbor_idx)});
                }
            }
        }
    }

    std::vector<std::pair<int, int>> path;
    if (came_from[target_idx] == -1 && start_idx != target_idx) {
        return {};
    }

    int curr = target_idx;
    while (curr != -1) {
        path.push_back({curr / D_c, curr % D_c});
        curr = came_from[curr];
    }
    std::reverse(path.begin(), path.end());
    return path;
}

std::vector<double> plan_height_profile(
    const std::vector<std::pair<int, int>>& coarse_path,
    const std::vector<uint16_t>& H_coarse,
    int D_c, double K, double safety_margin,
    double start_z, double target_z
) {
    size_t n = coarse_path.size();
    if (n == 0) return {};

    std::vector<double> Z(n);
    for (size_t i = 0; i < n; ++i) {
        int idx = coarse_path[i].first * D_c + coarse_path[i].second;
        Z[i] = H_coarse[idx] + safety_margin;
    }

    Z[0] = std::max(Z[0], start_z);
    Z[n-1] = std::max(Z[n-1], target_z);

    double max_slope = 0.2;

    for (size_t i = 1; i < n; ++i) {
        double dx = (coarse_path[i].first - coarse_path[i-1].first) * K;
        double dy = (coarse_path[i].second - coarse_path[i-1].second) * K;
        double d = std::sqrt(dx*dx + dy*dy);
        Z[i] = std::max(Z[i], Z[i-1] - max_slope * d);
    }

    for (int i = (int)n - 2; i >= 0; --i) {
        double dx = (coarse_path[i].first - coarse_path[i+1].first) * K;
        double dy = (coarse_path[i].second - coarse_path[i+1].second) * K;
        double d = std::sqrt(dx*dx + dy*dy);
        Z[i] = std::max(Z[i], Z[i+1] - max_slope * d);
    }

    return Z;
}

double get_max_height_on_line(const TerrainMap& map, double x1, double y1, double x2, double y2) {
    double dx = x2 - x1;
    double dy = y2 - y1;
    double dist = std::sqrt(dx*dx + dy*dy);
    if (dist < 1e-3) {
        Cell c = map.get_cell((int)x1, (int)y1);
        return (c.b != c.c) ? c.c : c.a;
    }
    
    double max_h = 0.0;
    int steps = (int)std::ceil(dist);
    for (int i = 0; i <= steps; ++i) {
        double t = (double)i / steps;
        double px = x1 + t * dx;
        double py = y1 + t * dy;
        Cell c = map.get_cell((int)px, (int)py);
        double h = (c.b != c.c) ? c.c : c.a;
        if (h > max_h) {
            max_h = h;
        }
    }
    return max_h;
}

std::vector<PathPoint> build_path_points(const std::vector<Vector3>& wps) {
    std::vector<PathPoint> pts;
    double dist = 0.0;
    pts.push_back({wps[0].x, wps[0].y, wps[0].z, 0.0});
    for (size_t i = 1; i < wps.size(); ++i) {
        double dx = wps[i].x - wps[i-1].x;
        double dy = wps[i].y - wps[i-1].y;
        double dz = wps[i].z - wps[i-1].z;
        dist += std::sqrt(dx*dx + dy*dy + dz*dz);
        pts.push_back({wps[i].x, wps[i].y, wps[i].z, dist});
    }
    return pts;
}

Vector3 get_point_at_s(const std::vector<PathPoint>& pts, double s) {
    if (pts.empty()) return {0.0, 0.0, 0.0};
    if (s <= 0.0) return {pts.front().x, pts.front().y, pts.front().z};
    if (s >= pts.back().s) return {pts.back().x, pts.back().y, pts.back().z};
    
    auto it = std::upper_bound(pts.begin(), pts.end(), s, [](double val, const PathPoint& p) {
        return val < p.s;
    });
    if (it == pts.end()) return {pts.back().x, pts.back().y, pts.back().z};
    
    const PathPoint& p1 = *(it - 1);
    const PathPoint& p2 = *it;
    double ratio = (s - p1.s) / (p2.s - p1.s);
    return {
        p1.x + ratio * (p2.x - p1.x),
        p1.y + ratio * (p2.y - p1.y),
        p1.z + ratio * (p2.z - p1.z)
    };
}

double find_closest_s(const std::vector<PathPoint>& pts, const Vector3& p) {
    double min_dist2 = std::numeric_limits<double>::infinity();
    double closest_s = 0.0;
    for (size_t i = 0; i < pts.size() - 1; ++i) {
        Vector3 A{pts[i].x, pts[i].y, pts[i].z};
        Vector3 B{pts[i+1].x, pts[i+1].y, pts[i+1].z};
        Vector3 AB = {B.x - A.x, B.y - A.y, B.z - A.z};
        double ab2 = AB.x*AB.x + AB.y*AB.y + AB.z*AB.z;
        if (ab2 < 1e-9) continue;
        
        Vector3 Ap = {p.x - A.x, p.y - A.y, p.z - A.z};
        double t = (Ap.x*AB.x + Ap.y*AB.y + Ap.z*AB.z) / ab2;
        t = std::clamp(t, 0.0, 1.0);
        
        Vector3 C = {A.x + AB.x * t, A.y + AB.y * t, A.z + AB.z * t};
        Vector3 diff = {p.x - C.x, p.y - C.y, p.z - C.z};
        double dist2 = diff.x*diff.x + diff.y*diff.y + diff.z*diff.z;
        if (dist2 < min_dist2) {
            min_dist2 = dist2;
            closest_s = pts[i].s + t * std::sqrt(ab2);
        }
    }
    return closest_s;
}

// Global configurations & state
int width, depth;
double height, gravity, max_thrust, fuel;
double start_x, start_y, start_z;
double target_x, target_y, target_z;
std::string terrain_file_path;
double D_total;

std::mutex state_mutex;
LanderState latest_state;
std::condition_variable state_cv;
std::atomic<bool> simulation_running{true};
std::atomic<bool> path_ready{false};
std::vector<PathPoint> path_points;

void path_planning_thread_func() {
    TerrainMap map;
    map.load(terrain_file_path, width, depth, (int)height);

    int max_dim = std::max(width, depth);
    int K = 100;
    if (max_dim > 1000) {
        K = max_dim / 500;
    }
    if (K < 1) K = 1;

    int W_c = (width + K - 1) / K;
    int D_c = (depth + K - 1) / K;

    std::vector<uint16_t> H_coarse(W_c * D_c, 0);
    if (map.cells != nullptr) {
        for (int cx = 0; cx < W_c; ++cx) {
            int x_start = cx * K;
            int x_end = std::min(width, (cx + 1) * K);
            for (int cy = 0; cy < D_c; ++cy) {
                int y_start = cy * K;
                int y_end = std::min(depth, (cy + 1) * K);
                uint16_t max_h = 0;
                for (int x = x_start; x < x_end; ++x) {
                    for (int y = y_start; y < y_end; ++y) {
                        Cell cell = map.get_cell(x, y);
                        uint16_t h = (cell.b != cell.c) ? cell.c : cell.a;
                        if (h > max_h) {
                            max_h = h;
                        }
                    }
                }
                H_coarse[cx * D_c + cy] = max_h;
            }
        }
    }

    int sx_c = std::clamp((int)(start_x / K), 0, W_c - 1);
    int sy_c = std::clamp((int)(start_y / K), 0, D_c - 1);
    int tx_c = std::clamp((int)(target_x / K), 0, W_c - 1);
    int ty_c = std::clamp((int)(target_y / K), 0, D_c - 1);

    double safety_margin = std::clamp(D_total * 0.2, 5.0, 15.0);
    auto coarse_path = plan_coarse_path(sx_c, sy_c, tx_c, ty_c, W_c, D_c, (double)K, H_coarse, height, safety_margin);

    std::vector<Vector3> wps;
    wps.push_back({start_x, start_y, start_z});

    if (coarse_path.empty() || coarse_path.size() <= 2) {
        double max_h = get_max_height_on_line(map, start_x, start_y, target_x, target_y);
        double z_mid = std::max({start_z, target_z, max_h + safety_margin});
        z_mid = std::min(z_mid, height - safety_margin);
        
        wps.push_back({start_x, start_y, z_mid});
        wps.push_back({target_x, target_y, z_mid});
    } else {
        auto Z = plan_height_profile(coarse_path, H_coarse, D_c, (double)K, safety_margin, start_z, target_z);
        
        wps.push_back({start_x, start_y, Z[0]});
        
        for (size_t i = 1; i < coarse_path.size() - 1; ++i) {
            double px = (coarse_path[i].first + 0.5) * K;
            double py = (coarse_path[i].second + 0.5) * K;
            wps.push_back({px, py, Z[i]});
        }
        
        wps.push_back({target_x, target_y, Z.back()});
    }
    wps.push_back({target_x, target_y, target_z});

    path_points = build_path_points(wps);
    path_ready = true;
}

ControlCommand compute_control(const LanderState& state) {
    double dist_xy = std::sqrt((state.x - target_x)*(state.x - target_x) + (state.y - target_y)*(state.y - target_y));
    double dist_z = std::abs(state.z - target_z);
    double speed = std::sqrt(state.vx*state.vx + state.vy*state.vy + state.vz*state.vz);
    double landing_radius = std::max(5.0, std::min((double)width, (double)depth) * 0.02);

    if (dist_xy < landing_radius * 0.5 && dist_z < 0.3 && speed < 2.0) {
        return {0.0, 0.0, 0.0, 0.0};
    }

    Vector3 target_pos;
    Vector3 target_vel;

    if (!path_ready) {
        target_pos = {start_x, start_y, start_z};
        target_vel = {0.0, 0.0, 0.0};
    } else {
        double closest_s = find_closest_s(path_points, {state.x, state.y, state.z});
        double total_s = path_points.back().s;
        double d_remain = std::max(0.0, total_s - closest_s);

        double v_max_cruise = std::min(150.0, D_total / 10.0 + 5.0);
        double a_dec = 0.1 * max_thrust;
        
        double v_fuel = 999.0;
        double term = (max_thrust * state.fuel) * (max_thrust * state.fuel) - 8.0 * gravity * d_remain;
        if (term >= 0.0) {
            v_fuel = (max_thrust * state.fuel + std::sqrt(term)) / 4.0;
        } else {
            v_fuel = std::sqrt(gravity * d_remain / 2.0);
        }
        
        double v_cruise = std::min({v_max_cruise, 0.8 * v_fuel, std::sqrt(2.0 * a_dec * d_remain)});
        v_cruise = std::max(2.0, v_cruise);

        double L = 15.0 + 1.5 * speed;
        L = std::min(L, d_remain);

        target_pos = get_point_at_s(path_points, closest_s + L);

        Vector3 dir = { target_pos.x - state.x, target_pos.y - state.y, target_pos.z - state.z };
        double dir_len = std::sqrt(dir.x*dir.x + dir.y*dir.y + dir.z*dir.z);
        if (dir_len > 1e-3) {
            target_vel = { dir.x / dir_len * v_cruise, dir.y / dir_len * v_cruise, dir.z / dir_len * v_cruise };
        } else {
            target_vel = {0.0, 0.0, 0.0};
        }

        if (dist_xy < landing_radius * 0.8) {
            target_pos = {target_x, target_y, target_z};
            double vz_target = std::max(-1.5, -0.5 * (state.z - target_z));
            target_vel = {0.0, 0.0, vz_target};
        }
    }

    double Kp_xy = 0.5;
    double Kv_xy = 1.414;
    double Kp_z = 0.6;
    double Kv_z = 1.5;

    double err_x = std::clamp(target_pos.x - state.x, -20.0, 20.0);
    double err_y = std::clamp(target_pos.y - state.y, -20.0, 20.0);
    double err_z = std::clamp(target_pos.z - state.z, -20.0, 20.0);

    double ax = Kp_xy * err_x + Kv_xy * (target_vel.x - state.vx);
    double ay = Kp_xy * err_y + Kv_xy * (target_vel.y - state.vy);
    double az = Kp_z * err_z + Kv_z * (target_vel.z - state.vz) + gravity;

    if (az < 0.0) az = 0.0;

    double a_mag = std::sqrt(ax*ax + ay*ay + az*az);
    double thrust_cmd = 0.0;
    Vector3 thrust_dir = {0, 0, 1};

    if (a_mag > 1e-6) {
        double clamped_mag = std::clamp(a_mag, 0.0, max_thrust);
        thrust_cmd = clamped_mag / max_thrust;
        thrust_dir = { ax / a_mag, ay / a_mag, az / a_mag };
    }

    double target_pitch = 0.0;
    double target_yaw = state.yaw;

    if (thrust_dir.z < 0.999) {
        target_pitch = std::acos(thrust_dir.z);
        target_yaw = std::atan2(-thrust_dir.y, -thrust_dir.x);
    }

    double K_theta = 1.0;
    double K_omega = 2.0;

    double pitch_err = target_pitch - state.pitch;
    double pitch_rate_desired = K_theta * pitch_err;
    pitch_rate_desired = std::clamp(pitch_rate_desired, -1.5, 1.5);
    double pitch_cmd = K_omega * (pitch_rate_desired - state.pitch_rate);
    pitch_cmd = std::clamp(pitch_cmd, -1.0, 1.0);

    double yaw_err = target_yaw - state.yaw;
    while (yaw_err < -PI) yaw_err += 2.0 * PI;
    while (yaw_err > PI) yaw_err -= 2.0 * PI;
    double yaw_rate_desired = K_theta * yaw_err;
    yaw_rate_desired = std::clamp(yaw_rate_desired, -1.5, 1.5);
    double yaw_cmd = K_omega * (yaw_rate_desired - state.yaw_rate);
    yaw_cmd = std::clamp(yaw_cmd, -1.0, 1.0);

    double roll_err = 0.0 - state.roll;
    while (roll_err < -PI) roll_err += 2.0 * PI;
    while (roll_err > PI) roll_err -= 2.0 * PI;
    double roll_rate_desired = K_theta * roll_err;
    roll_rate_desired = std::clamp(roll_rate_desired, -1.5, 1.5);
    double roll_cmd = K_omega * (roll_rate_desired - state.roll_rate);
    roll_cmd = std::clamp(roll_cmd, -1.0, 1.0);

    return { thrust_cmd, pitch_cmd, yaw_cmd, roll_cmd };
}

void input_thread_func() {
    std::string line;
    while (simulation_running) {
        if (!std::getline(std::cin, line)) {
            simulation_running = false;
            state_cv.notify_all();
            break;
        }
        if (line.empty()) continue;
        
        std::stringstream ss(line);
        LanderState state;
        if (ss >> state.x >> state.y >> state.z 
               >> state.vx >> state.vy >> state.vz 
               >> state.pitch >> state.yaw >> state.roll 
               >> state.pitch_rate >> state.yaw_rate >> state.roll_rate 
               >> state.fuel) {
            state.valid = true;
            {
                std::lock_guard<std::mutex> lock(state_mutex);
                latest_state = state;
            }
            state_cv.notify_one();
        }
    }
}

void control_thread_func() {
    while (simulation_running) {
        LanderState state;
        {
            std::unique_lock<std::mutex> lock(state_mutex);
            state_cv.wait(lock, [] { return !simulation_running || latest_state.valid; });
            if (!simulation_running) break;
            state = latest_state;
            latest_state.valid = false;
        }
        
        ControlCommand cmd = compute_control(state);
        
        std::cout << std::fixed << std::setprecision(6) 
                  << cmd.thrust << " " 
                  << cmd.pitch_cmd << " " 
                  << cmd.yaw_cmd << " " 
                  << cmd.roll_cmd << "\n";
        std::cout.flush();
    }
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    if (!(std::cin >> width >> depth >> height >> gravity >> max_thrust >> fuel)) {
        return 0;
    }
    if (!(std::cin >> start_x >> start_y >> start_z >> target_x >> target_y >> target_z)) {
        return 0;
    }
    
    std::string dummy;
    std::getline(std::cin, dummy); 

    if (!std::getline(std::cin, terrain_file_path)) {
        return 0;
    }
    while (!terrain_file_path.empty() && (terrain_file_path.back() == '\r' || terrain_file_path.back() == ' ')) {
        terrain_file_path.pop_back();
    }

    std::string state_marker;
    if (!std::getline(std::cin, state_marker)) {
        return 0;
    }

    D_total = std::sqrt((target_x - start_x)*(target_x - start_x) + (target_y - start_y)*(target_y - start_y));

    std::thread planning_thread(path_planning_thread_func);
    std::thread input_thread(input_thread_func);
    std::thread control_thread(control_thread_func);

    planning_thread.join();
    input_thread.join();
    control_thread.join();

    return 0;
}
