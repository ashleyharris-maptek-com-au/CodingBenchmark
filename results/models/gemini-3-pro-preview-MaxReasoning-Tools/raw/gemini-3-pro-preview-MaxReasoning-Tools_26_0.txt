use std::io::{self, Read, Write, BufReader, BufWriter};
use std::collections::HashSet;

// A simple pseudo-random number generator
struct Xorshift64 {
    state: u64,
}

impl Xorshift64 {
    fn new(seed: u64) -> Self {
        let mut seed = seed;
        if seed == 0 {
            seed = 1;
        }
        Self { state: seed }
    }
    
    fn next(&mut self) -> u64 {
        self.state ^= self.state << 13;
        self.state ^= self.state >> 7;
        self.state ^= self.state << 17;
        self.state
    }
    
    fn next_range(&mut self, min: usize, max: usize) -> usize {
        if min >= max {
            return min;
        }
        let val = self.next();
        min + (val % (max - min) as u64) as usize
    }
}

#[inline(always)]
fn sq_dist(a: &[f32; 3], b: &[f32; 3]) -> f32 {
    let dx = a[0] - b[0];
    let dy = a[1] - b[1];
    let dz = a[2] - b[2];
    dx * dx + dy * dy + dz * dz
}

// Flat KD-Tree structures
struct FlatKdNode {
    point: [f32; 3],
    index: u32,
    left: i32,
    right: i32,
}

fn build_kd_tree_rec(
    points: &mut [([f32; 3], usize)],
    depth: usize,
    nodes: &mut Vec<FlatKdNode>,
) -> i32 {
    if points.is_empty() {
        return -1;
    }
    
    let axis = depth % 3;
    let median_idx = points.len() / 2;
    
    points.select_nth_unstable_by(median_idx, |a, b| {
        a.0[axis].partial_cmp(&b.0[axis]).unwrap_or(std::cmp::Ordering::Equal)
    });
    
    let (median_point, original_idx) = points[median_idx];
    let node_idx = nodes.len() as i32;
    
    nodes.push(FlatKdNode {
        point: median_point,
        index: original_idx as u32,
        left: -1,
        right: -1,
    });
    
    let left_child = build_kd_tree_rec(&mut points[..median_idx], depth + 1, nodes);
    let right_child = build_kd_tree_rec(&mut points[median_idx + 1..], depth + 1, nodes);
    
    nodes[node_idx as usize].left = left_child;
    nodes[node_idx as usize].right = right_child;
    
    node_idx
}

fn nearest_neighbor(
    nodes: &[FlatKdNode],
    root_idx: i32,
    depth: usize,
    query: &[f32; 3],
    best_idx: &mut usize,
    best_dist_sq: &mut f32,
) {
    if root_idx == -1 {
        return;
    }
    
    let node = &nodes[root_idx as usize];
    let d_sq = sq_dist(query, &node.point);
    if d_sq < *best_dist_sq {
        *best_dist_sq = d_sq;
        *best_idx = node.index as usize;
    }
    
    let axis = depth % 3;
    let diff = query[axis] - node.point[axis];
    
    let (near_child, far_child) = if diff <= 0.0 {
        (node.left, node.right)
    } else {
        (node.right, node.left)
    };
    
    nearest_neighbor(nodes, near_child, depth + 1, query, best_idx, best_dist_sq);
    
    if diff * diff < *best_dist_sq {
        nearest_neighbor(nodes, far_child, depth + 1, query, best_idx, best_dist_sq);
    }
}

fn initialize_centroids(
    points: &[[f32; 3]],
    num_clusters: usize,
    rng: &mut Xorshift64,
) -> Vec<[f32; 3]> {
    let n = points.len();
    let mut centroids = Vec::with_capacity(num_clusters);
    
    if n <= num_clusters {
        for &p in points {
            centroids.push(p);
        }
        while centroids.len() < num_clusters {
            let base = points[rng.next_range(0, n)];
            let rx = (rng.next() as f32 / u64::MAX as f32 - 0.5) * 1e-5;
            let ry = (rng.next() as f32 / u64::MAX as f32 - 0.5) * 1e-5;
            let rz = (rng.next() as f32 / u64::MAX as f32 - 0.5) * 1e-5;
            centroids.push([base[0] + rx, base[1] + ry, base[2] + rz]);
        }
    } else {
        let mut selected = HashSet::new();
        while centroids.len() < num_clusters {
            let idx = rng.next_range(0, n);
            if selected.insert(idx) {
                centroids.push(points[idx]);
            }
        }
    }
    centroids
}

fn ensure_no_empty_centroids(
    points: &[[f32; 3]],
    centroids: &mut [[f32; 3]],
    rng: &mut Xorshift64,
) {
    let num_clusters = centroids.len();
    let use_kd = num_clusters > 64;
    
    let mut kd_nodes = Vec::new();
    let mut root_idx = -1;
    if use_kd {
        let mut kd_points: Vec<([f32; 3], usize)> = centroids.iter().copied().enumerate().map(|(i, c)| (c, i)).collect();
        root_idx = build_kd_tree_rec(&mut kd_points, 0, &mut kd_nodes);
    }
    
    let mut counts = vec![0; num_clusters];
    for p in points {
        let best_idx = if use_kd {
            let mut best_idx = 0;
            let mut best_dist_sq = f32::INFINITY;
            nearest_neighbor(&kd_nodes, root_idx, 0, p, &mut best_idx, &mut best_dist_sq);
            best_idx
        } else {
            let mut min_dist = f32::INFINITY;
            let mut min_idx = 0;
            for (idx, c) in centroids.iter().enumerate() {
                let d = sq_dist(p, c);
                if d < min_dist {
                    min_dist = d;
                    min_idx = idx;
                }
            }
            min_idx
        };
        counts[best_idx] += 1;
    }
    
    let mut empty_indices = Vec::new();
    for i in 0..num_clusters {
        if counts[i] == 0 {
            empty_indices.push(i);
        }
    }
    
    if empty_indices.is_empty() {
        return;
    }
    
    let mut selected_points = HashSet::new();
    for &empty_idx in &empty_indices {
        let mut attempts = 0;
        let mut found = false;
        while attempts < 100 {
            let r_idx = rng.next_range(0, points.len());
            if selected_points.insert(r_idx) {
                centroids[empty_idx] = points[r_idx];
                found = true;
                break;
            }
            attempts += 1;
        }
        if !found {
            let r_idx = rng.next_range(0, points.len());
            centroids[empty_idx] = points[r_idx];
        }
    }
}

fn run_kmeans(
    points: &[[f32; 3]],
    num_clusters: usize,
    max_iters: usize,
    rng: &mut Xorshift64,
) -> Vec<[f32; 3]> {
    let n = points.len();
    let mut centroids = initialize_centroids(points, num_clusters, rng);
    let mut new_centroids = vec![[0.0, 0.0, 0.0]; num_clusters];
    let mut prev_loss = f64::INFINITY;
    
    for _iter in 0..max_iters {
        let use_kd = num_clusters > 64;
        let mut kd_nodes = Vec::new();
        let mut root_idx = -1;
        if use_kd {
            let mut kd_points: Vec<([f32; 3], usize)> = centroids.iter().copied().enumerate().map(|(i, c)| (c, i)).collect();
            root_idx = build_kd_tree_rec(&mut kd_points, 0, &mut kd_nodes);
        }
        
        let num_threads = std::thread::available_parallelism().map(|p| p.get()).unwrap_or(4);
        let chunk_size = (n + num_threads - 1) / num_threads;
        
        struct ThreadResult {
            local_sums: Vec<[f64; 3]>,
            local_counts: Vec<usize>,
            loss: f64,
        }
        
        let centroids_ref = &centroids;
        let kd_nodes_ref = &kd_nodes;
        
        let thread_results = std::thread::scope(|s| {
            let mut handles = Vec::new();
            for chunk in points.chunks(chunk_size) {
                handles.push(s.spawn(move || {
                    let mut local_sums = vec![[0.0, 0.0, 0.0]; num_clusters];
                    let mut local_counts = vec![0; num_clusters];
                    let mut loss = 0.0;
                    
                    for p in chunk {
                        let best_idx = if use_kd {
                            let mut best_idx = 0;
                            let mut best_dist_sq = f32::INFINITY;
                            nearest_neighbor(kd_nodes_ref, root_idx, 0, p, &mut best_idx, &mut best_dist_sq);
                            loss += best_dist_sq as f64;
                            best_idx
                        } else {
                            let mut min_dist = f32::INFINITY;
                            let mut min_idx = 0;
                            for (idx, c) in centroids_ref.iter().enumerate() {
                                let d = sq_dist(p, c);
                                if d < min_dist {
                                    min_dist = d;
                                    min_idx = idx;
                                }
                            }
                            loss += min_dist as f64;
                            min_idx
                        };
                        
                        local_sums[best_idx][0] += p[0] as f64;
                        local_sums[best_idx][1] += p[1] as f64;
                        local_sums[best_idx][2] += p[2] as f64;
                        local_counts[best_idx] += 1;
                    }
                    
                    ThreadResult {
                        local_sums,
                        local_counts,
                        loss,
                    }
                }));
            }
            
            let mut global_sums = vec![[0.0, 0.0, 0.0]; num_clusters];
            let mut global_counts = vec![0; num_clusters];
            let mut total_loss = 0.0;
            
            for handle in handles {
                let res = handle.join().unwrap();
                total_loss += res.loss;
                for i in 0..num_clusters {
                    global_sums[i][0] += res.local_sums[i][0];
                    global_sums[i][1] += res.local_sums[i][1];
                    global_sums[i][2] += res.local_sums[i][2];
                    global_counts[i] += res.local_counts[i];
                }
            }
            (global_sums, global_counts, total_loss)
        });
        
        let (global_sums, global_counts, total_loss) = thread_results;
        
        let mut empty_clusters = Vec::new();
        for i in 0..num_clusters {
            if global_counts[i] == 0 {
                empty_clusters.push(i);
            } else {
                new_centroids[i][0] = (global_sums[i][0] / global_counts[i] as f64) as f32;
                new_centroids[i][1] = (global_sums[i][1] / global_counts[i] as f64) as f32;
                new_centroids[i][2] = (global_sums[i][2] / global_counts[i] as f64) as f32;
            }
        }
        
        if !empty_clusters.is_empty() {
            let mut local_rng = Xorshift64::new(rng.next());
            for &empty_idx in &empty_clusters {
                let r_idx = local_rng.next_range(0, n);
                new_centroids[empty_idx] = points[r_idx];
            }
        }
        
        let delta = (total_loss - prev_loss).abs();
        if delta < 1e-5 * total_loss || total_loss == 0.0 {
            centroids.copy_from_slice(&new_centroids);
            break;
        }
        prev_loss = total_loss;
        centroids.copy_from_slice(&new_centroids);
    }
    
    // Ensure no empty clusters at the very end
    ensure_no_empty_centroids(points, &mut centroids, rng);
    centroids
}

fn run_minibatch_kmeans(
    points: &[[f32; 3]],
    num_clusters: usize,
    max_iters: usize,
    batch_size: usize,
    rng: &mut Xorshift64,
) -> Vec<[f32; 3]> {
    let n = points.len();
    let mut centroids = initialize_centroids(points, num_clusters, rng);
    let mut counts = vec![0; num_clusters];
    
    for _iter in 0..max_iters {
        let use_kd = num_clusters > 64;
        let mut kd_nodes = Vec::new();
        let mut root_idx = -1;
        if use_kd {
            let mut kd_points: Vec<([f32; 3], usize)> = centroids.iter().copied().enumerate().map(|(i, c)| (c, i)).collect();
            root_idx = build_kd_tree_rec(&mut kd_points, 0, &mut kd_nodes);
        }
        
        let mut batch_indices = Vec::with_capacity(batch_size);
        for _ in 0..batch_size {
            batch_indices.push(rng.next_range(0, n));
        }
        
        let num_threads = std::thread::available_parallelism().map(|p| p.get()).unwrap_or(4);
        let chunk_size = (batch_size + num_threads - 1) / num_threads;
        
        struct BatchThreadResult {
            local_sums: Vec<[f64; 3]>,
            local_counts: Vec<usize>,
        }
        
        let centroids_ref = &centroids;
        let kd_nodes_ref = &kd_nodes;
        let points_ref = points;
        let batch_indices_ref = &batch_indices;
        
        let thread_results = std::thread::scope(|s| {
            let mut handles = Vec::new();
            for chunk_indices in batch_indices_ref.chunks(chunk_size) {
                handles.push(s.spawn(move || {
                    let mut local_sums = vec![[0.0, 0.0, 0.0]; num_clusters];
                    let mut local_counts = vec![0; num_clusters];
                    
                    for &idx in chunk_indices {
                        let p = &points_ref[idx];
                        let best_idx = if use_kd {
                            let mut best_idx = 0;
                            let mut best_dist_sq = f32::INFINITY;
                            nearest_neighbor(kd_nodes_ref, root_idx, 0, p, &mut best_idx, &mut best_dist_sq);
                            best_idx
                        } else {
                            let mut min_dist = f32::INFINITY;
                            let mut min_idx = 0;
                            for (idx, c) in centroids_ref.iter().enumerate() {
                                let d = sq_dist(p, c);
                                if d < min_dist {
                                    min_dist = d;
                                    min_idx = idx;
                                }
                            }
                            min_idx
                        };
                        
                        local_sums[best_idx][0] += p[0] as f64;
                        local_sums[best_idx][1] += p[1] as f64;
                        local_sums[best_idx][2] += p[2] as f64;
                        local_counts[best_idx] += 1;
                    }
                    
                    BatchThreadResult {
                        local_sums,
                        local_counts,
                    }
                }));
            }
            
            let mut global_sums = vec![[0.0, 0.0, 0.0]; num_clusters];
            let mut global_counts = vec![0; num_clusters];
            for handle in handles {
                let res = handle.join().unwrap();
                for i in 0..num_clusters {
                    global_sums[i][0] += res.local_sums[i][0];
                    global_sums[i][1] += res.local_sums[i][1];
                    global_sums[i][2] += res.local_sums[i][2];
                    global_counts[i] += res.local_counts[i];
                }
            }
            (global_sums, global_counts)
        });
        
        let (global_sums, global_counts) = thread_results;
        
        for i in 0..num_clusters {
            let c_i = global_counts[i];
            if c_i > 0 {
                counts[i] += c_i;
                let eta = 1.0 / counts[i] as f64;
                centroids[i][0] = (centroids[i][0] as f64 + eta * (global_sums[i][0] - c_i as f64 * centroids[i][0] as f64)) as f32;
                centroids[i][1] = (centroids[i][1] as f64 + eta * (global_sums[i][1] - c_i as f64 * centroids[i][1] as f64)) as f32;
                centroids[i][2] = (centroids[i][2] as f64 + eta * (global_sums[i][2] - c_i as f64 * centroids[i][2] as f64)) as f32;
            }
        }
        
        let mut empty_clusters = Vec::new();
        for i in 0..num_clusters {
            if counts[i] == 0 {
                empty_clusters.push(i);
            }
        }
        
        if !empty_clusters.is_empty() {
            for &empty_idx in &empty_clusters {
                let r_idx = rng.next_range(0, n);
                centroids[empty_idx] = points[r_idx];
                counts[empty_idx] = 1;
            }
        }
    }
    
    // Ensure no empty clusters at the very end
    ensure_no_empty_centroids(points, &mut centroids, rng);
    centroids
}

fn parse_header<R: Read>(reader: &mut R) -> io::Result<(usize, usize)> {
    let mut header = String::new();
    let mut buf = [0u8; 1];
    
    loop {
        reader.read_exact(&mut buf)?;
        let c = buf[0] as char;
        header.push(c);
        if c == '\n' {
            break;
        }
    }
    
    let parts: Vec<&str> = header.split_whitespace().collect();
    if parts.len() < 2 {
        return Err(io::Error::new(io::ErrorKind::InvalidData, "Invalid header format"));
    }
    
    let num_points = parts[0].parse::<usize>().map_err(|e| io::Error::new(io::ErrorKind::InvalidData, e))?;
    let num_clusters = parts[1].parse::<usize>().map_err(|e| io::Error::new(io::ErrorKind::InvalidData, e))?;
    
    Ok((num_points, num_clusters))
}

fn main() -> io::Result<()> {
    let stdin = io::stdin();
    let mut stdin_lock = stdin.lock();
    let mut reader = BufReader::new(&mut stdin_lock);
    
    let (num_points, num_clusters) = parse_header(&mut reader)?;
    
    let mut rng = Xorshift64::new(1337);
    
    // Determine the buffer/training size
    // We train on up to 5,000,000 points.
    let train_limit = 5_000_000;
    let train_size = std::cmp::min(num_points, train_limit);
    
    // Read the training points into memory
    let mut train_points = vec![[0.0f32; 3]; train_size];
    let mut raw_buf = vec![0u8; 12 * train_size];
    reader.read_exact(&mut raw_buf)?;
    
    for i in 0..train_size {
        let offset = i * 12;
        train_points[i][0] = f32::from_le_bytes([raw_buf[offset], raw_buf[offset+1], raw_buf[offset+2], raw_buf[offset+3]]);
        train_points[i][1] = f32::from_le_bytes([raw_buf[offset+4], raw_buf[offset+5], raw_buf[offset+6], raw_buf[offset+7]]);
        train_points[i][2] = f32::from_le_bytes([raw_buf[offset+8], raw_buf[offset+9], raw_buf[offset+10], raw_buf[offset+11]]);
    }
    drop(raw_buf);
    
    // Train centroids
    let centroids = if num_points <= 2_000_000 {
        // Run standard K-Means
        let max_iters = if num_clusters <= 100 { 30 } else { 15 };
        run_kmeans(&train_points, num_clusters, max_iters, &mut rng)
    } else {
        // Run Mini-Batch K-Means
        let (batch_size, max_iters) = if num_clusters <= 100 {
            (8192, 300)
        } else if num_clusters <= 1000 {
            (32768, 300)
        } else {
            (131072, 200)
        };
        run_minibatch_kmeans(&train_points, num_clusters, max_iters, batch_size, &mut rng)
    };
    
    // Prepare for assignment phase
    let use_kd = num_clusters > 64;
    let mut kd_nodes = Vec::new();
    let mut root_idx = -1;
    if use_kd {
        let mut kd_points: Vec<([f32; 3], usize)> = centroids.iter().copied().enumerate().map(|(i, c)| (c, i)).collect();
        root_idx = build_kd_tree_rec(&mut kd_points, 0, &mut kd_nodes);
    }
    
    let num_threads = std::thread::available_parallelism().map(|p| p.get()).unwrap_or(4);
    let centroids_ref = &centroids;
    let kd_nodes_ref = &kd_nodes;

    let stdout = io::stdout();
    let mut stdout_lock = stdout.lock();
    let mut writer = BufWriter::new(&mut stdout_lock);
    
    // 1. Assign and output the training points
    {
        let chunk_size = (train_size + num_threads - 1) / num_threads;
        let train_points_ref = &train_points;
        
        let train_assignments = std::thread::scope(|s| {
            let mut handles = Vec::new();
            for chunk in train_points_ref.chunks(chunk_size) {
                handles.push(s.spawn(move || {
                    let mut local_assigns = Vec::with_capacity(chunk.len());
                    for p in chunk {
                        let best_idx = if use_kd {
                            let mut best_idx = 0;
                            let mut best_dist_sq = f32::INFINITY;
                            nearest_neighbor(kd_nodes_ref, root_idx, 0, p, &mut best_idx, &mut best_dist_sq);
                            best_idx
                        } else {
                            let mut min_dist = f32::INFINITY;
                            let mut min_idx = 0;
                            for (idx, c) in centroids_ref.iter().enumerate() {
                                let d = sq_dist(p, c);
                                if d < min_dist {
                                    min_dist = d;
                                    min_idx = idx;
                                }
                            }
                            min_idx
                        };
                        local_assigns.push(best_idx as u32);
                    }
                    local_assigns
                }));
            }
            
            let mut all_assigns = Vec::with_capacity(train_size);
            for handle in handles {
                all_assigns.extend(handle.join().unwrap());
            }
            all_assigns
        });
        
        // Write training assignments
        let mut out_bytes = vec![0u8; train_size * 4];
        for (i, &c) in train_assignments.iter().enumerate() {
            let bytes = c.to_le_bytes();
            let offset = i * 4;
            out_bytes[offset] = bytes[0];
            out_bytes[offset+1] = bytes[1];
            out_bytes[offset+2] = bytes[2];
            out_bytes[offset+3] = bytes[3];
        }
        writer.write_all(&out_bytes)?;
    }
    
    // 2. Stream and assign the remaining points
    let remaining_points = num_points - train_size;
    if remaining_points > 0 {
        let stream_chunk_size = 131072;
        let mut stream_raw_buf = vec![0u8; stream_chunk_size * 12];
        
        let mut read_ptr = 0;
        while read_ptr < remaining_points {
            let current_chunk_size = std::cmp::min(stream_chunk_size, remaining_points - read_ptr);
            let bytes_to_read = current_chunk_size * 12;
            reader.read_exact(&mut stream_raw_buf[..bytes_to_read])?;
            
            // Parse raw bytes to points
            let mut chunk_points = vec![[0.0f32; 3]; current_chunk_size];
            for i in 0..current_chunk_size {
                let offset = i * 12;
                chunk_points[i][0] = f32::from_le_bytes([stream_raw_buf[offset], stream_raw_buf[offset+1], stream_raw_buf[offset+2], stream_raw_buf[offset+3]]);
                chunk_points[i][1] = f32::from_le_bytes([stream_raw_buf[offset+4], stream_raw_buf[offset+5], stream_raw_buf[offset+6], stream_raw_buf[offset+7]]);
                chunk_points[i][2] = f32::from_le_bytes([stream_raw_buf[offset+8], stream_raw_buf[offset+9], stream_raw_buf[offset+10], stream_raw_buf[offset+11]]);
            }
            
            // Assign in parallel
            let thread_chunk_size = (current_chunk_size + num_threads - 1) / num_threads;
            let chunk_points_ref = &chunk_points;
            
            let chunk_assignments = std::thread::scope(|s| {
                let mut handles = Vec::new();
                for c in chunk_points_ref.chunks(thread_chunk_size) {
                    handles.push(s.spawn(move || {
                        let mut local_assigns = Vec::with_capacity(c.len());
                        for p in c {
                            let best_idx = if use_kd {
                                let mut best_idx = 0;
                                let mut best_dist_sq = f32::INFINITY;
                                nearest_neighbor(kd_nodes_ref, root_idx, 0, p, &mut best_idx, &mut best_dist_sq);
                                best_idx
                            } else {
                                let mut min_dist = f32::INFINITY;
                                let mut min_idx = 0;
                                for (idx, c) in centroids_ref.iter().enumerate() {
                                    let d = sq_dist(p, c);
                                    if d < min_dist {
                                        min_dist = d;
                                        min_idx = idx;
                                    }
                                }
                                min_idx
                            };
                            local_assigns.push(best_idx as u32);
                        }
                        local_assigns
                    }));
                }
                
                let mut all_assigns = Vec::with_capacity(current_chunk_size);
                for handle in handles {
                    all_assigns.extend(handle.join().unwrap());
                }
                all_assigns
            });
            
            // Write chunk assignments
            let mut chunk_out_bytes = vec![0u8; current_chunk_size * 4];
            for (i, &c) in chunk_assignments.iter().enumerate() {
                let bytes = c.to_le_bytes();
                let offset = i * 4;
                chunk_out_bytes[offset] = bytes[0];
                chunk_out_bytes[offset+1] = bytes[1];
                chunk_out_bytes[offset+2] = bytes[2];
                chunk_out_bytes[offset+3] = bytes[3];
            }
            writer.write_all(&chunk_out_bytes)?;
            
            read_ptr += current_chunk_size;
        }
    }
    
    writer.flush()?;
    Ok(())
}
