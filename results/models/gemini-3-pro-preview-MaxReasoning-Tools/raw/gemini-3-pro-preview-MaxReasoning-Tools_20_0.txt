use std::io::{self, Read, Write};

#[derive(Clone)]
struct Point3D {
    x: f64,
    y: f64,
    z: f64,
}

impl Point3D {
    #[inline]
    fn coord(&self, axis: usize) -> f64 {
        match axis {
            0 => self.x,
            1 => self.y,
            2 => self.z,
            _ => unreachable!(),
        }
    }
}

struct KdNode {
    point_idx: usize,
    left: Option<usize>,
    right: Option<usize>,
}

fn build_kdtree(
    indices: &mut [usize],
    points: &[Point3D],
    depth: usize,
    nodes: &mut Vec<KdNode>,
) -> Option<usize> {
    if indices.is_empty() {
        return None;
    }
    let axis = depth % 3;
    let mid = indices.len() / 2;
    
    indices.select_nth_unstable_by(mid, |&a, &b| {
        points[a].coord(axis).partial_cmp(&points[b].coord(axis)).unwrap()
    });
    
    let point_idx = indices[mid];
    let node_idx = nodes.len();
    nodes.push(KdNode {
        point_idx,
        left: None,
        right: None,
    });
    
    let left_child = build_kdtree(&mut indices[..mid], points, depth + 1, nodes);
    let right_child = build_kdtree(&mut indices[mid + 1..], points, depth + 1, nodes);
    
    nodes[node_idx].left = left_child;
    nodes[node_idx].right = right_child;
    
    Some(node_idx)
}

struct HeapEntry {
    dist_sq: f64,
    point_idx: usize,
}

impl Eq for HeapEntry {}
impl PartialEq for HeapEntry {
    fn eq(&self, other: &Self) -> bool {
        self.dist_sq == other.dist_sq
    }
}
impl Ord for HeapEntry {
    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
        self.dist_sq.partial_cmp(&other.dist_sq).unwrap()
    }
}
impl PartialOrd for HeapEntry {
    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
        Some(self.cmp(other))
    }
}

struct MaxHeap {
    data: Vec<HeapEntry>,
    capacity: usize,
}

impl MaxHeap {
    fn new(capacity: usize) -> Self {
        Self {
            data: Vec::with_capacity(capacity + 1),
            capacity,
        }
    }
    
    fn push(&mut self, entry: HeapEntry) {
        if self.data.len() < self.capacity {
            self.data.push(entry);
            self.upheap(self.data.len() - 1);
        } else if entry.dist_sq < self.data[0].dist_sq {
            self.data[0] = entry;
            self.downheap(0);
        }
    }
    
    fn max_dist_sq(&self) -> f64 {
        if self.data.len() < self.capacity {
            f64::INFINITY
        } else {
            self.data[0].dist_sq
        }
    }
    
    fn upheap(&mut self, mut idx: usize) {
        while idx > 0 {
            let parent = (idx - 1) / 2;
            if self.data[idx].dist_sq > self.data[parent].dist_sq {
                self.data.swap(idx, parent);
                idx = parent;
            } else {
                break;
            }
        }
    }
    
    fn downheap(&mut self, mut idx: usize) {
        let len = self.data.len();
        while idx * 2 + 1 < len {
            let mut child = idx * 2 + 1;
            if child + 1 < len && self.data[child + 1].dist_sq > self.data[child].dist_sq {
                child += 1;
            }
            if self.data[idx].dist_sq < self.data[child].dist_sq {
                self.data.swap(idx, child);
                idx = child;
            } else {
                break;
            }
        }
    }
    
    fn into_indices(self) -> Vec<usize> {
        self.data.into_iter().map(|e| e.point_idx).collect()
    }
}

fn knn_search(
    node_idx: Option<usize>,
    target: &Point3D,
    points: &[Point3D],
    nodes: &[KdNode],
    depth: usize,
    heap: &mut MaxHeap,
    exclude_idx: usize,
) {
    let idx = match node_idx {
        Some(i) => i,
        None => return,
    };
    
    let node = &nodes[idx];
    let pt = &points[node.point_idx];
    
    if node.point_idx != exclude_idx {
        let dx = pt.x - target.x;
        let dy = pt.y - target.y;
        let dz = pt.z - target.z;
        let dist_sq = dx * dx + dy * dy + dz * dz;
        heap.push(HeapEntry {
            dist_sq,
            point_idx: node.point_idx,
        });
    }
    
    let axis = depth % 3;
    let diff = target.coord(axis) - pt.coord(axis);
    
    let (near, far) = if diff <= 0.0 {
        (node.left, node.right)
    } else {
        (node.right, node.left)
    };
    
    knn_search(near, target, points, nodes, depth + 1, heap, exclude_idx);
    
    if diff * diff < heap.max_dist_sq() {
        knn_search(far, target, points, nodes, depth + 1, heap, exclude_idx);
    }
}

struct Scanner<'a> {
    bytes: &'a [u8],
    pos: usize,
}

impl<'a> Scanner<'a> {
    fn new(bytes: &'a [u8]) -> Self {
        Self { bytes, pos: 0 }
    }
    
    #[inline]
    fn next_token(&mut self) -> Option<&'a [u8]> {
        while self.pos < self.bytes.len() && self.bytes[self.pos].is_ascii_whitespace() {
            self.pos += 1;
        }
        if self.pos >= self.bytes.len() {
            return None;
        }
        let start = self.pos;
        while self.pos < self.bytes.len() && !self.bytes[self.pos].is_ascii_whitespace() {
            self.pos += 1;
        }
        Some(&self.bytes[start..self.pos])
    }
    
    #[inline]
    fn next_string(&mut self) -> Option<String> {
        let token = self.next_token()?;
        Some(String::from_utf8_lossy(token).into_owned())
    }
    
    #[inline]
    fn next_f64(&mut self) -> Option<f64> {
        let token = self.next_token()?;
        let s = std::str::from_utf8(token).ok()?;
        s.parse::<f64>().ok()
    }
    
    #[inline]
    fn next_usize(&mut self) -> Option<usize> {
        let token = self.next_token()?;
        let s = std::str::from_utf8(token).ok()?;
        s.parse::<usize>().ok()
    }
    
    #[inline]
    fn next_property_value(&mut self) -> Option<f64> {
        let token = self.next_token()?;
        let s = std::str::from_utf8(token).ok()?;
        let s_trimmed = s.trim();
        if s_trimmed.eq_ignore_ascii_case("nan") 
            || s_trimmed.eq_ignore_ascii_case("na") 
            || s_trimmed.eq_ignore_ascii_case("null") 
            || s_trimmed == "*" 
            || s_trimmed == "-" 
        {
            None
        } else if let Ok(val) = s_trimmed.parse::<f64>() {
            if val.is_finite() {
                Some(val)
            } else {
                None
            }
        } else {
            None
        }
    }
}

fn median(vals: &mut [f64]) -> f64 {
    if vals.is_empty() {
        return 0.0;
    }
    let mid = vals.len() / 2;
    vals.select_nth_unstable_by(mid, |a, b| a.partial_cmp(b).unwrap());
    vals[mid]
}

struct Sample {
    hole_idx: usize,
    sample_idx: usize,
    coords: Point3D,
    values: Vec<Option<f64>>,
}

struct Hole {
    id: String,
}

struct Suspect {
    hole_id: String,
    sample_idx: usize,
    property_name: String,
    confidence: f64,
}

fn main() -> io::Result<()> {
    let mut bytes = Vec::new();
    io::stdin().read_to_end(&mut bytes)?;
    
    let mut scanner = Scanner::new(&bytes);
    
    let num_holes = match scanner.next_usize() {
        Some(n) => n,
        None => return Ok(()),
    };
    
    let num_properties = match scanner.next_usize() {
        Some(p) => p,
        None => return Ok(()),
    };
    
    let mut property_names = Vec::with_capacity(num_properties);
    for _ in 0..num_properties {
        if let Some(name) = scanner.next_string() {
            property_names.push(name);
        } else {
            return Ok(());
        }
    }
    
    let mut holes = Vec::with_capacity(num_holes);
    let mut samples = Vec::new();
    
    for hole_idx in 0..num_holes {
        let hole_id = match scanner.next_string() {
            Some(id) => id,
            None => break,
        };
        let start_x = scanner.next_f64().filter(|v| v.is_finite()).unwrap_or(0.0);
        let start_y = scanner.next_f64().filter(|v| v.is_finite()).unwrap_or(0.0);
        let start_z = scanner.next_f64().filter(|v| v.is_finite()).unwrap_or(0.0);
        let dir_x = scanner.next_f64().filter(|v| v.is_finite()).unwrap_or(0.0);
        let dir_y = scanner.next_f64().filter(|v| v.is_finite()).unwrap_or(0.0);
        let dir_z = scanner.next_f64().filter(|v| v.is_finite()).unwrap_or(0.0);
        let _length = scanner.next_f64().filter(|v| v.is_finite()).unwrap_or(0.0);
        let num_samples = scanner.next_usize().unwrap_or(0);
        
        let dir_len = (dir_x*dir_x + dir_y*dir_y + dir_z*dir_z).sqrt();
        let (ux, uy, uz) = if dir_len > 1e-8 {
            (dir_x / dir_len, dir_y / dir_len, dir_z / dir_len)
        } else {
            (0.0, 0.0, 0.0)
        };
        
        holes.push(Hole {
            id: hole_id,
        });
        
        for sample_idx in 0..num_samples {
            let depth = scanner.next_f64().filter(|v| v.is_finite()).unwrap_or(0.0);
            let mut values = Vec::with_capacity(num_properties);
            for _ in 0..num_properties {
                values.push(scanner.next_property_value());
            }
            
            let x = start_x + depth * ux;
            let y = start_y + depth * uy;
            let z = start_z + depth * uz;
            
            samples.push(Sample {
                hole_idx,
                sample_idx,
                coords: Point3D { x, y, z },
                values,
            });
        }
    }
    
    let num_samples = samples.len();
    if num_samples == 0 {
        println!("0");
        return Ok(());
    }
    
    // Compute global stats for each property
    let mut global_medians = vec![0.0; num_properties];
    let mut global_mads = vec![0.0; num_properties];
    let mut global_mins = vec![0.0; num_properties];
    let mut global_maxs = vec![0.0; num_properties];
    let mut property_has_data = vec![false; num_properties];
    
    for p in 0..num_properties {
        let mut p_vals: Vec<f64> = samples
            .iter()
            .filter_map(|s| s.values[p])
            .collect();
            
        if !p_vals.is_empty() {
            property_has_data[p] = true;
            let min_val = p_vals.iter().copied().fold(f64::INFINITY, f64::min);
            let max_val = p_vals.iter().copied().fold(f64::NEG_INFINITY, f64::max);
            global_mins[p] = min_val;
            global_maxs[p] = max_val;
            
            let med = median(&mut p_vals);
            global_medians[p] = med;
            
            let mut diffs: Vec<f64> = p_vals.iter().map(|&v| (v - med).abs()).collect();
            let mad = median(&mut diffs);
            global_mads[p] = mad;
        }
    }
    
    // Choose threshold adaptively based on number of samples
    let z_threshold = if num_samples <= 1000 {
        4.5
    } else if num_samples <= 10000 {
        5.0
    } else if num_samples <= 100000 {
        5.5
    } else if num_samples <= 500000 {
        6.0
    } else {
        7.0
    };
    
    // Build Kd-tree
    let kdtree_coords: Vec<Point3D> = samples.iter().map(|s| s.coords.clone()).collect();
    let mut indices: Vec<usize> = (0..num_samples).collect();
    let mut nodes = Vec::with_capacity(num_samples);
    let root = build_kdtree(&mut indices, &kdtree_coords, 0, &mut nodes);
    
    // Identify suspects
    // Decide whether to run sequentially or parallel based on sample count
    let suspects = if num_samples <= 50000 {
        // Sequential Strategy
        let mut local_suspects = Vec::new();
        let mut neighbor_vals = Vec::with_capacity(8);
        let mut diffs = Vec::with_capacity(8);
        
        for s_idx in 0..num_samples {
            let sample = &samples[s_idx];
            
            let mut has_any_val = false;
            for p in 0..num_properties {
                if sample.values[p].is_some() && property_has_data[p] {
                    has_any_val = true;
                    break;
                }
            }
            if !has_any_val {
                continue;
            }
            
            let mut heap = MaxHeap::new(8);
            knn_search(root, &sample.coords, &kdtree_coords, &nodes, 0, &mut heap, s_idx);
            let neighbors = heap.into_indices();
            
            for p in 0..num_properties {
                if !property_has_data[p] {
                    continue;
                }
                if let Some(val) = sample.values[p] {
                    neighbor_vals.clear();
                    for &n_idx in &neighbors {
                        if let Some(n_val) = samples[n_idx].values[p] {
                            neighbor_vals.push(n_val);
                        }
                    }
                    
                    if neighbor_vals.len() >= 3 {
                        let median_val = median(&mut neighbor_vals);
                        
                        diffs.clear();
                        for &v in &neighbor_vals {
                            diffs.push((v - median_val).abs());
                        }
                        let local_mad = median(&mut diffs);
                        
                        let local_std_est = 1.4826 * local_mad;
                        let global_std_est = 1.4826 * global_mads[p];
                        
                        let scale = local_std_est
                            .max(0.05 * global_std_est)
                            .max(1e-5 * (global_maxs[p] - global_mins[p]).max(1e-5));
                            
                        let mut min_diff = f64::INFINITY;
                        for &n_idx in &neighbors {
                            if let Some(n_val) = samples[n_idx].values[p] {
                                let dx = samples[n_idx].coords.x - sample.coords.x;
                                let dy = samples[n_idx].coords.y - sample.coords.y;
                                let dz = samples[n_idx].coords.z - sample.coords.z;
                                if dx*dx + dy*dy + dz*dz > 1e-8 {
                                    let d = (val - n_val).abs();
                                    if d < min_diff {
                                        min_diff = d;
                                    }
                                }
                            }
                        }
                        if min_diff.is_infinite() {
                            min_diff = (val - median_val).abs();
                        }
                        
                        let z_score = min_diff / scale;
                        
                        if z_score > z_threshold {
                            local_suspects.push(Suspect {
                                hole_id: holes[sample.hole_idx].id.clone(),
                                sample_idx: sample.sample_idx,
                                property_name: property_names[p].clone(),
                                confidence: z_score,
                            });
                        }
                    }
                }
            }
        }
        local_suspects
    } else {
        // Parallel Strategy
        let num_threads = std::thread::available_parallelism()
            .map(|x| x.get())
            .unwrap_or(4);
            
        let mut thread_results = Vec::with_capacity(num_threads);
        for _ in 0..num_threads {
            thread_results.push(Vec::new());
        }
        
        std::thread::scope(|s| {
            let chunk_size = (num_samples + num_threads - 1) / num_threads;
            for (t_idx, chunk_suspects) in thread_results.iter_mut().enumerate() {
                let start_idx = t_idx * chunk_size;
                let end_idx = (start_idx + chunk_size).min(num_samples);
                if start_idx >= end_idx {
                    continue;
                }
                
                let samples_ref = &samples;
                let coords_ref = &kdtree_coords;
                let nodes_ref = &nodes;
                let global_mads_ref = &global_mads;
                let global_mins_ref = &global_mins;
                let global_maxs_ref = &global_maxs;
                let property_has_data_ref = &property_has_data;
                let holes_ref = &holes;
                let property_names_ref = &property_names;
                let root_node = root;
                
                s.spawn(move || {
                    let mut neighbor_vals = Vec::with_capacity(8);
                    let mut diffs = Vec::with_capacity(8);
                    
                    for s_idx in start_idx..end_idx {
                        let sample = &samples_ref[s_idx];
                        
                        let mut has_any_val = false;
                        for p in 0..num_properties {
                            if sample.values[p].is_some() && property_has_data_ref[p] {
                                has_any_val = true;
                                break;
                            }
                        }
                        if !has_any_val {
                            continue;
                        }
                        
                        let mut heap = MaxHeap::new(8);
                        knn_search(root_node, &sample.coords, coords_ref, nodes_ref, 0, &mut heap, s_idx);
                        let neighbors = heap.into_indices();
                        
                        for p in 0..num_properties {
                            if !property_has_data_ref[p] {
                                continue;
                            }
                            if let Some(val) = sample.values[p] {
                                neighbor_vals.clear();
                                for &n_idx in &neighbors {
                                    if let Some(n_val) = samples_ref[n_idx].values[p] {
                                        neighbor_vals.push(n_val);
                                    }
                                }
                                
                                if neighbor_vals.len() >= 3 {
                                    let median_val = median(&mut neighbor_vals);
                                    
                                    diffs.clear();
                                    for &v in &neighbor_vals {
                                        diffs.push((v - median_val).abs());
                                    }
                                    let local_mad = median(&mut diffs);
                                    
                                    let local_std_est = 1.4826 * local_mad;
                                    let global_std_est = 1.4826 * global_mads_ref[p];
                                    
                                    let scale = local_std_est
                                        .max(0.05 * global_std_est)
                                        .max(1e-5 * (global_maxs_ref[p] - global_mins_ref[p]).max(1e-5));
                                        
                                    let mut min_diff = f64::INFINITY;
                                    for &n_idx in &neighbors {
                                        if let Some(n_val) = samples_ref[n_idx].values[p] {
                                            let dx = samples_ref[n_idx].coords.x - sample.coords.x;
                                            let dy = samples_ref[n_idx].coords.y - sample.coords.y;
                                            let dz = samples_ref[n_idx].coords.z - sample.coords.z;
                                            if dx*dx + dy*dy + dz*dz > 1e-8 {
                                                let d = (val - n_val).abs();
                                                if d < min_diff {
                                                    min_diff = d;
                                                }
                                            }
                                        }
                                    }
                                    if min_diff.is_infinite() {
                                        min_diff = (val - median_val).abs();
                                    }
                                    
                                    let z_score = min_diff / scale;
                                    
                                    if z_score > z_threshold {
                                        chunk_suspects.push(Suspect {
                                            hole_id: holes_ref[sample.hole_idx].id.clone(),
                                            sample_idx: sample.sample_idx,
                                            property_name: property_names_ref[p].clone(),
                                            confidence: z_score,
                                        });
                                    }
                                }
                            }
                        }
                    }
                });
            }
        });
        
        let mut suspects = Vec::new();
        for mut chunk_suspects in thread_results {
            suspects.append(&mut chunk_suspects);
        }
        suspects
    };
    
    // Sort suspects by confidence desc
    let mut suspects = suspects;
    suspects.sort_unstable_by(|a, b| b.confidence.partial_cmp(&a.confidence).unwrap());
    
    // Print suspects
    let stdout = io::stdout();
    let mut writer = io::BufWriter::new(stdout.lock());
    
    writeln!(writer, "{}", suspects.len())?;
    for s in suspects {
        writeln!(writer, "{} {} {} {:.4}", s.hole_id, s.sample_idx, s.property_name, s.confidence)?;
    }
    
    Ok(())
}
