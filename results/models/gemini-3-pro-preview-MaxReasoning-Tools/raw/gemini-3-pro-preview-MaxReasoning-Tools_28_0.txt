use std::io::{self, Read, Write, BufWriter};
use std::time::{Instant, Duration};

struct SimpleRng {
    state: u64,
}

impl SimpleRng {
    fn new(seed: u64) -> Self {
        let mut state = seed;
        if state == 0 {
            state = 1;
        }
        Self { state }
    }

    fn next_u64(&mut self) -> u64 {
        let mut x = self.state;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.state = x;
        x
    }

    fn gen_range(&mut self, min: usize, max: usize) -> usize {
        assert!(min < max);
        let range = (max - min) as u64;
        let val = self.next_u64() % range;
        min + val as usize
    }
}

struct BucketQueue {
    prev: Vec<u32>,
    next: Vec<u32>,
    bucket_head: Vec<u32>,
    curr_max_degree: usize,
}

const EMPTY: u32 = u32::MAX;

impl BucketQueue {
    fn new(num_vertices: usize, max_degree: usize) -> Self {
        Self {
            prev: vec![EMPTY; num_vertices],
            next: vec![EMPTY; num_vertices],
            bucket_head: vec![EMPTY; max_degree + 1],
            curr_max_degree: max_degree,
        }
    }

    fn insert(&mut self, u: usize, d: usize) {
        let head = self.bucket_head[d];
        if head != EMPTY {
            self.prev[head as usize] = u as u32;
        }
        self.next[u] = head;
        self.prev[u] = EMPTY;
        self.bucket_head[d] = u as u32;
        if d > self.curr_max_degree {
            self.curr_max_degree = d;
        }
    }

    fn remove(&mut self, u: usize, d: usize) {
        let p = self.prev[u];
        let n = self.next[u];
        if p != EMPTY {
            self.next[p as usize] = n;
        } else {
            self.bucket_head[d] = n;
        }
        if n != EMPTY {
            self.prev[n as usize] = p;
        }
        self.prev[u] = EMPTY;
        self.next[u] = EMPTY;
    }

    fn pop_max(&mut self) -> Option<(usize, usize)> {
        while self.curr_max_degree > 0 && self.bucket_head[self.curr_max_degree] == EMPTY {
            self.curr_max_degree -= 1;
        }
        if self.curr_max_degree == 0 {
            None
        } else {
            let u = self.bucket_head[self.curr_max_degree] as usize;
            let n = self.next[u];
            self.bucket_head[self.curr_max_degree] = n;
            if n != EMPTY {
                self.prev[n as usize] = EMPTY;
            }
            self.prev[u] = EMPTY;
            self.next[u] = EMPTY;
            Some((u, self.curr_max_degree))
        }
    }
}

fn add_to_cover_list(
    u: usize,
    cover_list: &mut Vec<usize>,
    index_in_cover_list: &mut [usize],
) {
    index_in_cover_list[u] = cover_list.len();
    cover_list.push(u);
}

fn remove_from_cover_list(
    u: usize,
    cover_list: &mut Vec<usize>,
    index_in_cover_list: &mut [usize],
) {
    let idx = index_in_cover_list[u];
    let last_u = cover_list[cover_list.len() - 1];
    cover_list[idx] = last_u;
    index_in_cover_list[last_u] = idx;
    cover_list.pop();
}

fn add_to_uncovered_edges(
    edge_idx: usize,
    uncovered_edges: &mut Vec<usize>,
    uncovered_edge_index: &mut [usize],
) {
    uncovered_edge_index[edge_idx] = uncovered_edges.len();
    uncovered_edges.push(edge_idx);
}

fn remove_from_uncovered_edges(
    edge_idx: usize,
    uncovered_edges: &mut Vec<usize>,
    uncovered_edge_index: &mut [usize],
) {
    let idx = uncovered_edge_index[edge_idx];
    let last_edge = uncovered_edges[uncovered_edges.len() - 1];
    uncovered_edges[idx] = last_edge;
    uncovered_edge_index[last_edge] = idx;
    uncovered_edges.pop();
}

fn pick_vertex_to_remove_bms(
    cover_list: &[usize],
    sum_uncovered_weights: &[u32],
    last_toggled: &[usize],
    step: usize,
    k: usize,
    tabu_tenure: usize,
    rng: &mut SimpleRng,
) -> usize {
    let mut best_v = None;
    let mut best_score = i32::MIN;
    let mut best_age = usize::MAX;
    
    let mut best_tabu_v = None;
    let mut best_tabu_score = i32::MIN;
    let mut best_tabu_age = usize::MAX;
    
    let sample_size = k.min(cover_list.len());
    for _ in 0..sample_size {
        let idx = rng.gen_range(0, cover_list.len());
        let v = cover_list[idx];
        
        let score = -(sum_uncovered_weights[v] as i32);
        let age = last_toggled[v];
        let is_tabu = step - age < tabu_tenure;
        
        if is_tabu {
            if best_tabu_v.is_none() || score > best_tabu_score || (score == best_tabu_score && age < best_tabu_age) {
                best_tabu_v = Some(v);
                best_tabu_score = score;
                best_tabu_age = age;
            }
        } else {
            if best_v.is_none() || score > best_score || (score == best_score && age < best_age) {
                best_v = Some(v);
                best_score = score;
                best_age = age;
            }
        }
    }
    
    best_v.or(best_tabu_v).expect("Cover list is not empty")
}

fn run_greedy(
    num_vertices: usize,
    offsets: &[usize],
    adj_edges: &[u32],
) -> Vec<bool> {
    let mut in_cover = vec![false; num_vertices];
    let mut removed = vec![false; num_vertices];
    let mut degree = vec![0u32; num_vertices];
    let mut max_deg = 0;
    for u in 0..num_vertices {
        let d = offsets[u + 1] - offsets[u];
        degree[u] = d as u32;
        if d > max_deg {
            max_deg = d;
        }
    }
    
    let mut queue = BucketQueue::new(num_vertices, max_deg);
    let mut deg1_queue = Vec::new();
    
    for u in 0..num_vertices {
        let d = degree[u] as usize;
        if d == 0 {
            removed[u] = true;
        } else {
            queue.insert(u, d);
            if d == 1 {
                deg1_queue.push(u);
            }
        }
    }
    
    let process_deg1 = |
        deg1_queue: &mut Vec<usize>,
        removed: &mut [bool],
        degree: &mut [u32],
        in_cover: &mut [bool],
        queue: &mut BucketQueue,
    | {
        while let Some(u) = deg1_queue.pop() {
            if removed[u] || degree[u] != 1 {
                continue;
            }
            
            let mut active_neighbor = None;
            for &v in &adj_edges[offsets[u]..offsets[u+1]] {
                if !removed[v as usize] {
                    active_neighbor = Some(v as usize);
                    break;
                }
            }
            
            if let Some(v) = active_neighbor {
                in_cover[v] = true;
                
                queue.remove(v, degree[v] as usize);
                removed[v] = true;
                
                for &w_u32 in &adj_edges[offsets[v]..offsets[v+1]] {
                    let w = w_u32 as usize;
                    if !removed[w] {
                        let old_d = degree[w] as usize;
                        queue.remove(w, old_d);
                        degree[w] -= 1;
                        let new_d = degree[w] as usize;
                        if new_d == 0 {
                            removed[w] = true;
                        } else {
                            queue.insert(w, new_d);
                            if new_d == 1 {
                                deg1_queue.push(w);
                            }
                        }
                    }
                }
            }
        }
    };
    
    process_deg1(&mut deg1_queue, &mut removed, &mut degree, &mut in_cover, &mut queue);
    
    while let Some((u, _)) = queue.pop_max() {
        if removed[u] {
            continue;
        }
        
        in_cover[u] = true;
        removed[u] = true;
        
        for &v_u32 in &adj_edges[offsets[u]..offsets[u+1]] {
            let v = v_u32 as usize;
            if !removed[v] {
                let old_d = degree[v] as usize;
                queue.remove(v, old_d);
                degree[v] -= 1;
                let new_d = degree[v] as usize;
                if new_d == 0 {
                    removed[v] = true;
                } else {
                    queue.insert(v, new_d);
                    if new_d == 1 {
                        deg1_queue.push(v);
                    }
                }
            }
        }
        
        process_deg1(&mut deg1_queue, &mut removed, &mut degree, &mut in_cover, &mut queue);
    }
    
    in_cover
}

fn remove_redundant(
    num_vertices: usize,
    offsets: &[usize],
    adj_edges: &[u32],
    in_cover: &mut [bool],
) {
    let mut cover_vertices: Vec<usize> = (0..num_vertices).filter(|&u| in_cover[u]).collect();
    cover_vertices.sort_by_key(|&u| offsets[u + 1] - offsets[u]);
    
    for u in cover_vertices {
        if !in_cover[u] {
            continue;
        }
        let mut can_remove = true;
        for &v in &adj_edges[offsets[u]..offsets[u+1]] {
            if !in_cover[v as usize] {
                can_remove = false;
                break;
            }
        }
        if can_remove {
            in_cover[u] = false;
        }
    }
}

fn local_search_thread(
    num_vertices: usize,
    offsets: &[usize],
    adj_edges: &[u32],
    vertex_edge_indices: &[u32],
    edges: &[(u32, u32)],
    initial_cover: &[bool],
    seed: u64,
    time_limit: Duration,
) -> Vec<bool> {
    let start_time = Instant::now();
    let mut rng = SimpleRng::new(seed);
    
    let mut in_cover = initial_cover.to_vec();
    let mut cover_list = Vec::with_capacity(num_vertices);
    let mut index_in_cover_list = vec![0; num_vertices];
    for u in 0..num_vertices {
        if in_cover[u] {
            index_in_cover_list[u] = cover_list.len();
            cover_list.push(u);
        }
    }
    
    let num_edges = edges.len();
    let mut edge_weights = vec![1u32; num_edges];
    let mut edge_cover_count = vec![0u8; num_edges];
    let mut uncovered_edges = Vec::with_capacity(num_edges);
    let mut uncovered_edge_index = vec![0; num_edges];
    
    for (edge_idx, &(u, v)) in edges.iter().enumerate() {
        let count = (in_cover[u as usize] as u8) + (in_cover[v as usize] as u8);
        edge_cover_count[edge_idx] = count;
        if count == 0 {
            uncovered_edge_index[edge_idx] = uncovered_edges.len();
            uncovered_edges.push(edge_idx);
        }
    }
    
    let mut sum_uncovered_weights = vec![0; num_vertices];
    for (edge_idx, &(u, v)) in edges.iter().enumerate() {
        let u_in = in_cover[u as usize];
        let v_in = in_cover[v as usize];
        let w = edge_weights[edge_idx];
        if !v_in {
            sum_uncovered_weights[u as usize] += w;
        }
        if !u_in {
            sum_uncovered_weights[v as usize] += w;
        }
    }
    
    let mut best_cover = in_cover.clone();
    let mut best_cover_size = cover_list.len();
    
    let k = 50;
    let tabu_tenure = 15;
    let mut last_toggled = vec![0usize; num_vertices];
    let mut step = 0;
    
    if !cover_list.is_empty() {
        let w = pick_vertex_to_remove_bms(
            &cover_list,
            &sum_uncovered_weights,
            &last_toggled,
            step,
            k,
            tabu_tenure,
            &mut rng,
        );
        
        in_cover[w] = false;
        remove_from_cover_list(w, &mut cover_list, &mut index_in_cover_list);
        last_toggled[w] = step;
        
        for idx in offsets[w]..offsets[w+1] {
            let edge_idx = vertex_edge_indices[idx] as usize;
            let neighbor = adj_edges[idx] as usize;
            
            let old_count = edge_cover_count[edge_idx];
            edge_cover_count[edge_idx] -= 1;
            if old_count == 1 {
                add_to_uncovered_edges(edge_idx, &mut uncovered_edges, &mut uncovered_edge_index);
            }
            
            sum_uncovered_weights[neighbor] += edge_weights[edge_idx];
        }
    }
    
    while start_time.elapsed() < time_limit {
        for _ in 0..1000 {
            step += 1;
            
            if uncovered_edges.is_empty() {
                if cover_list.len() < best_cover_size {
                    best_cover = in_cover.clone();
                    best_cover_size = cover_list.len();
                }
                
                if cover_list.is_empty() {
                    return best_cover;
                }
                let w = pick_vertex_to_remove_bms(
                    &cover_list,
                    &sum_uncovered_weights,
                    &last_toggled,
                    step,
                    k,
                    tabu_tenure,
                    &mut rng,
                );
                
                in_cover[w] = false;
                remove_from_cover_list(w, &mut cover_list, &mut index_in_cover_list);
                last_toggled[w] = step;
                
                for idx in offsets[w]..offsets[w+1] {
                    let edge_idx = vertex_edge_indices[idx] as usize;
                    let neighbor = adj_edges[idx] as usize;
                    
                    let old_count = edge_cover_count[edge_idx];
                    edge_cover_count[edge_idx] -= 1;
                    if old_count == 1 {
                        add_to_uncovered_edges(edge_idx, &mut uncovered_edges, &mut uncovered_edge_index);
                    }
                    
                    sum_uncovered_weights[neighbor] += edge_weights[edge_idx];
                }
                continue;
            }
            
            let rand_idx = rng.gen_range(0, uncovered_edges.len());
            let edge_idx = uncovered_edges[rand_idx];
            let (u_u32, v_u32) = edges[edge_idx];
            let u = u_u32 as usize;
            let v = v_u32 as usize;
            
            let u_score = sum_uncovered_weights[u];
            let v_score = sum_uncovered_weights[v];
            let x = if u_score != v_score {
                if u_score > v_score { u } else { v }
            } else {
                if last_toggled[u] < last_toggled[v] { u } else { v }
            };
            
            in_cover[x] = true;
            add_to_cover_list(x, &mut cover_list, &mut index_in_cover_list);
            last_toggled[x] = step;
            
            for idx in offsets[x]..offsets[x+1] {
                let e_idx = vertex_edge_indices[idx] as usize;
                let neighbor = adj_edges[idx] as usize;
                
                let old_count = edge_cover_count[e_idx];
                edge_cover_count[e_idx] += 1;
                if old_count == 0 {
                    remove_from_uncovered_edges(e_idx, &mut uncovered_edges, &mut uncovered_edge_index);
                }
                
                sum_uncovered_weights[neighbor] -= edge_weights[e_idx];
            }
            
            let y = pick_vertex_to_remove_bms(
                &cover_list,
                &sum_uncovered_weights,
                &last_toggled,
                step,
                k,
                tabu_tenure,
                &mut rng,
            );
            
            in_cover[y] = false;
            remove_from_cover_list(y, &mut cover_list, &mut index_in_cover_list);
            last_toggled[y] = step;
            
            for idx in offsets[y]..offsets[y+1] {
                let e_idx = vertex_edge_indices[idx] as usize;
                let neighbor = adj_edges[idx] as usize;
                
                let old_count = edge_cover_count[e_idx];
                edge_cover_count[e_idx] -= 1;
                if old_count == 1 {
                    add_to_uncovered_edges(e_idx, &mut uncovered_edges, &mut uncovered_edge_index);
                }
                
                sum_uncovered_weights[neighbor] += edge_weights[e_idx];
            }
            
            for &e_idx in &uncovered_edges {
                let (eu, ev) = edges[e_idx];
                edge_weights[e_idx] += 1;
                sum_uncovered_weights[eu as usize] += 1;
                sum_uncovered_weights[ev as usize] += 1;
            }
            
            if step % 10000 == 0 {
                for w in edge_weights.iter_mut() {
                    *w = (*w / 2).max(1);
                }
                sum_uncovered_weights.fill(0);
                for (e_idx, &(eu, ev)) in edges.iter().enumerate() {
                    let u_in = in_cover[eu as usize];
                    let v_in = in_cover[ev as usize];
                    let w = edge_weights[e_idx];
                    if !v_in {
                        sum_uncovered_weights[eu as usize] += w;
                    }
                    if !u_in {
                        sum_uncovered_weights[ev as usize] += w;
                    }
                }
            }
        }
    }
    
    best_cover
}

fn parse_input() -> (usize, usize, Vec<(u32, u32)>) {
    let mut stdin = io::stdin();
    let mut buffer = Vec::new();
    stdin.read_to_end(&mut buffer).unwrap();
    
    let mut cursor = 0;
    let mut next_usize = || -> Option<usize> {
        while cursor < buffer.len() && buffer[cursor] <= b' ' {
            cursor += 1;
        }
        if cursor >= buffer.len() {
            return None;
        }
        let mut val = 0;
        while cursor < buffer.len() && buffer[cursor] > b' ' {
            val = val * 10 + (buffer[cursor] - b'0') as usize;
            cursor += 1;
        }
        Some(val)
    };
    
    let num_vertices = next_usize().unwrap();
    let num_edges = next_usize().unwrap();
    
    let mut edges = Vec::with_capacity(num_edges);
    for _ in 0..num_edges {
        let u = next_usize().unwrap() as u32;
        let v = next_usize().unwrap() as u32;
        edges.push((u, v));
    }
    
    (num_vertices, num_edges, edges)
}

fn main() {
    let global_start = Instant::now();
    let (num_vertices, num_edges, edges) = parse_input();
    
    let mut degree = vec![0; num_vertices];
    for &(u, v) in &edges {
        degree[u as usize] += 1;
        degree[v as usize] += 1;
    }
    
    let mut offsets = vec![0; num_vertices + 1];
    for i in 0..num_vertices {
        offsets[i + 1] = offsets[i] + degree[i];
    }
    
    let mut current_offsets = offsets.clone();
    let mut adj_edges = vec![0; 2 * num_edges];
    let mut vertex_edge_indices = vec![0; 2 * num_edges];
    for (edge_idx, &(u, v)) in edges.iter().enumerate() {
        let u_idx = current_offsets[u as usize];
        adj_edges[u_idx] = v;
        vertex_edge_indices[u_idx] = edge_idx as u32;
        current_offsets[u as usize] += 1;
        
        let v_idx = current_offsets[v as usize];
        adj_edges[v_idx] = u;
        vertex_edge_indices[v_idx] = edge_idx as u32;
        current_offsets[v as usize] += 1;
    }
    
    let mut in_cover = run_greedy(num_vertices, &offsets, &adj_edges);
    remove_redundant(num_vertices, &offsets, &adj_edges, &mut in_cover);
    
    let elapsed = global_start.elapsed();
    let time_budget = Duration::from_millis(3500).saturating_sub(elapsed);
    
    let mut final_cover = if time_budget > Duration::from_millis(100) {
        let num_threads = 16;
        let seed_base = std::time::SystemTime::now()
            .duration_since(std::time::UNIX_EPOCH)
            .unwrap_or(Duration::ZERO)
            .as_nanos() as u64;
            
        let mut thread_covers = vec![in_cover.clone(); num_threads];
        
        std::thread::scope(|s| {
            let mut handles = Vec::new();
            for i in 0..num_threads {
                let seed = seed_base.wrapping_add(i as u64 * 997);
                let offsets_ref = &offsets;
                let adj_ref = &adj_edges;
                let edge_indices_ref = &vertex_edge_indices;
                let edges_ref = &edges;
                let initial_cover_ref = &in_cover;
                
                let handle = s.spawn(move || {
                    local_search_thread(
                        num_vertices,
                        offsets_ref,
                        adj_ref,
                        edge_indices_ref,
                        edges_ref,
                        initial_cover_ref,
                        seed,
                        time_budget,
                    )
                });
                handles.push(handle);
            }
            for (i, handle) in handles.into_iter().enumerate() {
                thread_covers[i] = handle.join().unwrap();
            }
        });
        
        let mut best_idx = 0;
        let mut best_size = usize::MAX;
        for (i, cov) in thread_covers.iter().enumerate() {
            let size = cov.iter().filter(|&&b| b).count();
            if size < best_size {
                best_size = size;
                best_idx = i;
            }
        }
        
        let mut best_cover = thread_covers[best_idx].clone();
        remove_redundant(num_vertices, &offsets, &adj_edges, &mut best_cover);
        best_cover
    } else {
        in_cover.clone()
    };
    
    let mut all_covered = true;
    for &(u, v) in &edges {
        if !final_cover[u as usize] && !final_cover[v as usize] {
            all_covered = false;
            break;
        }
    }
    
    if !all_covered {
        final_cover = in_cover;
    }
    
    // Print the cover
    let mut cover_vertices = Vec::new();
    for u in 0..num_vertices {
        if final_cover[u] {
            cover_vertices.push(u);
        }
    }
    
    let stdout = io::stdout();
    let mut writer = BufWriter::new(stdout.lock());
    
    writeln!(writer, "{}", cover_vertices.len()).unwrap();
    for (i, &v) in cover_vertices.iter().enumerate() {
        if i > 0 {
            write!(writer, " ").unwrap();
        }
        write!(writer, "{}", v).unwrap();
    }
    writeln!(writer).unwrap();
    writer.flush().unwrap();
}
