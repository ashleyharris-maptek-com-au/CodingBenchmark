use std::cmp::Ordering;
use std::collections::BinaryHeap;
use std::io::{self, BufRead, Write};
use std::sync::{Arc, Mutex};
use std::thread;
use std::time::{Duration, Instant};

struct StateUpdate {
    x: f64,
    y: f64,
    vx: f64,
    vy: f64,
    angle: f64,
    angular_vel: f64,
    fuel: f64,
    time: Instant,
}

#[derive(Copy, Clone, Eq, PartialEq)]
struct AStarState {
    cost: u32,
    x: i32,
    y: i32,
}

impl Ord for AStarState {
    fn cmp(&self, other: &Self) -> Ordering {
        other.cost.cmp(&self.cost)
    }
}

impl PartialOrd for AStarState {
    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        Some(self.cmp(other))
    }
}

fn read_metadata() -> Result<(usize, usize, f64, f64, f64, f64, f64, f64, f64), Box<dyn std::error::Error>> {
    let mut stdin = io::stdin().lock();
    let mut line = String::new();
    
    // Read first non-empty line
    while line.trim().is_empty() {
        line.clear();
        stdin.read_line(&mut line)?;
    }
    let parts: Vec<&str> = line.split_whitespace().collect();
    if parts.len() < 5 {
        return Err("Invalid first line".into());
    }
    let width: usize = parts[0].parse()?;
    let height: usize = parts[1].parse()?;
    let gravity: f64 = parts[2].parse()?;
    let max_thrust: f64 = parts[3].parse()?;
    let fuel: f64 = parts[4].parse()?;
    
    // Read second non-empty line
    line.clear();
    while line.trim().is_empty() {
        line.clear();
        stdin.read_line(&mut line)?;
    }
    let parts2: Vec<&str> = line.split_whitespace().collect();
    if parts2.len() < 4 {
        return Err("Invalid second line".into());
    }
    let start_x: f64 = parts2[0].parse()?;
    let start_y: f64 = parts2[1].parse()?;
    let target_x: f64 = parts2[2].parse()?;
    let target_y: f64 = parts2[3].parse()?;
    
    Ok((width, height, gravity, max_thrust, fuel, start_x, start_y, target_x, target_y))
}

fn parse_state_line(line: &str) -> Option<(f64, f64, f64, f64, f64, f64, f64)> {
    let parts: Vec<&str> = line.split_whitespace().collect();
    if parts.len() < 7 {
        return None;
    }
    let x = parts[0].parse().ok()?;
    let y = parts[1].parse().ok()?;
    let vx = parts[2].parse().ok()?;
    let vy = parts[3].parse().ok()?;
    let angle = parts[4].parse().ok()?;
    let angular_vel = parts[5].parse().ok()?;
    let fuel = parts[6].parse().ok()?;
    Some((x, y, vx, vy, angle, angular_vel, fuel))
}

fn get_lookahead_point(
    x: f64,
    y: f64,
    waypoints: &[(f64, f64)],
    lookahead_dist: f64,
) -> (f64, f64) {
    if waypoints.is_empty() {
        return (x, y);
    }
    if waypoints.len() == 1 {
        return waypoints[0];
    }
    
    let mut min_dist_sq = f64::MAX;
    let mut closest_segment_idx = 0;
    let mut closest_t = 0.0;
    
    for i in 0..(waypoints.len() - 1) {
        let a = waypoints[i];
        let b = waypoints[i+1];
        
        let ab_x = b.0 - a.0;
        let ab_y = b.1 - a.1;
        let ab_len_sq = ab_x * ab_x + ab_y * ab_y;
        if ab_len_sq < 1e-9 {
            continue;
        }
        
        let ap_x = x - a.0;
        let ap_y = y - a.1;
        
        let t = ((ap_x * ab_x + ap_y * ab_y) / ab_len_sq).clamp(0.0, 1.0);
        let cp_x = a.0 + t * ab_x;
        let cp_y = a.1 + t * ab_y;
        
        let dist_sq = (x - cp_x).powi(2) + (y - cp_y).powi(2);
        if dist_sq < min_dist_sq {
            min_dist_sq = dist_sq;
            closest_segment_idx = i;
            closest_t = t;
        }
    }
    
    let mut remaining_l = lookahead_dist;
    let mut curr_seg = closest_segment_idx;
    let mut curr_t = closest_t;
    
    loop {
        let a = waypoints[curr_seg];
        let b = waypoints[curr_seg + 1];
        
        let ab_x = b.0 - a.0;
        let ab_y = b.1 - a.1;
        let ab_len = (ab_x * ab_x + ab_y * ab_y).sqrt();
        
        let cp_x = a.0 + curr_t * ab_x;
        let cp_y = a.1 + curr_t * ab_y;
        
        let dist_to_b = (1.0 - curr_t) * ab_len;
        
        if remaining_l <= dist_to_b {
            if ab_len > 1e-9 {
                let ratio = remaining_l / ab_len;
                return (cp_x + ratio * ab_x, cp_y + ratio * ab_y);
            } else {
                return (cp_x, cp_y);
            }
        }
        
        remaining_l -= dist_to_b;
        curr_seg += 1;
        curr_t = 0.0;
        
        if curr_seg >= waypoints.len() - 1 {
            return waypoints[waypoints.len() - 1];
        }
    }
}

fn compute_control(
    x: f64,
    y: f64,
    vx: f64,
    vy: f64,
    angle: f64,
    angular_vel: f64,
    _fuel: f64,
    gravity: f64,
    max_thrust: f64,
    target_x: f64,
    target_y: f64,
    waypoints: &[(f64, f64)],
    is_obstacle_fn: &dyn Fn(i32, i32) -> bool,
) -> (f64, f64) {
    let dist_x = target_x - x;
    let dist_y = target_y - y;
    let dist_to_target = (dist_x * dist_x + dist_y * dist_y).sqrt();
    
    let is_landing = dist_y < 6.0 && dist_x.abs() < 4.0;
    
    let ax_max_hover = if max_thrust > gravity {
        (max_thrust * max_thrust - gravity * gravity).sqrt()
    } else {
        0.5 * max_thrust
    };
    let a_horizontal = (0.5 * ax_max_hover).clamp(1.0, 5.0);
    
    let (vx_ref, vy_ref) = if is_landing {
        let vx_ref = (dist_x * 1.5).clamp(-1.5, 1.5);
        let vy_ref = if dist_y < 1.0 {
            0.4
        } else if dist_y < 3.0 {
            0.8
        } else {
            1.2
        };
        (vx_ref, vy_ref)
    } else {
        let speed = (vx * vx + vy * vy).sqrt();
        let lookahead_dist = (speed * 0.8).clamp(3.0, 15.0);
        let (tx, ty) = get_lookahead_point(x, y, waypoints, lookahead_dist);
        
        let dx = tx - x;
        let dy = ty - y;
        
        let kp_pos = 0.8;
        let mut vx_ref = kp_pos * dx;
        let mut vy_ref = kp_pos * dy;
        
        let max_speed = (2.0 * a_horizontal * dist_to_target).sqrt().clamp(2.0, 12.0);
        let speed_ref = (vx_ref * vx_ref + vy_ref * vy_ref).sqrt();
        if speed_ref > max_speed {
            vx_ref = vx_ref * (max_speed / speed_ref);
            vy_ref = vy_ref * (max_speed / speed_ref);
        }
        (vx_ref, vy_ref)
    };
    
    let kp_vel = 2.0;
    let mut ax_cmd = kp_vel * (vx_ref - vx);
    let mut ay_cmd = kp_vel * (vy_ref - vy);
    
    // Local obstacle avoidance potential field
    let mut avoid_ax = 0.0;
    let mut avoid_ay = 0.0;
    let cx = x.floor() as i32;
    let cy = y.floor() as i32;
    for dy in -2..=2 {
        for dx in -2..=2 {
            let ox = cx + dx;
            let oy = cy + dy;
            if is_obstacle_fn(ox, oy) {
                let obs_x = ox as f64 + 0.5;
                let obs_y = oy as f64 + 0.5;
                let rx = x - obs_x;
                let ry = y - obs_y;
                let dist_sq = rx * rx + ry * ry;
                let dist = dist_sq.sqrt();
                if dist < 1.5 && dist > 0.01 {
                    let force = 4.0 * (1.5 - dist) / dist;
                    avoid_ax += (rx / dist) * force;
                    avoid_ay += (ry / dist) * force;
                }
            }
        }
    }
    
    let avoid_scale = (dist_to_target / 3.0).min(1.0);
    ax_cmd += avoid_ax * avoid_scale;
    ay_cmd += avoid_ay * avoid_scale;
    
    let fx = ax_cmd;
    let fy = (ay_cmd - gravity).min(0.0);
    
    let f_mag = (fx * fx + fy * fy).sqrt();
    let mut thrust_cmd = f_mag / max_thrust;
    
    let desired_angle = fx.atan2(-fy);
    
    let max_angle = if is_landing { 0.15 } else { 0.8 };
    let target_angle = desired_angle.clamp(-max_angle, max_angle);
    
    let mut angle_error = target_angle - angle;
    while angle_error > std::f64::consts::PI {
        angle_error -= 2.0 * std::f64::consts::PI;
    }
    while angle_error < -std::f64::consts::PI {
        angle_error += 2.0 * std::f64::consts::PI;
    }
    
    thrust_cmd = thrust_cmd * angle_error.cos().max(0.0);
    thrust_cmd = thrust_cmd.clamp(0.0, 1.0);
    
    let kp_angle = 8.0;
    let kd_angle = 4.0;
    let mut turn_cmd = kp_angle * angle_error - kd_angle * angular_vel;
    turn_cmd = turn_cmd.clamp(-1.0, 1.0);
    
    (thrust_cmd, turn_cmd)
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let (width, height, gravity, max_thrust, _fuel, start_x, start_y, target_x, target_y) = read_metadata()?;
    
    let bitset_size = (width * height + 63) / 64;
    let mut obstacles = vec![0u64; bitset_size];
    
    let mut stdin = io::stdin().lock();
    let mut row_idx = 0;
    while row_idx < height {
        let mut line = String::new();
        stdin.read_line(&mut line)?;
        let trimmed = line.trim_end_matches(|c| c == '\r' || c == '\n');
        if trimmed.is_empty() && row_idx == 0 {
            continue;
        }
        let mut col_idx = 0;
        for c in trimmed.chars() {
            if c.is_whitespace() {
                continue;
            }
            if col_idx >= width {
                break;
            }
            if c == '#' {
                let idx = row_idx as u64 * width as u64 + col_idx as u64;
                obstacles[(idx / 64) as usize] |= 1 << (idx % 64);
            }
            col_idx += 1;
        }
        if col_idx > 0 {
            row_idx += 1;
        }
    }
    
    let is_obstacle = |x: i32, y: i32| -> bool {
        if x < 0 || x >= width as i32 || y < 0 || y >= height as i32 {
            return true;
        }
        let idx = y as u64 * width as u64 + x as u64;
        (obstacles[(idx / 64) as usize] & (1 << (idx % 64))) != 0
    };
    
    let start_x_cell = start_x.floor() as i32;
    let start_y_cell = start_y.floor() as i32;
    let target_x_cell = target_x.floor() as i32;
    let target_y_cell = target_y.floor() as i32;
    
    let min_x = (start_x_cell.min(target_x_cell) - 300).max(0);
    let max_x = (start_x_cell.max(target_x_cell) + 300).min(width as i32 - 1);
    let min_y = (start_y_cell.min(target_y_cell) - 300).max(0);
    let max_y = (start_y_cell.max(target_y_cell) + 300).min(height as i32 - 1);
    
    let local_width = max_x - min_x + 1;
    let local_height = max_y - min_y + 1;
    let local_size = local_width as usize * local_height as usize;
    let use_flat = local_size <= 16_000_000;
    
    let start_key = (start_y_cell as u64) << 32 | (start_x_cell as u64);
    let target_key = (target_y_cell as u64) << 32 | (target_x_cell as u64);
    
    let mut pq = BinaryHeap::new();
    pq.push(AStarState { cost: 0, x: start_x_cell, y: start_y_cell });
    
    let mut found = false;
    let mut waypoints = Vec::new();
    let mut iterations = 0;
    
    if use_flat {
        let mut g_score = vec![u32::MAX; local_size];
        let mut came_from = vec![usize::MAX; local_size];
        let start_idx = (start_y_cell - min_y) as usize * local_width as usize + (start_x_cell - min_x) as usize;
        g_score[start_idx] = 0;
        
        while let Some(AStarState { cost: current_f, x: cx, y: cy }) = pq.pop() {
            iterations += 1;
            if iterations > 300_000 { break; }
            
            let idx = (cy - min_y) as usize * local_width as usize + (cx - min_x) as usize;
            let current_g = g_score[idx];
            if current_g == u32::MAX { continue; }
            
            let dist = (((target_x_cell - cx) as f64).powi(2) + ((target_y_cell - cy) as f64).powi(2)).sqrt();
            let h = (dist * 1000.0) as u32;
            if current_f > current_g + h { continue; }
            
            if cx == target_x_cell && cy == target_y_cell {
                found = true;
                break;
            }
            
            for dy in -1..=1 {
                for dx in -1..=1 {
                    if dx == 0 && dy == 0 { continue; }
                    let nx = cx + dx;
                    let ny = cy + dy;
                    
                    if nx < min_x || nx > max_x || ny < min_y || ny > max_y { continue; }
                    if is_obstacle(nx, ny) { continue; }
                    
                    if dx != 0 && dy != 0 {
                        if is_obstacle(cx + dx, cy) || is_obstacle(cx, cy + dy) {
                            continue;
                        }
                    }
                    
                    let mut penalty = 0;
                    for sdy in -1..=1 {
                        for sdx in -1..=1 {
                            if sdx == 0 && sdy == 0 { continue; }
                            if is_obstacle(nx + sdx, ny + sdy) {
                                penalty += 1500;
                            }
                        }
                    }
                    
                    let step_cost = if dx == 0 || dy == 0 { 1000 } else { 1414 };
                    let tent_g = current_g + step_cost + penalty;
                    
                    let n_idx = (ny - min_y) as usize * local_width as usize + (nx - min_x) as usize;
                    if tent_g < g_score[n_idx] {
                        g_score[n_idx] = tent_g;
                        came_from[n_idx] = idx;
                        let n_dist = (((target_x_cell - nx) as f64).powi(2) + ((target_y_cell - ny) as f64).powi(2)).sqrt();
                        let n_h = (n_dist * 1000.0) as u32;
                        pq.push(AStarState { cost: tent_g + n_h, x: nx, y: ny });
                    }
                }
            }
        }
        
        if found {
            let mut curr_x = target_x_cell;
            let mut curr_y = target_y_cell;
            while curr_x != start_x_cell || curr_y != start_y_cell {
                waypoints.push((curr_x as f64 + 0.5, curr_y as f64 + 0.5));
                let idx = (curr_y - min_y) as usize * local_width as usize + (curr_x - min_x) as usize;
                let p_idx = came_from[idx];
                if p_idx == usize::MAX { break; }
                curr_x = (p_idx % local_width as usize) as i32 + min_x;
                curr_y = (p_idx / local_width as usize) as i32 + min_y;
            }
            waypoints.push((start_x, start_y));
            waypoints.reverse();
            if !waypoints.is_empty() {
                let last_idx = waypoints.len() - 1;
                waypoints[last_idx] = (target_x, target_y);
            }
        }
    } else {
        let mut g_score = std::collections::HashMap::<u64, u32>::new();
        let mut came_from = std::collections::HashMap::<u64, u64>::new();
        g_score.insert(start_key, 0);
        
        while let Some(AStarState { cost: current_f, x: cx, y: cy }) = pq.pop() {
            iterations += 1;
            if iterations > 300_000 { break; }
            
            let current_key = (cy as u64) << 32 | (cx as u64);
            let current_g = *g_score.get(&current_key).unwrap_or(&u32::MAX);
            if current_g == u32::MAX { continue; }
            
            let dist = (((target_x_cell - cx) as f64).powi(2) + ((target_y_cell - cy) as f64).powi(2)).sqrt();
            let h = (dist * 1000.0) as u32;
            if current_f > current_g + h { continue; }
            
            if cx == target_x_cell && cy == target_y_cell {
                found = true;
                break;
            }
            
            for dy in -1..=1 {
                for dx in -1..=1 {
                    if dx == 0 && dy == 0 { continue; }
                    let nx = cx + dx;
                    let ny = cy + dy;
                    
                    if nx < min_x || nx > max_x || ny < min_y || ny > max_y { continue; }
                    if is_obstacle(nx, ny) { continue; }
                    
                    if dx != 0 && dy != 0 {
                        if is_obstacle(cx + dx, cy) || is_obstacle(cx, cy + dy) {
                            continue;
                        }
                    }
                    
                    let mut penalty = 0;
                    for sdy in -1..=1 {
                        for sdx in -1..=1 {
                            if sdx == 0 && sdy == 0 { continue; }
                            if is_obstacle(nx + sdx, ny + sdy) {
                                penalty += 1500;
                            }
                        }
                    }
                    
                    let step_cost = if dx == 0 || dy == 0 { 1000 } else { 1414 };
                    let tent_g = current_g + step_cost + penalty;
                    
                    let n_key = (ny as u64) << 32 | (nx as u64);
                    let old_g = *g_score.get(&n_key).unwrap_or(&u32::MAX);
                    if tent_g < old_g {
                        g_score.insert(n_key, tent_g);
                        came_from.insert(n_key, current_key);
                        let n_dist = (((target_x_cell - nx) as f64).powi(2) + ((target_y_cell - ny) as f64).powi(2)).sqrt();
                        let n_h = (n_dist * 1000.0) as u32;
                        pq.push(AStarState { cost: tent_g + n_h, x: nx, y: ny });
                    }
                }
            }
        }
        
        if found {
            let mut curr_key = target_key;
            while curr_key != start_key {
                let cx = (curr_key & 0xFFFFFFFF) as i32;
                let cy = (curr_key >> 32) as i32;
                waypoints.push((cx as f64 + 0.5, cy as f64 + 0.5));
                if let Some(&p_key) = came_from.get(&curr_key) {
                    curr_key = p_key;
                } else {
                    break;
                }
            }
            waypoints.push((start_x, start_y));
            waypoints.reverse();
            if !waypoints.is_empty() {
                let last_idx = waypoints.len() - 1;
                waypoints[last_idx] = (target_x, target_y);
            }
        }
    }
    
    if waypoints.is_empty() {
        waypoints = vec![(start_x, start_y), (target_x, target_y)];
    }
    
    let state_lock = Arc::new(Mutex::new(None::<StateUpdate>));
    let state_lock_clone = state_lock.clone();
    
    thread::spawn(move || {
        let stdin = io::stdin();
        let mut reader = io::BufReader::new(stdin);
        let mut line = String::new();
        loop {
            line.clear();
            match reader.read_line(&mut line) {
                Ok(0) => break,
                Ok(_) => {
                    if let Some(parsed) = parse_state_line(&line) {
                        let mut lock = state_lock_clone.lock().unwrap();
                        *lock = Some(StateUpdate {
                            x: parsed.0,
                            y: parsed.1,
                            vx: parsed.2,
                            vy: parsed.3,
                            angle: parsed.4,
                            angular_vel: parsed.5,
                            fuel: parsed.6,
                            time: Instant::now(),
                        });
                    }
                }
                Err(_) => break,
            }
        }
    });
    
    let loop_duration = Duration::from_millis(40);
    
    loop {
        let now = Instant::now();
        let state_opt = {
            let lock = state_lock.lock().unwrap();
            if let Some(ref s) = *lock {
                Some(StateUpdate {
                    x: s.x,
                    y: s.y,
                    vx: s.vx,
                    vy: s.vy,
                    angle: s.angle,
                    angular_vel: s.angular_vel,
                    fuel: s.fuel,
                    time: s.time,
                })
            } else {
                None
            }
        };
        
        if let Some(state) = state_opt {
            if now.duration_since(state.time) > Duration::from_secs(2) {
                break;
            }
            
            let dt = now.duration_since(state.time).as_secs_f64();
            let ext_vx = state.vx;
            let ext_vy = state.vy + gravity * dt;
            let ext_x = state.x + state.vx * dt;
            let ext_y = state.y + state.vy * dt + 0.5 * gravity * dt * dt;
            let ext_angle = state.angle + state.angular_vel * dt;
            let ext_angular_vel = state.angular_vel;
            
            let (thrust_cmd, turn_cmd) = compute_control(
                ext_x, ext_y, ext_vx, ext_vy, ext_angle, ext_angular_vel, state.fuel,
                gravity, max_thrust, target_x, target_y, &waypoints,
                &is_obstacle
            );
            
            println!("{:.4} {:.4}", thrust_cmd, turn_cmd);
            let _ = io::stdout().flush();
        } else {
            println!("0.0 0.0");
            let _ = io::stdout().flush();
        }
        
        let elapsed = now.elapsed();
        if elapsed < loop_duration {
            thread::sleep(loop_duration - elapsed);
        }
    }
    
    Ok(())
}
