using System;
using System.IO;
using System.Text;
using System.Collections.Generic;
using System.Diagnostics;

public class Program {
    // Custom Min-Heap for Dijkstra
    public class MinHeap {
        private long[] data;
        private int size;

        public MinHeap(int capacity) {
            data = new long[capacity];
            size = 0;
        }

        public void Push(int index, int cost) {
            if (size == data.Length) {
                Array.Resize(ref data, data.Length * 2);
            }
            long val = ((long)cost << 32) | (uint)index;
            data[size] = val;
            Up(size++);
        }

        public bool Pop(out int index, out int cost) {
            if (size == 0) {
                index = 0;
                cost = 0;
                return false;
            }
            long val = data[0];
            index = (int)(val & 0xFFFFFFFF);
            cost = (int)(val >> 32);
            data[0] = data[--size];
            Down(0);
            return true;
        }

        public void Clear() {
            size = 0;
        }

        public int Count => size;

        private void Up(int i) {
            while (i > 0) {
                int p = (i - 1) >> 1;
                if (data[i] >= data[p]) break;
                long tmp = data[i]; data[i] = data[p]; data[p] = tmp;
                i = p;
            }
        }

        private void Down(int i) {
            while ((i << 1) + 1 < size) {
                int j = (i << 1) + 1;
                if (j + 1 < size && data[j + 1] < data[j]) j++;
                if (data[i] <= data[j]) break;
                long tmp = data[i]; data[i] = data[j]; data[j] = tmp;
                i = j;
            }
        }
    }

    // Operation representation
    public struct Operation {
        public int N;
        public int DigX;
        public int DigY;
        public int DigZ;
        public List<int> Path;
        public int DestX;
        public int DestY;
    }

    static readonly int[] dx = { 1, -1, 0, 0 };
    static readonly int[] dy = { 0, 0, 1, -1 };

    public static void Main(string[] args) {
        Stopwatch stopwatch = Stopwatch.StartNew();

        // 1. Fast binary header parsing
        Stream stdin = Console.OpenStandardInput();
        List<byte> headerBytes = new List<byte>();
        while (true) {
            int b = stdin.ReadByte();
            if (b == -1) break;
            if (b == '\n') break;
            if (b == '\r') continue;
            headerBytes.Add((byte)b);
        }
        if (headerBytes.Count == 0) return;

        string header = Encoding.ASCII.GetString(headerBytes.ToArray());
        string[] tokens = header.Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
        int X = int.Parse(tokens[0]);
        int Y = int.Parse(tokens[1]);
        int Z = int.Parse(tokens[2]);
        double dxSize = double.Parse(tokens[3]);
        double dySize = double.Parse(tokens[4]);
        double dzSize = double.Parse(tokens[5]);

        // 2. Determine scale and parameters
        int K = 32;
        long totalVoxels = (long)X * Y;
        if (totalVoxels <= 10000) {
            K = 128;
        } else if (totalVoxels <= 1000000) {
            K = 64;
        } else {
            K = 32;
        }

        // Allocate memory structures
        short[] H = new short[X * Y];
        short[] initialH = new short[X * Y];
        byte[] voxelData = new byte[X * Y * K];
        bool[] hasValuable = new bool[X * Y];
        byte[] dugCount = new byte[X * Y];
        bool[] isOnPath = new bool[X * Y];

        // Temp column buffer
        byte[] tempColumn = new byte[Z];

        // Fast reader buffer
        byte[] readBuffer = new byte[65536];
        int bufferLen = 0;
        int bufferPtr = 0;

        int ReadByteFast() {
            if (bufferPtr >= bufferLen) {
                bufferPtr = 0;
                bufferLen = stdin.Read(readBuffer, 0, readBuffer.Length);
                if (bufferLen <= 0) return -1;
            }
            return readBuffer[bufferPtr++];
        }

        // Read all voxels
        for (int x = 0; x < X; x++) {
            for (int y = 0; y < Y; y++) {
                for (int z = 0; z < Z; z++) {
                    int b = ReadByteFast();
                    tempColumn[z] = (byte)b;
                }

                // Find top solid
                int height = -1;
                for (int z = Z - 1; z >= 0; z--) {
                    if ((tempColumn[z] & 15) > 0) { // mass > 0
                        height = z;
                        break;
                    }
                }

                int idx = x * Y + y;
                H[idx] = (short)height;
                initialH[idx] = (short)height;

                if (height >= 0) {
                    int baseIdx = idx * K;
                    bool hasVal = false;
                    for (int i = 0; i < K; i++) {
                        int z = height - i;
                        byte d = 0;
                        if (z >= 0) {
                            d = tempColumn[z];
                        }
                        voxelData[baseIdx + i] = d;
                        if ((d >> 4) > 0) {
                            hasVal = true;
                        }
                    }
                    hasValuable[idx] = hasVal;
                } else {
                    hasValuable[idx] = false;
                }
            }
        }

        // Dijkstra states
        int[] distToPlant = new int[X * Y];
        int[] parentToPlant = new int[X * Y];
        int[] distToDump = new int[X * Y];
        int[] parentToDump = new int[X * Y];
        int[] visitedToPlant = new int[X * Y];
        int visitTokenPlant = 0;
        int[] visitedToDump = new int[X * Y];
        int visitTokenDump = 0;

        MinHeap heap = new MinHeap(1024);
        List<Operation> operations = new List<Operation>();

        // Adaptive search radius R
        int maxDim = Math.Max(X, Y);
        int R = Math.Min(50, maxDim);

        bool needsDijkstra = true;
        int stepsSinceDijkstra = 0;

        // Adaptive update frequency S
        int S = 1;
        int activeAreaSize = (Math.Min(X - 1, R) + 1) * (Math.Min(Y - 1, R) + 1);
        if (activeAreaSize <= 2500) S = 1;
        else if (activeAreaSize <= 40000) S = 5;
        else if (activeAreaSize <= 250000) S = 20;
        else S = 100;

        int max_height_increase = (X * Y < 100) ? 10 : 2;

        while (true) {
            // Time check (4.5 minutes limit)
            if (stopwatch.ElapsedMilliseconds > 270000) {
                break;
            }

            int W_X = Math.Min(X - 1, R);
            int W_Y = Math.Min(Y - 1, R);

            if (needsDijkstra || stepsSinceDijkstra >= S) {
                visitTokenPlant++;
                heap.Clear();
                
                // Backward Dijkstra from (0, 0)
                heap.Push(0, 0);
                distToPlant[0] = 0;
                visitedToPlant[0] = visitTokenPlant;

                while (heap.Count > 0) {
                    heap.Pop(out int u, out int c);
                    if (c > distToPlant[u]) continue;

                    int ux = u / Y;
                    int uy = u % Y;
                    int uh = H[u];

                    for (int i = 0; i < 4; i++) {
                        int nx = ux + dx[i];
                        int ny = uy + dy[i];
                        if (nx >= 0 && nx <= W_X && ny >= 0 && ny <= W_Y) {
                            int v = nx * Y + ny;
                            int vh = H[v];
                            int diff = uh - vh; // forward transition is v -> u
                            if (Math.Abs(diff) <= 1) {
                                int stepCost = (diff == 1) ? 5 : ((diff == -1) ? 0 : 1);
                                int nextCost = c + stepCost;
                                if (visitedToPlant[v] != visitTokenPlant || nextCost < distToPlant[v]) {
                                    distToPlant[v] = nextCost;
                                    parentToPlant[v] = u;
                                    visitedToPlant[v] = visitTokenPlant;
                                    heap.Push(v, nextCost);
                                }
                            }
                        }
                    }
                }
                needsDijkstra = false;
                stepsSinceDijkstra = 0;
            }

            // Find best column
            int bestIdx = -1;
            double bestScore = -1e9;
            int bestVoxelIdx = -1;
            int bestVal = 0;
            int bestMass = 0;
            int bestColNeighbor = -1;
            int bestColTransportCost = 0;

            for (int x = 0; x <= W_X; x++) {
                for (int y = 0; y <= W_Y; y++) {
                    int idx = x * Y + y;
                    if (!hasValuable[idx]) continue;
                    
                    int minTransportCost = 1000000000;
                    int colNeighbor = -1;
                    
                    if (idx == 0) {
                        minTransportCost = 0;
                    } else {
                        int startH = H[idx] - 1;
                        for (int i = 0; i < 4; i++) {
                            int nx = x + dx[i];
                            int ny = y + dy[i];
                            if (nx >= 0 && nx <= W_X && ny >= 0 && ny <= W_Y) {
                                int v = nx * Y + ny;
                                if (visitedToPlant[v] == visitTokenPlant) {
                                    int diff = H[v] - startH;
                                    if (Math.Abs(diff) <= 1) {
                                        int stepCost = (diff == 1) ? 5 : ((diff == -1) ? 0 : 1);
                                        int totalCost = distToPlant[v] + stepCost;
                                        if (totalCost < minTransportCost) {
                                            minTransportCost = totalCost;
                                            colNeighbor = v;
                                        }
                                    }
                                }
                            }
                        }
                    }
                    
                    if (idx != 0 && colNeighbor == -1) continue; // Unreachable from plant

                    int dStart = dugCount[idx];
                    int mJunk = 0;
                    for (int i = dStart; i < K; i++) {
                        byte d = voxelData[idx * K + i];
                        int val = d >> 4;
                        int mass = d & 15;
                        if (val > 0) {
                            // Found valuable
                            double score = mass * (1000 * val - minTransportCost) - mJunk * 2.0;
                            if (score > bestScore) {
                                bestScore = score;
                                bestIdx = idx;
                                bestVoxelIdx = i;
                                bestVal = val;
                                bestMass = mass;
                                bestColNeighbor = colNeighbor;
                                bestColTransportCost = minTransportCost;
                            }
                            break; // Only evaluate the highest valuable voxel
                        } else {
                            mJunk += mass;
                        }
                    }
                }
            }

            if (bestScore <= 0) {
                // No profitable column found in active area. Try to expand R.
                if (R < maxDim) {
                    R += 50;
                    activeAreaSize = (Math.Min(X - 1, R) + 1) * (Math.Min(Y - 1, R) + 1);
                    if (activeAreaSize <= 2500) S = 1;
                    else if (activeAreaSize <= 40000) S = 5;
                    else if (activeAreaSize <= 250000) S = 20;
                    else S = 100;
                    needsDijkstra = true;
                    continue;
                } else {
                    break; // Fully searched, no more profit
                }
            }

            // Execute action for bestIdx
            int sx = bestIdx / Y;
            int sy = bestIdx % Y;
            int targetIdx = bestVoxelIdx;
            int currentDug = dugCount[bestIdx];

            if (currentDug == targetIdx) {
                // Top voxel is valuable! Dig it and move to (0, 0)
                int digZ = H[bestIdx];

                // Reconstruct path
                List<int> path = new List<int>();
                if (bestIdx != 0) {
                    path.Add(bestColNeighbor);
                    int curr = bestColNeighbor;
                    while (curr != 0) {
                        curr = parentToPlant[curr];
                        path.Add(curr);
                    }
                }

                // Dig
                H[bestIdx]--;
                dugCount[bestIdx]++;

                // Update hasValuable
                bool hasVal = false;
                for (int i = dugCount[bestIdx]; i < K; i++) {
                    if ((voxelData[bestIdx * K + i] >> 4) > 0) {
                        hasVal = true;
                        break;
                    }
                }
                hasValuable[bestIdx] = hasVal;

                // Record operation
                operations.Add(new Operation {
                    N = path.Count,
                    DigX = sx,
                    DigY = sy,
                    DigZ = digZ,
                    Path = path,
                    DestX = 0,
                    DestY = 0
                });

                stepsSinceDijkstra++;
            } else {
                // Top voxel is junk! Dig it and dump it nearby
                int digZ = H[bestIdx];

                // Dig
                H[bestIdx]--;
                dugCount[bestIdx]++;

                // Update hasValuable for source
                bool hasVal = false;
                for (int i = dugCount[bestIdx]; i < K; i++) {
                    if ((voxelData[bestIdx * K + i] >> 4) > 0) {
                        hasVal = true;
                        break;
                    }
                }
                hasValuable[bestIdx] = hasVal;

                // Mark path to avoid dumping on it
                isOnPath[bestIdx] = true;
                int markCurr = bestColNeighbor;
                while (markCurr != 0 && markCurr != -1) {
                    isOnPath[markCurr] = true;
                    markCurr = parentToPlant[markCurr];
                }
                isOnPath[0] = true;

                // Run forward Dijkstra to find nearest dump location
                visitTokenDump++;
                heap.Clear();
                heap.Push(bestIdx, 0);
                distToDump[bestIdx] = 0;
                visitedToDump[bestIdx] = visitTokenDump;

                int dumpIdx = -1;
                while (heap.Count > 0) {
                    heap.Pop(out int u, out int c);
                    if (c > distToDump[u]) continue;

                    // Check if valid dump location
                    if (u != bestIdx && u != 0 && !hasValuable[u] && !isOnPath[u] && (H[u] - initialH[u] < max_height_increase)) {
                        dumpIdx = u;
                        break;
                    }

                    int ux = u / Y;
                    int uy = u % Y;
                    int uh = H[u];

                    for (int i = 0; i < 4; i++) {
                        int nx = ux + dx[i];
                        int ny = uy + dy[i];
                        if (nx >= 0 && nx <= W_X && ny >= 0 && ny <= W_Y) {
                            int v = nx * Y + ny;
                            int vh = H[v];
                            int diff = vh - uh; // forward transition is u -> v
                            if (Math.Abs(diff) <= 1) {
                                int stepCost = (diff == 1) ? 5 : ((diff == -1) ? 0 : 1);
                                int nextCost = c + stepCost;
                                if (visitedToDump[v] != visitTokenDump || nextCost < distToDump[v]) {
                                    distToDump[v] = nextCost;
                                    parentToDump[v] = u;
                                    visitedToDump[v] = visitTokenDump;
                                    heap.Push(v, nextCost);
                                }
                            }
                        }
                    }
                }

                // Clear path marks
                isOnPath[bestIdx] = false;
                markCurr = bestColNeighbor;
                while (markCurr != 0 && markCurr != -1) {
                    isOnPath[markCurr] = false;
                    markCurr = parentToPlant[markCurr];
                }
                isOnPath[0] = false;

                if (dumpIdx == -1) {
                    // Failed to find dump location. Undo the dig.
                    H[bestIdx]++;
                    dugCount[bestIdx]--;
                    hasValuable[bestIdx] = true;
                    // Try to expand radius
                    if (R < maxDim) {
                        R += 50;
                        needsDijkstra = true;
                    } else {
                        break; // Stuck
                    }
                    continue;
                }

                // Reconstruct dump path
                List<int> dumpPath = new List<int>();
                int curr = dumpIdx;
                while (curr != bestIdx) {
                    dumpPath.Add(curr);
                    curr = parentToDump[curr];
                }
                dumpPath.Reverse();

                // Raise dump height
                H[dumpIdx]++;

                // Record operation
                operations.Add(new Operation {
                    N = dumpPath.Count,
                    DigX = sx,
                    DigY = sy,
                    DigZ = digZ,
                    Path = dumpPath,
                    DestX = dumpIdx / Y,
                    DestY = dumpIdx % Y
                });

                stepsSinceDijkstra++;
                needsDijkstra = true; // Heightmap changed significantly (dumped junk), re-run plant Dijkstra
            }
        }

        // Print output
        using (var sw = new StreamWriter(Console.OpenStandardOutput(), Encoding.ASCII, 65536)) {
            sw.WriteLine(operations.Count);
            for (int i = 0; i < operations.Count; i++) {
                var op = operations[i];
                sw.Write(op.N);
                sw.Write(' ');
                sw.Write(op.DigX);
                sw.Write(' ');
                sw.Write(op.DigY);
                sw.Write(' ');
                sw.Write(op.DigZ);
                for (int j = 0; j < op.Path.Count; j++) {
                    int idx = op.Path[j];
                    sw.Write(' ');
                    sw.Write(idx / Y);
                    sw.Write(' ');
                    sw.Write(idx % Y);
                }
                if (op.Path.Count == 0) {
                    sw.Write(' ');
                    sw.Write(op.DestX);
                    sw.Write(' ');
                    sw.Write(op.DestY);
                }
                sw.WriteLine();
            }
        }
    }
}
