import sys
import os

# Add parent directory to path so we can import 42
sys.path.append(os.path.abspath(".."))
import importlib
try:
    test42 = importlib.import_module("42")
    print("Imported 42.py successfully")
except Exception as e:
    print("Failed to import 42.py:", e)
    sys.exit(1)

shader_code = """; SPIR-V
; Version: 1.0
; Generator: Antigravity
               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %color %gl_out %outColor
               OpExecutionMode %main InputLines
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 4
               OpExecutionMode %main Invocations 1
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %color Location 2
               OpDecorate %outColor Location 0

       %void = OpTypeVoid
       %func = OpTypeFunction %void
      %float = OpTypeFloat 32
        %int = OpTypeInt 32 1
       %uint = OpTypeInt 32 0
     %v2float = OpTypeVector %float 2
     %v3float = OpTypeVector %float 3
     %v4float = OpTypeVector %float 4

     %uint_2 = OpConstant %uint 2
      %int_0 = OpConstant %int 0
      %int_1 = OpConstant %int 1
    %float_0 = OpConstant %float 0
   %float_03 = OpConstant %float 0.03

%gl_PerVertex_in = OpTypeStruct %v4float
  %arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
     %gl_in = OpVariable %ptr_in_arr_pv Input

  %ptr_in_v4 = OpTypePointer Input %v4float

   %arr_v3_2 = OpTypeArray %v3float %uint_2
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_2
     %color = OpVariable %ptr_in_arr_v3 Input
  %ptr_in_v3 = OpTypePointer Input %v3float

%gl_PerVertex_out = OpTypeStruct %v4float
 %ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
    %gl_out = OpVariable %ptr_out_pv Output
 %ptr_out_v4 = OpTypePointer Output %v4float

 %ptr_out_v3 = OpTypePointer Output %v3float
  %outColor = OpVariable %ptr_out_v3 Output

       %main = OpFunction %void None %func
      %entry = OpLabel

; 1. Read gl_in[0].gl_Position and gl_in[1].gl_Position
  %p0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
    %p0 = OpLoad %v4float %p0_ptr
  %p1_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
    %p1 = OpLoad %v4float %p1_ptr

; 2. Read color[0] and color[1]
  %c0_ptr = OpAccessChain %ptr_in_v3 %color %int_0
    %c0 = OpLoad %v3float %c0_ptr
  %c1_ptr = OpAccessChain %ptr_in_v3 %color %int_1
    %c1 = OpLoad %v3float %c1_ptr

; 3. Compute line direction: dir = p1 - p0
   %dir = OpFSub %v4float %p1 %p0

; 4. Compute perpendicular: perp = normalize(-dir.y, dir.x) * thickness, thickness = 0.03
 ; Extract x and y of dir
 %dir_x = OpCompositeExtract %float %dir 0
 %dir_y = OpCompositeExtract %float %dir 1
; Negate dir.y
 %neg_dir_y = OpFNegate %float %dir_y
; Construct vec2(-dir.y, dir.x)
%perp_dir = OpCompositeConstruct %v2float %neg_dir_y %dir_x
; Normalize perp_dir
%perp_norm = OpExtInst %v2float %1 Normalize %perp_dir
; Scale by thickness (0.03)
%perp_scaled = OpVectorTimesScalar %v2float %perp_norm %float_03

; Extract perp components and make a vec4 with z=0, w=0
%perp_x = OpCompositeExtract %float %perp_scaled 0
%perp_y = OpCompositeExtract %float %perp_scaled 1
%perp_v4 = OpCompositeConstruct %v4float %perp_x %perp_y %float_0 %float_0

; 5. Emit 4 vertices as triangle strip:
;    p0 + perp, p0 - perp, p1 + perp, p1 - perp

; Vertex 0: p0 + perp_v4
  %v0_pos = OpFAdd %v4float %p0 %perp_v4
 %gp_ptr0 = OpAccessChain %ptr_out_v4 %gl_out %int_0
            OpStore %gp_ptr0 %v0_pos
            OpStore %outColor %c0
            OpEmitVertex

; Vertex 1: p0 - perp_v4
  %v1_pos = OpFSub %v4float %p0 %perp_v4
 %gp_ptr1 = OpAccessChain %ptr_out_v4 %gl_out %int_0
            OpStore %gp_ptr1 %v1_pos
            OpStore %outColor %c0
            OpEmitVertex

; Vertex 2: p1 + perp_v4
  %v2_pos = OpFAdd %v4float %p1 %perp_v4
 %gp_ptr2 = OpAccessChain %ptr_out_v4 %gl_out %int_0
            OpStore %gp_ptr2 %v2_pos
            OpStore %outColor %c1
            OpEmitVertex

; Vertex 3: p1 - perp_v4
  %v3_pos = OpFSub %v4float %p1 %perp_v4
 %gp_ptr3 = OpAccessChain %ptr_out_v4 %gl_out %int_0
            OpStore %gp_ptr3 %v3_pos
            OpStore %outColor %c1
            OpEmitVertex

            OpEndPrimitive
            OpReturn
            OpFunctionEnd
"""

res = test42.gradeAnswer({"spirv_code": shader_code}, 3, "test_run")
print("Grade result:", res)
