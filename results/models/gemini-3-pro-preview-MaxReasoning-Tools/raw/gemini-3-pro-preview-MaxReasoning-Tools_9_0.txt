using System;
using System.Diagnostics;
using System.Collections.Generic;
using System.IO;
using System.Text;

class Solver {
    struct DPItem {
        public int OriginalIndex;
        public int Multiplier;
        public int Weight;
    }

    class FastScanner {
        private Stream stream;
        private byte[] buffer = new byte[1024 * 1024];
        private int head = 0;
        private int tail = 0;

        public FastScanner() {
            this.stream = Console.OpenStandardInput();
        }

        private byte Read() {
            if (head >= tail) {
                head = 0;
                tail = stream.Read(buffer, 0, buffer.Length);
                if (tail <= 0) return 0;
            }
            return buffer[head++];
        }

        public double NextDouble() {
            byte b = Read();
            while (b > 0 && b <= 32) b = Read();
            if (b == 0) return 0;
            bool neg = false;
            if (b == '-') {
                neg = true;
                b = Read();
            }
            double val = 0;
            while (b >= '0' && b <= '9') {
                val = val * 10 + (b - '0');
                b = Read();
            }
            if (b == '.') {
                b = Read();
                double div = 10;
                while (b >= '0' && b <= '9') {
                    val += (b - '0') / div;
                    div *= 10;
                    b = Read();
                }
            }
            return neg ? -val : val;
        }

        public int NextInt() {
            byte b = Read();
            while (b > 0 && b <= 32) b = Read();
            if (b == 0) return 0;
            bool neg = false;
            if (b == '-') {
                neg = true;
                b = Read();
            }
            int val = 0;
            while (b >= '0' && b <= '9') {
                val = val * 10 + (b - '0');
                b = Read();
            }
            return neg ? -val : val;
        }
    }

    static void Main() {
        FastScanner scanner = new FastScanner();
        int N = scanner.NextInt();
        if (N <= 0) {
            Console.WriteLine(0);
            return;
        }
        double stockLength = scanner.NextDouble();
        double[] originalLengths = new double[N];
        for (int i = 0; i < N; i++) {
            originalLengths[i] = scanner.NextDouble();
        }

        // Try to scale to integers
        int bestFactor = -1;
        int[] factors = { 1, 10, 100, 1000, 10000 };
        foreach (int f in factors) {
            double scaledStock = stockLength * f;
            if (Math.Abs(scaledStock - Math.Round(scaledStock)) > 1e-7) continue;
            
            bool ok = true;
            foreach (double len in originalLengths) {
                double scaledLen = len * f;
                if (Math.Abs(scaledLen - Math.Round(scaledLen)) > 1e-7) {
                    ok = false;
                    break;
                }
            }
            if (ok) {
                bestFactor = f;
                break;
            }
        }

        List<List<int>> packedBins = null;

        if (bestFactor != -1) {
            long scaledStockLength = (long)Math.Round(stockLength * bestFactor);
            if (scaledStockLength <= 2000000) {
                // Group cuts
                Dictionary<long, List<int>> groups = new Dictionary<long, List<int>>();
                for (int i = 0; i < N; i++) {
                    long scaledLen = (long)Math.Round(originalLengths[i] * bestFactor);
                    if (!groups.ContainsKey(scaledLen)) {
                        groups[scaledLen] = new List<int>();
                    }
                    groups[scaledLen].Add(i);
                }

                int K = groups.Count;
                if (K <= 2000) {
                    // We can try GPP
                    long[] uniqueLengths = new long[K];
                    int[] counts = new int[K];
                    List<int>[] indicesOfLength = new List<int>[K];
                    int idx = 0;
                    foreach (var kvp in groups) {
                        uniqueLengths[idx] = kvp.Key;
                        counts[idx] = kvp.Value.Count;
                        indicesOfLength[idx] = kvp.Value;
                        idx++;
                    }

                    packedBins = SolveGPP(uniqueLengths, counts, indicesOfLength, (int)scaledStockLength, out bool success);
                    if (!success) {
                        // If GPP failed (e.g. timeout or complexity limit), packedBins will be null
                        packedBins = null;
                    }
                }
            }
        }

        if (packedBins == null) {
            // Fallback to FFD
            packedBins = SolveFFD(originalLengths, stockLength);
        }

        // Output results
        using (var sw = new StreamWriter(Console.OpenStandardOutput(), new UTF8Encoding(false), 1024 * 1024)) {
            sw.WriteLine(packedBins.Count);
            for (int i = 0; i < packedBins.Count; i++) {
                var bin = packedBins[i];
                for (int j = 0; j < bin.Count; j++) {
                    sw.Write(bin[j]);
                    if (j < bin.Count - 1) sw.Write(' ');
                }
                sw.WriteLine();
            }
        }
    }

    static List<List<int>> SolveGPP(long[] uniqueLengths, int[] counts, List<int>[] indicesOfLength, int stockLength, out bool overallSuccess) {
        overallSuccess = true;
        int K = uniqueLengths.Length;
        List<List<int>> packedBins = new List<List<int>>();
        int[] currentIdx = new int[K];

        int[] dp = new int[stockLength + 1];
        int[] parent = new int[stockLength + 1];
        List<DPItem> dpItems = new List<DPItem>();

        int dpRuns = 0;
        long totalComplexityAccumulated = 0;

        while (true) {
            bool anyRemaining = false;
            for (int i = 0; i < K; i++) {
                if (counts[i] > 0) {
                    anyRemaining = true;
                    break;
                }
            }
            if (!anyRemaining) break;

            // Generate DP items
            dpItems.Clear();
            for (int i = 0; i < K; i++) {
                if (counts[i] <= 0) continue;
                long maxNeeded = stockLength / uniqueLengths[i];
                int limit = (int)Math.Min(counts[i], maxNeeded);

                int remaining = limit;
                int mult = 1;
                while (remaining > 0) {
                    int take = Math.Min(mult, remaining);
                    dpItems.Add(new DPItem {
                        OriginalIndex = i,
                        Multiplier = take,
                        Weight = take * (int)uniqueLengths[i]
                    });
                    remaining -= take;
                    mult *= 2;
                }
            }

            // Check complexity limits
            long runComplexity = (long)stockLength * dpItems.Count;
            totalComplexityAccumulated += runComplexity;
            // If accumulated complexity is too high, or too many runs, fallback to FFD for remaining
            if (totalComplexityAccumulated > 150000000 || dpRuns > 10000) {
                overallSuccess = false;
                break;
            }

            // Run DP
            for (int i = 0; i <= stockLength; i++) {
                dp[i] = -1;
                parent[i] = -1;
            }
            dp[0] = 0;

            int maxWeightReached = 0;
            for (int j = 0; j < dpItems.Count; j++) {
                int itemWeight = dpItems[j].Weight;
                int limit = Math.Min(stockLength, maxWeightReached + itemWeight);
                for (int w = limit; w >= itemWeight; w--) {
                    int prev = dp[w - itemWeight];
                    if (prev >= 0) {
                        int newVal = prev + itemWeight;
                        if (newVal > dp[w]) {
                            dp[w] = newVal;
                            parent[w] = j;
                        }
                    }
                }
                maxWeightReached = Math.Min(stockLength, maxWeightReached + itemWeight);
            }

            int bestW = 0;
            for (int w = stockLength; w >= 0; w--) {
                if (dp[w] > dp[bestW]) {
                    bestW = w;
                }
            }

            if (bestW == 0) {
                // Should not happen, fallback
                overallSuccess = false;
                break;
            }

            // Reconstruct pattern
            int[] patternCounts = new int[K];
            int currW = bestW;
            while (currW > 0) {
                int itemIdx = parent[currW];
                var item = dpItems[itemIdx];
                patternCounts[item.OriginalIndex] += item.Multiplier;
                currW -= item.Weight;
            }

            // Find max application of this pattern
            int maxApply = int.MaxValue;
            for (int i = 0; i < K; i++) {
                if (patternCounts[i] > 0) {
                    maxApply = Math.Min(maxApply, counts[i] / patternCounts[i]);
                }
            }

            if (maxApply == 0 || maxApply == int.MaxValue) {
                overallSuccess = false;
                break;
            }

            // Pack maxApply bins using this pattern
            for (int b = 0; b < maxApply; b++) {
                List<int> binCuts = new List<int>();
                for (int i = 0; i < K; i++) {
                    for (int c = 0; c < patternCounts[i]; c++) {
                        int cutOrigIdx = indicesOfLength[i][currentIdx[i]++];
                        binCuts.Add(cutOrigIdx);
                    }
                }
                packedBins.Add(binCuts);
            }

            // Update remaining counts
            for (int i = 0; i < K; i++) {
                counts[i] -= maxApply * patternCounts[i];
            }

            dpRuns++;
        }

        if (!overallSuccess) {
            return null;
        }

        return packedBins;
    }

    static List<List<int>> SolveFFD(double[] originalLengths, double stockLength) {
        int N = originalLengths.Length;
        double[] lengths = (double[])originalLengths.Clone();
        int[] indices = new int[N];
        for (int i = 0; i < N; i++) indices[i] = i;

        // Sort descending using negation trick
        for (int i = 0; i < N; i++) lengths[i] = -lengths[i];
        Array.Sort(lengths, indices);
        for (int i = 0; i < N; i++) lengths[i] = -lengths[i];

        int M = 1;
        while (M < N) M <<= 1;
        double[] tree = new double[2 * M];
        for (int i = 0; i < N; i++) {
            tree[M + i] = stockLength;
        }
        for (int i = M - 1; i >= 1; i--) {
            tree[i] = Math.Max(tree[2 * i], tree[2 * i + 1]);
        }

        int[] assignedBin = new int[N];
        int maxBinIdx = 0;

        for (int i = 0; i < N; i++) {
            double w = lengths[i];
            int binIdx = FindFirst(w, M, tree);
            if (binIdx == -1) {
                binIdx = maxBinIdx + 1;
            }
            assignedBin[indices[i]] = binIdx;
            if (binIdx > maxBinIdx) maxBinIdx = binIdx;
            
            double updatedCap = tree[M + binIdx] - w;
            if (updatedCap < 0) updatedCap = 0;
            Update(binIdx, updatedCap, M, tree);
        }

        int numBins = maxBinIdx + 1;
        List<List<int>> bins = new List<List<int>>(numBins);
        for (int i = 0; i < numBins; i++) {
            bins.Add(new List<int>());
        }
        for (int i = 0; i < N; i++) {
            bins[assignedBin[i]].Add(i);
        }
        return bins;
    }

    static int FindFirst(double w, int M, double[] tree) {
        if (tree[1] + 1e-9 < w) return -1;
        int idx = 1;
        while (idx < M) {
            if (tree[2 * idx] + 1e-9 >= w) {
                idx = 2 * idx;
            } else {
                idx = 2 * idx + 1;
            }
        }
        return idx - M;
    }

    static void Update(int binIdx, double newVal, int M, double[] tree) {
        int idx = M + binIdx;
        tree[idx] = newVal;
        int parent = idx / 2;
        while (parent >= 1) {
            double leftVal = tree[2 * parent];
            double rightVal = tree[2 * parent + 1];
            double maxVal = leftVal > rightVal ? leftVal : rightVal;
            if (tree[parent] == maxVal) {
                break;
            }
            tree[parent] = maxVal;
            parent /= 2;
        }
    }
}
