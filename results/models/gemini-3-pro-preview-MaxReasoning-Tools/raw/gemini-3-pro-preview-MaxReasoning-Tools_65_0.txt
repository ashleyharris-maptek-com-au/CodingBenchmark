import * as fs from 'fs';

interface TaskInput {
    id: string | number;
    duration: number;
    deps: (string | number)[];
    deadline: number;
    weight: number;
}

interface InputData {
    machineCount: number;
    tasks: TaskInput[];
}

interface ScheduleEntry {
    id: string | number;
    start: number;
    machine: number;
}

interface OutputSchedule {
    schedule: ScheduleEntry[];
}

// Helper: Min-Heap implementation for priority queue
class MinHeap<T> {
    private data: T[] = [];
    constructor(private compare: (a: T, b: T) => number) {}

    push(val: T) {
        this.data.push(val);
        this.up(this.data.length - 1);
    }

    pop(): T | undefined {
        if (this.data.length === 0) return undefined;
        const top = this.data[0];
        const bottom = this.data.pop()!;
        if (this.data.length > 0) {
            this.data[0] = bottom;
            this.down(0);
        }
        return top;
    }

    peek(): T | undefined {
        return this.data[0];
    }

    get length(): number {
        return this.data.length;
    }

    private up(i: number) {
        while (i > 0) {
            const p = (i - 1) >> 1;
            if (this.compare(this.data[i], this.data[p]) < 0) {
                const tmp = this.data[i];
                this.data[i] = this.data[p];
                this.data[p] = tmp;
                i = p;
            } else {
                break;
            }
        }
    }

    private down(i: number) {
        const len = this.data.length;
        while ((i << 1) + 1 < len) {
            let child = (i << 1) + 1;
            if (child + 1 < len && this.compare(this.data[child + 1], this.data[child]) < 0) {
                child++;
            }
            if (this.compare(this.data[child], this.data[i]) < 0) {
                const tmp = this.data[i];
                this.data[i] = this.data[child];
                this.data[child] = tmp;
                i = child;
            } else {
                break;
            }
        }
    }
}

// Decode a priority list of tasks into a valid schedule using Parallel SGS
function decodeSchedule(
    data: InputData,
    priorityMap: Map<string | number, number>
): { schedule: ScheduleEntry[]; makespan: number; weightedLateness: number; totalCost: number } {
    const N = data.tasks.length;
    const taskMap = new Map<string | number, TaskInput>();
    const successors = new Map<string | number, (string | number)[]>();
    const inDegree = new Map<string | number, number>();

    for (const task of data.tasks) {
        taskMap.set(task.id, task);
        successors.set(task.id, []);
        inDegree.set(task.id, 0);
    }

    for (const task of data.tasks) {
        for (const dep of task.deps) {
            successors.get(dep)!.push(task.id);
            inDegree.set(task.id, inDegree.get(task.id)! + 1);
        }
    }

    // Eligible heap: sorted by priority (higher priority first -> lower priority value in min heap)
    // We want to prioritize tasks with HIGHER priority numbers, so we compare (b, a)
    const eligible = new MinHeap<string | number>((a, b) => {
        const pA = priorityMap.get(a) ?? 0;
        const pB = priorityMap.get(b) ?? 0;
        return pB - pA; // Descending order of priority
    });

    for (const task of data.tasks) {
        if (inDegree.get(task.id) === 0) {
            eligible.push(task.id);
        }
    }

    // Events heap: sorted by time (ascending)
    interface FinishEvent {
        time: number;
        taskId: string | number;
        machineId: number;
    }
    const events = new MinHeap<FinishEvent>((a, b) => a.time - b.time);

    const freeMachines: number[] = [];
    for (let m = 0; m < data.machineCount; m++) {
        freeMachines.push(m);
    }

    const schedule: ScheduleEntry[] = [];
    let clock = 0;

    while (schedule.length < N) {
        // Process events at the current clock
        while (events.length > 0 && events.peek()!.time === clock) {
            const ev = events.pop()!;
            freeMachines.push(ev.machineId);
            for (const succ of successors.get(ev.taskId)!) {
                const newDeg = inDegree.get(succ)! - 1;
                inDegree.set(succ, newDeg);
                if (newDeg === 0) {
                    eligible.push(succ);
                }
            }
        }

        // Schedule as many tasks as possible
        while (freeMachines.length > 0 && eligible.length > 0) {
            const taskId = eligible.pop()!;
            const machineId = freeMachines.pop()!;
            const task = taskMap.get(taskId)!;

            schedule.push({
                id: taskId,
                start: clock,
                machine: machineId
            });

            events.push({
                time: clock + task.duration,
                taskId: taskId,
                machineId: machineId
            });
        }

        // Advance clock
        if (schedule.length < N) {
            if (events.length > 0) {
                clock = events.peek()!.time;
            } else {
                // If there are no events but we haven't scheduled all tasks, there is a cycle (invalid input)
                throw new Error("Cycle detected in dependencies!");
            }
        }
    }

    // Compute costs
    let makespan = 0;
    let weightedLateness = 0;
    for (const entry of schedule) {
        const task = taskMap.get(entry.id)!;
        const finish = entry.start + task.duration;
        if (finish > makespan) {
            makespan = finish;
        }
        const lateness = Math.max(0, finish - task.deadline);
        weightedLateness += task.weight * lateness;
    }

    return {
        schedule,
        makespan,
        weightedLateness,
        totalCost: makespan + weightedLateness
    };
}

// Generate some random scheduling test data
function generateTestData(numTasks: number, machineCount: number): InputData {
    const tasks: TaskInput[] = [];
    for (let i = 0; i < numTasks; i++) {
        const duration = Math.floor(Math.random() * 20) + 1;
        const deadline = Math.floor(Math.random() * 100) + 10;
        const weight = Math.floor(Math.random() * 5) + 1;
        tasks.push({
            id: `T${i}`,
            duration,
            deps: [],
            deadline,
            weight
        });
    }

    // Add some random dependencies (DAG)
    for (let i = 0; i < numTasks; i++) {
        const numDeps = Math.floor(Math.random() * 3);
        for (let d = 0; d < numDeps; d++) {
            const depIdx = Math.floor(Math.random() * i); // Only depend on previous tasks to guarantee DAG
            if (depIdx < i && !tasks[i].deps.includes(tasks[depIdx].id)) {
                tasks[i].deps.push(tasks[depIdx].id);
            }
        }
    }

    return {
        machineCount,
        tasks
    };
}

// Run the test
function run() {
    const data = generateTestData(100, 3);
    console.log(`Generated ${data.tasks.length} tasks with ${data.machineCount} machines.`);

    // 1. Uniform Priority Heuristic
    const priorityUniform = new Map<string | number, number>();
    for (const t of data.tasks) {
        priorityUniform.set(t.id, 0);
    }
    const resUniform = decodeSchedule(data, priorityUniform);
    console.log(`Uniform cost: ${resUniform.totalCost} (Makespan: ${resUniform.makespan}, Weighted Lateness: ${resUniform.weightedLateness})`);

    // 2. Earliest Deadline First Heuristic
    const priorityEDF = new Map<string | number, number>();
    for (const t of data.tasks) {
        priorityEDF.set(t.id, -t.deadline);
    }
    const resEDF = decodeSchedule(data, priorityEDF);
    console.log(`EDF cost: ${resEDF.totalCost} (Makespan: ${resEDF.makespan}, Weighted Lateness: ${resEDF.weightedLateness})`);
}

run();
