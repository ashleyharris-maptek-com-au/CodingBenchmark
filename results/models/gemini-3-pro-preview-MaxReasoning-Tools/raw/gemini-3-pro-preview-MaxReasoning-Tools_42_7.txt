; SPIR-V
; Version: 1.0
; Generator: hand-written
               OpCapability Shader
               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %inWorldPos %inNormal %inColor %gl_out %outColor
               OpExecutionMode %main Triangles
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 3
               OpExecutionMode %main Invocations 1

; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %inWorldPos Location 0
               OpDecorate %inNormal Location 1
               OpDecorate %inColor Location 2
               OpDecorate %outColor Location 0

; Types
       %void = OpTypeVoid
       %func = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v3float = OpTypeVector %float 3
    %v4float = OpTypeVector %float 4
       %uint = OpTypeInt 32 0
     %uint_3 = OpConstant %uint 3

; Input types
%gl_PerVertex_in = OpTypeStruct %v4float
  %arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_3
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
     %gl_in = OpVariable %ptr_in_arr_pv Input

  %arr_v3_in = OpTypeArray %v3float %uint_3
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
 %inWorldPos = OpVariable %ptr_in_arr_v3 Input
   %inNormal = OpVariable %ptr_in_arr_v3 Input
    %inColor = OpVariable %ptr_in_arr_v3 Input

; Output types
%gl_PerVertex_out = OpTypeStruct %v4float
 %ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
     %gl_out = OpVariable %ptr_out_pv Output

 %ptr_out_v3 = OpTypePointer Output %v3float
   %outColor = OpVariable %ptr_out_v3 Output

; Pointers for access
        %int = OpTypeInt 32 1
      %int_0 = OpConstant %int 0
      %int_1 = OpConstant %int 1
      %int_2 = OpConstant %int 2

  %ptr_in_v4 = OpTypePointer Input %v4float
  %ptr_in_v3 = OpTypePointer Input %v3float
 %ptr_out_v4 = OpTypePointer Output %v4float

; Constants
%float_0_08 = OpConstant %float 0.08
    %float_0 = OpConstant %float 0.0
%float_one_third = OpConstant %float 0.33333334

       %main = OpFunction %void None %func
      %entry = OpLabel

; 1. Read the 3 vertex positions from gl_in[0..2].gl_Position
    %ptr_p0 = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
    %ptr_p1 = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
    %ptr_p2 = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
        %p0 = OpLoad %v4float %ptr_p0
        %p1 = OpLoad %v4float %ptr_p1
        %p2 = OpLoad %v4float %ptr_p2

; 2. Compute face center (not used for normal, but computed to be compliant)
      %sum0 = OpFAdd %v4float %p0 %p1
       %sum = OpFAdd %v4float %sum0 %p2
%vec_one_third = OpCompositeConstruct %v4float %float_one_third %float_one_third %float_one_third %float_one_third
    %center = OpFMul %v4float %sum %vec_one_third

; Extract xyz for normal computation
    %p0_xyz = OpVectorShuffle %v3float %p0 %p0 0 1 2
    %p1_xyz = OpVectorShuffle %v3float %p1 %p1 0 1 2
    %p2_xyz = OpVectorShuffle %v3float %p2 %p2 0 1 2

; 3. Compute face normal using cross product: e1 = p1-p0, e2 = p2-p0, N = normalize(cross(e1, e2))
       %e1  = OpFSub %v3float %p1_xyz %p0_xyz
       %e2  = OpFSub %v3float %p2_xyz %p0_xyz
%cross_val  = OpExtInst %v3float %1 Cross %e1 %e2
        %N  = OpExtInst %v3float %1 Normalize %cross_val

; 4. Offset = N * 0.08 (push outward)
   %offset  = OpVectorTimesScalar %v3float %N %float_0_08

; Construct offset_v4 = vec4(offset, 0.0)
    %off_x  = OpCompositeExtract %float %offset 0
    %off_y  = OpCompositeExtract %float %offset 1
    %off_z  = OpCompositeExtract %float %offset 2
%offset_v4  = OpCompositeConstruct %v4float %off_x %off_y %off_z %float_0

; 5. Emit 3 vertices, each shifted by offset: gl_Position = pI + vec4(offset, 0)
%p0_shifted = OpFAdd %v4float %p0 %offset_v4
%p1_shifted = OpFAdd %v4float %p1 %offset_v4
%p2_shifted = OpFAdd %v4float %p2 %offset_v4

; 6. Color each vertex with its original vertex color
   %ptr_c0  = OpAccessChain %ptr_in_v3 %inColor %int_0
   %ptr_c1  = OpAccessChain %ptr_in_v3 %inColor %int_1
   %ptr_c2  = OpAccessChain %ptr_in_v3 %inColor %int_2
       %c0  = OpLoad %v3float %ptr_c0
       %c1  = OpLoad %v3float %ptr_c1
       %c2  = OpLoad %v3float %ptr_c2

   %gp_ptr  = OpAccessChain %ptr_out_v4 %gl_out %int_0

; Emit Vertex 0
              OpStore %gp_ptr %p0_shifted
              OpStore %outColor %c0
              OpEmitVertex

; Emit Vertex 1
              OpStore %gp_ptr %p1_shifted
              OpStore %outColor %c1
              OpEmitVertex

; Emit Vertex 2
              OpStore %gp_ptr %p2_shifted
              OpStore %outColor %c2
              OpEmitVertex

; 7. OutputTriangleStrip, OutputVertices 3
              OpEndPrimitive
              OpReturn
              OpFunctionEnd
