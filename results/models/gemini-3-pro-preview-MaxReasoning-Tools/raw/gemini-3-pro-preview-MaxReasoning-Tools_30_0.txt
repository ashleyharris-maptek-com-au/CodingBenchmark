import time
import numpy as np

def test_performance():
    print("Testing numpy performance for 3-SAT structure building...")
    N = 20_000_000
    M = 85_000_000
    
    # Simulate clause data
    start = time.time()
    clauses = np.random.randint(1, N + 1, size=(M, 3), dtype=np.int32)
    # randomly negate some
    neg = np.random.rand(M, 3) < 0.5
    clauses[neg] *= -1
    print(f"Generated dummy clauses: {time.time() - start:.2f}s")
    
    # Calculate literal keys
    start = time.time()
    lit_keys = 2 * np.abs(clauses) + (clauses < 0)
    flat_lits = lit_keys.ravel()
    print(f"Calculated literal keys: {time.time() - start:.2f}s")
    
    # Try bincount and offset
    start = time.time()
    counts = np.bincount(flat_lits, minlength=2*N + 2)
    offsets = np.zeros(2*N + 3, dtype=np.int32)
    np.cumsum(counts, out=offsets[1:])
    print(f"Bincount and cumsum: {time.time() - start:.2f}s")
    
    # Try argsort
    start = time.time()
    try:
        sort_idx = np.argsort(flat_lits)
        print(f"Argsort completed in: {time.time() - start:.2f}s")
    except Exception as e:
        print(f"Argsort failed: {e}")

if __name__ == "__main__":
    test_performance()
