import wgpu
import numpy as np
import subprocess
import os

def compile_glsl(glsl_code, output_spv="shader.spv"):
    with open("temp.comp", "w") as f:
        f.write(glsl_code)
    try:
        res = subprocess.run(
            ["glslangValidator", "-V", "temp.comp", "-o", output_spv],
            capture_output=True, text=True, check=True
        )
        print("Compilation success:", res.stdout)
    finally:
        if os.path.exists("temp.comp"):
            os.remove("temp.comp")

def run_compute(spv_path, input_data, N):
    adapter = wgpu.gpu.request_adapter(power_preference="high-performance")
    device = adapter.request_device()

    with open(spv_path, "rb") as f:
        spirv_bytes = f.read()

    shader_module = device.create_shader_module(code=spirv_bytes)

    input_bytes = input_data.astype(np.uint32).tobytes()
    buffer_in = device.create_buffer_with_data(
        data=input_bytes,
        usage=wgpu.BufferUsage.STORAGE
    )
    buffer_out = device.create_buffer(
        size=len(input_bytes),
        usage=wgpu.BufferUsage.STORAGE | wgpu.BufferUsage.COPY_SRC
    )
    uniform_data = np.array([N, 0, 0, 0], dtype=np.uint32)
    buffer_uni = device.create_buffer_with_data(
        data=uniform_data.tobytes(),
        usage=wgpu.BufferUsage.UNIFORM
    )

    pipeline = device.create_compute_pipeline(
        layout="auto",
        compute={"module": shader_module, "entry_point": "main"}
    )

    bind_group = device.create_bind_group(
        layout=pipeline.get_bind_group_layout(0),
        entries=[
            {"binding": 0, "resource": {"buffer": buffer_in, "offset": 0, "size": buffer_in.size}},
            {"binding": 1, "resource": {"buffer": buffer_out, "offset": 0, "size": buffer_out.size}},
            {"binding": 2, "resource": {"buffer": buffer_uni, "offset": 0, "size": buffer_uni.size}},
        ]
    )

    encoder = device.create_command_encoder()
    pass_encoder = encoder.begin_compute_pass()
    pass_encoder.set_pipeline(pipeline)
    pass_encoder.set_bind_group(0, bind_group)
    # The benchmark might dispatch (N + 255) // 256 workgroups. Let's do the same.
    workgroup_count = (N + 255) // 256
    pass_encoder.dispatch_workgroups(workgroup_count, 1, 1)
    pass_encoder.end()

    device.queue.submit([encoder.finish()])

    out_data = device.queue.read_buffer(buffer_out).cast("I")
    return np.array(out_data, dtype=np.uint32)

scan_glsl = """#version 450
layout(local_size_x = 256, local_size_y = 1, local_size_z = 1) in;

layout(set = 0, binding = 0, std430) readonly buffer InputBuffer {
    uint input_data[];
};

layout(set = 0, binding = 1, std430) buffer OutputBuffer {
    uint output_data[];
};

layout(set = 0, binding = 2, std140) uniform UniformBuffer {
    uint N;
    uint pad0;
    uint pad1;
    uint pad2;
};

shared uint shared_array[256];

void main() {
    if (gl_WorkGroupID.x > 0) {
        return;
    }

    uint t = gl_LocalInvocationID.x;
    uint prev_total = 0;
    uint num_chunks = (N + 255) / 256;

    for (uint c = 0; c < num_chunks; ++c) {
        uint idx = c * 256 + t;
        uint val = 0;
        if (idx < N) {
            val = input_data[idx];
        }
        shared_array[t] = val;
        barrier();

        for (uint stride = 1; stride < 256; stride *= 2) {
            uint temp = 0;
            if (t >= stride) {
                temp = shared_array[t - stride];
            }
            barrier();
            shared_array[t] += temp;
            barrier();
        }

        if (idx < N) {
            output_data[idx] = shared_array[t] + prev_total;
        }

        uint chunk_sum = shared_array[255];
        barrier();
        prev_total += chunk_sum;
    }
}
"""

if __name__ == "__main__":
    compile_glsl(scan_glsl)
    
    test_sizes = [256, 1024, 4096, 16384, 65536]
    for size in test_sizes:
        # Generate random inputs
        input_data = np.random.randint(1, 10, size=size, dtype=np.uint32)
        expected = np.cumsum(input_data, dtype=np.uint32)
        
        output = run_compute("shader.spv", input_data, size)
        
        if np.array_equal(expected, output):
            print(f"Size {size:5d}: Success!")
        else:
            print(f"Size {size:5d}: FAILED!")
            # Print first few mismatched elements
            mismatches = np.where(expected != output)[0]
            print("First mismatch at index:", mismatches[0])
            print("Expected:", expected[mismatches[0]:mismatches[0]+5])
            print("Got     :", output[mismatches[0]:mismatches[0]+5])
            break
