               OpCapability Geometry
  %glsl_std = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
               OpExecutionMode %main InputLinesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 4
               OpExecutionMode %main Invocations 1

               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %worldPos Location 0
               OpDecorate %normal Location 1
               OpDecorate %color Location 2

               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %outColor Location 0

       %void = OpTypeVoid
       %func = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v2float = OpTypeVector %float 2
    %v3float = OpTypeVector %float 3
    %v4float = OpTypeVector %float 4

       %uint = OpTypeInt 32 0
     %uint_4 = OpConstant %uint 4
        %int = OpTypeInt 32 1
      %int_0 = OpConstant %int 0
      %int_1 = OpConstant %int 1
      %int_2 = OpConstant %int 2
      %int_3 = OpConstant %int 3

 %ptr_in_v4 = OpTypePointer Input %v4float
 %ptr_in_v3 = OpTypePointer Input %v3float

%gl_PerVertex_in = OpTypeStruct %v4float
  %arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
     %gl_in = OpVariable %ptr_in_arr_pv Input

   %arr_v3_4 = OpTypeArray %v3float %uint_4
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_4
   %worldPos = OpVariable %ptr_in_arr_v3 Input
    %normal = OpVariable %ptr_in_arr_v3 Input
     %color = OpVariable %ptr_in_arr_v3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
    %gl_out = OpVariable %ptr_out_pv Output

%ptr_out_v3 = OpTypePointer Output %v3float
   %outColor = OpVariable %ptr_out_v3 Output

%ptr_out_v4 = OpTypePointer Output %v4float

 %float_thick = OpConstant %float 0.025

       %main = OpFunction %void None %func
      %entry = OpLabel

               ; 1. Read all 4 positions
   %p0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
       %p0 = OpLoad %v4float %p0_ptr

   %p1_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
       %p1 = OpLoad %v4float %p1_ptr

   %p2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
       %p2 = OpLoad %v4float %p2_ptr

   %p3_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
       %p3 = OpLoad %v4float %p3_ptr

               ; 2. Compute tangent at start: t0 = normalize(p2.pos - p0.pos)
    %p0_xy = OpVectorShuffle %v2float %p0 %p0 0 1
    %p2_xy = OpVectorShuffle %v2float %p2 %p2 0 1
    %diff0 = OpFSub %v2float %p2_xy %p0_xy
       %t0 = OpExtInst %v2float %glsl_std Normalize %diff0

               ; 3. Compute tangent at end: t1 = normalize(p3.pos - p1.pos)
    %p1_xy = OpVectorShuffle %v2float %p1 %p1 0 1
    %p3_xy = OpVectorShuffle %v2float %p3 %p3 0 1
    %diff1 = OpFSub %v2float %p3_xy %p1_xy
       %t1 = OpExtInst %v2float %glsl_std Normalize %diff1

               ; 4. Compute perpendiculars
     %t0_x = OpCompositeExtract %float %t0 0
     %t0_y = OpCompositeExtract %float %t0 1
 %neg_t0_y = OpFNegate %float %t0_y
    %perp0 = OpCompositeConstruct %v2float %neg_t0_y %t0_x

     %t1_x = OpCompositeExtract %float %t1 0
     %t1_y = OpCompositeExtract %float %t1 1
 %neg_t1_y = OpFNegate %float %t1_y
    %perp1 = OpCompositeConstruct %v2float %neg_t1_y %t1_x

               ; 5. Scale perpendiculars by thickness
    %disp0 = OpVectorTimesScalar %v2float %perp0 %float_thick
    %disp1 = OpVectorTimesScalar %v2float %perp1 %float_thick

               ; 6. Load colors from color[1] and color[2]
%color_1_ptr = OpAccessChain %ptr_in_v3 %color %int_1
  %color_1 = OpLoad %v3float %color_1_ptr

%color_2_ptr = OpAccessChain %ptr_in_v3 %color %int_2
  %color_2 = OpLoad %v3float %color_2_ptr

               ; Get output gl_Position pointer
%gl_Position_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0

               ; --- Emit Vertex 1: start + perp0*thickness ---
   %v1_xy = OpFAdd %v2float %p1_xy %disp0
   %v1_x  = OpCompositeExtract %float %v1_xy 0
   %v1_y  = OpCompositeExtract %float %v1_xy 1
   %p1_z  = OpCompositeExtract %float %p1 2
   %p1_w  = OpCompositeExtract %float %p1 3
  %v1_pos = OpCompositeConstruct %v4float %v1_x %v1_y %p1_z %p1_w
            OpStore %gl_Position_ptr %v1_pos
            OpStore %outColor %color_1
            OpEmitVertex

               ; --- Emit Vertex 2: start - perp0*thickness ---
   %v2_xy = OpFSub %v2float %p1_xy %disp0
   %v2_x  = OpCompositeExtract %float %v2_xy 0
   %v2_y  = OpCompositeExtract %float %v2_xy 1
  %v2_pos = OpCompositeConstruct %v4float %v2_x %v2_y %p1_z %p1_w
            OpStore %gl_Position_ptr %v2_pos
            OpStore %outColor %color_1
            OpEmitVertex

               ; --- Emit Vertex 3: end + perp1*thickness ---
   %v3_xy = OpFAdd %v2float %p2_xy %disp1
   %v3_x  = OpCompositeExtract %float %v3_xy 0
   %v3_y  = OpCompositeExtract %float %v3_xy 1
   %p2_z  = OpCompositeExtract %float %p2 2
   %p2_w  = OpCompositeExtract %float %p2 3
  %v3_pos = OpCompositeConstruct %v4float %v3_x %v3_y %p2_z %p2_w
            OpStore %gl_Position_ptr %v3_pos
            OpStore %outColor %color_2
            OpEmitVertex

               ; --- Emit Vertex 4: end - perp1*thickness ---
   %v4_xy = OpFSub %v2float %p2_xy %disp1
   %v4_x  = OpCompositeExtract %float %v4_xy 0
   %v4_y  = OpCompositeExtract %float %v4_xy 1
  %v2_z  = OpCompositeExtract %float %p2 2
  %v2_w  = OpCompositeExtract %float %p2 3
  %v4_pos = OpCompositeConstruct %v4float %v4_x %v4_y %v2_z %v2_w
            OpStore %gl_Position_ptr %v4_pos
            OpStore %outColor %color_2
            OpEmitVertex

            OpEndPrimitive
            OpReturn
            OpFunctionEnd
