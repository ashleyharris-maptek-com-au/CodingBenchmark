using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Runtime.CompilerServices;
using System.Threading;
using System.Threading.Tasks;

public enum FitHeuristic {
    BSSF, // Best Short Side Fit
    BLSF, // Best Long Side Fit
    BVF,  // Best Volume Fit
    BLP   // Bottom-Left-Front position
}

public enum BoxSortOrder {
    VolumeAsc,
    VolumeDesc,
    MaxDimAsc,
    MaxDimDesc,
    MinDimAsc,
    MinDimDesc,
    SumDimAsc,
    SumDimDesc,
    AreaAsc,
    AreaDesc,
    None
}

public struct Orientation {
    public int W, H, D;
    public Orientation(int w, int h, int d) {
        W = w;
        H = h;
        D = d;
    }
}

public struct Box {
    public int Index;
    public int W, H, D;
    public long Volume;
    public int MinDim, MidDim, MaxDim;
    public Orientation[] Orientations;

    public Box(int index, int w, int h, int d) {
        Index = index;
        W = w;
        H = h;
        D = d;
        Volume = (long)w * h * d;
        
        int[] dims = { w, h, d };
        Array.Sort(dims);
        MinDim = dims[0];
        MidDim = dims[1];
        MaxDim = dims[2];

        int count = 0;
        Orientation[] temp = new Orientation[6];
        Orientation[] perms = {
            new Orientation(w, h, d),
            new Orientation(w, d, h),
            new Orientation(h, w, d),
            new Orientation(h, d, w),
            new Orientation(d, w, h),
            new Orientation(d, h, w)
        };
        for (int i = 0; i < 6; i++) {
            bool dup = false;
            for (int j = 0; j < count; j++) {
                if (temp[j].W == perms[i].W && temp[j].H == perms[i].H && temp[j].D == perms[i].D) {
                    dup = true;
                    break;
                }
            }
            if (!dup) {
                temp[count++] = perms[i];
            }
        }
        Orientations = new Orientation[count];
        Array.Copy(temp, Orientations, count);
    }
}

public struct FreeSpace {
    public int X, Y, Z;
    public int W, H, D;
    public long Volume;
    public int MinDim, MidDim, MaxDim;

    public FreeSpace(int x, int y, int z, int w, int h, int d) {
        X = x;
        Y = y;
        Z = z;
        W = w;
        H = h;
        D = d;
        Volume = (long)w * h * d;

        int[] dims = { w, h, d };
        Array.Sort(dims);
        MinDim = dims[0];
        MidDim = dims[1];
        MaxDim = dims[2];
    }
}

public struct Placement {
    public int BoxIndex;
    public int X, Y, Z;
    public int W, H, D;
}

public struct SolverConfig {
    public BoxSortOrder SortOrder;
    public FitHeuristic Heuristic;
    public int MaxFreeSpaces;

    public SolverConfig(BoxSortOrder sortOrder, FitHeuristic heuristic, int maxFreeSpaces) {
        SortOrder = sortOrder;
        Heuristic = heuristic;
        MaxFreeSpaces = maxFreeSpaces;
    }
}

public class ScratchBuffer {
    public int[] SuffixMinMinDim;
    public int[] SuffixMinMidDim;
    public int[] SuffixMinMaxDim;
    public List<FreeSpace> FreeSpaces;
    public List<FreeSpace> Overlapping;
    public List<FreeSpace> NewCandidates;
    public List<Placement> Placements;
    public Box[] PerturbedBoxes;

    public ScratchBuffer(int maxBoxes) {
        SuffixMinMinDim = new int[maxBoxes];
        SuffixMinMidDim = new int[maxBoxes];
        SuffixMinMaxDim = new int[maxBoxes];
        FreeSpaces = new List<FreeSpace>(2048);
        Overlapping = new List<FreeSpace>(128);
        NewCandidates = new List<FreeSpace>(128);
        Placements = new List<Placement>(maxBoxes);
        PerturbedBoxes = new Box[maxBoxes];
    }

    public void Clear() {
        FreeSpaces.Clear();
        Overlapping.Clear();
        NewCandidates.Clear();
        Placements.Clear();
    }
}

public class FastScanner {
    private readonly Stream _stream;
    private readonly byte[] _buffer = new byte[1024 * 1024];
    private int _head = 0;
    private int _tail = 0;

    public FastScanner(Stream stream) {
        _stream = stream;
    }

    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    private byte Read() {
        if (_head >= _tail) {
            _head = 0;
            _tail = _stream.Read(_buffer, 0, _buffer.Length);
            if (_tail <= 0) return 0;
        }
        return _buffer[_head++];
    }

    public int NextInt() {
        byte b = Read();
        while (b > 0 && b <= 32) {
            b = Read();
        }
        if (b == 0) return -1;
        int val = 0;
        while (b >= 48 && b <= 57) {
            val = val * 10 + (b - 48);
            b = Read();
        }
        return val;
    }
}

public class Program {
    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    public static bool IsBetter(
        int leftoverWA, int leftoverHA, int leftoverDA, int yA, int xA, int zA, long fsVolumeA, long boxVolumeA,
        int leftoverWB, int leftoverHB, int leftoverDB, int yB, int xB, int zB, long fsVolumeB, long boxVolumeB,
        FitHeuristic heuristic) 
    {
        switch (heuristic) {
            case FitHeuristic.BSSF: {
                int scoreA = Math.Min(leftoverWA, Math.Min(leftoverHA, leftoverDA));
                int scoreB = Math.Min(leftoverWB, Math.Min(leftoverHB, leftoverDB));
                return scoreA < scoreB;
            }
            case FitHeuristic.BLSF: {
                int scoreA = Math.Max(leftoverWA, Math.Max(leftoverHA, leftoverDA));
                int scoreB = Math.Max(leftoverWB, Math.Max(leftoverHB, leftoverDB));
                return scoreA < scoreB;
            }
            case FitHeuristic.BVF: {
                long scoreA = fsVolumeA - boxVolumeA;
                long scoreB = fsVolumeB - boxVolumeB;
                return scoreA < scoreB;
            }
            case FitHeuristic.BLP: {
                if (yA != yB) return yA < yB;
                if (xA != xB) return xA < xB;
                return zA < zB;
            }
        }
        return false;
    }

    public static int PackWithOrder(int containerW, int containerH, int containerD, Box[] sortedBoxes, SolverConfig config, ScratchBuffer scratch, Stopwatch sw) {
        scratch.Clear();
        var freeSpaces = scratch.FreeSpaces;
        var placements = scratch.Placements;
        var overlapping = scratch.Overlapping;
        var newCandidates = scratch.NewCandidates;

        freeSpaces.Add(new FreeSpace(0, 0, 0, containerW, containerH, containerD));

        int n = sortedBoxes.Length;
        var suffixMinMinDim = scratch.SuffixMinMinDim;
        var suffixMinMidDim = scratch.SuffixMinMidDim;
        var suffixMinMaxDim = scratch.SuffixMinMaxDim;

        if (n > 0) {
            suffixMinMinDim[n - 1] = sortedBoxes[n - 1].MinDim;
            suffixMinMidDim[n - 1] = sortedBoxes[n - 1].MidDim;
            suffixMinMaxDim[n - 1] = sortedBoxes[n - 1].MaxDim;
            for (int i = n - 2; i >= 0; i--) {
                suffixMinMinDim[i] = Math.Min(sortedBoxes[i].MinDim, suffixMinMinDim[i + 1]);
                suffixMinMidDim[i] = Math.Min(sortedBoxes[i].MidDim, suffixMinMidDim[i + 1]);
                suffixMinMaxDim[i] = Math.Min(sortedBoxes[i].MaxDim, suffixMinMaxDim[i + 1]);
            }
        }

        for (int i = 0; i < n; i++) {
            if ((i & 255) == 0 && sw.ElapsedMilliseconds > 27500) {
                break;
            }

            Box box = sortedBoxes[i];

            int minRemainingMin = suffixMinMinDim[i];
            int minRemainingMid = suffixMinMidDim[i];
            int minRemainingMax = suffixMinMaxDim[i];

            int bestFreeSpaceIndex = -1;
            Orientation bestOrientation = new Orientation(0, 0, 0);
            
            int bestLeftoverW = 0, bestLeftoverH = 0, bestLeftoverD = 0;
            int bestY = 0, bestX = 0, bestZ = 0;
            long bestFSVolume = 0, bestBoxVolume = 0;

            int numFreeSpaces = freeSpaces.Count;
            for (int j = 0; j < numFreeSpaces; j++) {
                FreeSpace fs = freeSpaces[j];

                if (box.MinDim > fs.MinDim || box.MidDim > fs.MidDim || box.MaxDim > fs.MaxDim) {
                    continue;
                }

                var orients = box.Orientations;
                int numOrients = orients.Length;
                for (int k = 0; k < numOrients; k++) {
                    Orientation orient = orients[k];
                    if (orient.W <= fs.W && orient.H <= fs.H && orient.D <= fs.D) {
                        int leftoverW = fs.W - orient.W;
                        int leftoverH = fs.H - orient.H;
                        int leftoverD = fs.D - orient.D;
                        long boxVol = orient.W * (long)orient.H * orient.D;

                        bool isFirst = (bestFreeSpaceIndex == -1);
                        bool better = isFirst || IsBetter(
                            leftoverW, leftoverH, leftoverD, fs.Y, fs.X, fs.Z, fs.Volume, boxVol,
                            bestLeftoverW, bestLeftoverH, bestLeftoverD, bestY, bestX, bestZ, bestFSVolume, bestBoxVolume,
                            config.Heuristic
                        );

                        if (better) {
                            bestFreeSpaceIndex = j;
                            bestOrientation = orient;
                            bestLeftoverW = leftoverW;
                            bestLeftoverH = leftoverH;
                            bestLeftoverD = leftoverD;
                            bestY = fs.Y;
                            bestX = fs.X;
                            bestZ = fs.Z;
                            bestFSVolume = fs.Volume;
                            bestBoxVolume = boxVol;
                        }
                    }
                }
            }

            if (bestFreeSpaceIndex != -1) {
                FreeSpace targetFS = freeSpaces[bestFreeSpaceIndex];
                Placement placement;
                placement.BoxIndex = box.Index;
                placement.X = targetFS.X;
                placement.Y = targetFS.Y;
                placement.Z = targetFS.Z;
                placement.W = bestOrientation.W;
                placement.H = bestOrientation.H;
                placement.D = bestOrientation.D;
                placements.Add(placement);

                int bx = placement.X;
                int by = placement.Y;
                int bz = placement.Z;
                int bw = placement.W;
                int bh = placement.H;
                int bd = placement.D;

                overlapping.Clear();
                int fsCount = freeSpaces.Count;
                int writeIdx = 0;
                for (int j = 0; j < fsCount; j++) {
                    FreeSpace fs = freeSpaces[j];
                    if (bx < fs.X + fs.W && bx + bw > fs.X &&
                        by < fs.Y + fs.H && by + bh > fs.Y &&
                        bz < fs.Z + fs.D && bz + bd > fs.Z) {
                        overlapping.Add(fs);
                    } else {
                        freeSpaces[writeIdx++] = fs;
                    }
                }
                freeSpaces.RemoveRange(writeIdx, fsCount - writeIdx);

                newCandidates.Clear();
                int overlapCount = overlapping.Count;
                for (int j = 0; j < overlapCount; j++) {
                    FreeSpace fs = overlapping[j];
                    
                    if (bx > fs.X) {
                        int w = bx - fs.X;
                        if (w >= minRemainingMin) {
                            newCandidates.Add(new FreeSpace(fs.X, fs.Y, fs.Z, w, fs.H, fs.D));
                        }
                    }
                    if (bx + bw < fs.X + fs.W) {
                        int w = (fs.X + fs.W) - (bx + bw);
                        if (w >= minRemainingMin) {
                            newCandidates.Add(new FreeSpace(bx + bw, fs.Y, fs.Z, w, fs.H, fs.D));
                        }
                    }
                    if (by > fs.Y) {
                        int h = by - fs.Y;
                        if (h >= minRemainingMin) {
                            newCandidates.Add(new FreeSpace(fs.X, fs.Y, fs.Z, fs.W, h, fs.D));
                        }
                    }
                    if (by + bh < fs.Y + fs.H) {
                        int h = (fs.Y + fs.H) - (by + bh);
                        if (h >= minRemainingMin) {
                            newCandidates.Add(new FreeSpace(fs.X, by + bh, fs.Z, fs.W, h, fs.D));
                        }
                    }
                    if (bz > fs.Z) {
                        int d = bz - fs.Z;
                        if (d >= minRemainingMin) {
                            newCandidates.Add(new FreeSpace(fs.X, fs.Y, fs.Z, fs.W, fs.H, d));
                        }
                    }
                    if (bz + bd < fs.Z + fs.D) {
                        int d = (fs.Z + fs.D) - (bz + bd);
                        if (d >= minRemainingMin) {
                            newCandidates.Add(new FreeSpace(fs.X, fs.Y, bz + bd, fs.W, fs.H, d));
                        }
                    }
                }

                int candCount = newCandidates.Count;
                for (int j = 0; j < candCount; j++) {
                    FreeSpace c = newCandidates[j];

                    if (c.MinDim < minRemainingMin || c.MidDim < minRemainingMid || c.MaxDim < minRemainingMax) {
                        continue;
                    }

                    bool contained = false;
                    int currentFSCount = freeSpaces.Count;
                    for (int k = 0; k < currentFSCount; k++) {
                        FreeSpace existing = freeSpaces[k];
                        if (c.Volume > existing.Volume) continue;
                        if (c.X >= existing.X && c.X + c.W <= existing.X + existing.W &&
                            c.Y >= existing.Y && c.Y + c.H <= existing.Y + existing.H &&
                            c.Z >= existing.Z && c.Z + c.D <= existing.Z + existing.D) {
                            contained = true;
                            break;
                        }
                    }
                    if (contained) continue;

                    for (int k = 0; k < candCount; k++) {
                        if (j == k) continue;
                        FreeSpace other = newCandidates[k];
                        if (c.Volume > other.Volume) continue;
                        if (c.X >= other.X && c.X + c.W <= other.X + other.W &&
                            c.Y >= other.Y && c.Y + c.H <= other.Y + other.H &&
                            c.Z >= other.Z && c.Z + c.D <= other.Z + other.D) {
                            if (j < k && c.X == other.X && c.Y == other.Y && c.Z == other.Z &&
                                c.W == other.W && c.H == other.H && c.D == other.D) {
                                contained = true;
                                break;
                            }
                            contained = true;
                            break;
                        }
                    }

                    if (!contained) {
                        freeSpaces.Add(c);
                    }
                }

                if (freeSpaces.Count > config.MaxFreeSpaces) {
                    freeSpaces.Sort((a, b) => b.Volume.CompareTo(a.Volume));
                    freeSpaces.RemoveRange(config.MaxFreeSpaces, freeSpaces.Count - config.MaxFreeSpaces);
                }
            }
        }

        return placements.Count;
    }

    public static Box[] SortBoxes(Box[] originalBoxes, BoxSortOrder sortOrder) {
        Box[] boxes = (Box[])originalBoxes.Clone();
        switch (sortOrder) {
            case BoxSortOrder.VolumeAsc:
                Array.Sort(boxes, (a, b) => a.Volume.CompareTo(b.Volume));
                break;
            case BoxSortOrder.VolumeDesc:
                Array.Sort(boxes, (a, b) => b.Volume.CompareTo(a.Volume));
                break;
            case BoxSortOrder.MaxDimAsc:
                Array.Sort(boxes, (a, b) => a.MaxDim.CompareTo(b.MaxDim));
                break;
            case BoxSortOrder.MaxDimDesc:
                Array.Sort(boxes, (a, b) => b.MaxDim.CompareTo(a.MaxDim));
                break;
            case BoxSortOrder.MinDimAsc:
                Array.Sort(boxes, (a, b) => a.MinDim.CompareTo(b.MinDim));
                break;
            case BoxSortOrder.MinDimDesc:
                Array.Sort(boxes, (a, b) => b.MinDim.CompareTo(a.MinDim));
                break;
            case BoxSortOrder.SumDimAsc:
                Array.Sort(boxes, (a, b) => (a.W + a.H + a.D).CompareTo(b.W + b.H + b.D));
                break;
            case BoxSortOrder.SumDimDesc:
                Array.Sort(boxes, (a, b) => (b.W + b.H + b.D).CompareTo(a.W + a.H + a.D));
                break;
            case BoxSortOrder.AreaAsc:
                Array.Sort(boxes, (a, b) => ((long)a.W*a.H + (long)a.H*a.D + (long)a.D*a.W).CompareTo((long)b.W*b.H + (long)b.H*b.D + (long)b.D*b.W));
                break;
            case BoxSortOrder.AreaDesc:
                Array.Sort(boxes, (a, b) => ((long)b.W*b.H + (long)b.H*b.D + (long)b.D*b.W).CompareTo((long)a.W*a.H + (long)a.H*a.D + (long)a.D*a.W));
                break;
        }
        return boxes;
    }

    public static List<Placement> Pack(int containerW, int containerH, int containerD, Box[] originalBoxes, SolverConfig config, ScratchBuffer scratch, Stopwatch sw) {
        Box[] sorted = SortBoxes(originalBoxes, config.SortOrder);
        int count = PackWithOrder(containerW, containerH, containerD, sorted, config, scratch, sw);
        return new List<Placement>(scratch.Placements);
    }

    public static List<SolverConfig> GetConfigs(int N) {
        List<SolverConfig> configs = new List<SolverConfig>();
        
        int[] maxFsOpts;
        if (N <= 200) {
            maxFsOpts = new[] { 5000, 10000 };
        } else if (N <= 1000) {
            maxFsOpts = new[] { 2000, 4000 };
        } else if (N <= 5000) {
            maxFsOpts = new[] { 1000, 2000 };
        } else {
            maxFsOpts = new[] { 500, 1000 };
        }

        foreach (BoxSortOrder sort in Enum.GetValues(typeof(BoxSortOrder))) {
            foreach (FitHeuristic heur in Enum.GetValues(typeof(FitHeuristic))) {
                foreach (int maxFs in maxFsOpts) {
                    configs.Add(new SolverConfig(sort, heur, maxFs));
                }
            }
        }
        return configs;
    }

    public static void Main() {
        Stopwatch sw = Stopwatch.StartNew();

        int numCores = Environment.ProcessorCount;
        ThreadPool.SetMinThreads(numCores, numCores);

        // Read input
        var scanner = new FastScanner(Console.OpenStandardInput());
        int N = scanner.NextInt();
        if (N <= 0) {
            Console.WriteLine(0);
            return;
        }

        int containerW = scanner.NextInt();
        int containerH = scanner.NextInt();
        int containerD = scanner.NextInt();

        Box[] boxes = new Box[N];
        for (int i = 0; i < N; i++) {
            int w = scanner.NextInt();
            int h = scanner.NextInt();
            int d = scanner.NextInt();
            boxes[i] = new Box(i, w, h, d);
        }

        var configs = GetConfigs(N);

        List<Placement> bestPlacements = new List<Placement>();
        SolverConfig bestConfig = default;
        int bestCount = 0;
        object lockObj = new object();

        var parallelOptions = new ParallelOptions {
            MaxDegreeOfParallelism = numCores
        };

        ThreadLocal<ScratchBuffer> detScratch = new ThreadLocal<ScratchBuffer>(() => new ScratchBuffer(N));

        Parallel.ForEach(configs, parallelOptions, config => {
            if (sw.ElapsedMilliseconds > 27500) return;
            var scratch = detScratch.Value;
            var placements = Pack(containerW, containerH, containerD, boxes, config, scratch, sw);
            
            lock (lockObj) {
                if (placements.Count > bestCount) {
                    bestCount = placements.Count;
                    bestPlacements = placements;
                    bestConfig = config;
                }
            }
        });

        long totalLimitMs = 28000;

        if (bestCount < N && sw.ElapsedMilliseconds < totalLimitMs) {
            Box[] bestSortedBase = SortBoxes(boxes, bestConfig.SortOrder);

            ThreadLocal<ScratchBuffer> graspScratch = new ThreadLocal<ScratchBuffer>(() => new ScratchBuffer(N));
            ThreadLocal<Random> localRand = new ThreadLocal<Random>(() => new Random(Guid.NewGuid().GetHashCode()));

            Parallel.For(0, 1000000, parallelOptions, (iter, state) => {
                if (sw.ElapsedMilliseconds > totalLimitMs) {
                    state.Stop();
                    return;
                }

                Random threadRand = localRand.Value;
                ScratchBuffer scratch = graspScratch.Value;

                Array.Copy(bestSortedBase, scratch.PerturbedBoxes, bestSortedBase.Length);
                Box[] perturbed = scratch.PerturbedBoxes;
                
                int window = threadRand.Next(2, 15);
                for (int i = 0; i < perturbed.Length; i++) {
                    int target = i + threadRand.Next(-window, window + 1);
                    if (target >= 0 && target < perturbed.Length) {
                        var tmp = perturbed[i];
                        perturbed[i] = perturbed[target];
                        perturbed[target] = tmp;
                    }
                }

                int count = PackWithOrder(containerW, containerH, containerD, perturbed, bestConfig, scratch, sw);

                if (count > bestCount) {
                    lock (lockObj) {
                        if (count > bestCount) {
                            bestCount = count;
                            bestPlacements = new List<Placement>(scratch.Placements);
                        }
                    }
                }
            });
        }

        // Print output
        using (var writer = new StreamWriter(Console.OpenStandardOutput(), System.Text.Encoding.ASCII, 1024 * 1024)) {
            writer.WriteLine(bestPlacements.Count);
            int count = bestPlacements.Count;
            for (int i = 0; i < count; i++) {
                var p = bestPlacements[i];
                writer.Write(p.BoxIndex);
                writer.Write(' ');
                writer.Write(p.X);
                writer.Write(' ');
                writer.Write(p.Y);
                writer.Write(' ');
                writer.WriteLine(p.Z);
            }
        }
    }
}
