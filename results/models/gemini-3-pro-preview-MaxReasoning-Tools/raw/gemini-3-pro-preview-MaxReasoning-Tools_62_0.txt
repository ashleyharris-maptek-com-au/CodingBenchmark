declare const require: any;
declare const process: any;

const fs = require('fs');

interface Town {
    id: any;
    x: number;
    y: number;
    value: number;
}

interface InputData {
    width: number;
    height: number;
    grid: string[];
    ignitions: [number, number][];
    towns: Town[];
    cutBudget: number;
}

class Dinics {
    numNodes: number;
    head: Int32Array;
    next: Int32Array;
    to: Int32Array;
    cap: Float64Array;
    initCap!: Float64Array;
    edgeCount: number;

    level: Int32Array;
    work: Int32Array;
    queue: Int32Array;

    constructor(numNodes: number, maxEdges: number) {
        this.numNodes = numNodes;
        this.head = new Int32Array(numNodes).fill(-1);
        this.next = new Int32Array(maxEdges);
        this.to = new Int32Array(maxEdges);
        this.cap = new Float64Array(maxEdges);
        this.edgeCount = 0;

        this.level = new Int32Array(numNodes);
        this.work = new Int32Array(numNodes);
        this.queue = new Int32Array(numNodes);
    }

    addEdge(from: number, to: number, cap: number) {
        const e1 = this.edgeCount++;
        const e2 = this.edgeCount++;
        
        this.to[e1] = to;
        this.cap[e1] = cap;
        this.next[e1] = this.head[from];
        this.head[from] = e1;
        
        this.to[e2] = from;
        this.cap[e2] = 0;
        this.next[e2] = this.head[to];
        this.head[to] = e2;
    }

    bfs(S: number, T: number): boolean {
        this.level.fill(-1);
        this.level[S] = 0;
        let head = 0, tail = 0;
        this.queue[tail++] = S;
        
        while (head < tail) {
            const u = this.queue[head++];
            for (let e = this.head[u]; e !== -1; e = this.next[e]) {
                const v = this.to[e];
                if (this.level[v] === -1 && this.cap[e] > 1e-9) {
                    this.level[v] = this.level[u] + 1;
                    this.queue[tail++] = v;
                }
            }
        }
        return this.level[T] !== -1;
    }

    dfs(u: number, T: number, flow: number): number {
        if (u === T || flow < 1e-9) return flow;
        
        for (let e = this.work[u]; e !== -1; e = this.next[e]) {
            this.work[u] = e;
            const v = this.to[e];
            if (this.level[v] === this.level[u] + 1 && this.cap[e] > 1e-9) {
                const pushed = this.dfs(v, T, Math.min(flow, this.cap[e]));
                if (pushed > 1e-9) {
                    this.cap[e] -= pushed;
                    this.cap[e ^ 1] += pushed;
                    return pushed;
                }
            }
        }
        return 0;
    }

    maxFlow(S: number, T: number): number {
        let totalFlow = 0;
        while (this.bfs(S, T)) {
            this.work.set(this.head);
            while (true) {
                const pushed = this.dfs(S, T, Infinity);
                if (pushed < 1e-9) break;
                totalFlow += pushed;
            }
        }
        return totalFlow;
    }
}

function solveWildfire(data: InputData): [number, number][] {
    const { width, height, grid, ignitions, towns, cutBudget } = data;
    const startTime = Date.now();
    const timeLimit = 2500; // 2.5 seconds total runtime budget

    // 1. BFS forward from ignitions
    const visitedFromIgnitions = new Uint8Array(width * height);
    {
        const queue = new Int32Array(width * height);
        let head = 0, tail = 0;
        for (const [x, y] of ignitions) {
            if (x >= 0 && x < width && y >= 0 && y < height) {
                const idx = y * width + x;
                visitedFromIgnitions[idx] = 1;
                queue[tail++] = idx;
            }
        }
        
        while (head < tail) {
            const u = queue[head++];
            const ux = u % width;
            const uy = Math.floor(u / width);
            const dirs = [[0, 1], [0, -1], [1, 0], [-1, 0]];
            for (const [dx, dy] of dirs) {
                const vx = ux + dx;
                const vy = uy + dy;
                if (vx >= 0 && vx < width && vy >= 0 && vy < height) {
                    if (grid[vy][vx] === '.') {
                        const v = vy * width + vx;
                        if (visitedFromIgnitions[v] === 0) {
                            visitedFromIgnitions[v] = 1;
                            queue[tail++] = v;
                        }
                    }
                }
            }
        }
    }

    // 2. BFS backward from towns
    const visitedFromTowns = new Uint8Array(width * height);
    {
        const queue = new Int32Array(width * height);
        let head = 0, tail = 0;
        for (const town of towns) {
            const { x, y } = town;
            if (x >= 0 && x < width && y >= 0 && y < height) {
                const idx = y * width + x;
                visitedFromTowns[idx] = 1;
                queue[tail++] = idx;
            }
        }
        
        while (head < tail) {
            const u = queue[head++];
            const ux = u % width;
            const uy = Math.floor(u / width);
            const dirs = [[0, 1], [0, -1], [1, 0], [-1, 0]];
            for (const [dx, dy] of dirs) {
                const vx = ux + dx;
                const vy = uy + dy;
                if (vx >= 0 && vx < width && vy >= 0 && vy < height) {
                    if (grid[vy][vx] === '.') {
                        const v = vy * width + vx;
                        if (visitedFromTowns[v] === 0) {
                            visitedFromTowns[v] = 1;
                            queue[tail++] = v;
                        }
                    }
                }
            }
        }
    }

    // 3. Build active indices
    const activeIndices = new Int32Array(width * height).fill(-1);
    let M = 0;
    for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
            const idx = y * width + x;
            if (visitedFromIgnitions[idx] === 1 && visitedFromTowns[idx] === 1) {
                activeIndices[idx] = M++;
            }
        }
    }

    if (M === 0) {
        return [];
    }

    const activeCoords: [number, number][] = [];
    for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
            const idx = y * width + x;
            if (activeIndices[idx] !== -1) {
                activeCoords.push([x, y]);
            }
        }
    }

    const activeNeighbors: number[][] = Array.from({ length: M }, () => []);
    for (let i = 0; i < M; i++) {
        const [x, y] = activeCoords[i];
        const dirs = [[0, 1], [0, -1], [1, 0], [-1, 0]];
        for (const [dx, dy] of dirs) {
            const nx = x + dx;
            const ny = y + dy;
            if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                const nIdx = ny * width + nx;
                const nActive = activeIndices[nIdx];
                if (nActive !== -1) {
                    activeNeighbors[i].push(nActive);
                }
            }
        }
    }

    const isIgnitionActive = new Uint8Array(M);
    const activeIgnitions: number[] = [];
    for (const [x, y] of ignitions) {
        if (x >= 0 && x < width && y >= 0 && y < height) {
            const idx = y * width + x;
            const actIdx = activeIndices[idx];
            if (actIdx !== -1) {
                isIgnitionActive[actIdx] = 1;
                activeIgnitions.push(actIdx);
            }
        }
    }

    const isTownActive = new Uint8Array(M);
    const townValueAtCell = new Float64Array(M);
    for (const town of towns) {
        const { x, y, value } = town;
        if (x >= 0 && x < width && y >= 0 && y < height) {
            const idx = y * width + x;
            const actIdx = activeIndices[idx];
            if (actIdx !== -1) {
                townValueAtCell[actIdx] += value;
            }
        }
    }

    interface ActiveTown {
        index: number;
        value: number;
    }
    const activeTowns: ActiveTown[] = [];
    let totalActiveTownValue = 0;
    for (let i = 0; i < M; i++) {
        if (townValueAtCell[i] > 0) {
            isTownActive[i] = 1;
            activeTowns.push({ index: i, value: townValueAtCell[i] });
            totalActiveTownValue += townValueAtCell[i];
        }
    }

    // Allocate evaluation structures once to reuse
    const evalBurned = new Uint8Array(M);
    const evalQueue = new Int32Array(M);

    function evaluateCut(isCut: boolean[]): { score: number; boundary: number[] } {
        evalBurned.fill(0);
        let head = 0, tail = 0;
        
        for (const ig of activeIgnitions) {
            evalBurned[ig] = 1;
            evalQueue[tail++] = ig;
        }
        
        while (head < tail) {
            const u = evalQueue[head++];
            const neighbors = activeNeighbors[u];
            for (let i = 0; i < neighbors.length; i++) {
                const v = neighbors[i];
                if (evalBurned[v] === 0 && !isCut[v]) {
                    evalBurned[v] = 1;
                    evalQueue[tail++] = v;
                }
            }
        }
        
        let score = 0;
        for (let i = 0; i < activeTowns.length; i++) {
            const t = activeTowns[i];
            if (evalBurned[t.index] === 0) {
                score += t.value;
            }
        }
        
        const boundary: number[] = [];
        for (let i = 0; i < M; i++) {
            if (isCut[i]) {
                let adjacentToBurned = false;
                const neighbors = activeNeighbors[i];
                for (let j = 0; j < neighbors.length; j++) {
                    if (evalBurned[neighbors[j]] === 1) {
                        adjacentToBurned = true;
                        break;
                    }
                }
                if (adjacentToBurned) {
                    boundary.push(i);
                }
            }
        }
        
        return { score, boundary };
    }

    const greedyDist = new Int32Array(M);
    const greedyQueue = new Int32Array(M);
    const usefulness = new Float64Array(M);

    function greedyEnhance(isCut: boolean[], K: number): boolean[] {
        const currentCut = isCut.slice();
        let { score: currentScore, boundary } = evaluateCut(currentCut);
        
        const prunedCut = new Array(M).fill(false);
        for (const x of boundary) {
            prunedCut[x] = true;
        }
        
        let budgetUsed = boundary.length;
        
        while (budgetUsed < K) {
            greedyDist.fill(-1);
            let head = 0, tail = 0;
            
            for (const ig of activeIgnitions) {
                greedyDist[ig] = 0;
                greedyQueue[tail++] = ig;
            }
            
            while (head < tail) {
                const u = greedyQueue[head++];
                const neighbors = activeNeighbors[u];
                for (let i = 0; i < neighbors.length; i++) {
                    const v = neighbors[i];
                    if (greedyDist[v] === -1 && !prunedCut[v]) {
                        greedyDist[v] = greedyDist[u] + 1;
                        greedyQueue[tail++] = v;
                    }
                }
            }
            
            const burnedTowns: ActiveTown[] = [];
            for (const t of activeTowns) {
                if (greedyDist[t.index] !== -1) {
                    burnedTowns.push(t);
                }
            }
            
            if (burnedTowns.length === 0) {
                break;
            }
            
            usefulness.fill(0);
            for (const t of burnedTowns) {
                let curr = t.index;
                const path: number[] = [curr];
                while (greedyDist[curr] > 0) {
                    const neighbors = activeNeighbors[curr];
                    let nextCell = -1;
                    for (let i = 0; i < neighbors.length; i++) {
                        const n = neighbors[i];
                        if (greedyDist[n] === greedyDist[curr] - 1 && !prunedCut[n]) {
                            nextCell = n;
                            break;
                        }
                    }
                    if (nextCell === -1) break;
                    curr = nextCell;
                    path.push(curr);
                }
                
                for (const cell of path) {
                    if (isIgnitionActive[cell] || isTownActive[cell]) {
                        continue;
                    }
                    usefulness[cell] += t.value;
                }
            }
            
            let bestCell = -1;
            let maxUse = 0;
            for (let i = 0; i < M; i++) {
                if (!prunedCut[i] && !isIgnitionActive[i] && !isTownActive[i] && usefulness[i] > maxUse) {
                    maxUse = usefulness[i];
                    bestCell = i;
                }
            }
            
            if (bestCell === -1) {
                break;
            }
            
            prunedCut[bestCell] = true;
            budgetUsed++;
        }
        
        return prunedCut;
    }

    // 4. Build the Flow Graph for Dinic's
    const S = 2 * M;
    const T = 2 * M + 1;
    const maxEdges = 24 * M + 10;
    const dinic = new Dinics(2 * M + 2, maxEdges);
    const splitEdgeIndex = new Int32Array(M).fill(-1);

    for (let i = 0; i < M; i++) {
        let cap = 0;
        if (isIgnitionActive[i] || isTownActive[i]) {
            cap = Infinity;
        }
        splitEdgeIndex[i] = dinic.edgeCount;
        dinic.addEdge(2 * i, 2 * i + 1, cap);
    }

    for (let i = 0; i < M; i++) {
        const neighbors = activeNeighbors[i];
        for (let k = 0; k < neighbors.length; k++) {
            const j = neighbors[k];
            if (i < j) {
                dinic.addEdge(2 * i + 1, 2 * j, Infinity);
                dinic.addEdge(2 * j + 1, 2 * i, Infinity);
            }
        }
    }

    for (const ig of activeIgnitions) {
        dinic.addEdge(S, 2 * ig, Infinity);
    }

    for (const t of activeTowns) {
        dinic.addEdge(2 * t.index + 1, T, t.value);
    }

    const initCap = new Float64Array(dinic.cap);
    dinic.initCap = initCap;

    function runDinicWithLambda(lambda: number): boolean[] {
        dinic.cap.set(dinic.initCap);
        for (let i = 0; i < M; i++) {
            if (!isIgnitionActive[i] && !isTownActive[i]) {
                const e = splitEdgeIndex[i];
                dinic.cap[e] = lambda;
            }
        }
        
        dinic.maxFlow(S, T);
        
        const visited = new Uint8Array(2 * M + 2);
        const queue = dinic.queue;
        let head = 0, tail = 0;
        
        visited[S] = 1;
        queue[tail++] = S;
        
        while (head < tail) {
            const u = queue[head++];
            for (let e = dinic.head[u]; e !== -1; e = dinic.next[e]) {
                const v = dinic.to[e];
                if (visited[v] === 0 && dinic.cap[e] > 1e-9) {
                    visited[v] = 1;
                    queue[tail++] = v;
                }
            }
        }
        
        const isCut = new Array(M).fill(false);
        for (let i = 0; i < M; i++) {
            const u = 2 * i;
            const v = 2 * i + 1;
            if (visited[u] === 1 && visited[v] === 0) {
                isCut[i] = true;
            }
        }
        return isCut;
    }

    // 5. Binary search for lambda
    let low = 0;
    let high = totalActiveTownValue + 1;
    let bestCut = new Array(M).fill(false);
    let bestScore = -1;

    for (let iter = 0; iter < 18; iter++) {
        const mid = (low + high) / 2;
        const isCut = runDinicWithLambda(mid);
        const { score, boundary } = evaluateCut(isCut);
        const cutCount = boundary.length;
        
        if (cutCount <= cutBudget) {
            if (score > bestScore) {
                bestScore = score;
                bestCut = isCut;
            }
            high = mid;
        } else {
            low = mid;
        }
    }

    // Evaluate pure greedy cut as alternative starting point
    const emptyCut = new Array(M).fill(false);
    const pureGreedyCut = greedyEnhance(emptyCut, cutBudget);
    const { score: pureGreedyScore } = evaluateCut(pureGreedyCut);
    if (pureGreedyScore > bestScore) {
        bestScore = pureGreedyScore;
        bestCut = pureGreedyCut;
    }

    bestCut = greedyEnhance(bestCut, cutBudget);
    bestScore = evaluateCut(bestCut).score;

    // 7. Local search (1-swap optimization)
    function localSearch(isCut: boolean[], K: number, tLimit: number): boolean[] {
        let currentCut = isCut.slice();
        let { score: currentScore, boundary } = evaluateCut(currentCut);
        
        let bestCutSoFar = currentCut.slice();
        let bestScoreSoFar = currentScore;
        
        let improved = true;
        while (improved && (Date.now() - startTime) < tLimit) {
            improved = false;
            const activeCuts = boundary.slice();
            
            for (const c of activeCuts) {
                if ((Date.now() - startTime) >= tLimit) {
                    break;
                }
                
                const tempCut = currentCut.slice();
                tempCut[c] = false;
                
                const enhancedCut = greedyEnhance(tempCut, K);
                const { score: newScore, boundary: newBoundary } = evaluateCut(enhancedCut);
                
                if (newScore > bestScoreSoFar) {
                    bestScoreSoFar = newScore;
                    bestCutSoFar = enhancedCut;
                    currentCut = enhancedCut;
                    boundary = newBoundary;
                    improved = true;
                    break;
                }
            }
        }
        return bestCutSoFar;
    }

    bestCut = localSearch(bestCut, cutBudget, timeLimit);

    const { boundary } = evaluateCut(bestCut);
    const result: [number, number][] = [];
    for (const idx of boundary) {
        result.push(activeCoords[idx]);
    }
    return result;
}

function main() {
    const input: string = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);
    const firebreaks = solveWildfire(data);
    console.log(JSON.stringify({ firebreaks }));
}

main();
