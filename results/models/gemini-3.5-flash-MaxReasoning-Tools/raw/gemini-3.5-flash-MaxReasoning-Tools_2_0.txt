using System;
using System.Collections.Generic;
using System.IO;

public class Solver {
    // Fast standard input parser
    public class FastScanner {
        private readonly Stream stream;
        private readonly byte[] buffer = new byte[4 * 1024 * 1024];
        private int head = 0;
        private int tail = 0;

        public FastScanner() {
            stream = Console.OpenStandardInput();
        }

        private byte Read() {
            if (head >= tail) {
                head = 0;
                tail = stream.Read(buffer, 0, buffer.Length);
                if (tail <= 0) return 0;
            }
            return buffer[head++];
        }

        public int NextInt() {
            byte c = Read();
            while (c <= 32) {
                if (c == 0) return -1;
                c = Read();
            }
            bool neg = false;
            if (c == '-') {
                neg = true;
                c = Read();
            }
            int res = 0;
            while (c > 32) {
                if (c >= '0' && c <= '9') {
                    res = res * 10 + (c - '0');
                }
                c = Read();
            }
            return neg ? -res : res;
        }
    }

    // Fast standard output formatter
    public class FastWriter : IDisposable {
        private readonly Stream stream;
        private readonly byte[] buffer;
        private int head;

        public FastWriter() {
            stream = Console.OpenStandardOutput();
            buffer = new byte[4 * 1024 * 1024];
            head = 0;
        }

        public void WriteInt(int val) {
            if (head + 20 >= buffer.Length) {
                Flush();
            }
            if (val == 0) {
                buffer[head++] = (byte)'0';
                return;
            }
            if (val < 0) {
                buffer[head++] = (byte)'-';
                val = -val;
            }
            int start = head;
            while (val > 0) {
                buffer[head++] = (byte)('0' + (val % 10));
                val /= 10;
            }
            int end = head - 1;
            while (start < end) {
                byte temp = buffer[start];
                buffer[start] = buffer[end];
                buffer[end] = temp;
                start++;
                end--;
            }
        }

        public void WriteChar(char c) {
            if (head >= buffer.Length) {
                Flush();
            }
            buffer[head++] = (byte)c;
        }

        public void Flush() {
            if (head > 0) {
                stream.Write(buffer, 0, head);
                head = 0;
            }
        }

        public void Dispose() {
            Flush();
        }
    }

    // Custom Min-Heap for Dijkstra
    public class MinHeap {
        private int[] heap;
        private long[] priorities;
        private int size;

        public MinHeap(int capacity) {
            heap = new int[capacity];
            priorities = new long[capacity];
            size = 0;
        }

        public void Clear() {
            size = 0;
        }

        public int Size => size;

        public void Push(int element, long priority) {
            if (size == heap.Length) {
                Array.Resize(ref heap, heap.Length * 2);
                Array.Resize(ref priorities, priorities.Length * 2);
            }
            heap[size] = element;
            priorities[size] = priority;
            Up(size);
            size++;
        }

        public int Pop(out long priority) {
            int res = heap[0];
            priority = priorities[0];
            size--;
            if (size > 0) {
                heap[0] = heap[size];
                priorities[0] = priorities[size];
                Down(0);
            }
            return res;
        }

        private void Up(int i) {
            int val = heap[i];
            long prio = priorities[i];
            while (i > 0) {
                int p = (i - 1) >> 1;
                if (priorities[p] <= prio) break;
                heap[i] = heap[p];
                priorities[i] = priorities[p];
                i = p;
            }
            heap[i] = val;
            priorities[i] = prio;
        }

        private void Down(int i) {
            int val = heap[i];
            long prio = priorities[i];
            int half = size >> 1;
            while (i < half) {
                int left = (i << 1) + 1;
                int right = left + 1;
                int child = left;
                if (right < size && priorities[right] < priorities[left]) {
                    child = right;
                }
                if (priorities[child] >= prio) break;
                heap[i] = heap[child];
                priorities[i] = priorities[child];
                i = child;
            }
            heap[i] = val;
            priorities[i] = prio;
        }
    }

    // Candidate edge for Voronoi matching
    struct CandidateEdge : IComparable<CandidateEdge> {
        public int U;
        public int V;
        public long Weight;
        public int BoundaryU;
        public int BoundaryV;
        public int BoundaryEdgeId;

        public int CompareTo(CandidateEdge other) {
            if (Weight < other.Weight) return -1;
            if (Weight > other.Weight) return 1;
            return 0;
        }
    }

    static int[] head;
    static int[] next;
    static int[] to;
    static int[] weight;
    static int[] edgeId;
    static int edgeCount = 0;

    static void ArrayFill<T>(T[] array, T value) {
        for (int i = 0; i < array.Length; i++) {
            array[i] = value;
        }
    }

    static void AddEdge(int u, int v, int w, int id) {
        to[edgeCount] = v;
        weight[edgeCount] = w;
        edgeId[edgeCount] = id;
        next[edgeCount] = head[u];
        head[u] = edgeCount++;
    }

    static void Dijkstra(int src, long[] dist, int[] parentEdge) {
        ArrayFill(dist, long.MaxValue);
        ArrayFill(parentEdge, -1);
        dist[src] = 0;
        MinHeap heap = new MinHeap(1024);
        heap.Push(src, 0);
        while (heap.Size > 0) {
            long d;
            int u = heap.Pop(out d);
            if (d > dist[u]) continue;
            for (int e = head[u]; e != -1; e = next[e]) {
                int v = to[e];
                long nextDist = d + weight[e];
                if (nextDist < dist[v]) {
                    dist[v] = nextDist;
                    parentEdge[v] = edgeId[e];
                    heap.Push(v, nextDist);
                }
            }
        }
    }

    public static void Main(string[] args) {
        FastScanner scanner = new FastScanner();
        int N = scanner.NextInt();
        int M = scanner.NextInt();
        if (N == -1) return;

        head = new int[N];
        ArrayFill(head, -1);
        next = new int[2 * M];
        to = new int[2 * M];
        weight = new int[2 * M];
        edgeId = new int[2 * M];

        int[] edgeU = new int[M];
        int[] edgeV = new int[M];
        int[] edgeW = new int[M];

        for (int i = 0; i < M; i++) {
            int u = scanner.NextInt();
            int v = scanner.NextInt();
            int w = scanner.NextInt();
            edgeU[i] = u;
            edgeV[i] = v;
            edgeW[i] = w;
            AddEdge(u, v, w, i);
            AddEdge(v, u, w, i);
        }

        List<int> oddVertices = new List<int>();
        for (int i = 0; i < N; i++) {
            int deg = 0;
            for (int e = head[i]; e != -1; e = next[e]) {
                deg++;
            }
            if (deg % 2 != 0) {
                oddVertices.Add(i);
            }
        }

        List<int> duplicatedEdgeIds = new List<int>();

        if (oddVertices.Count > 0) {
            int k2 = oddVertices.Count;
            if (k2 <= 22) {
                // Exact matching using DP with bitmask
                long[,] cost = new long[k2, k2];
                int[][] parents = new int[k2][];
                for (int i = 0; i < k2; i++) {
                    parents[i] = new int[N];
                    long[] d = new long[N];
                    Dijkstra(oddVertices[i], d, parents[i]);
                    for (int j = 0; j < k2; j++) {
                        cost[i, j] = d[oddVertices[j]];
                    }
                }

                long[] dp = new long[1 << k2];
                int[] choice = new int[1 << k2];
                ArrayFill(dp, long.MaxValue);
                dp[0] = 0;

                int[] firstZero = new int[1 << k2];
                for (int mask = 0; mask < (1 << k2); mask++) {
                    int first = -1;
                    for (int b = 0; b < k2; b++) {
                        if ((mask & (1 << b)) == 0) {
                            first = b;
                            break;
                        }
                    }
                    firstZero[mask] = first;
                }

                for (int mask = 0; mask < (1 << k2); mask++) {
                    long curCost = dp[mask];
                    if (curCost == long.MaxValue) continue;
                    int i = firstZero[mask];
                    if (i == -1) continue;
                    for (int j = i + 1; j < k2; j++) {
                        if ((mask & (1 << j)) == 0) {
                            int nextMask = mask | (1 << i) | (1 << j);
                            long newCost = curCost + cost[i, j];
                            if (newCost < dp[nextMask]) {
                                dp[nextMask] = newCost;
                                choice[nextMask] = (i << 8) | j;
                            }
                        }
                    }
                }

                int curr = (1 << k2) - 1;
                while (curr > 0) {
                    int c = choice[curr];
                    int i = c >> 8;
                    int j = c & 0xFF;
                    
                    // Reconstruct path between i and j
                    int currNode = oddVertices[j];
                    int startNode = oddVertices[i];
                    while (currNode != startNode) {
                        int e = parents[i][currNode];
                        duplicatedEdgeIds.Add(e);
                        int u = edgeU[e];
                        int v = edgeV[e];
                        currNode = u == currNode ? v : u;
                    }
                    curr = curr & ~(1 << i) & ~(1 << j);
                }
            } else {
                // Heuristic matching for large graphs using Voronoi cells
                long[] dist = new long[N];
                int[] gen = new int[N];
                int[] parentEdge = new int[N];
                ArrayFill(dist, long.MaxValue);
                ArrayFill(gen, -1);
                ArrayFill(parentEdge, -1);

                MinHeap heap = new MinHeap(1024);
                for (int i = 0; i < k2; i++) {
                    int v = oddVertices[i];
                    dist[v] = 0;
                    gen[v] = v;
                    heap.Push(v, 0);
                }

                List<CandidateEdge> candidates = new List<CandidateEdge>();

                while (heap.Size > 0) {
                    long d;
                    int u = heap.Pop(out d);
                    if (d > dist[u]) continue;

                    for (int e = head[u]; e != -1; e = next[e]) {
                        int v = to[e];
                        long nextDist = d + weight[e];

                        if (gen[v] == -1) {
                            dist[v] = nextDist;
                            gen[v] = gen[u];
                            parentEdge[v] = edgeId[e];
                            heap.Push(v, nextDist);
                        } else if (gen[v] != gen[u]) {
                            // Prune boundary edges: only add when u < v to avoid duplicates
                            if (u < v) {
                                CandidateEdge ce;
                                ce.U = gen[u];
                                ce.V = gen[v];
                                ce.Weight = dist[u] + dist[v] + weight[e];
                                ce.BoundaryU = u;
                                ce.BoundaryV = v;
                                ce.BoundaryEdgeId = edgeId[e];
                                candidates.Add(ce);
                            }
                        }
                    }
                }

                CandidateEdge[] candArray = candidates.ToArray();
                Array.Sort(candArray);

                bool[] isMatched = new bool[N];
                foreach (var ce in candArray) {
                    if (!isMatched[ce.U] && !isMatched[ce.V]) {
                        isMatched[ce.U] = true;
                        isMatched[ce.V] = true;

                        // Reconstruct path
                        int curr = ce.BoundaryU;
                        while (curr != ce.U) {
                            int e = parentEdge[curr];
                            duplicatedEdgeIds.Add(e);
                            curr = edgeU[e] == curr ? edgeV[e] : edgeU[e];
                        }
                        curr = ce.BoundaryV;
                        while (curr != ce.V) {
                            int e = parentEdge[curr];
                            duplicatedEdgeIds.Add(e);
                            curr = edgeU[e] == curr ? edgeV[e] : edgeU[e];
                        }
                        duplicatedEdgeIds.Add(ce.BoundaryEdgeId);
                    }
                }

                List<int> leftovers = new List<int>();
                for (int i = 0; i < k2; i++) {
                    int v = oddVertices[i];
                    if (!isMatched[v]) {
                        leftovers.Add(v);
                    }
                }

                if (leftovers.Count > 0) {
                    bool[] isLeftover = new bool[N];
                    foreach (int v in leftovers) {
                        isLeftover[v] = true;
                    }

                    long[] cleanDist = new long[N];
                    int[] cleanParentEdge = new int[N];
                    int[] visitedVersion = new int[N];
                    int currentVersion = 0;
                    MinHeap cleanHeap = new MinHeap(1024);

                    foreach (int x in leftovers) {
                        if (isMatched[x]) continue;

                        currentVersion++;
                        cleanDist[x] = 0;
                        cleanParentEdge[x] = -1;
                        visitedVersion[x] = currentVersion;
                        cleanHeap.Clear();
                        cleanHeap.Push(x, 0);

                        int target = -1;
                        while (cleanHeap.Size > 0) {
                            long d;
                            int u = cleanHeap.Pop(out d);
                            long uDist = visitedVersion[u] == currentVersion ? cleanDist[u] : long.MaxValue;
                            if (d > uDist) continue;

                            if (u != x && isLeftover[u] && !isMatched[u]) {
                                target = u;
                                break;
                            }

                            for (int e = head[u]; e != -1; e = next[e]) {
                                int v = to[e];
                                long nextDist = d + weight[e];
                                long vDist = visitedVersion[v] == currentVersion ? cleanDist[v] : long.MaxValue;

                                if (nextDist < vDist) {
                                    cleanDist[v] = nextDist;
                                    visitedVersion[v] = currentVersion;
                                    cleanParentEdge[v] = edgeId[e];
                                    cleanHeap.Push(v, nextDist);
                                }
                            }
                        }

                        if (target != -1) {
                            isMatched[x] = true;
                            isMatched[target] = true;
                            int curr = target;
                            while (curr != x) {
                                int e = cleanParentEdge[curr];
                                duplicatedEdgeIds.Add(e);
                                curr = edgeU[e] == curr ? edgeV[e] : edgeU[e];
                            }
                        }
                    }
                }
            }
        }

        // Build multigraph with even degrees
        int M_prime = M + duplicatedEdgeIds.Count;
        List<int>[] adj = new List<int>[N];
        for (int i = 0; i < N; i++) {
            adj[i] = new List<int>();
        }

        int[] multiU = new int[M_prime];
        int[] multiV = new int[M_prime];

        for (int i = 0; i < M; i++) {
            multiU[i] = edgeU[i];
            multiV[i] = edgeV[i];
            adj[edgeU[i]].Add(i);
            adj[edgeV[i]].Add(i);
        }

        for (int i = 0; i < duplicatedEdgeIds.Count; i++) {
            int origId = duplicatedEdgeIds[i];
            int newId = M + i;
            multiU[newId] = edgeU[origId];
            multiV[newId] = edgeV[origId];
            adj[edgeU[origId]].Add(newId);
            adj[edgeV[origId]].Add(newId);
        }

        // Hierholzer's Algorithm (Iterative, stack overflow safe)
        int[] multiHead = new int[N];
        bool[] used = new bool[M_prime];
        Stack<int> currPath = new Stack<int>();
        List<int> eulerCircuit = new List<int>();

        currPath.Push(0);
        while (currPath.Count > 0) {
            int u = currPath.Peek();
            int nextEdge = -1;
            while (multiHead[u] < adj[u].Count) {
                int e = adj[u][multiHead[u]];
                if (!used[e]) {
                    nextEdge = e;
                    break;
                }
                multiHead[u]++;
            }
            if (nextEdge != -1) {
                used[nextEdge] = true;
                multiHead[u]++;
                int v = multiU[nextEdge] == u ? multiV[nextEdge] : multiU[nextEdge];
                currPath.Push(v);
            } else {
                eulerCircuit.Add(u);
                currPath.Pop();
            }
        }

        eulerCircuit.Reverse();

        // High performance output
        using (FastWriter writer = new FastWriter()) {
            writer.WriteInt(eulerCircuit.Count);
            writer.WriteChar('\n');
            for (int i = 0; i < eulerCircuit.Count; i++) {
                writer.WriteInt(eulerCircuit[i]);
                if (i < eulerCircuit.Count - 1) {
                    writer.WriteChar(' ');
                }
            }
            writer.WriteChar('\n');
        }
    }
}
