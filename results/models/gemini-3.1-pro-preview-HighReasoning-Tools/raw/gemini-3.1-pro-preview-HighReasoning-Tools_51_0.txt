import sys
import os
import math

# Add parent directory to path so we can import 51.py and autopilot_sim.py
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

import importlib
five_one = importlib.import_module("51")

# Let's import the autopilot function from answer.txt
answer_file = os.path.join(os.path.dirname(os.path.abspath(__file__)), "answer.txt")
with open(answer_file, "r") as f:
    code = f.read()

ns = {}
exec(compile(code, '<autopilot>', 'exec'), ns)
autopilot_fn = ns['autopilot_step']

def run_debug(sc_idx):
    runner, sc = five_one._setup_scenario(sc_idx)
    steps = sc.get('steps', 1200)
    
    # Custom simulation loop to record/print details
    dt = 0.05
    for step in range(steps):
        state = runner.sensors.build_state(runner.truth)
        state['target_alt_ft'] = runner.target_alt_ft
        state['target_hdg_deg'] = runner.target_hdg_deg
        state['target_ias_kt'] = runner.target_ias_kt

        # Call autopilot
        try:
            controls = autopilot_fn(state, dt)
            if not isinstance(controls, dict):
                controls = {}
        except Exception as e:
            print(f"Autopilot exception at step {step}: {e}")
            controls = {}

        # Sanitize controls
        for k in ['elevator', 'aileron', 'rudder', 'elevator_trim', 'speedbrake']:
            if k in controls:
                controls[k] = max(-1.0, min(1.0, float(controls[k])))
        for k in ['throttle_1', 'throttle_2']:
            if k in controls:
                controls[k] = max(0.0, min(1.0, float(controls[k])))
        if 'flaps' in controls:
            controls['flaps'] = max(0, min(40, float(controls['flaps'])))

        # Step physics
        runner.truth = five_one.physics_step(runner.truth, controls, dt)
        
        # Check crash conditions
        if runner.truth['z'] <= 0 and step > 10:
            print("CRASHED!")
            break
            
        V = math.sqrt(runner.truth['u']**2 + runner.truth['v']**2 + runner.truth['w']**2)
        ias_kt = five_one.ias_from_tas(V, runner.truth['z']) * five_one.MS2KT
        alt_ft = runner.truth['z'] * five_one.M2FT
        pitch_deg = runner.truth['theta'] * five_one.RAD2DEG
        roll_deg = runner.truth['phi'] * five_one.RAD2DEG
        vs_fpm = state.get('vertical_speed_fpm', 0)
        
        # Record history
        if step % 20 == 0:
            runner.history.append({
                'time': runner.truth['time'],
                'alt_ft': alt_ft,
                'ias_kt': ias_kt,
                'hdg_deg': runner.truth['psi'] * five_one.RAD2DEG,
                'pitch_deg': pitch_deg,
                'roll_deg': roll_deg,
                'vs_fpm': vs_fpm,
            })
            
        # Print info in the last 15 seconds (last 300 steps)
        if step >= steps - 300 and step % 20 == 0:
            p_int = ns['_S'].get('pitch_int', 0.0)
            a_int = ns['_S'].get('alt_int', 0.0)
            v_int = ns['_S'].get('vs_int', 0.0)
            in_lo = ns['_S'].get('in_level_off', False)
            print(f"Step {step:4d} | Alt: {alt_ft:5.0f} | VS: {vs_fpm:5.0f} | Pitch: {pitch_deg:4.1f} | IAS: {ias_kt:5.1f} | in_lo: {in_lo} | p_int: {p_int:5.2f} | a_int: {a_int:5.1f} | v_int: {v_int:5.2f} | elev: {controls.get('elevator',0):5.2f}")
            
    score, summary = runner.score_recovery()
    print(f"Final Score: {score:.4f} | Summary: {summary}")

if __name__ == '__main__':
    sc_idx = int(sys.argv[1]) if len(sys.argv) > 1 else 10
    run_debug(sc_idx)
