import random
import time
import sys

def solve_exact_cover_set(U, sets):
    S = len(sets)
    element_sets = [[] for _ in range(U)]
    for s_idx, s in enumerate(sets):
        for e in s:
            element_sets[e].append(s_idx)
            
    uncovered_elements = set(range(U))
    set_active = [True] * S
    element_counts = [len(element_sets[e]) for e in range(U)]
    solution = []
    calls = 0
    
    def backtrack():
        nonlocal calls
        calls += 1
        if not uncovered_elements:
            return True
            
        best_e = -1
        min_count = S + 1
        for e in uncovered_elements:
            count = element_counts[e]
            if count < min_count:
                min_count = count
                best_e = e
                if min_count == 0:
                    return False
                if min_count == 1:
                    break
                    
        if best_e == -1:
            return True
            
        candidates = [s for s in element_sets[best_e] if set_active[s]]
        for s in candidates:
            solution.append(s)
            
            deactivated = []
            for e in sets[s]:
                for s_prime in element_sets[e]:
                    if set_active[s_prime]:
                        set_active[s_prime] = False
                        deactivated.append(s_prime)
                        for elem in sets[s_prime]:
                            element_counts[elem] -= 1
                            
            for e in sets[s]:
                uncovered_elements.remove(e)
                
            if backtrack():
                return True
                
            for e in sets[s]:
                uncovered_elements.add(e)
                
            for s_prime in deactivated:
                set_active[s_prime] = True
                for elem in sets[s_prime]:
                    element_counts[elem] += 1
                    
            solution.pop()
            
        return False

    if backtrack():
        return sorted(solution), calls
    else:
        return None, calls

def solve_exact_cover_swap(U, sets):
    S = len(sets)
    element_sets = [[] for _ in range(U)]
    for s_idx, s in enumerate(sets):
        for e in s:
            element_sets[e].append(s_idx)
            
    uncovered_list = list(range(U))
    uncovered_pos = list(range(U))
    num_uncovered = U
    
    set_active = [True] * S
    element_counts = [len(element_sets[e]) for e in range(U)]
    solution = []
    calls = 0
    
    def backtrack(num_uncovered):
        nonlocal calls
        calls += 1
        if num_uncovered == 0:
            return True
            
        best_e = -1
        min_count = S + 1
        for i in range(num_uncovered):
            e = uncovered_list[i]
            count = element_counts[e]
            if count < min_count:
                min_count = count
                best_e = e
                if min_count == 0:
                    return False
                if min_count == 1:
                    break
                    
        if best_e == -1:
            return True
            
        candidates = [s for s in element_sets[best_e] if set_active[s]]
        for s in candidates:
            solution.append(s)
            
            deactivated = []
            for e in sets[s]:
                for s_prime in element_sets[e]:
                    if set_active[s_prime]:
                        set_active[s_prime] = False
                        deactivated.append(s_prime)
                        for elem in sets[s_prime]:
                            element_counts[elem] -= 1
                            
            for e in sets[s]:
                pos = uncovered_pos[e]
                num_uncovered -= 1
                last_el = uncovered_list[num_uncovered]
                uncovered_list[pos] = last_el
                uncovered_pos[last_el] = pos
                
            if backtrack(num_uncovered):
                return True
                
            for e in sets[s]:
                uncovered_list[num_uncovered] = e
                uncovered_pos[e] = num_uncovered
                num_uncovered += 1
                
            for s_prime in deactivated:
                set_active[s_prime] = True
                for elem in sets[s_prime]:
                    element_counts[elem] += 1
                    
            solution.pop()
            
        return False

    if backtrack(num_uncovered):
        return sorted(solution), calls
    else:
        return None, calls

def generate_planted_instance(U, S, density=0.1):
    elements = list(range(U))
    random.shuffle(elements)
    
    num_sol_sets = random.randint(max(1, U // 10), max(2, U // 2))
    sol_sets = [[] for _ in range(num_sol_sets)]
    for i, e in enumerate(elements):
        sol_sets[i % num_sol_sets].append(e)
        
    sol_sets = [s for s in sol_sets if s]
    num_sol_sets = len(sol_sets)
    
    other_sets = []
    remaining = S - num_sol_sets
    for _ in range(remaining):
        size = max(1, int(random.gauss(U * density, U * density * 0.5)))
        size = min(U, max(1, size))
        s = random.sample(range(U), size)
        other_sets.append(s)
        
    all_sets = sol_sets + other_sets
    indices = list(range(len(all_sets)))
    random.shuffle(indices)
    
    shuffled_sets = [all_sets[i] for i in indices]
    return shuffled_sets

def main():
    U = 200
    S = 1000
    print(f"Generating instance with U={U}, S={S}...")
    sets = generate_planted_instance(U, S, density=0.08)
    
    print("Running set-based algorithm...")
    t0 = time.time()
    sol_set, calls_set = solve_exact_cover_set(U, sets)
    t1 = time.time()
    time_set = t1 - t0
    print(f"Set-based: {time_set:.6f} seconds, {calls_set} calls")
    
    print("Running swap-based algorithm...")
    t0 = time.time()
    sol_swap, calls_swap = solve_exact_cover_swap(U, sets)
    t1 = time.time()
    time_swap = t1 - t0
    print(f"Swap-based: {time_swap:.6f} seconds, {calls_swap} calls")
    
    assert sol_set == sol_swap, f"Solutions differ! Set: {sol_set}, Swap: {sol_swap}"
    print("Solutions match!")

if __name__ == "__main__":
    main()
