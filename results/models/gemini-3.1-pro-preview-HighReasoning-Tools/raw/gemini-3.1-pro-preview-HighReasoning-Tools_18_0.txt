// Fast integer input parser using MSVC's _getchar_nolock
// This avoids standard C++ overhead and is safe under Windows platforms.
inline bool read_int(int& val) {
    int c = _getchar_nolock();
    while (c != EOF && c <= ' ') {
        c = _getchar_nolock();
    }
    if (c == EOF) return false;
    val = 0;
    while (c >= '0' && c <= '9') {
        val = val * 10 + (c - '0');
        c = _getchar_nolock();
    }
    return true;
}

// Binary edge writer to store discarded edges on disk during the streaming pass
struct EdgeWriter {
    FILE* f;
    EdgeWriter(const char* filepath) {
        f = fopen(filepath, "wb");
        if (f) {
            setvbuf(f, NULL, _IOFBF, 1024 * 1024); // 1 MB buffer for fast writing
        }
    }
    void write(int u, int v) {
        if (!f) return;
        int buf[2] = {u, v};
        fwrite(buf, sizeof(int), 2, f);
    }
    void close() {
        if (f) {
            fclose(f);
            f = nullptr;
        }
    }
    ~EdgeWriter() {
        close();
    }
};

// Binary edge reader to scan discarded edges during the final cut-edge determination
struct EdgeReader {
    FILE* f;
    EdgeReader(const char* filepath) {
        f = fopen(filepath, "rb");
        if (f) {
            setvbuf(f, NULL, _IOFBF, 1024 * 1024); // 1 MB buffer for fast reading
        }
    }
    bool read(int& u, int& v) {
        if (!f) return false;
        int buf[2];
        if (fread(buf, sizeof(int), 2, f) == 2) {
            u = buf[0];
            v = buf[1];
            return true;
        }
        return false;
    }
    void close() {
        if (f) {
            fclose(f);
            f = nullptr;
        }
    }
    ~EdgeReader() {
        close();
    }
};

// Disjoint Set Union (DSU) with iterative find to avoid stack overflow on deep trees
struct DSU {
    vector<int> parent;
    void init(int n) {
        parent.resize(n);
        for (int i = 0; i < n; ++i) parent[i] = i;
    }
    int find(int i) {
        int root = i;
        while (parent[root] != root) {
            root = parent[root];
        }
        int curr = i;
        while (curr != root) {
            int nxt = parent[curr];
            parent[curr] = root;
            curr = nxt;
        }
        return root;
    }
    bool unite(int i, int j) {
        int root_i = find(i);
        int root_j = find(j);
        if (root_i != root_j) {
            parent[root_i] = root_j;
            return true;
        }
        return false;
    }
};

// Stoer-Wagner algorithm: deterministic exact global min-cut for small graphs (N <= 2000)
// Time Complexity: O(V^3) but optimized using cache-friendly contiguous loops and active lists
struct StoerWagner {
    int N;
    vector<vector<int>> adj;
    vector<vector<int>> merged_vertices;
    vector<int> active_list;
    vector<int> pos_in_active;

    StoerWagner(int n, const vector<pair<int, int>>& edges) : N(n), adj(n, vector<int>(n, 0)), merged_vertices(n), active_list(n), pos_in_active(n) {
        for (auto& edge : edges) {
            int u = edge.first;
            int v = edge.second;
            if (u != v) {
                adj[u][v]++;
                adj[v][u]++;
            }
        }
        for (int i = 0; i < N; ++i) {
            merged_vertices[i].push_back(i);
            active_list[i] = i;
            pos_in_active[i] = i;
        }
    }

    void deactivate(int u) {
        int idx = pos_in_active[u];
        int last_v = active_list.back();
        active_list[idx] = last_v;
        pos_in_active[last_v] = idx;
        active_list.pop_back();
    }

    pair<int, vector<int>> solve() {
        int best_cut = 1e9;
        vector<int> best_partition;

        vector<int> w(N, 0);
        vector<bool> visited(N, false);

        while (active_list.size() > 1) {
            int R = active_list.size();
            
            fill(visited.begin(), visited.end(), true);
            for (int v : active_list) {
                w[v] = 0;
                visited[v] = false;
            }

            int last = -1, prev_last = -1;

            for (int step = 0; step < R; ++step) {
                int max_w = -1;
                int u = -1;
                for (int v : active_list) {
                    if (!visited[v] && w[v] > max_w) {
                        max_w = w[v];
                        u = v;
                    }
                }

                if (u == -1) {
                    // Graph is disconnected: min cut size is 0
                    best_cut = 0;
                    best_partition.clear();
                    for (int v : active_list) {
                        if (visited[v]) {
                            best_partition.insert(best_partition.end(), merged_vertices[v].begin(), merged_vertices[v].end());
                        }
                    }
                    return {0, best_partition};
                }

                visited[u] = true;
                prev_last = last;
                last = u;

                const int* adj_u = adj[u].data();
                for (int v = 0; v < N; ++v) {
                    if (!visited[v]) {
                        w[v] += adj_u[v];
                    }
                }
            }

            int cut_size = 0;
            for (int v : active_list) {
                if (v != last) {
                    cut_size += adj[last][v];
                }
            }

            if (cut_size < best_cut) {
                best_cut = cut_size;
                best_partition = merged_vertices[last];
            }

            // Merge 'last' into 'prev_last'
            for (int v : active_list) {
                if (v != last && v != prev_last) {
                    adj[prev_last][v] += adj[last][v];
                    adj[v][prev_last] += adj[v][last];
                }
            }

            merged_vertices[prev_last].insert(
                merged_vertices[prev_last].end(),
                merged_vertices[last].begin(),
                merged_vertices[last].end()
            );

            deactivate(last);
        }

        return {best_cut, best_partition};
    }
};

// Dinic's Max Flow algorithm on the certificate graph
// Equipped with DFS pruning optimization to guarantee optimal scaling
struct Dinic {
    struct Edge {
        int to;
        int cap;
        int flow;
        int rev;
    };

    int n;
    vector<vector<Edge>> adj;
    vector<int> level;
    vector<int> ptr;

    Dinic(int n) : n(n), adj(n), level(n), ptr(n) {}

    void add_undirected_edge(int u, int v, int cap = 1) {
        adj[u].push_back({v, cap, 0, (int)adj[v].size()});
        adj[v].push_back({u, cap, 0, (int)adj[u].size() - 1});
    }

    void reset_flow() {
        for (int i = 0; i < n; ++i) {
            for (auto& edge : adj[i]) {
                edge.flow = 0;
            }
        }
    }

    bool bfs(int s, int t) {
        fill(level.begin(), level.end(), -1);
        level[s] = 0;
        queue<int> q;
        q.push(s);
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            for (auto& edge : adj[v]) {
                if (edge.cap - edge.flow > 0 && level[edge.to] == -1) {
                    level[edge.to] = level[v] + 1;
                    q.push(edge.to);
                }
            }
        }
        return level[t] != -1;
    }

    int dfs(int v, int t, int pushed) {
        if (pushed == 0) return 0;
        if (v == t) return pushed;
        for (int& cid = ptr[v]; cid < adj[v].size(); ++cid) {
            auto& edge = adj[v][cid];
            int tr = edge.to;
            if (level[v] + 1 != level[tr] || edge.cap - edge.flow == 0) continue;
            int tr_pushed = dfs(tr, t, min(pushed, edge.cap - edge.flow));
            if (tr_pushed == 0) {
                level[tr] = -1; // Prune dead-ends in current BFS level
                continue;
            }
            edge.flow += tr_pushed;
            adj[tr][edge.rev].flow -= tr_pushed;
            return tr_pushed;
        }
        return 0;
    }

    int max_flow(int s, int t, int limit = 1e9) {
        int flow = 0;
        while (flow < limit && bfs(s, t)) {
            fill(ptr.begin(), ptr.end(), 0);
            while (int pushed = dfs(s, t, limit - flow)) {
                flow += pushed;
                if (flow >= limit) break;
            }
        }
        return flow;
    }

    vector<bool> get_cut_partition(int s) {
        vector<bool> visited(n, false);
        visited[s] = true;
        queue<int> q;
        q.push(s);
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            for (auto& edge : adj[v]) {
                if (edge.cap - edge.flow > 0 && !visited[edge.to]) {
                    visited[edge.to] = true;
                    q.push(edge.to);
                }
            }
        }
        return visited;
    }
};

int main() {
    // Fast standard streams synchronization
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int N, M;
    if (!read_int(N) || !read_int(M)) {
        return 0;
    }

    // --- ADAPTIVE LOGIC BRANCH 1: Small/Medium Scale ---
    if (N <= 2000) {
        vector<pair<int, int>> original_edges(M);
        for (int i = 0; i < M; ++i) {
            read_int(original_edges[i].first);
            read_int(original_edges[i].second);
        }

        StoerWagner sw(N, original_edges);
        auto [best_cut, best_partition] = sw.solve();

        vector<bool> in_partition(N, false);
        for (int v : best_partition) {
            in_partition[v] = true;
        }

        vector<pair<int, int>> cut_edges;
        for (auto& edge : original_edges) {
            if (in_partition[edge.first] != in_partition[edge.second]) {
                cut_edges.push_back(edge);
            }
        }

        cout << cut_edges.size() << "\n";
        for (auto& edge : cut_edges) {
            cout << edge.first << " " << edge.second << "\n";
        }
        return 0;
    }

    // --- ADAPTIVE LOGIC BRANCH 2: Large/Massive Scale ---
    // Uses streaming with DSU forest certificate to preserve connectivity certificates
    // and stores discarded edges on disk to support graphs of multi-GB sizes.
    const int k = 20; // 20 forests preserve connectivity cuts up to size 20
    vector<DSU> dsu(k);
    for (int i = 0; i < k; ++i) {
        dsu[i].init(N);
    }

    vector<pair<int, int>> certificate_edges;
    vector<int> degree(N, 0);

    const char* temp_filename = "temp_discarded_edges.bin";
    EdgeWriter writer(temp_filename);

    for (int i = 0; i < M; ++i) {
        int u, v;
        read_int(u);
        read_int(v);
        
        degree[u]++;
        degree[v]++;

        bool kept = false;
        for (int j = 0; j < k; ++j) {
            if (dsu[j].unite(u, v)) {
                kept = true;
                break;
            }
        }

        if (kept) {
            certificate_edges.push_back({u, v});
        } else {
            writer.write(u, v);
        }
    }
    writer.close();

    // Find the minimum degree vertex
    int min_deg_vertex = 0;
    for (int i = 1; i < N; ++i) {
        if (degree[i] < degree[min_deg_vertex]) {
            min_deg_vertex = i;
        }
    }
    int delta = degree[min_deg_vertex];

    // Initial min-cut is the cut separating the minimum degree vertex
    int best_cut = delta;
    vector<bool> best_partition_mask(N, false);
    best_partition_mask[min_deg_vertex] = true;

    // Early exit if the minimum cut is already 1 (cannot go lower for connected graphs)
    if (best_cut > 1) {
        Dinic solver(N);
        for (auto& edge : certificate_edges) {
            solver.add_undirected_edge(edge.first, edge.second, 1);
        }

        // Candidates for source vertices:
        // 1. Up to 10 vertices with the lowest degrees (small side of cut tends to have lower degrees)
        // 2. Up to 10 random vertices (to capture balanced cuts)
        vector<pair<int, int>> degree_vertex_pairs(N);
        for (int i = 0; i < N; ++i) {
            degree_vertex_pairs[i] = {degree[i], i};
        }
        sort(degree_vertex_pairs.begin(), degree_vertex_pairs.end());

        vector<int> candidates;
        for (int i = 0; i < min(N, 10); ++i) {
            candidates.push_back(degree_vertex_pairs[i].second);
        }

        std::mt19937 rng(1337);
        for (int i = 0; i < min(N, 10); ++i) {
            candidates.push_back(rng() % N);
        }

        // Deduplicate candidate sources
        sort(candidates.begin(), candidates.end());
        candidates.erase(unique(candidates.begin(), candidates.end()), candidates.end());

        for (int s : candidates) {
            int successful_runs = 0;
            int attempts = 0;
            // 5 random targets is mathematically robust (prob. of missing balanced cut < 2^-5 = 3%)
            while (successful_runs < 5 && attempts < 100) {
                attempts++;
                int t = rng() % N;
                if (t == s) continue;

                solver.reset_flow();
                int flow = solver.max_flow(s, t, best_cut);
                if (flow < best_cut) {
                    best_cut = flow;
                    best_partition_mask = solver.get_cut_partition(s);
                    if (best_cut == 1) {
                        break;
                    }
                }
                successful_runs++;
            }
            if (best_cut == 1) {
                break;
            }
        }
    }

    // Determine and output the cut edges
    vector<pair<int, int>> cut_edges;
    for (auto& edge : certificate_edges) {
        if (best_partition_mask[edge.first] != best_partition_mask[edge.second]) {
            cut_edges.push_back(edge);
        }
    }

    EdgeReader disk_reader(temp_filename);
    int u, v;
    while (disk_reader.read(u, v)) {
        if (best_partition_mask[u] != best_partition_mask[v]) {
            cut_edges.push_back({u, v});
        }
    }
    disk_reader.close();

    // Remove the temporary binary edge file
    remove(temp_filename);

    cout << cut_edges.size() << "\n";
    for (auto& edge : cut_edges) {
        cout << edge.first << " " << edge.second << "\n";
    }

    return 0;
}
