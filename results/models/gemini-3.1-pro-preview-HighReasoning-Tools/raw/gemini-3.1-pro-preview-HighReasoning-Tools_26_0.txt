use std::io::{self, Read, Write, BufReader, BufWriter};
use std::thread;
use std::time::Instant;

// Fast LCG Random Number Generator
struct SimpleRng {
    state: u64,
}

impl SimpleRng {
    fn new(seed: u64) -> Self {
        Self { state: seed }
    }
    
    fn next(&mut self) -> usize {
        self.state = self.state.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
        self.state as usize
    }
}

// Bounding box distance calculation helper
#[inline(always)]
fn sq_dist(a: &[f32; 3], b: &[f32; 3]) -> f32 {
    let dx = a[0] - b[0];
    let dy = a[1] - b[1];
    let dz = a[2] - b[2];
    dx * dx + dy * dy + dz * dz
}

// Flat KD-tree Node
struct KdNode {
    point: [f32; 3],
    id: u32,
    left: i32,
    right: i32,
    axis: u8,
}

// KD-tree for fast 3D nearest-neighbor search
struct KdTree {
    nodes: Vec<KdNode>,
    root: i32,
}

impl KdTree {
    fn build(centroids: &[[f32; 3]]) -> Self {
        let mut nodes = Vec::with_capacity(centroids.len());
        let mut indices: Vec<usize> = (0..centroids.len()).collect();
        let root = Self::build_recursive(&mut indices, centroids, &mut nodes);
        Self { nodes, root }
    }
    
    fn build_recursive(
        indices: &mut [usize],
        centroids: &[[f32; 3]],
        nodes: &mut Vec<KdNode>,
    ) -> i32 {
        if indices.is_empty() {
            return -1;
        }
        
        // Find split axis based on the dimension with the largest spread
        let mut min = [f32::INFINITY; 3];
        let mut max = [f32::NEG_INFINITY; 3];
        for &idx in indices.iter() {
            let p = centroids[idx];
            for i in 0..3 {
                if p[i] < min[i] { min[i] = p[i]; }
                if p[i] > max[i] { max[i] = p[i]; }
            }
        }
        
        let mut axis = 0;
        let mut max_spread = max[0] - min[0];
        for i in 1..3 {
            let spread = max[i] - min[i];
            if spread > max_spread {
                max_spread = spread;
                axis = i;
            }
        }
        
        let mid = indices.len() / 2;
        indices.select_nth_unstable_by(mid, |&a, &b| {
            centroids[a][axis].partial_cmp(&centroids[b][axis]).unwrap_or(std::cmp::Ordering::Equal)
        });
        
        let centroid_idx = indices[mid];
        let node_idx = nodes.len() as i32;
        nodes.push(KdNode {
            point: centroids[centroid_idx],
            id: centroid_idx as u32,
            left: -1,
            right: -1,
            axis: axis as u8,
        });
        
        let left = Self::build_recursive(&mut indices[..mid], centroids, nodes);
        let right = Self::build_recursive(&mut indices[mid + 1..], centroids, nodes);
        
        nodes[node_idx as usize].left = left;
        nodes[node_idx as usize].right = right;
        
        node_idx
    }
    
    #[inline(always)]
    fn find_nearest(&self, query: &[f32; 3]) -> (u32, f32) {
        let mut best_id = 0;
        let mut best_dist = f32::INFINITY;
        self.nearest_neighbor(self.root, query, &mut best_id, &mut best_dist);
        (best_id, best_dist)
    }
    
    fn nearest_neighbor(
        &self,
        root: i32,
        query: &[f32; 3],
        best_id: &mut u32,
        best_dist: &mut f32,
    ) {
        if root == -1 {
            return;
        }
        let node = &self.nodes[root as usize];
        let dist = sq_dist(query, &node.point);
        if dist < *best_dist {
            *best_dist = dist;
            *best_id = node.id;
        }
        
        let axis = node.axis as usize;
        let diff = query[axis] - node.point[axis];
        
        let (first, second) = if diff < 0.0 {
            (node.left, node.right)
        } else {
            (node.right, node.left)
        };
        
        self.nearest_neighbor(first, query, best_id, best_dist);
        
        if diff * diff < *best_dist {
            self.nearest_neighbor(second, query, best_id, best_dist);
        }
    }
}

// Randomly choose initial centroids from sample using candidate-based k-means++
fn initialize_centroids(sample: &[[f32; 3]], n_clusters: usize, rng: &mut SimpleRng) -> Vec<[f32; 3]> {
    let mut centroids = Vec::with_capacity(n_clusters);
    if sample.is_empty() {
        for _ in 0..n_clusters {
            centroids.push([0.0, 0.0, 0.0]);
        }
        return centroids;
    }
    
    if sample.len() <= n_clusters {
        for i in 0..n_clusters {
            centroids.push(sample[i % sample.len()]);
        }
        return centroids;
    }
    
    // Choose a candidate subset from the sample
    let candidate_size = std::cmp::min(sample.len(), 2 * n_clusters);
    let mut candidates = Vec::with_capacity(candidate_size);
    let mut indices: Vec<usize> = (0..sample.len()).collect();
    for i in (1..indices.len()).rev() {
        let j = rng.next() % (i + 1);
        indices.swap(i, j);
    }
    for i in 0..candidate_size {
        candidates.push(sample[indices[i]]);
    }
    
    // Run k-means++ on candidates
    // Pick first centroid randomly
    let first_idx = rng.next() % candidate_size;
    centroids.push(candidates[first_idx]);
    
    // Maintain the minimum distance squared from each candidate to any chosen centroid
    let mut min_dists = vec![f32::INFINITY; candidate_size];
    for i in 0..candidate_size {
        min_dists[i] = sq_dist(&candidates[i], &centroids[0]);
    }
    
    for _ in 1..n_clusters {
        let mut sum = 0.0f64;
        for &d in min_dists.iter() {
            sum += d as f64;
        }
        
        if sum == 0.0 {
            // All remaining candidates are identical to chosen centroids, just pick a random candidate
            let idx = rng.next() % candidate_size;
            centroids.push(candidates[idx]);
            continue;
        }
        
        // Generate random float between 0 and sum
        let random_fraction = (rng.next() % 1000000) as f64 / 1000000.0;
        let mut r = random_fraction * sum;
        let mut chosen_idx = 0;
        for (i, &d) in min_dists.iter().enumerate() {
            r -= d as f64;
            if r <= 0.0 {
                chosen_idx = i;
                break;
            }
        }
        
        centroids.push(candidates[chosen_idx]);
        
        // Update minimum distances
        let new_centroid = &centroids[centroids.len() - 1];
        for i in 0..candidate_size {
            let d = sq_dist(&candidates[i], new_centroid);
            if d < min_dists[i] {
                min_dists[i] = d;
            }
        }
    }
    
    centroids
}

// Parallel assignment step for standard k-means iteration
fn k_means_iteration(
    points: &[[f32; 3]],
    tree: &KdTree,
    n_clusters: usize,
    num_threads: usize,
) -> (Vec<[f32; 3]>, Vec<usize>) {
    let chunk_size = (points.len() + num_threads - 1) / num_threads;
    
    thread::scope(|s| {
        let mut handles = Vec::with_capacity(num_threads);
        for chunk in points.chunks(chunk_size) {
            handles.push(s.spawn(move || {
                let mut local_sums = vec![[0.0f32; 3]; n_clusters];
                let mut local_counts = vec![0usize; n_clusters];
                for pt in chunk {
                    let (centroid_id, _) = tree.find_nearest(pt);
                    let c = centroid_id as usize;
                    local_sums[c][0] += pt[0];
                    local_sums[c][1] += pt[1];
                    local_sums[c][2] += pt[2];
                    local_counts[c] += 1;
                }
                (local_sums, local_counts)
            }));
        }
        
        let mut global_sums = vec![[0.0f32; 3]; n_clusters];
        let mut global_counts = vec![0usize; n_clusters];
        for handle in handles {
            let (local_sums, local_counts) = handle.join().unwrap();
            for i in 0..n_clusters {
                global_sums[i][0] += local_sums[i][0];
                global_sums[i][1] += local_sums[i][1];
                global_sums[i][2] += local_sums[i][2];
                global_counts[i] += local_counts[i];
            }
        }
        (global_sums, global_counts)
    })
}

// Train centroids on the sample using accelerated standard k-means
fn train_centroids(
    sample: &[[f32; 3]],
    n_clusters: usize,
    num_threads: usize,
    rng: &mut SimpleRng,
) -> (Vec<[f32; 3]>, Vec<usize>) {
    let mut centroids = initialize_centroids(sample, n_clusters, rng);
    let mut counts = vec![0usize; n_clusters];
    
    // Calculate bounding box and coordinate spread for convergence check
    let mut min_val = [f32::INFINITY; 3];
    let mut max_val = [f32::NEG_INFINITY; 3];
    for pt in sample {
        for i in 0..3 {
            if pt[i] < min_val[i] { min_val[i] = pt[i]; }
            if pt[i] > max_val[i] { max_val[i] = pt[i]; }
        }
    }
    let dx = max_val[0] - min_val[0];
    let dy = max_val[1] - min_val[1];
    let dz = max_val[2] - min_val[2];
    let spread_sq = dx * dx + dy * dy + dz * dz;
    let tolerance = 1e-6 * spread_sq;
    
    let max_iterations = 25;
    for iter in 0..max_iterations {
        let tree = KdTree::build(&centroids);
        let (sums, new_counts) = k_means_iteration(sample, &tree, n_clusters, num_threads);
        
        let mut max_shift = 0.0f32;
        let mut updated_centroids = centroids.clone();
        
        for i in 0..n_clusters {
            if new_counts[i] > 0 {
                let new_centroid = [
                    sums[i][0] / new_counts[i] as f32,
                    sums[i][1] / new_counts[i] as f32,
                    sums[i][2] / new_counts[i] as f32,
                ];
                let shift = sq_dist(&centroids[i], &new_centroid);
                if shift > max_shift {
                    max_shift = shift;
                }
                updated_centroids[i] = new_centroid;
            } else {
                // Reinitialize empty cluster with a random sample point
                if !sample.is_empty() {
                    let random_pt = sample[rng.next() % sample.len()];
                    updated_centroids[i] = random_pt;
                }
                max_shift = f32::INFINITY;
            }
        }
        
        centroids = updated_centroids;
        counts = new_counts;
        
        if max_shift < tolerance {
            eprintln!("K-means converged at iteration {}", iter + 1);
            break;
        }
    }
    (centroids, counts)
}

// Parallel assignment for a batch during streaming
fn parallel_assign_batch(
    batch: &[[f32; 3]],
    tree: &KdTree,
    n_clusters: usize,
    num_threads: usize,
) -> (Vec<u32>, Vec<[f32; 3]>, Vec<usize>) {
    let chunk_size = (batch.len() + num_threads - 1) / num_threads;
    
    thread::scope(|s| {
        let mut handles = Vec::with_capacity(num_threads);
        for chunk in batch.chunks(chunk_size) {
            handles.push(s.spawn(move || {
                let mut assignments = Vec::with_capacity(chunk.len());
                let mut local_sums = vec![[0.0f32; 3]; n_clusters];
                let mut local_counts = vec![0usize; n_clusters];
                for pt in chunk {
                    let (id, _) = tree.find_nearest(pt);
                    assignments.push(id);
                    let c = id as usize;
                    local_sums[c][0] += pt[0];
                    local_sums[c][1] += pt[1];
                    local_sums[c][2] += pt[2];
                    local_counts[c] += 1;
                }
                (assignments, local_sums, local_counts)
            }));
        }
        
        let mut global_assignments = Vec::with_capacity(batch.len());
        let mut global_sums = vec![[0.0f32; 3]; n_clusters];
        let mut global_counts = vec![0usize; n_clusters];
        
        for handle in handles {
            let (assignments, local_sums, local_counts) = handle.join().unwrap();
            global_assignments.extend(assignments);
            for i in 0..n_clusters {
                global_sums[i][0] += local_sums[i][0];
                global_sums[i][1] += local_sums[i][1];
                global_sums[i][2] += local_sums[i][2];
                global_counts[i] += local_counts[i];
            }
        }
        (global_assignments, global_sums, global_counts)
    })
}

// Read header in text format
fn read_header<R: Read>(reader: &mut R) -> io::Result<(usize, usize)> {
    let mut header = String::new();
    let mut byte = [0u8; 1];
    while reader.read_exact(&mut byte).is_ok() {
        if byte[0] == b'\n' {
            break;
        }
        header.push(byte[0] as char);
    }
    let parts: Vec<&str> = header.split_whitespace().collect();
    if parts.len() < 2 {
        return Err(io::Error::new(io::ErrorKind::InvalidData, "Invalid header format"));
    }
    let num_points = parts[0].parse::<usize>().map_err(|e| io::Error::new(io::ErrorKind::InvalidData, e))?;
    let num_clusters = parts[1].parse::<usize>().map_err(|e| io::Error::new(io::ErrorKind::InvalidData, e))?;
    Ok((num_points, num_clusters))
}

// Fast read block helper
fn read_points_chunk<R: Read>(
    reader: &mut R,
    limit: usize,
) -> io::Result<Vec<[f32; 3]>> {
    let mut points = Vec::with_capacity(limit);
    let chunk_size = 65536;
    let mut byte_buf = vec![0u8; chunk_size * 12];
    
    let mut remaining = limit;
    while remaining > 0 {
        let to_read = std::cmp::min(remaining, chunk_size);
        let bytes_to_read = to_read * 12;
        
        let mut total_read = 0;
        while total_read < bytes_to_read {
            match reader.read(&mut byte_buf[total_read..bytes_to_read]) {
                Ok(0) => break,
                Ok(n) => total_read += n,
                Err(ref e) if e.kind() == io::ErrorKind::Interrupted => {}
                Err(e) => return Err(e),
            }
        }
        
        let points_read = total_read / 12;
        if points_read == 0 {
            break;
        }
        
        for i in 0..points_read {
            let offset = i * 12;
            let x = f32::from_le_bytes([
                byte_buf[offset],
                byte_buf[offset + 1],
                byte_buf[offset + 2],
                byte_buf[offset + 3],
            ]);
            let y = f32::from_le_bytes([
                byte_buf[offset + 4],
                byte_buf[offset + 5],
                byte_buf[offset + 6],
                byte_buf[offset + 7],
            ]);
            let z = f32::from_le_bytes([
                byte_buf[offset + 8],
                byte_buf[offset + 9],
                byte_buf[offset + 10],
                byte_buf[offset + 11],
            ]);
            points.push([x, y, z]);
        }
        
        if points_read < to_read {
            break;
        }
        remaining -= points_read;
    }
    
    Ok(points)
}

fn main() -> io::Result<()> {
    let mut reader = BufReader::new(io::stdin());
    
    // Read dataset header
    let (num_points, n_clusters) = read_header(&mut reader)?;
    if num_points == 0 {
        return Ok(());
    }
    
    let n_clusters = if n_clusters == 0 { 1 } else { n_clusters };
    
    let num_threads = std::thread::available_parallelism()
        .map(|n| n.get())
        .unwrap_or(4);
        
    eprintln!("Processing {} points, {} clusters, using {} threads", num_points, n_clusters, num_threads);
    
    let mut rng = SimpleRng::new(1337);
    
    let in_memory_threshold = 5_000_000;
    
    if num_points <= in_memory_threshold {
        // --- IN-MEMORY STRATEGY ---
        let start_time = Instant::now();
        let points = read_points_chunk(&mut reader, num_points)?;
        eprintln!("Loaded points in {:?}", start_time.elapsed());
        
        let train_start = Instant::now();
        let (centroids, _) = train_centroids(&points, n_clusters, num_threads, &mut rng);
        eprintln!("Trained centroids in {:?}", train_start.elapsed());
        
        let write_start = Instant::now();
        let tree = KdTree::build(&centroids);
        let stdout = io::stdout();
        let mut writer = BufWriter::new(stdout.lock());
        
        let chunk_size = 65536;
        for chunk in points.chunks(chunk_size) {
            let mut output_bytes = Vec::with_capacity(chunk.len() * 4);
            for pt in chunk {
                let (id, _) = tree.find_nearest(pt);
                output_bytes.extend_from_slice(&id.to_le_bytes());
            }
            writer.write_all(&output_bytes)?;
        }
        writer.flush()?;
        eprintln!("Wrote output in {:?}", write_start.elapsed());
    } else {
        // --- STREAMING STRATEGY ---
        // Dynamically choose sample size for training
        let sample_size = std::cmp::min(num_points, std::cmp::max(200 * n_clusters, 1_000_000));
        eprintln!("Streaming mode: training centroids on a sample of size {}", sample_size);
        
        let start_time = Instant::now();
        let sample = read_points_chunk(&mut reader, sample_size)?;
        eprintln!("Loaded training sample in {:?}", start_time.elapsed());
        
        let train_start = Instant::now();
        let (mut centroids, counts) = train_centroids(&sample, n_clusters, num_threads, &mut rng);
        eprintln!("Trained centroids in {:?}", train_start.elapsed());
        
        let mut tree = KdTree::build(&centroids);
        let stdout = io::stdout();
        let mut writer = BufWriter::new(stdout.lock());
        
        // Write assignments for the initial sample points first
        let write_start = Instant::now();
        let chunk_size = 65536;
        for chunk in sample.chunks(chunk_size) {
            let mut output_bytes = Vec::with_capacity(chunk.len() * 4);
            for pt in chunk {
                let (id, _) = tree.find_nearest(pt);
                output_bytes.extend_from_slice(&id.to_le_bytes());
            }
            writer.write_all(&output_bytes)?;
        }
        writer.flush()?;
        drop(sample); // Free memory immediately
        
        // Stream the remaining points in batches
        let batch_size = if num_points > 100_000_000 { 1_048_576 } else { 262_144 };
        let mut weights: Vec<f32> = counts.into_iter().map(|c| c as f32).collect();
        
        let mut remaining = num_points - sample_size;
        while remaining > 0 {
            let to_read = std::cmp::min(remaining, batch_size);
            let batch = read_points_chunk(&mut reader, to_read)?;
            if batch.is_empty() {
                break;
            }
            
            // Assign points in parallel and get statistics for updating
            let (assignments, local_sums, local_counts) = parallel_assign_batch(&batch, &tree, n_clusters, num_threads);
            
            // Online update of centroids (MacQueen's sequential k-means)
            for i in 0..n_clusters {
                let k = local_counts[i];
                if k > 0 {
                    let w = weights[i];
                    let new_w = w + k as f32;
                    centroids[i][0] = (w * centroids[i][0] + local_sums[i][0]) / new_w;
                    centroids[i][1] = (w * centroids[i][1] + local_sums[i][1]) / new_w;
                    centroids[i][2] = (w * centroids[i][2] + local_sums[i][2]) / new_w;
                    weights[i] = new_w;
                }
            }
            
            // Rebuild tree with updated centroids
            tree = KdTree::build(&centroids);
            
            // Output assignments for the batch
            let mut output_bytes = Vec::with_capacity(assignments.len() * 4);
            for id in assignments {
                output_bytes.extend_from_slice(&id.to_le_bytes());
            }
            writer.write_all(&output_bytes)?;
            
            remaining -= batch.len();
        }
        writer.flush()?;
        eprintln!("Streaming and assignments complete in {:?}", write_start.elapsed());
    }
    
    Ok(())
}
