// Define scale-independent global epsilon
double G_EPS = 1e-9;

struct Point3D {
    double x, y, z;
    Point3D operator+(const Point3D& o) const { return {x+o.x, y+o.y, z+o.z}; }
    Point3D operator-(const Point3D& o) const { return {x-o.x, y-o.y, z-o.z}; }
    Point3D operator*(double s) const { return {x*s, y*s, z*s}; }
    double dot(const Point3D& o) const { return x*o.x + y*o.y + z*o.z; }
    Point3D cross(const Point3D& o) const { return {y*o.z - z*o.y, z*o.x - x*o.z, x*o.y - y*o.x}; }
    double norm() const { return std::sqrt(x*x + y*y + z*z); }
    double dist(const Point3D& o) const { return (*this - o).norm(); }
};

struct Point2D {
    double x, y;
    Point2D operator-(const Point2D& o) const { return {x-o.x, y-o.y}; }
    double dot(const Point2D& o) const { return x*o.x + y*o.y; }
    double cross(const Point2D& o) const { return x*o.y - y*o.x; }
    double norm2() const { return x*x + y*y; }
    double norm() const { return std::sqrt(x*x + y*y); }
};

double ccw2d(const Point2D& A, const Point2D& B, const Point2D& C) {
    return (B.x - A.x) * (C.y - A.y) - (B.y - A.y) * (C.x - A.x);
}

bool point_in_segment_interior_2d(const Point2D& Q, const Point2D& A, const Point2D& B, double tol = -1.0) {
    if (tol < 0) tol = G_EPS;
    double cp = ccw2d(A, B, Q);
    double len = (B - A).norm();
    if (len < 1e-12) return false;
    if (std::abs(cp) > tol * len) return false;
    Point2D AB = B - A;
    double dot = (Q - A).dot(AB);
    double len2 = AB.norm2();
    return (dot > tol * len && dot < len2 - tol * len);
}

bool segments_intersect_proper_2d(const Point2D& A, const Point2D& B, const Point2D& C, const Point2D& D) {
    double cp1 = ccw2d(A, B, C);
    double cp2 = ccw2d(A, B, D);
    double cp3 = ccw2d(C, D, A);
    double cp4 = ccw2d(C, D, B);
    if (cp1 * cp2 < -G_EPS * G_EPS && cp3 * cp4 < -G_EPS * G_EPS) {
        return true;
    }
    return false;
}

double dist_to_segment(const Point3D& P, const Point3D& A, const Point3D& B) {
    Point3D AB = B - A;
    double len2 = AB.dot(AB);
    if (len2 < 1e-20) return P.dist(A);
    double t = (P - A).dot(AB) / len2;
    t = std::max(0.0, std::min(1.0, t));
    Point3D proj = A + AB * t;
    return P.dist(proj);
}

bool on_edge(const Point3D& P, const Point3D& A, const Point3D& B) {
    Point3D AB = B - A;
    double len = AB.norm();
    if (len < G_EPS) return false;
    double d = dist_to_segment(P, A, B);
    if (d > G_EPS * 20.0) return false;
    double t = (P - A).dot(AB) / (len * len);
    return (t > 1e-7 && t < 1.0 - 1e-7);
}

struct AABB {
    Point3D min_p, max_p;
    AABB() {
        min_p = {1e18, 1e18, 1e18};
        max_p = {-1e18, -1e18, -1e18};
    }
    void merge(const Point3D& p) {
        min_p.x = std::min(min_p.x, p.x);
        min_p.y = std::min(min_p.y, p.y);
        min_p.z = std::min(min_p.z, p.z);
        max_p.x = std::max(max_p.x, p.x);
        max_p.y = std::max(max_p.y, p.y);
        max_p.z = std::max(max_p.z, p.z);
    }
    void merge(const AABB& o) {
        min_p.x = std::min(min_p.x, o.min_p.x);
        min_p.y = std::min(min_p.y, o.min_p.y);
        min_p.z = std::min(min_p.z, o.min_p.z);
        max_p.x = std::max(max_p.x, o.max_p.x);
        max_p.y = std::max(max_p.y, o.max_p.y);
        max_p.z = std::max(max_p.z, o.max_p.z);
    }
    bool intersects(const AABB& o) const {
        return min_p.x <= o.max_p.x && max_p.x >= o.min_p.x &&
               min_p.y <= o.max_p.y && max_p.y >= o.min_p.y &&
               min_p.z <= o.max_p.z && max_p.z >= o.min_p.z;
    }
    bool intersects_ray(const Point3D& origin, const Point3D& dir) const {
        double tmin = -1e18, tmax = 1e18;
        auto check_axis = [&](double min_val, double max_val, double orig, double d) {
            if (std::abs(d) < 1e-12) {
                if (orig < min_val || orig > max_val) return false;
                return true;
            }
            double t1 = (min_val - orig) / d;
            double t2 = (max_val - orig) / d;
            if (t1 > t2) std::swap(t1, t2);
            tmin = std::max(tmin, t1);
            tmax = std::min(tmax, t2);
            return true;
        };
        if (!check_axis(min_p.x, max_p.x, origin.x, dir.x)) return false;
        if (!check_axis(min_p.y, max_p.y, origin.y, dir.y)) return false;
        if (!check_axis(min_p.z, max_p.z, origin.z, dir.z)) return false;
        return tmax >= tmin && tmax >= 0.0;
    }
};

struct Mesh {
    std::vector<Point3D> vertices;
    std::vector<std::vector<int>> vertex_int_idx;
    std::vector<std::vector<int>> faces;
};

struct AABBNode {
    AABB bbox;
    int left = -1;
    int right = -1;
};

class AABBTree {
public:
    std::vector<AABBNode> nodes;
    int root = -1;
    const Mesh& mesh;
    AABBTree(const Mesh& m) : mesh(m) {
        int n_faces = mesh.faces.size();
        if (n_faces == 0) return;
        std::vector<int> face_indices(n_faces);
        std::iota(face_indices.begin(), face_indices.end(), 0);
        std::vector<AABB> face_bboxes(n_faces);
        for (int i = 0; i < n_faces; ++i) {
            face_bboxes[i] = get_face_aabb(i);
        }
        nodes.reserve(n_faces * 2);
        root = build_recursive(face_indices, 0, n_faces, face_bboxes);
    }
    AABB get_face_aabb(int fid) const {
        AABB box;
        box.merge(mesh.vertices[mesh.faces[fid][0]]);
        box.merge(mesh.vertices[mesh.faces[fid][1]]);
        box.merge(mesh.vertices[mesh.faces[fid][2]]);
        return box;
    }
    int build_recursive(std::vector<int>& indices, int start, int end, const std::vector<AABB>& bboxes) {
        int count = end - start;
        int node_idx = nodes.size();
        nodes.push_back(AABBNode());
        AABB box;
        for (int i = start; i < end; ++i) {
            box.merge(bboxes[indices[i]]);
        }
        nodes[node_idx].bbox = box;
        if (count == 1) {
            nodes[node_idx].left = indices[start];
            nodes[node_idx].right = -1;
            return node_idx;
        }
        Point3D size = box.max_p - box.min_p;
        int axis = 0;
        if (size.y > size.x && size.y > size.z) axis = 1;
        else if (size.z > size.x && size.z > size.y) axis = 2;
        auto get_center = [&](int fid) {
            const AABB& b = bboxes[fid];
            if (axis == 0) return (b.min_p.x + b.max_p.x) * 0.5;
            if (axis == 1) return (b.min_p.y + b.max_p.y) * 0.5;
            return (b.min_p.z + b.max_p.z) * 0.5;
        };
        std::nth_element(indices.begin() + start, indices.begin() + start + count / 2, indices.begin() + end,
                         [&](int a, int b) {
                             return get_center(a) < get_center(b);
                         });
        int mid = start + count / 2;
        int left_child = build_recursive(indices, start, mid, bboxes);
        int right_child = build_recursive(indices, mid, end, bboxes);
        nodes[node_idx].left = left_child;
        nodes[node_idx].right = right_child;
        return node_idx;
    }
    void query_ray(const Point3D& origin, const Point3D& dir, std::vector<int>& out_faces) const {
        if (root == -1) return;
        query_ray_recursive(root, origin, dir, out_faces);
    }
    void query_ray_recursive(int node, const Point3D& origin, const Point3D& dir, std::vector<int>& out_faces) const {
        const auto& n = nodes[node];
        if (!n.bbox.intersects_ray(origin, dir)) return;
        if (n.right == -1) {
            out_faces.push_back(n.left);
            return;
        }
        query_ray_recursive(n.left, origin, dir, out_faces);
        query_ray_recursive(n.right, origin, dir, out_faces);
    }
};

struct GridCell {
    long long x, y, z;
    bool operator==(const GridCell& o) const {
        return x == o.x && y == o.y && z == o.z;
    }
};
struct GridCellHash {
    size_t operator()(const GridCell& c) const {
        return (c.x * 73856093) ^ (c.y * 19349663) ^ (c.z * 83492791);
    }
};

class SpatialHashGrid {
    double grid_size;
    std::unordered_map<GridCell, std::vector<int>, GridCellHash> grid;
    const std::vector<Point3D>& vertices;
public:
    SpatialHashGrid(double size, const std::vector<Point3D>& verts) 
        : grid_size(size), vertices(verts) {}
    int get_or_insert(const Point3D& p, std::vector<Point3D>& out_verts) {
        long long gx = std::floor(p.x / grid_size);
        long long gy = std::floor(p.y / grid_size);
        long long gz = std::floor(p.z / grid_size);
        double best_dist = grid_size;
        int best_idx = -1;
        for (long long dx = -1; dx <= 1; ++dx) {
            for (long long dy = -1; dy <= 1; ++dy) {
                for (long long dz = -1; dz <= 1; ++dz) {
                    GridCell cell = {gx + dx, gy + dy, gz + dz};
                    auto it = grid.find(cell);
                    if (it != grid.end()) {
                        for (int idx : it->second) {
                            double d = p.dist(vertices[idx]);
                            if (d < best_dist) {
                                best_dist = d;
                                best_idx = idx;
                            }
                        }
                    }
                }
            }
        }
        if (best_idx != -1) {
            return best_idx;
        }
        int new_idx = out_verts.size();
        const_cast<std::vector<Point3D>&>(vertices).push_back(p);
        GridCell cell = {gx, gy, gz};
        grid[cell].push_back(new_idx);
        return new_idx;
    }
};

bool tri_tri_intersect(const Point3D& A0, const Point3D& A1, const Point3D& A2,
                       const Point3D& B0, const Point3D& B1, const Point3D& B2,
                       Point3D& out_p1, Point3D& out_p2,
                       int& edge_p1_A, int& edge_p2_A,
                       int& edge_p1_B, int& edge_p2_B) {
    Point3D N2 = (B1 - B0).cross(B2 - B0);
    double len2 = N2.norm();
    if (len2 < 1e-12) return false;
    N2 = N2 * (1.0 / len2);
    double d_A0 = (A0 - B0).dot(N2);
    double d_A1 = (A1 - B0).dot(N2);
    double d_A2 = (A2 - B0).dot(N2);
    if ((d_A0 > G_EPS && d_A1 > G_EPS && d_A2 > G_EPS) ||
        (d_A0 < -G_EPS && d_A1 < -G_EPS && d_A2 < -G_EPS)) {
        return false;
    }
    Point3D N1 = (A1 - A0).cross(A2 - A0);
    double len1 = N1.norm();
    if (len1 < 1e-12) return false;
    N1 = N1 * (1.0 / len1);
    double d_B0 = (B0 - A0).dot(N1);
    double d_B1 = (B1 - A0).dot(N1);
    double d_B2 = (B2 - A0).dot(N1);
    if ((d_B0 > G_EPS && d_B1 > G_EPS && d_B2 > G_EPS) ||
        (d_B0 < -G_EPS && d_B1 < -G_EPS && d_B2 < -G_EPS)) {
        return false;
    }
    auto get_plane_intersection = [](const Point3D& u, const Point3D& v, double du, double dv, Point3D& out_pt) {
        if (std::abs(du - dv) < 1e-12) return false;
        double t = -du / (dv - du);
        if (t < -G_EPS || t > 1.0 + G_EPS) return false;
        t = std::max(0.0, std::min(1.0, t));
        out_pt = u + (v - u) * t;
        return true;
    };
    std::vector<std::pair<Point3D, int>> pts_A;
    Point3D tmp;
    if (get_plane_intersection(A0, A1, d_A0, d_A1, tmp)) pts_A.push_back({tmp, 0});
    if (get_plane_intersection(A1, A2, d_A1, d_A2, tmp)) pts_A.push_back({tmp, 1});
    if (get_plane_intersection(A2, A0, d_A2, d_A0, tmp)) pts_A.push_back({tmp, 2});
    std::vector<std::pair<Point3D, int>> unique_A;
    for (auto& p : pts_A) {
        bool dup = false;
        for (auto& u : unique_A) {
            if (p.first.dist(u.first) < G_EPS) { dup = true; break; }
        }
        if (!dup) unique_A.push_back(p);
    }
    if (unique_A.size() < 2) return false;
    Point3D P1 = unique_A[0].first;
    int edge_P1 = unique_A[0].second;
    Point3D P2 = unique_A[1].first;
    int edge_P2 = unique_A[1].second;
    
    std::vector<std::pair<Point3D, int>> pts_B;
    if (get_plane_intersection(B0, B1, d_B0, d_B1, tmp)) pts_B.push_back({tmp, 0});
    if (get_plane_intersection(B1, B2, d_B1, d_B2, tmp)) pts_B.push_back({tmp, 1});
    if (get_plane_intersection(B2, B0, d_B2, d_B0, tmp)) pts_B.push_back({tmp, 2});
    std::vector<std::pair<Point3D, int>> unique_B;
    for (auto& p : pts_B) {
        bool dup = false;
        for (auto& u : unique_B) {
            if (p.first.dist(u.first) < G_EPS) { dup = true; break; }
        }
        if (!dup) unique_B.push_back(p);
    }
    if (unique_B.size() < 2) return false;
    Point3D Q1 = unique_B[0].first;
    int edge_Q1 = unique_B[0].second;
    Point3D Q2 = unique_B[1].first;
    int edge_Q2 = unique_B[1].second;
    
    Point3D D = N1.cross(N2);
    double lenD = D.norm();
    if (lenD < 1e-12) return false;
    D = D * (1.0 / lenD);
    double t_P1 = P1.dot(D);
    double t_P2 = P2.dot(D);
    double t_Q1 = Q1.dot(D);
    double t_Q2 = Q2.dot(D);
    if (t_P1 > t_P2) { std::swap(t_P1, t_P2); std::swap(P1, P2); std::swap(edge_P1, edge_P2); }
    if (t_Q1 > t_Q2) { std::swap(t_Q1, t_Q2); std::swap(Q1, Q2); std::swap(edge_Q1, edge_Q2); }
    double t_start = std::max(t_P1, t_Q1);
    double t_end = std::min(t_P2, t_Q2);
    if (t_start > t_end - G_EPS) return false;
    
    edge_p1_A = -1; edge_p1_B = -1;
    if (t_P1 > t_Q1) {
        out_p1 = P1;
        edge_p1_A = edge_P1;
        if (std::abs(t_P1 - t_Q1) < G_EPS) edge_p1_B = edge_Q1;
        else if (std::abs(t_P1 - t_Q2) < G_EPS) edge_p1_B = edge_Q2;
    } else {
        out_p1 = Q1;
        edge_p1_B = edge_Q1;
        if (std::abs(t_Q1 - t_P1) < G_EPS) edge_p1_A = edge_P1;
        else if (std::abs(t_Q1 - t_P2) < G_EPS) edge_p1_A = edge_P2;
    }
    
    edge_p2_A = -1; edge_p2_B = -1;
    if (t_P2 < t_Q2) {
        out_p2 = P2;
        edge_p2_A = edge_P2;
        if (std::abs(t_P2 - t_Q1) < G_EPS) edge_p2_B = edge_Q1;
        else if (std::abs(t_P2 - t_Q2) < G_EPS) edge_p2_B = edge_Q2;
    } else {
        out_p2 = Q2;
        edge_p2_B = edge_Q2;
        if (std::abs(t_Q2 - t_P1) < G_EPS) edge_p2_A = edge_P1;
        else if (std::abs(t_Q2 - t_P2) < G_EPS) edge_p2_A = edge_P2;
    }
    return true;
}

int ray_triangle_intersect(const Point3D& P, const Point3D& dir, 
                           const Point3D& V0, const Point3D& V1, const Point3D& V2,
                           double& out_t, Point3D& out_ip) {
    Point3D edge1 = V1 - V0;
    Point3D edge2 = V2 - V0;
    Point3D h = dir.cross(edge2);
    double a = edge1.dot(h);
    if (std::abs(a) < 1e-10) {
        return 2;
    }
    double f = 1.0 / a;
    Point3D s = P - V0;
    double u = f * s.dot(h);
    if (u < -G_EPS || u > 1.0 + G_EPS) {
        return 0;
    }
    Point3D q = s.cross(edge1);
    double v = f * dir.dot(q);
    if (v < -G_EPS || u + v > 1.0 + G_EPS) {
        return 0;
    }
    double t = f * edge2.dot(q);
    if (t < G_EPS) {
        return 0;
    }
    if (u < G_EPS * 10.0 || v < G_EPS * 10.0 || (1.0 - u - v) < G_EPS * 10.0) {
        return 2;
    }
    out_t = t;
    out_ip = P + dir * t;
    return 1;
}

bool is_point_inside_mesh(const Point3D& P, const Mesh& M, const AABBTree& tree) {
    std::mt19937 gen(42);
    std::uniform_real_distribution<double> dis(-1.0, 1.0);
    while (true) {
        Point3D dir = {dis(gen), dis(gen), dis(gen)};
        double len = dir.norm();
        if (len < 1e-3) continue;
        dir = dir * (1.0 / len);
        std::vector<int> candidate_faces;
        tree.query_ray(P, dir, candidate_faces);
        int intersect_count = 0;
        bool degenerate = false;
        for (int fid : candidate_faces) {
            Point3D V0 = M.vertices[M.faces[fid][0]];
            Point3D V1 = M.vertices[M.faces[fid][1]];
            Point3D V2 = M.vertices[M.faces[fid][2]];
            double t;
            Point3D ip;
            int status = ray_triangle_intersect(P, dir, V0, V1, V2, t, ip);
            if (status == 1) {
                intersect_count++;
            } else if (status == 2) {
                degenerate = true;
                break;
            }
        }
        if (!degenerate) {
            return (intersect_count % 2) != 0;
        }
    }
}

double dist_to_bbox(const Point3D& P, const AABB& box) {
    double dx = std::max({0.0, box.min_p.x - P.x, P.x - box.max_p.x});
    double dy = std::max({0.0, box.min_p.y - P.y, P.y - box.max_p.y});
    double dz = std::max({0.0, box.min_p.z - P.z, P.z - box.max_p.z});
    return std::sqrt(dx*dx + dy*dy + dz*dz);
}

bool dist_to_triangle_less_than(const Point3D& P, const Point3D& V0, const Point3D& V1, const Point3D& V2, double snap_tol) {
    Point3D normal = (V1 - V0).cross(V2 - V0);
    double len = normal.norm();
    if (len < 1e-12) return false;
    normal = normal * (1.0 / len);
    double dist_to_plane = std::abs((P - V0).dot(normal));
    if (dist_to_plane >= snap_tol) return false;
    
    Point3D proj = P - normal * ((P - V0).dot(normal));
    
    double nx = std::abs(normal.x);
    double ny = std::abs(normal.y);
    double nz = std::abs(normal.z);
    int drop = 0;
    if (ny > nx && ny > nz) drop = 1;
    else if (nz > nx && nz > ny) drop = 2;
    
    Point2D p2d, v0_2d, v1_2d, v2_2d;
    if (drop == 0) {
        p2d = {proj.y, proj.z};
        v0_2d = {V0.y, V0.z};
        v1_2d = {V1.y, V1.z};
        v2_2d = {V2.y, V2.z};
    } else if (drop == 1) {
        p2d = {proj.x, proj.z};
        v0_2d = {V0.x, V0.z};
        v1_2d = {V1.x, V1.z};
        v2_2d = {V2.x, V2.z};
    } else {
        p2d = {proj.x, proj.y};
        v0_2d = {V0.x, V0.y};
        v1_2d = {V1.x, V1.y};
        v2_2d = {V2.x, V2.y};
    }
    
    double cp0 = ccw2d(v0_2d, v1_2d, p2d);
    double cp1 = ccw2d(v1_2d, v2_2d, p2d);
    double cp2 = ccw2d(v2_2d, v0_2d, p2d);
    
    double tol = -snap_tol * (v1_2d - v0_2d).norm();
    if ((cp0 >= tol && cp1 >= tol && cp2 >= tol) || (cp0 <= -tol && cp1 <= -tol && cp2 <= -tol)) {
        return true;
    }
    return false;
}

bool is_point_on_mesh_recursive(int node, const Point3D& P, const Mesh& M, const AABBTree& tree, double snap_tol) {
    const auto& n = tree.nodes[node];
    if (dist_to_bbox(P, n.bbox) >= snap_tol) return false;
    if (n.right == -1) {
        int fid = n.left;
        return dist_to_triangle_less_than(P, M.vertices[M.faces[fid][0]], M.vertices[M.faces[fid][1]], M.vertices[M.faces[fid][2]], snap_tol);
    }
    return is_point_on_mesh_recursive(n.left, P, M, tree, snap_tol) ||
           is_point_on_mesh_recursive(n.right, P, M, tree, snap_tol);
}

bool is_point_on_mesh(const Point3D& P, const Mesh& M, const AABBTree& tree, double snap_tol) {
    if (tree.root == -1) return false;
    return is_point_on_mesh_recursive(tree.root, P, M, tree, snap_tol);
}

void find_intersecting_pairs(const AABBTree& treeA, const AABBTree& treeB, std::vector<std::pair<int, int>>& out_pairs) {
    if (treeA.root == -1 || treeB.root == -1) return;
    std::vector<std::pair<int, int>> stack;
    stack.push_back({treeA.root, treeB.root});
    while (!stack.empty()) {
        auto [nA, nB] = stack.back();
        stack.pop_back();
        const auto& nodeA = treeA.nodes[nA];
        const auto& nodeB = treeB.nodes[nB];
        if (!nodeA.bbox.intersects(nodeB.bbox)) continue;
        if (nodeA.right == -1 && nodeB.right == -1) {
            out_pairs.push_back({nodeA.left, nodeB.left});
        } else if (nodeA.right == -1) {
            stack.push_back({nA, nodeB.left});
            stack.push_back({nA, nodeB.right});
        } else if (nodeB.right == -1) {
            stack.push_back({nodeA.left, nB});
            stack.push_back({nodeA.right, nB});
        } else {
            stack.push_back({nodeA.left, nodeB.left});
            stack.push_back({nodeA.left, nodeB.right});
            stack.push_back({nodeA.right, nodeB.left});
            stack.push_back({nodeA.right, nodeB.right});
        }
    }
}

struct IntersectionSegment {
    Point3D p, q;
    int idx_p, idx_q;
    int edge_p_A, edge_q_A;
    int edge_p_B, edge_q_B;
};

struct EdgeSplit {
    Point3D pt;
    int idx;
};

// Fast standard I/O for speed
inline double readDouble() {
    double val;
    std::cin >> val;
    return val;
}
inline int readInt() {
    int val;
    std::cin >> val;
    return val;
}inline long long make_edge_key(int u, int v) {
    if (u > v) std::swap(u, v);
    return ((long long)u << 32) | v;
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    int VA, FA, VB, FB;
    if (!(std::cin >> VA >> FA >> VB >> FB)) return 0;
    
    Mesh meshA, meshB;
    meshA.vertices.resize(VA);
    for (int i = 0; i < VA; ++i) {
        meshA.vertices[i].x = readDouble();
        meshA.vertices[i].y = readDouble();
        meshA.vertices[i].z = readDouble();
    }
    meshA.faces.resize(FA, std::vector<int>(3));
    for (int i = 0; i < FA; ++i) {
        meshA.faces[i][0] = readInt();
        meshA.faces[i][1] = readInt();
        meshA.faces[i][2] = readInt();
    }
    
    meshB.vertices.resize(VB);
    for (int i = 0; i < VB; ++i) {
        meshB.vertices[i].x = readDouble();
        meshB.vertices[i].y = readDouble();
        meshB.vertices[i].z = readDouble();
    }
    meshB.faces.resize(FB, std::vector<int>(3));
    for (int i = 0; i < FB; ++i) {
        meshB.faces[i][0] = readInt();
        meshB.faces[i][1] = readInt();
        meshB.faces[i][2] = readInt();
    }
    
    // Compute combined bounding box for scale-invariance
    AABB bbox;
    for (const auto& v : meshA.vertices) bbox.merge(v);
    for (const auto& v : meshB.vertices) bbox.merge(v);
    Point3D size = bbox.max_p - bbox.min_p;
    double max_size = std::max({size.x, size.y, size.z});
    if (max_size < 1e-9) max_size = 1.0;
    
    // Set global epsilon based on scale
    G_EPS = max_size * 1e-7;
    
    // Perturb mesh B to avoid degeneracies
    uint64_t state = 0x123456789abcdefULL;
    auto rand_double = [&state]() {
        state ^= state >> 12;
        state ^= state << 25;
        state ^= state >> 27;
        uint64_t val = state * 0x2545F4914F6CDD1DULL;
        return (double)(val & 0xFFFFFFFFFFFFFFF) / (double)0x1000000000000000ULL;
    };
    double perturb_scale = max_size * 1e-6;
    for (auto& v : meshB.vertices) {
        v.x += (rand_double() - 0.5) * perturb_scale;
        v.y += (rand_double() - 0.5) * perturb_scale;
        v.z += (rand_double() - 0.5) * perturb_scale;
    }
    
    // Build AABB trees
    AABBTree treeA(meshA);
    AABBTree treeB(meshB);
    
    double snap_tol = max_size * 5e-6;
    
    // Find candidate intersecting pairs
    std::vector<std::pair<int, int>> candidate_pairs;
    find_intersecting_pairs(treeA, treeB, candidate_pairs);
    
    std::vector<Point3D> IntPoints;
    SpatialHashGrid grid_IntPoints(snap_tol, IntPoints);
    std::set<std::pair<int, int>> IntSegments;
    
    std::vector<std::vector<IntersectionSegment>> meshA_int_segs(FA);
    std::vector<std::vector<IntersectionSegment>> meshB_int_segs(FB);
    
    for (auto& pair : candidate_pairs) {
        int fA = pair.first;
        int fB = pair.second;
        Point3D V0 = meshA.vertices[meshA.faces[fA][0]];
        Point3D V1 = meshA.vertices[meshA.faces[fA][1]];
        Point3D V2 = meshA.vertices[meshA.faces[fA][2]];
        Point3D W0 = meshB.vertices[meshB.faces[fB][0]];
        Point3D W1 = meshB.vertices[meshB.faces[fB][1]];
        Point3D W2 = meshB.vertices[meshB.faces[fB][2]];
        
        Point3D p1, p2;
        int ep1A, ep2A, ep1B, ep2B;
        if (tri_tri_intersect(V0, V1, V2, W0, W1, W2, p1, p2, ep1A, ep2A, ep1B, ep2B)) {
            int idx1 = grid_IntPoints.get_or_insert(p1, IntPoints);
            int idx2 = grid_IntPoints.get_or_insert(p2, IntPoints);
            if (idx1 != idx2) {
                Point3D snapped_p1 = IntPoints[idx1];
                Point3D snapped_p2 = IntPoints[idx2];
                IntSegments.insert({std::min(idx1, idx2), std::max(idx1, idx2)});
                meshA_int_segs[fA].push_back({snapped_p1, snapped_p2, idx1, idx2, ep1A, ep2A, ep1B, ep2B});
                meshB_int_segs[fB].push_back({snapped_p1, snapped_p2, idx1, idx2, ep1B, ep2B, ep1A, ep2A});
            }
        }
    }
    
    // If no intersections, handle simple cases
    if (IntSegments.empty()) {
        Point3D centroid_A = (meshA.vertices[meshA.faces[0][0]] + 
                              meshA.vertices[meshA.faces[0][1]] + 
                              meshA.vertices[meshA.faces[0][2]]) * (1.0 / 3.0);
        Point3D centroid_B = (meshB.vertices[meshB.faces[0][0]] + 
                              meshB.vertices[meshB.faces[0][1]] + 
                              meshB.vertices[meshB.faces[0][2]]) * (1.0 / 3.0);
        bool A_in_B = is_point_inside_mesh(centroid_A, meshB, treeB);
        if (A_in_B) {
            // Union is B
            std::cout << VB << " " << FB << "\n";
            for (auto& v : meshB.vertices) {
                std::cout << v.x << " " << v.y << " " << v.z << "\n";
            }
            for (auto& f : meshB.faces) {
                std::cout << f[0] << " " << f[1] << " " << f[2] << "\n";
            }
            return 0;
        }
        bool B_in_A = is_point_inside_mesh(centroid_B, meshA, treeA);
        if (B_in_A) {
            // Union is A
            std::cout << VA << " " << FA << "\n";
            for (auto& v : meshA.vertices) {
                std::cout << v.x << " " << v.y << " " << v.z << "\n";
            }
            for (auto& f : meshA.faces) {
                std::cout << f[0] << " " << f[1] << " " << f[2] << "\n";
            }
            return 0;
        }
        // Disjoint union
        std::cout << VA + VB << " " << FA + FB << "\n";
        for (auto& v : meshA.vertices) {
            std::cout << v.x << " " << v.y << " " << v.z << "\n";
        }
        for (auto& v : meshB.vertices) {
            std::cout << v.x << " " << v.y << " " << v.z << "\n";
        }
        for (auto& f : meshA.faces) {
            std::cout << f[0] << " " << f[1] << " " << f[2] << "\n";
        }
        for (auto& f : meshB.faces) {
            std::cout << f[0] + VA << " " << f[1] + VA << " " << f[2] + VA << "\n";
        }
        return 0;
    }
    
    auto split_mesh = [&](const Mesh& M, 
                          const std::vector<std::vector<IntersectionSegment>>& face_int_segs, 
                          const std::unordered_map<long long, std::vector<EdgeSplit>>& edge_splits,
                          Mesh& M_out) {
        SpatialHashGrid grid_M(snap_tol, M_out.vertices);
        
        for (size_t fid = 0; fid < M.faces.size(); ++fid) {
            Point3D V0 = M.vertices[M.faces[fid][0]];
            Point3D V1 = M.vertices[M.faces[fid][1]];
            Point3D V2 = M.vertices[M.faces[fid][2]];
            
            const auto& segs = face_int_segs[fid];
            int g0 = M.faces[fid][0];
            int g1 = M.faces[fid][1];
            int g2 = M.faces[fid][2];
            long long e0 = make_edge_key(g0, g1);
            long long e1 = make_edge_key(g1, g2);
            long long e2 = make_edge_key(g2, g0);
            
            bool has_splits = false;
            if (edge_splits.count(e0) || edge_splits.count(e1) || edge_splits.count(e2)) {
                has_splits = true;
            }
            
            if (segs.empty() && !has_splits) {
                // No intersections and no edge splits, just insert the face
                int g_idx0 = grid_M.get_or_insert(V0, M_out.vertices);
                int g_idx1 = grid_M.get_or_insert(V1, M_out.vertices);
                int g_idx2 = grid_M.get_or_insert(V2, M_out.vertices);
                if (M_out.vertex_int_idx.size() <= (size_t)std::max({g_idx0, g_idx1, g_idx2})) {
                    M_out.vertex_int_idx.resize(std::max({g_idx0, g_idx1, g_idx2}) + 1);
                }
                M_out.faces.push_back({g_idx0, g_idx1, g_idx2});
                continue;
            }
            
            std::vector<Point3D> unique_pts = {V0, V1, V2};
            std::vector<int> local_int_idx = {-1, -1, -1};
            
            // Insert edge splits first
            for (long long e_key : {e0, e1, e2}) {
                auto it = edge_splits.find(e_key);
                if (it != edge_splits.end()) {
                    for (const auto& split : it->second) {
                        int found = -1;
                        for (int i = 0; i < (int)unique_pts.size(); ++i) {
                            if (split.pt.dist(unique_pts[i]) < snap_tol) {
                                found = i;
                                break;
                            }
                        }
                        if (found == -1) {
                            unique_pts.push_back(split.pt);
                            local_int_idx.push_back(split.idx);
                        }
                    }
                }
            }
            
            // Insert face's own intersection points
            for (const auto& seg : segs) {
                for (auto& p_info : {std::make_pair(IntPoints[seg.idx_p], seg.idx_p), std::make_pair(IntPoints[seg.idx_q], seg.idx_q)}) {
                    Point3D pt = p_info.first;
                    int idx = p_info.second;
                    int found = -1;
                    for (int i = 0; i < (int)unique_pts.size(); ++i) {
                        if (pt.dist(unique_pts[i]) < snap_tol) {
                            found = i;
                            break;
                        }
                    }
                    if (found == -1) {
                        unique_pts.push_back(pt);
                        local_int_idx.push_back(idx);
                    } else {
                        if (local_int_idx[found] == -1) local_int_idx[found] = idx;
                    }
                }
            }
            
            std::vector<std::pair<int, int>> constraints;
            for (const auto& seg : segs) {
                int idx1 = -1, idx2 = -1;
                Point3D seg_p = IntPoints[seg.idx_p];
                Point3D seg_q = IntPoints[seg.idx_q];
                for (int i = 0; i < (int)unique_pts.size(); ++i) {
                    if (seg_p.dist(unique_pts[i]) < snap_tol) idx1 = i;
                    if (seg_q.dist(unique_pts[i]) < snap_tol) idx2 = i;
                }
                if (idx1 != -1 && idx2 != -1 && idx1 != idx2) {
                    constraints.push_back({idx1, idx2});
                }
            }
            
            auto add_edge_constraints = [&](int i1, int i2, const Point3D& A, const Point3D& B) {
                std::vector<std::pair<double, int>> pts_on_edge;
                pts_on_edge.push_back({0.0, i1});
                pts_on_edge.push_back({1.0, i2});
                Point3D AB = B - A;
                double len2 = AB.dot(AB);
                for (int i = 3; i < (int)unique_pts.size(); ++i) {
                    if (dist_to_segment(unique_pts[i], A, B) < snap_tol) {
                        double t = (unique_pts[i] - A).dot(AB) / len2;
                        pts_on_edge.push_back({t, i});
                    }
                }
                std::sort(pts_on_edge.begin(), pts_on_edge.end());
                for (size_t i = 0; i + 1 < pts_on_edge.size(); ++i) {
                    constraints.push_back({pts_on_edge[i].second, pts_on_edge[i+1].second});
                }
            };
            add_edge_constraints(0, 1, V0, V1);
            add_edge_constraints(1, 2, V1, V2);
            add_edge_constraints(2, 0, V2, V0);
            
            for (auto& c : constraints) {
                if (c.first > c.second) std::swap(c.first, c.second);
            }
            std::sort(constraints.begin(), constraints.end());
            constraints.erase(std::unique(constraints.begin(), constraints.end()), constraints.end());
            
            Point3D normal = (V1 - V0).cross(V2 - V0);
            double nx = std::abs(normal.x);
            double ny = std::abs(normal.y);
            double nz = std::abs(normal.z);
            int drop_coord = 0;
            if (ny > nx && ny > nz) drop_coord = 1;
            else if (nz > nx && nz > ny) drop_coord = 2;
            
            std::vector<Point2D> pts2d(unique_pts.size());
            for (size_t i = 0; i < unique_pts.size(); ++i) {
                if (drop_coord == 0) pts2d[i] = {unique_pts[i].y, unique_pts[i].z};
                else if (drop_coord == 1) pts2d[i] = {unique_pts[i].x, unique_pts[i].z};
                else pts2d[i] = {unique_pts[i].x, unique_pts[i].y};
            }
            
            std::vector<std::pair<int, int>> S = constraints;
            int n_pts = unique_pts.size();
            for (int i = 0; i < n_pts; ++i) {
                for (int j = i + 1; j < n_pts; ++j) {
                    bool exists = false;
                    for (const auto& seg : S) {
                        if ((seg.first == i && seg.second == j) || (seg.first == j && seg.second == i)) {
                            exists = true;
                            break;
                        }
                    }
                    if (exists) continue;
                    Point2D A = pts2d[i];
                    Point2D B = pts2d[j];
                    bool has_interior_point = false;
                    for (int k = 0; k < n_pts; ++k) {
                        if (k == i || k == j) continue;
                        if (point_in_segment_interior_2d(pts2d[k], A, B, snap_tol)) {
                            has_interior_point = true;
                            break;
                        }
                    }
                    if (has_interior_point) continue;
                    bool intersects = false;
                    for (const auto& seg : S) {
                        Point2D C = pts2d[seg.first];
                        Point2D D = pts2d[seg.second];
                        if (segments_intersect_proper_2d(A, B, C, D)) {
                            intersects = true;
                            break;
                        }
                    }
                    if (intersects) continue;
                    S.push_back({i, j});
                }
            }
            

            
            std::vector<std::vector<bool>> adj(n_pts, std::vector<bool>(n_pts, false));
            for (const auto& seg : S) {
                adj[seg.first][seg.second] = true;
                adj[seg.second][seg.first] = true;
            }
            
            std::vector<std::vector<int>> sub_triangles;
            for (int i = 0; i < n_pts; ++i) {
                for (int j = i + 1; j < n_pts; ++j) {
                    if (!adj[i][j]) continue;
                    for (int k = j + 1; k < n_pts; ++k) {
                        if (!adj[i][k] || !adj[j][k]) continue;
                        Point2D A = pts2d[i];
                        Point2D B = pts2d[j];
                        Point2D C = pts2d[k];
                        double area = ccw2d(A, B, C);
                        if (std::abs(area) < 1e-9) continue;
                        int t_i = i, t_j = j, t_k = k;
                        if (area < 0) {
                            std::swap(t_j, t_k);
                            area = -area;
                        }
                        bool point_inside = false;
                        for (int m = 0; m < n_pts; ++m) {
                            if (m == i || m == j || m == k) continue;
                            Point2D Q = pts2d[m];
                            if (ccw2d(pts2d[t_i], pts2d[t_j], Q) > 1e-9 &&
                                ccw2d(pts2d[t_j], pts2d[t_k], Q) > 1e-9 &&
                                ccw2d(pts2d[t_k], pts2d[t_i], Q) > 1e-9) {
                                point_inside = true;
                                break;
                            }
                        }
                        if (!point_inside) {
                            Point3D p_i = unique_pts[t_i];
                            Point3D p_j = unique_pts[t_j];
                            Point3D p_k = unique_pts[t_k];
                            Point3D sub_normal = (p_j - p_i).cross(p_k - p_i);
                            if (sub_normal.dot(normal) < 0) {
                                std::swap(t_j, t_k);
                            }
                            sub_triangles.push_back({t_i, t_j, t_k});
                        }
                    }
                }
            }
            
            std::vector<int> local_to_global(unique_pts.size());
            for (size_t lv = 0; lv < unique_pts.size(); ++lv) {
                int g_idx = grid_M.get_or_insert(unique_pts[lv], M_out.vertices);
                local_to_global[lv] = g_idx;
                if (M_out.vertex_int_idx.size() <= (size_t)g_idx) {
                    M_out.vertex_int_idx.resize(g_idx + 1);
                }
                if (local_int_idx[lv] != -1) {
                    auto& vec = M_out.vertex_int_idx[g_idx];
                    if (std::find(vec.begin(), vec.end(), local_int_idx[lv]) == vec.end()) {
                        vec.push_back(local_int_idx[lv]);
                    }
                }
            }
            for (auto& tri : sub_triangles) {
                M_out.faces.push_back({local_to_global[tri[0]], local_to_global[tri[1]], local_to_global[tri[2]]});
            }
        }
    };

    std::vector<std::vector<int>> point_to_faces_A(IntPoints.size());
    for (int fid = 0; fid < FA; ++fid) {
        for (const auto& seg : meshA_int_segs[fid]) {
            point_to_faces_A[seg.idx_p].push_back(fid);
            point_to_faces_A[seg.idx_q].push_back(fid);
        }
    }
    std::vector<std::vector<int>> point_to_faces_B(IntPoints.size());
    for (int fid = 0; fid < FB; ++fid) {
        for (const auto& seg : meshB_int_segs[fid]) {
            point_to_faces_B[seg.idx_p].push_back(fid);
            point_to_faces_B[seg.idx_q].push_back(fid);
        }
    }

    std::unordered_map<long long, std::vector<EdgeSplit>> edge_splits_A;
    std::unordered_map<long long, std::vector<EdgeSplit>> edge_splits_B;

    for (size_t i = 0; i < IntPoints.size(); ++i) {
        Point3D P = IntPoints[i];
        
        // Check vertex snapping for A
        double best_v_dist_A = snap_tol;
        Point3D best_v_pt_A = P;
        bool snapped_to_v_A = false;
        for (int fid : point_to_faces_A[i]) {
            for (int j = 0; j < 3; ++j) {
                Point3D V = meshA.vertices[meshA.faces[fid][j]];
                double d = P.dist(V);
                if (d < best_v_dist_A) {
                    best_v_dist_A = d;
                    best_v_pt_A = V;
                    snapped_to_v_A = true;
                }
            }
        }
        
        // Check vertex snapping for B
        double best_v_dist_B = snap_tol;
        Point3D best_v_pt_B = P;
        bool snapped_to_v_B = false;
        for (int fid : point_to_faces_B[i]) {
            for (int j = 0; j < 3; ++j) {
                Point3D V = meshB.vertices[meshB.faces[fid][j]];
                double d = P.dist(V);
                if (d < best_v_dist_B) {
                    best_v_dist_B = d;
                    best_v_pt_B = V;
                    snapped_to_v_B = true;
                }
            }
        }
        
        Point3D P_snapped = P;
        if (snapped_to_v_A && snapped_to_v_B) {
            if (best_v_dist_A < best_v_dist_B) P_snapped = best_v_pt_A;
            else P_snapped = best_v_pt_B;
        } else if (snapped_to_v_A) {
            P_snapped = best_v_pt_A;
        } else if (snapped_to_v_B) {
            P_snapped = best_v_pt_B;
        }
        IntPoints[i] = P_snapped;
        
        // Check edge snapping for A (only if not snapped to a vertex of A)
        double best_e_dist_A = snap_tol;
        long long best_e_key_A = -1;
        Point3D proj_A = IntPoints[i];
        
        if (!snapped_to_v_A) {
            for (int fid : point_to_faces_A[i]) {
                for (int j = 0; j < 3; ++j) {
                    int u = meshA.faces[fid][j];
                    int v = meshA.faces[fid][(j+1)%3];
                    Point3D U1 = meshA.vertices[u];
                    Point3D U2 = meshA.vertices[v];
                    double d = dist_to_segment(IntPoints[i], U1, U2);
                    if (d < best_e_dist_A) {
                        best_e_dist_A = d;
                        best_e_key_A = make_edge_key(u, v);
                        Point3D AB = U2 - U1;
                        double len2 = AB.dot(AB);
                        double t = (IntPoints[i] - U1).dot(AB) / len2;
                        t = std::max(0.0, std::min(1.0, t));
                        proj_A = U1 + AB * t;
                    }
                }
            }
        }
        
        // Check edge snapping for B (only if not snapped to a vertex of B)
        double best_e_dist_B = snap_tol;
        long long best_e_key_B = -1;
        Point3D proj_B = IntPoints[i];
        
        if (!snapped_to_v_B) {
            for (int fid : point_to_faces_B[i]) {
                for (int j = 0; j < 3; ++j) {
                    int u = meshB.faces[fid][j];
                    int v = meshB.faces[fid][(j+1)%3];
                    Point3D U1 = meshB.vertices[u];
                    Point3D U2 = meshB.vertices[v];
                    double d = dist_to_segment(IntPoints[i], U1, U2);
                    if (d < best_e_dist_B) {
                        best_e_dist_B = d;
                        best_e_key_B = make_edge_key(u, v);
                        Point3D AB = U2 - U1;
                        double len2 = AB.dot(AB);
                        double t = (IntPoints[i] - U1).dot(AB) / len2;
                        t = std::max(0.0, std::min(1.0, t));
                        proj_B = U1 + AB * t;
                    }
                }
            }
        }
        
        if (best_e_key_A != -1 || best_e_key_B != -1) {
            // Update IntPoints[i] to the closest projected point
            if (best_e_key_A != -1 && (best_e_key_B == -1 || best_e_dist_A < best_e_dist_B)) {
                IntPoints[i] = proj_A;
            } else {
                IntPoints[i] = proj_B;
            }
            
            // Add to edge splits
            if (best_e_key_A != -1) {
                auto& vec = edge_splits_A[best_e_key_A];
                bool exists = false;
                for (const auto& xp : vec) {
                    if (xp.pt.dist(IntPoints[i]) < snap_tol) { exists = true; break; }
                }
                if (!exists) vec.push_back({IntPoints[i], (int)i});
            }
            if (best_e_key_B != -1) {
                auto& vec = edge_splits_B[best_e_key_B];
                bool exists = false;
                for (const auto& xp : vec) {
                    if (xp.pt.dist(IntPoints[i]) < snap_tol) { exists = true; break; }
                }
                if (!exists) vec.push_back({IntPoints[i], (int)i});
            }
        }
    }

    int total_splits_A = 0;
    for (auto& pair : edge_splits_A) total_splits_A += pair.second.size();
    int total_splits_B = 0;
    for (auto& pair : edge_splits_B) total_splits_B += pair.second.size();

    Mesh MA_split, MB_split;
    split_mesh(meshA, meshA_int_segs, edge_splits_A, MA_split);
    split_mesh(meshB, meshB_int_segs, edge_splits_B, MB_split);

    auto build_adjacency = [&](const Mesh& M, std::vector<std::vector<int>>& adj_list) {
        int num_faces = M.faces.size();
        adj_list.assign(num_faces, std::vector<int>());
        struct EdgeShare {
            int u, v;
            int face_idx;
        };
        std::vector<EdgeShare> all_edges;
        all_edges.reserve(num_faces * 3);
        for (int i = 0; i < num_faces; ++i) {
            int u0 = M.faces[i][0];
            int u1 = M.faces[i][1];
            int u2 = M.faces[i][2];
            all_edges.push_back({std::min(u0, u1), std::max(u0, u1), i});
            all_edges.push_back({std::min(u1, u2), std::max(u1, u2), i});
            all_edges.push_back({std::min(u2, u0), std::max(u2, u0), i});
        }
        std::sort(all_edges.begin(), all_edges.end(), [](const EdgeShare& a, const EdgeShare& b) {
            if (a.u != b.u) return a.u < b.u;
            return a.v < b.v;
        });
        for (size_t i = 0; i < all_edges.size(); ) {
            size_t j = i + 1;
            while (j < all_edges.size() && all_edges[j].u == all_edges[i].u && all_edges[j].v == all_edges[i].v) {
                ++j;
            }
            int u = all_edges[i].u;
            int v = all_edges[i].v;
            bool is_int_edge = false;
            const auto& ids_u = M.vertex_int_idx[u];
            const auto& ids_v = M.vertex_int_idx[v];
            if (!ids_u.empty() && !ids_v.empty()) {
                for (int iu : ids_u) {
                    for (int iv : ids_v) {
                        auto p = std::make_pair(std::min(iu, iv), std::max(iu, iv));
                        if (IntSegments.count(p)) {
                            is_int_edge = true;
                            break;
                        }
                    }
                    if (is_int_edge) break;
                }
            }
            if (!is_int_edge) {
                for (size_t k1 = i; k1 < j; ++k1) {
                    for (size_t k2 = k1 + 1; k2 < j; ++k2) {
                        adj_list[all_edges[k1].face_idx].push_back(all_edges[k2].face_idx);
                        adj_list[all_edges[k2].face_idx].push_back(all_edges[k1].face_idx);
                    }
                }
            }
            i = j;
        }
    };
    
    std::vector<std::vector<int>> adj_list_A, adj_list_B;
    build_adjacency(MA_split, adj_list_A);
    build_adjacency(MB_split, adj_list_B);
    
    // Classify components of A
    int num_faces_A = MA_split.faces.size();
    std::vector<bool> visited_A(num_faces_A, false);
    std::vector<bool> keep_A(num_faces_A, false);
    for (int i = 0; i < num_faces_A; ++i) {
        if (visited_A[i]) continue;
        std::vector<int> component;
        std::vector<int> q;
        q.push_back(i);
        visited_A[i] = true;
        size_t head = 0;
        while (head < q.size()) {
            int curr = q[head++];
            component.push_back(curr);
            for (int neighbor : adj_list_A[curr]) {
                if (!visited_A[neighbor]) {
                    visited_A[neighbor] = true;
                    q.push_back(neighbor);
                }
            }
        }
        int rep_face = component[0];
        Point3D V0 = MA_split.vertices[MA_split.faces[rep_face][0]];
        Point3D V1 = MA_split.vertices[MA_split.faces[rep_face][1]];
        Point3D V2 = MA_split.vertices[MA_split.faces[rep_face][2]];
        Point3D centroid = (V0 + V1 + V2) * (1.0 / 3.0);
        bool keep = false;
        if (is_point_on_mesh(centroid, meshB, treeB, snap_tol)) {
            keep = true; // Keep A's coplanar boundary faces
        } else {
            bool inside = is_point_inside_mesh(centroid, meshB, treeB);
            keep = !inside;
        }
        for (int fid : component) {
            keep_A[fid] = keep;
        }
    }
    
    // Classify components of B
    int num_faces_B = MB_split.faces.size();
    std::vector<bool> visited_B(num_faces_B, false);
    std::vector<bool> keep_B(num_faces_B, false);
    for (int i = 0; i < num_faces_B; ++i) {
        if (visited_B[i]) continue;
        std::vector<int> component;
        std::vector<int> q;
        q.push_back(i);
        visited_B[i] = true;
        size_t head = 0;
        while (head < q.size()) {
            int curr = q[head++];
            component.push_back(curr);
            for (int neighbor : adj_list_B[curr]) {
                if (!visited_B[neighbor]) {
                    visited_B[neighbor] = true;
                    q.push_back(neighbor);
                }
            }
        }
        int rep_face = component[0];
        Point3D V0 = MB_split.vertices[MB_split.faces[rep_face][0]];
        Point3D V1 = MB_split.vertices[MB_split.faces[rep_face][1]];
        Point3D V2 = MB_split.vertices[MB_split.faces[rep_face][2]];
        Point3D centroid = (V0 + V1 + V2) * (1.0 / 3.0);
        bool keep = false;
        if (is_point_on_mesh(centroid, meshA, treeA, snap_tol)) {
            keep = false; // Discard B's coplanar boundary faces to avoid duplicates
        } else {
            bool inside = is_point_inside_mesh(centroid, meshA, treeA);
            keep = !inside;
        }
        for (int fid : component) {
            keep_B[fid] = keep;
        }
    }
    
    // Merge kept faces into final mesh
    Mesh final_mesh;
    SpatialHashGrid grid_final(snap_tol, final_mesh.vertices);
    
    auto add_kept_faces = [&](const Mesh& M, const std::vector<bool>& keep_face) {
        for (size_t i = 0; i < M.faces.size(); ++i) {
            if (!keep_face[i]) continue;
            std::vector<int> face_indices(3);
            for (int j = 0; j < 3; ++j) {
                Point3D p = M.vertices[M.faces[i][j]];
                int g_idx = grid_final.get_or_insert(p, final_mesh.vertices);
                face_indices[j] = g_idx;
            }
            if (face_indices[0] != face_indices[1] &&
                face_indices[1] != face_indices[2] &&
                face_indices[2] != face_indices[0]) {
                final_mesh.faces.push_back(face_indices);
            }
        }
    };
    
    add_kept_faces(MA_split, keep_A);
    add_kept_faces(MB_split, keep_B);
    
    // Print final union mesh
    std::cout << final_mesh.vertices.size() << " " << final_mesh.faces.size() << "\n";
    for (const auto& v : final_mesh.vertices) {
        std::cout << v.x << " " << v.y << " " << v.z << "\n";
    }
    for (const auto& f : final_mesh.faces) {
        std::cout << f[0] << " " << f[1] << " " << f[2] << "\n";
    }
    
    return 0;
}
