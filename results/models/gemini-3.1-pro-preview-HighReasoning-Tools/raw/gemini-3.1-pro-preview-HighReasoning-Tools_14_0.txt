    static void Main(string[] args) {
        // Parse input
        string firstLine = Console.ReadLine();
        while (string.IsNullOrWhiteSpace(firstLine)) {
            firstLine = Console.ReadLine();
        }
        string[] parts = firstLine.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        int H = int.Parse(parts[0]);
        int W = int.Parse(parts[1]);
        
        byte[] grid = new byte[H * W];
        bool[] initiallyUnknown = new bool[H * W];
        
        for (int r = 0; r < H; r++) {
            string line = Console.ReadLine();
            if (line == null) line = "";
            for (int c = 0; c < W; c++) {
                byte val = (c < line.Length) ? (byte)line[c] : (byte)' ';
                grid[r * W + c] = val;
                initiallyUnknown[r * W + c] = (val == (byte)' ');
            }
        }
        
        // Solve the board
        SolveBoard(grid, H, W);
        
        // Output results
        List<int> safeCells = new List<int>();
        for (int i = 0; i < H * W; i++) {
            if (initiallyUnknown[i] && grid[i] == (byte)'S') {
                safeCells.Add(i);
            }
        }
        
        Console.WriteLine(safeCells.Count);
        foreach (int idx in safeCells) {
            int r = idx / W;
            int c = idx % W;
            Console.WriteLine($"{c} {r}");
        }
    }

    static void SolveBoard(byte[] grid, int H, int W) {
        int maxComponentSizeForBacktracking;
        int maxStates;
        long totalCells = (long)H * W;
        
        if (totalCells <= 1000) {
            maxComponentSizeForBacktracking = 40;
            maxStates = 1000000;
        } else if (totalCells <= 100000) {
            maxComponentSizeForBacktracking = 35;
            maxStates = 200000;
        } else {
            maxComponentSizeForBacktracking = 25;
            maxStates = 20000;
        }
        
        Queue<int> queue = new Queue<int>();
        bool[] inQueue = new bool[H * W];
        bool[] isConstraintActive = new bool[H * W];
        List<int> activeConstraints = new List<int>();
        
        for (int i = 0; i < H * W; i++) {
            byte val = grid[i];
            if (val >= (byte)'0' && val <= (byte)'8') {
                if (HasUnknownNeighbor(grid, i, H, W)) {
                    queue.Enqueue(i);
                    inQueue[i] = true;
                    activeConstraints.Add(i);
                    isConstraintActive[i] = true;
                }
            }
        }
        
        // Allocate reuseable data structures for component finding to avoid GC pressure
        bool[] isFrontier = new bool[H * W];
        bool[] visited = new bool[H * W];
        List<int> modifiedFrontier = new List<int>();
        List<int> modifiedVisited = new List<int>();
        
        bool globalProgress = true;
        while (globalProgress) {
            globalProgress = false;
            
            bool propProgress = true;
            while (propProgress) {
                propProgress = false;
                
                if (RunL1(grid, H, W, queue, inQueue, isConstraintActive)) {
                    propProgress = true;
                }
                
                if (RunL2(grid, H, W, queue, inQueue, isConstraintActive, activeConstraints)) {
                    propProgress = true;
                }
            }
            
            List<List<int>> components = FindComponents(grid, H, W, activeConstraints, isConstraintActive, isFrontier, visited, modifiedFrontier, modifiedVisited);
            
            if (components.Count > 0) {
                ComponentResult[] results = new ComponentResult[components.Count];
                
                Parallel.For(0, components.Count, i => {
                    var comp = components[i];
                    if (comp.Count > maxComponentSizeForBacktracking) {
                        results[i] = SolveComponent(grid, comp, H, W, Math.Min(maxStates, 5000));
                    } else {
                        results[i] = SolveComponent(grid, comp, H, W, maxStates);
                    }
                });
                
                for (int i = 0; i < components.Count; i++) {
                    var res = results[i];
                    if (res != null) {
                        foreach (int safeIdx in res.SafeIndices) {
                            if (grid[safeIdx] == (byte)' ') {
                                grid[safeIdx] = (byte)'S';
                                OnCellResolved(grid, safeIdx, H, W, isConstraintActive);
                                EnqueueNeighbors(grid, safeIdx, H, W, queue, inQueue);
                                globalProgress = true;
                            }
                        }
                        foreach (int mineIdx in res.MineIndices) {
                            if (grid[mineIdx] == (byte)' ') {
                                grid[mineIdx] = (byte)'*';
                                OnCellResolved(grid, mineIdx, H, W, isConstraintActive);
                                EnqueueNeighbors(grid, mineIdx, H, W, queue, inQueue);
                                globalProgress = true;
                            }
                        }
                    }
                }
            }
        }
    }

    static bool RunL1(byte[] grid, int H, int W, Queue<int> queue, bool[] inQueue, bool[] isConstraintActive) {
        bool changed = false;
        while (queue.Count > 0) {
            int idx = queue.Dequeue();
            inQueue[idx] = false;
            
            byte val = grid[idx];
            if (val < (byte)'0' || val > (byte)'8') continue;
            int V = val - (byte)'0';
            
            Int8 uNeighbors = new Int8();
            int countM = 0;
            
            int r = idx / W;
            int c = idx % W;
            for (int dr = -1; dr <= 1; dr++) {
                for (int dc = -1; dc <= 1; dc++) {
                    if (dr == 0 && dc == 0) continue;
                    int nr = r + dr;
                    int nc = c + dc;
                    if (nr >= 0 && nr < H && nc >= 0 && nc < W) {
                        int nidx = nr * W + nc;
                        byte nval = grid[nidx];
                        if (nval == (byte)' ') {
                            uNeighbors.Add(nidx);
                        } else if (nval == (byte)'*') {
                            countM++;
                        }
                    }
                }
            }
            
            if (uNeighbors.Count == 0) {
                isConstraintActive[idx] = false;
                continue;
            }
            
            if (countM == V) {
                for (int i = 0; i < uNeighbors.Count; i++) {
                    int nidx = uNeighbors.Get(i);
                    grid[nidx] = (byte)'S';
                    OnCellResolved(grid, nidx, H, W, isConstraintActive);
                    EnqueueNeighbors(grid, nidx, H, W, queue, inQueue);
                }
                changed = true;
            } else if (countM + uNeighbors.Count == V) {
                for (int i = 0; i < uNeighbors.Count; i++) {
                    int nidx = uNeighbors.Get(i);
                    grid[nidx] = (byte)'*';
                    OnCellResolved(grid, nidx, H, W, isConstraintActive);
                    EnqueueNeighbors(grid, nidx, H, W, queue, inQueue);
                }
                changed = true;
            }
        }
        return changed;
    }

    static bool RunL2(byte[] grid, int H, int W, Queue<int> queue, bool[] inQueue, bool[] isConstraintActive, List<int> activeConstraints) {
        activeConstraints.RemoveAll(cidx => !isConstraintActive[cidx]);
        
        bool progress = false;
        for (int i = 0; i < activeConstraints.Count; i++) {
            int c1 = activeConstraints[i];
            if (!isConstraintActive[c1]) continue;
            
            Int8 u1 = GetUnknownNeighbors(grid, c1, H, W);
            if (u1.Count == 0) {
                isConstraintActive[c1] = false;
                continue;
            }
            
            int v1 = grid[c1] - (byte)'0';
            int m1 = CountFlaggedNeighbors(grid, c1, H, W);
            int need1 = v1 - m1;
            
            Int64 candidates = new Int64();
            
            for (int j = 0; j < u1.Count; j++) {
                int uIdx = u1.Get(j);
                Int8 adjConstraints = GetAdjacentConstraints(grid, uIdx, H, W);
                for (int k = 0; k < adjConstraints.Count; k++) {
                    int c2 = adjConstraints.Get(k);
                    if (c2 > c1 && isConstraintActive[c2]) {
                        bool exists = false;
                        for (int L = 0; L < candidates.Count; L++) {
                            if (candidates.Get(L) == c2) {
                                exists = true;
                                break;
                            }
                        }
                        if (!exists) {
                            candidates.Add(c2);
                        }
                    }
                }
            }
            
            for (int j = 0; j < candidates.Count; j++) {
                int c2 = candidates.Get(j);
                if (!isConstraintActive[c2]) continue;
                
                Int8 u2 = GetUnknownNeighbors(grid, c2, H, W);
                if (u2.Count == 0) {
                    isConstraintActive[c2] = false;
                    continue;
                }
                
                int v2 = grid[c2] - (byte)'0';
                int m2 = CountFlaggedNeighbors(grid, c2, H, W);
                int need2 = v2 - m2;
                
                Int8 intersection = new Int8();
                Int8 d1 = new Int8();
                Int8 d2 = new Int8();
                
                for (int k = 0; k < u1.Count; k++) {
                    int val1 = u1.Get(k);
                    bool inU2 = false;
                    for (int L = 0; L < u2.Count; L++) {
                        if (u2.Get(L) == val1) {
                            inU2 = true;
                            break;
                        }
                    }
                    if (inU2) {
                        intersection.Add(val1);
                    } else {
                        d1.Add(val1);
                    }
                }
                
                for (int k = 0; k < u2.Count; k++) {
                    int val2 = u2.Get(k);
                    bool inU1 = false;
                    for (int L = 0; L < u1.Count; L++) {
                        if (u1.Get(L) == val2) {
                            inU1 = true;
                            break;
                        }
                    }
                    if (!inU1) {
                        d2.Add(val2);
                    }
                }
                
                int lower_I = Math.Max(0, Math.Max(need1 - d1.Count, need2 - d2.Count));
                int upper_I = Math.Min(intersection.Count, Math.Min(need1, need2));
                
                if (lower_I > upper_I) continue;
                
                bool localChanged = false;
                
                int min_D1 = need1 - upper_I;
                int max_D1 = need1 - lower_I;
                if (min_D1 == d1.Count && d1.Count > 0) {
                    for (int k = 0; k < d1.Count; k++) {
                        int nidx = d1.Get(k);
                        if (grid[nidx] != (byte)'*') {
                            grid[nidx] = (byte)'*';
                            OnCellResolved(grid, nidx, H, W, isConstraintActive);
                            EnqueueNeighbors(grid, nidx, H, W, queue, inQueue);
                            localChanged = true;
                        }
                    }
                } else if (max_D1 == 0 && d1.Count > 0) {
                    for (int k = 0; k < d1.Count; k++) {
                        int nidx = d1.Get(k);
                        if (grid[nidx] != (byte)'S') {
                            grid[nidx] = (byte)'S';
                            OnCellResolved(grid, nidx, H, W, isConstraintActive);
                            EnqueueNeighbors(grid, nidx, H, W, queue, inQueue);
                            localChanged = true;
                        }
                    }
                }
                
                int min_D2 = need2 - upper_I;
                int max_D2 = need2 - lower_I;
                if (min_D2 == d2.Count && d2.Count > 0) {
                    for (int k = 0; k < d2.Count; k++) {
                        int nidx = d2.Get(k);
                        if (grid[nidx] != (byte)'*') {
                            grid[nidx] = (byte)'*';
                            OnCellResolved(grid, nidx, H, W, isConstraintActive);
                            EnqueueNeighbors(grid, nidx, H, W, queue, inQueue);
                            localChanged = true;
                        }
                    }
                } else if (max_D2 == 0 && d2.Count > 0) {
                    for (int k = 0; k < d2.Count; k++) {
                        int nidx = d2.Get(k);
                        if (grid[nidx] != (byte)'S') {
                            grid[nidx] = (byte)'S';
                            OnCellResolved(grid, nidx, H, W, isConstraintActive);
                            EnqueueNeighbors(grid, nidx, H, W, queue, inQueue);
                            localChanged = true;
                        }
                    }
                }
                
                if (lower_I == intersection.Count && intersection.Count > 0) {
                    for (int k = 0; k < intersection.Count; k++) {
                        int nidx = intersection.Get(k);
                        if (grid[nidx] != (byte)'*') {
                            grid[nidx] = (byte)'*';
                            OnCellResolved(grid, nidx, H, W, isConstraintActive);
                            EnqueueNeighbors(grid, nidx, H, W, queue, inQueue);
                            localChanged = true;
                        }
                    }
                } else if (upper_I == 0 && intersection.Count > 0) {
                    for (int k = 0; k < intersection.Count; k++) {
                        int nidx = intersection.Get(k);
                        if (grid[nidx] != (byte)'S') {
                            grid[nidx] = (byte)'S';
                            OnCellResolved(grid, nidx, H, W, isConstraintActive);
                            EnqueueNeighbors(grid, nidx, H, W, queue, inQueue);
                            localChanged = true;
                        }
                    }
                }
                
                if (localChanged) {
                    progress = true;
                }
            }
        }
        return progress;
    }

    static List<List<int>> FindComponents(
        byte[] grid, int H, int W, 
        List<int> activeConstraints, bool[] isConstraintActive,
        bool[] isFrontier, bool[] visited,
        List<int> modifiedFrontier, List<int> modifiedVisited
    ) {
        modifiedFrontier.Clear();
        modifiedVisited.Clear();
        
        List<int> frontierCells = new List<int>();
        
        foreach (int cidx in activeConstraints) {
            if (!isConstraintActive[cidx]) continue;
            
            Int8 uNeighbors = GetUnknownNeighbors(grid, cidx, H, W);
            for (int i = 0; i < uNeighbors.Count; i++) {
                int nidx = uNeighbors.Get(i);
                if (!isFrontier[nidx]) {
                    isFrontier[nidx] = true;
                    modifiedFrontier.Add(nidx);
                    frontierCells.Add(nidx);
                }
            }
        }
        
        List<List<int>> components = new List<List<int>>();
        Queue<int> bfsQueue = new Queue<int>();
        
        foreach (int startCell in frontierCells) {
            if (visited[startCell]) continue;
            
            List<int> comp = new List<int>();
            bfsQueue.Enqueue(startCell);
            visited[startCell] = true;
            modifiedVisited.Add(startCell);
            
            while (bfsQueue.Count > 0) {
                int curr = bfsQueue.Dequeue();
                comp.Add(curr);
                
                Int8 adjConstraints = GetAdjacentConstraints(grid, curr, H, W);
                
                for (int k = 0; k < adjConstraints.Count; k++) {
                    int cidx = adjConstraints.Get(k);
                    if (!isConstraintActive[cidx]) continue;
                    
                    Int8 uNeighbors = GetUnknownNeighbors(grid, cidx, H, W);
                    for (int L = 0; L < uNeighbors.Count; L++) {
                        int nidx = uNeighbors.Get(L);
                        if (!visited[nidx]) {
                            visited[nidx] = true;
                            modifiedVisited.Add(nidx);
                            bfsQueue.Enqueue(nidx);
                        }
                    }
                }
            }
            components.Add(comp);
        }
        
        foreach (int idx in modifiedFrontier) {
            isFrontier[idx] = false;
        }
        foreach (int idx in modifiedVisited) {
            visited[idx] = false;
        }
        
        return components;
    }

    static ComponentResult SolveComponent(byte[] grid, List<int> comp, int H, int W, int maxStates) {
        ComponentResult res = new ComponentResult();
        
        SolverState state = BuildSolverState(grid, comp, H, W);
        state.maxStates = maxStates;
        
        Search(state);
        
        if (!state.aborted && state.validSolutionsCount > 0) {
            res.Completed = true;
            for (int i = 0; i < state.vars.Length; i++) {
                if (state.canBeSafe[i] && !state.canBeMine[i]) {
                    res.SafeIndices.Add(state.vars[i]);
                }
                if (!state.canBeSafe[i] && state.canBeMine[i]) {
                    res.MineIndices.Add(state.vars[i]);
                }
            }
        } else {
            res.Completed = false;
        }
        
        return res;
    }

    static SolverState BuildSolverState(byte[] grid, List<int> comp, int H, int W) {
        int[] vars = comp.ToArray();
        int[] consts = GetAffectedConstraints(grid, comp, H, W);
        
        ConstraintState[] cStates = new ConstraintState[consts.Length];
        VarConsts[] varToConsts = new VarConsts[vars.Length];
        
        for (int i = 0; i < consts.Length; i++) {
            int cidx = consts[i];
            int val = grid[cidx] - (byte)'0';
            int m = CountFlaggedNeighbors(grid, cidx, H, W);
            
            Int8 uNeighbors = GetUnknownNeighbors(grid, cidx, H, W);
            
            cStates[i] = new ConstraintState {
                target = val - m,
                currentMines = 0,
                unassignedCount = 0,
                varCount = 0
            };
            
            for (int j = 0; j < uNeighbors.Count; j++) {
                int vIdx = GetVarIndex(vars, uNeighbors.Get(j));
                if (vIdx != -1) {
                    cStates[i].AddVar(vIdx);
                    cStates[i].unassignedCount++;
                    varToConsts[vIdx].Add(i);
                }
            }
        }
        
        int[] assignment = new int[vars.Length];
        for (int i = 0; i < vars.Length; i++) assignment[i] = -1;
        
        return new SolverState {
            vars = vars,
            assignment = assignment,
            cStates = cStates,
            varToConsts = varToConsts,
            canBeSafe = new bool[vars.Length],
            canBeMine = new bool[vars.Length],
            validSolutionsCount = 0,
            stateCount = 0,
            maxStates = 100000,
            aborted = false
        };
    }

    static int[] GetAffectedConstraints(byte[] grid, List<int> vars, int H, int W) {
        int maxConsts = 8 * vars.Count;
        int[] temp = new int[maxConsts];
        int count = 0;
        foreach (int v in vars) {
            Int8 adj = GetAdjacentConstraints(grid, v, H, W);
            for (int i = 0; i < adj.Count; i++) {
                temp[count++] = adj.Get(i);
            }
        }
        
        Array.Sort(temp, 0, count);
        int uniqueCount = 0;
        for (int i = 0; i < count; i++) {
            if (i == 0 || temp[i] != temp[i - 1]) {
                temp[uniqueCount++] = temp[i];
            }
        }
        
        int[] res = new int[uniqueCount];
        Array.Copy(temp, res, uniqueCount);
        return res;
    }

    static int GetVarIndex(int[] vars, int val) {
        for (int i = 0; i < vars.Length; i++) {
            if (vars[i] == val) return i;
        }
        return -1;
    }

    static void Search(SolverState s) {
        if (s.aborted) return;
        s.stateCount++;
        if (s.stateCount > s.maxStates) {
            s.aborted = true;
            return;
        }
        
        int vIdx = GetNextVar(s.cStates, s.assignment);
        if (vIdx == -1) {
            s.validSolutionsCount++;
            for (int i = 0; i < s.assignment.Length; i++) {
                if (s.assignment[i] == 0) s.canBeSafe[i] = true;
                else if (s.assignment[i] == 1) s.canBeMine[i] = true;
            }
            return;
        }
        
        for (int val = 0; val <= 1; val++) {
            bool valid = true;
            s.assignment[vIdx] = val;
            
            int constCount = s.varToConsts[vIdx].Count;
            for (int k = 0; k < constCount; k++) {
                int cIdx = s.varToConsts[vIdx].Get(k);
                s.cStates[cIdx].unassignedCount--;
                if (val == 1) {
                    s.cStates[cIdx].currentMines++;
                }
                
                int target = s.cStates[cIdx].target;
                int cur = s.cStates[cIdx].currentMines;
                int un = s.cStates[cIdx].unassignedCount;
                if (cur > target || cur + un < target) {
                    valid = false;
                }
            }
            
            if (valid) {
                Search(s);
            }
            
            for (int k = 0; k < constCount; k++) {
                int cIdx = s.varToConsts[vIdx].Get(k);
                s.cStates[cIdx].unassignedCount++;
                if (val == 1) {
                    s.cStates[cIdx].currentMines--;
                }
            }
            s.assignment[vIdx] = -1;
        }
    }

    static int GetNextVar(ConstraintState[] cStates, int[] assignment) {
        int bestVar = -1;
        int minUnassigned = int.MaxValue;
        
        for (int i = 0; i < cStates.Length; i++) {
            int uc = cStates[i].unassignedCount;
            if (uc > 0 && uc < minUnassigned) {
                for (int k = 0; k < cStates[i].varCount; k++) {
                    int vIdx = cStates[i].GetVar(k);
                    if (assignment[vIdx] == -1) {
                        bestVar = vIdx;
                        minUnassigned = uc;
                        break;
                    }
                }
            }
        }
        
        if (bestVar != -1) return bestVar;
        
        for (int i = 0; i < assignment.Length; i++) {
            if (assignment[i] == -1) return i;
        }
        
        return -1;
    }

    static bool HasUnknownNeighbor(byte[] grid, int idx, int H, int W) {
        int r = idx / W;
        int c = idx % W;
        for (int dr = -1; dr <= 1; dr++) {
            for (int dc = -1; dc <= 1; dc++) {
                if (dr == 0 && dc == 0) continue;
                int nr = r + dr;
                int nc = c + dc;
                if (nr >= 0 && nr < H && nc >= 0 && nc < W) {
                    if (grid[nr * W + nc] == (byte)' ') return true;
                }
            }
        }
        return false;
    }

    static Int8 GetUnknownNeighbors(byte[] grid, int idx, int H, int W) {
        Int8 neighbors = new Int8();
        int r = idx / W;
        int c = idx % W;
        for (int dr = -1; dr <= 1; dr++) {
            for (int dc = -1; dc <= 1; dc++) {
                if (dr == 0 && dc == 0) continue;
                int nr = r + dr;
                int nc = c + dc;
                if (nr >= 0 && nr < H && nc >= 0 && nc < W) {
                    int nidx = nr * W + nc;
                    if (grid[nidx] == (byte)' ') {
                        neighbors.Add(nidx);
                    }
                }
            }
        }
        return neighbors;
    }

    static int CountFlaggedNeighbors(byte[] grid, int idx, int H, int W) {
        int count = 0;
        int r = idx / W;
        int c = idx % W;
        for (int dr = -1; dr <= 1; dr++) {
            for (int dc = -1; dc <= 1; dc++) {
                if (dr == 0 && dc == 0) continue;
                int nr = r + dr;
                int nc = c + dc;
                if (nr >= 0 && nr < H && nc >= 0 && nc < W) {
                    if (grid[nr * W + nc] == (byte)'*') {
                        count++;
                    }
                }
            }
        }
        return count;
    }

    static Int8 GetAdjacentConstraints(byte[] grid, int idx, int H, int W) {
        Int8 constraints = new Int8();
        int r = idx / W;
        int c = idx % W;
        for (int dr = -1; dr <= 1; dr++) {
            for (int dc = -1; dc <= 1; dc++) {
                if (dr == 0 && dc == 0) continue;
                int nr = r + dr;
                int nc = c + dc;
                if (nr >= 0 && nr < H && nc >= 0 && nc < W) {
                    int nidx = nr * W + nc;
                    byte val = grid[nidx];
                    if (val >= (byte)'0' && val <= (byte)'8') {
                        constraints.Add(nidx);
                    }
                }
            }
        }
        return constraints;
    }

    static void EnqueueNeighbors(byte[] grid, int idx, int H, int W, Queue<int> queue, bool[] inQueue) {
        int r = idx / W;
        int c = idx % W;
        for (int dr = -1; dr <= 1; dr++) {
            for (int dc = -1; dc <= 1; dc++) {
                if (dr == 0 && dc == 0) continue;
                int nr = r + dr;
                int nc = c + dc;
                if (nr >= 0 && nr < H && nc >= 0 && nc < W) {
                    int nidx = nr * W + nc;
                    byte nval = grid[nidx];
                    if (nval >= (byte)'0' && nval <= (byte)'8') {
                        if (!inQueue[nidx]) {
                            queue.Enqueue(nidx);
                            inQueue[nidx] = true;
                        }
                    }
                }
            }
        }
    }

    static void OnCellResolved(byte[] grid, int idx, int H, int W, bool[] isConstraintActive) {
        int r = idx / W;
        int c = idx % W;
        for (int dr = -1; dr <= 1; dr++) {
            for (int dc = -1; dc <= 1; dc++) {
                if (dr == 0 && dc == 0) continue;
                int nr = r + dr;
                int nc = c + dc;
                if (nr >= 0 && nr < H && nc >= 0 && nc < W) {
                    int nidx = nr * W + nc;
                    byte val = grid[nidx];
                    if (val >= (byte)'0' && val <= (byte)'8') {
                        if (isConstraintActive[nidx]) {
                            if (!HasUnknownNeighbor(grid, nidx, H, W)) {
                                isConstraintActive[nidx] = false;
                            }
                        }
                    }
                }
            }
        }
    }
