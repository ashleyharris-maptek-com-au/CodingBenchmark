use std::io::{self, BufRead, Write};
use std::collections::VecDeque;

const WALL_MASK: u8 = 1 << 0;
const VISITED_FWD_MASK: u8 = 1 << 1;
const VISITED_BWD_MASK: u8 = 1 << 2;
const FWD_DIR_SHIFT: u32 = 3;
const BWD_DIR_SHIFT: u32 = 5;

const FWD_DIR_MASK: u8 = 0b00011000;
const BWD_DIR_MASK: u8 = 0b01100000;

const DIR_UP: u8 = 0;
const DIR_DOWN: u8 = 1;
const DIR_LEFT: u8 = 2;
const DIR_RIGHT: u8 = 3;

#[inline(always)]
fn check_and_visit_fwd(
    next: u32,
    dir_to_parent: u8,
    grid: &mut [u8],
    q_fwd: &mut VecDeque<u64>,
    next_coord: u32,
) -> Option<(u32, u32)> {
    let idx = next as usize;
    // SAFETY: next is mathematically guaranteed to be within bounds
    let val = unsafe { *grid.get_unchecked(idx) };
    if (val & WALL_MASK) != 0 {
        return None;
    }
    if (val & VISITED_FWD_MASK) != 0 {
        return None;
    }
    if (val & VISITED_BWD_MASK) != 0 {
        let dir_bits = dir_to_parent << FWD_DIR_SHIFT;
        unsafe {
            *grid.get_unchecked_mut(idx) |= VISITED_FWD_MASK | dir_bits;
        }
        return Some((next, next));
    }
    let dir_bits = dir_to_parent << FWD_DIR_SHIFT;
    unsafe {
        *grid.get_unchecked_mut(idx) |= VISITED_FWD_MASK | dir_bits;
    }
    q_fwd.push_back(((next as u64) << 32) | (next_coord as u64));
    None
}

#[inline(always)]
fn check_and_visit_bwd(
    next: u32,
    dir_to_parent: u8,
    grid: &mut [u8],
    q_bwd: &mut VecDeque<u64>,
    next_coord: u32,
) -> Option<(u32, u32)> {
    let idx = next as usize;
    // SAFETY: next is mathematically guaranteed to be within bounds
    let val = unsafe { *grid.get_unchecked(idx) };
    if (val & WALL_MASK) != 0 {
        return None;
    }
    if (val & VISITED_BWD_MASK) != 0 {
        return None;
    }
    if (val & VISITED_FWD_MASK) != 0 {
        let dir_bits = dir_to_parent << BWD_DIR_SHIFT;
        unsafe {
            *grid.get_unchecked_mut(idx) |= VISITED_BWD_MASK | dir_bits;
        }
        return Some((next, next));
    }
    let dir_bits = dir_to_parent << BWD_DIR_SHIFT;
    unsafe {
        *grid.get_unchecked_mut(idx) |= VISITED_BWD_MASK | dir_bits;
    }
    q_bwd.push_back(((next as u64) << 32) | (next_coord as u64));
    None
}

fn reconstruct_path(
    grid: &[u8],
    w: u32,
    fwd_connect: u32,
    bwd_connect: u32,
    start_idx: u32,
    goal_idx: u32,
) -> Vec<u32> {
    let mut fwd_path = Vec::new();
    let mut curr = fwd_connect;
    fwd_path.push(curr);
    while curr != start_idx {
        let parent_dir = (grid[curr as usize] & FWD_DIR_MASK) >> FWD_DIR_SHIFT;
        curr = match parent_dir {
            DIR_UP => curr - w,
            DIR_DOWN => curr + w,
            DIR_LEFT => curr - 1,
            DIR_RIGHT => curr + 1,
            _ => unreachable!(),
        };
        fwd_path.push(curr);
    }
    fwd_path.reverse();

    let mut bwd_path = Vec::new();
    let mut curr = bwd_connect;
    bwd_path.push(curr);
    while curr != goal_idx {
        let parent_dir = (grid[curr as usize] & BWD_DIR_MASK) >> BWD_DIR_SHIFT;
        curr = match parent_dir {
            DIR_UP => curr - w,
            DIR_DOWN => curr + w,
            DIR_LEFT => curr - 1,
            DIR_RIGHT => curr + 1,
            _ => unreachable!(),
        };
        bwd_path.push(curr);
    }

    fwd_path.extend(bwd_path.iter().skip(1));
    fwd_path
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let stdin = io::stdin();
    let mut reader = stdin.lock();

    let mut header = String::new();
    reader.read_line(&mut header)?;
    let parts: Vec<&str> = header.split_whitespace().collect();
    if parts.len() < 2 {
        return Ok(());
    }
    let h: u32 = parts[0].parse()?;
    let w: u32 = parts[1].parse()?;

    if h == 0 || w == 0 {
        return Ok(());
    }

    let mut grid = vec![0u8; (h as usize) * (w as usize)];
    let mut start_idx = None;
    let mut goal_idx = None;

    let mut buf = Vec::with_capacity(w as usize + 16);
    for r in 0..h {
        buf.clear();
        if reader.read_until(b'\n', &mut buf)? == 0 {
            break;
        }
        let limit = std::cmp::min(w as usize, buf.len());
        for c in 0..limit {
            let b = buf[c];
            let idx = r * w + (c as u32);
            match b {
                b'#' => {
                    grid[idx as usize] = WALL_MASK;
                }
                b'A' => {
                    start_idx = Some(idx);
                }
                b'B' => {
                    goal_idx = Some(idx);
                }
                _ => {}
            }
        }
    }

    let start_idx = match start_idx {
        Some(idx) => idx,
        None => return Ok(()),
    };
    let goal_idx = match goal_idx {
        Some(idx) => idx,
        None => return Ok(()),
    };

    if start_idx == goal_idx {
        let stdout = io::stdout();
        let mut writer = io::BufWriter::new(stdout.lock());
        writeln!(writer, "1")?;
        let x = start_idx % w;
        let y = start_idx / w;
        writeln!(writer, "{} {}", x, y)?;
        return Ok(());
    }

    let mut q_fwd = VecDeque::new();
    let mut q_bwd = VecDeque::new();

    grid[start_idx as usize] |= VISITED_FWD_MASK;
    grid[goal_idx as usize] |= VISITED_BWD_MASK;

    let start_y = start_idx / w;
    let start_x = start_idx % w;
    let start_coord = (start_y << 16) | start_x;

    let goal_y = goal_idx / w;
    let goal_x = goal_idx % w;
    let goal_coord = (goal_y << 16) | goal_x;

    q_fwd.push_back(((start_idx as u64) << 32) | (start_coord as u64));
    q_bwd.push_back(((goal_idx as u64) << 32) | (goal_coord as u64));

    let mut connection = None;

    while !q_fwd.is_empty() && !q_bwd.is_empty() {
        if q_fwd.len() <= q_bwd.len() {
            let entry = q_fwd.pop_front().unwrap();
            let curr = (entry >> 32) as u32;
            let coord = (entry & 0xFFFFFFFF) as u32;
            let curr_x = coord & 0xFFFF;
            let curr_y = coord >> 16;

            if curr_y > 0 {
                let next = curr - w;
                let next_coord = coord - 0x10000;
                if let Some(conn) = check_and_visit_fwd(next, DIR_DOWN, &mut grid, &mut q_fwd, next_coord) {
                    connection = Some(conn);
                    break;
                }
            }
            if curr_y < h - 1 {
                let next = curr + w;
                let next_coord = coord + 0x10000;
                if let Some(conn) = check_and_visit_fwd(next, DIR_UP, &mut grid, &mut q_fwd, next_coord) {
                    connection = Some(conn);
                    break;
                }
            }
            if curr_x > 0 {
                let next = curr - 1;
                let next_coord = coord - 1;
                if let Some(conn) = check_and_visit_fwd(next, DIR_RIGHT, &mut grid, &mut q_fwd, next_coord) {
                    connection = Some(conn);
                    break;
                }
            }
            if curr_x < w - 1 {
                let next = curr + 1;
                let next_coord = coord + 1;
                if let Some(conn) = check_and_visit_fwd(next, DIR_LEFT, &mut grid, &mut q_fwd, next_coord) {
                    connection = Some(conn);
                    break;
                }
            }
        } else {
            let entry = q_bwd.pop_front().unwrap();
            let curr = (entry >> 32) as u32;
            let coord = (entry & 0xFFFFFFFF) as u32;
            let curr_x = coord & 0xFFFF;
            let curr_y = coord >> 16;

            if curr_y > 0 {
                let next = curr - w;
                let next_coord = coord - 0x10000;
                if let Some(conn) = check_and_visit_bwd(next, DIR_DOWN, &mut grid, &mut q_bwd, next_coord) {
                    connection = Some(conn);
                    break;
                }
            }
            if curr_y < h - 1 {
                let next = curr + w;
                let next_coord = coord + 0x10000;
                if let Some(conn) = check_and_visit_bwd(next, DIR_UP, &mut grid, &mut q_bwd, next_coord) {
                    connection = Some(conn);
                    break;
                }
            }
            if curr_x > 0 {
                let next = curr - 1;
                let next_coord = coord - 1;
                if let Some(conn) = check_and_visit_bwd(next, DIR_RIGHT, &mut grid, &mut q_bwd, next_coord) {
                    connection = Some(conn);
                    break;
                }
            }
            if curr_x < w - 1 {
                let next = curr + 1;
                let next_coord = coord + 1;
                if let Some(conn) = check_and_visit_bwd(next, DIR_LEFT, &mut grid, &mut q_bwd, next_coord) {
                    connection = Some(conn);
                    break;
                }
            }
        }
    }

    if let Some((fwd_conn, bwd_conn)) = connection {
        let path = reconstruct_path(&grid, w, fwd_conn, bwd_conn, start_idx, goal_idx);
        let stdout = io::stdout();
        let mut writer = io::BufWriter::new(stdout.lock());
        writeln!(writer, "{}", path.len())?;
        for &idx in &path {
            let x = idx % w;
            let y = idx / w;
            writeln!(writer, "{} {}", x, y)?;
        }
    }

    Ok(())
}
