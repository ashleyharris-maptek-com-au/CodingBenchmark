#version 450

layout(local_size_x = 256) in;

// Each boid stored as two vec4s: [pos.xyz, 0], [vel.xyz, 0]
// Boid i is at index i*2 and i*2+1
layout(set = 0, binding = 0) readonly buffer InputBoids {
    vec4 inData[];   // length = N * 2
};

layout(set = 0, binding = 1) buffer OutputBoids {
    vec4 outData[];  // length = N * 2
};

layout(set = 0, binding = 2) uniform Params {
    uvec4 counts;      // counts.x = N
    vec4 radii;        // radii.x=dt, radii.y=sepRadius, radii.z=alignRadius, radii.w=cohRadius
    vec4 weights;      // weights.x=sepWeight, weights.y=alignWeight, weights.z=cohWeight, weights.w=maxSpeed
    vec4 bounds;       // bounds.x=boundSize, bounds.y=boundForce
};

void main() {
    uint index = gl_GlobalInvocationID.x;
    uint N = counts.x;
    if (index >= N) {
        return;
    }

    vec3 pos = inData[index * 2].xyz;
    vec3 vel = inData[index * 2 + 1].xyz;

    vec3 sepAccum = vec3(0.0);
    vec3 alignAccum = vec3(0.0);
    vec3 cohAccum = vec3(0.0);

    uint sepCount = 0;
    uint alignCount = 0;
    uint cohCount = 0;

    for (uint j = 0; j < N; j++) {
        if (index == j) {
            continue;
        }

        vec3 neighbor_pos = inData[j * 2].xyz;
        vec3 neighbor_vel = inData[j * 2 + 1].xyz;

        float dist = distance(pos, neighbor_pos);
        if (dist < 1e-6) {
            continue; // Skip zero-distance neighbors
        }

        // Separation (Rule 1)
        if (dist < radii.y) {
            sepAccum += (pos - neighbor_pos) / dist;
            sepCount++;
        }

        // Alignment (Rule 2)
        if (dist < radii.z) {
            alignAccum += neighbor_vel;
            alignCount++;
        }

        // Cohesion (Rule 3)
        if (dist < radii.w) {
            cohAccum += neighbor_pos;
            cohCount++;
        }
    }

    vec3 sepForce = vec3(0.0);
    if (sepCount > 0) {
        sepForce = sepAccum / float(sepCount);
    }

    vec3 alignForce = vec3(0.0);
    if (alignCount > 0) {
        alignForce = (alignAccum / float(alignCount)) - vel;
    }

    vec3 cohForce = vec3(0.0);
    if (cohCount > 0) {
        cohForce = (cohAccum / float(cohCount)) - pos;
    }

    // Boundary avoidance
    float boundSize = bounds.x;
    float boundForce = bounds.y;
    vec3 boundaryForce = vec3(0.0);

    if (abs(pos.x) > boundSize * 0.4) {
        boundaryForce.x = -pos.x * boundForce / (boundSize * 0.5);
    }
    if (abs(pos.y) > boundSize * 0.4) {
        boundaryForce.y = -pos.y * boundForce / (boundSize * 0.5);
    }
    if (abs(pos.z) > boundSize * 0.4) {
        boundaryForce.z = -pos.z * boundForce / (boundSize * 0.5);
    }

    // Total acceleration
    vec3 accel = sepForce * weights.x + alignForce * weights.y + cohForce * weights.z + boundaryForce;

    // Kinematic update
    vec3 newVel = vel + accel * radii.x; // dt
    float speed = length(newVel);
    if (speed > weights.w) { // maxSpeed
        newVel = (newVel / speed) * weights.w;
    }

    vec3 newPos = pos + newVel * radii.x; // dt

    // Write output
    outData[index * 2] = vec4(newPos, 0.0);
    outData[index * 2 + 1] = vec4(newVel, 0.0);
}
