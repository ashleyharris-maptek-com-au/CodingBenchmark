#[derive(Clone, Copy, Debug)]
struct Point {
    x: f64,
    y: f64,
}

struct OriginalPiece {
    _id: usize,
    vertices: Vec<Point>,
    area: f64,
}

#[derive(Clone)]
struct RotatedPieceVersion {
    _rotation_idx: usize,
    rotation_angle: f64,
    vertices: Vec<Point>, // normalized to start at (0, 0)
    width: f64,
    height: f64,
    p_min: Point, // original min coordinates after rotation
}

struct PlacedPiece {
    vertices: Vec<Point>,
    aabb: (Point, Point),
}

struct StockGridCell {
    is_inside: bool,
    is_outside: bool,
    edges: Vec<usize>,
}

struct StockGrid {
    xmin: f64,
    ymin: f64,
    xmax: f64,
    ymax: f64,
    stock_xmin: f64,
    stock_ymin: f64,
    stock_xmax: f64,
    stock_ymax: f64,
    cell_w: f64,
    cell_h: f64,
    cols: usize,
    rows: usize,
    cells: Vec<StockGridCell>,
}

struct PlacementGrid {
    xmin: f64,
    ymin: f64,
    cell_w: f64,
    cell_h: f64,
    cols: usize,
    rows: usize,
    cells: Vec<Vec<usize>>,
}

struct Lcg {
    state: u64,
}

impl Lcg {
    fn new(seed: u64) -> Self {
        Lcg { state: seed }
    }
    
    fn next_u32(&mut self) -> u32 {
        self.state = self.state.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
        (self.state >> 32) as u32
    }
    
    fn next_f64(&mut self) -> f64 {
        (self.next_u32() as f64) / (u32::MAX as f64)
    }
}

// Geometry helper functions
fn polygon_area(vertices: &[Point]) -> f64 {
    let mut area = 0.0;
    let n = vertices.len();
    let mut j = n - 1;
    for i in 0..n {
        area += (vertices[j].x + vertices[i].x) * (vertices[j].y - vertices[i].y);
        j = i;
    }
    area.abs() * 0.5
}

fn get_centroid(vertices: &[Point]) -> Point {
    let mut sum_x = 0.0;
    let mut sum_y = 0.0;
    for p in vertices {
        sum_x += p.x;
        sum_y += p.y;
    }
    Point {
        x: sum_x / vertices.len() as f64,
        y: sum_y / vertices.len() as f64,
    }
}

fn point_on_segment(p: Point, a: Point, b: Point) -> bool {
    let cross = (p.y - a.y) * (b.x - a.x) - (p.x - a.x) * (b.y - a.y);
    if cross.abs() > 1e-7 {
        return false;
    }
    let dot = (p.x - a.x) * (b.x - a.x) + (p.y - a.y) * (b.y - a.y);
    if dot < -1e-7 {
        return false;
    }
    let len_sq = (b.x - a.x) * (b.x - a.x) + (b.y - a.y) * (b.y - a.y);
    if dot > len_sq + 1e-7 {
        return false;
    }
    true
}

fn point_in_polygon(p: Point, poly: &[Point]) -> bool {
    let mut inside = false;
    let mut j = poly.len() - 1;
    for i in 0..poly.len() {
        let pi = poly[i];
        let pj = poly[j];
        if (pi.y > p.y) != (pj.y > p.y) {
            let intersect_x = (pj.x - pi.x) * (p.y - pi.y) / (pj.y - pi.y) + pi.x;
            if p.x < intersect_x {
                inside = !inside;
            }
        }
        j = i;
    }
    inside
}

fn point_in_polygon_or_boundary(p: Point, poly: &[Point]) -> bool {
    for i in 0..poly.len() {
        let a = poly[i];
        let b = poly[(i + 1) % poly.len()];
        if point_on_segment(p, a, b) {
            return true;
        }
    }
    point_in_polygon(p, poly)
}

fn point_strictly_in_polygon(p: Point, poly: &[Point]) -> bool {
    for i in 0..poly.len() {
        let a = poly[i];
        let b = poly[(i + 1) % poly.len()];
        if point_on_segment(p, a, b) {
            return false;
        }
    }
    point_in_polygon(p, poly)
}

fn segments_cross(a: Point, b: Point, c: Point, d: Point) -> bool {
    let cp1 = (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x);
    let cp2 = (b.x - a.x) * (d.y - a.y) - (b.y - a.y) * (d.x - a.x);
    let cp3 = (d.x - c.x) * (a.y - c.y) - (d.y - c.y) * (a.x - c.x);
    let cp4 = (d.x - c.x) * (b.y - c.y) - (d.y - c.y) * (b.x - c.x);

    let eps = 1e-7;
    if ((cp1 > eps && cp2 < -eps) || (cp1 < -eps && cp2 > eps)) &&
       ((cp3 > eps && cp4 < -eps) || (cp3 < -eps && cp4 > eps)) {
        return true;
    }
    false
}

fn segments_cross_or_touch(a: Point, b: Point, c: Point, d: Point) -> bool {
    let cp1 = (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x);
    let cp2 = (b.x - a.x) * (d.y - a.y) - (b.y - a.y) * (d.x - a.x);
    let cp3 = (d.x - c.x) * (a.y - c.y) - (d.y - c.y) * (a.x - c.x);
    let cp4 = (d.x - c.x) * (b.y - c.y) - (d.y - c.y) * (b.x - c.x);

    let s1 = cp1.signum();
    let s2 = cp2.signum();
    let s3 = cp3.signum();
    let s4 = cp4.signum();
    
    (s1 * s2 <= 0.0) && (s3 * s4 <= 0.0) && !(s1 == 0.0 && s2 == 0.0 && s3 == 0.0 && s4 == 0.0)
}

fn segment_intersects_box(a: Point, b: Point, xmin: f64, xmax: f64, ymin: f64, ymax: f64) -> bool {
    if a.x >= xmin && a.x <= xmax && a.y >= ymin && a.y <= ymax { return true; }
    if b.x >= xmin && b.x <= xmax && b.y >= ymin && b.y <= ymax { return true; }
    
    let p1 = Point { x: xmin, y: ymin };
    let p2 = Point { x: xmax, y: ymin };
    let p3 = Point { x: xmax, y: ymax };
    let p4 = Point { x: xmin, y: ymax };
    
    if segments_cross_or_touch(a, b, p1, p2) { return true; }
    if segments_cross_or_touch(a, b, p2, p3) { return true; }
    if segments_cross_or_touch(a, b, p3, p4) { return true; }
    if segments_cross_or_touch(a, b, p4, p1) { return true; }
    
    false
}

impl StockGrid {
    fn new(poly: &[Point], cols: usize, rows: usize) -> Self {
        let mut xmin = f64::MAX;
        let mut ymin = f64::MAX;
        let mut xmax = f64::MIN;
        let mut ymax = f64::MIN;
        for p in poly {
            if p.x < xmin { xmin = p.x; }
            if p.y < ymin { ymin = p.y; }
            if p.x > xmax { xmax = p.x; }
            if p.y > ymax { ymax = p.y; }
        }
        
        let stock_xmin = xmin;
        let stock_ymin = ymin;
        let stock_xmax = xmax;
        let stock_ymax = ymax;
        
        let dx = (xmax - xmin) * 0.01;
        let dy = (ymax - ymin) * 0.01;
        xmin -= dx;
        xmax += dx;
        ymin -= dy;
        ymax += dy;
        
        let cell_w = (xmax - xmin) / cols as f64;
        let cell_h = (ymax - ymin) / rows as f64;
        
        let mut cells = Vec::with_capacity(cols * rows);
        for _ in 0..(cols * rows) {
            cells.push(StockGridCell {
                is_inside: false,
                is_outside: false,
                edges: Vec::new(),
            });
        }
        
        let n = poly.len();
        for i in 0..n {
            let a = poly[i];
            let b = poly[(i + 1) % n];
            
            let seg_xmin = a.x.min(b.x);
            let seg_xmax = a.x.max(b.x);
            let seg_ymin = a.y.min(b.y);
            let seg_ymax = a.y.max(b.y);
            
            let col_start = (((seg_xmin - xmin) / cell_w).floor() as isize).max(0) as usize;
            let col_end = (((seg_xmax - xmin) / cell_w).floor() as isize).max(0) as usize;
            let col_end = col_end.min(cols - 1);
            
            let row_start = (((seg_ymin - ymin) / cell_h).floor() as isize).max(0) as usize;
            let row_end = (((seg_ymax - ymin) / cell_h).floor() as isize).max(0) as usize;
            let row_end = row_end.min(rows - 1);
            
            for r in row_start..=row_end {
                for c in col_start..=col_end {
                    let cell_xmin = xmin + c as f64 * cell_w;
                    let cell_xmax = cell_xmin + cell_w;
                    let cell_ymin = ymin + r as f64 * cell_h;
                    let cell_ymax = cell_ymin + cell_h;
                    
                    if segment_intersects_box(a, b, cell_xmin, cell_xmax, cell_ymin, cell_ymax) {
                        cells[r * cols + c].edges.push(i);
                    }
                }
            }
        }
        
        for r in 0..rows {
            for c in 0..cols {
                let idx = r * cols + c;
                if cells[idx].edges.is_empty() {
                    let cell_xmin = xmin + c as f64 * cell_w;
                    let cell_ymin = ymin + r as f64 * cell_h;
                    let center = Point {
                        x: cell_xmin + cell_w * 0.5,
                        y: cell_ymin + cell_h * 0.5,
                    };
                    if point_in_polygon(center, poly) {
                        cells[idx].is_inside = true;
                    } else {
                        cells[idx].is_outside = true;
                    }
                }
            }
        }
        
        StockGrid {
            xmin, ymin, xmax, ymax,
            stock_xmin, stock_ymin, stock_xmax, stock_ymax,
            cell_w, cell_h,
            cols, rows,
            cells,
        }
    }
    
    fn point_in_stock(&self, p: Point, poly: &[Point]) -> bool {
        if p.x < self.xmin || p.x > self.xmax || p.y < self.ymin || p.y > self.ymax {
            return false;
        }
        let c = (((p.x - self.xmin) / self.cell_w).floor() as isize).max(0) as usize;
        let c = c.min(self.cols - 1);
        let r = (((p.y - self.ymin) / self.cell_h).floor() as isize).max(0) as usize;
        let r = r.min(self.rows - 1);
        
        let idx = r * self.cols + c;
        let cell = &self.cells[idx];
        if cell.is_inside {
            return true;
        }
        if cell.is_outside {
            return false;
        }
        
        point_in_polygon_or_boundary(p, poly)
    }
    
    fn segment_intersects_stock(&self, a: Point, b: Point, poly: &[Point]) -> bool {
        let seg_xmin = a.x.min(b.x);
        let seg_xmax = a.x.max(b.x);
        let seg_ymin = a.y.min(b.y);
        let seg_ymax = a.y.max(b.y);
        
        if seg_xmin > self.xmax || seg_xmax < self.xmin || seg_ymin > self.ymax || seg_ymax < self.ymin {
            return true;
        }
        
        let col_start = (((seg_xmin - self.xmin) / self.cell_w).floor() as isize).max(0) as usize;
        let col_end = (((seg_xmax - self.xmin) / self.cell_w).floor() as isize).max(0) as usize;
        let col_end = col_end.min(self.cols - 1);
        
        let row_start = (((seg_ymin - self.ymin) / self.cell_h).floor() as isize).max(0) as usize;
        let row_end = (((seg_ymax - self.ymin) / self.cell_h).floor() as isize).max(0) as usize;
        let row_end = row_end.min(self.rows - 1);
        
        for r in row_start..=row_end {
            for c in col_start..=col_end {
                let idx = r * self.cols + c;
                let cell = &self.cells[idx];
                for &edge_idx in &cell.edges {
                    let sa = poly[edge_idx];
                    let sb = poly[(edge_idx + 1) % poly.len()];
                    if segments_cross(a, b, sa, sb) {
                        return true;
                    }
                }
            }
        }
        false
    }
}

fn piece_inside_stock(grid: &StockGrid, stock_poly: &[Point], piece_vertices: &[Point]) -> bool {
    for &p in piece_vertices {
        if !grid.point_in_stock(p, stock_poly) {
            return false;
        }
    }
    
    let n = piece_vertices.len();
    for i in 0..n {
        let a = piece_vertices[i];
        let b = piece_vertices[(i + 1) % n];
        let mid = Point { x: (a.x + b.x) * 0.5, y: (a.y + b.y) * 0.5 };
        if !grid.point_in_stock(mid, stock_poly) {
            return false;
        }
    }
    
    for i in 0..n {
        let a = piece_vertices[i];
        let b = piece_vertices[(i + 1) % n];
        if grid.segment_intersects_stock(a, b, stock_poly) {
            return false;
        }
    }
    
    true
}

impl PlacementGrid {
    fn new(xmin: f64, ymin: f64, xmax: f64, ymax: f64, cols: usize, rows: usize) -> Self {
        let cell_w = (xmax - xmin) / cols as f64;
        let cell_h = (ymax - ymin) / rows as f64;
        let mut cells = Vec::with_capacity(cols * rows);
        for _ in 0..(cols * rows) {
            cells.push(Vec::new());
        }
        PlacementGrid {
            xmin, ymin,
            cell_w, cell_h,
            cols, rows,
            cells,
        }
    }
    
    fn add_piece(&mut self, piece_idx: usize, aabb: (Point, Point)) {
        let (pmin, pmax) = aabb;
        let col_start = (((pmin.x - self.xmin) / self.cell_w).floor() as isize).max(0) as usize;
        let col_end = (((pmax.x - self.xmin) / self.cell_w).floor() as isize).max(0) as usize;
        let col_end = col_end.min(self.cols - 1);
        
        let row_start = (((pmin.y - self.ymin) / self.cell_h).floor() as isize).max(0) as usize;
        let row_end = (((pmax.y - self.ymin) / self.cell_h).floor() as isize).max(0) as usize;
        let row_end = row_end.min(self.rows - 1);
        
        for r in row_start..=row_end {
            for c in col_start..=col_end {
                self.cells[r * self.cols + c].push(piece_idx);
            }
        }
    }
    
    fn get_candidates(&self, aabb: (Point, Point), candidates: &mut Vec<usize>, visited: &mut [u32], epoch: u32) {
        let (pmin, pmax) = aabb;
        if pmin.x > self.xmin + self.cols as f64 * self.cell_w || pmax.x < self.xmin ||
           pmin.y > self.ymin + self.rows as f64 * self.cell_h || pmax.y < self.ymin {
            return;
        }
        
        let col_start = (((pmin.x - self.xmin) / self.cell_w).floor() as isize).max(0) as usize;
        let col_end = (((pmax.x - self.xmin) / self.cell_w).floor() as isize).max(0) as usize;
        let col_end = col_end.min(self.cols - 1);
        
        let row_start = (((pmin.y - self.ymin) / self.cell_h).floor() as isize).max(0) as usize;
        let row_end = (((pmax.y - self.ymin) / self.cell_h).floor() as isize).max(0) as usize;
        let row_end = row_end.min(self.rows - 1);
        
        for r in row_start..=row_end {
            for c in col_start..=col_end {
                let cell_pieces = &self.cells[r * self.cols + c];
                for &pid in cell_pieces {
                    if visited[pid] != epoch {
                        visited[pid] = epoch;
                        candidates.push(pid);
                    }
                }
            }
        }
    }
}

fn aabb_overlap(aabb1: (Point, Point), aabb2: (Point, Point)) -> bool {
    let (min1, max1) = aabb1;
    let (min2, max2) = aabb2;
    let eps = 1e-7;
    if max1.x < min2.x + eps || min1.x > max2.x - eps ||
       max1.y < min2.y + eps || min1.y > max2.y - eps {
        return false;
    }
    true
}

fn pieces_overlap(
    cand_vertices: &[Point],
    cand_aabb: (Point, Point),
    placed: &PlacedPiece
) -> bool {
    if !aabb_overlap(cand_aabb, placed.aabb) {
        return false;
    }
    
    let cand_centroid = get_centroid(cand_vertices);
    let placed_centroid = get_centroid(&placed.vertices);
    if (cand_centroid.x - placed_centroid.x).abs() < 1e-5 && (cand_centroid.y - placed_centroid.y).abs() < 1e-5 {
        return true;
    }
    
    let n1 = cand_vertices.len();
    let n2 = placed.vertices.len();
    
    // 1. Check edge intersections
    for i in 0..n1 {
        let a = cand_vertices[i];
        let b = cand_vertices[(i + 1) % n1];
        for j in 0..n2 {
            let c = placed.vertices[j];
            let d = placed.vertices[(j + 1) % n2];
            if segments_cross(a, b, c, d) {
                return true;
            }
        }
    }
    
    // 2. Check if cand is inside placed
    for &p in cand_vertices {
        if point_strictly_in_polygon(p, &placed.vertices) {
            return true;
        }
    }
    for i in 0..n1 {
        let a = cand_vertices[i];
        let b = cand_vertices[(i + 1) % n1];
        let mid = Point { x: (a.x + b.x) * 0.5, y: (a.y + b.y) * 0.5 };
        if point_strictly_in_polygon(mid, &placed.vertices) {
            return true;
        }
    }
    
    // 3. Check if placed is inside cand
    for &p in &placed.vertices {
        if point_strictly_in_polygon(p, cand_vertices) {
            return true;
        }
    }
    for j in 0..n2 {
        let c = placed.vertices[j];
        let d = placed.vertices[(j + 1) % n2];
        let mid = Point { x: (c.x + d.x) * 0.5, y: (c.y + d.y) * 0.5 };
        if point_strictly_in_polygon(mid, cand_vertices) {
            return true;
        }
    }
    
    false
}

struct StockState {
    placed_pieces: Vec<PlacedPiece>,
    placement_grid: PlacementGrid,
}

struct Solution {
    placements: Vec<PlacementResult>,
    num_stocks: usize,
    score: f64,
}

struct PlacementResult {
    piece_id: usize,
    stock_index: usize,
    x: f64,
    y: f64,
    rotation: f64,
}

fn find_placement_for_piece(
    stock_grid: &StockGrid,
    stock_poly: &[Point],
    placed_pieces: &[PlacedPiece],
    placement_grid: &PlacementGrid,
    versions: &[RotatedPieceVersion],
    visited: &mut [u32],
    epoch: &mut u32,
    candidates_buf: &mut Vec<usize>,
    step_divisor: f64,
) -> Option<(usize, Point, Point)> {
    for (v_idx, v) in versions.iter().enumerate() {
        let w = v.width;
        let h = v.height;
        
        let x_start = stock_grid.stock_xmin;
        let x_end = stock_grid.stock_xmax - w;
        let y_start = stock_grid.stock_ymin;
        let y_end = stock_grid.stock_ymax - h;
        
        if x_end < x_start || y_end < y_start {
            continue;
        }
        
        let min_step = (stock_grid.stock_xmax - stock_grid.stock_xmin).max(stock_grid.stock_ymax - stock_grid.stock_ymin) / 200.0;
        let base_step = (w.min(h) / 6.0).max(min_step);
        let step = base_step / step_divisor;
        
        let mut y = y_start;
        while y <= y_end + 1e-9 {
            let mut x = x_start;
            while x <= x_end + 1e-9 {
                let t_bl = Point { x, y };
                let cand_aabb = (t_bl, Point { x: x + w, y: y + h });
                
                *epoch += 1;
                candidates_buf.clear();
                placement_grid.get_candidates(cand_aabb, candidates_buf, visited, *epoch);
                
                let mut overlaps = false;
                let cand_vertices: Vec<Point> = v.vertices.iter().map(|p| {
                    Point { x: p.x + x, y: p.y + y }
                }).collect();
                
                let mut jump_x = None;
                for &pid in candidates_buf.iter() {
                    let placed = &placed_pieces[pid];
                    if pieces_overlap(&cand_vertices, cand_aabb, placed) {
                        overlaps = true;
                        jump_x = Some(placed.aabb.1.x);
                        break;
                    }
                }
                
                if !overlaps {
                    if piece_inside_stock(stock_grid, stock_poly, &cand_vertices) {
                        let t_output = Point {
                            x: x - v.p_min.x,
                            y: y - v.p_min.y,
                        };
                        return Some((v_idx, t_bl, t_output));
                    }
                }
                
                if overlaps {
                    if let Some(jx) = jump_x {
                        if jx > x {
                            x = jx;
                        } else {
                            x += step;
                        }
                    } else {
                        x += step;
                    }
                } else {
                    x += step;
                }
            }
            y += step;
        }
    }
    None
}

fn solve_single_order(
    stock_grid: &StockGrid,
    stock_poly: &[Point],
    pieces: &[OriginalPiece],
    all_versions: &[Vec<RotatedPieceVersion>],
    order: &[usize],
    max_stocks_to_check: usize,
) -> Solution {
    let np = pieces.len();
    let mut stocks: Vec<StockState> = Vec::new();
    let mut placements = Vec::with_capacity(np);
    
    // Buffers for grid queries to avoid re-allocating
    let mut visited = vec![0u32; np];
    let mut epoch = 0u32;
    let mut candidates_buf = Vec::with_capacity(64);
    
    for &p_idx in order {
        let versions = &all_versions[p_idx];
        let mut placed = false;
        
        // Try placing in existing stocks (from the most recent ones up to max_stocks_to_check)
        let num_existing = stocks.len();
        let start_idx = if num_existing > max_stocks_to_check {
            num_existing - max_stocks_to_check
        } else {
            0
        };
        
        for s_idx in start_idx..num_existing {
            let stock = &mut stocks[s_idx];
            if let Some((v_idx, t_bl, t_output)) = find_placement_for_piece(
                stock_grid,
                stock_poly,
                &stock.placed_pieces,
                &stock.placement_grid,
                versions,
                &mut visited,
                &mut epoch,
                &mut candidates_buf,
                1.0,
            ) {
                let v = &versions[v_idx];
                let placed_vertices: Vec<Point> = v.vertices.iter().map(|p| {
                    Point { x: p.x + t_bl.x, y: p.y + t_bl.y }
                }).collect();
                let aabb = (t_bl, Point { x: t_bl.x + v.width, y: t_bl.y + v.height });
                
                stock.placed_pieces.push(PlacedPiece {
                    vertices: placed_vertices,
                    aabb,
                });
                stock.placement_grid.add_piece(stock.placed_pieces.len() - 1, aabb);
                
                placements.push(PlacementResult {
                    piece_id: p_idx,
                    stock_index: s_idx,
                    x: t_output.x,
                    y: t_output.y,
                    rotation: v.rotation_angle,
                });
                placed = true;
                break;
            }
        }
        
        if !placed {
            // Must open a new stock
            let mut new_stock = StockState {
                placed_pieces: Vec::new(),
                placement_grid: PlacementGrid::new(
                    stock_grid.xmin,
                    stock_grid.ymin,
                    stock_grid.xmax,
                    stock_grid.ymax,
                    64,
                    64,
                ),
            };
            
            // Try to find placement with fallback refinements for empty stock
            let mut found = None;
            for divisor in &[1.0, 2.0, 4.0] {
                if let Some(res) = find_placement_for_piece(
                    stock_grid,
                    stock_poly,
                    &new_stock.placed_pieces,
                    &new_stock.placement_grid,
                    versions,
                    &mut visited,
                    &mut epoch,
                    &mut candidates_buf,
                    *divisor,
                ) {
                    found = Some(res);
                    break;
                }
            }
            
            let (v_idx, t_bl, t_output) = match found {
                Some(res) => res,
                None => {
                    // Fallback to origin if it absolutely doesn't fit (shouldn't happen on valid inputs)
                    (0, Point { x: stock_grid.xmin, y: stock_grid.ymin }, Point { x: 0.0, y: 0.0 })
                }
            };
            
            let v = &versions[v_idx];
            let placed_vertices: Vec<Point> = v.vertices.iter().map(|p| {
                Point { x: p.x + t_bl.x, y: p.y + t_bl.y }
            }).collect();
            let aabb = (t_bl, Point { x: t_bl.x + v.width, y: t_bl.y + v.height });
            
            new_stock.placed_pieces.push(PlacedPiece {
                vertices: placed_vertices,
                aabb,
            });
            new_stock.placement_grid.add_piece(new_stock.placed_pieces.len() - 1, aabb);
            
            let s_idx = stocks.len();
            stocks.push(new_stock);
            
            placements.push(PlacementResult {
                piece_id: p_idx,
                stock_index: s_idx,
                x: t_output.x,
                y: t_output.y,
                rotation: v.rotation_angle,
            });
        }
    }
    
    // Calculate score
    let num_stocks = stocks.len();
    let last_stock_area: f64 = if num_stocks > 0 {
        stocks[num_stocks - 1].placed_pieces.iter().map(|p| {
            polygon_area(&p.vertices)
        }).sum()
    } else {
        0.0
    };
    
    let total_piece_area: f64 = pieces.iter().map(|p| p.area).sum();
    let score = num_stocks as f64 + (last_stock_area / (total_piece_area + 1e-9));
    
    Solution {
        placements,
        num_stocks,
        score,
    }
}

fn read_input() -> (Vec<Point>, Vec<OriginalPiece>) {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines().map(|l| l.unwrap());
    
    let mut next_line = || {
        while let Some(line) = lines.next() {
            let trimmed = line.trim();
            if !trimmed.is_empty() {
                return Some(trimmed.to_string());
            }
        }
        None
    };
    
    let first_line = next_line().expect("Failed to read first line");
    let mut parts = first_line.split_whitespace();
    let sv: usize = parts.next().expect("Missing SV").parse().expect("Invalid SV");
    let np: usize = parts.next().expect("Missing NP").parse().expect("Invalid NP");
    
    let mut stock_poly = Vec::with_capacity(sv);
    for _ in 0..sv {
        let line = next_line().expect("Missing stock vertex line");
        let mut pts = line.split_whitespace();
        let x: f64 = pts.next().expect("Missing x").parse().expect("Invalid x");
        let y: f64 = pts.next().expect("Missing y").parse().expect("Invalid y");
        stock_poly.push(Point { x, y });
    }
    
    let mut pieces = Vec::with_capacity(np);
    for id in 0..np {
        let line = next_line().expect("Missing piece PV line");
        let pv: usize = line.parse().expect("Invalid PV");
        let mut vertices = Vec::with_capacity(pv);
        for _ in 0..pv {
            let line = next_line().expect("Missing piece vertex line");
            let mut pts = line.split_whitespace();
            let x: f64 = pts.next().expect("Missing x").parse().expect("Invalid x");
            let y: f64 = pts.next().expect("Missing y").parse().expect("Invalid y");
            vertices.push(Point { x, y });
        }
        let area = polygon_area(&vertices);
        pieces.push(OriginalPiece { _id: id, vertices, area });
    }
    
    (stock_poly, pieces)
}

fn main() {
    let start_time = Instant::now();
    let (stock_poly, pieces) = read_input();
    
    let np = pieces.len();
    if np == 0 {
        println!("0");
        return;
    }
    
    // Build stock grid (precomputing stock boundary cells for 100x speedup)
    let stock_grid = Arc::new(StockGrid::new(&stock_poly, 128, 128));
    
    // Choose number of rotations adaptively
    let k = if np < 50 {
        36 // every 10 degrees
    } else if np < 500 {
        12 // every 30 degrees
    } else if np < 2000 {
        8 // every 45 degrees
    } else {
        4 // every 90 degrees
    };
    
    // Pre-rotate all pieces
    let mut all_versions = Vec::with_capacity(np);
    for piece in &pieces {
        let mut versions = Vec::with_capacity(k);
        for r in 0..k {
            let angle = (2.0 * std::f64::consts::PI * r as f64) / k as f64;
            let cos_t = angle.cos();
            let sin_t = angle.sin();
            
            let rotated: Vec<Point> = piece.vertices.iter().map(|p| {
                Point {
                    x: p.x * cos_t - p.y * sin_t,
                    y: p.x * sin_t + p.y * cos_t,
                }
            }).collect();
            
            let mut xmin = f64::MAX;
            let mut ymin = f64::MAX;
            let mut xmax = f64::MIN;
            let mut ymax = f64::MIN;
            for p in &rotated {
                if p.x < xmin { xmin = p.x; }
                if p.y < ymin { ymin = p.y; }
                if p.x > xmax { xmax = p.x; }
                if p.y > ymax { ymax = p.y; }
            }
            
            let normalized: Vec<Point> = rotated.iter().map(|p| {
                Point { x: p.x - xmin, y: p.y - ymin }
            }).collect();
            
            versions.push(RotatedPieceVersion {
                _rotation_idx: r,
                rotation_angle: angle,
                vertices: normalized,
                width: xmax - xmin,
                height: ymax - ymin,
                p_min: Point { x: xmin, y: ymin },
            });
        }
        all_versions.push(versions);
    }
    
    let all_versions = Arc::new(all_versions);
    let stock_poly = Arc::new(stock_poly);
    let pieces = Arc::new(pieces);
    
    // Set up parallel search
    let num_threads = thread::available_parallelism().map(|n| n.get()).unwrap_or(64);
    
    // Choose how many stocks to check dynamically based on problem scale
    let max_stocks_to_check = if np < 200 {
        usize::MAX
    } else if np < 1000 {
        10
    } else {
        3
    };
    
    let best_solution = Arc::new(Mutex::new(None::<Solution>));
    let mut thread_handles = Vec::new();
    let time_limit = Duration::from_millis(4000);
    
    for t in 0..num_threads {
        let stock_grid = Arc::clone(&stock_grid);
        let stock_poly = Arc::clone(&stock_poly);
        let pieces = Arc::clone(&pieces);
        let all_versions = Arc::clone(&all_versions);
        let best_solution = Arc::clone(&best_solution);
        
        let handle = thread::spawn(move || {
            let seed = t as u64 * 1337 + 42;
            let mut lcg = Lcg::new(seed);
            let mut local_best: Option<Solution> = None;
            
            loop {
                let mut order: Vec<usize> = (0..np).collect();
                
                if t == 0 {
                    order.sort_by(|&a, &b| pieces[b].area.partial_cmp(&pieces[a].area).unwrap_or(std::cmp::Ordering::Equal));
                } else if t == 1 {
                    order.sort_by(|&a, &b| {
                        let score_a = pieces[a].area * pieces[a].vertices.len() as f64;
                        let score_b = pieces[b].area * pieces[b].vertices.len() as f64;
                        score_b.partial_cmp(&score_a).unwrap_or(std::cmp::Ordering::Equal)
                    });
                } else if t == 2 {
                    order.sort_by(|&a, &b| {
                        let v_a = &all_versions[a][0];
                        let v_b = &all_versions[b][0];
                        let max_dim_a = v_a.width.max(v_a.height);
                        let max_dim_b = v_b.width.max(v_b.height);
                        max_dim_b.partial_cmp(&max_dim_a).unwrap_or(std::cmp::Ordering::Equal)
                    });
                } else {
                    let noise_level = 0.01 + (lcg.next_f64() * 0.1);
                    order.sort_by(|&a, &b| {
                        let area_a = pieces[a].area;
                        let area_b = pieces[b].area;
                        let noise_a = area_a * (lcg.next_f64() - 0.5) * noise_level;
                        let noise_b = area_b * (lcg.next_f64() - 0.5) * noise_level;
                        (area_b + noise_b).partial_cmp(&(area_a + noise_a)).unwrap_or(std::cmp::Ordering::Equal)
                    });
                }
                
                let sol = solve_single_order(
                    &stock_grid,
                    &stock_poly,
                    &pieces,
                    &all_versions,
                    &order,
                    max_stocks_to_check,
                );
                
                match &local_best {
                    Some(best) => {
                        if sol.score < best.score {
                            local_best = Some(sol);
                        }
                    }
                    None => {
                        local_best = Some(sol);
                    }
                }
                
                if t <= 2 || start_time.elapsed() >= time_limit {
                    break;
                }
            }
            
            if let Some(sol) = local_best {
                let mut best = best_solution.lock().unwrap();
                match &*best {
                    Some(current_best) => {
                        if sol.score < current_best.score {
                            *best = Some(sol);
                        }
                    }
                    None => {
                        *best = Some(sol);
                    }
                }
            }
        });
        
        thread_handles.push(handle);
    }
    
    // Wait for all threads to finish
    for handle in thread_handles {
        let _ = handle.join();
    }
    
    // Output the best solution
    let final_best = best_solution.lock().unwrap();
    let sol = final_best.as_ref().expect("No solution found");
    
    println!("{}", sol.num_stocks);
    
    // We need to output the placements in the order of the pieces in the input
    let mut ordered_placements = vec![None; np];
    for p in &sol.placements {
        ordered_placements[p.piece_id] = Some(p);
    }
    
    for opt_p in ordered_placements {
        let p = opt_p.expect("Missing placement for piece");
        println!("{} {} {} {}", p.stock_index, p.x, p.y, p.rotation);
    }
}
