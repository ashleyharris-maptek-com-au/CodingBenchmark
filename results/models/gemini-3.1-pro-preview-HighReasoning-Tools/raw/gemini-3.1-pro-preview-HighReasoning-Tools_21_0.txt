struct Point {
    int D = 0;
    std::array<int16_t, 32> coords{};

    bool operator==(const Point& o) const {
        if (D != o.D) return false;
        for (int i = 0; i < D; ++i) {
            if (coords[i] != o.coords[i]) return false;
        }
        return true;
    }

    bool operator<(const Point& o) const {
        if (D != o.D) return D < o.D;
        for (int i = 0; i < D; ++i) {
            if (coords[i] != o.coords[i]) {
                return coords[i] < o.coords[i];
            }
        }
        return false;
    }
};

struct PointHash {
    std::size_t operator()(const Point& p) const {
        std::size_t h = 17;
        for (int i = 0; i < p.D; ++i) {
            h = h * 31 + p.coords[i];
        }
        return h;
    }
};

// Open-addressing hash set for points
struct PointSet {
    struct Slot {
        Point p;
        int visit_id = 0;
    };
    std::vector<Slot> table;
    int current_visit_id = 0;
    int mask = 0;

    PointSet(int capacity = 262144) {
        table.resize(capacity);
        mask = capacity - 1;
    }

    void clear() {
        current_visit_id++;
    }

    bool count(const Point& p) const {
        std::size_t h = PointHash{}(p);
        int idx = h & mask;
        while (table[idx].visit_id == current_visit_id) {
            if (table[idx].p == p) return true;
            idx = (idx + 1) & mask;
        }
        return false;
    }

    void insert(const Point& p) {
        std::size_t h = PointHash{}(p);
        int idx = h & mask;
        while (table[idx].visit_id == current_visit_id) {
            if (table[idx].p == p) return;
            idx = (idx + 1) & mask;
        }
        table[idx].p = p;
        table[idx].visit_id = current_visit_id;
    }
};

// Open-addressing hash map for snake body indices
struct PointMap {
    struct Slot {
        Point p;
        int val = -1;
        int visit_id = 0;
    };
    std::vector<Slot> table;
    int current_visit_id = 0;
    int mask = 0;

    PointMap(int capacity = 1024) {
        table.resize(capacity);
        mask = capacity - 1;
    }

    void clear() {
        current_visit_id++;
    }

    int get(const Point& p) const {
        std::size_t h = PointHash{}(p);
        int idx = h & mask;
        while (table[idx].visit_id == current_visit_id) {
            if (table[idx].p == p) return table[idx].val;
            idx = (idx + 1) & mask;
        }
        return -1;
    }

    void put(const Point& p, int val) {
        std::size_t h = PointHash{}(p);
        int idx = h & mask;
        while (table[idx].visit_id == current_visit_id) {
            if (table[idx].p == p) {
                // Keep the minimum value (first occurrence from head)
                return;
            }
            idx = (idx + 1) & mask;
        }
        table[idx].p = p;
        table[idx].val = val;
        table[idx].visit_id = current_visit_id;
    }
};

// Global environment and configuration
int D;
int max_turns;
std::vector<int> bounds;
std::unordered_set<Point, PointHash> food; // for tracking source of truth

PointSet obstacles_set(1024);
PointSet food_set(1024);

bool use_flat_grid = false;
long long total_cells = 0;
std::vector<int> strides;
std::vector<int> visited_flat;
std::vector<int> body_flat;

PointSet visited_set(262144);
PointMap body_map(1024);

inline int get_flat_index(const Point& p) {
    int idx = 0;
    for (int i = 0; i < D; ++i) {
        idx += p.coords[i] * strides[i];
    }
    return idx;
}

int flat_visit_id = 0;

struct Node {
    Point pos;
    Node* parent = nullptr;
    int t = 0;
};

std::vector<Node> node_pool(200000);
int node_pool_index = 0;

Node* create_node(const Point& pos, Node* parent, int t) {
    if (node_pool_index >= (int)node_pool.size()) {
        node_pool.resize(node_pool.size() * 2);
    }
    Node* n = &node_pool[node_pool_index++];
    n->pos = pos;
    n->parent = parent;
    n->t = t;
    return n;
}

inline bool is_body_blocked(const Point& p, int next_t, int L) {
    if (use_flat_grid) {
        int idx = get_flat_index(p);
        int val = body_flat[idx];
        return (val != -1 && val < L - next_t);
    } else {
        int val = body_map.get(p);
        return (val != -1 && val < L - next_t);
    }
}

void rebuild_food_set() {
    food_set.clear();
    for (const auto& p : food) {
        food_set.insert(p);
    }
}

int count_reachable_cells(const Point& start_pos, int start_t, int L, int max_cells = 1000) {
    struct SpaceNode {
        Point pos;
        int t;
    };
    std::queue<SpaceNode> q;
    q.push({start_pos, start_t});

    int count = 0;

    if (use_flat_grid) {
        flat_visit_id++;
        int start_idx = get_flat_index(start_pos);
        visited_flat[start_idx] = flat_visit_id;

        while (!q.empty() && count < max_cells) {
            SpaceNode curr = q.front();
            q.pop();
            count++;

            for (int axis = 0; axis < D; ++axis) {
                for (int dir : {-1, 1}) {
                    Point neighbor = curr.pos;
                    neighbor.coords[axis] += dir;

                    if (neighbor.coords[axis] < 0 || neighbor.coords[axis] >= bounds[axis]) continue;
                    if (obstacles_set.count(neighbor)) continue;

                    int idx = get_flat_index(neighbor);
                    if (visited_flat[idx] == flat_visit_id) continue;

                    int next_t = curr.t + 1;
                    int val = body_flat[idx];
                    if (val != -1 && val < L - next_t) continue;

                    visited_flat[idx] = flat_visit_id;
                    q.push({neighbor, next_t});
                }
            }
        }
    } else {
        visited_set.clear();
        visited_set.insert(start_pos);

        while (!q.empty() && count < max_cells) {
            SpaceNode curr = q.front();
            q.pop();
            count++;

            for (int axis = 0; axis < D; ++axis) {
                for (int dir : {-1, 1}) {
                    Point neighbor = curr.pos;
                    neighbor.coords[axis] += dir;

                    if (neighbor.coords[axis] < 0 || neighbor.coords[axis] >= bounds[axis]) continue;
                    if (obstacles_set.count(neighbor)) continue;
                    if (visited_set.count(neighbor)) continue;

                    int next_t = curr.t + 1;
                    int val = body_map.get(neighbor);
                    if (val != -1 && val < L - next_t) continue;

                    visited_set.insert(neighbor);
                    q.push({neighbor, next_t});
                }
            }
        }
    }
    return count;
}

Node* find_path(bool target_food, const Point& target_pos, const Point& head, int L) {
    node_pool_index = 0;
    std::queue<Node*> q;

    Node* start_node = create_node(head, nullptr, 0);
    q.push(start_node);

    Node* target_node = nullptr;

    if (use_flat_grid) {
        flat_visit_id++;
        int start_idx = get_flat_index(head);
        visited_flat[start_idx] = flat_visit_id;

        while (!q.empty()) {
            Node* curr = q.front();
            q.pop();

            if (target_food) {
                if (food_set.count(curr->pos)) {
                    target_node = curr;
                    break;
                }
            } else {
                if (curr->pos == target_pos) {
                    target_node = curr;
                    break;
                }
            }

            if (node_pool_index >= 30000) {
                break;
            }

            for (int axis = 0; axis < D; ++axis) {
                for (int dir : {-1, 1}) {
                    Point neighbor = curr->pos;
                    neighbor.coords[axis] += dir;

                    if (neighbor.coords[axis] < 0 || neighbor.coords[axis] >= bounds[axis]) continue;
                    if (obstacles_set.count(neighbor)) continue;

                    int idx = get_flat_index(neighbor);
                    if (visited_flat[idx] == flat_visit_id) continue;

                    int next_t = curr->t + 1;
                    int val = body_flat[idx];
                    if (val != -1 && val < L - next_t) continue;

                    visited_flat[idx] = flat_visit_id;
                    q.push(create_node(neighbor, curr, next_t));
                }
            }
        }
    } else {
        visited_set.clear();
        visited_set.insert(head);

        while (!q.empty()) {
            Node* curr = q.front();
            q.pop();

            if (target_food) {
                if (food_set.count(curr->pos)) {
                    target_node = curr;
                    break;
                }
            } else {
                if (curr->pos == target_pos) {
                    target_node = curr;
                    break;
                }
            }

            if (node_pool_index >= 30000) {
                break;
            }

            for (int axis = 0; axis < D; ++axis) {
                for (int dir : {-1, 1}) {
                    Point neighbor = curr->pos;
                    neighbor.coords[axis] += dir;

                    if (neighbor.coords[axis] < 0 || neighbor.coords[axis] >= bounds[axis]) continue;
                    if (obstacles_set.count(neighbor)) continue;
                    if (visited_set.count(neighbor)) continue;

                    int next_t = curr->t + 1;
                    int val = body_map.get(neighbor);
                    if (val != -1 && val < L - next_t) continue;

                    visited_set.insert(neighbor);
                    q.push(create_node(neighbor, curr, next_t));
                }
            }
        }
    }
    return target_node;
}

void simulate_body_after_path(const Node* target_node, const std::deque<Point>& current_body, std::deque<Point>& new_body, bool ate_food) {
    std::vector<Point> path;
    const Node* curr = target_node;
    while (curr != nullptr) {
        path.push_back(curr->pos);
        curr = curr->parent;
    }
    std::reverse(path.begin(), path.end());

    int k = path.size() - 1;
    int L = current_body.size();

    new_body.clear();
    for (int i = k; i >= 1; --i) {
        new_body.push_back(path[i]);
    }
    int remaining_limit = L - k + (ate_food ? 1 : 0);
    for (int i = 0; i < remaining_limit && i < L; ++i) {
        new_body.push_back(current_body[i]);
    }
}

void rebuild_body_flat(const std::deque<Point>& old_body, const std::deque<Point>& new_body) {
    for (const auto& p : old_body) {
        body_flat[get_flat_index(p)] = -1;
    }
    for (int i = 0; i < (int)new_body.size(); ++i) {
        const Point& p = new_body[i];
        int idx = get_flat_index(p);
        if (body_flat[idx] == -1) {
            body_flat[idx] = i;
        }
    }
}

void rebuild_body_map(const std::deque<Point>& old_body, const std::deque<Point>& new_body) {
    body_map.clear();
    for (int i = 0; i < (int)new_body.size(); ++i) {
        body_map.put(new_body[i], i);
    }
}

struct Move {
    int axis = 0;
    int direction = 1;
    bool is_deadly = false;
};

Move select_move(const std::deque<Point>& body) {
    Point head = body.front();

    int current_reach = count_reachable_cells(head, 0, body.size(), 1000);

    Node* food_node = find_path(true, Point(), head, body.size());

    if (food_node != nullptr) {
        std::deque<Point> sim_body;
        simulate_body_after_path(food_node, body, sim_body, true);

        if (use_flat_grid) {
            rebuild_body_flat(body, sim_body);
        } else {
            rebuild_body_map(body, sim_body);
        }

        int food_reach = count_reachable_cells(sim_body.front(), food_node->t, sim_body.size(), 1000);

        if (use_flat_grid) {
            rebuild_body_flat(sim_body, body);
        } else {
            rebuild_body_map(sim_body, body);
        }

        int safety_threshold = std::min(current_reach, (int)body.size() + 5);

        if (food_reach >= safety_threshold) {
            Node* curr = food_node;
            while (curr->parent != nullptr && curr->parent->parent != nullptr) {
                curr = curr->parent;
            }
            for (int i = 0; i < D; ++i) {
                if (curr->pos.coords[i] != head.coords[i]) {
                    return {i, curr->pos.coords[i] - head.coords[i], false};
                }
            }
        }
    }

    if (body.size() > 1) {
        Node* tail_node = find_path(false, body.back(), head, body.size());
        if (tail_node != nullptr) {
            Node* curr = tail_node;
            while (curr->parent != nullptr && curr->parent->parent != nullptr) {
                curr = curr->parent;
            }
            for (int i = 0; i < D; ++i) {
                if (curr->pos.coords[i] != head.coords[i]) {
                    return {i, curr->pos.coords[i] - head.coords[i], false};
                }
            }
        }
    }

    int best_axis = -1;
    int best_dir = 0;
    int max_space = -1;

    for (int axis = 0; axis < D; ++axis) {
        for (int dir : {-1, 1}) {
            Point neighbor = head;
            neighbor.coords[axis] += dir;

            if (neighbor.coords[axis] < 0 || neighbor.coords[axis] >= bounds[axis]) continue;
            if (obstacles_set.count(neighbor)) continue;

            if (is_body_blocked(neighbor, 1, body.size())) continue;

            std::deque<Point> sim_body;
            Node dummy_start{head, nullptr, 0};
            Node dummy_neighbor{neighbor, &dummy_start, 1};
            simulate_body_after_path(&dummy_neighbor, body, sim_body, false);

            if (use_flat_grid) {
                rebuild_body_flat(body, sim_body);
            } else {
                rebuild_body_map(body, sim_body);
            }

            int space = count_reachable_cells(neighbor, 1, sim_body.size(), 1000);

            if (use_flat_grid) {
                rebuild_body_flat(sim_body, body);
            } else {
                rebuild_body_map(sim_body, body);
            }

            if (space > max_space) {
                max_space = space;
                best_axis = axis;
                best_dir = dir;
            }
        }
    }

    if (best_axis != -1) {
        return {best_axis, best_dir, false};
    }

    for (int axis = 0; axis < D; ++axis) {
        for (int dir : {-1, 1}) {
            Point neighbor = head;
            neighbor.coords[axis] += dir;
            if (neighbor.coords[axis] >= 0 && neighbor.coords[axis] < bounds[axis] && !obstacles_set.count(neighbor)) {
                return {axis, dir, true};
            }
        }
    }

    return {0, 1, true};
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    if (!(std::cin >> D >> max_turns)) return 0;

    bounds.resize(D);
    for (int i = 0; i < D; ++i) {
        std::cin >> bounds[i];
    }

    int snake_length;
    std::cin >> snake_length;

    Point start_pos;
    start_pos.D = D;
    for (int i = 0; i < D; ++i) {
        std::cin >> start_pos.coords[i];
    }

    int F;
    if (!(std::cin >> F)) return 0;
    for (int i = 0; i < F; ++i) {
        Point p;
        p.D = D;
        for (int j = 0; j < D; ++j) {
            std::cin >> p.coords[j];
        }
        food.insert(p);
    }

    int O;
    if (!(std::cin >> O)) return 0;
    
    int obs_capacity = 1024;
    while (obs_capacity < O * 3) {
        obs_capacity *= 2;
    }
    obstacles_set = PointSet(obs_capacity);
    obstacles_set.current_visit_id = 1;

    for (int i = 0; i < O; ++i) {
        Point p;
        p.D = D;
        for (int j = 0; j < D; ++j) {
            std::cin >> p.coords[j];
        }
        obstacles_set.insert(p);
    }

    total_cells = 1;
    use_flat_grid = true;
    for (int i = 0; i < D; ++i) {
        if (total_cells > 10000000 / bounds[i]) {
            use_flat_grid = false;
            break;
        }
        total_cells *= bounds[i];
    }

    if (use_flat_grid) {
        strides.resize(D);
        int cur = 1;
        for (int i = 0; i < D; ++i) {
            strides[i] = cur;
            cur *= bounds[i];
        }
        visited_flat.assign(total_cells, 0);
        body_flat.assign(total_cells, -1);
    } else {
        int max_snake_len = snake_length + F;
        int capacity = 1024;
        while (capacity < max_snake_len * 3) {
            capacity *= 2;
        }
        body_map = PointMap(capacity);
    }

    int food_capacity = 1024;
    while (food_capacity < F * 3) {
        food_capacity *= 2;
    }
    food_set = PointSet(food_capacity);
    rebuild_food_set();

    std::deque<Point> body;
    for (int i = 0; i < snake_length; ++i) {
        body.push_back(start_pos);
    }

    if (use_flat_grid) {
        rebuild_body_flat(std::deque<Point>(), body);
    } else {
        rebuild_body_map(std::deque<Point>(), body);
    }

    for (int turn = 0; turn < max_turns; ++turn) {
        if (food.empty()) {
            break;
        }

        Move m = select_move(body);
        std::cout << m.axis << " " << m.direction << "\n";
        std::cout.flush();

        if (m.is_deadly) {
            break;
        }

        Point new_head = body.front();
        new_head.coords[m.axis] += m.direction;

        std::deque<Point> old_body = body;

        if (food.count(new_head)) {
            food.erase(new_head);
            rebuild_food_set();
            body.push_front(new_head);
        } else {
            body.push_front(new_head);
            body.pop_back();
        }

        if (use_flat_grid) {
            rebuild_body_flat(old_body, body);
        } else {
            rebuild_body_map(old_body, body);
        }
    }

    return 0;
}
