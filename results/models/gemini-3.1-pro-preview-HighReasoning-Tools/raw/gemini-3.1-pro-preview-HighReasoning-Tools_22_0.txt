    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    public int GetMaxHeight() {
        for (int r = Height - 1; r >= 0; r--) {
            if (GetRow(r) != 0) return r + 1;
        }
        return 0;
    }

    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    public void GetHolesAndBumpiness(out int holes, out int bumpiness) {
        holes = 0;
        ulong filledAbove = 0;
        for (int r = Height - 1; r >= 0; r--) {
            filledAbove |= GetRow(r);
            holes += TestGame.PopCount((~GetRow(r)) & filledAbove & FullRowMask);
        }

        bumpiness = 0;
        int[] heights = new int[Width];
        for (int c = 0; c < Width; c++) {
            for (int r = Height - 1; r >= 0; r--) {
                if ((GetRow(r) & (1UL << c)) != 0) {
                    heights[c] = r + 1;
                    break;
                }
            }
        }
        for (int c = 0; c < Width - 1; c++) {
            bumpiness += Math.Abs(heights[c] - heights[c + 1]);
        }
    }
}

    public static int PopCount(ulong x) {
        x -= (x >> 1) & 0x5555555555555555UL;
        x = (x & 0x3333333333333333UL) + ((x >> 2) & 0x3333333333333333UL);
        x = (x + (x >> 4)) & 0x0f0f0f0f0f0f0f0fUL;
        return (int)((x * 0x0101010101010101UL) >> 56);
    }

    static int PlacePiece(ref BitBoard bb, int pieceIndex, int rotation, int col) {
        Block[] coords = PIECES[pieceIndex][rotation % 4];
        int pieceHeight = 0;
        foreach (var b in coords) {
            if (b.R > pieceHeight) pieceHeight = b.R;
        }
        pieceHeight += 1;

        uint[] pieceMasks = new uint[4];
        for (int i = 0; i < coords.Length; i++) {
            pieceMasks[coords[i].R] |= (1U << (col + coords[i].C));
        }

        int landingRow = bb.Height;
        for (int startRow = bb.Height - 1; startRow >= 0; startRow--) {
            bool overlap = false;
            for (int r = 0; r < pieceHeight; r++) {
                int boardRow = startRow + r;
                if (boardRow < bb.Height && (bb.GetRow(boardRow) & pieceMasks[r]) != 0) {
                    overlap = true;
                    break;
                }
            }
            if (overlap) {
                landingRow = startRow + 1;
                break;
            }
        }
        if (landingRow == bb.Height) landingRow = 0;

        // Place
        for (int r = 0; r < pieceHeight; r++) {
            int boardRow = landingRow + r;
            if (boardRow < bb.Height) {
                bb.SetRow(boardRow, bb.GetRow(boardRow) | pieceMasks[r]);
            }
        }

        // Clear lines
        int linesCleared = 0;
        for (int r = 0; r < bb.Height; r++) {
            if ((bb.GetRow(r) & bb.FullRowMask) == bb.FullRowMask) {
                for (int r2 = r; r2 < bb.Height - 1; r2++) {
                    bb.SetRow(r2, bb.GetRow(r2 + 1));
                }
                bb.SetRow(bb.Height - 1, 0);
                linesCleared++;
                r--;
            }
        }
        return linesCleared;
    }

    static double EvaluateState(ref BitBoard bb, double landingHeight, int erodedCells, bool survivalMode) {
        int max_height = 0;
        for (int r = bb.Height - 1; r >= 0; r--) {
            if (bb.GetRow(r) != 0) {
                max_height = r + 1;
                break;
            }
        }

        int aggregate_height = 0;
        int holes = 0;
        int bumpiness = 0;
        int rowTransitions = 0;
        int colTransitions = 0;
        int wellSum = 0;

        if (max_height == 0) {
            colTransitions = bb.Width;
            double scoreEmpty = -4.5 * landingHeight + 3.4 * erodedCells - 9.3 * colTransitions;
            return scoreEmpty;
        }

        int[] heights = new int[bb.Width];
        bool[] heightFound = new bool[bb.Width];
        int columnsFoundCount = 0;
        int[] columnWellDepths = new int[bb.Width];
        ulong filledAbove = 0;

        colTransitions = PopCount(bb.GetRow(0) ^ bb.FullRowMask);

        for (int r = 0; r < max_height; r++) {
            ulong val = bb.GetRow(r);
            ulong flanked = 1UL | (val << 1) | (1UL << (bb.Width + 1));
            ulong diff = flanked ^ (flanked >> 1);
            ulong mask = (1UL << (bb.Width + 1)) - 1;
            rowTransitions += PopCount(diff & mask);

            if (r < bb.Height - 1) {
                colTransitions += PopCount(val ^ bb.GetRow(r + 1));
            }
        }

        for (int r = max_height - 1; r >= 0; r--) {
            ulong val = bb.GetRow(r);

            if (columnsFoundCount < bb.Width) {
                for (int c = 0; c < bb.Width; c++) {
                    if (!heightFound[c] && (val & (1UL << c)) != 0) {
                        heights[c] = r + 1;
                        heightFound[c] = true;
                        columnsFoundCount++;
                        aggregate_height += heights[c];
                    }
                }
            }

            filledAbove |= val;
            holes += PopCount((~val) & filledAbove & bb.FullRowMask);

            ulong wellsInRow = (~val) & ((val << 1) | 1UL) & ((val >> 1) | (1UL << (bb.Width - 1))) & bb.FullRowMask;
            for (int c = 0; c < bb.Width; c++) {
                if ((wellsInRow & (1UL << c)) != 0) {
                    columnWellDepths[c]++;
                    wellSum += columnWellDepths[c];
                } else {
                    columnWellDepths[c] = 0;
                }
            }
        }

        for (int c = 0; c < bb.Width - 1; c++) {
            bumpiness += Math.Abs(heights[c] - heights[c + 1]);
        }

        double wLH = -4.500158825082766;
        double wEPC = 3.4181268101392694;
        double wRT = -3.2178882868487753;
        double wCT = -9.348695305445199;
        double wH = -7.899265427351652;
        double wWS = -3.3855972247263626;
        double wMH = 0.0;
        double wB = 0.0;

        double heightRatio = (double)max_height / bb.Height;
        if (survivalMode || heightRatio > 0.6) {
            wH = -20.0;
            wCT = -15.0;
            wRT = -6.0;
            wLH = -8.0;
            wMH = -10.0;
            wB = -4.0;
            wEPC = 6.0;
        } else if (heightRatio > 0.3) {
            wH = -12.0;
            wCT = -10.0;
            wMH = -3.0;
            wB = -1.0;
            wLH = -5.0;
        }

        double score = wLH * landingHeight
                     + wEPC * erodedCells
                     + wRT * rowTransitions
                     + wCT * colTransitions
                     + wH * holes
                     + wWS * wellSum
                     + wMH * max_height
                     + wB * bumpiness;

        return score;
    }

    static (int bestRotation, int bestCol) GetBestMove(int pieceIndex, int depth, ref BitBoard bb) {
        double bestScore = double.NegativeInfinity;
        int bestRotation = 0;
        int bestCol = 0;
        int bbMaxHeight = bb.GetMaxHeight();

        // Calculate current holes and bumpiness
        bb.GetHolesAndBumpiness(out int currentHoles, out int currentBumpiness);

        // First pass: depth 1
        double bestDepth1Score = double.NegativeInfinity;
        int bestDepth1Rot = 0;
        int bestDepth1Col = 0;
        bool foundSafeMove = false;

        double bestOverallScore = double.NegativeInfinity;
        int bestOverallRot = 0;
        int bestOverallCol = 0;

        int uniqueRots = UNIQUE_ROTATIONS[pieceIndex];
        for (int rot = 0; rot < uniqueRots; rot++) {
            Block[] coords = PIECES[pieceIndex][rot];
            int pieceWidth = 0;
            int pieceHeight = 0;
            foreach (var b in coords) {
                if (b.C > pieceWidth) pieceWidth = b.C;
                if (b.R > pieceHeight) pieceHeight = b.R;
            }
            pieceWidth += 1;
            pieceHeight += 1;

            for (int col = 0; col <= bb.Width - pieceWidth; col++) {
                uint[] pieceMasks = new uint[4];
                for (int i = 0; i < coords.Length; i++) {
                    pieceMasks[coords[i].R] |= (1U << (col + coords[i].C));
                }

                int landingRow = bb.Height;
                int startScan = Math.Min(bb.Height - 1, bbMaxHeight);
                for (int startRow = startScan; startRow >= 0; startRow--) {
                    bool overlap = false;
                    for (int r = 0; r < pieceHeight; r++) {
                        int boardRow = startRow + r;
                        if (boardRow < bb.Height && (bb.GetRow(boardRow) & pieceMasks[r]) != 0) {
                            overlap = true;
                            break;
                        }
                    }
                    if (overlap) {
                        landingRow = startRow + 1;
                        break;
                    }
                }
                if (landingRow == bb.Height) landingRow = 0;

                if (landingRow + pieceHeight > bb.Height) {
                    continue;
                }

                BitBoard simulated = bb;
                for (int r = 0; r < pieceHeight; r++) {
                    int boardRow = landingRow + r;
                    simulated.SetRow(boardRow, simulated.GetRow(boardRow) | pieceMasks[r]);
                }

                int clearedCount = 0;
                int pieceCellsCleared = 0;
                for (int r = 0; r < pieceHeight; r++) {
                    int boardRow = landingRow + r;
                    if ((simulated.GetRow(boardRow) & simulated.FullRowMask) == simulated.FullRowMask) {
                        clearedCount++;
                        pieceCellsCleared += PopCount(pieceMasks[r]);
                    }
                }
                int erodedCells = clearedCount * pieceCellsCleared;

                for (int r = 0; r < simulated.Height; r++) {
                    if ((simulated.GetRow(r) & simulated.FullRowMask) == simulated.FullRowMask) {
                        for (int r2 = r; r2 < simulated.Height - 1; r2++) {
                            simulated.SetRow(r2, simulated.GetRow(r2 + 1));
                        }
                        simulated.SetRow(simulated.Height - 1, 0);
                        r--;
                    }
                }

                int simHoles = 0;
                ulong filledAbove = 0;
                for (int r = simulated.Height - 1; r >= 0; r--) {
                    filledAbove |= simulated.GetRow(r);
                    simHoles += PopCount((~simulated.GetRow(r)) & filledAbove & simulated.FullRowMask);
                }

                double score = EvaluateState(ref simulated, landingRow + PIECE_AVG_R[pieceIndex][rot], erodedCells, false);

                if (score > bestOverallScore) {
                    bestOverallScore = score;
                    bestOverallRot = rot;
                    bestOverallCol = col;
                }

                if (simHoles <= currentHoles) {
                    foundSafeMove = true;
                    if (score > bestDepth1Score) {
                        bestDepth1Score = score;
                        bestDepth1Rot = rot;
                        bestDepth1Col = col;
                    }
                }
            }
        }

        if (depth == 1 || bb.Width >= 16) {
            if (foundSafeMove && bestDepth1Score != double.NegativeInfinity) {
                return (bestDepth1Rot, bestDepth1Col);
            } else if (bestOverallScore != double.NegativeInfinity) {
                return (bestOverallRot, bestOverallCol);
            }
        }

        // Full Depth 2 Search
        for (int rot = 0; rot < uniqueRots; rot++) {
            Block[] coords = PIECES[pieceIndex][rot];
            int pieceWidth = 0;
            int pieceHeight = 0;
            foreach (var b in coords) {
                if (b.C > pieceWidth) pieceWidth = b.C;
                if (b.R > pieceHeight) pieceHeight = b.R;
            }
            pieceWidth += 1;
            pieceHeight += 1;

            for (int col = 0; col <= bb.Width - pieceWidth; col++) {
                uint[] pieceMasks = new uint[4];
                for (int i = 0; i < coords.Length; i++) {
                    pieceMasks[coords[i].R] |= (1U << (col + coords[i].C));
                }

                int landingRow = bb.Height;
                int startScan = Math.Min(bb.Height - 1, bbMaxHeight);
                for (int startRow = startScan; startRow >= 0; startRow--) {
                    bool overlap = false;
                    for (int r = 0; r < pieceHeight; r++) {
                        int boardRow = startRow + r;
                        if (boardRow < bb.Height && (bb.GetRow(boardRow) & pieceMasks[r]) != 0) {
                            overlap = true;
                            break;
                        }
                    }
                    if (overlap) {
                        landingRow = startRow + 1;
                        break;
                    }
                }
                if (landingRow == bb.Height) landingRow = 0;

                if (landingRow + pieceHeight > bb.Height) {
                    continue;
                }

                BitBoard simulated = bb;
                for (int r = 0; r < pieceHeight; r++) {
                    int boardRow = landingRow + r;
                    simulated.SetRow(boardRow, simulated.GetRow(boardRow) | pieceMasks[r]);
                }

                int clearedCount = 0;
                int pieceCellsCleared = 0;
                for (int r = 0; r < pieceHeight; r++) {
                    int boardRow = landingRow + r;
                    if ((simulated.GetRow(boardRow) & simulated.FullRowMask) == simulated.FullRowMask) {
                        clearedCount++;
                        pieceCellsCleared += PopCount(pieceMasks[r]);
                    }
                }
                int erodedCells = clearedCount * pieceCellsCleared;

                for (int r = 0; r < simulated.Height; r++) {
                    if ((simulated.GetRow(r) & simulated.FullRowMask) == simulated.FullRowMask) {
                        for (int r2 = r; r2 < simulated.Height - 1; r2++) {
                            simulated.SetRow(r2, simulated.GetRow(r2 + 1));
                        }
                        simulated.SetRow(simulated.Height - 1, 0);
                        r--;
                    }
                }

                int maxHeight = 0;
                for (int r = simulated.Height - 1; r >= 0; r--) {
                    if (simulated.GetRow(r) != 0) {
                        maxHeight = r + 1;
                        break;
                    }
                }
                bool survivalMode = (double)maxHeight / bb.Height > 0.7;

                double nextExpectation = 0;
                for (int nextPiece = 0; nextPiece < 7; nextPiece++) {
                    double bestNextScore = double.NegativeInfinity;
                    int nextUniqueRots = UNIQUE_ROTATIONS[nextPiece];
                    
                    for (int nextRot = 0; nextRot < nextUniqueRots; nextRot++) {
                        Block[] nextCoords = PIECES[nextPiece][nextRot];
                        int nextWidth = 0;
                        int nextHeight = 0;
                        foreach (var b in nextCoords) {
                            if (b.C > nextWidth) nextWidth = b.C;
                            if (b.R > nextHeight) nextHeight = b.R;
                        }
                        nextWidth += 1;
                        nextHeight += 1;

                        for (int nextCol = 0; nextCol <= simulated.Width - nextWidth; nextCol++) {
                            uint[] nextPieceMasks = new uint[4];
                            for (int i = 0; i < nextCoords.Length; i++) {
                                nextPieceMasks[nextCoords[i].R] |= (1U << (nextCol + nextCoords[i].C));
                            }

                            int nextLandingRow = simulated.Height;
                            int nextStartScan = Math.Min(simulated.Height - 1, maxHeight);
                            for (int startRow = nextStartScan; startRow >= 0; startRow--) {
                                bool overlap = false;
                                for (int r = 0; r < nextHeight; r++) {
                                    int boardRow = startRow + r;
                                    if (boardRow < simulated.Height && (simulated.GetRow(boardRow) & nextPieceMasks[r]) != 0) {
                                        overlap = true;
                                        break;
                                    }
                                }
                                if (overlap) {
                                    nextLandingRow = startRow + 1;
                                    break;
                                }
                            }
                            if (nextLandingRow == simulated.Height) nextLandingRow = 0;

                            if (nextLandingRow + nextHeight > simulated.Height) {
                                continue;
                            }

                            BitBoard nextSimulated = simulated;
                            for (int r = 0; r < nextHeight; r++) {
                                int boardRow = nextLandingRow + r;
                                nextSimulated.SetRow(boardRow, nextSimulated.GetRow(boardRow) | nextPieceMasks[r]);
                            }

                            int nextClearedCount = 0;
                            int nextPieceCellsCleared = 0;
                            for (int r = 0; r < nextHeight; r++) {
                                int boardRow = nextLandingRow + r;
                                if ((nextSimulated.GetRow(boardRow) & nextSimulated.FullRowMask) == nextSimulated.FullRowMask) {
                                    nextClearedCount++;
                                    nextPieceCellsCleared += PopCount(nextPieceMasks[r]);
                                }
                            }
                            int nextErodedCells = nextClearedCount * nextPieceCellsCleared;

                            for (int r = 0; r < nextSimulated.Height; r++) {
                                if ((nextSimulated.GetRow(r) & nextSimulated.FullRowMask) == nextSimulated.FullRowMask) {
                                    for (int r2 = r; r2 < nextSimulated.Height - 1; r2++) {
                                        nextSimulated.SetRow(r2, nextSimulated.GetRow(r2 + 1));
                                    }
                                    nextSimulated.SetRow(nextSimulated.Height - 1, 0);
                                    r--;
                                }
                            }

                            double nextStateScore = EvaluateState(ref nextSimulated, nextLandingRow + PIECE_AVG_R[nextPiece][nextRot], nextErodedCells, survivalMode);
                            if (nextStateScore > bestNextScore) {
                                    bestNextScore = nextStateScore;
                            }
                        }
                    }

                    if (bestNextScore == double.NegativeInfinity) {
                        bestNextScore = -10000.0;
                    }
                    nextExpectation += bestNextScore;
                }
                
                double currentMoveScore = EvaluateState(ref simulated, landingRow + PIECE_AVG_R[pieceIndex][rot], erodedCells, survivalMode) + (nextExpectation / 7.0);

                if (currentMoveScore > bestScore) {
                    bestScore = currentMoveScore;
                    bestRotation = rot;
                    bestCol = col;
                }
            }
        }

        if (bestScore == double.NegativeInfinity) {
            bestRotation = 0;
            bestCol = 0;
        }

        return (bestRotation, bestCol);
    }

    static void Main() {
        int width = 10;
        int height = 20;
        BitBoard bb = new BitBoard();
        bb.Width = width;
        bb.Height = height;
        bb.FullRowMask = (1UL << width) - 1;
        Random rand = new Random(12345);

        int numPieces = 50000;
        Stopwatch sw = Stopwatch.StartNew();
        int piecesPlaced = 0;
        int linesCleared = 0;

        for (int i = 0; i < numPieces; i++) {
            int pieceIndex = rand.Next(7);
            bb.GetHolesAndBumpiness(out int holes, out int bumpiness);
            int depth = (holes > 0 || bumpiness > 4 || bb.GetMaxHeight() >= height * 0.4) ? 2 : 1;
            
            (int rotation, int col) = GetBestMove(pieceIndex, depth, ref bb);
            
            int cleared = PlacePiece(ref bb, pieceIndex, rotation, col);
            linesCleared += cleared;
            piecesPlaced++;

            if (bb.GetMaxHeight() >= height) {
                Console.WriteLine($"Game Over at piece {i}! Max height exceeded.");
                break;
            }
        }
        sw.Stop();
        Console.WriteLine($"Placed {piecesPlaced} pieces, cleared {linesCleared} lines.");
        Console.WriteLine($"Total time: {sw.Elapsed.TotalSeconds:F2} seconds.");
        Console.WriteLine($"Average time per piece: {sw.Elapsed.TotalMilliseconds / piecesPlaced:F4} ms.");
    }
