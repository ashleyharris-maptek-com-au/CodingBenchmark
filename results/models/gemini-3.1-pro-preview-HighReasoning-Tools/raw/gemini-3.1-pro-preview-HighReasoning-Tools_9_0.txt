using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Threading.Tasks;

public struct Cut
{
    public int Id;
    public int Length;
    public double PerturbedLength;
}

/// <summary>
/// A fast, iterative segment tree built over the bin capacities.
/// Allows finding the leftmost bin with capacity >= val in O(log N) time,
/// and updating capacities in O(log N) time.
/// </summary>
public class SegmentTree
{
    public int M;
    public int[] Tree;

    public SegmentTree(int n, int initialValue)
    {
        // Find the next power of 2 >= n to make the tree a perfect binary tree
        M = 1;
        while (M < n) M <<= 1;

        Tree = new int[2 * M];
        for (int i = 0; i < n; i++)
        {
            Tree[M + i] = initialValue;
        }
        for (int i = n; i < M; i++)
        {
            Tree[M + i] = 0;
        }
        for (int i = M - 1; i >= 1; i--)
        {
            Tree[i] = Math.Max(Tree[i << 1], Tree[(i << 1) | 1]);
        }
    }

    /// <summary>
    /// Finds the leftmost bin with capacity >= val, subtracts val from it,
    /// updates the tree, and returns the bin index. Returns -1 if no bin fits.
    /// </summary>
    public int FindAndSubtract(int val)
    {
        if (Tree[1] < val)
        {
            return -1;
        }

        int node = 1;
        while (node < M)
        {
            int left = node << 1;
            if (Tree[left] >= val)
            {
                node = left;
            }
            else
            {
                node = left + 1;
            }
        }

        int binIdx = node - M;
        Tree[node] -= val;

        // Propagate updates to ancestors
        while (node > 1)
        {
            node >>= 1;
            Tree[node] = Math.Max(Tree[node << 1], Tree[(node << 1) | 1]);
        }

        return binIdx;
    }

    /// <summary>
    /// Sets the capacity of a specific bin and propagates the updates.
    /// </summary>
    public void SetCapacity(int binIdx, int val)
    {
        int node = M + binIdx;
        Tree[node] = val;
        while (node > 1)
        {
            node >>= 1;
            Tree[node] = Math.Max(Tree[node << 1], Tree[(node << 1) | 1]);
        }
    }

    /// <summary>
    /// Returns the capacity of a specific bin.
    /// </summary>
    public int GetCapacity(int binIdx)
    {
        return Tree[M + binIdx];
    }
}

public class Solver
{
    public static void Main(string[] args)
    {
        // Initialize custom fast input reader
        FastParser.Initialize();
        if (!FastParser.TryReadNextInt(out int n) || !FastParser.TryReadNextInt(out int stockLength))
        {
            return;
        }

        // Handle edge case: empty inputs
        if (n == 0)
        {
            Console.WriteLine(0);
            return;
        }

        Cut[] originalCuts = new Cut[n];
        long totalCutLength = 0;
        for (int i = 0; i < n; i++)
        {
            if (FastParser.TryReadNextInt(out int len))
            {
                originalCuts[i] = new Cut { Id = i, Length = len };
                totalCutLength += len;
            }
        }

        // Calculate absolute mathematical lower bound
        int lowerBound = (int)((totalCutLength + stockLength - 1) / stockLength);

        // Run the multi-threaded cutting stock solver
        var result = Solve(originalCuts, stockLength, n, lowerBound);

        // Output total number of stocks used
        Console.WriteLine(result.NumBins);
        
        // Group cut indices by bin
        List<int>[] bins = new List<int>[result.NumBins];
        for (int i = 0; i < result.NumBins; i++)
        {
            bins[i] = new List<int>();
        }
        for (int i = 0; i < n; i++)
        {
            bins[result.BinAssignments[i]].Add(i);
        }

        // Fast writing of output to stdout using a large buffer
        using (var writer = new StreamWriter(Console.OpenStandardOutput(), System.Text.Encoding.ASCII, 65536))
        {
            for (int i = 0; i < result.NumBins; i++)
            {
                for (int j = 0; j < bins[i].Count; j++)
                {
                    writer.Write(bins[i][j]);
                    if (j < bins[i].Count - 1)
                        writer.Write(' ');
                }
                writer.WriteLine();
            }
        }
    }

    public class SolutionResult
    {
        public int[] BinAssignments;
        public int NumBins;
    }

    /// <summary>
    /// Spawns parallel workers to explore different initial sorting configurations
    /// and optimize them using FFD and SBE (Single Bin Elimination) local search.
    /// </summary>
    public static SolutionResult Solve(Cut[] originalCuts, int stockLength, int n, int lowerBound)
    {
        int numThreads = Environment.ProcessorCount;
        
        // Adaptively decide the number of iterations per thread based on N
        int iterationsPerThread = 1;
        if (n <= 1000) iterationsPerThread = 100;
        else if (n <= 10000) iterationsPerThread = 20;
        else if (n <= 50000) iterationsPerThread = 5;

        object lockObj = new object();
        SolutionResult globalBest = null;
        bool lowerBoundReached = false;

        Parallel.For(0, numThreads, (t, state) =>
        {
            Random rand = new Random(t + 42);
            Cut[] localCuts = new Cut[n];
            Array.Copy(originalCuts, localCuts, n);

            SolutionResult localBest = null;

            for (int iter = 0; iter < iterationsPerThread; iter++)
            {
                // Check if another thread has already reached the optimal solution
                if (System.Threading.Volatile.Read(ref lowerBoundReached)) break;

                // Thread 0, iteration 0: standard FFD (deterministic)
                if (t == 0 && iter == 0)
                {
                    for (int i = 0; i < n; i++)
                    {
                        localCuts[i].PerturbedLength = localCuts[i].Length;
                    }
                }
                else
                {
                    // Perturb cut lengths slightly to explore different packing configurations
                    double epsilon = stockLength * 0.02 * rand.NextDouble();
                    for (int i = 0; i < n; i++)
                    {
                        localCuts[i].PerturbedLength = localCuts[i].Length + (rand.NextDouble() - 0.5) * epsilon;
                    }
                }

                // Sort descending by perturbed length
                Array.Sort(localCuts, (a, b) => b.PerturbedLength.CompareTo(a.PerturbedLength));

                // Run FFD (First Fit Decreasing) heuristic
                var (ffdAssignments, ffdNumBins) = RunFFD(localCuts, stockLength, n);

                // Run SBE (Single Bin Elimination) local search to reduce the number of bins
                var (sbeAssignments, sbeNumBins) = RunSBE(ffdAssignments, ffdNumBins, originalCuts, stockLength, n, lowerBound);

                if (localBest == null || sbeNumBins < localBest.NumBins)
                {
                    localBest = new SolutionResult
                    {
                        BinAssignments = sbeAssignments,
                        NumBins = sbeNumBins
                    };

                    // If we reached the theoretical lower bound, we can stop early
                    if (sbeNumBins <= lowerBound)
                    {
                        System.Threading.Volatile.Write(ref lowerBoundReached, true);
                        state.Stop();
                        break;
                    }
                }
            }

            lock (lockObj)
            {
                if (globalBest == null || localBest.NumBins < globalBest.NumBins)
                {
                    globalBest = localBest;
                }
            }
        });

        return globalBest;
    }

    /// <summary>
    /// Standard First Fit Decreasing heuristic using a segment tree.
    /// </summary>
    public static (int[] binAssignments, int numBins) RunFFD(Cut[] sortedCuts, int stockLength, int n)
    {
        SegmentTree tree = new SegmentTree(n, stockLength);
        int[] binAssignments = new int[n];
        int maxBinIdx = -1;

        for (int i = 0; i < n; i++)
        {
            int binIdx = tree.FindAndSubtract(sortedCuts[i].Length);
            if (binIdx == -1)
            {
                binIdx = n - 1;
            }
            binAssignments[sortedCuts[i].Id] = binIdx;
            if (binIdx > maxBinIdx)
            {
                maxBinIdx = binIdx;
            }
        }

        return (binAssignments, maxBinIdx + 1);
    }

    /// <summary>
    /// Single Bin Elimination local search.
    /// Identifies the least-filled bins and tries to reallocate their cuts to other bins.
    /// Uses segment tree queries to find placements quickly.
    /// </summary>
    public static (int[] binAssignments, int numBins) RunSBE(
        int[] initialAssignments, 
        int initialNumBins, 
        Cut[] originalCuts, 
        int stockLength, 
        int n,
        int lowerBound)
    {
        int[] binAssignments = (int[])initialAssignments.Clone();
        int numBins = initialNumBins;

        if (numBins <= lowerBound)
        {
            return (binAssignments, numBins);
        }

        // Initialize capacities of all active bins
        int[] capacities = new int[numBins];
        for (int i = 0; i < numBins; i++) capacities[i] = stockLength;
        for (int i = 0; i < n; i++)
        {
            capacities[binAssignments[i]] -= originalCuts[i].Length;
        }

        SegmentTree tree = new SegmentTree(numBins, stockLength);
        for (int i = 0; i < numBins; i++)
        {
            tree.SetCapacity(i, capacities[i]);
        }

        // Adjacency list representation using head and next arrays to avoid allocations
        int[] head = new int[initialNumBins];
        int[] next = new int[n];
        
        List<int> activeBins = new List<int>();
        for (int i = 0; i < numBins; i++)
        {
            activeBins.Add(i);
        }

        List<(int binIdx, int originalCap)> rollback = new List<(int, int)>();
        List<(int cutId, int newBin)> tempAssignments = new List<(int, int)>();

        bool improved = true;
        while (improved && numBins > lowerBound)
        {
            improved = false;

            for (int i = 0; i < initialNumBins; i++) head[i] = -1;
            for (int i = 0; i < n; i++)
            {
                int b = binAssignments[i];
                next[i] = head[b];
                head[b] = i;
            }

            // Sort active bins by remaining capacity descending (least filled first)
            activeBins.Sort((a, b) => tree.GetCapacity(b).CompareTo(tree.GetCapacity(a)));

            // Try to eliminate the least-filled bins
            int candidatesToTry = Math.Min(activeBins.Count, 200);

            for (int cIdx = 0; cIdx < candidatesToTry; cIdx++)
            {
                int candidateBin = activeBins[cIdx];
                
                int cutCount = 0;
                int curr = head[candidateBin];
                while (curr != -1)
                {
                    cutCount++;
                    curr = next[curr];
                }

                // If bin is empty or has too many cuts, we skip
                if (cutCount == 0 || cutCount > 15) 
                {
                    continue;
                }

                int[] tempCuts = new int[cutCount];
                curr = head[candidateBin];
                int idx = 0;
                while (curr != -1)
                {
                    tempCuts[idx++] = curr;
                    curr = next[curr];
                }

                // Sort candidate cuts descending
                Array.Sort(tempCuts, (a, b) => originalCuts[b].Length.CompareTo(originalCuts[a].Length));

                int originalCandidateCap = tree.GetCapacity(candidateBin);
                tree.SetCapacity(candidateBin, 0);

                rollback.Clear();
                tempAssignments.Clear();
                bool possible = true;

                for (int i = 0; i < cutCount; i++)
                {
                    int cutId = tempCuts[i];
                    int cutLen = originalCuts[cutId].Length;

                    int newBin = tree.FindAndSubtract(cutLen);
                    if (newBin == -1)
                    {
                        possible = false;
                        break;
                    }

                    rollback.Add((newBin, tree.GetCapacity(newBin) + cutLen));
                    tempAssignments.Add((cutId, newBin));
                }

                if (possible)
                {
                    for (int i = 0; i < tempAssignments.Count; i++)
                    {
                        binAssignments[tempAssignments[i].cutId] = tempAssignments[i].newBin;
                    }

                    activeBins.Remove(candidateBin);
                    numBins--;
                    improved = true;
                    break; // Rebuild and retry
                }
                else
                {
                    // Rollback capacity changes
                    for (int i = 0; i < rollback.Count; i++)
                    {
                        tree.SetCapacity(rollback[i].binIdx, rollback[i].originalCap);
                    }
                    tree.SetCapacity(candidateBin, originalCandidateCap);
                }
            }
        }

        // Compact bin indices to be contiguous starting from 0
        int[] compactedAssignments = new int[n];
        Dictionary<int, int> binMap = new Dictionary<int, int>();
        int nextNewBin = 0;

        for (int i = 0; i < n; i++)
        {
            int oldBin = binAssignments[i];
            if (!binMap.TryGetValue(oldBin, out int newBin))
            {
                newBin = nextNewBin++;
                binMap[oldBin] = newBin;
            }
            compactedAssignments[i] = newBin;
        }

        return (compactedAssignments, nextNewBin);
    }
}

/// <summary>
/// A fast I/O parser that reads integer values directly from the input stream.
/// Significantly outperforms standard Console.ReadLine() for large inputs.
/// </summary>
public static class FastParser
{
    private const int BufferSize = 1024 * 1024;
    private static readonly byte[] buffer = new byte[BufferSize];
    private static int bytesRead = 0;
    private static int bufferPointer = 0;
    private static Stream inputStream;

    public static void Initialize()
    {
        inputStream = Console.OpenStandardInput();
        bytesRead = 0;
        bufferPointer = 0;
    }

    private static byte ReadByte()
    {
        if (bufferPointer >= bytesRead)
        {
            bytesRead = inputStream.Read(buffer, 0, BufferSize);
            bufferPointer = 0;
            if (bytesRead == 0)
                return 0; // EOF
        }
        return buffer[bufferPointer++];
    }

    public static bool TryReadNextInt(out int result)
    {
        result = 0;
        byte b = ReadByte();
        while (b != 0 && b <= 32) // Skip whitespaces and newlines
        {
            b = ReadByte();
        }
        if (b == 0)
            return false;

        bool negative = false;
        if (b == '-')
        {
            negative = true;
            b = ReadByte();
        }

        while (b >= '0' && b <= '9')
        {
            result = result * 10 + (b - '0');
            b = ReadByte();
        }

        if (negative)
            result = -result;

        return true;
    }
}
