use std::io::{self, BufRead};

fn main() -> io::Result<()> {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let nv_line = match lines.next() {
        Some(Ok(line)) => line,
        _ => return Ok(()),
    };
    if nv_line.trim().is_empty() {
        return Ok(());
    }
    let nv: usize = match nv_line.trim().parse() {
        Ok(val) => val,
        _ => return Ok(()),
    };

    let mut poly = Vec::with_capacity(nv);
    for _ in 0..nv {
        let line = match lines.next() {
            Some(Ok(l)) => l,
            _ => return Ok(()),
        };
        let mut parts = line.split_whitespace();
        let x: f64 = parts.next().unwrap().parse().unwrap();
        let y: f64 = parts.next().unwrap().parse().unwrap();
        poly.push((x, y));
    }

    let sun_line = match lines.next() {
        Some(Ok(l)) => l,
        _ => return Ok(()),
    };
    let mut parts = sun_line.split_whitespace();
    let sx: f64 = parts.next().unwrap().parse().unwrap();
    let sy: f64 = parts.next().unwrap().parse().unwrap();
    let sz: f64 = parts.next().unwrap().parse().unwrap();

    solve(poly, [sx, sy, sz]);

    Ok(())
}

fn solve(poly: Vec<(f64, f64)>, s: [f64; 3]) {
    if poly.is_empty() {
        return;
    }

    let sz = s[2];
    let kx = s[0] / sz;
    let ky = s[1] / sz;

    // 1. Align c3 with s / |s|
    let s_len = (s[0]*s[0] + s[1]*s[1] + s[2]*s[2]).sqrt();
    let c3 = [s[0]/s_len, s[1]/s_len, s[2]/s_len];

    // 2. Find any perpendicular vectors u and v to c3
    let mut u = if c3[0].abs() > 0.9 {
        [0.0, 1.0, 0.0]
    } else {
        [1.0, 0.0, 0.0]
    };
    let dot = u[0]*c3[0] + u[1]*c3[1] + u[2]*c3[2];
    u = [u[0] - dot*c3[0], u[1] - dot*c3[1], u[2] - dot*c3[2]];
    let u_len = (u[0]*u[0] + u[1]*u[1] + u[2]*u[2]).sqrt();
    u = [u[0]/u_len, u[1]/u_len, u[2]/u_len];

    let v = [
        c3[1]*u[2] - c3[2]*u[1],
        c3[2]*u[0] - c3[0]*u[2],
        c3[0]*u[1] - c3[1]*u[0]
    ];

    // 3. Find the exact theta where len1 = len2
    let mut low = 0.0;
    let mut high = 90.0_f64.to_radians();
    let mut best_theta = 0.0;
    for _ in 0..60 {
        let mid = 0.5 * (low + high);
        let cos_t = mid.cos();
        let sin_t = mid.sin();
        let c1 = [
            cos_t * u[0] + sin_t * v[0],
            cos_t * u[1] + sin_t * v[1],
            cos_t * u[2] + sin_t * v[2],
        ];
        let c2 = [
            -sin_t * u[0] + cos_t * v[0],
            -sin_t * u[1] + cos_t * v[1],
            -sin_t * u[2] + cos_t * v[2],
        ];
        let p_c1 = [c1[0] - c1[2]*kx, c1[1] - c1[2]*ky];
        let p_c2 = [c2[0] - c2[2]*kx, c2[1] - c2[2]*ky];
        let len1 = (p_c1[0]*p_c1[0] + p_c1[1]*p_c1[1]).sqrt();
        let len2 = (p_c2[0]*p_c2[0] + p_c2[1]*p_c2[1]).sqrt();
        if len1 < len2 {
            low = mid;
        } else {
            high = mid;
        }
        best_theta = mid;
    }

    let cos_t = best_theta.cos();
    let sin_t = best_theta.sin();
    let c1 = [
        cos_t * u[0] + sin_t * v[0],
        cos_t * u[1] + sin_t * v[1],
        cos_t * u[2] + sin_t * v[2],
    ];
    let c2 = [
        -sin_t * u[0] + cos_t * v[0],
        -sin_t * u[1] + cos_t * v[1],
        -sin_t * u[2] + cos_t * v[2],
    ];
    let p_c1 = [c1[0] - c1[2]*kx, c1[1] - c1[2]*ky];
    let p_c2 = [c2[0] - c2[2]*kx, c2[1] - c2[2]*ky];

    // Compute transformation matrix A from uv to xy
    let e1 = p_c1;
    let e2 = p_c2;
    let d = e1[0]*e2[1] - e1[1]*e2[0];

    // Transform poly to u', v' coordinates
    let mut poly_uv = Vec::with_capacity(poly.len());
    for p in &poly {
        let x = p.0;
        let y = p.1;
        let u_coord = (e2[1]*x - e2[0]*y) / d;
        let v_coord = (-e1[1]*x + e1[0]*y) / d;
        let up = u_coord + v_coord;
        let vp = u_coord - v_coord;
        poly_uv.push((up, vp));
    }

    // Find bounding box in u', v'
    let mut u_min = poly_uv[0].0;
    let mut u_max = poly_uv[0].0;
    let mut v_min = poly_uv[0].1;
    let mut v_max = poly_uv[0].1;
    for p in &poly_uv {
        if p.0 < u_min { u_min = p.0; }
        if p.0 > u_max { u_max = p.0; }
        if p.1 < v_min { v_min = p.1; }
        if p.1 > v_max { v_max = p.1; }
    }

    // Grid cell size is 20.0 x 20.0 (since max scale S = 10.0, [-10, 10] x [-10, 10] in u', v' is 20x20)
    let w = 20.0;
    let h = 20.0;

    let mut centers = Vec::new();
    let mut single_cell_scale = None;

    if u_max - u_min <= w && v_max - v_min <= h {
        // Fits in a single cell!
        let uc = 0.5 * (u_min + u_max);
        let vc = 0.5 * (v_min + v_max);
        let size = (u_max - u_min).max(v_max - v_min);
        let scale = (0.5 * size + 1e-5).max(0.1).min(10.0);
        single_cell_scale = Some(scale);
        centers.push((uc, vc));
    } else {
        // Find boundary cells
        let mut boundary_cells = std::collections::HashSet::new();
        let n = poly_uv.len();
        
        let mut i_min_grid = i32::MAX;
        let mut i_max_grid = i32::MIN;
        let mut j_min_grid = i32::MAX;
        let mut j_max_grid = i32::MIN;

        for k in 0..n {
            let p1 = (poly_uv[k].0 - u_min, poly_uv[k].1 - v_min);
            let p2 = (poly_uv[(k+1)%n].0 - u_min, poly_uv[(k+1)%n].1 - v_min);
            let cells = get_intersecting_cells(p1, p2, w, h);
            for cell in cells {
                boundary_cells.insert(cell);
                if cell.0 < i_min_grid { i_min_grid = cell.0; }
                if cell.0 > i_max_grid { i_max_grid = cell.0; }
                if cell.1 < j_min_grid { j_min_grid = cell.1; }
                if cell.1 > j_max_grid { j_max_grid = cell.1; }
            }
        }

        // Bounding box dimensions in grid cells
        let grid_w = i_max_grid - i_min_grid + 3;
        let grid_h = j_max_grid - j_min_grid + 3;
        let bbox_size = grid_w as i64 * grid_h as i64;

        if bbox_size <= 100_000 {
            // Strategy 1: Outside flood fill (very fast, no point-in-polygon checks)
            let mut visited = vec![false; (grid_w * grid_h) as usize];
            let mut is_boundary = vec![false; (grid_w * grid_h) as usize];

            for &cell in &boundary_cells {
                let gi = cell.0 - i_min_grid + 1;
                let gj = cell.1 - j_min_grid + 1;
                if gi >= 0 && gi < grid_w && gj >= 0 && gj < grid_h {
                    is_boundary[(gi * grid_h + gj) as usize] = true;
                    visited[(gi * grid_h + gj) as usize] = true;
                }
            }

            let mut queue = std::collections::VecDeque::new();
            // Add outer boundary cells of the padded grid to queue
            for gi in 0..grid_w {
                for &gj in &[0, grid_h - 1] {
                    let idx = (gi * grid_h + gj) as usize;
                    if !is_boundary[idx] && !visited[idx] {
                        visited[idx] = true;
                        queue.push_back((gi, gj));
                    }
                }
            }
            for gj in 0..grid_h {
                for &gi in &[0, grid_w - 1] {
                    let idx = (gi * grid_h + gj) as usize;
                    if !is_boundary[idx] && !visited[idx] {
                        visited[idx] = true;
                        queue.push_back((gi, gj));
                    }
                }
            }

            while let Some((gi, gj)) = queue.pop_front() {
                let neighbors = [
                    (gi + 1, gj),
                    (gi - 1, gj),
                    (gi, gj + 1),
                    (gi, gj - 1),
                ];
                for &neigh in &neighbors {
                    let ngi = neigh.0;
                    let ngj = neigh.1;
                    if ngi >= 0 && ngi < grid_w && ngj >= 0 && ngj < grid_h {
                        let idx = (ngi * grid_h + ngj) as usize;
                        if !is_boundary[idx] && !visited[idx] {
                            visited[idx] = true;
                            queue.push_back((ngi, ngj));
                        }
                    }
                }
            }

            // All cells inside grid_w x grid_h that are NOT visited must be boundary or interior
            for gi in 0..grid_w {
                for gj in 0..grid_h {
                    let idx = (gi * grid_h + gj) as usize;
                    if !visited[idx] || is_boundary[idx] {
                        let cell_i = gi + i_min_grid - 1;
                        let cell_j = gj + j_min_grid - 1;
                        let uc = u_min + (cell_i as f64 + 0.5) * w;
                        let vc = v_min + (cell_j as f64 + 0.5) * h;
                        centers.push((uc, vc));
                    }
                }
            }
        } else {
            // Strategy 2: Inside flood fill (sparse/large bounding box)
            let mut visited = std::collections::HashSet::new();
            for &cell in &boundary_cells {
                let key = ((cell.0 as u64) << 32) | ((cell.1 as u32) as u64);
                visited.insert(key);
            }

            let mut interior_cells = std::collections::HashSet::new();
            let mut queue = std::collections::VecDeque::new();

            // Find neighbors of boundary cells that are inside the polygon
            for &cell in &boundary_cells {
                let neighbors = [
                    (cell.0 + 1, cell.1),
                    (cell.0 - 1, cell.1),
                    (cell.0, cell.1 + 1),
                    (cell.0, cell.1 - 1),
                ];
                for &neigh in &neighbors {
                    let n_key = ((neigh.0 as u64) << 32) | ((neigh.1 as u32) as u64);
                    if !visited.contains(&n_key) {
                        let uc = u_min + (neigh.0 as f64 + 0.5) * w;
                        let vc = v_min + (neigh.1 as f64 + 0.5) * h;
                        if point_in_polygon(uc, vc, &poly_uv) {
                            interior_cells.insert(n_key);
                            queue.push_back(neigh);
                        }
                        visited.insert(n_key);
                    }
                }
            }

            // BFS for interior cells
            while let Some(cell) = queue.pop_front() {
                let neighbors = [
                    (cell.0 + 1, cell.1),
                    (cell.0 - 1, cell.1),
                    (cell.0, cell.1 + 1),
                    (cell.0, cell.1 - 1),
                ];
                for &neigh in &neighbors {
                    let n_key = ((neigh.0 as u64) << 32) | ((neigh.1 as u32) as u64);
                    if !visited.contains(&n_key) {
                        interior_cells.insert(n_key);
                        queue.push_back(neigh);
                        visited.insert(n_key);
                    }
                }
            }

            // Collect all centers
            for &cell in &boundary_cells {
                let uc = u_min + (cell.0 as f64 + 0.5) * w;
                let vc = v_min + (cell.1 as f64 + 0.5) * h;
                centers.push((uc, vc));
            }
            for &key in &interior_cells {
                let i = (key >> 32) as i32;
                let j = key as u32 as i32;
                let uc = u_min + (i as f64 + 0.5) * w;
                let vc = v_min + (j as f64 + 0.5) * h;
                centers.push((uc, vc));
            }
        }
    }

    // Now output the tetrahedrons
    println!("{}", centers.len());

    let r_mat = [
        c1[0], c2[0], c3[0],
        c1[1], c2[1], c3[1],
        c1[2], c2[2], c3[2],
    ];
    let q = mat_to_quat(r_mat);

    let scale = match single_cell_scale {
        Some(s) => s,
        None => 10.0,
    };

    for (k, &c_uv) in centers.iter().enumerate() {
        let up = c_uv.0;
        let vp = c_uv.1;
        // Convert up, vp back to x0, y0
        let u_val = 0.5 * (up + vp);
        let v_val = 0.5 * (up - vp);
        let x0 = u_val * e1[0] + v_val * e2[0];
        let y0 = u_val * e1[1] + v_val * e2[1];

        // Unique z height
        let pz = 1000.0 + (k as f64) * 100.0;
        let px = x0 + pz * kx;
        let py = y0 + pz * ky;

        println!(
            "{:.9} {:.9} {:.9} {:.9} {:.9} {:.9} {:.9} {:.9}",
            px, py, pz, q[0], q[1], q[2], q[3], scale
        );
    }
}

fn get_intersecting_cells(
    p1: (f64, f64), p2: (f64, f64),
    w: f64, h: f64
) -> Vec<(i32, i32)> {
    let mut t_coords = Vec::new();
    t_coords.push(0.0);
    t_coords.push(1.0);

    let (x1, y1) = p1;
    let (x2, y2) = p2;

    if (x2 - x1).abs() > 1e-9 {
        let i_start = (x1 / w).floor() as i32;
        let i_end = (x2 / w).floor() as i32;
        let i_min = i_start.min(i_end);
        let i_max = i_start.max(i_end);
        for i in (i_min + 1)..=i_max {
            let t = (i as f64 * w - x1) / (x2 - x1);
            if t > 0.0 && t < 1.0 {
                t_coords.push(t);
            }
        }
    }

    if (y2 - y1).abs() > 1e-9 {
        let j_start = (y1 / h).floor() as i32;
        let j_end = (y2 / h).floor() as i32;
        let j_min = j_start.min(j_end);
        let j_max = j_start.max(j_end);
        for j in (j_min + 1)..=j_max {
            let t = (j as f64 * h - y1) / (y2 - y1);
            if t > 0.0 && t < 1.0 {
                t_coords.push(t);
            }
        }
    }

    t_coords.sort_by(|a, b| a.partial_cmp(b).unwrap());
    
    let mut cells = Vec::new();
    for k in 0..(t_coords.len() - 1) {
        let ta = t_coords[k];
        let tb = t_coords[k+1];
        if (tb - ta).abs() < 1e-9 {
            continue;
        }
        let xm = x1 + 0.5 * (ta + tb) * (x2 - x1);
        let ym = y1 + 0.5 * (ta + tb) * (y2 - y1);
        let i = (xm / w).floor() as i32;
        let j = (ym / h).floor() as i32;
        cells.push((i, j));
    }
    cells
}

fn point_in_polygon(u: f64, v: f64, poly: &[(f64, f64)]) -> bool {
    let mut inside = false;
    let n = poly.len();
    let mut j = n - 1;
    for i in 0..n {
        let (ui, vi) = poly[i];
        let (uj, vj) = poly[j];
        if ((vi > v) != (vj > v)) && (u < (uj - ui) * (v - vi) / (vj - vi) + ui) {
            inside = !inside;
        }
        j = i;
    }
    inside
}

fn mat_to_quat(r: [f64; 9]) -> [f64; 4] {
    let tr = r[0] + r[4] + r[8];
    if tr > 0.0 {
        let s = (tr + 1.0).sqrt() * 2.0;
        [
            0.25 * s,
            (r[7] - r[5]) / s,
            (r[2] - r[6]) / s,
            (r[3] - r[1]) / s,
        ]
    } else if r[0] > r[4] && r[0] > r[8] {
        let s = (1.0 + r[0] - r[4] - r[8]).sqrt() * 2.0;
        [
            (r[7] - r[5]) / s,
            0.25 * s,
            (r[1] + r[3]) / s,
            (r[2] + r[6]) / s,
        ]
    } else if r[4] > r[8] {
        let s = (1.0 + r[4] - r[0] - r[8]).sqrt() * 2.0;
        [
            (r[2] - r[6]) / s,
            (r[1] + r[3]) / s,
            0.25 * s,
            (r[5] + r[7]) / s,
        ]
    } else {
        let s = (1.0 + r[8] - r[0] - r[4]).sqrt() * 2.0;
        [
            (r[3] - r[1]) / s,
            (r[2] + r[6]) / s,
            (r[5] + r[7]) / s,
            0.25 * s,
        ]
    }
}
