using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;

public class Program
{
    // Fast Scanner class to optimize reading large inputs (N up to 2000) from stdin.
    public class FastScanner
    {
        private readonly TextReader _reader;
        private readonly char[] _buffer = new char[65536];
        private int _bufferLength = 0;
        private int _bufferPointer = 0;

        public FastScanner()
        {
            _reader = Console.In;
        }

        private bool ReadNextBuffer()
        {
            _bufferLength = _reader.Read(_buffer, 0, _buffer.Length);
            _bufferPointer = 0;
            return _bufferLength > 0;
        }

        private int ReadChar()
        {
            if (_bufferPointer >= _bufferLength)
            {
                if (!ReadNextBuffer()) return -1;
            }
            return _buffer[_bufferPointer++];
        }

        public int NextInt()
        {
            int c = ReadChar();
            while (c != -1 && c <= 32)
            {
                c = ReadChar();
            }
            if (c == -1) return -1;

            bool neg = false;
            if (c == '-')
            {
                neg = true;
                c = ReadChar();
            }

            int res = 0;
            while (c >= '0' && c <= '9')
            {
                res = res * 10 + (c - '0');
                c = ReadChar();
            }

            return neg ? -res : res;
        }
    }

    // Fast Xorshift Pseudo-Random Number Generator to optimize SA transitions.
    public class FastRandom
    {
        private uint _x;
        public FastRandom(uint seed)
        {
            _x = seed != 0 ? seed : 5489U;
        }
        public int Next(int max)
        {
            _x ^= _x << 13;
            _x ^= _x >> 17;
            _x ^= _x << 5;
            return (int)(_x % (uint)max);
        }
        public double NextDouble()
        {
            _x ^= _x << 13;
            _x ^= _x >> 17;
            _x ^= _x << 5;
            return (double)_x / uint.MaxValue;
        }
    }

    public static void Main()
    {
        // Enforce culture invariant parsing/formatting
        System.Threading.Thread.CurrentThread.CurrentCulture = System.Globalization.CultureInfo.InvariantCulture;

        FastScanner scanner = new FastScanner();
        int N = scanner.NextInt();
        if (N <= 0) return;

        int[] flow = new int[N * N];
        for (int i = 0; i < N * N; i++)
        {
            flow[i] = scanner.NextInt();
        }

        int[] dist = new int[N * N];
        for (int i = 0; i < N * N; i++)
        {
            dist[i] = scanner.NextInt();
        }

        Stopwatch stopwatch = new Stopwatch();
        stopwatch.Start();

        // Check if both flow and dist matrices are symmetric.
        // If they are symmetric, the swap delta evaluation runs twice as fast.
        bool isSymmetric = true;
        for (int i = 0; i < N; i++)
        {
            for (int j = i + 1; j < N; j++)
            {
                if (flow[i * N + j] != flow[j * N + i] || dist[i * N + j] != dist[j * N + i])
                {
                    isSymmetric = false;
                    break;
                }
            }
            if (!isSymmetric) break;
        }

        int[] bestP;
        long bestCost;

        if (N <= 10)
        {
            SolveExact(N, flow, dist, out bestP, out bestCost);
        }
        else
        {
            SolveHeuristic(N, flow, dist, isSymmetric, stopwatch, out bestP, out bestCost);
        }

        // Output results to stdout
        Console.WriteLine(bestCost);
        for (int i = 0; i < N; i++)
        {
            Console.Write(bestP[i] + (i == N - 1 ? "" : " "));
        }
        Console.WriteLine();
    }

    // Dynamic budget allocation based on the instance size N to prevent global timeouts
    // while ensuring maximum exploration time for larger, harder instances.
    private static double GetBudgetMs(int N)
    {
        if (N <= 10) return 100;
        if (N <= 20) return 500;
        if (N <= 50) return 2000;
        if (N <= 100) return 5000;
        if (N <= 200) return 15000;
        if (N <= 500) return 45000;
        if (N <= 1000) return 90000;
        return 280000; // Under 5 minutes (300 seconds) for N up to 2000
    }

    // Exact Backtracking DFS Solver for very small N (N <= 10).
    private static void SolveExact(int N, int[] flow, int[] dist, out int[] bestP, out long bestCost)
    {
        bestP = new int[N];
        for (int i = 0; i < N; i++) bestP[i] = i;
        bestCost = long.MaxValue;

        int[] currentP = new int[N];
        bool[] used = new bool[N];
        SolveExactDfs(0, N, flow, dist, currentP, used, ref bestCost, bestP);
    }

    private static void SolveExactDfs(int step, int N, int[] flow, int[] dist, int[] currentP, bool[] used, ref long bestCost, int[] bestP)
    {
        if (step == N)
        {
            long cost = ComputeCost(currentP, flow, dist, N);
            if (cost < bestCost)
            {
                bestCost = cost;
                Array.Copy(currentP, bestP, N);
            }
            return;
        }

        for (int v = 0; v < N; v++)
        {
            if (!used[v])
            {
                used[v] = true;
                currentP[step] = v;
                SolveExactDfs(step + 1, N, flow, dist, currentP, used, ref bestCost, bestP);
                used[v] = false;
            }
        }
    }

    // Multi-Start Simulated Annealing (MSSA) Metaheuristic Solver for N > 10.
    private static void SolveHeuristic(int N, int[] flow, int[] dist, bool isSymmetric, Stopwatch stopwatch, out int[] bestOverallP, out long bestOverallCost)
    {
        FastRandom rand = new FastRandom(1337);

        // Get initial permutation using sorting-based flow-distance alignment heuristic.
        int[] fdP = GetFdSortedPermutation(N, flow, dist);
        
        bestOverallP = new int[N];
        Array.Copy(fdP, bestOverallP, N);
        bestOverallCost = ComputeCost(fdP, flow, dist, N);

        double totalBudgetMs = GetBudgetMs(N); 
        int startId = 1;

        while (true)
        {
            double elapsed = stopwatch.Elapsed.TotalMilliseconds;
            double remaining = totalBudgetMs - elapsed;
            if (remaining < 500) break; // Terminate if insufficient time remains

            // Allocate time for this start using a geometric progression.
            double durationMs = Math.Max(200, remaining * 0.35);

            int[] startP;
            double tempScale;
            if (startId == 1)
            {
                startP = fdP;
                tempScale = 0.05; // Low temperature start to refine F-D constructive solution
            }
            else if (startId % 2 == 1)
            {
                startP = GetRandomizedFdPermutation(fdP, N, rand);
                tempScale = 0.3;  // Moderate temperature start for perturbation of F-D solution
            }
            else
            {
                startP = GetRandomPermutation(N, rand);
                tempScale = 0.5;  // Higher temperature start for global exploration from random state
            }

            double initialT = EstimateT0(N, startP, flow, dist, isSymmetric, rand) * tempScale;
            double finalT = initialT * 0.00001;
            if (finalT < 1e-6) finalT = 1e-6;

            RunSimulatedAnnealing(N, flow, dist, isSymmetric, startP, initialT, finalT, durationMs, rand, bestOverallP, ref bestOverallCost, stopwatch);

            startId++;
        }

        // Final local search polish on the best overall found permutation.
        PolishSolution(bestOverallP, ref bestOverallCost, flow, dist, N, isSymmetric, totalBudgetMs, stopwatch, rand);
    }

    // Constructive Heuristic: aligns high flow facilities to central locations with small distances.
    private static int[] GetFdSortedPermutation(int N, int[] flow, int[] dist)
    {
        long[] flowSums = new long[N];
        int[] facilities = new int[N];
        for (int i = 0; i < N; i++)
        {
            facilities[i] = i;
            long sum = 0;
            int offset = i * N;
            for (int j = 0; j < N; j++) sum += flow[offset + j];
            flowSums[i] = sum;
        }

        long[] negatedFlowSums = new long[N];
        for (int i = 0; i < N; i++) negatedFlowSums[i] = -flowSums[i];
        Array.Sort(negatedFlowSums, facilities);

        long[] distSums = new long[N];
        int[] locations = new int[N];
        for (int i = 0; i < N; i++)
        {
            locations[i] = i;
            long sum = 0;
            int offset = i * N;
            for (int j = 0; j < N; j++) sum += dist[offset + j];
            distSums[i] = sum;
        }
        Array.Sort(distSums, locations);

        int[] p = new int[N];
        for (int i = 0; i < N; i++)
        {
            p[facilities[i]] = locations[i];
        }
        return p;
    }

    private static int[] GetRandomizedFdPermutation(int[] fdP, int N, FastRandom rand)
    {
        int[] p = new int[N];
        Array.Copy(fdP, p, N);
        int swaps = Math.Max(1, N * 5 / 100);
        for (int s = 0; s < swaps; s++)
        {
            int u = rand.Next(N);
            int v = rand.Next(N);
            while (u == v) v = rand.Next(N);
            int tmp = p[u];
            p[u] = p[v];
            p[v] = tmp;
        }
        return p;
    }

    private static int[] GetRandomPermutation(int N, FastRandom rand)
    {
        int[] p = new int[N];
        for (int i = 0; i < N; i++) p[i] = i;
        for (int i = N - 1; i > 0; i--)
        {
            int j = rand.Next(i + 1);
            int tmp = p[i];
            p[i] = p[j];
            p[j] = tmp;
        }
        return p;
    }

    // Estimate the starting temperature T0 based on average positive delta values of random moves.
    private static double EstimateT0(int N, int[] p, int[] flow, int[] dist, bool isSymmetric, FastRandom rand)
    {
        long sumPosDelta = 0;
        int countPosDelta = 0;
        int samples = Math.Min(200, N * (N - 1) / 2);
        if (samples < 10) samples = 10;

        for (int i = 0; i < samples; i++)
        {
            int u = rand.Next(N);
            int v = rand.Next(N);
            while (u == v) v = rand.Next(N);

            long delta = EvaluateSwap(u, v, p, flow, dist, N, isSymmetric);
            if (delta > 0)
            {
                sumPosDelta += delta;
                countPosDelta++;
            }
        }

        if (countPosDelta > 0)
        {
            return (double)sumPosDelta / countPosDelta;
        }
        return 1.0;
    }

    // Core Simulated Annealing search loop.
    private static void RunSimulatedAnnealing(
        int N, int[] flow, int[] dist, bool isSymmetric,
        int[] initialP, double initialT, double finalT, double durationMs,
        FastRandom rand, int[] bestOverallP, ref long bestOverallCost,
        Stopwatch stopwatch)
    {
        int[] currentP = new int[N];
        Array.Copy(initialP, currentP, N);

        long currentCost = ComputeCost(currentP, flow, dist, N);

        int[] bestStartP = new int[N];
        Array.Copy(currentP, bestStartP, N);
        long bestStartCost = currentCost;

        double startTime = stopwatch.Elapsed.TotalMilliseconds;
        double endTime = startTime + durationMs;

        double T = initialT;
        int iter = 0;

        while (true)
        {
            iter++;
            if (iter % 2000 == 0) // Periodically update temperature and check timeout
            {
                double curTime = stopwatch.Elapsed.TotalMilliseconds;
                if (curTime >= endTime) break;

                double progress = (curTime - startTime) / durationMs;
                if (progress > 1.0) progress = 1.0;

                T = initialT * Math.Pow(finalT / initialT, progress);
            }

            int u = rand.Next(N);
            int v = rand.Next(N);
            while (u == v) v = rand.Next(N);

            long delta = EvaluateSwap(u, v, currentP, flow, dist, N, isSymmetric);

            if (delta <= 0)
            {
                int tmp = currentP[u];
                currentP[u] = currentP[v];
                currentP[v] = tmp;
                currentCost += delta;

                if (currentCost < bestStartCost)
                {
                    bestStartCost = currentCost;
                    Array.Copy(currentP, bestStartP, N);
                }
            }
            else
            {
                double pAccept = Math.Exp(-delta / T);
                if (rand.NextDouble() < pAccept)
                {
                    int tmp = currentP[u];
                    currentP[u] = currentP[v];
                    currentP[v] = tmp;
                    currentCost += delta;
                }
            }
        }

        if (bestStartCost < bestOverallCost)
        {
            bestOverallCost = bestStartCost;
            Array.Copy(bestStartP, bestOverallP, N);
        }
    }

    // Local improvement polishing to land in the exact local minimum basin.
    private static void PolishSolution(
        int[] p, ref long cost, int[] flow, int[] dist, int N, bool isSymmetric,
        double timeLimitMs, Stopwatch stopwatch, FastRandom rand)
    {
        if (N <= 300)
        {
            bool improved = true;
            for (int pass = 0; pass < 10 && improved; pass++)
            {
                improved = false;
                for (int i = 0; i < N; i++)
                {
                    if (stopwatch.Elapsed.TotalMilliseconds >= timeLimitMs) return;
                    for (int j = i + 1; j < N; j++)
                    {
                        long delta = EvaluateSwap(i, j, p, flow, dist, N, isSymmetric);
                        if (delta < 0)
                        {
                            int tmp = p[i];
                            p[i] = p[j];
                            p[j] = tmp;
                            cost += delta;
                            improved = true;
                        }
                    }
                }
            }
        }
        else
        {
            int maxFailures = 5 * N;
            int failures = 0;
            while (failures < maxFailures)
            {
                if (stopwatch.Elapsed.TotalMilliseconds >= timeLimitMs) return;

                int u = rand.Next(N);
                int v = rand.Next(N);
                while (u == v) v = rand.Next(N);

                long delta = EvaluateSwap(u, v, p, flow, dist, N, isSymmetric);
                if (delta < 0)
                {
                    int tmp = p[u];
                    p[u] = p[v];
                    p[v] = tmp;
                    cost += delta;
                    failures = 0;
                }
                else
                {
                    failures++;
                }
            }
        }
    }

    // Compute QAP cost from scratch.
    private static long ComputeCost(int[] p, int[] flow, int[] dist, int N)
    {
        long cost = 0;
        for (int i = 0; i < N; i++)
        {
            int pi = p[i];
            int flow_i_offset = i * N;
            for (int j = 0; j < N; j++)
            {
                cost += (long)flow[flow_i_offset + j] * dist[pi * N + p[j]];
            }
        }
        return cost;
    }

    // Highly optimized O(N) evaluation of a 2-opt swap delta.
    private static long EvaluateSwap(int u, int v, int[] p, int[] flow, int[] dist, int N, bool isSymmetric)
    {
        if (u > v) { int tmp = u; u = v; v = tmp; }

        int pu = p[u];
        int pv = p[v];
        long delta = 0;

        int flow_u_offset = u * N;
        int flow_v_offset = v * N;
        int dist_pu_offset = pu * N;
        int dist_pv_offset = pv * N;

        if (isSymmetric)
        {
            // Symmetric optimized loop (about twice as fast)
            for (int k = 0; k < u; k++)
            {
                int pk = p[k];
                delta += (long)(flow[flow_u_offset + k] - flow[flow_v_offset + k]) * (dist[dist_pv_offset + pk] - dist[dist_pu_offset + pk]);
            }
            for (int k = u + 1; k < v; k++)
            {
                int pk = p[k];
                delta += (long)(flow[flow_u_offset + k] - flow[flow_v_offset + k]) * (dist[dist_pv_offset + pk] - dist[dist_pu_offset + pk]);
            }
            for (int k = v + 1; k < N; k++)
            {
                int pk = p[k];
                delta += (long)(flow[flow_u_offset + k] - flow[flow_v_offset + k]) * (dist[dist_pv_offset + pk] - dist[dist_pu_offset + pk]);
            }

            delta *= 2;
            delta += (long)(flow[flow_u_offset + u] - flow[flow_v_offset + v]) * (dist[pv * N + pv] - dist[pu * N + pu]);
        }
        else
        {
            // General asymmetric loop
            int flow_k_offset = 0;
            for (int k = 0; k < u; k++)
            {
                int pk = p[k];
                int dist_pk_offset = pk * N;
                delta += (long)(flow[flow_u_offset + k] - flow[flow_v_offset + k]) * (dist[dist_pv_offset + pk] - dist[dist_pu_offset + pk])
                       + (long)(flow[flow_k_offset + u] - flow[flow_k_offset + v]) * (dist[dist_pk_offset + pv] - dist[dist_pk_offset + pu]);
                flow_k_offset += N;
            }
            flow_k_offset = (u + 1) * N;
            for (int k = u + 1; k < v; k++)
            {
                int pk = p[k];
                int dist_pk_offset = pk * N;
                delta += (long)(flow[flow_u_offset + k] - flow[flow_v_offset + k]) * (dist[dist_pv_offset + pk] - dist[dist_pu_offset + pk])
                       + (long)(flow[flow_k_offset + u] - flow[flow_k_offset + v]) * (dist[dist_pk_offset + pv] - dist[dist_pk_offset + pu]);
                flow_k_offset += N;
            }
            flow_k_offset = (v + 1) * N;
            for (int k = v + 1; k < N; k++)
            {
                int pk = p[k];
                int dist_pk_offset = pk * N;
                delta += (long)(flow[flow_u_offset + k] - flow[flow_v_offset + k]) * (dist[dist_pv_offset + pk] - dist[dist_pu_offset + pk])
                       + (long)(flow[flow_k_offset + u] - flow[flow_k_offset + v]) * (dist[dist_pk_offset + pv] - dist[dist_pk_offset + pu]);
                flow_k_offset += N;
            }

            delta += (long)(flow[flow_u_offset + u] - flow[flow_v_offset + v]) * (dist[pv * N + pv] - dist[pu * N + pu])
                   + (long)(flow[flow_u_offset + v] - flow[flow_v_offset + u]) * (dist[pv * N + pu] - dist[pu * N + pv]);
        }

        return delta;
    }
}
