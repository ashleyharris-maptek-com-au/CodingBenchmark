#include <iostream>
#include <vector>
#include <algorithm>
#include <chrono>
#include <random>
#include <numeric>

struct Rectangle {
    int id;
    int w, h;
    int board_idx;
    int x, y;
    int rotated;
    bool packed;
};

struct FreeRect {
    int x, y, w, h;
};

struct SkylineNode {
    int x, y, w;
};

struct Placement {
    int segment_idx;
    int x, y;
    bool rotated;
    int score;
};

struct Solution {
    int num_boards;
    std::vector<int> board_idx;
    std::vector<int> x;
    std::vector<int> y;
    std::vector<int> rotated;
};

std::vector<SkylineNode> skyline;
std::vector<FreeRect> waste_map;

Placement findSkylinePlacement(int rw, int rh, int W, int H) {
    Placement best_p;
    best_p.segment_idx = -1;
    best_p.score = std::numeric_limits<int>::max();
    
    for (size_t i = 0; i < skyline.size(); ++i) {
        // Try normal (not rotated)
        {
            int rx = skyline[i].x;
            if (rx + rw <= W) {
                int ry = 0;
                for (size_t k = i; k < skyline.size(); ++k) {
                    if (skyline[k].x >= rx + rw) break;
                    ry = std::max(ry, skyline[k].y);
                }
                if (ry + rh <= H) {
                    int score = ry + rh;
                    if (score < best_p.score || 
                        (score == best_p.score && ry < best_p.y) || 
                        (score == best_p.score && ry == best_p.y && rx < best_p.x)) {
                        best_p.segment_idx = i;
                        best_p.x = rx;
                        best_p.y = ry;
                        best_p.rotated = false;
                        best_p.score = score;
                    }
                }
            }
        }
        // Try rotated
        {
            int rx = skyline[i].x;
            if (rx + rh <= W) {
                int ry = 0;
                for (size_t k = i; k < skyline.size(); ++k) {
                    if (skyline[k].x >= rx + rh) break;
                    ry = std::max(ry, skyline[k].y);
                }
                if (ry + rw <= H) {
                    int score = ry + rw;
                    if (score < best_p.score || 
                        (score == best_p.score && ry < best_p.y) || 
                        (score == best_p.score && ry == best_p.y && rx < best_p.x)) {
                        best_p.segment_idx = i;
                        best_p.x = rx;
                        best_p.y = ry;
                        best_p.rotated = true;
                        best_p.score = score;
                    }
                }
            }
        }
    }
    return best_p;
}

void updateSkyline(int rx, int ry, int rw, int rh, int min_dim, int min_area) {
    int rx2 = rx + rw;
    int ry2 = ry + rh;
    
    std::vector<SkylineNode> next_skyline;
    next_skyline.reserve(skyline.size() + 2);
    
    bool inserted = false;
    
    for (const auto& node : skyline) {
        int nx1 = node.x;
        int nx2 = node.x + node.w;
        
        if (nx2 <= rx) {
            next_skyline.push_back(node);
        } else if (nx1 >= rx2) {
            if (!inserted) {
                next_skyline.push_back({rx, ry2, rw});
                inserted = true;
            }
            next_skyline.push_back(node);
        } else {
            int overlap_x1 = std::max(nx1, rx);
            int overlap_x2 = std::min(nx2, rx2);
            int waste_h = ry - node.y;
            if (waste_h > 0 && (overlap_x2 - overlap_x1) >= min_dim && waste_h >= min_dim) {
                waste_map.push_back({overlap_x1, node.y, overlap_x2 - overlap_x1, waste_h});
            }
            
            if (nx1 < rx) {
                next_skyline.push_back({nx1, node.y, rx - nx1});
            }
            if (nx2 > rx2) {
                if (!inserted) {
                    next_skyline.push_back({rx, ry2, rw});
                    inserted = true;
                }
                next_skyline.push_back({rx2, node.y, nx2 - rx2});
            }
        }
    }
    
    if (!inserted) {
        next_skyline.push_back({rx, ry2, rw});
    }
    
    skyline.clear();
    for (const auto& node : next_skyline) {
        if (skyline.empty()) {
            skyline.push_back(node);
        } else {
            auto& last = skyline.back();
            if (last.y == node.y) {
                last.w += node.w;
            } else {
                skyline.push_back(node);
            }
        }
    }
}

void solveWithStrategy(const std::vector<Rectangle>& initial_rects, int W, int H, int strategy, Solution& best_sol) {
    int N = initial_rects.size();
    std::vector<Rectangle> rects = initial_rects;
    
    std::vector<int> indices(N);
    std::iota(indices.begin(), indices.end(), 0);
    
    std::sort(indices.begin(), indices.end(), [&](int a, int b) {
        if (strategy == 0) {
            // Max Side Descending
            int ms_a = std::max(rects[a].w, rects[a].h);
            int ms_b = std::max(rects[b].w, rects[b].h);
            if (ms_a != ms_b) return ms_a > ms_b;
            int mi_a = std::min(rects[a].w, rects[a].h);
            int mi_b = std::min(rects[b].w, rects[b].h);
            if (mi_a != mi_b) return mi_a > mi_b;
            return rects[a].id < rects[b].id;
        } else if (strategy == 1) {
            // Area Descending
            int area_a = rects[a].w * rects[a].h;
            int area_b = rects[b].w * rects[b].h;
            if (area_a != area_b) return area_a > area_b;
            return std::max(rects[a].w, rects[a].h) > std::max(rects[b].w, rects[b].h);
        } else if (strategy == 2) {
            // Perimeter Descending
            int p_a = rects[a].w + rects[a].h;
            int p_b = rects[b].w + rects[b].h;
            if (p_a != p_b) return p_a > p_b;
            return rects[a].w * rects[a].h > rects[b].w * rects[b].h;
        } else {
            // Diagonal Descending
            long long diag_a = (long long)rects[a].w * rects[a].w + (long long)rects[a].h * rects[a].h;
            long long diag_b = (long long)rects[b].w * rects[b].w + (long long)rects[b].h * rects[b].h;
            if (diag_a != diag_b) return diag_a > diag_b;
            return rects[a].w * rects[a].h > rects[b].w * rects[b].h;
        }
    });
    
    int board_count = 0;
    std::vector<int> unpacked = indices;
    
    while (!unpacked.empty()) {
        int min_dim = std::numeric_limits<int>::max();
        int min_area = std::numeric_limits<int>::max();
        for (int idx : unpacked) {
            int dim = std::min(rects[idx].w, rects[idx].h);
            min_dim = std::min(min_dim, dim);
            min_area = std::min(min_area, rects[idx].w * rects[idx].h);
        }
        
        skyline = {{0, 0, W}};
        waste_map.clear();
        int max_free_w = W;
        int max_free_h = H;
        
        for (int idx : unpacked) {
            // O(1) Quick Filter
            if (!((rects[idx].w <= max_free_w && rects[idx].h <= max_free_h) ||
                  (rects[idx].h <= max_free_w && rects[idx].w <= max_free_h))) {
                continue;
            }
            
            // 1. Try waste map first
            int best_waste_idx = -1;
            int best_waste_score = std::numeric_limits<int>::max();
            bool best_waste_rot = false;
            
            for (size_t w_idx = 0; w_idx < waste_map.size(); ++w_idx) {
                const auto& F = waste_map[w_idx];
                if (rects[idx].w <= F.w && rects[idx].h <= F.h) {
                    int score = std::min(F.w - rects[idx].w, F.h - rects[idx].h);
                    if (score < best_waste_score) {
                        best_waste_score = score;
                        best_waste_idx = w_idx;
                        best_waste_rot = false;
                    }
                }
                if (rects[idx].h <= F.w && rects[idx].w <= F.h) {
                    int score = std::min(F.w - rects[idx].h, F.h - rects[idx].w);
                    if (score < best_waste_score) {
                        best_waste_score = score;
                        best_waste_idx = w_idx;
                        best_waste_rot = true;
                    }
                }
            }
            
            if (best_waste_idx != -1) {
                auto F = waste_map[best_waste_idx];
                int w = best_waste_rot ? rects[idx].h : rects[idx].w;
                int h = best_waste_rot ? rects[idx].w : rects[idx].h;
                
                rects[idx].x = F.x;
                rects[idx].y = F.y;
                rects[idx].rotated = best_waste_rot ? 1 : 0;
                rects[idx].board_idx = board_count;
                rects[idx].packed = true;
                
                FreeRect r1, r2;
                int leftover_w = F.w - w;
                int leftover_h = F.h - h;
                if (leftover_w > leftover_h) {
                    r1 = {F.x + w, F.y, leftover_w, F.h};
                    r2 = {F.x, F.y + h, w, leftover_h};
                } else {
                    r1 = {F.x, F.y + h, F.w, leftover_h};
                    r2 = {F.x + w, F.y, leftover_w, h};
                }
                
                waste_map[best_waste_idx] = waste_map.back();
                waste_map.pop_back();
                
                if (r1.w >= min_dim && r1.h >= min_dim && r1.w * r1.h >= min_area) waste_map.push_back(r1);
                if (r2.w >= min_dim && r2.h >= min_dim && r2.w * r2.h >= min_area) waste_map.push_back(r2);
                continue;
            }
            
            // 2. Try skyline
            Placement p = findSkylinePlacement(rects[idx].w, rects[idx].h, W, H);
            if (p.segment_idx != -1) {
                rects[idx].x = p.x;
                rects[idx].y = p.y;
                rects[idx].rotated = p.rotated ? 1 : 0;
                rects[idx].board_idx = board_count;
                rects[idx].packed = true;
                
                updateSkyline(p.x, p.y, p.rotated ? rects[idx].h : rects[idx].w, p.rotated ? rects[idx].w : rects[idx].h, min_dim, min_area);
                
                max_free_w = 0;
                max_free_h = 0;
                for (const auto& node : skyline) {
                    max_free_w = std::max(max_free_w, node.w);
                    max_free_h = std::max(max_free_h, H - node.y);
                }
            }
        }
        
        // Filter out packed rectangles
        size_t write_idx = 0;
        for (size_t read_idx = 0; read_idx < unpacked.size(); ++read_idx) {
            if (!rects[unpacked[read_idx]].packed) {
                unpacked[write_idx++] = unpacked[read_idx];
            }
        }
        unpacked.resize(write_idx);
        board_count++;
    }
    
    // Assign any remaining unpacked rectangles to new boards
    for (int i = 0; i < N; ++i) {
        if (!rects[i].packed) {
            rects[i].board_idx = board_count++;
            rects[i].x = 0;
            rects[i].y = 0;
            rects[i].rotated = 0;
            rects[i].packed = true;
        }
    }
    
    if (board_count < best_sol.num_boards) {
        best_sol.num_boards = board_count;
        for (int i = 0; i < N; ++i) {
            best_sol.board_idx[i] = rects[i].board_idx;
            best_sol.x[i] = rects[i].x;
            best_sol.y[i] = rects[i].y;
            best_sol.rotated[i] = rects[i].rotated;
        }
    }
}

int main() {
    // Fast I/O
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    int N, W, H;
    if (!(std::cin >> N >> W >> H)) return 0;
    
    std::vector<Rectangle> rects(N);
    for (int i = 0; i < N; ++i) {
        rects[i].id = i;
        std::cin >> rects[i].w >> rects[i].h;
        rects[i].packed = false;
        rects[i].board_idx = -1;
        rects[i].x = -1;
        rects[i].y = -1;
        rects[i].rotated = 0;
    }
    
    Solution best_sol;
    best_sol.num_boards = std::numeric_limits<int>::max();
    best_sol.board_idx.resize(N);
    best_sol.x.resize(N);
    best_sol.y.resize(N);
    best_sol.rotated.resize(N);
    
    // Dynamically select how many strategies to run based on N
    int max_strategies = 4;
    if (N >= 20000) {
        max_strategies = 3;
    }
    
    for (int strategy = 0; strategy < max_strategies; ++strategy) {
        // If N >= 20000, strategy 2 (Perimeter) is skipped to optimize time
        int actual_strategy = strategy;
        if (N >= 20000 && strategy == 2) {
            actual_strategy = 3; // Diagonal Descending
        }
        solveWithStrategy(rects, W, H, actual_strategy, best_sol);
    }
    
    std::cout << best_sol.num_boards << "\n";
    for (int i = 0; i < N; ++i) {
        std::cout << best_sol.board_idx[i] << " "
                  << best_sol.x[i] << " "
                  << best_sol.y[i] << " "
                  << best_sol.rotated[i] << "\n";
    }
    
    return 0;
}
