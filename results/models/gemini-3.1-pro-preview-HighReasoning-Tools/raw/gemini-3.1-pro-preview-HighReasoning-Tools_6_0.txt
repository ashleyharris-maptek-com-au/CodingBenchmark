using System;
using System.IO;
using System.Collections.Generic;
using System.Globalization;
using System.Diagnostics;

class Program
{
    static void Main(string[] args)
    {
        // Force invariant culture to ensure correct float parsing with dots
        System.Threading.Thread.CurrentThread.CurrentCulture = CultureInfo.InvariantCulture;

        // Read all inputs from standard input
        string input = Console.In.ReadToEnd();
        string[] tokens = input.Split(new char[] { ' ', '\r', '\n', '\t' }, StringSplitOptions.RemoveEmptyEntries);

        if (tokens.Length == 0) return;

        int N = int.Parse(tokens[0]);
        double mu = double.Parse(tokens[1]);

        double[] startOrbit = new double[6];
        for (int i = 0; i < 6; i++)
        {
            startOrbit[i] = double.Parse(tokens[2 + i]);
        }

        double[][] stationOrbits = new double[N][];
        int idx = 8;
        for (int i = 0; i < N; i++)
        {
            stationOrbits[i] = new double[6];
            for (int j = 0; j < 6; j++)
            {
                stationOrbits[i][j] = double.Parse(tokens[idx++]);
            }
        }

        // Compute symmetric distance matrix
        // dist[i, j] is the delta-V cost between station i and station j.
        // dist[N, i] and dist[i, N] is the cost between startOrbit and station i.
        // dist[N, N] is 0.
        double[,] dist = new double[N + 1, N + 1];
        for (int i = 0; i < N; i++)
        {
            dist[N, i] = EstimateTransferDeltaV(startOrbit, stationOrbits[i], mu);
            dist[i, N] = dist[N, i];
        }
        dist[N, N] = 0;

        for (int i = 0; i < N; i++)
        {
            dist[i, i] = 0;
            for (int j = i + 1; j < N; j++)
            {
                double d = EstimateTransferDeltaV(stationOrbits[i], stationOrbits[j], mu);
                dist[i, j] = d;
                dist[j, i] = d;
            }
        }

        int[] bestPath;
        if (N <= 20)
        {
            bestPath = SolveDP(dist, N);
        }
        else
        {
            // For N > 20, solve using greedy heuristic + 2-opt Hill Climbing + Multi-Start Simulated Annealing
            int[] initialPath = SolveGreedy(dist, N);
            
            int[] path = (int[])initialPath.Clone();
            Solve2OptHillClimbing(path, dist, N);

            bestPath = (int[])path.Clone();
            double bestCost = EvaluatePathCost(bestPath, dist, N);

            // Time budget: 22 seconds
            // Multi-start Simulated Annealing to avoid getting trapped in local minima
            int numRestarts = 3;
            double timePerRestart = 22.0 / numRestarts;

            for (int r = 0; r < numRestarts; r++)
            {
                int seed = 42 + r;
                int[] saPath = SolveSA(dist, N, path, timePerRestart, seed);
                double saCost = EvaluatePathCost(saPath, dist, N);
                if (saCost < bestCost)
                {
                    bestCost = saCost;
                    Array.Copy(saPath, bestPath, N);
                }
            }
        }

        // Output the result
        Console.WriteLine(string.Join(" ", bestPath));
    }

    static double EstimateTransferDeltaV(double[] orbit1, double[] orbit2, double mu)
    {
        double r1_x = orbit1[0], r1_y = orbit1[1], r1_z = orbit1[2];
        double v1_x = orbit1[3], v1_y = orbit1[4], v1_z = orbit1[5];

        double r2_x = orbit2[0], r2_y = orbit2[1], r2_z = orbit2[2];
        double v2_x = orbit2[3], v2_y = orbit2[4], v2_z = orbit2[5];

        double r1 = Math.Sqrt(r1_x * r1_x + r1_y * r1_y + r1_z * r1_z);
        double r2 = Math.Sqrt(r2_x * r2_x + r2_y * r2_y + r2_z * r2_z);
        double v1 = Math.Sqrt(v1_x * v1_x + v1_y * v1_y + v1_z * v1_z);
        double v2 = Math.Sqrt(v2_x * v2_x + v2_y * v2_y + v2_z * v2_z);

        // Angular momentum h = r x v
        double h1_x = r1_y * v1_z - r1_z * v1_y;
        double h1_y = r1_z * v1_x - r1_x * v1_z;
        double h1_z = r1_x * v1_y - r1_y * v1_x;

        double h2_x = r2_y * v2_z - r2_z * v2_y;
        double h2_y = r2_z * v2_x - r2_x * v2_z;
        double h2_z = r2_x * v2_y - r2_y * v2_x;

        double h1_mag = Math.Sqrt(h1_x * h1_x + h1_y * h1_y + h1_z * h1_z);
        double h2_mag = Math.Sqrt(h2_x * h2_x + h2_y * h2_y + h2_z * h2_z);

        double plane_change_angle = 0;
        if (h1_mag > 0 && h2_mag > 0)
        {
            double h1_nx = h1_x / h1_mag;
            double h1_ny = h1_y / h1_mag;
            double h1_nz = h1_z / h1_mag;

            double h2_nx = h2_x / h2_mag;
            double h2_ny = h2_y / h2_mag;
            double h2_nz = h2_z / h2_mag;

            double cos_plane_change = h1_nx * h2_nx + h1_ny * h2_ny + h1_nz * h2_nz;
            if (cos_plane_change < -1.0) cos_plane_change = -1.0;
            if (cos_plane_change > 1.0) cos_plane_change = 1.0;
            plane_change_angle = Math.Acos(cos_plane_change);
        }

        // Hohmann transfer for in-plane component
        double a_transfer = (r1 + r2) / 2.0;
        double in_plane_dv = 0;
        if (a_transfer > 0)
        {
            double v_t1 = Math.Sqrt(mu * (2.0 / r1 - 1.0 / a_transfer));
            double v_t2 = Math.Sqrt(mu * (2.0 / r2 - 1.0 / a_transfer));
            double v_c1 = Math.Sqrt(mu / r1);
            double v_c2 = Math.Sqrt(mu / r2);

            double dv1 = Math.Abs(v_t1 - v_c1);
            double dv2 = Math.Abs(v_c2 - v_t2);
            in_plane_dv = dv1 + dv2;
        }
        else
        {
            in_plane_dv = Math.Abs(v2 - v1);
        }

        // Plane change delta-V
        double v_at_node = Math.Sqrt(mu * (2.0 / r1 + 2.0 / r2) / 2.0);
        double plane_change_dv = 2.0 * v_at_node * Math.Sin(plane_change_angle / 2.0);

        double total_dv = in_plane_dv + plane_change_dv;
        double vel_magnitude_diff = Math.Abs(v2 - v1);
        double velocity_penalty = vel_magnitude_diff * 0.1;

        return total_dv + velocity_penalty;
    }

    static int[] SolveDP(double[,] dist, int N)
    {
        int numStates = 1 << N;
        float[,] dp = new float[numStates, N];
        sbyte[,] parent = new sbyte[numStates, N];

        for (int mask = 0; mask < numStates; mask++)
        {
            for (int u = 0; u < N; u++)
            {
                dp[mask, u] = float.MaxValue;
                parent[mask, u] = -1;
            }
        }

        for (int u = 0; u < N; u++)
        {
            dp[1 << u, u] = (float)dist[N, u];
        }

        for (int mask = 1; mask < numStates; mask++)
        {
            for (int u = 0; u < N; u++)
            {
                float currentCost = dp[mask, u];
                if (currentCost == float.MaxValue) continue;

                for (int v = 0; v < N; v++)
                {
                    if ((mask & (1 << v)) == 0)
                    {
                        float nextCost = currentCost + (float)dist[u, v];
                        int nextMask = mask | (1 << v);
                        if (nextCost < dp[nextMask, v])
                        {
                            dp[nextMask, v] = nextCost;
                            parent[nextMask, v] = (sbyte)u;
                        }
                    }
                }
            }
        }

        int fullMask = numStates - 1;
        float bestCost = float.MaxValue;
        int bestU = -1;
        for (int u = 0; u < N; u++)
        {
            if (dp[fullMask, u] < bestCost)
            {
                bestCost = dp[fullMask, u];
                bestU = u;
            }
        }

        int[] path = new int[N];
        int currU = bestU;
        int currMask = fullMask;
        for (int step = N - 1; step >= 0; step--)
        {
            path[step] = currU;
            int prevU = parent[currMask, currU];
            currMask ^= (1 << currU);
            currU = prevU;
        }

        return path;
    }

    static int[] SolveGreedy(double[,] dist, int N)
    {
        int[] path = new int[N];
        bool[] visited = new bool[N];
        int current = N;
        for (int step = 0; step < N; step++)
        {
            int next = -1;
            double bestDist = double.MaxValue;
            for (int i = 0; i < N; i++)
            {
                if (!visited[i])
                {
                    double d = dist[current, i];
                    if (d < bestDist)
                    {
                        bestDist = d;
                        next = i;
                    }
                }
            }
            visited[next] = true;
            path[step] = next;
            current = next;
        }
        return path;
    }

    static void Solve2OptHillClimbing(int[] path, double[,] dist, int N)
    {
        bool improved = true;
        while (improved)
        {
            improved = false;
            for (int i = 0; i < N; i++)
            {
                for (int j = i + 1; j < N; j++)
                {
                    double delta = Get2OptDelta(path, dist, N, i, j);
                    if (delta < -1e-9)
                    {
                        int l = i, r = j;
                        while (l < r)
                        {
                            int tmp = path[l];
                            path[l] = path[r];
                            path[r] = tmp;
                            l++;
                            r--;
                        }
                        improved = true;
                    }
                }
            }
        }
    }

    static int[] SolveSA(double[,] dist, int N, int[] initialPath, double timeLimitSeconds, int seed)
    {
        Random rng = new Random(seed);
        int[] currentPath = (int[])initialPath.Clone();
        double currentCost = EvaluatePathCost(currentPath, dist, N);

        int[] bestPath = (int[])currentPath.Clone();
        double bestCost = currentCost;

        Stopwatch sw = Stopwatch.StartNew();
        double timeLimitMs = timeLimitSeconds * 1000.0;

        // Estimate initial temperature T_start
        double sumPosDelta = 0;
        int countPosDelta = 0;
        for (int k = 0; k < 1000; k++)
        {
            int i = rng.Next(N);
            int j = rng.Next(N);
            if (i > j) { int tmp = i; i = j; j = tmp; }
            if (i == j) continue;

            double delta = Get2OptDelta(currentPath, dist, N, i, j);
            if (delta > 0)
            {
                sumPosDelta += delta;
                countPosDelta++;
            }
        }
        double avgDelta = countPosDelta > 0 ? (sumPosDelta / countPosDelta) : 10.0;
        double T_start = avgDelta * 0.5;
        double T_end = 0.0001;

        long iterations = 0;
        double elapsedMs = 0;

        while (true)
        {
            if (iterations % 16384 == 0)
            {
                elapsedMs = sw.Elapsed.TotalMilliseconds;
                if (elapsedMs >= timeLimitMs)
                    break;
            }

            iterations++;
            double progress = elapsedMs / timeLimitMs;
            double T = T_start * Math.Pow(T_end / T_start, progress);

            int i_idx = rng.Next(N);
            int j_idx = rng.Next(N);
            if (i_idx > j_idx) { int tmp = i_idx; i_idx = j_idx; j_idx = tmp; }
            if (i_idx == j_idx) continue;

            double delta = Get2OptDelta(currentPath, dist, N, i_idx, j_idx);

            bool accept = false;
            if (delta <= 0)
            {
                accept = true;
            }
            else
            {
                double p = Math.Exp(-delta / T);
                if (rng.NextDouble() < p)
                {
                    accept = true;
                }
            }

            if (accept)
            {
                int l = i_idx, r = j_idx;
                while (l < r)
                {
                    int tmp = currentPath[l];
                    currentPath[l] = currentPath[r];
                    currentPath[r] = tmp;
                    l++;
                    r--;
                }
                currentCost += delta;

                if (currentCost < bestCost)
                {
                    bestCost = currentCost;
                    Array.Copy(currentPath, bestPath, N);
                }
            }
        }

        return bestPath;
    }

    static double Get2OptDelta(int[] path, double[,] dist, int N, int i, int j)
    {
        if (i > 0 && j < N - 1)
        {
            return dist[path[i - 1], path[j]] + dist[path[i], path[j + 1]] 
                 - dist[path[i - 1], path[i]] - dist[path[j], path[j + 1]];
        }
        else if (i == 0 && j < N - 1)
        {
            return dist[N, path[j]] + dist[path[0], path[j + 1]] 
                 - dist[N, path[0]] - dist[path[j], path[j + 1]];
        }
        else if (i > 0 && j == N - 1)
        {
            return dist[path[i - 1], path[j]] 
                 - dist[path[i - 1], path[i]];
        }
        else // i == 0 && j == N - 1
        {
            return dist[N, path[j]] 
                 - dist[N, path[0]];
        }
    }

    static double EvaluatePathCost(int[] path, double[,] dist, int N)
    {
        double cost = dist[N, path[0]];
        for (int i = 0; i < N - 1; i++)
        {
            cost += dist[path[i], path[i + 1]];
        }
        return cost;
    }
}
