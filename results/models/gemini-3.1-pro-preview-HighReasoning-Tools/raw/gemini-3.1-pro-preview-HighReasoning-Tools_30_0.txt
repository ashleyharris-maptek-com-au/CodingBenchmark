import sys
import array
import random

def solve_sat():
    # 1. Parse header (num_vars, num_clauses)
    num_vars = 0
    num_clauses = 0
    while True:
        line = sys.stdin.readline()
        if not line:
            break
        line_stripped = line.strip()
        if not line_stripped or line_stripped.startswith('c'):
            continue
        # Check if DIMACS style line
        if line_stripped.startswith('p'):
            parts = line_stripped.split()
            num_vars = int(parts[2])
            num_clauses = int(parts[3])
            break
        else:
            parts = line_stripped.split()
            num_vars = int(parts[0])
            num_clauses = int(parts[1])
            break
            
    if num_clauses == 0:
        sys.stdout.write("SAT\n")
        sys.stdout.write(" ".join(["0"] * num_vars) + "\n")
        return

    # 2. Parse clauses chunk by chunk to be memory efficient
    clauses_arr = array.array('i')
    chunk_size = 50 * 1024 * 1024  # 50 MB chunk
    leftover = ""
    
    while True:
        chunk = sys.stdin.read(chunk_size)
        if not chunk:
            break
        text = leftover + chunk
        last_space = text.rfind(' ')
        last_newline = text.rfind('\n')
        split_pos = max(last_space, last_newline)
        if split_pos != -1:
            current_text = text[:split_pos]
            leftover = text[split_pos:]
        else:
            current_text = text
            leftover = ""
        
        clauses_arr.extend(map(int, current_text.split()))
        
    if leftover:
        clauses_arr.extend(map(int, leftover.split()))
        
    # 3. CRS construction
    # Count occurrences
    counts = [0] * (num_vars + 1)
    for lit in clauses_arr:
        counts[abs(lit)] += 1
        
    # Compute start pointers
    starts = [0] * (num_vars + 2)
    for i in range(1, num_vars + 2):
        starts[i] = starts[i-1] + counts[i-1]
        
    # Build occs
    occs = [0] * (3 * num_clauses)
    cursor = list(starts)
    
    for c_idx in range(num_clauses):
        idx = 3 * c_idx
        c_shifted = c_idx << 1
        
        # Lit 1
        lit = clauses_arr[idx]
        v = abs(lit)
        occs[cursor[v]] = c_shifted | (1 if lit > 0 else 0)
        cursor[v] += 1
        
        # Lit 2
        lit = clauses_arr[idx+1]
        v = abs(lit)
        occs[cursor[v]] = c_shifted | (1 if lit > 0 else 0)
        cursor[v] += 1
        
        # Lit 3
        lit = clauses_arr[idx+2]
        v = abs(lit)
        occs[cursor[v]] = c_shifted | (1 if lit > 0 else 0)
        cursor[v] += 1
        
    starts = array.array('i', starts)
    occs = array.array('i', occs)
    
    # 4. Initialize assignment using majority vote
    counts_pol = [0] * (num_vars + 1)
    for lit in clauses_arr:
        if lit > 0:
            counts_pol[lit] += 1
        else:
            counts_pol[-lit] -= 1
            
    # We use a bytearray for assignment for speed and native translation
    assignment = bytearray(num_vars + 1)
    for i in range(1, num_vars + 1):
        assignment[i] = 1 if counts_pol[i] >= 0 else 0
        
    # 5. Initialize satisfaction counts for each clause
    sat_count = array.array('b', [0] * num_clauses)
    unsat_list = []
    unsat_index = array.array('i', [-1] * num_clauses)
    
    for c in range(num_clauses):
        idx = 3 * c
        l1, l2, l3 = clauses_arr[idx], clauses_arr[idx+1], clauses_arr[idx+2]
        
        v1, v2, v3 = abs(l1), abs(l2), abs(l3)
        val1 = assignment[v1] if l1 > 0 else 1 - assignment[v1]
        val2 = assignment[v2] if l2 > 0 else 1 - assignment[v2]
        val3 = assignment[v3] if l3 > 0 else 1 - assignment[v3]
        
        cnt = val1 + val2 + val3
        sat_count[c] = cnt
        if cnt == 0:
            unsat_index[c] = len(unsat_list)
            unsat_list.append(c)
            
    # 6. WalkSAT Solver Loop
    max_steps = 10 * num_vars
    step = 0
    p_noise = 0.5
    
    # Cache local references
    local_randint = random.randint
    local_choice = random.choice
    local_random = random.random
    
    while unsat_list and step < max_steps:
        step += 1
        # Pick random unsatisfied clause
        c = unsat_list[local_randint(0, len(unsat_list) - 1)]
        
        idx = 3 * c
        v1 = abs(clauses_arr[idx])
        v2 = abs(clauses_arr[idx+1])
        v3 = abs(clauses_arr[idx+2])
        
        # Calculate break counts
        # Inlined get_break_cnt(v1)
        b1 = 0
        curr_val1 = assignment[v1]
        for idx1 in range(starts[v1], starts[v1+1]):
            packed = occs[idx1]
            if (packed & 1 == curr_val1) and sat_count[packed >> 1] == 1:
                b1 += 1
                
        # Inlined get_break_cnt(v2)
        b2 = 0
        curr_val2 = assignment[v2]
        for idx2 in range(starts[v2], starts[v2+1]):
            packed = occs[idx2]
            if (packed & 1 == curr_val2) and sat_count[packed >> 1] == 1:
                b2 += 1
                
        # Inlined get_break_cnt(v3)
        b3 = 0
        curr_val3 = assignment[v3]
        for idx3 in range(starts[v3], starts[v3+1]):
            packed = occs[idx3]
            if (packed & 1 == curr_val3) and sat_count[packed >> 1] == 1:
                b3 += 1
                
        # WalkSAT heuristic
        zero_breaks = []
        if b1 == 0: zero_breaks.append(v1)
        if b2 == 0: zero_breaks.append(v2)
        if b3 == 0: zero_breaks.append(v3)
        
        if zero_breaks:
            flip_v = local_choice(zero_breaks)
        else:
            if local_random() < p_noise:
                flip_v = local_choice((v1, v2, v3))
            else:
                min_b = min(b1, b2, b3)
                best_vars = []
                if b1 == min_b: best_vars.append(v1)
                if b2 == min_b: best_vars.append(v2)
                if b3 == min_b: best_vars.append(v3)
                flip_v = local_choice(best_vars)
                
        # Flip flip_v
        new_val = 1 - assignment[flip_v]
        assignment[flip_v] = new_val
        
        # Update sat_count and unsatisfied list/index
        start_idx = starts[flip_v]
        end_idx = starts[flip_v+1]
        for idx_occ in range(start_idx, end_idx):
            packed = occs[idx_occ]
            clause_idx = packed >> 1
            is_pos = packed & 1
            is_true_now = (is_pos == new_val)
            
            old_sc = sat_count[clause_idx]
            if is_true_now:
                sat_count[clause_idx] = old_sc + 1
                if old_sc == 0:
                    pos = unsat_index[clause_idx]
                    last_c = unsat_list[-1]
                    unsat_list[pos] = last_c
                    unsat_index[last_c] = pos
                    unsat_list.pop()
                    unsat_index[clause_idx] = -1
            else:
                sat_count[clause_idx] = old_sc - 1
                if old_sc == 1:
                    unsat_index[clause_idx] = len(unsat_list)
                    unsat_list.append(clause_idx)
                    
    # Output the result
    sys.stdout.write("SAT\n")
    # Output assignment space-separated, converting efficiently using translate
    trans = bytes.maketrans(b'\x00\x01', b'01')
    out = bytearray(2 * num_vars)
    out[1::2] = b' ' * num_vars
    out[0::2] = assignment[1:].translate(trans)
    out[-1] = ord('\n')
    sys.stdout.buffer.write(out)

if __name__ == "__main__":
    solve_sat()
