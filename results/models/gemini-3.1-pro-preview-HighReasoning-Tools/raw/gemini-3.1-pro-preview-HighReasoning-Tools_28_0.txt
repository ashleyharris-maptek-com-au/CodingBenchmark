struct FastParser<'a> {
    bytes: &'a [u8],
    pos: usize,
}

impl<'a> FastParser<'a> {
    fn new(bytes: &'a [u8]) -> Self {
        Self { bytes, pos: 0 }
    }

    #[inline(always)]
    fn next_usize(&mut self) -> Option<usize> {
        while self.pos < self.bytes.len() && self.bytes[self.pos] <= b' ' {
            self.pos += 1;
        }
        if self.pos >= self.bytes.len() {
            return None;
        }
        let mut val = 0;
        while self.pos < self.bytes.len() && self.bytes[self.pos] > b' ' {
            val = val * 10 + (self.bytes[self.pos] - b'0') as usize;
            self.pos += 1;
        }
        Some(val)
    }
}

struct FastRng {
    state: u64,
}

impl FastRng {
    fn new(seed: u64) -> Self {
        let mut seed = seed;
        if seed == 0 {
            seed = 1;
        }
        Self { state: seed }
    }

    #[inline(always)]
    fn next(&mut self) -> u64 {
        let mut x = self.state;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.state = x;
        x
    }
}

struct Graph {
    num_vertices: usize,
    num_edges: usize,
    edges: Vec<(usize, usize)>,
    v_beg_idx: Vec<usize>,
    v_edges: Vec<usize>,
    v_adj: Vec<usize>,
    v_degree: Vec<usize>,
}

struct SharedState {
    global_best_size: std::sync::atomic::AtomicUsize,
    best_cover: std::sync::Mutex<Vec<usize>>,
    stop_flag: std::sync::atomic::AtomicBool,
}

struct Solver {
    v_num: usize,
    e_num: usize,
    graph: std::sync::Arc<Graph>,
    v_in_c: Vec<bool>,
    dscore: Vec<i32>,
    time_stamp: Vec<u64>,
    remove_cand: Vec<usize>,
    index_in_remove_cand: Vec<usize>,
    remove_cand_size: usize,
    uncov_stack: Vec<usize>,
    index_in_uncov_stack: Vec<usize>,
    uncov_stack_fill_pointer: usize,
    c_size: usize,
    rng: FastRng,
}

impl Solver {
    fn new(graph: std::sync::Arc<Graph>, seed: u64) -> Self {
        let v_num = graph.num_vertices;
        let e_num = graph.num_edges;
        Self {
            v_num,
            e_num,
            graph,
            v_in_c: vec![false; v_num],
            dscore: vec![0; v_num],
            time_stamp: vec![0; v_num],
            remove_cand: vec![0; v_num],
            index_in_remove_cand: vec![0; v_num],
            remove_cand_size: 0,
            uncov_stack: vec![0; e_num],
            index_in_uncov_stack: vec![0; e_num],
            uncov_stack_fill_pointer: 0,
            c_size: 0,
            rng: FastRng::new(seed),
        }
    }

    fn init_sol(&mut self) {
        self.c_size = 0;
        for e in 0..self.e_num {
            let (v1, v2) = self.graph.edges[e];
            if !self.v_in_c[v1] && !self.v_in_c[v2] {
                if self.graph.v_degree[v1] > self.graph.v_degree[v2] {
                    self.v_in_c[v1] = true;
                } else {
                    self.v_in_c[v2] = true;
                }
                self.c_size += 1;
            }
        }

        for e in 0..self.e_num {
            let (v1, v2) = self.graph.edges[e];
            let in_1 = self.v_in_c[v1];
            let in_2 = self.v_in_c[v2];
            if in_1 && !in_2 {
                self.dscore[v1] -= 1;
            } else if in_2 && !in_1 {
                self.dscore[v2] -= 1;
            }
        }

        for v in 0..self.v_num {
            if self.v_in_c[v] && self.dscore[v] == 0 {
                self.remove(v);
                self.c_size -= 1;
            }
        }

        self.reset_remove_cand();
    }

    fn reset_remove_cand(&mut self) {
        let mut j = 0;
        for v in 0..self.v_num {
            if self.v_in_c[v] {
                self.remove_cand[j] = v;
                self.index_in_remove_cand[v] = j;
                j += 1;
            } else {
                self.index_in_remove_cand[v] = 0;
            }
        }
        self.remove_cand_size = j;
    }

    fn update_target_size(&mut self) {
        self.c_size -= 1;

        let mut best_remove_v = self.remove_cand[0];

        if self.dscore[best_remove_v] != 0 {
            for i in 0..(self.remove_cand_size - 1) {
                let v = self.remove_cand[i];

                if self.dscore[v] == 0 {
                    best_remove_v = v;
                    break;
                }

                if self.dscore[v] > self.dscore[best_remove_v] {
                    best_remove_v = v;
                }
            }
        }

        self.remove(best_remove_v);

        self.remove_cand_size -= 1;
        let last_remove_cand_v = self.remove_cand[self.remove_cand_size];
        let index = self.index_in_remove_cand[best_remove_v];
        self.remove_cand[index] = last_remove_cand_v;
        self.index_in_remove_cand[last_remove_cand_v] = index;
    }

    fn choose_remove_v(&mut self) -> usize {
        let cand_count = 50;
        let r = self.rng.next() as usize;
        let mut best_v = self.remove_cand[r % self.remove_cand_size];

        for _ in 0..(cand_count - 1) {
            let r_idx = self.rng.next() as usize;
            let v = self.remove_cand[r_idx % self.remove_cand_size];

            if self.dscore[v] < self.dscore[best_v] {
                continue;
            } else if self.dscore[v] > self.dscore[best_v] {
                best_v = v;
            } else if self.time_stamp[v] < self.time_stamp[best_v] {
                best_v = v;
            }
        }
        best_v
    }

    #[inline(always)]
    fn uncover(&mut self, e: usize) {
        self.index_in_uncov_stack[e] = self.uncov_stack_fill_pointer;
        self.uncov_stack[self.uncov_stack_fill_pointer] = e;
        self.uncov_stack_fill_pointer += 1;
    }

    #[inline(always)]
    fn cover(&mut self, e: usize) {
        self.uncov_stack_fill_pointer -= 1;
        let last_uncov_edge = self.uncov_stack[self.uncov_stack_fill_pointer];
        let index = self.index_in_uncov_stack[e];
        self.uncov_stack[index] = last_uncov_edge;
        self.index_in_uncov_stack[last_uncov_edge] = index;
    }

    fn add(&mut self, v: usize) {
        self.v_in_c[v] = true;
        self.dscore[v] = -self.dscore[v];

        let beg = self.graph.v_beg_idx[v];
        let edge_count = self.graph.v_degree[v];

        for i in 0..edge_count {
            let e = self.graph.v_edges[beg + i];
            let n = self.graph.v_adj[beg + i];

            if !self.v_in_c[n] {
                self.dscore[n] -= 1;
                self.cover(e);
            } else {
                self.dscore[n] += 1;
            }
        }
    }

    fn remove(&mut self, v: usize) {
        self.v_in_c[v] = false;
        self.dscore[v] = -self.dscore[v];

        let beg = self.graph.v_beg_idx[v];
        let edge_count = self.graph.v_degree[v];

        for i in 0..edge_count {
            let e = self.graph.v_edges[beg + i];
            let n = self.graph.v_adj[beg + i];

            if !self.v_in_c[n] {
                self.dscore[n] += 1;
                self.uncover(e);
            } else {
                self.dscore[n] -= 1;
            }
        }
    }

    fn cover_ls(&mut self, shared: &SharedState, start_time: &std::time::Instant, max_stagnation: u64) {
        let mut step: u64 = 1;
        let mut steps_since_improvement: u64 = 0;
        let try_step = 10000;

        let mut local_best_size = self.c_size;

        loop {
            if self.uncov_stack_fill_pointer == 0 {
                let current_size = self.c_size;

                if current_size < local_best_size {
                    local_best_size = current_size;
                    steps_since_improvement = 0;
                }

                let mut global_best = shared.global_best_size.load(std::sync::atomic::Ordering::Relaxed);
                while current_size < global_best {
                    match shared.global_best_size.compare_exchange_weak(
                        global_best,
                        current_size,
                        std::sync::atomic::Ordering::SeqCst,
                        std::sync::atomic::Ordering::Relaxed,
                    ) {
                        Ok(_) => {
                            let mut lock = shared.best_cover.lock().unwrap();
                            if lock.len() == 0 || current_size < lock.len() {
                                lock.clear();
                                for v in 0..self.v_num {
                                    if self.v_in_c[v] {
                                        lock.push(v);
                                    }
                                }
                            }
                            break;
                        }
                        Err(actual) => {
                            global_best = actual;
                        }
                    }
                }

                if shared.stop_flag.load(std::sync::atomic::Ordering::Relaxed) {
                    return;
                }

                self.update_target_size();
                continue;
            }

            if step % try_step == 0 {
                if shared.stop_flag.load(std::sync::atomic::Ordering::Relaxed) {
                    return;
                }
                if start_time.elapsed().as_secs_f64() > 4.5 {
                    shared.stop_flag.store(true, std::sync::atomic::Ordering::Relaxed);
                    return;
                }
                if steps_since_improvement > max_stagnation {
                    return;
                }
            }

            let remove_v = self.choose_remove_v();
            self.remove(remove_v);

            let r_edge_idx = (self.rng.next() as usize) % self.uncov_stack_fill_pointer;
            let e = self.uncov_stack[r_edge_idx];
            let (v1, v2) = self.graph.edges[e];

            let add_v = if self.dscore[v1] > self.dscore[v2] ||
                           (self.dscore[v1] == self.dscore[v2] && self.time_stamp[v1] < self.time_stamp[v2]) {
                v1
            } else {
                v2
            };

            self.add(add_v);

            let index = self.index_in_remove_cand[remove_v];
            self.index_in_remove_cand[remove_v] = 0;

            self.remove_cand[index] = add_v;
            self.index_in_remove_cand[add_v] = index;

            self.time_stamp[add_v] = step;
            self.time_stamp[remove_v] = step;

            step += 1;
            steps_since_improvement += 1;
        }
    }
}

fn main() {
    let start_time = std::time::Instant::now();

    let mut input_bytes = Vec::new();
    std::io::Read::read_to_end(&mut std::io::stdin(), &mut input_bytes).unwrap();
    let mut parser = FastParser::new(&input_bytes);

    let num_vertices = match parser.next_usize() {
        Some(v) => v,
        None => return,
    };
    let num_edges = parser.next_usize().unwrap();

    let mut edges = Vec::with_capacity(num_edges);
    let mut v_degree = vec![0; num_vertices];
    for _ in 0..num_edges {
        let u = parser.next_usize().unwrap();
        let v = parser.next_usize().unwrap();
        edges.push((u, v));
        v_degree[u] += 1;
        v_degree[v] += 1;
    }

    let mut v_beg_idx = Vec::with_capacity(num_vertices + 1);
    v_beg_idx.push(0);
    let mut acc = 0;
    for &deg in &v_degree {
        acc += deg;
        v_beg_idx.push(acc);
    }

    let total_deg = v_beg_idx[num_vertices];
    let mut v_edges = vec![0; total_deg];
    let mut v_adj = vec![0; total_deg];

    let mut v_degree_tmp = vec![0; num_vertices];
    for (e, &(v1, v2)) in edges.iter().enumerate() {
        v_edges[v_beg_idx[v1] + v_degree_tmp[v1]] = e;
        v_edges[v_beg_idx[v2] + v_degree_tmp[v2]] = e;

        v_adj[v_beg_idx[v1] + v_degree_tmp[v1]] = v2;
        v_adj[v_beg_idx[v2] + v_degree_tmp[v2]] = v1;

        v_degree_tmp[v1] += 1;
        v_degree_tmp[v2] += 1;
    }

    let graph = std::sync::Arc::new(Graph {
        num_vertices,
        num_edges,
        edges,
        v_beg_idx,
        v_edges,
        v_adj,
        v_degree,
    });

    let shared = std::sync::Arc::new(SharedState {
        global_best_size: std::sync::atomic::AtomicUsize::new(num_vertices),
        best_cover: std::sync::Mutex::new(Vec::new()),
        stop_flag: std::sync::atomic::AtomicBool::new(false),
    });

    let num_threads = 32;

    let max_stagnation = std::cmp::max(5_000_000 / num_threads as u64, (500 * num_vertices as u64) / num_threads as u64);

    let mut handles = Vec::new();

    for i in 0..num_threads {
        let graph_clone = graph.clone();
        let shared_clone = shared.clone();
        let seed = 123456789 + i as u64 * 987654321;

        let handle = std::thread::spawn(move || {
            let mut solver = Solver::new(graph_clone, seed);
            solver.init_sol();

            let current_size = solver.c_size;
            let mut global_best = shared_clone.global_best_size.load(std::sync::atomic::Ordering::Relaxed);
            while current_size < global_best {
                match shared_clone.global_best_size.compare_exchange_weak(
                    global_best,
                    current_size,
                    std::sync::atomic::Ordering::SeqCst,
                    std::sync::atomic::Ordering::Relaxed,
                ) {
                    Ok(_) => {
                        let mut lock = shared_clone.best_cover.lock().unwrap();
                        if lock.len() == 0 || current_size < lock.len() {
                            lock.clear();
                            for v in 0..solver.v_num {
                                if solver.v_in_c[v] {
                                    lock.push(v);
                                }
                            }
                        }
                        break;
                    }
                    Err(actual) => {
                        global_best = actual;
                    }
                }
            }

            solver.cover_ls(&shared_clone, &start_time, max_stagnation);
        });
        handles.push(handle);
    }

    for handle in handles {
        let _ = handle.join();
    }

    let final_cover = shared.best_cover.lock().unwrap().clone();
    println!("{}", final_cover.len());
    let mut first = true;
    for &v in &final_cover {
        if !first {
            print!(" ");
        }
        first = false;
        print!("{}", v);
    }
    println!();
}
