use std::io::{self, BufRead};

struct Lcg {
    state: u64,
}

impl Lcg {
    fn new(seed: u64) -> Self {
        Self { state: if seed == 0 { 1 } else { seed } }
    }
    fn next(&mut self) -> u64 {
        self.state ^= self.state << 13;
        self.state ^= self.state >> 7;
        self.state ^= self.state << 17;
        self.state
    }
    fn next_range(&mut self, min: usize, max: usize) -> usize {
        let range = max - min;
        min + (self.next() as usize % range)
    }
    fn shuffle<T>(&mut self, slice: &mut [T]) {
        for i in (1..slice.len()).rev() {
            let j = self.next_range(0, i + 1);
            slice.swap(i, j);
        }
    }
}

fn run_kernighan_lin(n: usize, adj: &[Vec<usize>], matrix: &[Vec<bool>], seed: u64) -> (u32, Vec<usize>) {
    let mut rng = Lcg::new(seed);
    let mut best_cut = u32::MAX;
    let mut best_partition = Vec::new();

    // Run 30 random starts
    for _ in 0..30 {
        let mut vertices: Vec<usize> = (0..n).collect();
        rng.shuffle(&mut vertices);
        let mut in_a = vec![false; n];
        for &v in &vertices[0..n/2] {
            in_a[v] = true;
        }

        let mut improved = true;
        while improved {
            improved = false;
            let mut d = vec![0i32; n];
            for i in 0..n {
                let mut ext = 0;
                let mut int = 0;
                for &neighbor in &adj[i] {
                    if in_a[neighbor] == in_a[i] {
                        int += 1;
                    } else {
                        ext += 1;
                    }
                }
                d[i] = ext - int;
            }

            let mut locked = vec![false; n];
            let mut swaps = Vec::new();
            let mut temp_in_a = in_a.clone();
            let mut temp_d = d.clone();

            for _ in 0..(n / 2) {
                let mut max_gain = i32::MIN;
                let mut best_a = 0;
                let mut best_b = 0;
                let mut found = false;

                for a in 0..n {
                    if temp_in_a[a] && !locked[a] {
                        for b in 0..n {
                            if !temp_in_a[b] && !locked[b] {
                                let w = if matrix[a][b] { 1 } else { 0 };
                                let gain = temp_d[a] + temp_d[b] - 2 * w;
                                if gain > max_gain {
                                    max_gain = gain;
                                    best_a = a;
                                    best_b = b;
                                    found = true;
                                }
                            }
                        }
                    }
                }

                if !found {
                    break;
                }

                locked[best_a] = true;
                locked[best_b] = true;
                temp_in_a[best_a] = false;
                temp_in_a[best_b] = true;
                swaps.push((best_a, best_b, max_gain));

                for i in 0..n {
                    if !locked[i] {
                        let w_a = if matrix[i][best_a] { 1 } else { 0 };
                        let w_b = if matrix[i][best_b] { 1 } else { 0 };
                        if temp_in_a[i] {
                            temp_d[i] += 2 * w_a - 2 * w_b;
                        } else {
                            temp_d[i] += 2 * w_b - 2 * w_a;
                        }
                    }
                }
            }

            let mut cum_gain = 0;
            let mut max_cum_gain = 0;
            let mut best_k = 0;
            for (i, &(_, _, gain)) in swaps.iter().enumerate() {
                cum_gain += gain;
                if cum_gain > max_cum_gain {
                    max_cum_gain = cum_gain;
                    best_k = i + 1;
                }
            }

            if max_cum_gain > 0 {
                for i in 0..best_k {
                    let (a, b, _) = swaps[i];
                    in_a.swap(a, b);
                }
                improved = true;
            }
        }

        let mut cut = 0;
        for i in 0..n {
            if in_a[i] {
                for &neighbor in &adj[i] {
                    if !in_a[neighbor] {
                        cut += 1;
                    }
                }
            }
        }

        if cut < best_cut {
            best_cut = cut;
            best_partition = (0..n).filter(|&i| in_a[i]).collect();
        }
    }

    (best_cut, best_partition)
}

struct Solver {
    n: usize,
    ordered_adj: Vec<Vec<usize>>,
    order: Vec<usize>,
    assignment: Vec<u8>,
    d_a: Vec<u32>,
    d_b: Vec<u32>,
    best_cut: u32,
    best_partition: Vec<usize>,
    p_a: Vec<u32>,
    p_b: Vec<u32>,
}

impl Solver {
    fn new(n: usize, adj: Vec<Vec<usize>>, initial_cut: u32, initial_partition: Vec<usize>) -> Self {
        let mut order = Vec::with_capacity(n);
        let mut inv_order = vec![0; n];
        let mut visited = vec![false; n];

        order.push(0);
        visited[0] = true;
        let mut conn = vec![0; n];
        for &v in &adj[0] {
            conn[v] += 1;
        }

        for _ in 1..n {
            let mut best_v = None;
            let mut max_conn = -1;
            for v in 0..n {
                if !visited[v] {
                    if conn[v] as i32 > max_conn {
                        max_conn = conn[v] as i32;
                        best_v = Some(v);
                    }
                }
            }
            let u = best_v.unwrap();
            visited[u] = true;
            order.push(u);
            for &v in &adj[u] {
                if !visited[v] {
                    conn[v] += 1;
                }
            }
        }

        for i in 0..n {
            inv_order[order[i]] = i;
        }

        let mut ordered_adj = vec![Vec::new(); n];
        for i in 0..n {
            let u = order[i];
            for &v in &adj[u] {
                ordered_adj[i].push(inv_order[v]);
            }
        }

        let mut assignment = vec![2; n];
        assignment[0] = 0;

        let mut d_a = vec![0; n];
        for &v_pos in &ordered_adj[0] {
            if v_pos > 0 {
                d_a[v_pos] += 1;
            }
        }

        Self {
            n,
            ordered_adj,
            order,
            assignment,
            d_a,
            d_b: vec![0; n],
            best_cut: initial_cut,
            best_partition: initial_partition,
            p_a: Vec::with_capacity(n),
            p_b: Vec::with_capacity(n),
        }
    }

    fn check_bound(&mut self, next_k: usize, cut_so_far: u32, size_a: usize, size_b: usize) -> bool {
        let a_rem = self.n / 2 - size_a;
        let b_rem = self.n / 2 - size_b;

        let mut simple_sum = 0;
        for j in next_k..self.n {
            simple_sum += std::cmp::min(self.d_a[j], self.d_b[j]);
        }

        if cut_so_far + simple_sum >= self.best_cut {
            return false;
        }

        self.p_a.clear();
        self.p_b.clear();
        let mut base_sum = 0;

        for j in next_k..self.n {
            let da = self.d_a[j];
            let db = self.d_b[j];
            if da >= db {
                base_sum += db;
                self.p_a.push(da - db);
            } else {
                base_sum += da;
                self.p_b.push(db - da);
            }
        }

        if self.p_a.len() > a_rem {
            self.p_a.sort_unstable();
            let needed = self.p_a.len() - a_rem;
            for i in 0..needed {
                base_sum += self.p_a[i];
            }
        } else if self.p_b.len() > b_rem {
            self.p_b.sort_unstable();
            let needed = self.p_b.len() - b_rem;
            for i in 0..needed {
                base_sum += self.p_b[i];
            }
        }

        cut_so_far + base_sum < self.best_cut
    }

    fn dfs(&mut self, k: usize, cut_so_far: u32, size_a: usize, size_b: usize) {
        if k == self.n {
            if cut_so_far < self.best_cut {
                self.best_cut = cut_so_far;
                self.best_partition.clear();
                for i in 0..self.n {
                    if self.assignment[i] == 0 {
                        self.best_partition.push(self.order[i]);
                    }
                }
            }
            return;
        }

        let a_rem = self.n / 2 - size_a;
        let b_rem = self.n / 2 - size_b;

        if a_rem == 0 {
            let mut final_cut = cut_so_far;
            for j in k..self.n {
                final_cut += self.d_a[j];
            }
            if final_cut < self.best_cut {
                self.best_cut = final_cut;
                self.best_partition.clear();
                for i in 0..self.n {
                    if i < k && self.assignment[i] == 0 {
                        self.best_partition.push(self.order[i]);
                    }
                }
            }
            return;
        }

        if b_rem == 0 {
            let mut final_cut = cut_so_far;
            for j in k..self.n {
                final_cut += self.d_b[j];
            }
            if final_cut < self.best_cut {
                self.best_cut = final_cut;
                self.best_partition.clear();
                for i in 0..self.n {
                    if (i < k && self.assignment[i] == 0) || i >= k {
                        self.best_partition.push(self.order[i]);
                    }
                }
            }
            return;
        }

        let prefer_a = self.d_a[k] >= self.d_b[k];

        if prefer_a {
            if size_a < self.n / 2 {
                let new_cut = cut_so_far + self.d_b[k];
                if new_cut < self.best_cut {
                    self.assignment[k] = 0;
                    for &v_pos in &self.ordered_adj[k] {
                        if v_pos > k {
                            self.d_a[v_pos] += 1;
                        }
                    }

                    if self.check_bound(k + 1, new_cut, size_a + 1, size_b) {
                        self.dfs(k + 1, new_cut, size_a + 1, size_b);
                    }

                    for &v_pos in &self.ordered_adj[k] {
                        if v_pos > k {
                            self.d_a[v_pos] -= 1;
                        }
                    }
                }
            }
            if size_b < self.n / 2 {
                let new_cut = cut_so_far + self.d_a[k];
                if new_cut < self.best_cut {
                    self.assignment[k] = 1;
                    for &v_pos in &self.ordered_adj[k] {
                        if v_pos > k {
                            self.d_b[v_pos] += 1;
                        }
                    }

                    if self.check_bound(k + 1, new_cut, size_a, size_b + 1) {
                        self.dfs(k + 1, new_cut, size_a, size_b + 1);
                    }

                    for &v_pos in &self.ordered_adj[k] {
                        if v_pos > k {
                            self.d_b[v_pos] -= 1;
                        }
                    }
                }
            }
        } else {
            if size_b < self.n / 2 {
                let new_cut = cut_so_far + self.d_a[k];
                if new_cut < self.best_cut {
                    self.assignment[k] = 1;
                    for &v_pos in &self.ordered_adj[k] {
                        if v_pos > k {
                            self.d_b[v_pos] += 1;
                        }
                    }

                    if self.check_bound(k + 1, new_cut, size_a, size_b + 1) {
                        self.dfs(k + 1, new_cut, size_a, size_b + 1);
                    }

                    for &v_pos in &self.ordered_adj[k] {
                        if v_pos > k {
                            self.d_b[v_pos] -= 1;
                        }
                    }
                }
            }
            if size_a < self.n / 2 {
                let new_cut = cut_so_far + self.d_b[k];
                if new_cut < self.best_cut {
                    self.assignment[k] = 0;
                    for &v_pos in &self.ordered_adj[k] {
                        if v_pos > k {
                            self.d_a[v_pos] += 1;
                        }
                    }

                    if self.check_bound(k + 1, new_cut, size_a + 1, size_b) {
                        self.dfs(k + 1, new_cut, size_a + 1, size_b);
                    }

                    for &v_pos in &self.ordered_adj[k] {
                        if v_pos > k {
                            self.d_a[v_pos] -= 1;
                        }
                    }
                }
            }
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = match lines.next() {
        Some(Ok(l)) => l,
        _ => return,
    };
    let parts: Vec<&str> = first_line.split_whitespace().collect();
    if parts.len() < 2 {
        return;
    }
    let n: usize = parts[0].parse().unwrap();
    let m: usize = parts[1].parse().unwrap();

    let mut adj = vec![Vec::new(); n];
    let mut matrix = vec![vec![false; n]; n];

    let mut edge_count = 0;
    while edge_count < m {
        if let Some(Ok(line)) = lines.next() {
            let pts: Vec<&str> = line.split_whitespace().collect();
            if pts.is_empty() {
                continue;
            }
            if pts.len() >= 2 {
                let u: usize = pts[0].parse().unwrap();
                let v: usize = pts[1].parse().unwrap();
                if u < n && v < n && u != v {
                    if !matrix[u][v] {
                        matrix[u][v] = true;
                        matrix[v][u] = true;
                        adj[u].push(v);
                        adj[v].push(u);
                    }
                }
            }
            edge_count += 1;
        } else {
            break;
        }
    }

    if n == 0 {
        println!("0");
        println!();
        return;
    }

    if n % 2 != 0 {
        panic!("n must be even");
    }

    // Run Kernighan-Lin heuristic to get an initial upper bound
    let (kl_cut, kl_part) = run_kernighan_lin(n, &adj, &matrix, 42);

    // Run exact Branch & Bound solver to find and prove the optimal minimum cut bisection
    let mut solver = Solver::new(n, adj, kl_cut, kl_part);
    if solver.n > 1 {
        solver.dfs(1, 0, 1, 0);
    }

    println!("{}", solver.best_cut);
    let mut part_a = solver.best_partition.clone();
    part_a.sort_unstable();
    for (i, &v) in part_a.iter().enumerate() {
        if i > 0 {
            print!(" ");
        }
        print!("{}", v);
    }
    println!();
}
