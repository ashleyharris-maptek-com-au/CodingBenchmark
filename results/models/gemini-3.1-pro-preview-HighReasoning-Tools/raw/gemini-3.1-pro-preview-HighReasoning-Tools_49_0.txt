#version 450
layout(local_size_x = 256, local_size_y = 1, local_size_z = 1) in;

layout(set = 0, binding = 0, std430) readonly buffer InputBuffer {
    uint data[];
} input_buf;

layout(set = 0, binding = 1, std430) buffer OutputBuffer {
    uint data[];
} output_buf;

layout(set = 0, binding = 2, std140) uniform UniformBuffer {
    uvec4 info; // info.x is N
} ubo;

shared uint temp[256];
shared uint shared_prev_block_sum;

void main() {
    uint g = gl_WorkGroupID.x;
    uint t = gl_LocalInvocationID.x;
    uint N = ubo.info.x;

    if (g * 256 >= N) {
        return;
    }

    // Step 1: Thread t sums the t-th element of all previous blocks
    uint prev_sum = 0;
    for (uint b = 0; b < g; ++b) {
        prev_sum += input_buf.data[b * 256 + t];
    }
    temp[t] = prev_sum;
    barrier();

    // Step 2: Shared memory reduction to get the sum of all previous blocks
    for (uint d = 128; d > 0; d /= 2) {
        if (t < d) {
            temp[t] += temp[t + d];
        }
        barrier();
    }
    if (t == 0) {
        shared_prev_block_sum = temp[0];
    }
    barrier();

    // Step 3: Load the current element of the block
    uint val = input_buf.data[g * 256 + t];
    temp[t] = val;
    barrier();

    // Step 4: Perform a local shared-memory Hillis-Steele scan
    for (uint d = 1; d < 256; d *= 2) {
        uint offset_val = 0;
        if (t >= d) {
            offset_val = temp[t - d];
        }
        barrier();
        if (t >= d) {
            temp[t] += offset_val;
        }
        barrier();
    }

    // Step 5: Add the sum of previous blocks to get the global inclusive scan
    output_buf.data[g * 256 + t] = temp[t] + shared_prev_block_sum;
}
