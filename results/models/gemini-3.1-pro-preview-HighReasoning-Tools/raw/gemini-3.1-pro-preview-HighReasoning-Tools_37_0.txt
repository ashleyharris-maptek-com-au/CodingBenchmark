using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Threading;

class Solver
{
    static void Main(string[] args)
    {
        // Run the main solver in a thread with a large stack (128 MB) to prevent stack overflow
        Thread thread = new Thread(Run, 128 * 1024 * 1024);
        thread.Start();
        thread.Join();
    }

    static void Run()
    {
        FastScanner scanner = new FastScanner();
        int n = scanner.NextInt();
        if (n <= 0)
        {
            Console.WriteLine(0);
            Console.WriteLine();
            return;
        }

        int m = scanner.NextInt();
        Graph G = new Graph(n);
        for (int i = 0; i < n; i++) G.Present[i] = true;

        for (int i = 0; i < m; i++)
        {
            int u = scanner.NextInt();
            int v = scanner.NextInt();
            if (u >= 0 && u < n && v >= 0 && v < n)
            {
                G.AddEdge(u, v);
            }
        }

        // Apply initial edge domination rule
        ApplyEdgeDomination(G);

        // Compute SCCs
        SCC scc = new SCC(G);
        scc.Run();

        List<int> globalSolution = new List<int>();

        foreach (var comp in scc.Components)
        {
            if (comp.Count == 1)
            {
                int u = comp[0];
                if (G.Out[u].Contains(u))
                {
                    globalSolution.Add(u);
                }
                continue;
            }

            // Map component vertices to 0..K-1
            int K = comp.Count;
            int[] originalToMapped = new int[n];
            for (int i = 0; i < n; i++) originalToMapped[i] = -1;
            for (int i = 0; i < K; i++)
            {
                originalToMapped[comp[i]] = i;
            }

            bool[] inComp = new bool[n];
            foreach (int v in comp) inComp[v] = true;

            Graph subG = new Graph(K);
            for (int i = 0; i < K; i++) subG.Present[i] = true;

            for (int i = 0; i < K; i++)
            {
                int u = comp[i];
                int mappedU = i;
                foreach (int w in G.Out[u])
                {
                    if (inComp[w])
                    {
                        int mappedW = originalToMapped[w];
                        subG.AddEdge(mappedU, mappedW);
                    }
                }
            }

            List<int> subSolutionMapped = SolveComponent(subG);
            foreach (int mappedV in subSolutionMapped)
            {
                globalSolution.Add(comp[mappedV]);
            }
        }

        // Output results
        Console.WriteLine(globalSolution.Count);
        Console.WriteLine(string.Join(" ", globalSolution));
    }

    static List<int> SolveComponent(Graph G)
    {
        int[] dist = new int[G.N];
        int[] parent = new int[G.N];
        int[] queue = new int[G.N];
        for (int i = 0; i < dist.Length; i++) dist[i] = -1;

        bestSolutionSize = int.MaxValue;
        bestSolution = null;

        BranchAndBound(G, new List<int>(), dist, parent, queue);

        return bestSolution ?? new List<int>();
    }

    static int bestSolutionSize;
    static List<int> bestSolution;

    static void BranchAndBound(Graph G, List<int> currentSolution, int[] dist, int[] parent, int[] queue)
    {
        // 1. Run Preprocessing
        Preprocessor prep = new Preprocessor(G, currentSolution);
        prep.Run();

        // 2. Check if we exceeded best solution
        if (currentSolution.Count >= bestSolutionSize)
        {
            return;
        }

        // 3. Compute lower bound
        int lb = GetDisjointCyclesLowerBound(G, dist, parent, queue);
        if (currentSolution.Count + lb >= bestSolutionSize)
        {
            return;
        }

        // 4. Find a shortest cycle to branch on
        List<int> cycle = FindShortestCycle(G, dist, parent, queue);
        if (cycle == null)
        {
            // No cycles! We found a valid feedback vertex set
            if (currentSolution.Count < bestSolutionSize)
            {
                bestSolutionSize = currentSolution.Count;
                bestSolution = new List<int>(currentSolution);
            }
            return;
        }

        // 5. Branch on the vertices of the cycle
        // Sort vertices of the cycle by degree in G (descending order)
        var sortedCycle = cycle.OrderByDescending(v => G.In[v].Count + G.Out[v].Count).ToList();

        foreach (int v in sortedCycle)
        {
            Graph nextG = G.Clone();
            List<int> nextSol = new List<int>(currentSolution);

            nextSol.Add(v);
            nextG.RemoveVertex(v);

            BranchAndBound(nextG, nextSol, dist, parent, queue);
        }
    }

    static int GetDisjointCyclesLowerBound(Graph G, int[] dist, int[] parent, int[] queue)
    {
        Graph tempG = G.Clone();
        int count = 0;
        while (true)
        {
            List<int> cycle = FindShortestCycle(tempG, dist, parent, queue);
            if (cycle == null) break;
            count++;
            foreach (int v in cycle)
            {
                tempG.RemoveVertex(v);
            }
            ApplySimpleReductions(tempG);
        }
        return count;
    }

    static void ApplySimpleReductions(Graph G)
    {
        Queue<int> q = new Queue<int>();
        for (int i = 0; i < G.N; i++)
        {
            if (G.Present[i] && (G.In[i].Count == 0 || G.Out[i].Count == 0))
            {
                q.Enqueue(i);
            }
        }
        while (q.Count > 0)
        {
            int u = q.Dequeue();
            if (!G.Present[u]) continue;

            List<int> neighbors = new List<int>();
            foreach (int v in G.In[u]) neighbors.Add(v);
            foreach (int v in G.Out[u]) neighbors.Add(v);

            G.RemoveVertex(u);

            foreach (int v in neighbors)
            {
                if (G.Present[v] && (G.In[v].Count == 0 || G.Out[v].Count == 0))
                {
                    q.Enqueue(v);
                }
            }
        }
    }

    static List<int> FindShortestCycle(Graph G, int[] dist, int[] parent, int[] queue)
    {
        int minLen = int.MaxValue;
        List<int> bestCycle = null;

        for (int s = 0; s < G.N; s++)
        {
            if (!G.Present[s]) continue;
            if (G.Out[s].Count == 0) continue;

            List<int> visited = new List<int>();

            int head = 0, tail = 0;
            queue[tail++] = s;
            dist[s] = 0;
            visited.Add(s);
            parent[s] = -1;

            bool foundForS = false;
            while (head < tail)
            {
                int u = queue[head++];
                int d = dist[u];

                if (d >= minLen - 1) break;

                foreach (int v in G.Out[u])
                {
                    if (v == s)
                    {
                        if (d + 1 < minLen)
                        {
                            minLen = d + 1;
                            List<int> cycle = new List<int>();
                            int curr = u;
                            while (curr != -1)
                            {
                                cycle.Add(curr);
                                curr = parent[curr];
                            }
                            bestCycle = cycle;
                            foundForS = true;
                            break;
                        }
                    }
                    else if (G.Present[v])
                    {
                        if (dist[v] == -1)
                        {
                            dist[v] = d + 1;
                            parent[v] = u;
                            queue[tail++] = v;
                            visited.Add(v);
                        }
                    }
                }
                if (foundForS) break;
            }

            foreach (int v in visited)
            {
                dist[v] = -1;
            }

            if (minLen == 2) break;
        }

        return bestCycle;
    }

    static void ApplyEdgeDomination(Graph G)
    {
        bool changed = true;
        while (changed)
        {
            changed = false;
            for (int u = 0; u < G.N; u++)
            {
                if (!G.Present[u]) continue;
                List<int> toRemove = new List<int>();
                foreach (int v in G.Out[u])
                {
                    if (v == u || !G.Present[v]) continue;

                    // Check if Out[v] is subset of Out[u]
                    bool outSub = true;
                    foreach (int w in G.Out[v])
                    {
                        if (!G.Out[u].Contains(w))
                        {
                            outSub = false;
                            break;
                        }
                    }

                    // Check if In[u] is subset of In[v]
                    bool inSub = true;
                    foreach (int w in G.In[u])
                    {
                        if (!G.In[v].Contains(w))
                        {
                            inSub = false;
                            break;
                        }
                    }

                    if (outSub || inSub)
                    {
                        toRemove.Add(v);
                    }
                }

                if (toRemove.Count > 0)
                {
                    foreach (int v in toRemove)
                    {
                        G.RemoveEdge(u, v);
                    }
                    changed = true;
                }
            }
        }
    }
}

class Graph
{
    public int N;
    public bool[] Present;
    public HashSet<int>[] In;
    public HashSet<int>[] Out;

    public Graph(int n)
    {
        N = n;
        Present = new bool[n];
        In = new HashSet<int>[n];
        Out = new HashSet<int>[n];
        for (int i = 0; i < n; i++)
        {
            In[i] = new HashSet<int>();
            Out[i] = new HashSet<int>();
        }
    }

    public Graph Clone()
    {
        Graph copy = new Graph(N);
        Array.Copy(Present, copy.Present, N);
        for (int i = 0; i < N; i++)
        {
            if (Present[i])
            {
                copy.In[i] = new HashSet<int>(In[i]);
                copy.Out[i] = new HashSet<int>(Out[i]);
            }
        }
        return copy;
    }

    public void AddEdge(int u, int v)
    {
        if (u == v)
        {
            Out[u].Add(u);
            In[u].Add(u);
            return;
        }
        Out[u].Add(v);
        In[v].Add(u);
    }

    public void RemoveEdge(int u, int v)
    {
        Out[u].Remove(v);
        In[v].Remove(u);
    }

    public void RemoveVertex(int u)
    {
        if (!Present[u]) return;
        Present[u] = false;
        foreach (int v in Out[u])
        {
            if (v != u) In[v].Remove(u);
        }
        foreach (int v in In[u])
        {
            if (v != u) Out[v].Remove(u);
        }
        Out[u].Clear();
        In[u].Clear();
    }
}

class Preprocessor
{
    public Graph G;
    public List<int> Solution;
    private Queue<int> queue;
    private bool[] inQueue;

    public Preprocessor(Graph g, List<int> solution)
    {
        G = g;
        Solution = solution;
        queue = new Queue<int>();
        inQueue = new bool[G.N];
        for (int i = 0; i < G.N; i++)
        {
            if (G.Present[i])
            {
                queue.Enqueue(i);
                inQueue[i] = true;
            }
        }
    }

    public void Enqueue(int u)
    {
        if (G.Present[u] && !inQueue[u])
        {
            queue.Enqueue(u);
            inQueue[u] = true;
        }
    }

    public bool Run()
    {
        bool changed = false;
        while (queue.Count > 0)
        {
            int u = queue.Dequeue();
            inQueue[u] = false;

            if (!G.Present[u]) continue;

            // 1. LOOP rule
            if (G.Out[u].Contains(u))
            {
                Solution.Add(u);
                List<int> neighbors = new List<int>();
                foreach (int v in G.In[u]) if (v != u) neighbors.Add(v);
                foreach (int v in G.Out[u]) if (v != u) neighbors.Add(v);

                G.RemoveVertex(u);
                changed = true;

                foreach (int v in neighbors) Enqueue(v);
                continue;
            }

            // 2. IN0 / OUT0 rule
            if (G.In[u].Count == 0 || G.Out[u].Count == 0)
            {
                List<int> neighbors = new List<int>();
                foreach (int v in G.In[u]) neighbors.Add(v);
                foreach (int v in G.Out[u]) neighbors.Add(v);

                G.RemoveVertex(u);
                changed = true;

                foreach (int v in neighbors) Enqueue(v);
                continue;
            }

            // 3. IN1 rule
            if (G.In[u].Count == 1)
            {
                int p = -1;
                foreach (int val in G.In[u]) { p = val; break; }
                List<int> successors = new List<int>(G.Out[u]);
                G.RemoveVertex(u);
                changed = true;

                foreach (int w in successors)
                {
                    G.AddEdge(p, w);
                }

                Enqueue(p);
                foreach (int w in successors) Enqueue(w);
                continue;
            }

            // 4. OUT1 rule
            if (G.Out[u].Count == 1)
            {
                int s = -1;
                foreach (int val in G.Out[u]) { s = val; break; }
                List<int> predecessors = new List<int>(G.In[u]);
                G.RemoveVertex(u);
                changed = true;

                foreach (int p in predecessors)
                {
                    G.AddEdge(p, s);
                }

                Enqueue(s);
                foreach (int p in predecessors) Enqueue(p);
                continue;
            }
        }
        return changed;
    }
}

class SCC
{
    private readonly Graph G;
    private readonly int[] dfn;
    private readonly int[] low;
    private readonly bool[] inStack;
    private readonly int[] stack;
    private int stackTop;
    private int timer;
    public List<List<int>> Components;

    public SCC(Graph g)
    {
        G = g;
        dfn = new int[G.N];
        low = new int[G.N];
        inStack = new bool[G.N];
        stack = new int[G.N];
        for (int i = 0; i < dfn.Length; i++) dfn[i] = -1;
        Components = new List<List<int>>();
    }

    public void Run()
    {
        for (int i = 0; i < G.N; i++)
        {
            if (G.Present[i] && dfn[i] == -1)
            {
                DFS(i);
            }
        }
    }

    private void DFS(int u)
    {
        dfn[u] = low[u] = timer++;
        stack[stackTop++] = u;
        inStack[u] = true;

        foreach (int v in G.Out[u])
        {
            if (!G.Present[v]) continue;
            if (dfn[v] == -1)
            {
                DFS(v);
                low[u] = Math.Min(low[u], low[v]);
            }
            else if (inStack[v])
            {
                low[u] = Math.Min(low[u], dfn[v]);
            }
        }

        if (low[u] == dfn[u])
        {
            List<int> component = new List<int>();
            while (true)
            {
                int v = stack[--stackTop];
                inStack[v] = false;
                component.Add(v);
                if (v == u) break;
            }
            Components.Add(component);
        }
    }
}

class FastScanner
{
    private readonly Stream stream;
    private readonly byte[] buffer = new byte[1024 * 1024];
    private int head, tail;

    public FastScanner()
    {
        stream = Console.OpenStandardInput();
    }

    private byte Read()
    {
        if (head >= tail)
        {
            head = 0;
            tail = stream.Read(buffer, 0, buffer.Length);
            if (tail <= 0) return 0;
        }
        return buffer[head++];
    }

    public int NextInt()
    {
        byte c = Read();
        while (c <= 32)
        {
            if (c == 0) return -1;
            c = Read();
        }
        int res = 0;
        while (c > 32)
        {
            if (c >= '0' && c <= '9')
            {
                res = res * 10 + c - '0';
            }
            c = Read();
        }
        return res;
    }
}
