using System;
using System.IO;
using System.Collections.Generic;
using System.Diagnostics;

struct Point
{
    public int X;
    public int Y;
    public Point(int x, int y) { X = x; Y = y; }
}

public class MinHeap<T>
{
    private struct Node
    {
        public T Item;
        public int Priority;
    }
    private Node[] _nodes;
    private int _size;

    public MinHeap(int capacity = 16)
    {
        _nodes = new Node[capacity];
    }

    public int Count => _size;

    public void Enqueue(T item, int priority)
    {
        if (_size == _nodes.Length)
        {
            Array.Resize(ref _nodes, _nodes.Length * 2);
        }
        _nodes[_size] = new Node { Item = item, Priority = priority };
        UpHeap(_size++);
    }

    public T Dequeue()
    {
        if (_size == 0) throw new InvalidOperationException("Heap is empty");
        T result = _nodes[0].Item;
        _nodes[0] = _nodes[--_size];
        _nodes[_size] = default;
        DownHeap(0);
        return result;
    }

    public void Clear()
    {
        _size = 0;
    }

    private void UpHeap(int i)
    {
        while (i > 0)
        {
            int p = (i - 1) / 2;
            if (_nodes[i].Priority >= _nodes[p].Priority) break;
            Node temp = _nodes[i];
            _nodes[i] = _nodes[p];
            _nodes[p] = temp;
            i = p;
        }
    }

    private void DownHeap(int i)
    {
        while (i * 2 + 1 < _size)
        {
            int l = i * 2 + 1;
            int r = l + 1;
            int child = (r < _size && _nodes[r].Priority < _nodes[l].Priority) ? r : l;
            if (_nodes[i].Priority <= _nodes[child].Priority) break;
            Node temp = _nodes[i];
            _nodes[i] = _nodes[child];
            _nodes[child] = temp;
            i = child;
        }
    }
}

class Program
{
    struct PathResult
    {
        public bool Success;
        public List<Point> Path;
        public int FuelCostMultiplier;
    }

    struct SavedOp
    {
        public int DigX;
        public int DigY;
        public int DigZ;
        public int PathStart;
        public int PathLength;
        public int DestX;
        public int DestY;
    }

    struct CandidateColumn
    {
        public int X;
        public int Y;
        public int TargetMinDepth;
        public double EstProfit;
    }

    static int X, Y, Z;
    static byte[][] columns;
    static int[] heights;
    static bool[] isCandidate;
    static bool[] isMinedFinished;

    static int epoch = 0;
    static int[] distTable;
    static int[] parentTable;
    static int[] visitedEpoch;
    static MinHeap<int> pq = new MinHeap<int>(1024);

    static int[] dx = { 1, -1, 0, 0 };
    static int[] dy = { 0, 0, 1, -1 };

    static int[] globalSteps = new int[1000000];
    static int globalStepsCount = 0;
    static List<SavedOp> ops = new List<SavedOp>();

    static byte GetVoxel(int tx, int ty, int tz)
    {
        int idx = tx * Y + ty;
        if (columns[idx] == null || tz >= columns[idx].Length) return 0;
        return columns[idx][tz];
    }

    static void SetVoxel(int tx, int ty, int tz, byte val)
    {
        int idx = tx * Y + ty;
        if (columns[idx] == null)
        {
            columns[idx] = new byte[tz + 1];
        }
        else if (tz >= columns[idx].Length)
        {
            Array.Resize(ref columns[idx], tz + 1);
        }
        columns[idx][tz] = val;
    }

    static void SaveOperation(int dx, int dy, int dz, List<Point> path, int destX, int destY)
    {
        int len = path.Count;
        if (globalStepsCount + len > globalSteps.Length)
        {
            int newSize = Math.Max(globalSteps.Length * 2, globalStepsCount + len);
            Array.Resize(ref globalSteps, newSize);
        }
        
        int start = globalStepsCount;
        for (int i = 0; i < len; i++)
        {
            globalSteps[start + i] = (path[i].X << 16) | path[i].Y;
        }
        globalStepsCount += len;
        
        ops.Add(new SavedOp
        {
            DigX = dx,
            DigY = dy,
            DigZ = dz,
            PathStart = start,
            PathLength = len,
            DestX = destX,
            DestY = destY
        });
    }

    static bool IsSafeToDump(int cx, int cy)
    {
        int idx = cx * Y + cy;
        if (isCandidate[idx] && !isMinedFinished[idx]) return false;
        return true;
    }

    static PathResult FindPathToPlant(Point start, int startHeight)
    {
        epoch++;
        pq.Clear();
        
        int startIdx = start.X * Y + start.Y;
        distTable[startIdx] = 0;
        visitedEpoch[startIdx] = epoch;
        pq.Enqueue(startIdx, 0);
        
        bool found = false;
        int destIdx = 0; // (0,0) is index 0
        
        int maxX = Math.Min(start.X + 10, X - 1);
        int maxY = Math.Min(start.Y + 10, Y - 1);
        
        while (pq.Count > 0)
        {
            int curr = pq.Dequeue();
            if (curr == destIdx)
            {
                found = true;
                break;
            }
            
            int currDist = distTable[curr];
            int cx = curr / Y;
            int cy = curr % Y;
            
            int currHeight = (cx == start.X && cy == start.Y) ? startHeight : heights[curr];
            
            for (int d = 0; d < 4; d++)
            {
                int nx = cx + dx[d];
                int ny = cy + dy[d];
                if (nx < 0 || nx > maxX || ny < 0 || ny > maxY) continue;
                
                int nIdx = nx * Y + ny;
                int nHeight = (nx == start.X && ny == start.Y) ? startHeight : heights[nIdx];
                if (Math.Abs(nHeight - currHeight) > 1) continue;
                
                int stepCost = 1;
                if (nHeight == currHeight + 1) stepCost = 5;
                else if (nHeight == currHeight - 1) stepCost = 0;
                
                int nextDist = currDist + stepCost;
                if (visitedEpoch[nIdx] != epoch || nextDist < distTable[nIdx])
                {
                    distTable[nIdx] = nextDist;
                    parentTable[nIdx] = curr;
                    visitedEpoch[nIdx] = epoch;
                    pq.Enqueue(nIdx, nextDist);
                }
            }
        }
        
        if (!found)
        {
            return new PathResult { Success = false };
        }
        
        List<Point> path = new List<Point>();
        int temp = destIdx;
        while (temp != startIdx)
        {
            path.Add(new Point(temp / Y, temp % Y));
            temp = parentTable[temp];
        }
        path.Reverse();
        
        return new PathResult
        {
            Success = true,
            Path = path,
            FuelCostMultiplier = distTable[destIdx]
        };
    }

    static PathResult FindPathToDump(Point start, int startHeight)
    {
        epoch++;
        pq.Clear();
        
        int startIdx = start.X * Y + start.Y;
        distTable[startIdx] = 0;
        visitedEpoch[startIdx] = epoch;
        pq.Enqueue(startIdx, 0);
        
        int foundIdx = -1;
        
        while (pq.Count > 0)
        {
            int curr = pq.Dequeue();
            int currDist = distTable[curr];
            
            int cx = curr / Y;
            int cy = curr % Y;
            
            if ((cx != start.X || cy != start.Y) && (cx != 0 || cy != 0) && IsSafeToDump(cx, cy))
            {
                foundIdx = curr;
                break;
            }
            
            if (currDist > 200) continue;
            
            int currHeight = (cx == start.X && cy == start.Y) ? startHeight : heights[curr];
            
            for (int d = 0; d < 4; d++)
            {
                int nx = cx + dx[d];
                int ny = cy + dy[d];
                if (nx < 0 || nx >= X || ny < 0 || ny >= Y) continue;
                
                int nIdx = nx * Y + ny;
                int nHeight = (nx == start.X && ny == start.Y) ? startHeight : heights[nIdx];
                if (Math.Abs(nHeight - currHeight) > 1) continue;
                
                int stepCost = 1;
                if (nHeight == currHeight + 1) stepCost = 5;
                else if (nHeight == currHeight - 1) stepCost = 0;
                
                int nextDist = currDist + stepCost;
                if (visitedEpoch[nIdx] != epoch || nextDist < distTable[nIdx])
                {
                    distTable[nIdx] = nextDist;
                    parentTable[nIdx] = curr;
                    visitedEpoch[nIdx] = epoch;
                    pq.Enqueue(nIdx, nextDist);
                }
            }
        }
        
        if (foundIdx == -1)
        {
            return new PathResult { Success = false };
        }
        
        List<Point> path = new List<Point>();
        int temp = foundIdx;
        while (temp != startIdx)
        {
            path.Add(new Point(temp / Y, temp % Y));
            temp = parentTable[temp];
        }
        path.Reverse();
        
        return new PathResult
        {
            Success = true,
            Path = path,
            FuelCostMultiplier = distTable[foundIdx]
        };
    }

    static void Main(string[] args)
    {
        Stopwatch globalSw = Stopwatch.StartNew();
        
        string headerLine = Console.ReadLine();
        if (string.IsNullOrWhiteSpace(headerLine)) return;
        string[] parts = headerLine.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        X = int.Parse(parts[0]);
        Y = int.Parse(parts[1]);
        Z = int.Parse(parts[2]);
        
        columns = new byte[X * Y][];
        heights = new int[X * Y];
        isCandidate = new bool[X * Y];
        isMinedFinished = new bool[X * Y];
        
        distTable = new int[X * Y];
        parentTable = new int[X * Y];
        visitedEpoch = new int[X * Y];
        
        using (Stream stdin = Console.OpenStandardInput())
        {
            var bufferedStdin = new BufferedStream(stdin, 4 * 1024 * 1024);
            byte[] tempBuf = new byte[Z];
            for (int x = 0; x < X; x++)
            {
                for (int y = 0; y < Y; y++)
                {
                    int bytesRead = 0;
                    while (bytesRead < Z)
                    {
                        int r = bufferedStdin.Read(tempBuf, bytesRead, Z - bytesRead);
                        if (r <= 0) break;
                        bytesRead += r;
                    }
                    
                    int h = 0;
                    for (int z = Z - 1; z >= 0; z--)
                    {
                        byte d = tempBuf[z];
                        int mass = d % 16;
                        if (mass > 0)
                        {
                            h = z + 1;
                            break;
                        }
                    }
                    
                    int idx = x * Y + y;
                    if (h > 0)
                    {
                        byte[] col = new byte[h];
                        Array.Copy(tempBuf, col, h);
                        columns[idx] = col;
                        heights[idx] = h;
                    }
                    else
                    {
                        columns[idx] = null;
                        heights[idx] = 0;
                    }
                }
            }
        }
        
        List<CandidateColumn> candidates = new List<CandidateColumn>();
        for (int x = 0; x < X; x++)
        {
            for (int y = 0; y < Y; y++)
            {
                int h = heights[x * Y + y];
                if (h == 0) continue;
                
                int bestZ = -1;
                double maxProfit = -1e9;
                double currentProfit = 0;
                
                for (int z = h - 1; z >= 0; z--)
                {
                    byte d = GetVoxel(x, y, z);
                    int mass = d % 16;
                    int val = d / 16;
                    
                    if (val > 0)
                    {
                        double revenue = mass * val * 1000.0;
                        double fuel = mass * (x + y) * 1.2;
                        currentProfit += (revenue - fuel);
                    }
                    else
                    {
                        double fuel = mass * 2.0;
                        currentProfit -= fuel;
                    }
                    
                    if (currentProfit > maxProfit)
                    {
                        maxProfit = currentProfit;
                        bestZ = z;
                    }
                }
                
                if (maxProfit > 0)
                {
                    candidates.Add(new CandidateColumn
                    {
                        X = x,
                        Y = y,
                        TargetMinDepth = bestZ,
                        EstProfit = maxProfit
                    });
                    isCandidate[x * Y + y] = true;
                }
            }
        }
        
        candidates.Sort((a, b) => b.EstProfit.CompareTo(a.EstProfit));
        
        for (int i = 0; i < candidates.Count; i++)
        {
            if (globalSw.ElapsedMilliseconds > 270000) break;
            
            var cand = candidates[i];
            Point coord = new Point(cand.X, cand.Y);
            
            int currentH = heights[cand.X * Y + cand.Y];
            if (currentH <= cand.TargetMinDepth) continue;
            
            var plantPath = FindPathToPlant(coord, currentH - 1);
            if (!plantPath.Success) continue;
            
            bool ok = true;
            while (heights[cand.X * Y + cand.Y] > cand.TargetMinDepth)
            {
                int digZ = heights[cand.X * Y + cand.Y] - 1;
                byte d = GetVoxel(cand.X, cand.Y, digZ);
                int mass = d % 16;
                int val = d / 16;
                
                if (val > 0)
                {
                    var pathRes = FindPathToPlant(coord, digZ);
                    if (pathRes.Success)
                    {
                        SaveOperation(cand.X, cand.Y, digZ, pathRes.Path, 0, 0);
                        heights[cand.X * Y + cand.Y]--;
                    }
                    else
                    {
                        var dumpRes = FindPathToDump(coord, digZ);
                        if (dumpRes.Success)
                        {
                            var dest = dumpRes.Path[dumpRes.Path.Count - 1];
                            SaveOperation(cand.X, cand.Y, digZ, dumpRes.Path, dest.X, dest.Y);
                            heights[cand.X * Y + cand.Y]--;
                            SetVoxel(dest.X, dest.Y, heights[dest.X * Y + dest.Y], d);
                            heights[dest.X * Y + dest.Y]++;
                        }
                        else
                        {
                            ok = false;
                            break;
                        }
                    }
                }
                else
                {
                    var dumpRes = FindPathToDump(coord, digZ);
                    if (dumpRes.Success)
                    {
                        var dest = dumpRes.Path[dumpRes.Path.Count - 1];
                        SaveOperation(cand.X, cand.Y, digZ, dumpRes.Path, dest.X, dest.Y);
                        heights[cand.X * Y + cand.Y]--;
                        SetVoxel(dest.X, dest.Y, heights[dest.X * Y + dest.Y], d);
                        heights[dest.X * Y + dest.Y]++;
                    }
                    else
                    {
                        ok = false;
                        break;
                    }
                }
            }
            
            if (ok)
            {
                isMinedFinished[cand.X * Y + cand.Y] = true;
            }
        }
        
        using (var stdout = new StreamWriter(new BufferedStream(Console.OpenStandardOutput(), 4 * 1024 * 1024)))
        {
            stdout.WriteLine(ops.Count);
            for (int i = 0; i < ops.Count; i++)
            {
                var op = ops[i];
                stdout.Write($"{op.PathLength} {op.DigX} {op.DigY} {op.DigZ}");
                for (int j = 0; j < op.PathLength; j++)
                {
                    int step = globalSteps[op.PathStart + j];
                    int sx = step >> 16;
                    int sy = step & 0xFFFF;
                    stdout.Write($" {sx} {sy}");
                }
                stdout.Write($" {op.DestX} {op.DestY}\n");
            }
        }
    }
}
