// Adaptive TSP solver.
//
// Strategy by problem size N:
//   N <= 16          : exact Held-Karp dynamic programming
//   N <= 200,000     : grid nearest-neighbour construction + 2-opt / Or-opt
//                      with k-nearest-neighbour candidate lists; iterated
//                      local search (double-bridge kicks) for N <= 50,000
//   N <= 20,000,000  : Hilbert-curve construction + 2-opt / Or-opt with
//                      candidate lists (time budgeted)
//   N  > 20,000,000  : pure Hilbert-curve ordering (multi-threaded quantise,
//                      parallel sort) - O(N log N), handles 1e9 cities
//
// Input : N, then N lines "x y"
// Output: visit order, 0-indexed, starting at city 0, return implicit.

#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <cmath>
#include <cstdint>
#include <vector>
#include <algorithm>
#include <chrono>
#include <thread>
#include <execution>
#ifdef _WIN32
#include <io.h>
#include <fcntl.h>
#endif

using std::vector;
typedef uint32_t u32;
typedef uint64_t u64;
typedef uint8_t u8;

static std::chrono::steady_clock::time_point T0;
static double elapsedS() {
    return std::chrono::duration<double>(std::chrono::steady_clock::now() - T0).count();
}

static int64_t N;
static vector<float> XS, YS;
static const float* Xp;
static const float* Yp;

static inline double dist(u32 a, u32 b) {
    double dx = (double)Xp[a] - (double)Xp[b];
    double dy = (double)Yp[a] - (double)Yp[b];
    return sqrt(dx * dx + dy * dy);
}

// ----------------------------- fast input ---------------------------------

static inline double parseF(const char*& p, const char* end) {
    while (p < end && (*p == ' ' || *p == '\n' || *p == '\r' || *p == '\t' || *p == ',')) ++p;
    bool neg = false;
    if (p < end && (*p == '+' || *p == '-')) { neg = (*p == '-'); ++p; }
    double v = 0;
    while (p < end && *p >= '0' && *p <= '9') { v = v * 10 + (*p - '0'); ++p; }
    if (p < end && *p == '.') {
        ++p;
        double f = 0, s = 1;
        while (p < end && *p >= '0' && *p <= '9') { f = f * 10 + (*p - '0'); s *= 10; ++p; }
        v += f / s;
    }
    if (p < end && (*p == 'e' || *p == 'E')) {
        ++p;
        bool en = false;
        if (p < end && (*p == '+' || *p == '-')) { en = (*p == '-'); ++p; }
        int ex = 0;
        while (p < end && *p >= '0' && *p <= '9') { ex = ex * 10 + (*p - '0'); ++p; }
        v *= pow(10.0, en ? -ex : ex);
    }
    return neg ? -v : v;
}

// ----------------------------- fast output --------------------------------

static vector<char> OB;
static size_t OI = 0;
static inline void flushOB() { fwrite(OB.data(), 1, OI, stdout); OI = 0; }
static inline void outNum(u64 v, char sep) {
    if (OI + 24 > OB.size()) flushOB();
    char t[20];
    int l = 0;
    do { t[l++] = (char)('0' + v % 10); v /= 10; } while (v);
    while (l) OB[OI++] = t[--l];
    OB[OI++] = sep;
}

static void outputTour(const vector<u32>& tour) {
    size_t n = tour.size(), z = 0;
    while (tour[z] != 0) ++z;
    for (size_t k = 0; k < n; k++) {
        size_t i = z + k;
        if (i >= n) i -= n;
        outNum(tour[i], (k + 1 == n) ? '\n' : ' ');
    }
    flushOB();
    fflush(stdout);
}

// ------------------------------ Held-Karp ---------------------------------

static void heldKarp() {
    int n = (int)N;
    int full = 1 << n;
    vector<double> dp((size_t)full * n, 1e30);
    vector<u8> par((size_t)full * n, 0);
    dp[(size_t)1 * n + 0] = 0.0;
    for (int mask = 1; mask < full; mask++) {
        if (!(mask & 1)) continue;
        for (int i = 0; i < n; i++) {
            if (!((mask >> i) & 1)) continue;
            double di = dp[(size_t)mask * n + i];
            if (di >= 1e29) continue;
            for (int j = 1; j < n; j++) {
                if ((mask >> j) & 1) continue;
                int m2 = mask | (1 << j);
                double v = di + dist((u32)i, (u32)j);
                if (v < dp[(size_t)m2 * n + j]) {
                    dp[(size_t)m2 * n + j] = v;
                    par[(size_t)m2 * n + j] = (u8)i;
                }
            }
        }
    }
    int fullm = full - 1;
    double bestv = 1e30;
    int bi = (n > 1) ? 1 : 0;
    for (int i = 1; i < n; i++) {
        double v = dp[(size_t)fullm * n + i] + dist((u32)i, 0);
        if (v < bestv) { bestv = v; bi = i; }
    }
    vector<int> order;
    int mask = fullm, i = bi;
    for (;;) {
        order.push_back(i);
        if (i == 0) break;
        int pi = par[(size_t)mask * n + i];
        mask ^= 1 << i;
        i = pi;
    }
    std::reverse(order.begin(), order.end()); // now starts at 0
    for (size_t k = 0; k < order.size(); k++)
        outNum((u64)order[k], (k + 1 == order.size()) ? '\n' : ' ');
    flushOB();
    fflush(stdout);
}

// ------------------------------ spatial grid -------------------------------

struct Grid {
    int nx = 1, ny = 1;
    double minx = 0, miny = 0, cell = 1.0;
    vector<u32> start; // size nx*ny+1, CSR offsets
    vector<u32> items; // size N, point ids bucketed by cell
    inline int cx(double x) const {
        int c = (int)((x - minx) / cell);
        if (c < 0) c = 0;
        if (c >= nx) c = nx - 1;
        return c;
    }
    inline int cy(double y) const {
        int c = (int)((y - miny) / cell);
        if (c < 0) c = 0;
        if (c >= ny) c = ny - 1;
        return c;
    }
};

template <class F>
static inline void forRing(const Grid& g, int cx0, int cy0, int r, F&& f) {
    if (r == 0) {
        if ((unsigned)cx0 < (unsigned)g.nx && (unsigned)cy0 < (unsigned)g.ny) f(cx0, cy0);
        return;
    }
    int x0 = cx0 - r, x1 = cx0 + r, y0 = cy0 - r, y1 = cy0 + r;
    for (int x = x0; x <= x1; x++) {
        if ((unsigned)x >= (unsigned)g.nx) continue;
        if ((unsigned)y0 < (unsigned)g.ny) f(x, y0);
        if ((unsigned)y1 < (unsigned)g.ny) f(x, y1);
    }
    for (int y = y0 + 1; y <= y1 - 1; y++) {
        if ((unsigned)y >= (unsigned)g.ny) continue;
        if ((unsigned)x0 < (unsigned)g.nx) f(x0, y);
        if ((unsigned)x1 < (unsigned)g.nx) f(x1, y);
    }
}

static void buildGrid(Grid& g) {
    size_t n = (size_t)N;
    double minx = 1e300, miny = 1e300, maxx = -1e300, maxy = -1e300;
    for (size_t i = 0; i < n; i++) {
        double x = Xp[i], y = Yp[i];
        if (x < minx) minx = x;
        if (x > maxx) maxx = x;
        if (y < miny) miny = y;
        if (y > maxy) maxy = y;
    }
    double w = maxx - minx, h = maxy - miny;
    double cell;
    if (w > 0 && h > 0) cell = sqrt(w * h * 2.0 / (double)n);
    else cell = std::max(w, h) / std::max(1.0, sqrt((double)n / 2.0));
    if (!(cell > 0) || !std::isfinite(cell)) cell = 1.0;
    int nx, ny;
    for (;;) {
        nx = (int)(w / cell) + 1;
        ny = (int)(h / cell) + 1;
        if (nx < 1) nx = 1;
        if (ny < 1) ny = 1;
        if ((double)nx * (double)ny <= 4.0 * (double)n + 64.0) break;
        cell *= 1.3;
    }
    g.nx = nx; g.ny = ny; g.minx = minx; g.miny = miny; g.cell = cell;
    size_t cells = (size_t)nx * (size_t)ny;
    g.start.assign(cells + 1, 0);
    vector<u32> cidx(n);
    for (size_t i = 0; i < n; i++) {
        int cx = g.cx(Xp[i]), cy = g.cy(Yp[i]);
        cidx[i] = (u32)((size_t)cy * nx + cx);
        g.start[cidx[i] + 1]++;
    }
    for (size_t c = 0; c < cells; c++) g.start[c + 1] += g.start[c];
    g.items.resize(n);
    vector<u32> fill(g.start.begin(), g.start.end() - 1);
    for (size_t i = 0; i < n; i++) g.items[fill[cidx[i]]++] = (u32)i;
}

// ------------------------------ kNN lists ----------------------------------

static void knnOne(const Grid& g, u32 i, int K, u32* out) {
    double px = Xp[i], py = Yp[i];
    int cx0 = g.cx(px), cy0 = g.cy(py);
    double nd[32];
    u32 ni[32];
    int cnt = 0;
    int rmax = std::max(g.nx, g.ny);
    for (int r = 0; r <= rmax; r++) {
        if (cnt == K) {
            double bound = ((double)r - 1.0) * g.cell;
            if (bound > 0 && nd[cnt - 1] <= bound * bound) break;
        }
        forRing(g, cx0, cy0, r, [&](int ax, int ay) {
            size_t c = (size_t)ay * g.nx + ax;
            for (u32 t = g.start[c]; t < g.start[c + 1]; t++) {
                u32 j = g.items[t];
                if (j == i) continue;
                double dx = Xp[j] - px, dy = Yp[j] - py;
                double d2 = dx * dx + dy * dy;
                if (cnt < K) {
                    int q = cnt++;
                    while (q > 0 && nd[q - 1] > d2) { nd[q] = nd[q - 1]; ni[q] = ni[q - 1]; q--; }
                    nd[q] = d2; ni[q] = j;
                } else if (d2 < nd[K - 1]) {
                    int q = K - 1;
                    while (q > 0 && nd[q - 1] > d2) { nd[q] = nd[q - 1]; ni[q] = ni[q - 1]; q--; }
                    nd[q] = d2; ni[q] = j;
                }
            }
        });
    }
    for (int q = 0; q < K; q++) out[q] = (q < cnt) ? ni[q] : UINT32_MAX;
}

static void buildKNN(const Grid& g, int K, vector<u32>& nbr) {
    size_t n = (size_t)N;
    nbr.resize(n * (size_t)K);
    unsigned T = std::thread::hardware_concurrency();
    if (T == 0) T = 4;
    if (T > 64) T = 64;
    if (n < 100000) T = 1;
    if (T == 1) {
        for (size_t i = 0; i < n; i++) knnOne(g, (u32)i, K, &nbr[i * (size_t)K]);
        return;
    }
    vector<std::thread> th;
    size_t chunk = (n + T - 1) / T;
    for (unsigned t = 0; t < T; t++) {
        size_t lo = (size_t)t * chunk, hi = std::min(n, lo + chunk);
        if (lo >= hi) break;
        th.emplace_back([&g, K, &nbr, lo, hi]() {
            for (size_t i = lo; i < hi; i++) knnOne(g, (u32)i, K, &nbr[i * (size_t)K]);
        });
    }
    for (auto& x : th) x.join();
}

// -------------------------- nearest-neighbour tour --------------------------

static void nnTour(const Grid& g, vector<u32>& tour) {
    u32 n = (u32)N;
    vector<u32> itemsM = g.items;
    vector<u32> posIn(n), cellOf(n);
    size_t cells = (size_t)g.nx * (size_t)g.ny;
    vector<u32> alive(cells);
    for (size_t c = 0; c < cells; c++) alive[c] = g.start[c + 1] - g.start[c];
    for (u32 idx = 0; idx < n; idx++) posIn[itemsM[idx]] = idx;
    for (u32 i = 0; i < n; i++)
        cellOf[i] = (u32)((size_t)g.cy(Yp[i]) * g.nx + g.cx(Xp[i]));

    auto removePt = [&](u32 p) {
        u32 c = cellOf[p];
        u32 last = g.start[c] + (--alive[c]);
        u32 pi = posIn[p];
        u32 q = itemsM[last];
        itemsM[pi] = q; posIn[q] = pi;
        itemsM[last] = p; posIn[p] = last;
    };

    tour.resize(n);
    u32 cur = 0;
    removePt(0);
    tour[0] = 0;
    int rmax = std::max(g.nx, g.ny);
    for (u32 s = 1; s < n; s++) {
        double px = Xp[cur], py = Yp[cur];
        int cx0 = g.cx(px), cy0 = g.cy(py);
        double best = 1e300;
        u32 bi = UINT32_MAX;
        for (int r = 0; r <= rmax; r++) {
            if (bi != UINT32_MAX) {
                double bound = ((double)r - 1.0) * g.cell;
                if (bound > 0 && best <= bound * bound) break;
            }
            forRing(g, cx0, cy0, r, [&](int ax, int ay) {
                size_t c = (size_t)ay * g.nx + ax;
                u32 b0 = g.start[c], b1 = b0 + alive[c];
                for (u32 t = b0; t < b1; t++) {
                    u32 j = itemsM[t];
                    double dx = Xp[j] - px, dy = Yp[j] - py;
                    double d2 = dx * dx + dy * dy;
                    if (d2 < best) { best = d2; bi = j; }
                }
            });
        }
        removePt(bi);
        tour[s] = bi;
        cur = bi;
    }
}

// ------------------------------ Hilbert tour --------------------------------

static inline u64 hilbertD(u32 x, u32 y) {
    u64 d = 0;
    for (u32 s = 1u << 15; s > 0; s >>= 1) {
        u32 rx = (x & s) ? 1u : 0u;
        u32 ry = (y & s) ? 1u : 0u;
        d += (u64)s * (u64)s * (u64)((3u * rx) ^ ry);
        if (ry == 0) {
            if (rx == 1) { x = 65535u - x; y = 65535u - y; }
            u32 t = x; x = y; y = t;
        }
    }
    return d;
}

static void hilbertTour(vector<u32>& tour) {
    size_t n = (size_t)N;
    double minx = 1e300, miny = 1e300, maxx = -1e300, maxy = -1e300;
    for (size_t i = 0; i < n; i++) {
        double x = Xp[i], y = Yp[i];
        if (x < minx) minx = x;
        if (x > maxx) maxx = x;
        if (y < miny) miny = y;
        if (y > maxy) maxy = y;
    }
    double w = maxx - minx, h = maxy - miny;
    double sx = (w > 0) ? 65535.0 / w : 0.0;
    double sy = (h > 0) ? 65535.0 / h : 0.0;
    vector<u64> keys(n);
    auto work = [&](size_t lo, size_t hi) {
        for (size_t i = lo; i < hi; i++) {
            double qxf = ((double)Xp[i] - minx) * sx;
            double qyf = ((double)Yp[i] - miny) * sy;
            u32 qx = (qxf <= 0) ? 0u : (qxf >= 65535.0 ? 65535u : (u32)qxf);
            u32 qy = (qyf <= 0) ? 0u : (qyf >= 65535.0 ? 65535u : (u32)qyf);
            keys[i] = (hilbertD(qx, qy) << 32) | (u64)i;
        }
    };
    unsigned T = std::thread::hardware_concurrency();
    if (T == 0) T = 4;
    if (T > 64) T = 64;
    if (n < 1000000 || T == 1) {
        work(0, n);
    } else {
        vector<std::thread> th;
        size_t chunk = (n + T - 1) / T;
        for (unsigned t = 0; t < T; t++) {
            size_t lo = (size_t)t * chunk, hi = std::min(n, lo + chunk);
            if (lo >= hi) break;
            th.emplace_back([&work, lo, hi]() { work(lo, hi); });
        }
        for (auto& x : th) x.join();
    }
    if (n >= 1000000)
        std::sort(std::execution::par_unseq, keys.begin(), keys.end());
    else
        std::sort(keys.begin(), keys.end());
    tour.resize(n);
    for (size_t i = 0; i < n; i++) tour[i] = (u32)(keys[i] & 0xffffffffu);
}

// ------------------------------ local search --------------------------------

static double tourLen(const vector<u32>& tour) {
    size_t n = tour.size();
    double L = 0;
    for (size_t i = 0; i < n; i++) L += dist(tour[i], tour[(i + 1 == n) ? 0 : i + 1]);
    return L;
}

static inline void cycRev(vector<u32>& tour, vector<u32>& pos, u32 i, u32 j, u32 n) {
    u32 len = ((j >= i) ? (j - i) : (j + n - i)) + 1;
    for (u32 k = 0; k < len / 2; k++) {
        u32 p = i + k; if (p >= n) p -= n;
        u32 q = (j >= k) ? (j - k) : (j + n - k);
        u32 a = tour[p], b = tour[q];
        tour[p] = b; tour[q] = a;
        pos[b] = p; pos[a] = q;
    }
}

// 2-opt with neighbour candidate lists and don't-look bits.
static double twoOptRun(vector<u32>& tour, vector<u32>& pos, const u32* nbr, int K,
                        double deadline, vector<u32>& stk, vector<u8>& inS,
                        u32 revLimit, double epsGain) {
    u32 n = (u32)tour.size();
    double total = 0;
    u64 ctr = 0;
    auto push = [&](u32 v) { if (!inS[v]) { inS[v] = 1; stk.push_back(v); } };
    while (!stk.empty()) {
        if (((++ctr) & 1023u) == 0 && elapsedS() > deadline) break;
        u32 a = stk.back(); stk.pop_back(); inS[a] = 0;
        bool again = true;
        while (again) {
            again = false;
            for (int dir = 0; dir < 2 && !again; dir++) {
                u32 pa = pos[a];
                u32 b = (dir == 0) ? tour[(pa + 1 == n) ? 0 : pa + 1]
                                   : tour[(pa == 0) ? n - 1 : pa - 1];
                double dab = dist(a, b);
                const u32* nb = nbr + (size_t)a * K;
                for (int t = 0; t < K; t++) {
                    u32 c = nb[t];
                    if (c == UINT32_MAX) break;
                    double dac = dist(a, c);
                    if (dac >= dab) break;
                    u32 pc = pos[c];
                    u32 d = (dir == 0) ? tour[(pc + 1 == n) ? 0 : pc + 1]
                                       : tour[(pc == 0) ? n - 1 : pc - 1];
                    if (c == b || d == a) continue;
                    double delta = dab + dist(c, d) - dac - dist(b, d);
                    if (delta > epsGain) {
                        u32 i, j;
                        if (dir == 0) { i = (pa + 1 == n) ? 0 : pa + 1; j = pc; }
                        else          { i = pa; j = (pc == 0) ? n - 1 : pc - 1; }
                        u32 len = ((j >= i) ? (j - i) : (j + n - i)) + 1;
                        u32 other = n - len;
                        if (std::min(len, other) > revLimit) continue;
                        if (len <= other) cycRev(tour, pos, i, j, n);
                        else cycRev(tour, pos, (j + 1 == n) ? 0 : j + 1, (i == 0) ? n - 1 : i - 1, n);
                        total += delta;
                        push(a); push(b); push(c); push(d);
                        again = true;
                        break;
                    }
                }
            }
        }
    }
    for (u32 v : stk) inS[v] = 0;
    stk.clear();
    return total;
}

// Or-opt (segment relocation, length 1..3, forward or reversed) on a
// doubly-linked-list tour representation. O(1) per applied move.
static double orOptRun(vector<u32>& nxt, vector<u32>& prv, const u32* nbr, int K,
                       double deadline, vector<u32>& stk, vector<u8>& inS,
                       double epsGain, u32 n) {
    if (n < 6) {
        for (u32 v : stk) inS[v] = 0;
        stk.clear();
        return 0;
    }
    double total = 0;
    u64 ctr = 0;
    auto push = [&](u32 v) { if (!inS[v]) { inS[v] = 1; stk.push_back(v); } };
    while (!stk.empty()) {
        if (((++ctr) & 511u) == 0 && elapsedS() > deadline) break;
        u32 s = stk.back(); stk.pop_back(); inS[s] = 0;
        bool moved = true;
        while (moved) {
            moved = false;
            u32 seg[3];
            seg[0] = s;
            u32 e = s;
            for (int L = 1; L <= 3 && !moved; L++) {
                if (L > 1) {
                    e = nxt[e];
                    seg[L - 1] = e;
                    if (e == prv[s]) break;
                }
                u32 pa = prv[s], ne = nxt[e];
                if (ne == s || pa == e) break;
                double base = dist(pa, s) + dist(e, ne) - dist(pa, ne);
                if (base + dist(s, e) <= epsGain) continue;
                double bestG = epsGain;
                u32 bestC = UINT32_MAX;
                bool bestRev = false;
                for (int side = 0; side < 2; side++) {
                    u32 bn = (side == 0) ? s : e;
                    if (side == 1 && e == s) break;
                    const u32* nb = nbr + (size_t)bn * K;
                    for (int t = 0; t < K; t++) {
                        u32 c = nb[t];
                        if (c == UINT32_MAX) break;
                        if (c == pa) continue;
                        bool inseg = false;
                        for (int q = 0; q < L; q++) if (seg[q] == c) { inseg = true; break; }
                        if (inseg) continue;
                        u32 nc = nxt[c];
                        double dcnc = dist(c, nc);
                        double gf = base + dcnc - dist(c, s) - dist(e, nc);
                        double gr = base + dcnc - dist(c, e) - dist(s, nc);
                        if (gf > bestG) { bestG = gf; bestC = c; bestRev = false; }
                        if (gr > bestG) { bestG = gr; bestC = c; bestRev = true; }
                    }
                }
                if (bestC != UINT32_MAX) {
                    u32 c = bestC, nc = nxt[c];
                    nxt[pa] = ne; prv[ne] = pa;
                    u32 ord[3];
                    for (int q = 0; q < L; q++) ord[q] = bestRev ? seg[L - 1 - q] : seg[q];
                    u32 prevn = c;
                    for (int q = 0; q < L; q++) { nxt[prevn] = ord[q]; prv[ord[q]] = prevn; prevn = ord[q]; }
                    nxt[prevn] = nc; prv[nc] = prevn;
                    total += bestG;
                    push(pa); push(ne); push(c); push(nc); push(s); push(e);
                    moved = true;
                }
            }
        }
    }
    for (u32 v : stk) inS[v] = 0;
    stk.clear();
    return total;
}

// Iterated local search: double-bridge kick + localized 2-opt repair.
static void ils(vector<u32>& tour, vector<u32>& pos, const u32* nbr, int K,
                double deadline, double epsGain, double& curLen,
                vector<u32>& stk, vector<u8>& inS, u32 revLimit) {
    u32 n = (u32)tour.size();
    if (n < 8) return;
    vector<u32> best = tour;
    double bestLen = curLen;
    vector<u32> tmp(n);
    u64 rng = 0x243F6A8885A308D3ULL;
    auto rnd = [&]() { rng ^= rng << 13; rng ^= rng >> 7; rng ^= rng << 17; return rng; };
    while (elapsedS() < deadline) {
        u32 r[3];
        r[0] = 1 + (u32)(rnd() % (n - 1));
        r[1] = 1 + (u32)(rnd() % (n - 1));
        r[2] = 1 + (u32)(rnd() % (n - 1));
        std::sort(r, r + 3);
        if (r[0] == r[1] || r[1] == r[2]) continue;
        u32 p1 = r[0], p2 = r[1], p3 = r[2];
        u32 A1 = tour[p1 - 1], B1 = tour[p1], B2 = tour[p2 - 1];
        u32 C1 = tour[p2], C2 = tour[p3 - 1], D1 = tour[p3];
        double delta = dist(A1, C1) + dist(C2, B1) + dist(B2, D1)
                     - dist(A1, B1) - dist(B2, C1) - dist(C2, D1);
        size_t idx = 0;
        memcpy(tmp.data(), tour.data(), (size_t)p1 * sizeof(u32)); idx = p1;
        memcpy(tmp.data() + idx, tour.data() + p2, (size_t)(p3 - p2) * sizeof(u32)); idx += p3 - p2;
        memcpy(tmp.data() + idx, tour.data() + p1, (size_t)(p2 - p1) * sizeof(u32)); idx += p2 - p1;
        memcpy(tmp.data() + idx, tour.data() + p3, (size_t)(n - p3) * sizeof(u32));
        tour.swap(tmp);
        for (u32 i = 0; i < n; i++) pos[tour[i]] = i;
        curLen += delta;
        u32 seeds[6] = { A1, B1, B2, C1, C2, D1 };
        for (u32 v : seeds) if (!inS[v]) { inS[v] = 1; stk.push_back(v); }
        curLen -= twoOptRun(tour, pos, nbr, K, deadline, stk, inS, revLimit, epsGain);
        if (curLen < bestLen - 1e-9) {
            bestLen = curLen;
            best = tour;
        } else {
            tour = best;
            for (u32 i = 0; i < n; i++) pos[tour[i]] = i;
            curLen = bestLen;
        }
    }
    tour = best;
    curLen = bestLen;
}

static void improve(vector<u32>& tour, const Grid& g, const vector<u32>& nbr, int K,
                    double deadline) {
    u32 n = (u32)tour.size();
    vector<u32> pos(n);
    for (u32 i = 0; i < n; i++) pos[tour[i]] = i;
    vector<u8> inS(n, 0);
    vector<u32> stk;
    stk.reserve(n);
    double epsGain = g.cell * 1e-7 + 1e-12;
    u32 revLimit = (n <= 2000000u) ? n : 1000000u;
    double prevLen = tourLen(tour);
    for (int round = 0; round < 6; round++) {
        if (elapsedS() > deadline) break;
        for (u32 i = 0; i < n; i++) {
            u32 v = tour[i];
            if (!inS[v]) { inS[v] = 1; stk.push_back(v); }
        }
        twoOptRun(tour, pos, nbr.data(), K, deadline, stk, inS, revLimit, epsGain);
        if (elapsedS() > deadline) break;
        {
            vector<u32> nxt(n), prv(n);
            for (u32 i = 0; i < n; i++) {
                u32 a = tour[i], b = tour[(i + 1 == n) ? 0 : i + 1];
                nxt[a] = b; prv[b] = a;
            }
            for (u32 i = 0; i < n; i++) {
                u32 v = tour[i];
                if (!inS[v]) { inS[v] = 1; stk.push_back(v); }
            }
            orOptRun(nxt, prv, nbr.data(), K, deadline, stk, inS, epsGain, n);
            u32 c = 0;
            for (u32 i = 0; i < n; i++) { tour[i] = c; pos[c] = i; c = nxt[c]; }
        }
        double len = tourLen(tour);
        if (prevLen - len <= prevLen * 1e-7 + epsGain) { prevLen = len; break; }
        prevLen = len;
    }
    if (n <= 200000) {
        double budget = (n <= 100) ? 5.0 : (n <= 2000) ? 20.0 : (n <= 20000) ? 60.0 : 90.0;
        double ilsDeadline = std::min(deadline, budget);
        if (elapsedS() < ilsDeadline)
            ils(tour, pos, nbr.data(), K, ilsDeadline, epsGain, prevLen, stk, inS, revLimit);
    }
}

int main() {
    T0 = std::chrono::steady_clock::now();
#ifdef _WIN32
    _setmode(_fileno(stdin), _O_BINARY);
    _setmode(_fileno(stdout), _O_BINARY);
#endif
    // Read all of stdin.
    size_t cap = (size_t)1 << 26, len = 0;
    char* buf = (char*)malloc(cap);
    if (!buf) return 1;
    for (;;) {
        if (len == cap) {
            cap <<= 1;
            char* nb = (char*)realloc(buf, cap);
            if (!nb) { free(buf); return 1; }
            buf = nb;
        }
        size_t r = fread(buf + len, 1, cap - len, stdin);
        if (r == 0) break;
        len += r;
    }
    const char* p = buf;
    const char* end = buf + len;
    N = (int64_t)parseF(p, end);
    OB.resize((size_t)1 << 22);
    if (N <= 0) { fputs("\n", stdout); free(buf); return 0; }
    XS.resize((size_t)N);
    YS.resize((size_t)N);
    for (int64_t i = 0; i < N; i++) {
        XS[(size_t)i] = (float)parseF(p, end);
        YS[(size_t)i] = (float)parseF(p, end);
    }
    free(buf);
    Xp = XS.data();
    Yp = YS.data();

    if (N == 1) { fputs("0\n", stdout); return 0; }
    if (N == 2) { fputs("0 1\n", stdout); return 0; }
    if (N <= 16) { heldKarp(); return 0; }

    const double IMPROVE_DEADLINE = 235.0;   // seconds of total runtime
    const int64_t IMPROVE_MAX = 20000000;    // local search only up to here

    vector<u32> tour;
    Grid g;
    bool gridBuilt = false;
    if (N <= 200000) {
        buildGrid(g);
        gridBuilt = true;
        nnTour(g, tour);
    } else {
        hilbertTour(tour);
    }

    if (N <= IMPROVE_MAX) {
        if (!gridBuilt) buildGrid(g);
        int K = (N <= 1000) ? 16 : (N <= 100000) ? 10 : (N <= 2000000) ? 8 : 6;
        if (K >= N) K = (int)N - 1;
        vector<u32> nbr;
        buildKNN(g, K, nbr);
        improve(tour, g, nbr, K, IMPROVE_DEADLINE);
    }

    outputTour(tour);
    return 0;
}
