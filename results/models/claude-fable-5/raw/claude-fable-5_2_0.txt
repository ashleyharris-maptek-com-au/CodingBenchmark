// Chinese Postman Problem solver.
//
// Strategy (adapts to scale):
//  1. Read graph, find odd-degree vertices T.
//  2. Make all degrees even by duplicating edges:
//     - |T| <= 16: exact minimum-weight matching over T (bitmask DP on
//       pairwise shortest-path distances), duplicate the shortest paths.
//     - moderate |T| (bounded work): greedy matching on pairwise
//       shortest-path distances (parallel Dijkstra, 64 cores available).
//     - huge |T| (epic graphs): linear-time T-join on a minimum spanning
//       tree - duplicate each MST edge whose subtree holds an odd number
//       of T-vertices.
//  3. Iterative Hierholzer Euler circuit from node 0 over the multigraph.
using System;
using System.Collections.Generic;
using System.IO;
using System.Threading.Tasks;

public static class Program
{
    static int N, M;
    static int[] eu, ev;
    static long[] ew;
    // CSR of the original graph
    static int[] off, ato, aeid;
    static long[] aw;

    public static void Main()
    {
        byte[] data; int len;
        using (var stdin = Console.OpenStandardInput())
        using (var ms = new MemoryStream())
        {
            stdin.CopyTo(ms, 1 << 20);
            data = ms.GetBuffer(); len = (int)ms.Length;
        }
        int pos = 0;
        N = NextInt(data, len, ref pos);
        M = NextInt(data, len, ref pos);
        if (N <= 0) N = 1;
        eu = new int[M]; ev = new int[M]; ew = new long[M];
        var deg = new int[N];
        for (int i = 0; i < M; i++)
        {
            int u = NextInt(data, len, ref pos);
            int v = NextInt(data, len, ref pos);
            long w = NextInt(data, len, ref pos);
            eu[i] = u; ev[i] = v; ew[i] = w;
            deg[u]++; deg[v]++;
        }

        var outStream = Console.OpenStandardOutput();
        if (M == 0)
        {
            var b = System.Text.Encoding.ASCII.GetBytes("0\n");
            outStream.Write(b, 0, b.Length); outStream.Flush();
            return;
        }

        BuildCsr(deg);

        var oddList = new List<int>();
        for (int v = 0; v < N; v++) if ((deg[v] & 1) != 0) oddList.Add(v);
        int t = oddList.Count;

        var dup = new List<int>(); // original edge ids to duplicate
        if (t > 0)
        {
            bool exact = t <= 16;
            bool matrixOk = t <= 1024
                && (long)t * M <= 1000000000L
                && (long)t * N <= 200000000L;
            if (exact || matrixOk) MatchViaShortestPaths(oddList.ToArray(), exact, dup);
            else TreeJoin(oddList, dup);
        }

        // ----- build multigraph (original + duplicated edges) -----
        int total = M + dup.Count;
        var mu = new int[total]; var mv = new int[total];
        Array.Copy(eu, mu, M); Array.Copy(ev, mv, M);
        for (int j = 0; j < dup.Count; j++)
        {
            int e = dup[j];
            mu[M + j] = eu[e]; mv[M + j] = ev[e];
        }
        var hoff = new int[N + 1];
        var hc = new int[N];
        for (int i = 0; i < total; i++) { hc[mu[i]]++; hc[mv[i]]++; }
        for (int v = 0; v < N; v++) hoff[v + 1] = hoff[v] + hc[v];
        var cur = new int[N];
        Array.Copy(hoff, cur, N);
        var hto = new int[2 * total]; var heid = new int[2 * total];
        for (int i = 0; i < total; i++)
        {
            int u = mu[i], v = mv[i];
            hto[cur[u]] = v; heid[cur[u]++] = i;
            hto[cur[v]] = u; heid[cur[v]++] = i;
        }

        // ----- Hierholzer (iterative) -----
        var used = new bool[total];
        var ptr = new int[N];
        Array.Copy(hoff, ptr, N);
        var stack = new int[total + 2]; int sp = 0;
        var circuit = new int[total + 1]; int cn = 0;
        stack[sp++] = 0;
        while (sp > 0)
        {
            int v = stack[sp - 1];
            int i = ptr[v];
            int end = hoff[v + 1];
            bool advanced = false;
            while (i < end)
            {
                int e = heid[i];
                if (!used[e])
                {
                    used[e] = true;
                    ptr[v] = i + 1;
                    stack[sp++] = hto[i];
                    advanced = true;
                    break;
                }
                i++;
            }
            if (!advanced)
            {
                ptr[v] = i;
                circuit[cn++] = v;
                sp--;
            }
        }

        WriteOut(outStream, circuit, cn);
    }

    static void BuildCsr(int[] deg)
    {
        off = new int[N + 1];
        for (int v = 0; v < N; v++) off[v + 1] = off[v] + deg[v];
        var cur = new int[N];
        Array.Copy(off, cur, N);
        ato = new int[2 * M]; aeid = new int[2 * M]; aw = new long[2 * M];
        for (int e = 0; e < M; e++)
        {
            int u = eu[e], v = ev[e];
            ato[cur[u]] = v; aw[cur[u]] = ew[e]; aeid[cur[u]++] = e;
            ato[cur[v]] = u; aw[cur[v]] = ew[e]; aeid[cur[v]++] = e;
        }
    }

    // ----- matching-based parity fix (small/medium odd sets) -----
    static void MatchViaShortestPaths(int[] odd, bool exact, List<int> dup)
    {
        int t = odd.Length;
        var dmat = new long[t * t];
        Parallel.For(0, t,
            () => new Dij(N),
            (i, _, st) =>
            {
                st.Run(odd[i], -1, false);
                for (int j = 0; j < t; j++) dmat[i * t + j] = st.dist[odd[j]];
                return st;
            },
            _ => { });

        var pu = new List<int>(); var pv = new List<int>();
        const long INF = long.MaxValue / 4;
        if (exact)
        {
            int full = (1 << t) - 1;
            var f = new long[1 << t];
            var choice = new int[1 << t];
            for (int mask = 1; mask <= full; mask++)
            {
                f[mask] = INF;
                int i = TrailingZeros(mask);
                int rest = mask & ~(1 << i);
                for (int j = i + 1; j < t; j++)
                {
                    if (((rest >> j) & 1) == 0) continue;
                    long prev = f[rest & ~(1 << j)];
                    if (prev >= INF) continue;
                    long d = dmat[i * t + j];
                    if (d >= INF) continue;
                    long cand = prev + d;
                    if (cand < f[mask]) { f[mask] = cand; choice[mask] = j; }
                }
            }
            int m2 = full;
            while (m2 != 0 && f[m2] < INF)
            {
                int i = TrailingZeros(m2);
                int j = choice[m2];
                pu.Add(i); pv.Add(j);
                m2 &= ~(1 << i); m2 &= ~(1 << j);
            }
        }
        else
        {
            int P = t * (t - 1) / 2;
            var keys = new long[P]; var ids = new int[P];
            int k = 0;
            for (int i = 0; i < t; i++)
                for (int j = i + 1; j < t; j++)
                { keys[k] = dmat[i * t + j]; ids[k] = i * t + j; k++; }
            Array.Sort(keys, ids);
            var matched = new bool[t];
            int rem = t;
            for (int q = 0; q < P && rem > 0; q++)
            {
                if (keys[q] >= INF) break;
                int i = ids[q] / t, j = ids[q] % t;
                if (matched[i] || matched[j]) continue;
                matched[i] = matched[j] = true;
                pu.Add(i); pv.Add(j); rem -= 2;
            }
        }

        int pc = pu.Count;
        var puA = pu.ToArray(); var pvA = pv.ToArray();
        var lists = new List<int>[pc];
        Parallel.For(0, pc,
            () => new Dij(N),
            (kk, _, st) =>
            {
                int s = odd[puA[kk]], g = odd[pvA[kk]];
                st.Run(s, g, true);
                var lst = new List<int>();
                int x = g;
                while (x != s)
                {
                    int e = st.parE[x];
                    if (e < 0) { lst.Clear(); break; }
                    lst.Add(e);
                    x = eu[e] == x ? ev[e] : eu[e];
                }
                lists[kk] = lst;
                return st;
            },
            _ => { });
        for (int kk = 0; kk < pc; kk++) dup.AddRange(lists[kk]);
    }

    // ----- MST-based T-join (huge odd sets) -----
    static void TreeJoin(List<int> oddList, List<int> dup)
    {
        var order = new int[M];
        for (int i = 0; i < M; i++) order[i] = i;
        var keys = (long[])ew.Clone();
        Array.Sort(keys, order);

        var dsu = new int[N];
        for (int i = 0; i < N; i++) dsu[i] = i;

        var tcnt = new int[N];
        var treeEdges = new List<int>(Math.Min(M, N));
        foreach (int e in order)
        {
            int a = eu[e]; while (dsu[a] != a) { dsu[a] = dsu[dsu[a]]; a = dsu[a]; }
            int b = ev[e]; while (dsu[b] != b) { dsu[b] = dsu[dsu[b]]; b = dsu[b]; }
            if (a == b) continue;
            dsu[a] = b;
            treeEdges.Add(e);
            tcnt[eu[e]]++; tcnt[ev[e]]++;
        }

        var toff = new int[N + 1];
        for (int v = 0; v < N; v++) toff[v + 1] = toff[v] + tcnt[v];
        var cur = new int[N];
        Array.Copy(toff, cur, N);
        int ta = treeEdges.Count * 2;
        var tto = new int[ta]; var teid = new int[ta];
        foreach (int e in treeEdges)
        {
            int u = eu[e], v = ev[e];
            tto[cur[u]] = v; teid[cur[u]++] = e;
            tto[cur[v]] = u; teid[cur[v]++] = e;
        }

        var parity = new bool[N];
        foreach (int v in oddList) parity[v] = true;
        var parent = new int[N]; var parentE = new int[N];
        var ord = new int[N]; int on = 0;
        var vis = new bool[N];
        var stk = new int[N];
        for (int r = 0; r < N; r++)
        {
            if (vis[r]) continue;
            int sp = 0; stk[sp++] = r; vis[r] = true;
            parent[r] = -1; parentE[r] = -1;
            while (sp > 0)
            {
                int v = stk[--sp];
                ord[on++] = v;
                for (int a = toff[v]; a < toff[v + 1]; a++)
                {
                    int w = tto[a];
                    if (vis[w]) continue;
                    vis[w] = true; parent[w] = v; parentE[w] = teid[a];
                    stk[sp++] = w;
                }
            }
        }
        // children appear after parents in DFS preorder, so reverse order
        // pushes parity up the tree correctly.
        for (int k = on - 1; k >= 0; k--)
        {
            int v = ord[k];
            if (parity[v] && parent[v] >= 0)
            {
                dup.Add(parentE[v]);
                parity[parent[v]] ^= true;
            }
        }
    }

    sealed class Dij
    {
        public long[] dist;
        public int[] parE;
        long[] hd; int[] hv; int hn;

        public Dij(int n)
        {
            dist = new long[n];
            parE = new int[n];
            hd = new long[1024]; hv = new int[1024];
        }

        public void Run(int src, int target, bool wantPar)
        {
            for (int i = 0; i < dist.Length; i++) dist[i] = long.MaxValue;
            if (wantPar) for (int i = 0; i < parE.Length; i++) parE[i] = -1;
            hn = 0;
            dist[src] = 0;
            Push(0, src);
            while (hn > 0)
            {
                long d = hd[0]; int v = hv[0];
                PopTop();
                if (d != dist[v]) continue;
                if (v == target) return;
                int e1 = off[v + 1];
                for (int a = off[v]; a < e1; a++)
                {
                    int to = ato[a];
                    long nd = d + aw[a];
                    if (nd < dist[to])
                    {
                        dist[to] = nd;
                        if (wantPar) parE[to] = aeid[a];
                        Push(nd, to);
                    }
                }
            }
        }

        void Push(long d, int v)
        {
            if (hn == hd.Length) { Array.Resize(ref hd, hn * 2); Array.Resize(ref hv, hn * 2); }
            int i = hn++;
            while (i > 0)
            {
                int p = (i - 1) >> 1;
                if (hd[p] <= d) break;
                hd[i] = hd[p]; hv[i] = hv[p]; i = p;
            }
            hd[i] = d; hv[i] = v;
        }

        void PopTop()
        {
            hn--;
            if (hn == 0) return;
            long ld = hd[hn]; int lv = hv[hn];
            int i = 0;
            while (true)
            {
                int c = 2 * i + 1;
                if (c >= hn) break;
                if (c + 1 < hn && hd[c + 1] < hd[c]) c++;
                if (hd[c] >= ld) break;
                hd[i] = hd[c]; hv[i] = hv[c]; i = c;
            }
            hd[i] = ld; hv[i] = lv;
        }
    }

    static int TrailingZeros(int x)
    {
        int r = 0;
        while ((x & 1) == 0) { x >>= 1; r++; }
        return r;
    }

    static int NextInt(byte[] b, int len, ref int p)
    {
        while (p < len && (b[p] < (byte)'0' || b[p] > (byte)'9') && b[p] != (byte)'-') p++;
        bool neg = false;
        if (p < len && b[p] == (byte)'-') { neg = true; p++; }
        int x = 0;
        while (p < len && b[p] >= (byte)'0' && b[p] <= (byte)'9') { x = x * 10 + (b[p] - (byte)'0'); p++; }
        return neg ? -x : x;
    }

    static void WriteOut(Stream os, int[] circuit, int cn)
    {
        var buf = new byte[1 << 16]; int bp = 0;
        var tmp = new byte[12];
        for (int i = 0; i < cn; i++)
        {
            int x = circuit[i];
            int tp = 12;
            if (x == 0) tmp[--tp] = (byte)'0';
            else while (x > 0) { tmp[--tp] = (byte)('0' + (x % 10)); x /= 10; }
            int need = (12 - tp) + 1;
            if (bp + need > buf.Length) { os.Write(buf, 0, bp); bp = 0; }
            for (int q = tp; q < 12; q++) buf[bp++] = tmp[q];
            buf[bp++] = (byte)(i + 1 == cn ? '\n' : ' ');
        }
        os.Write(buf, 0, bp);
        os.Flush();
    }
}
