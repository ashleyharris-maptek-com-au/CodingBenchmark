// 3D tetrahedron packing solver
// Strategy: axis-aligned, same-orientation regular tetrahedra placed on a
// regular grid whose spacing equals the canonical tetrahedron's bounding
// box. With identical orientation and bounding-box-sized spacing, the
// tetrahedra cannot overlap (their AABBs only touch at faces). The
// container is treated as the convex polyhedron defined by its vertices;
// half-spaces are extracted by brute force, with a bounding-box fallback
// for very large vertex counts. The grid is scaled up automatically for
// gigantic containers to stay within the time budget.

#include <iostream>
#include <vector>
#include <cmath>
#include <array>
#include <algorithm>
#include <chrono>
#include <cstdio>
#include <cstdlib>

struct Vec3 { double x, y, z; };

static inline double vdot(const Vec3& a, const Vec3& b) {
    return a.x * b.x + a.y * b.y + a.z * b.z;
}
static inline Vec3 vsub(const Vec3& a, const Vec3& b) {
    return { a.x - b.x, a.y - b.y, a.z - b.z };
}
static inline Vec3 vcross(const Vec3& a, const Vec3& b) {
    return { a.y * b.z - a.z * b.y,
             a.z * b.x - a.x * b.z,
             a.x * b.y - a.y * b.x };
}

struct Plane { double a, b, c, d; }; // interior: a*x + b*y + c*z <= d

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);

    auto t_start = std::chrono::steady_clock::now();

    int V;
    double edge;
    if (!(std::cin >> V >> edge)) {
        std::printf("0\n");
        return 0;
    }
    if (V < 4 || !(edge > 0.0)) {
        std::printf("0\n");
        return 0;
    }

    std::vector<Vec3> verts(V);
    for (int i = 0; i < V; ++i) {
        std::cin >> verts[i].x >> verts[i].y >> verts[i].z;
    }

    // Bounding box.
    Vec3 mn = verts[0], mx = verts[0];
    for (const auto& v : verts) {
        if (v.x < mn.x) mn.x = v.x; if (v.x > mx.x) mx.x = v.x;
        if (v.y < mn.y) mn.y = v.y; if (v.y > mx.y) mx.y = v.y;
        if (v.z < mn.z) mn.z = v.z; if (v.z > mx.z) mx.z = v.z;
    }

    // Half-spaces of the convex hull (brute force, O(V^4)).
    std::vector<Plane> planes;
    if (V <= 80) {
        // Tolerance scaled by the maximum coordinate magnitude.
        double cmax = 1.0;
        for (const auto& v : verts) {
            cmax = std::max(cmax, std::abs(v.x));
            cmax = std::max(cmax, std::abs(v.y));
            cmax = std::max(cmax, std::abs(v.z));
        }
        double tol = std::max(1e-7, cmax * 1e-9);

        for (int i = 0; i < V; ++i) {
            for (int j = i + 1; j < V; ++j) {
                for (int k = j + 1; k < V; ++k) {
                    Vec3 n = vcross(vsub(verts[j], verts[i]),
                                    vsub(verts[k], verts[i]));
                    double nrm2 = vdot(n, n);
                    if (nrm2 < 1e-24) continue;
                    double nrm = std::sqrt(nrm2);
                    n.x /= nrm; n.y /= nrm; n.z /= nrm;
                    double d = vdot(n, verts[i]);
                    int pos = 0, neg = 0;
                    for (int l = 0; l < V; ++l) {
                        if (l == i || l == j || l == k) continue;
                        double w = vdot(n, verts[l]) - d;
                        if (w >  tol) ++pos;
                        else if (w < -tol) ++neg;
                        if (pos > 0 && neg > 0) break;
                    }
                    if (pos > 0 && neg > 0) continue;
                    if (pos == 0 && neg == 0) continue;
                    if (pos > 0) { n.x = -n.x; n.y = -n.y; n.z = -n.z; d = -d; }
                    planes.push_back({ n.x, n.y, n.z, d });
                }
            }
        }
        // Dedup near-identical planes.
        std::vector<Plane> uniq;
        uniq.reserve(planes.size());
        for (const auto& p : planes) {
            bool dup = false;
            for (const auto& q : uniq) {
                if (std::abs(p.a - q.a) < 1e-5 &&
                    std::abs(p.b - q.b) < 1e-5 &&
                    std::abs(p.c - q.c) < 1e-5 &&
                    std::abs(p.d - q.d) < std::max(1e-5, std::abs(q.d) * 1e-7)) {
                    dup = true; break;
                }
            }
            if (!dup) uniq.push_back(p);
        }
        planes = std::move(uniq);
    }

    // Fallback: axis-aligned bounding box.
    if (planes.empty()) {
        planes.push_back({  1, 0, 0,  mx.x });
        planes.push_back({ -1, 0, 0, -mn.x });
        planes.push_back({  0, 1, 0,  mx.y });
        planes.push_back({  0,-1, 0, -mn.y });
        planes.push_back({  0, 0, 1,  mx.z });
        planes.push_back({  0, 0,-1, -mn.z });
    }

    // Canonical regular tetrahedron with given edge length, centroid at
    // origin. The bounding box is [-e*sqrt(3)/6, e*sqrt(3)/3] x [-e/2, e/2]
    // x [-h/4, 3h/4], where h = e*sqrt(2/3).
    const double h     = edge * std::sqrt(2.0 / 3.0);
    const double a_t   = edge * std::sqrt(3.0) / 3.0;   // +x vertex
    const double b_t   = edge * std::sqrt(3.0) / 6.0;   // |-x vertex|
    const double tx[4] = {  a_t, -b_t, -b_t, 0.0 };
    const double ty[4] = {  0.0,  edge / 2.0, -edge / 2.0, 0.0 };
    const double tz[4] = { -h / 4.0, -h / 4.0, -h / 4.0, 3.0 * h / 4.0 };

    auto inside_pt = [&planes](double px, double py, double pz) -> bool {
        for (const auto& pl : planes) {
            if (pl.a * px + pl.b * py + pl.c * pz > pl.d + 1e-9) return false;
        }
        return true;
    };

    // Grid spacing = tight tetra bounding-box (guarantees disjoint AABBs).
    double sx = edge * std::sqrt(3.0) / 2.0;
    double sy = edge;
    double sz = h;
    // Tiny epsilon so adjacent AABBs do not share a boundary slab.
    sx += sx * 1e-12 + 1e-12;
    sy += sy * 1e-12 + 1e-12;
    sz += sz * 1e-12 + 1e-12;

    // Auto-scale grid for very large containers.
    double dx = std::max(0.0, mx.x - mn.x);
    double dy = std::max(0.0, mx.y - mn.y);
    double dz = std::max(0.0, mx.z - mn.z);
    double bvol = dx * dy * dz;
    double cvol = sx * sy * sz;
    if (cvol < 1e-30) cvol = 1e-30;
    double est = bvol / cvol;

    const double MAX_CELLS = 8.0e6;
    if (est > MAX_CELLS) {
        double scale = std::cbrt(est / MAX_CELLS);
        sx *= scale; sy *= scale; sz *= scale;
    }

    const long long TIME_LIMIT_MS = 25000;
    const size_t    MAX_RESULTS   = 2'000'000;

    // Start positions: align first tet's AABB to the container's min corner.
    double x_start = mn.x + b_t;
    double y_start = mn.y + edge / 2.0;
    double z_start = mn.z + h / 4.0;

    std::vector<std::array<double, 3>> results;
    results.reserve(4096);

    bool stop = false;
    unsigned ticks = 0;

    for (double z = z_start; z <= mx.z + 1e-12 && !stop; z += sz) {
        for (double y = y_start; y <= mx.y + 1e-12 && !stop; y += sy) {
            if ((++ticks & 0x3F) == 0) {
                auto now = std::chrono::steady_clock::now();
                auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(
                              now - t_start).count();
                if (ms > TIME_LIMIT_MS) { stop = true; break; }
            }
            for (double x = x_start; x <= mx.x + 1e-12; x += sx) {
                bool ok = true;
                for (int t = 0; t < 4; ++t) {
                    if (!inside_pt(x + tx[t], y + ty[t], z + tz[t])) {
                        ok = false; break;
                    }
                }
                if (ok) {
                    results.push_back({ x, y, z });
                    if (results.size() >= MAX_RESULTS) { stop = true; break; }
                }
            }
        }
    }

    // Buffered output (large jobs may emit millions of lines).
    std::vector<char> outbuf;
    outbuf.reserve(results.size() * 96 + 32);
    char tmp[160];
    int n = std::snprintf(tmp, sizeof(tmp), "%zu\n", results.size());
    outbuf.insert(outbuf.end(), tmp, tmp + n);
    for (const auto& r : results) {
        n = std::snprintf(tmp, sizeof(tmp),
                          "%.9f %.9f %.9f 0 0 0\n",
                          r[0], r[1], r[2]);
        outbuf.insert(outbuf.end(), tmp, tmp + n);
    }
    std::fwrite(outbuf.data(), 1, outbuf.size(), stdout);
    std::fflush(stdout);
    return 0;
}
