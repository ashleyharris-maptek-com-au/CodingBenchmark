use std::io::{self, Read, Write, BufWriter};
use std::time::Instant;

const TIMEOUT_SECS: u64 = 270;

struct Solver {
    w: i32,
    h: i32,
    blocked: Vec<bool>,
    require_cycle: bool,
    target: usize,
    visited: Vec<bool>,
    deg: Vec<i32>,
    path: Vec<usize>,
    start: Instant,
    seen_gen: Vec<u32>,
    cur_gen: u32,
    bfs_stack: Vec<usize>,
}

impl Solver {
    #[inline]
    fn coords(&self, i: usize) -> (i32, i32) {
        ((i as i32) % self.w, (i as i32) / self.w)
    }

    #[inline(always)]
    fn fill_neighbors(&self, i: usize, out: &mut [usize; 4]) -> usize {
        let w = self.w as usize;
        let h = self.h as usize;
        let x = i % w;
        let y = i / w;
        let mut count = 0;
        if y + 1 < h {
            let ni = i + w;
            if !self.blocked[ni] { out[count] = ni; count += 1; }
        }
        if y >= 1 {
            let ni = i - w;
            if !self.blocked[ni] { out[count] = ni; count += 1; }
        }
        if x + 1 < w {
            let ni = i + 1;
            if !self.blocked[ni] { out[count] = ni; count += 1; }
        }
        if x >= 1 {
            let ni = i - 1;
            if !self.blocked[ni] { out[count] = ni; count += 1; }
        }
        count
    }

    #[inline]
    fn visit(&mut self, i: usize) {
        self.visited[i] = true;
        self.path.push(i);
        let mut nb = [0usize; 4];
        let cnt = self.fill_neighbors(i, &mut nb);
        for k in 0..cnt {
            self.deg[nb[k]] -= 1;
        }
    }

    #[inline]
    fn unvisit(&mut self, i: usize) {
        let mut nb = [0usize; 4];
        let cnt = self.fill_neighbors(i, &mut nb);
        for k in 0..cnt {
            self.deg[nb[k]] += 1;
        }
        self.path.pop();
        self.visited[i] = false;
    }

    #[inline]
    fn timed_out(&self) -> bool {
        self.start.elapsed().as_secs() >= TIMEOUT_SECS
    }

    fn connectivity_ok(&mut self, from: usize) -> bool {
        let remaining = self.target - self.path.len();
        if remaining == 0 { return true; }

        let mut nb = [0usize; 4];
        let cnt = self.fill_neighbors(from, &mut nb);
        let mut start_node = usize::MAX;
        for k in 0..cnt {
            if !self.visited[nb[k]] {
                start_node = nb[k];
                break;
            }
        }
        if start_node == usize::MAX { return false; }

        self.cur_gen = self.cur_gen.wrapping_add(1);
        if self.cur_gen == 0 {
            for v in self.seen_gen.iter_mut() { *v = 0; }
            self.cur_gen = 1;
        }
        let g = self.cur_gen;

        self.bfs_stack.clear();
        self.bfs_stack.push(start_node);
        self.seen_gen[start_node] = g;
        let mut count = 1usize;

        while let Some(node) = self.bfs_stack.pop() {
            let n = self.fill_neighbors(node, &mut nb);
            for k in 0..n {
                let m = nb[k];
                if !self.visited[m] && self.seen_gen[m] != g {
                    self.seen_gen[m] = g;
                    count += 1;
                    self.bfs_stack.push(m);
                }
            }
        }

        count == remaining
    }

    fn dfs(&mut self, current: usize) -> bool {
        if self.timed_out() { return false; }

        if self.path.len() == self.target {
            if self.require_cycle {
                let (cx, cy) = self.coords(current);
                return (cx == 0 && cy == 1) || (cx == 1 && cy == 0);
            }
            return true;
        }

        let mut nb = [0usize; 4];
        let cnt = self.fill_neighbors(current, &mut nb);

        let mut cands: [(i32, usize); 4] = [(0, 0); 4];
        let mut ccount = 0;
        for k in 0..cnt {
            let m = nb[k];
            if !self.visited[m] {
                cands[ccount] = (self.deg[m], m);
                ccount += 1;
            }
        }

        if ccount == 0 { return false; }

        if ccount > 1 {
            cands[..ccount].sort_by_key(|&(d, _)| d);
        }

        if self.require_cycle && self.path.len() + 1 == self.target {
            for k in 0..ccount {
                let next = cands[k].1;
                let (nx, ny) = self.coords(next);
                if !((nx == 0 && ny == 1) || (nx == 1 && ny == 0)) { continue; }
                self.visit(next);
                if self.dfs(next) { return true; }
                self.unvisit(next);
            }
            return false;
        }

        let mut reserved_for_cycle: usize = usize::MAX;
        if self.require_cycle {
            let mut unvisited_start_neighbors: [usize; 2] = [usize::MAX; 2];
            let mut usn_cnt = 0;
            if self.w >= 2 {
                let idx = 1usize;
                if !self.blocked[idx] && !self.visited[idx] {
                    unvisited_start_neighbors[usn_cnt] = idx; usn_cnt += 1;
                }
            }
            if self.h >= 2 {
                let idx = self.w as usize;
                if !self.blocked[idx] && !self.visited[idx] {
                    unvisited_start_neighbors[usn_cnt] = idx; usn_cnt += 1;
                }
            }
            if usn_cnt == 1 {
                reserved_for_cycle = unvisited_start_neighbors[0];
            } else if usn_cnt == 0 {
                return false;
            }
        }

        for k in 0..ccount {
            let next = cands[k].1;

            if next == reserved_for_cycle && self.path.len() + 1 != self.target {
                continue;
            }

            self.visit(next);

            let proceed = self.path.len() == self.target || self.connectivity_ok(next);

            if proceed && self.dfs(next) {
                return true;
            }

            self.unvisit(next);
        }

        false
    }
}

fn parity_check(w: i32, h: i32, blocked: &[bool], target: usize, require_cycle: bool) -> bool {
    let mut black = 0usize;
    let mut white = 0usize;
    for y in 0..h {
        for x in 0..w {
            let i = (y * w + x) as usize;
            if blocked[i] { continue; }
            if (x + y) % 2 == 0 { black += 1; } else { white += 1; }
        }
    }

    if require_cycle {
        if target % 2 != 0 { return false; }
        if black != white { return false; }
    } else {
        let expected_black = (target + 1) / 2;
        let expected_white = target / 2;
        if black != expected_black || white != expected_white { return false; }
    }

    true
}

fn deg_of(w: i32, h: i32, blocked: &[bool], x: i32, y: i32) -> i32 {
    let mut d = 0;
    for (dx, dy) in [(0i32, 1i32), (0, -1), (1, 0), (-1, 0)] {
        let nx = x + dx;
        let ny = y + dy;
        if nx >= 0 && nx < w && ny >= 0 && ny < h {
            let ni = (ny * w + nx) as usize;
            if !blocked[ni] { d += 1; }
        }
    }
    d
}

fn leaf_check(w: i32, h: i32, blocked: &[bool], require_cycle: bool, target: usize) -> bool {
    let mut leaf_count = 0;
    let mut start_leaf = false;
    for y in 0..h {
        for x in 0..w {
            let i = (y * w + x) as usize;
            if blocked[i] { continue; }
            let d = deg_of(w, h, blocked, x, y);
            if d == 0 {
                if !(x == 0 && y == 0 && target == 1) {
                    return false;
                }
            }
            if d == 1 {
                if x == 0 && y == 0 {
                    start_leaf = true;
                } else {
                    leaf_count += 1;
                }
            }
        }
    }

    if require_cycle {
        if start_leaf && target > 2 { return false; }
        if leaf_count > 0 { return false; }
    } else {
        if leaf_count > 1 { return false; }
    }

    let _ = start_leaf;
    true
}

fn connectivity_full(w: i32, h: i32, blocked: &[bool], target: usize) -> bool {
    let total = (w as usize) * (h as usize);
    if blocked[0] { return false; }
    let mut seen = vec![false; total];
    let mut stack = vec![0usize];
    seen[0] = true;
    let mut count = 1usize;
    while let Some(node) = stack.pop() {
        let x = (node as i32) % w;
        let y = (node as i32) / w;
        for (dx, dy) in [(0i32, 1i32), (0, -1), (1, 0), (-1, 0)] {
            let nx = x + dx;
            let ny = y + dy;
            if nx >= 0 && nx < w && ny >= 0 && ny < h {
                let ni = (ny * w + nx) as usize;
                if !blocked[ni] && !seen[ni] {
                    seen[ni] = true;
                    count += 1;
                    stack.push(ni);
                }
            }
        }
    }
    count == target
}

fn real_main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let stdout = io::stdout();
    let mut out = BufWriter::new(stdout.lock());

    let mut iter = input.split_ascii_whitespace();
    let w: i32 = match iter.next().and_then(|s| s.parse().ok()) {
        Some(v) => v,
        None => { writeln!(out, "0").unwrap(); return; }
    };
    let h: i32 = match iter.next().and_then(|s| s.parse().ok()) {
        Some(v) => v,
        None => { writeln!(out, "0").unwrap(); return; }
    };
    let obstacle_count: usize = match iter.next().and_then(|s| s.parse().ok()) {
        Some(v) => v,
        None => { writeln!(out, "0").unwrap(); return; }
    };
    let require_cycle_raw: i32 = match iter.next().and_then(|s| s.parse().ok()) {
        Some(v) => v,
        None => { writeln!(out, "0").unwrap(); return; }
    };
    let require_cycle = require_cycle_raw != 0;

    if w <= 0 || h <= 0 {
        writeln!(out, "0").unwrap();
        return;
    }

    let total = (w as usize) * (h as usize);
    let mut blocked = vec![false; total];
    for _ in 0..obstacle_count {
        let x: i32 = match iter.next().and_then(|s| s.parse().ok()) {
            Some(v) => v,
            None => break,
        };
        let y: i32 = match iter.next().and_then(|s| s.parse().ok()) {
            Some(v) => v,
            None => break,
        };
        if x >= 0 && x < w && y >= 0 && y < h {
            blocked[(y * w + x) as usize] = true;
        }
    }

    if blocked[0] {
        writeln!(out, "0").unwrap();
        return;
    }

    let target = blocked.iter().filter(|&&b| !b).count();

    if target == 0 {
        writeln!(out, "0").unwrap();
        return;
    }

    if target == 1 {
        if require_cycle {
            writeln!(out, "0").unwrap();
        } else {
            writeln!(out, "1").unwrap();
            writeln!(out, "0 0").unwrap();
        }
        return;
    }

    if !connectivity_full(w, h, &blocked, target) {
        writeln!(out, "0").unwrap();
        return;
    }

    if !parity_check(w, h, &blocked, target, require_cycle) {
        writeln!(out, "0").unwrap();
        return;
    }

    if !leaf_check(w, h, &blocked, require_cycle, target) {
        writeln!(out, "0").unwrap();
        return;
    }

    let mut solver = Solver {
        w,
        h,
        blocked,
        require_cycle,
        target,
        visited: vec![false; total],
        deg: vec![0i32; total],
        path: Vec::with_capacity(target),
        start: Instant::now(),
        seen_gen: vec![0u32; total],
        cur_gen: 0,
        bfs_stack: Vec::with_capacity(total),
    };

    for i in 0..total {
        if solver.blocked[i] { continue; }
        let mut nb = [0usize; 4];
        let cnt = solver.fill_neighbors(i, &mut nb);
        solver.deg[i] = cnt as i32;
    }

    solver.visit(0);

    let found = solver.dfs(0);

    if found {
        writeln!(out, "{}", solver.path.len()).unwrap();
        let path_len = solver.path.len();
        for k in 0..path_len {
            let idx = solver.path[k];
            let x = (idx as i32) % w;
            let y = (idx as i32) / w;
            writeln!(out, "{} {}", x, y).unwrap();
        }
    } else {
        writeln!(out, "0").unwrap();
    }
}

fn main() {
    let builder = std::thread::Builder::new().stack_size(128 * 1024 * 1024);
    let handle = builder.spawn(real_main).unwrap();
    handle.join().unwrap();
}
