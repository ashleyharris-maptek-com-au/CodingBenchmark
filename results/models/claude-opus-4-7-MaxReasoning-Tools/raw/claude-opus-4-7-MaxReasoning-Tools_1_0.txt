// Adaptive TSP solver - handles N from ~10 to 10,000,000+
// Strategy:
//   N <= 14         : Held-Karp dynamic programming (optimal)
//   N <= 2500       : Greedy NN + full-matrix 2-opt
//   N <= 300000     : Spatial-grid NN + neighbor-list 2-opt / Or-opt
//   N >  300000     : Hilbert space-filling curve (very fast, decent quality)
//
// Compiler target: MSVC cl.exe with /std:c++20
// Hardware target: x64 with AVX2 (no AVX-512), 12 cores, 32 GB RAM
// Time budget: 300 s (we self-cap at ~270 s and degrade gracefully)

#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
#include <chrono>
#include <cstdint>
#include <cstdio>
#include <cstdlib>
#include <numeric>
#include <random>
#include <thread>
#include <atomic>
#include <memory>
#include <cstring>
#include <execution>

using namespace std;
using namespace std::chrono;

using Clock_t = steady_clock;
static Clock_t::time_point g_start;
static double g_budget = 270.0;     // overall time cap (s)

static inline double elapsed_s() {
    return duration<double>(Clock_t::now() - g_start).count();
}
static inline bool timeup() {
    return elapsed_s() > g_budget;
}
static inline bool timeup_at(double cap) {
    return elapsed_s() > cap;
}

// ======================= Fast I/O =======================
struct FastReader {
    static constexpr size_t BUF_SIZE = 1 << 16;
    unsigned char buf[BUF_SIZE];
    size_t pos = 0, len = 0;
    FILE* fp;
    explicit FastReader(FILE* f) : fp(f) {}
    inline int gc() {
        if (pos >= len) {
            len = fread(buf, 1, BUF_SIZE, fp);
            pos = 0;
            if (len == 0) return -1;
        }
        return buf[pos++];
    }
    bool read_ll(long long& v) {
        v = 0;
        int c = gc();
        while (c != -1 && (c <= ' ')) c = gc();
        if (c == -1) return false;
        bool neg = false;
        if (c == '-') { neg = true; c = gc(); }
        else if (c == '+') { c = gc(); }
        if (c < '0' || c > '9') return false;
        while (c >= '0' && c <= '9') { v = v * 10 + (c - '0'); c = gc(); }
        if (neg) v = -v;
        return true;
    }
    bool read_float(float& v) {
        char nb[64];
        int idx = 0;
        int c = gc();
        while (c != -1 && (c <= ' ')) c = gc();
        if (c == -1) return false;
        while (c > ' ' && idx < 63) {
            nb[idx++] = (char)c;
            c = gc();
        }
        nb[idx] = 0;
        v = strtof(nb, nullptr);
        return true;
    }
};

struct FastWriter {
    static constexpr size_t BUF_SIZE = 1 << 16;
    char buf[BUF_SIZE];
    size_t pos = 0;
    FILE* fp;
    explicit FastWriter(FILE* f) : fp(f) {}
    ~FastWriter() { flush(); }
    void flush() { if (pos) { fwrite(buf, 1, pos, fp); pos = 0; } }
    inline void put(char c) { if (pos >= BUF_SIZE) flush(); buf[pos++] = c; }
    void put_int(int v) {
        if (v < 0) { put('-'); v = -v; }
        if (v == 0) { put('0'); return; }
        char t[12]; int n = 0;
        while (v > 0) { t[n++] = (char)('0' + (v % 10)); v /= 10; }
        while (n > 0) put(t[--n]);
    }
};

// ======================= Distance helpers =======================
static inline float dist_sq(float ax, float ay, float bx, float by) {
    float dx = ax - bx, dy = ay - by;
    return dx * dx + dy * dy;
}
static inline float dist_e(float ax, float ay, float bx, float by) {
    return sqrtf(dist_sq(ax, ay, bx, by));
}

// ======================= Tour validation / rotate-to-0 =======================
static void rotate_to_start_at_zero(vector<int>& tour) {
    auto it = find(tour.begin(), tour.end(), 0);
    if (it == tour.end() || it == tour.begin()) return;
    rotate(tour.begin(), it, tour.end());
}

// ======================= 1) Held-Karp DP (small N) =======================
static vector<int> solve_held_karp(const vector<float>& xs, const vector<float>& ys) {
    int N = (int)xs.size();
    vector<float> D((size_t)N * N);
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++)
            D[(size_t)i * N + j] = dist_e(xs[i], ys[i], xs[j], ys[j]);
    }

    int total = 1 << N;
    // dp[mask][last] = min cost to visit `mask` ending at `last`, starting at 0
    vector<float> dp((size_t)total * N, 1e30f);
    vector<int>   par((size_t)total * N, -1);
    dp[(size_t)1 * N + 0] = 0.0f;

    for (int mask = 1; mask < total; mask++) {
        if (!(mask & 1)) continue;
        for (int last = 0; last < N; last++) {
            if (!(mask & (1 << last))) continue;
            float v = dp[(size_t)mask * N + last];
            if (v >= 1e30f) continue;
            for (int nxt = 1; nxt < N; nxt++) {
                if (mask & (1 << nxt)) continue;
                int nmask = mask | (1 << nxt);
                float nd = v + D[(size_t)last * N + nxt];
                if (nd < dp[(size_t)nmask * N + nxt]) {
                    dp[(size_t)nmask * N + nxt] = nd;
                    par[(size_t)nmask * N + nxt] = last;
                }
            }
        }
    }

    int full = total - 1;
    int best_last = 1;
    float best = 1e30f;
    for (int last = 1; last < N; last++) {
        float v = dp[(size_t)full * N + last] + D[(size_t)last * N + 0];
        if (v < best) { best = v; best_last = last; }
    }

    vector<int> tour;
    tour.reserve(N);
    int mask = full;
    int cur = best_last;
    while (cur > 0) {
        tour.push_back(cur);
        int p = par[(size_t)mask * N + cur];
        mask ^= (1 << cur);
        cur = p;
    }
    tour.push_back(0);
    reverse(tour.begin(), tour.end());
    return tour;
}

// ======================= 2) NN + full-matrix 2-opt (small-medium) =======================
static vector<int> solve_full_matrix(const vector<float>& xs, const vector<float>& ys,
                                     double phase_cap) {
    int N = (int)xs.size();
    vector<float> D((size_t)N * N);
    for (int i = 0; i < N; i++) {
        D[(size_t)i * N + i] = 0;
        for (int j = i + 1; j < N; j++) {
            float d = dist_e(xs[i], ys[i], xs[j], ys[j]);
            D[(size_t)i * N + j] = D[(size_t)j * N + i] = d;
        }
    }

    // Greedy NN starting from city 0
    vector<int> tour(N);
    vector<char> vis(N, 0);
    tour[0] = 0; vis[0] = 1;
    for (int s = 1; s < N; s++) {
        int cur = tour[s - 1];
        int best = -1;
        float bestD = 1e30f;
        const float* row = &D[(size_t)cur * N];
        for (int j = 0; j < N; j++) {
            if (vis[j]) continue;
            if (row[j] < bestD) { bestD = row[j]; best = j; }
        }
        tour[s] = best;
        vis[best] = 1;
    }

    // 2-opt with full matrix (closes tour implicitly: edge (N-1, 0))
    bool improved = true;
    int pass = 0;
    while (improved && pass < 200 && !timeup_at(phase_cap)) {
        improved = false;
        pass++;
        for (int i = 0; i < N - 1; i++) {
            if ((i & 1023) == 0 && timeup_at(phase_cap)) break;
            int a = tour[i];
            int b = tour[i + 1];
            float dab = D[(size_t)a * N + b];
            const float* rowA = &D[(size_t)a * N];
            const float* rowB = &D[(size_t)b * N];
            for (int j = i + 2; j < N; j++) {
                int c = tour[j];
                int d_node = (j + 1 < N) ? tour[j + 1] : tour[0];
                if (j + 1 >= N && i == 0) continue;       // skip the wrap edge
                float dac = rowA[c];
                float dbd = rowB[d_node];
                float dcd = D[(size_t)c * N + d_node];
                if (dac + dbd + 1e-7f < dab + dcd) {
                    reverse(tour.begin() + i + 1, tour.begin() + j + 1);
                    improved = true;
                    b = tour[i + 1];
                    dab = rowA[b];
                }
            }
        }
    }

    rotate_to_start_at_zero(tour);
    return tour;
}

// ======================= 3) Spatial grid for NN / neighbor lists =======================
struct SpatialGrid {
    int G = 1;
    float minx = 0, miny = 0, cw = 1, ch = 1;
    vector<vector<int>> cells;

    void build(const vector<float>& xs, const vector<float>& ys, int gridSize) {
        int N = (int)xs.size();
        G = max(1, gridSize);
        minx = xs[0]; float maxx = xs[0];
        miny = ys[0]; float maxy = ys[0];
        for (int i = 1; i < N; i++) {
            if (xs[i] < minx) minx = xs[i];
            if (xs[i] > maxx) maxx = xs[i];
            if (ys[i] < miny) miny = ys[i];
            if (ys[i] > maxy) maxy = ys[i];
        }
        float W = maxx - minx;
        float H = maxy - miny;
        if (W < 1e-9f) W = 1.0f;
        if (H < 1e-9f) H = 1.0f;
        cw = W / G;
        ch = H / G;
        if (cw < 1e-12f) cw = 1e-12f;
        if (ch < 1e-12f) ch = 1e-12f;

        cells.assign((size_t)G * G, {});
        for (int i = 0; i < N; i++) {
            int cx = (int)((xs[i] - minx) / cw);
            int cy = (int)((ys[i] - miny) / ch);
            if (cx < 0) cx = 0; if (cx >= G) cx = G - 1;
            if (cy < 0) cy = 0; if (cy >= G) cy = G - 1;
            cells[(size_t)cy * G + cx].push_back(i);
        }
    }

    inline void getCell(float x, float y, int& cx, int& cy) const {
        cx = (int)((x - minx) / cw);
        cy = (int)((y - miny) / ch);
        if (cx < 0) cx = 0; if (cx >= G) cx = G - 1;
        if (cy < 0) cy = 0; if (cy >= G) cy = G - 1;
    }
};

// ======================= 3a) Grid-based nearest-neighbor tour =======================
static vector<int> solve_grid_nn(const vector<float>& xs, const vector<float>& ys) {
    int N = (int)xs.size();

    // Aim for ~4 points per cell on average.
    int gridSize = max(1, (int)sqrt((double)N / 4.0));
    gridSize = min(gridSize, 8192);

    SpatialGrid grid;
    grid.build(xs, ys, gridSize);
    int G = grid.G;
    float cw = grid.cw, ch = grid.ch;
    float minCell = min(cw, ch);

    vector<int> tour;
    tour.reserve(N);
    vector<char> visited(N, 0);

    int cur = 0;
    tour.push_back(0);
    visited[0] = 1;

    for (int step = 1; step < N; step++) {
        float bx = xs[cur], by = ys[cur];
        int ccx, ccy;
        grid.getCell(bx, by, ccx, ccy);

        int best = -1;
        float bestDsq = 1e30f;

        int maxR = G;
        int r = 0;
        while (r <= maxR) {
            int y0 = ccy - r, y1 = ccy + r;
            int x0 = ccx - r, x1 = ccx + r;
            if (y0 < 0) y0 = 0; if (y1 >= G) y1 = G - 1;
            if (x0 < 0) x0 = 0; if (x1 >= G) x1 = G - 1;

            for (int gy = y0; gy <= y1; gy++) {
                for (int gx = x0; gx <= x1; gx++) {
                    if (r > 0 && abs(gx - ccx) != r && abs(gy - ccy) != r) continue;
                    auto& cell = grid.cells[(size_t)gy * G + gx];
                    size_t k = 0;
                    while (k < cell.size()) {
                        int idx = cell[k];
                        if (visited[idx]) {
                            cell[k] = cell.back();
                            cell.pop_back();
                        } else {
                            float dx = xs[idx] - bx, dy = ys[idx] - by;
                            float dsq = dx * dx + dy * dy;
                            if (dsq < bestDsq) { bestDsq = dsq; best = idx; }
                            k++;
                        }
                    }
                }
            }

            if (best >= 0) {
                float bestD = sqrtf(bestDsq);
                int needed = (int)(bestD / minCell) + 1;
                if (needed < maxR) maxR = needed;
            }
            r++;
        }

        if (best < 0) {
            // Defensive fallback (shouldn't happen if data are well-formed)
            for (int i = 0; i < N; i++) {
                if (!visited[i]) { best = i; break; }
            }
        }

        tour.push_back(best);
        visited[best] = 1;
        cur = best;
    }

    return tour;
}

// ======================= 3b) K-nearest neighbors via grid =======================
static vector<vector<int>> build_knn(const vector<float>& xs, const vector<float>& ys,
                                     int k, double phase_cap) {
    int N = (int)xs.size();
    int gridSize = max(1, (int)sqrt((double)N / 4.0));
    gridSize = min(gridSize, 8192);
    SpatialGrid grid;
    grid.build(xs, ys, gridSize);
    int G = grid.G;
    float cw = grid.cw, ch = grid.ch;
    float minCell = min(cw, ch);

    vector<vector<int>> nbrs(N);
    int nthreads = (int)max(1u, std::thread::hardware_concurrency());
    nthreads = min(nthreads, 12);
    if (nthreads < 1) nthreads = 1;

    atomic<int> next_i(0);
    vector<thread> threads;
    for (int t = 0; t < nthreads; t++) {
        threads.emplace_back([&]() {
            vector<pair<float, int>> heap;
            heap.reserve((size_t)k + 8);
            while (true) {
                int i = next_i.fetch_add(1, memory_order_relaxed);
                if (i >= N) break;
                if ((i & 16383) == 0 && timeup_at(phase_cap)) return;
                heap.clear();
                float bx = xs[i], by = ys[i];
                int ccx, ccy;
                grid.getCell(bx, by, ccx, ccy);

                int maxR = G;
                for (int r = 0; r <= maxR; r++) {
                    int y0 = ccy - r, y1 = ccy + r;
                    int x0 = ccx - r, x1 = ccx + r;
                    if (y0 < 0) y0 = 0; if (y1 >= G) y1 = G - 1;
                    if (x0 < 0) x0 = 0; if (x1 >= G) x1 = G - 1;

                    for (int gy = y0; gy <= y1; gy++) {
                        for (int gx = x0; gx <= x1; gx++) {
                            if (r > 0 && abs(gx - ccx) != r && abs(gy - ccy) != r) continue;
                            const auto& cell = grid.cells[(size_t)gy * G + gx];
                            for (int idx : cell) {
                                if (idx == i) continue;
                                float dx = xs[idx] - bx, dy = ys[idx] - by;
                                float dsq = dx * dx + dy * dy;
                                if ((int)heap.size() < k) {
                                    heap.emplace_back(dsq, idx);
                                    push_heap(heap.begin(), heap.end());
                                } else if (dsq < heap.front().first) {
                                    pop_heap(heap.begin(), heap.end());
                                    heap.back() = {dsq, idx};
                                    push_heap(heap.begin(), heap.end());
                                }
                            }
                        }
                    }

                    if ((int)heap.size() >= k) {
                        float worstD = sqrtf(heap.front().first);
                        int needed = (int)(worstD / minCell) + 1;
                        if (needed < maxR) maxR = needed;
                    }
                }

                sort(heap.begin(), heap.end());
                nbrs[i].reserve(heap.size());
                for (auto& p : heap) nbrs[i].push_back(p.second);
            }
        });
    }
    for (auto& th : threads) th.join();
    return nbrs;
}

// ======================= 3c) 2-opt restricted to neighbor lists =======================
static void two_opt_neighbors(const vector<float>& xs, const vector<float>& ys,
                              vector<int>& tour, const vector<vector<int>>& nbrs,
                              double phase_cap) {
    int N = (int)tour.size();
    if (N < 4) return;

    vector<int> pos(N);
    for (int i = 0; i < N; i++) pos[tour[i]] = i;

    bool improved = true;
    int pass = 0;
    while (improved && pass < 40 && !timeup_at(phase_cap)) {
        improved = false;
        pass++;
        for (int i = 0; i < N; i++) {
            if ((i & 8191) == 0 && timeup_at(phase_cap)) break;
            int a = tour[i];
            int b = tour[(i + 1) % N];
            float dab = dist_e(xs[a], ys[a], xs[b], ys[b]);

            // Try replacing edge (a,b) with (a,c) + (b,d) where c is near a
            const auto& na = nbrs[a];
            for (int c : na) {
                if (c == a || c == b) continue;
                float dac = dist_e(xs[a], ys[a], xs[c], ys[c]);
                if (dac >= dab) break;  // nbrs sorted ascending - further c won't help

                int j = pos[c];
                int d = tour[(j + 1) % N];
                if (d == a || d == b) continue;

                float dcd = dist_e(xs[c], ys[c], xs[d], ys[d]);
                float dbd = dist_e(xs[b], ys[b], xs[d], ys[d]);
                if (dac + dbd + 1e-7f < dab + dcd) {
                    // Reverse segment tour[(i+1)..j]; handle wrap by reversing the shorter half
                    int lo = (i + 1) % N;
                    int hi = j;
                    if (lo <= hi) {
                        reverse(tour.begin() + lo, tour.begin() + hi + 1);
                        for (int k2 = lo; k2 <= hi; k2++) pos[tour[k2]] = k2;
                    } else {
                        // The segment wraps. Reverse the complementary segment instead.
                        // Tour reversal: reversing [hi+1 .. lo-1] is equivalent for a cycle.
                        int alt_lo = hi + 1;
                        int alt_hi = lo - 1;
                        if (alt_lo > alt_hi) continue;
                        reverse(tour.begin() + alt_lo, tour.begin() + alt_hi + 1);
                        for (int k2 = alt_lo; k2 <= alt_hi; k2++) pos[tour[k2]] = k2;
                    }
                    improved = true;
                    b = tour[(i + 1) % N];
                    dab = dist_e(xs[a], ys[a], xs[b], ys[b]);
                    break;
                }
            }
        }
    }
}

// ======================= 3d) Or-opt: relocate single cities (cheap, robust) =======================
static void or_opt_pass(const vector<float>& xs, const vector<float>& ys,
                        vector<int>& tour, const vector<vector<int>>& nbrs,
                        double phase_cap) {
    int N = (int)tour.size();
    if (N < 5) return;

    // Use doubly linked list over indices (city -> prev/next city in tour)
    vector<int> nxt(N), prv(N);
    for (int i = 0; i < N; i++) {
        nxt[tour[i]] = tour[(i + 1) % N];
        prv[tour[i]] = tour[(i - 1 + N) % N];
    }

    bool improved = true;
    int pass = 0;
    while (improved && pass < 6 && !timeup_at(phase_cap)) {
        improved = false;
        pass++;
        for (int b = 0; b < N; b++) {
            if ((b & 8191) == 0 && timeup_at(phase_cap)) break;
            int a = prv[b];
            int c = nxt[b];
            if (a == c) continue;
            float d_ab = dist_e(xs[a], ys[a], xs[b], ys[b]);
            float d_bc = dist_e(xs[b], ys[b], xs[c], ys[c]);
            float d_ac = dist_e(xs[a], ys[a], xs[c], ys[c]);
            float gain_remove = d_ab + d_bc - d_ac;
            if (gain_remove <= 1e-7f) continue;

            // Try inserting b between (p, q) where p is a neighbor of b
            int best_p = -1;
            float best_delta = 1e-7f;
            for (int p : nbrs[b]) {
                if (p == b || p == a || p == c) continue;
                int q = nxt[p];
                if (q == b) continue;
                float d_pq = dist_e(xs[p], ys[p], xs[q], ys[q]);
                float d_pb = dist_e(xs[p], ys[p], xs[b], ys[b]);
                float d_bq = dist_e(xs[b], ys[b], xs[q], ys[q]);
                float insert_cost = d_pb + d_bq - d_pq;
                float delta = gain_remove - insert_cost;
                if (delta > best_delta) {
                    best_delta = delta;
                    best_p = p;
                }
            }

            if (best_p >= 0) {
                int p = best_p;
                int q = nxt[p];
                // Remove b from between a and c
                nxt[a] = c; prv[c] = a;
                // Insert b between p and q
                nxt[p] = b; prv[b] = p;
                nxt[b] = q; prv[q] = b;
                improved = true;
            }
        }
    }

    // Walk the linked list to rebuild the tour
    vector<int> out;
    out.reserve(N);
    int cur = 0;
    for (int i = 0; i < N; i++) {
        out.push_back(cur);
        cur = nxt[cur];
    }
    tour.swap(out);
}

// ======================= 4) Hilbert curve for huge N =======================
// 2-D -> 1-D mapping using a 2^16 x 2^16 grid; Hilbert code fits in uint32_t.
// Maps (x, y) in [0, 2^16) x [0, 2^16) to a 32-bit Hilbert-curve index.
// Rotation uses the outer grid extent (N = 1<<16), not the current bit scale.
static inline uint32_t xy_to_hilbert(uint32_t x, uint32_t y) {
    constexpr uint32_t HILBERT_N = 1u << 16;  // outer grid side
    constexpr uint32_t HILBERT_M = HILBERT_N - 1u;
    uint32_t d = 0;
    for (uint32_t s = HILBERT_N >> 1; s > 0; s >>= 1) {
        uint32_t rx = (x & s) ? 1u : 0u;
        uint32_t ry = (y & s) ? 1u : 0u;
        d += s * s * ((3u * rx) ^ ry);
        if (ry == 0) {
            if (rx == 1) {
                x = HILBERT_M - x;
                y = HILBERT_M - y;
            }
            uint32_t t = x; x = y; y = t;
        }
    }
    return d;
}

static vector<int> solve_hilbert(const vector<float>& xs, const vector<float>& ys) {
    int N = (int)xs.size();
    float minx = xs[0], maxx = xs[0];
    float miny = ys[0], maxy = ys[0];
    for (int i = 1; i < N; i++) {
        if (xs[i] < minx) minx = xs[i];
        if (xs[i] > maxx) maxx = xs[i];
        if (ys[i] < miny) miny = ys[i];
        if (ys[i] > maxy) maxy = ys[i];
    }
    double W = (double)maxx - (double)minx;
    double H = (double)maxy - (double)miny;
    if (W < 1e-12) W = 1.0;
    if (H < 1e-12) H = 1.0;
    const double maxCoord = (double)((1u << 16) - 1);
    double sx = maxCoord / W;
    double sy = maxCoord / H;

    struct Entry { uint32_t key; int idx; };
    vector<Entry> entries(N);

    int nthreads = (int)max(1u, std::thread::hardware_concurrency());
    nthreads = min(nthreads, 12);
    int chunk = max(1, (N + nthreads - 1) / nthreads);

    vector<thread> threads;
    for (int t = 0; t < nthreads; t++) {
        int start = t * chunk;
        int end = min(N, start + chunk);
        if (start >= end) break;
        threads.emplace_back([&, start, end]() {
            for (int i = start; i < end; i++) {
                int xi_i = (int)((double)(xs[i] - minx) * sx);
                int yi_i = (int)((double)(ys[i] - miny) * sy);
                if (xi_i < 0) xi_i = 0; if (xi_i > 0xFFFF) xi_i = 0xFFFF;
                if (yi_i < 0) yi_i = 0; if (yi_i > 0xFFFF) yi_i = 0xFFFF;
                entries[i].key = xy_to_hilbert((uint32_t)xi_i, (uint32_t)yi_i);
                entries[i].idx = i;
            }
        });
    }
    for (auto& th : threads) th.join();

    // Parallel sort if available (falls back to serial on platforms without execution).
    try {
        std::sort(std::execution::par_unseq, entries.begin(), entries.end(),
                  [](const Entry& a, const Entry& b) { return a.key < b.key; });
    } catch (...) {
        std::sort(entries.begin(), entries.end(),
                  [](const Entry& a, const Entry& b) { return a.key < b.key; });
    }

    vector<int> tour(N);
    int startPos = 0;
    for (int i = 0; i < N; i++) {
        tour[i] = entries[i].idx;
        if (entries[i].idx == 0) startPos = i;
    }
    if (startPos != 0) {
        rotate(tour.begin(), tour.begin() + startPos, tour.end());
    }
    return tour;
}

// ======================= Top-level dispatcher =======================
static vector<int> solve_tsp(const vector<float>& xs, const vector<float>& ys) {
    int N = (int)xs.size();
    if (N <= 0) return {};
    if (N == 1) return {0};
    if (N == 2) return {0, 1};

    // Small: optimal DP
    if (N <= 14) {
        return solve_held_karp(xs, ys);
    }

    // Small-medium: NN + full-matrix 2-opt
    if (N <= 2500) {
        double cap = min(g_budget, 60.0 + elapsed_s());
        auto tour = solve_full_matrix(xs, ys, cap);
        return tour;
    }

    // Medium-large: grid NN + neighbor-list 2-opt + Or-opt
    if (N <= 300000) {
        auto tour = solve_grid_nn(xs, ys);
        rotate_to_start_at_zero(tour);

        if (timeup()) return tour;

        // Pick k by problem size to keep work bounded.
        int k = (N <= 5000) ? 20 : (N <= 50000) ? 15 : 10;

        double knn_cap = elapsed_s() + max(5.0, (g_budget - elapsed_s()) * 0.35);
        auto nbrs = build_knn(xs, ys, k, knn_cap);
        if (timeup()) return tour;

        // First pass of Or-opt cleans up bad NN insertions cheaply.
        double oropt_cap = elapsed_s() + max(5.0, (g_budget - elapsed_s()) * 0.30);
        or_opt_pass(xs, ys, tour, nbrs, oropt_cap);
        rotate_to_start_at_zero(tour);
        if (timeup()) return tour;

        // Then neighbor-restricted 2-opt to remove edge crossings.
        double twoopt_cap = g_budget - 5.0;
        two_opt_neighbors(xs, ys, tour, nbrs, twoopt_cap);
        rotate_to_start_at_zero(tour);
        return tour;
    }

    // Huge: Hilbert curve.  If we still have plenty of budget, run a cheap Or-opt
    // pass to remove the worst kinks in the space-filling traversal.
    auto tour = solve_hilbert(xs, ys);

    if (!timeup()) {
        double remaining = g_budget - elapsed_s();
        if (remaining > 30.0) {
            // Smaller k for larger N to keep work bounded.
            int k = (N <= 2000000) ? 8 : (N <= 5000000) ? 6 : 4;
            double knn_cap = elapsed_s() + remaining * 0.4;
            auto nbrs = build_knn(xs, ys, k, knn_cap);
            if (!timeup()) {
                double oropt_cap = g_budget - 10.0;
                or_opt_pass(xs, ys, tour, nbrs, oropt_cap);
            }
        }
    }
    rotate_to_start_at_zero(tour);
    return tour;
}

// ======================= main =======================
int main() {
    g_start = Clock_t::now();

    FastReader reader(stdin);
    long long Nll = 0;
    if (!reader.read_ll(Nll) || Nll <= 0) {
        return 0;
    }
    int N = (int)Nll;

    vector<float> xs(N), ys(N);
    for (int i = 0; i < N; i++) {
        reader.read_float(xs[i]);
        reader.read_float(ys[i]);
    }

    vector<int> tour = solve_tsp(xs, ys);

    // Guarantee that the tour is a valid permutation that starts at city 0.
    if ((int)tour.size() != N) {
        tour.assign(N, 0);
        for (int i = 0; i < N; i++) tour[i] = i;
    } else if (tour.empty() || tour[0] != 0) {
        rotate_to_start_at_zero(tour);
        if (tour.empty() || tour[0] != 0) {
            for (int i = 0; i < N; i++) tour[i] = i;
        }
    }

    FastWriter writer(stdout);
    for (int i = 0; i < N; i++) {
        if (i > 0) writer.put(' ');
        writer.put_int(tour[i]);
    }
    writer.put('\n');
    return 0;
}
