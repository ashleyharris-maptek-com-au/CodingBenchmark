use std::io::{self, Read, Write, BufWriter};

fn parse_num(input: &[u8], pos: &mut usize) -> usize {
    let n = input.len();
    while *pos < n && !input[*pos].is_ascii_digit() {
        *pos += 1;
    }
    let mut v = 0usize;
    while *pos < n && input[*pos].is_ascii_digit() {
        v = v * 10 + (input[*pos] - b'0') as usize;
        *pos += 1;
    }
    v
}

fn write_num(buf: &mut Vec<u8>, mut n: usize) {
    if n == 0 {
        buf.push(b'0');
        return;
    }
    let start = buf.len();
    while n > 0 {
        buf.push(b'0' + (n % 10) as u8);
        n /= 10;
    }
    buf[start..].reverse();
}

// State byte encoding for each padded cell:
//   0  = UNVISITED   (open, not yet visited by BFS)
//   1  = PARENT_UP   (move up by w_p to reach parent)
//   2  = PARENT_DOWN (move down by w_p to reach parent)
//   3  = PARENT_LEFT (move left by 1 to reach parent)
//   4  = PARENT_RIGHT(move right by 1 to reach parent)
//   5  = START_MARKER (path reconstruction terminator)
//   255= WALL        (cannot pass)
const UNVISITED: u8     = 0;
const PARENT_UP: u8     = 1;
const PARENT_DOWN: u8   = 2;
const PARENT_LEFT: u8   = 3;
const PARENT_RIGHT: u8  = 4;
const START_MARKER: u8  = 5;
const WALL: u8          = 0xFF;

fn main() {
    // Slurp stdin in one go - faster than line-by-line for large inputs.
    let mut input: Vec<u8> = Vec::new();
    io::stdin().read_to_end(&mut input).expect("failed to read stdin");

    let stdout = io::stdout();
    let mut out = BufWriter::with_capacity(1 << 20, stdout.lock());

    let n_input = input.len();
    let mut pos = 0usize;

    let h = parse_num(&input, &mut pos);
    let w = parse_num(&input, &mut pos);

    // Skip remainder of the first line (any trailing whitespace / \r / \n).
    while pos < n_input && input[pos] != b'\n' { pos += 1; }
    if pos < n_input { pos += 1; }

    if h == 0 || w == 0 {
        writeln!(out, "0").unwrap();
        return;
    }

    // Pad the grid with a 1-cell wall border on every side. This lets the
    // BFS inner loop access all four neighbours unconditionally - the walls
    // automatically block movement past the original boundary and we never
    // have to do any modulus, division or index-vs-bound checks per step.
    let w_p = w + 2;
    let h_p = h + 2;
    let total_p = h_p.checked_mul(w_p).expect("dimension overflow");

    // Sanity: indices must fit in u32 so the queue stays half-sized.
    // 50002 * 50002 = 2,500,200,004 which is < 2^32 = 4,294,967,296.
    assert!(total_p <= u32::MAX as usize, "maze too large for u32 indices");

    let mut state: Vec<u8> = vec![WALL; total_p];
    let mut start_idx: usize = usize::MAX;
    let mut goal_idx: usize  = usize::MAX;

    for y in 0..h {
        let row_start = pos;
        let mut row_end = pos;
        while row_end < n_input && input[row_end] != b'\n' { row_end += 1; }
        // Trim a trailing '\r' (Windows line endings).
        let actual_end = if row_end > row_start && input[row_end - 1] == b'\r' {
            row_end - 1
        } else {
            row_end
        };
        let row_len = actual_end - row_start;

        let base = (y + 1) * w_p + 1; // padded coordinate of original (0, y)
        for x in 0..w {
            // If a row is short, treat the missing tail as open space.
            let c = if x < row_len { input[row_start + x] } else { b' ' };
            let idx = base + x;
            match c {
                b'#' => {} // already WALL from the initial fill
                b'A' => {
                    start_idx = idx;
                    state[idx] = START_MARKER;
                }
                b'B' => {
                    goal_idx = idx;
                    state[idx] = UNVISITED;
                }
                _ => {
                    // Any other character (including ' ', '\t', etc.) is open.
                    state[idx] = UNVISITED;
                }
            }
        }
        pos = if row_end < n_input { row_end + 1 } else { row_end };
    }

    // Free the raw input buffer; we no longer need it.
    drop(input);

    if start_idx == usize::MAX || goal_idx == usize::MAX {
        writeln!(out, "0").unwrap();
        return;
    }

    // Trivial case: A and B in the same cell.
    if start_idx == goal_idx {
        let yp = start_idx / w_p;
        let xp = start_idx - yp * w_p;
        writeln!(out, "1").unwrap();
        writeln!(out, "{} {}", xp - 1, yp - 1).unwrap();
        return;
    }

    // BFS - shortest unweighted path on the 4-connected grid.
    // We store padded indices as u32 to halve queue memory vs usize.
    let mut queue: Vec<u32> = Vec::with_capacity(4096);
    queue.push(start_idx as u32);
    let mut qhead = 0usize;

    let mut found = false;
    // Raw pointer into state so we can bypass bounds checks in the hot loop.
    // Safe because every (nidx) we compute lives in the padded grid: walls
    // form a 1-cell border that catches all out-of-bounds neighbour steps.
    let state_ptr = state.as_mut_ptr();

    'bfs: while qhead < queue.len() {
        let idx = unsafe { *queue.get_unchecked(qhead) as usize };
        qhead += 1;

        unsafe {
            // Up
            {
                let nidx = idx - w_p;
                let s = state_ptr.add(nidx);
                if *s == UNVISITED {
                    *s = PARENT_DOWN;
                    if nidx == goal_idx { found = true; break 'bfs; }
                    queue.push(nidx as u32);
                }
            }
            // Down
            {
                let nidx = idx + w_p;
                let s = state_ptr.add(nidx);
                if *s == UNVISITED {
                    *s = PARENT_UP;
                    if nidx == goal_idx { found = true; break 'bfs; }
                    queue.push(nidx as u32);
                }
            }
            // Left
            {
                let nidx = idx - 1;
                let s = state_ptr.add(nidx);
                if *s == UNVISITED {
                    *s = PARENT_RIGHT;
                    if nidx == goal_idx { found = true; break 'bfs; }
                    queue.push(nidx as u32);
                }
            }
            // Right
            {
                let nidx = idx + 1;
                let s = state_ptr.add(nidx);
                if *s == UNVISITED {
                    *s = PARENT_LEFT;
                    if nidx == goal_idx { found = true; break 'bfs; }
                    queue.push(nidx as u32);
                }
            }
        }
    }

    // Queue no longer needed - free it before we allocate path storage.
    drop(queue);

    if !found {
        writeln!(out, "0").unwrap();
        return;
    }

    // Walk parent pointers from goal back to start.
    let mut path: Vec<u32> = Vec::new();
    let mut cur = goal_idx;
    loop {
        path.push(cur as u32);
        if cur == start_idx { break; }
        match state[cur] {
            PARENT_UP    => { cur -= w_p; }
            PARENT_DOWN  => { cur += w_p; }
            PARENT_LEFT  => { cur -= 1; }
            PARENT_RIGHT => { cur += 1; }
            _ => { break; } // unreachable for a well-formed BFS tree
        }
    }
    path.reverse();

    // State no longer needed.
    drop(state);

    // Header: number of steps in the path.
    writeln!(out, "{}", path.len()).unwrap();

    // Buffered formatting of "x y\n" lines. Flush every ~64 KiB so we don't
    // hold the entire output in a single growing Vec for huge paths.
    let mut buf: Vec<u8> = Vec::with_capacity(64 * 1024);
    for &p in &path {
        let p = p as usize;
        let yp = p / w_p;
        let xp = p - yp * w_p;
        let x = xp - 1; // unpad
        let y = yp - 1;
        write_num(&mut buf, x);
        buf.push(b' ');
        write_num(&mut buf, y);
        buf.push(b'\n');
        if buf.len() >= 64 * 1024 {
            out.write_all(&buf).unwrap();
            buf.clear();
        }
    }
    if !buf.is_empty() {
        out.write_all(&buf).unwrap();
    }
}
