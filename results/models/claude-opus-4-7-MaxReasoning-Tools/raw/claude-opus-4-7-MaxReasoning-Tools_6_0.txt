using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Globalization;
using System.Linq;
using System.Numerics;
using System.Text;

class Program
{
    static double mu = 398600.4418;
    static int N;
    static OrbElements startElem;
    static OrbElements[] elems;
    static double[,] dist;
    static double[] startDist;

    struct OrbElements
    {
        public double hx, hy, hz;
        public double a;
        public double e;
        public double hMag;
        public double rMag;
    }

    static OrbElements ComputeElements(double x, double y, double z, double vx, double vy, double vz)
    {
        var el = new OrbElements();
        el.rMag = Math.Sqrt(x * x + y * y + z * z);
        double vMag2 = vx * vx + vy * vy + vz * vz;
        el.hx = y * vz - z * vy;
        el.hy = z * vx - x * vz;
        el.hz = x * vy - y * vx;
        el.hMag = Math.Sqrt(el.hx * el.hx + el.hy * el.hy + el.hz * el.hz);
        double energy = vMag2 / 2.0 - mu / el.rMag;
        if (energy < -1e-10)
        {
            el.a = -mu / (2 * energy);
        }
        else
        {
            el.a = el.rMag;
        }
        if (el.a <= 0 || double.IsNaN(el.a) || double.IsInfinity(el.a)) el.a = el.rMag;
        double term = 1 - (el.hMag * el.hMag) / (mu * el.a);
        el.e = (term > 0) ? Math.Sqrt(term) : 0;
        if (double.IsNaN(el.e) || el.e > 1) el.e = 0;
        return el;
    }

    static double Cost(OrbElements a, OrbElements b)
    {
        double r1 = a.a, r2 = b.a;
        if (r1 <= 0 || r2 <= 0 || double.IsNaN(r1) || double.IsNaN(r2)) return 1e9;

        double sumR = r1 + r2;
        double v1 = Math.Sqrt(mu / r1);
        double v2 = Math.Sqrt(mu / r2);
        double vp = Math.Sqrt(2 * mu * r2 / (r1 * sumR));
        double va = Math.Sqrt(2 * mu * r1 / (r2 * sumR));
        double hohmann = Math.Abs(vp - v1) + Math.Abs(v2 - va);

        double dotH = a.hx * b.hx + a.hy * b.hy + a.hz * b.hz;
        double denom = a.hMag * b.hMag;
        double cosPlane = (denom > 1e-12) ? dotH / denom : 1.0;
        if (cosPlane > 1.0) cosPlane = 1.0;
        if (cosPlane < -1.0) cosPlane = -1.0;
        double planeAngle = Math.Acos(cosPlane);
        double vAvg = Math.Sqrt(mu / (sumR / 2.0));
        double planeChange = 2 * vAvg * Math.Sin(planeAngle / 2);

        double eccCost = Math.Abs(a.e - b.e) * vAvg * 0.3;

        double result = hohmann + planeChange + eccCost;
        if (double.IsNaN(result) || double.IsInfinity(result)) return 1e9;
        return result;
    }

    static void Main()
    {
        var sw = Stopwatch.StartNew();

        var raw = Console.In.ReadToEnd();
        if (raw.Length > 0 && raw[0] == '﻿') raw = raw.Substring(1);
        var tokens = raw.Split(new[] { ' ', '\t', '\n', '\r', ':', ',' }, StringSplitOptions.RemoveEmptyEntries);
        var nums = new List<double>();
        foreach (var t in tokens)
        {
            if (double.TryParse(t, NumberStyles.Float, CultureInfo.InvariantCulture, out double v))
            {
                nums.Add(v);
            }
        }

        int idx = 0;
        N = (int)nums[idx++];
        mu = nums[idx++];

        double[] startVec = new double[6];
        for (int i = 0; i < 6; i++) startVec[i] = nums[idx++];

        double[][] stations = new double[N][];
        for (int s = 0; s < N; s++)
        {
            stations[s] = new double[6];
            for (int i = 0; i < 6; i++) stations[s][i] = nums[idx++];
        }

        startElem = ComputeElements(startVec[0], startVec[1], startVec[2], startVec[3], startVec[4], startVec[5]);
        elems = new OrbElements[N];
        for (int i = 0; i < N; i++)
        {
            elems[i] = ComputeElements(stations[i][0], stations[i][1], stations[i][2],
                                        stations[i][3], stations[i][4], stations[i][5]);
        }

        dist = new double[N, N];
        startDist = new double[N];
        for (int i = 0; i < N; i++)
        {
            startDist[i] = Cost(startElem, elems[i]);
            for (int j = 0; j < N; j++)
            {
                if (i != j) dist[i, j] = Cost(elems[i], elems[j]);
            }
        }

        int[] order;
        if (N == 1) order = new int[] { 0 };
        else if (N <= 18) order = HeldKarp();
        else order = SolveHeuristic(sw);

        var sb = new StringBuilder();
        for (int i = 0; i < order.Length; i++)
        {
            if (i > 0) sb.Append(' ');
            sb.Append(order[i]);
        }
        Console.WriteLine(sb.ToString());
    }

    static int[] HeldKarp()
    {
        int n = N;
        int states = 1 << n;
        var dp = new double[states, n];
        var par = new short[states, n];
        for (int m = 0; m < states; m++)
        {
            for (int j = 0; j < n; j++)
            {
                dp[m, j] = double.MaxValue;
                par[m, j] = -1;
            }
        }
        for (int i = 0; i < n; i++) dp[1 << i, i] = startDist[i];

        int full = states - 1;
        for (int mask = 1; mask < states; mask++)
        {
            for (int last = 0; last < n; last++)
            {
                if ((mask & (1 << last)) == 0) continue;
                double c = dp[mask, last];
                if (c >= double.MaxValue) continue;
                int rem = full & ~mask;
                while (rem != 0)
                {
                    int bit = rem & -rem;
                    int nxt = BitOperations.TrailingZeroCount((uint)bit);
                    double nc = c + dist[last, nxt];
                    int nm = mask | bit;
                    if (nc < dp[nm, nxt])
                    {
                        dp[nm, nxt] = nc;
                        par[nm, nxt] = (short)last;
                    }
                    rem ^= bit;
                }
            }
        }

        double best = double.MaxValue;
        int bestEnd = 0;
        for (int i = 0; i < n; i++)
        {
            if (dp[full, i] < best) { best = dp[full, i]; bestEnd = i; }
        }

        int[] order = new int[n];
        int cur = bestEnd, m2 = full;
        for (int k = n - 1; k >= 0; k--)
        {
            order[k] = cur;
            int p = par[m2, cur];
            m2 ^= 1 << cur;
            cur = p;
        }
        return order;
    }

    static double TourCost(int[] order)
    {
        double s = startDist[order[0]];
        for (int i = 1; i < order.Length; i++) s += dist[order[i - 1], order[i]];
        return s;
    }

    static int[] NearestNeighbor(int firstStation)
    {
        int n = N;
        int[] order = new int[n];
        bool[] vis = new bool[n];
        order[0] = firstStation;
        vis[firstStation] = true;
        for (int k = 1; k < n; k++)
        {
            int last = order[k - 1];
            int nx = -1; double bc = double.MaxValue;
            for (int j = 0; j < n; j++)
            {
                if (!vis[j] && dist[last, j] < bc) { bc = dist[last, j]; nx = j; }
            }
            order[k] = nx;
            vis[nx] = true;
        }
        return order;
    }

    static int NearestToStart()
    {
        int n = N;
        int best = 0; double bc = double.MaxValue;
        for (int i = 0; i < n; i++) if (startDist[i] < bc) { bc = startDist[i]; best = i; }
        return best;
    }

    static bool TwoOpt(int[] order, Stopwatch sw, long limit)
    {
        int n = order.Length;
        bool any = false;
        bool improved = true;
        while (improved)
        {
            improved = false;
            if (sw.ElapsedMilliseconds > limit) break;
            for (int i = 0; i < n - 1; i++)
            {
                for (int j = i + 1; j < n; j++)
                {
                    int iCur = order[i];
                    int jCur = order[j];
                    int iPrev = i > 0 ? order[i - 1] : -1;
                    int jNext = j + 1 < n ? order[j + 1] : -1;

                    double e1Old, e1New;
                    if (iPrev < 0) { e1Old = startDist[iCur]; e1New = startDist[jCur]; }
                    else { e1Old = dist[iPrev, iCur]; e1New = dist[iPrev, jCur]; }

                    double e2Old, e2New;
                    if (jNext < 0) { e2Old = 0; e2New = 0; }
                    else { e2Old = dist[jCur, jNext]; e2New = dist[iCur, jNext]; }

                    if ((e1New + e2New) < (e1Old + e2Old) - 1e-9)
                    {
                        Array.Reverse(order, i, j - i + 1);
                        improved = true;
                        any = true;
                    }
                }
            }
        }
        return any;
    }

    static bool OrOpt(int[] order, Stopwatch sw, long limit)
    {
        int n = order.Length;
        bool any = false;
        bool improved = true;
        int maxL = Math.Min(3, n - 1);
        if (maxL < 1) return false;

        while (improved)
        {
            improved = false;
            if (sw.ElapsedMilliseconds > limit) break;

            for (int L = 1; L <= maxL && !improved; L++)
            {
                for (int i = 0; i + L <= n && !improved; i++)
                {
                    int seg0 = order[i];
                    int segE = order[i + L - 1];
                    int prev = i > 0 ? order[i - 1] : -1;
                    int next = i + L < n ? order[i + L] : -1;

                    double oldRemove, newRemove;
                    if (prev < 0)
                    {
                        oldRemove = startDist[seg0];
                        if (next < 0) newRemove = 0;
                        else { oldRemove += dist[segE, next]; newRemove = startDist[next]; }
                    }
                    else
                    {
                        oldRemove = dist[prev, seg0];
                        if (next < 0) newRemove = 0;
                        else { oldRemove += dist[segE, next]; newRemove = dist[prev, next]; }
                    }
                    double savings = oldRemove - newRemove;

                    int bestInsert = -1;
                    double bestDelta = 0;

                    for (int insertAt = 0; insertAt <= n - L; insertAt++)
                    {
                        if (insertAt == i) continue;

                        int leftNode, rightNode;
                        if (insertAt == 0) leftNode = -1;
                        else
                        {
                            int k = insertAt - 1;
                            leftNode = (k < i) ? order[k] : order[k + L];
                        }
                        if (insertAt == n - L) rightNode = -1;
                        else
                        {
                            int k = insertAt;
                            rightNode = (k < i) ? order[k] : order[k + L];
                        }

                        double oldInsert, newInsert;
                        if (leftNode < 0)
                        {
                            if (rightNode < 0) { oldInsert = 0; newInsert = startDist[seg0]; }
                            else { oldInsert = startDist[rightNode]; newInsert = startDist[seg0] + dist[segE, rightNode]; }
                        }
                        else
                        {
                            if (rightNode < 0) { oldInsert = 0; newInsert = dist[leftNode, seg0]; }
                            else { oldInsert = dist[leftNode, rightNode]; newInsert = dist[leftNode, seg0] + dist[segE, rightNode]; }
                        }
                        double cost = newInsert - oldInsert;

                        double netDelta = cost - savings;
                        if (netDelta < bestDelta - 1e-9)
                        {
                            bestDelta = netDelta;
                            bestInsert = insertAt;
                        }
                    }

                    if (bestInsert >= 0)
                    {
                        int[] seg = new int[L];
                        for (int k = 0; k < L; k++) seg[k] = order[i + k];

                        int[] reduced = new int[n - L];
                        int rIdx = 0;
                        for (int k = 0; k < n; k++)
                        {
                            if (k >= i && k < i + L) continue;
                            reduced[rIdx++] = order[k];
                        }

                        int oi = 0;
                        for (int k = 0; k < bestInsert; k++) order[oi++] = reduced[k];
                        for (int k = 0; k < L; k++) order[oi++] = seg[k];
                        for (int k = bestInsert; k < n - L; k++) order[oi++] = reduced[k];

                        improved = true;
                        any = true;
                    }
                }
            }
        }
        return any;
    }

    static int[] DoubleBridge(int[] order, Random rng)
    {
        int n = order.Length;
        if (n < 8)
        {
            int[] arr = (int[])order.Clone();
            int a = rng.Next(n), b = rng.Next(n);
            int tmp = arr[a]; arr[a] = arr[b]; arr[b] = tmp;
            return arr;
        }
        int p1 = 1 + rng.Next(n / 4);
        int p2 = p1 + 1 + rng.Next(n / 4);
        int p3 = p2 + 1 + rng.Next(n / 4);
        int[] newOrder = new int[n];
        int idx2 = 0;
        for (int i = 0; i < p1; i++) newOrder[idx2++] = order[i];
        for (int i = p3; i < n; i++) newOrder[idx2++] = order[i];
        for (int i = p2; i < p3; i++) newOrder[idx2++] = order[i];
        for (int i = p1; i < p2; i++) newOrder[idx2++] = order[i];
        return newOrder;
    }

    static int[] SolveHeuristic(Stopwatch sw)
    {
        long timeLimit = 240_000;
        int n = N;
        var rng = new Random(42);

        int[] best = NearestNeighbor(NearestToStart());
        TwoOpt(best, sw, timeLimit);
        OrOpt(best, sw, timeLimit);
        TwoOpt(best, sw, timeLimit);
        double bestCost = TourCost(best);

        var tried = new HashSet<int> { best[0] };

        int maxRestarts = Math.Min(20, n);
        for (int r = 0; r < maxRestarts && sw.ElapsedMilliseconds < timeLimit * 0.5; r++)
        {
            int start;
            int attempts = 0;
            do
            {
                start = rng.Next(n);
                attempts++;
            } while (tried.Contains(start) && attempts < 50);
            if (tried.Contains(start)) break;
            tried.Add(start);

            int[] cand = NearestNeighbor(start);
            TwoOpt(cand, sw, timeLimit);
            OrOpt(cand, sw, timeLimit);
            TwoOpt(cand, sw, timeLimit);
            double cc = TourCost(cand);
            if (cc < bestCost) { bestCost = cc; best = (int[])cand.Clone(); }
        }

        while (sw.ElapsedMilliseconds < timeLimit * 0.95)
        {
            int[] perturbed = DoubleBridge(best, rng);
            TwoOpt(perturbed, sw, timeLimit);
            if (sw.ElapsedMilliseconds >= timeLimit * 0.95) break;
            OrOpt(perturbed, sw, timeLimit);
            TwoOpt(perturbed, sw, timeLimit);
            double pc = TourCost(perturbed);
            if (pc < bestCost) { bestCost = pc; best = (int[])perturbed.Clone(); }
        }

        return best;
    }
}
