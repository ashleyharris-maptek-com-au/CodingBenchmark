using System;
using System.IO;
using System.Text;

class Program
{
    static void Main()
    {
        // ---------- Fast input ----------
        var stdin = Console.OpenStandardInput();
        var ms = new MemoryStream();
        stdin.CopyTo(ms);
        byte[] buf = ms.ToArray();
        int pos = 0;

        int ReadInt()
        {
            while (pos < buf.Length && (buf[pos] < (byte)'0' || buf[pos] > (byte)'9') && buf[pos] != (byte)'-') pos++;
            bool neg = false;
            if (pos < buf.Length && buf[pos] == (byte)'-') { neg = true; pos++; }
            int x = 0;
            while (pos < buf.Length && buf[pos] >= (byte)'0' && buf[pos] <= (byte)'9')
            {
                x = x * 10 + (buf[pos] - (byte)'0');
                pos++;
            }
            return neg ? -x : x;
        }

        int N = ReadInt();
        int M = ReadInt();

        if (N <= 0)
        {
            Console.Out.Write("0");
            return;
        }

        // ---------- Graph storage as half-edge linked lists ----------
        // Each undirected edge has two half-edges. Edge id = halfId >> 1.
        // We reserve room for up to (N-1) extra duplicated tree edges.
        int maxEdges = M + N + 8;
        int maxHalf = 2 * maxEdges;

        int[] head = new int[N];
        int[] nextE = new int[maxHalf];
        int[] toE = new int[maxHalf];
        bool[] usedE = new bool[maxEdges];

        for (int i = 0; i < N; i++) head[i] = -1;

        int hc = 0;
        int[] degree = new int[N];

        for (int i = 0; i < M; i++)
        {
            int u = ReadInt();
            int v = ReadInt();
            ReadInt(); // weight - any valid route is acceptable per problem statement

            toE[hc] = v;
            nextE[hc] = head[u];
            head[u] = hc++;

            toE[hc] = u;
            nextE[hc] = head[v];
            head[v] = hc++;

            degree[u]++;
            degree[v]++;
        }

        if (M == 0)
        {
            Console.Out.Write("0");
            return;
        }

        // ---------- BFS spanning tree from node 0 ----------
        int[] parent = new int[N];
        int[] bfsOrder = new int[N];
        bool[] visited = new bool[N];
        for (int i = 0; i < N; i++) parent[i] = -1;

        int[] qArr = new int[N];
        int qh = 0, qt = 0;
        qArr[qt++] = 0;
        visited[0] = true;
        int bfsCount = 0;

        while (qh < qt)
        {
            int u = qArr[qh++];
            bfsOrder[bfsCount++] = u;
            for (int h = head[u]; h != -1; h = nextE[h])
            {
                int v = toE[h];
                if (!visited[v])
                {
                    visited[v] = true;
                    parent[v] = u;
                    qArr[qt++] = v;
                }
            }
        }

        // ---------- T-join via spanning tree ----------
        // For each non-root vertex v, count odd-degree vertices in v's subtree.
        // If that count is odd, duplicate tree edge (parent(v), v).
        // This makes every vertex's degree even with at most N-1 duplications.
        int[] oddCount = new int[N];
        for (int i = 0; i < N; i++) oddCount[i] = (degree[i] & 1);

        for (int i = bfsCount - 1; i > 0; i--)
        {
            int v = bfsOrder[i];
            int p = parent[v];
            if (p >= 0) oddCount[p] += oddCount[v];
        }

        for (int i = 1; i < bfsCount; i++)
        {
            int v = bfsOrder[i];
            if ((oddCount[v] & 1) == 1)
            {
                int p = parent[v];
                toE[hc] = v;
                nextE[hc] = head[p];
                head[p] = hc++;

                toE[hc] = p;
                nextE[hc] = head[v];
                head[v] = hc++;

                degree[p]++;
                degree[v]++;
            }
        }

        // ---------- Iterative Hierholzer's algorithm from node 0 ----------
        int[] curHead = new int[N];
        Buffer.BlockCopy(head, 0, curHead, 0, N * sizeof(int));

        int totalEdges = hc >> 1;
        int routeLen = totalEdges + 1;

        int[] stk = new int[routeLen + 4];
        int sp = 0;
        stk[sp++] = 0;

        int[] route = new int[routeLen + 4];
        int rc = 0;

        while (sp > 0)
        {
            int v = stk[sp - 1];

            int h = curHead[v];
            while (h != -1 && usedE[h >> 1])
            {
                h = nextE[h];
            }
            curHead[v] = h;

            if (h == -1)
            {
                route[rc++] = v;
                sp--;
            }
            else
            {
                usedE[h >> 1] = true;
                curHead[v] = nextE[h];
                stk[sp++] = toE[h];
            }
        }

        // ---------- Fast output ----------
        var outStream = new BufferedStream(Console.OpenStandardOutput(), 1 << 16);

        // Optional length prefix (matches number of nodes listed).
        byte[] lenBytes = Encoding.ASCII.GetBytes(rc.ToString());
        outStream.Write(lenBytes, 0, lenBytes.Length);
        outStream.WriteByte((byte)'\n');

        byte[] tmp = new byte[12];
        for (int i = rc - 1; i >= 0; i--)
        {
            int val = route[i];
            int len = 0;
            if (val == 0)
            {
                tmp[len++] = (byte)'0';
            }
            else
            {
                int start = len;
                int t = val;
                while (t > 0) { tmp[len++] = (byte)('0' + (t % 10)); t /= 10; }
                int a = start, b = len - 1;
                while (a < b) { byte z = tmp[a]; tmp[a] = tmp[b]; tmp[b] = z; a++; b--; }
            }
            outStream.Write(tmp, 0, len);
            if (i > 0) outStream.WriteByte((byte)' ');
        }
        outStream.WriteByte((byte)'\n');
        outStream.Flush();
    }
}
