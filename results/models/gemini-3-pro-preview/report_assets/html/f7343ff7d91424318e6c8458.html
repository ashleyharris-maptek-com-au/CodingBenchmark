    // Custom Trailing Zero Count for 64-bit ulong using de Bruijn sequence
    private static readonly int[] DeBruijnTable64 = {
        0, 1, 56, 2, 57, 49, 28, 3, 61, 58, 42, 50, 38, 29, 17, 4,
        62, 47, 59, 36, 45, 43, 51, 22, 53, 39, 33, 30, 24, 18, 12, 5,
        63, 55, 48, 27, 60, 41, 37, 16, 46, 35, 44, 21, 52, 32, 23, 11,
        54, 26, 40, 15, 34, 20, 31, 10, 25, 14, 19, 9, 13, 8, 7, 6
    };

    public static int TrailingZeroCount(ulong v) {
        if (v == 0) return 64;
        ulong lsb = v & unchecked(~v + 1);
        return DeBruijnTable64[(lsb * 0x03F79D71B4CB0A89UL) >> 58];
    }

    public static int PopCount(ulong x) {
        x -= (x >> 1) & 0x5555555555555555UL;
        x = (x & 0x3333333333333333UL) + ((x >> 2) & 0x3333333333333333UL);
        x = (x + (x >> 4)) & 0x0f0f0f0f0f0f0f0fUL;
        return (int)((x * 0x0101010101010101UL) >> 56);
    }

    // Fast inline edge lookup via binary search
    static bool HasEdge(int[][] adj, int u, int v) {
        int[] neighbors = adj[u];
        if (neighbors.Length > adj[v].Length) {
            neighbors = adj[v];
            v = u;
        }
        int low = 0;
        int high = neighbors.Length - 1;
        while (low <= high) {
            int mid = (low + high) >>> 1;
            int val = neighbors[mid];
            if (val < v) {
                low = mid + 1;
            } else if (val > v) {
                high = mid - 1;
            } else {
                return true;
            }
        }
        return false;
    }

    // Fast I/O reader
    public class FastReader {
        private readonly Stream stream;
        private readonly byte[] buffer = new byte[65536];
        private int bufferPointer = 0;
        private int bytesRead = 0;

        public FastReader(Stream stream) {
            this.stream = stream;
        }

        private byte Read() {
            if (bufferPointer >= bytesRead) {
                bufferPointer = 0;
                bytesRead = stream.Read(buffer, 0, buffer.Length);
                if (bytesRead <= 0) return 0;
            }
            return buffer[bufferPointer++];
        }

        public int NextInt() {
            int ret = 0;
            byte c = Read();
            while (c <= 32) {
                if (c == 0) return -1;
                c = Read();
            }
            while (c > 32) {
                ret = ret * 10 + (c - '0');
                c = Read();
            }
            return ret;
        }
    }

    // Local search for clique improvement (1-opt swaps) without V-sized allocations
    static List<int> LocalSearchNoAlloc(List<int> initialClique, int[][] adj, int V) {
        var K = new List<int>(initialClique);
        K.Sort();
        
        bool improved = true;
        int stepsSinceImprovement = 0;
        List<int> affected = new List<int>();
        
        while (improved && stepsSinceImprovement < 1000) {
            improved = false;
            affected.Clear();
            
            foreach (int v in K) {
                foreach (int neighbor in adj[v]) {
                    if (K.BinarySearch(neighbor) < 0) {
                        affected.Add(neighbor);
                    }
                }
            }
            
            affected.Sort();
            int uniqueCount = 0;
            for (int i = 0; i < affected.Count; i++) {
                if (i == 0 || affected[i] != affected[i - 1]) {
                    affected[uniqueCount++] = affected[i];
                }
            }
            if (uniqueCount < affected.Count) {
                affected.RemoveRange(uniqueCount, affected.Count - uniqueCount);
            }
            
            int bestU = -1;
            foreach (int u in affected) {
                bool connectedToAll = true;
                foreach (int v in K) {
                    if (!HasEdge(adj, u, v)) {
                        connectedToAll = false;
                        break;
                    }
                }
                if (connectedToAll) {
                    bestU = u;
                    break;
                }
            }
            
            if (bestU != -1) {
                K.Add(bestU);
                K.Sort();
                improved = true;
                stepsSinceImprovement = 0;
                continue;
            }
            
            int swapU = -1;
            int swapW = -1;
            foreach (int u in affected) {
                int nonNeighborCount = 0;
                int nonNeighbor = -1;
                foreach (int w in K) {
                    if (!HasEdge(adj, u, w)) {
                        nonNeighborCount++;
                        nonNeighbor = w;
                        if (nonNeighborCount > 1) break;
                    }
                }
                if (nonNeighborCount == 1) {
                    swapU = u;
                    swapW = nonNeighbor;
                    break;
                }
            }
            
            if (swapU != -1) {
                K.Remove(swapW);
                K.Add(swapU);
                K.Sort();
                improved = true;
                stepsSinceImprovement++;
            }
        }
        return K;
    }

    static List<int> FindGreedyClique(int startVertex, int[][] adj) {
        var clique = new List<int> { startVertex };
        var candidates = new List<int>(adj[startVertex]);
        candidates.Sort((a, b) => adj[b].Length.CompareTo(adj[a].Length));
        
        foreach (var v in candidates) {
            bool ok = true;
            foreach (var cv in clique) {
                if (!HasEdge(adj, cv, v)) {
                    ok = false;
                    break;
                }
            }
            if (ok) {
                clique.Add(v);
            }
        }
        return clique;
    }

    static List<int> RunParallelHeuristic(int V, int[][] adj) {
        int[] sortedVertices = new int[V];
        for (int i = 0; i < V; i++) sortedVertices[i] = i;
        Array.Sort(sortedVertices, (a, b) => adj[b].Length.CompareTo(adj[a].Length));
        
        int numStarts = Math.Min(V, 200);
        List<int> bestClique = new List<int>();
        object lockObj = new object();
        
        Parallel.For(0, numStarts, idx => {
            int startVertex = sortedVertices[idx];
            var clique = FindGreedyClique(startVertex, adj);
            clique = LocalSearchNoAlloc(clique, adj, V);
            
            lock (lockObj) {
                if (clique.Count > bestClique.Count) {
                    bestClique = clique;
                }
            }
        });
        
        return bestClique;
    }

    struct DegeneracyResult {
        public int[] Order;
        public int Degeneracy;
    }

    static DegeneracyResult DegeneracyOrder(int V, int[][] adj) {
        int[] deg = new int[V];
        for (int i = 0; i < V; i++) deg[i] = adj[i].Length;
        
        int[] head = new int[V];
        for (int i = 0; i < V; i++) head[i] = -1;
        int[] next = new int[V];
        int[] prev = new int[V];
        
        for (int i = 0; i < V; i++) {
            int d = deg[i];
            next[i] = head[d];
            if (head[d] != -1) {
                prev[head[d]] = i;
            }
            prev[i] = -1;
            head[d] = i;
        }
        
        int[] order = new int[V];
        bool[] removed = new bool[V];
        int minDeg = 0;
        int degeneracy = 0;
        
        for (int step = 0; step < V; step++) {
            while (minDeg < V && head[minDeg] == -1) {
                minDeg++;
            }
            if (minDeg == V) break;
            
            int u = head[minDeg];
            head[minDeg] = next[u];
            if (next[u] != -1) {
                prev[next[u]] = -1;
            }
            
            removed[u] = true;
            order[step] = u;
            degeneracy = Math.Max(degeneracy, minDeg);
            
            foreach (int v in adj[u]) {
                if (!removed[v]) {
                    int dv = deg[v];
                    if (prev[v] != -1) {
                        next[prev[v]] = next[v];
                    } else {
                        head[dv] = next[v];
                    }
                    if (next[v] != -1) {
                        prev[next[v]] = prev[v];
                    }
                    
                    deg[v]--;
                    int ndv = deg[v];
                    next[v] = head[ndv];
                    if (head[ndv] != -1) {
                        prev[head[ndv]] = v;
                    }
                    prev[v] = -1;
                    head[ndv] = v;
                    
                    if (ndv < minDeg) {
                        minDeg = ndv;
                    }
                }
            }
        }
        
        return new DegeneracyResult { Order = order, Degeneracy = degeneracy };
    }

    // Exact Ostergard solver state (list-based)
    static int[] pool;
    static int[] levelStarts;
    static int[] counts;
    static int[] currentClique;
    static int[] c;
    static int maxFoundSize;
    static List<int> exactBestClique;
    static Stopwatch globalSw;

    static bool OstergardSearchList(int[][] adj, int level, int targetSize) {
        if (globalSw.ElapsedMilliseconds > 275000) { // 4 min 35 seconds
            throw new TimeoutException();
        }

        if (level + 1 == targetSize) {
            return true;
        }
        
        int start = levelStarts[level];
        int count = counts[level];
        
        if (level + 1 + count < targetSize) {
            return false;
        }
        
        int nextStart = start + count;
        levelStarts[level + 1] = nextStart;
        
        for (int j = 0; j < count; j++) {
            int v = pool[start + j];
            
            if (level + 1 + c[v] < targetSize) {
                break;
            }
            
            if (level + 1 + (count - j) < targetSize) {
                break;
            }
            
            int nextCount = 0;
            for (int k = j + 1; k < count; k++) {
                int w = pool[start + k];
                if (HasEdge(adj, v, w)) {
                    pool[nextStart + nextCount++] = w;
                }
            }
            
            counts[level + 1] = nextCount;
            currentClique[level + 1] = v;
            
            if (OstergardSearchList(adj, level + 1, targetSize)) {
                return true;
            }
        }
        
        return false;
    }

    // Exact Ostergard solver state (bitset-based)
    static ulong[] bitMatrix;
    static ulong[] bitPool;
    static int stride;

    static bool OstergardSearchBitset(int level, int targetSize) {
        if (globalSw.ElapsedMilliseconds > 275000) {
            throw new TimeoutException();
        }

        if (level == targetSize) {
            return true;
        }
        
        int startL = level * stride;
        int startNext = (level + 1) * stride;
        
        int remainingCount = 0;
        for (int j = 0; j < stride; j++) {
            remainingCount += PopCount(bitPool[startL + j]);
        }
        if (level + remainingCount < targetSize) {
            return false;
        }
        
        for (int j = 0; j < stride; j++) {
            ulong w = bitPool[startL + j];
            while (w != 0) {
                int tz = TrailingZeroCount(w);
                int v = (j << 6) + tz;
                
                if (level + c[v] < targetSize) {
                    return false;
                }
                
                int vWordIdx = v >> 6;
                for (int k = 0; k < vWordIdx; k++) {
                    bitPool[startNext + k] = 0;
                }
                ulong mask = (v & 63) == 63 ? 0UL : (~0UL << ((v & 63) + 1));
                bitPool[startNext + vWordIdx] = bitPool[startL + vWordIdx] & bitMatrix[v * stride + vWordIdx] & mask;
                for (int k = vWordIdx + 1; k < stride; k++) {
                    bitPool[startNext + k] = bitPool[startL + k] & bitMatrix[v * stride + k];
                }
                
                currentClique[level] = v;
                
                if (OstergardSearchBitset(level + 1, targetSize)) {
                    return true;
                }
                
                w &= w - 1UL;
            }
        }
        return false;
    }

    public static void Main() {
        globalSw = Stopwatch.StartNew();
        
        // Fast I/O Setup
        var stream = Console.OpenStandardInput();
        var reader = new FastReader(stream);
        
        int V = reader.NextInt();
        if (V <= 0) {
            Console.WriteLine(0);
            return;
        }
        int E = reader.NextInt();
        if (E < 0) E = 0;
        
        // Read edges into raw buffer to avoid multi-object list allocations
        int[] edgeU = new int[E];
        int[] edgeV = new int[E];
        int actualE = 0;
        for (int i = 0; i < E; i++) {
            int u = reader.NextInt();
            int v = reader.NextInt();
            if (u == -1 || v == -1) break;
            if (u == v) continue;
            if (u < 0 || u >= V || v < 0 || v >= V) continue;
            edgeU[actualE] = u;
            edgeV[actualE] = v;
            actualE++;
        }
        
        // Calculate degrees
        int[] deg = new int[V];
        for (int i = 0; i < actualE; i++) {
            deg[edgeU[i]]++;
            deg[edgeV[i]]++;
        }
        
        // Build CSR offsets
        int[] offset = new int[V + 1];
        for (int i = 0; i < V; i++) {
            offset[i + 1] = offset[i] + deg[i];
        }
        
        // Build CSR edges
        int[] csrEdges = new int[offset[V]];
        int[] curOffset = new int[V];
        Array.Copy(offset, curOffset, V);
        for (int i = 0; i < actualE; i++) {
            int u = edgeU[i];
            int v = edgeV[i];
            csrEdges[curOffset[u]++] = v;
            csrEdges[curOffset[v]++] = u;
        }
        
        // Sort and deduplicate CSR edges
        int[][] adj = new int[V][];
        int newTotalEdges = 0;
        int[] newOffset = new int[V + 1];
        for (int i = 0; i < V; i++) {
            newOffset[i] = newTotalEdges;
            int start = offset[i];
            int end = offset[i + 1];
            int count = end - start;
            if (count > 0) {
                Array.Sort(csrEdges, start, count);
                int uniqueCount = 0;
                for (int j = 0; j < count; j++) {
                    int val = csrEdges[start + j];
                    if (j == 0 || val != csrEdges[start + j - 1]) {
                        csrEdges[start + uniqueCount++] = val;
                    }
                }
                adj[i] = new int[uniqueCount];
                Array.Copy(csrEdges, start, adj[i], 0, uniqueCount);
                newTotalEdges += uniqueCount;
            } else {
                adj[i] = new int[0];
            }
        }
        newOffset[V] = newTotalEdges;
        
        // Free raw edge buffers early to reclaim memory
        edgeU = null;
        edgeV = null;
        csrEdges = null;
        offset = null;
        curOffset = null;
        newOffset = null;
        
        // 1. Run parallel heuristic search
        var heuristicClique = RunParallelHeuristic(V, adj);
        if (heuristicClique.Count == V) {
            Console.WriteLine(heuristicClique.Count);
            Console.WriteLine(string.Join(" ", heuristicClique));
            return;
        }
        
        // 2. Degeneracy ordering
        var degenResult = DegeneracyOrder(V, adj);
        int[] order = degenResult.Order;
        int degeneracy = degenResult.Degeneracy;
        
        if (heuristicClique.Count >= degeneracy + 1) {
            // Already matched maximum theoretical clique size
            Console.WriteLine(heuristicClique.Count);
            Console.WriteLine(string.Join(" ", heuristicClique));
            return;
        }
        
        // Map vertices to degeneracy positions
        int[] pos = new int[V];
        for (int i = 0; i < V; i++) {
            pos[order[i]] = i;
        }
        
        // Rename graph to match degeneracy ordering
        int[][] renamedAdj = new int[V][];
        List<int>[] renamedAdjList = new List<int>[V];
        for (int i = 0; i < V; i++) renamedAdjList[i] = new List<int>();
        for (int u = 0; u < V; u++) {
            int renamedU = pos[u];
            foreach (int v in adj[u]) {
                renamedAdjList[renamedU].Add(pos[v]);
            }
        }
        for (int i = 0; i < V; i++) {
            renamedAdjList[i].Sort();
            renamedAdj[i] = renamedAdjList[i].ToArray();
        }
        
        // Map heuristic clique to renamed vertices
        List<int> renamedHeuristic = new List<int>();
        foreach (int v in heuristicClique) {
            renamedHeuristic.Add(pos[v]);
        }
        renamedHeuristic.Sort();
        
        // Initialize exact solver parameters
        c = new int[V];
        currentClique = new int[V];
        maxFoundSize = heuristicClique.Count;
        exactBestClique = new List<int>(renamedHeuristic);
        
        // Decide solver mode: bitset for small dense, list-based for sparse/large
        bool useBitset = V <= 20000;
        
        try {
            if (useBitset) {
                stride = (V + 63) / 64;
                bitMatrix = new ulong[V * stride];
                for (int u = 0; u < V; u++) {
                    foreach (int v in renamedAdj[u]) {
                        bitMatrix[u * stride + (v >> 6)] |= 1UL << (v & 63);
                    }
                }
                bitPool = new ulong[(V + 1) * stride];
                
                for (int i = V - 1; i >= 0; i--) {
                    int targetSize = (i == V - 1) ? 1 : c[i + 1] + 1;
                    
                    int wordIdx = i >> 6;
                    int startNext = 0;
                    for (int j = 0; j < wordIdx; j++) {
                        bitPool[startNext + j] = 0;
                    }
                    ulong mask = (i & 63) == 63 ? 0UL : (~0UL << ((i & 63) + 1));
                    bitPool[startNext + wordIdx] = bitMatrix[i * stride + wordIdx] & mask;
                    for (int j = wordIdx + 1; j < stride; j++) {
                        bitPool[startNext + j] = bitMatrix[i * stride + j];
                    }
                    
                    currentClique[0] = i;
                    
                    if (OstergardSearchBitset(1, targetSize)) {
                        c[i] = targetSize;
                        if (targetSize > maxFoundSize) {
                            maxFoundSize = targetSize;
                            exactBestClique.Clear();
                            for (int k = 0; k < targetSize; k++) {
                                exactBestClique.Add(currentClique[k]);
                            }
                        }
                    } else {
                        c[i] = (i == V - 1) ? 0 : c[i + 1];
                    }
                }
            } else {
                // Prebuild adjLater for list-based solver
                int[][] adjLater = new int[V][];
                for (int i = 0; i < V; i++) {
                    List<int> later = new List<int>();
                    foreach (int v in renamedAdj[i]) {
                        if (v > i) later.Add(v);
                    }
                    adjLater[i] = later.ToArray();
                }
                
                pool = new int[V * 2];
                levelStarts = new int[V];
                counts = new int[V];
                
                for (int i = V - 1; i >= 0; i--) {
                    int targetSize = (i == V - 1) ? 1 : c[i + 1] + 1;
                    
                    levelStarts[0] = 0;
                    counts[0] = adjLater[i].Length;
                    Array.Copy(adjLater[i], 0, pool, 0, adjLater[i].Length);
                    
                    currentClique[0] = i;
                    
                    if (OstergardSearchList(renamedAdj, 0, targetSize)) {
                        c[i] = targetSize;
                        if (targetSize > maxFoundSize) {
                            maxFoundSize = targetSize;
                            exactBestClique.Clear();
                            for (int k = 0; k < targetSize; k++) {
                                exactBestClique.Add(currentClique[k]);
                            }
                        }
                    } else {
                        c[i] = (i == V - 1) ? 0 : c[i + 1];
                    }
                }
            }
        } catch (TimeoutException) {
            // Keep using the best clique found before timeout
        }
        
        // Map back to original indices
        List<int> finalClique = new List<int>();
        foreach (int v in exactBestClique) {
            finalClique.Add(order[v]);
        }
        finalClique.Sort();
        
        // Output result
        Console.WriteLine(finalClique.Count);
        Console.WriteLine(string.Join(" ", finalClique));
    }
