#[derive(Clone, Debug, PartialEq, Eq)]
struct BitSet {
    words: Vec<u64>,
}

impl BitSet {
    fn new(n: usize) -> Self {
        let num_words = (n + 63) / 64;
        BitSet { words: vec![0; num_words] }
    }

    fn set(&mut self, bit: usize) {
        let word = bit / 64;
        let pos = bit % 64;
        self.words[word] |= 1 << pos;
    }

    fn clear(&mut self, bit: usize) {
        let word = bit / 64;
        let pos = bit % 64;
        self.words[word] &= !(1 << pos);
    }

    fn test(&self, bit: usize) -> bool {
        let word = bit / 64;
        let pos = bit % 64;
        (self.words[word] & (1 << pos)) != 0
    }

    fn intersect(&self, other: &Self) -> Self {
        let mut res = BitSet { words: vec![0; self.words.len()] };
        for i in 0..self.words.len() {
            res.words[i] = self.words[i] & other.words[i];
        }
        res
    }

    fn count_ones(&self) -> u32 {
        let mut count = 0;
        for &w in &self.words {
            count += w.count_ones();
        }
        count
    }

    fn iter(&self) -> BitSetIter<'_> {
        BitSetIter {
            bitset: self,
            word_idx: 0,
            current_word: if self.words.is_empty() { 0 } else { self.words[0] },
        }
    }
}

struct BitSetIter<'a> {
    bitset: &'a BitSet,
    word_idx: usize,
    current_word: u64,
}

impl<'a> Iterator for BitSetIter<'a> {
    type Item = usize;

    fn next(&mut self) -> Option<Self::Item> {
        while self.current_word == 0 {
            self.word_idx += 1;
            if self.word_idx >= self.bitset.words.len() {
                return None;
            }
            self.current_word = self.bitset.words[self.word_idx];
        }
        let tz = self.current_word.trailing_zeros() as usize;
        let val = self.word_idx * 64 + tz;
        self.current_word &= self.current_word - 1;
        Some(val)
    }
}

fn compute_cut(part: &BitSet, adj: &[BitSet], n: usize) -> u32 {
    let mut cut = 0;
    let mut part1 = BitSet::new(n);
    for i in 0..n {
        if !part.test(i) {
            part1.set(i);
        }
    }
    for u in 0..n {
        if part.test(u) {
            cut += (adj[u].intersect(&part1)).count_ones();
        }
    }
    cut
}

fn kerninghan_lin(n: usize, adj: &[BitSet]) -> (u32, BitSet) {
    let mut best_cut = u32::MAX;
    let mut best_partition = BitSet::new(n);

    let mut rng = 123456789u64;
    let mut rand = || {
        rng = rng.wrapping_mul(6364136223846793005).wrapping_add(1);
        rng
    };

    let runs = if n < 50 { 20 } else { 10 };
    for _ in 0..runs {
        let mut vertices: Vec<usize> = (0..n).collect();
        for i in (1..n).rev() {
            let j = (rand() % (i as u64 + 1)) as usize;
            vertices.swap(i, j);
        }

        let mut part = BitSet::new(n);
        for i in 0..n / 2 {
            part.set(vertices[i]);
        }

        let mut improved = true;
        while improved {
            improved = false;
            let mut locked = BitSet::new(n);
            let mut swaps = Vec::with_capacity(n / 2);
            let mut current_part = part.clone();

            for _ in 0..n / 2 {
                let mut d = vec![0i32; n];
                for u in 0..n {
                    let is_in_0 = current_part.test(u);
                    let mut ext = 0;
                    let mut jnt = 0;
                    for v in 0..n {
                        if adj[u].test(v) {
                            if current_part.test(v) == is_in_0 {
                                jnt += 1;
                            } else {
                                ext += 1;
                            }
                        }
                    }
                    d[u] = ext - jnt;
                }

                let mut best_gain = i32::MIN;
                let mut best_u = usize::MAX;
                let mut best_v = usize::MAX;

                for u in 0..n {
                    if current_part.test(u) && !locked.test(u) {
                        for v in 0..n {
                            if !current_part.test(v) && !locked.test(v) {
                                let edge = if adj[u].test(v) { 1 } else { 0 };
                                let gain = d[u] + d[v] - 2 * edge;
                                if gain > best_gain {
                                    best_gain = gain;
                                    best_u = u;
                                    best_v = v;
                                }
                            }
                        }
                    }
                }

                if best_u == usize::MAX || best_v == usize::MAX {
                    break;
                }

                current_part.clear(best_u);
                current_part.set(best_v);
                locked.set(best_u);
                locked.set(best_v);
                swaps.push((best_u, best_v, best_gain));
            }

            let mut cum_gain = 0;
            let mut max_cum_gain = 0;
            let mut best_prefix_len = 0;
            for (i, &(_, _, gain)) in swaps.iter().enumerate() {
                cum_gain += gain;
                if cum_gain > max_cum_gain {
                    max_cum_gain = cum_gain;
                    best_prefix_len = i + 1;
                }
            }

            if max_cum_gain > 0 {
                for i in 0..best_prefix_len {
                    let (u, v, _) = swaps[i];
                    part.clear(u);
                    part.set(v);
                }
                improved = true;
            }
        }

        let cut = compute_cut(&part, adj, n);
        if cut < best_cut {
            best_cut = cut;
            best_partition = part;
        }
    }

    (best_cut, best_partition)
}

fn get_mcs_ordering(n: usize, adj: &[BitSet]) -> Vec<usize> {
    let mut ordered = Vec::with_capacity(n);
    let mut chosen = vec![false; n];
    let mut weight = vec![0; n];

    ordered.push(0);
    chosen[0] = true;
    for v in 0..n {
        if adj[0].test(v) {
            weight[v] += 1;
        }
    }

    for _ in 1..n {
        let mut max_w = -1;
        let mut best_v = usize::MAX;
        for v in 0..n {
            if !chosen[v] && weight[v] as i32 > max_w {
                max_w = weight[v] as i32;
                best_v = v;
            }
        }

        if best_v == usize::MAX {
            for v in 0..n {
                if !chosen[v] {
                    best_v = v;
                    break;
                }
            }
        }

        ordered.push(best_v);
        chosen[best_v] = true;
        for v in 0..n {
            if adj[best_v].test(v) {
                weight[v] += 1;
            }
        }
    }

    ordered
}

fn select_k_smallest(slice: &mut [i32], k: usize) -> i32 {
    if slice.len() <= 30 {
        for i in 1..slice.len() {
            let key = slice[i];
            let mut j = i;
            while j > 0 && slice[j - 1] > key {
                slice[j] = slice[j - 1];
                j -= 1;
            }
            slice[j] = key;
        }
        return slice[..=k].iter().sum();
    }
    let (left, &mut mid, _) = slice.select_nth_unstable(k);
    left.iter().sum::<i32>() + mid
}

struct Solver {
    n: usize,
    adj: Vec<BitSet>,
    deg: Vec<u32>,
    best_cut: u32,
    best_partition: BitSet,
    current_partition: BitSet,
    order: Vec<usize>,
}

impl Solver {
    fn new(n: usize, adj: Vec<BitSet>, initial_cut: u32, initial_partition: BitSet) -> Self {
        let order = get_mcs_ordering(n, &adj);
        let deg = adj.iter().map(|b| b.count_ones()).collect();
        Solver {
            n,
            adj,
            deg,
            best_cut: initial_cut,
            best_partition: initial_partition,
            current_partition: BitSet::new(n),
            order,
        }
    }

    fn solve(&mut self) {
        let mut d0 = vec![0u32; self.n];
        let mut d1 = vec![0u32; self.n];
        let mut diffs_buf = vec![0i32; self.n];

        // Fix the first vertex in the order to partition 0
        let first_v = self.order[0];
        self.current_partition.set(first_v);
        for v in self.adj[first_v].iter() {
            d0[v] += 1;
        }

        self.dfs(1, &mut d0, &mut d1, 1, 0, 0, &mut diffs_buf);
    }

    fn compute_lower_bound(
        &self,
        depth: usize,
        d0: &[u32],
        d1: &[u32],
        rem_0: usize,
        rem_1: usize,
        current_cut: u32,
        diffs_buf: &mut [i32],
    ) -> u32 {
        if rem_0 == 0 {
            let mut sum_d0 = 0;
            for idx in depth..self.n {
                let w = self.order[idx];
                sum_d0 += d0[w];
            }
            return current_cut + sum_d0;
        }
        if rem_1 == 0 {
            let mut sum_d1 = 0;
            for idx in depth..self.n {
                let w = self.order[idx];
                sum_d1 += d1[w];
            }
            return current_cut + sum_d1;
        }

        let mut sum_val_1 = 0;
        let len = self.n - depth;
        let slice = &mut diffs_buf[..len];

        for (i, idx) in (depth..self.n).enumerate() {
            let w = self.order[idx];
            let d0_w = d0[w];
            let d1_w = d1[w];
            let d_u = self.deg[w].saturating_sub(d0_w + d1_w);

            let c0 = if d_u + 1 >= rem_0 as u32 { d_u + 1 - rem_0 as u32 } else { 0 };
            let c1 = if d_u + 1 >= rem_1 as u32 { d_u + 1 - rem_1 as u32 } else { 0 };

            let val_0 = 2 * d1_w + c0;
            let val_1 = 2 * d0_w + c1;

            sum_val_1 += val_1;
            slice[i] = val_0 as i32 - val_1 as i32;
        }

        let sum_diffs = select_k_smallest(slice, rem_0 - 1);
        let double_cut = 2 * current_cut as i32 + sum_val_1 as i32 + sum_diffs;
        ((double_cut + 1) / 2) as u32
    }

    fn dfs(
        &mut self,
        depth: usize,
        d0: &mut [u32],
        d1: &mut [u32],
        size_0: usize,
        size_1: usize,
        current_cut: u32,
        diffs_buf: &mut [i32],
    ) {
        if depth == self.n {
            if size_0 == self.n / 2 && size_1 == self.n / 2 {
                if current_cut < self.best_cut {
                    self.best_cut = current_cut;
                    self.best_partition = self.current_partition.clone();
                }
            }
            return;
        }

        let rem_0 = self.n / 2 - size_0;
        let rem_1 = self.n / 2 - size_1;

        if rem_0 > self.n - depth || rem_1 > self.n - depth {
            return;
        }

        let lb = self.compute_lower_bound(depth, d0, d1, rem_0, rem_1, current_cut, diffs_buf);
        if lb >= self.best_cut {
            return;
        }

        let u = self.order[depth];
        let cut_inc_0 = d1[u];
        let cut_inc_1 = d0[u];

        let can_0 = rem_0 > 0;
        let can_1 = rem_1 > 0;

        if can_0 && can_1 {
            if cut_inc_0 <= cut_inc_1 {
                // Try 0 first
                self.current_partition.set(u);
                for v in self.adj[u].iter() {
                    d0[v] += 1;
                }
                self.dfs(depth + 1, d0, d1, size_0 + 1, size_1, current_cut + cut_inc_0, diffs_buf);
                for v in self.adj[u].iter() {
                    d0[v] -= 1;
                }

                // Try 1
                if current_cut + cut_inc_1 < self.best_cut {
                    self.current_partition.clear(u);
                    for v in self.adj[u].iter() {
                        d1[v] += 1;
                    }
                    self.dfs(depth + 1, d0, d1, size_0, size_1 + 1, current_cut + cut_inc_1, diffs_buf);
                    for v in self.adj[u].iter() {
                        d1[v] -= 1;
                    }
                }
            } else {
                // Try 1 first
                self.current_partition.clear(u);
                for v in self.adj[u].iter() {
                    d1[v] += 1;
                }
                self.dfs(depth + 1, d0, d1, size_0, size_1 + 1, current_cut + cut_inc_1, diffs_buf);
                for v in self.adj[u].iter() {
                    d1[v] -= 1;
                }

                // Try 0
                if current_cut + cut_inc_0 < self.best_cut {
                    self.current_partition.set(u);
                    for v in self.adj[u].iter() {
                        d0[v] += 1;
                    }
                    self.dfs(depth + 1, d0, d1, size_0 + 1, size_1, current_cut + cut_inc_0, diffs_buf);
                    for v in self.adj[u].iter() {
                        d0[v] -= 1;
                    }
                }
            }
        } else if can_0 {
            self.current_partition.set(u);
            for v in self.adj[u].iter() {
                d0[v] += 1;
            }
            self.dfs(depth + 1, d0, d1, size_0 + 1, size_1, current_cut + cut_inc_0, diffs_buf);
            for v in self.adj[u].iter() {
                d0[v] -= 1;
            }
        } else if can_1 {
            self.current_partition.clear(u);
            for v in self.adj[u].iter() {
                d1[v] += 1;
            }
            self.dfs(depth + 1, d0, d1, size_0, size_1 + 1, current_cut + cut_inc_1, diffs_buf);
            for v in self.adj[u].iter() {
                d1[v] -= 1;
            }
        }
    }
}

fn main() {
    let start_time = Instant::now();
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let mut first_line = String::new();
    while let Some(Ok(line)) = lines.next() {
        let trimmed = line.trim();
        if !trimmed.is_empty() {
            first_line = trimmed.to_string();
            break;
        }
    }

    if first_line.is_empty() {
        return;
    }

    let parts: Vec<&str> = first_line.split_whitespace().collect();
    if parts.len() < 2 {
        return;
    }

    let n: usize = parts[0].parse().unwrap();
    let m: usize = parts[1].parse().unwrap();

    let mut adj = vec![BitSet::new(n); n];
    let mut edge_count = 0;
    while edge_count < m {
        if let Some(Ok(line)) = lines.next() {
            let trimmed = line.trim();
            if trimmed.is_empty() {
                continue;
            }
            let edge_parts: Vec<&str> = trimmed.split_whitespace().collect();
            if edge_parts.len() < 2 {
                continue;
            }
            let u: usize = edge_parts[0].parse().unwrap();
            let v: usize = edge_parts[1].parse().unwrap();
            adj[u].set(v);
            adj[v].set(u);
            edge_count += 1;
        } else {
            break;
        }
    }

    let (kl_cut, kl_part) = kerninghan_lin(n, &adj);

    let mut solver = Solver::new(n, adj, kl_cut, kl_part);
    solver.solve();

    println!("{}", solver.best_cut);
    let mut part_a = Vec::new();
    for i in 0..n {
        if solver.best_partition.test(i) {
            part_a.push(i.to_string());
        }
    }
    println!("{}", part_a.join(" "));

    eprintln!("Time elapsed: {:?}", start_time.elapsed());
}
