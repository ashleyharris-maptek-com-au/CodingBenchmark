#![allow(non_snake_case)]

use std::io::{self, BufRead, Write};

const WALL_MASK: u8 = 0x01;
const FWD_MASK: u8 = 0x0E;
const FWD_SHIFT: u32 = 1;
const BWD_MASK: u8 = 0x70;
const BWD_SHIFT: u32 = 4;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
#[repr(u8)]
enum Dir {
    None = 0,
    Start = 1,
    Up = 2,
    Down = 3,
    Left = 4,
    Right = 5,
}

impl Dir {
    #[inline(always)]
    fn from_u8(v: u8) -> Self {
        match v {
            1 => Dir::Start,
            2 => Dir::Up,
            3 => Dir::Down,
            4 => Dir::Left,
            5 => Dir::Right,
            _ => Dir::None,
        }
    }
}

#[inline(always)]
fn set_fwd(cell: &mut u8, dir: Dir) {
    *cell = (*cell & !FWD_MASK) | ((dir as u8) << FWD_SHIFT);
}

#[inline(always)]
fn get_fwd(cell: u8) -> Dir {
    Dir::from_u8((cell & FWD_MASK) >> FWD_SHIFT)
}

#[inline(always)]
fn set_bwd(cell: &mut u8, dir: Dir) {
    *cell = (*cell & !BWD_MASK) | ((dir as u8) << BWD_SHIFT);
}

#[inline(always)]
fn get_bwd(cell: u8) -> Dir {
    Dir::from_u8((cell & BWD_MASK) >> BWD_SHIFT)
}

#[derive(Clone, Copy, Debug)]
struct Pos {
    y: u32,
    x: u32,
}

fn reconstruct_path(grid: &[u8], meet_idx: usize, W: usize) -> Vec<u32> {
    let mut fwd_path = Vec::new();
    let mut curr = meet_idx;
    loop {
        fwd_path.push(curr as u32);
        match get_fwd(grid[curr]) {
            Dir::Start => break,
            Dir::Up => curr -= W,
            Dir::Down => curr += W,
            Dir::Left => curr -= 1,
            Dir::Right => curr += 1,
            Dir::None => panic!("Invalid forward path"),
        }
    }
    fwd_path.reverse();

    let mut bwd_path = Vec::new();
    curr = meet_idx;
    loop {
        match get_bwd(grid[curr]) {
            Dir::Start => break,
            Dir::Up => curr -= W,
            Dir::Down => curr += W,
            Dir::Left => curr -= 1,
            Dir::Right => curr += 1,
            Dir::None => panic!("Invalid backward path"),
        }
        bwd_path.push(curr as u32);
    }

    fwd_path.extend(bwd_path);
    fwd_path
}

#[inline(always)]
fn try_visit_fwd(
    grid: &mut [u8],
    n_idx: usize,
    parent_dir: Dir,
    next_fwd: &mut Vec<Pos>,
    n_pos: Pos,
) -> Option<usize> {
    let cell = unsafe { *grid.get_unchecked(n_idx) };
    if (cell & WALL_MASK) != 0 {
        return None;
    }
    if get_fwd(cell) != Dir::None {
        return None;
    }
    unsafe {
        let cell_mut = grid.get_unchecked_mut(n_idx);
        set_fwd(cell_mut, parent_dir);
    }
    next_fwd.push(n_pos);

    let cell = unsafe { *grid.get_unchecked(n_idx) };
    if get_bwd(cell) != Dir::None {
        return Some(n_idx);
    }
    None
}

#[inline(always)]
fn try_visit_bwd(
    grid: &mut [u8],
    n_idx: usize,
    parent_dir: Dir,
    next_bwd: &mut Vec<Pos>,
    n_pos: Pos,
) -> Option<usize> {
    let cell = unsafe { *grid.get_unchecked(n_idx) };
    if (cell & WALL_MASK) != 0 {
        return None;
    }
    if get_bwd(cell) != Dir::None {
        return None;
    }
    unsafe {
        let cell_mut = grid.get_unchecked_mut(n_idx);
        set_bwd(cell_mut, parent_dir);
    }
    next_bwd.push(n_pos);

    let cell = unsafe { *grid.get_unchecked(n_idx) };
    if get_fwd(cell) != Dir::None {
        return Some(n_idx);
    }
    None
}

fn solve() -> io::Result<()> {
    let stdin = io::stdin();
    let mut reader = io::BufReader::with_capacity(262_144, stdin.lock());

    let mut first_line = String::new();
    reader.read_line(&mut first_line)?;
    let parts: Vec<&str> = first_line.split_whitespace().collect();
    if parts.len() < 2 {
        return Ok(());
    }
    let H: usize = parts[0].parse().unwrap();
    let W: usize = parts[1].parse().unwrap();

    let mut grid = vec![0u8; H * W];
    let mut start_idx = None;
    let mut end_idx = None;

    let mut line_buf = Vec::new();
    for y in 0..H {
        line_buf.clear();
        reader.read_until(b'\n', &mut line_buf)?;
        let mut len = line_buf.len();
        while len > 0 && (line_buf[len - 1] == b'\n' || line_buf[len - 1] == b'\r') {
            len -= 1;
        }
        let row = &line_buf[..len];
        let limit = W.min(row.len());
        let row_slice = &row[..limit];
        for (x, &c) in row_slice.iter().enumerate() {
            let idx = y * W + x;
            if c == b'#' {
                grid[idx] = WALL_MASK;
            } else if c == b'A' {
                start_idx = Some(idx);
            } else if c == b'B' {
                end_idx = Some(idx);
            }
        }
        for x in limit..W {
            let idx = y * W + x;
            grid[idx] = WALL_MASK;
        }
    }

    let start_idx = match start_idx {
        Some(idx) => idx,
        None => {
            println!("0");
            return Ok(());
        }
    };
    let end_idx = match end_idx {
        Some(idx) => idx,
        None => {
            println!("0");
            return Ok(());
        }
    };

    if start_idx == end_idx {
        println!("1");
        println!("{} {}", start_idx % W, start_idx / W);
        return Ok(());
    }

    let H_u32 = H as u32;
    let W_u32 = W as u32;

    let mut fwd_frontier = Vec::new();
    let mut next_fwd = Vec::new();

    let mut bwd_frontier = Vec::new();
    let mut next_bwd = Vec::new();

    fwd_frontier.push(Pos {
        y: (start_idx / W) as u32,
        x: (start_idx % W) as u32,
    });
    set_fwd(&mut grid[start_idx], Dir::Start);

    bwd_frontier.push(Pos {
        y: (end_idx / W) as u32,
        x: (end_idx % W) as u32,
    });
    set_bwd(&mut grid[end_idx], Dir::Start);

    let mut path = None;

    'outer: while !fwd_frontier.is_empty() && !bwd_frontier.is_empty() {
        if fwd_frontier.len() <= bwd_frontier.len() {
            next_fwd.clear();
            for &pos in &fwd_frontier {
                let y = pos.y;
                let x = pos.x;
                let idx = (y as usize) * W + (x as usize);

                // Up
                if y > 0 {
                    let n_pos = Pos { y: y - 1, x };
                    let n_idx = idx - W;
                    if let Some(meet) = try_visit_fwd(&mut grid, n_idx, Dir::Down, &mut next_fwd, n_pos) {
                        path = Some(reconstruct_path(&grid, meet, W));
                        break 'outer;
                    }
                }
                // Down
                if y < H_u32 - 1 {
                    let n_pos = Pos { y: y + 1, x };
                    let n_idx = idx + W;
                    if let Some(meet) = try_visit_fwd(&mut grid, n_idx, Dir::Up, &mut next_fwd, n_pos) {
                        path = Some(reconstruct_path(&grid, meet, W));
                        break 'outer;
                    }
                }
                // Left
                if x > 0 {
                    let n_pos = Pos { y, x: x - 1 };
                    let n_idx = idx - 1;
                    if let Some(meet) = try_visit_fwd(&mut grid, n_idx, Dir::Right, &mut next_fwd, n_pos) {
                        path = Some(reconstruct_path(&grid, meet, W));
                        break 'outer;
                    }
                }
                // Right
                if x < W_u32 - 1 {
                    let n_pos = Pos { y, x: x + 1 };
                    let n_idx = idx + 1;
                    if let Some(meet) = try_visit_fwd(&mut grid, n_idx, Dir::Left, &mut next_fwd, n_pos) {
                        path = Some(reconstruct_path(&grid, meet, W));
                        break 'outer;
                    }
                }
            }
            std::mem::swap(&mut fwd_frontier, &mut next_fwd);
        } else {
            next_bwd.clear();
            for &pos in &bwd_frontier {
                let y = pos.y;
                let x = pos.x;
                let idx = (y as usize) * W + (x as usize);

                // Up
                if y > 0 {
                    let n_pos = Pos { y: y - 1, x };
                    let n_idx = idx - W;
                    if let Some(meet) = try_visit_bwd(&mut grid, n_idx, Dir::Down, &mut next_bwd, n_pos) {
                        path = Some(reconstruct_path(&grid, meet, W));
                        break 'outer;
                    }
                }
                // Down
                if y < H_u32 - 1 {
                    let n_pos = Pos { y: y + 1, x };
                    let n_idx = idx + W;
                    if let Some(meet) = try_visit_bwd(&mut grid, n_idx, Dir::Up, &mut next_bwd, n_pos) {
                        path = Some(reconstruct_path(&grid, meet, W));
                        break 'outer;
                    }
                }
                // Left
                if x > 0 {
                    let n_pos = Pos { y, x: x - 1 };
                    let n_idx = idx - 1;
                    if let Some(meet) = try_visit_bwd(&mut grid, n_idx, Dir::Right, &mut next_bwd, n_pos) {
                        path = Some(reconstruct_path(&grid, meet, W));
                        break 'outer;
                    }
                }
                // Right
                if x < W_u32 - 1 {
                    let n_pos = Pos { y, x: x + 1 };
                    let n_idx = idx + 1;
                    if let Some(meet) = try_visit_bwd(&mut grid, n_idx, Dir::Left, &mut next_bwd, n_pos) {
                        path = Some(reconstruct_path(&grid, meet, W));
                        break 'outer;
                    }
                }
            }
            std::mem::swap(&mut bwd_frontier, &mut next_bwd);
        }
    }

    if let Some(p) = path {
        let stdout = io::stdout();
        let mut writer = io::BufWriter::with_capacity(262_144, stdout.lock());
        writeln!(writer, "{}", p.len())?;
        for idx in p {
            let idx = idx as usize;
            writeln!(writer, "{} {}", idx % W, idx / W)?;
        }
    } else {
        println!("0");
    }

    Ok(())
}

fn main() -> io::Result<()> {
    solve()
}
