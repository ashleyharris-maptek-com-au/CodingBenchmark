#version 450

layout(local_size_x = 256) in;

// Each boid stored as two vec4s: [pos.xyz, 0], [vel.xyz, 0]
// Boid i is at index i*2 and i*2+1
layout(set = 0, binding = 0) readonly buffer InputBoids {
    vec4 inData[];   // length = N * 2
};

layout(set = 0, binding = 1) buffer OutputBoids {
    vec4 outData[];  // length = N * 2
};

layout(set = 0, binding = 2) uniform Params {
    uvec4 counts;      // counts.x = N
    vec4 radii;        // radii.x=dt, radii.y=sepRadius, radii.z=alignRadius, radii.w=cohRadius
    vec4 weights;      // weights.x=sepWeight, weights.y=alignWeight, weights.z=cohWeight, weights.w=maxSpeed
    vec4 bounds;       // bounds.x=boundSize, bounds.y=boundForce
};

void main() {
    uint i = gl_GlobalInvocationID.x;
    uint N = counts.x;
    if (i >= N) {
        return;
    }

    vec3 pos = inData[i * 2].xyz;
    vec3 vel = inData[i * 2 + 1].xyz;

    float dt = radii.x;
    float sepRadius = radii.y;
    float alignRadius = radii.z;
    float cohRadius = radii.w;

    float sepWeight = weights.x;
    float alignWeight = weights.y;
    float cohWeight = weights.z;
    float maxSpeed = weights.w;

    float boundSize = bounds.x;
    float boundForce = bounds.y;

    vec3 sepSum = vec3(0.0);
    vec3 alignSum = vec3(0.0);
    vec3 cohSum = vec3(0.0);

    uint sepCount = 0;
    uint alignCount = 0;
    uint cohCount = 0;

    for (uint j = 0; j < N; j++) {
        if (i == j) {
            continue;
        }

        vec3 neighborPos = inData[j * 2].xyz;
        vec3 neighborVel = inData[j * 2 + 1].xyz;

        float dist = distance(pos, neighborPos);
        if (dist == 0.0) {
            continue;
        }

        // Separation (avoid crowding)
        if (dist < sepRadius) {
            sepSum += (pos - neighborPos) / dist;
            sepCount++;
        }

        // Alignment (align velocity)
        if (dist < alignRadius) {
            alignSum += neighborVel;
            alignCount++;
        }

        // Cohesion (stay close to neighbors)
        if (dist < cohRadius) {
            cohSum += neighborPos;
            cohCount++;
        }
    }

    vec3 sepForce = vec3(0.0);
    if (sepCount > 0) {
        sepForce = sepSum / float(sepCount);
    }

    vec3 alignForce = vec3(0.0);
    if (alignCount > 0) {
        alignForce = alignSum / float(alignCount);
    }

    vec3 cohForce = vec3(0.0);
    if (cohCount > 0) {
        vec3 cohAvgPos = cohSum / float(cohCount);
        cohForce = cohAvgPos - pos;
    }

    // Boundary force: push back when a component exceeds boundSize * 0.4
    vec3 boundaryForce = vec3(0.0);
    if (abs(pos.x) > boundSize * 0.4) {
        boundaryForce.x = -sign(pos.x) * boundForce * (abs(pos.x) / (boundSize * 0.5));
    }
    if (abs(pos.y) > boundSize * 0.4) {
        boundaryForce.y = -sign(pos.y) * boundForce * (abs(pos.y) / (boundSize * 0.5));
    }
    if (abs(pos.z) > boundSize * 0.4) {
        boundaryForce.z = -sign(pos.z) * boundForce * (abs(pos.z) / (boundSize * 0.5));
    }

    // Combine forces
    vec3 accel = (sepForce * sepWeight) + (alignForce * alignWeight) + (cohForce * cohWeight) + boundaryForce;

    // Update velocity and clamp speed to maxSpeed
    vel += accel * dt;
    float speed = length(vel);
    if (speed > maxSpeed) {
        vel = (vel / speed) * maxSpeed;
    }

    // Update position
    pos += vel * dt;

    // Write output data
    outData[i * 2] = vec4(pos, 0.0);
    outData[i * 2 + 1] = vec4(vel, 0.0);
}
