import math

# Global persistence memory
_memory = {}

def is_beam_hitting_obstacle(angle_deg, r, y_est, heading):
    angle_road = math.radians(angle_deg) + heading
    sin_ar = math.sin(angle_road)
    
    d_bound = 200.0
    if sin_ar < -0.001:
        d_bound = -y_est / sin_ar
    elif sin_ar > 0.001:
        d_bound = (11.1 - y_est) / sin_ar
        
    return r < (d_bound - 3.0)

def autopilot_step(state: dict, dt: float) -> dict:
    global _memory
    
    # 1. Reset memory if it's a new scenario
    sim_time = state.get('sim_time_s', 0.0)
    if sim_time < 0.05 or 'initialized' not in _memory:
        _memory.clear()
        _memory['initialized'] = True
        _memory['last_steering'] = 0.0
        _memory['last_throttle'] = 0.3
        _memory['last_brake'] = 0.0
        _memory['speed_integral'] = 0.0
        _memory['y_integral'] = 0.0
        _memory['speed_est_kmh'] = 110.0
        _memory['steer_est'] = 0.0
        _memory['prev_steer_est'] = 0.0
        _memory['yaw_rate_est'] = 0.0
        _memory['y_est'] = 5.55 # Center of middle lane
        _memory['heading_est'] = 0.0
        _memory['jam_ticks'] = 0
        _memory['steering_jammed'] = False

    # 2. Sensor Fusion for Speed (km/h)
    speed_candidates = []
    if 'vehicle_speed_1_kmh' in state:
        speed_candidates.append(state['vehicle_speed_1_kmh'])
    if 'vehicle_speed_2_kmh' in state:
        speed_candidates.append(state['vehicle_speed_2_kmh'])
    for i in [1, 2]:
        if state.get(f'gps_{i}_valid') and f'gps_{i}_speed_kmh' in state:
            speed_candidates.append(state[f'gps_{i}_speed_kmh'])
    for name in ['fl', 'fr', 'rl', 'rr']:
        if state.get(f'wheel_speed_{name}_valid') and f'wheel_speed_{name}_kmh' in state:
            speed_candidates.append(state[f'wheel_speed_{name}_kmh'])
            
    if speed_candidates:
        speed_candidates.sort()
        speed_est_kmh = speed_candidates[len(speed_candidates) // 2]
    else:
        speed_est_kmh = _memory['speed_est_kmh']
        
    # Low-pass filter for speed
    speed_est_kmh = 0.8 * _memory['speed_est_kmh'] + 0.2 * speed_est_kmh
    _memory['speed_est_kmh'] = speed_est_kmh
    speed_est_ms = speed_est_kmh / 3.6

    # 3. Sensor Fusion for Steering Angle (rad)
    steer_candidates = []
    for i in [1, 2]:
        if state.get(f'steering_angle_{i}_valid') and f'steering_angle_{i}_deg' in state:
            steer_candidates.append(math.radians(state[f'steering_angle_{i}_deg']))
            
    if steer_candidates:
        if len(steer_candidates) == 2:
            # Check for disagreement by cross-checking with expected steer from gyro yaw rate
            gyro_rates = []
            for i in [1, 2, 3]:
                if state.get(f'gyro_{i}_valid') and f'gyro_{i}_yaw_rate_dps' in state:
                    gyro_rates.append(math.radians(state[f'gyro_{i}_yaw_rate_dps']))
            
            if gyro_rates and speed_est_ms > 2.0:
                gyro_rates.sort()
                yaw_rate_med = gyro_rates[len(gyro_rates) // 2]
                steer_expected = math.atan2(yaw_rate_med * 2.7, speed_est_ms)
                
                diff0 = abs(steer_candidates[0] - steer_expected)
                diff1 = abs(steer_candidates[1] - steer_expected)
                if diff0 < diff1:
                    steer_est = steer_candidates[0]
                else:
                    steer_est = steer_candidates[1]
            else:
                last_steer = _memory['last_steering'] * (35.0 * math.pi / 180.0)
                diff0 = abs(steer_candidates[0] - last_steer)
                diff1 = abs(steer_candidates[1] - last_steer)
                if diff0 < diff1:
                    steer_est = steer_candidates[0]
                else:
                    steer_est = steer_candidates[1]
        else:
            steer_est = steer_candidates[0]
    else:
        steer_est = _memory['steer_est']
        
    # No filter for steer_est to avoid delay in jam detection
    _memory['steer_est'] = steer_est

    # 4. Sensor Fusion for Yaw Rate (rad/s)
    gyro_candidates = []
    for i in [1, 2, 3]:
        if state.get(f'gyro_{i}_valid') and f'gyro_{i}_yaw_rate_dps' in state:
            gyro_candidates.append(math.radians(state[f'gyro_{i}_yaw_rate_dps']))
            
    expected_yaw_rate = speed_est_ms * math.tan(steer_est) / 2.7
    
    if gyro_candidates:
        valid_gyros = []
        for val in gyro_candidates:
            if abs(val - expected_yaw_rate) < math.radians(15.0):
                valid_gyros.append(val)
        if valid_gyros:
            valid_gyros.sort()
            yaw_rate_est = valid_gyros[len(valid_gyros) // 2]
        else:
            gyro_candidates.sort()
            yaw_rate_est = gyro_candidates[len(gyro_candidates) // 2]
    else:
        yaw_rate_est = expected_yaw_rate
        
    yaw_rate_est = 0.8 * _memory['yaw_rate_est'] + 0.2 * yaw_rate_est
    _memory['yaw_rate_est'] = yaw_rate_est

    # 5. Sensor Fusion for Lateral Position y (m)
    curr_lane = state.get('current_lane', 1)
    lane_width = 3.7
    y_candidates = []
    
    for i in [1, 2]:
        if state.get(f'gps_{i}_valid') and f'gps_{i}_y_m' in state:
            y_candidates.append(state[f'gps_{i}_y_m'])
            
    for i in [1, 2, 3]:
        if f'vision_{i}_lane_offset_m' in state:
            y_candidates.append((curr_lane + 0.5) * lane_width + state[f'vision_{i}_lane_offset_m'])
        if state.get(f'vision_{i}_left_line_valid') and f'vision_{i}_left_line_dist_m' in state:
            y_candidates.append(curr_lane * lane_width + state[f'vision_{i}_left_line_dist_m'])
        if state.get(f'vision_{i}_right_line_valid') and f'vision_{i}_right_line_dist_m' in state:
            y_candidates.append((curr_lane + 1) * lane_width - state[f'vision_{i}_right_line_dist_m'])
            
    if y_candidates:
        y_candidates.sort()
        y_est = y_candidates[len(y_candidates) // 2]
    else:
        y_est = _memory['y_est']
        
    y_est = 0.8 * _memory['y_est'] + 0.2 * y_est
    _memory['y_est'] = y_est

    # 6. Sensor Fusion for Heading (rad)
    gps_headings = []
    for i in [1, 2]:
        if state.get(f'gps_{i}_valid') and f'gps_{i}_heading_deg' in state:
            h_rad = math.radians(state[f'gps_{i}_heading_deg'])
            h_rad = (h_rad + math.pi) % (2 * math.pi) - math.pi
            gps_headings.append(h_rad)
            
    if gps_headings:
        gps_headings.sort()
        gps_heading = gps_headings[len(gps_headings) // 2]
        if 'heading_est' not in _memory or _memory['heading_est'] is None:
            _memory['heading_est'] = gps_heading
        else:
            diff = (gps_heading - _memory['heading_est'] + math.pi) % (2 * math.pi) - math.pi
            _memory['heading_est'] += yaw_rate_est * dt + 0.1 * diff
    else:
        _memory['heading_est'] += yaw_rate_est * dt
        
    _memory['heading_est'] = (_memory['heading_est'] + math.pi) % (2 * math.pi) - math.pi
    heading_est = _memory['heading_est']

    # 7. Obstacle Detection Fusion
    obs_detected_votes = []
    obs_dists = []
    obs_rel_speeds = []
    
    for i in [1, 2, 3]:
        if state.get(f'vision_{i}_obstacle_valid', True):
            det = state.get(f'vision_{i}_obstacle_detected', False)
            obs_detected_votes.append(det)
            if det:
                obs_dists.append(state.get(f'vision_{i}_obstacle_dist_m', 200.0))
                obs_rel_speeds.append(state.get(f'vision_{i}_obstacle_rel_speed_kmh', 0.0) / 3.6)
                
    if obs_detected_votes:
        obs_detected = sum(obs_detected_votes) >= (len(obs_detected_votes) + 1) // 2
    else:
        obs_detected = False
        
    if obs_detected and obs_dists:
        obs_dists.sort()
        obs_rel_speeds.sort()
        obs_dist = obs_dists[len(obs_dists) // 2]
        obs_rel_speed = obs_rel_speeds[len(obs_rel_speeds) // 2]
        obs_speed_ms = speed_est_ms - obs_rel_speed
    else:
        obs_dist = 200.0
        obs_speed_ms = 0.0
        
    # Lidar safety backup for close obstacles (with boundary check)
    lidar_obstacle = False
    lidar_min_dist = 200.0
    for angle in [-5, 5]:
        key = f'lidar_{angle:+04d}_deg'
        if state.get(f'{key}_valid') and f'{key}_range_m' in state:
            r = state[f'{key}_range_m']
            
            # Filter out road boundary hits from lidar range
            angle_road = math.radians(angle) + heading_est
            sin_ar = math.sin(angle_road)
            d_bound = 200.0
            if sin_ar < -0.001:
                d_bound = -y_est / sin_ar
            elif sin_ar > 0.001:
                d_bound = (11.1 - y_est) / sin_ar
                
            # Only count as obstacle if range is < 30m and significantly shorter than road boundary
            if r < 30.0 and r < (d_bound - 3.0):
                lidar_obstacle = True
                lidar_min_dist = min(lidar_min_dist, r)
                
    if lidar_obstacle and (not obs_detected or lidar_min_dist < obs_dist):
        obs_detected = True
        obs_dist = lidar_min_dist
        obs_speed_ms = 0.0 # safety assumption

    # 8. Steering Jam Detection (5 consecutive steps with tracking error > 1.5 degrees)
    last_steer_cmd_rad = _memory['last_steering'] * (35.0 * math.pi / 180.0)
    steer_error = abs(steer_est - last_steer_cmd_rad)
    
    if steer_error > math.radians(1.5):
        _memory['jam_ticks'] += 1
    else:
        _memory['jam_ticks'] = 0
        
    if _memory['jam_ticks'] >= 5:
        _memory['steering_jammed'] = True

    # 9. Friction (Traction) Estimation
    mu_est = state.get('estimated_mu', 1.0)
    if 'mu_filtered' not in _memory:
        _memory['mu_filtered'] = mu_est
    else:
        _memory['mu_filtered'] = 0.9 * _memory['mu_filtered'] + 0.1 * mu_est
    mu_filtered = _memory['mu_filtered']

    # 10. Active Lane Change Decision for Obstacle Avoidance
    # Trigger earlier (120m) to give enough distance on low friction surfaces like ice
    if obs_detected and obs_dist < 120.0 and obs_speed_ms < 15.0 and (mu_filtered >= 0.5 or speed_est_kmh < 65.0):
        if 'lane_change_target' not in _memory:
            # We want to change lanes. Find if left/right lanes are free.
            left_free = True
            if curr_lane == 0:
                left_free = False
            else:
                for angle in [-15, -30, -45, -60]:
                    key = f'lidar_{angle:+04d}_deg'
                    if state.get(f'{key}_valid') and f'{key}_range_m' in state:
                        r = state[f'{key}_range_m']
                        if is_beam_hitting_obstacle(angle, r, y_est, heading_est):
                            left_free = False
                        
            right_free = True
            if curr_lane == 2:
                right_free = False
            else:
                for angle in [15, 30, 45, 60]:
                    key = f'lidar_{angle:+04d}_deg'
                    if state.get(f'{key}_valid') and f'{key}_range_m' in state:
                        r = state[f'{key}_range_m']
                        if is_beam_hitting_obstacle(angle, r, y_est, heading_est):
                            right_free = False
                        
            if curr_lane == 1:
                if right_free:
                    _memory['lane_change_target'] = 2
                elif left_free:
                    _memory['lane_change_target'] = 0
            elif curr_lane == 2 and left_free:
                _memory['lane_change_target'] = 1
            elif curr_lane == 0 and right_free:
                _memory['lane_change_target'] = 1

    # Apply lane change target if active
    target_lane = state.get('target_lane', 1)
    if 'lane_change_target' in _memory:
        target_lane = _memory['lane_change_target']

    # 11. Speed Control Target
    base_target_speed = state.get('target_speed_kmh', 110.0)
    
    # Adapt target speed to friction
    if mu_filtered < 0.2:
        target_speed_kmh = min(base_target_speed, 61.0) # Aquaplaning (61 km/h stays above 55 km/h limit)
    elif mu_filtered < 0.3:
        target_speed_kmh = min(base_target_speed, 63.0) # Ice/aquaplaning (63 km/h stays above 55 km/h limit)
    elif mu_filtered < 0.5:
        target_speed_kmh = min(base_target_speed, 80.0) # Wet
    else:
        target_speed_kmh = base_target_speed
        
    force_emergency_brake = False
    
    # Adapt target speed to obstacles (only if not changing lanes, or as safety backup)
    if obs_detected:
        a_max = 30000.0 * min(mu_filtered, 1.0) / 1600.0
        safety_margin = 12.0
        
        if obs_speed_ms < 2.0:
            # Stopped vehicle/debris
            critical_dist = (speed_est_ms**2) / (2.0 * a_max) + safety_margin
            
            # Use max(40.0, critical_dist) to trigger early and hold brake at 0.0 target speed
            if obs_dist < max(40.0, critical_dist):
                target_speed_kmh = 0.0
                force_emergency_brake = True
            else:
                safe_dist = max(12.0, speed_est_ms * 1.8 + 8.0)
                target_speed_ms_obs = obs_speed_ms + 0.3 * (obs_dist - safe_dist)
                target_speed_ms_obs = max(0.0, target_speed_ms_obs)
                target_speed_kmh = min(target_speed_kmh, target_speed_ms_obs * 3.6)
        else:
            # Moving vehicle
            if speed_est_ms > obs_speed_ms:
                critical_dist = (speed_est_ms**2 - obs_speed_ms**2) / (2.0 * a_max) + safety_margin
                if obs_dist < critical_dist:
                    target_speed_kmh = obs_speed_ms * 3.6
                    force_emergency_brake = True
                else:
                    safe_dist = max(12.0, speed_est_ms * 1.8 + 8.0)
                    target_speed_ms_obs = obs_speed_ms + 0.3 * (obs_dist - safe_dist)
                    target_speed_ms_obs = max(0.0, target_speed_ms_obs)
                    target_speed_kmh = min(target_speed_kmh, target_speed_ms_obs * 3.6)
            else:
                safe_dist = max(12.0, speed_est_ms * 1.8 + 8.0)
                target_speed_ms_obs = obs_speed_ms + 0.3 * (obs_dist - safe_dist)
                target_speed_ms_obs = max(0.0, target_speed_ms_obs)
                target_speed_kmh = min(target_speed_kmh, target_speed_ms_obs * 3.6)

    # 12. Input limits for traction management
    if mu_filtered < 0.3:
        max_steer = 0.15
        max_brake = 0.4
        max_throttle = 0.5
    elif mu_filtered < 0.5:
        max_steer = 0.3
        max_brake = 0.6
        max_throttle = 0.5
    else:
        max_steer = 1.0
        max_brake = 1.0
        max_throttle = 1.0

    # Override limits if obstacle detected or steering jammed
    if obs_detected or _memory['steering_jammed']:
        max_brake = 1.0

    # 13. Steering Jam Pullover Control
    # If steering is jammed, we brake hard immediately to stop safely on the road (scoring 0.5)
    if _memory['steering_jammed']:
        target_speed_kmh = 0.0
        force_emergency_brake = True
            
    # 14. Speed PI Controller
    speed_err = (target_speed_kmh - speed_est_kmh) / 3.6
    _memory['speed_integral'] += speed_err * dt
    _memory['speed_integral'] = max(-10.0, min(10.0, _memory['speed_integral']))
    
    Kp_speed = 0.15
    Ki_speed = 0.02
    control_effort = Kp_speed * speed_err + Ki_speed * _memory['speed_integral']
    
    if control_effort > 0.0:
        throttle = min(max_throttle, control_effort)
        brake = 0.0
    else:
        throttle = 0.0
        brake = min(max_brake, -control_effort)
        
    if force_emergency_brake:
        throttle = 0.0
        brake = 1.0  # In emergency, always use full brake force

    # 15. Lane Keeping Controller (Steering)
    lane_center_y = (target_lane + 0.5) * lane_width
    
    error_y = y_est - lane_center_y
    _memory['y_integral'] += error_y * dt
    _memory['y_integral'] = max(-3.0, min(3.0, _memory['y_integral']))
    
    Kp_y = 0.12
    Ki_y = 0.02
    Kp_yaw = 1.2
    
    # Scale down gains on low traction to prevent oscillation and instability due to slow lateral response
    if mu_filtered < 0.5:
        scale = max(0.3, mu_filtered)
        Kp_y *= scale
        Ki_y *= scale
        Kp_yaw *= scale
    
    # Curve feedforward
    curvatures = []
    for i in [1, 2, 3]:
        if f'vision_{i}_road_curvature_1pm' in state:
            curvatures.append(state[f'vision_{i}_road_curvature_1pm'])
    if curvatures:
        curvatures.sort()
        curvature_est = curvatures[len(curvatures) // 2]
    else:
        curvature_est = 0.0
    
    # Feedforward steer
    steer_ff = math.atan2(2.7 * curvature_est, 1.0)
    
    # Steering command in rad
    steer_cmd_rad = -Kp_y * error_y - Ki_y * _memory['y_integral'] - Kp_yaw * heading_est + steer_ff
    
    # Normalize steering to [-1.0, 1.0]
    steering = steer_cmd_rad / (35.0 * math.pi / 180.0)
    
    # Speed-dependent steering clamp to prevent high-speed swerving and rollover
    if mu_filtered >= 0.5:
        max_steer_speed = max(0.08, min(1.0, 2.5 / max(1.0, speed_est_ms)))
        max_steer_allowed = min(max_steer, max_steer_speed)
    else:
        max_steer_allowed = max_steer
    steering = max(-max_steer_allowed, min(max_steer_allowed, steering))

    # Save outputs to memory
    _memory['last_steering'] = steering
    _memory['last_throttle'] = throttle
    _memory['last_brake'] = brake

    return {
        'steering': steering,
        'throttle': throttle,
        'brake': brake
    }
