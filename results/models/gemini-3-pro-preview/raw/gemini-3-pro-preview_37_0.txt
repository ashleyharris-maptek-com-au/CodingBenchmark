using System;
using System.Collections.Generic;
using System.IO;

public class Program {
    class FastScanner {
        private readonly Stream _stream;
        private readonly byte[] _buf = new byte[1024 * 1024];
        private int _len, _ptr;

        public FastScanner() {
            _stream = Console.OpenStandardInput();
        }

        private byte ReadByte() {
            if (_ptr >= _len) {
                _ptr = 0;
                _len = _stream.Read(_buf, 0, _buf.Length);
                if (_len <= 0) return 0;
            }
            return _buf[_ptr++];
        }

        public int NextInt() {
            byte b = ReadByte();
            while (b > 0 && b <= 32) b = ReadByte();
            if (b == 0) return -1;
            int res = 0;
            while (b > 32) {
                if (b >= '0' && b <= '9') {
                    res = res * 10 + (b - '0');
                }
                b = ReadByte();
            }
            return res;
        }
    }

    public static void Main(string[] args) {
        FastScanner scanner = new FastScanner();
        int n = scanner.NextInt();
        if (n <= 0) {
            Console.WriteLine(0);
            Console.WriteLine();
            return;
        }
        int m = scanner.NextInt();

        HashSet<int>[] inNeighbors = new HashSet<int>[n];
        HashSet<int>[] outNeighbors = new HashSet<int>[n];
        for (int i = 0; i < n; i++) {
            inNeighbors[i] = new HashSet<int>();
            outNeighbors[i] = new HashSet<int>();
        }

        for (int i = 0; i < m; i++) {
            int u = scanner.NextInt();
            int v = scanner.NextInt();
            if (u >= 0 && u < n && v >= 0 && v < n) {
                outNeighbors[u].Add(v);
                inNeighbors[v].Add(u);
            }
        }

        bool[] removed = new bool[n];
        List<int> dfvs = new List<int>();
        Queue<int> reductionQueue = new Queue<int>();
        bool[] inQueue = new bool[n];

        for (int i = 0; i < n; i++) {
            if (inNeighbors[i].Count <= 1 || outNeighbors[i].Count <= 1 || outNeighbors[i].Contains(i)) {
                reductionQueue.Enqueue(i);
                inQueue[i] = true;
            }
        }

        while (reductionQueue.Count > 0) {
            int v = reductionQueue.Dequeue();
            inQueue[v] = false;
            if (removed[v]) continue;

            if (outNeighbors[v].Contains(v)) {
                dfvs.Add(v);
                RemoveVertex(v, inNeighbors, outNeighbors, removed, reductionQueue, inQueue);
                continue;
            }

            if (inNeighbors[v].Count == 0 || outNeighbors[v].Count == 0) {
                RemoveVertex(v, inNeighbors, outNeighbors, removed, reductionQueue, inQueue);
                continue;
            }

            if (inNeighbors[v].Count == 1) {
                int u = -1;
                foreach (int neighbor in inNeighbors[v]) { u = neighbor; }
                
                var outList = new List<int>(outNeighbors[v]);
                foreach (int w in outList) {
                    if (w == v) continue;
                    AddEdge(u, w, inNeighbors, outNeighbors, reductionQueue, inQueue);
                }
                RemoveVertex(v, inNeighbors, outNeighbors, removed, reductionQueue, inQueue);
                continue;
            }

            if (outNeighbors[v].Count == 1) {
                int w = -1;
                foreach (int neighbor in outNeighbors[v]) { w = neighbor; }

                var inList = new List<int>(inNeighbors[v]);
                foreach (int u in inList) {
                    if (u == v) continue;
                    AddEdge(u, w, inNeighbors, outNeighbors, reductionQueue, inQueue);
                }
                RemoveVertex(v, inNeighbors, outNeighbors, removed, reductionQueue, inQueue);
                continue;
            }
        }

        int[][] adj = new int[n][];
        for (int i = 0; i < n; i++) {
            if (removed[i]) {
                adj[i] = new int[0];
            } else {
                adj[i] = new int[outNeighbors[i].Count];
                outNeighbors[i].CopyTo(adj[i]);
            }
        }

        List<List<int>> sccs = FindSccs(adj, removed);
        foreach (var scc in sccs) {
            if (scc.Count <= 1) continue;
            List<int> sccDfvs = SolveSccDFVS(scc, adj, removed);
            dfvs.AddRange(sccDfvs);
        }

        Console.WriteLine(dfvs.Count);
        for (int i = 0; i < dfvs.Count; i++) {
            if (i > 0) Console.Write(" ");
            Console.Write(dfvs[i]);
        }
        Console.WriteLine();
    }

    static void RemoveVertex(int v, HashSet<int>[] inNeighbors, HashSet<int>[] outNeighbors, bool[] removed, Queue<int> queue, bool[] inQueue) {
        removed[v] = true;
        foreach (int u in inNeighbors[v]) {
            if (u != v) {
                outNeighbors[u].Remove(v);
                if (!inQueue[u] && (inNeighbors[u].Count <= 1 || outNeighbors[u].Count <= 1 || outNeighbors[u].Contains(u))) {
                    queue.Enqueue(u);
                    inQueue[u] = true;
                }
            }
        }
        foreach (int w in outNeighbors[v]) {
            if (w != v) {
                inNeighbors[w].Remove(v);
                if (!inQueue[w] && (inNeighbors[w].Count <= 1 || outNeighbors[w].Count <= 1 || outNeighbors[w].Contains(w))) {
                    queue.Enqueue(w);
                    inQueue[w] = true;
                }
            }
        }
        inNeighbors[v].Clear();
        outNeighbors[v].Clear();
    }

    static void AddEdge(int u, int w, HashSet<int>[] inNeighbors, HashSet<int>[] outNeighbors, Queue<int> queue, bool[] inQueue) {
        if (u == w) {
            if (!outNeighbors[u].Contains(u)) {
                outNeighbors[u].Add(u);
                inNeighbors[u].Add(u);
                if (!inQueue[u]) {
                    queue.Enqueue(u);
                    inQueue[u] = true;
                }
            }
            return;
        }
        if (!outNeighbors[u].Contains(w)) {
            outNeighbors[u].Add(w);
            inNeighbors[w].Add(u);
        }
    }

    static List<List<int>> FindSccs(int[][] adj, bool[] removed) {
        int n = adj.Length;
        var dfn = new int[n];
        var low = new int[n];
        var inStack = new bool[n];
        var tarjanStack = new int[n];
        int tarjanStackPtr = 0;
        int dfnId = 0;

        var sccs = new List<List<int>>();

        var dfsStack = new int[n];
        var edgeIdx = new int[n];
        var parent = new int[n];

        for (int i = 0; i < n; i++) {
            if (removed[i] || dfn[i] > 0) continue;

            int dfsStackPtr = 0;
            dfsStack[dfsStackPtr] = i;
            edgeIdx[i] = 0;
            dfnId++;
            dfn[i] = low[i] = dfnId;
            tarjanStack[tarjanStackPtr++] = i;
            inStack[i] = true;
            parent[i] = -1;

            while (dfsStackPtr >= 0) {
                int u = dfsStack[dfsStackPtr];
                var neighbors = adj[u];

                if (edgeIdx[u] < neighbors.Length) {
                    int v = neighbors[edgeIdx[u]];
                    edgeIdx[u]++;

                    if (removed[v]) continue;

                    if (dfn[v] == 0) {
                        dfnId++;
                        dfn[v] = low[v] = dfnId;
                        tarjanStack[tarjanStackPtr++] = v;
                        inStack[v] = true;
                        parent[v] = u;

                        dfsStackPtr++;
                        dfsStack[dfsStackPtr] = v;
                        edgeIdx[v] = 0;
                    }
                    else if (inStack[v]) {
                        low[u] = Math.Min(low[u], dfn[v]);
                    }
                }
                else {
                    if (low[u] == dfn[u]) {
                        var scc = new List<int>();
                        while (true) {
                            int v = tarjanStack[--tarjanStackPtr];
                            inStack[v] = false;
                            scc.Add(v);
                            if (v == u) break;
                        }
                        sccs.Add(scc);
                    }

                    int p = parent[u];
                    if (p != -1) {
                        low[p] = Math.Min(low[p], low[u]);
                    }

                    dfsStackPtr--;
                }
            }
        }

        return sccs;
    }

    static List<int> SolveSccDFVS(List<int> scc, int[][] adj, bool[] removed) {
        int n = adj.Length;
        var sccVertices = scc.ToArray();
        var inScc = new bool[n];
        foreach (int v in sccVertices) {
            inScc[v] = true;
        }

        var cycles = new List<List<int>>();

        var dist = new int[n];
        var parent = new int[n];
        var queue = new int[n];
        var vertexVisited = new int[n];
        int visitId = 0;

        var vertexToCycles = new List<int>[n];
        var vertexVisitedMHS = new int[n];
        int visitIdMHS = 0;

        while (true) {
            var s = SolveMHS(cycles, vertexToCycles, vertexVisitedMHS, ref visitIdMHS);

            foreach (int v in s) {
                removed[v] = true;
            }

            var newCyclesList = new List<List<int>>();
            var tempRemoved = new List<int>();

            while (true) {
                List<int> newCycle = null;
                for (int depth = 3; depth <= 8; depth += 2) {
                    foreach (var start in sccVertices) {
                        if (removed[start]) continue;
                        newCycle = FindShortestCycleContaining(start, depth, adj, removed, inScc, dist, parent, queue, ref visitId, vertexVisited);
                        if (newCycle != null) break;
                    }
                    if (newCycle != null) break;
                }

                if (newCycle == null) {
                    newCycle = FindCycleIterative(sccVertices, adj, removed, inScc);
                }

                if (newCycle == null) break;

                newCyclesList.Add(newCycle);
                foreach (int v in newCycle) {
                    removed[v] = true;
                    tempRemoved.Add(v);
                }
            }

            foreach (int v in tempRemoved) {
                removed[v] = false;
            }
            foreach (int v in s) {
                removed[v] = false;
            }

            if (newCyclesList.Count == 0) {
                foreach (int v in s) {
                    removed[v] = true;
                }
                return s;
            }

            foreach (var cycle in newCyclesList) {
                AddCycleAndMinimize(cycles, cycle);
            }
        }
    }

    class MhsContext {
        public List<int>[] CycleToVertices;
        public List<int>[] VertexToCycles;
        public int[] VertexVisited;
        public int VisitId;
        public int[] HitCount;
        public bool[] Selected;
        public List<int> Solution;
        public int ActiveCyclesCount;
        public int M;
    }

    static List<int> SolveMHS(List<List<int>> cycles, List<int>[] vertexToCycles, int[] vertexVisited, ref int visitId) {
        if (cycles.Count == 0) return new List<int>();

        cycles.Sort((a, b) => a.Count.CompareTo(b.Count));

        int m = cycles.Count;
        var cycleToVertices = new List<int>[m];
        for (int i = 0; i < m; i++) {
            cycleToVertices[i] = cycles[i];
        }

        foreach (var c in cycles) {
            foreach (int v in c) {
                if (vertexToCycles[v] == null) {
                    vertexToCycles[v] = new List<int>();
                }
            }
        }
        for (int i = 0; i < m; i++) {
            foreach (int v in cycles[i]) {
                vertexToCycles[v].Add(i);
            }
        }

        var ctx = new MhsContext {
            CycleToVertices = cycleToVertices,
            VertexToCycles = vertexToCycles,
            VertexVisited = vertexVisited,
            VisitId = visitId,
            HitCount = new int[m],
            Selected = new bool[vertexToCycles.Length],
            Solution = new List<int>(),
            ActiveCyclesCount = m,
            M = m
        };

        for (int budget = 0; ; budget++) {
            if (SearchMHS(budget, ctx)) {
                visitId = ctx.VisitId;
                foreach (var c in cycles) {
                    foreach (int v in c) {
                        vertexToCycles[v].Clear();
                    }
                }
                return ctx.Solution;
            }
        }
    }

    static bool SearchMHS(int budget, MhsContext ctx) {
        if (ctx.ActiveCyclesCount == 0) return true;
        if (budget == 0) return false;

        int lb = 0;
        ctx.VisitId++;
        for (int i = 0; i < ctx.M; i++) {
            if (ctx.HitCount[i] == 0) {
                bool disjoint = true;
                var verts = ctx.CycleToVertices[i];
                for (int j = 0; j < verts.Count; j++) {
                    if (ctx.VertexVisited[verts[j]] == ctx.VisitId) {
                        disjoint = false;
                        break;
                    }
                }
                if (disjoint) {
                    lb++;
                    for (int j = 0; j < verts.Count; j++) {
                        ctx.VertexVisited[verts[j]] = ctx.VisitId;
                    }
                }
            }
        }

        if (lb > budget) return false;

        int bestCycleIdx = -1;
        int minSize = int.MaxValue;
        for (int i = 0; i < ctx.M; i++) {
            if (ctx.HitCount[i] == 0) {
                int size = ctx.CycleToVertices[i].Count;
                if (size < minSize) {
                    minSize = size;
                    bestCycleIdx = i;
                    if (minSize == 1) break;
                }
            }
        }

        if (bestCycleIdx == -1) return true;

        var vertsToBranch = ctx.CycleToVertices[bestCycleIdx];
        int[] branchVerts = new int[vertsToBranch.Count];
        for (int i = 0; i < vertsToBranch.Count; i++) branchVerts[i] = vertsToBranch[i];

        Array.Sort(branchVerts, (a, b) => {
            int countA = 0;
            var listA = ctx.VertexToCycles[a];
            for (int i = 0; i < listA.Count; i++) if (ctx.HitCount[listA[i]] == 0) countA++;

            int countB = 0;
            var listB = ctx.VertexToCycles[b];
            for (int i = 0; i < listB.Count; i++) if (ctx.HitCount[listB[i]] == 0) countB++;

            return countB.CompareTo(countA);
        });

        foreach (int v in branchVerts) {
            ctx.Selected[v] = true;
            ctx.Solution.Add(v);
            var vCycles = ctx.VertexToCycles[v];
            for (int i = 0; i < vCycles.Count; i++) {
                int cIdx = vCycles[i];
                if (ctx.HitCount[cIdx] == 0) {
                    ctx.ActiveCyclesCount--;
                }
                ctx.HitCount[cIdx]++;
            }

            if (SearchMHS(budget - 1, ctx)) {
                return true;
            }

            for (int i = 0; i < vCycles.Count; i++) {
                int cIdx = vCycles[i];
                ctx.HitCount[cIdx]--;
                if (ctx.HitCount[cIdx] == 0) {
                    ctx.ActiveCyclesCount++;
                }
            }
            ctx.Solution.RemoveAt(ctx.Solution.Count - 1);
            ctx.Selected[v] = false;
        }

        return false;
    }

    static bool AddCycleAndMinimize(List<List<int>> cycles, List<int> newCycle) {
        var newCycleSet = new HashSet<int>(newCycle);
        foreach (var existing in cycles) {
            bool isSubset = true;
            foreach (int v in existing) {
                if (!newCycleSet.Contains(v)) {
                    isSubset = false;
                    break;
                }
            }
            if (isSubset) {
                return false;
            }
        }

        for (int i = cycles.Count - 1; i >= 0; i--) {
            var existing = cycles[i];
            bool isSuperset = true;
            foreach (int v in newCycle) {
                if (!existing.Contains(v)) {
                    isSuperset = false;
                    break;
                }
            }
            if (isSuperset) {
                cycles.RemoveAt(i);
            }
        }

        cycles.Add(newCycle);
        return true;
    }

    static List<int> FindShortestCycleContaining(int s, int maxDepth, int[][] adj, bool[] removed, bool[] inScc,
                                                 int[] dist, int[] parent, int[] queue, ref int visitId, int[] vertexVisited) {
        int head = 0, tail = 0;
        visitId++;

        foreach (int v in adj[s]) {
            if (removed[v] || !inScc[v]) continue;
            if (v == s) {
                return new List<int> { s };
            }
            if (tail >= 500) break;
            queue[tail++] = v;
            dist[v] = 1;
            parent[v] = s;
            vertexVisited[v] = visitId;
        }

        while (head < tail) {
            int u = queue[head++];
            int d = dist[u];

            if (d >= maxDepth) continue;

            foreach (int v in adj[u]) {
                if (removed[v] || !inScc[v]) continue;

                if (v == s) {
                    var cycle = new List<int>();
                    int curr = u;
                    while (curr != s) {
                        cycle.Add(curr);
                        curr = parent[curr];
                    }
                    cycle.Add(s);
                    cycle.Reverse();
                    return cycle;
                }

                if (vertexVisited[v] != visitId) {
                    if (tail >= 500) continue;
                    vertexVisited[v] = visitId;
                    dist[v] = d + 1;
                    parent[v] = u;
                    queue[tail++] = v;
                }
            }
        }

        return null;
    }

    static List<int> FindCycleIterative(int[] sccVertices, int[][] adj, bool[] removed, bool[] inScc) {
        int n = adj.Length;
        var visited = new bool[n];
        var inStack = new bool[n];
        var parent = new int[n];
        var stack = new int[n];
        var edgeIdx = new int[n];

        foreach (var start in sccVertices) {
            if (removed[start] || visited[start]) continue;

            int stackPtr = 0;
            stack[stackPtr] = start;
            edgeIdx[start] = 0;
            visited[start] = true;
            inStack[start] = true;

            while (stackPtr >= 0) {
                int u = stack[stackPtr];
                var neighbors = adj[u];
                if (edgeIdx[u] < neighbors.Length) {
                    int v = neighbors[edgeIdx[u]];
                    edgeIdx[u]++;

                    if (removed[v] || !inScc[v]) continue;

                    if (inStack[v]) {
                        var cycle = new List<int>();
                        int curr = u;
                        while (curr != v) {
                            cycle.Add(curr);
                            curr = parent[curr];
                        }
                        cycle.Add(v);
                        cycle.Reverse();
                        return cycle;
                    }

                    if (!visited[v]) {
                        visited[v] = true;
                        inStack[v] = true;
                        parent[v] = u;
                        stackPtr++;
                        stack[stackPtr] = v;
                        edgeIdx[v] = 0;
                    }
                }
                else {
                    inStack[u] = false;
                    stackPtr--;
                }
            }
        }
        return null;
    }
}
