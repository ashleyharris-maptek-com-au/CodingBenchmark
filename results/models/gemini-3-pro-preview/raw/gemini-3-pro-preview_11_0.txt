use std::io::{self, BufRead};
use std::time::Instant;

fn next_non_empty_line<I>(lines: &mut I) -> Option<String>
where
    I: Iterator<Item = String>,
{
    for line in lines {
        let trimmed = line.trim();
        if !trimmed.is_empty() {
            return Some(trimmed.to_string());
        }
    }
    None
}

fn read_input() -> (Vec<Point>, Vec<Vec<Point>>) {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines().map(|l| l.unwrap());
    
    let first_line = next_non_empty_line(&mut lines).expect("Expected SV and NP");
    let parts: Vec<&str> = first_line.split_whitespace().collect();
    let sv: usize = parts[0].parse().unwrap();
    let np: usize = parts[1].parse().unwrap();
    
    let mut stock_vertices = Vec::with_capacity(sv);
    for _ in 0..sv {
        let line = next_non_empty_line(&mut lines).expect("Expected stock vertex");
        let parts: Vec<&str> = line.split_whitespace().collect();
        let x: f64 = parts[0].parse().unwrap();
        let y: f64 = parts[1].parse().unwrap();
        stock_vertices.push(Point { x, y });
    }
    
    let mut pieces = Vec::with_capacity(np);
    for _ in 0..np {
        let pv_line = next_non_empty_line(&mut lines).expect("Expected PV");
        let pv: usize = pv_line.parse().unwrap();
        let mut piece_vertices = Vec::with_capacity(pv);
        for _ in 0..pv {
            let line = next_non_empty_line(&mut lines).expect("Expected piece vertex");
            let parts: Vec<&str> = line.split_whitespace().collect();
            let x: f64 = parts[0].parse().unwrap();
            let y: f64 = parts[1].parse().unwrap();
            piece_vertices.push(Point { x, y });
        }
        pieces.push(piece_vertices);
    }
    
    (stock_vertices, pieces)
}


#[derive(Debug, Clone, Copy, PartialEq)]
struct Point {
    x: f64,
    y: f64,
}

#[derive(Debug, Clone, Copy)]
struct AABB {
    min_x: f64,
    max_x: f64,
    min_y: f64,
    max_y: f64,
}

impl AABB {
    fn from_points(points: &[Point]) -> Self {
        let mut min_x = f64::INFINITY;
        let mut max_x = f64::NEG_INFINITY;
        let mut min_y = f64::INFINITY;
        let mut max_y = f64::NEG_INFINITY;
        for p in points {
            if p.x < min_x { min_x = p.x; }
            if p.x > max_x { max_x = p.x; }
            if p.y < min_y { min_y = p.y; }
            if p.y > max_y { max_y = p.y; }
        }
        AABB { min_x, max_x, min_y, max_y }
    }

    fn overlaps(&self, other: &Self) -> bool {
        self.min_x <= other.max_x + 1e-6 && self.max_x >= other.min_x - 1e-6 &&
        self.min_y <= other.max_y + 1e-6 && self.max_y >= other.min_y - 1e-6
    }
}

#[derive(Debug, Clone)]
struct Polygon {
    vertices: Vec<Point>,
    aabb: AABB,
    area: f64,
    interior_point: Point,
}

impl Polygon {
    fn new(mut vertices: Vec<Point>) -> Self {
        make_ccw(&mut vertices);
        let aabb = AABB::from_points(&vertices);
        let area = polygon_area(&vertices);
        let interior_point = find_interior_point(&vertices);
        Polygon { vertices, aabb, area, interior_point }
    }

    fn transform(&self, tx: f64, ty: f64, theta: f64) -> Self {
        let cos_t = theta.cos();
        let sin_t = theta.sin();
        let vertices: Vec<Point> = self.vertices.iter().map(|p| {
            Point {
                x: cos_t * p.x - sin_t * p.y + tx,
                y: sin_t * p.x + cos_t * p.y + ty,
            }
        }).collect();
        let aabb = AABB::from_points(&vertices);
        let interior_point = Point {
            x: cos_t * self.interior_point.x - sin_t * self.interior_point.y + tx,
            y: sin_t * self.interior_point.x + cos_t * self.interior_point.y + ty,
        };
        Polygon {
            vertices,
            aabb,
            area: self.area,
            interior_point,
        }
    }
}

fn ccw(a: Point, b: Point, c: Point) -> f64 {
    (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x)
}

fn polygon_area(vertices: &[Point]) -> f64 {
    let mut area = 0.0;
    let n = vertices.len();
    for i in 0..n {
        let j = (i + 1) % n;
        area += vertices[i].x * vertices[j].y - vertices[j].x * vertices[i].y;
    }
    0.5 * area.abs()
}

fn make_ccw(vertices: &mut Vec<Point>) {
    let mut area = 0.0;
    let n = vertices.len();
    for i in 0..n {
        let j = (i + 1) % n;
        area += vertices[i].x * vertices[j].y - vertices[j].x * vertices[i].y;
    }
    if area < 0.0 {
        vertices.reverse();
    }
}

fn point_in_triangle(p: Point, a: Point, b: Point, c: Point) -> bool {
    let cp1 = ccw(a, b, p);
    let cp2 = ccw(b, c, p);
    let cp3 = ccw(c, a, p);
    // Standard CCW triangle containment (including boundary)
    cp1 >= -1e-6 && cp2 >= -1e-6 && cp3 >= -1e-6
}

fn find_interior_point(vertices: &[Point]) -> Point {
    let n = vertices.len();
    if n < 3 {
        return Point { x: 0.0, y: 0.0 };
    }
    for i in 0..n {
        let prev = vertices[(i + n - 1) % n];
        let curr = vertices[i];
        let next = vertices[(i + 1) % n];
        
        let area = ccw(prev, curr, next);
        if area <= 1e-6 {
            continue; // Not convex corner
        }
        
        let mut has_vertex_inside = false;
        for j in 0..n {
            if j == (i + n - 1) % n || j == i || j == (i + 1) % n {
                continue;
            }
            if point_in_triangle(vertices[j], prev, curr, next) {
                has_vertex_inside = true;
                break;
            }
        }
        
        if !has_vertex_inside {
            return Point {
                x: (prev.x + curr.x + next.x) / 3.0,
                y: (prev.y + curr.y + next.y) / 3.0,
            };
        }
    }
    // Fallback to average
    let mut sx = 0.0;
    let mut sy = 0.0;
    for p in vertices {
        sx += p.x;
        sy += p.y;
    }
    Point { x: sx / n as f64, y: sy / n as f64 }
}

fn segments_cross(a: Point, b: Point, c: Point, d: Point) -> bool {
    let cp1 = ccw(a, b, c);
    let cp2 = ccw(a, b, d);
    let cp3 = ccw(c, d, a);
    let cp4 = ccw(c, d, b);
    
    if ((cp1 > 1e-6 && cp2 < -1e-6) || (cp1 < -1e-6 && cp2 > 1e-6)) &&
       ((cp3 > 1e-6 && cp4 < -1e-6) || (cp3 < -1e-6 && cp4 > 1e-6)) {
        return true;
    }
    false
}

fn is_point_on_segment(p: Point, a: Point, b: Point) -> bool {
    let cross = (p.y - a.y) * (b.x - a.x) - (p.x - a.x) * (b.y - a.y);
    if cross.abs() > 1e-6 {
        return false;
    }
    let min_x = f64::min(a.x, b.x);
    let max_x = f64::max(a.x, b.x);
    let min_y = f64::min(a.y, b.y);
    let max_y = f64::max(a.y, b.y);
    
    p.x >= min_x - 1e-6 && p.x <= max_x + 1e-6 && p.y >= min_y - 1e-6 && p.y <= max_y + 1e-6
}

fn is_point_on_boundary(p: Point, poly: &Polygon) -> bool {
    let n = poly.vertices.len();
    for i in 0..n {
        let a = poly.vertices[i];
        let b = poly.vertices[(i + 1) % n];
        if is_point_on_segment(p, a, b) {
            return true;
        }
    }
    false
}

fn point_in_polygon(p: Point, poly: &Polygon) -> bool {
    let mut inside = false;
    let n = poly.vertices.len();
    for i in 0..n {
        let j = (i + 1) % n;
        let vi = poly.vertices[i];
        let vj = poly.vertices[j];
        if ((vi.y > p.y) != (vj.y > p.y)) &&
           (p.x < (vj.x - vi.x) * (p.y - vi.y) / (vj.y - vi.y) + vi.x) {
            inside = !inside;
        }
    }
    inside
}

fn point_in_polygon_strict(p: Point, poly: &Polygon) -> bool {
    if is_point_on_boundary(p, poly) {
        return false;
    }
    point_in_polygon(p, poly)
}

fn point_in_convex_polygon(p: Point, vertices: &[Point]) -> bool {
    let n = vertices.len();
    for i in 0..n {
        let a = vertices[i];
        let b = vertices[(i + 1) % n];
        if ccw(a, b, p) < -1e-6 {
            return false;
        }
    }
    true
}

fn is_convex(vertices: &[Point]) -> bool {
    let n = vertices.len();
    if n < 3 {
        return false;
    }
    let mut sign = 0.0;
    for i in 0..n {
        let prev = vertices[(i + n - 1) % n];
        let curr = vertices[i];
        let next = vertices[(i + 1) % n];
        let cp = ccw(prev, curr, next);
        if cp.abs() > 1e-6 {
            if sign == 0.0 {
                sign = cp.signum();
            } else if cp.signum() != sign {
                return false;
            }
        }
    }
    true
}

fn get_rectangle_rotation(vertices: &[Point]) -> Option<(Point, f64)> {
    if vertices.len() != 4 {
        return None;
    }
    for i in 0..4 {
        let p1 = vertices[i];
        let p2 = vertices[(i + 1) % 4];
        let p3 = vertices[(i + 2) % 4];
        let v1 = Point { x: p1.x - p2.x, y: p1.y - p2.y };
        let v2 = Point { x: p3.x - p2.x, y: p3.y - p2.y };
        let dot = v1.x * v2.x + v1.y * v2.y;
        let len1 = (v1.x * v1.x + v1.y * v1.y).sqrt();
        let len2 = (v2.x * v2.x + v2.y * v2.y).sqrt();
        if (dot / (len1 * len2)).abs() > 1e-5 {
            return None;
        }
    }
    let v0 = vertices[0];
    let v1 = vertices[1];
    let dy = v1.y - v0.y;
    let dx = v1.x - v0.x;
    let alpha = dy.atan2(dx);
    Some((v0, alpha))
}

fn rotate_point(p: Point, theta: f64) -> Point {
    let cos_t = theta.cos();
    let sin_t = theta.sin();
    Point {
        x: cos_t * p.x - sin_t * p.y,
        y: sin_t * p.x + cos_t * p.y,
    }
}

fn polygons_overlap(poly1: &Polygon, poly2: &Polygon) -> bool {
    if !poly1.aabb.overlaps(&poly2.aabb) {
        return false;
    }
    
    if point_in_polygon_strict(poly1.interior_point, poly2) {
        return true;
    }
    if point_in_polygon_strict(poly2.interior_point, poly1) {
        return true;
    }
    
    let n1 = poly1.vertices.len();
    let n2 = poly2.vertices.len();
    for i in 0..n1 {
        let a = poly1.vertices[i];
        let b = poly1.vertices[(i + 1) % n1];
        for j in 0..n2 {
            let c = poly2.vertices[j];
            let d = poly2.vertices[(j + 1) % n2];
            if segments_cross(a, b, c, d) {
                return true;
            }
        }
    }
    
    for i in 0..n1 {
        let v = poly1.vertices[i];
        if point_in_polygon_strict(v, poly2) {
            return true;
        }
        let next = poly1.vertices[(i + 1) % n1];
        let mid = Point { x: 0.5 * (v.x + next.x), y: 0.5 * (v.y + next.y) };
        if point_in_polygon_strict(mid, poly2) {
            return true;
        }
    }
    
    for j in 0..n2 {
        let v = poly2.vertices[j];
        if point_in_polygon_strict(v, poly1) {
            return true;
        }
        let next = poly2.vertices[(j + 1) % n2];
        let mid = Point { x: 0.5 * (v.x + next.x), y: 0.5 * (v.y + next.y) };
        if point_in_polygon_strict(mid, poly1) {
            return true;
        }
    }
    
    false
}

enum StockType {
    Rectangle { w: f64, h: f64 },
    Convex,
    General,
}

struct SpatialGrid {
    cell_width: f64,
    cell_height: f64,
    min_x: f64,
    min_y: f64,
    cols: usize,
    rows: usize,
    cells: Vec<Vec<usize>>,
}

impl SpatialGrid {
    fn new(aabb: AABB, avg_w: f64, avg_h: f64) -> Self {
        let w = aabb.max_x - aabb.min_x;
        let h = aabb.max_y - aabb.min_y;
        let cols = (w / f64::max(avg_w, w / 128.0)).ceil() as usize;
        let cols = usize::max(1, usize::min(128, cols));
        let rows = (h / f64::max(avg_h, h / 128.0)).ceil() as usize;
        let rows = usize::max(1, usize::min(128, rows));
        
        let cell_width = w / cols as f64;
        let cell_height = h / rows as f64;
        
        SpatialGrid {
            cell_width,
            cell_height,
            min_x: aabb.min_x,
            min_y: aabb.min_y,
            cols,
            rows,
            cells: vec![Vec::new(); cols * rows],
        }
    }

    fn get_cell_range(&self, aabb: &AABB) -> (usize, usize, usize, usize) {
        let col_start = ((aabb.min_x - self.min_x) / self.cell_width).floor() as isize;
        let col_end = ((aabb.max_x - self.min_x) / self.cell_width).floor() as isize;
        let row_start = ((aabb.min_y - self.min_y) / self.cell_height).floor() as isize;
        let row_end = ((aabb.max_y - self.min_y) / self.cell_height).floor() as isize;
        
        let col_start = usize::max(0, usize::min(self.cols - 1, col_start as usize));
        let col_end = usize::max(0, usize::min(self.cols - 1, col_end as usize));
        let row_start = usize::max(0, usize::min(self.rows - 1, row_start as usize));
        let row_end = usize::max(0, usize::min(self.rows - 1, row_end as usize));
        
        (col_start, col_end, row_start, row_end)
    }

    fn insert(&mut self, idx: usize, aabb: &AABB) {
        let (col_start, col_end, row_start, row_end) = self.get_cell_range(aabb);
        for r in row_start..=row_end {
            for c in col_start..=col_end {
                self.cells[r * self.cols + c].push(idx);
            }
        }
    }

    fn query(&self, aabb: &AABB, out_indices: &mut Vec<usize>) {
        out_indices.clear();
        let (col_start, col_end, row_start, row_end) = self.get_cell_range(aabb);
        for r in row_start..=row_end {
            for c in col_start..=col_end {
                for &idx in &self.cells[r * self.cols + c] {
                    out_indices.push(idx);
                }
            }
        }
        out_indices.sort_unstable();
        out_indices.dedup();
    }
}

struct PlacedPiece {
    poly: Polygon,
    orig_idx: usize,
    rotation: f64,
    translation: Point,
}

struct StockState {
    placed_pieces: Vec<PlacedPiece>,
    grid: SpatialGrid,
    placed_area: f64,
}

fn add_candidate(candidates: &mut Vec<Point>, p: Point) {
    for c in candidates.iter() {
        if (c.x - p.x).abs() < 1e-4 && (c.y - p.y).abs() < 1e-4 {
            return;
        }
    }
    candidates.push(p);
}

fn main() {
    let _start_time = Instant::now();
    
    // Read input
    let (mut stock_vertices, pieces_vertices) = read_input();
    
    let mut rotation_info = None;
    if let Some((v0, alpha)) = get_rectangle_rotation(&stock_vertices) {
        rotation_info = Some((v0, alpha));
        // Rotate stock vertices
        for v in &mut stock_vertices {
            let p = Point { x: v.x - v0.x, y: v.y - v0.y };
            *v = rotate_point(p, -alpha);
        }
    }
    
    let stock = Polygon::new(stock_vertices);
    let stock_aabb = stock.aabb;
    let stock_area = stock.area;
    
    let stock_type = if rotation_info.is_some() {
        StockType::Rectangle { w: stock_aabb.max_x, h: stock_aabb.max_y }
    } else if is_convex(&stock.vertices) {
        StockType::Convex
    } else {
        StockType::General
    };
    
    // Prepare pieces
    let mut pieces = Vec::with_capacity(pieces_vertices.len());
    for (i, p_verts) in pieces_vertices.into_iter().enumerate() {
        let mut rotated_verts = p_verts;
        if let Some((_v0, alpha)) = rotation_info {
            for v in &mut rotated_verts {
                *v = rotate_point(*v, -alpha);
            }
        }
        let poly = Polygon::new(rotated_verts);
        pieces.push((i, poly));
    }
    
    let np = pieces.len();
    
    // Sort pieces by area descending
    pieces.sort_by(|a, b| {
        b.1.area.partial_cmp(&a.1.area).unwrap()
            .then(b.1.vertices.len().cmp(&a.1.vertices.len()))
    });
    
    // Compute average piece dimensions
    let mut sum_w = 0.0;
    let mut sum_h = 0.0;
    for (_, p) in &pieces {
        sum_w += p.aabb.max_x - p.aabb.min_x;
        sum_h += p.aabb.max_y - p.aabb.min_y;
    }
    let avg_w = sum_w / np as f64;
    let avg_h = sum_h / np as f64;
    
    // Placements list (orig_idx -> (stock_index, translation, rotation))
    let mut final_placements = vec![(0, Point { x: 0.0, y: 0.0 }, 0.0); np];
    
    let mut stocks: Vec<StockState> = Vec::new();
    
    // Adaptive parameters
    let num_rotations = if np < 200 {
        8
    } else if np < 1000 {
        4
    } else {
        2
    };
    
    let candidate_rotations: Vec<f64> = (0..num_rotations)
        .map(|i| i as f64 * (2.0 * std::f64::consts::PI / num_rotations as f64))
        .collect();
        
    let use_vertex_candidates = np < 1000;
    
    let mut query_buf = Vec::new();
    
    for (orig_idx, p_poly) in pieces {
        let p_area = p_poly.area;
        let mut placed = false;
        
        // Pre-rotate the piece for all candidate rotations
        let rotated_polys: Vec<Polygon> = candidate_rotations.iter()
            .map(|&rot| p_poly.transform(0.0, 0.0, rot))
            .collect();
            
        // Try to place on existing stocks
        for (_s_idx, s_state) in stocks.iter_mut().enumerate() {
            if s_state.placed_area + p_area > stock_area + 1e-6 {
                continue; // Pruning
            }
            
            // Generate candidates for this stock
            for (rot_idx, &rot) in candidate_rotations.iter().enumerate() {
                let p_rot = &rotated_polys[rot_idx];
                let p_aabb = p_rot.aabb;
                let ref_v = p_rot.vertices.iter()
                    .min_by(|a, b| a.x.partial_cmp(&b.x).unwrap().then(a.y.partial_cmp(&b.y).unwrap()))
                    .cloned()
                    .unwrap_or(Point { x: 0.0, y: 0.0 });
                
                let mut candidates = Vec::with_capacity(32);
                
                // 1. Bottom-left of stock AABB
                add_candidate(&mut candidates, Point {
                    x: stock_aabb.min_x - p_aabb.min_x,
                    y: stock_aabb.min_y - p_aabb.min_y,
                });
                
                // 2. Bounding box alignments with placed pieces
                for placed_p in &s_state.placed_pieces {
                    let o_aabb = placed_p.poly.aabb;
                    // Right of placed
                    add_candidate(&mut candidates, Point {
                        x: o_aabb.max_x - p_aabb.min_x,
                        y: o_aabb.min_y - p_aabb.min_y,
                    });
                    // Above placed
                    add_candidate(&mut candidates, Point {
                        x: o_aabb.min_x - p_aabb.min_x,
                        y: o_aabb.max_y - p_aabb.min_y,
                    });
                }
                
                // 3. Align reference vertex with stock and placed pieces vertices (only if allowed by adaptiveness)
                if use_vertex_candidates {
                    for &sv in &stock.vertices {
                        add_candidate(&mut candidates, Point {
                            x: sv.x - ref_v.x,
                            y: sv.y - ref_v.y,
                        });
                    }
                    for placed_p in &s_state.placed_pieces {
                        for &pv in &placed_p.poly.vertices {
                            add_candidate(&mut candidates, Point {
                                x: pv.x - ref_v.x,
                                y: pv.y - ref_v.y,
                            });
                        }
                    }
                }
                
                // Filter candidates: AABB must be inside stock AABB
                candidates.retain(|c| {
                    p_aabb.min_x + c.x >= stock_aabb.min_x - 1e-6 &&
                    p_aabb.max_x + c.x <= stock_aabb.max_x + 1e-6 &&
                    p_aabb.min_y + c.y >= stock_aabb.min_y - 1e-6 &&
                    p_aabb.max_y + c.y <= stock_aabb.max_y + 1e-6
                });
                
                // Sort candidates by Bottom-Left score
                candidates.sort_by(|a, b| {
                    let score_a = (a.y - stock_aabb.min_y) + 0.1 * (a.x - stock_aabb.min_x);
                    let score_b = (b.y - stock_aabb.min_y) + 0.1 * (b.x - stock_aabb.min_x);
                    score_a.partial_cmp(&score_b).unwrap()
                });
                
                // Evaluate candidates
                for c in candidates {
                    let trans_poly = p_rot.transform(c.x, c.y, 0.0);
                    
                    // A. Check containment in stock
                    let inside_stock = match stock_type {
                        StockType::Rectangle { w, h } => {
                            let mut ok = true;
                            for v in &trans_poly.vertices {
                                if v.x < -1e-6 || v.x > w + 1e-6 || v.y < -1e-6 || v.y > h + 1e-6 {
                                    ok = false;
                                    break;
                                }
                            }
                            ok
                        }
                        StockType::Convex => {
                            let mut ok = true;
                            for &v in &trans_poly.vertices {
                                if !point_in_convex_polygon(v, &stock.vertices) {
                                    ok = false;
                                    break;
                                }
                            }
                            ok
                        }
                        StockType::General => {
                            let mut ok = true;
                            for &v in &trans_poly.vertices {
                                if !point_in_polygon(v, &stock) && !is_point_on_boundary(v, &stock) {
                                    ok = false;
                                    break;
                                }
                            }
                            if ok {
                                let n1 = trans_poly.vertices.len();
                                let n2 = stock.vertices.len();
                                'outer: for i in 0..n1 {
                                    let a = trans_poly.vertices[i];
                                    let b = trans_poly.vertices[(i + 1) % n1];
                                    for j in 0..n2 {
                                        let c = stock.vertices[j];
                                        let d = stock.vertices[(j + 1) % n2];
                                        if segments_cross(a, b, c, d) {
                                            ok = false;
                                            break 'outer;
                                        }
                                    }
                                }
                            }
                            ok
                        }
                    };
                    
                    if !inside_stock {
                        continue;
                    }
                    
                    // B. Check overlap with already placed pieces
                    s_state.grid.query(&trans_poly.aabb, &mut query_buf);
                    let mut overlap = false;
                    for &idx in &query_buf {
                        if polygons_overlap(&trans_poly, &s_state.placed_pieces[idx].poly) {
                            overlap = true;
                            break;
                        }
                    }
                    
                    if !overlap {
                        // Place!
                        s_state.grid.insert(s_state.placed_pieces.len(), &trans_poly.aabb);
                        s_state.placed_area += p_area;
                        s_state.placed_pieces.push(PlacedPiece {
                            poly: trans_poly,
                            orig_idx,
                            rotation: rot,
                            translation: c,
                        });
                        placed = true;
                        break;
                    }
                }
                
                if placed {
                    break;
                }
            }
            
            if placed {
                break;
            }
        }
        
        if !placed {
            // Open a new stock!
            let mut s_grid = SpatialGrid::new(stock_aabb, avg_w, avg_h);
            
            // Try to place at bottom-left of stock AABB
            // Rotate the piece by 0.0 or find best initial rotation
            let mut best_c = Point { x: 0.0, y: 0.0 };
            let mut best_rot = 0.0;
            let mut best_aabb_area = f64::INFINITY;
            let mut found = false;
            
            for &rot in &candidate_rotations {
                let p_rot = p_poly.transform(0.0, 0.0, rot);
                let p_aabb = p_rot.aabb;
                let c = Point {
                    x: stock_aabb.min_x - p_aabb.min_x,
                    y: stock_aabb.min_y - p_aabb.min_y,
                };
                
                let trans_poly = p_rot.transform(c.x, c.y, 0.0);
                // Verify containment
                let inside_stock = match stock_type {
                    StockType::Rectangle { w, h } => {
                        let mut ok = true;
                        for v in &trans_poly.vertices {
                            if v.x < -1e-6 || v.x > w + 1e-6 || v.y < -1e-6 || v.y > h + 1e-6 {
                                ok = false;
                                break;
                            }
                        }
                        ok
                    }
                    _ => {
                        let mut ok = true;
                        for &v in &trans_poly.vertices {
                            if !point_in_polygon(v, &stock) && !is_point_on_boundary(v, &stock) {
                                ok = false;
                                break;
                            }
                        }
                        if ok {
                            let n1 = trans_poly.vertices.len();
                            let n2 = stock.vertices.len();
                            'outer2: for i in 0..n1 {
                                let a = trans_poly.vertices[i];
                                  let b = trans_poly.vertices[(i + 1) % n1];
                                for j in 0..n2 {
                                    let c = stock.vertices[j];
                                    let d = stock.vertices[(j + 1) % n2];
                                    if segments_cross(a, b, c, d) {
                                        ok = false;
                                        break 'outer2;
                                    }
                                }
                            }
                        }
                        ok
                    }
                };
                
                if inside_stock {
                    let aabb_w = p_aabb.max_x - p_aabb.min_x;
                    let aabb_h = p_aabb.max_y - p_aabb.min_y;
                    let aabb_area = aabb_w * aabb_h;
                    if aabb_area < best_aabb_area - 1e-6 {
                        best_aabb_area = aabb_area;
                        best_c = c;
                        best_rot = rot;
                        found = true;
                    }
                }
            }
            
            if !found {
                // If not fitting at bottom-left, just find first rotation that fits at any candidate
                // In practice, since it's the first piece on an empty stock, it should fit.
                // We will fall back to rot = 0.0, translation = 0.0
                best_rot = 0.0;
                let p_rot = p_poly.transform(0.0, 0.0, 0.0);
                best_c = Point {
                    x: stock_aabb.min_x - p_rot.aabb.min_x,
                    y: stock_aabb.min_y - p_rot.aabb.min_y,
                };
            }
            
            let trans_poly = p_poly.transform(best_c.x, best_c.y, best_rot);
            s_grid.insert(0, &trans_poly.aabb);
            
            stocks.push(StockState {
                placed_pieces: vec![PlacedPiece {
                    poly: trans_poly,
                    orig_idx,
                    rotation: best_rot,
                    translation: best_c,
                }],
                grid: s_grid,
                placed_area: p_area,
            });
        }
    }
    
    // Map placements back if stock was rotated
    for (s_idx, s_state) in stocks.iter().enumerate() {
        for placed_p in &s_state.placed_pieces {
            let final_rot;
            let final_trans;
            if let Some((v0, alpha)) = rotation_info {
                final_rot = normalize_angle(placed_p.rotation);
                let rotated_trans = rotate_point(placed_p.translation, alpha);
                final_trans = Point {
                    x: rotated_trans.x + v0.x,
                    y: rotated_trans.y + v0.y,
                };
            } else {
                final_rot = normalize_angle(placed_p.rotation);
                final_trans = placed_p.translation;
            }
            final_placements[placed_p.orig_idx] = (s_idx, final_trans, final_rot);
        }
    }
    
    // Output results
    println!("{}", stocks.len());
    for (s_idx, trans, rot) in final_placements {
        println!("{} {:.8} {:.8} {:.8}", s_idx, trans.x, trans.y, rot);
    }
}

fn normalize_angle(mut angle: f64) -> f64 {
    while angle > std::f64::consts::PI {
        angle -= 2.0 * std::f64::consts::PI;
    }
    while angle <= -std::f64::consts::PI {
        angle += 2.0 * std::f64::consts::PI;
    }
    angle
}
