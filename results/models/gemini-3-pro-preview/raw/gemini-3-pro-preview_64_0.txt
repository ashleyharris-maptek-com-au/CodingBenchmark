declare const require: any;
const fs = require('fs');

interface Drone {
    id: any;
    x: number;
    y: number;
    originalIndex: number; // to map back to output format order
}

interface InputData {
    probability: number[][];
    drones: { id: any; x: number; y: number }[];
    maxStepsPerDrone: number;
}

interface AStarNode {
    x: number;
    y: number;
    g: number;
    f: number;
    parent: AStarNode | null;
}

interface BFSNode {
    x: number;
    y: number;
    d: number;
    pathValue: number;
    parent: BFSNode | null;
}

class MinHeap<T> {
    private data: T[] = [];
    constructor(private compare: (a: T, b: T) => number) {}
    push(item: T) {
        this.data.push(item);
        this.up(this.data.length - 1);
    }
    pop(): T | undefined {
        if (this.data.length === 0) return undefined;
        const top = this.data[0];
        const bottom = this.data.pop()!;
        if (this.data.length > 0) {
            this.data[0] = bottom;
            this.down(0);
        }
        return top;
    }
    size(): number {
        return this.data.length;
    }
    private up(i: number) {
        while (i > 0) {
            const p = (i - 1) >> 1;
            if (this.compare(this.data[i], this.data[p]) < 0) {
                this.swap(i, p);
                i = p;
            } else {
                break;
            }
        }
    }
    private down(i: number) {
        const len = this.data.length;
        while ((i << 1) + 1 < len) {
            let child = (i << 1) + 1;
            if (child + 1 < len && this.compare(this.data[child + 1], this.data[child]) < 0) {
                child++;
            }
            if (this.compare(this.data[child], this.data[i]) < 0) {
                this.swap(i, child);
                i = child;
            } else {
                break;
            }
        }
    }
    private swap(i: number, j: number) {
        const temp = this.data[i];
        this.data[i] = this.data[j];
        this.data[j] = temp;
    }
}

// Global preallocated/reused arrays to avoid GC pressure and large allocations
let globalVisitedDist: Int32Array;
let globalMaxValAtDist: Float64Array;
let globalVisitedAStar: Int32Array;
let globalVisitedGrid: Uint8Array;

function findPathAStar(
    startX: number,
    startY: number,
    targetX: number,
    targetY: number,
    probability: number[][],
    W: number,
    H: number,
    maxSteps: number
): [number, number][] | null {
    const heap = new MinHeap<AStarNode>((a, b) => a.f - b.f);
    const visitedList: number[] = [];

    const h = (x: number, y: number) => Math.abs(x - targetX) + Math.abs(y - targetY);

    const startIdx = startY * W + startX;
    globalVisitedAStar[startIdx] = 0;
    visitedList.push(startIdx);

    const startNode: AStarNode = {
        x: startX,
        y: startY,
        g: 0,
        f: h(startX, startY),
        parent: null
    };
    heap.push(startNode);

    const dx = [1, -1, 0, 0];
    const dy = [0, 0, 1, -1];

    let foundNode: AStarNode | null = null;

    while (heap.size() > 0) {
        const curr = heap.pop()!;
        if (curr.x === targetX && curr.y === targetY) {
            foundNode = curr;
            break;
        }

        if (curr.g >= maxSteps) {
            continue;
        }

        for (let i = 0; i < 4; i++) {
            const nx = curr.x + dx[i];
            const ny = curr.y + dy[i];
            if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
                if (probability[ny][nx] !== -1) {
                    const ng = curr.g + 1;
                    const nidx = ny * W + nx;
                    const prevG = globalVisitedAStar[nidx];
                    if (prevG === -1 || ng < prevG) {
                        if (prevG === -1) {
                            visitedList.push(nidx);
                        }
                        globalVisitedAStar[nidx] = ng;
                        heap.push({
                            x: nx,
                            y: ny,
                            g: ng,
                            f: ng + h(nx, ny),
                            parent: curr
                        });
                    }
                }
            }
        }
    }

    // Reset visited array
    for (let i = 0; i < visitedList.length; i++) {
        globalVisitedAStar[visitedList[i]] = -1;
    }

    if (foundNode) {
        const path: [number, number][] = [];
        let node: AStarNode | null = foundNode;
        while (node.parent !== null) {
            path.push([node.x, node.y]);
            node = node.parent;
        }
        path.reverse();
        return path;
    }

    return null;
}

function findBestTargetBFS(
    startX: number,
    startY: number,
    maxSteps: number,
    probability: number[][],
    W: number,
    H: number,
    maxSearchDepth: number,
    alpha: number
): { x: number; y: number; path: [number, number][] } | null {
    const limit = Math.min(maxSteps, maxSearchDepth);
    if (limit <= 0) return null;

    const queue: BFSNode[] = [];

    const startIdx = startY * W + startX;
    globalVisitedDist[startIdx] = 0;
    globalMaxValAtDist[startIdx] = 0;

    const startNode: BFSNode = {
        x: startX,
        y: startY,
        d: 0,
        pathValue: 0,
        parent: null
    };

    queue.push(startNode);

    const dx = [1, -1, 0, 0];
    const dy = [0, 0, 1, -1];

    let head = 0;
    let bestTargetNode: BFSNode | null = null;
    let bestScore = -Infinity;

    while (head < queue.length) {
        const curr = queue[head++];

        const cellIdx = curr.y * W + curr.x;
        const cellProb = probability[curr.y][curr.x];
        if (curr.d > 0 && cellProb > 0 && globalVisitedGrid[cellIdx] === 0) {
            const score = curr.pathValue / Math.pow(curr.d, alpha);
            if (score > bestScore) {
                bestScore = score;
                bestTargetNode = curr;
            }
        }

        if (curr.d >= limit) {
            continue;
        }

        for (let i = 0; i < 4; i++) {
            const nx = curr.x + dx[i];
            const ny = curr.y + dy[i];
            if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
                if (probability[ny][nx] !== -1) {
                    const nd = curr.d + 1;
                    const nidx = ny * W + nx;
                    
                    let cellVal = probability[ny][nx];
                    if (globalVisitedGrid[nidx] === 1 || cellVal < 0) {
                        cellVal = 0;
                    } else if (cellVal > 0) {
                        let temp: BFSNode | null = curr;
                        while (temp !== null) {
                            if (temp.x === nx && temp.y === ny) {
                                cellVal = 0;
                                break;
                            }
                            temp = temp.parent;
                        }
                    }

                    const npathValue = curr.pathValue + cellVal;
                    const prevD = globalVisitedDist[nidx];

                    if (prevD === -1) {
                        globalVisitedDist[nidx] = nd;
                        globalMaxValAtDist[nidx] = npathValue;
                        queue.push({
                            x: nx,
                            y: ny,
                            d: nd,
                            pathValue: npathValue,
                            parent: curr
                        });
                    } else if (prevD === nd && npathValue > globalMaxValAtDist[nidx]) {
                        globalMaxValAtDist[nidx] = npathValue;
                        queue.push({
                            x: nx,
                            y: ny,
                            d: nd,
                            pathValue: npathValue,
                            parent: curr
                        });
                    }
                }
            }
        }
    }

    // Reset visited arrays
    for (let i = 0; i < queue.length; i++) {
        const node = queue[i];
        const idx = node.y * W + node.x;
        globalVisitedDist[idx] = -1;
        globalMaxValAtDist[idx] = -1;
    }

    if (bestTargetNode) {
        const path: [number, number][] = [];
        let node: BFSNode | null = bestTargetNode;
        while (node.parent !== null) {
            path.push([node.x, node.y]);
            node = node.parent;
        }
        path.reverse();
        return {
            x: bestTargetNode.x,
            y: bestTargetNode.y,
            path
        };
    }

    return null;
}

function findBestTargetGlobal(
    startX: number,
    startY: number,
    maxSteps: number,
    probability: number[][],
    W: number,
    H: number,
    activePositiveCells: [number, number][],
    alpha: number
): { x: number; y: number; path: [number, number][] } | null {
    interface Candidate {
        x: number;
        y: number;
        score: number;
    }
    const topCandidates: Candidate[] = [];

    for (let i = 0; i < activePositiveCells.length; i++) {
        const cell = activePositiveCells[i];
        const cx = cell[0];
        const cy = cell[1];
        if (globalVisitedGrid[cy * W + cx] === 1) continue;

        const dist = Math.abs(cx - startX) + Math.abs(cy - startY);
        if (dist === 0 || dist > maxSteps) continue;

        const val = probability[cy][cx];
        const score = val / Math.pow(dist, alpha);

        if (topCandidates.length < 5 || score > topCandidates[topCandidates.length - 1].score) {
            let insertIdx = topCandidates.length;
            while (insertIdx > 0 && score > topCandidates[insertIdx - 1].score) {
                insertIdx--;
            }
            topCandidates.splice(insertIdx, 0, { x: cx, y: cy, score });
            if (topCandidates.length > 5) {
                topCandidates.pop();
            }
        }
    }

    if (topCandidates.length === 0) return null;

    for (let i = 0; i < topCandidates.length; i++) {
        const cand = topCandidates[i];
        const path = findPathAStar(startX, startY, cand.x, cand.y, probability, W, H, maxSteps);
        if (path !== null) {
            return {
                x: cand.x,
                y: cand.y,
                path
            };
        }
    }

    return null;
}

function solveOnce(
    probability: number[][],
    drones: Drone[],
    maxSteps: number,
    W: number,
    H: number,
    initialActiveCells: [number, number][],
    alpha: number,
    maxSearchDepth: number
): [number, number][][] {
    globalVisitedGrid.fill(0);
    let activeCells = [...initialActiveCells];

    // Initialize paths with starting positions
    const paths: [number, number][][] = Array.from({ length: drones.length }, () => []);
    const currX = new Int32Array(drones.length);
    const currY = new Int32Array(drones.length);
    const remSteps = new Int32Array(drones.length);

    for (let i = 0; i < drones.length; i++) {
        const d = drones[i];
        paths[d.originalIndex].push([d.x, d.y]);
        globalVisitedGrid[d.y * W + d.x] = 1;
        currX[i] = d.x;
        currY[i] = d.y;
        remSteps[i] = maxSteps;
    }

    // Process drones sequentially
    for (let i = 0; i < drones.length; i++) {
        // Filter active cells list once per drone to remove already visited cells
        activeCells = activeCells.filter(ac => globalVisitedGrid[ac[1] * W + ac[0]] === 0);

        const d = drones[i];
        const origIdx = d.originalIndex;
        let x = currX[i];
        let y = currY[i];
        let steps = remSteps[i];

        while (steps > 0) {
            // 1. Try local BFS search first
            let target = findBestTargetBFS(x, y, steps, probability, W, H, maxSearchDepth, alpha);
            
            // 2. If no target found nearby, try global search using active cells
            if (!target) {
                target = findBestTargetGlobal(x, y, steps, probability, W, H, activeCells, alpha);
            }

            if (!target) {
                // No reachable targets left, pad the path by staying in place
                while (steps > 0) {
                    paths[origIdx].push([x, y]);
                    steps--;
                }
                break;
            }

            // Move to the target
            for (const cell of target.path) {
                paths[origIdx].push(cell);
                globalVisitedGrid[cell[1] * W + cell[0]] = 1;
                steps--;
            }
            x = target.x;
            y = target.y;
        }
    }

    return paths;
}

function calculateScore(paths: [number, number][][], probability: number[][], W: number, H: number): number {
    globalVisitedGrid.fill(0);
    let score = 0;
    for (let i = 0; i < paths.length; i++) {
        const path = paths[i];
        for (let j = 0; j < path.length; j++) {
            const p = path[j];
            const idx = p[1] * W + p[0];
            if (globalVisitedGrid[idx] === 0) {
                globalVisitedGrid[idx] = 1;
                const val = probability[p[1]][p[0]];
                if (val > 0) {
                    score += val;
                }
            }
        }
    }
    return score;
}

function main() {
    const input: string = fs.readFileSync(0, 'utf8');
    if (!input.trim()) {
        console.log(JSON.stringify({ paths: [] }));
        return;
    }
    const data: InputData = JSON.parse(input);

    const probability = data.probability;
    const dronesInput = data.drones;
    const maxSteps = data.maxStepsPerDrone;

    if (!probability || probability.length === 0 || !dronesInput || dronesInput.length === 0) {
        console.log(JSON.stringify({ paths: [] }));
        return;
    }

    const H = probability.length;
    const W = probability[0].length;
    const size = W * H;

    // Allocate global arrays
    globalVisitedDist = new Int32Array(size).fill(-1);
    globalMaxValAtDist = new Float64Array(size).fill(-1);
    globalVisitedAStar = new Int32Array(size).fill(-1);
    globalVisitedGrid = new Uint8Array(size);

    const drones: Drone[] = dronesInput.map((d, index) => ({
        id: d.id,
        x: d.x,
        y: d.y,
        originalIndex: index
    }));

    // Find all positive cells initially
    const initialActiveCells: [number, number][] = [];
    for (let y = 0; y < H; y++) {
        for (let x = 0; x < W; x++) {
            if (probability[y][x] > 0) {
                initialActiveCells.push([x, y]);
            }
        }
    }

    const startTime = Date.now();
    const timeLimitMs = 1500; // 1.5 seconds budget

    // Calculate dynamic search depth based on grid size and drone steps
    const dronesCount = drones.length;
    const expectedBFSCalls = (dronesCount * maxSteps) / 4;
    // We want expectedBFSCalls * (2 * depth^2) <= 250000 operations
    const calculatedDepth = Math.floor(Math.sqrt(250000 / Math.max(1, expectedBFSCalls)));
    const defaultMaxDepth = Math.max(5, Math.min(calculatedDepth, Math.min(maxSteps, 30)));

    let bestPaths: [number, number][][] | null = null;
    let bestScore = -Infinity;

    // Run first iteration deterministically with ordered drones and standard parameters
    const defaultAlpha = 1.3;
    const initialPaths = solveOnce(probability, drones, maxSteps, W, H, initialActiveCells, defaultAlpha, defaultMaxDepth);
    bestPaths = initialPaths;
    bestScore = calculateScore(initialPaths, probability, W, H);

    let iterations = 1;

    // Continue running randomized/shuffled iterations within the time limit
    while (Date.now() - startTime < timeLimitMs) {
        // Shuffle drones order
        const shuffledDrones = [...drones];
        for (let i = shuffledDrones.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            const temp = shuffledDrones[i];
            shuffledDrones[i] = shuffledDrones[j];
            shuffledDrones[j] = temp;
        }

        // Randomize parameters slightly
        const randomAlpha = 1.0 + Math.random() * 0.8; // alpha between 1.0 and 1.8
        const depthDiff = Math.max(2, Math.floor(defaultMaxDepth * 0.3));
        const randomMaxDepth = Math.max(3, defaultMaxDepth - depthDiff + Math.floor(Math.random() * (2 * depthDiff + 1)));

        const paths = solveOnce(probability, shuffledDrones, maxSteps, W, H, initialActiveCells, randomAlpha, randomMaxDepth);
        const score = calculateScore(paths, probability, W, H);

        if (score > bestScore) {
            bestScore = score;
            bestPaths = paths;
        }

        iterations++;
    }

    // Output the best paths
    console.log(JSON.stringify({ paths: bestPaths }));
}

main();
