const fs = require('fs');

function solve() {
    let input;
    try {
        input = fs.readFileSync(0, 'utf8');
    } catch (e) {
        return;
    }
    if (!input || input.trim() === '') return;
    
    let data;
    try {
        data = JSON.parse(input);
    } catch (e) {
        return;
    }
    
    const { width, height, grid, ignitions, towns, cutBudget } = data;
    if (!width || !height || !grid || !ignitions || !towns || cutBudget === undefined) {
        console.log(JSON.stringify({ firebreaks: [] }));
        return;
    }
    
    const N = width * height;
    
    // 1. Initial BFS to find reachable forest cells from ignitions
    const reachable = new Set();
    const queue = [];
    for (const [ix, iy] of ignitions) {
        const id = iy * width + ix;
        reachable.add(id);
        queue.push(id);
    }
    
    let qHead = 0;
    while (qHead < queue.length) {
        const curr = queue[qHead++];
        const cx = curr % width;
        const cy = Math.floor(curr / width);
        const neighbors = [
            [cx + 1, cy],
            [cx - 1, cy],
            [cx, cy + 1],
            [cx, cy - 1]
        ];
        for (const [nx, ny] of neighbors) {
            if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                if (grid[ny][nx] === '.') {
                    const nid = ny * width + nx;
                    if (!reachable.has(nid)) {
                        reachable.add(nid);
                        queue.push(nid);
                    }
                }
            }
        }
    }
    
    // Check which towns are reachable
    const reachableTowns = towns.filter(t => {
        const tid = t.y * width + t.x;
        return reachable.has(tid);
    });
    
    if (reachableTowns.length === 0 || cutBudget <= 0) {
        console.log(JSON.stringify({ firebreaks: [] }));
        return;
    }
    
    // Map reachable cells to indices
    const cellToIdx = new Map();
    const idxToCell = [];
    for (const id of reachable) {
        const idx = idxToCell.length;
        cellToIdx.set(id, idx);
        idxToCell.push(id);
    }
    
    const M = idxToCell.length;
    const S = 2 * M;
    const T = 2 * M + 1;
    const maxNodes = 2 * M + 2;
    
    const isIgnition = new Uint8Array(N);
    for (const [ix, iy] of ignitions) {
        isIgnition[iy * width + ix] = 1;
    }
    
    const cellValue = new Float64Array(N);
    for (const t of towns) {
        cellValue[t.y * width + t.x] += t.value;
    }
    
    // Count exact number of edges to avoid over-allocation
    let totalEdgesCount = 0;
    totalEdgesCount += 2 * M; // Internal node edges
    for (let idx = 0; idx < M; idx++) {
        const id = idxToCell[idx];
        if (isIgnition[id]) {
            totalEdgesCount += 2; // S -> 2*idx
        } else if (cellValue[id] > 0) {
            totalEdgesCount += 2; // 2*idx + 1 -> T
        }
        
        const cx = id % width;
        const cy = Math.floor(id / width);
        const neighbors = [
            [cx + 1, cy],
            [cx - 1, cy],
            [cx, cy + 1],
            [cx, cy - 1]
        ];
        for (const [nx, ny] of neighbors) {
            if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                const nid = ny * width + nx;
                if (cellToIdx.has(nid)) {
                    totalEdgesCount += 2; // Adjacency edge
                }
            }
        }
    }
    
    const head = new Int32Array(maxNodes).fill(-1);
    const to = new Int32Array(totalEdgesCount);
    const next = new Int32Array(totalEdgesCount);
    const cap = new Float64Array(totalEdgesCount);
    const flow = new Float64Array(totalEdgesCount);
    let edgeCount = 0;
    
    function addEdge(u, v, w) {
        to[edgeCount] = v;
        next[edgeCount] = head[u];
        cap[edgeCount] = w;
        flow[edgeCount] = 0;
        head[u] = edgeCount++;
        
        to[edgeCount] = u;
        next[edgeCount] = head[v];
        cap[edgeCount] = 0;
        flow[edgeCount] = 0;
        head[v] = edgeCount++;
        return edgeCount - 2;
    }
    
    const nodeEdgeIdx = new Int32Array(M).fill(-1);
    const townEdgeIdx = new Int32Array(M).fill(-1);
    
    for (let idx = 0; idx < M; idx++) {
        const id = idxToCell[idx];
        const isIgn = isIgnition[id];
        const val = cellValue[id];
        
        if (isIgn) {
            addEdge(2 * idx, 2 * idx + 1, Infinity);
            addEdge(S, 2 * idx, Infinity);
        } else if (val > 0) {
            addEdge(2 * idx, 2 * idx + 1, Infinity);
            const e = addEdge(2 * idx + 1, T, val);
            townEdgeIdx[idx] = e;
        } else {
            const e = addEdge(2 * idx, 2 * idx + 1, 0);
            nodeEdgeIdx[idx] = e;
        }
        
        const cx = id % width;
        const cy = Math.floor(id / width);
        const neighbors = [
            [cx + 1, cy],
            [cx - 1, cy],
            [cx, cy + 1],
            [cx, cy - 1]
        ];
        for (const [nx, ny] of neighbors) {
            if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                const nid = ny * width + nx;
                if (cellToIdx.has(nid)) {
                    const nidx = cellToIdx.get(nid);
                    addEdge(2 * idx + 1, 2 * nidx, Infinity);
                }
            }
        }
    }
    
    const level = new Int32Array(maxNodes);
    const ptr = new Int32Array(maxNodes);
    const q = new Int32Array(maxNodes);
    
    function bfsDinic() {
        level.fill(-1);
        level[S] = 0;
        let h = 0, t = 0;
        q[t++] = S;
        while (h < t) {
            const u = q[h++];
            for (let e = head[u]; e !== -1; e = next[e]) {
                const v = to[e];
                if (level[v] === -1 && cap[e] - flow[e] > 1e-9) {
                    level[v] = level[u] + 1;
                    q[t++] = v;
                }
            }
        }
        return level[T] !== -1;
    }
    
    function dfsDinic(u, pushed) {
        if (pushed <= 1e-9) return 0;
        if (u === T) return pushed;
        let totalPushed = 0;
        for (let e = ptr[u]; e !== -1; e = next[e]) {
            ptr[u] = e;
            const v = to[e];
            if (level[u] + 1 !== level[v] || cap[e] - flow[e] <= 1e-9) continue;
            const tr = dfsDinic(v, Math.min(pushed - totalPushed, cap[e] - flow[e]));
            if (tr > 1e-9) {
                flow[e] += tr;
                flow[e ^ 1] -= tr;
                totalPushed += tr;
                if (totalPushed >= pushed - 1e-9) break;
            }
        }
        return totalPushed;
    }
    
    function dinic() {
        let maxFlow = 0;
        while (bfsDinic()) {
            for (let i = 0; i < maxNodes; i++) {
                ptr[i] = head[i];
            }
            while (true) {
                const pushed = dfsDinic(S, Infinity);
                if (pushed <= 1e-9) break;
                maxFlow += pushed;
            }
        }
        return maxFlow;
    }
    
    const visited = new Uint8Array(maxNodes);
    function findReachableNodes() {
        visited.fill(0);
        visited[S] = 1;
        let h = 0, t = 0;
        q[t++] = S;
        while (h < t) {
            const u = q[h++];
            for (let e = head[u]; e !== -1; e = next[e]) {
                const v = to[e];
                if (!visited[v] && cap[e] - flow[e] > 1e-9) {
                    visited[v] = 1;
                    q[t++] = v;
                }
            }
        }
    }
    
    function solveForC(C) {
        flow.fill(0);
        for (let idx = 0; idx < M; idx++) {
            const e = nodeEdgeIdx[idx];
            if (e !== -1) {
                cap[e] = C;
                cap[e ^ 1] = 0;
            }
        }
        for (let idx = 0; idx < M; idx++) {
            const e = townEdgeIdx[idx];
            if (e !== -1) {
                cap[e] = cellValue[idxToCell[idx]];
                cap[e ^ 1] = 0;
            }
        }
        
        dinic();
        findReachableNodes();
        
        const cuts = [];
        for (let idx = 0; idx < M; idx++) {
            if (nodeEdgeIdx[idx] !== -1) {
                if (visited[2 * idx] === 1 && visited[2 * idx + 1] === 0) {
                    cuts.push(idxToCell[idx]);
                }
            }
        }
        
        let savedValue = 0;
        for (const t of towns) {
            const tid = t.y * width + t.x;
            if (!cellToIdx.has(tid)) {
                savedValue += t.value;
            } else {
                const idx = cellToIdx.get(tid);
                if (visited[2 * idx + 1] === 0) {
                    savedValue += t.value;
                }
            }
        }
        
        return { cuts, savedValue };
    }
    
    let totalReachableValue = 0;
    for (const t of reachableTowns) {
        totalReachableValue += t.value;
    }
    
    let bestCuts = [];
    let bestValue = -1;
    
    let low = 0;
    let high = totalReachableValue;
    for (let iter = 0; iter < 30; iter++) {
        const mid = (low + high) / 2;
        const { cuts, savedValue } = solveForC(mid);
        if (cuts.length <= cutBudget) {
            if (savedValue > bestValue || (savedValue === bestValue && cuts.length < bestCuts.length)) {
                bestValue = savedValue;
                bestCuts = cuts;
            }
            high = mid;
        } else {
            low = mid;
        }
    }
    
    // 2. Greedy search to top up remaining budget
    function getReachableCells(cuts) {
        const r = new Set();
        const qu = [];
        for (const [ix, iy] of ignitions) {
            const id = iy * width + ix;
            r.add(id);
            qu.push(id);
        }
        let h = 0;
        while (h < qu.length) {
            const curr = qu[h++];
            const cx = curr % width;
            const cy = Math.floor(curr / width);
            const neighbors = [
                [cx + 1, cy],
                [cx - 1, cy],
                [cx, cy + 1],
                [cx, cy - 1]
            ];
            for (const [nx, ny] of neighbors) {
                if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                    const nid = ny * width + nx;
                    if (grid[ny][nx] === '.' && !cuts.has(nid) && !r.has(nid)) {
                        r.add(nid);
                        qu.push(nid);
                    }
                }
            }
        }
        return r;
    }
    
    function getAdditionalCutsForTown(t, currentCuts) {
        const tid = t.y * width + t.x;
        if (!cellToIdx.has(tid)) return null;
        const targetIdx = cellToIdx.get(tid);
        
        flow.fill(0);
        for (let idx = 0; idx < M; idx++) {
            const e = nodeEdgeIdx[idx];
            if (e !== -1) {
                const cellId = idxToCell[idx];
                if (currentCuts.has(cellId)) {
                    cap[e] = 0;
                } else {
                    cap[e] = 1.0;
                }
                cap[e ^ 1] = 0;
            }
        }
        
        for (let idx = 0; idx < M; idx++) {
            const e = townEdgeIdx[idx];
            if (e !== -1) {
                if (idx === targetIdx) {
                    cap[e] = Infinity;
                } else {
                    cap[e] = 0;
                }
                cap[e ^ 1] = 0;
            }
        }
        
        const maxFlow = dinic();
        if (maxFlow >= 1e9) {
            return null;
        }
        
        findReachableNodes();
        
        const newCuts = [];
        for (let idx = 0; idx < M; idx++) {
            if (nodeEdgeIdx[idx] !== -1) {
                const cellId = idxToCell[idx];
                if (!currentCuts.has(cellId)) {
                    if (visited[2 * idx] === 1 && visited[2 * idx + 1] === 0) {
                        newCuts.push(cellId);
                    }
                }
            }
        }
        return newCuts;
    }
    
    const currentCuts = new Set(bestCuts);
    while (currentCuts.size < cutBudget) {
        const reachableCells = getReachableCells(currentCuts);
        const unsavedTowns = towns.filter(t => reachableCells.has(t.y * width + t.x));
        if (unsavedTowns.length === 0) break;
        
        let currentSavedValue = 0;
        for (const town of towns) {
            if (!reachableCells.has(town.y * width + town.x)) {
                currentSavedValue += town.value;
            }
        }
        
        unsavedTowns.sort((a, b) => b.value - a.value);
        const limit = Math.min(unsavedTowns.length, 15);
        let bestNextCuts = null;
        let bestScore = -1;
        
        for (let i = 0; i < limit; i++) {
            const t = unsavedTowns[i];
            const newCuts = getAdditionalCutsForTown(t, currentCuts);
            if (newCuts !== null && newCuts.length > 0) {
                const cost = newCuts.length;
                if (currentCuts.size + cost <= cutBudget) {
                    const tempCuts = new Set(currentCuts);
                    for (const cid of newCuts) tempCuts.add(cid);
                    const tempReachable = getReachableCells(tempCuts);
                    let tempSavedValue = 0;
                    for (const town of towns) {
                        if (!tempReachable.has(town.y * width + town.x)) {
                            tempSavedValue += town.value;
                        }
                    }
                    const benefit = tempSavedValue - currentSavedValue;
                    const score = benefit / cost;
                    if (score > bestScore) {
                        bestScore = score;
                        bestNextCuts = newCuts;
                    }
                }
            }
        }
        
        if (bestNextCuts === null) break;
        for (const cid of bestNextCuts) {
            currentCuts.add(cid);
        }
    }
    
    const finalCuts = [];
    for (const cid of currentCuts) {
        const cx = cid % width;
        const cy = Math.floor(cid / width);
        finalCuts.push([cx, cy]);
    }
    
    console.log(JSON.stringify({ firebreaks: finalCuts }));
}

solve();
