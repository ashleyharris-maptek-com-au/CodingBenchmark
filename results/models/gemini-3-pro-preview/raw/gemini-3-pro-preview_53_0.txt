```python
import math

# Globals
_EST_POS = None
_EST_VEL = None
_PREV_TIME = None
_PREV_CONTROLS = None
_INTEGRATED_ACCEL_BIAS = [0.0, 0.0, 0.0]

def estimate_relative_state(state: dict, dt: float):
    global _EST_POS, _EST_VEL, _PREV_TIME, _PREV_CONTROLS, _INTEGRATED_ACCEL_BIAS
    
    t = state.get('sim_time_s', 0.0)
    if _PREV_TIME is None or t < _PREV_TIME or t < 0.5:
        # Reset
        _EST_POS = None
        _EST_VEL = None
        _PREV_TIME = None
        _PREV_CONTROLS = None
        _INTEGRATED_ACCEL_BIAS = [0.0, 0.0, 0.0]

    # 1. Gather range finders
    valid_ranges = []
    for i in range(1, 4):
        if state.get(f'range_{i}_valid', False):
            r_val = state.get(f'range_{i}_m', 0.0)
            if r_val > 0.1:
                valid_ranges.append(r_val)
    
    r_est = None
    if valid_ranges:
        valid_ranges.sort()
        r_est = valid_ranges[len(valid_ranges) // 2]  # median

    # 2. Gather position candidates
    pos_candidates = []
    
    # Lidar
    if state.get('lidar_rel_x_valid', False) and state.get('lidar_rel_y_valid', False) and state.get('lidar_rel_z_valid', False):
        pos_candidates.append(('lidar', [state['lidar_rel_pos_x_m'], state['lidar_rel_pos_y_m'], state['lidar_rel_pos_z_m']]))
        
    # Radar
    if state.get('radar_rel_x_valid', False) and state.get('radar_rel_y_valid', False) and state.get('radar_rel_z_valid', False):
        pos_candidates.append(('radar', [state['radar_rel_pos_x_m'], state['radar_rel_pos_y_m'], state['radar_rel_pos_z_m']]))
        
    # Docking camera
    if state.get('dock_camera_active', False):
        if state.get('dock_camera_bearing_x_valid', False) and state.get('dock_camera_bearing_y_valid', False) and state.get('dock_camera_bearing_z_valid', False):
            cam_r = state.get('dock_camera_range_m', 0.0)
            if cam_r > 0.1:
                pos_candidates.append(('cam', [
                    state['dock_camera_bearing_x'] * cam_r,
                    state['dock_camera_bearing_y'] * cam_r,
                    state['dock_camera_bearing_z'] * cam_r
                ]))

    # Cross-check position candidates against r_est
    consistent_pos = []
    for name, p in pos_candidates:
        p_norm = math.sqrt(p[0]**2 + p[1]**2 + p[2]**2)
        if r_est is not None:
            tol = max(5.0, r_est * 0.10)
            if abs(p_norm - r_est) < tol:
                consistent_pos.append(p)
        else:
            consistent_pos.append(p)

    # Determine final position measurement
    pos_meas = None
    if consistent_pos:
        pos_meas = [sum(x)/len(consistent_pos) for x in zip(*consistent_pos)]
    elif pos_candidates:
        lidar_candidates = [p for name, p in pos_candidates if name == 'lidar']
        if lidar_candidates:
            pos_meas = lidar_candidates[0]
        else:
            pos_meas = pos_candidates[0][1]

    # 3. Gather velocity candidates
    vel_candidates = []
    if state.get('lidar_relv_x_valid', False) and state.get('lidar_relv_y_valid', False) and state.get('lidar_relv_z_valid', False):
        vel_candidates.append(('lidar', [state['lidar_rel_vel_x_ms'], state['lidar_rel_vel_y_ms'], state['lidar_rel_vel_z_ms']]))
        
    if state.get('radar_relv_x_valid', False) and state.get('radar_relv_y_valid', False) and state.get('radar_relv_z_valid', False):
        vel_candidates.append(('radar', [state['radar_rel_vel_x_ms'], state['radar_rel_vel_y_ms'], state['radar_rel_vel_z_ms']]))

    # Finite difference velocity
    vel_diff = None
    if _EST_POS is not None and dt > 0.0:
        vel_diff = [(pos_meas[i] - _EST_POS[i]) / dt for i in range(3)] if pos_meas is not None else None

    # Cross-check velocity candidates
    consistent_vel = []
    for name, v in vel_candidates:
        if vel_diff is not None:
            diff_norm = math.sqrt(sum((v[i] - vel_diff[i])**2 for i in range(3)))
            if diff_norm < 1.0:
                consistent_vel.append(v)
        else:
            consistent_vel.append(v)

    vel_meas = None
    if consistent_vel:
        vel_meas = [sum(x)/len(consistent_vel) for x in zip(*consistent_vel)]
    elif vel_candidates:
        lidar_vel_cand = [v for name, v in vel_candidates if name == 'lidar']
        if lidar_vel_cand:
            vel_meas = lidar_vel_cand[0]
        else:
            vel_meas = vel_candidates[0][1]
    elif vel_diff is not None:
        vel_meas = vel_diff

    # 4. Filter / Update State
    if _EST_POS is None:
        if pos_meas is not None:
            _EST_POS = list(pos_meas)
        else:
            _EST_POS = [0.0, 0.0, 0.0]
    else:
        # Propagate
        sma = state.get('ground_semi_major_axis_m', 6779000.0) if state.get('ground_semi_major_axis_m_valid', False) else 6779000.0
        if sma < 6e6 or sma > 8e6:
            sma = 6779000.0
        omega = math.sqrt(3.986004418e14 / sma**3)

        fuel = state.get('fuel_remaining_kg', 500.0)
        total_mass = 7000.0 + fuel
        ax, ay, az = 0.0, 0.0, 0.0
        if _PREV_CONTROLS is not None:
            tx = _PREV_CONTROLS.get('thrust_x', 0.0) * 220.0
            ty = _PREV_CONTROLS.get('thrust_y', 0.0) * 220.0
            tz = _PREV_CONTROLS.get('thrust_z', 0.0) * 220.0
            tx += _INTEGRATED_ACCEL_BIAS[0] * total_mass
            ty += _INTEGRATED_ACCEL_BIAS[1] * total_mass
            tz += _INTEGRATED_ACCEL_BIAS[2] * total_mass
            
            ax = tx / total_mass
            ay = ty / total_mass
            az = tz / total_mass

        x, y, z = _EST_POS
        vx, vy, vz = _EST_VEL if _EST_VEL is not None else [0.0, 0.0, 0.0]
        
        dx = vx * dt
        dy = vy * dt
        dz = vz * dt
        
        dvx = (3 * omega**2 * x + 2 * omega * vy + ax) * dt
        dvy = (-2 * omega * vx + ay) * dt
        dvz = (-omega**2 * z + az) * dt
        
        pos_prop = [x + dx, y + dy, z + dz]
        vel_prop = [vx + dvx, vy + dvy, vz + dvz]

        if pos_meas is not None:
            alpha_pos = 0.8
            _EST_POS = [pos_prop[i] + alpha_pos * (pos_meas[i] - pos_prop[i]) for i in range(3)]
        else:
            _EST_POS = pos_prop
            
        if vel_meas is not None:
            alpha_vel = 0.8
            _EST_VEL = [vel_prop[i] + alpha_vel * (vel_meas[i] - vel_prop[i]) for i in range(3)]
        else:
            _EST_VEL = vel_prop

        # Estimate acceleration bias
        if vel_meas is not None and _PREV_CONTROLS is not None and dt > 0.0:
            tx_cmd = _PREV_CONTROLS.get('thrust_x', 0.0) * 220.0 / total_mass
            ty_cmd = _PREV_CONTROLS.get('thrust_y', 0.0) * 220.0 / total_mass
            tz_cmd = _PREV_CONTROLS.get('thrust_z', 0.0) * 220.0 / total_mass
            
            dvx_grav = (3 * omega**2 * x + 2 * omega * vy) * dt
            dvy_grav = (-2 * omega * vx) * dt
            dvz_grav = (-omega**2 * z) * dt
            
            meas_dvx = vel_meas[0] - vx
            meas_dvy = vel_meas[1] - vy
            meas_dvz = vel_meas[2] - vz
            
            err_ax = (meas_dvx - dvx_grav) / dt - tx_cmd
            err_ay = (meas_dvy - dvy_grav) / dt - ty_cmd
            err_az = (meas_dvz - dvz_grav) / dt - tz_cmd
            
            bias_gain = 0.05
            _INTEGRATED_ACCEL_BIAS[0] += bias_gain * (err_ax - _INTEGRATED_ACCEL_BIAS[0])
            _INTEGRATED_ACCEL_BIAS[1] += bias_gain * (err_ay - _INTEGRATED_ACCEL_BIAS[1])
            _INTEGRATED_ACCEL_BIAS[2] += bias_gain * (err_az - _INTEGRATED_ACCEL_BIAS[2])
            
            max_bias_acc = 1.5 / total_mass
            for i in range(3):
                _INTEGRATED_ACCEL_BIAS[i] = max(-max_bias_acc, min(max_bias_acc, _INTEGRATED_ACCEL_BIAS[i]))

    if _EST_VEL is None:
        if vel_meas is not None:
            _EST_VEL = list(vel_meas)
        else:
            _EST_VEL = [0.0, 0.0, 0.0]

    _PREV_TIME = t
    return _EST_POS, _EST_VEL

def autopilot_step(state: dict, dt: float) -> dict:
    global _PREV_CONTROLS, _INTEGRATED_ACCEL_BIAS
    
    # 1. State Estimation
    pos, vel = estimate_relative_state(state, dt)
    x, y, z = pos
    vx, vy, vz = vel
    
    # Orbital rate
    sma = state.get('ground_semi_major_axis_m', 6779000.0) if state.get('ground_semi_major_axis_m_valid', False) else 6779000.0
    if sma < 6e6 or sma > 8e6:
        sma = 6779000.0
    omega = math.sqrt(3.986004418e14 / sma**3)
    
    # 2. Guidance Laws
    
    # Along-track guidance (y)
    abs_y = abs(y)
    if abs_y > 500.0:
        vy_cap = 1.5
    elif abs_y > 100.0:
        vy_cap = 0.8
    elif abs_y > 20.0:
        vy_cap = 0.3
    else:
        vy_cap = 0.12
        
    # Desired y velocity
    if y < -1.5:
        vy_des = max(0.12, min(vy_cap, -0.005 * y))
    elif y > 1.5:
        vy_des = min(-0.12, max(-vy_cap, -0.005 * y))
    else:
        vy_des = 0.0
        
    # Radial guidance (x)
    if abs_y >= 200.0:
        x_des = - vy_des / (1.5 * omega)
    else:
        if abs_y > 100.0:
            blend = (abs_y - 100.0) / 100.0
            x_des = blend * (- vy_des / (1.5 * omega))
        else:
            x_des = 0.0
            
    # Target radial velocity
    vx_des = - 0.01 * (x - x_des)
    vx_des = max(-0.5, min(0.5, vx_des))
    
    # Cross-track guidance (z)
    z_des = 0.0
    vz_des = 0.0
    
    # 3. Control Laws (Feedback Linearization)
    
    x_err = x - x_des
    x_err = max(-100.0, min(100.0, x_err))
    
    vx_err = vx - vx_des
    vx_err = max(-0.5, min(0.5, vx_err))
    
    vy_err = vy - vy_des
    vy_err = max(-0.5, min(0.5, vy_err))
    
    z_err = z - z_des
    z_err = max(-50.0, min(50.0, z_err))
    
    vz_err = vz - vz_des
    vz_err = max(-0.5, min(0.5, vz_err))
    
    acc_x_nom = - 0.0001 * x_err - 0.02 * vx_err
    acc_y_nom = - 0.02 * vy_err
    acc_z_nom = - 0.0001 * z_err - 0.02 * vz_err
    
    acc_x = acc_x_nom - 3 * omega**2 * x - 2 * omega * vy
    acc_y = acc_y_nom + 2 * omega * vx
    acc_z = acc_z_nom + omega**2 * z
    
    # 4. Thrust commands
    fuel = state.get('fuel_remaining_kg', 500.0)
    total_mass = 7000.0 + fuel
    
    thrust_x_nom = acc_x * total_mass
    thrust_y_nom = acc_y * total_mass
    thrust_z_nom = acc_z * total_mass
    
    bias_x = _INTEGRATED_ACCEL_BIAS[0] * total_mass
    bias_y = _INTEGRATED_ACCEL_BIAS[1] * total_mass
    bias_z = _INTEGRATED_ACCEL_BIAS[2] * total_mass
    
    thrust_x_cmd = thrust_x_nom - bias_x
    thrust_y_cmd = thrust_y_nom - bias_y
    thrust_z_cmd = thrust_z_nom - bias_z
    
    cmd_x = max(-1.0, min(1.0, thrust_x_cmd / 220.0))
    cmd_y = max(-1.0, min(1.0, thrust_y_cmd / 220.0))
    cmd_z = max(-1.0, min(1.0, thrust_z_cmd / 220.0))
    
    controls = {
        'thrust_x': cmd_x,
        'thrust_y': cmd_y,
        'thrust_z': cmd_z,
        'att_rate_roll': 0.0,
        'att_rate_pitch': 0.0,
        'att_rate_yaw': 0.0
    }
    
    _PREV_CONTROLS = controls
    return controls
```
