declare const require: any;
const fs = require('fs');

interface Package {
  id: string;
  x: number;
  y: number;
}

interface InputData {
  depot: [number, number];
  truckCount: number;
  packages: Package[];
}

function getDistance(p1: { x: number; y: number }, p2: { x: number; y: number }): number {
  const dx = p1.x - p2.x;
  const dy = p1.y - p2.y;
  return Math.sqrt(dx * dx + dy * dy);
}

function getRouteDistance(route: Package[], depotX: number, depotY: number): number {
  if (route.length === 0) return 0;
  let dist = 0;
  let currX = depotX;
  let currY = depotY;
  for (let i = 0; i < route.length; i++) {
    const p = route[i];
    const dx = p.x - currX;
    const dy = p.y - currY;
    dist += Math.sqrt(dx * dx + dy * dy);
    currX = p.x;
    currY = p.y;
  }
  const dx = depotX - currX;
  const dy = depotY - currY;
  dist += Math.sqrt(dx * dx + dy * dy);
  return dist;
}

function solveTSP(g: Package[], depotX: number, depotY: number): Package[] {
  const S = g.length;
  if (S === 0) return [];
  if (S === 1) return [g[0]];
  if (S === 2) return [g[0], g[1]];

  const points = [
    { x: depotX, y: depotY },
    ...g.map(p => ({ x: p.x, y: p.y }))
  ];
  const M = S + 1;
  const L = Math.min(20, S);

  const distMatrix: Float64Array | null = M <= 2000 ? new Float64Array(M * M) : null;
  if (distMatrix) {
    for (let i = 0; i < M; i++) {
      const pI = points[i];
      for (let j = 0; j < M; j++) {
        const pJ = points[j];
        const dx = pI.x - pJ.x;
        const dy = pI.y - pJ.y;
        distMatrix[i * M + j] = Math.sqrt(dx * dx + dy * dy);
      }
    }
  }

  const dist = (i: number, j: number): number => {
    if (distMatrix) {
      return distMatrix[i * M + j];
    }
    const pI = points[i];
    const pJ = points[j];
    const dx = pI.x - pJ.x;
    const dy = pI.y - pJ.y;
    return Math.sqrt(dx * dx + dy * dy);
  };

  const neighborList: number[][] = Array.from({ length: M }, () => []);
  if (M <= 100) {
    for (let i = 0; i < M; i++) {
      const candidates: { idx: number; d: number }[] = [];
      for (let j = 0; j < M; j++) {
        if (i !== j) {
          candidates.push({ idx: j, d: dist(i, j) });
        }
      }
      candidates.sort((a, b) => a.d - b.d);
      neighborList[i] = candidates.slice(0, L).map(c => c.idx);
    }
  } else {
    const minX = points.reduce((min, p) => p.x < min ? p.x : min, points[0].x);
    const maxX = points.reduce((max, p) => p.x > max ? p.x : max, points[0].x);
    const minY = points.reduce((min, p) => p.y < min ? p.y : min, points[0].y);
    const maxY = points.reduce((max, p) => p.y > max ? p.y : max, points[0].y);

    const B = Math.max(1, Math.floor(Math.sqrt(M / 10)));
    const cellWidth = maxX === minX ? 1.0 : (maxX - minX) / B;
    const cellHeight = maxY === minY ? 1.0 : (maxY - minY) / B;

    const grid: number[][][] = Array.from({ length: B }, () => Array.from({ length: B }, () => []));

    for (let i = 0; i < M; i++) {
      const cx = Math.min(B - 1, Math.floor((points[i].x - minX) / cellWidth));
      const cy = Math.min(B - 1, Math.floor((points[i].y - minY) / cellHeight));
      grid[cx][cy].push(i);
    }

    for (let i = 0; i < M; i++) {
      const cx = Math.min(B - 1, Math.floor((points[i].x - minX) / cellWidth));
      const cy = Math.min(B - 1, Math.floor((points[i].y - minY) / cellHeight));

      const candidates: number[] = [];
      let ring = 0;
      while (candidates.length < L + 1 && ring < B) {
        if (ring === 0) {
          candidates.push(...grid[cx][cy]);
        } else {
          for (let dx = -ring; dx <= ring; dx++) {
            const nx = cx + dx;
            if (nx >= 0 && nx < B) {
              const ny1 = cy - ring;
              if (ny1 >= 0 && ny1 < B) candidates.push(...grid[nx][ny1]);
              const ny2 = cy + ring;
              if (ny2 >= 0 && ny2 < B) candidates.push(...grid[nx][ny2]);
            }
          }
          for (let dy = -ring + 1; dy <= ring - 1; dy++) {
            const ny = cy + dy;
            if (ny >= 0 && ny < B) {
              const nx1 = cx - ring;
              if (nx1 >= 0 && nx1 < B) candidates.push(...grid[nx1][ny]);
              const nx2 = cx + ring;
              if (nx2 >= 0 && nx2 < B) candidates.push(...grid[nx2][ny]);
            }
          }
        }
        ring++;
      }

      const candWithDist = candidates
        .filter(idx => idx !== i)
        .map(idx => ({ idx, d: dist(i, idx) }));
      candWithDist.sort((a, b) => a.d - b.d);
      neighborList[i] = candWithDist.slice(0, L).map(c => c.idx);
    }
  }

  const tour: number[] = [0];
  const visited = new Uint8Array(M);
  visited[0] = 1;
  let curr = 0;
  for (let step = 1; step < M; step++) {
    let nextNode = -1;
    let minDist = Infinity;
    const neighbors = neighborList[curr];
    for (let k = 0; k < neighbors.length; k++) {
      const nb = neighbors[k];
      if (!visited[nb]) {
        nextNode = nb;
        break;
      }
    }
    if (nextNode === -1) {
      for (let j = 0; j < M; j++) {
        if (!visited[j]) {
          const d = dist(curr, j);
          if (d < minDist) {
            minDist = d;
            nextNode = j;
          }
        }
      }
    }
    tour.push(nextNode);
    visited[nextNode] = 1;
    curr = nextNode;
  }

  const pos = new Int32Array(M);
  for (let i = 0; i < M; i++) pos[tour[i]] = i;

  let improved = true;
  let pass = 0;
  const maxPasses = 100;
  while (improved && pass < maxPasses) {
    improved = false;
    pass++;
    for (let i = 0; i < M; i++) {
      const u = tour[i];
      const neighbors = neighborList[u];
      for (let k = 0; k < neighbors.length; k++) {
        const v = neighbors[k];
        const idx_v = pos[v];
        let i_swap = i;
        let j_swap = idx_v;
        if (i_swap > j_swap) {
          const temp = i_swap;
          i_swap = j_swap;
          j_swap = temp;
        }
        if (j_swap > i_swap + 1 && (j_swap !== M - 1 || i_swap !== 0)) {
          const a = tour[i_swap];
          const b = tour[i_swap + 1];
          const c = tour[j_swap];
          const d = tour[(j_swap + 1) % M];
          const delta = (dist(a, c) + dist(b, d)) - (dist(a, b) + dist(c, d));
          if (delta < -1e-9) {
            let start = i_swap + 1;
            let end = j_swap;
            while (start < end) {
              const xNode = tour[start];
              const yNode = tour[end];
              tour[start] = yNode;
              tour[end] = xNode;
              pos[yNode] = start;
              pos[xNode] = end;
              start++;
              end--;
            }
            improved = true;
          }
        }
      }
    }
  }

  const zeroIndex = tour.indexOf(0);
  const finalTour: number[] = [];
  if (zeroIndex === 0) {
    finalTour.push(...tour);
  } else {
    finalTour.push(...tour.slice(zeroIndex));
    finalTour.push(...tour.slice(0, zeroIndex));
  }

  return finalTour.slice(1).map(idx => g[idx - 1]);
}

function solve() {
  let input = '';
  try {
    input = fs.readFileSync(0, 'utf8');
  } catch (e) {
    return;
  }
  if (!input.trim()) {
    console.log(JSON.stringify({ routes: [] }));
    return;
  }
  let data: InputData;
  try {
    data = JSON.parse(input);
  } catch (e) {
    console.log(JSON.stringify({ routes: [] }));
    return;
  }

  const { depot, truckCount, packages } = data;
  if (!packages || packages.length === 0 || !truckCount || truckCount <= 0) {
    console.log(JSON.stringify({ routes: [] }));
    return;
  }

  const K = Math.min(truckCount, packages.length);
  const [depotX, depotY] = depot;
  const N = packages.length;

  let numCandidates = 1;
  if (N < 100) {
    numCandidates = 16;
  } else if (N < 1000) {
    numCandidates = 8;
  } else if (N < 5000) {
    numCandidates = 4;
  } else {
    numCandidates = 1;
  }

  const initialPackages = packages.map(p => {
    const angle = Math.atan2(p.y - depotY, p.x - depotX);
    return { p, angle };
  });

  initialPackages.sort((a, b) => a.angle - b.angle);

  let bestRoutes: Package[][] = [];
  let bestTotalDistance = Infinity;

  for (let cand = 0; cand < numCandidates; cand++) {
    const startIdx = Math.floor((cand * N) / numCandidates);
    const theta0 = initialPackages[startIdx].angle;

    const shiftedPackages = initialPackages.map(item => {
      let shifted = item.angle - theta0;
      if (shifted < -Math.PI) shifted += 2 * Math.PI;
      if (shifted > Math.PI) shifted -= 2 * Math.PI;
      return { p: item.p, shifted };
    });

    shiftedPackages.sort((a, b) => a.shifted - b.shifted);

    // Strategy 1: Equal partition
    const groupsEqual: Package[][] = Array.from({ length: K }, () => []);
    for (let i = 0; i < N; i++) {
      const groupIndex = Math.min(K - 1, Math.floor((i * K) / N));
      groupsEqual[groupIndex].push(shiftedPackages[i].p);
    }

    // Strategy 2: Gap partition
    let groupsGap: Package[][] | null = null;
    if (K > 1 && N > K) {
      const gaps: { index: number; size: number }[] = [];
      for (let i = 0; i < N - 1; i++) {
        const size = shiftedPackages[i + 1].shifted - shiftedPackages[i].shifted;
        gaps.push({ index: i, size });
      }
      gaps.sort((a, b) => b.size - a.size);
      const splitIndices = gaps.slice(0, K - 1).map(g => g.index);
      splitIndices.sort((a, b) => a - b);

      groupsGap = [];
      let lastSplit = -1;
      for (let k = 0; k < K - 1; k++) {
        const split = splitIndices[k];
        const group: Package[] = [];
        for (let i = lastSplit + 1; i <= split; i++) {
          group.push(shiftedPackages[i].p);
        }
        groupsGap.push(group);
        lastSplit = split;
      }
      const lastGroup: Package[] = [];
      for (let i = lastSplit + 1; i < N; i++) {
        lastGroup.push(shiftedPackages[i].p);
      }
      groupsGap.push(lastGroup);
    }

    const strategies = [groupsEqual];
    if (groupsGap) strategies.push(groupsGap);

    for (const groups of strategies) {
      const currentRoutes: Package[][] = [];
      let currentTotalDistance = 0;

      for (let gIdx = 0; gIdx < K; gIdx++) {
        const g = groups[gIdx];
        const tour = solveTSP(g, depotX, depotY);
        currentRoutes.push(tour);
        currentTotalDistance += getRouteDistance(tour, depotX, depotY);
      }

      if (currentTotalDistance < bestTotalDistance) {
        bestTotalDistance = currentTotalDistance;
        bestRoutes = currentRoutes;
      }
    }
  }

  const routes = bestRoutes.map(r => r.map(p => p.id));
  console.log(JSON.stringify({ routes }));
}

solve();
