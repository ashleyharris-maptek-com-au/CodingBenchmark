declare const require: any;
const fs = require('fs');

interface Robot {
  id: number;
  x: number;
  y: number;
}

interface Job {
  id: string | number;
  x: number;
  y: number;
  priority: number;
}

interface InputData {
  grid: string[];
  robots: Robot[];
  jobs: Job[];
}

function solve() {
  const input: string = fs.readFileSync(0, 'utf8');
  if (!input.trim()) {
    return;
  }
  const data: InputData = JSON.parse(input);
  const grid = data.grid;
  const robots = data.robots;
  const jobs = data.jobs;

  const H_orig = grid.length;
  if (H_orig === 0) {
    console.log(JSON.stringify({ assignments: [] }));
    return;
  }
  const W_orig = grid[0].length;
  if (W_orig === 0) {
    console.log(JSON.stringify({ assignments: [] }));
    return;
  }

  const H = H_orig + 2;
  const W = W_orig + 2;
  const size = W * H;
  const flatGrid = new Uint8Array(size);
  flatGrid.fill(35); // '#' is 35
  for (let y = 0; y < H_orig; y++) {
    const row = grid[y];
    for (let x = 0; x < W_orig; x++) {
      flatGrid[(y + 1) * W + (x + 1)] = row.charCodeAt(x);
    }
  }

  const comp = new Int32Array(size);
  const queue = new Int32Array(size);

  // Precompute connected components
  let nextCompId = 1;
  for (let i = 0; i < size; i++) {
    if (flatGrid[i] !== 35 && comp[i] === 0) {
      let head = 0;
      let tail = 0;
      queue[tail++] = i;
      comp[i] = nextCompId;
      while (head < tail) {
        const curr = queue[head++];
        let next = curr - 1;
        if (flatGrid[next] !== 35 && comp[next] === 0) {
          comp[next] = nextCompId;
          queue[tail++] = next;
        }
        next = curr + 1;
        if (flatGrid[next] !== 35 && comp[next] === 0) {
          comp[next] = nextCompId;
          queue[tail++] = next;
        }
        next = curr - W;
        if (flatGrid[next] !== 35 && comp[next] === 0) {
          comp[next] = nextCompId;
          queue[tail++] = next;
        }
        next = curr + W;
        if (flatGrid[next] !== 35 && comp[next] === 0) {
          comp[next] = nextCompId;
          queue[tail++] = next;
        }
      }
      nextCompId++;
    }
  }

  const numRobots = robots.length;
  const numJobs = jobs.length;
  const numPoints = numRobots + numJobs;

  const gridIdx = new Int32Array(numPoints);
  for (let r = 0; r < numRobots; r++) {
    gridIdx[r] = (robots[r].y + 1) * W + (robots[r].x + 1);
  }
  for (let j = 0; j < numJobs; j++) {
    gridIdx[numRobots + j] = (jobs[j].y + 1) * W + (jobs[j].x + 1);
  }

  // Common structures for BFS in algorithms
  const visited = new Int32Array(size);
  const dist = new Int32Array(size);
  let currentGen = 0;

  // Find maximum robot ID to size assignments array correctly
  const maxRobotId = numRobots > 0 ? Math.max(...robots.map(r => r.id)) : 0;

  // Helper for Greedy 1: Nearest Robot to Job (Priority Decending)
  function runGreedy1(): { assignments: (string | number)[][], cost: number } {
    const currentRobotPosPoint = new Int32Array(numRobots);
    for (let r = 0; r < numRobots; r++) {
      currentRobotPosPoint[r] = r;
    }
    const assignments: (string | number)[][] = Array.from({ length: maxRobotId + 1 }, () => []);
    let totalDist = 0;

    const jobsSorted = jobs.map((job, idx) => ({ job, idx }))
      .sort((a, b) => {
        if (b.job.priority !== a.job.priority) return b.job.priority - a.job.priority;
        return a.job.id < b.job.id ? -1 : (a.job.id > b.job.id ? 1 : 0);
      });

    const robotIdxAtGridIdxLocal = new Int32Array(size);

    for (const { job, idx } of jobsSorted) {
      const jobPoint = numRobots + idx;
      const startIdx = gridIdx[jobPoint];
      const startComp = comp[startIdx];
      if (startComp === 0) {
        assignments[robots[0].id].push(job.id);
        continue;
      }

      const robotsInComp: number[] = [];
      for (let r = 0; r < numRobots; r++) {
        if (comp[gridIdx[currentRobotPosPoint[r]]] === startComp) {
          robotsInComp.push(r);
        }
      }

      if (robotsInComp.length === 0) {
        assignments[robots[0].id].push(job.id);
        continue;
      }

      robotIdxAtGridIdxLocal.fill(-1);
      for (const r of robotsInComp) {
        robotIdxAtGridIdxLocal[gridIdx[currentRobotPosPoint[r]]] = r;
      }

      currentGen++;
      let head = 0, tail = 0;
      queue[tail++] = startIdx;
      visited[startIdx] = currentGen;
      dist[startIdx] = 0;
      let bestRobot = -1, bestRobotDist = 1e9;

      while (head < tail) {
        const curr = queue[head++];
        const d = dist[curr];
        if (d > bestRobotDist) break;
        const r = robotIdxAtGridIdxLocal[curr];
        if (r !== -1) {
          if (bestRobot === -1 || r < bestRobot) {
            bestRobot = r;
            bestRobotDist = d;
          }
        }
        const dNext = d + 1;
        let next = curr - 1;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
        next = curr + 1;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
        next = curr - W;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
        next = curr + W;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
      }

      if (bestRobot !== -1 && bestRobotDist < 1e9) {
        assignments[robots[bestRobot].id].push(job.id);
        totalDist += bestRobotDist;
        currentRobotPosPoint[bestRobot] = jobPoint;
      } else {
        assignments[robots[0].id].push(job.id);
      }
    }

    return { assignments, cost: totalDist };
  }

  // Helper for Greedy 2: Closest Job to Any Robot
  function runGreedy2(): { assignments: (string | number)[][], cost: number } {
    const unassigned = new Uint8Array(numJobs);
    unassigned.fill(1);
    let unassignedCount = numJobs;
    const currentRobotPosPoint = new Int32Array(numRobots);
    for (let r = 0; r < numRobots; r++) {
      currentRobotPosPoint[r] = r;
    }
    const assignments: (string | number)[][] = Array.from({ length: maxRobotId + 1 }, () => []);
    let totalDist = 0;

    const closestJob = new Int32Array(numRobots);
    const closestDist = new Int32Array(numRobots);
    const isCacheValid = new Uint8Array(numRobots);
    closestJob.fill(-1);
    closestDist.fill(1e9);
    isCacheValid.fill(0);

    const jobIdxAtGridIdxLocal = new Int32Array(size);

    function findClosestUnassignedJob(robotIdx: number) {
      const startPoint = currentRobotPosPoint[robotIdx];
      const startIdx = gridIdx[startPoint];
      const startComp = comp[startIdx];
      if (startComp === 0) {
        closestJob[robotIdx] = -1;
        closestDist[robotIdx] = 1e9;
        isCacheValid[robotIdx] = 1;
        return;
      }

      let hasJobsInComp = false;
      for (let j = 0; j < numJobs; j++) {
        if (unassigned[j] && comp[gridIdx[numRobots + j]] === startComp) {
          hasJobsInComp = true;
          break;
        }
      }
      if (!hasJobsInComp) {
        closestJob[robotIdx] = -1;
        closestDist[robotIdx] = 1e9;
        isCacheValid[robotIdx] = 1;
        return;
      }

      jobIdxAtGridIdxLocal.fill(-1);
      for (let j = 0; j < numJobs; j++) {
        if (unassigned[j] && comp[gridIdx[numRobots + j]] === startComp) {
          jobIdxAtGridIdxLocal[gridIdx[numRobots + j]] = j;
        }
      }

      currentGen++;
      let head = 0, tail = 0;
      queue[tail++] = startIdx;
      visited[startIdx] = currentGen;
      dist[startIdx] = 0;
      let bestJobIdx = -1, bestJobDist = 1e9;

      while (head < tail) {
        const curr = queue[head++];
        const d = dist[curr];
        if (d > bestJobDist) break;
        const jobIdx = jobIdxAtGridIdxLocal[curr];
        if (jobIdx !== -1) {
          if (bestJobIdx === -1) {
            bestJobIdx = jobIdx;
            bestJobDist = d;
          } else {
            const job1 = jobs[jobIdx];
            const job2 = jobs[bestJobIdx];
            if (job1.priority > job2.priority) {
              bestJobIdx = jobIdx;
            } else if (job1.priority === job2.priority) {
              if (job1.id < job2.id) bestJobIdx = jobIdx;
            }
          }
        }
        const dNext = d + 1;
        let next = curr - 1;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
        next = curr + 1;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
        next = curr - W;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
        next = curr + W;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
      }

      closestJob[robotIdx] = bestJobIdx;
      closestDist[robotIdx] = bestJobDist;
      isCacheValid[robotIdx] = 1;
    }

    while (unassignedCount > 0) {
      for (let r = 0; r < numRobots; r++) {
        if (!isCacheValid[r]) findClosestUnassignedJob(r);
      }

      let bestRobot = -1, bestJobIdx = -1, bestD = 1e9;
      for (let r = 0; r < numRobots; r++) {
        const d = closestDist[r];
        const j = closestJob[r];
        if (j !== -1 && d < bestD) {
          bestD = d; bestRobot = r; bestJobIdx = j;
        } else if (j !== -1 && d === bestD) {
          const job1 = jobs[j];
          const job2 = jobs[bestJobIdx];
          if (job1.priority > job2.priority) {
            bestRobot = r; bestJobIdx = j;
          } else if (job1.priority === job2.priority) {
            if (job1.id < job2.id) {
              bestRobot = r; bestJobIdx = j;
            } else if (job1.id === job2.id) {
              if (r < bestRobot) {
                bestRobot = r; bestJobIdx = j;
              }
            }
          }
        }
      }

      if (bestRobot === -1) {
        for (let j = 0; j < numJobs; j++) {
          if (unassigned[j]) {
            assignments[robots[0].id].push(jobs[j].id);
            unassigned[j] = 0;
            unassignedCount--;
          }
        }
        break;
      }

      assignments[robots[bestRobot].id].push(jobs[bestJobIdx].id);
      totalDist += bestD;
      unassigned[bestJobIdx] = 0;
      unassignedCount--;

      currentRobotPosPoint[bestRobot] = numRobots + bestJobIdx;
      isCacheValid[bestRobot] = 0;

      for (let r = 0; r < numRobots; r++) {
        if (closestJob[r] === bestJobIdx) isCacheValid[r] = 0;
      }
    }

    return { assignments, cost: totalDist };
  }

  // Helper for Greedy 3: Nearest Robot with Load-balancing (Priority Decending)
  function runGreedy3(): { assignments: (string | number)[][], cost: number } {
    const currentRobotPosPoint = new Int32Array(numRobots);
    const travelDist = new Int32Array(numRobots);
    for (let r = 0; r < numRobots; r++) {
      currentRobotPosPoint[r] = r;
    }
    const assignments: (string | number)[][] = Array.from({ length: maxRobotId + 1 }, () => []);
    let totalDist = 0;

    const jobsSorted = jobs.map((job, idx) => ({ job, idx }))
      .sort((a, b) => {
        if (b.job.priority !== a.job.priority) return b.job.priority - a.job.priority;
        return a.job.id < b.job.id ? -1 : (a.job.id > b.job.id ? 1 : 0);
      });

    const robotIdxAtGridIdxLocal = new Int32Array(size);
    const reached = new Uint8Array(numRobots);

    function findClosestRobotG3(jobIdx: number) {
      const startPoint = numRobots + jobIdx;
      const startIdx = gridIdx[startPoint];
      const startComp = comp[startIdx];
      if (startComp === 0) return { robotIdx: -1, dist: 1e9 };

      const robotsInComp: number[] = [];
      for (let r = 0; r < numRobots; r++) {
        if (comp[gridIdx[currentRobotPosPoint[r]]] === startComp) {
          robotsInComp.push(r);
        }
      }
      if (robotsInComp.length === 0) return { robotIdx: -1, dist: 1e9 };

      robotIdxAtGridIdxLocal.fill(-1);
      for (const r of robotsInComp) {
        robotIdxAtGridIdxLocal[gridIdx[currentRobotPosPoint[r]]] = r;
      }

      reached.fill(0);
      let minTravelSoFar = 1e9;
      for (const r of robotsInComp) {
        if (travelDist[r] < minTravelSoFar) minTravelSoFar = travelDist[r];
      }

      currentGen++;
      let head = 0, tail = 0;
      queue[tail++] = startIdx;
      visited[startIdx] = currentGen;
      dist[startIdx] = 0;
      let bestRobot = -1, bestCost = 1e9, bestRobotDist = 1e9;

      while (head < tail) {
        const curr = queue[head++];
        const d = dist[curr];
        if (d + minTravelSoFar >= bestCost) break;

        const r = robotIdxAtGridIdxLocal[curr];
        if (r !== -1) {
          reached[r] = 1;
          const cost = travelDist[r] + d;
          if (cost < bestCost || (cost === bestCost && r < bestRobot)) {
            bestCost = cost;
            bestRobot = r;
            bestRobotDist = d;
          }
          minTravelSoFar = 1e9;
          for (const ro of robotsInComp) {
            if (!reached[ro]) {
              if (travelDist[ro] < minTravelSoFar) minTravelSoFar = travelDist[ro];
            }
          }
        }

        const dNext = d + 1;
        let next = curr - 1;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
        next = curr + 1;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
        next = curr - W;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
        next = curr + W;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
      }
      return { robotIdx: bestRobot, dist: bestRobotDist };
    }

    for (const { job, idx } of jobsSorted) {
      const { robotIdx, dist } = findClosestRobotG3(idx);
      if (robotIdx !== -1 && dist < 1e9) {
        assignments[robots[robotIdx].id].push(job.id);
        totalDist += dist;
        travelDist[robotIdx] += dist;
        currentRobotPosPoint[robotIdx] = numRobots + idx;
      } else {
        assignments[robots[0].id].push(job.id);
      }
    }

    return { assignments, cost: totalDist };
  }

  // Helper for Greedy 4: Global Closest with Load-balancing
  function runGreedy4(): { assignments: (string | number)[][], cost: number } {
    const unassigned = new Uint8Array(numJobs);
    unassigned.fill(1);
    let unassignedCount = numJobs;
    const currentRobotPosPoint = new Int32Array(numRobots);
    const travelDist = new Int32Array(numRobots);
    for (let r = 0; r < numRobots; r++) {
      currentRobotPosPoint[r] = r;
    }
    const assignments: (string | number)[][] = Array.from({ length: maxRobotId + 1 }, () => []);
    let totalDist = 0;

    const closestJob = new Int32Array(numRobots);
    const closestDist = new Int32Array(numRobots);
    const isCacheValid = new Uint8Array(numRobots);
    closestJob.fill(-1);
    closestDist.fill(1e9);
    isCacheValid.fill(0);

    const jobIdxAtGridIdxLocal = new Int32Array(size);

    function findClosestUnassignedJob(robotIdx: number) {
      const startPoint = currentRobotPosPoint[robotIdx];
      const startIdx = gridIdx[startPoint];
      const startComp = comp[startIdx];
      if (startComp === 0) {
        closestJob[robotIdx] = -1;
        closestDist[robotIdx] = 1e9;
        isCacheValid[robotIdx] = 1;
        return;
      }

      let hasJobsInComp = false;
      for (let j = 0; j < numJobs; j++) {
        if (unassigned[j] && comp[gridIdx[numRobots + j]] === startComp) {
          hasJobsInComp = true;
          break;
        }
      }
      if (!hasJobsInComp) {
        closestJob[robotIdx] = -1;
        closestDist[robotIdx] = 1e9;
        isCacheValid[robotIdx] = 1;
        return;
      }

      jobIdxAtGridIdxLocal.fill(-1);
      for (let j = 0; j < numJobs; j++) {
        if (unassigned[j] && comp[gridIdx[numRobots + j]] === startComp) {
          jobIdxAtGridIdxLocal[gridIdx[numRobots + j]] = j;
        }
      }

      currentGen++;
      let head = 0, tail = 0;
      queue[tail++] = startIdx;
      visited[startIdx] = currentGen;
      dist[startIdx] = 0;
      let bestJobIdx = -1, bestJobDist = 1e9;

      while (head < tail) {
        const curr = queue[head++];
        const d = dist[curr];
        if (d > bestJobDist) break;
        const jobIdx = jobIdxAtGridIdxLocal[curr];
        if (jobIdx !== -1) {
          if (bestJobIdx === -1) {
            bestJobIdx = jobIdx;
            bestJobDist = d;
          } else {
            const job1 = jobs[jobIdx];
            const job2 = jobs[bestJobIdx];
            if (job1.priority > job2.priority) {
              bestJobIdx = jobIdx;
            } else if (job1.priority === job2.priority) {
              if (job1.id < job2.id) bestJobIdx = jobIdx;
            }
          }
        }
        const dNext = d + 1;
        let next = curr - 1;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
        next = curr + 1;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
        next = curr - W;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
        next = curr + W;
        if (flatGrid[next] !== 35 && visited[next] !== currentGen) {
          visited[next] = currentGen; dist[next] = dNext; queue[tail++] = next;
        }
      }

      closestJob[robotIdx] = bestJobIdx;
      closestDist[robotIdx] = bestJobDist;
      isCacheValid[robotIdx] = 1;
    }

    while (unassignedCount > 0) {
      for (let r = 0; r < numRobots; r++) {
        if (!isCacheValid[r]) findClosestUnassignedJob(r);
      }

      let bestRobot = -1, bestJobIdx = -1, bestCost = 1e9, bestD = 1e9;
      for (let r = 0; r < numRobots; r++) {
        const d = closestDist[r];
        const j = closestJob[r];
        if (j !== -1) {
          const cost = travelDist[r] + d;
          if (cost < bestCost) {
            bestCost = cost; bestD = d; bestRobot = r; bestJobIdx = j;
          } else if (cost === bestCost) {
            const job1 = jobs[j];
            const job2 = jobs[bestJobIdx];
            if (job1.priority > job2.priority) {
              bestRobot = r; bestJobIdx = j; bestD = d;
            } else if (job1.priority === job2.priority) {
              if (job1.id < job2.id) {
                bestRobot = r; bestJobIdx = j; bestD = d;
              } else if (job1.id === job2.id) {
                if (r < bestRobot) {
                  bestRobot = r; bestJobIdx = j; bestD = d;
                }
              }
            }
          }
        }
      }

      if (bestRobot === -1) {
        for (let j = 0; j < numJobs; j++) {
          if (unassigned[j]) {
            assignments[robots[0].id].push(jobs[j].id);
            unassigned[j] = 0;
            unassignedCount--;
          }
        }
        break;
      }

      assignments[robots[bestRobot].id].push(jobs[bestJobIdx].id);
      totalDist += bestD;
      travelDist[bestRobot] += bestD;
      unassigned[bestJobIdx] = 0;
      unassignedCount--;

      currentRobotPosPoint[bestRobot] = numRobots + bestJobIdx;
      isCacheValid[bestRobot] = 0;

      for (let r = 0; r < numRobots; r++) {
        if (closestJob[r] === bestJobIdx) isCacheValid[r] = 0;
      }
    }

    return { assignments, cost: totalDist };
  }

  // Time-bounded selection of the best greedy assignment
  const { performance } = require('perf_hooks');
  const startTime = performance.now();

  let bestAssignment: (string | number)[][] = [];
  let bestCost = Infinity;

  // Run Greedy 2 (usually best and very fast)
  {
    const res = runGreedy2();
    if (res.cost < bestCost) {
      bestCost = res.cost;
      bestAssignment = res.assignments;
    }
  }

  // Run Greedy 4
  if (performance.now() - startTime < 1500) {
    const res = runGreedy4();
    if (res.cost < bestCost) {
      bestCost = res.cost;
      bestAssignment = res.assignments;
    }
  }

  // Run Greedy 1
  if (performance.now() - startTime < 2500) {
    const res = runGreedy1();
    if (res.cost < bestCost) {
      bestCost = res.cost;
      bestAssignment = res.assignments;
    }
  }

  // Run Greedy 3
  if (performance.now() - startTime < 3500) {
    const res = runGreedy3();
    if (res.cost < bestCost) {
      bestCost = res.cost;
      bestAssignment = res.assignments;
    }
  }

  console.log(JSON.stringify({ assignments: bestAssignment }));
}

solve();
