using System;
using System.IO;
using System.Text;
using System.Collections.Generic;

class Program
{
    static int H, W;
    static byte[] grid;
    static byte[] mineNeed;
    static byte[] unknownCount;
    static List<int> activeCells = new List<int>();
    static Queue<int> changedCells = new Queue<int>();
    static List<int> safeCells = new List<int>();

    // Unsafe pointers to avoid bounds checking and optimize loops
    static unsafe byte* g;
    static unsafe byte* mn;
    static unsafe byte* uc;

    // Arrays for BFS visited
    static bool[] visited;
    static bool[] visitedR;

    // Temporary buffers to avoid allocations in inner loops
    static int[] bufA = new int[8];
    static int[] bufB = new int[8];
    static int[] bufR = new int[8];
    static int[] bufU = new int[8];
    static int[] checkedBs = new int[64];

    // Static buffers for overlap analysis to avoid allocations
    static int[] S = new int[8];
    static int[] APrime = new int[8];
    static int[] BPrime = new int[8];

    // Backtracking global variables
    static int[] backtrackAssignment;
    static int[] constrCount;
    static int[] constrSum;
    static int[] safeCount;
    static int[] mineCount;
    static int totalValidAssignments;
    static int stateCount;
    static int MaxStates;
    static int[][] constraintVars;
    static int[] constraintTargets;
    static List<int>[] varConstraints;

    struct StrategyParameters
    {
        public int MaxCompSize;
        public int MaxStates;
        public bool EnablePairwise;
        public bool EnableBacktrack;
    }

    static unsafe void Main()
    {
        // Use a buffered stream reader for fast input reading
        using var reader = new StreamReader(Console.OpenStandardInput(), Encoding.UTF8, false, 1048576);
        string firstLine = reader.ReadLine();
        if (string.IsNullOrEmpty(firstLine)) return;

        string[] parts = firstLine.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        if (parts.Length < 2) return;
        H = int.Parse(parts[0]);
        W = int.Parse(parts[1]);

        grid = new byte[H * W];
        for (int r = 0; r < H; r++)
        {
            string line = reader.ReadLine();
            if (line == null) break;
            for (int c = 0; c < W; c++)
            {
                char ch = line[c];
                int idx = r * W + c;
                if (ch == ' ') grid[idx] = 9; // Unknown
                else if (ch == '*') grid[idx] = 10; // Mine
                else if (ch >= '0' && ch <= '8') grid[idx] = (byte)(ch - '0'); // Revealed number
                else grid[idx] = 9;
            }
        }

        // Set adaptive parameters based on board size (complexity)
        long area = (long)H * W;
        StrategyParameters param = new StrategyParameters();
        if (area <= 1000) // Trivial
        {
            param.MaxCompSize = 35;
            param.MaxStates = 100000;
            param.EnablePairwise = true;
            param.EnableBacktrack = true;
        }
        else if (area <= 100000) // Small/Medium
        {
            param.MaxCompSize = 25;
            param.MaxStates = 10000;
            param.EnablePairwise = true;
            param.EnableBacktrack = true;
        }
        else if (area <= 2000000) // Large
        {
            param.MaxCompSize = 15;
            param.MaxStates = 2000;
            param.EnablePairwise = true;
            param.EnableBacktrack = true;
        }
        else // Massive / Ludicrous
        {
            param.MaxCompSize = 10;
            param.MaxStates = 500;
            param.EnablePairwise = true;
            param.EnableBacktrack = true;
        }

        mineNeed = new byte[H * W];
        unknownCount = new byte[H * W];
        activeCells = new List<int>(Math.Min((int)area, 1000000));

        fixed (byte* pGrid = grid)
        fixed (byte* pMineNeed = mineNeed)
        fixed (byte* pUnknownCount = unknownCount)
        {
            g = pGrid;
            mn = pMineNeed;
            uc = pUnknownCount;

            // 1. Process interior (r from 1 to H-2, c from 1 to W-2) to avoid boundary checks
            for (int r = 1; r < H - 1; r++)
            {
                int rW = r * W;
                int rPrevW = rW - W;
                int rNextW = rW + W;
                for (int c = 1; c < W - 1; c++)
                {
                    int idx = rW + c;
                    byte val = g[idx];
                    if (val <= 8)
                    {
                        int mCount = 0;
                        int uCount = 0;
                        byte n;

                        // Row r-1
                        n = g[rPrevW + c - 1]; if (n == 10) mCount++; else if (n == 9) uCount++;
                        n = g[rPrevW + c];     if (n == 10) mCount++; else if (n == 9) uCount++;
                        n = g[rPrevW + c + 1]; if (n == 10) mCount++; else if (n == 9) uCount++;

                        // Row r
                        n = g[rW + c - 1];     if (n == 10) mCount++; else if (n == 9) uCount++;
                        n = g[rW + c + 1];     if (n == 10) mCount++; else if (n == 9) uCount++;

                        // Row r+1
                        n = g[rNextW + c - 1]; if (n == 10) mCount++; else if (n == 9) uCount++;
                        n = g[rNextW + c];     if (n == 10) mCount++; else if (n == 9) uCount++;
                        n = g[rNextW + c + 1]; if (n == 10) mCount++; else if (n == 9) uCount++;

                        mn[idx] = (byte)Math.Max(0, val - mCount);
                        uc[idx] = (byte)uCount;
                        if (uCount > 0)
                        {
                            activeCells.Add(idx);
                        }
                    }
                }
            }

            // 2. Process boundary rows (r = 0 and r = H-1)
            ProcessRowBoundary(0);
            ProcessRowBoundary(H - 1);

            // 3. Process boundary columns (c = 0 and c = W-1) for interior rows
            for (int r = 1; r < H - 1; r++)
            {
                ProcessCellBoundary(r, 0);
                ProcessCellBoundary(r, W - 1);
            }

            // First pass: local rules
            for (int i = 0; i < activeCells.Count; i++)
            {
                int idx = activeCells[i];
                if (mn[idx] == 0)
                {
                    MarkUnknownNeighbors(idx, false);
                }
                else if (mn[idx] == uc[idx])
                {
                    MarkUnknownNeighbors(idx, true);
                }
            }
            PropagateQueue();

            bool mainProgress = true;
            while (mainProgress)
            {
                mainProgress = false;

                // 1. Pairwise analysis
                if (param.EnablePairwise)
                {
                    bool anyNewInPass = true;
                    while (anyNewInPass)
                    {
                        anyNewInPass = false;
                        FilterActiveCells();

                        for (int i = 0; i < activeCells.Count; i++)
                        {
                            int A = activeCells[i];
                            if (uc[A] == 0) continue;

                            int checkedCount = 0;
                            int lenA = GetUnknownNeighbors(A, bufA);

                            for (int ui = 0; ui < lenA; ui++)
                            {
                                int U = bufA[ui];
                                int lenU = GetRevealedNeighbors(U, bufB);
                                for (int rj = 0; rj < lenU; rj++)
                                {
                                    int B = bufB[rj];
                                    if (B <= A || uc[B] == 0) continue;

                                    bool alreadyChecked = false;
                                    for (int k = 0; k < checkedCount; k++)
                                    {
                                        if (checkedBs[k] == B)
                                        {
                                            alreadyChecked = true;
                                            break;
                                        }
                                    }
                                    if (alreadyChecked) continue;
                                    if (checkedCount < 64)
                                    {
                                        checkedBs[checkedCount++] = B;
                                    }

                                    bool found = AnalyzePair(A, B);
                                    if (found)
                                    {
                                        anyNewInPass = true;
                                        mainProgress = true;
                                        PropagateQueue();
                                        if (uc[A] == 0) break;
                                        lenA = GetUnknownNeighbors(A, bufA);
                                        ui = -1; // restart search for updated A
                                        break;
                                    }
                                }
                                if (uc[A] == 0) break;
                            }
                        }
                    }
                }

                if (mainProgress) continue;

                // 2. Backtracking pass
                if (param.EnableBacktrack)
                {
                    visited = new bool[H * W];
                    visitedR = new bool[H * W];

                    List<List<int>> componentsU = new List<List<int>>();
                    List<List<int>> componentsR = new List<List<int>>();

                    for (int idx = 0; idx < H * W; idx++)
                    {
                        if (g[idx] == 9 && !visited[idx])
                        {
                            if (!HasRevealedNeighbors(idx)) continue;

                            List<int> compU = new List<int>();
                            List<int> compR = new List<int>();
                            Queue<int> queue = new Queue<int>();

                            queue.Enqueue(idx);
                            visited[idx] = true;

                            while (queue.Count > 0)
                            {
                                int curr = queue.Dequeue();
                                compU.Add(curr);

                                int lenR = GetRevealedNeighbors(curr, bufR);
                                for (int i = 0; i < lenR; i++)
                                {
                                    int r = bufR[i];
                                    if (!visitedR[r])
                                    {
                                        visitedR[r] = true;
                                        compR.Add(r);

                                        int lenU = GetUnknownNeighbors(r, bufU);
                                        for (int j = 0; j < lenU; j++)
                                        {
                                            int u = bufU[j];
                                            if (!visited[u])
                                            {
                                                visited[u] = true;
                                                queue.Enqueue(u);
                                            }
                                        }
                                    }
                                }
                            }

                            for (int i = 0; i < compR.Count; i++)
                            {
                                visitedR[compR[i]] = false;
                            }

                            componentsU.Add(compU);
                            componentsR.Add(compR);
                        }
                    }

                    bool backtrackProgress = false;
                    for (int i = 0; i < componentsU.Count; i++)
                    {
                        List<int> compU = componentsU[i];
                        List<int> compR = componentsR[i];
                        if (compU.Count <= param.MaxCompSize)
                        {
                            int prevChangedCount = changedCells.Count;

                            ProcessComponent(compU, compR, param.MaxStates);

                            if (changedCells.Count > prevChangedCount)
                            {
                                PropagateQueue();
                                backtrackProgress = true;
                                mainProgress = true;
                                break;
                            }
                        }
                    }

                    visited = null;
                    visitedR = null;

                    if (backtrackProgress) continue;
                }
            }
        }

        // Print output
        Console.WriteLine(safeCells.Count);
        for (int i = 0; i < safeCells.Count; i++)
        {
            int idx = safeCells[i];
            int y = idx / W;
            int x = idx % W;
            Console.WriteLine(x + " " + y);
        }
    }

    static unsafe void ProcessRowBoundary(int r)
    {
        for (int c = 0; c < W; c++)
        {
            ProcessCellBoundary(r, c);
        }
    }

    static unsafe void ProcessCellBoundary(int r, int c)
    {
        int idx = r * W + c;
        byte val = g[idx];
        if (val <= 8)
        {
            int mCount = 0;
            int uCount = 0;
            for (int dy = -1; dy <= 1; dy++)
            {
                int ny = r + dy;
                if (ny < 0 || ny >= H) continue;
                int nyW = ny * W;
                for (int dx = -1; dx <= 1; dx++)
                {
                    if (dx == 0 && dy == 0) continue;
                    int nx = c + dx;
                    if (nx < 0 || nx >= W) continue;
                    int nidx = nyW + nx;
                    byte nval = g[nidx];
                    if (nval == 10) mCount++;
                    else if (nval == 9) uCount++;
                }
            }
            mn[idx] = (byte)Math.Max(0, val - mCount);
            uc[idx] = (byte)uCount;
            if (uCount > 0)
            {
                activeCells.Add(idx);
            }
        }
    }

    static unsafe void MarkCell(int idx, bool isMine)
    {
        if (g[idx] != 9) return;
        g[idx] = (byte)(isMine ? 10 : 11);
        if (!isMine)
        {
            safeCells.Add(idx);
        }
        changedCells.Enqueue(idx);
    }

    static unsafe void MarkUnknownNeighbors(int idx, bool isMine)
    {
        int cy = idx / W;
        int cx = idx % W;
        for (int dy = -1; dy <= 1; dy++)
        {
            int ny = cy + dy;
            if (ny < 0 || ny >= H) continue;
            int nyW = ny * W;
            for (int dx = -1; dx <= 1; dx++)
            {
                if (dx == 0 && dy == 0) continue;
                int nx = cx + dx;
                if (nx < 0 || nx >= W) continue;
                int nidx = nyW + nx;
                if (g[nidx] == 9)
                {
                    MarkCell(nidx, isMine);
                }
            }
        }
    }

    static unsafe void PropagateQueue()
    {
        while (changedCells.Count > 0)
        {
            int idx = changedCells.Dequeue();
            bool isMine = (g[idx] == 10);

            int cy = idx / W;
            int cx = idx % W;
            for (int dy = -1; dy <= 1; dy++)
            {
                int ny = cy + dy;
                if (ny < 0 || ny >= H) continue;
                int nyW = ny * W;
                for (int dx = -1; dx <= 1; dx++)
                {
                    if (dx == 0 && dy == 0) continue;
                    int nx = cx + dx;
                    if (nx < 0 || nx >= W) continue;
                    int nidx = nyW + nx;

                    byte nval = g[nidx];
                    if (nval <= 8)
                    {
                        uc[nidx]--;
                        if (isMine)
                        {
                            if (mn[nidx] > 0) mn[nidx]--;
                        }

                        if (uc[nidx] > 0)
                        {
                            if (mn[nidx] == 0)
                            {
                                MarkUnknownNeighbors(nidx, false);
                            }
                            else if (mn[nidx] == uc[nidx])
                            {
                                MarkUnknownNeighbors(nidx, true);
                            }
                        }
                    }
                }
            }
        }
    }

    static unsafe void FilterActiveCells()
    {
        int writeIdx = 0;
        for (int i = 0; i < activeCells.Count; i++)
        {
            int idx = activeCells[i];
            if (uc[idx] > 0)
            {
                activeCells[writeIdx++] = idx;
            }
        }
        activeCells.RemoveRange(writeIdx, activeCells.Count - writeIdx);
    }

    static unsafe int GetUnknownNeighbors(int idx, int[] buffer)
    {
        int count = 0;
        int cy = idx / W;
        int cx = idx % W;
        for (int dy = -1; dy <= 1; dy++)
        {
            int ny = cy + dy;
            if (ny < 0 || ny >= H) continue;
            int nyW = ny * W;
            for (int dx = -1; dx <= 1; dx++)
            {
                if (dx == 0 && dy == 0) continue;
                int nx = cx + dx;
                if (nx < 0 || nx >= W) continue;
                int nidx = nyW + nx;
                if (g[nidx] == 9)
                {
                    buffer[count++] = nidx;
                }
            }
        }
        return count;
    }

    static unsafe int GetRevealedNeighbors(int idx, int[] buffer)
    {
        int count = 0;
        int cy = idx / W;
        int cx = idx % W;
        for (int dy = -1; dy <= 1; dy++)
        {
            int ny = cy + dy;
            if (ny < 0 || ny >= H) continue;
            int nyW = ny * W;
            for (int dx = -1; dx <= 1; dx++)
            {
                if (dx == 0 && dy == 0) continue;
                int nx = cx + dx;
                if (nx < 0 || nx >= W) continue;
                int nidx = nyW + nx;
                byte val = g[nidx];
                if (val <= 8)
                {
                    buffer[count++] = nidx;
                }
            }
        }
        return count;
    }

    static unsafe bool HasRevealedNeighbors(int idx)
    {
        int cy = idx / W;
        int cx = idx % W;
        for (int dy = -1; dy <= 1; dy++)
        {
            int ny = cy + dy;
            if (ny < 0 || ny >= H) continue;
            int nyW = ny * W;
            for (int dx = -1; dx <= 1; dx++)
            {
                if (dx == 0 && dy == 0) continue;
                int nx = cx + dx;
                if (nx < 0 || nx >= W) continue;
                byte val = g[nyW + nx];
                if (val <= 8) return true;
            }
        }
        return false;
    }

    static unsafe bool AnalyzePair(int A, int B)
    {
        int lenA = GetUnknownNeighbors(A, bufA);
        int lenB = GetUnknownNeighbors(B, bufB);

        int lenS = 0, lenAP = 0, lenBP = 0;
        for (int i = 0; i < lenA; i++)
        {
            int u = bufA[i];
            bool inB = false;
            for (int j = 0; j < lenB; j++)
            {
                if (bufB[j] == u)
                {
                    inB = true;
                    break;
                }
            }
            if (inB)
            {
                S[lenS++] = u;
            }
            else
            {
                APrime[lenAP++] = u;
            }
        }

        for (int j = 0; j < lenB; j++)
        {
            int u = bufB[j];
            bool inA = false;
            for (int i = 0; i < lenA; i++)
            {
                if (bufA[i] == u)
                {
                    inA = true;
                    break;
                }
            }
            if (!inA)
            {
                BPrime[lenBP++] = u;
            }
        }

        int MA = mn[A];
        int MB = mn[B];

        int Lx = Math.Max(0, Math.Max(MA - lenAP, MB - lenBP));
        int Ux = Math.Min(lenS, Math.Min(MA, MB));

        bool foundNew = false;

        // Check A'
        if (lenAP > 0)
        {
            if (MA - Lx == 0)
            {
                for (int i = 0; i < lenAP; i++)
                {
                    if (g[APrime[i]] == 9)
                    {
                        MarkCell(APrime[i], false);
                        foundNew = true;
                    }
                }
            }
            else if (MA - Ux == lenAP)
            {
                for (int i = 0; i < lenAP; i++)
                {
                    if (g[APrime[i]] == 9)
                    {
                        MarkCell(APrime[i], true);
                        foundNew = true;
                    }
                }
            }
        }

        // Check B'
        if (lenBP > 0)
        {
            if (MB - Lx == 0)
            {
                for (int i = 0; i < lenBP; i++)
                {
                    if (g[BPrime[i]] == 9)
                    {
                        MarkCell(BPrime[i], false);
                        foundNew = true;
                    }
                }
            }
            else if (MB - Ux == lenBP)
            {
                for (int i = 0; i < lenBP; i++)
                {
                    if (g[BPrime[i]] == 9)
                    {
                        MarkCell(BPrime[i], true);
                        foundNew = true;
                    }
                }
            }
        }

        // Check S
        if (lenS > 0)
        {
            if (Lx == lenS)
            {
                for (int i = 0; i < lenS; i++)
                {
                    if (g[S[i]] == 9)
                    {
                        MarkCell(S[i], true);
                        foundNew = true;
                    }
                }
            }
            else if (Ux == 0)
            {
                for (int i = 0; i < lenS; i++)
                {
                    if (g[S[i]] == 9)
                    {
                        MarkCell(S[i], false);
                        foundNew = true;
                    }
                }
            }
        }

        return foundNew;
    }

    static unsafe void ProcessComponent(List<int> compU, List<int> compR, int maxStates)
    {
        int k = compU.Count;
        int m = compR.Count;

        Dictionary<int, int> cellToVar = new Dictionary<int, int>();
        for (int i = 0; i < k; i++)
        {
            cellToVar[compU[i]] = i;
        }

        constraintVars = new int[m][];
        constraintTargets = new int[m];
        varConstraints = new List<int>[k];
        for (int i = 0; i < k; i++) varConstraints[i] = new List<int>();

        int[] tmpU = new int[8];
        for (int j = 0; j < m; j++)
        {
            int r = compR[j];
            int lenU = GetUnknownNeighbors(r, tmpU);
            constraintVars[j] = new int[lenU];
            constraintTargets[j] = mn[r];
            for (int i = 0; i < lenU; i++)
            {
                int varIdx = cellToVar[tmpU[i]];
                constraintVars[j][i] = varIdx;
                varConstraints[varIdx].Add(j);
            }
        }

        backtrackAssignment = new int[k];
        constrCount = new int[m];
        constrSum = new int[m];
        safeCount = new int[k];
        mineCount = new int[k];
        totalValidAssignments = 0;
        stateCount = 0;
        MaxStates = maxStates;

        bool completed = SolveBacktrack(0, k, m);
        if (completed && totalValidAssignments > 0)
        {
            for (int i = 0; i < k; i++)
            {
                if (safeCount[i] == totalValidAssignments)
                {
                    MarkCell(compU[i], false);
                }
                else if (mineCount[i] == totalValidAssignments)
                {
                    MarkCell(compU[i], true);
                }
            }
        }
    }

    static bool SolveBacktrack(int varIdx, int k, int m)
    {
        if (stateCount > MaxStates) return false;

        if (varIdx == k)
        {
            totalValidAssignments++;
            for (int i = 0; i < k; i++)
            {
                if (backtrackAssignment[i] == 0) safeCount[i]++;
                else mineCount[i]++;
            }
            return true;
        }

        stateCount++;

        for (int val = 0; val <= 1; val++)
        {
            backtrackAssignment[varIdx] = val;
            bool valid = true;
            List<int> affected = varConstraints[varIdx];
            int numAffected = affected.Count;

            for (int cIdx = 0; cIdx < numAffected; cIdx++)
            {
                int c = affected[cIdx];
                constrCount[c]++;
                if (val == 1) constrSum[c]++;

                int target = constraintTargets[c];
                int assignedCount = constrCount[c];
                int assignedSum = constrSum[c];
                int totalVars = constraintVars[c].Length;

                if (assignedSum > target || assignedSum + (totalVars - assignedCount) < target)
                {
                    valid = false;
                }
            }

            if (valid)
            {
                bool ok = SolveBacktrack(varIdx + 1, k, m);
                if (!ok) return false;
            }

            for (int cIdx = 0; cIdx < numAffected; cIdx++)
            {
                int c = affected[cIdx];
                constrCount[c]--;
                if (val == 1) constrSum[c]--;
            }
        }

        return true;
    }
}
