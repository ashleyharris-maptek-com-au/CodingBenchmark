using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Globalization;

class Program
{
    const double EARTH_RADIUS = 6371.0;

    static void Main()
    {
        // Set culture to Invariant to ensure double parsing is correct
        System.Threading.Thread.CurrentThread.CurrentCulture = CultureInfo.InvariantCulture;

        string line1 = ReadNonEmptyLine();
        if (line1 == null) return;
        string[] tokens1 = line1.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        int N = int.Parse(tokens1[0]);
        double mu = double.Parse(tokens1[1]);

        string line2 = ReadNonEmptyLine();
        if (line2 == null) return;
        double[] startOrbit = parseOrbit(line2);

        double[][] stationOrbits = new double[N][];
        for (int i = 0; i < N; i++)
        {
            string line = ReadNonEmptyLine();
            if (line == null) break;
            stationOrbits[i] = parseOrbit(line);
        }

        // Solve OTSP
        int[] bestOrder = Solve(N, mu, startOrbit, stationOrbits);

        // Output visitation order
        Console.WriteLine(string.Join(" ", bestOrder));
    }

    static string ReadNonEmptyLine()
    {
        string line;
        while ((line = Console.ReadLine()) != null)
        {
            if (!string.IsNullOrWhiteSpace(line))
                return line.Trim();
        }
        return null;
    }

    static double[] parseOrbit(string line)
    {
        string[] tokens = line.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        double[] orbit = new double[6];
        for (int i = 0; i < 6; i++)
        {
            orbit[i] = double.Parse(tokens[i]);
        }
        return orbit;
    }

    struct OrbitCharacteristics
    {
        public int Index;
        public double Altitude;
        public double Inclination;
        public double EffectiveInclination;
        public bool IsRetrograde;
        public double SemiMajorAxis;
        public double Radius;
    }

    static OrbitCharacteristics GetOrbitCharacteristics(int index, double[] orbit, double mu, double earthRadius)
    {
        double x = orbit[0], y = orbit[1], z = orbit[2];
        double vx = orbit[3], vy = orbit[4], vz = orbit[5];

        double r = Math.Sqrt(x * x + y * y + z * z);
        double v = Math.Sqrt(vx * vx + vy * vy + vz * vz);

        double h_x = y * vz - z * vy;
        double h_y = z * vx - x * vz;
        double h_z = x * vy - y * vx;
        double h_mag = Math.Sqrt(h_x * h_x + h_y * h_y + h_z * h_z);

        double inclination = 0;
        bool isRetrograde = false;
        double effectiveInclination = 0;

        if (h_mag > 0)
        {
            double cosInclination = h_z / h_mag;
            if (cosInclination < -1.0) cosInclination = -1.0;
            if (cosInclination > 1.0) cosInclination = 1.0;
            inclination = Math.Acos(cosInclination) * (180.0 / Math.PI);
            isRetrograde = inclination > 90.0;
            effectiveInclination = isRetrograde ? (180.0 - inclination) : inclination;
        }

        double energy = v * v / 2.0 - mu / r;
        double a = r;
        if (Math.Abs(energy) > 1e-10)
        {
            a = -mu / (2.0 * energy);
        }

        return new OrbitCharacteristics
        {
            Index = index,
            Altitude = r - earthRadius,
            Inclination = inclination,
            EffectiveInclination = effectiveInclination,
            IsRetrograde = isRetrograde,
            SemiMajorAxis = a,
            Radius = r
        };
    }

    static List<List<int>> ClusterOrbits(double[][] stationOrbits, double mu, double earthRadius)
    {
        int N = stationOrbits.Length;
        List<OrbitCharacteristics> characteristics = new List<OrbitCharacteristics>();
        for (int i = 0; i < N; i++)
        {
            characteristics.Add(GetOrbitCharacteristics(i, stationOrbits[i], mu, earthRadius));
        }

        List<OrbitCharacteristics> progradeOrbits = new List<OrbitCharacteristics>();
        List<OrbitCharacteristics> retrogradeOrbits = new List<OrbitCharacteristics>();
        foreach (var charac in characteristics)
        {
            if (!charac.IsRetrograde)
                progradeOrbits.Add(charac);
            else
                retrogradeOrbits.Add(charac);
        }

        List<List<int>> clusters = new List<List<int>>();

        // Cluster prograde
        if (progradeOrbits.Count > 0)
        {
            progradeOrbits.Sort((a, b) => a.Altitude.CompareTo(b.Altitude));
            List<int> currentCluster = new List<int>();
            double? lastAltitude = null;
            double? lastInclination = null;

            foreach (var charac in progradeOrbits)
            {
                if (lastAltitude == null || Math.Abs(charac.Altitude - lastAltitude.Value) > 500.0
                    || Math.Abs(charac.EffectiveInclination - lastInclination.Value) > 10.0)
                {
                    if (currentCluster.Count > 0)
                    {
                        clusters.Add(new List<int>(currentCluster));
                    }
                    currentCluster.Clear();
                    currentCluster.Add(charac.Index);
                }
                else
                {
                    currentCluster.Add(charac.Index);
                }
                lastAltitude = charac.Altitude;
                lastInclination = charac.EffectiveInclination;
            }
            if (currentCluster.Count > 0)
            {
                clusters.Add(new List<int>(currentCluster));
            }
        }

        // Cluster retrograde
        if (retrogradeOrbits.Count > 0)
        {
            retrogradeOrbits.Sort((a, b) => a.Altitude.CompareTo(b.Altitude));
            List<int> currentCluster = new List<int>();
            double? lastAltitude = null;
            double? lastInclination = null;

            foreach (var charac in retrogradeOrbits)
            {
                if (lastAltitude == null || Math.Abs(charac.Altitude - lastAltitude.Value) > 500.0
                    || Math.Abs(charac.EffectiveInclination - lastInclination.Value) > 10.0)
                {
                    if (currentCluster.Count > 0)
                    {
                        clusters.Add(new List<int>(currentCluster));
                    }
                    currentCluster.Clear();
                    currentCluster.Add(charac.Index);
                }
                else
                {
                    currentCluster.Add(charac.Index);
                }
                lastAltitude = charac.Altitude;
                lastInclination = charac.EffectiveInclination;
            }
            if (currentCluster.Count > 0)
            {
                clusters.Add(new List<int>(currentCluster));
            }
        }

        return clusters;
    }

    static int[] GetBaselineSolution(int N, double mu, double[] startOrbit, double[][] stationOrbits)
    {
        if (N <= 2)
        {
            if (N == 1) return new int[] { 0 };
            double dv1 = EstimateTransferDeltaV(startOrbit, stationOrbits[0], mu) +
                         EstimateTransferDeltaV(stationOrbits[0], stationOrbits[1], mu);
            double dv2 = EstimateTransferDeltaV(startOrbit, stationOrbits[1], mu) +
                         EstimateTransferDeltaV(stationOrbits[1], stationOrbits[0], mu);
            if (dv1 <= dv2) return new int[] { 0, 1 };
            else return new int[] { 1, 0 };
        }

        double earthRadius = 6371.0;
        List<List<int>> clusters = ClusterOrbits(stationOrbits, mu, earthRadius);

        List<List<int>> remainingClusters = new List<List<int>>();
        foreach (var c in clusters) remainingClusters.Add(new List<int>(c));

        List<List<int>> clusterOrder = new List<List<int>>();
        double[] currentOrbit = startOrbit;

        while (remainingClusters.Count > 0)
        {
            List<int> bestCluster = null;
            double bestCost = double.MaxValue;

            foreach (var cluster in remainingClusters)
            {
                foreach (int stationIdx in cluster)
                {
                    double cost = EstimateTransferDeltaV(currentOrbit, stationOrbits[stationIdx], mu);
                    if (cost < bestCost)
                    {
                        bestCost = cost;
                        bestCluster = cluster;
                    }
                }
            }

            clusterOrder.Add(bestCluster);
            remainingClusters.Remove(bestCluster);
            currentOrbit = stationOrbits[bestCluster[bestCluster.Count - 1]];
        }

        List<int> finalOrder = new List<int>();
        currentOrbit = startOrbit;

        foreach (var cluster in clusterOrder)
        {
            if (cluster.Count == 1)
            {
                int stationIdx = cluster[0];
                finalOrder.Add(stationIdx);
                currentOrbit = stationOrbits[stationIdx];
            }
            else
            {
                List<int> clusterRemaining = new List<int>(cluster);
                while (clusterRemaining.Count > 0)
                {
                    int bestStation = -1;
                    double bestDv = double.MaxValue;

                    foreach (int stationIdx in clusterRemaining)
                    {
                        double dv = EstimateTransferDeltaV(currentOrbit, stationOrbits[stationIdx], mu);
                        if (dv < bestDv)
                        {
                            bestDv = dv;
                            bestStation = stationIdx;
                        }
                    }

                    finalOrder.Add(bestStation);
                    clusterRemaining.Remove(bestStation);
                    currentOrbit = stationOrbits[bestStation];
                }
            }
        }

        return finalOrder.ToArray();
    }

    static double EstimateTransferDeltaV(double[] orbit1, double[] orbit2, double mu)
    {
        double r1_x = orbit1[0], r1_y = orbit1[1], r1_z = orbit1[2];
        double v1_x = orbit1[3], v1_y = orbit1[4], v1_z = orbit1[5];
        double r2_x = orbit2[0], r2_y = orbit2[1], r2_z = orbit2[2];
        double v2_x = orbit2[3], v2_y = orbit2[4], v2_z = orbit2[5];

        double r1 = Math.Sqrt(r1_x * r1_x + r1_y * r1_y + r1_z * r1_z);
        double r2 = Math.Sqrt(r2_x * r2_x + r2_y * r2_y + r2_z * r2_z);
        double v1 = Math.Sqrt(v1_x * v1_x + v1_y * v1_y + v1_z * v1_z);
        double v2 = Math.Sqrt(v2_x * v2_x + v2_y * v2_y + v2_z * v2_z);

        double h1_x = r1_y * v1_z - r1_z * v1_y;
        double h1_y = r1_z * v1_x - r1_x * v1_z;
        double h1_z = r1_x * v1_y - r1_y * v1_x;

        double h2_x = r2_y * v2_z - r2_z * v2_y;
        double h2_y = r2_z * v2_x - r2_x * v2_z;
        double h2_z = r2_x * v2_y - r2_y * v2_x;

        double h1_mag = Math.Sqrt(h1_x * h1_x + h1_y * h1_y + h1_z * h1_z);
        double h2_mag = Math.Sqrt(h2_x * h2_x + h2_y * h2_y + h2_z * h2_z);

        double plane_change_angle = 0;
        if (h1_mag > 0 && h2_mag > 0)
        {
            double h1_nx = h1_x / h1_mag;
            double h1_ny = h1_y / h1_mag;
            double h1_nz = h1_z / h1_mag;

            double h2_nx = h2_x / h2_mag;
            double h2_ny = h2_y / h2_mag;
            double h2_nz = h2_z / h2_mag;

            double cos_plane_change = h1_nx * h2_nx + h1_ny * h2_ny + h1_nz * h2_nz;
            if (cos_plane_change < -1.0) cos_plane_change = -1.0;
            if (cos_plane_change > 1.0) cos_plane_change = 1.0;
            plane_change_angle = Math.Acos(cos_plane_change);
        }

        double in_plane_dv = 0;
        double a_transfer = (r1 + r2) / 2.0;
        if (a_transfer > 0)
        {
            double v_t1 = Math.Sqrt(mu * (2.0 / r1 - 1.0 / a_transfer));
            double v_t2 = Math.Sqrt(mu * (2.0 / r2 - 1.0 / a_transfer));
            double v_c1 = Math.Sqrt(mu / r1);
            double v_c2 = Math.Sqrt(mu / r2);

            double dv1 = Math.Abs(v_t1 - v_c1);
            double dv2 = Math.Abs(v_c2 - v_t2);
            in_plane_dv = dv1 + dv2;
        }
        else
        {
            in_plane_dv = Math.Abs(v2 - v1);
        }

        double v_at_node = Math.Sqrt(mu * (2.0 / r1 + 2.0 / r2) / 2.0);
        double plane_change_dv = 2.0 * v_at_node * Math.Sin(plane_change_angle / 2.0);

        double total_dv = in_plane_dv + plane_change_dv;
        double vel_magnitude_diff = Math.Abs(v2 - v1);
        double velocity_penalty = vel_magnitude_diff * 0.1;

        return total_dv + velocity_penalty;
    }

    static int[] Solve(int N, double mu, double[] startOrbit, double[][] stationOrbits)
    {
        double[][] costMatrix = new double[N][];
        for (int i = 0; i < N; i++)
        {
            costMatrix[i] = new double[N];
        }
        for (int i = 0; i < N; i++)
        {
            for (int j = i + 1; j < N; j++)
            {
                double c = EstimateTransferDeltaV(stationOrbits[i], stationOrbits[j], mu);
                costMatrix[i][j] = c;
                costMatrix[j][i] = c;
            }
        }

        double[] startCost = new double[N];
        for (int i = 0; i < N; i++)
        {
            startCost[i] = EstimateTransferDeltaV(startOrbit, stationOrbits[i], mu);
        }

        if (N <= 20)
        {
            return SolveExactDP(N, costMatrix, startCost);
        }
        else
        {
            return SolveHeuristic(N, costMatrix, startCost, mu, startOrbit, stationOrbits);
        }
    }

    static int[] SolveExactDP(int N, double[][] costMatrix, double[] startCost)
    {
        int numStates = 1 << N;
        double[] dp = new double[numStates * N];
        int[] parent = new int[numStates * N];

        for (int i = 0; i < dp.Length; i++)
        {
            dp[i] = double.MaxValue;
            parent[i] = -1;
        }

        for (int i = 0; i < N; i++)
        {
            dp[(1 << i) * N + i] = startCost[i];
        }

        for (int mask = 1; mask < numStates; mask++)
        {
            for (int u = 0; u < N; u++)
            {
                if ((mask & (1 << u)) == 0) continue;
                double currentVal = dp[mask * N + u];
                if (currentVal == double.MaxValue) continue;

                for (int v = 0; v < N; v++)
                {
                    if ((mask & (1 << v)) != 0) continue;
                    int nextMask = mask | (1 << v);
                    double newVal = currentVal + costMatrix[u][v];
                    int nextIdx = nextMask * N + v;
                    if (newVal < dp[nextIdx])
                    {
                        dp[nextIdx] = newVal;
                        parent[nextIdx] = u;
                    }
                }
            }
        }

        double minTotalCost = double.MaxValue;
        int bestLastNode = -1;
        int finalMask = numStates - 1;
        for (int i = 0; i < N; i++)
        {
            double cost = dp[finalMask * N + i];
            if (cost < minTotalCost)
            {
                minTotalCost = cost;
                bestLastNode = i;
            }
        }

        int[] path = new int[N];
        int currNode = bestLastNode;
        int currMask = finalMask;
        for (int step = N - 1; step >= 0; step--)
        {
            path[step] = currNode;
            int prevNode = parent[currMask * N + currNode];
            currMask ^= (1 << currNode);
            currNode = prevNode;
        }
        return path;
    }

    static int[] SolveHeuristic(int N, double[][] costMatrix, double[] startCost, double mu, double[] startOrbit, double[][] stationOrbits)
    {
        int[] bestPath = new int[N];
        double bestCost = double.MaxValue;

        Action<int[]> updateBest = (path) =>
        {
            double cost = GetPathCost(path, costMatrix, startCost, N);
            if (cost < bestCost)
            {
                bestCost = cost;
                Array.Copy(path, bestPath, N);
            }
        };

        // 1. Baseline clustering solution
        int[] baselinePath = GetBaselineSolution(N, mu, startOrbit, stationOrbits);
        updateBest(baselinePath);
        TwoOpt(baselinePath, costMatrix, startCost, N);
        updateBest(baselinePath);

        // 2. Nearest Neighbor starting from Start
        int[] nnPath = GetGreedyPath(costMatrix, startCost, N);
        updateBest(nnPath);
        TwoOpt(nnPath, costMatrix, startCost, N);
        updateBest(nnPath);

        // 3. Nearest Neighbor starting from each of the N stations
        for (int i = 0; i < N; i++)
        {
            int[] path = GetGreedyPathFromNode(costMatrix, startCost, N, i);
            TwoOpt(path, costMatrix, startCost, N);
            updateBest(path);
        }

        // 4. Random starts with 2-opt until time limit is reached or max iterations
        Random rand = new Random(42);
        System.Diagnostics.Stopwatch sw = System.Diagnostics.Stopwatch.StartNew();
        long maxSearchTimeMs = 2000; // Search for at most 2 seconds
        int iterations = 0;
        int maxIterations = 2000;

        while (sw.ElapsedMilliseconds < maxSearchTimeMs && iterations < maxIterations)
        {
            int[] path = GetRandomPath(N, rand);
            TwoOpt(path, costMatrix, startCost, N);
            updateBest(path);
            iterations++;
        }

        // 5. Simulated Annealing starting from bestPath for 3 seconds
        int[] saPath = new int[N];
        Array.Copy(bestPath, saPath, N);
        SimulatedAnnealing(saPath, costMatrix, startCost, N, rand, 3000);
        updateBest(saPath);

        return bestPath;
    }

    static double GetPathCost(int[] path, double[][] cost, double[] startCost, int N)
    {
        double total = startCost[path[0]];
        for (int i = 0; i < N - 1; i++)
        {
            total += cost[path[i]][path[i + 1]];
        }
        return total;
    }

    static void TwoOpt(int[] path, double[][] cost, double[] startCost, int N)
    {
        double currentCost = GetPathCost(path, cost, startCost, N);
        bool improved = true;
        while (improved)
        {
            improved = false;
            for (int i = 0; i < N - 1; i++)
            {
                for (int j = i + 1; j < N; j++)
                {
                    Array.Reverse(path, i, j - i + 1);
                    double newCost = GetPathCost(path, cost, startCost, N);
                    if (newCost < currentCost - 1e-9)
                    {
                        currentCost = newCost;
                        improved = true;
                    }
                    else
                    {
                        Array.Reverse(path, i, j - i + 1);
                    }
                }
            }
        }
    }

    static void SimulatedAnnealing(int[] path, double[][] cost, double[] startCost, int N, Random rand, double timeBudgetMs)
    {
        double currentCost = GetPathCost(path, cost, startCost, N);
        int[] currentPath = new int[N];
        Array.Copy(path, currentPath, N);

        int[] bestPath = new int[N];
        Array.Copy(path, bestPath, N);
        double bestCost = currentCost;

        double initialTemp = 10.0;
        double finalTemp = 0.001;
        double temp = initialTemp;

        System.Diagnostics.Stopwatch sw = System.Diagnostics.Stopwatch.StartNew();

        while (sw.ElapsedMilliseconds < timeBudgetMs)
        {
            double progress = (double)sw.ElapsedMilliseconds / timeBudgetMs;
            temp = initialTemp * Math.Pow(finalTemp / initialTemp, progress);

            int moveType = rand.Next(2);
            if (moveType == 0)
            {
                int i = rand.Next(N - 1);
                int j = rand.Next(i + 1, N);

                Array.Reverse(currentPath, i, j - i + 1);
                double newCost = GetPathCost(currentPath, cost, startCost, N);
                double delta = newCost - currentCost;

                if (delta < 0 || rand.NextDouble() < Math.Exp(-delta / temp))
                {
                    currentCost = newCost;
                    if (currentCost < bestCost)
                    {
                        bestCost = currentCost;
                        Array.Copy(currentPath, bestPath, N);
                    }
                }
                else
                {
                    Array.Reverse(currentPath, i, j - i + 1);
                }
            }
            else
            {
                int i = rand.Next(N);
                int j = rand.Next(N);
                if (i == j) continue;

                int node = currentPath[i];
                if (i < j)
                {
                    for (int k = i; k < j; k++) currentPath[k] = currentPath[k + 1];
                    currentPath[j] = node;
                }
                else
                {
                    for (int k = i; k > j; k--) currentPath[k] = currentPath[k - 1];
                    currentPath[j] = node;
                }

                double newCost = GetPathCost(currentPath, cost, startCost, N);
                double delta = newCost - currentCost;

                if (delta < 0 || rand.NextDouble() < Math.Exp(-delta / temp))
                {
                    currentCost = newCost;
                    if (currentCost < bestCost)
                    {
                        bestCost = currentCost;
                        Array.Copy(currentPath, bestPath, N);
                    }
                }
                else
                {
                    if (i < j)
                    {
                        for (int k = j; k > i; k--) currentPath[k] = currentPath[k - 1];
                        currentPath[i] = node;
                    }
                    else
                    {
                        for (int k = j; k < i; k++) currentPath[k] = currentPath[k + 1];
                        currentPath[i] = node;
                    }
                }
            }
        }
        Array.Copy(bestPath, path, N);
    }

    static int[] GetGreedyPath(double[][] cost, double[] startCost, int N)
    {
        int[] path = new int[N];
        bool[] visited = new bool[N];
        int bestNext = -1;
        double minD = double.MaxValue;
        for (int i = 0; i < N; i++)
        {
            if (startCost[i] < minD)
            {
                minD = startCost[i];
                bestNext = i;
            }
        }
        path[0] = bestNext;
        visited[bestNext] = true;

        for (int step = 1; step < N; step++)
        {
            int curr = path[step - 1];
            bestNext = -1;
            minD = double.MaxValue;
            for (int i = 0; i < N; i++)
            {
                if (!visited[i] && cost[curr][i] < minD)
                {
                    minD = cost[curr][i];
                    bestNext = i;
                }
            }
            path[step] = bestNext;
            visited[bestNext] = true;
        }
        return path;
    }

    static int[] GetGreedyPathFromNode(double[][] cost, double[] startCost, int N, int startNode)
    {
        int[] path = new int[N];
        bool[] visited = new bool[N];
        path[0] = startNode;
        visited[startNode] = true;

        for (int step = 1; step < N; step++)
        {
            int curr = path[step - 1];
            int bestNext = -1;
            double minD = double.MaxValue;
            for (int i = 0; i < N; i++)
            {
                if (!visited[i] && cost[curr][i] < minD)
                {
                    minD = cost[curr][i];
                    bestNext = i;
                }
            }
            path[step] = bestNext;
            visited[bestNext] = true;
        }
        return path;
    }

    static int[] GetRandomPath(int N, Random rand)
    {
        int[] path = new int[N];
        for (int i = 0; i < N; i++) path[i] = i;
        for (int i = N - 1; i > 0; i--)
        {
            int j = rand.Next(i + 1);
            int temp = path[i];
            path[i] = path[j];
            path[j] = temp;
        }
        return path;
    }
}
