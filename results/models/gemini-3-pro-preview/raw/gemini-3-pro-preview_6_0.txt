using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Globalization;
using System.Text.RegularExpressions;

namespace OrbitalTSP
{
    struct Vector3
    {
        public double X, Y, Z;
        public Vector3(double x, double y, double z) { X = x; Y = y; Z = z; }
        public static Vector3 Cross(Vector3 a, Vector3 b) =>
            new Vector3(a.Y * b.Z - a.Z * b.Y, a.Z * b.X - a.X * b.Z, a.X * b.Y - a.Y * b.X);
        public static double Dot(Vector3 a, Vector3 b) => a.X * b.X + a.Y * b.Y + a.Z * b.Z;
        public static Vector3 operator -(Vector3 a, Vector3 b) => new Vector3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
        public static Vector3 operator +(Vector3 a, Vector3 b) => new Vector3(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
        public static Vector3 operator *(Vector3 a, double d) => new Vector3(a.X * d, a.Y * d, a.Z * d);
        public static Vector3 operator /(Vector3 a, double d) => new Vector3(a.X / d, a.Y / d, a.Z / d);
        public double Length() => Math.Sqrt(X * X + Y * Y + Z * Z);
    }

    class Orbit
    {
        public Vector3 R, V, H, E;
        public double P;

        public Orbit(double x, double y, double z, double vx, double vy, double vz, double mu)
        {
            R = new Vector3(x, y, z);
            V = new Vector3(vx, vy, vz);
            H = Vector3.Cross(R, V);
            P = Vector3.Dot(H, H) / mu;
            Vector3 v_cross_h = Vector3.Cross(V, H);
            E = (v_cross_h / mu) - (R / R.Length());
        }
    }

    class Program
    {
        static double[] ParseLine(string line)
        {
            if (string.IsNullOrWhiteSpace(line)) return new double[0];
            var matches = Regex.Matches(line, @"[-+]?[0-9]*\.?[0-9]+(?:[eE][-+]?[0-9]+)?");
            var nums = new List<double>();
            foreach(Match m in matches)
            {
                nums.Add(double.Parse(m.Value, CultureInfo.InvariantCulture));
            }
            return nums.ToArray();
        }

        static double[] ReadNextLine()
        {
            while (true)
            {
                string line = Console.ReadLine();
                if (line == null) return null;
                var nums = ParseLine(line);
                if (nums.Length > 0) return nums;
            }
        }

        static double Distance(Orbit a, Orbit b, double mu)
        {
            double r_m = (a.P + b.P) / 2.0;
            if (r_m < 1e-6) r_m = 1e-6;
            double v_m = Math.Sqrt(mu / r_m);
            double dh = (a.H - b.H).Length();
            double de = (a.E - b.E).Length();
            return (dh / r_m) + v_m * de;
        }

        static double TotalCost(int[] p, double[,] dist)
        {
            double cost = 0;
            for (int i = 0; i < p.Length - 1; i++)
            {
                cost += dist[p[i], p[i + 1]];
            }
            return cost;
        }

        static void Main(string[] args)
        {
            var line1 = ReadNextLine();
            if (line1 == null || line1.Length < 2) return;
            int N = (int)line1[0];
            double mu = line1[1];

            var line2 = ReadNextLine();
            if (line2 == null || line2.Length < 6) return;
            Orbit startOrbit = new Orbit(line2[0], line2[1], line2[2], line2[3], line2[4], line2[5], mu);

            Orbit[] stations = new Orbit[N];
            for (int i = 0; i < N; i++)
            {
                var line = ReadNextLine();
                if (line == null || line.Length < 6) break;
                stations[i] = new Orbit(line[0], line[1], line[2], line[3], line[4], line[5], mu);
            }

            double[,] dist = new double[N + 1, N + 1];
            for (int i = 0; i <= N; i++)
            {
                Orbit o1 = i == 0 ? startOrbit : stations[i - 1];
                for (int j = 0; j <= N; j++)
                {
                    Orbit o2 = j == 0 ? startOrbit : stations[j - 1];
                    if (i == j) dist[i, j] = 0;
                    else dist[i, j] = Distance(o1, o2, mu);
                }
            }

            int[] bestPath = new int[N + 1];
            double bestCost = double.MaxValue;

            for (int firstNode = 1; firstNode <= N; firstNode++)
            {
                int[] p = new int[N + 1];
                p[0] = 0;
                p[1] = firstNode;
                bool[] vis = new bool[N + 1];
                vis[0] = true;
                vis[firstNode] = true;
                int currNode = firstNode;
                
                for (int step = 2; step <= N; step++)
                {
                    double minD = double.MaxValue;
                    int nxt = -1;
                    for (int i = 1; i <= N; i++)
                    {
                        if (!vis[i] && dist[currNode, i] < minD)
                        {
                            minD = dist[currNode, i];
                            nxt = i;
                        }
                    }
                    if (nxt != -1)
                    {
                        p[step] = nxt;
                        vis[nxt] = true;
                        currNode = nxt;
                    }
                }
                
                bool imp = true;
                while(imp)
                {
                    imp = false;
                    for (int i = 1; i < N; i++)
                    {
                        for (int j = i + 1; j <= N; j++)
                        {
                            int n1 = p[i - 1];
                            int n2 = p[i];
                            int n3 = p[j];
                            int n4 = j + 1 <= N ? p[j + 1] : -1;

                            double cCost = dist[n1, n2];
                            if (n4 != -1) cCost += dist[n3, n4];

                            double nCost = dist[n1, n3];
                            if (n4 != -1) nCost += dist[n2, n4];

                            if (nCost < cCost - 1e-9)
                            {
                                Array.Reverse(p, i, j - i + 1);
                                imp = true;
                            }
                        }
                    }
                }
                
                double cost = TotalCost(p, dist);
                if (cost < bestCost)
                {
                    bestCost = cost;
                    bestPath = (int[])p.Clone();
                }
            }

            Stopwatch sw = Stopwatch.StartNew();
            double timeLimit = 4.5;
            Random rng = new Random(42);
            double temp = 10.0;
            double coolingRate = 0.99999;
            int[] saPath = (int[])bestPath.Clone();
            double saCost = bestCost;
            int acceptCount = 0;
            
            while (sw.Elapsed.TotalSeconds < timeLimit)
            {
                int i = rng.Next(1, N + 1);
                int j = rng.Next(1, N + 1);
                if (i == j) continue;
                if (i > j) { int tempIdx = i; i = j; j = tempIdx; }

                int n1 = saPath[i - 1];
                int n2 = saPath[i];
                int n3 = saPath[j];
                int n4 = j + 1 <= N ? saPath[j + 1] : -1;

                double dCost = dist[n1, n3] - dist[n1, n2];
                if (n4 != -1)
                {
                    dCost += dist[n2, n4] - dist[n3, n4];
                }

                if (dCost < 0 || Math.Exp(-dCost / temp) > rng.NextDouble())
                {
                    Array.Reverse(saPath, i, j - i + 1);
                    saCost += dCost;
                    acceptCount++;

                    if (acceptCount % 10000 == 0) saCost = TotalCost(saPath, dist);

                    if (saCost < bestCost)
                    {
                        saCost = TotalCost(saPath, dist);
                        if (saCost < bestCost)
                        {
                            bestCost = saCost;
                            bestPath = (int[])saPath.Clone();
                        }
                    }
                }

                temp *= coolingRate;
                if (temp < 1e-5) temp = 10.0;
            }

            int[] result = new int[N];
            for (int i = 1; i <= N; i++)
            {
                result[i - 1] = bestPath[i] - 1;
            }

            Console.WriteLine(string.Join(" ", result));
        }
    }
}
