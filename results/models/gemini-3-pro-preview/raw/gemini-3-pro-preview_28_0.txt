use std::io::{self, Read, Write};

struct SplitMix64 {
    state: u64,
}

impl SplitMix64 {
    fn new(seed: u64) -> Self {
        Self { state: seed }
    }

    fn next(&mut self) -> u64 {
        self.state = self.state.wrapping_add(0x9e3779b97f4a7c15);
        let mut z = self.state;
        z = (z ^ (z >> 30)).wrapping_mul(0xbf58476d1ce4e5b9);
        z = (z ^ (z >> 27)).wrapping_mul(0x94d049bb133111eb);
        z ^ (z >> 31)
    }

    fn next_range(&mut self, limit: usize) -> usize {
        if limit == 0 {
            return 0;
        }
        ((self.next() as u128 * limit as u128) >> 64) as usize
    }
}

struct FastReader {
    buffer: Vec<u8>,
    pos: usize,
}

impl FastReader {
    fn new() -> Self {
        let mut buffer = Vec::new();
        io::stdin().read_to_end(&mut buffer).unwrap();
        Self { buffer, pos: 0 }
    }

    fn next_u32(&mut self) -> Option<u32> {
        while self.pos < self.buffer.len() && self.buffer[self.pos] <= b' ' {
            self.pos += 1;
        }
        if self.pos >= self.buffer.len() {
            return None;
        }
        let mut res = 0u32;
        while self.pos < self.buffer.len() && self.buffer[self.pos] > b' ' {
            res = res * 10 + (self.buffer[self.pos] - b'0') as u32;
            self.pos += 1;
        }
        Some(res)
    }
}

struct Edge {
    u: u32,
    v: u32,
}

struct FastVC {
    v_num: usize,
    e_num: usize,
    edges: Vec<Edge>,
    dscore: Vec<i32>,
    time_stamp: Vec<i64>,
    v_beg_idx: Vec<usize>,
    v_edges: Vec<u32>,
    v_adj: Vec<u32>,
    v_degree: Vec<u32>,
    
    // solution state
    c_size: usize,
    v_in_c: Vec<bool>,
    remove_cand: Vec<u32>,
    index_in_remove_cand: Vec<u32>,
    remove_cand_size: usize,
    
    best_c_size: usize,
    best_v_in_c: Vec<bool>,
    
    // uncovered edge stack
    uncov_stack: Vec<u32>,
    uncov_stack_fill_pointer: usize,
    index_in_uncov_stack: Vec<u32>,
    
    rng: SplitMix64,
}

impl FastVC {
    fn new(v_num: usize, raw_edges: Vec<Edge>, seed: u64) -> Self {
        let e_num = raw_edges.len();
        
        let mut v_degree = vec![0u32; v_num];
        for edge in &raw_edges {
            v_degree[edge.u as usize] += 1;
            v_degree[edge.v as usize] += 1;
        }
        
        let mut v_beg_idx = Vec::with_capacity(v_num + 1);
        v_beg_idx.push(0);
        let mut sum = 0;
        for &deg in &v_degree {
            sum += deg as usize;
            v_beg_idx.push(sum);
        }
        
        let total_deg = sum;
        let mut v_edges = vec![0u32; total_deg];
        let mut v_adj = vec![0u32; total_deg];
        
        let mut v_degree_tmp = vec![0usize; v_num];
        for (e, edge) in raw_edges.iter().enumerate() {
            let u = edge.u as usize;
            let v = edge.v as usize;
            
            let pos_u = v_beg_idx[u] + v_degree_tmp[u];
            v_edges[pos_u] = e as u32;
            v_adj[pos_u] = v as u32;
            v_degree_tmp[u] += 1;
            
            let pos_v = v_beg_idx[v] + v_degree_tmp[v];
            v_edges[pos_v] = e as u32;
            v_adj[pos_v] = u as u32;
            v_degree_tmp[v] += 1;
        }
        
        let mut solver = Self {
            v_num,
            e_num,
            edges: raw_edges,
            dscore: vec![0; v_num],
            time_stamp: vec![0; v_num],
            v_beg_idx,
            v_edges,
            v_adj,
            v_degree,
            c_size: 0,
            v_in_c: vec![false; v_num],
            remove_cand: vec![0; v_num],
            index_in_remove_cand: vec![0; v_num],
            remove_cand_size: 0,
            best_c_size: 0,
            best_v_in_c: vec![false; v_num],
            uncov_stack: vec![0; e_num],
            uncov_stack_fill_pointer: 0,
            index_in_uncov_stack: vec![0; e_num],
            rng: SplitMix64::new(seed),
        };
        
        solver.init_sol();
        solver
    }

    fn init_sol(&mut self) {
        self.c_size = 0;
        for e in 0..self.e_num {
            let u = self.edges[e].u as usize;
            let v = self.edges[e].v as usize;
            
            if !self.v_in_c[u] && !self.v_in_c[v] {
                if self.v_degree[u] > self.v_degree[v] {
                    self.v_in_c[u] = true;
                } else {
                    self.v_in_c[v] = true;
                }
                self.c_size += 1;
            }
        }
        
        self.uncov_stack_fill_pointer = 0;
        
        // calculate dscores
        for e in 0..self.e_num {
            let u = self.edges[e].u as usize;
            let v = self.edges[e].v as usize;
            
            if self.v_in_c[u] && !self.v_in_c[v] {
                self.dscore[u] -= 1;
            } else if self.v_in_c[v] && !self.v_in_c[u] {
                self.dscore[v] -= 1;
            }
        }
        
        // remove redundant vertices
        for v in 0..self.v_num {
            if self.v_in_c[v] && self.dscore[v] == 0 {
                self.remove(v);
                self.c_size -= 1;
            }
        }
        
        self.update_best_sol();
        self.reset_remove_cand();
    }

    fn remove(&mut self, v: usize) {
        self.v_in_c[v] = false;
        self.dscore[v] = -self.dscore[v];
        
        let start_idx = self.v_beg_idx[v];
        let edge_count = self.v_degree[v] as usize;
        
        let v_edges = &self.v_edges;
        let v_adj = &self.v_adj;
        let v_in_c = &self.v_in_c;
        let dscore = &mut self.dscore;
        let index_in_uncov_stack = &mut self.index_in_uncov_stack;
        let uncov_stack = &mut self.uncov_stack;
        let uncov_stack_fill_pointer = &mut self.uncov_stack_fill_pointer;
        
        let v_edges_slice = &v_edges[start_idx..start_idx + edge_count];
        let v_adj_slice = &v_adj[start_idx..start_idx + edge_count];
        
        for i in 0..edge_count {
            let e = v_edges_slice[i] as usize;
            let n = v_adj_slice[i] as usize;
            
            if !v_in_c[n] {
                dscore[n] += 1;
                
                // uncover(e)
                index_in_uncov_stack[e] = *uncov_stack_fill_pointer as u32;
                uncov_stack[*uncov_stack_fill_pointer] = e as u32;
                *uncov_stack_fill_pointer += 1;
            } else {
                dscore[n] -= 1;
            }
        }
    }

    fn add(&mut self, v: usize) {
        self.v_in_c[v] = true;
        self.dscore[v] = -self.dscore[v];
        
        let start_idx = self.v_beg_idx[v];
        let edge_count = self.v_degree[v] as usize;
        
        let v_edges = &self.v_edges;
        let v_adj = &self.v_adj;
        let v_in_c = &self.v_in_c;
        let dscore = &mut self.dscore;
        let index_in_uncov_stack = &mut self.index_in_uncov_stack;
        let uncov_stack = &mut self.uncov_stack;
        let uncov_stack_fill_pointer = &mut self.uncov_stack_fill_pointer;
        
        let v_edges_slice = &v_edges[start_idx..start_idx + edge_count];
        let v_adj_slice = &v_adj[start_idx..start_idx + edge_count];
        
        for i in 0..edge_count {
            let e = v_edges_slice[i] as usize;
            let n = v_adj_slice[i] as usize;
            
            if !v_in_c[n] {
                dscore[n] -= 1;
                
                // cover(e)
                *uncov_stack_fill_pointer -= 1;
                let last_uncov_edge = uncov_stack[*uncov_stack_fill_pointer] as usize;
                let index = index_in_uncov_stack[e] as usize;
                uncov_stack[index] = last_uncov_edge as u32;
                index_in_uncov_stack[last_uncov_edge] = index as u32;
            } else {
                dscore[n] += 1;
            }
        }
    }

    fn update_best_sol(&mut self) {
        self.best_v_in_c.clone_from(&self.v_in_c);
        self.best_c_size = self.c_size;
    }

    fn reset_remove_cand(&mut self) {
        let mut j = 0;
        for v in 0..self.v_num {
            if self.v_in_c[v] {
                self.remove_cand[j] = v as u32;
                self.index_in_remove_cand[v] = j as u32;
                j += 1;
            } else {
                self.index_in_remove_cand[v] = 0;
            }
        }
        self.remove_cand_size = j;
    }

    fn update_target_size(&mut self) -> bool {
        if self.c_size == 0 {
            return false;
        }
        self.c_size -= 1;
        
        let mut best_remove_v = self.remove_cand[0] as usize;
        if self.dscore[best_remove_v] != 0 {
            for i in 1..self.remove_cand_size {
                let v = self.remove_cand[i] as usize;
                if self.dscore[v] == 0 {
                    best_remove_v = v;
                    break;
                }
                if class_higher_dscore(self.dscore[v], self.dscore[best_remove_v]) {
                    best_remove_v = v;
                }
            }
        }
        
        self.remove(best_remove_v);
        
        self.remove_cand_size -= 1;
        let last_remove_cand_v = self.remove_cand[self.remove_cand_size];
        let index = self.index_in_remove_cand[best_remove_v] as usize;
        self.remove_cand[index] = last_remove_cand_v;
        self.index_in_remove_cand[last_remove_cand_v as usize] = index as u32;
        true
    }

    fn choose_remove_v(&mut self) -> usize {
        let cand_count = 50;
        let mut best_v = self.remove_cand[self.rng.next_range(self.remove_cand_size)] as usize;
        
        for _ in 0..cand_count - 1 {
            let v = self.remove_cand[self.rng.next_range(self.remove_cand_size)] as usize;
            
            if self.dscore[v] < self.dscore[best_v] {
                continue;
            } else if self.dscore[v] > self.dscore[best_v] {
                best_v = v;
            } else if self.time_stamp[v] < self.time_stamp[best_v] {
                best_v = v;
            }
        }
        best_v
    }

    fn cover_ls(&mut self, cutoff_time: std::time::Duration) {
        let start = std::time::Instant::now();
        let mut step: i64 = 1;
        let try_step = 10;
        
        loop {
            if self.uncov_stack_fill_pointer == 0 {
                self.update_best_sol();
                if !self.update_target_size() {
                    break;
                }
                continue;
            }
            
            if step % try_step == 0 {
                if start.elapsed() >= cutoff_time {
                    break;
                }
            }
            
            let remove_v = self.choose_remove_v();
            self.remove(remove_v);
            
            let e = self.uncov_stack[self.rng.next_range(self.uncov_stack_fill_pointer)] as usize;
            let v1 = self.edges[e].u as usize;
            let v2 = self.edges[e].v as usize;
            
            let add_v = if self.dscore[v1] > self.dscore[v2] || 
                           (self.dscore[v1] == self.dscore[v2] && self.time_stamp[v1] < self.time_stamp[v2]) {
                v1
            } else {
                v2
            };
            
            self.add(add_v);
            
            let index = self.index_in_remove_cand[remove_v] as usize;
            self.index_in_remove_cand[remove_v] = 0;
            
            self.remove_cand[index] = add_v as u32;
            self.index_in_remove_cand[add_v] = index as u32;
            
            self.time_stamp[add_v] = step;
            self.time_stamp[remove_v] = step;
            
            step += 1;
        }
    }
}

#[inline]
fn class_higher_dscore(d1: i32, d2: i32) -> bool {
    d1 > d2
}

fn main() {
    let mut reader = FastReader::new();
    let v_num_opt = reader.next_u32();
    let e_num_opt = reader.next_u32();
    
    if v_num_opt.is_none() || e_num_opt.is_none() {
        return;
    }
    
    let v_num = v_num_opt.unwrap() as usize;
    let e_num = e_num_opt.unwrap() as usize;
    
    if e_num == 0 {
        println!("0");
        println!();
        return;
    }
    
    let mut raw_edges = Vec::with_capacity(e_num);
    for _ in 0..e_num {
        let u = reader.next_u32().unwrap();
        let v = reader.next_u32().unwrap();
        raw_edges.push(Edge { u, v });
    }
    
    let seed = 123456789u64;
    
    let mut solver = FastVC::new(v_num, raw_edges, seed);
    
    let cutoff = std::time::Duration::from_millis(2500);
    solver.cover_ls(cutoff);
    
    let mut out_vertices = Vec::new();
    for v in 0..v_num {
        if solver.best_v_in_c[v] {
            out_vertices.push(v);
        }
    }
    
    let stdout = io::stdout();
    let mut handle = io::BufWriter::new(stdout.lock());
    
    writeln!(handle, "{}", out_vertices.len()).unwrap();
    for (i, &v) in out_vertices.iter().enumerate() {
        if i > 0 {
            write!(handle, " ").unwrap();
        }
        write!(handle, "{}", v).unwrap();
    }
    writeln!(handle).unwrap();
}
