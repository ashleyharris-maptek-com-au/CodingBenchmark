// Portable bit manipulation intrinsics for C++17
inline int portable_popcount64(uint64_t x) {
#ifdef _MSC_VER
    return (int)__popcnt64(x);
#else
    return __builtin_popcountll(x);
#endif
}

inline int portable_ctz64(uint64_t x) {
    if (x == 0) return 64;
#ifdef _MSC_VER
    unsigned long index;
    _BitScanForward64(&index, x);
    return (int)index;
#else
    return __builtin_ctzll(x);
#endif
}

// Fast I/O reader using getchar()
class FastReader {
    bool eof = false;

public:
    int read_int() {
        if (eof) return -1;
        int c = getchar();
        while (c != EOF && c <= 32) {
            c = getchar();
        }
        if (c == EOF) {
            eof = true;
            return -1;
        }
        int res = 0;
        while (c > 32) {
            res = res * 10 + (c - '0');
            c = getchar();
        }
        if (c == EOF) {
            eof = true;
        }
        return res;
    }
};

// Represents a set of colors. Supports up to 1024 colors with 16 64-bit words.
struct ColorSet {
    static constexpr int WORD_COUNT = 16;
    uint64_t mask[WORD_COUNT] = {0};

    void set(int color) {
        if (color < 0 || color >= WORD_COUNT * 64) return;
        mask[color >> 6] |= (1ULL << (color & 63));
    }

    bool test(int color) const {
        if (color < 0 || color >= WORD_COUNT * 64) return false;
        return (mask[color >> 6] & (1ULL << (color & 63))) != 0;
    }

    int count() const {
        int cnt = 0;
        for (int i = 0; i < WORD_COUNT; ++i) {
            cnt += portable_popcount64(mask[i]);
        }
        return cnt;
    }

    int first_unset() const {
        for (int i = 0; i < WORD_COUNT; ++i) {
            if (mask[i] != ~0ULL) {
                return (i << 6) + portable_ctz64(~mask[i]);
            }
        }
        return WORD_COUNT * 64;
    }
};

// CSR graph representation
struct Graph {
    int n = 0;
    int m = 0;
    std::vector<int> degrees;
    std::vector<int> offsets;
    std::vector<int> edges;

    void build(int num_v, const std::vector<std::pair<int, int>>& input_edges) {
        n = num_v;
        m = input_edges.size();
        degrees.assign(n, 0);
        for (const auto& edge : input_edges) {
            degrees[edge.first]++;
            degrees[edge.second]++;
        }
        offsets.resize(n + 1);
        offsets[0] = 0;
        for (int i = 0; i < n; ++i) {
            offsets[i + 1] = offsets[i] + degrees[i];
        }
        edges.resize(2 * m);
        std::vector<int> current_offsets = offsets;
        for (const auto& edge : input_edges) {
            int u = edge.first;
            int v = edge.second;
            edges[current_offsets[u]++] = v;
            edges[current_offsets[v]++] = u;
        }

        // Clean graph: remove self loops and duplicate edges
        std::vector<int> new_edges;
        new_edges.reserve(edges.size());
        std::vector<int> new_offsets(n + 1, 0);

        for (int i = 0; i < n; ++i) {
            int start = offsets[i];
            int end = offsets[i + 1];
            std::sort(edges.begin() + start, edges.begin() + end);
            int last = -1;
            int count = 0;
            for (int j = start; j < end; ++j) {
                int v = edges[j];
                if (v != i && v != last) {
                    new_edges.push_back(v);
                    last = v;
                    count++;
                }
            }
            new_offsets[i + 1] = new_offsets[i] + count;
        }
        edges = std::move(new_edges);
        offsets = std::move(new_offsets);
        for (int i = 0; i < n; ++i) {
            degrees[i] = offsets[i + 1] - offsets[i];
        }
        m = edges.size() / 2;
    }
};

// Greedy coloring helper
bool greedy_coloring(const Graph& g, const std::vector<int>& order, int k, std::vector<int>& colors, std::atomic<bool>& stop_flag) {
    colors.assign(g.n, -1);
    int count = 0;
    for (int u : order) {
        if ((count++ & 1023) == 0) {
            if (stop_flag.load(std::memory_order_relaxed)) {
                return false;
            }
        }
        ColorSet neighbors;
        for (int i = g.offsets[u]; i < g.offsets[u + 1]; ++i) {
            int v = g.edges[i];
            int c = colors[v];
            if (c >= 0) {
                neighbors.set(c);
            }
        }
        int assigned = neighbors.first_unset();
        if (assigned >= k) {
            return false;
        }
        colors[u] = assigned;
    }
    return true;
}

// Local Search (Min-Conflicts / WalkSAT hybrid)
bool local_search_coloring(const Graph& g, int k, std::vector<int>& colors, int max_iterations, unsigned int seed, std::atomic<bool>& stop_flag) {
    std::mt19937 rng(seed);
    
    for (int i = 0; i < g.n; ++i) {
        if (colors[i] < 0 || colors[i] >= k) {
            colors[i] = rng() % k;
        }
    }

    std::vector<int> neighbor_color_count(g.n * k, 0);
    for (int u = 0; u < g.n; ++u) {
        for (int i = g.offsets[u]; i < g.offsets[u + 1]; ++i) {
            int v = g.edges[i];
            int cv = colors[v];
            neighbor_color_count[u * k + cv]++;
        }
    }

    std::vector<int> conflicted_vertices;
    std::vector<int> conflicted_index(g.n, -1);
    
    auto add_conflicted = [&](int v) {
        if (conflicted_index[v] != -1) return;
        conflicted_index[v] = conflicted_vertices.size();
        conflicted_vertices.push_back(v);
    };

    auto remove_conflicted = [&](int v) {
        int idx = conflicted_index[v];
        if (idx == -1) return;
        int last_v = conflicted_vertices.back();
        conflicted_vertices[idx] = last_v;
        conflicted_index[last_v] = idx;
        conflicted_vertices.pop_back();
        conflicted_index[v] = -1;
    };

    for (int u = 0; u < g.n; ++u) {
        int cu = colors[u];
        if (neighbor_color_count[u * k + cu] > 0) {
            add_conflicted(u);
        }
    }

    std::uniform_real_distribution<double> prob_dist(0.0, 1.0);
    const double p_walk = 0.1;

    // Allocate stack buffer for best colors to avoid allocation in inner loop
    int stack_best_colors[1024];
    std::vector<int> heap_best_colors;
    int* best_arr = stack_best_colors;
    if (k > 1024) {
        heap_best_colors.resize(k);
        best_arr = heap_best_colors.data();
    }

    for (int iter = 0; iter < max_iterations; ++iter) {
        if ((iter & 1023) == 0) {
            if (stop_flag.load(std::memory_order_relaxed)) {
                return false;
            }
        }
        if (conflicted_vertices.empty()) {
            return true;
        }

        int u_idx = rng() % conflicted_vertices.size();
        int u = conflicted_vertices[u_idx];

        int current_color = colors[u];
        int new_color = -1;

        if (prob_dist(rng) < p_walk) {
            if (k > 1) {
                new_color = rng() % (k - 1);
                if (new_color >= current_color) {
                    new_color++;
                }
            } else {
                new_color = 0;
            }
        } else {
            int min_conflicts = g.degrees[u] + 1;
            int best_colors_count = 0;

            for (int c = 0; c < k; ++c) {
                int conflicts = neighbor_color_count[u * k + c];
                if (conflicts < min_conflicts) {
                    min_conflicts = conflicts;
                    best_colors_count = 0;
                    best_arr[best_colors_count++] = c;
                } else if (conflicts == min_conflicts) {
                    if (best_colors_count < 1024 || k > 1024) {
                        best_arr[best_colors_count++] = c;
                    }
                }
            }

            if (best_colors_count == 1) {
                new_color = best_arr[0];
            } else if (best_colors_count > 1) {
                new_color = best_arr[rng() % best_colors_count];
            }
        }

        if (new_color != current_color && new_color != -1) {
            colors[u] = new_color;

            for (int i = g.offsets[u]; i < g.offsets[u + 1]; ++i) {
                int v = g.edges[i];
                int cv = colors[v];
                bool was_v_conflicted = (neighbor_color_count[v * k + cv] > 0);

                neighbor_color_count[v * k + current_color]--;
                neighbor_color_count[v * k + new_color]++;

                bool is_v_conflicted = (neighbor_color_count[v * k + cv] > 0);

                if (was_v_conflicted && !is_v_conflicted) {
                    remove_conflicted(v);
                } else if (!was_v_conflicted && is_v_conflicted) {
                    add_conflicted(v);
                }
            }

            bool is_u_conflicted = (neighbor_color_count[u * k + new_color] > 0);
            if (is_u_conflicted) {
                add_conflicted(u);
            } else {
                remove_conflicted(u);
            }
        }
    }

    return conflicted_vertices.empty();
}

// DSATUR-seeded Local Search solver
bool dsatur_seeded_local_search(const Graph& g, int k, std::vector<int>& colors, double noise_range, int max_iters, unsigned int seed, std::atomic<bool>& stop_flag) {
    std::mt19937 rng(seed);
    colors.assign(g.n, -1);
    std::vector<ColorSet> neighbor_colors(g.n);
    std::vector<int> sat_degree(g.n, 0);

    // Max 1024 buckets for saturation degrees (capped at 1023)
    std::vector<std::priority_queue<std::pair<double, int>>> buckets(1024);

    std::vector<double> priorities(g.n);
    std::uniform_real_distribution<double> dist(-noise_range, noise_range);
    for (int i = 0; i < g.n; ++i) {
        double noise = (noise_range > 0.0) ? dist(rng) : 0.0;
        priorities[i] = g.degrees[i] + noise;
        buckets[0].push({priorities[i], i});
    }

    int max_sat = 0;
    int colored_count = 0;

    while (max_sat >= 0) {
        if (buckets[max_sat].empty()) {
            max_sat--;
            continue;
        }
        auto [pri, u] = buckets[max_sat].top();
        buckets[max_sat].pop();

        if (colors[u] != -1) continue;
        int current_bucket = std::min(1023, sat_degree[u]);
        if (current_bucket != max_sat) continue;

        if ((colored_count & 1023) == 0) {
            if (stop_flag.load(std::memory_order_relaxed)) {
                return false;
            }
        }

        int assigned = neighbor_colors[u].first_unset();
        colors[u] = assigned;
        colored_count++;

        for (int i = g.offsets[u]; i < g.offsets[u + 1]; ++i) {
            int v = g.edges[i];
            if (colors[v] == -1) {
                if (!neighbor_colors[v].test(assigned)) {
                    neighbor_colors[v].set(assigned);
                    int new_sat = neighbor_colors[v].count();
                    sat_degree[v] = new_sat;
                    int new_bucket = std::min(1023, new_sat);
                    buckets[new_bucket].push({priorities[v], v});
                    if (new_bucket > max_sat) {
                        max_sat = new_bucket;
                    }
                }
            }
        }
    }

    int max_color_used = 0;
    for (int c : colors) {
        if (c > max_color_used) max_color_used = c;
    }

    if (max_color_used < k) {
        return true;
    }

    // Map colors >= k to random colors in [0, k-1]
    for (int i = 0; i < g.n; ++i) {
        if (colors[i] >= k) {
            colors[i] = rng() % k;
        }
    }

    return local_search_coloring(g, k, colors, max_iters, seed, stop_flag);
}

// Worker thread routine
void worker(int thread_id, const Graph& g, int k, unsigned int seed, 
            std::atomic<bool>& found, std::mutex& sol_mtx, std::vector<int>& final_sol,
            std::atomic<int>& active_threads) {
    active_threads++;
    std::mt19937 rng(seed);
    std::vector<int> colors(g.n, -1);
    bool success = false;

    int strategy = thread_id % 4;
    int max_iters = std::max(500000, g.n * 100);

    if (strategy == 0) {
        double noise = (thread_id == 0) ? 0.0 : (1.0 + (thread_id / 4) * 2.0);
        success = dsatur_seeded_local_search(g, k, colors, noise, max_iters, seed, found);
    } else if (strategy == 1) {
        std::vector<int> order(g.n);
        std::iota(order.begin(), order.end(), 0);
        double noise_range = (thread_id / 4) * 2.0;
        if (noise_range > 0.0) {
            std::vector<double> priorities(g.n);
            std::uniform_real_distribution<double> dist(-noise_range, noise_range);
            for (int i = 0; i < g.n; ++i) {
                priorities[i] = g.degrees[i] + dist(rng);
            }
            std::sort(order.begin(), order.end(), [&](int a, int b) {
                return priorities[a] > priorities[b];
            });
        } else {
            std::sort(order.begin(), order.end(), [&](int a, int b) {
                return g.degrees[a] > g.degrees[b];
            });
        }

        std::vector<int> initial_colors(g.n, -1);
        // Welsh-Powell greedy coloring
        if (greedy_coloring(g, order, k, initial_colors, found)) {
            success = true;
            colors = std::move(initial_colors);
        } else {
            // Map colors >= k to random colors
            for (int i = 0; i < g.n; ++i) {
                if (initial_colors[i] >= k || initial_colors[i] < 0) {
                    initial_colors[i] = rng() % k;
                }
            }
            success = local_search_coloring(g, k, initial_colors, max_iters, seed, found);
            if (success) {
                colors = std::move(initial_colors);
            }
        }
    } else if (strategy == 2) {
        std::vector<int> rand_colors(g.n);
        for (int i = 0; i < g.n; ++i) {
            rand_colors[i] = rng() % k;
        }
        success = local_search_coloring(g, k, rand_colors, max_iters, seed, found);
        if (success) {
            colors = std::move(rand_colors);
        }
    } else if (strategy == 3) {
        std::vector<int> order(g.n);
        std::iota(order.begin(), order.end(), 0);
        int max_trials = std::max(500, 500000 / g.n);
        for (int trial = 0; trial < max_trials && !found.load(std::memory_order_relaxed); ++trial) {
            double noise_range = 1.0 + (rng() % 10) * 2.0;
            std::vector<double> priorities(g.n);
            std::uniform_real_distribution<double> dist(-noise_range, noise_range);
            for (int i = 0; i < g.n; ++i) {
                priorities[i] = g.degrees[i] + dist(rng);
            }
            std::sort(order.begin(), order.end(), [&](int a, int b) {
                return priorities[a] > priorities[b];
            });

            if (greedy_coloring(g, order, k, colors, found)) {
                success = true;
                break;
            }
        }
    }

    if (success) {
        bool expected = false;
        if (found.compare_exchange_strong(expected, true)) {
            std::lock_guard<std::mutex> lock(sol_mtx);
            final_sol = std::move(colors);
        }
    }
    active_threads--;
}

bool validate_solution(const Graph& g, int k, const std::vector<int>& colors) {
    if (colors.size() != (size_t)g.n) return false;
    for (int i = 0; i < g.n; ++i) {
        if (colors[i] < 0 || colors[i] >= k) return false;
    }
    for (int u = 0; u < g.n; ++u) {
        for (int i = g.offsets[u]; i < g.offsets[u + 1]; ++i) {
            int v = g.edges[i];
            if (colors[u] == colors[v]) return false;
        }
    }
    return true;
}

void print_solution(const std::vector<int>& colors) {
    std::vector<char> out_buf;
    out_buf.reserve(colors.size() * 5);
    char temp[12];
    for (int c : colors) {
        int len = 0;
        int val = c;
        if (val == 0) {
            temp[len++] = '0';
        } else {
            int start = len;
            while (val > 0) {
                temp[len++] = '0' + (val % 10);
                val /= 10;
            }
            std::reverse(temp + start, temp + len);
        }
        for (int i = 0; i < len; ++i) {
            out_buf.push_back(temp[i]);
        }
        out_buf.push_back(' ');
    }
    if (!out_buf.empty()) {
        out_buf.back() = '\n';
        fwrite(out_buf.data(), 1, out_buf.size(), stdout);
    }
}

int main() {
    // Fast I/O
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    FastReader reader;
    int n = reader.read_int();
    int m = reader.read_int();
    int k = reader.read_int();

    if (n <= 0) {
        return 0;
    }

    std::vector<std::pair<int, int>> input_edges;
    input_edges.reserve(m);
    for (int i = 0; i < m; ++i) {
        int u = reader.read_int();
        int v = reader.read_int();
        if (u >= 0 && u < n && v >= 0 && v < n && u != v) {
            input_edges.push_back({u, v});
        }
    }

    Graph g;
    g.build(n, input_edges);

    if (k <= 0) {
        std::vector<int> fallback(n, 0);
        print_solution(fallback);
        return 0;
    }

    std::atomic<bool> found{false};
    std::mutex sol_mtx;
    std::vector<int> final_sol;
    std::atomic<int> active_threads{0};

    // Adaptive threading: use standard hardware threads, but limit memory consumption on huge graphs
    unsigned int num_cores = std::thread::hardware_concurrency();
    if (num_cores == 0) num_cores = 4;
    
    // Adaptive thread allocation
    int max_threads = num_cores;
    if (g.n <= 100) max_threads = 1;
    else if (g.n <= 1000) max_threads = std::min(max_threads, 4);
    else if (g.n <= 10000) max_threads = std::min(max_threads, 8);
    else if (g.n <= 100000) max_threads = std::min(max_threads, 16);
    else if (g.n <= 500000) max_threads = std::min(max_threads, 32);
    else max_threads = std::min(max_threads, 8);

    std::random_device rd;
    std::mt19937 rng(rd());

    std::vector<std::thread> threads;
    for (int i = 0; i < max_threads; ++i) {
        threads.emplace_back(worker, i, std::ref(g), k, rng(), std::ref(found), std::ref(sol_mtx), std::ref(final_sol), std::ref(active_threads));
    }

    // Monitor time limit (5 minutes total, exit early to be safe at 290 seconds)
    auto start_time = std::chrono::steady_clock::now();
    while (!found.load() && active_threads.load() > 0) {
        auto current_time = std::chrono::steady_clock::now();
        auto elapsed = std::chrono::duration_cast<std::chrono::seconds>(current_time - start_time).count();
        if (elapsed > 290) {
            found.store(true);
            break;
        }
        std::this_thread::sleep_for(std::chrono::milliseconds(50));
    }

    // Join all threads
    for (auto& t : threads) {
        t.join();
    }

    if (found.load() && !final_sol.empty() && validate_solution(g, k, final_sol)) {
        print_solution(final_sol);
    } else {
        // Try one final quick deterministic fallback just in case
        std::vector<int> fallback_sol;
        std::vector<int> order(g.n);
        std::iota(order.begin(), order.end(), 0);
        std::sort(order.begin(), order.end(), [&](int a, int b) {
            return g.degrees[a] > g.degrees[b];
        });
        std::atomic<bool> dummy_stop{false};
        if (greedy_coloring(g, order, k, fallback_sol, dummy_stop)) {
            print_solution(fallback_sol);
        } else {
            // Ultimate fallback to avoid incomplete/missing output
            for (int i = 0; i < g.n; ++i) {
                if (fallback_sol[i] < 0 || fallback_sol[i] >= k) {
                    fallback_sol[i] = 0;
                }
            }
            print_solution(fallback_sol);
        }
    }

    return 0;
}
