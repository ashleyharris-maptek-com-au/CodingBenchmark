import sys

# Increase recursion depth limit for large inputs
sys.setrecursionlimit(200000)

def solve(X, Y, solution=None):
    if solution is None:
        solution = []
    if not X:
        yield list(solution)
        return

    # Choose the element with the fewest remaining sets containing it (MRV heuristic)
    c = min(X, key=lambda c: len(X[c]))
    for r in list(X[c]):
        solution.append(r)
        cols = select(X, Y, r)
        yield from solve(X, Y, solution)
        deselect(X, Y, r, cols)
        solution.pop()

def select(X, Y, r):
    cols = []
    for j in Y[r]:
        for i in X[j]:
            for k in Y[i]:
                if k != j:
                    X[k].remove(i)
        cols.append(X.pop(j))
    return cols

def deselect(X, Y, r, cols):
    for j in reversed(Y[r]):
        X[j] = cols.pop()
        for i in X[j]:
            for k in Y[i]:
                if k != j:
                    X[k].add(i)

def parse_input():
    lines = sys.stdin.read().splitlines()
    header_idx = -1
    for i, line in enumerate(lines):
        if line.strip():
            header_idx = i
            break
    if header_idx == -1:
        return None, None, []
    
    header = lines[header_idx].strip().split()
    if len(header) < 2:
        return None, None, []
    
    try:
        U = int(header[0])
        S = int(header[1])
    except ValueError:
        return None, None, []
        
    sets = []
    curr_idx = header_idx + 1
    for _ in range(S):
        if curr_idx < len(lines):
            line = lines[curr_idx]
            sets.append([int(x) for x in line.strip().split()])
            curr_idx += 1
        else:
            sets.append([])
    return U, S, sets

def main():
    U, S, sets = parse_input()
    if U is None:
        print("NO SOLUTION")
        return

    # X maps element -> set of set-IDs containing it
    X = {i: set() for i in range(U)}
    # Y maps set-ID -> list of elements
    Y = {}
    
    for r, s in enumerate(sets):
        Y[r] = s
        for val in s:
            if val in X:
                X[val].add(r)

    # Check if there's any element not covered by any set initially
    # If so, there's no solution
    for i in range(U):
        if not X[i]:
            print("NO SOLUTION")
            return

    generator = solve(X, Y)
    try:
        sol = next(generator)
        print("SOLUTION")
        print(" ".join(map(str, sorted(sol))))
    except StopIteration:
        print("NO SOLUTION")

if __name__ == "__main__":
    main()
