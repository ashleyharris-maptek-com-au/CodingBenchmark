using System;
using System.IO;
using System.Text;
using System.Diagnostics;
using System.Collections.Generic;

class Program {
    struct Element {
        public int NodeId;
        public int Cost;
    }

    class MinHeap {
        Element[] heap;
        int size;
        int[] pos;

        public MinHeap(int maxNodes) {
            heap = new Element[maxNodes];
            pos = new int[maxNodes];
            for (int i = 0; i < maxNodes; i++) pos[i] = -1;
            size = 0;
        }

        public void Clear() {
            size = 0;
        }

        public bool IsEmpty => size == 0;

        public void PushOrDecreaseKey(int nodeId, int cost) {
            int idx = pos[nodeId];
            if (idx == -1) {
                heap[size] = new Element { NodeId = nodeId, Cost = cost };
                pos[nodeId] = size;
                SiftUp(size);
                size++;
            } else {
                if (cost < heap[idx].Cost) {
                    heap[idx].Cost = cost;
                    SiftUp(idx);
                }
            }
        }

        public int Pop(out int cost) {
            int rootId = heap[0].NodeId;
            cost = heap[0].Cost;
            pos[rootId] = -1;
            size--;
            if (size > 0) {
                heap[0] = heap[size];
                pos[heap[0].NodeId] = 0;
                SiftDown(0);
            }
            return rootId;
        }

        private void SiftUp(int idx) {
            Element val = heap[idx];
            while (idx > 0) {
                int p = (idx - 1) / 2;
                if (heap[p].Cost <= val.Cost) break;
                heap[idx] = heap[p];
                pos[heap[idx].NodeId] = idx;
                idx = p;
            }
            heap[idx] = val;
            pos[val.NodeId] = idx;
        }

        private void SiftDown(int idx) {
            Element val = heap[idx];
            while (idx * 2 + 1 < size) {
                int left = idx * 2 + 1;
                int right = left + 1;
                int best = left;
                if (right < size && heap[right].Cost < heap[left].Cost) {
                    best = right;
                }
                if (heap[best].Cost >= val.Cost) break;
                heap[idx] = heap[best];
                pos[heap[idx].NodeId] = idx;
                idx = best;
            }
            heap[idx] = val;
            pos[val.NodeId] = idx;
        }
    }

    struct Point {
        public int X;
        public int Y;
        public Point(int x, int y) { X = x; Y = y; }
    }

    class Column {
        public int X;
        public int Y;
        public byte[] Voxels;
        public int Height;
        public int ValuableCount;
        public int MinValuableZ;
        public double Potential;
        public bool IsDumpSite;
        
        // Caching nearest dump site info
        public int CachedDumpCost = -1; // -1: uncalculated, -2: unreachable, >=0: cost
        public List<Point> CachedDumpPath = null;
    }

    class Operation {
        public int DigX, DigY, DigZ;
        public List<Point> Path;
        public int DestX, DestY;
    }

    static int W, H, Z;
    static double bx, by, bz;
    static Column[] columns;
    static int R;

    // Dijkstra structures
    static MinHeap minHeap;
    static int[] distFromPlant;
    static int[] parentFromPlant;
    static int[] genFromPlant;
    static int currentGenFromPlant = 0;

    static int[] distDump;
    static int[] parentDump;
    static int[] genDump;
    static int currentGenDump = 0;

    static int[] dx = { 0, 0, 1, -1 };
    static int[] dy = { 1, -1, 0, 0 };

    static void Main(string[] args) {
        // Optimize standard I/O performance
        Stream stdin = Console.OpenStandardInput();
        BufferedStream bufStdin = new BufferedStream(stdin, 16 * 1024 * 1024); // 16MB buffer
        
        string headerLine = ReadLine(bufStdin);
        if (string.IsNullOrEmpty(headerLine)) return;
        
        string[] parts = headerLine.Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
        W = ParseInt(parts[0]);
        H = ParseInt(parts[1]);
        Z = ParseInt(parts[2]);
        bx = ParseDouble(parts[3]);
        by = ParseDouble(parts[4]);
        bz = ParseDouble(parts[5]);

        // Dynamically compute active region radius R to keep memory and CPU low
        R = W + H;
        for (int r = 1; r < W + H; r++) {
            long cols = 0;
            for (int x = 0; x < W; x++) {
                if (x > r) break;
                int max_y = Math.Min(H - 1, r - x);
                cols += (max_y + 1);
            }
            if (cols > 100000) {
                R = r - 1;
                break;
            }
        }
        R = Math.Max(R, 10);

        int maxNodes = W * H;
        columns = new Column[maxNodes];

        byte[] colBuffer = new byte[Z];
        for (int x = 0; x < W; x++) {
            for (int y = 0; y < H; y++) {
                // Read Z bytes for this column
                int bytesRead = 0;
                while (bytesRead < Z) {
                    int r = bufStdin.Read(colBuffer, bytesRead, Z - bytesRead);
                    if (r <= 0) break;
                    bytesRead += r;
                }

                if (x + y <= R) {
                    Column col = new Column();
                    col.X = x;
                    col.Y = y;
                    col.Voxels = new byte[Z];
                    Array.Copy(colBuffer, col.Voxels, Z);
                    
                    col.Height = -1;
                    col.ValuableCount = 0;
                    for (int z = Z - 1; z >= 0; z--) {
                        byte d = col.Voxels[z];
                        int mass = d % 16;
                        if (mass > 0) {
                            if (col.Height == -1) col.Height = z;
                            int val = d / 16;
                            if (val > 0) col.ValuableCount++;
                        }
                    }

                    columns[x * H + y] = col;
                }
            }
        }

        // Initialize Potentials and Dump sites
        for (int x = 0; x < W; x++) {
            for (int y = 0; y < H; y++) {
                Column col = columns[x * H + y];
                if (col != null) {
                    UpdatePotential(col);
                }
            }
        }

        // Check if there's at least one dump site. If not, convert the lowest potential one.
        int dumpCount = 0;
        for (int x = 0; x < W; x++) {
            for (int y = 0; y < H; y++) {
                Column col = columns[x * H + y];
                if (col != null && col.IsDumpSite) dumpCount++;
            }
        }
        if (dumpCount == 0) {
            Column bestCandidate = null;
            double minPotential = double.MaxValue;
            for (int x = 0; x < W; x++) {
                for (int y = 0; y < H; y++) {
                    if (x == 0 && y == 0) continue;
                    Column col = columns[x * H + y];
                    if (col != null) {
                        if (col.Potential < minPotential) {
                            minPotential = col.Potential;
                            bestCandidate = col;
                        }
                    }
                }
            }
            if (bestCandidate != null) {
                bestCandidate.IsDumpSite = true;
                bestCandidate.Potential = 0;
                bestCandidate.ValuableCount = 0;
            }
        }

        // Initialize Dijkstra structures
        minHeap = new MinHeap(maxNodes);
        distFromPlant = new int[maxNodes];
        parentFromPlant = new int[maxNodes];
        genFromPlant = new int[maxNodes];

        distDump = new int[maxNodes];
        parentDump = new int[maxNodes];
        genDump = new int[maxNodes];

        List<Operation> operations = new List<Operation>();
        Stopwatch sw = Stopwatch.StartNew();

        // Main Simulation Loop
        while (operations.Count < 100000 && sw.ElapsedMilliseconds < 260000) {
            // Step 1: Recompute Dijkstra from (0, 0)
            RunDijkstraFromPlant();

            // Step 2: Evaluate scores for all mine site columns
            Column bestCol = null;
            double bestScore = -1e9;
            List<Point> bestPath = null;
            int bestPathCost = 0;
            bool bestIsValuable = false;

            for (int x = 0; x < W; x++) {
                if (x > R) break;
                for (int y = 0; y <= R - x; y++) {
                    if (y >= H) break;
                    Column col = columns[x * H + y];
                    if (col == null || col.IsDumpSite || col.Height == -1) continue;

                    int z = col.Height;
                    byte d = col.Voxels[z];
                    int mass = d % 16;
                    int val = d / 16;

                    if (val > 0) {
                        // Top voxel is valuable. Find path to (0, 0).
                        if (x == 0 && y == 0) {
                            // Already at plant
                            double score = (double)mass * val * 1000;
                            if (score > bestScore) {
                                bestScore = score;
                                bestCol = col;
                                bestPath = new List<Point>();
                                bestPathCost = 0;
                                bestIsValuable = true;
                            }
                        } else {
                            // Find path using neighbors and distFromPlant
                            int minNeighborCost = 2000000000;
                            int bestN = -1;
                            for (int i = 0; i < 4; i++) {
                                int nx = x + dx[i];
                                int ny = y + dy[i];
                                if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
                                    Column nCol = columns[nx * H + ny];
                                    if (nCol != null) {
                                        int nv = nx * H + ny;
                                        if (genFromPlant[nv] == currentGenFromPlant) {
                                            int dh = (nCol.Height + 1) - z;
                                            if (dh >= -1 && dh <= 1) {
                                                int transitionCost = (dh == 1) ? 5 : (dh == 0 ? 1 : 0);
                                                int total = transitionCost + distFromPlant[nv];
                                                if (total < minNeighborCost) {
                                                    minNeighborCost = total;
                                                    bestN = nv;
                                                }
                                            }
                                        }
                                    }
                                }
                            }

                            if (bestN != -1) {
                                double score = (double)mass * val * 1000 - (double)minNeighborCost * mass;
                                if (score > bestScore) {
                                    bestScore = score;
                                    bestCol = col;
                                    bestIsValuable = true;
                                    bestPathCost = minNeighborCost;
                                    
                                    // Reconstruct path
                                    bestPath = new List<Point>();
                                    int curr = bestN;
                                    while (curr != 0) {
                                        bestPath.Add(new Point(curr / H, curr % H));
                                        curr = parentFromPlant[curr];
                                    }
                                    bestPath.Add(new Point(0, 0));
                                }
                            }
                        }
                    } else {
                        // Top voxel is junk. Find path to nearest dump site.
                        if (col.CachedDumpCost == -1) {
                            int cost;
                            List<Point> path = FindPathToNearestDump(x, y, z, out cost);
                            if (path != null) {
                                col.CachedDumpCost = cost;
                                col.CachedDumpPath = path;
                            } else {
                                col.CachedDumpCost = -2;
                            }
                        }

                        if (col.CachedDumpCost >= 0) {
                            double score = -col.CachedDumpCost * mass + col.Potential / (col.Height - col.MinValuableZ + 1);
                            if (score > bestScore) {
                                bestScore = score;
                                bestCol = col;
                                bestIsValuable = false;
                                bestPath = col.CachedDumpPath;
                                bestPathCost = col.CachedDumpCost;
                            }
                        }
                    }
                }
            }

            // If no profitable or possible move exists, we terminate the simulation
            if (bestCol == null || bestScore <= 0) {
                break;
            }

            // Execute the best operation
            int oldX = bestCol.X;
            int oldY = bestCol.Y;
            int zDig = bestCol.Height;
            byte voxelData = bestCol.Voxels[zDig];
            int massDig = voxelData % 16;
            int valDig = voxelData / 16;

            // Perform Dig
            bestCol.Voxels[zDig] = 0;
            bestCol.Height = -1;
            for (int nz = zDig - 1; nz >= 0; nz--) {
                if ((bestCol.Voxels[nz] % 16) > 0) {
                    bestCol.Height = nz;
                    break;
                }
            }
            if (valDig > 0) bestCol.ValuableCount--;
            UpdatePotential(bestCol);

            Operation op = new Operation();
            op.DigX = oldX;
            op.DigY = oldY;
            op.DigZ = zDig;
            op.Path = bestPath;

            if (bestIsValuable) {
                op.DestX = 0;
                op.DestY = 0;
                InvalidateCacheNear(oldX, oldY);
            } else {
                // Dump at destination
                Point dest = bestPath[bestPath.Count - 1];
                op.DestX = dest.X;
                op.DestY = dest.Y;

                Column dCol = columns[dest.X * H + dest.Y];
                int newZ = dCol.Height + 1;
                if (newZ >= dCol.Voxels.Length) {
                    Array.Resize(ref dCol.Voxels, dCol.Voxels.Length * 2);
                }
                dCol.Voxels[newZ] = voxelData;
                dCol.Height = newZ;
                UpdatePotential(dCol);

                InvalidateCacheNear(oldX, oldY);
                InvalidateCacheNear(dest.X, dest.Y);
            }

            operations.Add(op);
        }

        // Print output
        Stream stdout = Console.OpenStandardOutput();
        using (StreamWriter writer = new StreamWriter(stdout, Encoding.ASCII, 65536)) {
            writer.WriteLine(operations.Count);
            foreach (var op in operations) {
                int K = op.Path.Count;
                if (K == 0) {
                    writer.Write(0);
                    writer.Write(' ');
                    writer.Write(op.DigX);
                    writer.Write(' ');
                    writer.Write(op.DigY);
                    writer.Write(' ');
                    writer.Write(op.DigZ);
                    writer.Write(' ');
                    writer.Write(op.DestX);
                    writer.Write(' ');
                    writer.Write(op.DestY);
                    writer.WriteLine();
                } else {
                    int N = K - 1;
                    writer.Write(N);
                    writer.Write(' ');
                    writer.Write(op.DigX);
                    writer.Write(' ');
                    writer.Write(op.DigY);
                    writer.Write(' ');
                    writer.Write(op.DigZ);
                    for (int i = 0; i < N; i++) {
                        writer.Write(' ');
                        writer.Write(op.Path[i].X);
                        writer.Write(' ');
                        writer.Write(op.Path[i].Y);
                    }
                    writer.Write(' ');
                    writer.Write(op.DestX);
                    writer.Write(' ');
                    writer.Write(op.DestY);
                    writer.WriteLine();
                }
            }
        }
    }

    static void UpdatePotential(Column col) {
        if (col.ValuableCount == 0) {
            col.Potential = 0;
            col.IsDumpSite = (col.X != 0 || col.Y != 0); // (0, 0) can never be a dump site
            return;
        }

        int z_min = -1;
        for (int z = 0; z <= col.Height; z++) {
            byte d = col.Voxels[z];
            int mass = d % 16;
            int val = d / 16;
            if (mass > 0 && val > 0) {
                z_min = z;
                break;
            }
        }

        if (z_min == -1) {
            col.Potential = 0;
            col.IsDumpSite = (col.X != 0 || col.Y != 0);
            return;
        }

        long valuableValue = 0;
        long valuableMass = 0;
        long totalJunkMass = 0;

        for (int z = z_min; z <= col.Height; z++) {
            byte d = col.Voxels[z];
            int mass = d % 16;
            int val = d / 16;
            if (mass > 0) {
                if (val > 0) {
                    valuableMass += mass;
                    valuableValue += (long)mass * val * 1000;
                } else {
                    totalJunkMass += mass;
                }
            }
        }

        double estimatedCost = valuableMass * (col.X + col.Y) * 1.2 + totalJunkMass * Math.Min(col.X + col.Y, 5) * 1.2;
        col.Potential = valuableValue - estimatedCost;
        col.MinValuableZ = z_min;
        col.IsDumpSite = false;
    }

    static void InvalidateCacheNear(int x, int y) {
        int r = 15;
        int xMin = Math.Max(0, x - r);
        int xMax = Math.Min(W - 1, x + r);
        int yMin = Math.Max(0, y - r);
        int yMax = Math.Min(H - 1, y + r);
        for (int cx = xMin; cx <= xMax; cx++) {
            for (int cy = yMin; cy <= yMax; cy++) {
                Column col = columns[cx * H + cy];
                if (col != null) {
                    col.CachedDumpCost = -1;
                    col.CachedDumpPath = null;
                }
            }
        }
    }

    static void RunDijkstraFromPlant() {
        currentGenFromPlant++;
        minHeap.Clear();

        int startNode = 0;
        distFromPlant[startNode] = 0;
        genFromPlant[startNode] = currentGenFromPlant;
        minHeap.PushOrDecreaseKey(startNode, 0);

        while (!minHeap.IsEmpty) {
            int uCost;
            int u = minHeap.Pop(out uCost);
            int ux = u / H;
            int uy = u % H;
            Column uCol = columns[ux * H + uy];
            int uH = uCol.Height;

            for (int i = 0; i < 4; i++) {
                int vx = ux + dx[i];
                int vy = uy + dy[i];
                if (vx >= 0 && vx < W && vy >= 0 && vy < H) {
                    Column vCol = columns[vx * H + vy];
                    if (vCol == null) continue;

                    int v = vx * H + vy;
                    int vH = vCol.Height;
                    int dh = uH - vH;
                    if (dh >= -1 && dh <= 1) {
                        int stepCost = (dh == 1) ? 5 : (dh == 0 ? 1 : 0);
                        int newCost = uCost + stepCost;

                        bool visited = (genFromPlant[v] == currentGenFromPlant);
                        if (!visited || newCost < distFromPlant[v]) {
                            distFromPlant[v] = newCost;
                            genFromPlant[v] = currentGenFromPlant;
                            parentFromPlant[v] = u;
                            minHeap.PushOrDecreaseKey(v, newCost);
                        }
                    }
                }
            }
        }
    }

    static List<Point> FindPathToNearestDump(int startX, int startY, int z, out int pathCost) {
        currentGenDump++;
        minHeap.Clear();

        int startNode = startX * H + startY;
        distDump[startNode] = 0;
        genDump[startNode] = currentGenDump;
        minHeap.PushOrDecreaseKey(startNode, 0);

        int destNode = -1;
        pathCost = 0;

        while (!minHeap.IsEmpty) {
            int uCost;
            int u = minHeap.Pop(out uCost);
            int ux = u / H;
            int uy = u % H;
            Column uCol = columns[ux * H + uy];

            if (uCol.IsDumpSite && u != startNode) {
                destNode = u;
                pathCost = uCost;
                break;
            }

            int uH = uCol.Height;

            for (int i = 0; i < 4; i++) {
                int vx = ux + dx[i];
                int vy = uy + dy[i];
                if (vx >= 0 && vx < W && vy >= 0 && vy < H) {
                    Column vCol = columns[vx * H + vy];
                    if (vCol == null) continue;

                    int v = vx * H + vy;
                    int vH = vCol.Height;
                    
                    int dh;
                    if (u == startNode) {
                        dh = (vH + 1) - z;
                    } else {
                        dh = vH - uH;
                    }

                    if (dh >= -1 && dh <= 1) {
                        int stepCost = (dh == 1) ? 5 : (dh == 0 ? 1 : 0);
                        int newCost = uCost + stepCost;

                        bool visited = (genDump[v] == currentGenDump);
                        if (!visited || newCost < distDump[v]) {
                            distDump[v] = newCost;
                            genDump[v] = currentGenDump;
                            parentDump[v] = u;
                            minHeap.PushOrDecreaseKey(v, newCost);
                        }
                    }
                }
            }
        }

        if (destNode == -1) {
            pathCost = 0;
            return null;
        }

        List<Point> path = new List<Point>();
        int curr = destNode;
        while (curr != startNode) {
            path.Add(new Point(curr / H, curr % H));
            curr = parentDump[curr];
        }
        path.Reverse();
        return path;
    }

    static string ReadLine(Stream stream) {
        List<byte> bytes = new List<byte>();
        while (true) {
            int b = stream.ReadByte();
            if (b == -1 || b == '\n') break;
            if (b != '\r') bytes.Add((byte)b);
        }
        return Encoding.ASCII.GetString(bytes.ToArray());
    }

    static int ParseInt(string s) {
        StringBuilder sb = new StringBuilder();
        foreach (char c in s) {
            if (char.IsDigit(c) || c == '-') sb.Append(c);
        }
        return int.Parse(sb.ToString(), System.Globalization.CultureInfo.InvariantCulture);
    }

    static double ParseDouble(string s) {
        StringBuilder sb = new StringBuilder();
        foreach (char c in s) {
            if (char.IsDigit(c) || c == '-' || c == '.' || c == ',') sb.Append(c);
        }
        string val = sb.ToString().Replace(',', '.');
        return double.Parse(val, System.Globalization.CultureInfo.InvariantCulture);
    }
}
