#pragma pack(push, 1)
struct Transition {
    int next_state;
    int next_trans;
    char c;
};

struct FlatTransition {
    int next_state;
    char c;
};
#pragma pack(pop)

struct State {
    int len;
    int link;
    int head;
};

struct MinimalState {
    int len;
    int link;
};

int get_trans(int u, char c, const std::vector<State>& state, const std::vector<Transition>& trans) {
    int head = state[u].head;
    unsigned char uc = (unsigned char)c;
    while (head != -1) {
        if ((unsigned char)trans[head].c == uc) {
            return trans[head].next_state;
        }
        head = trans[head].next_trans;
    }
    return -1;
}

void add_trans(int u, int v, char c, std::vector<State>& state, std::vector<Transition>& trans) {
    int new_trans_idx = (int)trans.size();
    trans.push_back({v, state[u].head, c});
    state[u].head = new_trans_idx;
}

void redirect_trans(int u, char c, int new_target, const std::vector<State>& state, std::vector<Transition>& trans) {
    int head = state[u].head;
    unsigned char uc = (unsigned char)c;
    while (head != -1) {
        if ((unsigned char)trans[head].c == uc) {
            trans[head].next_state = new_target;
            return;
        }
        head = trans[head].next_trans;
    }
}

void sam_extend(char c, int idx, std::vector<State>& state, std::vector<Transition>& trans, int& sz, int& last, std::vector<int>& first_end) {
    int cur = sz++;
    state.push_back({state[last].len + 1, -1, -1});
    first_end.push_back(idx);
    
    int p = last;
    while (p != -1 && get_trans(p, c, state, trans) == -1) {
        add_trans(p, cur, c, state, trans);
        p = state[p].link;
    }
    
    if (p == -1) {
        state[cur].link = 0;
    } else {
        int q = get_trans(p, c, state, trans);
        if (state[p].len + 1 == state[q].len) {
            state[cur].link = q;
        } else {
            int clone = sz++;
            state.push_back({state[p].len + 1, state[q].link, -1});
            first_end.push_back(first_end[q]);
            
            // Copy transitions
            int head = state[q].head;
            while (head != -1) {
                add_trans(clone, trans[head].next_state, trans[head].c, state, trans);
                head = trans[head].next_trans;
            }
            
            // Redirect transitions
            while (p != -1 && get_trans(p, c, state, trans) == q) {
                redirect_trans(p, c, clone, state, trans);
                p = state[p].link;
            }
            state[q].link = state[cur].link = clone;
        }
    }
    last = cur;
}

#else
__attribute__((always_inline)) inline
#endif
int get_flat_trans(int u, char c, const std::vector<int>& trans_start, const std::vector<uint16_t>& trans_count, const std::vector<FlatTransition>& flat_trans) {
    int start = trans_start[u];
    int count = trans_count[u];
    if (count == 0) return -1;
    
    unsigned char uc = (unsigned char)c;
    if (count <= 8) {
        for (int i = 0; i < count; i++) {
            if ((unsigned char)flat_trans[start + i].c == uc) {
                return flat_trans[start + i].next_state;
            }
        }
        return -1;
    }
    
    int low = 0;
    int high = count - 1;
    while (low <= high) {
        int mid = low + (high - low) / 2;
        unsigned char mid_c = (unsigned char)flat_trans[start + mid].c;
        if (mid_c == uc) {
            return flat_trans[start + mid].next_state;
        } else if (mid_c < uc) {
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }
    return -1;
}

void match_and_propagate(
    const std::string& S,
    int sz,
    const std::vector<MinimalState>& min_state,
    const std::vector<int>& trans_start,
    const std::vector<uint16_t>& trans_count,
    const std::vector<FlatTransition>& flat_trans,
    const std::vector<int>& order,
    std::vector<int>& temp_max,
    std::vector<int>& thread_min,
    const int root_trans[256]
) {
    std::fill(temp_max.begin(), temp_max.end(), 0);
    
    int curr = 0;
    int curr_len = 0;
    for (char c : S) {
        int next = (curr == 0) ? root_trans[(unsigned char)c] : get_flat_trans(curr, c, trans_start, trans_count, flat_trans);
        while (curr != -1 && next == -1) {
            curr = min_state[curr].link;
            if (curr != -1) {
                curr_len = min_state[curr].len;
                next = (curr == 0) ? root_trans[(unsigned char)c] : get_flat_trans(curr, c, trans_start, trans_count, flat_trans);
            }
        }
        if (curr == -1) {
            curr = 0;
            curr_len = 0;
        } else {
            curr = next;
            curr_len++;
        }
        if (curr_len > temp_max[curr]) {
            temp_max[curr] = curr_len;
        }
    }
    
    for (int u : order) {
        if (u == 0) continue;
        int p = min_state[u].link;
        if (p != -1) {
            int val = std::min(min_state[p].len, temp_max[u]);
            if (val > temp_max[p]) {
                temp_max[p] = val;
            }
        }
    }
    
    for (int i = 0; i < sz; i++) {
        if (thread_min[i] > temp_max[i]) {
            thread_min[i] = temp_max[i];
        }
    }
}

std::vector<int> get_propagation_order(int sz, int L1, const std::vector<MinimalState>& min_state) {
    std::vector<int> head_bucket(L1 + 1, -1);
    std::vector<int> next_bucket(sz, -1);
    
    for (int i = 0; i < sz; i++) {
        int l = min_state[i].len;
        if (l >= 0 && l <= L1) {
            next_bucket[i] = head_bucket[l];
            head_bucket[l] = i;
        }
    }
    
    std::vector<int> order;
    order.reserve(sz);
    for (int l = L1; l >= 0; l--) {
        int curr = head_bucket[l];
        while (curr != -1) {
            order.push_back(curr);
            curr = next_bucket[curr];
        }
    }
    return order;
}

// Trivial/Small strategy
void solve_trivial(const std::vector<std::string>& strings) {
    // Find shortest string
    int shortest_idx = 0;
    for (size_t i = 1; i < strings.size(); i++) {
        if (strings[i].length() < strings[shortest_idx].length()) {
            shortest_idx = (int)i;
        }
    }
    
    const std::string& base = strings[shortest_idx];
    int base_len = (int)base.length();
    
    std::string best_substr = "";
    
    // We check substrings starting with the longest possible length
    for (int len = base_len; len >= 1; len--) {
        for (int start = 0; start <= base_len - len; start++) {
            std::string sub = base.substr(start, len);
            bool possible = true;
            for (size_t i = 0; i < strings.size(); i++) {
                if ((int)i == shortest_idx) continue;
                if (strings[i].find(sub) == std::string::npos) {
                    possible = false;
                    break;
                }
            }
            if (possible) {
                std::cout << sub << "\n";
                return;
            }
        }
    }
    std::cout << "\n";
}

int main() {
    // Optimize standard input/output streams for speed
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    int N;
    if (!(std::cin >> N)) {
        std::cout << "\n";
        return 0;
    }
    
    std::string dummy;
    std::getline(std::cin, dummy); // consume remainder of first line
    
    std::vector<std::string> strings;
    strings.reserve(N);
    
    for (int i = 0; i < N; i++) {
        std::string s;
        if (!std::getline(std::cin, s)) {
            break;
        }
        // Remove trailing carriage returns and newlines
        while (!s.empty() && (s.back() == '\r' || s.back() == '\n')) {
            s.pop_back();
        }
        strings.push_back(std::move(s));
    }
    
    if (strings.empty()) {
        std::cout << "\n";
        return 0;
    }
    
    // If N = 1, the LCS is the string itself
    if (strings.size() == 1) {
        std::cout << strings[0] << "\n";
        return 0;
    }
    
    // If any string is empty, the LCS is empty
    for (const auto& s : strings) {
        if (s.empty()) {
            std::cout << "\n";
            return 0;
        }
    }
    
    // Sort strings so that strings[0] is the shortest string (base string for SAM)
    std::sort(strings.begin(), strings.end(), [](const std::string& a, const std::string& b) {
        return a.length() < b.length();
    });
    
    int L1 = (int)strings[0].length();
    
    // Adaptive Logic: choose different strategies based on problem scale
    if (L1 <= 100 && strings.size() <= 5) {
        solve_trivial(strings);
        return 0;
    }
    
    // Build Suffix Automaton of strings[0]
    std::vector<State> state;
    std::vector<Transition> trans;
    std::vector<int> first_end;
    
    state.reserve(2 * L1);
    first_end.reserve(2 * L1);
    trans.reserve(3 * L1);
    
    state.push_back({0, -1, -1});
    first_end.push_back(-1);
    int sz = 1;
    int last = 0;
    
    for (int i = 0; i < L1; i++) {
        sam_extend(strings[0][i], i, state, trans, sz, last, first_end);
    }
    
    // Flatten Suffix Automaton to compact contiguous arrays
    std::vector<int> trans_start(sz, 0);
    std::vector<uint16_t> trans_count(sz, 0);
    
    for (int i = 0; i < sz; i++) {
        int head = state[i].head;
        int count = 0;
        while (head != -1) {
            count++;
            head = trans[head].next_trans;
        }
        trans_count[i] = (uint16_t)count;
    }
    
    int total_trans = 0;
    for (int i = 0; i < sz; i++) {
        trans_start[i] = total_trans;
        total_trans += trans_count[i];
    }
    
    std::vector<FlatTransition> flat_trans(total_trans);
    std::vector<int> current_offset = trans_start;
    
    for (int i = 0; i < sz; i++) {
        int head = state[i].head;
        while (head != -1) {
            int offset = current_offset[i]++;
            flat_trans[offset] = {trans[head].next_state, trans[head].c};
            head = trans[head].next_trans;
        }
        
        std::sort(flat_trans.begin() + trans_start[i], flat_trans.begin() + trans_start[i] + trans_count[i],
                  [](const FlatTransition& a, const FlatTransition& b) {
                      return (unsigned char)a.c < (unsigned char)b.c;
                  });
    }
    
    // Discard large build-only structures
    trans.clear();
    trans.shrink_to_fit();
    
    std::vector<MinimalState> min_state(sz);
    for (int i = 0; i < sz; i++) {
        min_state[i] = {state[i].len, state[i].link};
    }
    state.clear();
    state.shrink_to_fit();
    
    // Get propagation order
    std::vector<int> order = get_propagation_order(sz, L1, min_state);
    
    // Prepare fast root transitions array
    int root_trans[256];
    std::fill(std::begin(root_trans), std::end(root_trans), -1);
    if (sz > 0) {
        for (int i = 0; i < trans_count[0]; i++) {
            unsigned char uc = (unsigned char)flat_trans[trans_start[0] + i].c;
            root_trans[uc] = flat_trans[trans_start[0] + i].next_state;
        }
    }
    
    // Decide number of worker threads
    int num_strings_to_match = (int)strings.size() - 1;
    unsigned int hardware_threads = std::thread::hardware_concurrency();
    if (hardware_threads == 0) hardware_threads = 4;
    
    size_t state_count = sz;
    size_t mem_per_thread = state_count * 2LL * sizeof(int);
    int max_threads_allowed = (int)(8LL * 1024 * 1024 * 1024 / mem_per_thread);
    if (max_threads_allowed < 1) max_threads_allowed = 1;
    
    int num_workers = std::min({(int)hardware_threads, num_strings_to_match, max_threads_allowed, 16});
    if (num_workers < 1) num_workers = 1;
    
    std::vector<std::vector<int>> thread_min(num_workers, std::vector<int>(sz));
    for (int w = 0; w < num_workers; w++) {
        for (int i = 0; i < sz; i++) {
            thread_min[w][i] = min_state[i].len;
        }
    }
    std::vector<std::vector<int>> thread_temp_max(num_workers, std::vector<int>(sz, 0));
    
    std::vector<std::vector<int>> worker_tasks(num_workers);
    for (int i = 1; i < (int)strings.size(); i++) {
        worker_tasks[(i - 1) % num_workers].push_back(i);
    }
    
    std::vector<std::thread> threads;
    threads.reserve(num_workers);
    
    for (int w = 0; w < num_workers; w++) {
        threads.push_back(std::thread([w, &worker_tasks, &strings, sz, &min_state, &trans_start, &trans_count, &flat_trans, &order, &thread_min, &thread_temp_max, &root_trans]() {
            for (int idx : worker_tasks[w]) {
                match_and_propagate(strings[idx], sz, min_state, trans_start, trans_count, flat_trans, order, thread_temp_max[w], thread_min[w], root_trans);
            }
        }));
    }
    
    for (auto& th : threads) {
        th.join();
    }
    
    // Parallel merge of results into global_min
    std::vector<int> global_min(sz);
    std::vector<std::thread> merge_threads;
    merge_threads.reserve(num_workers);
    
    int chunk_size = (sz + num_workers - 1) / num_workers;
    for (int w = 0; w < num_workers; w++) {
        int start = w * chunk_size;
        int end = std::min(start + chunk_size, sz);
        if (start >= end) break;
        
        merge_threads.push_back(std::thread([start, end, num_workers, &thread_min, &global_min, &min_state]() {
            for (int i = start; i < end; i++) {
                int val = min_state[i].len;
                for (int w2 = 0; w2 < num_workers; w2++) {
                    if (thread_min[w2][i] < val) {
                        val = thread_min[w2][i];
                    }
                }
                global_min[i] = val;
            }
        }));
    }
    
    for (auto& th : merge_threads) {
        th.join();
    }
    
    // Find state with maximum common length
    int max_lcs_len = 0;
    int best_state = 0;
    for (int i = 0; i < sz; i++) {
        if (global_min[i] > max_lcs_len) {
            max_lcs_len = global_min[i];
            best_state = i;
        }
    }
    
    if (max_lcs_len == 0) {
        std::cout << "\n";
    } else {
        int end_pos = first_end[best_state];
        int start_pos = end_pos - max_lcs_len + 1;
        std::cout << strings[0].substr(start_pos, max_lcs_len) << "\n";
    }
    
    return 0;
}
