#include <iostream>
#include <vector>
#include <algorithm>
#include <limits>
#include <chrono>
#include <thread>
#include <future>
#include <numeric>

// Struct definitions
struct Rectangle {
    int id; // original index
    int w;
    int h;
};

struct Placement {
    int board_idx;
    int x;
    int y;
    int rotated; // 0 or 1
};

enum class SortOrder {
    AreaDesc,
    MaxSideDesc,
    MinSideDesc,
    PerimeterDesc,
    WidthDesc,
    HeightDesc,
    LongerSideRatioDesc,
    WidthHeightDesc,
    HeightWidthDesc
};

enum class SkylineHeuristic {
    BottomLeft,
    MinWaste
};

struct Segment {
    int x;
    int y;
    int w;
};

struct FreeRect {
    int x;
    int y;
    int w;
    int h;
};

struct PackResult {
    int num_boards;
    std::vector<Placement> placements;
    int last_board_max_y;
};

// Main packing function for a single configuration
PackResult Pack(const std::vector<Rectangle>& input_rects, int board_width, int board_height, SortOrder sort_order, SkylineHeuristic skyline_heuristic) {
    int N = input_rects.size();
    
    // Sort the rectangles
    std::vector<Rectangle> rects = input_rects;
    if (sort_order == SortOrder::AreaDesc) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            return (long long)a.w * a.h > (long long)b.w * b.h;
        });
    } else if (sort_order == SortOrder::MaxSideDesc) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            return std::max(a.w, a.h) > std::max(b.w, b.h);
        });
    } else if (sort_order == SortOrder::MinSideDesc) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            return std::min(a.w, a.h) > std::min(b.w, b.h);
        });
    } else if (sort_order == SortOrder::PerimeterDesc) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            return a.w + a.h > b.w + b.h;
        });
    } else if (sort_order == SortOrder::WidthDesc) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            return a.w > b.w;
        });
    } else if (sort_order == SortOrder::HeightDesc) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            return a.h > b.h;
        });
    } else if (sort_order == SortOrder::LongerSideRatioDesc) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            double rA = (double)std::max(a.w, a.h) * std::max(a.w, a.h) / std::min(a.w, a.h);
            double rB = (double)std::max(b.w, b.h) * std::max(b.w, b.h) / std::min(b.w, b.h);
            return rA > rB;
        });
    } else if (sort_order == SortOrder::WidthHeightDesc) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            if (a.w != b.w) return a.w > b.w;
            return a.h > b.h;
        });
    } else if (sort_order == SortOrder::HeightWidthDesc) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            if (a.h != b.h) return a.h > b.h;
            return a.w > b.w;
        });
    }
    
    std::vector<int> unpacked_indices(N);
    for (int i = 0; i < N; ++i) unpacked_indices[i] = i;
    
    std::vector<Placement> placements(N);
    int board_count = 0;
    
    std::vector<int> rect_min_dim(N);
    for (int i = 0; i < N; ++i) {
        rect_min_dim[i] = std::min(rects[i].w, rects[i].h);
    }
    
    std::vector<int> suffix_min_dim(N);
    if (N > 0) {
        suffix_min_dim[N - 1] = rect_min_dim[N - 1];
        for (int i = N - 2; i >= 0; --i) {
            suffix_min_dim[i] = std::min(rect_min_dim[i], suffix_min_dim[i + 1]);
        }
    }
    
    int last_board_max_y = 0;
    
    while (!unpacked_indices.empty()) {
        board_count++;
        int board_idx = board_count - 1;
        
        std::vector<Segment> skyline;
        skyline.push_back({0, 0, board_width});
        int skyline_min_y = 0;
        
        std::vector<FreeRect> waste_map;
        
        auto InsertSortedWaste = [&](const FreeRect& r) {
            int r_area = r.w * r.h;
            auto it = std::lower_bound(waste_map.begin(), waste_map.end(), r_area, [](const FreeRect& a, int val) {
                return a.w * a.h > val;
            });
            waste_map.insert(it, r);
        };
        
        std::vector<int> next_unpacked_indices;
        next_unpacked_indices.reserve(unpacked_indices.size());
        
        bool board_full = false;
        
        for (size_t k = 0; k < unpacked_indices.size(); ++k) {
            int idx = unpacked_indices[k];
            
            if (board_full) {
                next_unpacked_indices.push_back(idx);
                continue;
            }
            
            int current_min_val = suffix_min_dim[idx];
            
            // Fast check if board is full
            if (skyline_min_y + current_min_val > board_height) {
                bool waste_map_can_fit = false;
                for (const auto& r : waste_map) {
                    if (std::min(r.w, r.h) >= current_min_val) {
                        waste_map_can_fit = true;
                        break;
                    }
                }
                if (!waste_map_can_fit) {
                    board_full = true;
                    next_unpacked_indices.push_back(idx);
                    continue;
                }
            }
            
            const auto& rect = rects[idx];
            int w = rect.w;
            int h = rect.h;
            
            // 1. Try waste map first
            int best_waste_idx = -1;
            int wx = 0, wy = 0;
            bool w_rotated = false;
            int min_waste = std::numeric_limits<int>::max();
            
            for (int i = (int)waste_map.size() - 1; i >= 0; --i) {
                const auto& r = waste_map[i];
                int r_area = r.w * r.h;
                if (r_area - w * h >= min_waste) {
                    break; // Pruning: all remaining free rects have larger area
                }
                // Try normal
                if (w <= r.w && h <= r.h) {
                    int waste = r_area - w * h;
                    if (waste < min_waste) {
                        min_waste = waste;
                        best_waste_idx = i;
                        wx = r.x;
                        wy = r.y;
                        w_rotated = false;
                        if (min_waste == 0) break;
                    }
                }
                // Try rotated
                if (h <= r.w && w <= r.h) {
                    int waste = r_area - w * h;
                    if (waste < min_waste) {
                        min_waste = waste;
                        best_waste_idx = i;
                        wx = r.x;
                        wy = r.y;
                        w_rotated = true;
                        if (min_waste == 0) break;
                    }
                }
            }
            
            if (best_waste_idx != -1) {
                int pw = w_rotated ? h : w;
                int ph = w_rotated ? w : h;
                
                placements[rect.id] = {board_idx, wx, wy, w_rotated ? 1 : 0};
                
                auto F = waste_map[best_waste_idx];
                waste_map.erase(waste_map.begin() + best_waste_idx);
                
                int right_w = F.w - pw;
                int top_h = F.h - ph;
                if (right_w > 0 && top_h > 0) {
                    if (right_w > top_h) {
                        if (right_w >= current_min_val && F.h >= current_min_val) {
                            InsertSortedWaste({F.x + pw, F.y, right_w, F.h});
                        }
                        if (pw >= current_min_val && top_h >= current_min_val) {
                            InsertSortedWaste({F.x, F.y + ph, pw, top_h});
                        }
                    } else {
                        if (right_w >= current_min_val && ph >= current_min_val) {
                            InsertSortedWaste({F.x + pw, F.y, right_w, ph});
                        }
                        if (F.w >= current_min_val && top_h >= current_min_val) {
                            InsertSortedWaste({F.x, F.y + ph, F.w, top_h});
                        }
                    }
                } else if (right_w > 0) {
                    if (right_w >= current_min_val && F.h >= current_min_val) {
                        InsertSortedWaste({F.x + pw, F.y, right_w, F.h});
                    }
                } else if (top_h > 0) {
                    if (F.w >= current_min_val && top_h >= current_min_val) {
                        InsertSortedWaste({F.x, F.y + ph, F.w, top_h});
                    }
                }
                
                if (waste_map.size() > 2000) {
                    waste_map.resize(1000);
                }
                continue;
            }
            
            // 2. Try skyline
            int best_seg_idx = -1;
            int sx = 0, sy = 0;
            int best_score1 = std::numeric_limits<int>::max();
            int best_score2 = std::numeric_limits<int>::max();
            bool s_rotated = false;
            
            auto check_skyline_fit = [&](int rw, int rh, bool rotated) {
                for (size_t i = 0; i < skyline.size(); ++i) {
                    int rx = skyline[i].x;
                    if (rx + rw > board_width) break;
                    
                    if (skyline_heuristic == SkylineHeuristic::BottomLeft) {
                        if (skyline[i].y + rh >= best_score1) continue;
                    }
                    
                    int max_y = 0;
                    int waste = 0;
                    size_t k = i;
                    bool fits = true;
                    while (k < skyline.size() && skyline[k].x < rx + rw) {
                        int old_max_y = max_y;
                        max_y = std::max(max_y, skyline[k].y);
                        if (max_y + rh > board_height) {
                            fits = false;
                            break;
                        }
                        if (max_y > old_max_y && k > i) {
                            int width_before = skyline[k].x - rx;
                            waste += (max_y - old_max_y) * width_before;
                        }
                        int overlap_w = std::min(rx + rw, skyline[k].x + skyline[k].w) - skyline[k].x;
                        waste += (max_y - skyline[k].y) * overlap_w;
                        
                        if (skyline_heuristic == SkylineHeuristic::MinWaste) {
                            if (waste >= best_score1) {
                                fits = false;
                                break;
                            }
                        }
                        k++;
                    }
                    
                    if (fits) {
                        int score1 = 0;
                        int score2 = 0;
                        if (skyline_heuristic == SkylineHeuristic::BottomLeft) {
                            score1 = max_y + rh;
                            score2 = rx;
                        } else { // MinWaste
                            score1 = waste;
                            score2 = max_y + rh;
                        }
                        
                        if (score1 < best_score1 || (score1 == best_score1 && score2 < best_score2)) {
                            best_score1 = score1;
                            best_score2 = score2;
                            best_seg_idx = i;
                            sx = rx;
                            sy = max_y;
                            s_rotated = rotated;
                        }
                    }
                }
            };
            
            check_skyline_fit(w, h, false);
            if (w != h) {
                check_skyline_fit(h, w, true);
            }
            
            if (best_seg_idx != -1) {
                int pw = s_rotated ? h : w;
                int ph = s_rotated ? w : h;
                
                placements[rect.id] = {board_idx, sx, sy, s_rotated ? 1 : 0};
                
                // Add waste under the new rectangle to waste_map
                int rx = sx;
                int rw = pw;
                size_t k = best_seg_idx;
                while (k < skyline.size() && skyline[k].x < rx + rw) {
                    int seg_left = skyline[k].x;
                    int seg_right = seg_left + skyline[k].w;
                    int overlap_left = std::max(rx, seg_left);
                    int overlap_right = std::min(rx + rw, seg_right);
                    int overlap_w = overlap_right - overlap_left;
                    int gap_h = sy - skyline[k].y;
                    if (gap_h > 0 && overlap_w >= current_min_val) {
                        InsertSortedWaste({overlap_left, skyline[k].y, overlap_w, gap_h});
                    }
                    k++;
                }
                
                // Update skyline
                Segment new_seg = {sx, sy + ph, pw};
                size_t i = best_seg_idx;
                size_t j = i;
                while (j < skyline.size() && skyline[j].x < sx + pw) {
                    j++;
                }
                
                bool keep_right_part = false;
                Segment right_part;
                if (skyline[j-1].x + skyline[j-1].w > sx + pw) {
                    keep_right_part = true;
                    int rx_end = sx + pw;
                    right_part.x = rx_end;
                    right_part.y = skyline[j-1].y;
                    right_part.w = (skyline[j-1].x + skyline[j-1].w) - rx_end;
                }
                
                skyline.erase(skyline.begin() + i, skyline.begin() + j);
                skyline.insert(skyline.begin() + i, new_seg);
                if (keep_right_part) {
                    skyline.insert(skyline.begin() + i + 1, right_part);
                }
                
                // Merge adjacent segments
                for (size_t m = 0; m + 1 < skyline.size(); ) {
                    if (skyline[m].y == skyline[m+1].y) {
                        skyline[m].w += skyline[m+1].w;
                        skyline.erase(skyline.begin() + m + 1);
                    } else {
                        m++;
                    }
                }
                
                // Recalculate skyline_min_y
                skyline_min_y = skyline[0].y;
                for (size_t m = 1; m < skyline.size(); ++m) {
                    skyline_min_y = std::min(skyline_min_y, skyline[m].y);
                }
                
                if (waste_map.size() > 2000) {
                    waste_map.resize(1000);
                }
            } else {
                next_unpacked_indices.push_back(idx);
            }
        }
        
        if (next_unpacked_indices.empty()) {
            // Find max y of the last board
            for (const auto& seg : skyline) {
                last_board_max_y = std::max(last_board_max_y, seg.y);
            }
        }
        
        unpacked_indices = std::move(next_unpacked_indices);
    }
    
    return {board_count, placements, last_board_max_y};
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    int N, board_width, board_height;
    if (!(std::cin >> N >> board_width >> board_height)) {
        return 0;
    }
    
    std::vector<Rectangle> rects(N);
    for (int i = 0; i < N; ++i) {
        rects[i].id = i;
        std::cin >> rects[i].w >> rects[i].h;
    }
    
    // Configurations to test
    struct Config {
        SortOrder sort_order;
        SkylineHeuristic skyline_heuristic;
    };
    
    std::vector<Config> configs = {
        {SortOrder::AreaDesc, SkylineHeuristic::MinWaste},
        {SortOrder::AreaDesc, SkylineHeuristic::BottomLeft},
        {SortOrder::MaxSideDesc, SkylineHeuristic::MinWaste},
        {SortOrder::MaxSideDesc, SkylineHeuristic::BottomLeft},
        {SortOrder::MinSideDesc, SkylineHeuristic::MinWaste},
        {SortOrder::MinSideDesc, SkylineHeuristic::BottomLeft},
        {SortOrder::PerimeterDesc, SkylineHeuristic::MinWaste},
        {SortOrder::PerimeterDesc, SkylineHeuristic::BottomLeft},
        {SortOrder::WidthDesc, SkylineHeuristic::MinWaste},
        {SortOrder::HeightDesc, SkylineHeuristic::MinWaste},
        {SortOrder::LongerSideRatioDesc, SkylineHeuristic::MinWaste},
        {SortOrder::LongerSideRatioDesc, SkylineHeuristic::BottomLeft},
        {SortOrder::WidthHeightDesc, SkylineHeuristic::MinWaste},
        {SortOrder::WidthHeightDesc, SkylineHeuristic::BottomLeft},
        {SortOrder::HeightWidthDesc, SkylineHeuristic::MinWaste},
        {SortOrder::HeightWidthDesc, SkylineHeuristic::BottomLeft}
    };
    
    std::vector<std::future<PackResult>> futures;
    for (const auto& config : configs) {
        futures.push_back(std::async(std::launch::async, Pack, std::ref(rects), board_width, board_height, config.sort_order, config.skyline_heuristic));
    }
    
    PackResult best_result;
    best_result.num_boards = std::numeric_limits<int>::max();
    best_result.last_board_max_y = std::numeric_limits<int>::max();
    
    for (auto& f : futures) {
        PackResult res = f.get();
        if (res.num_boards < best_result.num_boards || 
            (res.num_boards == best_result.num_boards && res.last_board_max_y < best_result.last_board_max_y)) {
            best_result = std::move(res);
        }
    }
    
    std::cout << best_result.num_boards << "\n";
    for (int i = 0; i < N; ++i) {
        std::cout << best_result.placements[i].board_idx << " "
                  << best_result.placements[i].x << " "
                  << best_result.placements[i].y << " "
                  << best_result.placements[i].rotated << "\n";
    }
    
    return 0;
}
