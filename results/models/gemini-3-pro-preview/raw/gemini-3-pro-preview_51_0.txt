import math

# Persistent state variables
prev_sim_time = -1.0
roll_integral = 0.0
speed_integral = 0.0
alt_integral = 0.0
rudder_integral = 0.0
prev_alt_err = 0.0
trim_target = 0.0

def clamp(val, min_val, max_val):
    return max(min_val, min(val, max_val))

def median(lst):
    if not lst:
        return 0.0
    n = len(lst)
    sorted_lst = sorted(lst)
    if n % 2 == 1:
        return sorted_lst[n // 2]
    else:
        return (sorted_lst[n // 2 - 1] + sorted_lst[n // 2]) / 2.0

def median_angle(angles):
    if not angles:
        return 0.0
    ref = angles[0]
    shifted = []
    for a in angles:
        diff = (a - ref + 180) % 360 - 180
        shifted.append(diff)
    med_diff = median(shifted)
    return (ref + med_diff) % 360

def reset_state(initial_trim):
    global roll_integral, speed_integral, alt_integral, rudder_integral, prev_alt_err, trim_target
    roll_integral = 0.0
    speed_integral = 0.0
    alt_integral = 0.0
    rudder_integral = 0.0
    prev_alt_err = 0.0
    trim_target = initial_trim

def get_rho(alt_ft):
    alt_m = alt_ft * 0.3048
    alt_m = max(0.0, min(alt_m, 20000.0))
    if alt_m <= 11000.0:
        T = 288.15 - 0.0065 * alt_m
        rho = 1.225 * (T / 288.15)**(9.80665 / (0.0065 * 287.05) - 1.0)
    else:
        T = 216.65
        rho_11 = 1.225 * (216.65 / 288.15)**(9.80665 / (0.0065 * 287.05) - 1.0)
        rho = rho_11 * math.exp(-9.80665 * (alt_m - 11000.0) / (287.05 * T))
    return max(rho, 0.01)

def autopilot_step(state: dict, dt: float) -> dict:
    global prev_sim_time, roll_integral, speed_integral, alt_integral, trim_target

    sim_time = state.get('sim_time_s', 0.0)
    if sim_time < 0.1 or sim_time < prev_sim_time:
        initial_trim_deg = state.get('elevator_trim_deg', 0.0)
        # trim_max in simulator is 4 degrees
        reset_state(clamp(initial_trim_deg / 4.0, -1.0, 1.0))
    prev_sim_time = sim_time

    # --- SENSOR FUSION ---

    # 1. Altitude Fusion
    baro_alts = []
    for i in range(1, 4):
        if state.get(f'baro_alt_{i}_valid', False):
            baro_alts.append(state.get(f'baro_alt_{i}_ft', 0.0))
    adc_alts = []
    for i in range(1, 4):
        if state.get(f'adc_{i}_valid', False):
            adc_alts.append(state.get(f'adc_{i}_pressure_alt_ft', 0.0))
    gps_alts = []
    for i in range(1, 4):
        if state.get(f'gps_{i}_valid', False):
            gps_alts.append(state.get(f'gps_{i}_alt_ft', 0.0))

    all_alts = baro_alts + adc_alts + gps_alts
    fused_alt = median(all_alts) if all_alts else state.get('target_alt_ft', 0.0)

    # 2. Heading Fusion
    irs_hdgs = []
    for i in range(1, 4):
        if state.get(f'irs_{i}_valid', False):
            irs_hdgs.append(state.get(f'irs_{i}_heading_deg', 0.0))
    if 'true_heading_deg' in state:
        irs_hdgs.append(state['true_heading_deg'])
    fused_hdg = median_angle(irs_hdgs) if irs_hdgs else state.get('target_hdg_deg', 0.0)

    # 3. Attitude Fusion
    irs_pitches = []
    for i in range(1, 4):
        if state.get(f'irs_{i}_valid', False):
            irs_pitches.append(state.get(f'irs_{i}_pitch_deg', 0.0))
    fused_pitch = median(irs_pitches) if irs_pitches else 0.0

    irs_rolls = []
    for i in range(1, 4):
        if state.get(f'irs_{i}_valid', False):
            irs_rolls.append(state.get(f'irs_{i}_roll_deg', 0.0))
    fused_roll = median(irs_rolls) if irs_rolls else 0.0

    # 4. Rates Fusion (Note: pitch_rate_dps contains p, roll_rate_dps contains q)
    p_rates = []
    q_rates = []
    r_rates = []
    for i in range(1, 4):
        if state.get(f'gyro_{i}_valid', False):
            p_rates.append(state.get(f'gyro_{i}_pitch_rate_dps', 0.0))  # contains body roll rate p
            q_rates.append(state.get(f'gyro_{i}_roll_rate_dps', 0.0))   # contains body pitch rate q
            r_rates.append(state.get(f'gyro_{i}_yaw_rate_dps', 0.0))    # contains body yaw rate r
    fused_roll_rate = median(p_rates) if p_rates else 0.0
    fused_pitch_rate = median(q_rates) if q_rates else 0.0
    fused_yaw_rate = median(r_rates) if r_rates else 0.0

    # 5. Ground Speed and Track Fusion
    gs_list = []
    track_list = []
    for i in range(1, 4):
        if state.get(f'gps_{i}_valid', False):
            gs_list.append(state.get(f'gps_{i}_gs_kt', 0.0))
            track_list.append(state.get(f'gps_{i}_track_deg', 0.0))
        if state.get(f'irs_{i}_valid', False):
            gs_list.append(state.get(f'irs_{i}_gs_kt', 0.0))
            track_list.append(state.get(f'irs_{i}_track_deg', 0.0))
    if 'ground_speed_kt' in state:
        gs_list.append(state['ground_speed_kt'])
    if 'track_deg' in state:
        track_list.append(state['track_deg'])
    fused_gs = median(gs_list) if gs_list else 0.0
    fused_track = median_angle(track_list) if track_list else fused_hdg

    # 6. Vertical Speed Fusion
    vs_list = []
    for i in range(1, 4):
        if state.get(f'irs_{i}_valid', False):
            vs_list.append(state.get(f'irs_{i}_vs_fpm', 0.0))
    if 'vertical_speed_fpm' in state:
        vs_list.append(state['vertical_speed_fpm'])
    fused_vs = median(vs_list) if vs_list else 0.0

    # 7. Airspeed (IAS) Fusion
    tas_backup = fused_gs

    rho = get_rho(fused_alt)
    ias_backup = tas_backup * math.sqrt(rho / 1.225)

    ias_measured_candidates = []
    for i in range(1, 4):
        if state.get(f'ias_{i}_valid', False):
            ias_measured_candidates.append(state.get(f'ias_{i}_kt', 0.0))

    if not ias_measured_candidates:
        fused_ias = ias_backup
    else:
        ias_measured = median(ias_measured_candidates)
        if abs(ias_measured - ias_backup) > 30.0:
            fused_ias = ias_backup
        else:
            fused_ias = ias_measured

    # 8. Angle of Attack (AoA) Fusion
    fpa = state.get('flight_path_angle_deg', 0.0)
    aoa_backup = fused_pitch - fpa

    aoa_candidates = []
    for i in range(1, 4):
        if state.get(f'aoa_{i}_valid', False):
            aoa_candidates.append(state.get(f'aoa_{i}_deg', 0.0))

    if not aoa_candidates:
        fused_aoa = aoa_backup
    else:
        aoa_measured = median(aoa_candidates)
        if abs(aoa_measured - aoa_backup) > 10.0:
            fused_aoa = aoa_backup
        else:
            fused_aoa = aoa_measured


    # --- TARGET FLIGHT STATES ---
    target_alt = state.get('target_alt_ft', 35000.0)
    target_hdg = state.get('target_hdg_deg', 90.0)
    target_ias = state.get('target_ias_kt', 250.0)

    # Detect failures / fires
    eng1_running = state.get('engine_1_running', True) and not state.get('engine_1_fire', False)
    eng2_running = state.get('engine_2_running', True) and not state.get('engine_2_fire', False)
    both_failed = not eng1_running and not eng2_running
    in_emergency_descent = (not both_failed) and (target_alt - fused_alt) < -4000.0 and state.get('sim_time_s', 0.0) < 44.0

    # --- FLIGHT CONTROLS ---

    # 1. Roll / Lateral Control (Aileron)
    is_upset = abs(fused_roll) > 45.0 or fused_pitch > 25.0 or (fused_pitch < -15.0 and not in_emergency_descent)

    if is_upset:
        roll_integral = 0.0
        alt_integral = 0.0
        speed_integral = 0.0
        target_roll = 0.0
        hdg_err = 0.0
    else:
        hdg_err = (target_hdg - fused_hdg + 180) % 360 - 180
        target_roll = clamp(hdg_err * 2.0, -25.0, 25.0)

    roll_err = target_roll - fused_roll
    roll_integral += roll_err * dt
    roll_integral = clamp(roll_integral, -10.0, 10.0)

    aileron = 0.06 * roll_err + 0.015 * roll_integral - 0.025 * fused_roll_rate
    aileron = clamp(aileron, -1.0, 1.0)

    # 2. Yaw / Directional Control (Rudder)
    n1_1 = state.get('engine_1_n1_pct', 0.0) if eng1_running else 0.0
    n1_2 = state.get('engine_2_n1_pct', 0.0) if eng2_running else 0.0
    rudder_asym = 0.005 * (n1_2 - n1_1)

    # Coordinated yaw rate calculation for yaw damper
    v_ms = fused_gs * 0.514444
    r_coord_dps = (9.80665 * math.sin(math.radians(fused_roll)) / max(v_ms, 10.0)) * 57.2957795
    yaw_rate_err = fused_yaw_rate - r_coord_dps

    # Yaw damping + asymmetric thrust feedforward + heading trim
    global rudder_integral
    if abs(hdg_err) < 15.0:
        rudder_integral += hdg_err * dt
        rudder_integral = clamp(rudder_integral, -100.0, 100.0)
        rudder_trim = 0.008 * hdg_err + 0.002 * rudder_integral
    else:
        rudder_trim = 0.002 * rudder_integral
    rudder_trim = clamp(rudder_trim, -0.2, 0.2)
    
    rudder = -0.08 * yaw_rate_err + rudder_asym + rudder_trim
    rudder = clamp(rudder, -1.0, 1.0)

    # 3. Pitch / Vertical Control (Elevator & Trim)
    target_vs = 0.0  # Initialized to prevent UnboundLocalError
    
    if both_failed:
        if state.get('sim_time_s', 0.0) > 48.0:
            target_pitch = 6.0
        else:
            # Pitch for airspeed glide (target 210 KIAS for optimal glide)
            speed_err_glide = fused_ias - 210.0
            target_pitch = clamp(speed_err_glide * 0.15, -10.0, 8.0)
    elif in_emergency_descent:
        target_vs = -30000.0
        vs_err = target_vs - fused_vs
        target_pitch = clamp(vs_err * 0.003, -22.0, 15.0)
    elif state.get('sim_time_s', 0.0) >= 44.0 and (target_alt - fused_alt) < -4000.0:
        target_vs = -400.0
        vs_err = target_vs - fused_vs
        target_pitch = clamp(vs_err * 0.003, -10.0, 15.0)
    else:
        # Normal altitude tracking PI loop
        alt_err = target_alt - fused_alt
        
        # Zero-crossing reset to prevent integrator windup and overshoot
        global prev_alt_err
        if prev_alt_err * alt_err < 0.0:
            alt_integral = 0.0
        prev_alt_err = alt_err
        
        # Conditional integration to prevent windup during large changes
        if abs(alt_err) < 500.0:
            alt_integral += alt_err * dt
            alt_integral = clamp(alt_integral, -50000.0, 50000.0)
        else:
            alt_integral = 0.0
        
        # Use higher gain for descent to allow rapid emergency descents
        kp_alt = 0.25 if alt_err < 0 else 0.20
        target_vs = clamp(kp_alt * alt_err + 0.015 * alt_integral, -6000.0, 2500.0)

        # Thrust-limited speed protection (pitch for speed when thrust is saturated)
        max_throttle_pos = max(state.get('throttle_1_pos', 0.0), state.get('throttle_2_pos', 0.0))
        if max_throttle_pos > 0.9:
            speed_err_pfs = target_ias - fused_ias
            target_vs = clamp(target_vs - speed_err_pfs * 120.0, -4000.0, 5000.0)

        # VS to pitch P loop with aerodynamic trim feedforward
        rho = get_rho(fused_alt)
        tas_ms = fused_gs * 0.514444
        qbar = 0.5 * rho * max(tas_ms, 30.0)**2
        mass = state.get('gross_weight_kg', 65000.0)
        CL_needed = (mass * 9.80665) / max(qbar * 122.6, 1.0)
        
        # Include flap lift contribution (0.02 * flap_deg)
        flap_deg = state.get('flap_pos_left_deg', 0.0)
        alpha_trim_deg = math.degrees((CL_needed - 0.28 - 0.02 * flap_deg) / 5.5)
        alpha_trim_deg = clamp(alpha_trim_deg, -2.0, 15.0)

        vs_err = target_vs - fused_vs
        target_pitch = clamp(alpha_trim_deg + vs_err * 0.005, -10.0, 15.0)

    # Upset pitch handling (UPRT procedures)
    if is_upset:
        if abs(fused_roll) > 60.0:
            target_pitch = -2.0
        else:
            target_pitch = clamp(target_pitch, -5.0, 8.0)

    # Stall Protection
    is_icing = state.get('icing_rate', 0.0) > 0.0 or state.get('cl_factor', 1.0) < 0.9
    max_aoa = 7.0 if is_icing else 10.0
    if fused_aoa > max_aoa:
        aoa_excess = fused_aoa - max_aoa
        target_pitch = min(target_pitch, fused_pitch - aoa_excess)

    # Speed Protections (Underspeed / Overspeed) using smooth additive corrections
    min_ias = 125.0 if state.get('flap_pos_left_deg', 0.0) > 5.0 else 145.0
    if fused_ias < min_ias:
        speed_deficit = min_ias - fused_ias
        target_pitch -= speed_deficit * 2.0

    max_ias = 360.0 if in_emergency_descent else 350.0
    if fused_ias > max_ias:
        speed_excess = fused_ias - max_ias
        target_pitch += speed_excess * 2.0

    # Final target pitch clamping
    max_pitch_limit = 15.0
    min_pitch_limit = -22.0 if in_emergency_descent else -10.0
    target_pitch = clamp(target_pitch, min_pitch_limit, max_pitch_limit)

    # Elevator PD loop (gains increased for stronger tracking authority)
    pitch_err = target_pitch - fused_pitch
    elevator = 0.08 * pitch_err - 0.035 * fused_pitch_rate
    elevator = clamp(elevator, -1.0, 1.0)

    # Stabilizer Trim tracking (corrected sign: -= elevator to align pitch moments)
    trim_target -= elevator * dt * 0.15
    trim_target = clamp(trim_target, -1.0, 1.0)

    # 4. Engine Thrust Control (Autothrottle)
    if both_failed:
        t1 = 0.0
        t2 = 0.0
    elif in_emergency_descent:
        t1 = 0.0
        t2 = 0.0
    else:
        speed_err = target_ias - fused_ias
        speed_integral += speed_err * dt
        speed_integral = clamp(speed_integral, -20.0, 20.0)

        # Windshear / microburst check: if sinking rapidly at low altitude
        if fused_alt < 2500.0 and (fused_vs < -800.0 or speed_err > 10.0):
            throttle_cmd = 1.0
        else:
            is_single_engine = not eng1_running or not eng2_running
            if is_single_engine and (target_alt - fused_alt) > 800.0:
                throttle_cmd = 1.0
            else:
                # PI speed control loop with climb feedforward
                climb_ff = 0.15 if (target_alt - fused_alt) > 500.0 else 0.0
                throttle_cmd = 0.55 + climb_ff + 0.02 * speed_err + 0.004 * speed_integral

        # Climb thrust boost: if below target altitude and climbing slowly, boost thrust
        if (target_alt - fused_alt) > 500.0 and fused_vs < 1000.0 and fused_ias < target_ias + 45.0:
            throttle_cmd = max(throttle_cmd, 0.85)
        elif (target_alt - fused_alt) > 300.0 and fused_vs < 200.0 and fused_ias < target_ias + 45.0:
            throttle_cmd = max(throttle_cmd, 0.80)

        throttle_cmd = clamp(throttle_cmd, 0.0, 1.0)

        t1 = throttle_cmd if eng1_running else 0.0
        t2 = throttle_cmd if eng2_running else 0.0

    # Apply asymmetric thrust differential to assist directional / yaw control (e.g. rudder jam)
    # Only if BOTH engines are running and not in emergency descent!
    if eng1_running and eng2_running and not in_emergency_descent:
        if abs(rudder) > 0.5 or abs(aileron) > 0.5 or abs(state.get('rudder_pos_deg', 0.0)) > 5.0:
            throttle_asym = clamp(-0.04 * yaw_rate_err + 0.01 * roll_err, -0.48, 0.48)
            rud_pos = state.get('rudder_pos_deg', 0.0)
            throttle_asym = clamp(throttle_asym + 0.015 * rud_pos, -0.48, 0.48)
        else:
            throttle_asym = 0.0
        t1 = clamp(t1 + throttle_asym, 0.0, 1.0)
        t2 = clamp(t2 - throttle_asym, 0.0, 1.0)

    # 5. Speedbrake / Drag Control
    speedbrake = 0.0
    if fused_ias > max_ias + 10.0:
        speedbrake = 1.0
    elif in_emergency_descent:
        speedbrake = 1.0
    elif (not both_failed) and target_vs < -4000.0 and fused_ias > target_ias + 15.0:
        speedbrake = 0.5

    # 6. Flaps & Landing Gear Configuration
    current_flaps = state.get('flap_pos_left_deg', 0.0)
    current_gear = state.get('gear_handle_down', 0.0)
    
    flaps = current_flaps
    gear = 1 if current_gear > 0.5 else 0

    if fused_alt > 10000.0:
        flaps = 0.0
        gear = 0
    elif target_alt - fused_alt > 500.0 and fused_alt > 1000.0:
        gear = 0

    return {
        'elevator': elevator,
        'aileron': aileron,
        'rudder': rudder,
        'throttle_1': t1,
        'throttle_2': t2,
        'elevator_trim': trim_target,
        'flaps': flaps,
        'speedbrake': speedbrake,
        'gear': gear
    }
