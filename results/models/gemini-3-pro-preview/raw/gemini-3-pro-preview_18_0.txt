// High-performance buffered reader for fast I/O
class FastScanner {
private:
    static const int SZ = 65536 * 4;
    char buf[SZ];
    int head = 0;
    int tail = 0;

    char nextChar() {
        if (head >= tail) {
            head = 0;
            tail = fread(buf, 1, SZ, stdin);
            if (tail <= 0) return EOF;
        }
        return buf[head++];
    }

public:
    int readInt() {
        char c = nextChar();
        while (c != EOF && c <= 32) {
            c = nextChar();
        }
        if (c == EOF) return -1;
        int res = 0;
        while (c > 32) {
            if (c >= '0' && c <= '9') {
                res = res * 10 + (c - '0');
            }
            c = nextChar();
        }
        return res;
    }
};

struct Edge {
    int u, v;
};

// Flat Adjacency List for the original graph to minimize memory overhead
struct FlatGraph {
    int N;
    vector<int> head;
    vector<int> next;
    vector<int> to;
    vector<int> edge_idx;

    void init(int n, int m) {
        N = n;
        head.assign(n + 1, -1);
        next.clear(); next.reserve(2 * m);
        to.clear(); to.reserve(2 * m);
        edge_idx.clear(); edge_idx.reserve(2 * m);
    }

    void add_edge(int u, int v, int idx) {
        to.push_back(v);
        edge_idx.push_back(idx);
        next.push_back(head[u]);
        head[u] = (int)to.size() - 1;
    }
};

// Explicit-stack DFS state for bridge finding
struct DFSState {
    int u;
    int parent_edge_idx;
    int edge_ptr;
};

// Find a bridge (1-cut) in O(V + E) time
bool find_bridge_flat(const FlatGraph& g, int N, int M, pair<int, int>& bridge_edge, int& bridge_idx) {
    vector<int> tin(N, -1), low(N, -1);
    int timer = 0;
    
    vector<DFSState> stack;
    stack.reserve(N);

    for (int start = 0; start < N; ++start) {
        if (tin[start] != -1) continue;

        stack.push_back({start, -1, g.head[start]});
        tin[start] = low[start] = timer++;

        while (!stack.empty()) {
            auto& state = stack.back();
            int u = state.u;
            int p_edge = state.parent_edge_idx;
            
            if (state.edge_ptr != -1) {
                int e = state.edge_ptr;
                int v = g.to[e];
                int idx = g.edge_idx[e];
                state.edge_ptr = g.next[e];

                if (idx == p_edge) continue;

                if (tin[v] != -1) {
                    low[u] = min(low[u], tin[v]);
                } else {
                    tin[v] = low[v] = timer++;
                    stack.push_back({v, idx, g.head[v]});
                }
            } else {
                stack.pop_back();
                if (!stack.empty()) {
                    int p = stack.back().u;
                    low[p] = min(low[p], low[u]);
                    if (low[u] > tin[p]) {
                        bridge_idx = p_edge;
                        bridge_edge = {p, u};
                        return true;
                    }
                }
            }
        }
    }
    return false;
}

// Custom heap for localized Maximum Adjacency Search
struct IndexedMaxHeap {
    vector<int> heap;
    vector<int> pos;
    const vector<int>& weight;

    IndexedMaxHeap(int n, const vector<int>& w) : weight(w) {
        pos.assign(n, -1);
    }

    bool empty() const {
        return heap.empty();
    }

    void push_or_update(int u) {
        if (pos[u] == -1) {
            pos[u] = heap.size();
            heap.push_back(u);
            up(heap.size() - 1);
        } else {
            up(pos[u]);
        }
    }

    int pop_max() {
        int res = heap[0];
        pos[res] = -1;
        if (heap.size() > 1) {
            heap[0] = heap.back();
            pos[heap[0]] = 0;
            heap.pop_back();
            down(0);
        } else {
            heap.pop_back();
        }
        return res;
    }

    void clear() {
        for (int u : heap) {
            pos[u] = -1;
        }
        heap.clear();
    }

private:
    void swap_nodes(int i, int j) {
        swap(heap[i], heap[j]);
        pos[heap[i]] = i;
        pos[heap[j]] = j;
    }

    void up(int i) {
        while (i > 0) {
            int p = (i - 1) / 2;
            if (weight[heap[i]] > weight[heap[p]]) {
                swap_nodes(i, p);
                i = p;
            } else {
                break;
            }
        }
    }

    void down(int i) {
        int n = heap.size();
        while (2 * i + 1 < n) {
            int left = 2 * i + 1;
            int right = 2 * i + 2;
            int best = left;
            if (right < n && weight[heap[right]] > weight[heap[left]]) {
                best = right;
            }
            if (weight[heap[best]] > weight[heap[i]]) {
                swap_nodes(i, best);
                i = best;
            } else {
                break;
            }
        }
    }
};

// Local Maximum Adjacency Search starting from start_v
void run_local_mas(const FlatGraph& g, int start_v, int N, int& best_cut, vector<int>& best_partition) {
    static vector<int> weight;
    static vector<bool> in_S;
    static vector<int> modified;
    static IndexedMaxHeap heap(0, weight);

    if (weight.size() < N) {
        weight.assign(N, 0);
        in_S.assign(N, false);
        heap.pos.assign(N, -1);
    }
    modified.clear();
    heap.clear();

    in_S[start_v] = true;
    modified.push_back(start_v);
    
    int current_cut_size = 0;
    for (int e = g.head[start_v]; e != -1; e = g.next[e]) {
        int v = g.to[e];
        current_cut_size++;
        if (!in_S[v]) {
            if (weight[v] == 0) {
                modified.push_back(v);
            }
            weight[v]++;
            heap.push_or_update(v);
        }
    }

    vector<int> current_S = {start_v};
    if (current_cut_size < best_cut) {
        best_cut = current_cut_size;
        best_partition = current_S;
    }

    const int MaxK = min(300, N - 1);
    for (int step = 0; step < MaxK && !heap.empty(); ++step) {
        int u = heap.pop_max();
        in_S[u] = true;
        current_S.push_back(u);

        for (int e = g.head[u]; e != -1; e = g.next[e]) {
            int v = g.to[e];
            if (in_S[v]) {
                current_cut_size--;
            } else {
                current_cut_size++;
                if (weight[v] == 0) {
                    modified.push_back(v);
                }
                weight[v]++;
                heap.push_or_update(v);
            }
        }

        if (current_S.size() < N && current_cut_size < best_cut) {
            best_cut = current_cut_size;
            best_partition = current_S;
        }
    }

    // Reset global arrays
    for (int u : modified) {
        weight[u] = 0;
        in_S[u] = false;
        heap.pos[u] = -1;
    }
}

// Flat Adjacency List for Dinic's Max Flow
struct DinicFlatGraph {
    int N;
    vector<int> head;
    vector<int> next;
    vector<int> to;
    vector<int> cap;
    vector<int> flow;
    vector<int> rev;

    DinicFlatGraph(int n, int m_twice) {
        N = n;
        head.assign(n + 1, -1);
        next.reserve(m_twice);
        to.reserve(m_twice);
        cap.reserve(m_twice);
        flow.reserve(m_twice);
        rev.reserve(m_twice);
    }

    void add_undirected_edge(int u, int v, int c) {
        // u -> v
        to.push_back(v);
        cap.push_back(c);
        flow.push_back(0);
        next.push_back(head[u]);
        int idx_u = (int)to.size() - 1;
        head[u] = idx_u;

        // v -> u
        to.push_back(u);
        cap.push_back(c);
        flow.push_back(0);
        next.push_back(head[v]);
        int idx_v = (int)to.size() - 1;
        head[v] = idx_v;

        rev.push_back(idx_v);
        rev.push_back(idx_u);
    }
};

vector<int> flow_level;
vector<int> flow_ptr;

bool dinic_bfs(const DinicFlatGraph& g, int s, int t) {
    fill(flow_level.begin(), flow_level.end(), -1);
    flow_level[s] = 0;
    queue<int> q;
    q.push(s);
    while (!q.empty()) {
        int v = q.front();
        q.pop();
        for (int e = g.head[v]; e != -1; e = g.next[e]) {
            int tr = g.to[e];
            if (g.cap[e] - g.flow[e] > 0 && flow_level[tr] == -1) {
                flow_level[tr] = flow_level[v] + 1;
                q.push(tr);
            }
        }
    }
    return flow_level[t] != -1;
}

int dinic_dfs(DinicFlatGraph& g, int v, int t, int pushed) {
    if (pushed == 0) return 0;
    if (v == t) return pushed;
    for (int& e = flow_ptr[v]; e != -1; e = g.next[e]) {
        int tr = g.to[e];
        if (flow_level[v] + 1 != flow_level[tr] || g.cap[e] - g.flow[e] == 0) continue;
        int push = dinic_dfs(g, tr, t, min(pushed, g.cap[e] - g.flow[e]));
        if (push == 0) continue;
        g.flow[e] += push;
        g.flow[g.rev[e]] -= push;
        return push;
    }
    return 0;
}

int run_dinic(DinicFlatGraph& g, int s, int t, int cap_limit) {
    fill(g.flow.begin(), g.flow.end(), 0);
    int flow = 0;
    flow_level.resize(g.N);
    flow_ptr.resize(g.N);
    while (flow < cap_limit && dinic_bfs(g, s, t)) {
        for (int i = 0; i < g.N; ++i) {
            flow_ptr[i] = g.head[i];
        }
        while (int pushed = dinic_dfs(g, s, t, cap_limit - flow)) {
            flow += pushed;
            if (flow >= cap_limit) break;
        }
    }
    return flow;
}

vector<int> get_reachable_vertices(const DinicFlatGraph& fg, int s, int t, bool& valid) {
    vector<bool> visited(fg.N, false);
    queue<int> q;
    q.push(s);
    visited[s] = true;
    vector<int> S;
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        S.push_back(u);
        for (int e = fg.head[u]; e != -1; e = fg.next[e]) {
            if (fg.cap[e] - fg.flow[e] > 0 && !visited[fg.to[e]]) {
                visited[fg.to[e]] = true;
                q.push(fg.to[e]);
            }
        }
    }
    valid = !visited[t];
    return S;
}

void run_bfs_distances(const FlatGraph& g, int s, vector<int>& dists) {
    fill(dists.begin(), dists.end(), -1);
    dists[s] = 0;
    queue<int> q;
    q.push(s);
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        int d = dists[u];
        for (int e = g.head[u]; e != -1; e = g.next[e]) {
            int v = g.to[e];
            if (dists[v] == -1) {
                dists[v] = d + 1;
                q.push(v);
            }
        }
    }
}

// Stoer-Wagner exact algorithm for N <= 2000
int run_exact_stoer_wagner(int N, const vector<Edge>& edges, vector<int>& best_partition) {
    vector<vector<int>> adj(N, vector<int>(N, 0));
    for (const auto& e : edges) {
        adj[e.u][e.v]++;
        adj[e.v][e.u]++;
    }

    vector<vector<int>> groups(N);
    for (int i = 0; i < N; ++i) {
        groups[i].push_back(i);
    }

    vector<int> active_list(N);
    iota(active_list.begin(), active_list.end(), 0);

    int best_cut = 1e9;

    vector<int> w(N);
    vector<bool> in_A(N);

    for (int phase = 0; phase < N - 1; ++phase) {
        fill(w.begin(), w.end(), 0);
        fill(in_A.begin(), in_A.end(), false);

        int start = active_list[0];
        in_A[start] = true;
        for (int v : active_list) {
            w[v] = adj[start][v];
        }

        int last = -1, prev = -1;
        int active_size = active_list.size();

        for (int step = 0; step < active_size - 1; ++step) {
            int next_v = -1;
            int max_w = -1;
            for (int v : active_list) {
                if (!in_A[v]) {
                    if (w[v] > max_w) {
                        max_w = w[v];
                        next_v = v;
                    }
                }
            }
            if (next_v == -1) break;
            in_A[next_v] = true;
            prev = last;
            last = next_v;

            for (int v : active_list) {
                if (!in_A[v]) {
                    w[v] += adj[next_v][v];
                }
            }
        }

        if (prev == -1) {
            for (int v : active_list) {
                if (v != last) {
                    prev = v;
                    break;
                }
            }
        }

        int cut_val = w[last];
        if (cut_val < best_cut) {
            best_cut = cut_val;
            best_partition = groups[last];
        }

        for (int v : active_list) {
            if (v != prev && v != last) {
                adj[prev][v] += adj[last][v];
                adj[v][prev] += adj[v][last];
            }
        }
        groups[prev].insert(groups[prev].end(), groups[last].begin(), groups[last].end());

        active_list.erase(remove(active_list.begin(), active_list.end(), last), active_list.end());
    }

    return best_cut;
}

int main() {
    // Fast I/O
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    FastScanner scanner;
    int N = scanner.readInt();
    int M = scanner.readInt();
    if (N == -1) return 0;

    vector<Edge> edges;
    edges.reserve(M);
    
    vector<int> deg(N, 0);
    for (int i = 0; i < M; ++i) {
        int u = scanner.readInt();
        int v = scanner.readInt();
        edges.push_back({u, v});
        deg[u]++;
        deg[v]++;
    }

    // Adaptive threshold for exact vs heuristic
    if (N <= 2000) {
        vector<int> best_partition;
        run_exact_stoer_wagner(N, edges, best_partition);
        
        vector<bool> in_partition(N, false);
        for (int v : best_partition) {
            in_partition[v] = true;
        }

        vector<Edge> cut_edges;
        for (const auto& e : edges) {
            if (in_partition[e.u] != in_partition[e.v]) {
                cut_edges.push_back(e);
            }
        }

        cout << cut_edges.size() << "\n";
        for (const auto& e : cut_edges) {
            cout << e.u << " " << e.v << "\n";
        }
        return 0;
    }

    // Construct flat graph
    FlatGraph g;
    g.init(N, M);
    for (int i = 0; i < M; ++i) {
        g.add_edge(edges[i].u, edges[i].v, i);
        g.add_edge(edges[i].v, edges[i].u, i);
    }

    // Initialize best cut using minimum degree vertex
    int min_deg = 1e9;
    int min_deg_v = -1;
    for (int i = 0; i < N; ++i) {
        if (deg[i] < min_deg) {
            min_deg = deg[i];
            min_deg_v = i;
        }
    }

    int best_cut = min_deg;
    vector<int> best_partition = {min_deg_v};

    // If min degree is 1, it's already a minimum cut
    if (best_cut == 1) {
        vector<bool> in_partition(N, false);
        in_partition[min_deg_v] = true;
        vector<Edge> cut_edges;
        for (const auto& e : edges) {
            if (in_partition[e.u] != in_partition[e.v]) {
                cut_edges.push_back(e);
            }
        }
        cout << cut_edges.size() << "\n";
        for (const auto& e : cut_edges) {
            cout << e.u << " " << e.v << "\n";
        }
        return 0;
    }

    // Check for bridges (1-cut)
    pair<int, int> bridge_edge;
    int bridge_idx = -1;
    if (find_bridge_flat(g, N, M, bridge_edge, bridge_idx)) {
        cout << 1 << "\n";
        cout << bridge_edge.first << " " << bridge_edge.second << "\n";
        return 0;
    }

    // Since there are no bridges, min cut is at least 2.
    // If our min degree cut is 2, it is optimal.
    if (best_cut == 2) {
        vector<bool> in_partition(N, false);
        in_partition[min_deg_v] = true;
        vector<Edge> cut_edges;
        for (const auto& e : edges) {
            if (in_partition[e.u] != in_partition[e.v]) {
                cut_edges.push_back(e);
            }
        }
        cout << cut_edges.size() << "\n";
        for (const auto& e : cut_edges) {
            cout << e.u << " " << e.v << "\n";
        }
        return 0;
    }

    // Localized MAS Search from low-degree vertices
    vector<pair<int, int>> sorted_deg;
    sorted_deg.reserve(N);
    for (int i = 0; i < N; ++i) {
        sorted_deg.push_back({deg[i], i});
    }
    int search_limit = min(50, N);
    nth_element(sorted_deg.begin(), sorted_deg.begin() + search_limit, sorted_deg.end());
    sorted_deg.resize(search_limit);

    for (int i = 0; i < search_limit; ++i) {
        int u = sorted_deg[i].second;
        run_local_mas(g, u, N, best_cut, best_partition);
        if (best_cut == 2) break;
    }

    // Run some random starting vertices too
    if (best_cut > 2) {
        mt19937 rng(42);
        for (int i = 0; i < 50; ++i) {
            int u = rng() % N;
            run_local_mas(g, u, N, best_cut, best_partition);
            if (best_cut == 2) break;
        }
    }

    // Global Max-Flow Capped Search (Furthest Vertex Heuristic)
    if (best_cut > 2) {
        DinicFlatGraph flow_g(N, 2 * M);
        for (int i = 0; i < M; ++i) {
            flow_g.add_undirected_edge(edges[i].u, edges[i].v, 1);
        }

        vector<int> dists(N);
        mt19937 rng(12345);

        const int num_iterations = 5;
        for (int iter = 0; iter < num_iterations && best_cut > 2; ++iter) {
            int s = rng() % N;
            run_bfs_distances(g, s, dists);

            int max_d = 0;
            for (int i = 0; i < N; ++i) {
                max_d = max(max_d, dists[i]);
            }
            if (max_d <= 0) continue;

            vector<int> candidates;
            for (int i = 0; i < N; ++i) {
                if (dists[i] >= max_d - 2 && dists[i] > 0) {
                    candidates.push_back(i);
                }
            }

            shuffle(candidates.begin(), candidates.end(), rng);
            int test_sinks = min(3, (int)candidates.size());
            for (int k = 0; k < test_sinks && best_cut > 2; ++k) {
                int t = candidates[k];
                int flow = run_dinic(flow_g, s, t, best_cut - 1);
                if (flow < best_cut) {
                    bool valid = false;
                    vector<int> S = get_reachable_vertices(flow_g, s, t, valid);
                    if (valid) {
                        best_cut = flow;
                        best_partition = S;
                    }
                }
            }
        }
    }

    // Output results
    vector<bool> in_partition(N, false);
    for (int v : best_partition) {
        in_partition[v] = true;
    }

    vector<Edge> cut_edges;
    for (const auto& e : edges) {
        if (in_partition[e.u] != in_partition[e.v]) {
            cut_edges.push_back(e);
        }
    }

    cout << cut_edges.size() << "\n";
    for (const auto& e : cut_edges) {
        cout << e.u << " " << e.v << "\n";
    }

    return 0;
}
