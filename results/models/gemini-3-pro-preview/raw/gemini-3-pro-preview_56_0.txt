const fs = require('fs');

function distance(x1, y1, x2, y2) {
  const dx = x1 - x2;
  const dy = y1 - y2;
  return Math.sqrt(dx * dx + dy * dy);
}

function solve() {
  const globalStartTime = performance.now();
  const timeLimitMs = 800; // Time limit for optimization in milliseconds

  function isTimeout() {
    return performance.now() - globalStartTime > timeLimitMs;
  }

  let input;
  try {
    input = fs.readFileSync(0, 'utf8');
  } catch (e) {
    process.exit(1);
  }
  
  if (!input || input.trim() === "") {
    process.exit(1);
  }
  
  let data;
  try {
    data = JSON.parse(input);
  } catch (e) {
    process.exit(1);
  }
  
  const depot = data.depot;
  const truckCount = data.truckCount;
  const packages = data.packages;
  
  if (!depot || typeof truckCount !== 'number' || !Array.isArray(packages)) {
    process.exit(1);
  }
  
  const N = packages.length;
  if (N === 0) {
    console.log(JSON.stringify({ routes: [] }));
    return;
  }
  
  const k = Math.min(truckCount, N);
  if (k === 0) {
    console.log(JSON.stringify({ routes: [] }));
    return;
  }
  
  // Coordinates arrays
  const x = new Float64Array(N + 1);
  const y = new Float64Array(N + 1);
  const ids = new Array(N);
  
  for (let i = 0; i < N; i++) {
    x[i] = packages[i].x;
    y[i] = packages[i].y;
    ids[i] = packages[i].id;
  }
  x[N] = depot[0];
  y[N] = depot[1];
  
  function getDistance(i, j) {
    const dx = x[i] - x[j];
    const dy = y[i] - y[j];
    return Math.sqrt(dx * dx + dy * dy);
  }
  
  // K-Means clustering to partition packages into k groups
  const centroidsX = new Float64Array(k);
  const centroidsY = new Float64Array(k);
  
  // Choose packages furthest apart as initial centroids (K-Means++ style)
  const chosen = new Set();
  let firstCentroid = -1;
  let maxD = -1;
  for (let i = 0; i < N; i++) {
    const d = getDistance(i, N);
    if (d > maxD) {
      maxD = d;
      firstCentroid = i;
    }
  }
  centroidsX[0] = x[firstCentroid];
  centroidsY[0] = y[firstCentroid];
  chosen.add(firstCentroid);
  
  for (let c = 1; c < k; c++) {
    let bestPkg = -1;
    let maxMinDist = -1;
    for (let i = 0; i < N; i++) {
      if (chosen.has(i)) continue;
      let minDist = Infinity;
      for (let j = 0; j < c; j++) {
        const dx = x[i] - centroidsX[j];
        const dy = y[i] - centroidsY[j];
        const d = Math.sqrt(dx * dx + dy * dy);
        if (d < minDist) minDist = d;
      }
      if (minDist > maxMinDist) {
        maxMinDist = minDist;
        bestPkg = i;
      }
    }
    centroidsX[c] = x[bestPkg];
    centroidsY[c] = y[bestPkg];
    chosen.add(bestPkg);
  }
  
  let assignments = new Int32Array(N);
  let changed = true;
  let iter = 0;
  const maxIter = 10;
  
  while (changed && iter < maxIter) {
    changed = false;
    iter++;
    
    for (let i = 0; i < N; i++) {
      let bestC = -1;
      let minDist = Infinity;
      for (let c = 0; c < k; c++) {
        const dx = x[i] - centroidsX[c];
        const dy = y[i] - centroidsY[c];
        const d = Math.sqrt(dx * dx + dy * dy);
        if (d < minDist) {
          minDist = d;
          bestC = c;
        }
      }
      if (assignments[i] !== bestC) {
        assignments[i] = bestC;
        changed = true;
      }
    }
    
    if (changed) {
      const counts = new Int32Array(k);
      const sumX = new Float64Array(k);
      const sumY = new Float64Array(k);
      for (let i = 0; i < N; i++) {
        const c = assignments[i];
        sumX[c] += x[i];
        sumY[c] += y[i];
        counts[c]++;
      }
      for (let c = 0; c < k; c++) {
        if (counts[c] > 0) {
          centroidsX[c] = sumX[c] / counts[c];
          centroidsY[c] = sumY[c] / counts[c];
        }
      }
    }
  }
  
  let routes = [];
  for (let r = 0; r < k; r++) {
    routes.push([]);
  }
  for (let i = 0; i < N; i++) {
    routes[assignments[i]].push(i);
  }
  
  // Fill empty routes to ensure all starting routes are non-empty
  for (let r = 0; r < k; r++) {
    if (routes[r].length === 0) {
      let largestR = -1;
      let maxLen = 0;
      for (let j = 0; j < k; j++) {
        if (routes[j].length > maxLen) {
          maxLen = routes[j].length;
          largestR = j;
        }
      }
      if (maxLen > 1) {
        const p = routes[largestR].pop();
        routes[r].push(p);
      }
    }
  }
  
  // Nearest Neighbor routing to initialize
  for (let r = 0; r < k; r++) {
    routes[r] = nearestNeighbor(routes[r]);
  }
  
  function nearestNeighbor(route) {
    if (route.length <= 1) return route;
    const unvisited = new Set(route);
    const newRoute = [];
    let current = N;
    while (unvisited.size > 0) {
      if (isTimeout()) break;
      let nearest = -1;
      let minDist = Infinity;
      for (const node of unvisited) {
        const d = getDistance(current, node);
        if (d < minDist) {
          minDist = d;
          nearest = node;
        }
      }
      newRoute.push(nearest);
      unvisited.delete(nearest);
      current = nearest;
    }
    for (const node of unvisited) {
      newRoute.push(node);
    }
    return newRoute;
  }
  
  function twoOpt(route) {
    const M = route.length;
    if (M <= 1) return route;
    
    const T = new Int32Array(M + 2);
    T[0] = N;
    for (let i = 0; i < M; i++) {
      T[i + 1] = route[i];
    }
    T[M + 1] = N;
    
    let improved = true;
    let passes = 0;
    const maxPasses = 10;
    
    while (improved && passes < maxPasses && !isTimeout()) {
      improved = false;
      passes++;
      for (let i = 1; i <= M - 1; i++) {
        if (isTimeout()) break;
        let u = T[i - 1];
        let v = T[i];
        let dist_u_v = getDistance(u, v);
        
        for (let j = i + 1; j <= M; j++) {
          const w = T[j];
          const z = T[j + 1];
          
          const oldDist = dist_u_v + getDistance(w, z);
          const newDist = getDistance(u, w) + getDistance(v, z);
          
          if (newDist < oldDist - 1e-9) {
            let l = i, r = j;
            while (l < r) {
              const temp = T[l];
              T[l] = T[r];
              T[r] = temp;
              l++;
              r--;
            }
            improved = true;
            v = T[i];
            dist_u_v = getDistance(u, v);
          }
        }
      }
    }
    
    const newRoute = [];
    for (let i = 1; i <= M; i++) {
      newRoute.push(T[i]);
    }
    return newRoute;
  }
  
  // Optimize each route with 2-opt initially
  for (let r = 0; r < k; r++) {
    if (isTimeout()) break;
    routes[r] = twoOpt(routes[r]);
  }
  
  // Inter-route optimization (relocate and swap)
  let outerImproved = true;
  while (outerImproved && !isTimeout()) {
    outerImproved = false;
    let interImproved = true;
    const modified = new Uint8Array(k);
    
    while (interImproved && !isTimeout()) {
      interImproved = false;
      
      for (let r1 = 0; r1 < k; r1++) {
        if (isTimeout()) break;
        const route1 = routes[r1];
        const M1 = route1.length;
        if (M1 === 0) continue;
        
        const T1 = [N, ...route1, N];
        
        for (let i = 1; i <= M1; i++) {
          const p1 = T1[i];
          
          // Relocation (allows routes to become empty)
          if (M1 >= 1) {
            for (let r2 = 0; r2 < k; r2++) {
              if (r1 === r2) continue;
              const route2 = routes[r2];
              const M2 = route2.length;
              const T2 = [N, ...route2, N];
              
              for (let j = 0; j <= M2; j++) {
                const cost_remove = getDistance(T1[i-1], T1[i+1]) - (getDistance(T1[i-1], p1) + getDistance(p1, T1[i+1]));
                const cost_insert = getDistance(T2[j], p1) + getDistance(p1, T2[j+1]) - getDistance(T2[j], T2[j+1]);
                const change = cost_remove + cost_insert;
                
                if (change < -1e-9) {
                  route1.splice(i - 1, 1);
                  route2.splice(j, 0, p1);
                  modified[r1] = 1;
                  modified[r2] = 1;
                  interImproved = true;
                  outerImproved = true;
                  break;
                }
              }
              if (interImproved) break;
            }
          }
          if (interImproved) break;
          
          // Swap
          for (let r2 = 0; r2 < k; r2++) {
            if (r1 === r2) continue;
            const route2 = routes[r2];
            const M2 = route2.length;
            if (M2 === 0) continue;
            const T2 = [N, ...route2, N];
            
            for (let j = 1; j <= M2; j++) {
              const p2 = T2[j];
              const cost1 = getDistance(T1[i-1], p2) + getDistance(p2, T1[i+1]) - (getDistance(T1[i-1], p1) + getDistance(p1, T1[i+1]));
              const cost2 = getDistance(T2[j-1], p1) + getDistance(p1, T2[j+1]) - (getDistance(T2[j-1], p2) + getDistance(p2, T2[j+1]));
              const change = cost1 + cost2;
              
              if (change < -1e-9) {
                route1[i - 1] = p2;
                route2[j - 1] = p1;
                modified[r1] = 1;
                modified[r2] = 1;
                interImproved = true;
                outerImproved = true;
                break;
              }
            }
            if (interImproved) break;
          }
          if (interImproved) break;
        }
        if (interImproved) break;
      }
    }
    
    // Re-run 2-opt on modified routes
    for (let r = 0; r < k; r++) {
      if (isTimeout()) break;
      if (modified[r] === 1) {
        routes[r] = twoOpt(routes[r]);
      }
    }
  }
  
  // Format output
  const outputRoutes = [];
  for (let r = 0; r < k; r++) {
    if (routes[r].length > 0) {
      outputRoutes.push(routes[r].map(idx => ids[idx]));
    }
  }
  
  console.log(JSON.stringify({ routes: outputRoutes }));
}

solve();
