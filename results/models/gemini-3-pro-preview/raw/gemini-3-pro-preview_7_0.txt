struct csgjs_vector {
    float x, y, z;
    csgjs_vector() : x(0.0f), y(0.0f), z(0.0f) {}
    explicit csgjs_vector(float x, float y, float z) : x(x), y(y), z(z) {}
};

inline static csgjs_vector operator + (const csgjs_vector & a, const csgjs_vector & b) { return csgjs_vector(a.x + b.x, a.y + b.y, a.z + b.z); }
inline static csgjs_vector operator - (const csgjs_vector & a, const csgjs_vector & b) { return csgjs_vector(a.x - b.x, a.y - b.y, a.z - b.z); }
inline static csgjs_vector operator * (const csgjs_vector & a, float b) { return csgjs_vector(a.x * b, a.y * b, a.z * b); }
inline static csgjs_vector operator / (const csgjs_vector & a, float b) { return a * (1.0f / b); }
inline static float dot(const csgjs_vector & a, const csgjs_vector & b) { return a.x * b.x + a.y * b.y + a.z * b.z; }
inline static float length(const csgjs_vector & a) { return sqrtf(dot(a, a)); }
inline static csgjs_vector unit(const csgjs_vector & a) {
    float len = length(a);
    if (len < 1e-8f) return csgjs_vector(0, 0, 0);
    return a / len;
}
inline static csgjs_vector cross(const csgjs_vector & a, const csgjs_vector & b) { return csgjs_vector(a.y * b.z - a.z * b.y, a.z * b.x - a.x * b.z, a.x * b.y - a.y * b.x); }

struct csgjs_vertex {
    csgjs_vector pos;
};

inline static csgjs_vertex interpolate(const csgjs_vertex & a, const csgjs_vertex & b, float t) {
    csgjs_vertex ret;
    ret.pos = a.pos + (b.pos - a.pos) * t;
    return ret;
}

struct csgjs_model {
    std::vector<csgjs_vertex> vertices;
    std::vector<int> indices;
};

struct AABB {
    csgjs_vector min_p, max_p;
    AABB() {
        min_p = csgjs_vector(1e30f, 1e30f, 1e30f);
        max_p = csgjs_vector(-1e30f, -1e30f, -1e30f);
    }
    AABB(const csgjs_vector& p) : min_p(p), max_p(p) {}
    void extend(const csgjs_vector& p) {
        min_p.x = std::min(min_p.x, p.x);
        min_p.y = std::min(min_p.y, p.y);
        min_p.z = std::min(min_p.z, p.z);
        max_p.x = std::max(max_p.x, p.x);
        max_p.y = std::max(max_p.y, p.y);
        max_p.z = std::max(max_p.z, p.z);
    }
    void extend(const AABB& b) {
        extend(b.min_p);
        extend(b.max_p);
    }
    bool intersects(const AABB& b) const {
        return min_p.x <= b.max_p.x && max_p.x >= b.min_p.x &&
               min_p.y <= b.max_p.y && max_p.y >= b.min_p.y &&
               min_p.z <= b.max_p.z && max_p.z >= b.min_p.z;
    }
};

struct AABBNode {
    AABB box;
    int left = -1;
    int right = -1;
    int tri_start = -1;
    int tri_end = -1;
};

bool ray_box_intersect(const csgjs_vector& ray_origin, const csgjs_vector& ray_dir, const AABB& box) {
    float tmin = -1e30f, tmax = 1e30f;
    
    auto clip_axis = [&](float origin, float dir, float min_val, float max_val) {
        if (fabs(dir) < 1e-8f) {
            if (origin < min_val || origin > max_val) return false;
            return true;
        }
        float t1 = (min_val - origin) / dir;
        float t2 = (max_val - origin) / dir;
        if (t1 > t2) std::swap(t1, t2);
        tmin = std::max(tmin, t1);
        tmax = std::min(tmax, t2);
        return tmin <= tmax;
    };
    
    if (!clip_axis(ray_origin.x, ray_dir.x, box.min_p.x, box.max_p.x)) return false;
    if (!clip_axis(ray_origin.y, ray_dir.y, box.min_p.y, box.max_p.y)) return false;
    if (!clip_axis(ray_origin.z, ray_dir.z, box.min_p.z, box.max_p.z)) return false;
    
    return tmax >= 0.0f;
}

struct AABBTree {
    std::vector<AABBNode> nodes;
    std::vector<int> tri_indices;
    const csgjs_model* model;
    
    void query_box(int node_idx, const AABB& query, std::vector<int>& result) const {
        if (node_idx < 0 || node_idx >= (int)nodes.size()) return;
        if (!nodes[node_idx].box.intersects(query)) return;
        
        if (nodes[node_idx].left == -1) {
            for (int i = nodes[node_idx].tri_start; i < nodes[node_idx].tri_end; ++i) {
                result.push_back(tri_indices[i]);
            }
        } else {
            query_box(nodes[node_idx].left, query, result);
            query_box(nodes[node_idx].right, query, result);
        }
    }

    void query_ray(const csgjs_vector& ray_origin, const csgjs_vector& ray_dir,
                   int node_idx, std::vector<int>& result) const {
        if (node_idx < 0 || node_idx >= (int)nodes.size()) return;
        if (!ray_box_intersect(ray_origin, ray_dir, nodes[node_idx].box)) return;
        
        if (nodes[node_idx].left == -1) {
            for (int i = nodes[node_idx].tri_start; i < nodes[node_idx].tri_end; ++i) {
                result.push_back(tri_indices[i]);
            }
        } else {
            query_ray(ray_origin, ray_dir, nodes[node_idx].left, result);
            query_ray(ray_origin, ray_dir, nodes[node_idx].right, result);
        }
    }
};

AABBTree build_tree(const csgjs_model& model) {
    AABBTree tree;
    tree.model = &model;
    int num_tris = model.indices.size() / 3;
    if (num_tris == 0) return tree;
    
    struct TriInfo {
        int idx;
        AABB box;
        csgjs_vector centroid;
    };
    std::vector<TriInfo> tris(num_tris);
    for (int i = 0; i < num_tris; ++i) {
        tris[i].idx = i;
        const auto& v1 = model.vertices[model.indices[3 * i]].pos;
        const auto& v2 = model.vertices[model.indices[3 * i + 1]].pos;
        const auto& v3 = model.vertices[model.indices[3 * i + 2]].pos;
        tris[i].box = AABB(v1);
        tris[i].box.extend(v2);
        tris[i].box.extend(v3);
        tris[i].centroid = (v1 + v2 + v3) / 3.0f;
    }
    
    tree.tri_indices.resize(num_tris);
    for (int i = 0; i < num_tris; ++i) tree.tri_indices[i] = i;
    
    auto build_recursive = [&](auto& self, int start, int end) -> int {
        int node_idx = tree.nodes.size();
        tree.nodes.push_back(AABBNode());
        
        AABB box;
        for (int i = start; i < end; ++i) {
            box.extend(tris[tree.tri_indices[i]].box);
        }
        tree.nodes[node_idx].box = box;
        
        int count = end - start;
        if (count <= 4) {
            tree.nodes[node_idx].tri_start = start;
            tree.nodes[node_idx].tri_end = end;
            return node_idx;
        }
        
        csgjs_vector extent = box.max_p - box.min_p;
        int axis = 0;
        if (extent.y > extent.x && extent.y > extent.z) axis = 1;
        else if (extent.z > extent.x && extent.z > extent.y) axis = 2;
        
        int mid = start + count / 2;
        std::nth_element(tree.tri_indices.begin() + start, tree.tri_indices.begin() + mid, tree.tri_indices.begin() + end, [&](int a, int b) {
            if (axis == 0) return tris[a].centroid.x < tris[b].centroid.x;
            if (axis == 1) return tris[a].centroid.y < tris[b].centroid.y;
            return tris[a].centroid.z < tris[b].centroid.z;
        });
        
        tree.nodes[node_idx].left = self(self, start, mid);
        tree.nodes[node_idx].right = self(self, mid, end);
        return node_idx;
    };
    
    build_recursive(build_recursive, 0, num_tris);
    return tree;
}

bool is_point_inside_mesh(const csgjs_vector& p, const AABBTree& tree, unsigned int& seed) {
    if (tree.nodes.empty()) return false;
    float diag = length(tree.nodes[0].box.max_p - tree.nodes[0].box.min_p);
    if (diag < 1e-5f) diag = 1.0f;
    
    auto rand_float = [&](unsigned int& s) {
        s = s * 1664525 + 1013904223;
        return (float)(s & 0xFFFFFF) / 16777216.0f;
    };
    
    for (int attempt = 0; attempt < 15; ++attempt) {
        float theta = rand_float(seed) * 2.0f * 3.14159265f;
        float phi = acos(1.0f - 2.0f * rand_float(seed));
        csgjs_vector dir(sin(phi) * cos(theta), sin(phi) * sin(theta), cos(phi));
        
        std::vector<int> candidates;
        tree.query_ray(p, dir, 0, candidates);
        
        bool degenerate = false;
        std::vector<float> intersection_ts;
        
        for (int tri_idx : candidates) {
            const auto& v0 = tree.model->vertices[tree.model->indices[3 * tri_idx]].pos;
            const auto& v1 = tree.model->vertices[tree.model->indices[3 * tri_idx + 1]].pos;
            const auto& v2 = tree.model->vertices[tree.model->indices[3 * tri_idx + 2]].pos;
            
            csgjs_vector edge1 = v1 - v0;
            csgjs_vector edge2 = v2 - v0;
            csgjs_vector h = cross(dir, edge2);
            float a = dot(edge1, h);
            if (fabs(a) < 1e-8f) {
                continue;
            }
            
            float f = 1.0f / a;
            csgjs_vector s = p - v0;
            float u = f * dot(s, h);
            
            csgjs_vector q = cross(s, edge1);
            float v = f * dot(dir, q);
            float t = f * dot(edge2, q);
            
            if (u < -1e-5f || u > 1.0f + 1e-5f) continue;
            if (v < -1e-5f || u + v > 1.0f + 1e-5f) continue;
            
            if (fabs(t) < 1e-5f * diag) {
                degenerate = true;
                break;
            }
            
            if (t >= 1e-5f * diag) {
                const float boundary_eps = 1e-4f;
                if (u < boundary_eps || v < boundary_eps || (1.0f - u - v) < boundary_eps) {
                    degenerate = true;
                    break;
                }
                intersection_ts.push_back(t);
            }
        }
        
        if (degenerate) continue;
        
        std::sort(intersection_ts.begin(), intersection_ts.end());
        int unique_count = 0;
        float last_t = -1.0f;
        for (float t : intersection_ts) {
            if (last_t < 0.0f || t - last_t > 1e-5f * diag) {
                unique_count++;
                last_t = t;
            } else {
                degenerate = true;
                break;
            }
        }
        
        if (degenerate) continue;
        
        return (unique_count % 2) != 0;
    }
    return false;
}

bool intersect_triangles(const csgjs_vertex& a1, const csgjs_vertex& a2, const csgjs_vertex& a3,
                         const csgjs_vertex& b1, const csgjs_vertex& b2, const csgjs_vertex& b3,
                         csgjs_vector& out_p1, csgjs_vector& out_p2) {
    csgjs_vector nA = cross(a2.pos - a1.pos, a3.pos - a1.pos);
    float len_nA = length(nA);
    if (len_nA < 1e-8f) return false;
    nA = nA / len_nA;
    float wA = dot(nA, a1.pos);
    
    csgjs_vector nB = cross(b2.pos - b1.pos, b3.pos - b1.pos);
    float len_nB = length(nB);
    if (len_nB < 1e-8f) return false;
    nB = nB / len_nB;
    float wB = dot(nB, b1.pos);
    
    csgjs_vector d = cross(nA, nB);
    float len_d = length(d);
    if (len_d < 1e-6f) return false;
    
    float da1 = dot(nB, a1.pos) - wB;
    float da2 = dot(nB, a2.pos) - wB;
    float da3 = dot(nB, a3.pos) - wB;
    
    if ((da1 > 0 && da2 > 0 && da3 > 0) || (da1 < 0 && da2 < 0 && da3 < 0)) return false;
    
    float db1 = dot(nA, b1.pos) - wA;
    float db2 = dot(nA, b2.pos) - wA;
    float db3 = dot(nA, b3.pos) - wA;
    
    if ((db1 > 0 && db2 > 0 && db3 > 0) || (db1 < 0 && db2 < 0 && db3 < 0)) return false;
    
    csgjs_vector ptsA[2];
    int countA = 0;
    
    auto intersect_edge_plane = [&](const csgjs_vector& v1, const csgjs_vector& v2, float d1, float d2) {
        if ((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) {
            float t = -d1 / (d2 - d1);
            if (countA < 2) {
                ptsA[countA++] = v1 + (v2 - v1) * t;
            }
        }
    };
    
    intersect_edge_plane(a1.pos, a2.pos, da1, da2);
    intersect_edge_plane(a2.pos, a3.pos, da2, da3);
    intersect_edge_plane(a3.pos, a1.pos, da3, da1);
    
    if (countA < 2) return false;
    
    csgjs_vector ptsB[2];
    int countB = 0;
    
    auto intersect_edge_plane_B = [&](const csgjs_vector& v1, const csgjs_vector& v2, float d1, float d2) {
        if ((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) {
            float t = -d1 / (d2 - d1);
            if (countB < 2) {
                ptsB[countB++] = v1 + (v2 - v1) * t;
            }
        }
    };
    
    intersect_edge_plane_B(b1.pos, b2.pos, db1, db2);
    intersect_edge_plane_B(b2.pos, b3.pos, db2, db3);
    intersect_edge_plane_B(b3.pos, b1.pos, db3, db1);
    
    if (countB < 2) return false;
    
    csgjs_vector dir = d / len_d;
    int axis = 0;
    if (fabs(dir.y) > fabs(dir.x) && fabs(dir.y) > fabs(dir.z)) axis = 1;
    else if (fabs(dir.z) > fabs(dir.x) && fabs(dir.z) > fabs(dir.y)) axis = 2;
    
    auto get_coord = [&](const csgjs_vector& p) {
        if (axis == 0) return p.x;
        if (axis == 1) return p.y;
        return p.z;
    };
    
    float al1 = get_coord(ptsA[0]);
    float al2 = get_coord(ptsA[1]);
    if (al1 > al2) { std::swap(al1, al2); std::swap(ptsA[0], ptsA[1]); }
    
    float bl1 = get_coord(ptsB[0]);
    float bl2 = get_coord(ptsB[1]);
    if (bl1 > bl2) { std::swap(bl1, bl2); std::swap(ptsB[0], ptsB[1]); }
    
    float start = std::max(al1, bl1);
    float end = std::min(al2, bl2);
    
    if (start + 1e-5f >= end) return false;
    
    auto interpolate_3d = [&](const csgjs_vector& p1, const csgjs_vector& p2, float l1, float l2, float target_l) {
        if (fabs(l2 - l1) < 1e-8f) return p1;
        float t = (target_l - l1) / (l2 - l1);
        return p1 + (p2 - p1) * t;
    };
    
    out_p1 = interpolate_3d(ptsA[0], ptsA[1], al1, al2, start);
    out_p2 = interpolate_3d(ptsA[0], ptsA[1], al1, al2, end);
    return true;
}

struct Polygon {
    std::vector<csgjs_vertex> vertices;
};

void clip_polygon_by_plane(const Polygon& poly, const csgjs_vector& plane_normal, float plane_w,
                           Polygon& front, Polygon& back, float diag) {
    int n = poly.vertices.size();
    if (n == 0) return;
    
    float eps = 1e-5f * diag;
    std::vector<float> dists(n);
    bool has_front = false;
    bool has_back = false;
    for (int i = 0; i < n; ++i) {
        dists[i] = dot(plane_normal, poly.vertices[i].pos) - plane_w;
        if (dists[i] > eps) has_front = true;
        if (dists[i] < -eps) has_back = true;
    }
    
    if (!has_front && !has_back) {
        front = poly;
        back.vertices.clear();
        return;
    }
    if (has_front && !has_back) {
        front = poly;
        back.vertices.clear();
        return;
    }
    if (!has_front && has_back) {
        front.vertices.clear();
        back = poly;
        return;
    }
    
    std::vector<csgjs_vertex> f_verts, b_verts;
    for (int i = 0; i < n; ++i) {
        int j = (i + 1) % n;
        const auto& vi = poly.vertices[i];
        const auto& vj = poly.vertices[j];
        float di = dists[i];
        float dj = dists[j];
        
        if (di >= -eps) f_verts.push_back(vi);
        if (di <= eps) b_verts.push_back(vi);
        
        if ((di > eps && dj < -eps) || (di < -eps && dj > eps)) {
            float t = di / (di - dj);
            csgjs_vertex v = interpolate(vi, vj, t);
            f_verts.push_back(v);
            b_verts.push_back(v);
        }
    }
    
    auto clean_poly = [eps](const std::vector<csgjs_vertex>& verts) -> Polygon {
        Polygon res;
        for (const auto& v : verts) {
            if (res.vertices.empty() || length(res.vertices.back().pos - v.pos) > eps) {
                res.vertices.push_back(v);
            }
        }
        if (res.vertices.size() >= 3 && length(res.vertices.back().pos - res.vertices.front().pos) <= eps) {
            res.vertices.pop_back();
        }
        return res;
    };
    
    front = clean_poly(f_verts);
    back = clean_poly(b_verts);
}

void insert_intersection_points(Polygon& poly, const std::vector<csgjs_vector>& P_ints, float diag) {
    std::vector<csgjs_vertex> new_verts;
    int n = poly.vertices.size();
    if (n < 3) return;
    
    float eps = 1e-4f * diag;
    float zero_eps = 1e-8f * diag;
    
    for (int i = 0; i < n; ++i) {
        new_verts.push_back(poly.vertices[i]);
        int j = (i + 1) % n;
        const auto& vi = poly.vertices[i];
        const auto& vj = poly.vertices[j];
        
        struct InsertPoint {
            csgjs_vector pos;
            float dist;
        };
        std::vector<InsertPoint> to_insert;
        
        csgjs_vector edge = vj.pos - vi.pos;
        float edge_len = length(edge);
        if (edge_len < zero_eps) continue;
        csgjs_vector edge_dir = edge / edge_len;
        
        for (const auto& p : P_ints) {
            csgjs_vector diff = p - vi.pos;
            float projection = dot(diff, edge_dir);
            if (projection > eps && projection < edge_len - eps) {
                csgjs_vector perp = diff - edge_dir * projection;
                if (length(perp) < eps) {
                    to_insert.push_back({p, projection});
                }
            }
        }
        
        if (!to_insert.empty()) {
            std::sort(to_insert.begin(), to_insert.end(), [](const InsertPoint& a, const InsertPoint& b) {
                return a.dist < b.dist;
            });
            for (const auto& ip : to_insert) {
                float t = ip.dist / edge_len;
                csgjs_vertex v_new = interpolate(vi, vj, t);
                v_new.pos = ip.pos;
                new_verts.push_back(v_new);
            }
        }
    }
    poly.vertices = new_verts;
}

std::vector<Polygon> triangulate_convex_polygon(const Polygon& poly, float diag) {
    std::vector<Polygon> result;
    int n = poly.vertices.size();
    if (n < 3) return result;
    
    float eps = 1e-4f;
    float zero_eps = 1e-11f * diag * diag;
    
    std::vector<int> corners;
    for (int i = 0; i < n; ++i) {
        int prev = (i - 1 + n) % n;
        int next = (i + 1) % n;
        csgjs_vector e1 = unit(poly.vertices[i].pos - poly.vertices[prev].pos);
        csgjs_vector e2 = unit(poly.vertices[next].pos - poly.vertices[i].pos);
        if (length(cross(e1, e2)) > eps) {
            corners.push_back(i);
        }
    }
    
    if (corners.size() < 3) return result;
    
    int num_corners = corners.size();
    for (int i = 2; i < num_corners; ++i) {
        int idx0 = corners[0];
        int idx1 = corners[i-1];
        int idx2 = corners[i];
        
        std::vector<int> sub_poly_indices;
        
        if (i == 2) {
            int curr = idx0;
            while (curr != idx1) {
                sub_poly_indices.push_back(curr);
                curr = (curr + 1) % n;
            }
        } else {
            sub_poly_indices.push_back(idx0);
        }
        
        int curr = idx1;
        while (curr != idx2) {
            sub_poly_indices.push_back(curr);
            curr = (curr + 1) % n;
        }
        
        sub_poly_indices.push_back(idx2);
        
        if (i == num_corners - 1) {
            int curr = (idx2 + 1) % n;
            while (curr != idx0) {
                sub_poly_indices.push_back(curr);
                curr = (curr + 1) % n;
            }
        }
        
        // Find best fan center for the sub-polygon
        int num_sub = sub_poly_indices.size();
        int best_center = -1;
        float max_min_area = -1.0f;
        std::vector<Polygon> best_tris;
        
        for (int c_idx = 0; c_idx < num_sub; ++c_idx) {
            std::vector<Polygon> candidate_tris;
            float min_area = 1e30f;
            bool ok = true;
            
            for (int k = 1; k < num_sub - 1; ++k) {
                int idx_a = sub_poly_indices[c_idx];
                int idx_b = sub_poly_indices[(c_idx + k) % num_sub];
                int idx_c = sub_poly_indices[(c_idx + k + 1) % num_sub];
                
                Polygon tri;
                tri.vertices = { poly.vertices[idx_a], poly.vertices[idx_b], poly.vertices[idx_c] };
                
                csgjs_vector edge1 = tri.vertices[1].pos - tri.vertices[0].pos;
                csgjs_vector edge2 = tri.vertices[2].pos - tri.vertices[0].pos;
                float area = length(cross(edge1, edge2));
                if (area < zero_eps) {
                    ok = false;
                    break;
                }
                min_area = std::min(min_area, area);
                candidate_tris.push_back(tri);
            }
            
            if (ok && min_area > max_min_area) {
                max_min_area = min_area;
                best_center = c_idx;
                best_tris = candidate_tris;
            }
        }
        
        if (best_center != -1) {
            result.insert(result.end(), best_tris.begin(), best_tris.end());
        } else {
            // Fallback to simple fanning if all centers are collinear
            for (int k = 2; k < num_sub; ++k) {
                Polygon tri;
                tri.vertices = {
                    poly.vertices[sub_poly_indices[0]],
                    poly.vertices[sub_poly_indices[k-1]],
                    poly.vertices[sub_poly_indices[k]]
                };
                csgjs_vector edge1 = tri.vertices[1].pos - tri.vertices[0].pos;
                csgjs_vector edge2 = tri.vertices[2].pos - tri.vertices[0].pos;
                if (length(cross(edge1, edge2)) > zero_eps) {
                    result.push_back(tri);
                }
            }
        }
    }
    return result;
}

struct GridDeduplicator {
    float cell_size;
    struct Cell {
        int x, y, z;
        bool operator==(const Cell& o) const { return x == o.x && y == o.y && z == o.z; }
    };
    struct CellHash {
        size_t operator()(const Cell& c) const {
            return (size_t)(c.x * 73856093 ^ c.y * 19349663 ^ c.z * 83492791);
        }
    };
    std::unordered_map<Cell, std::vector<int>, CellHash> grid;
    std::vector<csgjs_vector> unique_verts;
    
    GridDeduplicator(float tol) : cell_size(tol) {}
    
    int get_or_insert(const csgjs_vector& p) {
        Cell cell = { (int)floor(p.x / cell_size), (int)floor(p.y / cell_size), (int)floor(p.z / cell_size) };
        
        float tol_sq = cell_size * cell_size;
        
        for (int dx = -1; dx <= 1; ++dx) {
            for (int dy = -1; dy <= 1; ++dy) {
                for (int dz = -1; dz <= 1; ++dz) {
                    Cell neighbor = { cell.x + dx, cell.y + dy, cell.z + dz };
                    auto it = grid.find(neighbor);
                    if (it != grid.end()) {
                        for (int idx : it->second) {
                            csgjs_vector diff = unique_verts[idx] - p;
                            if (dot(diff, diff) < tol_sq) {
                                return idx;
                            }
                        }
                    }
                }
            }
        }
        
        int new_idx = unique_verts.size();
        unique_verts.push_back(p);
        grid[cell].push_back(new_idx);
        return new_idx;
    }
};

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    int VA, FA, VB, FB;
    if (!(std::cin >> VA >> FA >> VB >> FB)) return 0;
    
    csgjs_model meshA, meshB;
    meshA.vertices.resize(VA);
    for (int i = 0; i < VA; ++i) {
        std::cin >> meshA.vertices[i].pos.x >> meshA.vertices[i].pos.y >> meshA.vertices[i].pos.z;
    }
    meshA.indices.resize(FA * 3);
    for (int i = 0; i < FA * 3; ++i) {
        std::cin >> meshA.indices[i];
    }
    
    meshB.vertices.resize(VB);
    for (int i = 0; i < VB; ++i) {
        std::cin >> meshB.vertices[i].pos.x >> meshB.vertices[i].pos.y >> meshB.vertices[i].pos.z;
    }
    meshB.indices.resize(FB * 3);
    for (int i = 0; i < FB * 3; ++i) {
        std::cin >> meshB.indices[i];
    }
    
    // 1. Compute bounding box and diagonal size
    AABB full_box;
    for (const auto& v : meshA.vertices) full_box.extend(v.pos);
    for (const auto& v : meshB.vertices) full_box.extend(v.pos);
    float diag = length(full_box.max_p - full_box.min_p);
    if (diag < 1e-5f) diag = 1.0f;
    
    // 2. Perturb B slightly to avoid coplanar and degenerate issues
    float perturbation_scale = 1.5e-4f * diag;
    csgjs_vector translation_vector(1.3f * perturbation_scale, -0.7f * perturbation_scale, 0.9f * perturbation_scale);
    for (auto& v : meshB.vertices) {
        v.pos = v.pos + translation_vector;
    }
    
    // 3. Build trees
    AABBTree treeA = build_tree(meshA);
    AABBTree treeB = build_tree(meshB);
    
    // 4. Find intersecting pairs
    int num_trisA = FA;
    int num_trisB = FB;
    
    std::vector<std::vector<int>> intersectsA(num_trisA);
    std::vector<std::vector<int>> intersectsB(num_trisB);
    
    #pragma omp parallel
    {
        #pragma omp for nowait
        for (int i = 0; i < num_trisA; ++i) {
            const auto& v1 = meshA.vertices[meshA.indices[3 * i]].pos;
            const auto& v2 = meshA.vertices[meshA.indices[3 * i + 1]].pos;
            const auto& v3 = meshA.vertices[meshA.indices[3 * i + 2]].pos;
            AABB tri_box(v1);
            tri_box.extend(v2);
            tri_box.extend(v3);
            
            std::vector<int> candidates;
            treeB.query_box(0, tri_box, candidates);
            
            for (int triB : candidates) {
                csgjs_vector p1, p2;
                const auto& bv1 = meshB.vertices[meshB.indices[3 * triB]];
                const auto& bv2 = meshB.vertices[meshB.indices[3 * triB + 1]];
                const auto& bv3 = meshB.vertices[meshB.indices[3 * triB + 2]];
                const auto& av1 = meshA.vertices[meshA.indices[3 * i]];
                const auto& av2 = meshA.vertices[meshA.indices[3 * i + 1]];
                const auto& av3 = meshA.vertices[meshA.indices[3 * i + 2]];
                
                if (intersect_triangles(av1, av2, av3, bv1, bv2, bv3, p1, p2)) {
                    #pragma omp critical
                    {
                        intersectsA[i].push_back(triB);
                        intersectsB[triB].push_back(i);
                    }
                }
            }
        }
    }
    
    // 5. Check if disjoint/fully enclosed
    bool any_intersection = false;
    for (int i = 0; i < num_trisA; ++i) {
        if (!intersectsA[i].empty()) { any_intersection = true; break; }
    }
    
    AABB boxA = treeA.nodes[0].box;
    AABB boxB = treeB.nodes[0].box;
    
    std::vector<Polygon> final_triangles;
    
    unsigned int global_seed = 12345;
    
    if (!any_intersection) {
        // Disjoint or enclosed
        // Cast a ray from meshA's first centroid to meshB
        csgjs_vector centroidA = (meshA.vertices[meshA.indices[0]].pos + meshA.vertices[meshA.indices[1]].pos + meshA.vertices[meshA.indices[2]].pos) / 3.0f;
        csgjs_vector centroidB = (meshB.vertices[meshB.indices[0]].pos + meshB.vertices[meshB.indices[1]].pos + meshB.vertices[meshB.indices[2]].pos) / 3.0f;
        
        bool A_in_B = is_point_inside_mesh(centroidA, treeB, global_seed);
        bool B_in_A = is_point_inside_mesh(centroidB, treeA, global_seed);
        
        if (A_in_B) {
            // Union is B
            final_triangles.resize(num_trisB);
            for (int i = 0; i < num_trisB; ++i) {
                final_triangles[i].vertices = {
                    meshB.vertices[meshB.indices[3 * i]],
                    meshB.vertices[meshB.indices[3 * i + 1]],
                    meshB.vertices[meshB.indices[3 * i + 2]]
                };
            }
        } else if (B_in_A) {
            // Union is A
            final_triangles.resize(num_trisA);
            for (int i = 0; i < num_trisA; ++i) {
                final_triangles[i].vertices = {
                    meshA.vertices[meshA.indices[3 * i]],
                    meshA.vertices[meshA.indices[3 * i + 1]],
                    meshA.vertices[meshA.indices[3 * i + 2]]
                };
            }
        } else {
            // Disjoint, union is A + B
            final_triangles.resize(num_trisA + num_trisB);
            for (int i = 0; i < num_trisA; ++i) {
                final_triangles[i].vertices = {
                    meshA.vertices[meshA.indices[3 * i]],
                    meshA.vertices[meshA.indices[3 * i + 1]],
                    meshA.vertices[meshA.indices[3 * i + 2]]
                };
            }
            for (int i = 0; i < num_trisB; ++i) {
                final_triangles[num_trisA + i].vertices = {
                    meshB.vertices[meshB.indices[3 * i]],
                    meshB.vertices[meshB.indices[3 * i + 1]],
                    meshB.vertices[meshB.indices[3 * i + 2]]
                };
            }
        }
    } else {
        // 5. Clipping Stage: clip A's triangles by B's planes, and B's by A's planes
        std::vector<std::vector<Polygon>> A_polys(num_trisA);
        #pragma omp parallel for
        for (int i = 0; i < num_trisA; ++i) {
            const auto& av1 = meshA.vertices[meshA.indices[3 * i]];
            const auto& av2 = meshA.vertices[meshA.indices[3 * i + 1]];
            const auto& av3 = meshA.vertices[meshA.indices[3 * i + 2]];
            
            if (intersectsA[i].empty()) {
                A_polys[i] = { { {av1, av2, av3} } };
            } else {
                std::vector<std::pair<csgjs_vector, float>> plane_list;
                for (int triB : intersectsA[i]) {
                    const auto& bv1 = meshB.vertices[meshB.indices[3 * triB]];
                    const auto& bv2 = meshB.vertices[meshB.indices[3 * triB + 1]];
                    const auto& bv3 = meshB.vertices[meshB.indices[3 * triB + 2]];
                    
                    csgjs_vector p1, p2;
                    if (intersect_triangles(av1, av2, av3, bv1, bv2, bv3, p1, p2)) {
                        csgjs_vector n = cross(bv2.pos - bv1.pos, bv3.pos - bv1.pos);
                        float len_n = length(n);
                        if (len_n > 1e-8f) {
                            n = n / len_n;
                            float w = dot(n, bv1.pos);
                            plane_list.push_back({n, w});
                        }
                    }
                }
                
                std::vector<Polygon> polys = { { {av1, av2, av3} } };
                for (const auto& plane : plane_list) {
                    std::vector<Polygon> next_polys;
                    for (const auto& poly : polys) {
                        Polygon f, b;
                        clip_polygon_by_plane(poly, plane.first, plane.second, f, b, diag);
                        if (f.vertices.size() >= 3) next_polys.push_back(f);
                        if (b.vertices.size() >= 3) next_polys.push_back(b);
                    }
                    polys = next_polys;
                }
                A_polys[i] = polys;
            }
        }
        
        std::vector<std::vector<Polygon>> B_polys(num_trisB);
        #pragma omp parallel for
        for (int i = 0; i < num_trisB; ++i) {
            const auto& bv1 = meshB.vertices[meshB.indices[3 * i]];
            const auto& bv2 = meshB.vertices[meshB.indices[3 * i + 1]];
            const auto& bv3 = meshB.vertices[meshB.indices[3 * i + 2]];
            
            if (intersectsB[i].empty()) {
                B_polys[i] = { { {bv1, bv2, bv3} } };
            } else {
                std::vector<std::pair<csgjs_vector, float>> plane_list;
                for (int triA : intersectsB[i]) {
                    const auto& av1 = meshA.vertices[meshA.indices[3 * triA]];
                    const auto& av2 = meshA.vertices[meshA.indices[3 * triA + 1]];
                    const auto& av3 = meshA.vertices[meshA.indices[3 * triA + 2]];
                    
                    csgjs_vector p1, p2;
                    if (intersect_triangles(av1, av2, av3, bv1, bv2, bv3, p1, p2)) {
                        csgjs_vector n = cross(av2.pos - av1.pos, av3.pos - av1.pos);
                        float len_n = length(n);
                        if (len_n > 1e-8f) {
                            n = n / len_n;
                            float w = dot(n, av1.pos);
                            plane_list.push_back({n, w});
                        }
                    }
                }
                
                std::vector<Polygon> polys = { { {bv1, bv2, bv3} } };
                for (const auto& plane : plane_list) {
                    std::vector<Polygon> next_polys;
                    for (const auto& poly : polys) {
                        Polygon f, b;
                        clip_polygon_by_plane(poly, plane.first, plane.second, f, b, diag);
                        if (f.vertices.size() >= 3) next_polys.push_back(f);
                        if (b.vertices.size() >= 3) next_polys.push_back(b);
                    }
                    polys = next_polys;
                }
                B_polys[i] = polys;
            }
        }
        
        // 6. Gather all unique sub-polygon vertices
        std::vector<csgjs_vector> all_sub_verts;
        for (int i = 0; i < num_trisA; ++i) {
            for (const auto& poly : A_polys[i]) {
                for (const auto& v : poly.vertices) {
                    all_sub_verts.push_back(v.pos);
                }
            }
        }
        for (int i = 0; i < num_trisB; ++i) {
            for (const auto& poly : B_polys[i]) {
                for (const auto& v : poly.vertices) {
                    all_sub_verts.push_back(v.pos);
                }
            }
        }
        
        std::vector<csgjs_vector> global_unique_P_ints;
        if (!all_sub_verts.empty()) {
            int M = all_sub_verts.size();
            struct SweepVert {
                csgjs_vector pos;
                int orig_idx;
            };
            std::vector<SweepVert> sverts(M);
            for (int i = 0; i < M; ++i) {
                sverts[i] = { all_sub_verts[i], i };
            }
            std::sort(sverts.begin(), sverts.end(), [](const SweepVert& a, const SweepVert& b) {
                if (fabs(a.pos.x - b.pos.x) > 1e-9f) return a.pos.x < b.pos.x;
                if (fabs(a.pos.y - b.pos.y) > 1e-9f) return a.pos.y < b.pos.y;
                return a.pos.z < b.pos.z;
            });
            
            std::vector<int> to_unique(M);
            float tol = 1e-5f * diag;
            float tol_sq = tol * tol;
            for (int i = 0; i < M; ++i) {
                bool found = false;
                int found_idx = -1;
                for (int j = i - 1; j >= 0; --j) {
                    if (sverts[i].pos.x - sverts[j].pos.x > tol) break;
                    csgjs_vector diff = sverts[i].pos - sverts[j].pos;
                    if (dot(diff, diff) < tol_sq) {
                        found = true;
                        found_idx = to_unique[sverts[j].orig_idx];
                        break;
                    }
                }
                if (found) {
                    to_unique[sverts[i].orig_idx] = found_idx;
                } else {
                    int new_idx = global_unique_P_ints.size();
                    global_unique_P_ints.push_back(sverts[i].pos);
                    to_unique[sverts[i].orig_idx] = new_idx;
                }
            }
        }
        
        // 7. Query AABB trees to associate unique vertices with original triangles
        std::vector<std::vector<csgjs_vector>> A_insertions(num_trisA);
        float query_tol = 1e-4f * diag;
        #pragma omp parallel for
        for (int p_idx = 0; p_idx < (int)global_unique_P_ints.size(); ++p_idx) {
            const auto& p = global_unique_P_ints[p_idx];
            AABB point_box;
            point_box.min_p = p - csgjs_vector(query_tol, query_tol, query_tol);
            point_box.max_p = p + csgjs_vector(query_tol, query_tol, query_tol);
            std::vector<int> candidates;
            treeA.query_box(0, point_box, candidates);
            for (int tri_idx : candidates) {
                #pragma omp critical
                {
                    A_insertions[tri_idx].push_back(p);
                }
            }
        }
        
        std::vector<std::vector<csgjs_vector>> B_insertions(num_trisB);
        #pragma omp parallel for
        for (int p_idx = 0; p_idx < (int)global_unique_P_ints.size(); ++p_idx) {
            const auto& p = global_unique_P_ints[p_idx];
            AABB point_box;
            point_box.min_p = p - csgjs_vector(query_tol, query_tol, query_tol);
            point_box.max_p = p + csgjs_vector(query_tol, query_tol, query_tol);
            std::vector<int> candidates;
            treeB.query_box(0, point_box, candidates);
            for (int tri_idx : candidates) {
                #pragma omp critical
                {
                    B_insertions[tri_idx].push_back(p);
                }
            }
        }
        
        // 8. Insert points, classify, and triangulate
        
        // Process A
        #pragma omp parallel
        {
            unsigned int seed = 12345 + omp_get_thread_num() * 1111;
            std::vector<Polygon> local_tris;
            
            #pragma omp for nowait
            for (int i = 0; i < num_trisA; ++i) {
                for (auto& poly : A_polys[i]) {
                    insert_intersection_points(poly, A_insertions[i], diag);
                    std::vector<Polygon> sub_tris = triangulate_convex_polygon(poly, diag);
                    for (const auto& stri : sub_tris) {
                        csgjs_vector c = (stri.vertices[0].pos + stri.vertices[1].pos + stri.vertices[2].pos) / 3.0f;
                        if (!is_point_inside_mesh(c, treeB, seed)) {
                            local_tris.push_back(stri);
                        }
                    }
                }
            }
            
            #pragma omp critical
            {
                final_triangles.insert(final_triangles.end(), local_tris.begin(), local_tris.end());
            }
        }
        
        // Process B
        #pragma omp parallel
        {
            unsigned int seed = 54321 + omp_get_thread_num() * 1111;
            std::vector<Polygon> local_tris;
            
            #pragma omp for nowait
            for (int i = 0; i < num_trisB; ++i) {
                for (auto& poly : B_polys[i]) {
                    insert_intersection_points(poly, B_insertions[i], diag);
                    std::vector<Polygon> sub_tris = triangulate_convex_polygon(poly, diag);
                    for (const auto& stri : sub_tris) {
                        csgjs_vector c = (stri.vertices[0].pos + stri.vertices[1].pos + stri.vertices[2].pos) / 3.0f;
                        if (!is_point_inside_mesh(c, treeA, seed)) {
                            local_tris.push_back(stri);
                        }
                    }
                }
            }
            
            #pragma omp critical
            {
                final_triangles.insert(final_triangles.end(), local_tris.begin(), local_tris.end());
            }
        }
    }
    
    // 9. Deduplicate vertices of the final output mesh using sweep-line
    std::vector<csgjs_vector> final_all_verts;
    final_all_verts.reserve(final_triangles.size() * 3);
    for (const auto& tri : final_triangles) {
        final_all_verts.push_back(tri.vertices[0].pos);
        final_all_verts.push_back(tri.vertices[1].pos);
        final_all_verts.push_back(tri.vertices[2].pos);
    }
    
    std::vector<csgjs_vector> unique_verts;
    std::vector<int> final_indices;
    if (!final_all_verts.empty()) {
        int M = final_all_verts.size();
        struct SweepVert {
            csgjs_vector pos;
            int orig_idx;
        };
        std::vector<SweepVert> sverts(M);
        for (int i = 0; i < M; ++i) {
            sverts[i] = { final_all_verts[i], i };
        }
        std::sort(sverts.begin(), sverts.end(), [](const SweepVert& a, const SweepVert& b) {
            if (fabs(a.pos.x - b.pos.x) > 1e-9f) return a.pos.x < b.pos.x;
            if (fabs(a.pos.y - b.pos.y) > 1e-9f) return a.pos.y < b.pos.y;
            return a.pos.z < b.pos.z;
        });
        
        std::vector<int> to_unique(M);
        float tol = 1e-5f * diag;
        float tol_sq = tol * tol;
        for (int i = 0; i < M; ++i) {
            bool found = false;
            int found_idx = -1;
            for (int j = i - 1; j >= 0; --j) {
                if (sverts[i].pos.x - sverts[j].pos.x > tol) break;
                csgjs_vector diff = sverts[i].pos - sverts[j].pos;
                if (dot(diff, diff) < tol_sq) {
                    found = true;
                    found_idx = to_unique[sverts[j].orig_idx];
                    break;
                }
            }
            if (found) {
                to_unique[sverts[i].orig_idx] = found_idx;
            } else {
                int new_idx = unique_verts.size();
                unique_verts.push_back(sverts[i].pos);
                to_unique[sverts[i].orig_idx] = new_idx;
            }
        }
        
        final_indices.reserve(final_triangles.size() * 3);
        for (size_t i = 0; i < final_triangles.size(); ++i) {
            int idx0 = to_unique[3 * i];
            int idx1 = to_unique[3 * i + 1];
            int idx2 = to_unique[3 * i + 2];
            if (idx0 != idx1 && idx1 != idx2 && idx2 != idx0) {
                final_indices.push_back(idx0);
                final_indices.push_back(idx1);
                final_indices.push_back(idx2);
            }
        }
    }
    
    // 10. Output results
    std::cout << unique_verts.size() << " " << (final_indices.size() / 3) << "\n";
    for (const auto& v : unique_verts) {
        std::cout << v.x << " " << v.y << " " << v.z << "\n";
    }
    for (size_t i = 0; i < final_indices.size(); i += 3) {
        std::cout << final_indices[i] << " " << final_indices[i+1] << " " << final_indices[i+2] << "\n";
    }
    
    return 0;
}
