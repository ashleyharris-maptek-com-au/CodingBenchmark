fn segments_intersect(p1: [f64; 2], p2: [f64; 2], q1: [f64; 2], q2: [f64; 2]) -> bool {
    let d = (p2[0] - p1[0]) * (q2[1] - q1[1]) - (p2[1] - p1[1]) * (q2[0] - q1[0]);
    if d.abs() < 1e-9 {
        return false;
    }
    let u = ((q1[0] - p1[0]) * (q2[1] - q1[1]) - (q1[1] - p1[1]) * (q2[0] - q1[0])) / d;
    let v = ((q1[0] - p1[0]) * (p2[1] - p1[1]) - (q1[1] - p1[1]) * (p2[0] - p1[0])) / d;
    u >= 0.0 && u <= 1.0 && v >= 0.0 && v <= 1.0
}

fn segment_intersects_cell(
    a: [f64; 2],
    b: [f64; 2],
    ci: i32,
    cj: i32,
    w_cell: f64,
    h_cell: f64,
    dx: f64,
    dy: f64,
) -> bool {
    let u_min = (ci as f64) * w_cell + dx;
    let u_max = ((ci + 1) as f64) * w_cell + dx;
    let v_min = (cj as f64) * h_cell + dy;
    let v_max = ((cj + 1) as f64) * h_cell + dy;
    
    let left_1 = [u_min, v_min];
    let left_2 = [u_min, v_max];
    if segments_intersect(a, b, left_1, left_2) { return true; }
    
    let right_1 = [u_max, v_min];
    let right_2 = [u_max, v_max];
    if segments_intersect(a, b, right_1, right_2) { return true; }
    
    let bottom_1 = [u_min, v_min];
    let bottom_2 = [u_max, v_min];
    if segments_intersect(a, b, bottom_1, bottom_2) { return true; }
    
    let top_1 = [u_min, v_max];
    let top_2 = [u_max, v_max];
    if segments_intersect(a, b, top_1, top_2) { return true; }
    
    false
}

fn point_in_polygon(p: [f64; 2], poly: &[[f64; 2]]) -> bool {
    let mut inside = false;
    let n = poly.len();
    let mut j = n - 1;
    for i in 0..n {
        if ((poly[i][1] > p[1]) != (poly[j][1] > p[1])) &&
           (p[0] < (poly[j][0] - poly[i][0]) * (p[1] - poly[i][1]) / (poly[j][1] - poly[i][1]) + poly[i][0]) {
            inside = !inside;
        }
        j = i;
    }
    inside
}

fn mat_to_quat(r: [[f64; 3]; 3]) -> [f64; 4] {
    let tr = r[0][0] + r[1][1] + r[2][2];
    if tr > 0.0 {
        let s = (tr + 1.0).sqrt() * 2.0;
        let qw = 0.25 * s;
        let qx = (r[2][1] - r[1][2]) / s;
        let qy = (r[0][2] - r[2][0]) / s;
        let qz = (r[1][0] - r[0][1]) / s;
        [qw, qx, qy, qz]
    } else if r[0][0] > r[1][1] && r[0][0] > r[2][2] {
        let s = (1.0 + r[0][0] - r[1][1] - r[2][2]).sqrt() * 2.0;
        let qw = (r[2][1] - r[1][2]) / s;
        let qx = 0.25 * s;
        let qy = (r[0][1] + r[1][0]) / s;
        let qz = (r[0][2] + r[2][0]) / s;
        [qw, qx, qy, qz]
    } else if r[1][1] > r[2][2] {
        let s = (1.0 + r[1][1] - r[0][0] - r[2][2]).sqrt() * 2.0;
        let qw = (r[0][2] - r[2][0]) / s;
        let qx = (r[0][1] + r[1][0]) / s;
        let qy = 0.25 * s;
        let qz = (r[1][2] + r[2][1]) / s;
        [qw, qx, qy, qz]
    } else {
        let s = (1.0 + r[2][2] - r[0][0] - r[1][1]).sqrt() * 2.0;
        let qw = (r[1][0] - r[0][1]) / s;
        let qx = (r[0][2] + r[2][0]) / s;
        let qy = (r[1][2] + r[2][1]) / s;
        let qz = 0.25 * s;
        [qw, qx, qy, qz]
    }
}

fn find_grid_cover(
    poly_uv: &[[f64; 2]],
    phi: f64,
    dx: f64,
    dy: f64,
    w_cell: f64,
    h_cell: f64,
) -> Vec<(i32, i32)> {
    let cos_p = phi.cos();
    let sin_p = phi.sin();
    
    let mut rotated_poly = Vec::with_capacity(poly_uv.len());
    let mut u_min = f64::INFINITY;
    let mut u_max = f64::NEG_INFINITY;
    let mut v_min = f64::INFINITY;
    let mut v_max = f64::NEG_INFINITY;
    
    for p in poly_uv {
        let u_rot = p[0] * cos_p + p[1] * sin_p;
        let v_rot = -p[0] * sin_p + p[1] * cos_p;
        rotated_poly.push([u_rot, v_rot]);
        
        if u_rot < u_min { u_min = u_rot; }
        if u_rot > u_max { u_max = u_rot; }
        if v_rot < v_min { v_min = v_rot; }
        if v_rot > v_max { v_max = v_rot; }
    }
    
    let i_start = ((u_min - dx) / w_cell).floor() as i32;
    let i_end = ((u_max - dx) / w_cell).floor() as i32;
    let j_start = ((v_min - dy) / h_cell).floor() as i32;
    let j_end = ((v_max - dy) / h_cell).floor() as i32;
    
    let mut cell_has_vertex = std::collections::HashSet::new();
    for p in &rotated_poly {
        let ci = ((p[0] - dx) / w_cell).floor() as i32;
        let cj = ((p[1] - dy) / h_cell).floor() as i32;
        cell_has_vertex.insert((ci, cj));
    }
    
    let mut cell_edges = std::collections::HashMap::new();
    let n = rotated_poly.len();
    for k in 0..n {
        let a = rotated_poly[k];
        let b = rotated_poly[(k + 1) % n];
        
        let u_min_e = a[0].min(b[0]);
        let u_max_e = a[0].max(b[0]);
        let v_min_e = a[1].min(b[1]);
        let v_max_e = a[1].max(b[1]);
        
        let ei_start = ((u_min_e - dx) / w_cell).floor() as i32;
        let ei_end = ((u_max_e - dx) / w_cell).floor() as i32;
        let ej_start = ((v_min_e - dy) / h_cell).floor() as i32;
        let ej_end = ((v_max_e - dy) / h_cell).floor() as i32;
        
        for ci in ei_start..=ei_end {
            for cj in ej_start..=ej_end {
                cell_edges.entry((ci, cj)).or_insert_with(Vec::new).push((a, b));
            }
        }
    }
    
    let mut active_cells = Vec::new();
    for ci in i_start..=i_end {
        for cj in j_start..=j_end {
            if cell_has_vertex.contains(&(ci, cj)) {
                active_cells.push((ci, cj));
                continue;
            }
            
            let mut intersects = false;
            if let Some(edges) = cell_edges.get(&(ci, cj)) {
                for &(a, b) in edges {
                    if segment_intersects_cell(a, b, ci, cj, w_cell, h_cell, dx, dy) {
                        intersects = true;
                        break;
                    }
                }
            }
            if intersects {
                active_cells.push((ci, cj));
                continue;
            }
            
            let u_c = (ci as f64 + 0.5) * w_cell + dx;
            let v_c = (cj as f64 + 0.5) * h_cell + dy;
            if point_in_polygon([u_c, v_c], &rotated_poly) {
                active_cells.push((ci, cj));
            }
        }
    }
    
    active_cells
}

fn search_best_grid_cover(
    poly_uv: &[[f64; 2]],
    w_cell: f64,
    h_cell: f64,
) -> (f64, f64, f64, Vec<(i32, i32)>) {
    let mut u_min = f64::INFINITY;
    let mut u_max = f64::NEG_INFINITY;
    let mut v_min = f64::INFINITY;
    let mut v_max = f64::NEG_INFINITY;
    for p in poly_uv {
        if p[0] < u_min { u_min = p[0]; }
        if p[0] > u_max { u_max = p[0]; }
        if p[1] < v_min { v_min = p[1]; }
        if p[1] > v_max { v_max = p[1]; }
    }
    let w_est = u_max - u_min;
    let h_est = v_max - v_min;
    let n_cells_est = (w_est / w_cell) * (h_est / h_cell);
    
    let (num_angles, num_offsets) = if n_cells_est <= 20.0 {
        (24, 4)
    } else if n_cells_est <= 200.0 {
        (12, 3)
    } else if n_cells_est <= 2000.0 {
        (6, 2)
    } else {
        (1, 1)
    };
    
    let mut best_phi = 0.0;
    let mut best_dx = 0.0;
    let mut best_dy = 0.0;
    let mut best_cells = Vec::new();
    let mut min_count = usize::MAX;
    
    for a in 0..num_angles {
        let phi = (a as f64) * std::f64::consts::PI / (num_angles as f64);
        for ox in 0..num_offsets {
            let dx = (ox as f64) * w_cell / (num_offsets as f64);
            for oy in 0..num_offsets {
                let dy = (oy as f64) * h_cell / (num_offsets as f64);
                
                let cells = find_grid_cover(poly_uv, phi, dx, dy, w_cell, h_cell);
                if cells.len() < min_count {
                    min_count = cells.len();
                    best_phi = phi;
                    best_dx = dx;
                    best_dy = dy;
                    best_cells = cells;
                }
            }
        }
    }
    
    (best_phi, best_dx, best_dy, best_cells)
}

fn solve(poly: Vec<[f64; 2]>, sun: [f64; 3]) {
    let sx = sun[0];
    let sy = sun[1];
    let sz = sun[2];
    
    let s_len = (sx*sx + sy*sy + sz*sz).sqrt();
    let c3 = [sx / s_len, sy / s_len, sz / s_len];
    let horiz_len = (sx*sx + sy*sy).sqrt();
    
    let (u1, u2, e1, e2, l_factor) = if horiz_len < 1e-9 {
        let u1 = [1.0, 0.0, 0.0];
        let u2 = [0.0, 1.0, 0.0];
        let e1 = [1.0, 0.0];
        let e2 = [0.0, 1.0];
        let l_factor = 1.0;
        (u1, u2, e1, e2, l_factor)
    } else {
        let u1 = [-sy / horiz_len, sx / horiz_len, 0.0];
        let u2 = [
            c3[1] * u1[2] - c3[2] * u1[1],
            c3[2] * u1[0] - c3[0] * u1[2],
            c3[0] * u1[1] - c3[1] * u1[0],
        ];
        let e1 = [-sy / horiz_len, sx / horiz_len];
        let e2 = [sx / horiz_len, sy / horiz_len];
        let l_factor = s_len / sz.abs();
        (u1, u2, e1, e2, l_factor)
    };
    
    let mut poly_uv = Vec::with_capacity(poly.len());
    for p in &poly {
        let u = p[0] * e1[0] + p[1] * e1[1];
        let v = p[0] * e2[0] + p[1] * e2[1];
        poly_uv.push([u, v]);
    }
    
    // Check if single tetrahedron can cover it
    let mut best_single_phi = 0.0;
    let mut min_single_scale = f64::INFINITY;
    let mut best_single_bbox = (0.0, 0.0, 0.0, 0.0);
    
    let num_angles_single = 180;
    for a in 0..num_angles_single {
        let phi = (a as f64) * std::f64::consts::PI / (num_angles_single as f64);
        let cos_p = phi.cos();
        let sin_p = phi.sin();
        
        let mut u_min = f64::INFINITY;
        let mut u_max = f64::NEG_INFINITY;
        let mut v_min = f64::INFINITY;
        let mut v_max = f64::NEG_INFINITY;
        
        for p in &poly_uv {
            let u_rot = p[0] * cos_p + p[1] * sin_p;
            let v_rot = -p[0] * sin_p + p[1] * cos_p;
            if u_rot < u_min { u_min = u_rot; }
            if u_rot > u_max { u_max = u_rot; }
            if v_rot < v_min { v_min = v_rot; }
            if v_rot > v_max { v_max = v_rot; }
        }
        
        let w = u_max - u_min;
        let h = v_max - v_min;
        
        let scale = (w / 2.0f64.sqrt()).max(h / (l_factor * 2.0f64.sqrt()));
        if scale < min_single_scale {
            min_single_scale = scale;
            best_single_phi = phi;
            best_single_bbox = (u_min, u_max, v_min, v_max);
        }
    }
    
    if min_single_scale <= 10.0 {
        // Output single tetrahedron
        let scale = min_single_scale.min(10.0);
        let (u_min, u_max, v_min, v_max) = best_single_bbox;
        let u_c = (u_min + u_max) / 2.0;
        let v_c = (v_min + v_max) / 2.0;
        
        let tx = u_c * best_single_phi.cos() - v_c * best_single_phi.sin();
        let ty = u_c * best_single_phi.sin() + v_c * best_single_phi.cos();
        
        let pz = 0.0;
        let px = tx * e1[0] + ty * e2[0] + pz * sx / sz;
        let py = tx * e1[1] + ty * e2[1] + pz * sy / sz;
        
        let theta = best_single_phi + std::f64::consts::FRAC_PI_4;
        let cos_t = theta.cos();
        let sin_t = theta.sin();
        let c1 = [
            cos_t * u1[0] + sin_t * u2[0],
            cos_t * u1[1] + sin_t * u2[1],
            cos_t * u1[2] + sin_t * u2[2],
        ];
        let c2 = [
            -sin_t * u1[0] + cos_t * u2[0],
            -sin_t * u1[1] + cos_t * u2[1],
            -sin_t * u1[2] + cos_t * u2[2],
        ];
        let r = [
            [c1[0], c2[0], c3[0]],
            [c1[1], c2[1], c3[1]],
            [c1[2], c2[2], c3[2]],
        ];
        let q = mat_to_quat(r);
        
        println!("1");
        println!(
            "{:.9} {:.9} {:.9} {:.9} {:.9} {:.9} {:.9} {:.9}",
            px, py, pz, q[0], q[1], q[2], q[3], scale
        );
        return;
    }
    
    // Grid Cover
    let w_cell = 10.0 * 2.0f64.sqrt();
    let h_cell = 10.0 * l_factor * 2.0f64.sqrt();
    
    let (best_phi, best_dx, best_dy, mut best_cells) =
        search_best_grid_cover(&poly_uv, w_cell, h_cell);
        
    if best_cells.is_empty() {
        best_cells.push((0, 0));
    }
    
    println!("{}", best_cells.len());
    let theta = best_phi + std::f64::consts::FRAC_PI_4;
    let cos_t = theta.cos();
    let sin_t = theta.sin();
    let c1 = [
        cos_t * u1[0] + sin_t * u2[0],
        cos_t * u1[1] + sin_t * u2[1],
        cos_t * u1[2] + sin_t * u2[2],
    ];
    let c2 = [
        -sin_t * u1[0] + cos_t * u2[0],
        -sin_t * u1[1] + cos_t * u2[1],
        -sin_t * u1[2] + cos_t * u2[2],
    ];
    let r = [
        [c1[0], c2[0], c3[0]],
        [c1[1], c2[1], c3[1]],
        [c1[2], c2[2], c3[2]],
    ];
    let q = mat_to_quat(r);
    
    let cos_bp = best_phi.cos();
    let sin_bp = best_phi.sin();
    let z_step = 2.0 * 0.707 * 10.0 + 0.1;
    
    for (k, &(ci, cj)) in best_cells.iter().enumerate() {
        let u_c = (ci as f64 + 0.5) * w_cell + best_dx;
        let v_c = (cj as f64 + 0.5) * h_cell + best_dy;
        
        let tx = u_c * cos_bp - v_c * sin_bp;
        let ty = u_c * sin_bp + v_c * cos_bp;
        
        let pz = (k as f64) * z_step;
        let px = tx * e1[0] + ty * e2[0] + pz * sx / sz;
        let py = tx * e1[1] + ty * e2[1] + pz * sy / sz;
        
        println!(
            "{:.9} {:.9} {:.9} {:.9} {:.9} {:.9} {:.9} 10.0",
            px, py, pz, q[0], q[1], q[2], q[3]
        );
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();
    
    let nv_line = match lines.next() {
        Some(Ok(line)) => line,
        _ => return,
    };
    let nv: usize = match nv_line.trim().parse() {
        Ok(val) => val,
        _ => return,
    };
    
    let mut poly = Vec::with_capacity(nv);
    for _ in 0..nv {
        let line = match lines.next() {
            Some(Ok(l)) => l,
            _ => return,
        };
        let parts: Vec<&str> = line.split_whitespace().collect();
        if parts.len() < 2 {
            return;
        }
        let x: f64 = parts[0].parse().unwrap();
        let y: f64 = parts[1].parse().unwrap();
        poly.push([x, y]);
    }
    
    let sun_line = match lines.next() {
        Some(Ok(l)) => l,
        _ => return,
    };
    let sun_parts: Vec<&str> = sun_line.split_whitespace().collect();
    if sun_parts.len() < 3 {
        return;
    }
    let sx: f64 = sun_parts[0].parse().unwrap();
    let sy: f64 = sun_parts[1].parse().unwrap();
    let sz: f64 = sun_parts[2].parse().unwrap();
    
    solve(poly, [sx, sy, sz]);
}
