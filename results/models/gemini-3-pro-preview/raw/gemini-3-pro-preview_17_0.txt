    public struct Point3D
    {

    }

    public enum BoxComparerType { VolumeAsc, MaxDimAsc, SumDimAsc, MinDimAsc, AreaAsc, VolumeDesc, MaxDimDesc }
    public enum EPComparerType { Z_Y_X, X_Y_Z, Y_Z_X, Sum }
    public enum ScoreType { Z_Y_X, X_Y_Z, Y_Z_X, EnclosingVolume, DistanceSq, FirstFit }

    public class BoxComparer : IComparer<Box>
    {
        public readonly BoxComparerType type;
        public BoxComparer(BoxComparerType type) { this.type = type; }
        
        public int Compare(Box a, Box b)
        {
            switch (type)
            {
                case BoxComparerType.VolumeAsc:
                    {
                        int cmp = a.Volume.CompareTo(b.Volume);
                        if (cmp != 0) return cmp;
                        long maxA = Math.Max(a.W, Math.Max(a.H, a.D));
                        long maxB = Math.Max(b.W, Math.Max(b.H, b.D));
                        return maxA.CompareTo(maxB);
                    }
                case BoxComparerType.MaxDimAsc:
                    {
                        long maxA = Math.Max(a.W, Math.Max(a.H, a.D));
                        long maxB = Math.Max(b.W, Math.Max(b.H, b.D));
                        int cmp = maxA.CompareTo(maxB);
                        if (cmp != 0) return cmp;
                        return a.Volume.CompareTo(b.Volume);
                    }
                case BoxComparerType.SumDimAsc:
                    {
                        long sumA = a.W + a.H + a.D;
                        long sumB = b.W + b.H + b.D;
                        int cmp = sumA.CompareTo(sumB);
                        if (cmp != 0) return cmp;
                        return a.Volume.CompareTo(b.Volume);
                    }
                case BoxComparerType.MinDimAsc:
                    {
                        long minA = Math.Min(a.W, Math.Min(a.H, a.D));
                        long minB = Math.Min(b.W, Math.Min(b.H, b.D));
                        int cmp = minA.CompareTo(minB);
                        if (cmp != 0) return cmp;
                        return a.Volume.CompareTo(b.Volume);
                    }
                case BoxComparerType.AreaAsc:
                    {
                        long areaA = a.W * a.H + a.H * a.D + a.D * a.W;
                        long areaB = b.W * b.H + b.H * b.D + b.D * b.W;
                        int cmp = areaA.CompareTo(areaB);
                        if (cmp != 0) return cmp;
                        return a.Volume.CompareTo(b.Volume);
                    }
                case BoxComparerType.VolumeDesc:
                    {
                        int cmp = b.Volume.CompareTo(a.Volume);
                        if (cmp != 0) return cmp;
                        long maxA = Math.Max(a.W, Math.Max(a.H, a.D));
                        long maxB = Math.Max(b.W, Math.Max(b.H, b.D));
                        return maxB.CompareTo(maxA);
                    }
                case BoxComparerType.MaxDimDesc:
                    {
                        long maxA = Math.Max(a.W, Math.Max(a.H, a.D));
                        long maxB = Math.Max(b.W, Math.Max(b.H, b.D));
                        int cmp = maxB.CompareTo(maxA);
                        if (cmp != 0) return cmp;
                        return b.Volume.CompareTo(a.Volume);
                    }
                default:
                    return a.Id.CompareTo(b.Id);
            }
        }
    }

    public class EPComparer : IComparer<Point3D>
    {
        public readonly EPComparerType type;
        public EPComparer(EPComparerType type) { this.type = type; }
        
        public int Compare(Point3D a, Point3D b)
        {
            switch (type)
            {
                case EPComparerType.Z_Y_X:
                    if (a.Z != b.Z) return a.Z.CompareTo(b.Z);
                    if (a.Y != b.Y) return a.Y.CompareTo(b.Y);
                    return a.X.CompareTo(b.X);
                case EPComparerType.X_Y_Z:
                    if (a.X != b.X) return a.X.CompareTo(b.X);
                    if (a.Y != b.Y) return a.Y.CompareTo(b.Y);
                    return a.Z.CompareTo(b.Z);
                case EPComparerType.Y_Z_X:
                    if (a.Y != b.Y) return a.Y.CompareTo(b.Y);
                    if (a.Z != b.Z) return a.Z.CompareTo(b.Z);
                    return a.X.CompareTo(b.X);
                case EPComparerType.Sum:
                    long sumA = a.X + a.Y + a.Z;
                    long sumB = b.X + b.Y + b.Z;
                    if (sumA != sumB) return sumA.CompareTo(sumB);
                    if (a.Z != b.Z) return a.Z.CompareTo(b.Z);
                    if (a.Y != b.Y) return a.Y.CompareTo(b.Y);
                    return a.X.CompareTo(b.X);
                default:
                    if (a.Z != b.Z) return a.Z.CompareTo(b.Z);
                    if (a.Y != b.Y) return a.Y.CompareTo(b.Y);
                    return a.X.CompareTo(b.X);
            }
        }
    }

    public struct PlacementScore : IComparable<PlacementScore>
    {
        public long Val1;
        public long Val2;
        public long Val3;
        
        public PlacementScore(long v1, long v2, long v3)
        {
            Val1 = v1;
            Val2 = v2;
            Val3 = v3;
        }
        
        public int CompareTo(PlacementScore other)
        {
            if (Val1 != other.Val1) return Val1.CompareTo(other.Val1);
            if (Val2 != other.Val2) return Val2.CompareTo(other.Val2);
            return Val3.CompareTo(other.Val3);
        }
    }

    public class SpatialGrid
    {
        private readonly int gx, gy, gz;
        private readonly long W, H, D;
        private readonly List<BoxPlacement>[,,] grid;
        
        public SpatialGrid(long W, long H, long D, int resolution)
        {
            this.W = W;
            this.H = H;
            this.D = D;
            
            gx = (int)Math.Max(1, Math.Min(W, resolution));
            gy = (int)Math.Max(1, Math.Min(H, resolution));
            gz = (int)Math.Max(1, Math.Min(D, resolution));
            
            grid = new List<BoxPlacement>[gx, gy, gz];
        }
        
        private int GetCellX(long x)
        {
            if (x <= 0) return 0;
            if (W <= 0) return 0;
            int idx = (int)((double)x * gx / W);
            return idx >= gx ? gx - 1 : idx;
        }
        
        private int GetCellY(long y)
        {
            if (y <= 0) return 0;
            if (H <= 0) return 0;
            int idx = (int)((double)y * gy / H);
            return idx >= gy ? gy - 1 : idx;
        }
        
        private int GetCellZ(long z)
        {
            if (z <= 0) return 0;
            if (D <= 0) return 0;
            int idx = (int)((double)z * gz / D);
            return idx >= gz ? gz - 1 : idx;
        }
        
        public void Insert(BoxPlacement box)
        {
            int xMin = GetCellX(box.X);
            int xMax = GetCellX(box.X + box.W - 1);
            int yMin = GetCellY(box.Y);
            int yMax = GetCellY(box.Y + box.H - 1);
            int zMin = GetCellZ(box.Z);
            int zMax = GetCellZ(box.Z + box.D - 1);
            
            for (int x = xMin; x <= xMax; x++)
            {
                for (int y = yMin; y <= yMax; y++)
                {
                    for (int z = zMin; z <= zMax; z++)
                    {
                        if (grid[x, y, z] == null)
                        {
                            grid[x, y, z] = new List<BoxPlacement>();
                        }
                        grid[x, y, z].Add(box);
                    }
                }
            }
        }
        
        public bool Overlaps(long x, long y, long z, long w, long h, long d)
        {
            int xMin = GetCellX(x);
            int xMax = GetCellX(x + w - 1);
            int yMin = GetCellY(y);
            int yMax = GetCellY(y + h - 1);
            int zMin = GetCellZ(z);
            int zMax = GetCellZ(z + d - 1);
            
            for (int cx = xMin; cx <= xMax; cx++)
            {
                for (int cy = yMin; cy <= yMax; cy++)
                {
                    for (int cz = zMin; cz <= zMax; cz++)
                    {
                        var list = grid[cx, cy, cz];
                        if (list != null)
                        {
                            for (int i = 0; i < list.Count; i++)
                            {
                                var other = list[i];
                                if (x < other.X + other.W && x + w > other.X &&
                                    y < other.Y + other.H && y + h > other.Y &&
                                    z < other.Z + other.D && z + d > other.Z)
                                {
                                    return true;
                                }
                            }
                        }
                    }
                }
            }
            return false;
        }
    }

    public class PackingConfig
    {
        public BoxComparer BoxComparer;
        public EPComparer EPComparer;
        public ScoreType ScoreType;
        public int MaxEPsToSearch;
        public int SpatialGridRes;
    }

    public class PackingResult
    {
        public List<BoxPlacement> Placements;
        public PackingConfig Config;
    }

    public class Program
    {
        private static volatile bool allPackedFound = false;

        public static UniqueOrientations GetUniqueOrientations(long w, long h, long d)
        {
            UniqueOrientations u = new UniqueOrientations();
            AddOrientation(ref u, w, h, d);
            AddOrientation(ref u, w, d, h);
            AddOrientation(ref u, h, w, d);
            AddOrientation(ref u, h, d, w);
            AddOrientation(ref u, d, w, h);
            AddOrientation(ref u, d, h, w);
            return u;
        }

        private static void AddOrientation(ref UniqueOrientations u, long w, long h, long d)
        {
            if (u.Count > 0 && u.O1.W == w && u.O1.H == h && u.O1.D == d) return;
            if (u.Count > 1 && u.O2.W == w && u.O2.H == h && u.O2.D == d) return;
            if (u.Count > 2 && u.O3.W == w && u.O3.H == h && u.O3.D == d) return;
            if (u.Count > 3 && u.O4.W == w && u.O4.H == h && u.O4.D == d) return;
            if (u.Count > 4 && u.O5.W == w && u.O5.H == h && u.O5.D == d) return;
            if (u.Count > 5 && u.O6.W == w && u.O6.H == h && u.O6.D == d) return;
            
            switch (u.Count)
            {
                case 0: u.O1 = new Orientation(w, h, d); break;
                case 1: u.O2 = new Orientation(w, h, d); break;
                case 2: u.O3 = new Orientation(w, h, d); break;
                case 3: u.O4 = new Orientation(w, h, d); break;
                case 4: u.O5 = new Orientation(w, h, d); break;
                case 5: u.O6 = new Orientation(w, h, d); break;
            }
            u.Count++;
        }

        public static bool IsPointInsideBox(long px, long py, long pz, BoxPlacement box)
        {
            return px >= box.X && px < box.X + box.W &&
                   py >= box.Y && py < box.Y + box.H &&
                   pz >= box.Z && pz < box.Z + box.D;
        }

        public static PlacementScore CalculateScore(long x, long y, long z, long w, long h, long d, ScoreType type, long currentMaxX, long currentMaxY, long currentMaxZ)
        {
            long fx = x + w;
            long fy = y + h;
            long fz = z + d;
            
            switch (type)
            {
                case ScoreType.Z_Y_X:
                    return new PlacementScore(fz, fy, fx);
                case ScoreType.X_Y_Z:
                    return new PlacementScore(fx, fy, fz);
                case ScoreType.Y_Z_X:
                    return new PlacementScore(fy, fz, fx);
                case ScoreType.EnclosingVolume:
                    long newMaxX = Math.Max(currentMaxX, fx);
                    long newMaxY = Math.Max(currentMaxY, fy);
                    long newMaxZ = Math.Max(currentMaxZ, fz);
                    long vol = newMaxX * newMaxY * newMaxZ;
                    long sum = fx + fy + fz;
                    return new PlacementScore(vol, sum, 0);
                case ScoreType.DistanceSq:
                    long distSq = fx * fx + fy * fy + fz * fz;
                    return new PlacementScore(distSq, 0, 0);
                case ScoreType.FirstFit:
                default:
                    return new PlacementScore(0, 0, 0);
            }
        }

        public static PackingResult RunPacking(List<Box> originalBoxes, long W, long H, long D, PackingConfig config, Stopwatch stopwatch)
        {
            if (allPackedFound) return null;

            var boxes = new List<Box>(originalBoxes);
            boxes.Sort(config.BoxComparer);
            
            var packed = new List<BoxPlacement>();
            var epList = new List<Point3D>();
            epList.Add(new Point3D(0, 0, 0));
            
            var grid = new SpatialGrid(W, H, D, config.SpatialGridRes);
            var epComparer = config.EPComparer;
            
            long currentMaxX = 0;
            long currentMaxY = 0;
            long currentMaxZ = 0;
            
            int boxCount = 0;
            foreach (var box in boxes)
            {
                boxCount++;
                if (boxCount % 128 == 0)
                {
                    if (allPackedFound) return null;
                    if (stopwatch.ElapsedMilliseconds > 28000) break;
                }
                
                bool found = false;
                Point3D bestEP = new Point3D();
                Orientation bestOrient = new Orientation();
                PlacementScore bestScore = new PlacementScore(long.MaxValue, long.MaxValue, long.MaxValue);
                
                int epsToSearch = Math.Min(epList.Count, config.MaxEPsToSearch);
                for (int i = 0; i < epsToSearch; i++)
                {
                    var ep = epList[i];
                    var orientations = GetUniqueOrientations(box.W, box.H, box.D);
                    
                    for (int oIdx = 0; oIdx < orientations.Count; oIdx++)
                    {
                        Orientation orient;
                        switch (oIdx)
                        {
                            case 0: orient = orientations.O1; break;
                            case 1: orient = orientations.O2; break;
                            case 2: orient = orientations.O3; break;
                            case 3: orient = orientations.O4; break;
                            case 4: orient = orientations.O5; break;
                            default: orient = orientations.O6; break;
                        }
                        
                        if (ep.X + orient.W <= W && ep.Y + orient.H <= H && ep.Z + orient.D <= D)
                        {
                            if (!grid.Overlaps(ep.X, ep.Y, ep.Z, orient.W, orient.H, orient.D))
                            {
                                PlacementScore score = CalculateScore(ep.X, ep.Y, ep.Z, orient.W, orient.H, orient.D, config.ScoreType, currentMaxX, currentMaxY, currentMaxZ);
                                if (!found || score.CompareTo(bestScore) < 0)
                                {
                                    found = true;
                                    bestEP = ep;
                                    bestOrient = orient;
                                    bestScore = score;
                                }
                            }
                        }
                    }
                    
                    if (found && config.ScoreType == ScoreType.FirstFit)
                    {
                        break;
                    }
                }
                
                if (found)
                {
                    var placement = new BoxPlacement
                    {
                        Id = box.Id,
                        X = bestEP.X,
                        Y = bestEP.Y,
                        Z = bestEP.Z,
                        W = bestOrient.W,
                        H = bestOrient.H,
                        D = bestOrient.D
                    };
                    
                    packed.Add(placement);
                    grid.Insert(placement);
                    
                    currentMaxX = Math.Max(currentMaxX, placement.X + placement.W);
                    currentMaxY = Math.Max(currentMaxY, placement.Y + placement.H);
                    currentMaxZ = Math.Max(currentMaxZ, placement.Z + placement.D);
                    
                    int epIndex = epList.BinarySearch(bestEP, epComparer);
                    if (epIndex >= 0)
                    {
                        epList.RemoveAt(epIndex);
                    }
                    
                    for (int i = 0; i < epList.Count; i++)
                    {
                        if (IsPointInsideBox(epList[i].X, epList[i].Y, epList[i].Z, placement))
                        {
                            epList.RemoveAt(i);
                            i--;
                        }
                    }
                    
                    long[] newX = { placement.X + placement.W, placement.X, placement.X };
                    long[] newY = { placement.Y, placement.Y + placement.H, placement.Y };
                    long[] newZ = { placement.Z, placement.Z, placement.Z + placement.D };
                    
                    for (int k = 0; k < 3; k++)
                    {
                        long px = newX[k];
                        long py = newY[k];
                        long pz = newZ[k];
                        
                        if (px < W && py < H && pz < D)
                        {
                            if (!grid.Overlaps(px, py, pz, 1, 1, 1))
                            {
                                var newEP = new Point3D(px, py, pz);
                                int insertIndex = epList.BinarySearch(newEP, epComparer);
                                if (insertIndex < 0)
                                {
                                    epList.Insert(~insertIndex, newEP);
                                }
                            }
                        }
                    }
                }
            }
            
            if (packed.Count == originalBoxes.Count)
            {
                allPackedFound = true;
            }
            
            return new PackingResult { Placements = packed, Config = config };
        }

        public static List<PackingConfig> GenerateConfigs(int N)
        {
            var list = new List<PackingConfig>();
            
            int baseMaxEPs = 200;
            int baseGridRes = 32;
            if (N <= 100) { baseMaxEPs = 2000; baseGridRes = 16; }
            else if (N <= 1000) { baseMaxEPs = 600; baseGridRes = 24; }
            else if (N <= 5000) { baseMaxEPs = 300; baseGridRes = 32; }
            else { baseMaxEPs = 150; baseGridRes = 32; }
            
            BoxComparerType[] boxTypes;
            EPComparerType[] epTypes;
            ScoreType[] scoreTypes;

            if (N > 2000)
            {
                boxTypes = new[] {
                    BoxComparerType.VolumeAsc,
                    BoxComparerType.MaxDimAsc,
                    BoxComparerType.AreaAsc
                };
                epTypes = new[] {
                    EPComparerType.Z_Y_X,
                    EPComparerType.Sum
                };
                scoreTypes = new[] {
                    ScoreType.Z_Y_X,
                    ScoreType.EnclosingVolume,
                    ScoreType.FirstFit
                };
            }
            else
            {
                boxTypes = new[] {
                    BoxComparerType.VolumeAsc,
                    BoxComparerType.MaxDimAsc,
                    BoxComparerType.SumDimAsc,
                    BoxComparerType.MinDimAsc,
                    BoxComparerType.AreaAsc,
                    BoxComparerType.VolumeDesc
                };
                epTypes = new[] {
                    EPComparerType.Z_Y_X,
                    EPComparerType.X_Y_Z,
                    EPComparerType.Y_Z_X,
                    EPComparerType.Sum
                };
                scoreTypes = new[] {
                    ScoreType.Z_Y_X,
                    ScoreType.X_Y_Z,
                    ScoreType.EnclosingVolume,
                    ScoreType.DistanceSq,
                    ScoreType.FirstFit
                };
            }
            
            foreach (var bt in boxTypes)
            {
                foreach (var et in epTypes)
                {
                    bool isDesc = (bt == BoxComparerType.VolumeDesc || bt == BoxComparerType.MaxDimDesc);
                    foreach (var st in scoreTypes)
                    {
                        if (isDesc && st != ScoreType.FirstFit && st != ScoreType.EnclosingVolume)
                            continue;
                        
                        list.Add(new PackingConfig
                        {
                            BoxComparer = new BoxComparer(bt),
                            EPComparer = new EPComparer(et),
                            ScoreType = st,
                            MaxEPsToSearch = baseMaxEPs,
                            SpatialGridRes = baseGridRes
                        });
                    }
                }
            }
            
            return list;
        }

        public static long[] ParseLine(string line)
        {
            if (string.IsNullOrWhiteSpace(line)) return null;
            var parts = line.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
            var res = new long[parts.Length];
            for (int i = 0; i < parts.Length; i++)
            {
                res[i] = long.Parse(parts[i]);
            }
            return res;
        }

        public static void Main(string[] args)
        {
            if (args.Length > 0 && args[0] == "test")
            {
                RunSelfTests();
                return;
            }

            string firstLine = Console.ReadLine();
            while (string.IsNullOrWhiteSpace(firstLine))
            {
                firstLine = Console.ReadLine();
                if (firstLine == null) return;
            }

            var firstParts = ParseLine(firstLine);
            int N = (int)firstParts[0];
            long W = firstParts[1];
            long H = firstParts[2];
            long D = firstParts[3];

            var boxes = new List<Box>(N);
            for (int i = 0; i < N; i++)
            {
                string line = Console.ReadLine();
                while (string.IsNullOrWhiteSpace(line))
                {
                    line = Console.ReadLine();
                    if (line == null) break;
                }
                if (line == null) break;
                var parts = ParseLine(line);
                long w = parts[0];
                long h = parts[1];
                long d = parts[2];
                boxes.Add(new Box
                {
                    Id = i,
                    W = w,
                    H = h,
                    D = d,
                    Volume = w * h * d
                });
            }

            var stopwatch = Stopwatch.StartNew();
            var configs = GenerateConfigs(N);
            PackingResult bestResult = null;
            object lockObj = new object();

            Parallel.ForEach(configs, config =>
            {
                if (allPackedFound) return;
                var result = RunPacking(boxes, W, H, D, config, stopwatch);
                if (result != null)
                {
                    lock (lockObj)
                    {
                        if (bestResult == null || result.Placements.Count > bestResult.Placements.Count)
                        {
                            bestResult = result;
                        }
                    }
                }
            });

            if (bestResult != null)
            {
                Console.WriteLine(bestResult.Placements.Count);
                foreach (var p in bestResult.Placements)
                {
                    Console.WriteLine($"{p.Id} {p.X} {p.Y} {p.Z}");
                }
            }
            else
            {
                Console.WriteLine(0);
            }
        }

            sw.Restart();

            Parallel.ForEach(configs, config =>

                var res = RunPacking(boxes, W, H, D, config, sw);

                    lock (lockObj)
                    {
                        if (best == null || res.Placements.Count > best.Placements.Count) best = res;
                    }
                }
            });
            Console.WriteLine($"Packed: {best.Placements.Count} / {boxes.Count}");
            Console.WriteLine($"Best Config: BoxSorter={best.Config.BoxComparer.type}, EP={best.Config.EPComparer.type}, Score={best.Config.ScoreType}");
            Console.WriteLine($"Time elapsed: {sw.ElapsedMilliseconds} ms");
            valid = VerifyPacking(boxes, best.Placements, W, H, D);
            Console.WriteLine($"Valid: {valid}");
        }

        private static bool VerifyPacking(List<Box> originalBoxes, List<BoxPlacement> placements, long W, long H, long D)
        {
            var grid = new SpatialGrid(W, H, D, 32);
            var usedIds = new HashSet<int>();
            
            foreach (var p in placements)
            {
                if (usedIds.Contains(p.Id))
                {
                    Console.WriteLine($"Error: Box {p.Id} placed multiple times!");
                    return false;
                }
                usedIds.Add(p.Id);
                
                if (p.X < 0 || p.X + p.W > W || p.Y < 0 || p.Y + p.H > H || p.Z < 0 || p.Z + p.D > D)
                {
                    Console.WriteLine($"Error: Box {p.Id} at ({p.X},{p.Y},{p.Z}) size ({p.W},{p.H},{p.D}) is out of bounds for container ({W},{H},{D})!");
                    return false;
                }
                
                var orig = originalBoxes[p.Id];
                long[] placedDims = { p.W, p.H, p.D };
                long[] origDims = { orig.W, orig.H, orig.D };
                Array.Sort(placedDims);
                Array.Sort(origDims);
                if (placedDims[0] != origDims[0] || placedDims[1] != origDims[1] || placedDims[2] != origDims[2])
                {
                    Console.WriteLine($"Error: Placed dimensions of box {p.Id} ({p.W},{p.H},{p.D}) do not match original ({orig.W},{orig.H},{orig.D})!");
                    return false;
                }
                
                if (grid.Overlaps(p.X, p.Y, p.Z, p.W, p.H, p.D))
                {
                    Console.WriteLine($"Error: Box {p.Id} at ({p.X},{p.Y},{p.Z}) size ({p.W},{p.H},{p.D}) overlaps with another box!");
                    return false;
                }
                grid.Insert(p);
            }
            
            return true;
        }
