const fs = require('fs');

function computeCriticalLengths(tasks, children, N) {
  const inDegree = new Array(N).fill(0);
  for (let i = 0; i < N; i++) {
    const ch = children[i];
    for (let j = 0; j < ch.length; j++) {
      inDegree[ch[j]]++;
    }
  }
  
  const queue = [];
  for (let i = 0; i < N; i++) {
    if (inDegree[i] === 0) {
      queue.push(i);
    }
  }
  
  const order = [];
  let head = 0;
  while (head < queue.length) {
    const u = queue[head++];
    order.push(u);
    const ch = children[u];
    for (let j = 0; j < ch.length; j++) {
      const v = ch[j];
      inDegree[v]--;
      if (inDegree[v] === 0) {
        queue.push(v);
      }
    }
  }
  
  if (order.length < N) {
    return new Array(N).fill(0);
  }
  
  const critical = new Array(N).fill(0);
  for (let i = N - 1; i >= 0; i--) {
    const u = order[i];
    let maxChildCritical = 0;
    const ch = children[u];
    for (let j = 0; j < ch.length; j++) {
      const v = ch[j];
      if (critical[v] > maxChildCritical) {
        maxChildCritical = critical[v];
      }
    }
    critical[u] = tasks[u].duration + maxChildCritical;
  }
  return critical;
}

function getPriorityKey(rule, task, criticalVal, depReady) {
  const duration = task.duration;
  const deadline = task.deadline;
  const weight = task.weight;
  const id = task.id;
  
  if (rule === "deadline") {
    return [deadline, -weight, duration, id];
  }
  if (rule === "weighted_deadline") {
    return [deadline / Math.max(1, weight), duration, id];
  }
  if (rule === "shortest") {
    return [duration, deadline, id];
  }
  if (rule === "weighted_shortest") {
    return [duration / Math.max(1, weight), deadline, id];
  }
  if (rule === "critical") {
    return [-criticalVal, deadline, id];
  }
  if (rule === "slack") {
    return [deadline - depReady - duration - criticalVal, deadline, id];
  }
  if (rule === "weight") {
    return [-weight, deadline, id];
  }
  if (rule === "longest") {
    return [-duration, deadline, id];
  }
  if (rule === "weighted_longest") {
    return [-weight * duration, deadline, id];
  }
  return [id];
}

function compareKeys(keyA, keyB) {
  const len = Math.min(keyA.length, keyB.length);
  for (let i = 0; i < len; i++) {
    if (keyA[i] < keyB[i]) return -1;
    if (keyA[i] > keyB[i]) return 1;
  }
  return keyA.length - keyB.length;
}

function getDepReadyTime(task, finish) {
  let ready = 0;
  const deps = task.deps;
  for (let i = 0; i < deps.length; i++) {
    const f = finish[deps[i]];
    if (f > ready) {
      ready = f;
    }
  }
  return ready;
}

function scheduleWithRule(tasks, machineCount, children, critical, rule, N) {
  const depsCount = new Array(N);
  for (let i = 0; i < N; i++) {
    depsCount[i] = tasks[i].deps.length;
  }
  
  const machines = machineCount;
  const machineFree = new Array(machines).fill(0);
  const finish = new Array(N).fill(0);
  
  const ready = [];
  const unscheduled = new Array(N).fill(true);
  
  for (let i = 0; i < N; i++) {
    if (depsCount[i] === 0) {
      ready.push(i);
    }
  }
  
  const order = [];
  
  while (ready.length > 0) {
    let bestIdx = 0;
    let bestTid = ready[0];
    let bestKey = getPriorityKey(rule, tasks[bestTid], critical[bestTid], getDepReadyTime(tasks[bestTid], finish));
    
    for (let i = 1; i < ready.length; i++) {
      const tid = ready[i];
      const key = getPriorityKey(rule, tasks[tid], critical[tid], getDepReadyTime(tasks[tid], finish));
      if (compareKeys(key, bestKey) < 0) {
        bestIdx = i;
        bestTid = tid;
        bestKey = key;
      }
    }
    
    ready[bestIdx] = ready[ready.length - 1];
    ready.pop();
    
    unscheduled[bestTid] = false;
    
    const depReady = getDepReadyTime(tasks[bestTid], finish);
    
    let bestM = 0;
    let earliestStart = machineFree[0] > depReady ? machineFree[0] : depReady;
    for (let m = 1; m < machines; m++) {
      const start = machineFree[m] > depReady ? machineFree[m] : depReady;
      if (start < earliestStart) {
        earliestStart = start;
        bestM = m;
      }
    }
    
    const start = earliestStart;
    const end = start + tasks[bestTid].duration;
    finish[bestTid] = end;
    machineFree[bestM] = end;
    
    order.push(bestTid);
    
    const ch = children[bestTid];
    for (let j = 0; j < ch.length; j++) {
      const cid = ch[j];
      depsCount[cid]--;
      if (depsCount[cid] === 0 && unscheduled[cid]) {
        ready.push(cid);
      }
    }
  }
  
  if (order.length < N) {
    return null;
  }
  
  return { order };
}

function scheduleAndGetCost(tasks, machineCount, order, N) {
  const machines = machineCount;
  const machineFree = new Array(machines).fill(0);
  const finish = new Array(N).fill(0);
  let lateness = 0;
  
  for (let i = 0; i < N; i++) {
    const tid = order[i];
    const task = tasks[tid];
    let ready = 0;
    const deps = task.deps;
    for (let j = 0; j < deps.length; j++) {
      const f = finish[deps[j]];
      if (f > ready) {
        ready = f;
      }
    }
    
    let bestM = 0;
    let earliestStart = machineFree[0] > ready ? machineFree[0] : ready;
    for (let m = 1; m < machines; m++) {
      const freeTime = machineFree[m];
      const start = freeTime > ready ? freeTime : ready;
      if (start < earliestStart) {
        earliestStart = start;
        bestM = m;
      }
    }
    
    const start = earliestStart;
    const end = start + task.duration;
    finish[tid] = end;
    machineFree[bestM] = end;
    
    const l = end - task.deadline;
    if (l > 0) {
      lateness += l * task.weight;
    }
  }
  
  let makespan = machineFree[0];
  for (let m = 1; m < machines; m++) {
    if (machineFree[m] > makespan) {
      makespan = machineFree[m];
    }
  }
  
  const cost = lateness + makespan * 0.05;
  return cost;
}

function swapAdjacentPerturb(order, tasks, N) {
  const idx = Math.floor(Math.random() * (N - 1));
  const u = order[idx];
  const v = order[idx + 1];
  
  const uDeps = tasks[v].deps;
  for (let i = 0; i < uDeps.length; i++) {
    if (uDeps[i] === u) {
      return null;
    }
  }
  
  const nextOrder = order.slice();
  nextOrder[idx] = v;
  nextOrder[idx + 1] = u;
  return nextOrder;
}

function insertPerturb(order, tasks, children, N, pos) {
  for (let i = 0; i < N; i++) {
    pos[order[i]] = i;
  }
  
  const idx = Math.floor(Math.random() * N);
  const u = order[idx];
  
  let maxDepIdx = -1;
  const deps = tasks[u].deps;
  for (let i = 0; i < deps.length; i++) {
    const d = deps[i];
    if (pos[d] > maxDepIdx) {
      maxDepIdx = pos[d];
    }
  }
  
  let minChildIdx = N;
  const uChildren = children[u];
  for (let i = 0; i < uChildren.length; i++) {
    const c = uChildren[i];
    if (pos[c] < minChildIdx) {
      minChildIdx = pos[c];
    }
  }
  
  const low = maxDepIdx + 1;
  const high = minChildIdx - 1;
  if (low >= high) {
    return null;
  }
  
  let newIdx = low + Math.floor(Math.random() * (high - low + 1));
  if (newIdx === idx) {
    if (high > low) {
      while (newIdx === idx) {
        newIdx = low + Math.floor(Math.random() * (high - low + 1));
      }
    } else {
      return null;
    }
  }
  
  const nextOrder = order.slice();
  nextOrder.splice(idx, 1);
  nextOrder.splice(newIdx, 0, u);
  return nextOrder;
}

function getBasicTopologicalOrder(tasks, children, N) {
  const inDegree = new Array(N).fill(0);
  for (let i = 0; i < N; i++) {
    const ch = children[i];
    for (let j = 0; j < ch.length; j++) {
      inDegree[ch[j]]++;
    }
  }
  
  const queue = [];
  for (let i = 0; i < N; i++) {
    if (inDegree[i] === 0) {
      queue.push(i);
    }
  }
  
  const order = [];
  while (queue.length > 0) {
    queue.sort((a, b) => a - b);
    const u = queue.shift();
    order.push(u);
    const ch = children[u];
    for (let j = 0; j < ch.length; j++) {
      const v = ch[j];
      inDegree[v]--;
      if (inDegree[v] === 0) {
        queue.push(v);
      }
    }
  }
  return order;
}

function solve(data) {
  const N = data.tasks.length;
  if (N === 0) {
    return { order: [] };
  }
  
  const idMap = new Map();
  const revIdMap = new Array(N);
  for (let i = 0; i < N; i++) {
    const origId = data.tasks[i].id;
    idMap.set(origId, i);
    revIdMap[i] = origId;
  }
  
  const tasks = new Array(N);
  for (let i = 0; i < N; i++) {
    const t = data.tasks[i];
    const mappedDeps = [];
    for (let j = 0; j < t.deps.length; j++) {
      const mappedDep = idMap.get(t.deps[j]);
      if (mappedDep !== undefined) {
        mappedDeps.push(mappedDep);
      }
    }
    tasks[i] = {
      id: i,
      originalId: t.id,
      duration: Number(t.duration),
      deps: mappedDeps,
      deadline: Number(t.deadline),
      weight: Number(t.weight)
    };
  }
  
  const children = Array.from({ length: N }, () => []);
  for (let i = 0; i < N; i++) {
    const deps = tasks[i].deps;
    for (let j = 0; j < deps.length; j++) {
      children[deps[j]].push(i);
    }
  }
  
  const critical = computeCriticalLengths(tasks, children, N);
  
  const rules = [
    "id",
    "deadline",
    "weighted_deadline",
    "shortest",
    "weighted_shortest",
    "critical",
    "slack",
    "weight",
    "longest",
    "weighted_longest"
  ];
  
  const seeds = [];
  const seenCosts = new Set();
  
  function addSeed(order, cost) {
    if (order === null) return;
    if (!seenCosts.has(cost)) {
      seenCosts.add(cost);
      seeds.push({ order, cost });
    }
  }
  
  for (let i = 0; i < rules.length; i++) {
    const result = scheduleWithRule(tasks, data.machineCount, children, critical, rules[i], N);
    if (result !== null) {
      const { order } = result;
      const cost = scheduleAndGetCost(tasks, data.machineCount, order, N);
      addSeed(order, cost);
    }
  }
  
  const basicOrder = getBasicTopologicalOrder(tasks, children, N);
  const basicCost = scheduleAndGetCost(tasks, data.machineCount, basicOrder, N);
  addSeed(basicOrder, basicCost);
  
  if (seeds.length === 0) {
    return { order: data.tasks.map(t => t.id) };
  }
  
  seeds.sort((a, b) => a.cost - b.cost);
  
  let bestOrder = seeds[0].order;
  let bestCost = seeds[0].cost;
  
  let timeLimit = 100;
  if (N > 50) {
    timeLimit = 300;
  }
  if (N > 200) {
    timeLimit = 800;
  }
  if (N > 500) {
    timeLimit = 2000;
  }
  if (N > 1000) {
    timeLimit = 4000;
  }
  
  const numRuns = Math.min(seeds.length, 3);
  const runTimeBudget = timeLimit / numRuns;
  const posBuf = new Array(N);
  
  for (let run = 0; run < numRuns; run++) {
    const seed = seeds[run];
    let currentOrder = seed.order.slice();
    let currentCost = seed.cost;
    
    const T_start = seed.cost * 0.02;
    const T_end = seed.cost * 0.0001;
    
    const runStartTime = Date.now();
    while (true) {
      const elapsed = Date.now() - runStartTime;
      if (elapsed >= runTimeBudget) {
        break;
      }
      
      const progress = elapsed / runTimeBudget;
      const T = T_start * Math.pow(T_end / T_start, progress);
      
      let nextOrder = null;
      if (Math.random() < 0.5) {
        nextOrder = swapAdjacentPerturb(currentOrder, tasks, N);
      } else {
        nextOrder = insertPerturb(currentOrder, tasks, children, N, posBuf);
      }
      
      if (nextOrder === null) {
        continue;
      }
      
      const nextCost = scheduleAndGetCost(tasks, data.machineCount, nextOrder, N);
      const diff = nextCost - currentCost;
      
      if (diff <= 0) {
        currentOrder = nextOrder;
        currentCost = nextCost;
        if (nextCost < bestCost) {
          bestCost = nextCost;
          bestOrder = nextOrder;
        }
      } else {
        const prob = Math.exp(-diff / T);
        if (Math.random() < prob) {
          currentOrder = nextOrder;
          currentCost = nextCost;
        }
      }
      
      if (currentCost > bestCost * 1.1) {
        currentOrder = bestOrder.slice();
        currentCost = bestCost;
      }
    }
  }
  
  const finalOrder = new Array(N);
  for (let i = 0; i < N; i++) {
    finalOrder[i] = revIdMap[bestOrder[i]];
  }
  
  return { order: finalOrder };
}

function fallbackOrder(data) {
  try {
    const tasks = data.tasks;
    const order = [];
    const visited = new Set();
    const temp = new Set();
    
    const taskMap = {};
    for (const t of tasks) {
      taskMap[t.id] = t;
    }
    
    function visit(id) {
      if (visited.has(id)) return;
      if (temp.has(id)) return;
      temp.add(id);
      const t = taskMap[id];
      if (t && t.deps) {
        for (const d of t.deps) {
          visit(d);
        }
      }
      temp.delete(id);
      visited.add(id);
      order.push(id);
    }
    
    for (const t of tasks) {
      visit(t.id);
    }
    return order;
  } catch (e) {
    return data && data.tasks ? data.tasks.map(t => t.id) : [];
  }
}
