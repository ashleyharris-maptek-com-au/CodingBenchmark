const fs = require('fs');

function solve() {
    // Read input from stdin
    let input;
    try {
        input = fs.readFileSync(0, 'utf8');
    } catch (err) {
        // Fallback for environment issues
        return;
    }
    if (!input || input.trim() === '') {
        return;
    }
    const data = JSON.parse(input);

    if (!data.robots || data.robots.length === 0) {
        console.log(JSON.stringify({ assignments: [] }));
        return;
    }

    const R = data.robots.length;
    const J = data.jobs ? data.jobs.length : 0;

    // Grid details
    const grid = data.grid;
    const H_orig = grid.length;
    const W_orig = grid[0].length;
    const W = W_orig + 2;
    const H = H_orig + 2;

    const BLOCKED_VAL = 1000000000;
    const visitedOrBlocked = new Int32Array(W * H);

    // Fill borders as blocked
    for (let x = 0; x < W; x++) {
        visitedOrBlocked[x] = BLOCKED_VAL;
        visitedOrBlocked[(H - 1) * W + x] = BLOCKED_VAL;
    }
    for (let y = 0; y < H; y++) {
        visitedOrBlocked[y * W] = BLOCKED_VAL;
        visitedOrBlocked[y * W + (W - 1)] = BLOCKED_VAL;
    }

    // Fill inside grid from input ('.' is 46, '#' is 35)
    for (let y = 0; y < H_orig; y++) {
        const row = grid[y];
        const rowOffset = (y + 1) * W;
        for (let x = 0; x < W_orig; x++) {
            if (row.charCodeAt(x) === 35) {
                visitedOrBlocked[rowOffset + (x + 1)] = BLOCKED_VAL;
            }
        }
    }

    // Map targets: Robots first (0 to R-1), then Jobs (R to R+J-1)
    const targets = [];
    const targetIndex = new Int32Array(W * H);

    for (let i = 0; i < R; i++) {
        const r = data.robots[i];
        const cellIdx = (r.y + 1) * W + (r.x + 1);
        targets.push({
            type: 'robot',
            id: r.id,
            x: r.x,
            y: r.y,
            cellIdx
        });
        targetIndex[cellIdx] = i + 1;
    }

    for (let i = 0; i < J; i++) {
        const j = data.jobs[i];
        const cellIdx = (j.y + 1) * W + (j.x + 1);
        targets.push({
            type: 'job',
            id: j.id,
            x: j.x,
            y: j.y,
            priority: typeof j.priority === 'number' ? j.priority : 0,
            cellIdx
        });
        targetIndex[cellIdx] = R + i + 1;
    }

    const N = R + J;

    // Component labeling
    const component = new Int32Array(W * H);
    let nextCompId = 1;
    const compQueue = new Int32Array(W * H);
    const compVisited = new Uint8Array(W * H);

    for (let i = 0; i < W * H; i++) {
        if (visitedOrBlocked[i] !== BLOCKED_VAL && compVisited[i] === 0) {
            let head = 0;
            let tail = 0;
            compVisited[i] = 1;
            component[i] = nextCompId;
            compQueue[tail++] = i;
            
            while (head < tail) {
                const curr = compQueue[head++];
                
                const left = curr - 1;
                if (visitedOrBlocked[left] !== BLOCKED_VAL && compVisited[left] === 0) {
                    compVisited[left] = 1;
                    component[left] = nextCompId;
                    compQueue[tail++] = left;
                }
                const right = curr + 1;
                if (visitedOrBlocked[right] !== BLOCKED_VAL && compVisited[right] === 0) {
                    compVisited[right] = 1;
                    component[right] = nextCompId;
                    compQueue[tail++] = right;
                }
                const up = curr - W;
                if (visitedOrBlocked[up] !== BLOCKED_VAL && compVisited[up] === 0) {
                    compVisited[up] = 1;
                    component[up] = nextCompId;
                    compQueue[tail++] = up;
                }
                const down = curr + W;
                if (visitedOrBlocked[down] !== BLOCKED_VAL && compVisited[down] === 0) {
                    compVisited[down] = 1;
                    component[down] = nextCompId;
                    compQueue[tail++] = down;
                }
            }
            nextCompId++;
        }
    }

    const compTargetCounts = new Int32Array(nextCompId);
    for (let i = 0; i < N; i++) {
        const idx = targets[i].cellIdx;
        const compId = component[idx];
        if (compId > 0) {
            compTargetCounts[compId]++;
        }
    }

    // Lazy BFS setup
    const distMatrix = new Array(N);
    const bfsRun = new Uint8Array(N);
    const dist = new Int32Array(W * H);
    const queue = new Int32Array(W * H);
    let visitedToken = 0;

    function runBFS(srcTargetIdx) {
        bfsRun[srcTargetIdx] = 1;
        const row = new Int32Array(N).fill(-1);
        distMatrix[srcTargetIdx] = row;
        
        const startIdx = targets[srcTargetIdx].cellIdx;
        if (visitedOrBlocked[startIdx] === BLOCKED_VAL) {
            return;
        }
        
        visitedToken++;
        let head = 0;
        let tail = 0;
        
        dist[startIdx] = 0;
        visitedOrBlocked[startIdx] = visitedToken;
        queue[tail++] = startIdx;
        
        let foundTargets = 0;
        const compId = component[startIdx];
        const targetCount = compTargetCounts[compId];
        
        while (head < tail) {
            const curr = queue[head++];
            const d = dist[curr];
            
            const tid = targetIndex[curr];
            if (tid > 0) {
                row[tid - 1] = d;
                foundTargets++;
                if (foundTargets === targetCount) {
                    break;
                }
            }
            
            const nextD = d + 1;
            
            const left = curr - 1;
            if (visitedOrBlocked[left] < visitedToken) {
                visitedOrBlocked[left] = visitedToken;
                dist[left] = nextD;
                queue[tail++] = left;
            }
            const right = curr + 1;
            if (visitedOrBlocked[right] < visitedToken) {
                visitedOrBlocked[right] = visitedToken;
                dist[right] = nextD;
                queue[tail++] = right;
            }
            const up = curr - W;
            if (visitedOrBlocked[up] < visitedToken) {
                visitedOrBlocked[up] = visitedToken;
                dist[up] = nextD;
                queue[tail++] = up;
            }
            const down = curr + W;
            if (visitedOrBlocked[down] < visitedToken) {
                visitedOrBlocked[down] = visitedToken;
                dist[down] = nextD;
                queue[tail++] = down;
            }
        }
    }

    function getDistance(i, j) {
        if (i === j) return 0;
        if (i < R && j < R) return -1;
        
        const jobIdx = i >= R ? i : j;
        const otherIdx = i >= R ? j : i;
        
        if (bfsRun[jobIdx] === 0) {
            runBFS(jobIdx);
        }
        return distMatrix[jobIdx][otherIdx];
    }

    function getDistanceWithPenalty(i, j) {
        const d = getDistance(i, j);
        return d === -1 ? 99999999 : d;
    }

    function computePathCost(robotIdx, path) {
        if (path.length === 0) return 0;
        let cost = getDistanceWithPenalty(robotIdx, path[0]);
        for (let i = 0; i < path.length - 1; i++) {
            cost += getDistanceWithPenalty(path[i], path[i + 1]);
        }
        return cost;
    }

    function computeTotalCost(paths) {
        let total = 0;
        for (let r = 0; r < R; r++) {
            total += computePathCost(r, paths[r]);
        }
        return total;
    }

    // 1. Build Priority Greedy Heuristic
    function buildPriorityGreedy() {
        const sortedJobIndices = Array.from({ length: J }, (_, i) => R + i);
        sortedJobIndices.sort((a, b) => {
            const pa = targets[a].priority;
            const pb = targets[b].priority;
            if (pb !== pa) return pb - pa;
            return a - b;
        });

        const paths = Array.from({ length: R }, () => []);
        const robotEndIdx = Array.from({ length: R }, (_, r) => r);

        for (const jobIdx of sortedJobIndices) {
            let bestRobot = 0;
            let bestDist = Infinity;
            for (let r = 0; r < R; r++) {
                const d = getDistance(robotEndIdx[r], jobIdx);
                if (d !== -1 && d < bestDist) {
                    bestDist = d;
                    bestRobot = r;
                }
            }
            if (bestDist === Infinity) {
                let fallbackRobot = 0;
                let fallbackDist = Infinity;
                for (let r = 0; r < R; r++) {
                    const d = getDistance(r, jobIdx);
                    if (d !== -1 && d < fallbackDist) {
                        fallbackDist = d;
                        fallbackRobot = r;
                    }
                }
                paths[fallbackRobot].push(jobIdx);
                robotEndIdx[fallbackRobot] = jobIdx;
            } else {
                paths[bestRobot].push(jobIdx);
                robotEndIdx[bestRobot] = jobIdx;
            }
        }
        return paths;
    }

    // 2. Build Distance Greedy Heuristic
    function buildDistanceGreedy() {
        const paths = Array.from({ length: R }, () => []);
        const robotEndIdx = Array.from({ length: R }, (_, r) => r);
        const unassigned = new Set(Array.from({ length: J }, (_, i) => R + i));
        
        while (unassigned.size > 0) {
            let bestRobot = -1;
            let bestJob = -1;
            let bestDist = Infinity;
            
            for (let r = 0; r < R; r++) {
                const endIdx = robotEndIdx[r];
                for (const jobIdx of unassigned) {
                    const d = getDistance(endIdx, jobIdx);
                    if (d !== -1 && d < bestDist) {
                        bestDist = d;
                        bestRobot = r;
                        bestJob = jobIdx;
                    }
                }
            }
            
            if (bestDist === Infinity) {
                for (const jobIdx of unassigned) {
                    let fallbackRobot = 0;
                    let fallbackDist = Infinity;
                    for (let r = 0; r < R; r++) {
                        const d = getDistance(r, jobIdx);
                        if (d !== -1 && d < fallbackDist) {
                            fallbackDist = d;
                            fallbackRobot = r;
                        }
                    }
                    paths[fallbackRobot].push(jobIdx);
                }
                break;
            } else {
                paths[bestRobot].push(bestJob);
                robotEndIdx[bestRobot] = bestJob;
                unassigned.delete(bestJob);
            }
        }
        return paths;
    }

    // 3. Optimize paths using Local Search with a safety timer
    const startTime = Date.now();
    const TIME_LIMIT = 8000; // 8 seconds

    function optimize(paths) {
        let improved = true;
        while (improved) {
            improved = false;
            
            if (Date.now() - startTime > TIME_LIMIT) {
                break;
            }
            
            // 2-opt loops
            let improved2opt = true;
            while (improved2opt) {
                improved2opt = false;
                if (Date.now() - startTime > TIME_LIMIT) break;
                for (let r = 0; r < R; r++) {
                    const path = paths[r];
                    const K = path.length;
                    if (K < 2) continue;
                    for (let i = 0; i < K - 1; i++) {
                        for (let j = i + 1; j < K; j++) {
                            const p_prev = i === 0 ? r : path[i - 1];
                            const p_i = path[i];
                            const p_j = path[j];
                            const p_next = j === K - 1 ? null : path[j + 1];
                            
                            const d1 = getDistanceWithPenalty(p_prev, p_i);
                            const d2 = p_next === null ? 0 : getDistanceWithPenalty(p_j, p_next);
                            
                            const r1 = getDistanceWithPenalty(p_prev, p_j);
                            const r2 = p_next === null ? 0 : getDistanceWithPenalty(p_i, p_next);
                            
                            const costDiff = r1 + r2 - (d1 + d2);
                            if (costDiff < -1e-6) {
                                let left = i, right = j;
                                while (left < right) {
                                    const tmp = path[left];
                                    path[left] = path[right];
                                    path[right] = tmp;
                                    left++;
                                    right--;
                                }
                                improved2opt = true;
                                improved = true;
                            }
                        }
                    }
                }
            }
            
            if (Date.now() - startTime > TIME_LIMIT) break;
            
            // Relocate loops
            let improvedRelocate = false;
            for (let r1 = 0; r1 < R; r1++) {
                for (let r2 = 0; r2 < R; r2++) {
                    if (r1 === r2) continue;
                    const path1 = paths[r1];
                    const path2 = paths[r2];
                    for (let i = 0; i < path1.length; i++) {
                        const jobIdx = path1[i];
                        const p_prev = i === 0 ? r1 : path1[i - 1];
                        const p_next = i === path1.length - 1 ? null : path1[i + 1];
                        
                        const d_remove_old = getDistanceWithPenalty(p_prev, jobIdx) + (p_next === null ? 0 : getDistanceWithPenalty(jobIdx, p_next));
                        const d_remove_new = p_next === null ? 0 : getDistanceWithPenalty(p_prev, p_next);
                        const cost_change_1 = d_remove_new - d_remove_old;
                        
                        let bestJ = -1;
                        let bestInsertCost = Infinity;
                        for (let j = 0; j <= path2.length; j++) {
                            const q_prev = j === 0 ? r2 : path2[j - 1];
                            const q_next = j === path2.length ? null : path2[j];
                            const d_insert_old = q_next === null ? 0 : getDistanceWithPenalty(q_prev, q_next);
                            const d_insert_new = getDistanceWithPenalty(q_prev, jobIdx) + (q_next === null ? 0 : getDistanceWithPenalty(jobIdx, q_next));
                            const cost_change_2 = d_insert_new - d_insert_old;
                            const total_change = cost_change_1 + cost_change_2;
                            if (total_change < bestInsertCost) {
                                bestInsertCost = total_change;
                                bestJ = j;
                            }
                        }
                        if (bestInsertCost < -1e-6) {
                            path1.splice(i, 1);
                            path2.splice(bestJ, 0, jobIdx);
                            improvedRelocate = true;
                            improved = true;
                            break;
                        }
                    }
                    if (improvedRelocate) break;
                }
                if (improvedRelocate) break;
            }
            if (improvedRelocate) continue;
            
            if (Date.now() - startTime > TIME_LIMIT) break;
            
            // Swap loops
            let improvedSwap = false;
            for (let r1 = 0; r1 < R - 1; r1++) {
                for (let r2 = r1 + 1; r2 < R; r2++) {
                    const path1 = paths[r1];
                    const path2 = paths[r2];
                    for (let i = 0; i < path1.length; i++) {
                        for (let j = 0; j < path2.length; j++) {
                            const p_prev = i === 0 ? r1 : path1[i - 1];
                            const p_i = path1[i];
                            const p_next = i === path1.length - 1 ? null : path1[i + 1];
                            
                            const q_prev = j === 0 ? r2 : path2[j - 1];
                            const q_j = path2[j];
                            const q_next = j === path2.length - 1 ? null : path2[j + 1];
                            
                            const old_cost_1 = getDistanceWithPenalty(p_prev, p_i) + (p_next === null ? 0 : getDistanceWithPenalty(p_i, p_next));
                            const new_cost_1 = getDistanceWithPenalty(p_prev, q_j) + (p_next === null ? 0 : getDistanceWithPenalty(q_j, p_next));
                            
                            const old_cost_2 = getDistanceWithPenalty(q_prev, q_j) + (q_next === null ? 0 : getDistanceWithPenalty(q_j, q_next));
                            const new_cost_2 = getDistanceWithPenalty(q_prev, p_i) + (q_next === null ? 0 : getDistanceWithPenalty(p_i, q_next));
                            
                            const total_change = (new_cost_1 - old_cost_1) + (new_cost_2 - old_cost_2);
                            if (total_change < -1e-6) {
                                path1[i] = q_j;
                                path2[j] = p_i;
                                improvedSwap = true;
                                improved = true;
                                break;
                            }
                        }
                        if (improvedSwap) break;
                    }
                    if (improvedSwap) break;
                }
                if (improvedSwap) break;
            }
        }
    }

    // Build and optimize both candidates
    const pathsPG = buildPriorityGreedy();
    optimize(pathsPG);
    const costPG = computeTotalCost(pathsPG);

    const pathsDG = buildDistanceGreedy();
    optimize(pathsDG);
    const costDG = computeTotalCost(pathsDG);

    const bestPaths = costPG < costDG ? pathsPG : pathsDG;

    // Build assignments matching the requested output format
    function getRobotIdAsIndex(robot, index) {
        const id = robot.id;
        if (typeof id === 'number') {
            return id;
        }
        const parsed = parseInt(id);
        if (!isNaN(parsed) && String(parsed) === String(id).trim()) {
            return parsed;
        }
        return index;
    }

    const maxIndex = data.robots.reduce((max, r, idx) => {
        return Math.max(max, getRobotIdAsIndex(r, idx));
    }, 0);

    const assignments = Array.from({ length: maxIndex + 1 }, () => []);

    for (let i = 0; i < R; i++) {
        const robot = data.robots[i];
        const robotIdxInAssignments = getRobotIdAsIndex(robot, i);
        const path = bestPaths[i];
        assignments[robotIdxInAssignments] = path.map(jobTargetIdx => targets[jobTargetIdx].id);
    }

    console.log(JSON.stringify({ assignments }));
}

solve();
