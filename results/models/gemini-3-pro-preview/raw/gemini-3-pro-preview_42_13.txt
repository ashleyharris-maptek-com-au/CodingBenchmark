OpCapability Shader
OpCapability Geometry
%glsl = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %gl_out %outColor %worldPos %normal %color
OpExecutionMode %main InputTrianglesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 12
OpExecutionMode %main Invocations 1

; Decorations
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position

OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2

OpDecorate %outColor Location 0

OpDecorate %ubo_struct Block
OpMemberDecorate %ubo_struct 0 ColMajor
OpMemberDecorate %ubo_struct 0 MatrixStride 16
OpMemberDecorate %ubo_struct 0 Offset 0
OpMemberDecorate %ubo_struct 1 ColMajor
OpMemberDecorate %ubo_struct 1 MatrixStride 16
OpMemberDecorate %ubo_struct 1 Offset 64
OpMemberDecorate %ubo_struct 2 ColMajor
OpMemberDecorate %ubo_struct 2 MatrixStride 16
OpMemberDecorate %ubo_struct 2 Offset 128
OpMemberDecorate %ubo_struct 3 Offset 192
OpMemberDecorate %ubo_struct 4 Offset 208
OpMemberDecorate %ubo_struct 5 Offset 224

OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0

; Types
%void = OpTypeVoid
%void_func = OpTypeFunction %void
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%mat4 = OpTypeMatrix %v4float 4
%uint = OpTypeInt 32 0

; Constants
%uint_6 = OpConstant %uint 6
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3
%int_4 = OpConstant %int 4
%int_5 = OpConstant %int 5
%float_0_5 = OpConstant %float 0.5
%float_0_004 = OpConstant %float 0.004

; Pointers
%ptr_in_v4float = OpTypePointer Input %v4float
%ptr_in_v3float = OpTypePointer Input %v3float
%ptr_out_v4float = OpTypePointer Output %v4float
%ptr_out_v3float = OpTypePointer Output %v3float

; Input variables

%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_6

%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input

; Output variables

%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

%outColor = OpVariable %ptr_out_v3float Output

%label = OpLabel

; =====================================================================
; --- Edge 0 (0 -> 2) ---
; =====================================================================
; Load positions
%ptr_P0 = OpAccessChain %ptr_in_v4float %gl_in %int_0 %int_0
%P0 = OpLoad %v4float %ptr_P0
%ptr_P2 = OpAccessChain %ptr_in_v4float %gl_in %int_2 %int_0
%P2 = OpLoad %v4float %ptr_P2

; Load colors
%ptr_C0 = OpAccessChain %ptr_in_v3float %color %int_0
%C0 = OpLoad %v3float %ptr_C0
%ptr_C2 = OpAccessChain %ptr_in_v3float %color %int_2
%C2 = OpLoad %v3float %ptr_C2

; Compute average color
%color_sum0 = OpFAdd %v3float %C0 %C2
%avg_color0 = OpVectorTimesScalar %v3float %color_sum0 %float_0_5

; Extract components of P0
%x0 = OpCompositeExtract %float %P0 0
%y0 = OpCompositeExtract %float %P0 1
%z0 = OpCompositeExtract %float %P0 2
%w0 = OpCompositeExtract %float %P0 3

; Extract components of P2
%x2 = OpCompositeExtract %float %P2 0
%y2 = OpCompositeExtract %float %P2 1
%z2 = OpCompositeExtract %float %P2 2
%w2 = OpCompositeExtract %float %P2 3

; Compute NDC coordinates
%ndc0_x = OpFDiv %float %x0 %w0
%ndc0_y = OpFDiv %float %y0 %w0
%ndc0 = OpCompositeConstruct %v2float %ndc0_x %ndc0_y

%ndc2_x = OpFDiv %float %x2 %w2
%ndc2_y = OpFDiv %float %y2 %w2
%ndc2 = OpCompositeConstruct %v2float %ndc2_x %ndc2_y

; Direction vector
%dir0 = OpFSub %v2float %ndc2 %ndc0
%dir_norm0 = OpExtInst %v2float %glsl Normalize %dir0

; Normal vector
%dir_norm0_x = OpCompositeExtract %float %dir_norm0 0
%dir_norm0_y = OpCompositeExtract %float %dir_norm0 1
%neg_dir_norm0_y = OpFNegate %float %dir_norm0_y
%normal0 = OpCompositeConstruct %v2float %neg_dir_norm0_y %dir_norm0_x

; Offset
%offset0 = OpVectorTimesScalar %v2float %normal0 %float_0_004
%offset0_x = OpCompositeExtract %float %offset0 0
%offset0_y = OpCompositeExtract %float %offset0 1

; Offset scaled by W
%offset0_w0_x = OpFMul %float %offset0_x %w0
%offset0_w0_y = OpFMul %float %offset0_y %w0

%offset0_w2_x = OpFMul %float %offset0_x %w2
%offset0_w2_y = OpFMul %float %offset0_y %w2

; --- Emit 4 vertices for Edge 0 ---

; Vertex 0_0: P0 + offset0
%pos0_0_x = OpFAdd %float %x0 %offset0_w0_x
%pos0_0_y = OpFAdd %float %y0 %offset0_w0_y
%pos0_0 = OpCompositeConstruct %v4float %pos0_0_x %pos0_0_y %z0 %w0
%ptr_out_pos_0_0 = OpAccessChain %ptr_out_v4float %gl_out %int_0
OpStore %ptr_out_pos_0_0 %pos0_0
OpStore %outColor %avg_color0
OpEmitVertex

; Vertex 0_1: P0 - offset0
%neg_offset0_w0_x = OpFNegate %float %offset0_w0_x
%neg_offset0_w0_y = OpFNegate %float %offset0_w0_y
%pos0_1_x = OpFAdd %float %x0 %neg_offset0_w0_x
%pos0_1_y = OpFAdd %float %y0 %neg_offset0_w0_y
%pos0_1 = OpCompositeConstruct %v4float %pos0_1_x %pos0_1_y %z0 %w0
%ptr_out_pos_0_1 = OpAccessChain %ptr_out_v4float %gl_out %int_0
OpStore %ptr_out_pos_0_1 %pos0_1
OpStore %outColor %avg_color0
OpEmitVertex

; Vertex 0_2: P2 + offset0
%pos0_2_x = OpFAdd %float %x2 %offset0_w2_x
%pos0_2_y = OpFAdd %float %y2 %offset0_w2_y
%pos0_2 = OpCompositeConstruct %v4float %pos0_2_x %pos0_2_y %z2 %w2
%ptr_out_pos_0_2 = OpAccessChain %ptr_out_v4float %gl_out %int_0
OpStore %ptr_out_pos_0_2 %pos0_2
OpStore %outColor %avg_color0
OpEmitVertex

; Vertex 0_3: P2 - offset0
%neg_offset0_w2_x = OpFNegate %float %offset0_w2_x
%neg_offset0_w2_y = OpFNegate %float %offset0_w2_y
%pos0_3_x = OpFAdd %float %x2 %neg_offset0_w2_x
%pos0_3_y = OpFAdd %float %y2 %neg_offset0_w2_y
%pos0_3 = OpCompositeConstruct %v4float %pos0_3_x %pos0_3_y %z2 %w2
%ptr_out_pos_0_3 = OpAccessChain %ptr_out_v4float %gl_out %int_0
OpStore %ptr_out_pos_0_3 %pos0_3
OpStore %outColor %avg_color0

OpEndPrimitive
