/*
 * 3D Bin Packing Solver in C#
 * 
 * Key Features and Design:
 * 1. Adaptive Scaling & Voxelization:
 *    - Automatically chooses voxel resolution (dx) dynamically based on container volume and polyhedron dimensions.
 *    - Caps the grid size at ~12 million cells to guarantee sub-second execution and small memory footprint (approx. 15MB) for extreme scales.
 *    - Scales up resolution for small/medium scales to maximize precision.
 * 2. Robust Polyhedron Inside Checking:
 *    - Implements Möller-Trumbore ray-triangle intersection.
 *    - Employs 3-ray majority voting to eliminate numerical edge cases and singularities.
 *    - Samples edges and faces to ensure exact boundary coverage.
 * 3. 24 Unique Standard 3D Orientations:
 *    - Precomputes rotated basis arrays and unique quaternions to allow snug packing in all valid 90-degree rotations.
 * 4. Hybrid Adaptive Search:
 *    - Grid-Packing: Computes the maximum regular packing density of the best orientation in O(1) time.
 *    - Extreme Points (EP) Packer: Uses the stable Extreme Points heuristic sorted layer-by-layer (z-first) to pack items snugly.
 *    - Score-based Fit: Employs a contact area metric to maximize snug fit against container walls and already packed items.
 *    - For small/medium scales (< 1500 items), tries both Pure EP and Hybrid to select the absolute best packing count.
 *    - For large/extreme scales (>= 1500 items), seeds the grid with the optimal regular grid packing and fills the boundary slabs.
 */

using System;
using System.IO;
using System.Collections.Generic;
using System.Globalization;
using System.Numerics;

public class Program
{
    public class OrientationInfo
    {
        public int Index { get; set; }
        public Quaternion Rotation { get; set; }
        public Vector3 MinRotatedPoint { get; set; }
        public List<(int x, int y, int z)> Voxels { get; set; }
        public int Width { get; set; }
        public int Height { get; set; }
        public int Depth { get; set; }
    }

    public class PlacedItem
    {
        public int OrientationIndex { get; set; }
        public Quaternion Rotation { get; set; }
        public Vector3 Translation { get; set; }
        public int Px { get; set; }
        public int Py { get; set; }
        public int Pz { get; set; }
    }

    public class BitGrid3D
    {
        private readonly ulong[] _data;
        public int Width { get; }
        public int Height { get; }
        public int Depth { get; }
        private readonly int _strideY;
        private readonly int _strideZ;

        public BitGrid3D(int width, int height, int depth)
        {
            Width = width;
            Height = height;
            Depth = depth;
            _strideY = width;
            _strideZ = width * height;
            long totalBits = (long)width * height * depth;
            long size = (totalBits + 63) / 64;
            _data = new ulong[size];
        }

        public bool Get(int x, int y, int z)
        {
            long index = (long)z * _strideZ + (long)y * _strideY + x;
            int wordIndex = (int)(index >> 6);
            int bitIndex = (int)(index & 63);
            return (_data[wordIndex] & (1UL << bitIndex)) != 0;
        }

        public void Set(int x, int y, int z, bool value)
        {
            long index = (long)z * _strideZ + (long)y * _strideY + x;
            int wordIndex = (int)(index >> 6);
            int bitIndex = (int)(index & 63);
            if (value)
                _data[wordIndex] |= (1UL << bitIndex);
            else
                _data[wordIndex] &= ~(1UL << bitIndex);
        }
    }

    public class ExtremePointComparer : IComparer<(int x, int y, int z)>
    {
        public int Compare((int x, int y, int z) a, (int x, int y, int z) b)
        {
            if (a.z != b.z) return a.z.CompareTo(b.z);
            if (a.y != b.y) return a.y.CompareTo(b.y);
            return a.x.CompareTo(b.x);
        }
    }

    public static void Main()
    {
        CultureInfo.DefaultThreadCurrentCulture = CultureInfo.InvariantCulture;
        CultureInfo.DefaultThreadCurrentUICulture = CultureInfo.InvariantCulture;

        Vector3[] vertices;
        int[][] faces;
        float W, H, D;

        try
        {
            ParseInput(out vertices, out faces, out W, out H, out D);
        }
        catch (Exception)
        {
            Console.WriteLine(0);
            return;
        }

        if (vertices.Length == 0 || faces.Length == 0 || W <= 0 || H <= 0 || D <= 0)
        {
            Console.WriteLine(0);
            return;
        }

        var orientationsList = GenerateUniqueOrientations();

        float minPolyDim = float.MaxValue;
        foreach (var q in orientationsList)
        {
            float minX = float.MaxValue, minY = float.MaxValue, minZ = float.MaxValue;
            float maxX = float.MinValue, maxY = float.MinValue, maxZ = float.MinValue;
            foreach (var v in vertices)
            {
                Vector3 r = Vector3.Transform(v, q);
                minX = Math.Min(minX, r.X);
                minY = Math.Min(minY, r.Y);
                minZ = Math.Min(minZ, r.Z);
                maxX = Math.Max(maxX, r.X);
                maxY = Math.Max(maxY, r.Y);
                maxZ = Math.Max(maxZ, r.Z);
            }
            float dxVal = maxX - minX;
            float dyVal = maxY - minY;
            float dzVal = maxZ - minZ;
            minPolyDim = Math.Min(minPolyDim, Math.Min(dxVal, Math.Min(dyVal, dzVal)));
        }

        if (minPolyDim <= 0)
        {
            minPolyDim = 1.0f;
        }

        float dx = minPolyDim / 6.0f;
        double totalCells = ((double)W / dx) * ((double)H / dx) * ((double)D / dx);

        if (totalCells > 12000000.0)
        {
            dx = (float)Math.Pow((double)W * H * D / 12000000.0, 1.0 / 3.0);
        }
        else if (totalCells < 500000.0)
        {
            float optimalDx = (float)Math.Pow((double)W * H * D / 1000000.0, 1.0 / 3.0);
            if (optimalDx < dx)
            {
                dx = Math.Max(optimalDx, minPolyDim / 12.0f);
            }
        }
        dx = Math.Max(dx, 1e-4f);

        var orientations = PrecomputeOrientations(vertices, faces, dx, orientationsList);

        int NX = (int)Math.Floor(W / dx);
        int NY = (int)Math.Floor(H / dx);
        int NZ = (int)Math.Floor(D / dx);

        if (NX <= 0 || NY <= 0 || NZ <= 0)
        {
            Console.WriteLine(0);
            return;
        }

        OrientationInfo bestGridOri = null;
        int maxGridCount = 0;
        int gx = 0, gy = 0, gz = 0;

        foreach (var ori in orientations)
        {
            if (ori.Width > 0 && ori.Height > 0 && ori.Depth > 0)
            {
                int nx = NX / ori.Width;
                int ny = NY / ori.Height;
                int nz = NZ / ori.Depth;
                int count = nx * ny * nz;
                if (count > maxGridCount)
                {
                    maxGridCount = count;
                    bestGridOri = ori;
                    gx = nx;
                    gy = ny;
                    gz = nz;
                }
            }
        }

        List<PlacedItem> finalPacking = null;

        if (maxGridCount < 1500)
        {
            BitGrid3D gridPure = new BitGrid3D(NX, NY, NZ);
            List<PlacedItem> pureEPPacking = RunEPPacker(NX, NY, NZ, orientations, gridPure, new List<(int x, int y, int z)> { (0, 0, 0) }, dx);

            List<PlacedItem> hybridPacking = null;
            if (maxGridCount > 0 && bestGridOri != null)
            {
                BitGrid3D gridHybrid = new BitGrid3D(NX, NY, NZ);
                var hybridInitial = new List<PlacedItem>();
                for (int i = 0; i < gx; i++)
                {
                    for (int j = 0; j < gy; j++)
                    {
                        for (int k = 0; k < gz; k++)
                        {
                            int px = i * bestGridOri.Width;
                            int py = j * bestGridOri.Height;
                            int pz = k * bestGridOri.Depth;

                            for (int vIdx = 0; vIdx < bestGridOri.Voxels.Count; vIdx++)
                            {
                                var v = bestGridOri.Voxels[vIdx];
                                gridHybrid.Set(px + v.x, py + v.y, pz + v.z, true);
                            }

                            Vector3 translation = new Vector3(
                                px * dx - bestGridOri.MinRotatedPoint.X,
                                py * dx - bestGridOri.MinRotatedPoint.Y,
                                pz * dx - bestGridOri.MinRotatedPoint.Z
                            );

                            hybridInitial.Add(new PlacedItem
                            {
                                OrientationIndex = bestGridOri.Index,
                                Rotation = bestGridOri.Rotation,
                                Translation = translation,
                                Px = px,
                                Py = py,
                                Pz = pz
                            });
                        }
                    }
                }

                var seeds = new List<(int x, int y, int z)>();
                if (gx * bestGridOri.Width < NX) seeds.Add((gx * bestGridOri.Width, 0, 0));
                if (gy * bestGridOri.Height < NY) seeds.Add((0, gy * bestGridOri.Height, 0));
                if (gz * bestGridOri.Depth < NZ) seeds.Add((0, 0, gz * bestGridOri.Depth));

                var extraHybrid = RunEPPacker(NX, NY, NZ, orientations, gridHybrid, seeds, dx);
                hybridInitial.AddRange(extraHybrid);
                hybridPacking = hybridInitial;
            }

            if (hybridPacking != null && hybridPacking.Count > pureEPPacking.Count)
            {
                finalPacking = hybridPacking;
            }
            else
            {
                finalPacking = pureEPPacking;
            }
        }
        else
        {
            BitGrid3D gridHybrid = new BitGrid3D(NX, NY, NZ);
            finalPacking = new List<PlacedItem>();
            for (int i = 0; i < gx; i++)
            {
                for (int j = 0; j < gy; j++)
                {
                    for (int k = 0; k < gz; k++)
                    {
                        int px = i * bestGridOri.Width;
                        int py = j * bestGridOri.Height;
                        int pz = k * bestGridOri.Depth;

                        for (int vIdx = 0; vIdx < bestGridOri.Voxels.Count; vIdx++)
                        {
                            var v = bestGridOri.Voxels[vIdx];
                            gridHybrid.Set(px + v.x, py + v.y, pz + v.z, true);
                        }

                        Vector3 translation = new Vector3(
                            px * dx - bestGridOri.MinRotatedPoint.X,
                            py * dx - bestGridOri.MinRotatedPoint.Y,
                            pz * dx - bestGridOri.MinRotatedPoint.Z
                        );

                        finalPacking.Add(new PlacedItem
                        {
                            OrientationIndex = bestGridOri.Index,
                            Rotation = bestGridOri.Rotation,
                            Translation = translation,
                            Px = px,
                            Py = py,
                            Pz = pz
                        });
                    }
                }
            }

            var seeds = new List<(int x, int y, int z)>();
            if (gx * bestGridOri.Width < NX) seeds.Add((gx * bestGridOri.Width, 0, 0));
            if (gy * bestGridOri.Height < NY) seeds.Add((0, gy * bestGridOri.Height, 0));
            if (gz * bestGridOri.Depth < NZ) seeds.Add((0, 0, gz * bestGridOri.Depth));

            var extraHybrid = RunEPPacker(NX, NY, NZ, orientations, gridHybrid, seeds, dx);
            finalPacking.AddRange(extraHybrid);
        }

        Console.WriteLine(finalPacking.Count);
        foreach (var item in finalPacking)
        {
            Console.WriteLine($"{item.Translation.X.ToString("F6", CultureInfo.InvariantCulture)} {item.Translation.Y.ToString("F6", CultureInfo.InvariantCulture)} {item.Translation.Z.ToString("F6", CultureInfo.InvariantCulture)} {item.Rotation.W.ToString("F6", CultureInfo.InvariantCulture)} {item.Rotation.X.ToString("F6", CultureInfo.InvariantCulture)} {item.Rotation.Y.ToString("F6", CultureInfo.InvariantCulture)} {item.Rotation.Z.ToString("F6", CultureInfo.InvariantCulture)}");
        }
    }

    public static int Clamp(int value, int min, int max)
    {
        return (value < min) ? min : (value > max ? max : value);
    }

    public static bool RayTriangleIntersect(Vector3 rayOrigin, Vector3 rayDir, Vector3 v0, Vector3 v1, Vector3 v2, out float t)
    {
        t = 0;
        Vector3 edge1 = v1 - v0;
        Vector3 edge2 = v2 - v0;
        Vector3 h = Vector3.Cross(rayDir, edge2);
        float a = Vector3.Dot(edge1, h);
        if (a > -1e-6f && a < 1e-6f)
            return false;

        float f = 1.0f / a;
        Vector3 s = rayOrigin - v0;
        float u = f * Vector3.Dot(s, h);
        if (u < 0.0f || u > 1.0f)
            return false;

        Vector3 q = Vector3.Cross(s, edge1);
        float v = f * Vector3.Dot(rayDir, q);
        if (v < 0.0f || u + v > 1.0f)
            return false;

        t = f * Vector3.Dot(edge2, q);
        return t > 1e-6f;
    }

    public static List<(int x, int y, int z)> Voxelize(Vector3[] vertices, int[][] faces, float dx)
    {
        var voxelSet = new HashSet<(int x, int y, int z)>();

        float minX = float.MaxValue, minY = float.MaxValue, minZ = float.MaxValue;
        float maxX = float.MinValue, maxY = float.MinValue, maxZ = float.MinValue;
        foreach (var v in vertices)
        {
            minX = Math.Min(minX, v.X);
            minY = Math.Min(minY, v.Y);
            minZ = Math.Min(minZ, v.Z);
            maxX = Math.Max(maxX, v.X);
            maxY = Math.Max(maxY, v.Y);
            maxZ = Math.Max(maxZ, v.Z);
        }

        int nx = (int)Math.Ceiling((maxX - minX) / dx);
        int ny = (int)Math.Ceiling((maxY - minY) / dx);
        int nz = (int)Math.Ceiling((maxZ - minZ) / dx);

        nx = Math.Max(nx, 1);
        ny = Math.Max(ny, 1);
        nz = Math.Max(nz, 1);

        Vector3[] rayDirs = new Vector3[]
        {
            Vector3.Normalize(new Vector3(1.0f, 0.0123f, 0.00789f)),
            Vector3.Normalize(new Vector3(0.0115f, 1.0f, 0.00912f)),
            Vector3.Normalize(new Vector3(0.00831f, 0.0134f, 1.0f))
        };

        for (int ix = 0; ix < nx; ix++)
        {
            for (int iy = 0; iy < ny; iy++)
            {
                for (int iz = 0; iz < nz; iz++)
                {
                    Vector3 center = new Vector3(
                        minX + (ix + 0.5f) * dx,
                        minY + (iy + 0.5f) * dx,
                        minZ + (iz + 0.5f) * dx
                    );

                    int oddVotes = 0;
                    for (int r = 0; r < 3; r++)
                    {
                        Vector3 dir = rayDirs[r];
                        int intersectionCount = 0;
                        foreach (var face in faces)
                        {
                            Vector3 v0 = vertices[face[0]];
                            Vector3 v1 = vertices[face[1]];
                            Vector3 v2 = vertices[face[2]];

                            if (RayTriangleIntersect(center, dir, v0, v1, v2, out float t))
                            {
                                intersectionCount++;
                            }
                        }
                        if (intersectionCount % 2 == 1)
                        {
                            oddVotes++;
                        }
                    }

                    if (oddVotes >= 2)
                    {
                        voxelSet.Add((ix, iy, iz));
                    }
                }
            }
        }

        foreach (var face in faces)
        {
            Vector3 v0 = vertices[face[0]];
            Vector3 v1 = vertices[face[1]];
            Vector3 v2 = vertices[face[2]];

            float l1 = Vector3.Distance(v0, v1);
            float l2 = Vector3.Distance(v0, v2);

            int stepsU = (int)Math.Ceiling(l1 / (dx * 0.5f));
            int stepsV = (int)Math.Ceiling(l2 / (dx * 0.5f));

            stepsU = Math.Max(stepsU, 1);
            stepsV = Math.Max(stepsV, 1);

            for (int iu = 0; iu <= stepsU; iu++)
            {
                float u = (float)iu / stepsU;
                for (int jv = 0; jv <= stepsV - iu; jv++)
                {
                    float v = (float)jv / stepsV;
                    Vector3 p = v0 + u * (v1 - v0) + v * (v2 - v0);

                    int ix = (int)Math.Floor((p.X - minX) / dx);
                    int iy = (int)Math.Floor((p.Y - minY) / dx);
                    int iz = (int)Math.Floor((p.Z - minZ) / dx);

                    ix = Clamp(ix, 0, nx - 1);
                    iy = Clamp(iy, 0, ny - 1);
                    iz = Clamp(iz, 0, nz - 1);

                    voxelSet.Add((ix, iy, iz));
                }
            }
        }

        return new List<(int x, int y, int z)>(voxelSet);
    }

    public static List<Quaternion> GenerateUniqueOrientations()
    {
        var orientations = new List<Quaternion>();
        var uniqueRotatedBases = new List<(Vector3, Vector3)>();

        float[] angles = { 0f, 90f, 180f, 270f };
        foreach (float ax in angles)
        {
            foreach (float ay in angles)
            {
                foreach (float az in angles)
                {
                    float rx = ax * (float)(Math.PI / 180.0);
                    float ry = ay * (float)(Math.PI / 180.0);
                    float rz = az * (float)(Math.PI / 180.0);

                    Quaternion qx = Quaternion.CreateFromAxisAngle(Vector3.UnitX, rx);
                    Quaternion qy = Quaternion.CreateFromAxisAngle(Vector3.UnitY, ry);
                    Quaternion qz = Quaternion.CreateFromAxisAngle(Vector3.UnitZ, rz);

                    Quaternion q = qz * qy * qx;
                    q = Quaternion.Normalize(q);

                    Vector3 rotX = Vector3.Transform(Vector3.UnitX, q);
                    Vector3 rotY = Vector3.Transform(Vector3.UnitY, q);

                    Vector3 rxKey = new Vector3((float)Math.Round(rotX.X), (float)Math.Round(rotX.Y), (float)Math.Round(rotX.Z));
                    Vector3 ryKey = new Vector3((float)Math.Round(rotY.X), (float)Math.Round(rotY.Y), (float)Math.Round(rotY.Z));

                    bool exists = false;
                    foreach (var basis in uniqueRotatedBases)
                    {
                        if (Vector3.Distance(basis.Item1, rxKey) < 0.1f && Vector3.Distance(basis.Item2, ryKey) < 0.1f)
                        {
                            exists = true;
                            break;
                        }
                    }

                    if (!exists)
                    {
                        uniqueRotatedBases.Add((rxKey, ryKey));
                        orientations.Add(q);
                    }
                }
            }
        }
        return orientations;
    }

    public static List<OrientationInfo> PrecomputeOrientations(Vector3[] originalVertices, int[][] faces, float dx, List<Quaternion> orientationsList)
    {
        var list = new List<OrientationInfo>();

        for (int i = 0; i < orientationsList.Count; i++)
        {
            var q = orientationsList[i];

            Vector3[] rotated = new Vector3[originalVertices.Length];
            float minX = float.MaxValue, minY = float.MaxValue, minZ = float.MaxValue;
            for (int j = 0; j < originalVertices.Length; j++)
            {
                rotated[j] = Vector3.Transform(originalVertices[j], q);
                minX = Math.Min(minX, rotated[j].X);
                minY = Math.Min(minY, rotated[j].Y);
                minZ = Math.Min(minZ, rotated[j].Z);
            }

            Vector3 minRotated = new Vector3(minX, minY, minZ);

            Vector3[] shifted = new Vector3[originalVertices.Length];
            for (int j = 0; j < originalVertices.Length; j++)
            {
                shifted[j] = rotated[j] - minRotated;
            }

            var voxels = Voxelize(shifted, faces, dx);

            int wb = 0, hb = 0, db = 0;
            foreach (var v in voxels)
            {
                wb = Math.Max(wb, v.x + 1);
                hb = Math.Max(hb, v.y + 1);
                db = Math.Max(db, v.z + 1);
            }

            list.Add(new OrientationInfo
            {
                Index = i,
                Rotation = q,
                MinRotatedPoint = minRotated,
                Voxels = voxels,
                Width = wb,
                Height = hb,
                Depth = db
            });
        }

        return list;
    }

    public static List<PlacedItem> RunEPPacker(
        int NX, int NY, int NZ, 
        List<OrientationInfo> orientations, 
        BitGrid3D grid, 
        List<(int x, int y, int z)> initialEPs,
        float dx)
    {
        var placedItems = new List<PlacedItem>();
        var eps = new SortedSet<(int x, int y, int z)>(new ExtremePointComparer());
        foreach (var ep in initialEPs)
        {
            if (ep.x < NX && ep.y < NY && ep.z < NZ)
                eps.Add(ep);
        }

        if (eps.Count == 0)
            eps.Add((0, 0, 0));

        int maxIterations = 200000;
        int iterations = 0;

        while (eps.Count > 0 && iterations++ < maxIterations)
        {
            var ep = eps.Min;
            eps.Remove(ep);

            if (grid.Get(ep.x, ep.y, ep.z))
                continue;

            OrientationInfo bestOri = null;
            int bestScore = -1;

            foreach (var ori in orientations)
            {
                if (ep.x + ori.Width > NX || ep.y + ori.Height > NY || ep.z + ori.Depth > NZ)
                    continue;

                bool hasCollision = false;
                for (int i = 0; i < ori.Voxels.Count; i++)
                {
                    var v = ori.Voxels[i];
                    if (grid.Get(ep.x + v.x, ep.y + v.y, ep.z + v.z))
                    {
                        hasCollision = true;
                        break;
                    }
                }

                if (!hasCollision)
                {
                    int score = 0;
                    for (int i = 0; i < ori.Voxels.Count; i++)
                    {
                        var v = ori.Voxels[i];
                        int cx = ep.x + v.x;
                        int cy = ep.y + v.y;
                        int cz = ep.z + v.z;

                        if (cx - 1 < 0 || grid.Get(cx - 1, cy, cz)) score++;
                        if (cx + 1 >= NX || grid.Get(cx + 1, cy, cz)) score++;
                        if (cy - 1 < 0 || grid.Get(cx, cy - 1, cz)) score++;
                        if (cy + 1 >= NY || grid.Get(cx, cy + 1, cz)) score++;
                        if (cz - 1 < 0 || grid.Get(cx, cy, cz - 1)) score++;
                        if (cz + 1 >= NZ || grid.Get(cx, cy, cz + 1)) score++;
                    }

                    if (score > bestScore)
                    {
                        bestScore = score;
                        bestOri = ori;
                    }
                }
            }

            if (bestOri != null)
            {
                int px = ep.x;
                int py = ep.y;
                int pz = ep.z;

                for (int i = 0; i < bestOri.Voxels.Count; i++)
                {
                    var v = bestOri.Voxels[i];
                    grid.Set(px + v.x, py + v.y, pz + v.z, true);
                }

                Vector3 translation = new Vector3(
                    px * dx - bestOri.MinRotatedPoint.X,
                    py * dx - bestOri.MinRotatedPoint.Y,
                    pz * dx - bestOri.MinRotatedPoint.Z
                );

                placedItems.Add(new PlacedItem
                {
                    OrientationIndex = bestOri.Index,
                    Rotation = bestOri.Rotation,
                    Translation = translation,
                    Px = px,
                    Py = py,
                    Pz = pz
                });

                var newEp1 = (x: px + bestOri.Width, y: py, z: pz);
                var newEp2 = (x: px, y: py + bestOri.Height, z: pz);
                var newEp3 = (x: px, y: py, z: pz + bestOri.Depth);

                if (newEp1.x < NX && !grid.Get(newEp1.x, newEp1.y, newEp1.z)) eps.Add(newEp1);
                if (newEp2.y < NY && !grid.Get(newEp2.x, newEp2.y, newEp2.z)) eps.Add(newEp2);
                if (newEp3.z < NZ && !grid.Get(newEp3.x, newEp3.y, newEp3.z)) eps.Add(newEp3);

                var toRemove = new List<(int x, int y, int z)>();
                foreach (var existingEp in eps)
                {
                    if (existingEp.x >= px && existingEp.x < px + bestOri.Width &&
                        existingEp.y >= py && existingEp.y < py + bestOri.Height &&
                        existingEp.z >= pz && existingEp.z < pz + bestOri.Depth)
                    {
                        toRemove.Add(existingEp);
                    }
                }
                foreach (var r in toRemove)
                {
                    eps.Remove(r);
                }
            }
        }

        return placedItems;
    }

    private static void ParseInput(out Vector3[] vertices, out int[][] faces, out float W, out float H, out float D)
    {
        var tokens = new List<string>();
        string line;
        while ((line = Console.ReadLine()) != null)
        {
            var parts = line.Split((char[])null, StringSplitOptions.RemoveEmptyEntries);
            foreach (var p in parts)
            {
                tokens.Add(p);
            }
        }

        if (tokens.Count == 0)
        {
            vertices = new Vector3[0];
            faces = new int[0][];
            W = H = D = 0;
            return;
        }

        int ptr = 0;
        int V = int.Parse(tokens[ptr++]);
        int F = int.Parse(tokens[ptr++]);

        vertices = new Vector3[V];
        for (int i = 0; i < V; i++)
        {
            float x = float.Parse(tokens[ptr++], CultureInfo.InvariantCulture);
            float y = float.Parse(tokens[ptr++], CultureInfo.InvariantCulture);
            float z = float.Parse(tokens[ptr++], CultureInfo.InvariantCulture);
            vertices[i] = new Vector3(x, y, z);
        }

        faces = new int[F][];
        int minIndex = int.MaxValue;
        int maxIndex = int.MinValue;
        for (int i = 0; i < F; i++)
        {
            int a = int.Parse(tokens[ptr++]);
            int b = int.Parse(tokens[ptr++]);
            int c = int.Parse(tokens[ptr++]);
            faces[i] = new int[] { a, b, c };
            minIndex = Math.Min(minIndex, Math.Min(a, Math.Min(b, c)));
            maxIndex = Math.Max(maxIndex, Math.Max(a, Math.Max(b, c)));
        }

        if (maxIndex == V && minIndex == 1)
        {
            for (int i = 0; i < F; i++)
            {
                faces[i][0]--;
                faces[i][1]--;
                faces[i][2]--;
            }
        }

        W = float.Parse(tokens[ptr++], CultureInfo.InvariantCulture);
        H = float.Parse(tokens[ptr++], CultureInfo.InvariantCulture);
        D = float.Parse(tokens[ptr++], CultureInfo.InvariantCulture);
    }
