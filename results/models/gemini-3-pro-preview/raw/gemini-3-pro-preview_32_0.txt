#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>
#include <numeric>
#include <chrono>
#include <iomanip>
#include <cmath>

const long long INF = 1e18;

struct Edge {
    int u, v;
    long long weight;
};

struct OriginalEdge {
    int to;
    long long weight;
};

struct DSU {
    std::vector<int> parent;
    DSU(int n) {
        parent.resize(n);
        for (int i = 0; i < n; ++i) parent[i] = i;
    }
    int find(int i) {
        if (parent[i] == i)
            return i;
        return parent[i] = find(parent[i]);
    }
    bool unite(int i, int j) {
        int root_i = find(i);
        int root_j = find(j);
        if (root_i != root_j) {
            parent[root_i] = root_j;
            return true;
        }
        return false;
    }
};

// Prunes a set of edges to remove non-terminal leaves
std::vector<Edge> prune_tree(const std::vector<Edge>& edges, const std::vector<bool>& is_terminal) {
    if (edges.empty()) return {};
    
    std::vector<int> vertices;
    for (const auto& e : edges) {
        vertices.push_back(e.u);
        vertices.push_back(e.v);
    }
    std::sort(vertices.begin(), vertices.end());
    vertices.erase(std::unique(vertices.begin(), vertices.end()), vertices.end());
    
    int n_sub = vertices.size();
    auto get_local_id = [&](int x) {
        return std::lower_bound(vertices.begin(), vertices.end(), x) - vertices.begin();
    };
    
    std::vector<int> deg(n_sub, 0);
    std::vector<std::vector<std::pair<int, long long>>> adj(n_sub);
    
    for (const auto& e : edges) {
        int lu = get_local_id(e.u);
        int lv = get_local_id(e.v);
        adj[lu].push_back({lv, e.weight});
        adj[lv].push_back({lu, e.weight});
        deg[lu]++;
        deg[lv]++;
    }
    
    std::vector<int> q;
    std::vector<bool> in_q(n_sub, false);
    for (int i = 0; i < n_sub; ++i) {
        int x = vertices[i];
        if (!is_terminal[x] && deg[i] <= 1) {
            q.push_back(i);
            in_q[i] = true;
        }
    }
    
    int head = 0;
    while (head < (int)q.size()) {
        int u = q[head++];
        for (auto& edge : adj[u]) {
            int v = edge.first;
            if (deg[v] > 0) {
                deg[v]--;
                int orig_v = vertices[v];
                if (!is_terminal[orig_v] && deg[v] <= 1 && !in_q[v]) {
                    q.push_back(v);
                    in_q[v] = true;
                }
            }
        }
        deg[u] = 0;
    }
    
    std::vector<Edge> pruned;
    for (const auto& e : edges) {
        int lu = get_local_id(e.u);
        int lv = get_local_id(e.v);
        if (deg[lu] > 0 && deg[lv] > 0) {
            pruned.push_back(e);
        }
    }
    return pruned;
}

std::vector<Edge> induced_mst_and_prune(std::vector<Edge> edges, const std::vector<bool>& is_terminal) {
    if (edges.empty()) return {};
    
    std::vector<int> vertices;
    for (const auto& e : edges) {
        vertices.push_back(e.u);
        vertices.push_back(e.v);
    }
    std::sort(vertices.begin(), vertices.end());
    vertices.erase(std::unique(vertices.begin(), vertices.end()), vertices.end());
    
    int n_sub = vertices.size();
    auto get_local_id = [&](int x) {
        return std::lower_bound(vertices.begin(), vertices.end(), x) - vertices.begin();
    };
    
    std::sort(edges.begin(), edges.end(), [](const Edge& a, const Edge& b) {
        return a.weight < b.weight;
    });
    
    DSU dsu(n_sub);
    std::vector<Edge> mst_edges;
    for (const auto& e : edges) {
        int lu = get_local_id(e.u);
        int lv = get_local_id(e.v);
        if (dsu.unite(lu, lv)) {
            mst_edges.push_back(e);
        }
    }
    
    return prune_tree(mst_edges, is_terminal);
}

std::vector<Edge> run_sph(int V, int K, const std::vector<std::vector<OriginalEdge>>& adj, 
                         const std::vector<int>& terminals, const std::vector<bool>& is_terminal, 
                         int start_terminal) {
    std::vector<bool> in_tree(V, false);
    std::vector<long long> dist(V, INF);
    std::vector<int> parent_node(V, -1);
    std::vector<long long> parent_edge_weight(V, 0);
    
    using PQ_Element = std::pair<long long, int>;
    std::priority_queue<PQ_Element, std::vector<PQ_Element>, std::greater<PQ_Element>> pq;
    
    int terminals_connected = 0;
    
    in_tree[start_terminal] = true;
    dist[start_terminal] = 0;
    pq.push({0, start_terminal});
    terminals_connected++;
    
    std::vector<Edge> tree_edges;
    
    while (terminals_connected < K) {
        int next_terminal = -1;
        while (!pq.empty()) {
            auto [d, u] = pq.top();
            pq.pop();
            
            if (d > dist[u]) continue;
            
            if (is_terminal[u] && !in_tree[u]) {
                next_terminal = u;
                break;
            }
            
            for (const auto& edge : adj[u]) {
                int v = edge.to;
                long long w = edge.weight;
                if (dist[u] + w < dist[v]) {
                    dist[v] = dist[u] + w;
                    parent_node[v] = u;
                    parent_edge_weight[v] = w;
                    pq.push({dist[v], v});
                }
            }
        }
        
        if (next_terminal == -1) {
            break; // Graph is disconnected
        }
        
        int curr = next_terminal;
        std::vector<int> path;
        while (!in_tree[curr]) {
            path.push_back(curr);
            tree_edges.push_back({curr, parent_node[curr], parent_edge_weight[curr]});
            curr = parent_node[curr];
        }
        
        for (int u : path) {
            in_tree[u] = true;
            dist[u] = 0;
            pq.push({0, u});
            if (is_terminal[u]) {
                terminals_connected++;
            }
        }
    }
    
    if (terminals_connected < K) {
        return {}; // Return empty if we cannot connect all terminals
    }
    return tree_edges;
}

struct DW_Solver {
    int V, K;
    std::vector<int> terminals;
    std::vector<std::vector<OriginalEdge>> adj;
    std::vector<bool> is_terminal;

    std::vector<std::vector<long long>> dp;
    std::vector<std::vector<char>> type;
    std::vector<std::vector<int>> split_sub;
    std::vector<std::vector<int>> edge_parent;

    DW_Solver(int V, int K, const std::vector<int>& terminals, 
              const std::vector<std::vector<OriginalEdge>>& adj, 
              const std::vector<bool>& is_terminal) 
        : V(V), K(K), terminals(terminals), adj(adj), is_terminal(is_terminal) {}

    void solve(std::vector<Edge>& out_edges, long long& out_weight) {
        int num_states = 1 << K;
        dp.assign(num_states, std::vector<long long>(V, INF));
        type.assign(num_states, std::vector<char>(V, 0));
        split_sub.assign(num_states, std::vector<int>(V, 0));
        edge_parent.assign(num_states, std::vector<int>(V, -1));

        for (int i = 0; i < K; ++i) {
            dp[1 << i][terminals[i]] = 0;
        }

        using PQ_Element = std::pair<long long, int>;

        for (int mask = 1; mask < num_states; ++mask) {
            if ((mask & (mask - 1)) != 0) {
                int lsb = mask & -mask;
                for (int submask = (mask - 1) & mask; submask > 0; submask = (submask - 1) & mask) {
                    if (submask & lsb) {
                        int sibling = mask ^ submask;
                        for (int u = 0; u < V; ++u) {
                            long long val = dp[submask][u] + dp[sibling][u];
                            if (val < dp[mask][u]) {
                                dp[mask][u] = val;
                                type[mask][u] = 1; // SPLIT
                                split_sub[mask][u] = submask;
                            }
                        }
                    }
                }
            }

            std::priority_queue<PQ_Element, std::vector<PQ_Element>, std::greater<PQ_Element>> pq;
            for (int u = 0; u < V; ++u) {
                if (dp[mask][u] < INF) {
                    pq.push({dp[mask][u], u});
                }
            }

            while (!pq.empty()) {
                auto [d, u] = pq.top();
                pq.pop();

                if (d > dp[mask][u]) continue;

                for (const auto& edge : adj[u]) {
                    int v = edge.to;
                    long long w = edge.weight;
                    if (dp[mask][u] + w < dp[mask][v]) {
                        dp[mask][v] = dp[mask][u] + w;
                        type[mask][v] = 2; // EDGE
                        edge_parent[mask][v] = u;
                        pq.push({dp[mask][v], v});
                    }
                }
            }
        }

        int root = -1;
        long long min_w = INF;
        for (int u = 0; u < V; ++u) {
            if (dp[num_states - 1][u] < min_w) {
                min_w = dp[num_states - 1][u];
                root = u;
            }
        }

        if (min_w >= INF || root == -1) {
            out_weight = INF;
            return;
        }

        out_weight = min_w;
        std::vector<std::pair<int, int>> recon_edges;
        reconstruct(num_states - 1, root, recon_edges);

        for (auto p : recon_edges) {
            int u = p.first;
            int v = p.second;
            long long w = INF;
            for (const auto& edge : adj[u]) {
                if (edge.to == v) {
                    w = std::min(w, edge.weight);
                }
            }
            if (w < INF) {
                out_edges.push_back({u, v, w});
            }
        }
    }

    void reconstruct(int mask, int u, std::vector<std::pair<int, int>>& recon_edges) {
        if ((mask & (mask - 1)) == 0) {
            int t_idx = -1;
            for (int i = 0; i < K; ++i) {
                if (mask & (1 << i)) {
                    t_idx = i;
                    break;
                }
            }
            if (u == terminals[t_idx]) {
                return;
            }
        }

        if (type[mask][u] == 1) {
            int sub = split_sub[mask][u];
            reconstruct(sub, u, recon_edges);
            reconstruct(mask ^ sub, u, recon_edges);
        } else if (type[mask][u] == 2) {
            int p = edge_parent[mask][u];
            recon_edges.push_back({u, p});
            reconstruct(mask, p, recon_edges);
        }
    }
};

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    int V, E, K;
    if (!(std::cin >> V >> E >> K)) return 0;

    struct InputEdge {
        int u, v;
        long long weight;
    };

    std::vector<InputEdge> input_edges;
    input_edges.reserve(E);
    for (int i = 0; i < E; ++i) {
        int u, v;
        long long w;
        if (std::cin >> u >> v >> w) {
            if (u == v) continue; // Skip self loops
            if (u > v) std::swap(u, v);
            input_edges.push_back({u, v, w});
        }
    }

    // Filter duplicate edges, keeping the one with minimum weight
    std::sort(input_edges.begin(), input_edges.end(), [](const InputEdge& a, const InputEdge& b) {
        if (a.u != b.u) return a.u < b.u;
        if (a.v != b.v) return a.v < b.v;
        return a.weight < b.weight;
    });

    std::vector<InputEdge> filtered_edges;
    filtered_edges.reserve(input_edges.size());
    for (const auto& e : input_edges) {
        if (filtered_edges.empty() || filtered_edges.back().u != e.u || filtered_edges.back().v != e.v) {
            filtered_edges.push_back(e);
        }
    }

    std::vector<int> terminals(K);
    std::vector<bool> is_terminal(V, false);
    for (int i = 0; i < K; ++i) {
        std::cin >> terminals[i];
        is_terminal[terminals[i]] = true;
    }

    // Handle K = 1 edge case
    if (K == 1) {
        std::cout << "0 0\n";
        return 0;
    }

    // Build adjacency list from filtered edges
    std::vector<std::vector<OriginalEdge>> adj(V);
    for (const auto& e : filtered_edges) {
        adj[e.u].push_back({e.v, e.weight});
        adj[e.v].push_back({e.u, e.weight});
    }

    // Determine whether to use Dreyfus-Wagner exact solver
    // Threshold: 3^K * V <= 1.5e8
    bool use_dw = false;
    if (K <= 14) {
        double ops = std::pow(3.0, K) * V;
        if (ops <= 1.5e8) {
            use_dw = true;
        }
    }

    std::vector<Edge> final_edges;
    long long final_weight = INF;

    if (use_dw) {
        DW_Solver dw(V, K, terminals, adj, is_terminal);
        std::vector<Edge> dw_edges;
        long long dw_weight = INF;
        dw.solve(dw_edges, dw_weight);
        if (dw_weight < INF) {
            final_edges = induced_mst_and_prune(dw_edges, is_terminal);
            final_weight = 0;
            for (const auto& e : final_edges) final_weight += e.weight;
        }
    }

    // If exact solver was not run or failed (disconnected graph), run SPH + Union-MST
    if (final_weight >= INF) {
        // Run SPH starting from multiple terminals
        // Scale number of runs dynamically based on graph size to ensure fast execution
        int max_runs = 50;
        if (filtered_edges.size() > 200000) max_runs = 20;
        if (filtered_edges.size() > 1000000) max_runs = 5;
        
        int runs = std::min(K, max_runs);
        
        struct RunResult {
            std::vector<Edge> edges;
            long long weight;
        };
        std::vector<RunResult> results;
        
        for (int i = 0; i < runs; ++i) {
            int start_term = terminals[i];
            auto sph_edges = run_sph(V, K, adj, terminals, is_terminal, start_term);
            if (sph_edges.empty()) continue; // Disconnected component
            
            auto pruned = induced_mst_and_prune(sph_edges, is_terminal);
            long long w = 0;
            for (const auto& e : pruned) w += e.weight;
            results.push_back({pruned, w});
        }
        
        if (!results.empty()) {
            // Find the best individual run
            int best_idx = 0;
            for (int i = 1; i < (int)results.size(); ++i) {
                if (results[i].weight < results[best_idx].weight) {
                    best_idx = i;
                }
            }
            final_weight = results[best_idx].weight;
            final_edges = results[best_idx].edges;
            
            // Try Union-MST of the top runs to see if we can improve
            std::sort(results.begin(), results.end(), [](const RunResult& a, const RunResult& b) {
                return a.weight < b.weight;
            });
            
            int num_union = std::min((int)results.size(), 10);
            std::vector<Edge> union_edges;
            for (int i = 0; i < num_union; ++i) {
                for (const auto& e : results[i].edges) {
                    union_edges.push_back(e);
                }
            }
            
            // Unique the union edges to speed up Kruskal's
            for (auto& e : union_edges) {
                if (e.u > e.v) std::swap(e.u, e.v);
            }
            std::sort(union_edges.begin(), union_edges.end(), [](const Edge& a, const Edge& b) {
                if (a.u != b.u) return a.u < b.u;
                if (a.v != b.v) return a.v < b.v;
                return a.weight < b.weight;
            });
            union_edges.erase(std::unique(union_edges.begin(), union_edges.end(), [](const Edge& a, const Edge& b) {
                return a.u == b.u && a.v == b.v;
            }), union_edges.end());
            
            auto union_mst_edges = induced_mst_and_prune(union_edges, is_terminal);
            long long union_weight = 0;
            for (const auto& e : union_mst_edges) union_weight += e.weight;
            
            if (union_weight < final_weight && union_weight > 0) {
                final_weight = union_weight;
                final_edges = union_mst_edges;
            }
        }
    }

    if (final_weight >= INF) {
        // Could not connect terminals
        std::cout << "0 0\n";
    } else {
        std::cout << final_weight << " " << final_edges.size() << "\n";
        for (const auto& e : final_edges) {
            std::cout << e.u << " " << e.v << "\n";
        }
    }

    return 0;
}
