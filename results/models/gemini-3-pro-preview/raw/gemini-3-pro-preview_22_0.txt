using System;
using System.Collections.Generic;
using System.Text;

namespace TetrisAI
{
    public struct Point
    {
        public int X;
        public int Y;
        public Point(int x, int y) { X = x; Y = y; }
    }

    public class Program
    {
        // 4 rotations for all 7 standard Tetris pieces
        public static readonly Dictionary<char, Point[][]> Pieces = new Dictionary<char, Point[][]>
        {
            { 'I', new Point[][] {
                new Point[] { new Point(0,0), new Point(1,0), new Point(2,0), new Point(3,0) },
                new Point[] { new Point(0,0), new Point(0,1), new Point(0,2), new Point(0,3) },
                new Point[] { new Point(0,0), new Point(1,0), new Point(2,0), new Point(3,0) },
                new Point[] { new Point(0,0), new Point(0,1), new Point(0,2), new Point(0,3) }
            }},
            { 'O', new Point[][] {
                new Point[] { new Point(0,0), new Point(1,0), new Point(0,1), new Point(1,1) },
                new Point[] { new Point(0,0), new Point(1,0), new Point(0,1), new Point(1,1) },
                new Point[] { new Point(0,0), new Point(1,0), new Point(0,1), new Point(1,1) },
                new Point[] { new Point(0,0), new Point(1,0), new Point(0,1), new Point(1,1) }
            }},
            { 'T', new Point[][] {
                new Point[] { new Point(0,0), new Point(1,0), new Point(2,0), new Point(1,1) },
                new Point[] { new Point(0,0), new Point(0,1), new Point(0,2), new Point(1,1) },
                new Point[] { new Point(0,1), new Point(1,0), new Point(1,1), new Point(2,1) },
                new Point[] { new Point(0,1), new Point(1,0), new Point(1,1), new Point(1,2) }
            }},
            { 'S', new Point[][] {
                new Point[] { new Point(0,0), new Point(1,0), new Point(1,1), new Point(2,1) },
                new Point[] { new Point(0,1), new Point(0,2), new Point(1,0), new Point(1,1) },
                new Point[] { new Point(0,0), new Point(1,0), new Point(1,1), new Point(2,1) },
                new Point[] { new Point(0,1), new Point(0,2), new Point(1,0), new Point(1,1) }
            }},
            { 'Z', new Point[][] {
                new Point[] { new Point(0,1), new Point(1,1), new Point(1,0), new Point(2,0) },
                new Point[] { new Point(0,0), new Point(0,1), new Point(1,1), new Point(1,2) },
                new Point[] { new Point(0,1), new Point(1,1), new Point(1,0), new Point(2,0) },
                new Point[] { new Point(0,0), new Point(0,1), new Point(1,1), new Point(1,2) }
            }},
            { 'J', new Point[][] {
                new Point[] { new Point(0,0), new Point(1,0), new Point(2,0), new Point(0,1) },
                new Point[] { new Point(0,0), new Point(0,1), new Point(0,2), new Point(1,2) },
                new Point[] { new Point(0,1), new Point(1,1), new Point(2,1), new Point(2,0) },
                new Point[] { new Point(0,0), new Point(1,0), new Point(1,1), new Point(1,2) }
            }},
            { 'L', new Point[][] {
                new Point[] { new Point(0,0), new Point(1,0), new Point(2,0), new Point(2,1) },
                new Point[] { new Point(0,0), new Point(0,1), new Point(0,2), new Point(1,0) },
                new Point[] { new Point(0,0), new Point(0,1), new Point(1,1), new Point(2,1) },
                new Point[] { new Point(0,2), new Point(1,0), new Point(1,1), new Point(1,2) }
            }}
        };

        public static void Main(string[] args)
        {
            // Read initial board configuration
            string tokenWidth = ReadNextToken();
            if (string.IsNullOrEmpty(tokenWidth)) return;
            int width = int.Parse(tokenWidth);

            string tokenHeight = ReadNextToken();
            if (string.IsNullOrEmpty(tokenHeight)) return;
            int height = int.Parse(tokenHeight);

            string tokenNumPieces = ReadNextToken();
            if (string.IsNullOrEmpty(tokenNumPieces)) return;
            int numPieces = int.Parse(tokenNumPieces);

            // Internally track the board representation
            bool[,] grid = new bool[width, height];

            for (int p = 0; p < numPieces; p++)
            {
                string pieceToken = ReadNextToken();
                if (string.IsNullOrEmpty(pieceToken)) break;

                char pieceType = char.ToUpper(pieceToken[0]);
                if (!Pieces.ContainsKey(pieceType))
                {
                    // Fallback for unknown piece types
                    Console.WriteLine("0 0");
                    Console.Out.Flush();
                    string res = ReadNextToken();
                    if (string.IsNullOrEmpty(res) || res == "gameover") break;
                    continue;
                }

                double bestScore = double.NegativeInfinity;
                int bestRot = 0;
                int bestCol = 0;
                bool foundMove = false;

                Point[][] rotations = Pieces[pieceType];
                for (int rot = 0; rot < 4; rot++)
                {
                    Point[] blocks = rotations[rot];
                    int maxDx = 0;
                    for (int i = 0; i < 4; i++)
                        if (blocks[i].X > maxDx) maxDx = blocks[i].X;

                    // Try all possible valid horizontal columns
                    for (int col = 0; col <= width - 1 - maxDx; col++)
                    {
                        int yBase = GetLandingHeight(grid, width, height, blocks, col);
                        if (yBase < 0) continue;

                        // Check if placement overflows the board (game over)
                        bool gameOverPlacement = false;
                        for (int i = 0; i < 4; i++)
                        {
                            if (yBase + blocks[i].Y >= height)
                            {
                                gameOverPlacement = true;
                                break;
                            }
                        }

                        if (gameOverPlacement)
                        {
                            // In case of no other moves, we still consider it with low preference
                            double score = -10000000.0 + yBase;
                            if (score > bestScore)
                            {
                                bestScore = score;
                                bestRot = rot;
                                bestCol = col;
                                foundMove = true;
                            }
                            continue;
                        }

                        // Simulate placement on a copy of the board
                        bool[,] nextGrid = (bool[,])grid.Clone();
                        for (int i = 0; i < 4; i++)
                        {
                            nextGrid[col + blocks[i].X, yBase + blocks[i].Y] = true;
                        }

                        // Count cleared lines and piece blocks eliminated
                        int linesCleared = 0;
                        int blocksCleared = 0;
                        for (int y = 0; y < height; y++)
                        {
                            bool full = true;
                            for (int x = 0; x < width; x++)
                            {
                                if (!nextGrid[x, y]) { full = false; break; }
                            }
                            if (full)
                            {
                                linesCleared++;
                                for (int i = 0; i < 4; i++)
                                {
                                    if (yBase + blocks[i].Y == y)
                                        blocksCleared++;
                                }
                            }
                        }

                        // Shift lines down on the simulated board
                        if (linesCleared > 0)
                        {
                            bool[,] clearedGrid = new bool[width, height];
                            int targetY = 0;
                            for (int y = 0; y < height; y++)
                            {
                                bool full = true;
                                for (int x = 0; x < width; x++)
                                {
                                    if (!nextGrid[x, y]) { full = false; break; }
                                }
                                if (!full)
                                {
                                    for (int x = 0; x < width; x++)
                                    {
                                        clearedGrid[x, targetY] = nextGrid[x, y];
                                    }
                                    targetY++;
                                }
                            }
                            nextGrid = clearedGrid;
                        }

                        // Evaluate the simulated board state
                        double placementScore = EvaluateBoard(nextGrid, width, height, yBase, linesCleared, blocksCleared);
                        if (placementScore > bestScore)
                        {
                            bestScore = placementScore;
                            bestRot = rot;
                            bestCol = col;
                            foundMove = true;
                        }
                    }
                }

                if (!foundMove)
                {
                    bestRot = 0;
                    bestCol = 0;
                }

                // Output rotation and leftmost column
                Console.WriteLine($"{bestRot} {bestCol}");
                Console.Out.Flush();

                // Wait for the result from the game engine
                string resultToken = ReadNextToken();
                if (string.IsNullOrEmpty(resultToken) || resultToken == "gameover")
                {
                    break;
                }

                // Update our actual board representation
                Point[] chosenBlocks = rotations[bestRot];
                int finalYBase = GetLandingHeight(grid, width, height, chosenBlocks, bestCol);
                if (finalYBase >= 0)
                {
                    for (int i = 0; i < 4; i++)
                    {
                        int px = bestCol + chosenBlocks[i].X;
                        int py = finalYBase + chosenBlocks[i].Y;
                        if (px >= 0 && px < width && py >= 0 && py < height)
                        {
                            grid[px, py] = true;
                        }
                    }

                    // Apply line clears to our actual board
                    bool[,] newGrid = new bool[width, height];
                    int newY = 0;
                    for (int y = 0; y < height; y++)
                    {
                        bool full = true;
                        for (int x = 0; x < width; x++)
                        {
                            if (!grid[x, y]) { full = false; break; }
                        }
                        if (!full)
                        {
                            for (int x = 0; x < width; x++)
                            {
                                newGrid[x, newY] = grid[x, y];
                            }
                            newY++;
                        }
                    }
                    grid = newGrid;
                }
            }
        }

        // Calculate landing height for a piece dropped straight down in a column
        private static int GetLandingHeight(bool[,] grid, int width, int height, Point[] blocks, int col)
        {
            int maxDx = 0;
            for (int i = 0; i < 4; i++)
                if (blocks[i].X > maxDx) maxDx = blocks[i].X;

            int[] dyMin = new int[maxDx + 1];
            for (int i = 0; i <= maxDx; i++) dyMin[i] = int.MaxValue;
            for (int i = 0; i < 4; i++)
            {
                int dx = blocks[i].X;
                if (blocks[i].Y < dyMin[dx])
                    dyMin[dx] = blocks[i].Y;
            }

            int yBase = 0;
            for (int dx = 0; dx <= maxDx; dx++)
            {
                int boardCol = col + dx;
                if (boardCol < 0 || boardCol >= width) return -1;

                int topHeight = 0;
                for (int y = height - 1; y >= 0; y--)
                {
                    if (grid[boardCol, y])
                    {
                        topHeight = y + 1;
                        break;
                    }
                }

                int val = topHeight - dyMin[dx];
                if (val > yBase)
                    yBase = val;
            }

            return yBase;
        }

        // Heuristic function evaluates board features: height, holes, transitions, and wells.
        // It features adaptive strategies to switch behavior based on current stack height and board complexity.
        private static double EvaluateBoard(bool[,] grid, int width, int height, int landingHeight, int linesCleared, int blocksCleared)
        {
            int[] heights = new int[width];
            int maxH = 0;
            int aggH = 0;
            for (int x = 0; x < width; x++)
            {
                heights[x] = 0;
                for (int y = height - 1; y >= 0; y--)
                {
                    if (grid[x, y])
                    {
                        heights[x] = y + 1;
                        break;
                    }
                }
                aggH += heights[x];
                if (heights[x] > maxH) maxH = heights[x];
            }

            int holes = 0;
            for (int x = 0; x < width; x++)
            {
                bool foundOccupied = false;
                for (int y = height - 1; y >= 0; y--)
                {
                    if (grid[x, y]) foundOccupied = true;
                    else if (foundOccupied) holes++;
                }
            }

            int bumpiness = 0;
            for (int x = 0; x < width - 1; x++)
            {
                bumpiness += Math.Abs(heights[x] - heights[x + 1]);
            }

            int rowTrans = 0;
            for (int y = 0; y < height; y++)
            {
                bool prev = true;
                for (int x = 0; x < width; x++)
                {
                    bool curr = grid[x, y];
                    if (curr != prev) rowTrans++;
                    prev = curr;
                }
                if (!prev) rowTrans++;
            }

            int colTrans = 0;
            for (int x = 0; x < width; x++)
            {
                bool prev = true;
                for (int y = 0; y < height; y++)
                {
                    bool curr = grid[x, y];
                    if (curr != prev) colTrans++;
                    prev = curr;
                }
                if (prev) colTrans++;
            }

            int wellSum = 0;
            for (int x = 0; x < width; x++)
            {
                int leftHeight = (x == 0) ? height : heights[x - 1];
                int rightHeight = (x == width - 1) ? height : heights[x + 1];
                int depth = Math.Max(0, Math.Min(leftHeight, rightHeight) - heights[x]);
                wellSum += depth * (depth + 1) / 2;
            }

            double w_lh, w_eroded, w_rt, w_ct, w_holes, w_wells, w_bump, w_maxH, w_aggH;

            double maxPercent = (double)maxH / height;

            // Strategy Selector (Adaptive Logic)
            if (maxPercent > 0.8)
            {
                // Panic Mode: prioritize line clears and reducing stack height immediately to avoid top-out.
                w_lh = -1.0;
                w_eroded = 15.0;
                w_rt = -2.0;
                w_ct = -4.0;
                w_holes = -15.0;
                w_wells = -1.0;
                w_bump = -1.0;
                w_maxH = -10.0;
                w_aggH = -1.0;
            }
            else if (maxPercent > 0.4 || width < 8)
            {
                // Defensive Mode: focus on flatness, minimizing holes, and preserving clean layout.
                w_lh = -4.5;
                w_eroded = 3.4;
                w_rt = -3.2;
                w_ct = -9.3;
                w_holes = -12.0;
                w_wells = -3.4;
                w_bump = -1.0;
                w_maxH = -1.0;
                w_aggH = -0.5;
            }
            else
            {
                // Balanced / Strategic Mode: standard Pierre Dellacherie weights for long-term safe building.
                w_lh = -4.500158825082766;
                w_eroded = 3.4181268101392694;
                w_rt = -3.2178882868487753;
                w_ct = -9.348695305445199;
                w_holes = -7.899265427351652;
                w_wells = -3.385597252632798;
                w_bump = -0.5;
                w_maxH = 0.0;
                w_aggH = 0.0;
            }

            double score = w_lh * landingHeight
                         + w_eroded * (linesCleared * blocksCleared)
                         + w_rt * rowTrans
                         + w_ct * colTrans
                         + w_holes * holes
                         + w_wells * wellSum
                         + w_bump * bumpiness
                         + w_maxH * maxH
                         + w_aggH * aggH;

            return score;
        }

        // Robust token-based reader to parse whitespace-separated streams
        private static string ReadNextToken()
        {
            var sb = new StringBuilder();
            int ch;
            while ((ch = Console.Read()) != -1)
            {
                char c = (char)ch;
                if (!char.IsWhiteSpace(c))
                {
                    sb.Append(c);
                    break;
                }
            }
            if (ch == -1) return null;
            while ((ch = Console.Read()) != -1)
            {
                char c = (char)ch;
                if (char.IsWhiteSpace(c))
                {
                    break;
                }
                sb.Append(c);
            }
            return sb.ToString();
        }
    }
