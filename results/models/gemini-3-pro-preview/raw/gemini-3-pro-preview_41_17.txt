OpCapability Shader
%glsl = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %worldPos %normal %uv %tangent %color %outColor
OpExecutionMode %main OriginUpperLeft

OpName %main "main"
OpName %worldPos "worldPos"
OpName %normal "normal"
OpName %uv "uv"
OpName %tangent "tangent"
OpName %color "color"
OpName %outColor "outColor"
OpName %UBO "UBO"
OpMemberName %UBO 0 "model"
OpMemberName %UBO 1 "view"
OpMemberName %UBO 2 "projection"
OpMemberName %UBO 3 "lightPos"
OpMemberName %UBO 4 "cameraPos"
OpMemberName %UBO 5 "params"
OpName %ubo "ubo"

OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %uv Location 2
OpDecorate %tangent Location 3
OpDecorate %color Location 4
OpDecorate %outColor Location 0
OpMemberDecorate %UBO 0 ColMajor
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 0 MatrixStride 16
OpMemberDecorate %UBO 1 ColMajor
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 1 MatrixStride 16
OpMemberDecorate %UBO 2 ColMajor
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 2 MatrixStride 16
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %UBO Block
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0

%void = OpTypeVoid
%void_fn = OpTypeFunction %void
%float = OpTypeFloat 32
%vec2 = OpTypeVector %float 2
%vec3 = OpTypeVector %float 3
%vec4 = OpTypeVector %float 4
%mat4 = OpTypeMatrix %vec4 4
%UBO = OpTypeStruct %mat4 %mat4 %mat4 %vec4 %vec4 %vec4
%bool = OpTypeBool
%v3bool = OpTypeVector %bool 3

%_ptr_Input_vec2 = OpTypePointer Input %vec2
%_ptr_Input_vec3 = OpTypePointer Input %vec3
%_ptr_Output_vec4 = OpTypePointer Output %vec4
%_ptr_Uniform_UBO = OpTypePointer Uniform %UBO
%_ptr_Uniform_vec4 = OpTypePointer Uniform %vec4

%int = OpTypeInt 32 1
%int_3 = OpConstant %int 3

%worldPos = OpVariable %_ptr_Input_vec3 Input
%normal = OpVariable %_ptr_Input_vec3 Input
%uv = OpVariable %_ptr_Input_vec2 Input
%tangent = OpVariable %_ptr_Input_vec3 Input
%color = OpVariable %_ptr_Input_vec3 Input
%outColor = OpVariable %_ptr_Output_vec4 Output
%ubo = OpVariable %_ptr_Uniform_UBO Uniform

%float_0_0 = OpConstant %float 0.0
%float_0_5 = OpConstant %float 0.5
%float_1_0 = OpConstant %float 1.0
%float_2_0 = OpConstant %float 2.0
%float_8_0 = OpConstant %float 8.0
%float_16_0 = OpConstant %float 16.0
%float_0_05 = OpConstant %float 0.05
%float_0_1 = OpConstant %float 0.1
%float_0_4 = OpConstant %float 0.4
%float_0_6 = OpConstant %float 0.6
%float_0_7 = OpConstant %float 0.7
%float_0_2 = OpConstant %float 0.2

%vec2_8_16 = OpConstantComposite %vec2 %float_8_0 %float_16_0
%vec3_brick = OpConstantComposite %vec3 %float_0_7 %float_0_2 %float_0_1
%vec3_mortar = OpConstantComposite %vec3 %float_0_7 %float_0_7 %float_0_7

%main = OpFunction %void None %void_fn
%entry = OpLabel

; Load inputs
%uv_val = OpLoad %vec2 %uv
%worldPos_val = OpLoad %vec3 %worldPos
%normal_val = OpLoad %vec3 %normal

; Load lightPos from UBO
%lightPos_ptr = OpAccessChain %_ptr_Uniform_vec4 %ubo %int_3
%lightPos_val = OpLoad %vec4 %lightPos_ptr

; Calculate scaledUV = uv * vec2(8.0, 16.0)
%scaledUV = OpFMul %vec2 %uv_val %vec2_8_16

; Extract y component of scaledUV
%scaledUV_y = OpCompositeExtract %float %scaledUV 1

; Compute row = floor(scaledUV_y)
%row = OpExtInst %float %glsl Floor %scaledUV_y

; Compute oddRow = row - 2.0 * floor(row * 0.5)
%half_row = OpFMul %float %row %float_0_5
%floor_half_row = OpExtInst %float %glsl Floor %half_row
%twice_floor_half_row = OpFMul %float %floor_half_row %float_2_0
%oddRow = OpFSub %float %row %twice_floor_half_row

; Extract x component of scaledUV
%scaledUV_x = OpCompositeExtract %float %scaledUV 0

; Compute offset = oddRow * 0.5
%offset_x = OpFMul %float %oddRow %float_0_5
%scaledUV_x_new = OpFAdd %float %scaledUV_x %offset_x

; Reconstruct scaledUV_new = vec2(scaledUV_x_new, scaledUV_y)
%scaledUV_new = OpCompositeConstruct %vec2 %scaledUV_x_new %scaledUV_y

; Compute floor(scaledUV_new)
%floor_scaledUV = OpExtInst %vec2 %glsl Floor %scaledUV_new

; Compute localUV = scaledUV_new - floor(scaledUV_new)
%localUV = OpFSub %vec2 %scaledUV_new %floor_scaledUV

; Extract localUV components
%localUV_x = OpCompositeExtract %float %localUV 0
%localUV_y = OpCompositeExtract %float %localUV 1

; check if it is brick or mortar
%cond_x = OpFOrdGreaterThanEqual %bool %localUV_x %float_0_05
%cond_y = OpFOrdGreaterThanEqual %bool %localUV_y %float_0_1
%isBrick_bool = OpLogicalAnd %bool %cond_x %cond_y
%isBrick_vec3 = OpCompositeConstruct %v3bool %isBrick_bool %isBrick_bool %isBrick_bool
%diffuseColor = OpSelect %vec3 %isBrick_vec3 %vec3_brick %vec3_mortar

; Normalize normal
%N = OpExtInst %vec3 %glsl Normalize %normal_val

; Extract lightPos.xyz
%lightPos_xyz = OpVectorShuffle %vec3 %lightPos_val %lightPos_val 0 1 2

; L_unnormalized = lightPos.xyz - worldPos
%L_unnormalized = OpFSub %vec3 %lightPos_xyz %worldPos_val

; L = normalize(L_unnormalized)
%L = OpExtInst %vec3 %glsl Normalize %L_unnormalized

; dot_NL = dot(N, L)
%dot_NL = OpDot %float %N %L

; diffuse = max(dot_NL, 0.0)
%diffuse = OpExtInst %float %glsl FMax %dot_NL %float_0_0

; factor = 0.4 + 0.6 * diffuse
%scaled_diffuse = OpFMul %float %diffuse %float_0_6
%factor = OpFAdd %float %float_0_4 %scaled_diffuse

; shadedColor = diffuseColor * factor
%shadedColor = OpVectorTimesScalar %vec3 %diffuseColor %factor

; outColor = vec4(shadedColor, 1.0)
%final_color = OpCompositeConstruct %vec4 %shadedColor %float_1_0
OpStore %outColor %final_color

OpReturn
OpFunctionEnd
