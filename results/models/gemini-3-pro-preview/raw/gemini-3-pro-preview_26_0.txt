use std::io::{self, Write, BufReader, BufWriter};
use std::fs::File;

const TEMP_FILE_PATH: &str = "temp_points.bin";

struct SimpleRng {
    state: u64,
}

impl SimpleRng {
    fn new(seed: u64) -> Self {
        Self { state: if seed == 0 { 1 } else { seed } }
    }

    fn next_u32(&mut self) -> u32 {
        self.state ^= self.state << 13;
        self.state ^= self.state >> 7;
        self.state ^= self.state << 17;
        self.state as u32
    }
}

struct TempFileCleanup {
    path: &'static str,
    active: bool,
}

impl Drop for TempFileCleanup {
    fn drop(&mut self) {
        if self.active {
            let _ = std::fs::remove_file(self.path);
        }
    }
}

struct MetaCentroid {
    coords: [f32; 3],
    child_indices: Vec<usize>,
}

struct CentroidTree {
    meta_centroids: Vec<MetaCentroid>,
    centroids: Vec<f32>,
    use_hierarchy: bool,
}

impl CentroidTree {
    fn find_nearest(&self, x: f32, y: f32, z: f32) -> usize {
        if !self.use_hierarchy {
            let mut min_dist_sq = f32::INFINITY;
            let mut best_idx = 0;
            let n = self.centroids.len() / 3;
            for i in 0..n {
                let cx = self.centroids[i * 3];
                let cy = self.centroids[i * 3 + 1];
                let cz = self.centroids[i * 3 + 2];
                let dx = x - cx;
                let dy = y - cy;
                let dz = z - cz;
                let d = dx*dx + dy*dy + dz*dz;
                if d < min_dist_sq {
                    min_dist_sq = d;
                    best_idx = i;
                }
            }
            best_idx
        } else {
            let mut min_meta_dist_sq1 = f32::INFINITY;
            let mut best_meta_idx1 = 0;
            let mut min_meta_dist_sq2 = f32::INFINITY;
            let mut best_meta_idx2 = 0;
            
            for (i, mc) in self.meta_centroids.iter().enumerate() {
                let dx = x - mc.coords[0];
                let dy = y - mc.coords[1];
                let dz = z - mc.coords[2];
                let d = dx*dx + dy*dy + dz*dz;
                if d < min_meta_dist_sq1 {
                    min_meta_dist_sq2 = min_meta_dist_sq1;
                    best_meta_idx2 = best_meta_idx1;
                    min_meta_dist_sq1 = d;
                    best_meta_idx1 = i;
                } else if d < min_meta_dist_sq2 {
                    min_meta_dist_sq2 = d;
                    best_meta_idx2 = i;
                }
            }
            
            let mut min_dist_sq = f32::INFINITY;
            let mut best_idx = 0;
            
            let mc1 = &self.meta_centroids[best_meta_idx1];
            for &idx in &mc1.child_indices {
                let cx = self.centroids[idx * 3];
                let cy = self.centroids[idx * 3 + 1];
                let cz = self.centroids[idx * 3 + 2];
                let dx = x - cx;
                let dy = y - cy;
                let dz = z - cz;
                let d = dx*dx + dy*dy + dz*dz;
                if d < min_dist_sq {
                    min_dist_sq = d;
                    best_idx = idx;
                }
            }
            
            if !self.meta_centroids.is_empty() && best_meta_idx2 != best_meta_idx1 {
                let mc2 = &self.meta_centroids[best_meta_idx2];
                for &idx in &mc2.child_indices {
                    let cx = self.centroids[idx * 3];
                    let cy = self.centroids[idx * 3 + 1];
                    let cz = self.centroids[idx * 3 + 2];
                    let dx = x - cx;
                    let dy = y - cy;
                    let dz = z - cz;
                    let d = dx*dx + dy*dy + dz*dz;
                    if d < min_dist_sq {
                        min_dist_sq = d;
                        best_idx = idx;
                    }
                }
            }
            
            best_idx
        }
    }
}

fn initialize_centroids(points: &[f32], n_clusters: usize, rng: &mut SimpleRng) -> Vec<f32> {
    let n_points = points.len() / 3;
    let mut centroids = Vec::with_capacity(n_clusters * 3);
    
    if n_points >= n_clusters {
        let first_idx = (rng.next_u32() as usize) % n_points;
        centroids.push(points[first_idx * 3]);
        centroids.push(points[first_idx * 3 + 1]);
        centroids.push(points[first_idx * 3 + 2]);
        
        let mut chosen_indices = std::collections::HashSet::with_capacity(n_clusters);
        chosen_indices.insert(first_idx);
        
        let candidates = std::cmp::max(5, std::cmp::min(64, 2_000_000 / n_clusters));
        for _ in 1..n_clusters {
            let mut best_pt_idx = 0;
            let mut max_min_dist_sq = -1.0;
            
            for _ in 0..candidates {
                let candidate_idx = (rng.next_u32() as usize) % n_points;
                if chosen_indices.contains(&candidate_idx) {
                    continue;
                }
                
                let cx = points[candidate_idx * 3];
                let cy = points[candidate_idx * 3 + 1];
                let cz = points[candidate_idx * 3 + 2];
                
                let mut min_dist_sq = f32::INFINITY;
                let n_existing = centroids.len() / 3;
                for j in 0..n_existing {
                    let ex = centroids[j * 3];
                    let ey = centroids[j * 3 + 1];
                    let ez = centroids[j * 3 + 2];
                    let dx = cx - ex;
                    let dy = cy - ey;
                    let dz = cz - ez;
                    let d = dx*dx + dy*dy + dz*dz;
                    if d < min_dist_sq {
                        min_dist_sq = d;
                    }
                }
                
                if min_dist_sq > max_min_dist_sq {
                    max_min_dist_sq = min_dist_sq;
                    best_pt_idx = candidate_idx;
                }
            }
            
            chosen_indices.insert(best_pt_idx);
            centroids.push(points[best_pt_idx * 3]);
            centroids.push(points[best_pt_idx * 3 + 1]);
            centroids.push(points[best_pt_idx * 3 + 2]);
        }
    } else {
        for i in 0..n_clusters {
            let idx = i % n_points;
            centroids.push(points[idx * 3]);
            centroids.push(points[idx * 3 + 1]);
            centroids.push(points[idx * 3 + 2]);
        }
    }
    centroids
}

fn run_kmeans(
    points: &[f32],
    n_clusters: usize,
    max_iters: usize,
    rng: &mut SimpleRng,
) -> Vec<f32> {
    let n_points = points.len() / 3;
    if n_points == 0 {
        return vec![0.0; n_clusters * 3];
    }
    
    let mut centroids = initialize_centroids(points, n_clusters, rng);
    let mut assignments = vec![0; n_points];
    let mut new_centroids = vec![0.0; n_clusters * 3];
    let mut counts = vec![0; n_clusters];
    
    for _iter in 0..max_iters {
        new_centroids.fill(0.0);
        counts.fill(0);
        
        let mut changed = 0;
        
        for i in 0..n_points {
            let x = points[i * 3];
            let y = points[i * 3 + 1];
            let z = points[i * 3 + 2];
            
            let mut min_dist_sq = f32::INFINITY;
            let mut best_idx = 0;
            
            for j in 0..n_clusters {
                let cx = centroids[j * 3];
                let cy = centroids[j * 3 + 1];
                let cz = centroids[j * 3 + 2];
                let dx = x - cx;
                let dy = y - cy;
                let dz = z - cz;
                let d = dx*dx + dy*dy + dz*dz;
                if d < min_dist_sq {
                    min_dist_sq = d;
                    best_idx = j;
                }
            }
            
            if assignments[i] != best_idx {
                assignments[i] = best_idx;
                changed += 1;
            }
            
            new_centroids[best_idx * 3] += x;
            new_centroids[best_idx * 3 + 1] += y;
            new_centroids[best_idx * 3 + 2] += z;
            counts[best_idx] += 1;
        }
        
        for j in 0..n_clusters {
            let c = counts[j];
            if c > 0 {
                centroids[j * 3] = new_centroids[j * 3] / c as f32;
                centroids[j * 3 + 1] = new_centroids[j * 3 + 1] / c as f32;
                centroids[j * 3 + 2] = new_centroids[j * 3 + 2] / c as f32;
            } else {
                let random_pt_idx = (rng.next_u32() as usize) % n_points;
                centroids[j * 3] = points[random_pt_idx * 3];
                centroids[j * 3 + 1] = points[random_pt_idx * 3 + 1];
                centroids[j * 3 + 2] = points[random_pt_idx * 3 + 2];
            }
        }
        
        if changed == 0 {
            break;
        }
    }
    
    centroids
}

fn parallel_assign_and_accumulate(
    points: &[f32],
    centroids: &[f32],
    assignments: &mut [usize],
    n_clusters: usize,
    num_threads: usize,
) -> (Vec<f32>, Vec<usize>, usize) {
    let n_points = points.len() / 3;
    let chunk_size = (n_points + num_threads - 1) / num_threads;
    
    let mut thread_new_centroids = vec![vec![0.0; n_clusters * 3]; num_threads];
    let mut thread_counts = vec![vec![0; n_clusters]; num_threads];
    let mut thread_changed = vec![0; num_threads];
    
    let mut assignments_chunks: Vec<&mut [usize]> = Vec::new();
    let mut remaining = assignments;
    while !remaining.is_empty() {
        let (left, right) = remaining.split_at_mut(std::cmp::min(chunk_size, remaining.len()));
        assignments_chunks.push(left);
        remaining = right;
    }
    
    std::thread::scope(|s| {
        let points_chunks = points.chunks(chunk_size * 3);
        let assignments_chunks_iter = assignments_chunks.into_iter();
        let new_centroids_iter = thread_new_centroids.iter_mut();
        let counts_iter = thread_counts.iter_mut();
        let changed_iter = thread_changed.iter_mut();
        
        for ((((points_chunk, assignments_chunk), new_centroids_ref), counts_ref), changed_ref) in
            points_chunks
                .zip(assignments_chunks_iter)
                .zip(new_centroids_iter)
                .zip(counts_iter)
                .zip(changed_iter)
        {
            let centroids_ref = &centroids;
            
            s.spawn(move || {
                let chunk_n_points = points_chunk.len() / 3;
                for i in 0..chunk_n_points {
                    let x = points_chunk[i * 3];
                    let y = points_chunk[i * 3 + 1];
                    let z = points_chunk[i * 3 + 2];
                    
                    let mut min_dist_sq = f32::INFINITY;
                    let mut best_idx = 0;
                    
                    for j in 0..n_clusters {
                        let cx = centroids_ref[j * 3];
                        let cy = centroids_ref[j * 3 + 1];
                        let cz = centroids_ref[j * 3 + 2];
                        let dx = x - cx;
                        let dy = y - cy;
                        let dz = z - cz;
                        let d = dx*dx + dy*dy + dz*dz;
                        if d < min_dist_sq {
                            min_dist_sq = d;
                            best_idx = j;
                        }
                    }
                    
                    if assignments_chunk[i] != best_idx {
                        assignments_chunk[i] = best_idx;
                        *changed_ref += 1;
                    }
                    
                    new_centroids_ref[best_idx * 3] += x;
                    new_centroids_ref[best_idx * 3 + 1] += y;
                    new_centroids_ref[best_idx * 3 + 2] += z;
                    counts_ref[best_idx] += 1;
                }
            });
        }
    });
    
    let mut global_new_centroids = vec![0.0; n_clusters * 3];
    let mut global_counts = vec![0; n_clusters];
    let mut global_changed = 0;
    
    for t_idx in 0..num_threads {
        global_changed += thread_changed[t_idx];
        for j in 0..n_clusters {
            global_new_centroids[j * 3] += thread_new_centroids[t_idx][j * 3];
            global_new_centroids[j * 3 + 1] += thread_new_centroids[t_idx][j * 3 + 1];
            global_new_centroids[j * 3 + 2] += thread_new_centroids[t_idx][j * 3 + 2];
            global_counts[j] += thread_counts[t_idx][j];
        }
    }
    
    (global_new_centroids, global_counts, global_changed)
}

fn run_kmeans_parallel(
    points: &[f32],
    n_clusters: usize,
    max_iters: usize,
    rng: &mut SimpleRng,
    num_threads: usize,
) -> Vec<f32> {
    let n_points = points.len() / 3;
    if n_points == 0 {
        return vec![0.0; n_clusters * 3];
    }
    
    let mut centroids = initialize_centroids(points, n_clusters, rng);
    let mut assignments = vec![0; n_points];
    
    for _iter in 0..max_iters {
        let (new_centroids, counts, changed) = parallel_assign_and_accumulate(
            points,
            &centroids,
            &mut assignments,
            n_clusters,
            num_threads,
        );
        
        let mut centroids_updated = centroids.clone();
        for j in 0..n_clusters {
            let c = counts[j];
            if c > 0 {
                centroids_updated[j * 3] = new_centroids[j * 3] / c as f32;
                centroids_updated[j * 3 + 1] = new_centroids[j * 3 + 1] / c as f32;
                centroids_updated[j * 3 + 2] = new_centroids[j * 3 + 2] / c as f32;
            } else {
                let random_pt_idx = (rng.next_u32() as usize) % n_points;
                centroids_updated[j * 3] = points[random_pt_idx * 3];
                centroids_updated[j * 3 + 1] = points[random_pt_idx * 3 + 1];
                centroids_updated[j * 3 + 2] = points[random_pt_idx * 3 + 2];
            }
        }
        
        centroids = centroids_updated;
        if changed == 0 {
            break;
        }
    }
    
    centroids
}

fn build_centroid_tree(centroids: Vec<f32>, use_hierarchy: bool, rng: &mut SimpleRng) -> CentroidTree {
    let k = centroids.len() / 3;
    if !use_hierarchy || k <= 200 {
        return CentroidTree {
            meta_centroids: Vec::new(),
            centroids,
            use_hierarchy: false,
        };
    }
    
    let m = (k as f32).sqrt().round() as usize;
    let m = std::cmp::max(2, m);
    
    let meta_centroids_coords = run_kmeans(&centroids, m, 10, rng);
    
    let mut meta_centroids = Vec::with_capacity(m);
    for i in 0..m {
        meta_centroids.push(MetaCentroid {
            coords: [
                meta_centroids_coords[i * 3],
                meta_centroids_coords[i * 3 + 1],
                meta_centroids_coords[i * 3 + 2],
            ],
            child_indices: Vec::new(),
        });
    }
    
    for i in 0..k {
        let cx = centroids[i * 3];
        let cy = centroids[i * 3 + 1];
        let cz = centroids[i * 3 + 2];
        
        let mut min_dist_sq = f32::INFINITY;
        let mut best_meta_idx = 0;
        for j in 0..m {
            let mc = &meta_centroids[j];
            let dx = cx - mc.coords[0];
            let dy = cy - mc.coords[1];
            let dz = cz - mc.coords[2];
            let d = dx*dx + dy*dy + dz*dz;
            if d < min_dist_sq {
                min_dist_sq = d;
                best_meta_idx = j;
            }
        }
        meta_centroids[best_meta_idx].child_indices.push(i);
    }
    
    CentroidTree {
        meta_centroids,
        centroids,
        use_hierarchy: true,
    }
}

fn parallel_assign_batch(
    points_batch: &[f32],
    tree: &CentroidTree,
    assignments_batch: &mut [u32],
    start_point_idx: usize,
    n_clusters: usize,
    num_threads: usize,
) {
    let n_points = points_batch.len() / 3;
    let chunk_size = (n_points + num_threads - 1) / num_threads;
    
    let mut assignments_chunks: Vec<&mut [u32]> = Vec::new();
    let mut remaining = assignments_batch;
    while !remaining.is_empty() {
        let (left, right) = remaining.split_at_mut(std::cmp::min(chunk_size, remaining.len()));
        assignments_chunks.push(left);
        remaining = right;
    }
    
    std::thread::scope(|s| {
        for (t_idx, (points_chunk, assignments_chunk)) in points_batch
            .chunks(chunk_size * 3)
            .zip(assignments_chunks.into_iter())
            .enumerate()
        {
            let tree_ref = &tree;
            let thread_start_idx = start_point_idx + t_idx * chunk_size;
            
            s.spawn(move || {
                let chunk_n_points = points_chunk.len() / 3;
                for i in 0..chunk_n_points {
                    let global_idx = thread_start_idx + i;
                    let cluster_idx = if global_idx < n_clusters {
                        global_idx as u32
                    } else {
                        let x = points_chunk[i * 3];
                        let y = points_chunk[i * 3 + 1];
                        let z = points_chunk[i * 3 + 2];
                        tree_ref.find_nearest(x, y, z) as u32
                    };
                    assignments_chunk[i] = cluster_idx;
                }
            });
        }
    });
}

fn read_exact_or_eof<R: std::io::Read>(reader: &mut R, mut buf: &mut [u8]) -> std::io::Result<usize> {
    let mut total_read = 0;
    while !buf.is_empty() {
        match reader.read(buf) {
            Ok(0) => break,
            Ok(n) => {
                total_read += n;
                buf = &mut buf[n..];
            }
            Err(ref e) if e.kind() == std::io::ErrorKind::Interrupted => {}
            Err(e) => return Err(e),
        }
    }
    Ok(total_read)
}

fn bytes_to_f32_le(bytes: &[u8], floats: &mut [f32]) {
    assert_eq!(bytes.len(), floats.len() * 4);
    for (i, chunk) in bytes.chunks_exact(4).enumerate() {
        let arr = [chunk[0], chunk[1], chunk[2], chunk[3]];
        floats[i] = f32::from_le_bytes(arr);
    }
}

fn write_assignments_batch<W: std::io::Write>(
    writer: &mut W,
    assignments: &[u32],
    byte_buf: &mut [u8],
) -> std::io::Result<()> {
    assert_eq!(byte_buf.len(), assignments.len() * 4);
    for (i, &idx) in assignments.iter().enumerate() {
        let bytes = idx.to_le_bytes();
        byte_buf[i * 4] = bytes[0];
        byte_buf[i * 4 + 1] = bytes[1];
        byte_buf[i * 4 + 2] = bytes[2];
        byte_buf[i * 4 + 3] = bytes[3];
    }
    writer.write_all(byte_buf)?;
    Ok(())
}

fn read_header<R: std::io::BufRead>(reader: &mut R) -> std::io::Result<(usize, usize)> {
    let mut line = String::new();
    loop {
        line.clear();
        let bytes_read = reader.read_line(&mut line)?;
        if bytes_read == 0 {
            return Err(std::io::Error::new(
                std::io::ErrorKind::UnexpectedEof,
                "EOF reached before reading header",
            ));
        }
        let trimmed = line.trim();
        if !trimmed.is_empty() {
            let parts: Vec<&str> = trimmed.split_whitespace().collect();
            if parts.len() >= 2 {
                let num_points = parts[0].parse::<usize>().map_err(|e| {
                    std::io::Error::new(std::io::ErrorKind::InvalidData, e)
                })?;
                let num_clusters = parts[1].parse::<usize>().map_err(|e| {
                    std::io::Error::new(std::io::ErrorKind::InvalidData, e)
                })?;
                return Ok((num_points, num_clusters));
            }
        }
    }
}

fn main() -> std::io::Result<()> {
    let mut stdin_reader = BufReader::new(io::stdin());
    let (num_points, n_clusters) = read_header(&mut stdin_reader)?;
    
    if num_points == 0 {
        return Ok(());
    }
    
    let num_threads = std::thread::available_parallelism()
        .map(|n| n.get())
        .unwrap_or(4);
    let num_threads = std::cmp::max(1, num_threads);
    
    let mut rng = SimpleRng::new(42);
    
    // Adaptive sample size
    let max_sample_size = std::cmp::max(100_000, std::cmp::min(2_000_000, 2_000_000_000 / n_clusters));
    let sample_size = std::cmp::min(num_points, max_sample_size);
    let step = if num_points > sample_size {
        num_points / sample_size
    } else {
        1
    };
    
    let batch_points_limit = 1_200_000;
    
    // Memory decision: under 30 million points can be stored in RAM safely
    let use_disk = num_points > 30_000_000;
    
    if use_disk {
        // Try creating temp file
        let file_res = File::create(TEMP_FILE_PATH);
        match file_res {
            Ok(temp_file) => {
                let mut cleanup = TempFileCleanup {
                    path: TEMP_FILE_PATH,
                    active: true,
                };
                
                let mut temp_writer = BufWriter::with_capacity(8 * 1024 * 1024, temp_file);
                let mut sample_points = Vec::with_capacity(sample_size * 3);
                
                let mut global_idx = 0;
                let mut batch_bytes = vec![0u8; batch_points_limit * 12];
                let mut batch_floats = vec![0.0f32; batch_points_limit * 3];
                
                while global_idx < num_points {
                    let to_read_points = std::cmp::min(batch_points_limit, num_points - global_idx);
                    let bytes_to_read = to_read_points * 12;
                    let n_read_bytes = read_exact_or_eof(&mut stdin_reader, &mut batch_bytes[..bytes_to_read])?;
                    if n_read_bytes == 0 {
                        break;
                    }
                    let n_points_read = n_read_bytes / 12;
                    
                    temp_writer.write_all(&batch_bytes[..n_points_read * 12])?;
                    
                    bytes_to_f32_le(&batch_bytes[..n_points_read * 12], &mut batch_floats[..n_points_read * 3]);
                    
                    for i in 0..n_points_read {
                        let cur_global_idx = global_idx + i;
                        if cur_global_idx % step == 0 && sample_points.len() / 3 < sample_size {
                            sample_points.push(batch_floats[i * 3]);
                            sample_points.push(batch_floats[i * 3 + 1]);
                            sample_points.push(batch_floats[i * 3 + 2]);
                        }
                    }
                    
                    global_idx += n_points_read;
                    if n_points_read < to_read_points {
                        break;
                    }
                }
                
                temp_writer.flush()?;
                drop(temp_writer);
                
                let centroids = run_kmeans_parallel(
                    &sample_points,
                    n_clusters,
                    15,
                    &mut rng,
                    num_threads,
                );
                
                let tree = build_centroid_tree(centroids, n_clusters > 200, &mut rng);
                
                let temp_read_file = File::open(TEMP_FILE_PATH)?;
                let mut temp_reader = BufReader::with_capacity(8 * 1024 * 1024, temp_read_file);
                let mut stdout_writer = BufWriter::with_capacity(8 * 1024 * 1024, io::stdout());
                
                let mut assignments_batch = vec![0u32; batch_points_limit];
                let mut assignments_byte_buf = vec![0u8; batch_points_limit * 4];
                
                let mut out_global_idx = 0;
                while out_global_idx < num_points {
                    let to_read_points = std::cmp::min(batch_points_limit, num_points - out_global_idx);
                    let bytes_to_read = to_read_points * 12;
                    let n_read_bytes = read_exact_or_eof(&mut temp_reader, &mut batch_bytes[..bytes_to_read])?;
                    if n_read_bytes == 0 {
                        break;
                    }
                    let n_points_read = n_read_bytes / 12;
                    
                    bytes_to_f32_le(&batch_bytes[..n_points_read * 12], &mut batch_floats[..n_points_read * 3]);
                    
                    parallel_assign_batch(
                        &batch_floats[..n_points_read * 3],
                        &tree,
                        &mut assignments_batch[..n_points_read],
                        out_global_idx,
                        n_clusters,
                        num_threads,
                    );
                    
                    write_assignments_batch(
                        &mut stdout_writer,
                        &assignments_batch[..n_points_read],
                        &mut assignments_byte_buf[..n_points_read * 4],
                    )?;
                    
                    out_global_idx += n_points_read;
                    if n_points_read < to_read_points {
                        break;
                    }
                }
                
                stdout_writer.flush()?;
                drop(temp_reader);
                
                let _ = std::fs::remove_file(TEMP_FILE_PATH);
                cleanup.active = false;
            }
            Err(_) => {
                // Disk creation failed, fallback to in-memory
                run_in_memory(
                    &mut stdin_reader,
                    num_points,
                    n_clusters,
                    sample_size,
                    step,
                    batch_points_limit,
                    num_threads,
                    &mut rng,
                )?;
            }
        }
    } else {
        run_in_memory(
            &mut stdin_reader,
            num_points,
            n_clusters,
            sample_size,
            step,
            batch_points_limit,
            num_threads,
            &mut rng,
        )?;
    }
    
    Ok(())
}

fn run_in_memory<R: std::io::BufRead>(
    reader: &mut R,
    num_points: usize,
    n_clusters: usize,
    sample_size: usize,
    step: usize,
    batch_points_limit: usize,
    num_threads: usize,
    rng: &mut SimpleRng,
) -> std::io::Result<()> {
    let mut points = Vec::with_capacity(num_points * 3);
    
    let mut batch_bytes = vec![0u8; batch_points_limit * 12];
    let mut batch_floats = vec![0.0f32; batch_points_limit * 3];
    
    let mut global_idx = 0;
    while global_idx < num_points {
        let to_read_points = std::cmp::min(batch_points_limit, num_points - global_idx);
        let bytes_to_read = to_read_points * 12;
        let n_read_bytes = read_exact_or_eof(reader, &mut batch_bytes[..bytes_to_read])?;
        if n_read_bytes == 0 {
            break;
        }
        let n_points_read = n_read_bytes / 12;
        bytes_to_f32_le(&batch_bytes[..n_points_read * 12], &mut batch_floats[..n_points_read * 3]);
        
        points.extend_from_slice(&batch_floats[..n_points_read * 3]);
        global_idx += n_points_read;
        if n_points_read < to_read_points {
            break;
        }
    }
    
    let mut sample_points = Vec::with_capacity(sample_size * 3);
    for i in 0..sample_size {
        let idx = (i * step) % num_points;
        sample_points.push(points[idx * 3]);
        sample_points.push(points[idx * 3 + 1]);
        sample_points.push(points[idx * 3 + 2]);
    }
    
    let centroids = run_kmeans_parallel(
        &sample_points,
        n_clusters,
        15,
        rng,
        num_threads,
    );
    
    let tree = build_centroid_tree(centroids, n_clusters > 200, rng);
    
    let mut stdout_writer = BufWriter::with_capacity(8 * 1024 * 1024, io::stdout());
    let mut assignments_batch = vec![0u32; batch_points_limit];
    let mut assignments_byte_buf = vec![0u8; batch_points_limit * 4];
    
    let mut out_global_idx = 0;
    while out_global_idx < num_points {
        let chunk_size = std::cmp::min(batch_points_limit, num_points - out_global_idx);
        
        parallel_assign_batch(
            &points[out_global_idx * 3..(out_global_idx + chunk_size) * 3],
            &tree,
            &mut assignments_batch[..chunk_size],
            out_global_idx,
            n_clusters,
            num_threads,
        );
        
        write_assignments_batch(
            &mut stdout_writer,
            &assignments_batch[..chunk_size],
            &mut assignments_byte_buf[..chunk_size * 4],
        )?;
        
        out_global_idx += chunk_size;
    }
    
    stdout_writer.flush()?;
    Ok(())
}
