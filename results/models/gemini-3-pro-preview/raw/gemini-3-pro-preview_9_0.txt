using System;
using System.IO;
using System.Collections.Generic;
using System.Threading.Tasks;

public static class FastIO
{
    private const int BufferSize = 1024 * 1024;
    private static readonly byte[] Buffer = new byte[BufferSize];
    private static int _bufferLength = 0;
    private static int _bufferPosition = 0;
    private static Stream _stream;

    public static void Initialize()
    {
        _stream = Console.OpenStandardInput();
    }

    private static byte ReadByte()
    {
        if (_bufferPosition >= _bufferLength)
        {
            _bufferLength = _stream.Read(Buffer, 0, BufferSize);
            _bufferPosition = 0;
            if (_bufferLength == 0)
                return 0; // EOF
        }
        return Buffer[_bufferPosition++];
    }

    public static bool TryReadInt(out int value)
    {
        value = 0;
        byte b = ReadByte();
        while (b != 0 && b <= 32) // skip whitespace
        {
            b = ReadByte();
        }
        if (b == 0)
            return false;
        
        while (b > 32)
        {
            value = value * 10 + (b - '0');
            b = ReadByte();
        }
        return true;
    }

    public static bool TryReadLong(out long value)
    {
        value = 0;
        byte b = ReadByte();
        while (b != 0 && b <= 32) // skip whitespace
        {
            b = ReadByte();
        }
        if (b == 0)
            return false;
        
        while (b > 32)
        {
            value = value * 10 + (b - '0');
            b = ReadByte();
        }
        return true;
    }
}

public struct Cut
{
    public int Index;
    public long Length;
}

public struct PerturbedCut : IComparable<PerturbedCut>
{
    public int Index;
    public long Length;
    public double SortKey;

    public int CompareTo(PerturbedCut other)
    {
        // Sort descending by SortKey
        return other.SortKey.CompareTo(this.SortKey);
    }
}

class Program
{
    static void Main(string[] args)
    {
        FastIO.Initialize();

        if (!FastIO.TryReadInt(out int n))
            return;
        if (!FastIO.TryReadLong(out long stockLength))
            return;

        Cut[] cuts = new Cut[n];
        for (int i = 0; i < n; i++)
        {
            if (FastIO.TryReadLong(out long len))
            {
                cuts[i] = new Cut { Index = i, Length = len };
            }
            else
            {
                n = i;
                Array.Resize(ref cuts, n);
                break;
            }
        }

        if (n == 0)
        {
            Console.WriteLine(0);
            return;
        }

        // Dynamically adjust trials and passes based on N to balance quality and execution time
        int numTrials;
        int numPasses;
        if (n <= 10000)
        {
            numTrials = 256;
            numPasses = 5;
        }
        else if (n <= 100000)
        {
            numTrials = 128;
            numPasses = 4;
        }
        else if (n <= 500000)
        {
            numTrials = 64;
            numPasses = 3;
        }
        else if (n <= 2000000)
        {
            numTrials = 32;
            numPasses = 2;
        }
        else
        {
            numTrials = 8;
            numPasses = 2;
        }

        object bestLock = new object();
        int bestNumStocks = int.MaxValue;
        int[] bestAssignment = null;

        Parallel.For(0, numTrials, trialIndex =>
        {
            double noiseLevel;
            if (trialIndex == 0)
            {
                noiseLevel = 0.0;
            }
            else
            {
                double ratio = (double)trialIndex / numTrials;
                if (ratio < 0.1) noiseLevel = 0.0001;
                else if (ratio < 0.25) noiseLevel = 0.001;
                else if (ratio < 0.5) noiseLevel = 0.01;
                else if (ratio < 0.75) noiseLevel = 0.05;
                else if (ratio < 0.9) noiseLevel = 0.1;
                else noiseLevel = 0.2;
            }

            var (numStocks, assignment) = RunSolver(cuts, stockLength, noiseLevel, trialIndex, numPasses);

            lock (bestLock)
            {
                if (numStocks < bestNumStocks)
                {
                    bestNumStocks = numStocks;
                    bestAssignment = assignment;
                }
            }
        });

        // Reconstruct groups for printing
        int treeSize = 1;
        while (treeSize < n)
        {
            treeSize <<= 1;
        }

        int[] firstCutInBin = new int[treeSize];
        for (int i = 0; i < firstCutInBin.Length; i++) firstCutInBin[i] = -1;
        int[] nextCutInBin = new int[n];

        for (int i = 0; i < n; i++)
        {
            int b = bestAssignment[i];
            nextCutInBin[i] = firstCutInBin[b];
            firstCutInBin[b] = i;
        }

        // Print output using buffered writer
        using (var writer = new StreamWriter(Console.OpenStandardOutput(), System.Text.Encoding.ASCII, 1024 * 1024))
        {
            writer.WriteLine(bestNumStocks);
            List<int> tempIndices = new List<int>();
            for (int b = 0; b < treeSize; b++)
            {
                if (firstCutInBin[b] != -1)
                {
                    tempIndices.Clear();
                    int cutIdx = firstCutInBin[b];
                    while (cutIdx != -1)
                    {
                        tempIndices.Add(cutIdx);
                        cutIdx = nextCutInBin[cutIdx];
                    }
                    tempIndices.Sort();
                    for (int i = 0; i < tempIndices.Count; i++)
                    {
                        if (i > 0)
                            writer.Write(' ');
                        writer.Write(tempIndices[i]);
                    }
                    writer.WriteLine();
                }
            }
        }
    }

    static (int NumStocks, int[] Assignment) RunSolver(Cut[] originalCuts, long stockLength, double noiseLevel, int seed, int numPasses)
    {
        int n = originalCuts.Length;
        int treeSize = 1;
        while (treeSize < n)
        {
            treeSize <<= 1;
        }

        // Iterative segment tree
        long[] tree = new long[2 * treeSize];
        for (int i = 0; i < tree.Length; i++) tree[i] = stockLength;
        
        long[] binCapacities = new long[treeSize];
        for (int i = 0; i < binCapacities.Length; i++) binCapacities[i] = stockLength;

        int FindAndDecreaseIterative(long value)
        {
            if (tree[1] < value) return -1;
            
            int node = 1;
            while (node < treeSize)
            {
                int leftNode = 2 * node;
                int rightNode = 2 * node + 1;
                if (tree[leftNode] >= value)
                {
                    node = leftNode;
                }
                else
                {
                    node = rightNode;
                }
            }
            
            int binIdx = node - treeSize;
            tree[node] -= value;
            binCapacities[binIdx] -= value;
            
            node /= 2;
            while (node > 0)
            {
                tree[node] = Math.Max(tree[2 * node], tree[2 * node + 1]);
                node /= 2;
            }
            
            return binIdx;
        }

        void UpdateCapacityIterative(int binIdx, long delta)
        {
            int node = treeSize + binIdx;
            tree[node] += delta;
            binCapacities[binIdx] += delta;
            
            node /= 2;
            while (node > 0)
            {
                tree[node] = Math.Max(tree[2 * node], tree[2 * node + 1]);
                node /= 2;
            }
        }

        // Perturb and sort cuts
        PerturbedCut[] perturbedCuts = new PerturbedCut[n];
        var rand = new Random(seed);
        for (int i = 0; i < n; i++)
        {
            double noise = 1.0 + (rand.NextDouble() - 0.5) * noiseLevel;
            perturbedCuts[i] = new PerturbedCut
            {
                Index = originalCuts[i].Index,
                Length = originalCuts[i].Length,
                SortKey = originalCuts[i].Length * noise
            };
        }
        Array.Sort(perturbedCuts);

        // Run First Fit Decreasing (FFD)
        int[] cutBinAssignment = new int[n];
        for (int i = 0; i < n; i++)
        {
            int binIdx = FindAndDecreaseIterative(perturbedCuts[i].Length);
            cutBinAssignment[perturbedCuts[i].Index] = binIdx;
        }

        // Setup evacuation structures
        int[] firstCutInBin = new int[treeSize];
        for (int i = 0; i < firstCutInBin.Length; i++) firstCutInBin[i] = -1;
        int[] nextCutInBin = new int[n];

        for (int i = 0; i < n; i++)
        {
            int b = cutBinAssignment[i];
            nextCutInBin[i] = firstCutInBin[b];
            firstCutInBin[b] = i;
        }

        List<int> activeBins = new List<int>();
        List<int> tempCutIndices = new List<int>();
        List<int> tempTargetBins = new List<int>();

        // Run evacuation passes
        for (int pass = 0; pass < numPasses; pass++)
        {
            activeBins.Clear();
            for (int b = 0; b < treeSize; b++)
            {
                if (firstCutInBin[b] != -1)
                {
                    activeBins.Add(b);
                }
            }

            // Sort active bins descending by remaining capacity (emptiest bins first)
            activeBins.Sort((x, y) => binCapacities[y].CompareTo(binCapacities[x]));

            for (int i = 0; i < activeBins.Count; i++)
            {
                int b_idx = activeBins[i];
                if (firstCutInBin[b_idx] == -1) continue;

                long originalCap = binCapacities[b_idx];
                UpdateCapacityIterative(b_idx, -originalCap);

                tempCutIndices.Clear();
                tempTargetBins.Clear();

                int cutIdx = firstCutInBin[b_idx];
                bool success = true;
                while (cutIdx != -1)
                {
                    long cutLen = originalCuts[cutIdx].Length;
                    int target = FindAndDecreaseIterative(cutLen);
                    if (target == -1)
                    {
                        success = false;
                        break;
                    }
                    tempCutIndices.Add(cutIdx);
                    tempTargetBins.Add(target);
                    cutIdx = nextCutInBin[cutIdx];
                }

                if (success)
                {
                    for (int k = 0; k < tempCutIndices.Count; k++)
                    {
                        int c = tempCutIndices[k];
                        int target = tempTargetBins[k];
                        nextCutInBin[c] = firstCutInBin[target];
                        firstCutInBin[target] = c;
                    }
                    firstCutInBin[b_idx] = -1;
                }
                else
                {
                    UpdateCapacityIterative(b_idx, originalCap);
                    for (int k = 0; k < tempCutIndices.Count; k++)
                    {
                        int target = tempTargetBins[k];
                        long cutLen = originalCuts[tempCutIndices[k]].Length;
                        UpdateCapacityIterative(target, cutLen);
                    }
                }
            }
        }

        // Build final assignment
        int finalNumStocks = 0;
        int[] finalAssignment = new int[n];
        for (int b = 0; b < treeSize; b++)
        {
            if (firstCutInBin[b] != -1)
            {
                int cutIdx = firstCutInBin[b];
                while (cutIdx != -1)
                {
                    finalAssignment[cutIdx] = finalNumStocks;
                    cutIdx = nextCutInBin[cutIdx];
                }
                finalNumStocks++;
            }
        }

        return (finalNumStocks, finalAssignment);
    }
}
