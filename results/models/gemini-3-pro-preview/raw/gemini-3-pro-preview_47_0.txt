fn rotl(x: u32, shift: u32) -> u32 {
    return (x << shift) | (x >> (32u - shift));
}

@group(0) @binding(0) var<storage, read>       baseData : array<u32, 16>;
@group(0) @binding(1) var<storage, read_write>  result   : array<u32, 4>;
@group(0) @binding(2) var<uniform>              params   : vec4<u32>;
// params.x = difficulty, params.y = range_start, params.z = range_size

@compute @workgroup_size(256)
fn main(@builtin(global_invocation_id) gid : vec3<u32>) {
    let difficulty = params.x;
    let range_start = params.y;
    let range_size = params.z;

    if (gid.x >= range_size) {
        return;
    }

    let nonce = range_start + gid.x;

    // Initialize state: state[i] = (baseData[i] ^ nonce) ^ (nonce + i)
    var state : array<u32, 16>;
    for (var i = 0u; i < 16u; i = i + 1u) {
        state[i] = (baseData[i] ^ nonce) ^ (nonce + i);
    }

    // 8 rounds of quarter-round mixing (ChaCha-like)
    for (var round = 0u; round < 8u; round = round + 1u) {
        // Group 1: state[0..3]
        state[0u] = state[0u] + state[1u]; state[3u] = state[3u] ^ state[0u]; state[3u] = rotl(state[3u], 16u);
        state[2u] = state[2u] + state[3u]; state[1u] = state[1u] ^ state[2u]; state[1u] = rotl(state[1u], 12u);
        state[0u] = state[0u] + state[1u]; state[3u] = state[3u] ^ state[0u]; state[3u] = rotl(state[3u], 8u);
        state[2u] = state[2u] + state[3u]; state[1u] = state[1u] ^ state[2u]; state[1u] = rotl(state[1u], 7u);

        // Group 2: state[4..7]
        state[4u] = state[4u] + state[5u]; state[7u] = state[7u] ^ state[4u]; state[7u] = rotl(state[7u], 16u);
        state[6u] = state[6u] + state[7u]; state[5u] = state[5u] ^ state[6u]; state[5u] = rotl(state[5u], 12u);
        state[4u] = state[4u] + state[5u]; state[7u] = state[7u] ^ state[4u]; state[7u] = rotl(state[7u], 8u);
        state[6u] = state[6u] + state[7u]; state[5u] = state[5u] ^ state[6u]; state[5u] = rotl(state[5u], 7u);

        // Group 3: state[8..11]
        state[8u] = state[8u] + state[9u]; state[11u] = state[11u] ^ state[8u]; state[11u] = rotl(state[11u], 16u);
        state[10u] = state[10u] + state[11u]; state[9u] = state[9u] ^ state[10u]; state[9u] = rotl(state[9u], 12u);
        state[8u] = state[8u] + state[9u]; state[11u] = state[11u] ^ state[8u]; state[11u] = rotl(state[11u], 8u);
        state[10u] = state[10u] + state[11u]; state[9u] = state[9u] ^ state[10u]; state[9u] = rotl(state[9u], 7u);

        // Group 4: state[12..15]
        state[12u] = state[12u] + state[13u]; state[15u] = state[15u] ^ state[12u]; state[15u] = rotl(state[15u], 16u);
        state[14u] = state[14u] + state[15u]; state[13u] = state[13u] ^ state[14u]; state[13u] = rotl(state[13u], 12u);
        state[12u] = state[12u] + state[13u]; state[15u] = state[15u] ^ state[12u]; state[15u] = rotl(state[15u], 8u);
        state[14u] = state[14u] + state[15u]; state[13u] = state[13u] ^ state[14u]; state[13u] = rotl(state[13u], 7u);
    }

    // Fold 16 words to 2
    let h0 = state[0u] ^ state[2u] ^ state[4u] ^ state[6u] ^ state[8u] ^ state[10u] ^ state[12u] ^ state[14u];
    let h1 = state[1u] ^ state[3u] ^ state[5u] ^ state[7u] ^ state[9u] ^ state[11u] ^ state[13u] ^ state[15u];

    // Trailing zero count of the 64-bit value: [h0 | (h1 << 32)]
    var trailing_zeros = 0u;
    let tz0 = countTrailingZeros(h0);
    if (tz0 < 32u) {
        trailing_zeros = tz0;
    } else {
        trailing_zeros = 32u + countTrailingZeros(h1);
    }

    if (trailing_zeros >= difficulty) {
        result[0u] = 1u;
        result[1u] = nonce;
        result[2u] = h0;
        result[3u] = h1;
    }
}
