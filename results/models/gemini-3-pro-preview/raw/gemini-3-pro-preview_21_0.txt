#include <iostream>
#include <vector>
#include <deque>
#include <unordered_set>
#include <unordered_map>
#include <queue>
#include <cmath>
#include <algorithm>
#include <array>
#include <utility>

int D;
std::array<int, 32> bounds{};

struct Point {
    std::array<int, 32> coords{};
    
    bool operator==(const Point& other) const {
        for (int i = 0; i < D; ++i) {
            if (coords[i] != other.coords[i]) return false;
        }
        return true;
    }
    
    bool operator<(const Point& other) const {
        for (int i = 0; i < D; ++i) {
            if (coords[i] != other.coords[i]) {
                return coords[i] < other.coords[i];
            }
        }
        return false;
    }
};

struct PointHash {
    std::size_t operator()(const Point& p) const {
        std::size_t h = 17;
        for (int i = 0; i < D; ++i) {
            h = h * 31 + p.coords[i];
        }
        return h;
    }
};

int manhattan_distance(const Point& a, const Point& b) {
    int dist = 0;
    for (int i = 0; i < D; ++i) {
        dist += std::abs(a.coords[i] - b.coords[i]);
    }
    return dist;
}

struct SearchNode {
    Point pos;
    int g;
    int f;
    int parent_idx;
};

std::unordered_map<Point, int, PointHash> build_occupied_map(const std::deque<Point>& snake) {
    std::unordered_map<Point, int, PointHash> m;
    m.reserve(snake.size());
    int L = snake.size();
    for (int i = 0; i < L; ++i) {
        const auto& p = snake[i];
        int val = L - 1 - i;
        auto it = m.find(p);
        if (it == m.end()) {
            m[p] = val;
        } else {
            it->second = std::max(it->second, val);
        }
    }
    return m;
}

bool is_path_safe(
    const std::vector<Point>& path,
    const std::unordered_set<Point, PointHash>& obstacles,
    const std::deque<Point>& snake,
    std::unordered_set<Point, PointHash>& visited_reused
) {
    int K = path.size() - 1; // path[0] is start, path[K] is food
    int L = snake.size();
    int new_L = L + 1; // since we eat food
    
    // Compute total free cells in the grid (capped to avoid overflow)
    long long total_grid_cells = 1;
    for (int i = 0; i < D; ++i) {
        total_grid_cells *= bounds[i];
        if (total_grid_cells > 1000000) {
            total_grid_cells = 1000000;
            break;
        }
    }
    int free_cells = total_grid_cells - obstacles.size();
    int target_count = std::min(new_L, free_cells);

    // Build the set of occupied positions at the end of the path (step K)
    visited_reused.clear();
    
    // Add obstacles
    for (const auto& o : obstacles) {
        visited_reused.insert(o);
    }
    
    // Add path segments
    for (int i = 0; i < new_L; ++i) {
        int idx = K - i;
        if (idx >= 0) {
            visited_reused.insert(path[idx]);
        } else {
            int snake_idx = -idx - 1;
            if (snake_idx < L) {
                visited_reused.insert(snake[snake_idx]);
            }
        }
    }
    
    // Copy occupied set to a local set and clear visited_reused for BFS
    std::unordered_set<Point, PointHash> occupied = std::move(visited_reused);
    visited_reused.clear();
    
    // BFS from path[K] (the new head)
    std::queue<Point> q;
    q.push(path[K]);
    visited_reused.insert(path[K]);
    
    int count = 0;
    while (!q.empty()) {
        Point curr = q.front();
        q.pop();
        count++;
        
        if (count >= target_count) {
            return true;
        }
        
        for (int axis = 0; axis < D; ++axis) {
            for (int dir : {-1, 1}) {
                Point neighbor = curr;
                neighbor.coords[axis] += dir;
                
                if (neighbor.coords[axis] < 0 || neighbor.coords[axis] >= bounds[axis]) {
                    continue;
                }
                if (occupied.count(neighbor)) {
                    continue;
                }
                if (visited_reused.count(neighbor)) {
                    continue;
                }
                
                visited_reused.insert(neighbor);
                q.push(neighbor);
            }
        }
    }
    
    return count >= target_count;
}

std::vector<Point> find_path(
    const Point& start,
    const std::vector<Point>& food_list,
    const std::unordered_set<Point, PointHash>& food_set,
    const std::unordered_set<Point, PointHash>& obstacles,
    const std::deque<Point>& snake,
    const std::unordered_map<Point, int, PointHash>& snake_occupied_until
) {
    if (food_set.empty()) return {};

    using PQElement = std::pair<int, int>; // {f, index}
    std::priority_queue<PQElement, std::vector<PQElement>, std::greater<PQElement>> pq;
    
    std::vector<SearchNode> nodes;
    nodes.reserve(10000);
    
    std::unordered_map<Point, int, PointHash> visited;
    visited.reserve(10000);
    
    auto get_h = [&](const Point& p) {
        int min_d = 1e9;
        for (const auto& f : food_list) {
            min_d = std::min(min_d, manhattan_distance(p, f));
        }
        return min_d;
    };
    
    int start_h = get_h(start);
    nodes.push_back({start, 0, start_h, -1});
    pq.push({start_h, 0});
    visited[start] = 0;
    
    int target_node_idx = -1;
    const int max_expanded = 15000;
    int expanded_count = 0;
    
    std::unordered_set<Point, PointHash> visited_reused;
    visited_reused.reserve(1000);
    int safety_checks_count = 0;
    
    while (!pq.empty()) {
        auto [curr_f, curr_idx] = pq.top();
        pq.pop();
        
        SearchNode curr_node = nodes[curr_idx];
        
        if (food_set.count(curr_node.pos)) {
            // Reconstruct path to check safety
            std::vector<Point> temp_path;
            int temp_curr = curr_idx;
            while (temp_curr != -1) {
                temp_path.push_back(nodes[temp_curr].pos);
                temp_curr = nodes[temp_curr].parent_idx;
            }
            std::reverse(temp_path.begin(), temp_path.end());
            
            safety_checks_count++;
            if (safety_checks_count > 10 || is_path_safe(temp_path, obstacles, snake, visited_reused)) {
                target_node_idx = curr_idx;
                break;
            }
        }
        
        if (curr_node.g > visited[curr_node.pos]) {
            continue;
        }
        
        if (++expanded_count > max_expanded) {
            break;
        }
        
        for (int axis = 0; axis < D; ++axis) {
            for (int dir : {-1, 1}) {
                Point neighbor = curr_node.pos;
                neighbor.coords[axis] += dir;
                
                if (neighbor.coords[axis] < 0 || neighbor.coords[axis] >= bounds[axis]) {
                    continue;
                }
                
                if (obstacles.count(neighbor)) {
                    continue;
                }
                
                int next_g = curr_node.g + 1;
                
                auto it = snake_occupied_until.find(neighbor);
                if (it != snake_occupied_until.end() && it->second >= next_g) {
                    continue;
                }
                
                bool hits_path = false;
                if (visited.count(neighbor)) {
                    int parent_check = curr_idx;
                    int steps_back = 1;
                    int L = snake.size();
                    while (parent_check != -1 && steps_back < L) {
                        if (nodes[parent_check].pos == neighbor) {
                            hits_path = true;
                            break;
                        }
                        parent_check = nodes[parent_check].parent_idx;
                        steps_back++;
                    }
                }
                if (hits_path) {
                    continue;
                }
                
                auto vit = visited.find(neighbor);
                if (vit == visited.end() || next_g < vit->second) {
                    visited[neighbor] = next_g;
                    int h = get_h(neighbor);
                    int f = next_g + h;
                    nodes.push_back({neighbor, next_g, f, curr_idx});
                    pq.push({f, (int)nodes.size() - 1});
                }
            }
        }
    }
    
    std::vector<Point> path;
    if (target_node_idx != -1) {
        int curr = target_node_idx;
        while (curr != -1) {
            path.push_back(nodes[curr].pos);
            curr = nodes[curr].parent_idx;
        }
        std::reverse(path.begin(), path.end());
    }
    return path;
}

int count_reachable_cells(
    const Point& start_bfs,
    const std::unordered_set<Point, PointHash>& obstacles,
    const std::unordered_map<Point, int, PointHash>& snake_occupied_until,
    std::unordered_set<Point, PointHash>& visited_reused
) {
    visited_reused.clear();
    std::queue<std::pair<Point, int>> q;
    
    q.push({start_bfs, 1});
    visited_reused.insert(start_bfs);
    
    int count = 0;
    const int max_bfs_nodes = 500;
    
    while (!q.empty() && count < max_bfs_nodes) {
        auto [curr, depth] = q.front();
        q.pop();
        count++;
        
        for (int axis = 0; axis < D; ++axis) {
            for (int dir : {-1, 1}) {
                Point neighbor = curr;
                neighbor.coords[axis] += dir;
                
                if (neighbor.coords[axis] < 0 || neighbor.coords[axis] >= bounds[axis]) {
                    continue;
                }
                if (obstacles.count(neighbor)) {
                    continue;
                }
                if (visited_reused.count(neighbor)) {
                    continue;
                }
                
                int next_depth = depth + 1;
                auto it = snake_occupied_until.find(neighbor);
                if (it != snake_occupied_until.end() && it->second >= next_depth) {
                    continue;
                }
                
                visited_reused.insert(neighbor);
                q.push({neighbor, next_depth});
            }
        }
    }
    return count;
}

struct SurvivalMove {
    int axis;
    int dir;
    int reachable_count;
    int food_dist;
};

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    int max_turns;
    if (!(std::cin >> D >> max_turns)) return 0;
    
    for (int i = 0; i < D; ++i) {
        std::cin >> bounds[i];
    }
    
    int snake_length;
    std::cin >> snake_length;
    
    Point start_pos;
    for (int i = 0; i < D; ++i) {
        std::cin >> start_pos.coords[i];
    }
    
    std::deque<Point> snake;
    for (int i = 0; i < snake_length; ++i) {
        snake.push_back(start_pos);
    }
    
    int F;
    if (!(std::cin >> F)) return 0;
    
    std::unordered_set<Point, PointHash> food_set;
    food_set.reserve(F);
    std::vector<Point> food_list;
    food_list.reserve(F);
    
    for (int i = 0; i < F; ++i) {
        Point f;
        for (int j = 0; j < D; ++j) {
            std::cin >> f.coords[j];
        }
        food_set.insert(f);
        food_list.push_back(f);
    }
    
    int O;
    if (!(std::cin >> O)) return 0;
    
    std::unordered_set<Point, PointHash> obstacles;
    obstacles.reserve(O);
    for (int i = 0; i < O; ++i) {
        Point o;
        for (int j = 0; j < D; ++j) {
            std::cin >> o.coords[j];
        }
        obstacles.insert(o);
    }
    
    std::vector<Point> active_path;
    int path_idx = 0;
    
    for (int turn = 0; turn < max_turns; ++turn) {
        if (food_set.empty()) {
            break;
        }
        
        auto snake_occupied_until = build_occupied_map(snake);
        
        // Check if we need to plan a new path
        bool need_replan = true;
        if (path_idx + 1 < (int)active_path.size()) {
            Point next_p = active_path[path_idx + 1];
            bool path_valid = true;
            
            for (int i = 0; i < D; ++i) {
                if (next_p.coords[i] < 0 || next_p.coords[i] >= bounds[i]) {
                    path_valid = false;
                    break;
                }
            }
            if (path_valid && obstacles.count(next_p)) {
                path_valid = false;
            }
            if (path_valid) {
                auto it = snake_occupied_until.find(next_p);
                if (it != snake_occupied_until.end() && it->second >= 1) {
                    path_valid = false;
                }
            }
            
            if (path_valid) {
                need_replan = false;
                path_idx++;
            }
        }
        
        if (need_replan) {
            active_path = find_path(snake.front(), food_list, food_set, obstacles, snake, snake_occupied_until);
            path_idx = 0;
        }
        
        int chosen_axis = -1;
        int chosen_dir = 0;
        
        if (path_idx + 1 < (int)active_path.size()) {
            Point p0 = active_path[path_idx];
            Point p1 = active_path[path_idx + 1];
            for (int i = 0; i < D; ++i) {
                if (p0.coords[i] != p1.coords[i]) {
                    chosen_axis = i;
                    chosen_dir = p1.coords[i] - p0.coords[i];
                    break;
                }
            }
        } else {
            std::unordered_set<Point, PointHash> survival_visited;
            survival_visited.reserve(1000);
            std::vector<SurvivalMove> candidates;
            for (int axis = 0; axis < D; ++axis) {
                for (int dir : {-1, 1}) {
                    Point neighbor = snake.front();
                    neighbor.coords[axis] += dir;
                    
                    if (neighbor.coords[axis] < 0 || neighbor.coords[axis] >= bounds[axis]) {
                        continue;
                    }
                    if (obstacles.count(neighbor)) {
                        continue;
                    }
                    auto it = snake_occupied_until.find(neighbor);
                    if (it != snake_occupied_until.end() && it->second >= 1) {
                        continue;
                    }
                    
                    int reach = count_reachable_cells(neighbor, obstacles, snake_occupied_until, survival_visited);
                    int f_dist = 1e9;
                    for (const auto& f : food_list) {
                        f_dist = std::min(f_dist, manhattan_distance(neighbor, f));
                    }
                    candidates.push_back({axis, dir, reach, f_dist});
                }
            }
            
            if (candidates.empty()) {
                for (int axis = 0; axis < D; ++axis) {
                    Point neighbor = snake.front();
                    if (neighbor.coords[axis] + 1 < bounds[axis]) {
                        chosen_axis = axis;
                        chosen_dir = 1;
                        break;
                    } else if (neighbor.coords[axis] - 1 >= 0) {
                        chosen_axis = axis;
                        chosen_dir = -1;
                        break;
                    }
                }
                if (chosen_axis == -1) {
                    chosen_axis = 0;
                    chosen_dir = 1;
                }
            } else {
                std::sort(candidates.begin(), candidates.end(), [](const SurvivalMove& a, const SurvivalMove& b) {
                    if (a.reachable_count != b.reachable_count) {
                        return a.reachable_count > b.reachable_count;
                      }
                    return a.food_dist < b.food_dist;
                });
                chosen_axis = candidates[0].axis;
                chosen_dir = candidates[0].dir;
            }
            
            active_path.clear();
            path_idx = 0;
        }
        
        std::cout << chosen_axis << " " << chosen_dir << "\n";
        std::cout.flush();
        
        Point new_head = snake.front();
        new_head.coords[chosen_axis] += chosen_dir;
        
        if (food_set.count(new_head)) {
            food_set.erase(new_head);
            food_list.erase(std::remove(food_list.begin(), food_list.end(), new_head), food_list.end());
            snake.push_front(new_head);
        } else {
            snake.push_front(new_head);
            snake.pop_back();
        }
    }
    
    return 0;
}
