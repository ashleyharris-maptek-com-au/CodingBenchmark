using System;
using System.IO;
using System.Text;
using System.Diagnostics;

class Program
{
    static void Main()
    {
        // Use fast scanner for input reading
        FastScanner scanner = new FastScanner(Console.OpenStandardInput());
        
        int N = scanner.NextInt();
        if (N <= 0 || scanner.EOF)
        {
            Console.WriteLine(0);
            Console.WriteLine();
            return;
        }

        int[][] flow = new int[N][];
        for (int i = 0; i < N; i++)
        {
            flow[i] = new int[N];
            for (int j = 0; j < N; j++)
            {
                flow[i][j] = scanner.NextInt();
            }
        }

        int[][] dist = new int[N][];
        for (int i = 0; i < N; i++)
        {
            dist[i] = new int[N];
            for (int j = 0; j < N; j++)
            {
                dist[i][j] = scanner.NextInt();
            }
        }

        // Compute transposes for cache-friendly column access
        int[][] flowT = new int[N][];
        int[][] distT = new int[N][];
        for (int i = 0; i < N; i++)
        {
            flowT[i] = new int[N];
            distT[i] = new int[N];
        }
        for (int i = 0; i < N; i++)
        {
            for (int j = 0; j < N; j++)
            {
                flowT[i][j] = flow[j][i];
                distT[i][j] = dist[j][i];
            }
        }

        // Determine parameters based on N
        long timeLimitMs = 290000; // default 290 seconds (4.8 minutes)
        int maxNonImprovingRestarts = 15;
        if (N > 1000)
        {
            timeLimitMs = 45000;
            maxNonImprovingRestarts = 4;
        }
        else if (N > 500)
        {
            timeLimitMs = 60000;
            maxNonImprovingRestarts = 6;
        }
        else if (N > 200)
        {
            timeLimitMs = 30000;
            maxNonImprovingRestarts = 10;
        }
        else if (N > 100)
        {
            timeLimitMs = 15000;
            maxNonImprovingRestarts = 15;
        }
        else
        {
            timeLimitMs = 5000;
            maxNonImprovingRestarts = 20;
        }

        Stopwatch stopwatch = Stopwatch.StartNew();

        // 1. Greedy Initialization
        int[] p = GreedyInit(N, flow, dist);
        long cost = CalculateTotalCost(p, flow, dist);

        int[] bestPerm = (int[])p.Clone();
        long bestCost = cost;

        int[] I = new int[N];
        for (int i = 0; i < N; i++) I[i] = i;

        Random rand = new Random(42);
        long maxChecks = Math.Min((long)N * (N - 1) / 2, 100000L);

        // 2. Initial Local Search on greedy permutation
        LocalSearch(p, ref cost, I, rand, stopwatch, timeLimitMs, flow, flowT, dist, distT, maxChecks);
        if (cost < bestCost)
        {
            bestCost = cost;
            Array.Copy(p, bestPerm, N);
        }

        int basePert = Math.Max(4, N / 100);
        int pertStrength = basePert;
        int maxPert = Math.Max(20, N / 10);
        int nonImprovingRestarts = 0;

        // 3. Iterated Local Search Loop
        while (stopwatch.ElapsedMilliseconds < timeLimitMs && nonImprovingRestarts < maxNonImprovingRestarts)
        {
            int[] currPerm = (int[])bestPerm.Clone();
            long currCost = bestCost;

            // Perturb
            Perturb(currPerm, ref currCost, pertStrength, rand, flow, flowT, dist, distT);

            // Local Search
            LocalSearch(currPerm, ref currCost, I, rand, stopwatch, timeLimitMs, flow, flowT, dist, distT, maxChecks);

            if (currCost < bestCost)
            {
                bestCost = currCost;
                Array.Copy(currPerm, bestPerm, N);
                pertStrength = basePert;
                nonImprovingRestarts = 0; // reset
            }
            else
            {
                pertStrength++;
                if (pertStrength > maxPert)
                {
                    // Random restart
                    for (int i = 0; i < N; i++) currPerm[i] = i;
                    // Shuffle
                    for (int i = 0; i < N; i++)
                    {
                        int r = rand.Next(i, N);
                        int temp = currPerm[i];
                        currPerm[i] = currPerm[r];
                        currPerm[r] = temp;
                    }
                    currCost = CalculateTotalCost(currPerm, flow, dist);
                    LocalSearch(currPerm, ref currCost, I, rand, stopwatch, timeLimitMs, flow, flowT, dist, distT, maxChecks);

                    if (currCost < bestCost)
                    {
                        bestCost = currCost;
                        Array.Copy(currPerm, bestPerm, N);
                        nonImprovingRestarts = 0;
                    }
                    else
                    {
                        nonImprovingRestarts++;
                    }
                    pertStrength = basePert;
                }
            }
        }

        // Print final result
        Console.WriteLine(bestCost);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < N; i++)
        {
            sb.Append(bestPerm[i]);
            if (i < N - 1) sb.Append(' ');
        }
        Console.WriteLine(sb.ToString());
    }

    static int[] GreedyInit(int N, int[][] flow, int[][] dist)
    {
        long[] F = new long[N];
        for (int i = 0; i < N; i++)
        {
            for (int j = 0; j < N; j++)
            {
                F[i] += (long)flow[i][j] + flow[j][i];
            }
        }

        long[] D = new long[N];
        for (int i = 0; i < N; i++)
        {
            for (int j = 0; j < N; j++)
            {
                D[i] += (long)dist[i][j] + dist[j][i];
            }
        }

        int[] facIndices = new int[N];
        int[] locIndices = new int[N];
        for (int i = 0; i < N; i++)
        {
            facIndices[i] = i;
            locIndices[i] = i;
        }

        Array.Sort(facIndices, (a, b) => F[b].CompareTo(F[a]));
        Array.Sort(locIndices, (a, b) => D[a].CompareTo(D[b]));

        int[] p = new int[N];
        for (int k = 0; k < N; k++)
        {
            p[facIndices[k]] = locIndices[k];
        }
        return p;
    }

    static long CalculateTotalCost(int[] p, int[][] flow, int[][] dist)
    {
        int N = p.Length;
        long total = 0;
        for (int i = 0; i < N; i++)
        {
            int pi = p[i];
            int[] flowI = flow[i];
            for (int j = 0; j < N; j++)
            {
                total += (long)flowI[j] * dist[pi][p[j]];
            }
        }
        return total;
    }

    static long CalculateDelta(int u, int v, int[] p, int[][] flow, int[][] flowT, int[][] dist, int[][] distT)
    {
        int N = p.Length;
        int[] flowU = flow[u];
        int[] flowV = flow[v];
        int[] flowTu = flowT[u];
        int[] flowTv = flowT[v];

        int pu = p[u];
        int pv = p[v];
        int[] distPu = dist[pu];
        int[] distPv = dist[pv];
        int[] distTpu = distT[pu];
        int[] distTpv = distT[pv];

        long delta = 0;
        delta += (long)flowU[u] * ((long)distPv[pv] - distPu[pu]);
        delta += (long)flowV[v] * ((long)distPu[pu] - distPv[pv]);
        delta += ((long)flowU[v] - flowV[u]) * ((long)distPv[pu] - distPu[pv]);

        for (int i = 0; i < N; i++)
        {
            if (i == u || i == v) continue;
            int pi = p[i];
            delta += ((long)flowU[i] - flowV[i]) * ((long)distPv[pi] - distPu[pi]);
            delta += ((long)flowTu[i] - flowTv[i]) * ((long)distTpv[pi] - distTpu[pi]);
        }
        return delta;
    }

    static void LocalSearch(int[] p, ref long cost, int[] I, Random rand, Stopwatch stopwatch, long timeLimitMs, int[][] flow, int[][] flowT, int[][] dist, int[][] distT, long maxChecks)
    {
        int N = p.Length;
        bool improved = true;
        while (improved)
        {
            improved = false;
            if (stopwatch.ElapsedMilliseconds > timeLimitMs)
            {
                break;
            }

            // Shuffle I
            for (int i = 0; i < N; i++)
            {
                int r = rand.Next(i, N);
                int temp = I[i];
                I[i] = I[r];
                I[r] = temp;
            }

            long checks = 0;
            for (int i = 0; i < N; i++)
            {
                int u = I[i];
                for (int j = i + 1; j < N; j++)
                {
                    int v = I[j];
                    checks++;
                    if (checks > maxChecks)
                    {
                        break;
                    }

                    long delta = CalculateDelta(u, v, p, flow, flowT, dist, distT);
                    if (delta < 0)
                    {
                        int temp = p[u];
                        p[u] = p[v];
                        p[v] = temp;
                        cost += delta;
                        improved = true;
                        break;
                    }
                }
                if (improved || checks > maxChecks) break;
            }
        }
    }

    static void Perturb(int[] p, ref long cost, int numSwaps, Random rand, int[][] flow, int[][] flowT, int[][] dist, int[][] distT)
    {
        int N = p.Length;
        for (int step = 0; step < numSwaps; step++)
        {
            int u = rand.Next(N);
            int v = rand.Next(N);
            while (u == v)
            {
                v = rand.Next(N);
            }
            long delta = CalculateDelta(u, v, p, flow, flowT, dist, distT);
            int temp = p[u];
            p[u] = p[v];
            p[v] = temp;
            cost += delta;
        }
    }
}

class FastScanner
{
    private readonly Stream stream;
    private readonly byte[] buffer = new byte[65536];
    private int bufferLen = 0;
    private int bufferPtr = 0;
    public bool EOF { get; private set; } = false;

    public FastScanner(Stream stream)
    {
        this.stream = stream;
    }

    private byte Read()
    {
        if (EOF) return 0;
        if (bufferPtr >= bufferLen)
        {
            bufferLen = stream.Read(buffer, 0, buffer.Length);
            bufferPtr = 0;
            if (bufferLen <= 0)
            {
                EOF = true;
                return 0;
            }
        }
        return buffer[bufferPtr++];
    }

    public int NextInt()
    {
        int c = Read();
        while (c <= 32)
        {
            if (EOF) return 0;
            c = Read();
        }
        bool neg = false;
        if (c == '-')
        {
            neg = true;
            c = Read();
        }
        int res = 0;
        while (c > 32)
        {
            res = res * 10 + c - '0';
            c = Read();
        }
        return neg ? -res : res;
    }
}
