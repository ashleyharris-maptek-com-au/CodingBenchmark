// Global constants
const double M_PI = 3.14159265358979323846;

// Physics and map parameters
int width = 0, depth = 0, height = 0;
double gravity = 0.0, max_thrust = 0.0, fuel = 0.0;
double start_x = 0.0, start_y = 0.0, start_z = 0.0;
double target_x = 0.0, target_y = 0.0, target_z = 0.0;
std::string terrain_file_path;

// Memory mapping variables
HANDLE hFile = INVALID_HANDLE_VALUE;
HANDLE hMapping = NULL;
void* pData = nullptr;

struct TerrainCell {
    uint16_t a, b, c;
};

TerrainCell get_cell(int x, int y) {
    if (x < 0 || x >= width || y < 0 || y >= depth) {
        return {0, 0, 0};
    }
    if (!pData) return {0, 0, 0};
    uint16_t* ptr = (uint16_t*)pData + 3 * ((size_t)x * depth + y);
    return {ptr[0], ptr[1], ptr[2]};
}

double get_solid_height_below(double x, double y, double z) {
    int ix = std::clamp((int)std::floor(x), 0, width - 1);
    int iy = std::clamp((int)std::floor(y), 0, depth - 1);
    TerrainCell cell = get_cell(ix, iy);
    if (z >= cell.c) {
        if (cell.b < cell.c) {
            return (double)cell.c;
        } else {
            return (double)cell.a;
        }
    } else {
        return (double)cell.a;
    }
}

double get_solid_height_above(double x, double y, double z) {
    int ix = std::clamp((int)std::floor(x), 0, width - 1);
    int iy = std::clamp((int)std::floor(y), 0, depth - 1);
    TerrainCell cell = get_cell(ix, iy);
    if (z < cell.a) {
        return (double)cell.a;
    }
    if (cell.b < cell.c) {
        if (z < cell.b) {
            return (double)cell.b;
        }
    }
    return (double)height;
}

bool is_point_collision(double cx, double cy, double cz) {
    if (cx < 0.0 || cx >= width || cy < 0.0 || cy >= depth || cz < 0.0 || cz >= height) {
        return true;
    }
    int ix = (int)cx;
    int iy = (int)cy;
    TerrainCell cell = get_cell(ix, iy);
    if (cz < cell.a) return true;
    if (cell.b < cell.c && cz >= cell.b && cz < cell.c) return true;
    return false;
}

bool is_collision_free(double x1, double y1, double z1, double x2, double y2, double z2, double step_size = 2.0) {
    double dx = x2 - x1;
    double dy = y2 - y1;
    double dz = z2 - z1;
    double len = std::sqrt(dx*dx + dy*dy + dz*dz);
    if (len < 1e-5) return !is_point_collision(x1, y1, z1);
    
    int num_steps = (int)std::ceil(len / step_size);
    for (int i = 0; i <= num_steps; i++) {
        double t = (double)i / num_steps;
        double cx = x1 + t * dx;
        double cy = y1 + t * dy;
        double cz = z1 + t * dz;
        if (is_point_collision(cx, cy, cz)) {
            return false;
        }
    }
    return true;
}

struct Point3D {
    double x, y, z;
    double distance(const Point3D& o) const {
        return std::sqrt((x-o.x)*(x-o.x) + (y-o.y)*(y-o.y) + (z-o.z)*(z-o.z));
    }
};

double S = 10.0;     // horizontal grid cell size
double S_z = 10.0;   // vertical grid cell size

struct GridCoord {
    int x, y, z;
    bool operator==(const GridCoord& o) const {
        return x == o.x && y == o.y && z == o.z;
    }
};

struct GridCoordHash {
    size_t operator()(const GridCoord& c) const {
        return std::hash<int>{}(c.x) ^ (std::hash<int>{}(c.y) << 1) ^ (std::hash<int>{}(c.z) << 2);
    }
};

struct AStarNode {
    GridCoord coord;
    double g_score;
    double f_score;
    
    bool operator>(const AStarNode& o) const {
        return f_score > o.f_score;
    }
};

void init_grid_sizes() {
    S = std::max(5.0, std::max((double)width, (double)depth) / 250.0);
    S_z = std::max(5.0, (double)height / 50.0);
}

int get_gx(double x) { return (int)std::floor(x / S); }
int get_gy(double y) { return (int)std::floor(y / S); }
int get_gz(double z) { return (int)std::floor(z / S_z); }

Point3D get_pos(const GridCoord& c, const GridCoord& start_gc, const GridCoord& target_gc) {
    if (c == start_gc) {
        return {start_x, start_y, start_z};
    }
    if (c == target_gc) {
        return {target_x, target_y, target_z};
    }
    return {(c.x + 0.5) * S, (c.y + 0.5) * S, (c.z + 0.5) * S_z};
}

bool try_elevated_path(double sx, double sy, double sz, double tx, double ty, double tz, std::vector<Point3D>& path, double step_size) {
    double min_z = std::max(sz, tz) + 10.0;
    double max_z = (double)height - 10.0;
    if (min_z >= max_z) return false;
    
    double step_z = std::max(5.0, (max_z - min_z) / 40.0);
    for (double z_safe = min_z; z_safe <= max_z; z_safe += step_z) {
        if (!is_collision_free(sx, sy, sz, sx, sy, z_safe, step_size)) continue;
        if (!is_collision_free(tx, ty, z_safe, tx, ty, tz, step_size)) continue;
        if (!is_collision_free(sx, sy, z_safe, tx, ty, z_safe, step_size)) continue;
        
        path = { {sx, sy, sz}, {sx, sy, z_safe}, {tx, ty, z_safe}, {tx, ty, tz} };
        return true;
    }
    return false;
}

std::vector<Point3D> run_astar() {
    init_grid_sizes();
    double step_size = std::max(2.0, S / 80.0);
    
    GridCoord start_gc = { get_gx(start_x), get_gy(start_y), get_gz(start_z) };
    GridCoord target_gc = { get_gx(target_x), get_gy(target_y), get_gz(target_z) };
    
    Point3D start_pos = {start_x, start_y, start_z};
    Point3D target_pos = {target_x, target_y, target_z};
    
    // Quick straight line check
    if (is_collision_free(start_x, start_y, start_z, target_x, target_y, target_z, step_size)) {
        return { start_pos, target_pos };
    }
    
    // Quick elevated path check
    std::vector<Point3D> elevated_path;
    if (try_elevated_path(start_x, start_y, start_z, target_x, target_y, target_z, elevated_path, step_size)) {
        return elevated_path;
    }
    
    // Fallback to A*
    std::priority_queue<AStarNode, std::vector<AStarNode>, std::greater<AStarNode>> open_set;
    std::unordered_map<GridCoord, GridCoord, GridCoordHash> came_from;
    std::unordered_map<GridCoord, double, GridCoordHash> g_score;
    
    g_score[start_gc] = 0.0;
    open_set.push({start_gc, 0.0, start_pos.distance(target_pos)});
    
    int grid_w = (int)std::ceil(width / S);
    int grid_d = (int)std::ceil(depth / S);
    int grid_h = (int)std::ceil(height / S_z);
    
    int iterations = 0;
    const int MAX_ITERATIONS = 12000;
    
    GridCoord best_coord = start_gc;
    double best_dist = start_pos.distance(target_pos);
    
    while (!open_set.empty()) {
        iterations++;
        if (iterations > MAX_ITERATIONS) {
            break;
        }
        
        AStarNode current = open_set.top();
        open_set.pop();
        
        GridCoord curr_gc = current.coord;
        Point3D curr_pos = get_pos(curr_gc, start_gc, target_gc);
        
        if (curr_gc == target_gc) {
            best_coord = curr_gc;
            break;
        }
        
        double d_to_target = curr_pos.distance(target_pos);
        if (d_to_target < best_dist) {
            best_dist = d_to_target;
            best_coord = curr_gc;
        }
        
        // 26-neighbors
        for (int dx = -1; dx <= 1; dx++) {
            for (int dy = -1; dy <= 1; dy++) {
                for (int dz = -1; dz <= 1; dz++) {
                    if (dx == 0 && dy == 0 && dz == 0) continue;
                    
                    GridCoord neighbor_gc = { curr_gc.x + dx, curr_gc.y + dy, curr_gc.z + dz };
                    
                    if (neighbor_gc.x < 0 || neighbor_gc.x >= grid_w ||
                        neighbor_gc.y < 0 || neighbor_gc.y >= grid_d ||
                        neighbor_gc.z < 0 || neighbor_gc.z >= grid_h) {
                        continue;
                    }
                    
                    Point3D neighbor_pos = get_pos(neighbor_gc, start_gc, target_gc);
                    
                    if (!is_collision_free(curr_pos.x, curr_pos.y, curr_pos.z,
                                           neighbor_pos.x, neighbor_pos.y, neighbor_pos.z, step_size)) {
                        continue;
                    }
                    
                    double edge_weight = curr_pos.distance(neighbor_pos);
                    double tentative_g = g_score[curr_gc] + edge_weight;
                    
                    if (g_score.find(neighbor_gc) == g_score.end() || tentative_g < g_score[neighbor_gc]) {
                        came_from[neighbor_gc] = curr_gc;
                        g_score[neighbor_gc] = tentative_g;
                        double f = tentative_g + neighbor_pos.distance(target_pos);
                        open_set.push({neighbor_gc, tentative_g, f});
                    }
                }
            }
        }
    }
    
    std::vector<Point3D> path;
    GridCoord curr = best_coord;
    while (true) {
        path.push_back(get_pos(curr, start_gc, target_gc));
        if (curr == start_gc) break;
        if (came_from.find(curr) == came_from.end()) break;
        curr = came_from[curr];
    }
    std::reverse(path.begin(), path.end());
    
    if (!(best_coord == target_gc)) {
        path.push_back(target_pos);
    }
    
    return path;
}

std::vector<Point3D> simplify_path(const std::vector<Point3D>& path, double step_size) {
    if (path.size() <= 2) return path;
    std::vector<Point3D> simplified;
    simplified.push_back(path.front());
    
    size_t curr = 0;
    while (curr < path.size() - 1) {
        size_t next = curr + 1;
        for (size_t i = path.size(); i > curr + 1; i--) {
            size_t idx = i - 1;
            if (is_collision_free(path[curr].x, path[curr].y, path[curr].z,
                                  path[idx].x, path[idx].y, path[idx].z, step_size)) {
                next = idx;
                break;
            }
        }
        simplified.push_back(path[next]);
        curr = next;
    }
    return simplified;
}

std::vector<Point3D> waypoints;
std::mutex waypoint_mutex;
std::atomic<bool> path_ready{false};

void pathfinder_thread_func() {
    std::vector<Point3D> raw_path = run_astar();
    double step_size = std::max(2.0, S / 80.0);
    std::vector<Point3D> smooth_path = simplify_path(raw_path, step_size);
    
    {
        std::lock_guard<std::mutex> lock(waypoint_mutex);
        waypoints = smooth_path;
        path_ready = true;
    }
    
    // Print path to stderr
    std::cerr << "Planned path with " << smooth_path.size() << " waypoints:" << std::endl;
    for (size_t i = 0; i < smooth_path.size(); i++) {
        std::cerr << "  W" << i << ": (" << smooth_path[i].x << ", " << smooth_path[i].y << ", " << smooth_path[i].z << ")" << std::endl;
    }
}

bool map_terrain_file() {
    std::filesystem::path p(terrain_file_path);
    if (!std::filesystem::exists(p)) {
        std::cerr << "Terrain file does not exist: " << terrain_file_path << std::endl;
        return false;
    }
    
    hFile = CreateFileW(p.c_str(), GENERIC_READ, FILE_SHARE_READ, NULL, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);
    if (hFile == INVALID_HANDLE_VALUE) {
        std::cerr << "Failed to open terrain file: (Error: " << GetLastError() << ")" << std::endl;
        return false;
    }
    
    hMapping = CreateFileMappingW(hFile, NULL, PAGE_READONLY, 0, 0, NULL);
    if (hMapping == NULL) {
        std::cerr << "Failed to create file mapping: " << GetLastError() << std::endl;
        CloseHandle(hFile);
        return false;
    }
    
    pData = MapViewOfFile(hMapping, FILE_MAP_READ, 0, 0, 0);
    if (pData == nullptr) {
        std::cerr << "Failed to map view of file: " << GetLastError() << std::endl;
        CloseHandle(hMapping);
        CloseHandle(hFile);
        return false;
    }
    
    std::cerr << "Successfully memory-mapped terrain file." << std::endl;
    return true;
}

void unmap_terrain_file() {
    if (pData) {
        UnmapViewOfFile(pData);
        pData = nullptr;
    }
    if (hMapping) {
        CloseHandle(hMapping);
        hMapping = NULL;
    }
    if (hFile != INVALID_HANDLE_VALUE) {
        CloseHandle(hFile);
        hFile = INVALID_HANDLE_VALUE;
    }
}

bool read_header() {
    if (!(std::cin >> width >> depth >> height >> gravity >> max_thrust >> fuel)) {
        return false;
    }
    if (!(std::cin >> start_x >> start_y >> start_z >> target_x >> target_y >> target_z)) {
        return false;
    }
    
    std::string dummy;
    std::getline(std::cin, dummy);
    
    if (!std::getline(std::cin, terrain_file_path)) {
        return false;
    }
    while (!terrain_file_path.empty() && (terrain_file_path.back() == '\r' || terrain_file_path.back() == ' ' || terrain_file_path.back() == '\n')) {
        terrain_file_path.pop_back();
    }
    
    std::string state_line;
    while (std::getline(std::cin, state_line)) {
        while (!state_line.empty() && (state_line.back() == '\r' || state_line.back() == ' ' || state_line.back() == '\n')) {
            state_line.pop_back();
        }
        if (state_line == "STATE") {
            break;
        }
    }
    
    return true;
}

int main() {
    // Fast I/O
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    if (!read_header()) {
        std::cerr << "Failed to read header from stdin." << std::endl;
        return 1;
    }
    
    if (!map_terrain_file()) {
        std::cerr << "Failed to map terrain file." << std::endl;
        return 1;
    }
    
    double x, y, z, vx, vy, vz;
    double pitch, yaw, roll;
    double pitch_rate, yaw_rate, roll_rate;
    double fuel_remaining;
    
    bool first_update = true;
    std::thread path_thread;
    
    std::cout << std::fixed << std::setprecision(6);
    
    size_t current_waypoint_index = 0;
    
    while (std::cin >> x >> y >> z >> vx >> vy >> vz >> pitch >> yaw >> roll >> pitch_rate >> yaw_rate >> roll_rate >> fuel_remaining) {
        if (first_update) {
            first_update = false;
            path_thread = std::thread(pathfinder_thread_func);
        }
        
        std::vector<Point3D> local_waypoints;
        bool ready = path_ready.load();
        if (ready) {
            std::lock_guard<std::mutex> lock(waypoint_mutex);
            local_waypoints = waypoints;
        }
        
        double v_des_x = 0.0;
        double v_des_y = 0.0;
        double v_des_z = 0.0;
        
        Point3D pos = {x, y, z};
        
        if (ready && !local_waypoints.empty()) {
            size_t target_idx = current_waypoint_index;
            for (size_t i = local_waypoints.size(); i > current_waypoint_index; i--) {
                size_t idx = i - 1;
                double step = std::max(2.0, S / 80.0);
                if (is_collision_free(x, y, z, local_waypoints[idx].x, local_waypoints[idx].y, local_waypoints[idx].z, step)) {
                    target_idx = idx;
                    break;
                }
            }
            current_waypoint_index = std::max(current_waypoint_index, target_idx);
            
            double dist_to_wp = pos.distance(local_waypoints[current_waypoint_index]);
            double reach_dist = (current_waypoint_index == local_waypoints.size() - 1) ? 2.0 : std::max(10.0, S * 0.15);
            if (dist_to_wp < reach_dist && current_waypoint_index < local_waypoints.size() - 1) {
                current_waypoint_index++;
            }
            
            Point3D target_wp = local_waypoints[current_waypoint_index];
            Point3D dir = {target_wp.x - x, target_wp.y - y, target_wp.z - z};
            double dist = pos.distance(target_wp);
            
            double landing_radius = std::max(5.0, std::min((double)width, (double)depth) * 0.02);
            double dx = target_x - x;
            double dy = target_y - y;
            double horizontal_dist = std::sqrt(dx*dx + dy*dy);
            
            if (horizontal_dist < 0.8 * landing_radius) {
                double k_pos = 0.5;
                v_des_x = k_pos * (target_x - x);
                v_des_y = k_pos * (target_y - y);
                
                if (z > target_z + 0.2) {
                    v_des_z = -1.2;
                } else if (z < target_z - 0.2) {
                    v_des_z = 1.0;
                } else {
                    v_des_z = -0.5;
                }
                
                double landing_speed_limit = 4.0;
                double speed_horiz = std::sqrt(v_des_x*v_des_x + v_des_y*v_des_y);
                if (speed_horiz > landing_speed_limit) {
                    v_des_x = (v_des_x / speed_horiz) * landing_speed_limit;
                    v_des_y = (v_des_y / speed_horiz) * landing_speed_limit;
                }
            } else {
                double target_speed = 0.0;
                if (current_waypoint_index == local_waypoints.size() - 1) {
                    target_speed = std::min(100.0, std::max(5.0, std::sqrt(2.0 * (0.2 * max_thrust) * dist)));
                } else {
                    target_speed = std::min(80.0, std::max(20.0, S * 0.1));
                }
                
                if (dist > 1e-5) {
                    v_des_x = (dir.x / dist) * target_speed;
                    v_des_y = (dir.y / dist) * target_speed;
                    v_des_z = (dir.z / dist) * target_speed;
                }
            }
        }
        
        double tau = 0.8;
        double ax_des = (v_des_x - vx) / tau;
        double ay_des = (v_des_y - vy) / tau;
        double az_des = (v_des_z - vz) / tau + gravity;
        
        double fx = ax_des;
        double fy = ay_des;
        double fz = az_des;
        
        double solid_below = get_solid_height_below(x, y, z);
        double dist_to_ground = z - solid_below;
        if (vz < -0.5) {
            double a_req = (vz * vz) / (2.0 * std::max(1.0, dist_to_ground - 4.0));
            double fz_min = a_req + gravity;
            fz = std::max(fz, fz_min);
        }
        
        double solid_above = get_solid_height_above(x, y, z);
        double dist_to_ceiling = solid_above - z;
        if (vz > 0.5) {
            double a_req_up = (vz * vz) / (2.0 * std::max(1.0, dist_to_ceiling - 4.0));
            double fz_max = gravity - a_req_up;
            fz = std::min(fz, fz_max);
        }
        
        double f_mag = std::sqrt(fx*fx + fy*fy + fz*fz);
        double ux = 0.0, uy = 0.0, uz = 1.0;
        if (f_mag > 1e-5) {
            ux = fx / f_mag;
            uy = fy / f_mag;
            uz = fz / f_mag;
        }
        
        double u_horiz = std::sqrt(ux*ux + uy*uy);
        u_horiz = std::clamp(u_horiz, 0.0, 1.0);
        
        double pitch_target = 0.0;
        double yaw_target = yaw;
        
        if (u_horiz > 1e-5) {
            double max_pitch = 0.5;
            if (max_thrust > 0.0) {
                double ratio = (1.05 * gravity) / max_thrust;
                if (ratio < 1.0) {
                    max_pitch = std::min(0.5, std::acos(ratio));
                } else {
                    max_pitch = 0.05;
                }
            }
            pitch_target = std::min(max_pitch, std::asin(u_horiz));
            yaw_target = std::atan2(-uy, -ux);
        }
        
        double thrust_cmd = f_mag / max_thrust;
        double cos_pitch = std::cos(pitch);
        if (cos_pitch > 0.1) {
            double thrust_for_z = fz / (max_thrust * cos_pitch);
            thrust_cmd = std::max(thrust_cmd, thrust_for_z);
        }
        thrust_cmd = std::clamp(thrust_cmd, 0.0, 1.0);
        
        double kp_ang = 6.0;
        double kd_ang = 1.5;
        
        double pitch_error = pitch_target - pitch;
        double desired_pitch_rate = kp_ang * pitch_error - kd_ang * pitch_rate;
        double pitch_cmd = (desired_pitch_rate / 0.95 - pitch_rate) / 0.2;
        pitch_cmd = std::clamp(pitch_cmd, -1.0, 1.0);
        
        double yaw_error = yaw_target - yaw;
        while (yaw_error > M_PI) yaw_error -= 2.0 * M_PI;
        while (yaw_error < -M_PI) yaw_error += 2.0 * M_PI;
        
        double desired_yaw_rate = kp_ang * yaw_error - kd_ang * yaw_rate;
        double yaw_cmd = (desired_yaw_rate / 0.95 - yaw_rate) / 0.2;
        yaw_cmd = std::clamp(yaw_cmd, -1.0, 1.0);
        
        double roll_error = 0.0 - roll;
        double desired_roll_rate = kp_ang * roll_error - kd_ang * roll_rate;
        double roll_cmd = (desired_roll_rate / 0.95 - roll_rate) / 0.2;
        roll_cmd = std::clamp(roll_cmd, -1.0, 1.0);
        
        std::cout << thrust_cmd << " " << pitch_cmd << " " << yaw_cmd << " " << roll_cmd << "\n";
        std::cout.flush();
    }
    
    if (path_thread.joinable()) {
        path_thread.join();
    }
    
    unmap_terrain_file();
    return 0;
}
