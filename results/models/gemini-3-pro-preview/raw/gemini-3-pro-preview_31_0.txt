use std::collections::HashMap;
use std::io::{self, Read, Write, BufWriter};

// -------------------------------------------------------------
// Fast I/O Reader
// -------------------------------------------------------------
struct FastReader {
    buffer: Vec<u8>,
    pos: usize,
}

impl FastReader {
    fn new() -> Self {
        let mut buffer = Vec::new();
        io::stdin().read_to_end(&mut buffer).expect("Failed to read stdin");
        FastReader { buffer, pos: 0 }
    }

    #[inline(always)]
    fn next_token(&mut self) -> Option<&[u8]> {
        while self.pos < self.buffer.len() && self.buffer[self.pos] <= b' ' {
            self.pos += 1;
        }
        if self.pos >= self.buffer.len() {
            return None;
        }
        let start = self.pos;
        while self.pos < self.buffer.len() && self.buffer[self.pos] > b' ' {
            self.pos += 1;
        }
        Some(&self.buffer[start..self.pos])
    }

    #[inline(always)]
    fn next_i64(&mut self) -> Option<i64> {
        let token = self.next_token()?;
        let mut val = 0i64;
        let mut neg = false;
        let mut start = 0;
        if token[0] == b'-' {
            neg = true;
            start = 1;
        } else if token[0] == b'+' {
            start = 1;
        }
        for &b in &token[start..] {
            val = val * 10 + (b - b'0') as i64;
        }
        if neg {
            val = -val;
        }
        Some(val)
    }

    #[inline(always)]
    fn next_u64(&mut self) -> Option<u64> {
        let token = self.next_token()?;
        let mut val = 0u64;
        let mut start = 0;
        if token[0] == b'+' {
            start = 1;
        }
        for &b in &token[start..] {
            val = val * 10 + (b - b'0') as u64;
        }
        Some(val)
    }
}

// -------------------------------------------------------------
// Simple Xorshift RNG
// -------------------------------------------------------------
struct SimpleRng {
    state: u64,
}

impl SimpleRng {
    fn new(seed: u64) -> Self {
        let s = if seed == 0 { 1 } else { seed };
        SimpleRng { state: s }
    }

    fn next_u64(&mut self) -> u64 {
        let mut x = self.state;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.state = x;
        x
    }

    fn shuffle<T>(&mut self, slice: &mut [T]) {
        let n = slice.len();
        if n <= 1 {
            return;
        }
        for i in (1..n).rev() {
            let j = (self.next_u64() % (i as u64 + 1)) as usize;
            slice.swap(i, j);
        }
    }
}

// -------------------------------------------------------------
// Custom BitSet
// -------------------------------------------------------------
struct BitSet {
    data: Vec<u64>,
}

impl BitSet {
    fn new(size_bits: usize) -> Self {
        let num_words = (size_bits + 63) / 64;
        BitSet {
            data: vec![0; num_words],
        }
    }

    #[inline(always)]
    fn set(&mut self, bit: usize) {
        let word = bit / 64;
        let pos = bit % 64;
        self.data[word] |= 1 << pos;
    }

    #[inline(always)]
    fn get(&self, bit: usize) -> bool {
        let word = bit / 64;
        let pos = bit % 64;
        (self.data[word] & (1 << pos)) != 0
    }

    #[inline(always)]
    fn shift_left_or(&mut self, shift: usize, active_words: usize) {
        let word_shift = shift / 64;
        let bit_shift = shift % 64;
        let len = self.data.len();
        let limit = active_words.min(len);
        
        if bit_shift == 0 {
            for i in (word_shift..limit).rev() {
                self.data[i] |= self.data[i - word_shift];
            }
        } else {
            let carry_shift = 64 - bit_shift;
            for i in (word_shift..limit).rev() {
                let val = self.data[i - word_shift] << bit_shift;
                let carry = if i > word_shift {
                    self.data[i - word_shift - 1] >> carry_shift
                } else {
                    0
                };
                self.data[i] |= val | carry;
            }
        }
    }

    #[inline(always)]
    fn shift_right_or(&mut self, shift: usize, active_words: usize) {
        let word_shift = shift / 64;
        let bit_shift = shift % 64;
        let len = self.data.len();
        let limit = active_words.min(len);
        
        if bit_shift == 0 {
            for i in 0..(limit.saturating_sub(word_shift)) {
                self.data[i] |= self.data[i + word_shift];
            }
        } else {
            let carry_shift = 64 - bit_shift;
            for i in 0..(limit.saturating_sub(word_shift)) {
                let val = self.data[i + word_shift] >> bit_shift;
                let carry = if i + word_shift + 1 < limit {
                    self.data[i + word_shift + 1] << carry_shift
                } else {
                    0
                };
                self.data[i] |= val | carry;
            }
        }
    }
}

// -------------------------------------------------------------
// Binary Decomposition for Grouping Duplicates
// -------------------------------------------------------------
#[derive(Clone)]
struct DecomposedElement {
    val: i64,
    indices: Vec<usize>,
}

fn decompose_elements(elements: &[(i64, usize)]) -> Vec<DecomposedElement> {
    let mut groups: HashMap<i64, Vec<usize>> = HashMap::new();
    for &(val, idx) in elements {
        groups.entry(val).or_default().push(idx);
    }
    
    let mut decomposed = Vec::new();
    for (&val, indices) in &groups {
        let n = indices.len();
        let mut remaining = n;
        let mut power = 1;
        let mut idx_offset = 0;
        while remaining > 0 {
            let count = if remaining >= power {
                power
            } else {
                remaining
            };
            let chunk_indices = indices[idx_offset..idx_offset + count].to_vec();
            decomposed.push(DecomposedElement {
                val: val * count as i64,
                indices: chunk_indices,
            });
            remaining -= count;
            idx_offset += count;
            power *= 2;
        }
    }
    decomposed
}

// -------------------------------------------------------------
// Meet-in-the-Middle Solver
// -------------------------------------------------------------
fn meet_in_the_middle(elements: &[(i64, usize)], target: i64) -> Option<Vec<usize>> {
    let n = elements.len();
    if n == 0 {
        if target == 0 {
            return Some(vec![]);
        } else {
            return None;
        }
    }
    
    let mid = n / 2;
    let (left, right) = elements.split_at(mid);
    
    let mut left_sums = Vec::with_capacity(1 << left.len());
    generate_sums(left, 0, 0, 0, &mut left_sums);
    left_sums.sort_unstable_by_key(|&(sum, _)| sum);
    
    search_right(right, 0, 0, 0, target, &left_sums, left)
}

fn generate_sums(
    sub_elements: &[(i64, usize)],
    idx: usize,
    current_sum: i64,
    current_mask: u64,
    out: &mut Vec<(i64, u64)>,
) {
    if idx == sub_elements.len() {
        out.push((current_sum, current_mask));
        return;
    }
    generate_sums(sub_elements, idx + 1, current_sum, current_mask, out);
    generate_sums(
        sub_elements,
        idx + 1,
        current_sum + sub_elements[idx].0,
        current_mask | (1 << idx),
        out,
    );
}

fn search_right(
    right: &[(i64, usize)],
    idx: usize,
    current_sum: i64,
    current_mask: u64,
    target: i64,
    left_sums: &[(i64, u64)],
    left: &[(i64, usize)],
) -> Option<Vec<usize>> {
    if idx == right.len() {
        let needed = target - current_sum;
        if let Ok(l_idx) = left_sums.binary_search_by_key(&needed, |&(sum, _)| sum) {
            let mut result = Vec::new();
            let l_mask = left_sums[l_idx].1;
            for i in 0..left.len() {
                if (l_mask & (1 << i)) != 0 {
                    result.push(left[i].1);
                }
            }
            for i in 0..right.len() {
                if (current_mask & (1 << i)) != 0 {
                    result.push(right[i].1);
                }
            }
            return Some(result);
        }
        return None;
    }
    
    if let Some(res) = search_right(right, idx + 1, current_sum, current_mask, target, left_sums, left) {
        return Some(res);
    }
    if let Some(res) = search_right(
        right,
        idx + 1,
        current_sum + right[idx].0,
        current_mask | (1 << idx),
        target,
        left_sums,
        left,
    ) {
        return Some(res);
    }
    
    None
}

// -------------------------------------------------------------
// Backtracking Solver
// -------------------------------------------------------------
fn solve_backtracking(elements: &[(i64, usize)], target: i64, max_nodes: usize) -> Option<Vec<usize>> {
    let n = elements.len();
    
    let mut sorted_elements = elements.to_vec();
    sorted_elements.sort_unstable_by_key(|&(val, _)| std::cmp::Reverse(val.abs()));
    
    let mut suffix_min = vec![0i64; n + 1];
    let mut suffix_max = vec![0i64; n + 1];
    let mut cur_min = 0i64;
    let mut cur_max = 0i64;
    for i in (0..n).rev() {
        let val = sorted_elements[i].0;
        if val < 0 {
            cur_min += val;
        } else {
            cur_max += val;
        }
        suffix_min[i] = cur_min;
        suffix_max[i] = cur_max;
    }
    
    let mut current_subset = Vec::new();
    let mut nodes_visited = 0;
    if search_backtracking(
        0,
        0,
        target,
        &sorted_elements,
        &suffix_min,
        &suffix_max,
        &mut current_subset,
        &mut nodes_visited,
        max_nodes,
    ) {
        Some(current_subset)
    } else {
        None
    }
}

fn search_backtracking(
    index: usize,
    current_sum: i64,
    target: i64,
    elements: &[(i64, usize)],
    suffix_min: &[i64],
    suffix_max: &[i64],
    current_subset: &mut Vec<usize>,
    nodes_visited: &mut usize,
    max_nodes: usize,
) -> bool {
    *nodes_visited += 1;
    if *nodes_visited > max_nodes {
        return false;
    }
    if current_sum == target {
        return true;
    }
    if index >= elements.len() {
        return false;
    }
    
    let (val, orig_idx) = elements[index];
    let next_min = current_sum + val + suffix_min[index + 1];
    let next_max = current_sum + val + suffix_max[index + 1];
    if target >= next_min && target <= next_max {
        current_subset.push(orig_idx);
        if search_backtracking(
            index + 1,
            current_sum + val,
            target,
            elements,
            suffix_min,
            suffix_max,
            current_subset,
            nodes_visited,
            max_nodes,
        ) {
            return true;
        }
        current_subset.pop();
    }
    
    let next_min = current_sum + suffix_min[index + 1];
    let next_max = current_sum + suffix_max[index + 1];
    if target >= next_min && target <= next_max {
        if search_backtracking(
            index + 1,
            current_sum,
            target,
            elements,
            suffix_min,
            suffix_max,
            current_subset,
            nodes_visited,
            max_nodes,
        ) {
            return true;
        }
    }
    
    false
}

// -------------------------------------------------------------
// Randomized Greedy Solver (Signed)
// -------------------------------------------------------------
fn solve_randomized_greedy_signed(
    elements: &[(i64, usize)],
    target: i64,
    num_trials: usize,
) -> Option<Vec<usize>> {
    let mut rng = SimpleRng::new(54321);
    let n = elements.len();
    let mut temp_elements = elements.to_vec();
    
    for _ in 0..num_trials {
        rng.shuffle(&mut temp_elements);
        
        let mut used = vec![false; n];
        let mut current_sum = 0i64;
        
        for i in 0..n {
            let val = temp_elements[i].0;
            let current_dist = (current_sum - target).abs();
            let new_dist = (current_sum + val - target).abs();
            if new_dist < current_dist {
                current_sum += val;
                used[i] = true;
            }
        }
        
        if current_sum == target {
            let mut res = Vec::new();
            for i in 0..n {
                if used[i] {
                    res.push(temp_elements[i].1);
                }
            }
            return Some(res);
        }
        
        let diff = target - current_sum;
        let mut unused = Vec::new();
        for i in 0..n {
            if !used[i] {
                unused.push((temp_elements[i].0, temp_elements[i].1));
            }
        }
        
        unused.sort_unstable_by_key(|&(val, _)| val);
        if let Ok(idx) = unused.binary_search_by_key(&diff, |&(val, _)| val) {
            let mut res = Vec::new();
            for i in 0..n {
                if used[i] {
                    res.push(temp_elements[i].1);
                }
            }
            res.push(unused[idx].1);
            return Some(res);
        }
        
        if unused.len() >= 2 {
            let mut l = 0;
            let mut r = unused.len() - 1;
            while l < r {
                let sum = unused[l].0 + unused[r].0;
                if sum == diff {
                    let mut res = Vec::new();
                    for i in 0..n {
                        if used[i] {
                            res.push(temp_elements[i].1);
                        }
                    }
                    res.push(unused[l].1);
                    res.push(unused[r].1);
                    return Some(res);
                } else if sum < diff {
                    l += 1;
                } else {
                    r -= 1;
                }
            }
        }
    }
    
    None
}

// -------------------------------------------------------------
// DP + D&C Solver (Signed)
// -------------------------------------------------------------
fn get_reachable_sums_signed(elements: &[DecomposedElement], min_sum: i64, max_sum: i64) -> BitSet {
    let size = (max_sum - min_sum + 1) as usize;
    let mut dp = BitSet::new(size);
    let offset = (-min_sum) as usize;
    dp.set(offset);
    
    let mut cur_max = 0i64;
    
    for val in elements {
        let v = val.val;
        if v > 0 {
            cur_max += v;
            let max_idx = (cur_max - min_sum) as usize;
            let active_words = (max_idx / 64) + 1;
            dp.shift_left_or(v as usize, active_words);
        } else if v < 0 {
            let max_idx = (cur_max - min_sum) as usize;
            let active_words = (max_idx / 64) + 1;
            dp.shift_right_or((-v) as usize, active_words);
        }
    }
    dp
}

fn solve_dp_dc_helper_signed(
    elements: &[DecomposedElement],
    target: i64,
    out_indices: &mut Vec<usize>,
) -> bool {
    if target == 0 {
        return true;
    }
    if elements.is_empty() {
        return false;
    }
    if elements.len() == 1 {
        if elements[0].val == target {
            out_indices.extend(&elements[0].indices);
            return true;
        } else {
            return false;
        }
    }
    
    let mid = elements.len() / 2;
    let (left, right) = elements.split_at(mid);
    
    let mut left_min = 0i64;
    let mut left_max = 0i64;
    for val in left {
        let v = val.val;
        if v < 0 {
            left_min += v;
        } else {
            left_max += v;
        }
    }
    let left_reachable = get_reachable_sums_signed(left, left_min, left_max);
    
    let mut right_min = 0i64;
    let mut right_max = 0i64;
    for val in right {
        let v = val.val;
        if v < 0 {
            right_min += v;
        } else {
            right_max += v;
        }
    }
    let right_reachable = get_reachable_sums_signed(right, right_min, right_max);
    
    let s_start = left_min.max(target - right_max);
    let s_end = left_max.min(target - right_min);
    
    for s in s_start..=s_end {
        let left_idx = (s - left_min) as usize;
        let right_idx = (target - s - right_min) as usize;
        if left_reachable.get(left_idx) && right_reachable.get(right_idx) {
            let left_ok = solve_dp_dc_helper_signed(left, s, out_indices);
            let right_ok = solve_dp_dc_helper_signed(right, target - s, out_indices);
            return left_ok && right_ok;
        }
    }
    
    false
}

fn solve_dp_dc_signed(elements: &[DecomposedElement], target: i64) -> Option<Vec<usize>> {
    let mut result = Vec::new();
    if solve_dp_dc_helper_signed(elements, target, &mut result) {
        Some(result)
    } else {
        None
    }
}

// -------------------------------------------------------------
// DP + D&C Solver (Unsigned - Optimised for non-negative only)
// -------------------------------------------------------------
fn get_reachable_sums_unsigned(elements: &[DecomposedElement], target: u64) -> BitSet {
    let mut dp = BitSet::new((target + 1) as usize);
    dp.set(0);
    
    let mut current_sum = 0u64;
    for val in elements {
        let v = val.val as u64;
        current_sum = (current_sum + v).min(target);
        let active_words = (current_sum / 64) as usize + 1;
        dp.shift_left_or(v as usize, active_words);
    }
    dp
}

fn solve_dp_dc_helper_unsigned(
    elements: &[DecomposedElement],
    target: u64,
    out_indices: &mut Vec<usize>,
) -> bool {
    if target == 0 {
        return true;
    }
    if elements.is_empty() {
        return false;
    }
    if elements.len() == 1 {
        if elements[0].val as u64 == target {
            out_indices.extend(&elements[0].indices);
            return true;
        } else {
            return false;
        }
    }
    
    let mid = elements.len() / 2;
    let (left, right) = elements.split_at(mid);
    
    let left_reachable = get_reachable_sums_unsigned(left, target);
    let right_reachable = get_reachable_sums_unsigned(right, target);
    
    for s in 0..=target {
        if left_reachable.get(s as usize) && right_reachable.get((target - s) as usize) {
            let left_ok = solve_dp_dc_helper_unsigned(left, s, out_indices);
            let right_ok = solve_dp_dc_helper_unsigned(right, target - s, out_indices);
            return left_ok && right_ok;
        }
    }
    
    false
}

fn solve_dp_dc_unsigned(elements: &[DecomposedElement], target: u64) -> Option<Vec<usize>> {
    let mut result = Vec::new();
    if solve_dp_dc_helper_unsigned(elements, target, &mut result) {
        Some(result)
    } else {
        None
    }
}

// -------------------------------------------------------------
// Large Set fallback via Sub-DP (non-negative only)
// -------------------------------------------------------------
fn solve_large_set_via_sub_dp(
    elements: &[(i64, usize)],
    target: i64,
) -> Option<Vec<usize>> {
    let mut sorted = elements.to_vec();
    sorted.sort_unstable_by_key(|&(val, _)| val);
    
    let mut sub_elements = Vec::new();
    let mut current_sum = 0i64;
    for &(val, idx) in &sorted {
        if val > target {
            continue;
        }
        sub_elements.push((val, idx));
        current_sum += val;
        if current_sum >= target + 500000 {
            break;
        }
    }
    
    if current_sum < target {
        return None;
    }
    
    let decomposed = decompose_elements(&sub_elements);
    solve_dp_dc_unsigned(&decomposed, target as u64)
}

// -------------------------------------------------------------
// General Master Solver
// -------------------------------------------------------------
fn solve(elements: Vec<(i64, usize)>, target: i64) -> Option<Vec<usize>> {
    if target == 0 {
        return Some(vec![]);
    }
    if elements.is_empty() {
        return None;
    }
    
    for &(val, idx) in &elements {
        if val == target {
            return Some(vec![idx]);
        }
    }
    
    let total_sum: i64 = elements.iter().map(|&(val, _)| val).sum();
    if total_sum == target {
        return Some(elements.iter().map(|&(_, idx)| idx).collect());
    }
    
    let has_neg = elements.iter().any(|&(v, _)| v < 0);
    
    let mut min_sum = 0i64;
    let mut max_sum = 0i64;
    for &(val, _) in &elements {
        if val < 0 {
            min_sum += val;
        } else {
            max_sum += val;
        }
    }
    if target < min_sum || target > max_sum {
        return None;
    }
    
    let n = elements.len();
    
    if n <= 36 {
        return meet_in_the_middle(&elements, target);
    }
    
    if let Some(res) = solve_backtracking(&elements, target, 5000000) {
        return Some(res);
    }
    
    let num_trials = if n <= 1000 {
        1000
    } else if n <= 10000 {
        500
    } else if n <= 100000 {
        100
    } else {
        10
    };
    
    if let Some(res) = solve_randomized_greedy_signed(&elements, target, num_trials) {
        return Some(res);
    }
    
    if !has_neg && total_sum > target + 500000 {
        if let Some(res) = solve_large_set_via_sub_dp(&elements, target) {
            return Some(res);
        }
    }
    
    let range = max_sum - min_sum;
    if range <= 500_000_000 {
        let decomposed = decompose_elements(&elements);
        if has_neg {
            return solve_dp_dc_signed(&decomposed, target);
        } else {
            return solve_dp_dc_unsigned(&decomposed, target as u64);
        }
    }
    
    None
}

// -------------------------------------------------------------
// Main execution
// -------------------------------------------------------------
fn main() {
    let mut reader = FastReader::new();
    let num_numbers = match reader.next_u64() {
        Some(val) => val as usize,
        None => return,
    };
    let target = match reader.next_i64() {
        Some(val) => val,
        None => return,
    };
    
    let mut elements = Vec::with_capacity(num_numbers);
    for idx in 0..num_numbers {
        if let Some(val) = reader.next_i64() {
            elements.push((val, idx));
        } else {
            break;
        }
    }
    
    if let Some(mut result) = solve(elements, target) {
        result.sort_unstable();
        let stdout = io::stdout();
        let mut writer = BufWriter::new(stdout.lock());
        writeln!(writer, "YES").unwrap();
        write!(writer, "[").unwrap();
        for (i, &idx) in result.iter().enumerate() {
            if i > 0 {
                write!(writer, " ").unwrap();
            }
            write!(writer, "{}", idx).unwrap();
        }
        writeln!(writer, "]").unwrap();
        writer.flush().unwrap();
    } else {
        println!("NO");
    }
}
