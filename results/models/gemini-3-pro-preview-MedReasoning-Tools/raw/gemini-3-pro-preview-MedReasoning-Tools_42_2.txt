import sys
import os
import re
import importlib.util

sys.path.append(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
spec = importlib.util.spec_from_file_location("test42", "../42.py")
test42 = importlib.util.module_from_spec(spec)
spec.loader.exec_module(test42)

shader_path = "answer.txt"
with open(shader_path, "r") as f:
    orig_shader = f.read()

# Let's test factors from 0.5 to 1.0
best_score = 0.0
best_factor = None
best_explanation = ""

for factor in [0.5, 0.6, 0.7, 0.75, 0.8, 0.85, 0.9, 0.95, 1.0]:
    # Replace the factor in the shader code
    # %float_08 = OpConstant %float 0.8
    modified_shader = re.sub(
        r"%float_08 = OpConstant %float 0\.8",
        f"%float_08 = OpConstant %float {factor}",
        orig_shader
    )
    result = {"spirv_code": modified_shader}
    score, explanation = test42.gradeAnswer(result, 2, f"test_engine_{factor}")
    print(f"Factor: {factor:.2f} -> Score: {score:.6f} ({explanation})")
    if score > best_score:
        best_score = score
        best_factor = factor
        best_explanation = explanation

print(f"\nBest Factor: {best_factor} with Score: {best_score}")
