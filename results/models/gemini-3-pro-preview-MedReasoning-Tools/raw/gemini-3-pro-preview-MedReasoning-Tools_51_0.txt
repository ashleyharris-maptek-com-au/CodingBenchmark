import sys
import os
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
import importlib
m51 = importlib.import_module("51")

sc_id = 11
if len(sys.argv) > 1:
    sc_id = int(sys.argv[1])

runner, scenario = m51._setup_scenario(sc_id)

with open("answer.txt", "r") as f:
    code = f.read()

ns = {}
exec(compile(code, '<autopilot>', 'exec'), ns)
autopilot_fn = ns['autopilot_step']

print(f"Starting simulation loop for Scenario {sc_id}: {scenario['name']}...")
steps = scenario.get('steps', 1200)
for step in range(steps):
    state = runner.sensors.build_state(runner.truth)
    state['target_alt_ft'] = runner.target_alt_ft
    state['target_hdg_deg'] = runner.target_hdg_deg
    state['target_ias_kt'] = runner.target_ias_kt
    
    controls = autopilot_fn(state, 0.05)
    if not isinstance(controls, dict):
        controls = {}
    
    # Extract persistent variables from AP_STATE for debugging
    ap_state = ns['AP_STATE']
    pitch_int = ap_state.integrators.get('pitch', 0.0)
    speed_int = ap_state.integrators.get('speed', 0.0)
    
    # Re-calculate intermediate variables
    ias_val = state.get('ias_1_kt', 0.0)
    aoa_val = state.get('aoa_1_deg', 0.0)
    
    # Sanitize controls
    for k in ['elevator', 'aileron', 'rudder', 'elevator_trim', 'speedbrake']:
        if k in controls:
            controls[k] = max(-1.0, min(1.0, float(controls[k])))
    for k in ['throttle_1', 'throttle_2']:
        if k in controls:
            controls[k] = max(0.0, min(1.0, float(controls[k])))
    if 'flaps' in controls:
        controls['flaps'] = max(0, min(40, float(controls['flaps'])))
        
    runner.truth = m51.physics_step(runner.truth, controls, 0.05)
    
    # Calculate G-load for printing
    ax_g = state.get('accel_1_x_g', 0.0)
    ay_g = state.get('accel_1_y_g', 0.0)
    az_g = state.get('accel_1_z_g', -1.0)
    g_load = -az_g
    
    if step % 100 == 0 or step == steps - 1:
        print(f"Step {step:4d}: Alt={runner.truth['z']*3.28084:7.1f} ft (tgt {runner.target_alt_ft:.0f}), Pitch={runner.truth['theta']*57.2958:5.1f} deg, Roll={runner.truth['phi']*57.2958:5.1f} deg, IAS={ias_val:5.1f} kt (tgt {runner.target_ias_kt:.0f}), AoA={aoa_val:4.1f} deg, Elevator={controls.get('elevator', 0):.3f}, Throttle={controls.get('throttle_2', 0):.3f}, VS={state.get('vertical_speed_fpm', 0):.0f}")
    
    if runner.truth['z'] <= 0 and step > 10:
        print(f"Crashed at step {step}!")
        break
        
score, explanation = runner.score_recovery()
print(f"Final Score: {score:.2f}")
print(f"Explanation: {explanation}")
