#include <iostream>
#include <vector>
#include <algorithm>
#include <chrono>
#include <random>
#include <string>

using namespace std;

struct Rectangle {
    int id;
    int w, h;
};

struct Placement {
    int board_index;
    int x, y;
    int rotated; // 0 or 1
};

struct SkylineSegment {
    int x;
    int y;
    int width;
};

struct WasteRect {
    int x;
    int y;
    int w;
    int h;
};

struct Board {
    int width, height;
    std::vector<SkylineSegment> skyline;
    std::vector<WasteRect> waste_map;
    
    Board(int w, int h) : width(w), height(h) {
        skyline.push_back({0, 0, w});
    }
};

enum class Heuristic {
    BottomLeft,
    MinWaste
};

bool try_pack_in_waste_map(Board& board, int rw, int rh, int& out_x, int& out_y, int& out_rotated) {
    int best_idx = -1;
    int min_waste_area = 1e9;
    bool rotate_placed = false;

    for (size_t i = 0; i < board.waste_map.size(); ++i) {
        const auto& wr = board.waste_map[i];
        if (rw <= wr.w && rh <= wr.h) {
            int area = wr.w * wr.h;
            if (area < min_waste_area) {
                min_waste_area = area;
                best_idx = i;
                rotate_placed = false;
            }
        }
        if (rw != rh && rh <= wr.w && rw <= wr.h) {
            int area = wr.w * wr.h;
            if (area < min_waste_area) {
                min_waste_area = area;
                best_idx = i;
                rotate_placed = true;
            }
        }
    }

    if (best_idx != -1) {
        auto wr = board.waste_map[best_idx];
        board.waste_map.erase(board.waste_map.begin() + best_idx);

        int prw = rotate_placed ? rh : rw;
        int prh = rotate_placed ? rw : rh;

        out_x = wr.x;
        out_y = wr.y;
        out_rotated = rotate_placed ? 1 : 0;

        int w1 = wr.w - prw;
        int h1 = prh;
        int w2 = wr.w;
        int h2 = wr.h - prh;

        if (w1 > 0 && h1 > 0 && board.waste_map.size() < 2000) {
            board.waste_map.push_back({wr.x + prw, wr.y, w1, h1});
        }
        if (w2 > 0 && h2 > 0 && board.waste_map.size() < 2000) {
            board.waste_map.push_back({wr.x, wr.y + prh, w2, h2});
        }
        return true;
    }
    return false;
}

bool try_pack_on_skyline(Board& board, int rw, int rh, Heuristic heuristic, int& out_x, int& out_y, int& out_rotated) {
    int best_score1 = 1e9;
    int best_score2 = 1e9;
    int best_segment_idx = -1;
    bool best_rotated = false;
    int best_y_level = -1;

    auto evaluate_placement = [&](int segment_idx, int prw, int prh, bool rotated) {
        const auto& seg = board.skyline[segment_idx];
        if (seg.x + prw > board.width) return;

        int y_level = 0;
        int w_covered = 0;
        for (size_t j = segment_idx; j < board.skyline.size(); ++j) {
            y_level = std::max(y_level, board.skyline[j].y);
            w_covered += board.skyline[j].width;
            if (w_covered >= prw) break;
        }

        if (y_level + prh > board.height) return;

        int score1 = 0;
        int score2 = 0;
        if (heuristic == Heuristic::BottomLeft) {
            score1 = y_level + prh;
            score2 = seg.x;
        } else if (heuristic == Heuristic::MinWaste) {
            int waste_area = 0;
            int x_curr = seg.x;
            for (size_t j = segment_idx; j < board.skyline.size(); ++j) {
                int next_x = x_curr + board.skyline[j].width;
                int w_intersect = std::min(next_x, seg.x + prw) - std::max(x_curr, seg.x);
                if (w_intersect > 0) {
                    waste_area += (y_level - board.skyline[j].y) * w_intersect;
                }
                x_curr = next_x;
                if (x_curr >= seg.x + prw) break;
            }
            score1 = waste_area;
            score2 = y_level;
        }

        if (score1 < best_score1 || (score1 == best_score1 && score2 < best_score2)) {
            best_score1 = score1;
            best_score2 = score2;
            best_segment_idx = segment_idx;
            best_rotated = rotated;
            best_y_level = y_level;
        }
    };

    for (size_t i = 0; i < board.skyline.size(); ++i) {
        evaluate_placement(i, rw, rh, false);
        if (rw != rh) {
            evaluate_placement(i, rh, rw, true);
        }
    }

    if (best_segment_idx != -1) {
        int prw = best_rotated ? rh : rw;
        int prh = best_rotated ? rw : rh;
        int target_x = board.skyline[best_segment_idx].x;
        int target_y = best_y_level;

        out_x = target_x;
        out_y = target_y;
        out_rotated = best_rotated ? 1 : 0;

        int x_curr = target_x;
        for (size_t j = best_segment_idx; j < board.skyline.size(); ++j) {
            int next_x = x_curr + board.skyline[j].width;
            int w_intersect = std::min(next_x, target_x + prw) - std::max(x_curr, target_x);
            if (w_intersect > 0) {
                int waste_h = target_y - board.skyline[j].y;
                if (waste_h > 0 && board.waste_map.size() < 2000) {
                    board.waste_map.push_back({std::max(x_curr, target_x), board.skyline[j].y, w_intersect, waste_h});
                }
            }
            x_curr = next_x;
            if (x_curr >= target_x + prw) break;
        }

        std::vector<SkylineSegment> new_skyline;
        for (size_t j = 0; j < board.skyline.size(); ++j) {
            const auto& seg = board.skyline[j];
            if (seg.x + seg.width <= target_x) {
                new_skyline.push_back(seg);
            } else {
                break;
            }
        }

        for (size_t j = 0; j < board.skyline.size(); ++j) {
            const auto& seg = board.skyline[j];
            if (seg.x < target_x && seg.x + seg.width > target_x) {
                new_skyline.push_back({seg.x, seg.y, target_x - seg.x});
                break;
            }
        }

        new_skyline.push_back({target_x, target_y + prh, prw});

        for (size_t j = 0; j < board.skyline.size(); ++j) {
            const auto& seg = board.skyline[j];
            if (seg.x < target_x + prw && seg.x + seg.width > target_x + prw) {
                new_skyline.push_back({target_x + prw, seg.y, seg.x + seg.width - (target_x + prw)});
                break;
            }
        }

        for (size_t j = 0; j < board.skyline.size(); ++j) {
            const auto& seg = board.skyline[j];
            if (seg.x >= target_x + prw) {
                new_skyline.push_back(seg);
            }
        }

        board.skyline.clear();
        if (!new_skyline.empty()) {
            board.skyline.push_back(new_skyline[0]);
            for (size_t j = 1; j < new_skyline.size(); ++j) {
                if (new_skyline[j].y == board.skyline.back().y) {
                    board.skyline.back().width += new_skyline[j].width;
                } else {
                    board.skyline.push_back(new_skyline[j]);
                }
            }
        }

        return true;
    }

    return false;
}

vector<Placement> solve_packing(const vector<Rectangle>& rects, int board_width, int board_height, Heuristic heuristic, int& out_num_boards) {
    vector<Placement> placements(rects.size());
    vector<bool> packed(rects.size(), false);
    int num_boards = 0;
    int packed_count = 0;
    int N = rects.size();

    vector<int> unpacked_indices;
    unpacked_indices.reserve(N);
    for (int i = 0; i < N; ++i) {
        unpacked_indices.push_back(i);
    }

    while (packed_count < N) {
        Board board(board_width, board_height);
        
        vector<int> current_round_indices;
        current_round_indices.reserve(unpacked_indices.size());
        for (int idx : unpacked_indices) {
            if (!packed[idx]) {
                current_round_indices.push_back(idx);
            }
        }
        unpacked_indices = current_round_indices;

        for (int idx : unpacked_indices) {
            int rx = 0, ry = 0, rot = 0;
            if (try_pack_in_waste_map(board, rects[idx].w, rects[idx].h, rx, ry, rot)) {
                placements[rects[idx].id] = {num_boards, rx, ry, rot};
                packed[idx] = true;
                packed_count++;
            } else if (try_pack_on_skyline(board, rects[idx].w, rects[idx].h, heuristic, rx, ry, rot)) {
                placements[rects[idx].id] = {num_boards, rx, ry, rot};
                packed[idx] = true;
                packed_count++;
            }
        }
        num_boards++;
    }
    
    out_num_boards = num_boards;
    return placements;
}

void sort_rects(vector<Rectangle>& rects, int criterion) {
    if (criterion == 0) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            int area_a = a.w * a.h;
            int area_b = b.w * b.h;
            if (area_a != area_b) return area_a > area_b;
            return a.id < b.id;
        });
    } else if (criterion == 1) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            int max_a = std::max(a.w, a.h);
            int max_b = std::max(b.w, b.h);
            if (max_a != max_b) return max_a > max_b;
            int min_a = std::min(a.w, a.h);
            int min_b = std::min(b.w, b.h);
            if (min_a != min_b) return min_a > min_b;
            return a.id < b.id;
        });
    } else if (criterion == 2) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            if (a.h != b.h) return a.h > b.h;
            if (a.w != b.w) return a.w > b.w;
            return a.id < b.id;
        });
    } else if (criterion == 3) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            if (a.w != b.w) return a.w > b.w;
            if (a.h != b.h) return a.h > b.h;
            return a.id < b.id;
        });
    } else if (criterion == 4) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            int p_a = a.w + a.h;
            int p_b = b.w + b.h;
            if (p_a != p_b) return p_a > p_b;
            return a.id < b.id;
        });
    } else if (criterion == 5) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            int max_a = std::max(a.w, a.h);
            int max_b = std::max(b.w, b.h);
            if (max_a != max_b) return max_a > max_b;
            int area_a = a.w * a.h;
            int area_b = b.w * b.h;
            if (area_a != area_b) return area_a > area_b;
            return a.id < b.id;
        });
    } else if (criterion == 6) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            int min_a = std::min(a.w, a.h);
            int min_b = std::min(b.w, b.h);
            if (min_a != min_b) return min_a > min_b;
            int max_a = std::max(a.w, a.h);
            int max_b = std::max(b.w, b.h);
            if (max_a != max_b) return max_a > max_b;
            return a.id < b.id;
        });
    } else if (criterion == 7) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            double ratio_a = (double)std::max(a.w, a.h) / std::max(1, std::min(a.w, a.h));
            double ratio_b = (double)std::max(b.w, b.h) / std::max(1, std::min(b.w, b.h));
            if (std::abs(ratio_a - ratio_b) > 1e-5) return ratio_a > ratio_b;
            int area_a = a.w * a.h;
            int area_b = b.w * b.h;
            if (area_a != area_b) return area_a > area_b;
            return a.id < b.id;
        });
    }
}

int main() {
    int N = 50000;
    int W = 2000, H = 2000;
    vector<Rectangle> original_rects(N);
    mt19937 rng(42);
    uniform_int_distribution<int> dist_w(10, 100);
    uniform_int_distribution<int> dist_h(10, 100);

    for (int i = 0; i < N; ++i) {
        original_rects[i] = {i, dist_w(rng), dist_h(rng)};
    }

    vector<string> sort_names = {
        "AreaDesc", "MaxDimDesc", "HeightDesc", "WidthDesc",
        "PerimeterDesc", "MaxDimAreaDesc", "MinDimDesc", "AspectRatioDesc"
    };

    for (int crit = 0; crit < 8; ++crit) {
        for (int h_type = 0; h_type < 2; ++h_type) {
            vector<Rectangle> rects = original_rects;
            sort_rects(rects, crit);

            Heuristic heuristic = (h_type == 0) ? Heuristic::BottomLeft : Heuristic::MinWaste;
            string h_name = (h_type == 0) ? "BottomLeft" : "MinWaste";

            auto start = chrono::high_resolution_clock::now();
            int num_boards = 0;
            auto placements = solve_packing(rects, W, H, heuristic, num_boards);
            auto end = chrono::high_resolution_clock::now();
            auto duration = chrono::duration_cast<chrono::milliseconds>(end - start).count();

            cout << "Sort: " << sort_names[crit] << " | Heuristic: " << h_name 
                 << " | Boards: " << num_boards << " | Time: " << duration << " ms" << endl;
        }
    }

    return 0;
}
