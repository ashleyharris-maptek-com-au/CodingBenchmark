import math

# Global state to persist estimator across steps within a scenario.
# Re-initializes automatically when a new scenario is detected (via a fuel jump).
global HISTORY
HISTORY = {
    'initialized': False,
    'est_x': 0.0, 'est_y': 0.0, 'est_z': 0.0,
    'est_vx': 0.0, 'est_vy': 0.0, 'est_vz': 0.0,
    'est_roll': 0.0, 'est_pitch': 0.0, 'est_yaw': 0.0,
    'prev_thrust_x': 0.0, 'prev_thrust_y': 0.0, 'prev_thrust_z': 0.0,
    'prev_selected_roll': None,
    'prev_selected_pitch': None,
    'prev_selected_yaw': None,
    'bias_x': 0.0, 'bias_y': 0.0, 'bias_z': 0.0,
    'prev_fuel': -1.0,
}

def autopilot_step(state: dict, dt: float) -> dict:
    global HISTORY

    # 1. SCENARIO DETECTOR & RE-INITIALIZATION
    current_fuel = state.get('fuel_remaining_kg', 500.0)
    prev_fuel = HISTORY.get('prev_fuel', -1.0)
    if current_fuel > prev_fuel + 5.0:
        # Scenario reset or new scenario detected
        HISTORY = {
            'initialized': False,
            'est_x': 0.0, 'est_y': 0.0, 'est_z': 0.0,
            'est_vx': 0.0, 'est_vy': 0.0, 'est_vz': 0.0,
            'est_roll': 0.0, 'est_pitch': 0.0, 'est_yaw': 0.0,
            'prev_thrust_x': 0.0, 'prev_thrust_y': 0.0, 'prev_thrust_z': 0.0,
            'prev_selected_roll': None,
            'prev_selected_pitch': None,
            'prev_selected_yaw': None,
            'bias_x': 0.0, 'bias_y': 0.0, 'bias_z': 0.0,
            'prev_fuel': current_fuel,
        }
    else:
        HISTORY['prev_fuel'] = current_fuel

    # 2. DEFINE HELPER FUNCTIONS
    def median_fallback(lst, default=0.0):
        if not lst:
            return default
        lst_sorted = sorted(lst)
        n_len = len(lst_sorted)
        if n_len % 2 == 1:
            return lst_sorted[n_len // 2]
        else:
            return (lst_sorted[n_len // 2 - 1] + lst_sorted[n_len // 2]) / 2.0

    def get_consensus(raw_vals, pred_val):
        if not raw_vals:
            return pred_val
        if len(raw_vals) == 1:
            val = raw_vals[0]
            if pred_val is not None:
                thresh = max(5.0, 0.2 * abs(pred_val))
                if abs(val - pred_val) < thresh:
                    return val
                else:
                    return pred_val
            else:
                return val
        # Clustering
        clusters = []
        for v in raw_vals:
            thresh = max(2.0, 0.1 * abs(v))
            cluster = [x for x in raw_vals if abs(x - v) <= thresh]
            clusters.append(cluster)
        best_cluster = max(clusters, key=len)
        mean_val = sum(best_cluster) / len(best_cluster)
        
        if len(best_cluster) == 1 and pred_val is not None:
            closest_val = min(raw_vals, key=lambda x: abs(x - pred_val))
            thresh = max(5.0, 0.2 * abs(pred_val))
            if abs(closest_val - pred_val) < thresh:
                return closest_val
            return pred_val
        return mean_val

    # 3. ATTITUDE STATE ESTIMATOR & ATTITUDE RATE SENSOR FUSION
    # Gather star tracker angles
    st_rolls = []
    st_pitches = []
    st_yaws = []
    if state.get('star_tracker_1_roll_valid'):
        st_rolls.append(state['star_tracker_1_roll_deg'])
        st_pitches.append(state['star_tracker_1_pitch_deg'])
        st_yaws.append(state['star_tracker_1_yaw_deg'])
    if state.get('star_tracker_2_roll_valid'):
        st_rolls.append(state['star_tracker_2_roll_deg'])
        st_pitches.append(state['star_tracker_2_pitch_deg'])
        st_yaws.append(state['star_tracker_2_yaw_deg'])
        
    st_roll = median_fallback(st_rolls, None)
    st_pitch = median_fallback(st_pitches, None)
    st_yaw = median_fallback(st_yaws, None)

    # Gather gyro rates (dps)
    gyro_rolls = []
    gyro_pitches = []
    gyro_yaws = []
    for i in (1, 2, 3):
        if state.get(f'gyro_{i}_roll_valid'):
            gyro_rolls.append(state[f'gyro_{i}_roll_dps'])
        if state.get(f'gyro_{i}_pitch_valid'):
            gyro_pitches.append(state[f'gyro_{i}_pitch_dps'])
        if state.get(f'gyro_{i}_yaw_valid'):
            gyro_yaws.append(state[f'gyro_{i}_yaw_dps'])

    # Determine consensus star tracker reading
    selected_roll, selected_pitch, selected_yaw = None, None, None
    t1_valid = state.get('star_tracker_1_roll_valid')
    t2_valid = state.get('star_tracker_2_roll_valid')
    t1_val = (state.get('star_tracker_1_roll_deg'), state.get('star_tracker_1_pitch_deg'), state.get('star_tracker_1_yaw_deg'))
    t2_val = (state.get('star_tracker_2_roll_deg'), state.get('star_tracker_2_pitch_deg'), state.get('star_tracker_2_yaw_deg'))

    if not HISTORY['initialized']:
        # Initial step attitude
        if t1_valid and t2_valid:
            selected_roll = (t1_val[0] + t2_val[0]) / 2.0
            selected_pitch = (t1_val[1] + t2_val[1]) / 2.0
            selected_yaw = (t1_val[2] + t2_val[2]) / 2.0
        elif t1_valid:
            selected_roll, selected_pitch, selected_yaw = t1_val
        elif t2_valid:
            selected_roll, selected_pitch, selected_yaw = t2_val
        
        HISTORY['est_roll'] = selected_roll if selected_roll is not None else 0.0
        HISTORY['est_pitch'] = selected_pitch if selected_pitch is not None else 0.0
        HISTORY['est_yaw'] = selected_yaw if selected_yaw is not None else 0.0
    else:
        # Propagate attitude from previous step using estimated rates
        roll_rate_avg = median_fallback(gyro_rolls, 0.0)
        pitch_rate_avg = median_fallback(gyro_pitches, 0.0)
        yaw_rate_avg = median_fallback(gyro_yaws, 0.0)
        
        prop_roll = HISTORY['est_roll'] + roll_rate_avg * dt
        prop_pitch = HISTORY['est_pitch'] + pitch_rate_avg * dt
        prop_yaw = HISTORY['est_yaw'] + yaw_rate_avg * dt

        if t1_valid and t2_valid:
            diff_roll = abs(t1_val[0] - t2_val[0])
            if diff_roll < 2.0:
                selected_roll = (t1_val[0] + t2_val[0]) / 2.0
                selected_pitch = (t1_val[1] + t2_val[1]) / 2.0
                selected_yaw = (t1_val[2] + t2_val[2]) / 2.0
            else:
                # Silent failure check: choose the one closer to propagation
                d1 = abs(t1_val[0] - prop_roll)
                d2 = abs(t2_val[0] - prop_roll)
                if d1 < d2:
                    selected_roll, selected_pitch, selected_yaw = t1_val
                else:
                    selected_roll, selected_pitch, selected_yaw = t2_val
        elif t1_valid:
            selected_roll, selected_pitch, selected_yaw = t1_val
        elif t2_valid:
            selected_roll, selected_pitch, selected_yaw = t2_val

        # Low-pass filter for attitude angles
        alpha_att = 0.2
        if selected_roll is not None:
            HISTORY['est_roll'] = (1 - alpha_att) * prop_roll + alpha_att * selected_roll
            HISTORY['est_pitch'] = (1 - alpha_att) * prop_pitch + alpha_att * selected_pitch
            HISTORY['est_yaw'] = (1 - alpha_att) * prop_yaw + alpha_att * selected_yaw
        else:
            HISTORY['est_roll'] = prop_roll
            HISTORY['est_pitch'] = prop_pitch
            HISTORY['est_yaw'] = prop_yaw

    # Attitude rates estimation (incorporating star tracker differentiation to cancel gyro biases/drift)
    has_st_rate = (HISTORY['initialized'] and 
                   selected_roll is not None and 
                   HISTORY.get('prev_selected_roll') is not None)
    
    st_rates = None
    if has_st_rate:
        st_rates = (
            (selected_roll - HISTORY['prev_selected_roll']) / dt,
            (selected_pitch - HISTORY['prev_selected_pitch']) / dt,
            (selected_yaw - HISTORY['prev_selected_yaw']) / dt
        )

    # Reject gyro outliers that diverge from star tracker derivative
    filtered_gyro_rolls = []
    filtered_gyro_pitches = []
    filtered_gyro_yaws = []
    
    for val in gyro_rolls:
        if has_st_rate:
            if abs(val - st_rates[0]) < 2.0:
                filtered_gyro_rolls.append(val)
        else:
            filtered_gyro_rolls.append(val)
            
    for val in gyro_pitches:
        if has_st_rate:
            if abs(val - st_rates[1]) < 2.0:
                filtered_gyro_pitches.append(val)
        else:
            filtered_gyro_pitches.append(val)
            
    for val in gyro_yaws:
        if has_st_rate:
            if abs(val - st_rates[2]) < 2.0:
                filtered_gyro_yaws.append(val)
        else:
            filtered_gyro_yaws.append(val)
            
    roll_rate_dps = median_fallback(filtered_gyro_rolls, st_rates[0] if has_st_rate else 0.0)
    pitch_rate_dps = median_fallback(filtered_gyro_pitches, st_rates[1] if has_st_rate else 0.0)
    yaw_rate_dps = median_fallback(filtered_gyro_yaws, st_rates[2] if has_st_rate else 0.0)

    # Save selected star tracker angles for the next step differentiation
    HISTORY['prev_selected_roll'] = selected_roll
    HISTORY['prev_selected_pitch'] = selected_pitch
    HISTORY['prev_selected_yaw'] = selected_yaw

    # 4. ORBITAL MEAN MOTION (n) & SPACECRAFT MASS ESTIMATION
    semi_major_axis = 6778000.0  # Default LEO
    speed = 7670.0
    if state.get('ground_semi_major_axis_m_valid') and state['ground_semi_major_axis_m'] > 1e6:
        semi_major_axis = state['ground_semi_major_axis_m']
    if state.get('ground_speed_ms_valid') and state['ground_speed_ms'] > 1000.0:
        speed = state['ground_speed_ms']
    n = speed / semi_major_axis

    # Spacecraft Mass & Command-to-Acceleration scaling (Thruster force: 220 N)
    fuel = state.get('fuel_remaining_kg', 500.0)
    mass = 4000.0 + fuel
    a_max = 220.0 / mass

    # 5. PHYSICS-BASED PROPAGATION (PREDICTION STEP)
    # Collect accelerometers (non-gravitational acceleration measurements)
    accel_x_list = []
    accel_y_list = []
    accel_z_list = []
    for i in (1, 2, 3):
        if state.get(f'accel_{i}_x_valid'):
            accel_x_list.append(state[f'accel_{i}_x_ms2'])
        if state.get(f'accel_{i}_y_valid'):
            accel_y_list.append(state[f'accel_{i}_y_ms2'])
        if state.get(f'accel_{i}_z_valid'):
            accel_z_list.append(state[f'accel_{i}_z_ms2'])

    prev_tx = HISTORY.get('prev_thrust_x', 0.0)
    prev_ty = HISTORY.get('prev_thrust_y', 0.0)
    prev_tz = HISTORY.get('prev_thrust_z', 0.0)
    
    thrust_accel_x = prev_tx * a_max
    thrust_accel_y = prev_ty * a_max
    thrust_accel_z = prev_tz * a_max

    ax = get_consensus(accel_x_list, thrust_accel_x)
    ay = get_consensus(accel_y_list, thrust_accel_y)
    az = get_consensus(accel_z_list, thrust_accel_z)

    # Estimate uncommanded thrust / bias (ADRC)
    bias_x = ax - thrust_accel_x
    bias_y = ay - thrust_accel_y
    bias_z = az - thrust_accel_z

    if not HISTORY['initialized']:
        HISTORY['bias_x'] = 0.0
        HISTORY['bias_y'] = 0.0
        HISTORY['bias_z'] = 0.0
    else:
        alpha_bias = 0.05
        HISTORY['bias_x'] = (1 - alpha_bias) * HISTORY['bias_x'] + alpha_bias * bias_x
        HISTORY['bias_y'] = (1 - alpha_bias) * HISTORY['bias_y'] + alpha_bias * bias_y
        HISTORY['bias_z'] = (1 - alpha_bias) * HISTORY['bias_z'] + alpha_bias * bias_z

    pred_x, pred_y, pred_z = 0.0, 0.0, 0.0
    pred_vx, pred_vy, pred_vz = 0.0, 0.0, 0.0

    if HISTORY['initialized']:
        # Clohessy-Wiltshire (CW) equations propagation
        pred_x = HISTORY['est_x'] + HISTORY['est_vx'] * dt
        pred_y = HISTORY['est_y'] + HISTORY['est_vy'] * dt
        pred_z = HISTORY['est_z'] + HISTORY['est_vz'] * dt
        
        pred_vx = HISTORY['est_vx'] + (3.0 * n**2 * HISTORY['est_x'] + 2.0 * n * HISTORY['est_vy'] + ax) * dt
        pred_vy = HISTORY['est_vy'] + (-2.0 * n * HISTORY['est_vx'] + ay) * dt
        pred_vz = HISTORY['est_vz'] + (-n**2 * HISTORY['est_z'] + az) * dt

    # 6. SENSOR MEASUREMENT EXTRACTION (POSITION & VELOCITY)
    # Docking camera processing (resolving 3D position by rotating camera frame to LVLH)
    cam_pos = [None, None, None]
    if state.get('dock_camera_active'):
        r_cam = state.get('dock_camera_range_m')
        if r_cam is not None and r_cam > 0:
            bx = state.get('dock_camera_bearing_x')
            by = state.get('dock_camera_bearing_y')
            bz = state.get('dock_camera_bearing_z')
            if (state.get('dock_camera_bearing_x_valid') and 
                state.get('dock_camera_bearing_y_valid') and 
                state.get('dock_camera_bearing_z_valid') and
                bx is not None and by is not None and bz is not None):
                
                norm = math.sqrt(bx**2 + by**2 + bz**2)
                if norm > 1e-5:
                    bx_n, by_n, bz_n = bx/norm, by/norm, bz/norm
                    p_body = [bx_n * r_cam, by_n * r_cam, bz_n * r_cam]
                    
                    # Rotate from body to LVLH using R_transpose
                    r_val = math.radians(HISTORY['est_roll'])
                    p_val = math.radians(HISTORY['est_pitch'])
                    y_val = math.radians(HISTORY['est_yaw'])
                    cr, sr = math.cos(r_val), math.sin(r_val)
                    cp, sp = math.cos(p_val), math.sin(p_val)
                    cy, sy = math.cos(y_val), math.sin(y_val)
                    
                    px_LVLH = (cp*cy)*p_body[0] + (sr*sp*cy - cr*sy)*p_body[1] + (cr*sp*cy + sr*sy)*p_body[2]
                    py_LVLH = (cp*sy)*p_body[0] + (sr*sp*sy + cr*cy)*p_body[1] + (cr*sp*sy - sr*cy)*p_body[2]
                    pz_LVLH = (-sp)*p_body[0] + (sr*cp)*p_body[1] + (cr*cp)*p_body[2]
                    cam_pos = [px_LVLH, py_LVLH, pz_LVLH]

    # Gather relative position measurements
    raw_x_vals = []
    raw_y_vals = []
    raw_z_vals = []

    if state.get('radar_rel_x_valid'): raw_x_vals.append(state['radar_rel_pos_x_m'])
    if state.get('radar_rel_y_valid'): raw_y_vals.append(state['radar_rel_pos_y_m'])
    if state.get('radar_rel_z_valid'): raw_z_vals.append(state['radar_rel_pos_z_m'])
        
    if state.get('lidar_rel_x_valid'): raw_x_vals.append(state['lidar_rel_pos_x_m'])
    if state.get('lidar_rel_y_valid'): raw_y_vals.append(state['lidar_rel_pos_y_m'])
    if state.get('lidar_rel_z_valid'): raw_z_vals.append(state['lidar_rel_pos_z_m'])
        
    if state.get('gps_1_x_valid'): raw_x_vals.append(state['gps_1_pos_x_m'])
    if state.get('gps_1_y_valid'): raw_y_vals.append(state['gps_1_pos_y_m'])
    if state.get('gps_1_z_valid'): raw_z_vals.append(state['gps_1_pos_z_m'])
        
    if state.get('gps_2_x_valid'): raw_x_vals.append(state['gps_2_pos_x_m'])
    if state.get('gps_2_y_valid'): raw_y_vals.append(state['gps_2_pos_y_m'])
    if state.get('gps_2_z_valid'): raw_z_vals.append(state['gps_2_pos_z_m'])
        
    if cam_pos[0] is not None: raw_x_vals.append(cam_pos[0])
    if cam_pos[1] is not None: raw_y_vals.append(cam_pos[1])
    if cam_pos[2] is not None: raw_z_vals.append(cam_pos[2])

    # Consensus range from range finders
    rf_ranges = []
    if state.get('range_1_valid'): rf_ranges.append(state['range_1_m'])
    if state.get('range_2_valid'): rf_ranges.append(state['range_2_m'])
    if state.get('range_3_valid'): rf_ranges.append(state['range_3_m'])
    if state.get('dock_camera_active') and state.get('dock_camera_range_m') is not None:
        rf_ranges.append(state['dock_camera_range_m'])

    # Determine consensus range
    consensus_range = None
    if rf_ranges:
        if len(rf_ranges) == 1:
            consensus_range = rf_ranges[0]
        else:
            r_clusters = []
            for r in rf_ranges:
                r_cluster = [x for x in rf_ranges if abs(x - r) <= max(2.0, 0.1 * r)]
                r_clusters.append(r_cluster)
            consensus_range = sum(max(r_clusters, key=len)) / len(max(r_clusters, key=len))

    # Determine consensus position
    est_x = get_consensus(raw_x_vals, pred_x if HISTORY['initialized'] else None)
    est_y = get_consensus(raw_y_vals, pred_y if HISTORY['initialized'] else None)
    est_z = get_consensus(raw_z_vals, pred_z if HISTORY['initialized'] else None)
    
    if est_x is None: est_x = 0.0
    if est_y is None: est_y = 0.0
    if est_z is None: est_z = 0.0

    # Gather relative velocity measurements
    raw_vx_vals = []
    raw_vy_vals = []
    raw_vz_vals = []

    if state.get('radar_relv_x_valid'): raw_vx_vals.append(state['radar_rel_vel_x_ms'])
    if state.get('radar_relv_y_valid'): raw_vy_vals.append(state['radar_rel_vel_y_ms'])
    if state.get('radar_relv_z_valid'): raw_vz_vals.append(state['radar_rel_vel_z_ms'])
        
    if state.get('lidar_relv_x_valid'): raw_vx_vals.append(state['lidar_rel_vel_x_ms'])
    if state.get('lidar_relv_y_valid'): raw_vy_vals.append(state['lidar_rel_vel_y_ms'])
    if state.get('lidar_relv_z_valid'): raw_vz_vals.append(state['lidar_rel_vel_z_ms'])

    # Add numeric derivatives of position as independent velocity sensors
    if HISTORY['initialized']:
        raw_vx_vals.append((est_x - HISTORY['est_x']) / dt)
        raw_vy_vals.append((est_y - HISTORY['est_y']) / dt)
        raw_vz_vals.append((est_z - HISTORY['est_z']) / dt)

    est_vx = get_consensus(raw_vx_vals, pred_vx if HISTORY['initialized'] else None)
    est_vy = get_consensus(raw_vy_vals, pred_vy if HISTORY['initialized'] else None)
    est_vz = get_consensus(raw_vz_vals, pred_vz if HISTORY['initialized'] else None)

    if est_vx is None: est_vx = 0.0
    if est_vy is None: est_vy = 0.0
    if est_vz is None: est_vz = 0.0

    # 7. SENSOR FUSION STATE UPDATE (ALPHA-BETA FILTERING)
    if not HISTORY['initialized']:
        HISTORY['est_x'] = est_x
        HISTORY['est_y'] = est_y
        HISTORY['est_z'] = est_z
        HISTORY['est_vx'] = est_vx
        HISTORY['est_vy'] = est_vy
        HISTORY['est_vz'] = est_vz
        HISTORY['initialized'] = True
    else:
        # Determine sensor spread to dynamically adjust filtering gains (Kalman Gain adaptation)
        spreads = []
        for vals in [raw_x_vals, raw_y_vals, raw_z_vals]:
            valid_vals = [v for v in vals if v is not None]
            if len(valid_vals) >= 2:
                spreads.append(max(valid_vals) - min(valid_vals))
        max_spread = max(spreads) if spreads else 0.0

        if max_spread > 20.0:
            alpha_pos = 0.08
            alpha_vel = 0.06
        elif max_spread > 5.0:
            alpha_pos = 0.2
            alpha_vel = 0.15
        else:
            alpha_pos = 0.4
            alpha_vel = 0.3

        # Update estimates with filtering
        HISTORY['est_x'] = (1 - alpha_pos) * pred_x + alpha_pos * est_x
        HISTORY['est_y'] = (1 - alpha_pos) * pred_y + alpha_pos * est_y
        HISTORY['est_z'] = (1 - alpha_pos) * pred_z + alpha_pos * est_z
        
        HISTORY['est_vx'] = (1 - alpha_vel) * pred_vx + alpha_vel * est_vx
        HISTORY['est_vy'] = (1 - alpha_vel) * pred_vy + alpha_vel * est_vy
        HISTORY['est_vz'] = (1 - alpha_vel) * pred_vz + alpha_vel * est_vz

        # Anchor 3D position estimate to range finders distance consensus
        if consensus_range is not None and len(rf_ranges) >= 2 and consensus_range > 0:
            est_r = math.sqrt(HISTORY['est_x']**2 + HISTORY['est_y']**2 + HISTORY['est_z']**2)
            if est_r > 1.0:
                ratio = consensus_range / est_r
                if 0.5 < ratio < 2.0:
                    scale = 0.95 * 1.0 + 0.05 * ratio
                    HISTORY['est_x'] *= scale
                    HISTORY['est_y'] *= scale
                    HISTORY['est_z'] *= scale

    # Read current state values from HISTORY
    x = HISTORY['est_x']
    y = HISTORY['est_y']
    z = HISTORY['est_z']
    vx = HISTORY['est_vx']
    vy = HISTORY['est_vy']
    vz = HISTORY['est_vz']

    # 8. GUIDANCE & CONTROL (ORBITAL MANEUVERING & CW DYNAMICS)
    range_to_target = math.sqrt(x**2 + y**2 + z**2)

    # Multi-regime approach velocity limit schedule
    if range_to_target > 1000.0:
        max_speed = 3.0
    elif range_to_target > 500.0:
        max_speed = 2.0
    elif range_to_target > 200.0:
        max_speed = 1.0
    elif range_to_target > 50.0:
        max_speed = 0.4
    elif range_to_target > 15.0:
        max_speed = 0.15
    else:
        max_speed = 0.08

    # Target along-track velocity to close the y gap
    v_y_target = -0.025 * y
    v_y_target = max(-max_speed, min(max_speed, v_y_target))

    # Maintain minimum approach velocity to guarantee docking success and prevent stopping
    if range_to_target >= 2.0:
        if y < 0:
            v_y_target = max(0.05, v_y_target)
        else:
            v_y_target = min(-0.05, v_y_target)
    else:
        # Hold station at dock
        v_y_target = 0.0

    # Hohmann-like transfer drift orbit calculation
    smooth_factor = max(0.0, min(1.0, (range_to_target - 20.0) / 80.0))
    x_target = - (2.0 * v_y_target) / (3.0 * n) * smooth_factor

    # Velocity-limiting controls for radial and cross-track
    v_x_target = -0.02 * (x - x_target)
    max_radial_speed = max(0.05, 0.25 * max_speed)
    v_x_target = max(-max_radial_speed, min(max_radial_speed, v_x_target))

    v_z_target = -0.02 * z
    max_cross_speed = max(0.05, 0.25 * max_speed)
    v_z_target = max(-max_cross_speed, min(max_cross_speed, v_z_target))

    if range_to_target < 2.0:
        v_x_target = 0.0
        v_z_target = 0.0

    # Required control accelerations
    # Dual-mode controller: Hohmann orbit coupling far away, direct linear close up
    # We smoothly transition between them based on range to target
    alpha_regime = max(0.0, min(1.0, (range_to_target - 50.0) / 100.0))

    # Direct controller accelerations
    Kd_x = 0.1
    Kd_y = 0.15
    Kd_z = 0.1

    ax_req_direct = -Kd_x * (vx - v_x_target)
    ay_req_direct = -Kd_y * (vy - v_y_target)
    az_req_direct = -Kd_z * (vz - v_z_target)

    ux_direct = ax_req_direct - 3.0 * n**2 * x - 2.0 * n * vy
    uy_direct = ay_req_direct + 2.0 * n * vx
    uz_direct = az_req_direct + n**2 * z

    # Hohmann controller accelerations (along-track controls altitude error, radial controls velocity damping)
    Kp_alt = 0.0005
    Kd_radial = 0.05
    ux_hohmann = -Kd_radial * vx
    uy_hohmann = Kp_alt * (x_target - x)
    uz_hohmann = uz_direct  # Cross-track is identical in both modes

    # Fuse the control accelerations based on the regime
    ux = (1 - alpha_regime) * ux_direct + alpha_regime * ux_hohmann - HISTORY['bias_x']
    uy = (1 - alpha_regime) * uy_direct + alpha_regime * uy_hohmann - HISTORY['bias_y']
    uz = (1 - alpha_regime) * uz_direct + alpha_regime * uz_hohmann - HISTORY['bias_z']

    # Rotate command vector from LVLH frame to spacecraft body frame
    r_val = math.radians(HISTORY['est_roll'])
    p_val = math.radians(HISTORY['est_pitch'])
    y_val = math.radians(HISTORY['est_yaw'])
    cr, sr = math.cos(r_val), math.sin(r_val)
    cp, sp = math.cos(p_val), math.sin(p_val)
    cy, sy = math.cos(y_val), math.sin(y_val)

    ux_body = cp*cy * ux + cp*sy * uy - sp * uz
    uy_body = (sr*sp*cy - cr*sy) * ux + (sr*sp*sy + cr*cy) * uy + sr*cp * uz
    uz_body = (cr*sp*cy + sr*sy) * ux + (cr*sp*sy - sr*cy) * uy + cr*cp * uz

    # Scale acceleration to thrust command (clamped from -1.0 to 1.0)
    thrust_x = ux_body / a_max
    thrust_y = uy_body / a_max
    thrust_z = uz_body / a_max

    # Fuel-saving deadbands (widened under low fuel conditions)
    fuel_pct = state.get('fuel_remaining_pct', 1.0)
    if fuel < 60.0 or fuel_pct < 0.12:
        deadband = 0.05
    else:
        deadband = 0.005

    if abs(thrust_x) < deadband: thrust_x = 0.0
    if abs(thrust_y) < deadband: thrust_y = 0.0
    if abs(thrust_z) < deadband: thrust_z = 0.0

    thrust_x = max(-1.0, min(1.0, thrust_x))
    thrust_y = max(-1.0, min(1.0, thrust_y))
    thrust_z = max(-1.0, min(1.0, thrust_z))

    HISTORY['prev_thrust_x'] = thrust_x
    HISTORY['prev_thrust_y'] = thrust_y
    HISTORY['prev_thrust_z'] = thrust_z

    # 9. ATTITUDE CONTROLLER (PD RANGING COMMANDS TO MAINTAIN ALIGNMENT)
    roll_rad = math.radians(HISTORY['est_roll'])
    pitch_rad = math.radians(HISTORY['est_pitch'])
    yaw_rad = math.radians(HISTORY['est_yaw'])
    
    roll_rate_rad = math.radians(roll_rate_dps)
    pitch_rate_rad = math.radians(pitch_rate_dps)
    yaw_rate_rad = math.radians(yaw_rate_dps)

    Kp_att = 0.05
    Kd_att = 0.3

    cmd_roll = -Kp_att * roll_rad - Kd_att * roll_rate_rad
    cmd_pitch = -Kp_att * pitch_rad - Kd_att * pitch_rate_rad
    cmd_yaw = -Kp_att * yaw_rad - Kd_att * yaw_rate_rad

    LIMIT = 0.035
    cmd_roll = max(-LIMIT, min(LIMIT, cmd_roll))
    cmd_pitch = max(-LIMIT, min(LIMIT, cmd_pitch))
    cmd_yaw = max(-LIMIT, min(LIMIT, cmd_yaw))

    return {
        'thrust_x': thrust_x,
        'thrust_y': thrust_y,
        'thrust_z': thrust_z,
        'att_rate_roll': cmd_roll,
        'att_rate_pitch': cmd_pitch,
        'att_rate_yaw': cmd_yaw
    }
