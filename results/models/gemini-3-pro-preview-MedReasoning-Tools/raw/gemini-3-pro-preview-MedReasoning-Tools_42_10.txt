; SPIR-V
; Version: 1.0
               OpCapability Geometry
          %ext = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
               OpExecutionMode %main InputLinesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 10
               OpExecutionMode %main Invocations 1
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %worldPos Location 0
               OpDecorate %normal Location 1
               OpDecorate %color Location 2
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %outColor Location 0

       %void = OpTypeVoid
       %func = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v3float = OpTypeVector %float 3
    %v4float = OpTypeVector %float 4
       %uint = OpTypeInt 32 0
     %uint_4 = OpConstant %uint 4
        %int = OpTypeInt 32 1
      %int_0 = OpConstant %int 0
      %int_1 = OpConstant %int 1
      %int_2 = OpConstant %int 2
      %int_3 = OpConstant %int 3
    %float_0 = OpConstant %float 0.0
    %float_0_25 = OpConstant %float 0.25
    %float_0_5 = OpConstant %float 0.5
    %float_0_75 = OpConstant %float 0.75
    %float_1 = OpConstant %float 1.0
    %float_0_02 = OpConstant %float 0.02

%gl_PerVertex_in = OpTypeStruct %v4float
  %arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
     %gl_in = OpVariable %ptr_in_arr_pv Input

   %arr_v3_4 = OpTypeArray %v3float %uint_4
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_4
   %worldPos = OpVariable %ptr_in_arr_v3 Input
     %normal = OpVariable %ptr_in_arr_v3 Input
      %color = OpVariable %ptr_in_arr_v3 Input
%ptr_in_v3 = OpTypePointer Input %v3float

%gl_PerVertex_out = OpTypeStruct %v4float
 %ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
    %gl_out = OpVariable %ptr_out_pv Output
 %ptr_out_v4 = OpTypePointer Output %v4float

 %ptr_out_v3 = OpTypePointer Output %v3float
   %outColor = OpVariable %ptr_out_v3 Output

       %main = OpFunction %void None %func
      %entry = OpLabel

               ; Load points
      %p0_ptr = OpAccessChain %ptr_in_v3 %worldPos %int_0
          %p0 = OpLoad %v3float %p0_ptr
      %p1_ptr = OpAccessChain %ptr_in_v3 %worldPos %int_1
          %p1 = OpLoad %v3float %p1_ptr
      %p2_ptr = OpAccessChain %ptr_in_v3 %worldPos %int_2
          %p2 = OpLoad %v3float %p2_ptr
      %p3_ptr = OpAccessChain %ptr_in_v3 %worldPos %int_3
          %p3 = OpLoad %v3float %p3_ptr

               ; Load colors
      %c1_ptr = OpAccessChain %ptr_in_v3 %color %int_1
          %c1 = OpLoad %v3float %c1_ptr
      %c2_ptr = OpAccessChain %ptr_in_v3 %color %int_2
          %c2 = OpLoad %v3float %c2_ptr

               ; Compute tangents
          %t0 = OpFSub %v3float %p2 %p0
          %t1 = OpFSub %v3float %p3 %p1

               ; Output pointer for Position
      %gp_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0

               ; Step 0 (t = 0.0)
               ; Compute position
         %p1_s_0 = OpVectorTimesScalar %v3float %p1 %float_1
         %p2_s_0 = OpVectorTimesScalar %v3float %p2 %float_0
          %p_t_0 = OpFAdd %v3float %p1_s_0 %p2_s_0

               ; Compute tangent
         %t0_s_0 = OpVectorTimesScalar %v3float %t0 %float_1
         %t1_s_0 = OpVectorTimesScalar %v3float %t1 %float_0
          %t_t_0 = OpFAdd %v3float %t0_s_0 %t1_s_0

               ; Normalize tangent
          %u_t_0 = OpExtInst %v3float %ext Normalize %t_t_0

               ; Compute perpendicular vector
          %u_x_0 = OpCompositeExtract %float %u_t_0 0
          %u_y_0 = OpCompositeExtract %float %u_t_0 1
       %neg_u_y_0 = OpFNegate %float %u_y_0
       %perp_u_0 = OpCompositeConstruct %v3float %neg_u_y_0 %u_x_0 %float_0
         %perp_0 = OpVectorTimesScalar %v3float %perp_u_0 %float_0_02

               ; Compute position + perp and position - perp
     %pos_plus_0 = OpFAdd %v3float %p_t_0 %perp_0
    %pos_minus_0 = OpFSub %v3float %p_t_0 %perp_0

    %pos_plus4_0 = OpCompositeConstruct %v4float %pos_plus_0 %float_1
   %pos_minus4_0 = OpCompositeConstruct %v4float %pos_minus_0 %float_1

               ; Compute color
         %c1_s_0 = OpVectorTimesScalar %v3float %c1 %float_1
         %c2_s_0 = OpVectorTimesScalar %v3float %c2 %float_0
          %c_t_0 = OpFAdd %v3float %c1_s_0 %c2_s_0

               ; Emit first vertex (plus)
               OpStore %gp_ptr %pos_plus4_0
               OpStore %outColor %c_t_0
               OpEmitVertex

               ; Emit second vertex (minus)
               OpStore %gp_ptr %pos_minus4_0
               OpStore %outColor %c_t_0
               OpEmitVertex

               ; Step 1 (t = 0.25)
               ; Compute position
         %p1_s_1 = OpVectorTimesScalar %v3float %p1 %float_0_75
         %p2_s_1 = OpVectorTimesScalar %v3float %p2 %float_0_25
          %p_t_1 = OpFAdd %v3float %p1_s_1 %p2_s_1

               ; Compute tangent
         %t0_s_1 = OpVectorTimesScalar %v3float %t0 %float_0_75
         %t1_s_1 = OpVectorTimesScalar %v3float %t1 %float_0_25
          %t_t_1 = OpFAdd %v3float %t0_s_1 %t1_s_1

               ; Normalize tangent
          %u_t_1 = OpExtInst %v3float %ext Normalize %t_t_1

               ; Compute perpendicular vector
          %u_x_1 = OpCompositeExtract %float %u_t_1 0
          %u_y_1 = OpCompositeExtract %float %u_t_1 1
       %neg_u_y_1 = OpFNegate %float %u_y_1
       %perp_u_1 = OpCompositeConstruct %v3float %neg_u_y_1 %u_x_1 %float_0
         %perp_1 = OpVectorTimesScalar %v3float %perp_u_1 %float_0_02

               ; Compute position + perp and position - perp
     %pos_plus_1 = OpFAdd %v3float %p_t_1 %perp_1
    %pos_minus_1 = OpFSub %v3float %p_t_1 %perp_1

    %pos_plus4_1 = OpCompositeConstruct %v4float %pos_plus_1 %float_1
   %pos_minus4_1 = OpCompositeConstruct %v4float %pos_minus_1 %float_1

               ; Compute color
         %c1_s_1 = OpVectorTimesScalar %v3float %c1 %float_0_75
         %c2_s_1 = OpVectorTimesScalar %v3float %c2 %float_0_25
          %c_t_1 = OpFAdd %v3float %c1_s_1 %c2_s_1

               ; Emit first vertex (plus)
               OpStore %gp_ptr %pos_plus4_1
               OpStore %outColor %c_t_1
               OpEmitVertex

               ; Emit second vertex (minus)
               OpStore %gp_ptr %pos_minus4_1
               OpStore %outColor %c_t_1
               OpEmitVertex

               ; Step 2 (t = 0.5)
               ; Compute position
         %p1_s_2 = OpVectorTimesScalar %v3float %p1 %float_0_5
         %p2_s_2 = OpVectorTimesScalar %v3float %p2 %float_0_5
          %p_t_2 = OpFAdd %v3float %p1_s_2 %p2_s_2

               ; Compute tangent
         %t0_s_2 = OpVectorTimesScalar %v3float %t0 %float_0_5
         %t1_s_2 = OpVectorTimesScalar %v3float %t1 %float_0_5
          %t_t_2 = OpFAdd %v3float %t0_s_2 %t1_s_2

               ; Normalize tangent
          %u_t_2 = OpExtInst %v3float %ext Normalize %t_t_2

               ; Compute perpendicular vector
          %u_x_2 = OpCompositeExtract %float %u_t_2 0
          %u_y_2 = OpCompositeExtract %float %u_t_2 1
       %neg_u_y_2 = OpFNegate %float %u_y_2
       %perp_u_2 = OpCompositeConstruct %v3float %neg_u_y_2 %u_x_2 %float_0
         %perp_2 = OpVectorTimesScalar %v3float %perp_u_2 %float_0_02

               ; Compute position + perp and position - perp
     %pos_plus_2 = OpFAdd %v3float %p_t_2 %perp_2
    %pos_minus_2 = OpFSub %v3float %p_t_2 %perp_2

    %pos_plus4_2 = OpCompositeConstruct %v4float %pos_plus_2 %float_1
   %pos_minus4_2 = OpCompositeConstruct %v4float %pos_minus_2 %float_1

               ; Compute color
         %c1_s_2 = OpVectorTimesScalar %v3float %c1 %float_0_5
         %c2_s_2 = OpVectorTimesScalar %v3float %c2 %float_0_5
          %c_t_2 = OpFAdd %v3float %c1_s_2 %c2_s_2

               ; Emit first vertex (plus)
               OpStore %gp_ptr %pos_plus4_2
               OpStore %outColor %c_t_2
               OpEmitVertex

               ; Emit second vertex (minus)
               OpStore %gp_ptr %pos_minus4_2
               OpStore %outColor %c_t_2
               OpEmitVertex

               ; Step 3 (t = 0.75)
               ; Compute position
         %p1_s_3 = OpVectorTimesScalar %v3float %p1 %float_0_25
         %p2_s_3 = OpVectorTimesScalar %v3float %p2 %float_0_75
          %p_t_3 = OpFAdd %v3float %p1_s_3 %p2_s_3

               ; Compute tangent
         %t0_s_3 = OpVectorTimesScalar %v3float %t0 %float_0_25
         %t1_s_3 = OpVectorTimesScalar %v3float %t1 %float_0_75
          %t_t_3 = OpFAdd %v3float %t0_s_3 %t1_s_3

               ; Normalize tangent
          %u_t_3 = OpExtInst %v3float %ext Normalize %t_t_3

               ; Compute perpendicular vector
          %u_x_3 = OpCompositeExtract %float %u_t_3 0
          %u_y_3 = OpCompositeExtract %float %u_t_3 1
       %neg_u_y_3 = OpFNegate %float %u_y_3
       %perp_u_3 = OpCompositeConstruct %v3float %neg_u_y_3 %u_x_3 %float_0
         %perp_3 = OpVectorTimesScalar %v3float %perp_u_3 %float_0_02

               ; Compute position + perp and position - perp
     %pos_plus_3 = OpFAdd %v3float %p_t_3 %perp_3
    %pos_minus_3 = OpFSub %v3float %p_t_3 %perp_3

    %pos_plus4_3 = OpCompositeConstruct %v4float %pos_plus_3 %float_1
   %pos_minus4_3 = OpCompositeConstruct %v4float %pos_minus_3 %float_1

               ; Compute color
         %c1_s_3 = OpVectorTimesScalar %v3float %c1 %float_0_25
         %c2_s_3 = OpVectorTimesScalar %v3float %c2 %float_0_75
          %c_t_3 = OpFAdd %v3float %c1_s_3 %c2_s_3

               ; Emit first vertex (plus)
               OpStore %gp_ptr %pos_plus4_3
               OpStore %outColor %c_t_3
               OpEmitVertex

               ; Emit second vertex (minus)
               OpStore %gp_ptr %pos_minus4_3
               OpStore %outColor %c_t_3
               OpEmitVertex

               ; Step 4 (t = 1.0)
               ; Compute position
         %p1_s_4 = OpVectorTimesScalar %v3float %p1 %float_0
         %p2_s_4 = OpVectorTimesScalar %v3float %p2 %float_1
          %p_t_4 = OpFAdd %v3float %p1_s_4 %p2_s_4

               ; Compute tangent
         %t0_s_4 = OpVectorTimesScalar %v3float %t0 %float_0
         %t1_s_4 = OpVectorTimesScalar %v3float %t1 %float_1
          %t_t_4 = OpFAdd %v3float %t0_s_4 %t1_s_4

               ; Normalize tangent
          %u_t_4 = OpExtInst %v3float %ext Normalize %t_t_4

               ; Compute perpendicular vector
          %u_x_4 = OpCompositeExtract %float %u_t_4 0
          %u_y_4 = OpCompositeExtract %float %u_t_4 1
       %neg_u_y_4 = OpFNegate %float %u_y_4
       %perp_u_4 = OpCompositeConstruct %v3float %neg_u_y_4 %u_x_4 %float_0
         %perp_4 = OpVectorTimesScalar %v3float %perp_u_4 %float_0_02

               ; Compute position + perp and position - perp
     %pos_plus_4 = OpFAdd %v3float %p_t_4 %perp_4
    %pos_minus_4 = OpFSub %v3float %p_t_4 %perp_4

    %pos_plus4_4 = OpCompositeConstruct %v4float %pos_plus_4 %float_1
   %pos_minus4_4 = OpCompositeConstruct %v4float %pos_minus_4 %float_1

               ; Compute color
         %c1_s_4 = OpVectorTimesScalar %v3float %c1 %float_0
         %c2_s_4 = OpVectorTimesScalar %v3float %c2 %float_1
          %c_t_4 = OpFAdd %v3float %c1_s_4 %c2_s_4

               ; Emit first vertex (plus)
               OpStore %gp_ptr %pos_plus4_4
               OpStore %outColor %c_t_4
               OpEmitVertex

               ; Emit second vertex (minus)
               OpStore %gp_ptr %pos_minus4_4
               OpStore %outColor %c_t_4
               OpEmitVertex

               OpEndPrimitive
               OpReturn
               OpFunctionEnd
