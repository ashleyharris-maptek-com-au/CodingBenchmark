; SPIR-V
; Version: 1.0
; Generator: Antigravity
               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %color %gl_out %outColor
               OpExecutionMode %main InputLinesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 8
               OpExecutionMode %main Invocations 1

; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %color Location 2
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %outColor Location 0

       %void = OpTypeVoid
       %func = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v2float = OpTypeVector %float 2
    %v3float = OpTypeVector %float 3
    %v4float = OpTypeVector %float 4

       %uint = OpTypeInt 32 0
     %uint_4 = OpConstant %uint 4
        %int = OpTypeInt 32 1
      %int_0 = OpConstant %int 0
      %int_1 = OpConstant %int 1
      %int_2 = OpConstant %int 2
      %int_3 = OpConstant %int 3

; Input types
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
     %gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3_in = OpTypeArray %v3float %uint_4
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
      %color = OpVariable %ptr_in_arr_v3 Input

; Output types
%gl_PerVertex_out = OpTypeStruct %v4float
 %ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
     %gl_out = OpVariable %ptr_out_pv Output

 %ptr_out_v3 = OpTypePointer Output %v3float
   %outColor = OpVariable %ptr_out_v3 Output

; Pointers for access chains
  %ptr_in_v4 = OpTypePointer Input %v4float
  %ptr_in_v3 = OpTypePointer Input %v3float
 %ptr_out_v4 = OpTypePointer Output %v4float

; Constants used in math
    %float_0 = OpConstant %float 0
  %float_0_5 = OpConstant %float 0.5
    %float_1 = OpConstant %float 1
   %float_0_02 = OpConstant %float 0.02
  %float_1_3 = OpConstant %float 0.333333333
  %float_2_3 = OpConstant %float 0.666666667

       %main = OpFunction %void None %func
      %entry = OpLabel
; Read positions
    %p0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
        %p0 = OpLoad %v4float %p0_ptr
    %p1_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
     %start = OpLoad %v4float %p1_ptr
    %p2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
       %end = OpLoad %v4float %p2_ptr
    %p3_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
        %p3 = OpLoad %v4float %p3_ptr

; Adjacency tangents
        %v0 = OpCompositeExtract %float %p0 0
        %v1 = OpCompositeExtract %float %p0 1
      %p0_2d = OpCompositeConstruct %v2float %v0 %v1

        %v2 = OpCompositeExtract %float %start 0
        %v3 = OpCompositeExtract %float %start 1
    %start_2d = OpCompositeConstruct %v2float %v2 %v3

        %v4 = OpCompositeExtract %float %end 0
        %v5 = OpCompositeExtract %float %end 1
      %end_2d = OpCompositeConstruct %v2float %v4 %v5

        %v6 = OpCompositeExtract %float %p3 0
        %v7 = OpCompositeExtract %float %p3 1
      %p3_2d = OpCompositeConstruct %v2float %v6 %v7

     %diff0 = OpFSub %v2float %end_2d %p0_2d
        %t0 = OpExtInst %v2float %1 Normalize %diff0

     %diff1 = OpFSub %v2float %p3_2d %start_2d
        %t1 = OpExtInst %v2float %1 Normalize %diff1

     %t0_x = OpCompositeExtract %float %t0 0
     %t0_y = OpCompositeExtract %float %t0 1
    %neg_t0_y = OpFNegate %float %t0_y
     %perp0 = OpCompositeConstruct %v2float %neg_t0_y %t0_x

     %t1_x = OpCompositeExtract %float %t1 0
     %t1_y = OpCompositeExtract %float %t1 1
    %neg_t1_y = OpFNegate %float %t1_y
     %perp1 = OpCompositeConstruct %v2float %neg_t1_y %t1_x

; Read colors
    %c1_ptr = OpAccessChain %ptr_in_v3 %color %int_1
        %c1 = OpLoad %v3float %c1_ptr
    %c2_ptr = OpAccessChain %ptr_in_v3 %color %int_2
        %c2 = OpLoad %v3float %c2_ptr

; Interpolate using mix formula
; pos_1_3 = start * 2/3 + end * 1/3
   %start_2_3 = OpVectorTimesScalar %v4float %start %float_2_3
   %end_1_3 = OpVectorTimesScalar %v4float %end %float_1_3
   %pos_1_3 = OpFAdd %v4float %start_2_3 %end_1_3

; perp_1_3 = perp0 * 2/3 + perp1 * 1/3
   %perp0_2_3 = OpVectorTimesScalar %v2float %perp0 %float_2_3
   %perp1_1_3 = OpVectorTimesScalar %v2float %perp1 %float_1_3
   %perp_1_3 = OpFAdd %v2float %perp0_2_3 %perp1_1_3

; col_1_3 = c1 * 2/3 + c2 * 1/3
   %c1_2_3 = OpVectorTimesScalar %v3float %c1 %float_2_3
   %c2_1_3 = OpVectorTimesScalar %v3float %c2 %float_1_3
   %col_1_3 = OpFAdd %v3float %c1_2_3 %c2_1_3

; pos_2_3 = start * 1/3 + end * 2/3
   %start_1_3 = OpVectorTimesScalar %v4float %start %float_1_3
   %end_2_3 = OpVectorTimesScalar %v4float %end %float_2_3
   %pos_2_3 = OpFAdd %v4float %start_1_3 %end_2_3

; perp_2_3 = perp0 * 1/3 + perp1 * 2/3
   %perp0_1_3 = OpVectorTimesScalar %v2float %perp0 %float_1_3
   %perp1_2_3 = OpVectorTimesScalar %v2float %perp1 %float_2_3
   %perp_2_3 = OpFAdd %v2float %perp0_1_3 %perp1_2_3

; col_2_3 = c1 * 1/3 + c2 * 2/3
   %c1_1_3 = OpVectorTimesScalar %v3float %c1 %float_1_3
   %c2_2_3 = OpVectorTimesScalar %v3float %c2 %float_2_3
   %col_2_3 = OpFAdd %v3float %c1_1_3 %c2_2_3

; Offsets
   %perp0_scaled = OpVectorTimesScalar %v2float %perp0 %float_0_02
   %perp0_x = OpCompositeExtract %float %perp0_scaled 0
   %perp0_y = OpCompositeExtract %float %perp0_scaled 1
   %offset0 = OpCompositeConstruct %v4float %perp0_x %perp0_y %float_0 %float_0

   %pos_v0 = OpFAdd %v4float %start %offset0
   %pos_v1 = OpFSub %v4float %start %offset0

   %perp_1_3_scaled = OpVectorTimesScalar %v2float %perp_1_3 %float_0_02
   %perp_1_3_x = OpCompositeExtract %float %perp_1_3_scaled 0
   %perp_1_3_y = OpCompositeExtract %float %perp_1_3_scaled 1
   %offset_1_3 = OpCompositeConstruct %v4float %perp_1_3_x %perp_1_3_y %float_0 %float_0

   %pos_v2 = OpFAdd %v4float %pos_1_3 %offset_1_3
   %pos_v3 = OpFSub %v4float %pos_1_3 %offset_1_3

   %perp_2_3_scaled = OpVectorTimesScalar %v2float %perp_2_3 %float_0_02
   %perp_2_3_x = OpCompositeExtract %float %perp_2_3_scaled 0
   %perp_2_3_y = OpCompositeExtract %float %perp_2_3_scaled 1
   %offset_2_3 = OpCompositeConstruct %v4float %perp_2_3_x %perp_2_3_y %float_0 %float_0

   %pos_v4 = OpFAdd %v4float %pos_2_3 %offset_2_3
   %pos_v5 = OpFSub %v4float %pos_2_3 %offset_2_3

   %perp1_scaled = OpVectorTimesScalar %v2float %perp1 %float_0_02
   %perp1_x = OpCompositeExtract %float %perp1_scaled 0
   %perp1_y = OpCompositeExtract %float %perp1_scaled 1
   %offset1 = OpCompositeConstruct %v4float %perp1_x %perp1_y %float_0 %float_0

   %pos_v6 = OpFAdd %v4float %end %offset1
   %pos_v7 = OpFSub %v4float %end %offset1

; Emit
   %gp_out_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0

; Quad 1
               OpStore %gp_out_ptr %pos_v0
               OpStore %outColor %c1
               OpEmitVertex

               OpStore %gp_out_ptr %pos_v1
               OpStore %outColor %c1
               OpEmitVertex

               OpStore %gp_out_ptr %pos_v2
               OpStore %outColor %col_1_3
               OpEmitVertex

               OpStore %gp_out_ptr %pos_v3
               OpStore %outColor %col_1_3
               OpEmitVertex

               OpEndPrimitive

; Quad 2
               OpStore %gp_out_ptr %pos_v4
               OpStore %outColor %col_2_3
               OpEmitVertex

               OpStore %gp_out_ptr %pos_v5
               OpStore %outColor %col_2_3
               OpEmitVertex

               OpStore %gp_out_ptr %pos_v6
               OpStore %outColor %c2
               OpEmitVertex

               OpStore %gp_out_ptr %pos_v7
               OpStore %outColor %c2
               OpEmitVertex

               OpEndPrimitive

               OpReturn
               OpFunctionEnd
