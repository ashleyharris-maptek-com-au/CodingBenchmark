const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    if (!input || input.trim() === '') {
        return;
    }
    const data = JSON.parse(input);
    const width = data.probability[0].length;
    const height = data.probability.length;
    const drones = data.drones;
    const maxSteps = data.maxStepsPerDrone;
    const probability = data.probability;

    // Helper to get cells in radius R (Manhattan distance)
    function getNeighborsInRadius(x, y, R) {
        const neighbors = [];
        for (let dy = -R; dy <= R; dy++) {
            const limit = R - Math.abs(dy);
            for (let dx = -limit; dx <= limit; dx++) {
                const nx = x + dx;
                const ny = y + dy;
                if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                    neighbors.push([nx, ny]);
                }
            }
        }
        return neighbors;
    }

    // BFS to find shortest paths from start to all cells
    function bfs(startX, startY) {
        const dist = Array.from({ length: height }, () => new Int32Array(width).fill(-1));
        const parent = Array.from({ length: height }, () => new Int32Array(width).fill(-1));
        const queueX = new Int32Array(width * height);
        const queueY = new Int32Array(width * height);
        let head = 0, tail = 0;

        queueX[tail] = startX;
        queueY[tail] = startY;
        dist[startY][startX] = 0;
        tail++;

        const dx = [1, -1, 0, 0];
        const dy = [0, 0, 1, -1];

        while (head < tail) {
            const cx = queueX[head];
            const cy = queueY[head];
            head++;

            const d = dist[cy][cx];
            for (let i = 0; i < 4; i++) {
                const nx = cx + dx[i];
                const ny = cy + dy[i];
                if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                    if (probability[ny][nx] >= 0 && dist[ny][nx] === -1) {
                        dist[ny][nx] = d + 1;
                        parent[ny][nx] = cy * width + cx;
                        queueX[tail] = nx;
                        queueY[tail] = ny;
                        tail++;
                    }
                }
            }
        }
        return { dist, parent };
    }

    // Evaluate coverage score for a planned set of paths
    function getCoverageScore(paths) {
        const covered = Array.from({ length: height }, () => new Uint8Array(width));
        let score = 0;
        for (let d = 0; d < paths.length; d++) {
            const path = paths[d];
            for (let i = 0; i < path.length; i++) {
                const [x, y] = path[i];
                if (!covered[y][x]) {
                    covered[y][x] = 1;
                    const val = probability[y][x];
                    if (val > 0) {
                        score += val;
                    }
                }
            }
        }
        return score;
    }

    // Run the solver with specific parameters
    function runSimulation(R, beta, K) {
        // Precompute neighbors in radius R
        const neighborsGrid = [];
        for (let y = 0; y < height; y++) {
            const row = [];
            for (let x = 0; x < width; x++) {
                row.push(getNeighborsInRadius(x, y, R));
            }
            neighborsGrid.push(row);
        }

        const isCovered = Array.from({ length: height }, () => new Uint8Array(width));
        for (let d = 0; d < drones.length; d++) {
            isCovered[drones[d].y][drones[d].x] = 1;
        }

        // Initialize density map
        const density = Array.from({ length: height }, () => new Float32Array(width));
        for (let y = 0; y < height; y++) {
            for (let x = 0; x < width; x++) {
                let sum = 0;
                const nb = neighborsGrid[y][x];
                for (let i = 0; i < nb.length; i++) {
                    const [nx, ny] = nb[i];
                    if (!isCovered[ny][nx] && probability[ny][nx] > 0) {
                        sum += probability[ny][nx];
                    }
                }
                density[y][x] = sum;
            }
        }

        const paths = [];
        for (let d = 0; d < drones.length; d++) {
            const drone = drones[d];
            let cx = drone.x;
            let cy = drone.y;
            const path = [[cx, cy]];

            let targetX = -1;
            let targetY = -1;
            let pathToTarget = [];
            let pathIdx = 0;
            let stepsSinceLastEval = 0;

            for (let step = 0; step < maxSteps; step++) {
                const remSteps = maxSteps - step;

                let needNewTarget = false;
                if (targetX === -1 || targetY === -1) {
                    needNewTarget = true;
                } else if (cx === targetX && cy === targetY) {
                    needNewTarget = true;
                } else if (isCovered[targetY][targetX] || density[targetY][targetX] <= 0) {
                    needNewTarget = true;
                } else if (stepsSinceLastEval >= K) {
                    needNewTarget = true;
                } else if (pathIdx >= pathToTarget.length) {
                    needNewTarget = true;
                }

                if (needNewTarget) {
                    const { dist, parent } = bfs(cx, cy);

                    let bestTX = -1, bestTY = -1;
                    let maxScore = -1;

                    for (let y = 0; y < height; y++) {
                        for (let x = 0; x < width; x++) {
                            const dVal = dist[y][x];
                            if (dVal > 0 && dVal <= remSteps) {
                                const dens = density[y][x];
                                if (dens > 0) {
                                    const score = dens / (dVal + beta);
                                    if (score > maxScore) {
                                        maxScore = score;
                                        bestTX = x;
                                        bestTY = y;
                                    }
                                }
                            }
                        }
                    }

                    if (bestTX !== -1) {
                        targetX = bestTX;
                        targetY = bestTY;
                        const tempPath = [];
                        let tx = targetX, ty = targetY;
                        while (tx !== cx || ty !== cy) {
                            tempPath.push([tx, ty]);
                            const p = parent[ty][tx];
                            tx = p % width;
                            ty = Math.floor(p / width);
                        }
                        tempPath.reverse();
                        pathToTarget = tempPath;
                        pathIdx = 0;
                        stepsSinceLastEval = 0;
                    } else {
                        targetX = -1;
                        targetY = -1;
                        pathToTarget = [];
                        pathIdx = 0;
                    }
                }

                let nextX = cx;
                let nextY = cy;

                if (pathIdx < pathToTarget.length) {
                    const nextCell = pathToTarget[pathIdx];
                    nextX = nextCell[0];
                    nextY = nextCell[1];
                    pathIdx++;
                    stepsSinceLastEval++;
                } else {
                    const dx = [1, -1, 0, 0];
                    const dy = [0, 0, 1, -1];
                    let bestNeighborX = cx;
                    let bestNeighborY = cy;
                    let maxP = -1;

                    for (let i = 0; i < 4; i++) {
                        const nx = cx + dx[i];
                        const ny = cy + dy[i];
                        if (nx >= 0 && nx < width && ny >= 0 && ny < height && probability[ny][nx] >= 0) {
                            const pVal = isCovered[ny][nx] ? 0 : probability[ny][nx];
                            if (pVal > maxP) {
                                maxP = pVal;
                                bestNeighborX = nx;
                                bestNeighborY = ny;
                            }
                        }
                    }
                    nextX = bestNeighborX;
                    nextY = bestNeighborY;
                    stepsSinceLastEval = 0;
                }

                cx = nextX;
                cy = nextY;
                path.push([cx, cy]);

                if (!isCovered[cy][cx]) {
                    isCovered[cy][cx] = 1;
                    const val = probability[cy][cx];
                    if (val > 0) {
                        const nb = neighborsGrid[cy][cx];
                        for (let i = 0; i < nb.length; i++) {
                            const [nx, ny] = nb[i];
                            density[ny][nx] -= val;
                        }
                    }
                }
            }
            paths.push(path);
        }
        return paths;
    }

    const configs = [
        { R: 1, beta: 1, K: 1 },
        { R: 1, beta: 3, K: 3 },
        { R: 2, beta: 1.5, K: 2 },
        { R: 2, beta: 3, K: 3 },
        { R: 3, beta: 2, K: 3 },
        { R: 3, beta: 5, K: 5 },
        { R: 4, beta: 5, K: 5 },
        { R: 5, beta: 10, K: 5 }
    ];

    let bestPaths = null;
    let bestScore = -1;

    for (let c = 0; c < configs.length; c++) {
        const { R, beta, K } = configs[c];
        const paths = runSimulation(R, beta, K);
        const score = getCoverageScore(paths);
        if (score > bestScore) {
            bestScore = score;
            bestPaths = paths;
        }
    }

    console.log(JSON.stringify({ paths: bestPaths }));
}

solve();
