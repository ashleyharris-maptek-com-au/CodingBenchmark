// Custom fast parser using fread
class FastInput {
    static const int BUF_SIZE = 1 << 20;
    std::vector<char> buf;
    int buf_pos = 0;
    int buf_len = 0;

    char get_char() {
        if (buf.empty()) buf.resize(BUF_SIZE);
        if (buf_pos >= buf_len) {
            buf_len = fread(buf.data(), 1, BUF_SIZE, stdin);
            buf_pos = 0;
            if (buf_len == 0) return EOF;
        }
        return buf[buf_pos++];
    }

public:
    int read_int() {
        int x = 0;
        char c = get_char();
        while (c <= 32 && c != EOF) c = get_char();
        if (c == EOF) return 0;
        bool neg = false;
        if (c == '-') {
            neg = true;
            c = get_char();
        }
        while (c >= '0' && c <= '9') {
            x = x * 10 + (c - '0');
            c = get_char();
        }
        return neg ? -x : x;
    }

    double read_double() {
        char c = get_char();
        while (c <= 32 && c != EOF) c = get_char();
        if (c == EOF) return 0.0;
        bool neg = false;
        if (c == '-') {
            neg = true;
            c = get_char();
        }
        double x = 0.0;
        while (c >= '0' && c <= '9') {
            x = x * 10.0 + (c - '0');
            c = get_char();
        }
        if (c == '.') {
            c = get_char();
            double dec = 0.1;
            while (c >= '0' && c <= '9') {
                x += (c - '0') * dec;
                dec *= 0.1;
                c = get_char();
            }
        }
        if (c == 'e' || c == 'E') {
            c = get_char();
            bool exp_neg = false;
            if (c == '-') {
                exp_neg = true;
                c = get_char();
            } else if (c == '+') {
                c = get_char();
            }
            int exp = 0;
            while (c >= '0' && c <= '9') {
                exp = exp * 10 + (c - '0');
                c = get_char();
            }
            if (exp_neg) {
                x /= std::pow(10.0, exp);
            } else {
                x *= std::pow(10.0, exp);
            }
        }
        return neg ? -x : x;
    }
};

class FastOutput {
    static const int BUF_SIZE = 1 << 20;
    std::vector<char> buf;
    int buf_pos = 0;

    void flush() {
        if (buf_pos > 0) {
            fwrite(buf.data(), 1, buf_pos, stdout);
            buf_pos = 0;
        }
    }

    void put_char(char c) {
        if (buf.empty()) buf.resize(BUF_SIZE);
        if (buf_pos >= BUF_SIZE) flush();
        buf[buf_pos++] = c;
    }

public:
    ~FastOutput() {
        flush();
    }

    void write_int(long long x) {
        if (x < 0) {
            put_char('-');
            x = -x;
        }
        char temp[30];
        int idx = 0;
        do {
            temp[idx++] = (x % 10) + '0';
            x /= 10;
        } while (x > 0);
        while (idx > 0) {
            put_char(temp[--idx]);
        }
    }

    void write_double(double x) {
        if (std::isnan(x)) {
            write_string("nan");
            return;
        }
        if (std::isinf(x)) {
            if (x < 0) put_char('-');
            write_string("inf");
            return;
        }
        if (x < 0.0) {
            put_char('-');
            x = -x;
        }
        double integer_part;
        double fractional_part = std::modf(x, &integer_part);
        
        unsigned long long int_val = (unsigned long long)integer_part;
        write_int((long long)int_val);
        
        put_char('.');
        
        for (int i = 0; i < 10; ++i) {
            fractional_part *= 10.0;
            int digit = (int)std::floor(fractional_part);
            put_char(digit + '0');
            fractional_part -= digit;
        }
    }

    void write_string(const char* s) {
        while (*s) {
            put_char(*s++);
        }
    }
    
    void write_newline() {
        put_char('\n');
    }
    
    void write_space() {
        put_char(' ');
    }
};

struct Point3D {
    double x, y, z;
    Point3D operator+(const Point3D& o) const { return {x + o.x, y + o.y, z + o.z}; }
    Point3D operator-(const Point3D& o) const { return {x - o.x, y - o.y, z - o.z}; }
    Point3D operator*(double s) const { return {x * s, y * s, z * s}; }
    double dot(const Point3D& o) const { return x * o.x + y * o.y + z * o.z; }
    Point3D cross(const Point3D& o) const {
        return {
            y * o.z - z * o.y,
            z * o.x - x * o.z,
            x * o.y - y * o.x
        };
    }
    double normSq() const { return x * x + y * y + z * z; }
    double norm() const { return std::sqrt(normSq()); }
};

struct Face {
    int v[3];
};

struct Mesh {
    std::vector<Point3D> vertices;
    std::vector<Face> faces;
};

struct AABB {
    double min_x, min_y, min_z;
    double max_x, max_y, max_z;
    bool intersects(const AABB& other) const {
        return min_x <= other.max_x && max_x >= other.min_x &&
               min_y <= other.max_y && max_y >= other.min_y &&
               min_z <= other.max_z && max_z >= other.min_z;
    }
    bool contains(const Point3D& p) const {
        return p.x >= min_x && p.x <= max_x &&
               p.y >= min_y && p.y <= max_y &&
               p.z >= min_z && p.z <= max_z;
    }
};

struct BVHNode {
    AABB box;
    int left_child = -1;
    int right_child = -1;
    std::vector<int> face_indices;
    bool is_leaf() const { return left_child == -1 && right_child == -1; }
};

class BVH {
public:
    std::vector<BVHNode> nodes;
    int root = -1;
    
    void build(const Mesh& mesh) {
        std::vector<int> indices(mesh.faces.size());
        std::iota(indices.begin(), indices.end(), 0);
        nodes.clear();
        if (indices.empty()) return;
        root = build_recursive(mesh, indices, 0, indices.size());
    }
    
    template<typename Callback>
    void intersect_ray(const Point3D& p, const Point3D& dir, Callback callback) const {
        if (root == -1) return;
        intersect_ray_recursive(root, p, dir, callback);
    }
    
    template<typename Callback>
    void intersect_box(const AABB& box, Callback callback) const {
        if (root == -1) return;
        intersect_box_recursive(root, box, callback);
    }

private:
    int build_recursive(const Mesh& mesh, std::vector<int>& indices, int start, int end) {
        int node_idx = nodes.size();
        nodes.emplace_back();
        
        AABB box = compute_aabb(mesh, indices, start, end);
        nodes[node_idx].box = box;
        
        int count = end - start;
        if (count <= 4) {
            nodes[node_idx].face_indices.assign(indices.begin() + start, indices.begin() + end);
            return node_idx;
        }
        
        double dx = box.max_x - box.min_x;
        double dy = box.max_y - box.min_y;
        double dz = box.max_z - box.min_z;
        int axis = 0;
        if (dy > dx && dy > dz) axis = 1;
        else if (dz > dx && dz > dy) axis = 2;
        
        auto get_centroid = [&](int idx) {
            const auto& F = mesh.faces[idx];
            return (mesh.vertices[F.v[0]] + mesh.vertices[F.v[1]] + mesh.vertices[F.v[2]]) * (1.0 / 3.0);
        };
        
        std::sort(indices.begin() + start, indices.begin() + end, [&](int a, int b) {
            Point3D ca = get_centroid(a);
            Point3D cb = get_centroid(b);
            if (axis == 0) return ca.x < cb.x;
            if (axis == 1) return ca.y < cb.y;
            return ca.z < cb.z;
        });
        
        int mid = start + count / 2;
        int left = build_recursive(mesh, indices, start, mid);
        int right = build_recursive(mesh, indices, mid, end);
        
        nodes[node_idx].left_child = left;
        nodes[node_idx].right_child = right;
        return node_idx;
    }
    
    AABB compute_aabb(const Mesh& mesh, const std::vector<int>& indices, int start, int end) {
        AABB box = {
            std::numeric_limits<double>::infinity(), std::numeric_limits<double>::infinity(), std::numeric_limits<double>::infinity(),
            -std::numeric_limits<double>::infinity(), -std::numeric_limits<double>::infinity(), -std::numeric_limits<double>::infinity()
        };
        for (int i = start; i < end; ++i) {
            const auto& F = mesh.faces[indices[i]];
            for (int j = 0; j < 3; ++j) {
                const auto& v = mesh.vertices[F.v[j]];
                box.min_x = std::min(box.min_x, v.x);
                box.min_y = std::min(box.min_y, v.y);
                box.min_z = std::min(box.min_z, v.z);
                box.max_x = std::max(box.max_x, v.x);
                box.max_y = std::max(box.max_y, v.y);
                box.max_z = std::max(box.max_z, v.z);
            }
        }
        return box;
    }
    
    bool ray_intersects_aabb(const Point3D& p, const Point3D& dir, const AABB& box) const {
        double tmin = 0.0, tmax = std::numeric_limits<double>::infinity();
        
        if (std::abs(dir.x) < 1e-15) {
            if (p.x < box.min_x || p.x > box.max_x) return false;
        } else {
            double t1 = (box.min_x - p.x) / dir.x;
            double t2 = (box.max_x - p.x) / dir.x;
            if (t1 > t2) std::swap(t1, t2);
            tmin = std::max(tmin, t1);
            tmax = std::min(tmax, t2);
            if (tmin > tmax) return false;
        }
        
        if (std::abs(dir.y) < 1e-15) {
            if (p.y < box.min_y || p.y > box.max_y) return false;
        } else {
            double t1 = (box.min_y - p.y) / dir.y;
            double t2 = (box.max_y - p.y) / dir.y;
            if (t1 > t2) std::swap(t1, t2);
            tmin = std::max(tmin, t1);
            tmax = std::min(tmax, t2);
            if (tmin > tmax) return false;
        }
        
        if (std::abs(dir.z) < 1e-15) {
            if (p.z < box.min_z || p.z > box.max_z) return false;
        } else {
            double t1 = (box.min_z - p.z) / dir.z;
            double t2 = (box.max_z - p.z) / dir.z;
            if (t1 > t2) std::swap(t1, t2);
            tmin = std::max(tmin, t1);
            tmax = std::min(tmax, t2);
            if (tmin > tmax) return false;
        }
        return true;
    }
    
    template<typename Callback>
    void intersect_ray_recursive(int node_idx, const Point3D& p, const Point3D& dir, Callback callback) const {
        const auto& node = nodes[node_idx];
        if (!ray_intersects_aabb(p, dir, node.box)) return;
        
        if (node.is_leaf()) {
            for (int idx : node.face_indices) {
                callback(idx);
            }
        } else {
            intersect_ray_recursive(node.left_child, p, dir, callback);
            intersect_ray_recursive(node.right_child, p, dir, callback);
        }
    }
    
    template<typename Callback>
    void intersect_box_recursive(int node_idx, const AABB& box, Callback callback) const {
        const auto& node = nodes[node_idx];
        if (!node.box.intersects(box)) return;
        
        if (node.is_leaf()) {
            for (int idx : node.face_indices) {
                callback(idx);
            }
        } else {
            intersect_box_recursive(node.left_child, box, callback);
            intersect_box_recursive(node.right_child, box, callback);
        }
    }
};

void intersect_bvhs(const Mesh& meshA, const BVH& bvhA, int nodeA,
                    const Mesh& meshB, const BVH& bvhB, int nodeB,
                    std::vector<std::pair<int, int>>& intersecting_pairs) {
    const auto& nA = bvhA.nodes[nodeA];
    const auto& nB = bvhB.nodes[nodeB];
    if (!nA.box.intersects(nB.box)) return;
    
    if (nA.is_leaf() && nB.is_leaf()) {
        for (int idxA : nA.face_indices) {
            for (int idxB : nB.face_indices) {
                intersecting_pairs.push_back({idxA, idxB});
            }
        }
    } else if (nA.is_leaf()) {
        intersect_bvhs(meshA, bvhA, nodeA, meshB, bvhB, nB.left_child, intersecting_pairs);
        intersect_bvhs(meshA, bvhA, nodeA, meshB, bvhB, nB.right_child, intersecting_pairs);
    } else if (nB.is_leaf()) {
        intersect_bvhs(meshA, bvhA, nA.left_child, meshB, bvhB, nodeB, intersecting_pairs);
        intersect_bvhs(meshA, bvhA, nA.right_child, meshB, bvhB, nodeB, intersecting_pairs);
    } else {
        double volA = (nA.box.max_x - nA.box.min_x) * (nA.box.max_y - nA.box.min_y) * (nA.box.max_z - nA.box.min_z);
        double volB = (nB.box.max_x - nB.box.min_x) * (nB.box.max_y - nB.box.min_y) * (nB.box.max_z - nB.box.min_z);
        if (volA > volB) {
            intersect_bvhs(meshA, bvhA, nA.left_child, meshB, bvhB, nodeB, intersecting_pairs);
            intersect_bvhs(meshA, bvhA, nA.right_child, meshB, bvhB, nodeB, intersecting_pairs);
        } else {
            intersect_bvhs(meshA, bvhA, nodeA, meshB, bvhB, nB.left_child, intersecting_pairs);
            intersect_bvhs(meshA, bvhA, nodeA, meshB, bvhB, nB.right_child, intersecting_pairs);
        }
    }
}

struct Triangle3D {
    Point3D p[3];
};

double distSq(const Point3D& a, const Point3D& b) {
    double dx = a.x - b.x;
    double dy = a.y - b.y;
    double dz = a.z - b.z;
    return dx * dx + dy * dy + dz * dz;
}

std::vector<Point3D> get_plane_intersection_points(const Triangle3D& T, const Point3D& N2, double d2) {
    double d[3];
    for (int i = 0; i < 3; ++i) {
        d[i] = T.p[i].dot(N2) + d2;
    }
    std::vector<Point3D> pts;
    for (int i = 0; i < 3; ++i) {
        int next = (i + 1) % 3;
        if (std::abs(d[i]) < 1e-9) {
            pts.push_back(T.p[i]);
        } else if ((d[i] > 0 && d[next] < 0) || (d[i] < 0 && d[next] > 0)) {
            double s = -d[i] / (d[next] - d[i]);
            s = std::max(0.0, std::min(1.0, s));
            pts.push_back(T.p[i] + (T.p[next] - T.p[i]) * s);
        }
    }
    std::vector<Point3D> unique_pts;
    for (const auto& p : pts) {
        bool duplicate = false;
        for (const auto& up : unique_pts) {
            if ((p - up).normSq() < 1e-18) {
                duplicate = true;
                break;
            }
        }
        if (!duplicate) {
            unique_pts.push_back(p);
        }
    }
    return unique_pts;
}

bool intersect_triangles(const Triangle3D& T1, const Triangle3D& T2, Point3D& pStart, Point3D& pEnd) {
    Point3D N1 = (T1.p[1] - T1.p[0]).cross(T1.p[2] - T1.p[0]);
    double len1 = N1.norm();
    if (len1 < 1e-12) return false;
    N1 = N1 * (1.0 / len1);
    double d1 = -N1.dot(T1.p[0]);
    
    double d2_vals[3];
    bool has_pos2 = false, has_neg2 = false;
    for (int i = 0; i < 3; ++i) {
        d2_vals[i] = N1.dot(T2.p[i]) + d1;
        if (d2_vals[i] > 1e-9) has_pos2 = true;
        else if (d2_vals[i] < -1e-9) has_neg2 = true;
    }
    if (!has_pos2 || !has_neg2) return false;
    
    Point3D N2 = (T2.p[1] - T2.p[0]).cross(T2.p[2] - T2.p[0]);
    double len2 = N2.norm();
    if (len2 < 1e-12) return false;
    N2 = N2 * (1.0 / len2);
    double d2 = -N2.dot(T2.p[0]);
    
    double d1_vals[3];
    bool has_pos1 = false, has_neg1 = false;
    for (int i = 0; i < 3; ++i) {
        d1_vals[i] = N2.dot(T1.p[i]) + d2;
        if (d1_vals[i] > 1e-9) has_pos1 = true;
        else if (d1_vals[i] < -1e-9) has_neg1 = true;
    }
    if (!has_pos1 || !has_neg1) return false;
    
    std::vector<Point3D> A_pts = get_plane_intersection_points(T1, N2, d2);
    std::vector<Point3D> B_pts = get_plane_intersection_points(T2, N1, d1);
    if (A_pts.size() != 2 || B_pts.size() != 2) return false;
    
    Point3D D = N1.cross(N2);
    double tA1 = A_pts[0].dot(D);
    double tA2 = A_pts[1].dot(D);
    Point3D A1 = A_pts[0], A2 = A_pts[1];
    if (tA1 > tA2) { std::swap(tA1, tA2); std::swap(A1, A2); }
    
    double tB1 = B_pts[0].dot(D);
    double tB2 = B_pts[1].dot(D);
    Point3D B1 = B_pts[0], B2 = B_pts[1];
    if (tB1 > tB2) { std::swap(tB1, tB2); std::swap(B1, B2); }
    
    double tStart = std::max(tA1, tB1);
    double tEnd = std::min(tA2, tB2);
    
    if (tStart < tEnd - 1e-9) {
        pStart = (tA1 > tB1) ? A1 : B1;
        pEnd = (tA2 < tB2) ? A2 : B2;
        return true;
    }
    return false;
}

struct Point2D {
    double x, y;
    Point2D operator+(const Point2D& o) const { return {x + o.x, y + o.y}; }
    Point2D operator-(const Point2D& o) const { return {x - o.x, y - o.y}; }
    Point2D operator*(double s) const { return {x * s, y * s}; }
};

bool point_on_segment_2D(const Point2D& p, const Point2D& a, const Point2D& b, double& s) {
    Point2D ab = b - a;
    double lenSq = ab.x * ab.x + ab.y * ab.y;
    if (lenSq < 1e-16) return false;
    Point2D ap = p - a;
    double dot = ap.x * ab.x + ap.y * ab.y;
    s = dot / lenSq;
    if (s < -1e-8 || s > 1.0 + 1e-8) return false;
    double perp_x = ap.x - s * ab.x;
    double perp_y = ap.y - s * ab.y;
    double distSq = perp_x * perp_x + perp_y * perp_y;
    return distSq < 1e-12;
}

bool intersect_segments_2D(const Point2D& a, const Point2D& b, const Point2D& c, const Point2D& d, Point2D& out_p) {
    double denom = (b.x - a.x) * (d.y - c.y) - (b.y - a.y) * (d.x - c.x);
    if (std::abs(denom) < 1e-12) return false;
    
    double t = ((c.x - a.x) * (d.y - c.y) - (c.y - a.y) * (d.x - c.x)) / denom;
    double u = ((c.x - a.x) * (b.y - a.y) - (c.y - a.y) * (b.x - a.x)) / denom;
    
    if (t >= -1e-9 && t <= 1.0 + 1e-9 && u >= -1e-9 && u <= 1.0 + 1e-9) {
        out_p = a + (b - a) * std::max(0.0, std::min(1.0, t));
        return true;
    }
    return false;
}

bool point_strictly_in_triangle_2D(const Point2D& p, const Point2D& a, const Point2D& b, const Point2D& c) {
    auto sign = [](const Point2D& p1, const Point2D& p2, const Point2D& p3) {
        return (p1.x - p3.x) * (p2.y - p3.y) - (p2.x - p3.x) * (p1.y - p3.y);
    };
    double d1 = sign(p, a, b);
    double d2 = sign(p, b, c);
    double d3 = sign(p, c, a);
    
    if (d1 > 1e-9 && d2 > 1e-9 && d3 > 1e-9) return true;
    if (d1 < -1e-9 && d2 < -1e-9 && d3 < -1e-9) return true;
    return false;
}

bool are_triangles_coplanar(const Triangle3D& T1, const Triangle3D& T2, const Point3D& N1, double d1) {
    Point3D N2 = (T2.p[1] - T2.p[0]).cross(T2.p[2] - T2.p[0]);
    double len2 = N2.norm();
    if (len2 < 1e-12) return false;
    N2 = N2 * (1.0 / len2);
    
    double cos_theta = std::abs(N1.dot(N2));
    if (cos_theta < 0.99999) return false;
    
    for (int i = 0; i < 3; ++i) {
        if (std::abs(T2.p[i].dot(N1) + d1) > 1e-7) return false;
    }
    return true;
}

void handle_coplanar_intersection(const Triangle3D& T1, const Triangle3D& T2,
                                  const Point3D& N1, double d1,
                                  std::vector<std::pair<Point3D, Point3D>>& new_constraints) {
    Point3D U;
    if (std::abs(N1.x) > 0.9) {
        U = Point3D{0.0, 1.0, 0.0}.cross(N1);
    } else {
        U = Point3D{1.0, 0.0, 0.0}.cross(N1);
    }
    U = U * (1.0 / U.norm());
    Point3D V = N1.cross(U);
    
    auto proj = [&](const Point3D& P) -> Point2D {
        return {P.dot(U), P.dot(V)};
    };
    auto unproj = [&](const Point2D& p2) -> Point3D {
        return U * p2.x + V * p2.y - N1 * d1;
    };
    
    Point2D p1[3] = {proj(T1.p[0]), proj(T1.p[1]), proj(T1.p[2])};
    Point2D p2[3] = {proj(T2.p[0]), proj(T2.p[1]), proj(T2.p[2])};
    
    for (int e1 = 0; e1 < 3; ++e1) {
        Point2D a = p1[e1];
        Point2D b = p1[(e1 + 1) % 3];
        
        std::vector<Point2D> pts_on_edge;
        pts_on_edge.push_back(a);
        pts_on_edge.push_back(b);
        
        for (int e2 = 0; e2 < 3; ++e2) {
            Point2D c = p2[e2];
            Point2D d = p2[(e2 + 1) % 3];
            Point2D ip;
            if (intersect_segments_2D(a, b, c, d, ip)) {
                pts_on_edge.push_back(ip);
            }
        }
        
        std::vector<Point2D> unique_pts;
        for (const auto& p : pts_on_edge) {
            bool duplicate = false;
            for (const auto& up : unique_pts) {
                double dx = p.x - up.x;
                double dy = p.y - up.y;
                if (dx * dx + dy * dy < 1e-16) { duplicate = true; break; }
            }
            if (!duplicate) unique_pts.push_back(p);
        }
        
        Point2D ab = b - a;
        std::sort(unique_pts.begin(), unique_pts.end(), [&](const Point2D& x, const Point2D& y) {
            double sx = (x - a).x * ab.x + (x - a).y * ab.y;
            double sy = (y - a).x * ab.x + (y - a).y * ab.y;
            return sx < sy;
        });
        
        for (size_t i = 0; i + 1 < unique_pts.size(); ++i) {
            Point2D m = (unique_pts[i] + unique_pts[i + 1]) * 0.5;
            if (point_strictly_in_triangle_2D(m, p2[0], p2[1], p2[2])) {
                new_constraints.push_back({unproj(unique_pts[i]), unproj(unique_pts[i + 1])});
            }
        }
    }
    
    for (int e2 = 0; e2 < 3; ++e2) {
        Point2D a = p2[e2];
        Point2D b = p2[(e2 + 1) % 3];
        
        std::vector<Point2D> pts_on_edge;
        pts_on_edge.push_back(a);
        pts_on_edge.push_back(b);
        
        for (int e1 = 0; e1 < 3; ++e1) {
            Point2D c = p1[e1];
            Point2D d = p1[(e1 + 1) % 3];
            Point2D ip;
            if (intersect_segments_2D(a, b, c, d, ip)) {
                pts_on_edge.push_back(ip);
            }
        }
        
        std::vector<Point2D> unique_pts;
        for (const auto& p : pts_on_edge) {
            bool duplicate = false;
            for (const auto& up : unique_pts) {
                double dx = p.x - up.x;
                double dy = p.y - up.y;
                if (dx * dx + dy * dy < 1e-16) { duplicate = true; break; }
            }
            if (!duplicate) unique_pts.push_back(p);
        }
        
        Point2D ab = b - a;
        std::sort(unique_pts.begin(), unique_pts.end(), [&](const Point2D& x, const Point2D& y) {
            double sx = (x - a).x * ab.x + (x - a).y * ab.y;
            double sy = (y - a).x * ab.x + (y - a).y * ab.y;
            return sx < sy;
        });
        
        for (size_t i = 0; i + 1 < unique_pts.size(); ++i) {
            Point2D m = (unique_pts[i] + unique_pts[i + 1]) * 0.5;
            if (point_strictly_in_triangle_2D(m, p1[0], p1[1], p1[2])) {
                new_constraints.push_back({unproj(unique_pts[i]), unproj(unique_pts[i + 1])});
            }
        }
    }
}

struct Triangle {
    int v[3];
};

std::vector<Triangle> triangulate_polygon_2D(const std::vector<int>& poly, const std::vector<Point2D>& pts) {
    std::vector<Triangle> tris;
    std::vector<int> active = poly;
    
    auto is_convex = [&](int prev, int curr, int next) {
        Point2D a = pts[prev];
        Point2D b = pts[curr];
        Point2D c = pts[next];
        double area = (b.x - a.x) * (c.y - a.y) - (c.x - a.x) * (b.y - a.y);
        return area > 1e-12;
    };
    
    auto point_in_triangle = [&](const Point2D& p, const Point2D& a, const Point2D& b, const Point2D& c) {
        auto sign = [](const Point2D& p1, const Point2D& p2, const Point2D& p3) {
            return (p1.x - p3.x) * (p2.y - p3.y) - (p2.x - p3.x) * (p1.y - p3.y);
        };
        double d1 = sign(p, a, b);
        double d2 = sign(p, b, c);
        double d3 = sign(p, c, a);
        bool has_neg = (d1 < -1e-12) || (d2 < -1e-12) || (d3 < -1e-12);
        bool has_pos = (d1 > 1e-12) || (d2 > 1e-12) || (d3 > 1e-12);
        return !(has_neg && has_pos);
    };

    int n = active.size();
    int count = 0;
    while (n > 3 && count < 2 * n) {
        bool clipped = false;
        for (int i = 0; i < n; ++i) {
            int prev = active[(i - 1 + n) % n];
            int curr = active[i];
            int next = active[(i + 1) % n];
            
            if (is_convex(prev, curr, next)) {
                bool has_point_inside = false;
                for (int j = 0; j < n; ++j) {
                    if (active[j] == prev || active[j] == curr || active[j] == next) continue;
                    if (point_in_triangle(pts[active[j]], pts[prev], pts[curr], pts[next])) {
                        has_point_inside = true;
                        break;
                    }
                }
                if (!has_point_inside) {
                    tris.push_back({prev, curr, next});
                    active.erase(active.begin() + i);
                    n = active.size();
                    clipped = true;
                    break;
                }
            }
        }
        if (!clipped) {
            for (int i = 0; i < n; ++i) {
                int prev = active[(i - 1 + n) % n];
                int curr = active[i];
                int next = active[(i + 1) % n];
                if (is_convex(prev, curr, next)) {
                    tris.push_back({prev, curr, next});
                    active.erase(active.begin() + i);
                    n = active.size();
                    clipped = true;
                    break;
                }
            }
            if (!clipped) {
                tris.push_back({active[0], active[1], active[2]});
                active.erase(active.begin() + 1);
                n = active.size();
            }
        }
        count++;
    }
    if (n == 3) {
        tris.push_back({active[0], active[1], active[2]});
    }
    return tris;
}

std::vector<std::array<Point3D, 3>> triangulate_face_with_constraints(
    const Triangle3D& T,
    const std::vector<std::pair<Point3D, Point3D>>& constraints)
{


    Point3D N = (T.p[1] - T.p[0]).cross(T.p[2] - T.p[0]);
    double area3D = N.norm();
    if (area3D < 1e-12) return {};
    N = N * (1.0 / area3D);
    
    Point3D U;
    if (std::abs(N.x) > 0.9) {
        U = Point3D{0.0, 1.0, 0.0}.cross(N);
    } else {
        U = Point3D{1.0, 0.0, 0.0}.cross(N);
    }
    U = U * (1.0 / U.norm());
    Point3D V = N.cross(U);
    double plane_d = T.p[0].dot(N);
    
    auto proj = [&](const Point3D& P) -> Point2D {
        return {P.dot(U), P.dot(V)};
    };
    auto unproj = [&](const Point2D& p2) -> Point3D {
        return U * p2.x + V * p2.y + N * plane_d;
    };
    
    std::vector<Point2D> pts2d;
    pts2d.push_back(proj(T.p[0]));
    pts2d.push_back(proj(T.p[1]));
    pts2d.push_back(proj(T.p[2]));
    
    for (const auto& c : constraints) {
        pts2d.push_back(proj(c.first));
        pts2d.push_back(proj(c.second));
    }
    
    std::vector<Point2D> unique_pts;
    std::vector<int> map_to_unique(pts2d.size(), -1);
    for (int i = 0; i < (int)pts2d.size(); ++i) {
        int found = -1;
        for (int j = 0; j < (int)unique_pts.size(); ++j) {
            double dx = unique_pts[j].x - pts2d[i].x;
            double dy = unique_pts[j].y - pts2d[i].y;
            if (dx * dx + dy * dy < 1e-16) {
                found = j;
                break;
            }
        }
        if (found == -1) {
            found = unique_pts.size();
            unique_pts.push_back(pts2d[i]);
        }
        map_to_unique[i] = found;
    }
    

    
    int v0_idx = map_to_unique[0];
    int v1_idx = map_to_unique[1];
    int v2_idx = map_to_unique[2];
    
    struct SimpleEdge {
        int u, v;
        bool operator==(const SimpleEdge& o) const {
            return (u == o.u && v == o.v) || (u == o.v && v == o.u);
        }
    };
    std::vector<SimpleEdge> edges;
    auto add_edge_unique = [&](int u, int v) {
        if (u == v) return;
        SimpleEdge e{std::min(u, v), std::max(u, v)};
        for (const auto& existing : edges) {
            if (existing == e) return;
        }
        edges.push_back(e);
    };
    
    std::vector<std::vector<std::pair<double, int>>> edge_splits(3);
    edge_splits[0].push_back({0.0, v0_idx});
    edge_splits[0].push_back({1.0, v1_idx});
    edge_splits[1].push_back({0.0, v1_idx});
    edge_splits[1].push_back({1.0, v2_idx});
    edge_splits[2].push_back({0.0, v2_idx});
    edge_splits[2].push_back({1.0, v0_idx});
    
    for (int idx = 3; idx < (int)pts2d.size(); ++idx) {
        int pt_idx = map_to_unique[idx];
        if (pt_idx == v0_idx || pt_idx == v1_idx || pt_idx == v2_idx) continue;
        
        Point2D p = unique_pts[pt_idx];
        for (int e = 0; e < 3; ++e) {
            int a_idx = (e == 0) ? v0_idx : ((e == 1) ? v1_idx : v2_idx);
            int b_idx = (e == 0) ? v1_idx : ((e == 1) ? v2_idx : v0_idx);
            Point2D a = unique_pts[a_idx];
            Point2D b = unique_pts[b_idx];
            
            double s;
            if (point_on_segment_2D(p, a, b, s)) {
                edge_splits[e].push_back({s, pt_idx});
                break;
            }
        }
    }
    
    for (int e = 0; e < 3; ++e) {
        auto& splits = edge_splits[e];
        std::sort(splits.begin(), splits.end());
        std::vector<std::pair<double, int>> unique_splits;
        for (const auto& sp : splits) {
            if (unique_splits.empty() || sp.first - unique_splits.back().first > 1e-8) {
                unique_splits.push_back(sp);
            }
        }
        for (size_t i = 0; i + 1 < unique_splits.size(); ++i) {
            add_edge_unique(unique_splits[i].second, unique_splits[i + 1].second);
        }
    }
    
    for (size_t i = 0; i < constraints.size(); ++i) {
        int u = map_to_unique[3 + 2 * i];
        int v = map_to_unique[3 + 2 * i + 1];
        add_edge_unique(u, v);
    }
    

    
    std::vector<std::vector<int>> adj(unique_pts.size());
    for (const auto& e : edges) {
        adj[e.u].push_back(e.v);
        adj[e.v].push_back(e.u);
    }
    
    for (int i = 0; i < (int)unique_pts.size(); ++i) {
        Point2D base = unique_pts[i];
        std::sort(adj[i].begin(), adj[i].end(), [&](int a, int b) {
            Point2D va = unique_pts[a] - base;
            Point2D vb = unique_pts[b] - base;
            double ang_a = std::atan2(va.y, va.x);
            double ang_b = std::atan2(vb.y, vb.x);
            return ang_a < ang_b;
        });
    }
    
    struct HashEdge {
        size_t operator()(const std::pair<int, int>& p) const {
            return std::hash<int>()(p.first) ^ std::hash<int>()(p.second);
        }
    };
    std::unordered_set<std::pair<int, int>, HashEdge> visited;
    std::vector<std::vector<int>> CCW_faces;
    
    for (const auto& edge : edges) {
        for (int step = 0; step < 2; ++step) {
            int start_u = (step == 0) ? edge.u : edge.v;
            int start_v = (step == 0) ? edge.v : edge.u;
            
            if (visited.count({start_u, start_v})) continue;
            
            std::vector<int> face;
            face.push_back(start_u);
            
            int curr = start_v;
            int prev = start_u;
            bool ok = true;
            
            while (curr != start_u) {
                face.push_back(curr);
                visited.insert({prev, curr});
                
                const auto& neighbors = adj[curr];
                auto it = std::find(neighbors.begin(), neighbors.end(), prev);
                if (it == neighbors.end()) { ok = false; break; }
                
                int idx = std::distance(neighbors.begin(), it);
                int next_idx = (idx - 1 + neighbors.size()) % neighbors.size();
                int next_v = neighbors[next_idx];
                
                prev = curr;
                curr = next_v;
                
                if (face.size() > unique_pts.size() * 2) { ok = false; break; }
            }
            if (ok) {
                visited.insert({prev, curr});
                double area = 0.0;
                int n = face.size();
                for (int i = 0; i < n; ++i) {
                    Point2D p1 = unique_pts[face[i]];
                    Point2D p2 = unique_pts[face[(i + 1) % n]];
                    area += p1.x * p2.y - p2.x * p1.y;
                }
                if (area > 1e-12) {
                    CCW_faces.push_back(face);
                }
            }
        }
    }
    
    std::vector<std::array<Point3D, 3>> result_triangles;
    for (const auto& face : CCW_faces) {
        if (face.size() < 3) continue;
        auto tris = triangulate_polygon_2D(face, unique_pts);
        for (const auto& tri : tris) {
            result_triangles.push_back({
                unproj(unique_pts[tri.v[0]]),
                unproj(unique_pts[tri.v[1]]),
                unproj(unique_pts[tri.v[2]])
            });
        }
    }
    

    
    return result_triangles;
}

void deduplicate_constraints(std::vector<std::pair<Point3D, Point3D>>& constraints) {
    std::vector<std::pair<Point3D, Point3D>> unique_c;
    for (const auto& c : constraints) {
        Point3D p1 = c.first;
        Point3D p2 = c.second;
        if (p2.x < p1.x || (p2.x == p1.x && p2.y < p1.y) || (p2.x == p1.x && p2.y == p1.y && p2.z < p1.z)) {
            std::swap(p1, p2);
        }
        
        bool duplicate = false;
        for (const auto& uc : unique_c) {
            if ((p1 - uc.first).normSq() < 1e-16 && (p2 - uc.second).normSq() < 1e-16) {
                duplicate = true;
                break;
            }
        }
        if (!duplicate) {
            unique_c.push_back({p1, p2});
        }
    }
    constraints = unique_c;
}

bool is_point_inside_mesh(const Point3D& p, const Mesh& mesh, const BVH& bvh) {
    std::mt19937 gen(42);
    std::uniform_real_distribution<double> dis(-1.0, 1.0);
    
    for (int attempt = 0; attempt < 5; ++attempt) {
        Point3D dir = {1.0, 0.0, 0.0};
        if (attempt > 0) {
            dir = {dis(gen), dis(gen), dis(gen)};
            double len = dir.norm();
            if (len > 1e-6) {
                dir = dir * (1.0 / len);
            } else {
                dir = {1.0, 0.0, 0.0};
            }
        }
        
        std::vector<double> t_values;
        bool edge_hit = false;
        
        auto intersect_callback = [&](int face_idx) {
            const auto& F = mesh.faces[face_idx];
            Point3D v0 = mesh.vertices[F.v[0]];
            Point3D v1 = mesh.vertices[F.v[1]];
            Point3D v2 = mesh.vertices[F.v[2]];
            
            Point3D edge1 = v1 - v0;
            Point3D edge2 = v2 - v0;
            Point3D h = dir.cross(edge2);
            double a = edge1.dot(h);
            if (std::abs(a) < 1e-12) return;
            
            double f = 1.0 / a;
            Point3D s = p - v0;
            double u = f * s.dot(h);
            if (u < -1e-9 || u > 1.0 + 1e-9) return;
            
            Point3D q = s.cross(edge1);
            double v = f * dir.dot(q);
            if (v < -1e-9 || u + v > 1.0 + 1e-9) return;
            
            double t = f * edge2.dot(q);
            if (t > 1e-9) {
                if (u < 1e-9 || v < 1e-9 || (1.0 - u - v) < 1e-9) {
                    edge_hit = true;
                }
                t_values.push_back(t);
            }
        };
        
        bvh.intersect_ray(p, dir, intersect_callback);
        
        if (edge_hit) {
            continue;
        }
        
        std::sort(t_values.begin(), t_values.end());
        int unique_count = 0;
        double last_t = -1.0;
        for (double t : t_values) {
            if (last_t < 0 || t - last_t > 1e-9) {
                unique_count++;
                last_t = t;
            }
        }
        
        return (unique_count % 2) == 1;
    }
    
    return false;
}

std::vector<std::vector<int>> build_face_adjacency(const Mesh& mesh) {
    struct Edge {
        int u, v;
        bool operator==(const Edge& o) const {
            return u == o.u && v == o.v;
        }
    };
    struct EdgeHash {
        size_t operator()(const Edge& e) const {
            return std::hash<int>()(e.u) ^ std::hash<int>()(e.v);
        }
    };
    
    std::unordered_map<Edge, std::vector<int>, EdgeHash> edge_to_faces;
    for (int i = 0; i < (int)mesh.faces.size(); ++i) {
        const auto& F = mesh.faces[i];
        for (int j = 0; j < 3; ++j) {
            int u = F.v[j];
            int v = F.v[(j + 1) % 3];
            edge_to_faces[Edge{std::min(u, v), std::max(u, v)}].push_back(i);
        }
    }
    
    std::vector<std::vector<int>> adj(mesh.faces.size());
    for (const auto& pair : edge_to_faces) {
        const auto& faces = pair.second;
        if (faces.size() == 2) {
            adj[faces[0]].push_back(faces[1]);
            adj[faces[1]].push_back(faces[0]);
        }
    }
    return adj;
}

struct GridMerge {
    double eps = 1e-8;
    struct BucketKey {
        long long x, y, z;
        bool operator==(const BucketKey& o) const {
            return x == o.x && y == o.y && z == o.z;
        }
    };
    struct BucketKeyHash {
        size_t operator()(const BucketKey& k) const {
            return (k.x * 73856093) ^ (k.y * 19349663) ^ (k.z * 83492791);
        }
    };
    
    std::unordered_map<BucketKey, std::vector<int>, BucketKeyHash> grid;
    std::vector<Point3D> unique_vertices;
    
    int get_or_create_vertex(const Point3D& p) {
        long long bx = (long long)std::floor(p.x / eps);
        long long by = (long long)std::floor(p.y / eps);
        long long bz = (long long)std::floor(p.z / eps);
        
        for (long long dx = -1; dx <= 1; ++dx) {
            for (long long dy = -1; dy <= 1; ++dy) {
                for (long long dz = -1; dz <= 1; ++dz) {
                    BucketKey key{bx + dx, by + dy, bz + dz};
                    auto it = grid.find(key);
                    if (it != grid.end()) {
                        for (int idx : it->second) {
                            const auto& candidate = unique_vertices[idx];
                            double dx_val = candidate.x - p.x;
                            double dy_val = candidate.y - p.y;
                            double dz_val = candidate.z - p.z;
                            if (dx_val * dx_val + dy_val * dy_val + dz_val * dz_val < eps * eps) {
                                return idx;
                            }
                        }
                    }
                }
            }
        }
        
        int new_idx = unique_vertices.size();
        unique_vertices.push_back(p);
        grid[BucketKey{bx, by, bz}].push_back(new_idx);
        return new_idx;
    }
};

        Triangle3D T1 = {meshA.vertices[FA_face.v[0]], meshA.vertices[FA_face.v[1]], meshA.vertices[FA_face.v[2]]};
        
        const auto& FB_face = meshB.faces[j];
        Triangle3D T2 = {meshB.vertices[FB_face.v[0]], meshB.vertices[FB_face.v[1]], meshB.vertices[FB_face.v[2]]};
        
        Point3D N1 = (T1.p[1] - T1.p[0]).cross(T1.p[2] - T1.p[0]);
        double len1 = N1.norm();
        if (len1 < 1e-12) continue;
        N1 = N1 * (1.0 / len1);
        double d1 = -N1.dot(T1.p[0]);
        
        if (are_triangles_coplanar(T1, T2, N1, d1)) {
            // Check if normals point in the same direction
            Point3D N2 = (T2.p[1] - T2.p[0]).cross(T2.p[2] - T2.p[0]);
            double len2 = N2.norm();
            if (len2 > 1e-12) {
                N2 = N2 * (1.0 / len2);
                if (N1.dot(N2) > 0.9) {
                    B_coplanar_same_normal[j] = true;
                }
            }
            std::vector<std::pair<Point3D, Point3D>> coplanar_constraints;
            handle_coplanar_intersection(T1, T2, N1, d1, coplanar_constraints);
            if (!coplanar_constraints.empty()) {
                I_A[i] = true;
                I_B[j] = true;
                for (const auto& c : coplanar_constraints) {
                    segments_on_A[i].push_back(c);
                    segments_on_B[j].push_back(c);
                }
            }
        } else {
            Point3D pStart, pEnd;
            if (intersect_triangles(T1, T2, pStart, pEnd)) {
                I_A[i] = true;
                I_B[j] = true;
                segments_on_A[i].push_back({pStart, pEnd});
                segments_on_B[j].push_back({pStart, pEnd});
            }
        }
    }
    
    std::vector<std::array<Point3D, 3>> kept_triangles;
    
    // Process intersecting faces of A
    for (int i = 0; i < FA; ++i) {
        if (I_A[i]) {
            deduplicate_constraints(segments_on_A[i]);
            const auto& F = meshA.faces[i];
            Triangle3D T = {meshA.vertices[F.v[0]], meshA.vertices[F.v[1]], meshA.vertices[F.v[2]]};
            Point3D N = (T.p[1] - T.p[0]).cross(T.p[2] - T.p[0]);
            double len = N.norm();
            if (len > 1e-12) N = N * (1.0 / len);
            
            auto sub_tris = triangulate_face_with_constraints(T, segments_on_A[i]);
            for (const auto& sub_t : sub_tris) {
                Point3D centroid = (sub_t[0] + sub_t[1] + sub_t[2]) * (1.0 / 3.0);
                Point3D test_pt = centroid + N * 1e-8;
                bool insideB = is_point_inside_mesh(test_pt, meshB, bvhB);

                if (!insideB) {
                    kept_triangles.push_back(sub_t);
                }
            }
        }
    }
    
    // Process intersecting faces of B
    for (int j = 0; j < FB; ++j) {
        if (I_B[j]) {
            deduplicate_constraints(segments_on_B[j]);
            const auto& F = meshB.faces[j];
            Triangle3D T = {meshB.vertices[F.v[0]], meshB.vertices[F.v[1]], meshB.vertices[F.v[2]]};
            Point3D N = (T.p[1] - T.p[0]).cross(T.p[2] - T.p[0]);
            double len = N.norm();
            if (len > 1e-12) N = N * (1.0 / len);
            
            auto sub_tris = triangulate_face_with_constraints(T, segments_on_B[j]);
            double perturb_sign = B_coplanar_same_normal[j] ? -1.0 : 1.0;
            for (const auto& sub_t : sub_tris) {
                Point3D centroid = (sub_t[0] + sub_t[1] + sub_t[2]) * (1.0 / 3.0);
                Point3D test_pt = centroid + N * (perturb_sign * 1e-8);
                if (!is_point_inside_mesh(test_pt, meshA, bvhA)) {
                    kept_triangles.push_back(sub_t);
                }
            }
        }
    }
    
    std::cerr << "Classifying non-intersecting faces of A..." << std::endl;
    std::vector<std::vector<int>> adjA = build_face_adjacency(meshA);
    std::vector<bool> visitedA(FA, false);
    for (int i = 0; i < FA; ++i) {
        if (I_A[i] || visitedA[i]) continue;
        
        std::vector<int> component;
        std::queue<int> q;
        q.push(i);
        visitedA[i] = true;
        
        while (!q.empty()) {
            int curr = q.front();
            q.pop();
            component.push_back(curr);
            
            for (int neighbor : adjA[curr]) {
                if (!I_A[neighbor] && !visitedA[neighbor]) {
                    visitedA[neighbor] = true;
                    q.push(neighbor);
                }
            }
        }
        
        int rep = component[0];
        const auto& F = meshA.faces[rep];
        Point3D centroid = (meshA.vertices[F.v[0]] + meshA.vertices[F.v[1]] + meshA.vertices[F.v[2]]) * (1.0 / 3.0);
        Point3D N = (meshA.vertices[F.v[1]] - meshA.vertices[F.v[0]]).cross(meshA.vertices[F.v[2]] - meshA.vertices[F.v[0]]);
        double len = N.norm();
        if (len > 1e-12) N = N * (1.0 / len);
        Point3D test_pt = centroid + N * 1e-8;
        
        if (!is_point_inside_mesh(test_pt, meshB, bvhB)) {
            for (int idx : component) {
                const auto& F_comp = meshA.faces[idx];
                kept_triangles.push_back({
                    meshA.vertices[F_comp.v[0]],
                    meshA.vertices[F_comp.v[1]],
                    meshA.vertices[F_comp.v[2]]
                });
            }
        }
    }
    
    std::cerr << "Classifying non-intersecting faces of B..." << std::endl;
    std::vector<std::vector<int>> adjB = build_face_adjacency(meshB);
    std::vector<bool> visitedB(FB, false);
    for (int j = 0; j < FB; ++j) {
        if (I_B[j] || visitedB[j]) continue;
        
        std::vector<int> component;
        std::queue<int> q;
        q.push(j);
        visitedB[j] = true;
        
        while (!q.empty()) {
            int curr = q.front();
            q.pop();
            component.push_back(curr);
            
            for (int neighbor : adjB[curr]) {
                if (!I_B[neighbor] && !visitedB[neighbor]) {
                    visitedB[neighbor] = true;
                    q.push(neighbor);
                }
            }
        }
        
        int rep = component[0];
        const auto& F = meshB.faces[rep];
        Point3D centroid = (meshB.vertices[F.v[0]] + meshB.vertices[F.v[1]] + meshB.vertices[F.v[2]]) * (1.0 / 3.0);
        Point3D N = (meshB.vertices[F.v[1]] - meshB.vertices[F.v[0]]).cross(meshB.vertices[F.v[2]] - meshB.vertices[F.v[0]]);
        double len = N.norm();
        if (len > 1e-12) N = N * (1.0 / len);
        Point3D test_pt = centroid + N * 1e-8;
        
        if (!is_point_inside_mesh(test_pt, meshA, bvhA)) {
            for (int idx : component) {
                const auto& F_comp = meshB.faces[idx];
                kept_triangles.push_back({
                    meshB.vertices[F_comp.v[0]],
                    meshB.vertices[F_comp.v[1]],
                    meshB.vertices[F_comp.v[2]]
                });
            }
        }
    }
    
    std::cerr << "Merging vertices. Kept count: " << kept_triangles.size() << std::endl;
    GridMerge merger;
    merger.grid.reserve(kept_triangles.size() * 3);
    
    struct IndexedTriangle {
        int v[3];
        int original_parity;
    };
    std::vector<IndexedTriangle> indexed_triangles;
    indexed_triangles.reserve(kept_triangles.size());
    
    for (const auto& t : kept_triangles) {
        int idx0 = merger.get_or_create_vertex(t[0]);
        int idx1 = merger.get_or_create_vertex(t[1]);
        int idx2 = merger.get_or_create_vertex(t[2]);
        if (idx0 == idx1 || idx1 == idx2 || idx2 == idx0) continue;
        
        int s0 = idx0, s1 = idx1, s2 = idx2;
        int parity = 1;
        if (s0 > s1) { std::swap(s0, s1); parity = -parity; }
        if (s1 > s2) { std::swap(s1, s2); parity = -parity; }
        if (s0 > s1) { std::swap(s0, s1); parity = -parity; }
        
        indexed_triangles.push_back({ {idx0, idx1, idx2}, parity });
    }
    
    struct TriSignature {
        int v[3];
        bool operator==(const TriSignature& o) const {
            return v[0] == o.v[0] && v[1] == o.v[1] && v[2] == o.v[2];
        }
    };
    struct TriSigHash {
        size_t operator()(const TriSignature& ts) const {
            return std::hash<int>()(ts.v[0]) ^ std::hash<int>()(ts.v[1]) ^ std::hash<int>()(ts.v[2]);
        }
    };
    
    std::unordered_map<TriSignature, std::vector<IndexedTriangle>, TriSigHash> groups;
    for (const auto& itri : indexed_triangles) {
        int s[3] = {itri.v[0], itri.v[1], itri.v[2]};
        std::sort(s, s + 3);
        groups[TriSignature{ {s[0], s[1], s[2]} }].push_back(itri);
    }
    
    std::vector<Face> final_faces;
    for (const auto& pair : groups) {
        const auto& list = pair.second;
        if (list.size() == 1) {
            final_faces.push_back({ {list[0].v[0], list[0].v[1], list[0].v[2]} });
        } else if (list.size() >= 2) {
            int sum_parity = 0;
            for (const auto& item : list) {
                sum_parity += item.original_parity;
            }
            if (sum_parity > 0) {
                for (const auto& item : list) {
                    if (item.original_parity == 1) {
                        final_faces.push_back({ {item.v[0], item.v[1], item.v[2]} });
                        break;
                    }
                }
            } else if (sum_parity < 0) {
                for (const auto& item : list) {
                    if (item.original_parity == -1) {
                        final_faces.push_back({ {item.v[0], item.v[1], item.v[2]} });
                        break;
                    }
                }
            }
        }
    }
    
    // Write output
    FastOutput out;
    out.write_int((long long)merger.unique_vertices.size());
    out.write_space();
    out.write_int((long long)final_faces.size());
    out.write_newline();
    
    for (const auto& v : merger.unique_vertices) {
        out.write_double(v.x);
        out.write_space();
        out.write_double(v.y);
        out.write_space();
        out.write_double(v.z);
        out.write_newline();
    }
    for (const auto& f : final_faces) {
        out.write_int((long long)f.v[0]);
        out.write_space();
        out.write_int((long long)f.v[1]);
        out.write_space();
        out.write_int((long long)f.v[2]);
        out.write_newline();
    }
    
    return 0;
}
