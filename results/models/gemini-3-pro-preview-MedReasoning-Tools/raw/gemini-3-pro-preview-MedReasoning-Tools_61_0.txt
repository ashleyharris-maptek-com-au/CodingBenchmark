declare const require: any;
const fs = require('fs');

interface Package {
  id: number;
  x: number;
  y: number;
}

interface InputData {
  depot: [number, number];
  truckCount: number;
  packages: Package[];
}

function main() {
  const input: string = fs.readFileSync(0, 'utf8');
  if (!input.trim()) {
    console.log(JSON.stringify({ routes: [] }));
    return;
  }
  const data: InputData = JSON.parse(input);
  const depot = data.depot;
  const truckCount = data.truckCount;
  const packages = data.packages;
  const n = packages.length;

  if (n === 0) {
    console.log(JSON.stringify({ routes: [] }));
    return;
  }

  // Pre-extract coordinates for faster access
  const ptsX = new Float64Array(n);
  const ptsY = new Float64Array(n);
  const ids = new Int32Array(n);
  for (let i = 0; i < n; i++) {
    ptsX[i] = packages[i].x;
    ptsY[i] = packages[i].y;
    ids[i] = packages[i].id;
  }

  // Define helper functions
  const dist = (x1: number, y1: number, x2: number, y2: number): number => {
    const dx = x1 - x2;
    const dy = y1 - y2;
    return Math.sqrt(dx * dx + dy * dy);
  };

  const getRouteDist = (route: number[]): number => {
    if (route.length === 0) return 0;
    let total = 0;
    let curX = depot[0];
    let curY = depot[1];
    for (let i = 0; i < route.length; i++) {
      const idx = route[i];
      const px = ptsX[idx];
      const py = ptsY[idx];
      total += dist(curX, curY, px, py);
      curX = px;
      curY = py;
    }
    total += dist(curX, curY, depot[0], depot[1]);
    return total;
  };

  const getRoutesTotalDist = (routes: number[][]): number => {
    let total = 0;
    for (let i = 0; i < routes.length; i++) {
      total += getRouteDist(routes[i]);
    }
    return total;
  };

  // Candidates list
  let bestRoutes: number[][] = [];
  let bestDist = Infinity;

  const updateBest = (routes: number[][]) => {
    const d = getRoutesTotalDist(routes);
    if (d < bestDist) {
      bestDist = d;
      bestRoutes = routes;
    }
  };

  // Candidate 1: Polar angle sort (1 route)
  const angles = new Float64Array(n);
  for (let i = 0; i < n; i++) {
    angles[i] = Math.atan2(ptsY[i] - depot[1], ptsX[i] - depot[0]);
  }
  const polarIndices = new Int32Array(n);
  for (let i = 0; i < n; i++) polarIndices[i] = i;
  polarIndices.sort((a, b) => angles[a] - angles[b]);
  updateBest([[...polarIndices]]);

  // Candidate 2: Hilbert sort (1 route)
  let minX = depot[0], maxX = depot[0];
  let minY = depot[1], maxY = depot[1];
  for (let i = 0; i < n; i++) {
    if (ptsX[i] < minX) minX = ptsX[i];
    if (ptsX[i] > maxX) maxX = ptsX[i];
    if (ptsY[i] < minY) minY = ptsY[i];
    if (ptsY[i] > maxY) maxY = ptsY[i];
  }
  const rangeX = maxX - minX || 1.0;
  const rangeY = maxY - minY || 1.0;

  function xy2d(x: number, y: number): number {
    let d = 0;
    let rx = 0;
    let ry = 0;
    for (let s = 32768; s > 0; s = s >> 1) {
      rx = (x & s) > 0 ? 1 : 0;
      ry = (y & s) > 0 ? 1 : 0;
      d += s * s * ((3 * rx) ^ ry);
      if (ry === 0) {
        if (rx === 1) {
          x = s - 1 - x;
          y = s - 1 - y;
        }
        let temp = x;
        x = y;
        y = temp;
      }
    }
    return d;
  }

  const hilbertDists = new Float64Array(n);
  for (let i = 0; i < n; i++) {
    const px = Math.floor(((ptsX[i] - minX) / rangeX) * 65535);
    const py = Math.floor(((ptsY[i] - minY) / rangeY) * 65535);
    hilbertDists[i] = xy2d(px, py);
  }
  const hilbertIndices = new Int32Array(n);
  for (let i = 0; i < n; i++) hilbertIndices[i] = i;
  hilbertIndices.sort((a, b) => hilbertDists[a] - hilbertDists[b]);
  updateBest([[...hilbertIndices]]);

  // Candidate 3: Polar angle split into truckCount routes
  const polarSplit: number[][] = [];
  for (let t = 0; t < truckCount; t++) {
    const start = Math.floor((n * t) / truckCount);
    const end = Math.floor((n * (t + 1)) / truckCount);
    const chunk: number[] = [];
    for (let i = start; i < end; i++) chunk.push(polarIndices[i]);
    if (chunk.length > 0) {
      // Run nearest neighbor within chunk for small n
      if (n <= 3600) {
        const nnChunk: number[] = [];
        const visited = new Uint8Array(n);
        let curX = depot[0];
        let curY = depot[1];
        for (let step = 0; step < chunk.length; step++) {
          let bestIdx = -1;
          let bestD = Infinity;
          for (let k = 0; k < chunk.length; k++) {
            const idx = chunk[k];
            if (visited[idx] === 0) {
              const d = dist(curX, curY, ptsX[idx], ptsY[idx]);
              if (d < bestD) {
                bestD = d;
                bestIdx = idx;
              }
            }
          }
          if (bestIdx !== -1) {
            nnChunk.push(bestIdx);
            visited[bestIdx] = 1;
            curX = ptsX[bestIdx];
            curY = ptsY[bestIdx];
          }
        }
        polarSplit.push(nnChunk);
      } else {
        polarSplit.push(chunk);
      }
    }
  }
  updateBest(polarSplit);

  // Candidate 4: Hilbert split into truckCount routes
  const hilbertSplit: number[][] = [];
  for (let t = 0; t < truckCount; t++) {
    const start = Math.floor((n * t) / truckCount);
    const end = Math.floor((n * (t + 1)) / truckCount);
    const chunk: number[] = [];
    for (let i = start; i < end; i++) chunk.push(hilbertIndices[i]);
    if (chunk.length > 0) {
      hilbertSplit.push(chunk);
    }
  }
  updateBest(hilbertSplit);

  // Candidate 5: Nearest Neighbor (1 route) for small n
  if (n <= 2000) {
    const nnIndices: number[] = [];
    const visited = new Uint8Array(n);
    let curX = depot[0];
    let curY = depot[1];
    for (let step = 0; step < n; step++) {
      let bestIdx = -1;
      let bestD = Infinity;
      for (let i = 0; i < n; i++) {
        if (visited[i] === 0) {
          const d = dist(curX, curY, ptsX[i], ptsY[i]);
          if (d < bestD) {
            bestD = d;
            bestIdx = i;
          }
        }
      }
      if (bestIdx !== -1) {
        nnIndices.push(bestIdx);
        visited[bestIdx] = 1;
        curX = ptsX[bestIdx];
        curY = ptsY[bestIdx];
      }
    }
    updateBest([nnIndices]);
  }

  // 2-opt local search refinement
  const run2Opt = (route: number[], maxIter: number): number[] => {
    const len = route.length;
    if (len <= 3) return route;
    const tour = new Int32Array(len + 2);
    tour[0] = -1;
    for (let i = 0; i < len; i++) tour[i + 1] = route[i];
    tour[len + 1] = -1;

    const getX = (idx: number) => idx === -1 ? depot[0] : ptsX[idx];
    const getY = (idx: number) => idx === -1 ? depot[1] : ptsY[idx];

    let improved = true;
    let iter = 0;
    while (improved && iter < maxIter) {
      improved = false;
      iter++;
      for (let i = 1; i < len; i++) {
        for (let j = i + 1; j <= len; j++) {
          const a = tour[i - 1];
          const b = tour[i];
          const c = tour[j];
          const d = tour[j + 1];

          const ax = getX(a), ay = getY(a);
          const bx = getX(b), by = getY(b);
          const cx = getX(c), cy = getY(c);
          const dx = getX(d), dy = getY(d);

          const d1 = dist(ax, ay, bx, by) + dist(cx, cy, dx, dy);
          const d2 = dist(ax, ay, cx, cy) + dist(bx, by, dx, dy);

          if (d2 < d1 - 1e-9) {
            let l = i, r = j;
            while (l < r) {
              const tmp = tour[l];
              tour[l] = tour[r];
              tour[r] = tmp;
              l++;
              r--;
            }
            improved = true;
          }
        }
      }
    }
    const res: number[] = [];
    for (let i = 1; i <= len; i++) res.push(tour[i]);
    return res;
  };

  // Run 2-opt refinement on bestRoutes
  if (n <= 1000) {
    bestRoutes = bestRoutes.map(r => run2Opt(r, 200));
  } else if (n <= 10000) {
    bestRoutes = bestRoutes.map(r => run2Opt(r, 20));
  }

  // Format response matching original package IDs
  const finalRoutes = bestRoutes.map(r => r.map(idx => ids[idx]));
  console.log(JSON.stringify({ routes: finalRoutes }));
}

main();
