import math

# Persistent state dictionary to maintain history across calls
_PERSISTENT_STATE = {}

def robust_median(values):
    if not values:
        return None
    sorted_val = sorted(values)
    n = len(sorted_val)
    if n % 2 == 1:
        return sorted_val[n // 2]
    else:
        return (sorted_val[n // 2 - 1] + sorted_val[n // 2]) / 2.0

def get_lidar_key(angle):
    sign = "+" if angle >= 0 else "-"
    abs_angle = abs(angle)
    return f"lidar_{sign}{abs_angle:03d}_deg"

def fuse_vision_field(state, field_name, valid_name, default_value=None):
    candidates = []
    # Try using valid flag
    for i in [1, 2, 3]:
        val = state.get(f"vision_{i}_{field_name}")
        vld = state.get(f"vision_{i}_{valid_name}")
        if vld and val is not None:
            candidates.append(val)
    # If no candidates from valid flags, try any available (in case valid flags are wrong/flickering)
    if not candidates:
        for i in [1, 2, 3]:
            val = state.get(f"vision_{i}_{field_name}")
            if val is not None:
                candidates.append(val)
    if candidates:
        return robust_median(candidates)
    return default_value

def fuse_vision_category(state, field_name, valid_name, default_value="solid"):
    candidates = []
    for i in [1, 2, 3]:
        val = state.get(f"vision_{i}_{field_name}")
        vld = state.get(f"vision_{i}_{valid_name}")
        if vld and val is not None:
            candidates.append(val)
    if not candidates:
        for i in [1, 2, 3]:
            val = state.get(f"vision_{i}_{field_name}")
            if val is not None:
                candidates.append(val)
    if not candidates:
        return default_value
    # Vote
    counts = {}
    for c in candidates:
        counts[c] = counts.get(c, 0) + 1
    return max(counts, key=counts.get)

def is_lane_safe(target_lane, current_lane, lidar_data):
    if target_lane == current_lane:
        return True
    
    # Check side lidar beams to ensure the target lane is clear
    # If target is to the right
    if target_lane > current_lane:
        check_angles = [30, 45, 60, 90]
    else:
        check_angles = [-30, -45, -60, -90]
        
    for angle in check_angles:
        if angle in lidar_data:
            beam = lidar_data[angle]
            # If a beam is valid and range is less than 15.0m, it's unsafe to change lanes
            if beam['range'] < 15.0:
                return False
    return True

def autopilot_step(state: dict, dt: float) -> dict:
    global _PERSISTENT_STATE
    
    # Extract simulation time
    sim_time = state.get("sim_time_s", 0.0)
    prev_sim_time = _PERSISTENT_STATE.get('sim_time', 0.0)
    
    # Reset persistent state if simulation time goes backward or is zero (new scenario starting)
    if not _PERSISTENT_STATE or sim_time < prev_sim_time or sim_time == 0.0:
        _PERSISTENT_STATE = {
            'speed_kmh': None,
            'speed_integral': 0.0,
            'lane_offset': 0.0,
            'lane_offset_dot': 0.0,
            'steer_integral': 0.0,
            'prev_steer': 0.0,
            'prev_throttle': 0.0,
            'prev_brake': 0.0,
            'road_curvature': 0.0,
            'sign_multiplier': None,
            'sim_time': sim_time,
        }
    else:
        _PERSISTENT_STATE['sim_time'] = sim_time

    # Protect against dt <= 0
    if dt <= 0:
        dt = 0.01

    # ----------------------------------------------------
    # 1. Accelerometer Fusion
    # ----------------------------------------------------
    accel_long_candidates = []
    accel_lat_candidates = []
    for i in [1, 2, 3]:
        if state.get(f"accel_{i}_valid") and state.get(f"accel_{i}_longitudinal_ms2") is not None:
            accel_long_candidates.append(state[f"accel_{i}_longitudinal_ms2"])
        if state.get(f"accel_{i}_valid") and state.get(f"accel_{i}_lateral_ms2") is not None:
            accel_lat_candidates.append(state[f"accel_{i}_lateral_ms2"])
            
    if not accel_long_candidates:
        for i in [1, 2, 3]:
            if state.get(f"accel_{i}_longitudinal_ms2") is not None:
                accel_long_candidates.append(state[f"accel_{i}_longitudinal_ms2"])
    if not accel_lat_candidates:
        for i in [1, 2, 3]:
            if state.get(f"accel_{i}_lateral_ms2") is not None:
                accel_lat_candidates.append(state[f"accel_{i}_lateral_ms2"])
                
    accel_long = robust_median(accel_long_candidates) or 0.0
    accel_lat = robust_median(accel_lat_candidates) or 0.0

    # ----------------------------------------------------
    # 2. Speed Fusion
    # ----------------------------------------------------
    speed_candidates = []
    for w in ['fl', 'fr', 'rl', 'rr']:
        if state.get(f"wheel_speed_{w}_valid") and state.get(f"wheel_speed_{w}_kmh") is not None:
            speed_candidates.append(state[f"wheel_speed_{w}_kmh"])
    for g in [1, 2]:
        if state.get(f"gps_{g}_valid") and state.get(f"gps_{g}_speed_kmh") is not None:
            speed_candidates.append(state[f"gps_{g}_speed_kmh"])
    for v in [1, 2]:
        if state.get(f"vehicle_speed_{v}_kmh") is not None:
            speed_candidates.append(state[f"vehicle_speed_{v}_kmh"])

    if not speed_candidates:
        # If all valid flags are false, collect any non-None speed values
        for w in ['fl', 'fr', 'rl', 'rr']:
            if state.get(f"wheel_speed_{w}_kmh") is not None:
                speed_candidates.append(state[f"wheel_speed_{w}_kmh"])
        for g in [1, 2]:
            if state.get(f"gps_{g}_speed_kmh") is not None:
                speed_candidates.append(state[f"gps_{g}_speed_kmh"])

    prev_speed = _PERSISTENT_STATE.get('speed_kmh')
    if prev_speed is not None:
        # Filter outliers that deviate too much from previous speed (accel limit check)
        max_diff = max(10.0, 60.0 * dt)
        filtered_candidates = [c for c in speed_candidates if abs(c - prev_speed) < max_diff]
        if filtered_candidates:
            speed_kmh = robust_median(filtered_candidates)
        else:
            # Fallback to accelerometer dead reckoning
            speed_kmh = max(0.0, prev_speed + accel_long * 3.6 * dt)
    else:
        # First iteration initialization
        if speed_candidates:
            speed_kmh = robust_median(speed_candidates)
        else:
            speed_kmh = state.get('target_speed_kmh', 100.0)

    _PERSISTENT_STATE['speed_kmh'] = speed_kmh

    # ----------------------------------------------------
    # 3. Gyroscope Fusion (Yaw Rate)
    # ----------------------------------------------------
    gyro_candidates = []
    for i in [1, 2, 3]:
        if state.get(f"gyro_{i}_valid") and state.get(f"gyro_{i}_yaw_rate_dps") is not None:
            gyro_candidates.append(state[f"gyro_{i}_yaw_rate_dps"])
    if not gyro_candidates:
        for i in [1, 2, 3]:
            if state.get(f"gyro_{i}_yaw_rate_dps") is not None:
                gyro_candidates.append(state[f"gyro_{i}_yaw_rate_dps"])
                
    yaw_rate_dps = robust_median(gyro_candidates) or 0.0

    # ----------------------------------------------------
    # 4. Lidar Parser
    # ----------------------------------------------------
    lidar_data = {}
    for angle in [5, 15, 30, 45, 60, 90, -5, -15, -30, -45, -60, -90]:
        key_base = get_lidar_key(angle)
        r_val = state.get(f"{key_base}_range_m")
        v_val = state.get(f"{key_base}_valid")
        if r_val is not None and r_val > 0.0:
            lidar_data[angle] = {
                'range': r_val,
                'valid': bool(v_val)
            }

    # ----------------------------------------------------
    # 5. Vision Obstacle Fusion & Lidar Cross-Check
    # ----------------------------------------------------
    vision_obstacles = []
    for i in [1, 2, 3]:
        detected = state.get(f"vision_{i}_obstacle_detected")
        valid = state.get(f"vision_{i}_obstacle_valid")
        dist = state.get(f"vision_{i}_obstacle_dist_m")
        rel_speed = state.get(f"vision_{i}_obstacle_rel_speed_kmh")
        if detected and dist is not None:
            vision_obstacles.append({
                'index': i,
                'dist': dist,
                'rel_speed': rel_speed if rel_speed is not None else 0.0,
                'valid': bool(valid)
            })

    def check_lidar_confirmation(dist, tolerance=5.0):
        confirmations = 0
        total_valid_front_beams = 0
        for angle in [-15, -5, 5, 15]:
            if angle in lidar_data:
                total_valid_front_beams += 1
                if abs(lidar_data[angle]['range'] - dist) <= tolerance or lidar_data[angle]['range'] < dist:
                    confirmations += 1
        if total_valid_front_beams == 0:
            return True # Can't crosscheck due to lidar failure, assume True for safety
        return confirmations > 0

    obstacle_detected = False
    obstacle_dist = float('inf')
    obstacle_rel_speed = 0.0

    if vision_obstacles:
        valid_detections = [o for o in vision_obstacles if o['valid']]
        target_detections = valid_detections if valid_detections else vision_obstacles
        
        if len(target_detections) >= 2:
            obstacle_detected = True
            obstacle_dist = robust_median([o['dist'] for o in target_detections])
            obstacle_rel_speed = robust_median([o['rel_speed'] for o in target_detections])
        elif len(target_detections) == 1:
            det = target_detections[0]
            is_confirmed = check_lidar_confirmation(det['dist'], tolerance=max(5.0, det['dist'] * 0.15))
            if is_confirmed or det['dist'] < 15.0 or (det['valid'] and det['dist'] < 40.0):
                obstacle_detected = True
                obstacle_dist = det['dist']
                obstacle_rel_speed = det['rel_speed']

    # Crosscheck: If Lidar sees something very close directly in front of us, respect it
    lidar_min_front_dist = float('inf')
    for angle in [-5, 5]:
        if angle in lidar_data:
            beam = lidar_data[angle]
            if beam['range'] < 70.0:
                if beam['range'] < lidar_min_front_dist:
                    lidar_min_front_dist = beam['range']

    if lidar_min_front_dist < obstacle_dist:
        # Lidar obstacle is closer than what vision reported or vision missed it entirely
        obstacle_detected = True
        obstacle_dist = lidar_min_front_dist
        # If relative speed is unknown, estimate from previous steps or assume stationary
        obstacle_rel_speed = min(0.0, -speed_kmh)

    # ----------------------------------------------------
    # 6. Lane Offset Estimator & Sign Convention Auto-Detection
    # ----------------------------------------------------
    raw_lane_offset = fuse_vision_field(state, "lane_offset_m", "left_line_valid", default_value=None)
    
    # Auto-detect sign multiplier to ensure positive offset means to the right
    sign_multiplier = _PERSISTENT_STATE.get('sign_multiplier')
    if sign_multiplier is None:
        sign_multiplier = 1.0 # Default fallback
        for i in [1, 2, 3]:
            offset = state.get(f"vision_{i}_lane_offset_m")
            left_d = state.get(f"vision_{i}_left_line_dist_m")
            right_d = state.get(f"vision_{i}_right_line_dist_m")
            left_v = state.get(f"vision_{i}_left_line_valid")
            right_v = state.get(f"vision_{i}_right_line_valid")
            if offset is not None and left_v and right_v and left_d is not None and right_d is not None:
                calc = (left_d - right_d) / 2.0
                if abs(calc) > 0.05 and abs(offset) > 0.05:
                    if (offset > 0 and calc < 0) or (offset < 0 and calc > 0):
                        sign_multiplier = -1.0
                    else:
                        sign_multiplier = 1.0
                    _PERSISTENT_STATE['sign_multiplier'] = sign_multiplier
                    break

    prev_offset = _PERSISTENT_STATE.get('lane_offset', 0.0)
    prev_offset_dot = _PERSISTENT_STATE.get('lane_offset_dot', 0.0)
    
    # Determine vision validity in this frame
    vision_valid = False
    for i in [1, 2, 3]:
        if state.get(f"vision_{i}_left_line_valid") or state.get(f"vision_{i}_right_line_valid"):
            vision_valid = True
            break

    if raw_lane_offset is not None and vision_valid:
        # Smooth the reading
        alpha = 0.3
        lane_offset = alpha * raw_lane_offset + (1.0 - alpha) * prev_offset
        raw_dot = (lane_offset - prev_offset) / dt
        lane_offset_dot = 0.3 * raw_dot + 0.7 * prev_offset_dot
    else:
        # Graceful degradation (Flickering / Blind dead reckoning)
        lane_offset = prev_offset + prev_offset_dot * dt
        lane_offset_dot = prev_offset_dot * 0.95 # Slow decay

    _PERSISTENT_STATE['lane_offset'] = lane_offset
    _PERSISTENT_STATE['lane_offset_dot'] = lane_offset_dot

    # ----------------------------------------------------
    # 7. Traction & Environmental Scaling
    # ----------------------------------------------------
    mu = state.get("estimated_mu", 1.0)
    if mu is None:
        mu = 1.0
    rain = state.get("rain_sensor", 0.0) or 0.0
    temp = state.get("temperature_c", 20.0) or 20.0
    
    # Environmental crosscheck
    if temp < 1.0 and rain > 0.1:
        mu = min(mu, 0.25)
    elif rain > 0.5:
        mu = min(mu, 0.40)
        
    traction_scale = min(1.0, max(0.15, mu))

    # ----------------------------------------------------
    # 8. Lane Determination & Lane Change Decision
    # ----------------------------------------------------
    current_lane = state.get("current_lane", 1)
    req_target_lane = state.get("target_lane", current_lane)
    final_target_lane = req_target_lane

    # Abort unsafe lane change
    if req_target_lane != current_lane:
        if not is_lane_safe(req_target_lane, current_lane, lidar_data):
            final_target_lane = current_lane

    # Obstacle avoidance lane change: if blocked in current lane, look for escape
    if obstacle_detected and final_target_lane == current_lane and obstacle_dist < 40.0:
        if current_lane == 1:
            if is_lane_safe(0, 1, lidar_data):
                final_target_lane = 0
            elif is_lane_safe(2, 1, lidar_data):
                final_target_lane = 2
        elif current_lane == 0:
            if is_lane_safe(1, 0, lidar_data):
                final_target_lane = 1
        elif current_lane == 2:
            if is_lane_safe(1, 2, lidar_data):
                final_target_lane = 1

    # ----------------------------------------------------
    # 9. Speed Control
    # ----------------------------------------------------
    desired_speed = state.get("target_speed_kmh", 100.0)
    speed_limit = fuse_vision_field(state, "speed_limit_kmh", "speed_limit_valid", default_value=None)
    if speed_limit is not None and speed_limit > 0:
        desired_speed = min(desired_speed, speed_limit)

    # Slow down for low-mu conditions
    desired_speed = desired_speed * (0.4 + 0.6 * traction_scale)
    # Slow down if we are driving blind due to vision failure
    if not vision_valid:
        desired_speed = desired_speed * 0.75

    # Safe following distance
    speed_m_s = max(1.0, speed_kmh / 3.6)
    reaction_time = 1.8 / traction_scale
    d_buffer = 5.0 / traction_scale
    d_safe = reaction_time * speed_m_s + d_buffer

    # Calculate target speed based on obstacle tracking
    obstacle_speed_kmh = max(0.0, speed_kmh + obstacle_rel_speed)
    obstacle_speed_m_s = obstacle_speed_kmh / 3.6

    if obstacle_detected:
        dist_error = obstacle_dist - d_safe
        if dist_error < 0:
            ratio = max(0.0, obstacle_dist - d_buffer) / max(1.0, d_safe - d_buffer)
            target_speed_obs = obstacle_speed_kmh * ratio
            if obstacle_dist < d_buffer:
                target_speed_obs = 0.0
        else:
            if obstacle_rel_speed < -5.0:
                target_speed_obs = obstacle_speed_kmh + 0.1 * dist_error * 3.6
            else:
                target_speed_obs = desired_speed
        control_target_speed = min(desired_speed, target_speed_obs)
    else:
        control_target_speed = desired_speed

    # Speed PID controller
    speed_error = control_target_speed - speed_kmh
    speed_integral = _PERSISTENT_STATE.get('speed_integral', 0.0)
    prev_throttle = _PERSISTENT_STATE.get('prev_throttle', 0.0)
    
    # Windup guard
    if 0.0 < prev_throttle < 1.0 or (prev_throttle >= 1.0 and speed_error < 0) or (prev_throttle <= 0.0 and speed_error > 0):
        speed_integral += speed_error * dt
    _PERSISTENT_STATE['speed_integral'] = speed_integral

    Kp_speed = 0.04
    Ki_speed = 0.004
    throttle_pid = Kp_speed * speed_error + Ki_speed * speed_integral
    throttle_cmd = max(0.0, min(1.0, throttle_pid))

    # Deceleration-based Braking
    brake_cmd = 0.0
    if obstacle_detected:
        if speed_m_s > obstacle_speed_m_s or obstacle_dist < d_safe:
            target_dist = d_buffer if obstacle_speed_m_s < 1.0 else d_safe
            dist_to_target = max(1.0, obstacle_dist - target_dist)
            
            # Physics-based deceleration required: v^2 - v0^2 = 2 * a * d
            required_decel = (speed_m_s**2 - obstacle_speed_m_s**2) / (2.0 * dist_to_target)
            
            if obstacle_dist < d_safe:
                # Add proportional penalty for violating safety distance
                required_decel += 1.5 * (d_safe - obstacle_dist) / traction_scale
                
            if required_decel > 0.3:
                throttle_cmd = 0.0
                max_decel = 8.0 * traction_scale
                brake_cmd = max(0.0, min(1.0, required_decel / max_decel))
                
                if obstacle_dist < max(5.0, d_buffer):
                    brake_cmd = 1.0
                    throttle_cmd = 0.0

    # Over-speeding / Stuck-throttle brake recovery
    if speed_error < -2.0:
        throttle_cmd = 0.0
        # Ride the brakes to hold the speed
        brake_cmd = max(brake_cmd, min(0.8, -speed_error * 0.05))

    # ----------------------------------------------------
    # 10. Steering Control (Lateral PID + Curvature FF + Gyro Damping)
    # ----------------------------------------------------
    pos_right_m = lane_offset * sign_multiplier
    target_pos_right_m = (final_target_lane - current_lane) * 3.6
    lateral_error = pos_right_m - target_pos_right_m
    lateral_error_dot = lane_offset_dot * sign_multiplier

    steer_integral = _PERSISTENT_STATE.get('steer_integral', 0.0)
    prev_steer = _PERSISTENT_STATE.get('prev_steer', 0.0)
    
    if abs(prev_steer) < 1.0 or (prev_steer >= 1.0 and lateral_error < 0) or (prev_steer <= -1.0 and lateral_error > 0):
        steer_integral += lateral_error * dt
    _PERSISTENT_STATE['steer_integral'] = steer_integral

    # Adjust PID gains based on speed sensitivity
    gain_scale = 20.0 / speed_m_s
    Kp_steer = 0.18 * gain_scale
    Kd_steer = 0.12 * gain_scale
    Ki_steer = 0.015 * gain_scale

    # Yaw damping term (helps with crosswinds & oscillations)
    yaw_rate_rad = math.radians(yaw_rate_dps)
    damping_term = -0.15 * gain_scale * yaw_rate_rad

    # Curvature feedforward
    raw_curvature = fuse_vision_field(state, "road_curvature_1pm", "left_line_valid", default_value=0.0)
    prev_curvature = _PERSISTENT_STATE.get('road_curvature', 0.0)
    road_curvature = 0.2 * raw_curvature + 0.8 * prev_curvature
    _PERSISTENT_STATE['road_curvature'] = road_curvature
    ff_term = road_curvature * 25.0

    steering_cmd = -(Kp_steer * lateral_error + Kd_steer * lateral_error_dot + Ki_steer * steer_integral) + damping_term + ff_term
    steering_cmd = max(-1.0, min(1.0, steering_cmd))

    # Steering rate limits based on traction
    max_steer_rate = 3.0 * traction_scale
    max_steer_change = max_steer_rate * dt
    steering_change = steering_cmd - prev_steer
    steering_cmd = prev_steer + max(-max_steer_change, min(max_steer_change, steering_change))

    # ----------------------------------------------------
    # 11. Traction Protection limits
    # ----------------------------------------------------
    max_throttle_limit = 0.2 + 0.8 * traction_scale
    max_brake_limit = 0.2 + 0.8 * traction_scale
    
    throttle_cmd = min(max_throttle_limit, throttle_cmd)
    brake_cmd = min(max_brake_limit, brake_cmd)

    _PERSISTENT_STATE['prev_steer'] = steering_cmd
    _PERSISTENT_STATE['prev_throttle'] = throttle_cmd
    _PERSISTENT_STATE['prev_brake'] = brake_cmd

    return {
        "steering": steering_cmd,
        "throttle": throttle_cmd,
        "brake": brake_cmd
    }
