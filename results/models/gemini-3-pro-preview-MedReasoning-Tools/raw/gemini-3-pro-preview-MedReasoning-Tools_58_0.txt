const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    if (!input.trim()) {
        console.log(JSON.stringify({ assignments: [] }));
        return;
    }
    const data = JSON.parse(input);
    const grid = data.grid;
    const robots = data.robots;
    const jobs = data.jobs;

    if (!grid || grid.length === 0 || !robots || robots.length === 0 || !jobs || jobs.length === 0) {
        console.log(JSON.stringify({ assignments: [] }));
        return;
    }

    const H = grid.length;
    const W = grid[0].length;

    const blocked = new Uint8Array(W * H);
    for (let y = 0; y < H; y++) {
        const row = grid[y];
        for (let x = 0; x < W; x++) {
            if (row[x] === '#') {
                blocked[y * W + x] = 1;
            }
        }
    }

    robots.forEach((r, idx) => {
        r.nodeIdx = idx;
    });
    jobs.forEach((j, idx) => {
        j.nodeIdx = robots.length + idx;
    });

    const nodes = [];
    for (const r of robots) {
        nodes.push(r);
    }
    for (const j of jobs) {
        nodes.push(j);
    }

    const N = nodes.length;
    const maxRobotId = Math.max(...robots.map(r => r.id));

    let tourId = 0;
    const visited = new Int32Array(W * H);
    const dist = new Int32Array(W * H);
    const queue = new Int32Array(W * H);
    const cachedDists = new Array(N).fill(null);

    function getDistancesFromNode(nodeIdx) {
        if (cachedDists[nodeIdx] !== null) {
            return cachedDists[nodeIdx];
        }

        const node = nodes[nodeIdx];
        tourId++;
        let head = 0;
        let tail = 0;

        const startIdx = node.y * W + node.x;
        visited[startIdx] = tourId;
        dist[startIdx] = 0;
        queue[tail++] = startIdx;

        const dx = [0, 0, 1, -1];
        const dy = [1, -1, 0, 0];

        while (head < tail) {
            const u = queue[head++];
            const ux = u % W;
            const uy = Math.floor(u / W);
            const d = dist[u];

            for (let i = 0; i < 4; i++) {
                const vx = ux + dx[i];
                const vy = uy + dy[i];
                if (vx >= 0 && vx < W && vy >= 0 && vy < H) {
                    const v = vy * W + vx;
                    if (!blocked[v] && visited[v] !== tourId) {
                        visited[v] = tourId;
                        dist[v] = d + 1;
                        queue[tail++] = v;
                    }
                }
            }
        }

        const dists = new Int32Array(N);
        for (let i = 0; i < N; i++) {
            const targetIdx = nodes[i].y * W + nodes[i].x;
            if (visited[targetIdx] === tourId) {
                dists[i] = dist[targetIdx];
            } else {
                dists[i] = -1;
            }
        }

        cachedDists[nodeIdx] = dists;
        return dists;
    }

    function getDistance(uIdx, vIdx) {
        const dists = getDistancesFromNode(uIdx);
        return dists[vIdx];
    }

    function solvePFG() {
        const assignments = Array.from({ length: maxRobotId + 1 }, () => []);
        const sortedJobs = [...jobs].sort((a, b) => {
            if (b.priority !== a.priority) {
                return b.priority - a.priority;
            }
            return a.id - b.id;
        });

        const robotCurrentNode = robots.map(r => r.nodeIdx);

        for (const job of sortedJobs) {
            let minD = Infinity;
            let bestRIdx = -1;

            for (let r = 0; r < robots.length; r++) {
                const currNode = robotCurrentNode[r];
                const d = getDistance(currNode, job.nodeIdx);
                if (d !== -1 && d < minD) {
                    minD = d;
                    bestRIdx = r;
                }
            }

            if (bestRIdx === -1) {
                bestRIdx = 0;
            }

            const robot = robots[bestRIdx];
            assignments[robot.id].push(job.id);
            robotCurrentNode[bestRIdx] = job.nodeIdx;
        }

        return assignments;
    }

    function solveCFG() {
        const assignments = Array.from({ length: maxRobotId + 1 }, () => []);
        const assigned = new Uint8Array(jobs.length);

        const robotCurrentNode = robots.map(r => r.nodeIdx);
        const closestJobIdx = new Int32Array(robots.length).fill(-1);
        const closestDist = new Int32Array(robots.length).fill(-1);

        function updateClosest(r) {
            let minDist = Infinity;
            let bestJobIdx = -1;
            const currNode = robotCurrentNode[r];
            const dists = getDistancesFromNode(currNode);

            for (let j = 0; j < jobs.length; j++) {
                if (!assigned[j]) {
                    const jobNodeIdx = jobs[j].nodeIdx;
                    const d = dists[jobNodeIdx];
                    if (d !== -1 && d < minDist) {
                        minDist = d;
                        bestJobIdx = j;
                    }
                }
            }
            closestJobIdx[r] = bestJobIdx;
            closestDist[r] = bestJobIdx === -1 ? Infinity : minDist;
        }

        for (let r = 0; r < robots.length; r++) {
            updateClosest(r);
        }

        for (let step = 0; step < jobs.length; step++) {
            let minDist = Infinity;
            let bestRIdx = -1;

            for (let r = 0; r < robots.length; r++) {
                if (closestDist[r] < minDist) {
                    minDist = closestDist[r];
                    bestRIdx = r;
                }
            }

            if (bestRIdx === -1 || minDist === Infinity) {
                const fallbackRobot = robots[0];
                for (let j = 0; j < jobs.length; j++) {
                    if (!assigned[j]) {
                        assignments[fallbackRobot.id].push(jobs[j].id);
                        assigned[j] = 1;
                    }
                }
                break;
            }

            const bestJIdx = closestJobIdx[bestRIdx];
            const job = jobs[bestJIdx];
            const robot = robots[bestRIdx];

            assignments[robot.id].push(job.id);
            assigned[bestJIdx] = 1;
            robotCurrentNode[bestRIdx] = job.nodeIdx;

            updateClosest(bestRIdx);
            for (let r = 0; r < robots.length; r++) {
                if (r !== bestRIdx && closestJobIdx[r] === bestJIdx) {
                    updateClosest(r);
                }
            }
        }

        return assignments;
    }

    function solvePGCFG() {
        const assignments = Array.from({ length: maxRobotId + 1 }, () => []);

        const groups = new Map();
        for (const job of jobs) {
            if (!groups.has(job.priority)) {
                groups.set(job.priority, []);
            }
            groups.get(job.priority).push(job);
        }

        const sortedPriorities = Array.from(groups.keys()).sort((a, b) => b - a);
        const robotCurrentNode = robots.map(r => r.nodeIdx);

        for (const priority of sortedPriorities) {
            const groupJobs = groups.get(priority);
            const groupAssigned = new Uint8Array(groupJobs.length);

            const closestJobIdxInGroup = new Int32Array(robots.length).fill(-1);
            const closestDist = new Int32Array(robots.length).fill(-1);

            const updateClosestForGroup = (r) => {
                let minDist = Infinity;
                let bestJobIdx = -1;
                const currNode = robotCurrentNode[r];
                const dists = getDistancesFromNode(currNode);

                for (let j = 0; j < groupJobs.length; j++) {
                    if (!groupAssigned[j]) {
                        const jobNodeIdx = groupJobs[j].nodeIdx;
                        const d = dists[jobNodeIdx];
                        if (d !== -1 && d < minDist) {
                            minDist = d;
                            bestJobIdx = j;
                        }
                    }
                }
                closestJobIdxInGroup[r] = bestJobIdx;
                closestDist[r] = bestJobIdx === -1 ? Infinity : minDist;
            };

            for (let r = 0; r < robots.length; r++) {
                updateClosestForGroup(r);
            }

            for (let step = 0; step < groupJobs.length; step++) {
                let minDist = Infinity;
                let bestRIdx = -1;
                for (let r = 0; r < robots.length; r++) {
                    if (closestDist[r] < minDist) {
                        minDist = closestDist[r];
                        bestRIdx = r;
                    }
                }

                if (bestRIdx === -1 || minDist === Infinity) {
                    const fallbackRobot = robots[0];
                    for (let j = 0; j < groupJobs.length; j++) {
                        if (!groupAssigned[j]) {
                            assignments[fallbackRobot.id].push(groupJobs[j].id);
                            groupAssigned[j] = 1;
                        }
                    }
                    break;
                }

                const bestJIdx = closestJobIdxInGroup[bestRIdx];
                const job = groupJobs[bestJIdx];
                const robot = robots[bestRIdx];

                assignments[robot.id].push(job.id);
                groupAssigned[bestJIdx] = 1;
                robotCurrentNode[bestRIdx] = job.nodeIdx;

                updateClosestForGroup(bestRIdx);
                for (let r = 0; r < robots.length; r++) {
                    if (r !== bestRIdx && closestJobIdxInGroup[r] === bestJIdx) {
                        updateClosestForGroup(r);
                    }
                }
            }
        }

        return assignments;
    }

    function evaluateAssignment(assignments) {
        let totalTravel = 0;
        const jobMap = new Map();
        for (const job of jobs) {
            jobMap.set(job.id, job);
        }

        for (let r = 0; r < robots.length; r++) {
            const robot = robots[r];
            const path = assignments[robot.id] || [];
            if (path.length === 0) continue;

            let currNodeIdx = robot.nodeIdx;
            for (const jobId of path) {
                const job = jobMap.get(jobId);
                const d = getDistance(currNodeIdx, job.nodeIdx);
                if (d !== -1) {
                    totalTravel += d;
                }
                currNodeIdx = job.nodeIdx;
            }
        }
        return totalTravel;
    }

    function optimizeLocalSearch(assignments, startTime, maxTimeMs) {
        let improved = true;
        const jobMap = new Map();
        for (const job of jobs) {
            jobMap.set(job.id, job);
        }

        function getPathDistance(robot, path) {
            if (path.length === 0) return 0;
            let distSum = 0;
            let currNodeIdx = robot.nodeIdx;
            for (const jobId of path) {
                const job = jobMap.get(jobId);
                const d = getDistance(currNodeIdx, job.nodeIdx);
                if (d !== -1) {
                    distSum += d;
                }
                currNodeIdx = job.nodeIdx;
            }
            return distSum;
        }

        while (improved) {
            if (Date.now() - startTime > maxTimeMs) {
                break;
            }
            improved = false;

            // 1. Intra-route optimization (swap consecutive)
            for (let r = 0; r < robots.length; r++) {
                const robot = robots[r];
                const path = assignments[robot.id];
                if (!path || path.length < 2) continue;

                for (let i = 0; i < path.length - 1; i++) {
                    const oldDist = getPathDistance(robot, path);

                    const temp = path[i];
                    path[i] = path[i + 1];
                    path[i + 1] = temp;

                    const newDist = getPathDistance(robot, path);

                    if (newDist < oldDist) {
                        improved = true;
                    } else {
                        const temp2 = path[i];
                        path[i] = path[i + 1];
                        path[i + 1] = temp2;
                    }
                }
            }

            // 2. Inter-route optimization (move a job to another robot)
            for (let r1 = 0; r1 < robots.length; r1++) {
                const robot1 = robots[r1];
                const path1 = assignments[robot1.id];
                if (!path1 || path1.length === 0) continue;

                for (let r2 = 0; r2 < robots.length; r2++) {
                    if (r1 === r2) continue;
                    const robot2 = robots[r2];
                    const path2 = assignments[robot2.id] || [];

                    for (let i = 0; i < path1.length; i++) {
                        const jobId = path1[i];

                        for (let j = 0; j <= path2.length; j++) {
                            const oldDist1 = getPathDistance(robot1, path1);
                            const oldDist2 = getPathDistance(robot2, path2);

                            const newPath1 = path1.filter((_, idx) => idx !== i);
                            const newPath2 = [...path2];
                            newPath2.splice(j, 0, jobId);

                            const newDist1 = getPathDistance(robot1, newPath1);
                            const newDist2 = getPathDistance(robot2, newPath2);

                            if (newDist1 + newDist2 < oldDist1 + oldDist2) {
                                assignments[robot1.id] = newPath1;
                                assignments[robot2.id] = newPath2;
                                improved = true;
                                break;
                            }
                        }
                        if (improved) break;
                    }
                    if (improved) break;
                }
                if (improved) break;
            }
        }
    }

    const strategies = [
        { name: 'CFG', solve: solveCFG },
        { name: 'PFG', solve: solvePFG },
        { name: 'PGCFG', solve: solvePGCFG }
    ];

    let bestAssignment = null;
    let minTotalTravel = Infinity;

    for (const strategy of strategies) {
        const assignments = strategy.solve();
        const initialTravel = evaluateAssignment(assignments);
        optimizeLocalSearch(assignments, Date.now(), 600); // 600ms per strategy
        const travel = evaluateAssignment(assignments);
        console.warn(`Strategy: ${strategy.name}, Initial Travel: ${initialTravel}, Optimized Travel: ${travel}, Assignments: ${JSON.stringify(assignments)}`);
        if (travel < minTotalTravel) {
            minTotalTravel = travel;
            bestAssignment = assignments;
        }
    }

    console.log(JSON.stringify({ assignments: bestAssignment }));
}

solve();
