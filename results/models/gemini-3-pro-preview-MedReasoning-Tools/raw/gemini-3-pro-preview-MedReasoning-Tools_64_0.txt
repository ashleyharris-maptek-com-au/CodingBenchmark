declare const require: any;
const fs = require('fs');

interface Point {
    x: number;
    y: number;
}

interface Drone {
    id: number;
    x: number;
    y: number;
}

interface PathNode {
    idx: number;
    parent: PathNode | null;
}

interface State {
    x: number;
    y: number;
    idx: number;
    score: number;
    evalScore: number;
    path: PathNode;
}

// Generate a random test grid
function generateGrid(W: number, H: number, obstacleProb: number = 0.1, maxProb: number = 100): number[][] {
    const grid: number[][] = [];
    for (let y = 0; y < H; y++) {
        const row: number[] = [];
        for (let x = 0; x < W; x++) {
            if (Math.random() < obstacleProb) {
                row.push(-1);
            } else {
                row.push(Math.random() < 0.3 ? Math.floor(Math.random() * maxProb) : 0);
            }
        }
        grid.push(row);
    }
    return grid;
}

// Reference Greedy (simultaneous/sequential step-by-step)
function runGreedyReference(
    probability: number[][],
    drones: Drone[],
    maxSteps: number
): { paths: number[][][]; score: number } {
    const H = probability.length;
    const W = probability[0].length;
    const remaining = probability.map(row => [...row]);
    
    const paths: number[][][] = drones.map(d => [[d.x, d.y]]);
    let totalScore = 0;
    
    // Initial positions
    for (const d of drones) {
        if (remaining[d.y][d.x] > 0) {
            totalScore += remaining[d.y][d.x];
            remaining[d.y][d.x] = 0;
        }
    }
    
    const dy = [0, -1, 1, 0, 0]; // stay, up, down, left, right
    const dx = [0, 0, 0, -1, 1];
    
    for (let step = 0; step < maxSteps; step++) {
        for (let i = 0; i < drones.length; i++) {
            const path = paths[i];
            const [cx, cy] = path[path.length - 1];
            
            let bestX = cx;
            let bestY = cy;
            let bestVal = -2;
            
            for (let d = 0; d < 5; d++) {
                const nx = cx + dx[d];
                const ny = cy + dy[d];
                if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
                    const val = remaining[ny][nx];
                    if (val !== -1) {
                        if (val > bestVal) {
                            bestVal = val;
                            bestX = nx;
                            bestY = ny;
                        }
                    }
                }
            }
            
            path.push([bestX, bestY]);
            if (remaining[bestY][bestX] > 0) {
                totalScore += remaining[bestY][bestX];
                remaining[bestY][bestX] = 0;
            }
        }
    }
    return { paths, score: totalScore };
}

// Diffuse remaining probability to compute a potential field
function computePotential(
    W: number,
    H: number,
    isObstacle: Uint8Array,
    remaining: Float32Array,
    iterations: number
): Float32Array {
    const size = H * W;
    let potential = new Float32Array(size);
    potential.set(remaining);
    
    let nextPotential = new Float32Array(size);
    
    for (let iter = 0; iter < iterations; iter++) {
        nextPotential.fill(0);
        for (let y = 0; y < H; y++) {
            const rowOffset = y * W;
            for (let x = 0; x < W; x++) {
                const idx = rowOffset + x;
                if (isObstacle[idx]) continue;
                
                let sumNeighbors = 0;
                
                // up
                if (y > 0) {
                    const uIdx = idx - W;
                    if (!isObstacle[uIdx]) sumNeighbors += potential[uIdx];
                }
                // down
                if (y < H - 1) {
                    const dIdx = idx + W;
                    if (!isObstacle[dIdx]) sumNeighbors += potential[dIdx];
                }
                // left
                if (x > 0) {
                    const lIdx = idx - 1;
                    if (!isObstacle[lIdx]) sumNeighbors += potential[lIdx];
                }
                // right
                if (x < W - 1) {
                    const rIdx = idx + 1;
                    if (!isObstacle[rIdx]) sumNeighbors += potential[rIdx];
                }
                
                nextPotential[idx] = potential[idx] * 0.9 + 0.05 * sumNeighbors;
            }
        }
        // swap
        const temp = potential;
        potential = nextPotential;
        nextPotential = temp;
    }
    
    return potential;
}

// Check if linked list path contains target cell index
function pathContains(node: PathNode | null, target: number): boolean {
    let curr: PathNode | null = node;
    while (curr !== null) {
        if (curr.idx === target) return true;
        curr = curr.parent;
    }
    return false;
}

// Advanced Optimizer using LNS and Beam Search
function runAdvancedOptimizer(
    probability: number[][],
    drones: Drone[],
    maxSteps: number,
    timeLimitMs: number = 3000
): { paths: number[][][]; score: number } {
    const startTime = Date.now();
    const H = probability.length;
    const W = probability[0].length;
    const size = H * W;
    
    // Flat arrays for speed
    const origProb = new Float32Array(size);
    const isObstacle = new Uint8Array(size);
    for (let y = 0; y < H; y++) {
        const rowOffset = y * W;
        for (let x = 0; x < W; x++) {
            const idx = rowOffset + x;
            const val = probability[y][x];
            if (val === -1) {
                isObstacle[idx] = 1;
            } else {
                origProb[idx] = val;
            }
        }
    }
    
    // Initial paths using greedy reference
    const initGreedy = runGreedyReference(probability, drones, maxSteps);
    let bestPaths = initGreedy.paths.map(p => p.map(pt => [...pt]));
    let bestScore = initGreedy.score;
    
    // Track coverage count globally
    const visitedCount = new Int32Array(size);
    const activeRemaining = new Float32Array(origProb);
    
    // Initialize coverage tracker
    for (const path of bestPaths) {
        for (const [x, y] of path) {
            const idx = y * W + x;
            if (visitedCount[idx] === 0) {
                activeRemaining[idx] = 0;
            }
            visitedCount[idx]++;
        }
    }
    
    const dy = [0, -1, 1, 0, 0];
    const dx = [0, 0, 0, -1, 1];
    
    // Large Neighborhood Search (LNS) iterations
    let iterations = 0;
    const beamSize = 15;
    const potentialWeight = 1.0;
    
    let totalPotentialTime = 0;
    let totalBeamTime = 0;
    
    while (Date.now() - startTime < timeLimitMs) {
        iterations++;
        
        // Recompute potential field once per LNS iteration based on remaining probability
        const iterPotentialIter = Math.min(maxSteps, 10);
        const t0 = Date.now();
        const potential = computePotential(W, H, isObstacle, activeRemaining, iterPotentialIter);
        totalPotentialTime += Date.now() - t0;
        
        const t1 = Date.now();
        // Replan each drone
        for (let i = 0; i < drones.length; i++) {
            // Check time limit inside loop
            if (Date.now() - startTime >= timeLimitMs) break;
            
            const drone = drones[i];
            const oldPath = bestPaths[i];
            
            // 1. Remove this drone's path from visitedCount and restore activeRemaining
            for (const [x, y] of oldPath) {
                const idx = y * W + x;
                visitedCount[idx]--;
                if (visitedCount[idx] === 0) {
                    activeRemaining[idx] = origProb[idx];
                }
            }
            
            // 2. Replan path using Beam Search
            const startIdx = drone.y * W + drone.x;
            const startScore = activeRemaining[startIdx];
            let beam: State[] = [{
                x: drone.x,
                y: drone.y,
                idx: startIdx,
                score: startScore,
                evalScore: startScore + potential[startIdx] * Math.min(maxSteps, 8) * potentialWeight,
                path: { idx: startIdx, parent: null }
            }];
            
            for (let step = 0; step < maxSteps; step++) {
                const candidates: State[] = [];
                const remainingSteps = maxSteps - step;
                const minRemSteps = Math.min(remainingSteps, 8);
                
                for (const state of beam) {
                    for (const d of [0, 1, 2, 3, 4]) { // stay, up, down, left, right
                        const nx = state.x + dx[d];
                        const ny = state.y + dy[d];
                        if (nx >= 0 && nx < W && ny >= 0 && ny < H) {
                            const nidx = ny * W + nx;
                            if (!isObstacle[nidx]) {
                                let gain = 0;
                                if (activeRemaining[nidx] > 0) {
                                    if (!pathContains(state.path, nidx)) {
                                        gain = activeRemaining[nidx];
                                    }
                                }
                                const nscore = state.score + gain;
                                const evalScore = nscore + potential[nidx] * minRemSteps * potentialWeight;
                                
                                candidates.push({
                                    x: nx,
                                    y: ny,
                                    idx: nidx,
                                    score: nscore,
                                    evalScore: evalScore,
                                    path: { idx: nidx, parent: state.path }
                                });
                            }
                        }
                    }
                }
                
                if (candidates.length === 0) break;
                
                // Sort candidates by evaluation score
                candidates.sort((a, b) => b.evalScore - a.evalScore);
                
                // Keep top B, using diversity heuristic (max 2 states per coordinate)
                const nextBeam: State[] = [];
                for (let cIdx = 0; cIdx < candidates.length; cIdx++) {
                    const cand = candidates[cIdx];
                    let count = 0;
                    for (let j = 0; j < nextBeam.length; j++) {
                        if (nextBeam[j].idx === cand.idx) {
                            count++;
                        }
                    }
                    if (count < 2) {
                        nextBeam.push(cand);
                        if (nextBeam.length >= beamSize) break;
                    }
                }
                beam = nextBeam;
            }
            
            // Best path found by Beam Search
            const bestState = beam[0];
            const newPathPoints: number[][] = [];
            let curr: PathNode | null = bestState.path;
            while (curr !== null) {
                newPathPoints.push([curr.idx % W, Math.floor(curr.idx / W)]);
                curr = curr.parent;
            }
            newPathPoints.reverse();
            
            // 3. Add the new path to visitedCount and remove from activeRemaining
            bestPaths[i] = newPathPoints;
            for (const [x, y] of newPathPoints) {
                const idx = y * W + x;
                if (visitedCount[idx] === 0) {
                    activeRemaining[idx] = 0;
                }
                visitedCount[idx]++;
            }
        }
        totalBeamTime += Date.now() - t1;
    }
    
    console.log(`Potential time: ${totalPotentialTime}ms, Beam time: ${totalBeamTime}ms`);
    
    // Evaluate final score
    const finalScore = evaluatePaths(bestPaths);
    
    // Helper to calculate total score of a set of paths
    function evaluatePaths(paths: number[][][]): number {
        const visited = new Uint8Array(size);
        let score = 0;
        for (const path of paths) {
            for (const [x, y] of path) {
                const idx = y * W + x;
                if (!visited[idx]) {
                    visited[idx] = 1;
                    if (origProb[idx] > 0) {
                        score += origProb[idx];
                    }
                }
            }
        }
        return score;
    }
    
    console.log(`LNS completed ${iterations} iterations in ${Date.now() - startTime}ms`);
    return { paths: bestPaths, score: finalScore };
}

// Run test comparison
function test() {
    const W = 500;
    const H = 500;
    const numDrones = 50;
    const maxSteps = 100;
    
    console.log(`Generating grid ${W}x${H}...`);
    const probability = generateGrid(W, H, 0.1, 100);
    const drones: Drone[] = [];
    // Place drones at random non-obstacle cells
    for (let i = 0; i < numDrones; i++) {
        let x = 0, y = 0;
        do {
            x = Math.floor(Math.random() * W);
            y = Math.floor(Math.random() * H);
        } while (probability[y][x] === -1);
        drones.push({ id: i, x, y });
    }
    
    console.log("Running Greedy Reference...");
    const ref = runGreedyReference(probability, drones, maxSteps);
    console.log(`Greedy Reference Score: ${ref.score}`);
    
    console.log("Running Advanced Optimizer...");
    const adv = runAdvancedOptimizer(probability, drones, maxSteps, 2000);
    console.log(`Advanced Optimizer Score: ${adv.score}`);
    
    const imp = ((adv.score - ref.score) / ref.score) * 100;
    console.log(`Improvement: ${imp.toFixed(2)}%`);
}

test();
