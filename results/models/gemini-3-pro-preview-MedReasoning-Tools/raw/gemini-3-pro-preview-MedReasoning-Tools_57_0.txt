const fs = require('fs');

function solve() {
    let input;
    try {
        input = fs.readFileSync(0, 'utf8');
    } catch (e) {
        return;
    }
    if (!input) return;
    const data = JSON.parse(input);

    const width = data.width;
    const height = data.height;
    const grid = data.grid;
    const ignitions = data.ignitions;
    const towns = data.towns;
    const cutBudget = data.cutBudget;

    if (cutBudget <= 0) {
        console.log(JSON.stringify({ firebreaks: [] }));
        return;
    }

    const ignitionSet = new Set();
    for (const [x, y] of ignitions) {
        if (x >= 0 && x < width && y >= 0 && y < height) {
            ignitionSet.add(y * width + x);
        }
    }

    // Run BFS to find all reachable cells from ignitions
    const reachable = new Uint8Array(width * height);
    const queue = new Int32Array(width * height);
    let headQ = 0, tailQ = 0;

    for (const cellIdx of ignitionSet) {
        if (grid[Math.floor(cellIdx / width)][cellIdx % width] === '.') {
            reachable[cellIdx] = 1;
            queue[tailQ++] = cellIdx;
        }
    }

    const dx = [1, -1, 0, 0];
    const dy = [0, 0, 1, -1];

    while (headQ < tailQ) {
        const curr = queue[headQ++];
        const cx = curr % width;
        const cy = Math.floor(curr / width);
        for (let i = 0; i < 4; i++) {
            const nx = cx + dx[i];
            const ny = cy + dy[i];
            if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                if (grid[ny][nx] === '.') {
                    const nidx = ny * width + nx;
                    if (!reachable[nidx]) {
                        reachable[nidx] = 1;
                        queue[tailQ++] = nidx;
                    }
                }
            }
        }
    }

    const R = [];
    const gridToFlow = new Int32Array(width * height);
    gridToFlow.fill(-1);
    for (let i = 0; i < width * height; i++) {
        if (reachable[i]) {
            gridToFlow[i] = R.length;
            R.push(i);
        }
    }

    const M = R.length;
    if (M === 0) {
        console.log(JSON.stringify({ firebreaks: [] }));
        return;
    }

    const cellToTown = new Map();
    for (const t of towns) {
        const cellIdx = t.y * width + t.x;
        if (cellIdx >= 0 && cellIdx < width * height && gridToFlow[cellIdx] >= 0) {
            if (cellToTown.has(cellIdx)) {
                const existing = cellToTown.get(cellIdx);
                existing.value += t.value;
            } else {
                cellToTown.set(cellIdx, {
                    cellIdx: cellIdx,
                    flowIdx: gridToFlow[cellIdx],
                    value: t.value
                });
            }
        }
    }

    const activeTowns = Array.from(cellToTown.values());

    if (activeTowns.length === 0) {
        console.log(JSON.stringify({ firebreaks: [] }));
        return;
    }

    const INF = 1000000000;
    const numNodes = 2 * M + 2;
    const S_star = 2 * M;
    const T_star = 2 * M + 1;

    const head = new Int32Array(numNodes);
    head.fill(-1);

    const maxEdges = 16 * M + activeTowns.length * 2 + 100;
    const to = new Int32Array(maxEdges);
    const cap = new Int32Array(maxEdges);
    const flow = new Int32Array(maxEdges);
    const next = new Int32Array(maxEdges);
    let edgeCount = 0;

    function addEdge(u, v, c) {
        to[edgeCount] = v;
        cap[edgeCount] = c;
        flow[edgeCount] = 0;
        next[edgeCount] = head[u];
        head[u] = edgeCount++;

        to[edgeCount] = u;
        cap[edgeCount] = 0;
        flow[edgeCount] = 0;
        next[edgeCount] = head[v];
        head[v] = edgeCount++;
    }

    for (let i = 0; i < M; i++) {
        const cellIdx = R[i];
        const isIgnition = ignitionSet.has(cellIdx);
        const isTown = cellToTown.has(cellIdx);
        const capacity = (isIgnition || isTown) ? INF : 1;
        addEdge(2 * i, 2 * i + 1, capacity);
    }

    for (const cellIdx of ignitionSet) {
        const flowIdx = gridToFlow[cellIdx];
        if (flowIdx >= 0) {
            addEdge(S_star, 2 * flowIdx, INF);
        }
    }

    const townEdgeIdx = new Int32Array(activeTowns.length);
    for (let i = 0; i < activeTowns.length; i++) {
        const t = activeTowns[i];
        townEdgeIdx[i] = edgeCount;
        addEdge(2 * t.flowIdx + 1, T_star, 0);
    }

    for (let i = 0; i < M; i++) {
        const cx = R[i] % width;
        const cy = Math.floor(R[i] / width);
        for (let k = 0; k < 4; k++) {
            const nx = cx + dx[k];
            const ny = cy + dy[k];
            if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                const nidx = ny * width + nx;
                const j = gridToFlow[nidx];
                if (j >= 0) {
                    addEdge(2 * i + 1, 2 * j, INF);
                }
            }
        }
    }

    const level = new Int32Array(numNodes);
    const queueArr = new Int32Array(numNodes);
    const work = new Int32Array(numNodes);
    const pathNodes = new Int32Array(numNodes);
    const pathEdges = new Int32Array(numNodes);

    function bfsDinic(source, sink) {
        level.fill(-1);
        level[source] = 0;
        let headQ = 0, tailQ = 0;
        queueArr[tailQ++] = source;
        while (headQ < tailQ) {
            const u = queueArr[headQ++];
            for (let e = head[u]; e !== -1; e = next[e]) {
                const v = to[e];
                if (cap[e] - flow[e] > 0 && level[v] === -1) {
                    level[v] = level[u] + 1;
                    queueArr[tailQ++] = v;
                }
            }
        }
        return level[sink] !== -1;
    }

    function runDinic(source, sink, limit) {
        let totalFlow = 0;
        while (totalFlow <= limit) {
            if (!bfsDinic(source, sink)) break;
            work.set(head);
            
            let depth = 0;
            pathNodes[0] = source;
            
            while (true) {
                let u = pathNodes[depth];
                if (u === sink) {
                    let pathFlow = limit - totalFlow + 1;
                    for (let d = 0; d < depth; d++) {
                        const e = pathEdges[d];
                        const tr = cap[e] - flow[e];
                        if (tr < pathFlow) pathFlow = tr;
                    }
                    
                    for (let d = 0; d < depth; d++) {
                        const e = pathEdges[d];
                        flow[e] += pathFlow;
                        flow[e ^ 1] -= pathFlow;
                    }
                    
                    totalFlow += pathFlow;
                    if (totalFlow > limit) break;
                    
                    let newDepth = 0;
                    for (let d = 0; d < depth; d++) {
                        const e = pathEdges[d];
                        if (cap[e] - flow[e] === 0) {
                            newDepth = d;
                            break;
                        }
                    }
                    depth = newDepth;
                } else {
                    let foundEdge = -1;
                    for (let e = work[u]; e !== -1; e = next[e]) {
                        work[u] = e;
                        const v = to[e];
                        if (level[v] === level[u] + 1 && cap[e] - flow[e] > 0) {
                            foundEdge = e;
                            break;
                        }
                    }
                    if (foundEdge !== -1) {
                        pathEdges[depth] = foundEdge;
                        pathNodes[depth + 1] = to[foundEdge];
                        depth++;
                    } else {
                        level[u] = -1;
                        if (depth === 0) break;
                        depth--;
                    }
                }
            }
        }
        return totalFlow;
    }

    function getReachableNodes(source) {
        const visited = new Uint8Array(numNodes);
        const q = new Int32Array(numNodes);
        let headQ = 0, tailQ = 0;
        visited[source] = 1;
        q[tailQ++] = source;
        while (headQ < tailQ) {
            const u = q[headQ++];
            for (let e = head[u]; e !== -1; e = next[e]) {
                const v = to[e];
                if (cap[e] - flow[e] > 0 && !visited[v]) {
                    visited[v] = 1;
                    q[tailQ++] = v;
                }
            }
        }
        return visited;
    }

    const useMultiPass = (M * activeTowns.length <= 150000);
    const committedTowns = new Set();
    const savedFlow = new Int32Array(flow.length);
    let currentFlow = 0;

    if (useMultiPass) {
        while (true) {
            let bestTownIdx = -1;
            let bestEff = -1;
            let bestDeltaCost = Infinity;
            let bestVal = -1;

            savedFlow.set(flow);

            for (let i = 0; i < activeTowns.length; i++) {
                if (committedTowns.has(i)) continue;

                cap[townEdgeIdx[i]] = INF;
                let limit = cutBudget - currentFlow;
                let inc = runDinic(S_star, T_star, limit);

                if (inc <= limit) {
                    let deltaCost = inc;
                    let val = activeTowns[i].value;
                    if (deltaCost === 0) {
                        if (bestDeltaCost > 0 || val > bestVal) {
                            bestTownIdx = i;
                            bestEff = Infinity;
                            bestDeltaCost = 0;
                            bestVal = val;
                        }
                    } else {
                        let eff = val / deltaCost;
                        if (bestDeltaCost > 0 && (eff > bestEff || (eff === bestEff && val > bestVal))) {
                            bestTownIdx = i;
                            bestEff = eff;
                            bestDeltaCost = deltaCost;
                            bestVal = val;
                        }
                    }
                }

                flow.set(savedFlow);
                cap[townEdgeIdx[i]] = 0;
            }

            if (bestTownIdx === -1) {
                break;
            }

            committedTowns.add(bestTownIdx);
            cap[townEdgeIdx[bestTownIdx]] = INF;
            let inc = runDinic(S_star, T_star, cutBudget - currentFlow);
            currentFlow += inc;
        }
    } else {
        const townCosts = new Int32Array(activeTowns.length);
        for (let i = 0; i < activeTowns.length; i++) {
            savedFlow.set(flow);
            cap[townEdgeIdx[i]] = INF;
            let cost = runDinic(S_star, T_star, cutBudget);
            townCosts[i] = cost;
            flow.set(savedFlow);
            cap[townEdgeIdx[i]] = 0;
        }

        const indices = [];
        for (let i = 0; i < activeTowns.length; i++) {
            if (townCosts[i] <= cutBudget) {
                indices.push(i);
            }
        }

        indices.sort((a, b) => {
            const costA = townCosts[a];
            const costB = townCosts[b];
            const valA = activeTowns[a].value;
            const valB = activeTowns[b].value;
            
            const ratioA = costA === 0 ? Infinity : valA / costA;
            const ratioB = costB === 0 ? Infinity : valB / costB;
            if (ratioA !== ratioB) return ratioB - ratioA;
            return valB - valA;
        });

        for (const i of indices) {
            savedFlow.set(flow);
            cap[townEdgeIdx[i]] = INF;
            let limit = cutBudget - currentFlow;
            let inc = runDinic(S_star, T_star, limit);
            if (inc <= limit) {
                committedTowns.add(i);
                currentFlow += inc;
            } else {
                flow.set(savedFlow);
                cap[townEdgeIdx[i]] = 0;
            }
        }
    }

    const reachableNodes = getReachableNodes(S_star);
    const firebreaks = [];
    for (let i = 0; i < M; i++) {
        const cellIdx = R[i];
        const isIgnition = ignitionSet.has(cellIdx);
        const isTown = cellToTown.has(cellIdx);
        if (!isIgnition && !isTown) {
            const inNode = 2 * i;
            const outNode = 2 * i + 1;
            if (reachableNodes[inNode] && !reachableNodes[outNode]) {
                const x = cellIdx % width;
                const y = Math.floor(cellIdx / width);
                firebreaks.push([x, y]);
            }
        }
    }

    console.log(JSON.stringify({ firebreaks }));
}

solve();
