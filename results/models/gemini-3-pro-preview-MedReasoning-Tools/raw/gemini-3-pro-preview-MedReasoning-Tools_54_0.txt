import sys
import os
import traceback

# Add parent directory to sys.path to import 54.py and autoland_sim.py
sys.path.insert(0, os.path.abspath(os.path.join(os.path.dirname(__file__), "..")))

import importlib
import importlib.util
spec = importlib.util.spec_from_file_location("test_54", os.path.abspath(os.path.join(os.path.dirname(__file__), "..", "54.py")))
test_54 = importlib.util.module_from_spec(spec)
spec.loader.exec_module(test_54)

def main():
    code_path = "answer.txt"
    if not os.path.exists(code_path):
        print(f"Error: {code_path} not found.")
        return

    with open(code_path, "r", encoding="utf-8") as f:
        code = f.read()

    result = {"python_code": code}
    scenarios_count = len(test_54.SCENARIOS)
    
    print(f"Loaded {scenarios_count} scenarios.")
    scores = []
    
    for i in range(scenarios_count):
        sc = test_54.SCENARIOS[i]
        print(f"\n--- Running Scenario {i}: {sc['name']} ---")
        try:
            score, details = test_54.gradeAnswer(result, i, "test_engine")
            print(f"Scenario {i} Score: {score}")
            print(f"Details: {details}")
            scores.append((i, score, details))
        except Exception as e:
            print(f"Scenario {i} Error: {e}")
            traceback.print_exc()
            scores.append((i, 0.0, f"Error: {e}"))

    print("\n================ SUMMARY ================")
    total_score = sum(s[1] for s in scores)
    for i, score, details in scores:
        print(f"Scenario {i:2d}: {score:.2f} | {details}")
    print(f"Total Score: {total_score:.2f} / {scenarios_count:.2f}")

if __name__ == "__main__":
    main()
