```glsl
#version 450

layout(local_size_x = 256) in;

// Each body is stored as two consecutive vec4s:
//   vec4(position.xyz, mass)
//   vec4(velocity.xyz, 0.0)
// So body i starts at index i*2 in the buffer.

layout(set = 0, binding = 0) readonly buffer InputBodies {
    vec4 inData[];   // length = N * 2
};

layout(set = 0, binding = 1) buffer OutputBodies {
    vec4 outData[];  // length = N * 2
};

layout(set = 0, binding = 2) uniform Params {
    uvec4 counts;    // counts.x = N (number of bodies)
    vec4 physics;    // physics.x = dt, physics.y = softening, physics.z = G
};

void main() {
    uint i = gl_GlobalInvocationID.x;
    uint N = counts.x;
    
    if (i >= N) {
        return;
    }

    // Read current body data
    vec4 posMass_i = inData[i * 2 + 0];
    vec4 velData_i = inData[i * 2 + 1];

    vec3 pos_i = posMass_i.xyz;
    float mass_i = posMass_i.w;
    vec3 vel_i = velData_i.xyz;

    float dt = physics.x;
    float softening = physics.y;
    float G = physics.z;

    vec3 acc = vec3(0.0);

    // Compute gravitational acceleration from all other bodies
    for (uint j = 0; j < N; ++j) {
        if (i == j) {
            continue;
        }
        vec4 posMass_j = inData[j * 2 + 0];
        vec3 pos_j = posMass_j.xyz;
        float mass_j = posMass_j.w;

        vec3 r = pos_j - pos_i;
        float dist_sq = dot(r, r) + softening * softening;
        float dist = sqrt(dist_sq);
        
        float force_mag = G * mass_j / dist_sq;
        acc += force_mag * r / dist;
    }

    // Update velocity and position
    vec3 new_vel = vel_i + acc * dt;
    vec3 new_pos = pos_i + new_vel * dt;

    // Write output data
    outData[i * 2 + 0] = vec4(new_pos, mass_i);
    outData[i * 2 + 1] = vec4(new_vel, 0.0);
}
```
