declare const require: any;
declare const process: any;
const fs = require('fs');

interface Town {
  id: any;
  x: number;
  y: number;
  value: number;
}

interface InputJSON {
  width: number;
  height: number;
  grid: string[];
  ignitions: [number, number][];
  towns: Town[];
  cutBudget: number;
}

function solve() {
  let input: string;
  try {
    input = fs.readFileSync(0, 'utf8');
  } catch (err) {
    // Fail silently if stdin is empty or fails
    console.log(JSON.stringify({ firebreaks: [] }));
    return;
  }

  let data: InputJSON;
  try {
    data = JSON.parse(input);
  } catch (err) {
    console.log(JSON.stringify({ firebreaks: [] }));
    return;
  }

  const { width, height, grid, ignitions, towns, cutBudget } = data;
  if (!width || !height || !grid || !ignitions || !towns || cutBudget === undefined) {
    console.log(JSON.stringify({ firebreaks: [] }));
    return;
  }

  const N = width * height;

  // Setup grid strings properly (handle array of strings or split single string)
  let gridStrings: string[];
  if (Array.isArray(grid)) {
    gridStrings = grid;
  } else if (typeof grid === 'string') {
    gridStrings = (grid as string).split('\n').filter(s => s.trim().length > 0);
  } else {
    console.log(JSON.stringify({ firebreaks: [] }));
    return;
  }

  // Pre-identify town cells and values
  const townValue = new Float64Array(N);
  const isTownCell = new Uint8Array(N);
  for (const town of towns) {
    const cell = town.y * width + town.x;
    if (cell >= 0 && cell < N) {
      townValue[cell] += town.value;
      isTownCell[cell] = 1;
    }
  }

  // Pre-identify ignition cells
  const isIgnitionCell = new Uint8Array(N);
  const ignitionsCells: number[] = [];
  for (const [ix, iy] of ignitions) {
    const cell = iy * width + ix;
    if (cell >= 0 && cell < N) {
      isIgnitionCell[cell] = 1;
      ignitionsCells.push(cell);
    }
  }

  // Setup capacity array
  // 0 = water/rock (~), 1 = forest (.), 2 = unrestricted (ignitions and towns)
  const capacity = new Uint8Array(N);
  for (let u = 0; u < N; u++) {
    const x = u % width;
    const y = Math.floor(u / width);
    if (y < gridStrings.length && x < gridStrings[y].length) {
      const char = gridStrings[y][x];
      if (char === '~') {
        capacity[u] = 0;
      } else {
        capacity[u] = 1;
      }
    } else {
      capacity[u] = 0;
    }
  }

  for (const cell of ignitionsCells) {
    capacity[cell] = 2;
  }
  for (let u = 0; u < N; u++) {
    if (isTownCell[u] && capacity[u] > 0) {
      capacity[u] = 2;
    }
  }

  // Build adjacency list for all cells
  const neighborCount = new Int32Array(N);
  const neighbors = new Int32Array(4 * N);
  const dirs = [[1, 0], [-1, 0], [0, 1], [0, -1]];

  for (let y = 0; y < height; y++) {
    for (let x = 0; x < width; x++) {
      const u = y * width + x;
      if (capacity[u] === 0) continue;

      let count = 0;
      for (const [dx, dy] of dirs) {
        const nx = x + dx;
        const ny = y + dy;
        if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
          const v = ny * width + nx;
          if (capacity[v] > 0) {
            neighbors[4 * u + count] = v;
            count++;
          }
        }
      }
      neighborCount[u] = count;
    }
  }

  // BFS 1: Find all cells reachable from ignitions in original grid (no cuts)
  const reachableFromIgnition = new Uint8Array(N);
  {
    const q = new Int32Array(N);
    let head = 0;
    let tail = 0;
    for (const s of ignitionsCells) {
      q[tail++] = s;
      reachableFromIgnition[s] = 1;
    }
    while (head < tail) {
      const u = q[head++];
      const count = neighborCount[u];
      const offset = 4 * u;
      for (let i = 0; i < count; i++) {
        const v = neighbors[offset + i];
        if (!reachableFromIgnition[v]) {
          reachableFromIgnition[v] = 1;
          q[tail++] = v;
        }
      }
    }
  }

  // Find otherwise-burned towns (towns reachable from ignitions)
  const otherwiseBurnedTownCells: number[] = [];
  let totalOtherwiseBurnedValue = 0;
  for (let u = 0; u < N; u++) {
    if (isTownCell[u] && reachableFromIgnition[u]) {
      otherwiseBurnedTownCells.push(u);
      totalOtherwiseBurnedValue += townValue[u];
    }
  }

  if (otherwiseBurnedTownCells.length === 0 || totalOtherwiseBurnedValue === 0) {
    console.log(JSON.stringify({ firebreaks: [] }));
    return;
  }

  // BFS 2: Find all cells that can reach any otherwise burned town
  const canReachTown = new Uint8Array(N);
  {
    const q = new Int32Array(N);
    let head = 0;
    let tail = 0;
    for (const t of otherwiseBurnedTownCells) {
      q[tail++] = t;
      canReachTown[t] = 1;
    }
    while (head < tail) {
      const u = q[head++];
      const count = neighborCount[u];
      const offset = 4 * u;
      for (let i = 0; i < count; i++) {
        const v = neighbors[offset + i];
        if (!canReachTown[v]) {
          canReachTown[v] = 1;
          q[tail++] = v;
        }
      }
    }
  }

  // Active region consists of cells that are in both reachable sets
  const inActiveRegion = new Uint8Array(N);
  for (let u = 0; u < N; u++) {
    if (reachableFromIgnition[u] && canReachTown[u]) {
      inActiveRegion[u] = 1;
    }
  }

  // Allocate global flow arrays (reused across all passes)
  const flow = new Uint8Array(N);
  const flowInFrom = new Int32Array(N);
  const flowOutTo = new Int32Array(N);
  flowInFrom.fill(-1);
  flowOutTo.fill(-1);

  // Allocate global BFS arrays (reused)
  const queue = new Int32Array(2 * N);
  const visited = new Int32Array(2 * N);
  const parent = new Int32Array(2 * N);
  let visitedToken = 0;

  const isTargetTown = new Uint8Array(N);

  // Helper: backward BFS from target towns to ignitions
  function findAugmentingPath(): number[] | null {
    let foundSourceState = -1;
    let head = 0;
    let tail = 0;

    visitedToken++;

    for (let u = 0; u < N; u++) {
      if (isTargetTown[u] && inActiveRegion[u]) {
        const state = 2 * u; // (u, IN)
        queue[tail++] = state;
        visited[state] = visitedToken;
        parent[state] = -1;
      }
    }

    while (head < tail) {
      const curr = queue[head++];
      const type = curr & 1; // 0 = IN, 1 = OUT
      const u = curr >> 1;

      if (type === 0) {
        // (u, IN)
        // 1. If flow[u] === 1: transition to (u, OUT)
        if (flow[u] === 1) {
          const nextState = 2 * u + 1;
          if (visited[nextState] !== visitedToken) {
            visited[nextState] = visitedToken;
            parent[nextState] = curr;
            queue[tail++] = nextState;
          }
        }
        // 2. Transition to (p, OUT) for all active neighbors p of u
        const count = neighborCount[u];
        const offset = 4 * u;
        for (let i = 0; i < count; i++) {
          const p = neighbors[offset + i];
          if (inActiveRegion[p] === 0) continue;
          const nextState = 2 * p + 1; // (p, OUT)
          if (visited[nextState] !== visitedToken) {
            visited[nextState] = visitedToken;
            parent[nextState] = curr;
            queue[tail++] = nextState;
          }
        }
      } else {
        // (u, OUT)
        // Check if u is an ignition cell
        if (isIgnitionCell[u]) {
          foundSourceState = curr;
          break;
        }

        // Transitions from (u, OUT):
        // 1. If flow[u] === 0: transition to (u, IN)
        if (flow[u] === 0) {
          const nextState = 2 * u; // (u, IN)
          if (visited[nextState] !== visitedToken) {
            visited[nextState] = visitedToken;
            parent[nextState] = curr;
            queue[tail++] = nextState;
          }
        } else {
          // 2. If flow[u] === 1: transition to (q, IN) where q = flowOutTo[u]
          const q = flowOutTo[u];
          const nextState = 2 * q; // (q, IN)
          if (visited[nextState] !== visitedToken) {
            visited[nextState] = visitedToken;
            parent[nextState] = curr;
            queue[tail++] = nextState;
          }
        }
      }
    }

    if (foundSourceState === -1) return null;

    // Reconstruct path
    const path: number[] = [];
    let curr = foundSourceState;
    while (curr !== -1) {
      path.push(curr);
      curr = parent[curr];
    }
    return path;
  }

  function augmentPathDirect(path: number[]) {
    for (let i = 0; i < path.length - 1; i++) {
      const sCurr = path[i];
      const sNext = path[i+1];
      const u = sCurr >> 1;
      const v = sNext >> 1;
      const typeCurr = sCurr & 1;
      const typeNext = sNext & 1;

      if (u === v) {
        if (typeCurr === 0 && typeNext === 1) {
          if (capacity[u] === 1) flow[u] = 1;
        } else if (typeCurr === 1 && typeNext === 0) {
          if (capacity[u] === 1) flow[u] = 0;
        }
      } else {
        if (typeCurr === 1 && typeNext === 0) {
          if (capacity[u] === 1) flowOutTo[u] = v;
          if (capacity[v] === 1) flowInFrom[v] = u;
        }
      }
    }
  }

  // Rollback mechanism
  const modificationLog: { cell: number, field: 'flow' | 'in' | 'out', oldVal: number }[] = [];

  function setFlow(cell: number, val: number) {
    modificationLog.push({ cell, field: 'flow', oldVal: flow[cell] });
    flow[cell] = val;
  }
  function setFlowIn(cell: number, val: number) {
    modificationLog.push({ cell, field: 'in', oldVal: flowInFrom[cell] });
    flowInFrom[cell] = val;
  }
  function setFlowOut(cell: number, val: number) {
    modificationLog.push({ cell, field: 'out', oldVal: flowOutTo[cell] });
    flowOutTo[cell] = val;
  }

  function rollback() {
    while (modificationLog.length > 0) {
      const mod = modificationLog.pop()!;
      if (mod.field === 'flow') {
        flow[mod.cell] = mod.oldVal;
      } else if (mod.field === 'in') {
        flowInFrom[mod.cell] = mod.oldVal;
      } else if (mod.field === 'out') {
        flowOutTo[mod.cell] = mod.oldVal;
      }
    }
  }

  function augmentPathLogged(path: number[]) {
    for (let i = 0; i < path.length - 1; i++) {
      const sCurr = path[i];
      const sNext = path[i+1];
      const u = sCurr >> 1;
      const v = sNext >> 1;
      const typeCurr = sCurr & 1;
      const typeNext = sNext & 1;

      if (u === v) {
        if (typeCurr === 0 && typeNext === 1) {
          if (capacity[u] === 1) setFlow(u, 1);
        } else if (typeCurr === 1 && typeNext === 0) {
          if (capacity[u] === 1) setFlow(u, 0);
        }
      } else {
        if (typeCurr === 1 && typeNext === 0) {
          if (capacity[u] === 1) setFlowOut(u, v);
          if (capacity[v] === 1) setFlowIn(v, u);
        }
      }
    }
  }

  function getMinCutSize(t: number): number {
    flow.fill(0);
    flowInFrom.fill(-1);
    flowOutTo.fill(-1);
    isTargetTown.fill(0);
    isTargetTown[t] = 1;

    let flowValue = 0;
    while (true) {
      const path = findAugmentingPath();
      if (!path) break;

      let hasCap1 = false;
      for (const state of path) {
        const u = state >> 1;
        if (capacity[u] === 1) {
          hasCap1 = true;
          break;
        }
      }
      if (!hasCap1) {
        return Infinity;
      }

      augmentPathDirect(path);
      flowValue++;
    }
    return flowValue;
  }

  function findCuts(): number[] {
    const reachable = new Uint8Array(2 * N);
    const q = new Int32Array(2 * N);
    let head = 0;
    let tail = 0;

    for (const s of ignitionsCells) {
      if (inActiveRegion[s]) {
        const state = 2 * s + 1; // (s, OUT)
        q[tail++] = state;
        reachable[state] = 1;
      }
    }

    while (head < tail) {
      const curr = q[head++];
      const type = curr & 1;
      const u = curr >> 1;

      if (type === 0) {
        if (flow[u] === 0) {
          const nextState = 2 * u + 1;
          if (!reachable[nextState]) {
            reachable[nextState] = 1;
            q[tail++] = nextState;
          }
        } else {
          const p = flowInFrom[u];
          const nextState = 2 * p + 1;
          if (!reachable[nextState]) {
            reachable[nextState] = 1;
            q[tail++] = nextState;
          }
        }
      } else {
        const count = neighborCount[u];
        const offset = 4 * u;
        for (let i = 0; i < count; i++) {
          const v = neighbors[offset + i];
          if (inActiveRegion[v] === 0) continue;
          const nextState = 2 * v;
          if (!reachable[nextState]) {
            reachable[nextState] = 1;
            q[tail++] = nextState;
          }
        }
        if (flow[u] === 1) {
          const nextState = 2 * u;
          if (!reachable[nextState]) {
            reachable[nextState] = 1;
            q[tail++] = nextState;
          }
        }
      }
    }

    const cuts: number[] = [];
    for (let u = 0; u < N; u++) {
      if (capacity[u] === 1 && inActiveRegion[u]) {
        if (reachable[2 * u] && !reachable[2 * u + 1]) {
          cuts.push(u);
        }
      }
    }
    return cuts;
  }

  interface TownInfo {
    cell: number;
    value: number;
    minCut: number;
  }

  const townInfos: TownInfo[] = [];
  for (const tCell of otherwiseBurnedTownCells) {
    const c_t = getMinCutSize(tCell);
    if (c_t < Infinity) {
      townInfos.push({
        cell: tCell,
        value: townValue[tCell],
        minCut: c_t
      });
    }
  }

  function runGreedyPass(orderedTowns: TownInfo[]): { savedValue: number, cuts: number[] } {
    flow.fill(0);
    flowInFrom.fill(-1);
    flowOutTo.fill(-1);
    isTargetTown.fill(0);

    let flowValue = 0;
    let savedValue = 0;

    for (const town of orderedTowns) {
      const tCell = town.cell;
      isTargetTown[tCell] = 1;

      let tempFlowValue = flowValue;
      let success = true;
      modificationLog.length = 0;

      while (true) {
        const path = findAugmentingPath();
        if (!path) break;
        augmentPathLogged(path);
        tempFlowValue++;
        if (tempFlowValue > cutBudget) {
          success = false;
          break;
        }
      }

      if (success) {
        flowValue = tempFlowValue;
        modificationLog.length = 0;
        savedValue += town.value;
      } else {
        rollback();
        isTargetTown[tCell] = 0;
      }
    }

    const cuts = findCuts();
    return { savedValue, cuts };
  }

  // Try different orderings and select the best one
  let bestSavedValue = -1;
  let bestCuts: number[] = [];

  // Ordering 1: value / minCut descending
  const order1 = [...townInfos].sort((a, b) => {
    const scoreA = a.value / a.minCut;
    const scoreB = b.value / b.minCut;
    if (Math.abs(scoreA - scoreB) > 1e-9) return scoreB - scoreA;
    return b.value - a.value;
  });

  // Ordering 2: value descending
  const order2 = [...townInfos].sort((a, b) => b.value - a.value);

  // Ordering 3: value / (minCut * minCut) descending
  const order3 = [...townInfos].sort((a, b) => {
    const scoreA = a.value / (a.minCut * a.minCut);
    const scoreB = b.value / (b.minCut * b.minCut);
    if (Math.abs(scoreA - scoreB) > 1e-9) return scoreB - scoreA;
    return b.value - a.value;
  });

  const orders = [order1, order2, order3];

  // Also try 20 randomized orderings to escape local optima
  for (let i = 0; i < 20; i++) {
    const orderRand = [...townInfos].sort((a, b) => {
      // Perturb the score slightly
      const scoreA = (a.value * (0.8 + 0.4 * Math.random())) / a.minCut;
      const scoreB = (b.value * (0.8 + 0.4 * Math.random())) / b.minCut;
      return scoreB - scoreA;
    });
    orders.push(orderRand);
  }

  for (const order of orders) {
    const res = runGreedyPass(order);
    if (res.savedValue > bestSavedValue || (res.savedValue === bestSavedValue && res.cuts.length < bestCuts.length)) {
      bestSavedValue = res.savedValue;
      bestCuts = res.cuts;
    }
  }

  const firebreaks = bestCuts.map(cell => [cell % width, Math.floor(cell / width)]);
  console.log(JSON.stringify({ firebreaks }));
}

solve();
