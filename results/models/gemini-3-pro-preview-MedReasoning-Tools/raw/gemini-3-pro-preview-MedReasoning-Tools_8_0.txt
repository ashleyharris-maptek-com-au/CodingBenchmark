use std::collections::VecDeque;
use std::io::{self, BufRead, Write};

fn write_usize<W: Write>(writer: &mut W, mut n: usize) -> io::Result<()> {
    if n == 0 {
        return writer.write_all(b"0");
    }
    let mut buf = [0u8; 20];
    let mut i = 20;
    while n > 0 {
        i -= 1;
        buf[i] = b'0' + (n % 10) as u8;
        n /= 10;
    }
    writer.write_all(&buf[i..])
}

fn write_coord<W: Write>(writer: &mut W, x: usize, y: usize) -> io::Result<()> {
    write_usize(writer, x)?;
    writer.write_all(b" ")?;
    write_usize(writer, y)?;
    writer.write_all(b"\n")
}

fn main() {
    let stdin = io::stdin();
    let mut reader = io::BufReader::with_capacity(8 * 1024 * 1024, stdin.lock());

    let mut line = String::new();
    if reader.read_line(&mut line).is_err() {
        return;
    }
    let mut parts = line.split_whitespace();
    let h_str = match parts.next() {
        Some(s) => s,
        None => return,
    };
    let w_str = match parts.next() {
        Some(s) => s,
        None => return,
    };
    let H: usize = match h_str.parse() {
        Ok(v) => v,
        Err(_) => return,
    };
    let W: usize = match w_str.parse() {
        Ok(v) => v,
        Err(_) => return,
    };

    if H == 0 || W == 0 {
        return;
    }

    let mut grid = vec![0u8; H * W];
    let mut start_idx = None;
    let mut goal_idx = None;

    let mut buf = Vec::with_capacity(W + 2);
    for r in 0..H {
        buf.clear();
        if reader.read_until(b'\n', &mut buf).is_err() {
            return;
        }
        while buf.last() == Some(&b'\n') || buf.last() == Some(&b'\r') {
            buf.pop();
        }
        let start_offset = r * W;
        let len = buf.len().min(W);
        for c in 0..len {
            let val = buf[c];
            grid[start_offset + c] = val;
            if val == b'A' {
                start_idx = Some(start_offset + c);
            } else if val == b'B' {
                goal_idx = Some(start_offset + c);
            }
        }
        for c in len..W {
            grid[start_offset + c] = b' ';
        }
    }

    let start_idx = match start_idx {
        Some(idx) => idx,
        None => return,
    };
    let goal_idx = match goal_idx {
        Some(idx) => idx,
        None => return,
    };

    if start_idx == goal_idx {
        let mut writer = io::BufWriter::new(io::stdout().lock());
        writeln!(writer, "1").unwrap();
        let x = start_idx % W;
        let y = start_idx / W;
        let _ = write_coord(&mut writer, x, y);
        return;
    }

    let mut q_f = VecDeque::new();
    let mut q_b = VecDeque::new();

    let start_x = start_idx % W;
    let start_y = start_idx / W;
    let goal_x = goal_idx % W;
    let goal_y = goal_idx / W;

    q_f.push_back(((start_y as u64) << 32) | (start_x as u64));
    q_b.push_back(((goal_y as u64) << 32) | (goal_x as u64));

    let mut intersection = None;

    while !q_f.is_empty() && !q_b.is_empty() {
        if q_f.len() <= q_b.len() {
            let curr = q_f.pop_front().unwrap();
            let cx = (curr & 0xFFFFFFFF) as usize;
            let cy = (curr >> 32) as usize;
            let curr_idx = cy * W + cx;

            // UP
            if cy > 0 {
                let next_idx = curr_idx - W;
                let val = grid[next_idx];
                if val != b'#' {
                    if val == b'B' || val == b'u' || val == b'd' || val == b'l' || val == b'r' {
                        intersection = Some((curr_idx, next_idx));
                        break;
                    } else if val == b' ' {
                        grid[next_idx] = b'D';
                        q_f.push_back((((cy - 1) as u64) << 32) | (cx as u64));
                    }
                }
            }
            // DOWN
            if cy + 1 < H {
                let next_idx = curr_idx + W;
                let val = grid[next_idx];
                if val != b'#' {
                    if val == b'B' || val == b'u' || val == b'd' || val == b'l' || val == b'r' {
                        intersection = Some((curr_idx, next_idx));
                        break;
                    } else if val == b' ' {
                        grid[next_idx] = b'U';
                        q_f.push_back((((cy + 1) as u64) << 32) | (cx as u64));
                    }
                }
            }
            // LEFT
            if cx > 0 {
                let next_idx = curr_idx - 1;
                let val = grid[next_idx];
                if val != b'#' {
                    if val == b'B' || val == b'u' || val == b'd' || val == b'l' || val == b'r' {
                        intersection = Some((curr_idx, next_idx));
                        break;
                    } else if val == b' ' {
                        grid[next_idx] = b'R';
                        q_f.push_back(((cy as u64) << 32) | ((cx - 1) as u64));
                    }
                }
            }
            // RIGHT
            if cx + 1 < W {
                let next_idx = curr_idx + 1;
                let val = grid[next_idx];
                if val != b'#' {
                    if val == b'B' || val == b'u' || val == b'd' || val == b'l' || val == b'r' {
                        intersection = Some((curr_idx, next_idx));
                        break;
                    } else if val == b' ' {
                        grid[next_idx] = b'L';
                        q_f.push_back(((cy as u64) << 32) | ((cx + 1) as u64));
                    }
                }
            }
        } else {
            let curr = q_b.pop_front().unwrap();
            let cx = (curr & 0xFFFFFFFF) as usize;
            let cy = (curr >> 32) as usize;
            let curr_idx = cy * W + cx;

            // UP
            if cy > 0 {
                let next_idx = curr_idx - W;
                let val = grid[next_idx];
                if val != b'#' {
                    if val == b'A' || val == b'U' || val == b'D' || val == b'L' || val == b'R' {
                        intersection = Some((next_idx, curr_idx));
                        break;
                    } else if val == b' ' {
                        grid[next_idx] = b'd';
                        q_b.push_back((((cy - 1) as u64) << 32) | (cx as u64));
                    }
                }
            }
            // DOWN
            if cy + 1 < H {
                let next_idx = curr_idx + W;
                let val = grid[next_idx];
                if val != b'#' {
                    if val == b'A' || val == b'U' || val == b'D' || val == b'L' || val == b'R' {
                        intersection = Some((next_idx, curr_idx));
                        break;
                    } else if val == b' ' {
                        grid[next_idx] = b'u';
                        q_b.push_back((((cy + 1) as u64) << 32) | (cx as u64));
                    }
                }
            }
            // LEFT
            if cx > 0 {
                let next_idx = curr_idx - 1;
                let val = grid[next_idx];
                if val != b'#' {
                    if val == b'A' || val == b'U' || val == b'D' || val == b'L' || val == b'R' {
                        intersection = Some((next_idx, curr_idx));
                        break;
                    } else if val == b' ' {
                        grid[next_idx] = b'r';
                        q_b.push_back(((cy as u64) << 32) | ((cx - 1) as u64));
                    }
                }
            }
            // RIGHT
            if cx + 1 < W {
                let next_idx = curr_idx + 1;
                let val = grid[next_idx];
                if val != b'#' {
                    if val == b'A' || val == b'U' || val == b'D' || val == b'L' || val == b'R' {
                        intersection = Some((next_idx, curr_idx));
                        break;
                    } else if val == b' ' {
                        grid[next_idx] = b'l';
                        q_b.push_back(((cy as u64) << 32) | ((cx + 1) as u64));
                    }
                }
            }
        }
    }

    let (f_idx, b_idx) = match intersection {
        Some(pair) => pair,
        None => {
            // No path exists, output nothing or 0
            return;
        }
    };

    // Reconstruct Forward path: from f_idx back to start_idx
    let mut path_f = Vec::new();
    let mut p = f_idx;
    while p != start_idx {
        path_f.push(p);
        let px = p % W;
        let py = p / W;
        p = match grid[p] {
            b'U' => (py - 1) * W + px,
            b'D' => (py + 1) * W + px,
            b'L' => py * W + (px - 1),
            b'R' => py * W + (px + 1),
            _ => unreachable!(),
        };
    }
    path_f.push(start_idx);
    path_f.reverse();

    // Reconstruct Backward path: from b_idx back to goal_idx
    let mut path_b = Vec::new();
    let mut p = b_idx;
    while p != goal_idx {
        path_b.push(p);
        let px = p % W;
        let py = p / W;
        p = match grid[p] {
            b'u' => (py - 1) * W + px,
            b'd' => (py + 1) * W + px,
            b'l' => py * W + (px - 1),
            b'r' => py * W + (px + 1),
            _ => unreachable!(),
        };
    }
    path_b.push(goal_idx);

    let total_len = path_f.len() + path_b.len();
    let mut writer = io::BufWriter::new(io::stdout().lock());
    let _ = write_usize(&mut writer, total_len);
    let _ = writer.write_all(b"\n");

    for &idx in &path_f {
        let x = idx % W;
        let y = idx / W;
        let _ = write_coord(&mut writer, x, y);
    }
    for &idx in &path_b {
        let x = idx % W;
        let y = idx / W;
        let _ = write_coord(&mut writer, x, y);
    }
}
