#include <iostream>
#include <vector>
#include <numeric>
#include <algorithm>
#include <chrono>
#include <cmath>

// Wikipedia Hilbert curve implementation
void rot(int n, int *x, int *y, int rx, int ry) {
    if (ry == 0) {
        if (rx == 1) {
            *x = n - 1 - *x;
            *y = n - 1 - *y;
        }
        int t = *x;
        *x = *y;
        *y = t;
    }
}

uint64_t xy2d(int n, int x, int y) {
    int rx, ry, s;
    uint64_t d = 0;
    for (s = n / 2; s > 0; s /= 2) {
        rx = (x & s) > 0;
        ry = (y & s) > 0;
        d += (uint64_t)s * s * ((3 * rx) ^ ry);
        rot(s, &x, &y, rx, ry);
    }
    return d;
}

int main() {
    int n = 32; // grid size 32x32
    std::cout << "Testing Hilbert coordinates for 4x4 grid:" << std::endl;
    for (int y = 0; y < 4; ++y) {
        for (int x = 0; x < 4; ++x) {
            std::cout << "x=" << x << ", y=" << y << " -> d=" << xy2d(4, x, y) << std::endl;
        }
    }
    return 0;
}
