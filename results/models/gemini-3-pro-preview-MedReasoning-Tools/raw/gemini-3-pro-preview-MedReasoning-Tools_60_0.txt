const fs = require('fs');

// Simple, high-performance binary Max-Heap for Kahn's algorithm
class MaxHeap {
  constructor(compare) {
    this.data = [];
    this.compare = compare;
  }
  push(val) {
    this.data.push(val);
    this.up(this.data.length - 1);
  }
  pop() {
    if (this.data.length === 0) return null;
    const top = this.data[0];
    const bottom = this.data.pop();
    if (this.data.length > 0) {
      this.data[0] = bottom;
      this.down(0);
    }
    return top;
  }
  up(i) {
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (this.compare(this.data[i], this.data[p]) > 0) {
        const tmp = this.data[i];
        this.data[i] = this.data[p];
        this.data[p] = tmp;
        i = p;
      } else {
        break;
      }
    }
  }
  down(i) {
    const len = this.data.length;
    while ((i << 1) + 1 < len) {
      let best = i;
      const left = (i << 1) + 1;
      const right = left + 1;
      if (this.compare(this.data[left], this.data[best]) > 0) {
        best = left;
      }
      if (right < len && this.compare(this.data[right], this.data[best]) > 0) {
        best = right;
      }
      if (best !== i) {
        const tmp = this.data[i];
        this.data[i] = this.data[best];
        this.data[best] = tmp;
        i = best;
      } else {
        break;
      }
    }
  }
  size() {
    return this.data.length;
  }
}

// Generates a topological order using Kahn's algorithm biased by a priority function
function generateOrder(tasks, children, initialInDegrees, compareTasks) {
  const n = tasks.length;
  const inDegrees = new Int32Array(initialInDegrees);
  const heap = new MaxHeap(compareTasks);
  for (let i = 0; i < n; i++) {
    if (inDegrees[i] === 0) {
      heap.push(i);
    }
  }

  const order = new Int32Array(n);
  let idx = 0;
  while (idx < n) {
    if (heap.size() > 0) {
      const u = heap.pop();
      order[idx++] = u;
      const ch = children[u];
      for (let i = 0; i < ch.length; i++) {
        const v = ch[i];
        inDegrees[v]--;
        if (inDegrees[v] === 0) {
          heap.push(v);
        }
      }
    } else {
      // Cycle breaker fallback to guarantee topological order is complete
      let bestTask = -1;
      let minInDegree = Infinity;
      for (let i = 0; i < n; i++) {
        if (inDegrees[i] > 0) {
          if (inDegrees[i] < minInDegree) {
            minInDegree = inDegrees[i];
            bestTask = i;
          }
        }
      }
      if (bestTask !== -1) {
        inDegrees[bestTask] = 0;
        heap.push(bestTask);
      } else {
        break;
      }
    }
  }
  return order;
}

// Pre-allocated workspace arrays to prevent GC overhead during search
let completionTimes;
let machineReadyTimes;
let sortedMachines;

function initWorkspace(n, machineCount) {
  completionTimes = new Int32Array(n);
  machineReadyTimes = new Int32Array(machineCount);
  sortedMachines = new Int32Array(machineCount);
}

// List scheduling with binary search for workstation assignment and zero object allocations
function evaluateScheduleZeroAlloc(orderIndices, tasks, machineCount, startTimes, machines) {
  const n = tasks.length;
  
  completionTimes.fill(0);
  machineReadyTimes.fill(0);
  for (let i = 0; i < machineCount; i++) {
    sortedMachines[i] = i;
  }

  for (let i = 0; i < n; i++) {
    const taskIdx = orderIndices[i];
    const task = tasks[taskIdx];
    
    let earliestStart = 0;
    const deps = task.deps;
    for (let j = 0; j < deps.length; j++) {
      const depComp = completionTimes[deps[j]];
      if (depComp > earliestStart) {
        earliestStart = depComp;
      }
    }

    let k = 0;
    if (machineReadyTimes[sortedMachines[0]] > earliestStart) {
      k = 0;
    } else {
      let low = 0, high = machineCount - 1;
      while (low <= high) {
        const mid = (low + high) >> 1;
        if (machineReadyTimes[sortedMachines[mid]] <= earliestStart) {
          k = mid;
          low = mid + 1;
        } else {
          high = mid - 1;
        }
      }
    }

    const bestMachine = sortedMachines[k];
    const ready = machineReadyTimes[bestMachine];
    const start = ready > earliestStart ? ready : earliestStart;
    const comp = start + task.duration;

    completionTimes[taskIdx] = comp;
    machineReadyTimes[bestMachine] = comp;
    startTimes[taskIdx] = start;
    machines[taskIdx] = bestMachine;

    // Re-insert machine into sorted order
    let idx = k;
    while (idx < machineCount - 1 && machineReadyTimes[sortedMachines[idx + 1]] < comp) {
      sortedMachines[idx] = sortedMachines[idx + 1];
      idx++;
    }
    sortedMachines[idx] = bestMachine;
  }

  let weightedLateness = 0;
  let makespan = 0;
  for (let i = 0; i < n; i++) {
    const task = tasks[i];
    const comp = completionTimes[i];
    if (comp > makespan) {
      makespan = comp;
    }
    const lateness = comp - task.deadline;
    if (lateness > 0) {
      weightedLateness += lateness * task.weight;
    }
  }

  return weightedLateness + makespan;
}

function main() {
  const startTime = Date.now();
  const TIME_LIMIT = 1200; // 1.2s budget for optimization

  const input = fs.readFileSync(0, 'utf8');
  if (!input || input.trim() === '') {
    return;
  }
  const data = JSON.parse(input);

  const machineCount = data.machineCount;
  const n = data.tasks.length;

  if (n === 0) {
    console.log(JSON.stringify({ schedule: [] }));
    return;
  }

  const idToIndex = new Map();
  for (let i = 0; i < n; i++) {
    idToIndex.set(data.tasks[i].id, i);
  }

  const tasks = data.tasks.map((t, idx) => ({
    index: idx,
    id: t.id,
    duration: t.duration,
    deps: (t.deps || []).map(depId => idToIndex.get(depId)),
    deadline: t.deadline,
    weight: t.weight
  }));

  const initialInDegrees = new Int32Array(n);
  const children = Array.from({ length: n }, () => []);
  for (let i = 0; i < n; i++) {
    const deps = tasks[i].deps;
    initialInDegrees[i] = deps.length;
    for (let j = 0; j < deps.length; j++) {
      children[deps[j]].push(i);
    }
  }

  // Pre-calculate clean topological sort for metric propagation
  const q = [];
  for (let i = 0; i < n; i++) {
    if (initialInDegrees[i] === 0) {
      q.push(i);
    }
  }
  const topo = [];
  let head = 0;
  while (head < q.length) {
    const u = q[head++];
    topo.push(u);
    const ch = children[u];
    for (let i = 0; i < ch.length; i++) {
      const v = ch[i];
      initialInDegrees[v]--;
      if (initialInDegrees[v] === 0) {
        q.push(v);
      }
    }
  }

  // Restore initialInDegrees
  for (let i = 0; i < n; i++) {
    initialInDegrees[i] = tasks[i].deps.length;
  }

  // Compute DP bounds over the DAG
  const depth = new Array(n).fill(1);
  const downstreamDuration = new Array(n);
  const downstreamWeight = new Array(n);
  for (let i = 0; i < n; i++) {
    downstreamDuration[i] = tasks[i].duration;
    downstreamWeight[i] = tasks[i].weight;
  }

  for (let i = topo.length - 1; i >= 0; i--) {
    const u = topo[i];
    const ch = children[u];
    let maxD = 0;
    let maxDD = 0;
    let sumDW = 0;
    for (let j = 0; j < ch.length; j++) {
      const v = ch[j];
      if (depth[v] > maxD) maxD = depth[v];
      if (downstreamDuration[v] > maxDD) maxDD = downstreamDuration[v];
      sumDW += downstreamWeight[v];
    }
    depth[u] = 1 + maxD;
    downstreamDuration[u] = tasks[u].duration + maxDD;
    downstreamWeight[u] = tasks[u].weight + sumDW;
  }

  // Normalization variables
  let maxDD = 1, maxW = 1, maxDL = 1, maxDW = 1, maxWSPT = 1;
  const wspt = new Array(n);
  for (let i = 0; i < n; i++) {
    wspt[i] = tasks[i].weight / tasks[i].duration;
    if (downstreamDuration[i] > maxDD) maxDD = downstreamDuration[i];
    if (tasks[i].weight > maxW) maxW = tasks[i].weight;
    if (tasks[i].deadline > maxDL) maxDL = tasks[i].deadline;
    if (downstreamWeight[i] > maxDW) maxDW = downstreamWeight[i];
    if (wspt[i] > maxWSPT) maxWSPT = wspt[i];
  }

  let bestScore = Infinity;
  const bestStartTimes = new Int32Array(n);
  const bestMachines = new Int32Array(n);

  const currentStartTimes = new Int32Array(n);
  const currentMachines = new Int32Array(n);

  initWorkspace(n, machineCount);

  // Deterministic standard dispatching policies
  const strategies = [
    (a, b) => {
      const diff = downstreamDuration[a] - downstreamDuration[b];
      if (diff !== 0) return diff;
      return tasks[b].deadline - tasks[a].deadline;
    },
    (a, b) => {
      const diff = tasks[b].deadline - tasks[a].deadline;
      if (diff !== 0) return diff;
      return downstreamDuration[a] - downstreamDuration[b];
    },
    (a, b) => {
      const diff = tasks[a].weight - tasks[b].weight;
      if (diff !== 0) return diff;
      return downstreamDuration[a] - downstreamDuration[b];
    },
    (a, b) => {
      const diff = wspt[a] - wspt[b];
      if (Math.abs(diff) > 1e-9) return diff;
      return downstreamDuration[a] - downstreamDuration[b];
    },
    (a, b) => {
      const diff = downstreamWeight[a] - downstreamWeight[b];
      if (diff !== 0) return diff;
      return tasks[b].deadline - tasks[a].deadline;
    }
  ];

  for (const comp of strategies) {
    const order = generateOrder(tasks, children, initialInDegrees, comp);
    const score = evaluateScheduleZeroAlloc(order, tasks, machineCount, currentStartTimes, currentMachines);
    if (score < bestScore) {
      bestScore = score;
      bestStartTimes.set(currentStartTimes);
      bestMachines.set(currentMachines);
    }
  }

  // Randomized parameter search within time budget
  const globalSearchLimit = startTime + TIME_LIMIT;
  while (Date.now() < globalSearchLimit) {
    const c1 = Math.random() * 2 - 1;
    const c2 = Math.random() * 2 - 1;
    const c3 = Math.random() * 2 - 1;
    const c4 = Math.random() * 2 - 1;
    const c5 = Math.random() * 2 - 1;

    const compareRand = (a, b) => {
      const scoreA = c1 * (downstreamDuration[a] / maxDD) 
                   + c2 * (tasks[a].weight / maxW)
                   - c3 * (tasks[a].deadline / maxDL)
                   + c4 * (downstreamWeight[a] / maxDW)
                   + c5 * (wspt[a] / maxWSPT);
      const scoreB = c1 * (downstreamDuration[b] / maxDD)
                   + c2 * (tasks[b].weight / maxW)
                   - c3 * (tasks[b].deadline / maxDL)
                   + c4 * (downstreamWeight[b] / maxDW)
                   + c5 * (wspt[b] / maxWSPT);
      if (scoreA !== scoreB) return scoreA - scoreB;
      return b - a;
    };

    const order = generateOrder(tasks, children, initialInDegrees, compareRand);
    const score = evaluateScheduleZeroAlloc(order, tasks, machineCount, currentStartTimes, currentMachines);
    if (score < bestScore) {
      bestScore = score;
      bestStartTimes.set(currentStartTimes);
      bestMachines.set(currentMachines);
    }
  }

  // Output formatting
  const finalSchedule = new Array(n);
  for (let i = 0; i < n; i++) {
    finalSchedule[i] = {
      id: tasks[i].id,
      start: bestStartTimes[i],
      machine: bestMachines[i]
    };
  }

  console.log(JSON.stringify({ schedule: finalSchedule }));
}

main();
