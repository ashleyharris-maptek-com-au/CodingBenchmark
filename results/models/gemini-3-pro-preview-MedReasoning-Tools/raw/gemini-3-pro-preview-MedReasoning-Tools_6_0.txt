using System;
using System.IO;
using System.Linq;
using System.Collections.Generic;
using System.Diagnostics;

class Program
{
    static void Main(string[] args)
    {
        using (var reader = new StreamReader(Console.OpenStandardInput(8192)))
        {
            string firstLine = reader.ReadLine();
            if (string.IsNullOrWhiteSpace(firstLine)) return;

            string[] parts = firstLine.Trim().Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
            int N = int.Parse(parts[0]);
            double mu = double.Parse(parts[1]);

            string secondLine = reader.ReadLine();
            double[] startOrbit = ParseOrbit(secondLine);

            double[][] stationOrbits = new double[N][];
            for (int i = 0; i < N; i++)
            {
                stationOrbits[i] = ParseOrbit(reader.ReadLine());
            }

            int[] visitOrder = SolveOTSP(N, mu, startOrbit, stationOrbits);

            Console.WriteLine(string.Join(" ", visitOrder));
        }
    }

    static double[] ParseOrbit(string line)
    {
        if (string.IsNullOrWhiteSpace(line)) return new double[6];
        string[] parts = line.Trim().Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        double[] orbit = new double[6];
        for (int i = 0; i < 6; i++)
        {
            orbit[i] = double.Parse(parts[i]);
        }
        return orbit;
    }

    static int[] SolveOTSP(int N, double mu, double[] startOrbit, double[][] stationOrbits)
    {
        double[] distStart = new double[N];
        for (int i = 0; i < N; i++)
        {
            distStart[i] = EstimateTransferDeltaV(startOrbit, stationOrbits[i], mu);
            if (double.IsNaN(distStart[i])) Console.Error.WriteLine($"distStart[{i}] is NaN");
        }

        double[][] dist = new double[N][];
        for (int i = 0; i < N; i++)
        {
            dist[i] = new double[N];
            for (int j = 0; j < N; j++)
            {
                if (i == j) dist[i][j] = 0;
                else
                {
                    dist[i][j] = EstimateTransferDeltaV(stationOrbits[i], stationOrbits[j], mu);
                    if (double.IsNaN(dist[i][j])) Console.Error.WriteLine($"dist[{i}][{j}] is NaN");
                }
            }
        }

        if (N <= 20)
        {
            return SolveDP(N, distStart, dist);
        }
        else
        {
            return SolveSA(N, distStart, dist);
        }
    }

    static int[] SolveDP(int N, double[] distStart, double[][] dist)
    {
        int numStates = (1 << N);
        double[] dp = new double[numStates * N];
        for (int i = 0; i < dp.Length; i++) dp[i] = double.MaxValue;

        for (int v = 0; v < N; v++)
        {
            dp[(1 << v) * N + v] = distStart[v];
        }

        for (int mask = 1; mask < numStates; mask++)
        {
            for (int v = 0; v < N; v++)
            {
                if ((mask & (1 << v)) == 0) continue;
                double currentVal = dp[mask * N + v];
                if (currentVal == double.MaxValue) continue;

                for (int u = 0; u < N; u++)
                {
                    if ((mask & (1 << u)) != 0) continue;
                    int nextMask = mask | (1 << u);
                    double nextCost = currentVal + dist[v][u];
                    int nextState = nextMask * N + u;
                    if (nextCost < dp[nextState])
                    {
                        dp[nextState] = nextCost;
                    }
                }
            }
        }

        int bestLast = -1;
        double minCost = double.MaxValue;
        int finalMask = numStates - 1;
        for (int v = 0; v < N; v++)
        {
            double cost = dp[finalMask * N + v];
            if (cost < minCost)
            {
                minCost = cost;
                bestLast = v;
            }
        }

        int[] path = new int[N];
        int currentMask = finalMask;
        int currentLast = bestLast;
        path[N - 1] = currentLast;

        for (int step = N - 2; step >= 0; step--)
        {
            int prevMask = currentMask ^ (1 << currentLast);
            int prevLast = -1;
            double targetCost = dp[currentMask * N + currentLast];

            for (int u = 0; u < N; u++)
            {
                if ((prevMask & (1 << u)) == 0) continue;
                double diff = Math.Abs(dp[prevMask * N + u] + dist[u][currentLast] - targetCost);
                if (diff < 1e-9)
                {
                    prevLast = u;
                    break;
                }
            }

            if (prevLast == -1)
            {
                double bestDiff = double.MaxValue;
                for (int u = 0; u < N; u++)
                {
                    if ((prevMask & (1 << u)) == 0) continue;
                    double diff = Math.Abs(dp[prevMask * N + u] + dist[u][currentLast] - targetCost);
                    if (diff < bestDiff)
                    {
                        bestDiff = diff;
                        prevLast = u;
                    }
                }
            }

            path[step] = prevLast;
            currentMask = prevMask;
            currentLast = prevLast;
        }

        return path;
    }

    static int[] SolveSA(int N, double[] distStart, double[][] dist)
    {
        Random rng = new Random(42);

        // Nearest Neighbor Greedy Initialization
        int[] bestGlobalTour = new int[N];
        bool[] visited = new bool[N];
        int current = -1;
        for (int step = 0; step < N; step++)
        {
            int nextNode = -1;
            double bestDist = double.MaxValue;
            for (int i = 0; i < N; i++)
            {
                if (!visited[i])
                {
                    double d = (current == -1) ? distStart[i] : dist[current][i];
                    if (d < bestDist)
                    {
                        bestDist = d;
                        nextNode = i;
                    }
                }
            }
            if (nextNode == -1)
            {
                Console.Error.WriteLine($"Greedy failed: nextNode is -1 at step {step}, current is {current}");
                // Find first unvisited as fallback
                for (int i = 0; i < N; i++)
                {
                    if (!visited[i])
                    {
                        nextNode = i;
                        break;
                    }
                }
            }
            bestGlobalTour[step] = nextNode;
            visited[nextNode] = true;
            current = nextNode;
        }

        Console.Error.WriteLine("Greedy tour: " + string.Join(" ", bestGlobalTour));

        double bestGlobalCost = EvaluateTour(bestGlobalTour, distStart, dist);
        bestGlobalCost = Refine2Opt(bestGlobalTour, distStart, dist, bestGlobalCost);

        var stopwatch = Stopwatch.StartNew();
        long timeLimitMs = N > 50 ? 2500 : 1500;

        int restartCount = 0;
        int[] currentTour = new int[N];
        int[] bestRestartTour = new int[N];

        while (stopwatch.ElapsedMilliseconds < timeLimitMs)
        {
            if (restartCount == 0)
            {
                Array.Copy(bestGlobalTour, currentTour, N);
            }
            else
            {
                for (int i = 0; i < N; i++) currentTour[i] = i;
                for (int i = N - 1; i > 0; i--)
                {
                    int j = rng.Next(0, i + 1);
                    int tempVal = currentTour[i];
                    currentTour[i] = currentTour[j];
                    currentTour[j] = tempVal;
                }
            }

            double currentCost = EvaluateTour(currentTour, distStart, dist);
            Array.Copy(currentTour, bestRestartTour, N);
            double bestRestartCost = currentCost;

            double initialTemp = 100.0;
            double finalTemp = 0.001;
            int iterations = 100000;
            double coolingRate = Math.Pow(finalTemp / initialTemp, 1.0 / iterations);
            double temp = initialTemp;

            for (int iter = 0; iter < iterations; iter++)
            {
                int i = rng.Next(0, N);
                int j = rng.Next(0, N);
                if (i == j) continue;
                if (i > j) { int tmp = i; i = j; j = tmp; }

                double delta = 0;
                if (i > 0 && j < N - 1)
                {
                    delta = - dist[currentTour[i - 1]][currentTour[i]]
                            - dist[currentTour[j]][currentTour[j + 1]]
                            + dist[currentTour[i - 1]][currentTour[j]]
                            + dist[currentTour[i]][currentTour[j + 1]];
                }
                else if (i == 0 && j < N - 1)
                {
                    delta = - distStart[currentTour[0]]
                            - dist[currentTour[j]][currentTour[j + 1]]
                            + distStart[currentTour[j]]
                            + dist[currentTour[0]][currentTour[j + 1]];
                }
                else if (i > 0 && j == N - 1)
                {
                    delta = - dist[currentTour[i - 1]][currentTour[i]]
                            + dist[currentTour[i - 1]][currentTour[j]];
                }
                else if (i == 0 && j == N - 1)
                {
                    delta = - distStart[currentTour[0]]
                            + distStart[currentTour[j]];
                }

                if (delta < 0 || rng.NextDouble() < Math.Exp(-delta / temp))
                {
                    Array.Reverse(currentTour, i, j - i + 1);
                    currentCost += delta;

                    if (currentCost < bestRestartCost)
                    {
                        bestRestartCost = currentCost;
                        Array.Copy(currentTour, bestRestartTour, N);
                    }
                }

                temp *= coolingRate;
            }

            bestRestartCost = Refine2Opt(bestRestartTour, distStart, dist, bestRestartCost);

            if (bestRestartCost < bestGlobalCost)
            {
                bestGlobalCost = bestRestartCost;
                Array.Copy(bestRestartTour, bestGlobalTour, N);
            }

            restartCount++;
        }

        return bestGlobalTour;
    }

    static double Refine2Opt(int[] tour, double[] distStart, double[][] dist, double currentCost)
    {
        int N = tour.Length;
        bool improved = true;
        while (improved)
        {
            improved = false;
            for (int i = 0; i < N; i++)
            {
                for (int j = i + 1; j < N; j++)
                {
                    double delta = 0;
                    if (i > 0 && j < N - 1)
                    {
                        delta = - dist[tour[i - 1]][tour[i]]
                                - dist[tour[j]][tour[j + 1]]
                                + dist[tour[i - 1]][tour[j]]
                                + dist[tour[i]][tour[j + 1]];
                    }
                    else if (i == 0 && j < N - 1)
                    {
                        delta = - distStart[tour[0]]
                                - dist[tour[j]][tour[j + 1]]
                                + distStart[tour[j]]
                                + dist[tour[0]][tour[j + 1]];
                    }
                    else if (i > 0 && j == N - 1)
                    {
                        delta = - dist[tour[i - 1]][tour[i]]
                                + dist[tour[i - 1]][tour[j]];
                    }
                    else if (i == 0 && j == N - 1)
                    {
                        delta = - distStart[tour[0]]
                                + distStart[tour[j]];
                    }

                    if (delta < -1e-9)
                    {
                        Array.Reverse(tour, i, j - i + 1);
                        currentCost += delta;
                        improved = true;
                    }
                }
            }
        }
        return currentCost;
    }

    static double EvaluateTour(int[] tour, double[] distStart, double[][] dist)
    {
        double cost = distStart[tour[0]];
        for (int i = 0; i < tour.Length - 1; i++)
        {
            cost += dist[tour[i]][tour[i + 1]];
        }
        return cost;
    }

    static double EstimateTransferDeltaV(double[] orbit1, double[] orbit2, double mu)
    {
        double r1_x = orbit1[0];
        double r1_y = orbit1[1];
        double r1_z = orbit1[2];
        double v1_x = orbit1[3];
        double v1_y = orbit1[4];
        double v1_z = orbit1[5];

        double r2_x = orbit2[0];
        double r2_y = orbit2[1];
        double r2_z = orbit2[2];
        double v2_x = orbit2[3];
        double v2_y = orbit2[4];
        double v2_z = orbit2[5];

        double r1 = Math.Sqrt(r1_x * r1_x + r1_y * r1_y + r1_z * r1_z);
        double r2 = Math.Sqrt(r2_x * r2_x + r2_y * r2_y + r2_z * r2_z);
        double v1 = Math.Sqrt(v1_x * v1_x + v1_y * v1_y + v1_z * v1_z);
        double v2 = Math.Sqrt(v2_x * v2_x + v2_y * v2_y + v2_z * v2_z);

        double h1_x = r1_y * v1_z - r1_z * v1_y;
        double h1_y = r1_z * v1_x - r1_x * v1_z;
        double h1_z = r1_x * v1_y - r1_y * v1_x;

        double h2_x = r2_y * v2_z - r2_z * v2_y;
        double h2_y = r2_z * v2_x - r2_x * v2_z;
        double h2_z = r2_x * v2_y - r2_y * v2_x;

        double h1_mag = Math.Sqrt(h1_x * h1_x + h1_y * h1_y + h1_z * h1_z);
        double h2_mag = Math.Sqrt(h2_x * h2_x + h2_y * h2_y + h2_z * h2_z);

        double plane_change_angle = 0;
        if (h1_mag > 0 && h2_mag > 0)
        {
            double h1_nx = h1_x / h1_mag;
            double h1_ny = h1_y / h1_mag;
            double h1_nz = h1_z / h1_mag;

            double h2_nx = h2_x / h2_mag;
            double h2_ny = h2_y / h2_mag;
            double h2_nz = h2_z / h2_mag;

            double cos_plane_change = h1_nx * h2_nx + h1_ny * h2_ny + h1_nz * h2_nz;
            cos_plane_change = Math.Max(-1.0, Math.Min(1.0, cos_plane_change));
            plane_change_angle = Math.Acos(cos_plane_change);
        }

        double a_transfer = (r1 + r2) / 2.0;
        double in_plane_dv = 0;
        if (a_transfer > 0)
        {
            double v_t1 = Math.Sqrt(mu * (2.0 / r1 - 1.0 / a_transfer));
            double v_t2 = Math.Sqrt(mu * (2.0 / r2 - 1.0 / a_transfer));
            double v_c1 = Math.Sqrt(mu / r1);
            double v_c2 = Math.Sqrt(mu / r2);

            double dv1 = Math.Abs(v_t1 - v_c1);
            double dv2 = Math.Abs(v_c2 - v_t2);
            in_plane_dv = dv1 + dv2;
        }
        else
        {
            in_plane_dv = Math.Abs(v2 - v1);
        }

        double v_at_node = Math.Sqrt(mu * (2.0 / r1 + 2.0 / r2) / 2.0);
        double plane_change_dv = 2.0 * v_at_node * Math.Sin(plane_change_angle / 2.0);

        double total_dv = in_plane_dv + plane_change_dv;

        double vel_magnitude_diff = Math.Abs(v2 - v1);
        double velocity_penalty = vel_magnitude_diff * 0.1;

        return total_dv + velocity_penalty;
    }
}
