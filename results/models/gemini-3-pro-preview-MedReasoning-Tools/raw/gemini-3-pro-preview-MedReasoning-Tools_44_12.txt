#version 450

layout(location = 0) in vec3 worldPos;
layout(location = 1) in vec3 normal;
layout(location = 2) in vec2 uv;
layout(location = 3) in vec3 tangent;
layout(location = 4) in vec3 color;

layout(set = 0, binding = 0) uniform UBO {
    mat4 model;
    mat4 view;
    mat4 projection;
    vec4 lightPos;
    vec4 cameraPos;
    vec4 params;
} ubo;

layout(location = 0) out vec4 outColor;

void main() {
    // Normalize the interpolated normal vector
    vec3 N = normalize(normal);
    
    // Light direction vector from worldPos to light position
    vec3 L = normalize(ubo.lightPos.xyz - worldPos);
    
    // Diffuse shading term (Lambertian)
    float d = max(dot(N, L), 0.0);
    
    // Ambient colors for the hemisphere model
    vec3 skyColor = vec3(0.4, 0.6, 1.0);
    vec3 groundColor = vec3(0.3, 0.15, 0.05);
    
    // Blending factor based on normalized surface normal's Y-coordinate
    float blend = N.y * 0.5 + 0.5;
    
    // Interpolate ambient lighting between ground and sky
    vec3 ambient = mix(groundColor, skyColor, blend);
    
    // Add the scaled diffuse term
    vec3 finalColor = ambient + 0.3 * d;
    
    // Output final color with opaque alpha
    outColor = vec4(finalColor, 1.0);
}
