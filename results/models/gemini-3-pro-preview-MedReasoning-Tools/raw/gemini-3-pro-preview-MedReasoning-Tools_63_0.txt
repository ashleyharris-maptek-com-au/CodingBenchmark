declare const require: any;
const fs = require('fs');

interface Robot {
    id: number;
    x: number;
    y: number;
}

interface Job {
    id: any;
    x: number;
    y: number;
    priority: number;
}

interface InputData {
    grid: string[];
    robots: Robot[];
    jobs: Job[];
}

class GridDistanceFinder {
    width: number;
    height: number;
    grid: string[];
    interestMap: Map<number, number>;
    interestPoints: number[];
    cache: Map<number, Int32Array>;
    visited: Int32Array;
    bfsId: number;
    queue: Int32Array;
    dist: Int32Array;

    constructor(grid: string[], robots: Robot[], jobs: Job[]) {
        this.grid = grid;
        this.height = grid.length;
        this.width = grid[0].length;
        this.cache = new Map();
        
        const w = this.width;
        this.visited = new Int32Array(w * this.height);
        this.bfsId = 0;
        this.queue = new Int32Array(w * this.height);
        this.dist = new Int32Array(w * this.height);

        const interestSet = new Set<number>();
        for (const r of robots) {
            interestSet.add(r.y * w + r.x);
        }
        for (const j of jobs) {
            interestSet.add(j.y * w + j.x);
        }

        this.interestPoints = Array.from(interestSet);
        this.interestMap = new Map();
        for (let i = 0; i < this.interestPoints.length; i++) {
            this.interestMap.set(this.interestPoints[i], i);
        }
    }

    getDistance(sx: number, sy: number, tx: number, ty: number): number {
        if (sx < 0 || sx >= this.width || sy < 0 || sy >= this.height) return 1e9;
        if (tx < 0 || tx >= this.width || ty < 0 || ty >= this.height) return 1e9;

        const w = this.width;
        const sIdx = sy * w + sx;
        const tIdx = ty * w + tx;
        if (sIdx === tIdx) return 0;

        const tInterestIdx = this.interestMap.get(tIdx);
        if (tInterestIdx === undefined) return 1e9;

        let dists = this.cache.get(sIdx);
        if (dists) return dists[tInterestIdx];

        const sInterestIdx = this.interestMap.get(sIdx);
        if (sInterestIdx !== undefined) {
            const dists2 = this.cache.get(tIdx);
            if (dists2) return dists2[sInterestIdx];
        }

        dists = this.runBfs(sx, sy);
        this.cache.set(sIdx, dists);
        return dists[tInterestIdx];
    }

    private runBfs(sx: number, sy: number): Int32Array {
        const w = this.width;
        const h = this.height;

        this.bfsId++;
        const visited = this.visited;
        const queue = this.queue;
        const dist = this.dist;
        const grid = this.grid;

        let head = 0;
        let tail = 0;

        const startIdx = sy * w + sx;
        visited[startIdx] = this.bfsId;
        dist[startIdx] = 0;
        queue[tail++] = startIdx;

        const dirs = [-1, 1, -w, w];

        while (head < tail) {
            const curr = queue[head++];
            const currDist = dist[curr];

            const cx = curr % w;
            const cy = Math.floor(curr / w);

            for (let i = 0; i < 4; i++) {
                const diff = dirs[i];
                const next = curr + diff;

                if (i === 0 && cx === 0) continue;
                if (i === 1 && cx === w - 1) continue;
                if (i === 2 && cy === 0) continue;
                if (i === 3 && cy === h - 1) continue;

                if (visited[next] !== this.bfsId) {
                    visited[next] = this.bfsId;
                    const nx = next % w;
                    const ny = Math.floor(next / w);
                    if (grid[ny][nx] !== '#') {
                        dist[next] = currDist + 1;
                        queue[tail++] = next;
                    }
                }
            }
        }

        const numInterest = this.interestPoints.length;
        const res = new Int32Array(numInterest);
        for (let i = 0; i < numInterest; i++) {
            const pt = this.interestPoints[i];
            if (visited[pt] === this.bfsId) {
                res[i] = dist[pt];
            } else {
                res[i] = 1e9;
            }
        }
        return res;
    }
}

const jobCompare = (a: Job, b: Job) => {
    if (b.priority !== a.priority) {
        return b.priority - a.priority;
    }
    if (typeof a.id === 'number' && typeof b.id === 'number') {
        return a.id - b.id;
    }
    return String(a.id).localeCompare(String(b.id));
};

function runGreedyAssignment(robots: Robot[], jobs: Job[], distanceFinder: GridDistanceFinder) {
    const sortedJobs = [...jobs].sort(jobCompare);
    
    let maxRobotId = 0;
    for (const r of robots) {
        if (r.id > maxRobotId) {
            maxRobotId = r.id;
        }
    }
    
    const assignments: any[][] = Array.from({ length: maxRobotId + 1 }, () => []);

    const robotPositions = new Map<number, { x: number, y: number }>();
    for (const r of robots) {
        robotPositions.set(r.id, { x: r.x, y: r.y });
    }

    for (const job of sortedJobs) {
        let bestRobotId = -1;
        let minDistance = Infinity;

        for (const robot of robots) {
            const pos = robotPositions.get(robot.id)!;
            const dist = distanceFinder.getDistance(pos.x, pos.y, job.x, job.y);
            if (dist < minDistance) {
                minDistance = dist;
                bestRobotId = robot.id;
            } else if (dist === minDistance) {
                if (bestRobotId === -1 || robot.id < bestRobotId) {
                    bestRobotId = robot.id;
                }
            }
        }

        if (bestRobotId !== -1) {
            assignments[bestRobotId].push(job.id);
            robotPositions.set(bestRobotId, { x: job.x, y: job.y });
        }
    }

    return assignments;
}

function computeRobotPathDistance(robot: Robot, jobIds: any[], distanceFinder: GridDistanceFinder, jobMap: Map<any, Job>): number {
    if (jobIds.length === 0) return 0;
    
    let totalDist = 0;
    let currX = robot.x;
    let currY = robot.y;
    
    for (const jobId of jobIds) {
        const job = jobMap.get(jobId)!;
        const dist = distanceFinder.getDistance(currX, currY, job.x, job.y);
        totalDist += dist;
        currX = job.x;
        currY = job.y;
    }
    
    return totalDist;
}

function computeTotalDistance(robots: Robot[], assignments: any[][], distanceFinder: GridDistanceFinder, jobMap: Map<any, Job>): number {
    let total = 0;
    for (const robot of robots) {
        total += computeRobotPathDistance(robot, assignments[robot.id], distanceFinder, jobMap);
    }
    return total;
}

function optimizeRobotPath(robot: Robot, jobIds: any[], distanceFinder: GridDistanceFinder, jobMap: Map<any, Job>): any[] {
    const n = jobIds.length;
    if (n <= 1) return jobIds;

    const getCoords = (idx: number) => {
        if (idx === -1) return { x: robot.x, y: robot.y };
        const job = jobMap.get(jobIds[idx])!;
        return { x: job.x, y: job.y };
    };

    const getDist = (i: number, j: number) => {
        const p1 = getCoords(i);
        const p2 = getCoords(j);
        return distanceFinder.getDistance(p1.x, p1.y, p2.x, p2.y);
    };

    if (n <= 8) {
        let bestPath = [...jobIds];
        let bestDist = Infinity;

        const permute = (arr: any[], memo: any[] = []) => {
            if (arr.length === 0) {
                let dist = 0;
                let lastIdx = -1;
                for (let i = 0; i < memo.length; i++) {
                    const idx = jobIds.indexOf(memo[i]);
                    dist += getDist(lastIdx, idx);
                    lastIdx = idx;
                }
                if (dist < bestDist) {
                    bestDist = dist;
                    bestPath = [...memo];
                }
                return;
            }
            for (let i = 0; i < arr.length; i++) {
                const curr = arr.slice();
                const next = curr.splice(i, 1);
                permute(curr.slice(), memo.concat(next));
            }
        };

        permute(jobIds);
        return bestPath;
    }

    let currentPath = [...jobIds];
    let improved = true;
    
    while (improved) {
        improved = false;
        for (let i = -1; i < n - 1; i++) {
            for (let j = i + 2; j < n; j++) {
                const oldDist1 = getDist(i, i + 1);
                const oldDist2 = (j + 1 < n) ? getDist(j, j + 1) : 0;
                const newDist1 = getDist(i, j);
                const newDist2 = (j + 1 < n) ? getDist(i + 1, j + 1) : 0;

                if (newDist1 + newDist2 < oldDist1 + oldDist2) {
                    let left = i + 1;
                    let right = j;
                    while (left < right) {
                        const temp = currentPath[left];
                        currentPath[left] = currentPath[right];
                        currentPath[right] = temp;
                        left++;
                        right--;
                    }
                    improved = true;
                }
            }
        }
    }

    return currentPath;
}

function optimizeAssignments(
    robots: Robot[],
    jobs: Job[],
    initialAssignments: any[][],
    distanceFinder: GridDistanceFinder,
    jobMap: Map<any, Job>,
    robotMap: Map<number, Robot>,
    timeLimitMs: number
) {
    const startTime = Date.now();
    let currentAssignments = initialAssignments.map(arr => [...arr]);
    
    for (const robot of robots) {
        currentAssignments[robot.id] = optimizeRobotPath(robot, currentAssignments[robot.id], distanceFinder, jobMap);
    }

    let bestAssignments = currentAssignments.map(arr => [...arr]);
    let bestTotalDist = computeTotalDistance(robots, bestAssignments, distanceFinder, jobMap);

    const robotIds = robots.map(r => r.id);

    while (Date.now() - startTime < timeLimitMs) {
        const isSwap = Math.random() < 0.3;

        if (isSwap) {
            if (robotIds.length < 2) continue;
            const r1 = robotIds[Math.floor(Math.random() * robotIds.length)];
            const r2 = robotIds[Math.floor(Math.random() * robotIds.length)];
            if (r1 === r2) continue;

            const path1 = currentAssignments[r1];
            const path2 = currentAssignments[r2];
            if (path1.length === 0 || path2.length === 0) continue;

            const idx1 = Math.floor(Math.random() * path1.length);
            const idx2 = Math.floor(Math.random() * path2.length);

            const job1 = path1[idx1];
            const job2 = path2[idx2];

            const newPath1 = [...path1];
            newPath1[idx1] = job2;
            const newPath2 = [...path2];
            newPath2[idx2] = job1;

            const robot1 = robotMap.get(r1)!;
            const robot2 = robotMap.get(r2)!;

            const optPath1 = optimizeRobotPath(robot1, newPath1, distanceFinder, jobMap);
            const optPath2 = optimizeRobotPath(robot2, newPath2, distanceFinder, jobMap);

            const oldDist = computeRobotPathDistance(robot1, path1, distanceFinder, jobMap) +
                            computeRobotPathDistance(robot2, path2, distanceFinder, jobMap);
            const newDist = computeRobotPathDistance(robot1, optPath1, distanceFinder, jobMap) +
                            computeRobotPathDistance(robot2, optPath2, distanceFinder, jobMap);

            if (newDist < oldDist) {
                currentAssignments[r1] = optPath1;
                currentAssignments[r2] = optPath2;
                
                const totalDist = computeTotalDistance(robots, currentAssignments, distanceFinder, jobMap);
                if (totalDist < bestTotalDist) {
                    bestTotalDist = totalDist;
                    bestAssignments = currentAssignments.map(arr => [...arr]);
                }
            }
        } else {
            const r1 = robotIds[Math.floor(Math.random() * robotIds.length)];
            const path1 = currentAssignments[r1];
            if (path1.length === 0) continue;

            const r2 = robotIds[Math.floor(Math.random() * robotIds.length)];
            
            const idx = Math.floor(Math.random() * path1.length);
            const job = path1[idx];

            const newPath1 = path1.filter((_, i) => i !== idx);

            const newPath2 = [...currentAssignments[r2]];
            const insertIdx = Math.floor(Math.random() * (newPath2.length + 1));
            newPath2.splice(insertIdx, 0, job);

            const robot1 = robotMap.get(r1)!;
            const robot2 = robotMap.get(r2)!;

            const optPath1 = optimizeRobotPath(robot1, newPath1, distanceFinder, jobMap);
            const optPath2 = optimizeRobotPath(robot2, newPath2, distanceFinder, jobMap);

            const oldDist = computeRobotPathDistance(robot1, path1, distanceFinder, jobMap) +
                            computeRobotPathDistance(robot2, currentAssignments[r2], distanceFinder, jobMap);
            const newDist = computeRobotPathDistance(robot1, optPath1, distanceFinder, jobMap) +
                            computeRobotPathDistance(robot2, optPath2, distanceFinder, jobMap);

            if (newDist < oldDist) {
                currentAssignments[r1] = optPath1;
                currentAssignments[r2] = optPath2;

                const totalDist = computeTotalDistance(robots, currentAssignments, distanceFinder, jobMap);
                if (totalDist < bestTotalDist) {
                    bestTotalDist = totalDist;
                    bestAssignments = currentAssignments.map(arr => [...arr]);
                }
            }
        }
    }

    return bestAssignments;
}

function main() {
    const input: string = fs.readFileSync(0, 'utf8');
    const data: InputData = JSON.parse(input);

    if (!data.robots || data.robots.length === 0) {
        console.log(JSON.stringify({ assignments: [] }));
        return;
    }

    const distanceFinder = new GridDistanceFinder(data.grid, data.robots, data.jobs);

    const jobMap = new Map<any, Job>();
    for (const job of data.jobs) {
        jobMap.set(job.id, job);
    }

    const robotMap = new Map<number, Robot>();
    for (const robot of data.robots) {
        robotMap.set(robot.id, robot);
    }

    const greedyAssignments = runGreedyAssignment(data.robots, data.jobs, distanceFinder);

    let timeLimitMs = 1500;
    const numJobs = data.jobs.length;
    const numRobots = data.robots.length;
    if (numRobots <= 1) {
        timeLimitMs = 0;
    } else if (numJobs <= 5) {
        timeLimitMs = 50;
    } else if (numJobs <= 20) {
        timeLimitMs = 200;
    } else if (numJobs <= 100) {
        timeLimitMs = 800;
    }

    const finalAssignments = optimizeAssignments(
        data.robots,
        data.jobs,
        greedyAssignments,
        distanceFinder,
        jobMap,
        robotMap,
        timeLimitMs
    );

    let maxRobotId = 0;
    for (const r of data.robots) {
        if (r.id > maxRobotId) {
            maxRobotId = r.id;
        }
    }

    const assignmentsOutput: any[][] = Array.from({ length: maxRobotId + 1 }, () => []);
    for (const robot of data.robots) {
        assignmentsOutput[robot.id] = finalAssignments[robot.id] || [];
    }

    console.log(JSON.stringify({ assignments: assignmentsOutput }));
}

main();
