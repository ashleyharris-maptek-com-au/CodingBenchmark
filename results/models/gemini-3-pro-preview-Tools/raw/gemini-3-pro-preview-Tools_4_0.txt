import sys
import os
import time

# Add parent directory to path to import 4.py and native_compiler
sys.path.insert(0, os.path.abspath(".."))

import importlib
# Import 4.py as a module
try:
    problem_module = importlib.import_module("4")
except Exception as e:
    print(f"Error importing 4.py: {e}")
    sys.exit(1)

def run_tests():
    # Read answer.txt
    if not os.path.exists("answer.txt"):
        print("answer.txt not found!")
        return

    with open("answer.txt", "r", encoding="utf-8") as f:
        code = f.read()

    result = {"cpp_code": code}

    print("=" * 60)
    print("Running local packing benchmark tests...")
    print("=" * 60)

    total_score = 0.0
    num_cases = len(problem_module.CONTAINER_CONFIGS)

    for subpass in range(num_cases):
        name = problem_module.CONTAINER_CONFIGS[subpass][0]
        print(f"Subpass {subpass}: {name} - Starting...")
        t0 = time.time()
        try:
            score, explanation = problem_module.gradeAnswer(result, subpass, "test_engine")
            elapsed = time.time() - t0
            print(f"Subpass {subpass}: {name} -> Score: {score:.3f} | {explanation} (took {elapsed:.2f}s)")
            total_score += score
        except Exception as e:
            import traceback
            traceback.print_exc()
            print(f"Subpass {subpass}: {name} -> FAILED with exception: {e}")
        print("-" * 60)

    print(f"Completed. Total score: {total_score:.3f} / {num_cases:.1f}")

if __name__ == "__main__":
    run_tests()
