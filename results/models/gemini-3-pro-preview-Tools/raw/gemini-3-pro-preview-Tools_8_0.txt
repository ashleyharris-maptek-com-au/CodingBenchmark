use std::collections::VecDeque;
use std::io::{self, BufRead, BufReader, BufWriter, Write};

#[derive(Clone, Copy, Debug)]
struct Cell {
    r: u32,
    c: u32,
}

#[inline(always)]
fn get_parent(u: usize, val: u8, w: usize) -> usize {
    match val {
        2 | 7 => u - w,
        3 | 8 => u + w,
        4 | 9 => u - 1,
        5 | 10 => u + 1,
        _ => unreachable!(),
    }
}

#[inline]
fn write_usize<W: Write>(out: &mut W, mut val: usize) -> io::Result<()> {
    if val == 0 {
        return out.write_all(b"0");
    }
    let mut buf = [0u8; 20];
    let mut idx = 20;
    while val > 0 {
        idx -= 1;
        buf[idx] = b'0' + (val % 10) as u8;
        val /= 10;
    }
    out.write_all(&buf[idx..])
}

#[inline]
fn write_coord<W: Write>(out: &mut W, c: usize, r: usize) -> io::Result<()> {
    write_usize(out, c)?;
    out.write_all(b" ")?;
    write_usize(out, r)?;
    out.write_all(b"\n")
}

fn main() -> io::Result<()> {
    let stdin = io::stdin();
    // Use an 8MB buffer for reading
    let mut reader = BufReader::with_capacity(8 * 1024 * 1024, stdin.lock());

    let mut first_line = String::new();
    reader.read_line(&mut first_line)?;
    let parts: Vec<&str> = first_line.split_whitespace().collect();
    if parts.len() < 2 {
        println!("0");
        return Ok(());
    }
    let h: usize = parts[0].parse().unwrap();
    let w: usize = parts[1].parse().unwrap();

    let total_cells = match h.checked_mul(w) {
        Some(val) => val,
        None => {
            println!("0");
            return Ok(());
        }
    };

    let mut grid = Vec::new();
    if grid.try_reserve_exact(total_cells).is_err() {
        println!("0");
        return Ok(());
    }
    grid.resize(total_cells, 0u8);

    let mut a_idx = 0;
    let mut b_idx = 0;
    let mut a_found = false;
    let mut b_found = false;
    let mut buf = Vec::new();

    for r in 0..h {
        buf.clear();
        reader.read_until(b'\n', &mut buf)?;
        if buf.len() < w {
            println!("0");
            return Ok(());
        }
        for c in 0..w {
            let b = buf[c];
            let idx = r * w + c;
            match b {
                b'#' => grid[idx] = 0xFF,
                b' ' => grid[idx] = 0,
                b'A' => {
                    grid[idx] = 1;
                    a_idx = idx;
                    a_found = true;
                }
                b'B' => {
                    grid[idx] = 6;
                    b_idx = idx;
                    b_found = true;
                }
                _ => grid[idx] = 0xFF,
            }
        }
    }

    if !a_found || !b_found {
        println!("0");
        return Ok(());
    }

    if a_idx == b_idx {
        let r = a_idx / w;
        let c = a_idx % w;
        println!("1\n{} {}", c, r);
        return Ok(());
    }

    let mut q_f = VecDeque::with_capacity(65536);
    let mut q_b = VecDeque::with_capacity(65536);

    q_f.push_back(Cell { r: (a_idx / w) as u32, c: (a_idx % w) as u32 });
    q_b.push_back(Cell { r: (b_idx / w) as u32, c: (b_idx % w) as u32 });

    let mut meeting_f: Option<usize> = None;
    let mut meeting_b: Option<usize> = None;

    'search: while !q_f.is_empty() && !q_b.is_empty() {
        if q_f.len() <= q_b.len() {
            let curr = q_f.pop_front().unwrap();
            let curr_idx = curr.r as usize * w + curr.c as usize;

            if curr.r > 0 {
                let n_idx = curr_idx - w;
                let val = grid[n_idx];
                if val == 0 {
                    grid[n_idx] = 3;
                    q_f.push_back(Cell { r: curr.r - 1, c: curr.c });
                } else if val >= 6 && val <= 10 {
                    meeting_f = Some(curr_idx);
                    meeting_b = Some(n_idx);
                    break 'search;
                }
            }
            if (curr.r as usize) < h - 1 {
                let n_idx = curr_idx + w;
                let val = grid[n_idx];
                if val == 0 {
                    grid[n_idx] = 2;
                    q_f.push_back(Cell { r: curr.r + 1, c: curr.c });
                } else if val >= 6 && val <= 10 {
                    meeting_f = Some(curr_idx);
                    meeting_b = Some(n_idx);
                    break 'search;
                }
            }
            if curr.c > 0 {
                let n_idx = curr_idx - 1;
                let val = grid[n_idx];
                if val == 0 {
                    grid[n_idx] = 5;
                    q_f.push_back(Cell { r: curr.r, c: curr.c - 1 });
                } else if val >= 6 && val <= 10 {
                    meeting_f = Some(curr_idx);
                    meeting_b = Some(n_idx);
                    break 'search;
                }
            }
            if (curr.c as usize) < w - 1 {
                let n_idx = curr_idx + 1;
                let val = grid[n_idx];
                if val == 0 {
                    grid[n_idx] = 4;
                    q_f.push_back(Cell { r: curr.r, c: curr.c + 1 });
                } else if val >= 6 && val <= 10 {
                    meeting_f = Some(curr_idx);
                    meeting_b = Some(n_idx);
                    break 'search;
                }
            }
        } else {
            let curr = q_b.pop_front().unwrap();
            let curr_idx = curr.r as usize * w + curr.c as usize;

            if curr.r > 0 {
                let n_idx = curr_idx - w;
                let val = grid[n_idx];
                if val == 0 {
                    grid[n_idx] = 8;
                    q_b.push_back(Cell { r: curr.r - 1, c: curr.c });
                } else if val >= 1 && val <= 5 {
                    meeting_f = Some(n_idx);
                    meeting_b = Some(curr_idx);
                    break 'search;
                }
            }
            if (curr.r as usize) < h - 1 {
                let n_idx = curr_idx + w;
                let val = grid[n_idx];
                if val == 0 {
                    grid[n_idx] = 7;
                    q_b.push_back(Cell { r: curr.r + 1, c: curr.c });
                } else if val >= 1 && val <= 5 {
                    meeting_f = Some(n_idx);
                    meeting_b = Some(curr_idx);
                    break 'search;
                }
            }
            if curr.c > 0 {
                let n_idx = curr_idx - 1;
                let val = grid[n_idx];
                if val == 0 {
                    grid[n_idx] = 10;
                    q_b.push_back(Cell { r: curr.r, c: curr.c - 1 });
                } else if val >= 1 && val <= 5 {
                    meeting_f = Some(n_idx);
                    meeting_b = Some(curr_idx);
                    break 'search;
                }
            }
            if (curr.c as usize) < w - 1 {
                let n_idx = curr_idx + 1;
                let val = grid[n_idx];
                if val == 0 {
                    grid[n_idx] = 9;
                    q_b.push_back(Cell { r: curr.r, c: curr.c + 1 });
                } else if val >= 1 && val <= 5 {
                    meeting_f = Some(n_idx);
                    meeting_b = Some(curr_idx);
                    break 'search;
                }
            }
        }
    }

    if let (Some(mf), Some(mb)) = (meeting_f, meeting_b) {
        let mut path_f = Vec::new();
        let mut curr = mf;
        loop {
            path_f.push(curr as u32);
            let val = grid[curr];
            if val == 1 {
                break;
            }
            curr = get_parent(curr, val, w);
        }
        path_f.reverse();

        let mut path_b = Vec::new();
        curr = mb;
        loop {
            path_b.push(curr as u32);
            let val = grid[curr];
            if val == 6 {
                break;
            }
            curr = get_parent(curr, val, w);
        }

        let total_steps = path_f.len() + path_b.len();
        let stdout = io::stdout();
        // Use an 8MB buffer for writing
        let mut out = BufWriter::with_capacity(8 * 1024 * 1024, stdout.lock());
        
        write_usize(&mut out, total_steps)?;
        out.write_all(b"\n")?;

        for idx in path_f {
            let idx = idx as usize;
            let r = idx / w;
            let c = idx % w;
            write_coord(&mut out, c, r)?;
        }
        for idx in path_b {
            let idx = idx as usize;
            let r = idx / w;
            let c = idx % w;
            write_coord(&mut out, c, r)?;
        }
        out.flush()?;
    } else {
        println!("0");
    }

    Ok(())
}
