use std::io::{self, Read, BufWriter, Write};
use std::thread;

// Seedable LCG random number generator to avoid external dependencies
struct Lcg {
    state: u64,
}

impl Lcg {
    fn new(seed: u64) -> Self {
        Self { state: seed }
    }
    
    fn next_f32(&mut self) -> f32 {
        self.state = self.state.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
        ((self.state >> 32) as u32) as f32 / u32::MAX as f32
    }
}

// Disjoint Set Union for component detection
struct Dsu {
    parent: Vec<usize>,
}

impl Dsu {
    fn new(n: usize) -> Self {
        Self {
            parent: (0..n).collect(),
        }
    }
    
    fn find(&mut self, i: usize) -> usize {
        let mut root = i;
        while root != self.parent[root] {
            root = self.parent[root];
        }
        let mut curr = i;
        while curr != root {
            let nxt = self.parent[curr];
            self.parent[curr] = root;
            curr = nxt;
        }
        root
    }
    
    fn union(&mut self, i: usize, j: usize) {
        let root_i = self.find(i);
        let root_j = self.find(j);
        if root_i != root_j {
            self.parent[root_i] = root_j;
        }
    }
}

fn cell_hash(cx: i32, cy: i32) -> usize {
    ((cx as usize).wrapping_mul(73856093) ^ (cy as usize).wrapping_mul(19349663))
}

fn main() {
    let mut input = String::new();
    if io::stdin().read_to_string(&mut input).is_err() {
        return;
    }
    let mut tokens = input.split_whitespace();
    
    let n_str = match tokens.next() {
        Some(s) => s,
        None => return,
    };
    let n: usize = n_str.parse().unwrap();
    
    let m_str = match tokens.next() {
        Some(s) => s,
        None => return,
    };
    let m: usize = m_str.parse().unwrap();
    
    let mut edges = Vec::with_capacity(m);
    for _ in 0..m {
        if let (Some(u_str), Some(v_str)) = (tokens.next(), tokens.next()) {
            let u: usize = u_str.parse().unwrap();
            let v: usize = v_str.parse().unwrap();
            edges.push((u, v));
        } else {
            break;
        }
    }
    
    // 1. Deduplicate and clean edges
    let mut unique_edges = Vec::with_capacity(edges.len());
    for (u, v) in edges {
        if u != v && u < n && v < n {
            unique_edges.push((u.min(v), u.max(v)));
        }
    }
    unique_edges.sort_unstable();
    unique_edges.dedup();
    
    // 2. Component detection using DSU
    let mut dsu = Dsu::new(n);
    for &(u, v) in &unique_edges {
        dsu.union(u, v);
    }
    
    let mut component_map = std::collections::HashMap::new();
    for i in 0..n {
        let root = dsu.find(i);
        component_map.entry(root).or_insert_with(Vec::new).push(i);
    }
    
    let mut components: Vec<Vec<usize>> = component_map.into_values().collect();
    components.sort_unstable_by(|a, b| b.len().cmp(&a.len()));
    
    // 3. Initialize positions
    let mut x = vec![0.0f32; n];
    let mut y = vec![0.0f32; n];
    
    let mut centers = vec![(0.0f32, 0.0f32); components.len()];
    if components.len() > 1 {
        let mut current_radius = 0.0f32;
        let mut current_angle = 0.0f32;
        for j in 0..components.len() {
            if j == 0 {
                centers[j] = (0.0f32, 0.0f32);
                let size = components[j].len() as f32;
                current_radius = 2.0 * size.sqrt() * 15.0;
            } else {
                centers[j] = (
                    current_radius * current_angle.cos(),
                    current_radius * current_angle.sin()
                );
                current_angle += 0.5f32;
                let comp_size = components[j].len() as f32;
                current_radius += (comp_size.sqrt() + 1.0f32) * 15.0f32;
            }
        }
    }
    
    let mut rng = Lcg::new(42);
    for j in 0..components.len() {
        let (cx, cy) = centers[j];
        let comp_size = components[j].len() as f32;
        let r = 2.0 * comp_size.sqrt() * 5.0 + 5.0;
        for &u in &components[j] {
            let angle = rng.next_f32() * 2.0 * std::f32::consts::PI;
            let dist = rng.next_f32().sqrt() * r;
            x[u] = cx + dist * angle.cos();
            y[u] = cy + dist * angle.sin();
        }
    }
    
    // 4. Force-directed layout parameters
    let l_param = 15.0f32;
    let k_sq = l_param * l_param;
    let cell_size = 2.0 * l_param;
    let gravity = 0.05 * l_param;
    
    // Select number of iterations and mode based on N
    let (iterations, use_grid) = if n <= 500 {
        let iters = if n <= 15 { 500 } else if n <= 100 { 400 } else { 300 };
        (iters, false)
    } else {
        let iters = if n <= 5000 { 150 } else if n <= 20000 { 80 } else if n <= 100000 { 40 } else { 20 };
        (iters, true)
    };
    
    let initial_temp = (0.1 * (n as f32).sqrt() * l_param).min(1.5 * l_param);
    let mut temp = initial_temp;
    let cooling_factor = 0.95f32;
    
    // We can pre-allocate force vectors
    let mut fx = vec![0.0f32; n];
    let mut fy = vec![0.0f32; n];
    
    // Get available parallelism
    let available_cores = thread::available_parallelism().map(|n| n.get()).unwrap_or(1);
    // Limit thread count based on size
    let num_threads = if n > 1000 { available_cores.min(64) } else { 1 };
    
    // Build adjacency list representation for attractive force
    let mut adj = vec![Vec::new(); n];
    for &(u, v) in &unique_edges {
        adj[u].push(v);
        adj[v].push(u);
    }
    
    // Spatial hashing structures (reused across iterations)
    let table_size = if use_grid {
        (n * 2).next_power_of_two()
    } else {
        2
    };
    let mut head = vec![u32::MAX; table_size];
    let mut next = vec![u32::MAX; n];
    let mut node_cell_x = vec![0i32; n];
    let mut node_cell_y = vec![0i32; n];
    
    for _iter in 0..iterations {
        // Clear forces
        fx.fill(0.0);
        fy.fill(0.0);
        
        // A. Attractive forces (along edges)
        // Since it's O(M), we can do it sequentially or in parallel.
        // For simplicity and speed, sequential is very fast.
        for u in 0..n {
            let xu = x[u];
            let yu = y[u];
            for &v in &adj[u] {
                // To avoid double computing, we only process each edge once
                if u < v {
                    let dx = x[v] - xu;
                    let dy = y[v] - yu;
                    let d_sq = dx * dx + dy * dy;
                    let d = d_sq.sqrt();
                    if d > 1e-4 {
                        let f_a = d_sq / l_param;
                        let f_ax = dx * (f_a / d);
                        let f_ay = dy * (f_a / d);
                        fx[u] += f_ax;
                        fy[u] += f_ay;
                        fx[v] -= f_ax;
                        fy[v] -= f_ay;
                    } else {
                        // Small kick to separate overlapping connected nodes
                        let mut r = Lcg::new((u as u64) ^ (v as u64) ^ (_iter as u64));
                        let angle = r.next_f32() * 2.0 * std::f32::consts::PI;
                        let kick = 0.1;
                        fx[u] -= kick * angle.cos();
                        fy[u] -= kick * angle.sin();
                        fx[v] += kick * angle.cos();
                        fy[v] += kick * angle.sin();
                    }
                }
            }
        }
        
        // B. Repulsive forces
        if !use_grid {
            // Full O(N^2) pairwise repulsive forces
            for i in 0..n {
                let xi = x[i];
                let yi = y[i];
                let mut rfx = 0.0;
                let mut rfy = 0.0;
                for j in 0..n {
                    if i != j {
                        let dx = xi - x[j];
                        let dy = yi - y[j];
                        let dist_sq = dx * dx + dy * dy;
                        let dist_sq = if dist_sq < 1e-4 { 1e-4 } else { dist_sq };
                        let factor = k_sq / dist_sq;
                        rfx += dx * factor;
                        rfy += dy * factor;
                    }
                }
                fx[i] += rfx;
                fy[i] += rfy;
            }
        } else {
            // Update node cells
            for i in 0..n {
                node_cell_x[i] = (x[i] / cell_size).floor() as i32;
                node_cell_y[i] = (y[i] / cell_size).floor() as i32;
            }
            
            // Build spatial hash map
            head.fill(u32::MAX);
            for i in 0..n {
                let h = cell_hash(node_cell_x[i], node_cell_y[i]) & (table_size - 1);
                next[i] = head[h];
                head[h] = i as u32;
            }
            
            // Parallel/Sequential repulsive forces using spatial hash
            if num_threads <= 1 {
                // Sequential
                for i in 0..n {
                    let xi = x[i];
                    let yi = y[i];
                    let cx = node_cell_x[i];
                    let cy = node_cell_y[i];
                    let mut rfx = 0.0;
                    let mut rfy = 0.0;
                    
                    // Query 9 cells
                    for ncy in (cy - 1)..=(cy + 1) {
                        for ncx in (cx - 1)..=(cx + 1) {
                            let nh = cell_hash(ncx, ncy) & (table_size - 1);
                            let mut curr = head[nh];
                            while curr != u32::MAX {
                                let j = curr as usize;
                                if j != i && node_cell_x[j] == ncx && node_cell_y[j] == ncy {
                                    let dx = xi - x[j];
                                    let dy = yi - y[j];
                                    let dist_sq = dx * dx + dy * dy;
                                    if dist_sq < cell_size * cell_size {
                                        let dist_sq = if dist_sq < 1e-4 { 1e-4 } else { dist_sq };
                                        let factor = k_sq / dist_sq;
                                        rfx += dx * factor;
                                        rfy += dy * factor;
                                    }
                                }
                                curr = next[j];
                            }
                        }
                    }
                    fx[i] += rfx;
                    fy[i] += rfy;
                }
            } else {
                // Parallel implementation using thread::scope
                let chunk_size = (n + num_threads - 1) / num_threads;
                let mut fx_chunks = fx.chunks_mut(chunk_size);
                let mut fy_chunks = fy.chunks_mut(chunk_size);
                
                thread::scope(|s| {
                    for chunk_idx in 0..num_threads {
                        let chunk_fx = match fx_chunks.next() {
                            Some(c) => c,
                            None => break,
                        };
                        let chunk_fy = fy_chunks.next().unwrap();
                        let start_i = chunk_idx * chunk_size;
                        
                        let x_ref = &x;
                        let y_ref = &y;
                        let ncx_ref = &node_cell_x;
                        let ncy_ref = &node_cell_y;
                        let head_ref = &head;
                        let next_ref = &next;
                        
                        s.spawn(move || {
                            for (offset, (rfx, rfy)) in chunk_fx.iter_mut().zip(chunk_fy.iter_mut()).enumerate() {
                                let i = start_i + offset;
                                let xi = x_ref[i];
                                let yi = y_ref[i];
                                let cx = ncx_ref[i];
                                let cy = ncy_ref[i];
                                
                                for ncy in (cy - 1)..=(cy + 1) {
                                    for ncx in (cx - 1)..=(cx + 1) {
                                        let nh = cell_hash(ncx, ncy) & (table_size - 1);
                                        let mut curr = head_ref[nh];
                                        while curr != u32::MAX {
                                            let j = curr as usize;
                                            if j != i && ncx_ref[j] == ncx && ncy_ref[j] == ncy {
                                                let dx = xi - x_ref[j];
                                                let dy = yi - y_ref[j];
                                                let dist_sq = dx * dx + dy * dy;
                                                if dist_sq < cell_size * cell_size {
                                                    let dist_sq = if dist_sq < 1e-4 { 1e-4 } else { dist_sq };
                                                    let factor = k_sq / dist_sq;
                                                    *rfx += dx * factor;
                                                    *rfy += dy * factor;
                                                }
                                            }
                                            curr = next_ref[j];
                                        }
                                    }
                                }
                            }
                        });
                    }
                });
            }
        }
        
        // C. Gravity force and updating positions
        for i in 0..n {
            let xi = x[i];
            let yi = y[i];
            let g_dist_sq = xi * xi + yi * yi;
            let g_dist = g_dist_sq.sqrt();
            if g_dist > 1e-4 {
                fx[i] -= gravity * (xi / g_dist);
                fy[i] -= gravity * (yi / g_dist);
            }
            
            let mut dx = fx[i];
            let mut dy = fy[i];
            let len = (dx * dx + dy * dy).sqrt();
            if len > temp {
                let ratio = temp / len;
                dx *= ratio;
                dy *= ratio;
            }
            x[i] += dx;
            y[i] += dy;
        }
        
        temp *= cooling_factor;
    }
    
    // 5. Output results
    let stdout = io::stdout();
    let mut writer = BufWriter::new(stdout.lock());
    writeln!(writer, "{}", n).unwrap();
    for i in 0..n {
        writeln!(writer, "{:.4} {:.4}", x[i], y[i]).unwrap();
    }
}
