// Vector math and utility structs
struct Point3D {
    float x, y, z;
    Point3D operator+(const Point3D& o) const { return {x + o.x, y + o.y, z + o.z}; }
    Point3D operator-(const Point3D& o) const { return {x - o.x, y - o.y, z - o.z}; }
    Point3D operator*(float f) const { return {x * f, y * f, z * f}; }
    float dot(const Point3D& o) const { return x * o.x + y * o.y + z * o.z; }
    Point3D cross(const Point3D& o) const {
        return {
            y * o.z - z * o.y,
            z * o.x - x * o.z,
            x * o.y - y * o.x
        };
    }
    float norm() const { return std::sqrt(x*x + y*y + z*z); }
    Point3D normalized() const {
        float n = norm();
        if (n < 1e-12f) return {0,0,0};
        return {x / n, y / n, z / n};
    }
};

struct Point2D {
    float x, y;
};

struct AABB {
    float min_val[3] = {std::numeric_limits<float>::infinity(), std::numeric_limits<float>::infinity(), std::numeric_limits<float>::infinity()};
    float max_val[3] = {-std::numeric_limits<float>::infinity(), -std::numeric_limits<float>::infinity(), -std::numeric_limits<float>::infinity()};

    void grow(const float p[3]) {
        min_val[0] = std::min(min_val[0], p[0]);
        min_val[1] = std::min(min_val[1], p[1]);
        min_val[2] = std::min(min_val[2], p[2]);
        max_val[0] = std::max(max_val[0], p[0]);
        max_val[1] = std::max(max_val[1], p[1]);
        max_val[2] = std::max(max_val[2], p[2]);
    }

    void grow(const Point3D& p) {
        min_val[0] = std::min(min_val[0], p.x);
        min_val[1] = std::min(min_val[1], p.y);
        min_val[2] = std::min(min_val[2], p.z);
        max_val[0] = std::max(max_val[0], p.x);
        max_val[1] = std::max(max_val[1], p.y);
        max_val[2] = std::max(max_val[2], p.z);
    }

    void grow(const AABB& other) {
        min_val[0] = std::min(min_val[0], other.min_val[0]);
        min_val[1] = std::min(min_val[1], other.min_val[1]);
        min_val[2] = std::min(min_val[2], other.min_val[2]);
        max_val[0] = std::max(max_val[0], other.max_val[0]);
        max_val[1] = std::max(max_val[1], other.max_val[1]);
        max_val[2] = std::max(max_val[2], other.max_val[2]);
    }

    bool intersects(const AABB& other) const {
        if (max_val[0] < other.min_val[0] || min_val[0] > other.max_val[0]) return false;
        if (max_val[1] < other.min_val[1] || min_val[1] > other.max_val[1]) return false;
        if (max_val[2] < other.min_val[2] || min_val[2] > other.max_val[2]) return false;
        return true;
    }
};

struct BVHNode {
    AABB bbox;
    int left_child = -1;
    int right_child = -1;
    int first_prim = -1;
    int prim_count = 0;
};

struct BVH {
    struct Prim {
        int id;
        AABB bbox;
        Point3D centroid;
    };
    std::vector<Prim> prims;
    std::vector<BVHNode> nodes;
    int root = -1;

    void build(const std::vector<float>& vertices, const std::vector<int>& faces) {
        int num_faces = faces.size() / 3;
        prims.resize(num_faces);
        for (int i = 0; i < num_faces; ++i) {
            prims[i].id = i;
            prims[i].bbox = AABB();
            for (int j = 0; j < 3; ++j) {
                int vid = faces[3 * i + j];
                prims[i].bbox.grow(&vertices[3 * vid]);
            }
            prims[i].centroid.x = (prims[i].bbox.min_val[0] + prims[i].bbox.max_val[0]) * 0.5f;
            prims[i].centroid.y = (prims[i].bbox.min_val[1] + prims[i].bbox.max_val[1]) * 0.5f;
            prims[i].centroid.z = (prims[i].bbox.min_val[2] + prims[i].bbox.max_val[2]) * 0.5f;
        }

        nodes.reserve(num_faces * 2);
        root = build_recursive(0, num_faces);
    }

    int build_recursive(int start, int end) {
        int node_idx = nodes.size();
        nodes.emplace_back();

        AABB bbox;
        AABB centroid_bbox;
        for (int i = start; i < end; ++i) {
            bbox.grow(prims[i].bbox);
            centroid_bbox.grow(prims[i].centroid);
        }
        nodes[node_idx].bbox = bbox;

        int count = end - start;
        if (count <= 4) {
            nodes[node_idx].first_prim = start;
            nodes[node_idx].prim_count = count;
            return node_idx;
        }

        int axis = 0;
        float span = centroid_bbox.max_val[0] - centroid_bbox.min_val[0];
        if (centroid_bbox.max_val[1] - centroid_bbox.min_val[1] > span) {
            axis = 1;
            span = centroid_bbox.max_val[1] - centroid_bbox.min_val[1];
        }
        if (centroid_bbox.max_val[2] - centroid_bbox.min_val[2] > span) {
            axis = 2;
        }

        float split_val = 0.5f * (axis == 0 ? (centroid_bbox.min_val[0] + centroid_bbox.max_val[0]) :
                                  (axis == 1 ? (centroid_bbox.min_val[1] + centroid_bbox.max_val[1]) :
                                               (centroid_bbox.min_val[2] + centroid_bbox.max_val[2])));

        int mid = start;
        for (int i = start; i < end; ++i) {
            float val = (axis == 0 ? prims[i].centroid.x : (axis == 1 ? prims[i].centroid.y : prims[i].centroid.z));
            if (val < split_val) {
                std::swap(prims[i], prims[mid]);
                mid++;
            }
        }

        if (mid == start || mid == end) {
            mid = start + count / 2;
        }

        int left = build_recursive(start, mid);
        int right = build_recursive(mid, end);
        nodes[node_idx].left_child = left;
        nodes[node_idx].right_child = right;
        return node_idx;
    }
};

// BVH collision traversal
void collide_bvhs(const BVH& bvh_A, int node_A, const BVH& bvh_B, int node_B,
                  const std::vector<float>& verts_A, const std::vector<int>& faces_A,
                  const std::vector<float>& verts_B, const std::vector<int>& faces_B,
                  std::vector<std::pair<int, int>>& intersection_pairs) {
    if (node_A < 0 || node_B < 0) return;
    const auto& nA = bvh_A.nodes[node_A];
    const auto& nB = bvh_B.nodes[node_B];

    if (!nA.bbox.intersects(nB.bbox)) return;

    if (nA.prim_count > 0 && nB.prim_count > 0) {
        for (int i = 0; i < nA.prim_count; ++i) {
            int prim_A = bvh_A.prims[nA.first_prim + i].id;
            for (int j = 0; j < nB.prim_count; ++j) {
                int prim_B = bvh_B.prims[nB.first_prim + j].id;
                if (bvh_A.prims[nA.first_prim + i].bbox.intersects(bvh_B.prims[nB.first_prim + j].bbox)) {
                    intersection_pairs.push_back({prim_A, prim_B});
                }
            }
        }
        return;
    }

    if (nA.prim_count > 0) {
        collide_bvhs(bvh_A, node_A, bvh_B, nB.left_child, verts_A, faces_A, verts_B, faces_B, intersection_pairs);
        collide_bvhs(bvh_A, node_A, bvh_B, nB.right_child, verts_A, faces_A, verts_B, faces_B, intersection_pairs);
    } else if (nB.prim_count > 0) {
        collide_bvhs(bvh_A, nA.left_child, bvh_B, node_B, verts_A, faces_A, verts_B, faces_B, intersection_pairs);
        collide_bvhs(bvh_A, nA.right_child, bvh_B, node_B, verts_A, faces_A, verts_B, faces_B, intersection_pairs);
    } else {
        float sizeA = std::max({nA.bbox.max_val[0] - nA.bbox.min_val[0], nA.bbox.max_val[1] - nA.bbox.min_val[1], nA.bbox.max_val[2] - nA.bbox.min_val[2]});
        float sizeB = std::max({nB.bbox.max_val[0] - nB.bbox.min_val[0], nB.bbox.max_val[1] - nB.bbox.min_val[1], nB.bbox.max_val[2] - nB.bbox.min_val[2]});
        if (sizeA > sizeB) {
            collide_bvhs(bvh_A, nA.left_child, bvh_B, node_B, verts_A, faces_A, verts_B, faces_B, intersection_pairs);
            collide_bvhs(bvh_A, nA.right_child, bvh_B, node_B, verts_A, faces_A, verts_B, faces_B, intersection_pairs);
        } else {
            collide_bvhs(bvh_A, node_A, bvh_B, nB.left_child, verts_A, faces_A, verts_B, faces_B, intersection_pairs);
            collide_bvhs(bvh_A, node_A, bvh_B, nB.right_child, verts_A, faces_A, verts_B, faces_B, intersection_pairs);
        }
    }
}

// Ray-AABB intersection
bool ray_intersects_aabb(const float O[3], const float D[3], const AABB& bbox, float& t_min, float& t_max) {
    t_min = 0.0f;
    t_max = std::numeric_limits<float>::infinity();
    for (int i = 0; i < 3; ++i) {
        if (std::abs(D[i]) < 1e-9f) {
            if (O[i] < bbox.min_val[i] || O[i] > bbox.max_val[i]) return false;
        } else {
            float ood = 1.0f / D[i];
            float t1 = (bbox.min_val[i] - O[i]) * ood;
            float t2 = (bbox.max_val[i] - O[i]) * ood;
            if (t1 > t2) std::swap(t1, t2);
            t_min = std::max(t_min, t1);
            t_max = std::min(t_max, t2);
            if (t_min > t_max) return false;
        }
    }
    return true;
}

enum RayResult {
    INTERSECT,
    MISS,
    DEGENERATE
};

// Möller-Trumbore ray-triangle intersection
RayResult ray_intersects_triangle_robust(const float O[3], const float D[3],
                                         const float V0[3], const float V1[3], const float V2[3],
                                         float& t) {
    float edge1[3] = {V1[0] - V0[0], V1[1] - V0[1], V1[2] - V0[2]};
    float edge2[3] = {V2[0] - V0[0], V2[1] - V0[1], V2[2] - V0[2]};
    float h[3] = {D[1] * edge2[2] - D[2] * edge2[1],
                  D[2] * edge2[0] - D[0] * edge2[2],
                  D[0] * edge2[1] - D[1] * edge2[0]};
    float a = edge1[0] * h[0] + edge1[1] * h[1] + edge1[2] * h[2];
    if (std::abs(a) < 1e-9f) return MISS;

    float f = 1.0f / a;
    float s[3] = {O[0] - V0[0], O[1] - V0[1], O[2] - V0[2]};
    float u = f * (s[0] * h[0] + s[1] * h[1] + s[2] * h[2]);
    if (u < -1e-5f || u > 1.00001f) return MISS;

    float q[3] = {s[1] * edge1[2] - s[2] * edge1[1],
                  s[2] * edge1[0] - s[0] * edge1[2],
                  s[0] * edge1[1] - s[1] * edge1[0]};
    float v = f * (D[0] * q[0] + D[1] * q[1] + D[2] * q[2]);
    if (v < -1e-5f || u + v > 1.00001f) return MISS;

    float eps = 1e-3f;
    if (std::abs(u) < eps || std::abs(1.0f - u) < eps ||
        std::abs(v) < eps || std::abs(1.0f - (u + v)) < eps) {
        return DEGENERATE;
    }

    t = f * (edge2[0] * q[0] + edge2[1] * q[1] + edge2[2] * q[2]);
    if (t < eps) return MISS;

    return INTERSECT;
}

bool count_ray_intersections(const BVH& bvh, int node_idx,
                             const std::vector<float>& vertices, const std::vector<int>& faces,
                             const float O[3], const float D[3], int& count) {
    if (node_idx < 0) return true;
    const auto& node = bvh.nodes[node_idx];

    float t_min, t_max;
    if (!ray_intersects_aabb(O, D, node.bbox, t_min, t_max)) {
        return true;
    }

    if (node.prim_count > 0) {
        for (int i = 0; i < node.prim_count; ++i) {
            int prim_id = bvh.prims[node.first_prim + i].id;
            const float* v0 = &vertices[3 * faces[3 * prim_id + 0]];
            const float* v1 = &vertices[3 * faces[3 * prim_id + 1]];
            const float* v2 = &vertices[3 * faces[3 * prim_id + 2]];
            float t;
            RayResult res = ray_intersects_triangle_robust(O, D, v0, v1, v2, t);
            if (res == DEGENERATE) {
                return false;
            } else if (res == INTERSECT) {
                count++;
            }
        }
        return true;
    }

    if (!count_ray_intersections(bvh, node.left_child, vertices, faces, O, D, count)) return false;
    if (!count_ray_intersections(bvh, node.right_child, vertices, faces, O, D, count)) return false;
    return true;
}

bool is_inside_mesh(const BVH& bvh, const std::vector<float>& vertices, const std::vector<int>& faces,
                    const float p[3]) {
    std::mt19937 gen(42);
    std::uniform_real_distribution<float> dis(-1.0f, 1.0f);

    for (int retry = 0; retry < 15; ++retry) {
        float D[3] = {dis(gen), dis(gen), dis(gen)};
        float len = std::sqrt(D[0]*D[0] + D[1]*D[1] + D[2]*D[2]);
        if (len < 1e-4f) continue;
        D[0] /= len; D[1] /= len; D[2] /= len;

        int count = 0;
        if (count_ray_intersections(bvh, bvh.root, vertices, faces, p, D, count)) {
            return (count % 2) == 1;
        }
    }
    return false;
}

// Triangle-triangle intersection segment calculation
int intersect_triangle_plane(const Point3D V[3], const Point3D& n, const Point3D& p0, Point3D pts[2]) {
    float d[3];
    for (int i = 0; i < 3; ++i) {
        d[i] = n.dot(V[i] - p0);
    }

    int num_pts = 0;
    for (int i = 0; i < 3; ++i) {
        int next = (i + 1) % 3;
        float di = d[i];
        float dnext = d[next];
        if ((di > 0.0f && dnext < 0.0f) || (di < 0.0f && dnext > 0.0f)) {
            float t = di / (di - dnext);
            pts[num_pts++] = V[i] + (V[next] - V[i]) * t;
        } else if (std::abs(di) < 1e-9f) {
            pts[num_pts++] = V[i];
        }
        if (num_pts == 2) break;
    }
    if (num_pts == 2 && (pts[0] - pts[1]).norm() < 1e-6f) {
        num_pts = 1;
    }
    return num_pts;
}

bool intersect_triangles(const Point3D A[3], const Point3D B[3], Point3D& out_P1, Point3D& out_P2) {
    Point3D nA = (A[1] - A[0]).cross(A[2] - A[0]);
    float lenA = nA.norm();
    if (lenA < 1e-12f) return false;
    nA = nA * (1.0f / lenA);

    Point3D nB = (B[1] - B[0]).cross(B[2] - B[0]);
    float lenB = nB.norm();
    if (lenB < 1e-12f) return false;
    nB = nB * (1.0f / lenB);

    Point3D D = nA.cross(nB);
    float lenD = D.norm();
    if (lenD < 1e-6f) return false;
    D = D * (1.0f / lenD);

    Point3D ptsB[2];
    int numB = intersect_triangle_plane(B, nA, A[0], ptsB);
    if (numB < 2) return false;

    Point3D ptsA[2];
    int numA = intersect_triangle_plane(A, nB, B[0], ptsA);
    if (numA < 2) return false;

    float b1 = ptsB[0].dot(D);
    float b2 = ptsB[1].dot(D);
    if (b1 > b2) std::swap(b1, b2);

    float a1 = ptsA[0].dot(D);
    float a2 = ptsA[1].dot(D);
    if (a1 > a2) std::swap(a1, a2);

    float u1 = std::max(a1, b1);
    float u2 = std::min(a2, b2);

    if (u1 >= u2 - 1e-5f) return false;

    out_P1 = ptsB[0] + (ptsB[1] - ptsB[0]) * ((u1 - b1) / (b2 - b1 + 1e-12f));
    out_P2 = ptsB[0] + (ptsB[1] - ptsB[0]) * ((u2 - b1) / (b2 - b1 + 1e-12f));
    return true;
}

// 2D projection structure
struct Projection {
    int drop_axis;
    bool flip_y;

    Point2D project(const Point3D& p) const {
        Point2D p2;
        if (drop_axis == 0) p2 = {p.y, p.z};
        else if (drop_axis == 1) p2 = {p.x, p.z};
        else p2 = {p.x, p.y};
        if (flip_y) p2.y = -p2.y;
        return p2;
    }
};

int get_drop_axis(const Point3D& n) {
    float absx = std::abs(n.x);
    float absy = std::abs(n.y);
    float absz = std::abs(n.z);
    if (absx > absy && absx > absz) return 0;
    if (absy > absz) return 1;
    return 2;
}

float signed_area_2d(const Point2D& a, const Point2D& b, const Point2D& c) {
    return 0.5f * ((b.x - a.x) * (c.y - a.y) - (c.x - a.x) * (b.y - a.y));
}

Projection get_projection(const Point3D& V0, const Point3D& V1, const Point3D& V2) {
    Point3D n = (V1 - V0).cross(V2 - V0).normalized();
    int drop_axis = get_drop_axis(n);
    Projection proj;
    proj.drop_axis = drop_axis;
    proj.flip_y = false;
    
    Point2D p0 = proj.project(V0);
    Point2D p1 = proj.project(V1);
    Point2D p2 = proj.project(V2);
    if (signed_area_2d(p0, p1, p2) < 0.0f) {
        proj.flip_y = true;
    }
    return proj;
}

bool point_on_segment_2d(const Point2D& P, const Point2D& A, const Point2D& B, float eps) {
    Point2D AB = {B.x - A.x, B.y - A.y};
    Point2D AP = {P.x - A.x, P.y - A.y};
    float len_sq = AB.x*AB.x + AB.y*AB.y;
    if (len_sq < 1e-12f) return false;
    float t = (AP.x*AB.x + AP.y*AB.y) / len_sq;
    if (t < 1e-4f || t > 1.0f - 1e-4f) return false;
    Point2D projection = {A.x + AB.x * t, A.y + AB.y * t};
    float dx = P.x - projection.x;
    float dy = P.y - projection.y;
    return std::sqrt(dx*dx + dy*dy) < eps;
}


int find_next_half_edge(int u, int v, const std::vector<std::vector<int>>& adj, const std::vector<Point2D>& V2d) {
    const auto& neighbors = adj[v];
    if (neighbors.empty()) return -1;
    if (neighbors.size() == 1) return neighbors[0];
    
    float dx_in = V2d[v].x - V2d[u].x;
    float dy_in = V2d[v].y - V2d[u].y;
    float theta_in = std::atan2(dy_in, dx_in);
    
    float min_diff = 10.0f;
    int best_w = -1;
    for (int w : neighbors) {
        if (w == u) continue;
        float dx_out = V2d[w].x - V2d[v].x;
        float dy_out = V2d[w].y - V2d[v].y;
        float theta_out = std::atan2(dy_out, dx_out);
        
        float diff = theta_out - theta_in;
        while (diff <= 0.0f) diff += 2.0f * 3.1415926535f;
        while (diff > 2.0f * 3.1415926535f) diff -= 2.0f * 3.1415926535f;
        
        if (diff < min_diff) {
            min_diff = diff;
            best_w = w;
        }
    }
    if (best_w == -1) {
        best_w = u;
    }
    return best_w;
}

bool is_convex_angle(const Point2D& a, const Point2D& b, const Point2D& c) {
    return signed_area_2d(a, b, c) > 1e-6f;
}

bool point_in_triangle_2d(const Point2D& p, const Point2D& a, const Point2D& b, const Point2D& c) {
    float s1 = signed_area_2d(a, b, p);
    float s2 = signed_area_2d(b, c, p);
    float s3 = signed_area_2d(c, a, p);
    float eps = -1e-6f;
    return (s1 >= eps && s2 >= eps && s3 >= eps);
}


std::vector<std::vector<int>> triangulate_polygon(const std::vector<int>& indices, const std::vector<Point2D>& V2d) {
    std::vector<std::vector<int>> triangles;
    std::vector<int> poly = indices;

    int safety_counter = 0;
    while (poly.size() > 3 && safety_counter < 1000) {
        safety_counter++;
        bool ear_found = false;
        for (size_t i = 0; i < poly.size(); ++i) {
            int prev = poly[(i + poly.size() - 1) % poly.size()];
            int curr = poly[i];
            int next = poly[(i + 1) % poly.size()];

            if (!is_convex_angle(V2d[prev], V2d[curr], V2d[next])) {
                continue;
            }

            bool has_vertex_inside = false;
            for (size_t j = 0; j < poly.size(); ++j) {
                int v = poly[j];
                if (v == prev || v == curr || v == next) continue;
                if (point_in_triangle_2d(V2d[v], V2d[prev], V2d[curr], V2d[next])) {
                    has_vertex_inside = true;
                    break;
                }
            }

            if (!has_vertex_inside) {
                triangles.push_back({prev, curr, next});
                poly.erase(poly.begin() + i);
                ear_found = true;
                break;
            }
        }
        if (!ear_found) {
            for (size_t i = 0; i < poly.size(); ++i) {
                int prev = poly[(i + poly.size() - 1) % poly.size()];
                int curr = poly[i];
                int next = poly[(i + 1) % poly.size()];
                triangles.push_back({prev, curr, next});
                poly.erase(poly.begin() + i);
                ear_found = true;
                break;
            }
            if (!ear_found) break;
        }
    }
    if (poly.size() == 3) {
        triangles.push_back({poly[0], poly[1], poly[2]});
    }
    return triangles;
}

// Dual Graph Helper structures
struct HalfEdge {
    int u, v;
    int tri_id;
    int local_edge;
};

std::vector<int> build_dual_graph(int num_verts, const std::vector<int>& faces) {
    int num_faces = faces.size() / 3;
    std::vector<HalfEdge> hes;
    hes.reserve(3 * num_faces);
    for (int i = 0; i < num_faces; ++i) {
        for (int j = 0; j < 3; ++j) {
            int u = faces[3 * i + j];
            int v = faces[3 * i + (j + 1) % 3];
            hes.push_back({u, v, i, j});
        }
    }

    std::sort(hes.begin(), hes.end(), [](const HalfEdge& a, const HalfEdge& b) {
        int au = a.u, av = a.v; if (au > av) std::swap(au, av);
        int bu = b.u, bv = b.v; if (bu > bv) std::swap(bu, bv);
        if (au != bu) return au < bu;
        return av < bv;
    });

    std::vector<int> tri_adj(3 * num_faces, -1);
    for (size_t i = 0; i + 1 < hes.size(); ) {
        int au = hes[i].u, av = hes[i].v; if (au > av) std::swap(au, av);
        int bu = hes[i+1].u, bv = hes[i+1].v; if (bu > bv) std::swap(bu, bv);
        if (au == bu && av == bv) {
            int tri1 = hes[i].tri_id;
            int edge1 = hes[i].local_edge;
            int tri2 = hes[i+1].tri_id;
            int edge2 = hes[i+1].local_edge;
            tri_adj[3 * tri1 + edge1] = tri2;
            tri_adj[3 * tri2 + edge2] = tri1;
            i += 2;
        } else {
            i += 1;
        }
    }
    return tri_adj;
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    int VA, FA, VB, FB;
    if (!(std::cin >> VA >> FA >> VB >> FB)) return 0;

    std::vector<float> verts_A(3 * VA);
    for (int i = 0; i < 3 * VA; ++i) std::cin >> verts_A[i];

    std::vector<int> faces_A(3 * FA);
    for (int i = 0; i < 3 * FA; ++i) std::cin >> faces_A[i];

    std::vector<float> verts_B(3 * VB);
    for (int i = 0; i < 3 * VB; ++i) std::cin >> verts_B[i];

    std::vector<int> faces_B(3 * FB);
    for (int i = 0; i < 3 * FB; ++i) std::cin >> faces_B[i];

    // Compute bounding box and apply deterministic tiny perturbation to B to break degeneracies
    AABB total_bbox;
    for (int i = 0; i < VA; ++i) total_bbox.grow(&verts_A[3 * i]);
    for (int i = 0; i < VB; ++i) total_bbox.grow(&verts_B[3 * i]);

    float dx = total_bbox.max_val[0] - total_bbox.min_val[0];
    float dy = total_bbox.max_val[1] - total_bbox.min_val[1];
    float dz = total_bbox.max_val[2] - total_bbox.min_val[2];
    float total_scale = std::sqrt(dx*dx + dy*dy + dz*dz);
    if (total_scale < 1e-6f) total_scale = 1.0f;

    float perturb_scale = total_scale * 1e-8f;
    for (int i = 0; i < VB; ++i) {
        uint32_t x = (uint32_t)i;
        x = ((x >> 16) ^ x) * 0x45d9f3b;
        x = ((x >> 16) ^ x) * 0x45d9f3b;
        x = (x >> 16) ^ x;
        float rx = ((x & 0xFF) / 255.0f) * 2.0f - 1.0f;
        float ry = (((x >> 8) & 0xFF) / 255.0f) * 2.0f - 1.0f;
        float rz = (((x >> 16) & 0xFF) / 255.0f) * 2.0f - 1.0f;
        verts_B[3 * i + 0] += rx * perturb_scale;
        verts_B[3 * i + 1] += ry * perturb_scale;
        verts_B[3 * i + 2] += rz * perturb_scale;
    }

    // Build BVHs
    BVH bvh_A, bvh_B;
    bvh_A.build(verts_A, faces_A);
    bvh_B.build(verts_B, faces_B);

    // Collision detection
    std::vector<std::pair<int, int>> candidate_pairs;
    collide_bvhs(bvh_A, bvh_A.root, bvh_B, bvh_B.root, verts_A, faces_A, verts_B, faces_B, candidate_pairs);

    // Actual triangle-triangle intersection
    std::vector<std::vector<std::pair<Point3D, Point3D>>> intersection_segments_A(FA);
    std::vector<std::vector<std::pair<Point3D, Point3D>>> intersection_segments_B(FB);
    std::vector<bool> intersecting_A(FA, false);
    std::vector<bool> intersecting_B(FB, false);

    for (const auto& pair : candidate_pairs) {
        int triA = pair.first;
        int triB = pair.second;

        Point3D A[3] = {
            {verts_A[3 * faces_A[3 * triA + 0] + 0], verts_A[3 * faces_A[3 * triA + 0] + 1], verts_A[3 * faces_A[3 * triA + 0] + 2]},
            {verts_A[3 * faces_A[3 * triA + 1] + 0], verts_A[3 * faces_A[3 * triA + 1] + 1], verts_A[3 * faces_A[3 * triA + 1] + 2]},
            {verts_A[3 * faces_A[3 * triA + 2] + 0], verts_A[3 * faces_A[3 * triA + 2] + 1], verts_A[3 * faces_A[3 * triA + 2] + 2]}
        };

        Point3D B[3] = {
            {verts_B[3 * faces_B[3 * triB + 0] + 0], verts_B[3 * faces_B[3 * triB + 0] + 1], verts_B[3 * faces_B[3 * triB + 0] + 2]},
            {verts_B[3 * faces_B[3 * triB + 1] + 0], verts_B[3 * faces_B[3 * triB + 1] + 1], verts_B[3 * faces_B[3 * triB + 1] + 2]},
            {verts_B[3 * faces_B[3 * triB + 2] + 0], verts_B[3 * faces_B[3 * triB + 2] + 1], verts_B[3 * faces_B[3 * triB + 2] + 2]}
        };

        Point3D P1, P2;
        if (intersect_triangles(A, B, P1, P2)) {
            intersection_segments_A[triA].push_back({P1, P2});
            intersection_segments_B[triB].push_back({P1, P2});
            intersecting_A[triA] = true;
            intersecting_B[triB] = true;
        }
    }

    // Build dual graphs and classify connected components of non-intersecting faces
    std::vector<int> tri_adj_A = build_dual_graph(VA, faces_A);
    std::vector<int> component_id_A(FA, -1);
    std::vector<bool> component_outside_A;
    int comp_count_A = 0;

    for (int i = 0; i < FA; ++i) {
        if (intersecting_A[i] || component_id_A[i] != -1) continue;

        std::vector<int> q;
        q.push_back(i);
        component_id_A[i] = comp_count_A;
        size_t head = 0;
        while (head < q.size()) {
            int curr = q[head++];
            for (int j = 0; j < 3; ++j) {
                int neighbor = tri_adj_A[3 * curr + j];
                if (neighbor != -1 && !intersecting_A[neighbor] && component_id_A[neighbor] == -1) {
                    component_id_A[neighbor] = comp_count_A;
                    q.push_back(neighbor);
                }
            }
        }

        int rep_tri = i;
        float centroid[3] = {
            (verts_A[3 * faces_A[3 * rep_tri + 0] + 0] + verts_A[3 * faces_A[3 * rep_tri + 1] + 0] + verts_A[3 * faces_A[3 * rep_tri + 2] + 0]) / 3.0f,
            (verts_A[3 * faces_A[3 * rep_tri + 0] + 1] + verts_A[3 * faces_A[3 * rep_tri + 1] + 1] + verts_A[3 * faces_A[3 * rep_tri + 2] + 1]) / 3.0f,
            (verts_A[3 * faces_A[3 * rep_tri + 0] + 2] + verts_A[3 * faces_A[3 * rep_tri + 1] + 2] + verts_A[3 * faces_A[3 * rep_tri + 2] + 2]) / 3.0f
        };

        // Normal of representative triangle
        Point3D v0{verts_A[3 * faces_A[3 * rep_tri + 0] + 0], verts_A[3 * faces_A[3 * rep_tri + 0] + 1], verts_A[3 * faces_A[3 * rep_tri + 0] + 2]};
        Point3D v1{verts_A[3 * faces_A[3 * rep_tri + 1] + 0], verts_A[3 * faces_A[3 * rep_tri + 1] + 1], verts_A[3 * faces_A[3 * rep_tri + 1] + 2]};
        Point3D v2{verts_A[3 * faces_A[3 * rep_tri + 2] + 0], verts_A[3 * faces_A[3 * rep_tri + 2] + 1], verts_A[3 * faces_A[3 * rep_tri + 2] + 2]};
        Point3D normal = (v1 - v0).cross(v2 - v0).normalized();

        float perturbed_point[3] = {
            centroid[0] + normal.x * total_scale * 1e-4f,
            centroid[1] + normal.y * total_scale * 1e-4f,
            centroid[2] + normal.z * total_scale * 1e-4f
        };

        bool outside = !is_inside_mesh(bvh_B, verts_B, faces_B, perturbed_point);
        component_outside_A.push_back(outside);
        comp_count_A++;
    }

    std::vector<int> tri_adj_B = build_dual_graph(VB, faces_B);
    std::vector<int> component_id_B(FB, -1);
    std::vector<bool> component_outside_B;
    int comp_count_B = 0;

    for (int i = 0; i < FB; ++i) {
        if (intersecting_B[i] || component_id_B[i] != -1) continue;

        std::vector<int> q;
        q.push_back(i);
        component_id_B[i] = comp_count_B;
        size_t head = 0;
        while (head < q.size()) {
            int curr = q[head++];
            for (int j = 0; j < 3; ++j) {
                int neighbor = tri_adj_B[3 * curr + j];
                if (neighbor != -1 && !intersecting_B[neighbor] && component_id_B[neighbor] == -1) {
                    component_id_B[neighbor] = comp_count_B;
                    q.push_back(neighbor);
                }
            }
        }

        int rep_tri = i;
        float centroid[3] = {
            (verts_B[3 * faces_B[3 * rep_tri + 0] + 0] + verts_B[3 * faces_B[3 * rep_tri + 1] + 0] + verts_B[3 * faces_B[3 * rep_tri + 2] + 0]) / 3.0f,
            (verts_B[3 * faces_B[3 * rep_tri + 0] + 1] + verts_B[3 * faces_B[3 * rep_tri + 1] + 1] + verts_B[3 * faces_B[3 * rep_tri + 2] + 1]) / 3.0f,
            (verts_B[3 * faces_B[3 * rep_tri + 0] + 2] + verts_B[3 * faces_B[3 * rep_tri + 1] + 2] + verts_B[3 * faces_B[3 * rep_tri + 2] + 2]) / 3.0f
        };

        Point3D v0{verts_B[3 * faces_B[3 * rep_tri + 0] + 0], verts_B[3 * faces_B[3 * rep_tri + 0] + 1], verts_B[3 * faces_B[3 * rep_tri + 0] + 2]};
        Point3D v1{verts_B[3 * faces_B[3 * rep_tri + 1] + 0], verts_B[3 * faces_B[3 * rep_tri + 1] + 1], verts_B[3 * faces_B[3 * rep_tri + 1] + 2]};
        Point3D v2{verts_B[3 * faces_B[3 * rep_tri + 2] + 0], verts_B[3 * faces_B[3 * rep_tri + 2] + 1], verts_B[3 * faces_B[3 * rep_tri + 2] + 2]};
        Point3D normal = (v1 - v0).cross(v2 - v0).normalized();

        float perturbed_point[3] = {
            centroid[0] + normal.x * total_scale * 1e-4f,
            centroid[1] + normal.y * total_scale * 1e-4f,
            centroid[2] + normal.z * total_scale * 1e-4f
        };

        bool outside = !is_inside_mesh(bvh_A, verts_A, faces_A, perturbed_point);
        component_outside_B.push_back(outside);
        comp_count_B++;
    }

    // Assembly output structures
    std::vector<Point3D> out_raw_verts;
    std::vector<int> out_raw_faces;

    auto add_triangle = [&](const Point3D& p0, const Point3D& p1, const Point3D& p2) {
        int base = out_raw_verts.size();
        out_raw_verts.push_back(p0);
        out_raw_verts.push_back(p1);
        out_raw_verts.push_back(p2);
        out_raw_faces.push_back(base);
        out_raw_faces.push_back(base + 1);
        out_raw_faces.push_back(base + 2);
    };

    // Keep outside non-intersecting faces of A
    for (int i = 0; i < FA; ++i) {
        if (!intersecting_A[i]) {
            int comp = component_id_A[i];
            if (comp != -1 && component_outside_A[comp]) {
                Point3D p0{verts_A[3 * faces_A[3 * i + 0] + 0], verts_A[3 * faces_A[3 * i + 0] + 1], verts_A[3 * faces_A[3 * i + 0] + 2]};
                Point3D p1{verts_A[3 * faces_A[3 * i + 1] + 0], verts_A[3 * faces_A[3 * i + 1] + 1], verts_A[3 * faces_A[3 * i + 1] + 2]};
                Point3D p2{verts_A[3 * faces_A[3 * i + 2] + 0], verts_A[3 * faces_A[3 * i + 2] + 1], verts_A[3 * faces_A[3 * i + 2] + 2]};
                add_triangle(p0, p1, p2);
            }
        }
    }

    // Keep outside non-intersecting faces of B
    for (int i = 0; i < FB; ++i) {
        if (!intersecting_B[i]) {
            int comp = component_id_B[i];
            if (comp != -1 && component_outside_B[comp]) {
                Point3D p0{verts_B[3 * faces_B[3 * i + 0] + 0], verts_B[3 * faces_B[3 * i + 0] + 1], verts_B[3 * faces_B[3 * i + 0] + 2]};
                Point3D p1{verts_B[3 * faces_B[3 * i + 1] + 0], verts_B[3 * faces_B[3 * i + 1] + 1], verts_B[3 * faces_B[3 * i + 1] + 2]};
                Point3D p2{verts_B[3 * faces_B[3 * i + 2] + 0], verts_B[3 * faces_B[3 * i + 2] + 1], verts_B[3 * faces_B[3 * i + 2] + 2]};
                add_triangle(p0, p1, p2);
            }
        }
    }

    // Subdivide and classify intersecting faces of A
    for (int i = 0; i < FA; ++i) {
        if (!intersecting_A[i]) continue;

        Point3D T0{verts_A[3 * faces_A[3 * i + 0] + 0], verts_A[3 * faces_A[3 * i + 0] + 1], verts_A[3 * faces_A[3 * i + 0] + 2]};
        Point3D T1{verts_A[3 * faces_A[3 * i + 1] + 0], verts_A[3 * faces_A[3 * i + 1] + 1], verts_A[3 * faces_A[3 * i + 1] + 2]};
        Point3D T2{verts_A[3 * faces_A[3 * i + 2] + 0], verts_A[3 * faces_A[3 * i + 2] + 1], verts_A[3 * faces_A[3 * i + 2] + 2]};

        float triangle_scale = std::max({(T1 - T0).norm(), (T2 - T1).norm(), (T0 - T2).norm()});
        float eps_local = 1e-5f * triangle_scale;

        std::vector<Point3D> V_list = {T0, T1, T2};
        for (const auto& seg : intersection_segments_A[i]) {
            V_list.push_back(seg.first);
            V_list.push_back(seg.second);
        }

        std::vector<Point3D> unique_verts = {T0, T1, T2};
        std::vector<int> vertex_map(V_list.size());
        vertex_map[0] = 0; vertex_map[1] = 1; vertex_map[2] = 2;

        for (size_t k = 3; k < V_list.size(); ++k) {
            int target = -1;
            for (size_t j = 0; j < unique_verts.size(); ++j) {
                if ((V_list[k] - unique_verts[j]).norm() < eps_local) {
                    target = j;
                    break;
                }
            }
            if (target != -1) {
                vertex_map[k] = target;
            } else {
                vertex_map[k] = unique_verts.size();
                unique_verts.push_back(V_list[k]);
            }
        }

        std::vector<std::pair<int, int>> unique_segs;
        for (size_t k = 0; k < intersection_segments_A[i].size(); ++k) {
            int u = vertex_map[3 + 2 * k];
            int v = vertex_map[3 + 2 * k + 1];
            if (u != v) {
                if (u > v) std::swap(u, v);
                unique_segs.push_back({u, v});
            }
        }
        std::sort(unique_segs.begin(), unique_segs.end());
        unique_segs.erase(std::unique(unique_segs.begin(), unique_segs.end()), unique_segs.end());

        Projection proj = get_projection(unique_verts[0], unique_verts[1], unique_verts[2]);
        std::vector<Point2D> unique_verts_2d(unique_verts.size());
        for (size_t k = 0; k < unique_verts.size(); ++k) {
            unique_verts_2d[k] = proj.project(unique_verts[k]);
        }

        std::vector<int> split_pts[3];
        float eps_2d = 1e-4f;
        for (size_t k = 3; k < unique_verts.size(); ++k) {
            if (point_on_segment_2d(unique_verts_2d[k], unique_verts_2d[0], unique_verts_2d[1], eps_2d)) {
                split_pts[0].push_back(k);
            } else if (point_on_segment_2d(unique_verts_2d[k], unique_verts_2d[1], unique_verts_2d[2], eps_2d)) {
                split_pts[1].push_back(k);
            } else if (point_on_segment_2d(unique_verts_2d[k], unique_verts_2d[2], unique_verts_2d[0], eps_2d)) {
                split_pts[2].push_back(k);
            }
        }

        std::sort(split_pts[0].begin(), split_pts[0].end(), [&](int a, int b) {
            return (unique_verts[a] - unique_verts[0]).norm() < (unique_verts[b] - unique_verts[0]).norm();
        });
        std::sort(split_pts[1].begin(), split_pts[1].end(), [&](int a, int b) {
            return (unique_verts[a] - unique_verts[1]).norm() < (unique_verts[b] - unique_verts[1]).norm();
        });
        std::sort(split_pts[2].begin(), split_pts[2].end(), [&](int a, int b) {
            return (unique_verts[a] - unique_verts[2]).norm() < (unique_verts[b] - unique_verts[2]).norm();
        });

        std::vector<std::pair<int, int>> pslg_edges = unique_segs;
        auto add_boundary_edges = [&](int start, int end, const std::vector<int>& splits) {
            int prev = start;
            for (int pt : splits) {
                int u = prev, v = pt;
                if (u > v) std::swap(u, v);
                pslg_edges.push_back({u, v});
                prev = pt;
            }
            int u = prev, v = end;
            if (u > v) std::swap(u, v);
            pslg_edges.push_back({u, v});
        };

        add_boundary_edges(0, 1, split_pts[0]);
        add_boundary_edges(1, 2, split_pts[1]);
        add_boundary_edges(2, 0, split_pts[2]);

        std::sort(pslg_edges.begin(), pslg_edges.end());
        pslg_edges.erase(std::unique(pslg_edges.begin(), pslg_edges.end()), pslg_edges.end());

        std::vector<std::vector<int>> adj(unique_verts.size());
        for (const auto& edge : pslg_edges) {
            adj[edge.first].push_back(edge.second);
            adj[edge.second].push_back(edge.first);
        }

        std::vector<std::vector<bool>> visited(unique_verts.size());
        for (size_t k = 0; k < unique_verts.size(); ++k) {
            visited[k].resize(adj[k].size(), false);
        }

        Point3D normal = (T1 - T0).cross(T2 - T0).normalized();

        for (size_t u = 0; u < unique_verts.size(); ++u) {
            for (size_t k = 0; k < adj[u].size(); ++k) {
                if (visited[u][k]) continue;

                std::vector<int> loop;
                int curr_u = u;
                int curr_v = adj[u][k];
                int curr_k = k;

                bool success = true;
                int safety_loop = 0;
                while (!visited[curr_u][curr_k] && safety_loop < 1000) {
                    safety_loop++;
                    visited[curr_u][curr_k] = true;
                    loop.push_back(curr_u);

                    int next_w = find_next_half_edge(curr_u, curr_v, adj, unique_verts_2d);
                    if (next_w == -1) {
                        success = false;
                        break;
                    }

                    auto it = std::find(adj[curr_v].begin(), adj[curr_v].end(), next_w);
                    if (it == adj[curr_v].end()) {
                        success = false;
                        break;
                    }
                    int next_k = std::distance(adj[curr_v].begin(), it);

                    curr_u = curr_v;
                    curr_v = next_w;
                    curr_k = next_k;
                }

                if (success && loop.size() >= 3) {
                    float area = 0.0f;
                    for (size_t idx = 0; idx < loop.size(); ++idx) {
                        int i0 = loop[idx];
                        int i1 = loop[(idx + 1) % loop.size()];
                        area += (unique_verts_2d[i0].x * unique_verts_2d[i1].y - unique_verts_2d[i1].x * unique_verts_2d[i0].y);
                    }
                    area *= 0.5f;

                    if (area > 1e-6f) {
                        auto sub_tris = triangulate_polygon(loop, unique_verts_2d);
                        for (const auto& st : sub_tris) {
                            Point3D p0 = unique_verts[st[0]];
                            Point3D p1 = unique_verts[st[1]];
                            Point3D p2 = unique_verts[st[2]];

                            bool m1 = (std::abs(p0.x - 0.5f) < 1e-3f && std::abs(p0.y - 0.5f) < 1e-3f && std::abs(p0.z - 0.5f) < 1e-3f) ||
                                      (std::abs(p0.x - 0.0f) < 1e-3f && std::abs(p0.y - 0.5f) < 1e-3f && std::abs(p0.z - 0.5f) < 1e-3f) ||
                                      (std::abs(p0.x + 0.5f) < 1e-3f && std::abs(p0.y - 0.5f) < 1e-3f && std::abs(p0.z - 0.5f) < 1e-3f);
                            bool m2 = (std::abs(p1.x - 0.5f) < 1e-3f && std::abs(p1.y - 0.5f) < 1e-3f && std::abs(p1.z - 0.5f) < 1e-3f) ||
                                      (std::abs(p1.x - 0.0f) < 1e-3f && std::abs(p1.y - 0.5f) < 1e-3f && std::abs(p1.z - 0.5f) < 1e-3f) ||
                                      (std::abs(p1.x + 0.5f) < 1e-3f && std::abs(p1.y - 0.5f) < 1e-3f && std::abs(p1.z - 0.5f) < 1e-3f);
                            bool m3 = (std::abs(p2.x - 0.5f) < 1e-3f && std::abs(p2.y - 0.5f) < 1e-3f && std::abs(p2.z - 0.5f) < 1e-3f) ||
                                      (std::abs(p2.x - 0.0f) < 1e-3f && std::abs(p2.y - 0.5f) < 1e-3f && std::abs(p2.z - 0.5f) < 1e-3f) ||
                                      (std::abs(p2.x + 0.5f) < 1e-3f && std::abs(p2.y - 0.5f) < 1e-3f && std::abs(p2.z - 0.5f) < 1e-3f);
                            if (m1 && m2 && m3) {
                                std::cout << "FOUND DEG TRIANGLE IN A! Loop: ";
                                for (int idx : loop) {
                                    std::cout << idx << " (" << unique_verts_2d[idx].x << "," << unique_verts_2d[idx].y << ") ";
                                }
                                std::cout << "\n";
                            }

                            float centroid[3] = {
                                (p0.x + p1.x + p2.x) / 3.0f,
                                (p0.y + p1.y + p2.y) / 3.0f,
                                (p0.z + p1.z + p2.z) / 3.0f
                            };

                            float test_pt[3] = {
                                centroid[0] + normal.x * total_scale * 1e-4f,
                                centroid[1] + normal.y * total_scale * 1e-4f,
                                centroid[2] + normal.z * total_scale * 1e-4f
                            };

                            if (!is_inside_mesh(bvh_B, verts_B, faces_B, test_pt)) {
                                add_triangle(p0, p1, p2);
                            }
                        }
                    }
                }
            }
        }
    }

    // Subdivide and classify intersecting faces of B
    for (int i = 0; i < FB; ++i) {
        if (!intersecting_B[i]) continue;

        Point3D T0{verts_B[3 * faces_B[3 * i + 0] + 0], verts_B[3 * faces_B[3 * i + 0] + 1], verts_B[3 * faces_B[3 * i + 0] + 2]};
        Point3D T1{verts_B[3 * faces_B[3 * i + 1] + 0], verts_B[3 * faces_B[3 * i + 1] + 1], verts_B[3 * faces_B[3 * i + 1] + 2]};
        Point3D T2{verts_B[3 * faces_B[3 * i + 2] + 0], verts_B[3 * faces_B[3 * i + 2] + 1], verts_B[3 * faces_B[3 * i + 2] + 2]};

        float triangle_scale = std::max({(T1 - T0).norm(), (T2 - T1).norm(), (T0 - T2).norm()});
        float eps_local = 1e-5f * triangle_scale;

        std::vector<Point3D> V_list = {T0, T1, T2};
        for (const auto& seg : intersection_segments_B[i]) {
            V_list.push_back(seg.first);
            V_list.push_back(seg.second);
        }

        std::vector<Point3D> unique_verts = {T0, T1, T2};
        std::vector<int> vertex_map(V_list.size());
        vertex_map[0] = 0; vertex_map[1] = 1; vertex_map[2] = 2;

        for (size_t k = 3; k < V_list.size(); ++k) {
            int target = -1;
            for (size_t j = 0; j < unique_verts.size(); ++j) {
                if ((V_list[k] - unique_verts[j]).norm() < eps_local) {
                    target = j;
                    break;
                }
            }
            if (target != -1) {
                vertex_map[k] = target;
            } else {
                vertex_map[k] = unique_verts.size();
                unique_verts.push_back(V_list[k]);
            }
        }

        std::vector<std::pair<int, int>> unique_segs;
        for (size_t k = 0; k < intersection_segments_B[i].size(); ++k) {
            int u = vertex_map[3 + 2 * k];
            int v = vertex_map[3 + 2 * k + 1];
            if (u != v) {
                if (u > v) std::swap(u, v);
                unique_segs.push_back({u, v});
            }
        }
        std::sort(unique_segs.begin(), unique_segs.end());
        unique_segs.erase(std::unique(unique_segs.begin(), unique_segs.end()), unique_segs.end());

        Projection proj = get_projection(unique_verts[0], unique_verts[1], unique_verts[2]);
        std::vector<Point2D> unique_verts_2d(unique_verts.size());
        for (size_t k = 0; k < unique_verts.size(); ++k) {
            unique_verts_2d[k] = proj.project(unique_verts[k]);
        }

        std::vector<int> split_pts[3];
        float eps_2d = 1e-4f;
        for (size_t k = 3; k < unique_verts.size(); ++k) {
            if (point_on_segment_2d(unique_verts_2d[k], unique_verts_2d[0], unique_verts_2d[1], eps_2d)) {
                split_pts[0].push_back(k);
            } else if (point_on_segment_2d(unique_verts_2d[k], unique_verts_2d[1], unique_verts_2d[2], eps_2d)) {
                split_pts[1].push_back(k);
            } else if (point_on_segment_2d(unique_verts_2d[k], unique_verts_2d[2], unique_verts_2d[0], eps_2d)) {
                split_pts[2].push_back(k);
            }
        }

        std::sort(split_pts[0].begin(), split_pts[0].end(), [&](int a, int b) {
            return (unique_verts[a] - unique_verts[0]).norm() < (unique_verts[b] - unique_verts[0]).norm();
        });
        std::sort(split_pts[1].begin(), split_pts[1].end(), [&](int a, int b) {
            return (unique_verts[a] - unique_verts[1]).norm() < (unique_verts[b] - unique_verts[1]).norm();
        });
        std::sort(split_pts[2].begin(), split_pts[2].end(), [&](int a, int b) {
            return (unique_verts[a] - unique_verts[2]).norm() < (unique_verts[b] - unique_verts[2]).norm();
        });

        std::vector<std::pair<int, int>> pslg_edges = unique_segs;
        auto add_boundary_edges = [&](int start, int end, const std::vector<int>& splits) {
            int prev = start;
            for (int pt : splits) {
                int u = prev, v = pt;
                if (u > v) std::swap(u, v);
                pslg_edges.push_back({u, v});
                prev = pt;
            }
            int u = prev, v = end;
            if (u > v) std::swap(u, v);
            pslg_edges.push_back({u, v});
        };

        add_boundary_edges(0, 1, split_pts[0]);
        add_boundary_edges(1, 2, split_pts[1]);
        add_boundary_edges(2, 0, split_pts[2]);

        std::sort(pslg_edges.begin(), pslg_edges.end());
        pslg_edges.erase(std::unique(pslg_edges.begin(), pslg_edges.end()), pslg_edges.end());

        std::vector<std::vector<int>> adj(unique_verts.size());
        for (const auto& edge : pslg_edges) {
            adj[edge.first].push_back(edge.second);
            adj[edge.second].push_back(edge.first);
        }

        std::vector<std::vector<bool>> visited(unique_verts.size());
        for (size_t k = 0; k < unique_verts.size(); ++k) {
            visited[k].resize(adj[k].size(), false);
        }

        Point3D normal = (T1 - T0).cross(T2 - T0).normalized();

        for (size_t u = 0; u < unique_verts.size(); ++u) {
            for (size_t k = 0; k < adj[u].size(); ++k) {
                if (visited[u][k]) continue;

                std::vector<int> loop;
                int curr_u = u;
                int curr_v = adj[u][k];
                int curr_k = k;

                bool success = true;
                int safety_loop = 0;
                while (!visited[curr_u][curr_k] && safety_loop < 1000) {
                    safety_loop++;
                    visited[curr_u][curr_k] = true;
                    loop.push_back(curr_u);

                    int next_w = find_next_half_edge(curr_u, curr_v, adj, unique_verts_2d);
                    if (next_w == -1) {
                        success = false;
                        break;
                    }

                    auto it = std::find(adj[curr_v].begin(), adj[curr_v].end(), next_w);
                    if (it == adj[curr_v].end()) {
                        success = false;
                        break;
                    }
                    int next_k = std::distance(adj[curr_v].begin(), it);

                    curr_u = curr_v;
                    curr_v = next_w;
                    curr_k = next_k;
                }

                if (success && loop.size() >= 3) {
                    float area = 0.0f;
                    for (size_t idx = 0; idx < loop.size(); ++idx) {
                        int i0 = loop[idx];
                        int i1 = loop[(idx + 1) % loop.size()];
                        area += (unique_verts_2d[i0].x * unique_verts_2d[i1].y - unique_verts_2d[i1].x * unique_verts_2d[i0].y);
                    }
                    area *= 0.5f;

                    if (area > 1e-6f) {
                        auto sub_tris = triangulate_polygon(loop, unique_verts_2d);
                        for (const auto& st : sub_tris) {
                            Point3D p0 = unique_verts[st[0]];
                            Point3D p1 = unique_verts[st[1]];
                            Point3D p2 = unique_verts[st[2]];

                            float centroid[3] = {
                                (p0.x + p1.x + p2.x) / 3.0f,
                                (p0.y + p1.y + p2.y) / 3.0f,
                                (p0.z + p1.z + p2.z) / 3.0f
                            };

                            float test_pt[3] = {
                                centroid[0] + normal.x * total_scale * 1e-4f,
                                centroid[1] + normal.y * total_scale * 1e-4f,
                                centroid[2] + normal.z * total_scale * 1e-4f
                            };

                            if (!is_inside_mesh(bvh_A, verts_A, faces_A, test_pt)) {
                                add_triangle(p0, p1, p2);
                            }
                        }
                    }
                }
            }
        }
    }

    // Deduplicate output vertices
    int out_n = out_raw_verts.size();
    std::vector<int> sorted_indices(out_n);
    for (int i = 0; i < out_n; ++i) sorted_indices[i] = i;

    std::sort(sorted_indices.begin(), sorted_indices.end(), [&](int a, int b) {
        if (out_raw_verts[a].x != out_raw_verts[b].x) return out_raw_verts[a].x < out_raw_verts[b].x;
        if (out_raw_verts[a].y != out_raw_verts[b].y) return out_raw_verts[a].y < out_raw_verts[b].y;
        return out_raw_verts[a].z < out_raw_verts[b].z;
    });

    std::vector<int> merge_map(out_n, -1);
    std::vector<Point3D> out_verts;
    float final_eps = total_scale * 1e-5f;

    for (int i = 0; i < out_n; ++i) {
        int idx = sorted_indices[i];
        if (merge_map[idx] != -1) continue;

        int rep_idx = out_verts.size();
        out_verts.push_back(out_raw_verts[idx]);
        merge_map[idx] = rep_idx;

        for (int j = i + 1; j < out_n; ++j) {
            int comp_idx = sorted_indices[j];
            if (out_raw_verts[comp_idx].x - out_raw_verts[idx].x > final_eps) {
                break;
            }
            if (merge_map[comp_idx] == -1 && (out_raw_verts[comp_idx] - out_raw_verts[idx]).norm() < final_eps) {
                merge_map[comp_idx] = rep_idx;
            }
        }
    }

    std::vector<int> out_faces;
    out_faces.reserve(out_raw_faces.size());
    for (size_t i = 0; i < out_raw_faces.size() / 3; ++i) {
        int v0 = merge_map[out_raw_faces[3 * i + 0]];
        int v1 = merge_map[out_raw_faces[3 * i + 1]];
        int v2 = merge_map[out_raw_faces[3 * i + 2]];
        if (v0 != v1 && v1 != v2 && v2 != v0) {
            out_faces.push_back(v0);
            out_faces.push_back(v1);
            out_faces.push_back(v2);
        }
    }

    std::cout << out_verts.size() << " " << out_faces.size() / 3 << "\n";
    for (const auto& v : out_verts) {
        std::cout << v.x << " " << v.y << " " << v.z << "\n";
    }
    for (size_t i = 0; i < out_faces.size() / 3; ++i) {
        std::cout << out_faces[3 * i + 0] << " " << out_faces[3 * i + 1] << " " << out_faces[3 * i + 2] << "\n";
    }

    return 0;
}
