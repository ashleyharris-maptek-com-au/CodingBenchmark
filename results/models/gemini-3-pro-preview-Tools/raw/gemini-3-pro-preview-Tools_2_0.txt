using System;
using System.IO;

public class Solver
{
    // Adjacency list representation using flat arrays
    static int[] head;
    static int[] to;
    static int[] next;
    static int[] edgeId;
    static int edgeCount;

    // Edge weights and traversal counts
    static int[] weight;
    static int[] traversalCounts;

    // Degrees of vertices
    static int[] degree;

    // For Dijkstra
    static int[] distGen;
    static long[] dist;
    static int[] parent;
    static int[] parentEdgeId;
    static int currentGen;

    // Priority Queue
    static PriorityQueue pq;

    public class PriorityQueue
    {
        private struct Node
        {
            public int Vertex;
            public long Distance;
        }

        private Node[] heap;
        private int size;

        public PriorityQueue(int capacity)
        {
            heap = new Node[capacity];
            size = 0;
        }

        public int Count => size;

        public void Clear()
        {
            size = 0;
        }

        public void Enqueue(int vertex, long distance)
        {
            if (size == heap.Length)
            {
                Array.Resize(ref heap, heap.Length * 2);
            }
            heap[size] = new Node { Vertex = vertex, Distance = distance };
            UpHeap(size);
            size++;
        }

        public int Dequeue(out long distance)
        {
            int res = heap[0].Vertex;
            distance = heap[0].Distance;
            size--;
            if (size > 0)
            {
                heap[0] = heap[size];
                DownHeap(0);
            }
            return res;
        }

        private void UpHeap(int i)
        {
            var node = heap[i];
            while (i > 0)
            {
                int p = (i - 1) / 2;
                if (heap[p].Distance <= node.Distance) break;
                heap[i] = heap[p];
                i = p;
            }
            heap[i] = node;
        }

        private void DownHeap(int i)
        {
            var node = heap[i];
            while (i * 2 + 1 < size)
            {
                int left = i * 2 + 1;
                int right = left + 1;
                int child = left;
                if (right < size && heap[right].Distance < heap[left].Distance)
                {
                    child = right;
                }
                if (heap[child].Distance >= node.Distance) break;
                heap[i] = heap[child];
                i = child;
            }
            heap[i] = node;
        }
    }

    public class FastScanner
    {
        private readonly Stream stream;
        private readonly byte[] buffer = new byte[1024 * 1024];
        private int bufferLength = 0;
        private int bufferPointer = 0;

        public FastScanner(Stream stream)
        {
            this.stream = stream;
        }

        private byte Read()
        {
            if (bufferPointer >= bufferLength)
            {
                bufferLength = stream.Read(buffer, 0, buffer.Length);
                bufferPointer = 0;
                if (bufferLength <= 0) return 0;
            }
            return buffer[bufferPointer++];
        }

        public int NextInt()
        {
            int res = 0;
            byte b = Read();
            while (b <= 32)
            {
                if (b == 0) return -1; // EOF
                b = Read();
            }
            bool neg = false;
            if (b == '-')
            {
                neg = true;
                b = Read();
            }
            while (b > 32)
            {
                res = res * 10 + (b - '0');
                b = Read();
            }
            return neg ? -res : res;
        }
    }

    public class FastWriter
    {
        private readonly Stream stream;
        private readonly byte[] buffer = new byte[1024 * 1024];
        private int ptr = 0;

        public FastWriter(Stream stream)
        {
            this.stream = stream;
        }

        public void WriteInt(int val)
        {
            if (val == 0)
            {
                WriteByte((byte)'0');
                return;
            }
            if (val < 0)
            {
                WriteByte((byte)'-');
                val = -val;
            }
            int len = 0;
            byte[] temp = new byte[12];
            while (val > 0)
            {
                temp[len++] = (byte)('0' + (val % 10));
                val /= 10;
            }
            for (int i = len - 1; i >= 0; i--)
            {
                WriteByte(temp[i]);
            }
        }

        public void WriteByte(byte b)
        {
            if (ptr >= buffer.Length)
            {
                Flush();
            }
            buffer[ptr++] = b;
        }

        public void Flush()
        {
            if (ptr > 0)
            {
                stream.Write(buffer, 0, ptr);
                ptr = 0;
            }
        }
    }

    public static void Main()
    {
        FastScanner scanner = new FastScanner(Console.OpenStandardInput());
        int N = scanner.NextInt();
        if (N == -1) return;
        int M = scanner.NextInt();

        head = new int[N];
        for (int i = 0; i < N; i++) head[i] = -1;
        to = new int[2 * M];
        next = new int[2 * M];
        edgeId = new int[2 * M];
        edgeCount = 0;

        weight = new int[M];
        traversalCounts = new int[M];
        for (int i = 0; i < M; i++) traversalCounts[i] = 1;
        degree = new int[N];

        for (int i = 0; i < M; i++)
        {
            int u = scanner.NextInt();
            int v = scanner.NextInt();
            int w = scanner.NextInt();
            weight[i] = w;

            // Add edge u -> v
            to[edgeCount] = v;
            edgeId[edgeCount] = i;
            next[edgeCount] = head[u];
            head[u] = edgeCount++;

            // Add edge v -> u
            to[edgeCount] = u;
            edgeId[edgeCount] = i;
            next[edgeCount] = head[v];
            head[v] = edgeCount++;

            degree[u]++;
            degree[v]++;
        }

        MatchOddVertices(N, M);

        int routeLength;
        int[] route = Hierholzer(N, M, out routeLength);

        FastWriter writer = new FastWriter(Console.OpenStandardOutput());
        for (int i = routeLength - 1; i >= 0; i--)
        {
            writer.WriteInt(route[i]);
            if (i > 0)
            {
                writer.WriteByte((byte)' ');
            }
        }
        writer.WriteByte((byte)'\n');
        writer.Flush();
    }

    static void MatchOddVertices(int N, int M)
    {
        bool[] isOdd = new bool[N];
        bool[] matched = new bool[N];
        int oddCount = 0;
        for (int i = 0; i < N; i++)
        {
            if (degree[i] % 2 != 0)
            {
                isOdd[i] = true;
                oddCount++;
            }
        }

        if (oddCount == 0) return;

        distGen = new int[N];
        dist = new long[N];
        parent = new int[N];
        parentEdgeId = new int[N];
        currentGen = 0;

        pq = new PriorityQueue(1024);

        for (int u = 0; u < N; u++)
        {
            if (isOdd[u] && !matched[u])
            {
                currentGen++;
                dist[u] = 0;
                distGen[u] = currentGen;
                parent[u] = -1;
                pq.Clear();
                pq.Enqueue(u, 0);

                int matchedVertex = -1;

                while (pq.Count > 0)
                {
                    long d;
                    int curr = pq.Dequeue(out d);

                    long currentDist = (distGen[curr] == currentGen) ? dist[curr] : long.MaxValue;
                    if (d > currentDist) continue;

                    if (curr != u && isOdd[curr] && !matched[curr])
                    {
                        matchedVertex = curr;
                        break;
                    }

                    for (int eIdx = head[curr]; eIdx != -1; eIdx = next[eIdx])
                    {
                        int toNode = to[eIdx];
                        int eId = edgeId[eIdx];
                        int w = weight[eId];
                        long newDist = d + w;

                        long oldDist = (distGen[toNode] == currentGen) ? dist[toNode] : long.MaxValue;
                        if (newDist < oldDist)
                        {
                            dist[toNode] = newDist;
                            distGen[toNode] = currentGen;
                            parent[toNode] = curr;
                            parentEdgeId[toNode] = eId;
                            pq.Enqueue(toNode, newDist);
                        }
                    }
                }

                if (matchedVertex != -1)
                {
                    matched[u] = true;
                    matched[matchedVertex] = true;

                    // Trace back and duplicate edges
                    int curr = matchedVertex;
                    while (curr != u)
                    {
                        int eId = parentEdgeId[curr];
                        traversalCounts[eId]++;
                        curr = parent[curr];
                    }
                }
            }
        }
    }

    static int[] Hierholzer(int N, int M, out int routeLength)
    {
        long totalEdgesLong = 0;
        for (int i = 0; i < M; i++)
        {
            totalEdgesLong += traversalCounts[i];
        }

        int totalEdges = (int)totalEdgesLong;
        int[] circuit = new int[totalEdges + 1];
        int circuitPtr = 0;

        int[] curEdge = new int[N];
        for (int i = 0; i < N; i++)
        {
            curEdge[i] = head[i];
        }

        int[] stack = new int[totalEdges + 5];
        int stackPtr = 0;

        stack[stackPtr++] = 0;

        while (stackPtr > 0)
        {
            int u = stack[stackPtr - 1];

            bool found = false;
            while (curEdge[u] != -1)
            {
                int eIdx = curEdge[u];
                int id = edgeId[eIdx];
                if (traversalCounts[id] > 0)
                {
                    traversalCounts[id]--;
                    int v = to[eIdx];
                    stack[stackPtr++] = v;
                    found = true;
                    break;
                }
                else
                {
                    curEdge[u] = next[eIdx];
                }
            }

            if (!found)
            {
                stackPtr--;
                circuit[circuitPtr++] = u;
            }
        }

        routeLength = circuitPtr;
        return circuit;
    }
}
