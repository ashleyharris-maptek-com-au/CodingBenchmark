using System;
using System.Diagnostics;
using System.IO;

// Chinese Postman Problem (Route Inspection) solver.
//
// Pipeline:
//   1. Fast byte-level stdin parsing.
//   2. Find odd-degree vertices. Pair them cheaply using rounds of
//      multi-source Dijkstra: grow shortest-path regions from every unmatched
//      odd vertex simultaneously; edges where two regions meet yield candidate
//      pairings (cost = dist[u] + w + dist[v]); accept candidates greedily by
//      cost. Repeat a few rounds for stragglers.
//   3. Any odd vertices still unmatched (pathological cases / time pressure)
//      are fixed via a spanning-tree T-join: duplicate a tree edge iff its
//      subtree contains an odd number of remaining odd vertices. O(N).
//   4. Reduce edge multiplicities mod 2 (keeping >= 1) so overlapping
//      duplicated paths don't inflate the tour.
//   5. Iterative Hierholzer to emit the Eulerian circuit from node 0.
//
// Scales: all core steps are O((N+M) log N) per matching round with a small
// constant number of rounds, and O(N+M) otherwise.
static class ChinesePostman
{
    static int N, M;
    static int[] eu, ev, ew;              // edge endpoints and weights
    static int[] adjOff, adjTo, adjEdge;  // CSR adjacency (both directions)
    static int[] mult;                    // times each edge must be traversed

    // binary min-heap (parallel arrays)
    static long[] heapD;
    static int[] heapV;
    static int heapSize;

    static void HeapPush(long d, int v)
    {
        if (heapSize == heapD.Length)
        {
            Array.Resize(ref heapD, heapSize * 2);
            Array.Resize(ref heapV, heapSize * 2);
        }
        int i = heapSize++;
        heapD[i] = d; heapV[i] = v;
        while (i > 0)
        {
            int p = (i - 1) >> 1;
            if (heapD[p] <= heapD[i]) break;
            long td = heapD[p]; heapD[p] = heapD[i]; heapD[i] = td;
            int tv = heapV[p]; heapV[p] = heapV[i]; heapV[i] = tv;
            i = p;
        }
    }

    static void HeapPop(out long d, out int v)
    {
        d = heapD[0]; v = heapV[0];
        heapSize--;
        if (heapSize > 0)
        {
            heapD[0] = heapD[heapSize]; heapV[0] = heapV[heapSize];
            int i = 0;
            while (true)
            {
                int l = 2 * i + 1, r = l + 1, s = i;
                if (l < heapSize && heapD[l] < heapD[s]) s = l;
                if (r < heapSize && heapD[r] < heapD[s]) s = r;
                if (s == i) break;
                long td = heapD[s]; heapD[s] = heapD[i]; heapD[i] = td;
                int tv = heapV[s]; heapV[s] = heapV[i]; heapV[i] = tv;
                i = s;
            }
        }
    }

    static void Main()
    {
        var sw = Stopwatch.StartNew();

        // ---- read all of stdin ----
        byte[] buf;
        using (var stdin = Console.OpenStandardInput())
        using (var ms = new MemoryStream())
        {
            byte[] chunk = new byte[1 << 20];
            int r;
            while ((r = stdin.Read(chunk, 0, chunk.Length)) > 0) ms.Write(chunk, 0, r);
            buf = ms.ToArray();
        }
        int pos = 0;
        Func<int> nextInt = delegate
        {
            while (pos < buf.Length && (buf[pos] < (byte)'0' || buf[pos] > (byte)'9') && buf[pos] != (byte)'-') pos++;
            bool neg = false;
            if (pos < buf.Length && buf[pos] == (byte)'-') { neg = true; pos++; }
            int x = 0;
            while (pos < buf.Length && buf[pos] >= (byte)'0' && buf[pos] <= (byte)'9')
            { x = x * 10 + (buf[pos] - (byte)'0'); pos++; }
            return neg ? -x : x;
        };

        N = nextInt(); M = nextInt();
        eu = new int[M]; ev = new int[M]; ew = new int[M];
        for (int i = 0; i < M; i++) { eu[i] = nextInt(); ev[i] = nextInt(); ew[i] = nextInt(); }

        var stdout = new StreamWriter(Console.OpenStandardOutput(), System.Text.Encoding.ASCII, 1 << 20);
        if (M == 0) { stdout.Write("1\n0\n"); stdout.Flush(); return; }

        // ---- CSR adjacency ----
        adjOff = new int[N + 2];
        for (int i = 0; i < M; i++) { adjOff[eu[i] + 1]++; adjOff[ev[i] + 1]++; }
        for (int i = 0; i < N; i++) adjOff[i + 1] += adjOff[i];
        adjTo = new int[2 * M]; adjEdge = new int[2 * M];
        int[] cursor = new int[N];
        Array.Copy(adjOff, cursor, N);
        for (int i = 0; i < M; i++)
        {
            int a = eu[i], b = ev[i];
            adjTo[cursor[a]] = b; adjEdge[cursor[a]] = i; cursor[a]++;
            adjTo[cursor[b]] = a; adjEdge[cursor[b]] = i; cursor[b]++;
        }

        mult = new int[M];
        for (int i = 0; i < M; i++) mult[i] = 1;

        // ---- odd-degree vertices ----
        bool[] unmatchedOdd = new bool[N];
        int oddCount = 0;
        for (int v = 0; v < N; v++)
            if (((adjOff[v + 1] - adjOff[v]) & 1) == 1) { unmatchedOdd[v] = true; oddCount++; }

        // ---- matching rounds: multi-source Dijkstra + greedy pairing ----
        if (oddCount > 0)
        {
            long[] dist = new long[N];
            int[] srcOf = new int[N];
            int[] parEdge = new int[N];
            heapD = new long[Math.Max(1024, 2 * M + 8)];
            heapV = new int[heapD.Length];
            long[] candCost = new long[M];
            int[] candEdge = new int[M];

            int round = 0;
            while (oddCount > 0 && round < 24 && sw.ElapsedMilliseconds < 18000)
            {
                round++;
                for (int v = 0; v < N; v++) { dist[v] = long.MaxValue; srcOf[v] = -1; parEdge[v] = -1; }
                heapSize = 0;
                for (int v = 0; v < N; v++)
                    if (unmatchedOdd[v]) { dist[v] = 0; srcOf[v] = v; HeapPush(0, v); }

                while (heapSize > 0)
                {
                    long d; int u;
                    HeapPop(out d, out u);
                    if (d > dist[u]) continue;
                    int s = srcOf[u];
                    for (int k = adjOff[u]; k < adjOff[u + 1]; k++)
                    {
                        int w2 = adjTo[k];
                        long nd = d + ew[adjEdge[k]];
                        if (nd < dist[w2])
                        {
                            dist[w2] = nd; srcOf[w2] = s; parEdge[w2] = adjEdge[k];
                            HeapPush(nd, w2);
                        }
                    }
                }

                int nc = 0;
                for (int i = 0; i < M; i++)
                {
                    int su = srcOf[eu[i]], sv = srcOf[ev[i]];
                    if (su >= 0 && sv >= 0 && su != sv)
                    {
                        candCost[nc] = dist[eu[i]] + dist[ev[i]] + ew[i];
                        candEdge[nc] = i; nc++;
                    }
                }
                if (nc == 0) break; // remaining odds unreachable from each other
                Array.Sort(candCost, candEdge, 0, nc);

                int matchedNow = 0;
                for (int j = 0; j < nc && oddCount > 0; j++)
                {
                    int e = candEdge[j];
                    int su = srcOf[eu[e]], sv = srcOf[ev[e]];
                    if (!unmatchedOdd[su] || !unmatchedOdd[sv]) continue;
                    unmatchedOdd[su] = false; unmatchedOdd[sv] = false;
                    oddCount -= 2; matchedNow += 2;
                    // duplicate the meeting edge and both shortest-path halves
                    mult[e]++;
                    int x = eu[e];
                    while (x != su) { int pe = parEdge[x]; mult[pe]++; x = (eu[pe] == x) ? ev[pe] : eu[pe]; }
                    x = ev[e];
                    while (x != sv) { int pe = parEdge[x]; mult[pe]++; x = (eu[pe] == x) ? ev[pe] : eu[pe]; }
                }
                if (matchedNow == 0) break;
            }
        }

        // ---- fallback: spanning-tree T-join for any leftover odd vertices ----
        if (oddCount > 0)
        {
            int[] order = new int[N];
            int[] parE = new int[N];
            int[] parV = new int[N];
            bool[] seen = new bool[N];
            int qh = 0, qt = 0;
            order[qt++] = 0; seen[0] = true; parE[0] = -1; parV[0] = -1;
            while (qh < qt)
            {
                int u = order[qh++];
                for (int k = adjOff[u]; k < adjOff[u + 1]; k++)
                {
                    int w2 = adjTo[k];
                    if (!seen[w2]) { seen[w2] = true; parE[w2] = adjEdge[k]; parV[w2] = u; order[qt++] = w2; }
                }
            }
            int[] parity = new int[N];
            for (int v = 0; v < N; v++) if (unmatchedOdd[v]) parity[v] = 1;
            for (int idx = qt - 1; idx >= 1; idx--)
            {
                int v = order[idx];
                if ((parity[v] & 1) == 1) { mult[parE[v]]++; parity[parV[v]] ^= 1; }
            }
        }

        // ---- reduce multiplicities: same parity, at least one traversal ----
        for (int i = 0; i < M; i++)
            if (mult[i] > 2) mult[i] = 2 - (mult[i] & 1); // odd -> 1, even -> 2

        long total = 0;
        for (int i = 0; i < M; i++) total += mult[i];

        // ---- Hierholzer (iterative) ----
        int[] remaining = mult; // consumed in place
        int[] ptr = new int[N];
        Array.Copy(adjOff, ptr, N);
        int[] stack = new int[total + 2];
        int[] circuit = new int[total + 1];
        int sp = 0, ci = 0;
        stack[sp++] = 0;
        while (sp > 0)
        {
            int v = stack[sp - 1];
            int end = adjOff[v + 1];
            int p = ptr[v];
            while (p < end && remaining[adjEdge[p]] == 0) p++;
            ptr[v] = p;
            if (p < end)
            {
                remaining[adjEdge[p]]--;
                stack[sp++] = adjTo[p];
            }
            else
            {
                circuit[ci++] = v;
                sp--;
            }
        }

        // ---- output ----
        char[] tmp = new char[12];
        stdout.Write(ci);
        stdout.Write('\n');
        for (int i = ci - 1; i >= 0; i--)
        {
            int x = circuit[i];
            int t = 12;
            if (x == 0) tmp[--t] = '0';
            else while (x > 0) { tmp[--t] = (char)('0' + x % 10); x /= 10; }
            stdout.Write(tmp, t, 12 - t);
            stdout.Write(i == 0 ? '\n' : ' ');
        }
        stdout.Flush();
    }
}
