// Adaptive Travelling Salesman Problem solver.
//
// Strategy by scale (N = number of cities):
//   N <= 13          : exact Held-Karp dynamic programming (optimal tour)
//   all other N      : Hilbert space-filling-curve construction (O(N log N)),
//                      then grid-based k-nearest-neighbour 2-opt improvement
//                      with don't-look bits, shorter-side segment reversal and
//                      a reversal-length cap that scales with N.
//   N > 20,000,000   : Hilbert construction only (already near-linear and
//                      memory-lean; guarantees completion well inside 300 s).
//
// Input : line 1 N, then N lines "x y" (floats) on stdin.
// Output: one line, N space-separated city indices starting with city 0.
//
// Target: MSVC 19.5x, /std:c++20, x64. Uses only standard headers plus
// <io.h>/<fcntl.h> for binary stdin/stdout on Windows (guarded for MSVC).

#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <cstdint>
#include <cmath>
#include <vector>
#include <algorithm>
#include <chrono>
#include <thread>
#include <string>

#if defined(_MSC_VER)
#include <io.h>
#include <fcntl.h>
#endif

#if defined(_MSC_VER)
#include <execution>
#define HAVE_PAR_SORT 1
#endif

using Clock = std::chrono::steady_clock;
static Clock::time_point g_start;

static double elapsedSec() {
    return std::chrono::duration<double>(Clock::now() - g_start).count();
}

// ---------------------------------------------------------------- input ----

static std::vector<char> readAllStdin() {
#if defined(_MSC_VER)
    _setmode(_fileno(stdin), _O_BINARY);
#endif
    std::vector<char> buf;
    buf.reserve(1 << 22);
    const size_t CH = 1 << 22;
    size_t sz = 0;
    for (;;) {
        buf.resize(sz + CH);
        size_t got = fread(buf.data() + sz, 1, CH, stdin);
        sz += got;
        if (got < CH) break;
    }
    buf.resize(sz);
    buf.push_back('\0');
    return buf;
}

// Fast float parser: [+-]digits[.digits][eE[+-]digits]
static inline double parseFloat(const char*& p) {
    while (*p && (*p == ' ' || *p == '\t' || *p == '\r' || *p == '\n' || *p == ',')) ++p;
    bool neg = false;
    if (*p == '-') { neg = true; ++p; }
    else if (*p == '+') ++p;
    double v = 0.0;
    while (*p >= '0' && *p <= '9') { v = v * 10.0 + (*p - '0'); ++p; }
    if (*p == '.') {
        ++p;
        double f = 0.0, scale = 1.0;
        while (*p >= '0' && *p <= '9') { f = f * 10.0 + (*p - '0'); scale *= 10.0; ++p; }
        v += f / scale;
    }
    if (*p == 'e' || *p == 'E') {
        ++p;
        bool eneg = false;
        if (*p == '-') { eneg = true; ++p; }
        else if (*p == '+') ++p;
        int ex = 0;
        while (*p >= '0' && *p <= '9') { ex = ex * 10 + (*p - '0'); ++p; }
        v *= std::pow(10.0, eneg ? -ex : ex);
    }
    return neg ? -v : v;
}

static inline long long parseInt(const char*& p) {
    while (*p && (*p < '0' || *p > '9') && *p != '-') ++p;
    bool neg = false;
    if (*p == '-') { neg = true; ++p; }
    long long v = 0;
    while (*p >= '0' && *p <= '9') { v = v * 10 + (*p - '0'); ++p; }
    return neg ? -v : v;
}

// --------------------------------------------------------------- output ----

static void writeTour(const std::vector<int>& tour) {
#if defined(_MSC_VER)
    _setmode(_fileno(stdout), _O_BINARY);
#endif
    const size_t n = tour.size();
    std::vector<char> out;
    out.resize(n * 9 + 16); // avg guess; grown below if needed
    size_t cap = out.size();
    size_t o = 0;
    char tmp[16];
    for (size_t i = 0; i < n; ++i) {
        if (o + 16 > cap) { cap = cap * 2; out.resize(cap); }
        unsigned v = (unsigned)tour[i];
        int len = 0;
        do { tmp[len++] = char('0' + v % 10); v /= 10; } while (v);
        while (len) out[o++] = tmp[--len];
        out[o++] = (i + 1 < n) ? ' ' : '\n';
    }
    fwrite(out.data(), 1, o, stdout);
    fflush(stdout);
}

// ------------------------------------------------------------- geometry ----

static std::vector<float> X, Y;

static inline double dist(int a, int b) {
    double dx = (double)X[a] - X[b];
    double dy = (double)Y[a] - Y[b];
    return std::sqrt(dx * dx + dy * dy);
}

// ------------------------------------------------------------ Held-Karp ----

static std::vector<int> heldKarp(int n) {
    // Exact DP over subsets, n <= 13. Path starts at 0; return implicit.
    const int FULL = 1 << n;
    std::vector<double> D(n * n);
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < n; ++j) D[i * n + j] = dist(i, j);

    const double INF = 1e300;
    std::vector<double> dp((size_t)FULL * n, INF);
    std::vector<int8_t> par((size_t)FULL * n, -1);
    dp[(size_t)1 * n + 0] = 0.0;
    for (int mask = 1; mask < FULL; ++mask) {
        if (!(mask & 1)) continue;
        for (int last = 0; last < n; ++last) {
            if (!(mask & (1 << last))) continue;
            double cur = dp[(size_t)mask * n + last];
            if (cur >= INF) continue;
            for (int nxt = 1; nxt < n; ++nxt) {
                if (mask & (1 << nxt)) continue;
                int m2 = mask | (1 << nxt);
                double nd = cur + D[last * n + nxt];
                if (nd < dp[(size_t)m2 * n + nxt]) {
                    dp[(size_t)m2 * n + nxt] = nd;
                    par[(size_t)m2 * n + nxt] = (int8_t)last;
                }
            }
        }
    }
    // close the cycle back to 0
    int best = -1; double bestc = INF;
    int full = FULL - 1;
    for (int last = 1; last < n; ++last) {
        double c = dp[(size_t)full * n + last] + D[last * n + 0];
        if (c < bestc) { bestc = c; best = last; }
    }
    std::vector<int> tour(n);
    int mask = full, cur = (n == 1) ? 0 : best;
    for (int i = n - 1; i >= 0; --i) {
        tour[i] = cur;
        int p = par[(size_t)mask * n + cur];
        mask ^= (1 << cur);
        cur = p;
    }
    return tour;
}

// ---------------------------------------------------------- Hilbert sort ----

// Map (x,y) in [0,2^16)^2 to Hilbert distance (32-bit).
static inline uint32_t hilbertD(uint32_t x, uint32_t y) {
    uint32_t rx, ry, d = 0;
    for (uint32_t s = 1u << 15; s > 0; s >>= 1) {
        rx = (x & s) ? 1u : 0u;
        ry = (y & s) ? 1u : 0u;
        d += s * s * ((3u * rx) ^ ry);
        // rotate
        if (ry == 0) {
            if (rx == 1) { x = s - 1 - x; y = s - 1 - y; }
            uint32_t t = x; x = y; y = t;
        }
    }
    return d;
}

static std::vector<int> hilbertTour(int n) {
    float minx = X[0], maxx = X[0], miny = Y[0], maxy = Y[0];
    for (int i = 1; i < n; ++i) {
        minx = std::min(minx, X[i]); maxx = std::max(maxx, X[i]);
        miny = std::min(miny, Y[i]); maxy = std::max(maxy, Y[i]);
    }
    double sx = (maxx > minx) ? 65535.0 / ((double)maxx - minx) : 0.0;
    double sy = (maxy > miny) ? 65535.0 / ((double)maxy - miny) : 0.0;

    std::vector<uint64_t> keys(n);
    int hw = (int)std::thread::hardware_concurrency();
    if (hw < 1) hw = 1;
    int nthreads = (n > 200000) ? std::min(hw, 32) : 1;
    auto fill = [&](int lo, int hi) {
        for (int i = lo; i < hi; ++i) {
            uint32_t gx = (uint32_t)((X[i] - minx) * sx + 0.5);
            uint32_t gy = (uint32_t)((Y[i] - miny) * sy + 0.5);
            keys[i] = ((uint64_t)hilbertD(gx, gy) << 32) | (uint32_t)i;
        }
    };
    if (nthreads == 1) fill(0, n);
    else {
        std::vector<std::thread> th;
        int chunk = (n + nthreads - 1) / nthreads;
        for (int t = 0; t < nthreads; ++t) {
            int lo = t * chunk, hi = std::min(n, lo + chunk);
            if (lo < hi) th.emplace_back(fill, lo, hi);
        }
        for (auto& t : th) t.join();
    }
#if defined(HAVE_PAR_SORT)
    if (n > 100000) std::sort(std::execution::par_unseq, keys.begin(), keys.end());
    else std::sort(keys.begin(), keys.end());
#else
    std::sort(keys.begin(), keys.end());
#endif
    std::vector<int> tour(n);
    for (int i = 0; i < n; ++i) tour[i] = (int)(keys[i] & 0xffffffffu);
    return tour;
}

// ----------------------------------------------- k-nearest via uniform grid -

struct Grid {
    int gw = 1, gh = 1;
    float minx = 0, miny = 0;
    double invCell = 1.0;
    std::vector<int> cellStart; // size gw*gh+1
    std::vector<int> items;     // size n
};

static Grid buildGrid(int n) {
    Grid g;
    float minx = X[0], maxx = X[0], miny = Y[0], maxy = Y[0];
    for (int i = 1; i < n; ++i) {
        minx = std::min(minx, X[i]); maxx = std::max(maxx, X[i]);
        miny = std::min(miny, Y[i]); maxy = std::max(maxy, Y[i]);
    }
    double w = std::max(1e-12, (double)maxx - minx);
    double h = std::max(1e-12, (double)maxy - miny);
    // aim for ~2 points per cell
    double targetCells = std::max(1.0, n / 2.0);
    double cell = std::sqrt(w * h / targetCells);
    if (cell <= 0 || !std::isfinite(cell)) cell = 1.0;
    int gw = (int)(w / cell) + 1;
    int gh = (int)(h / cell) + 1;
    gw = std::max(1, std::min(gw, 65536));
    gh = std::max(1, std::min(gh, 65536));
    g.gw = gw; g.gh = gh; g.minx = minx; g.miny = miny;
    g.invCell = 1.0 / cell;
    // recompute invCell so all points map inside
    g.invCell = std::min((gw - 1e-6) / w, (gh - 1e-6) / h);
    if (!std::isfinite(g.invCell) || g.invCell <= 0) g.invCell = 1.0;

    std::vector<int> cnt((size_t)gw * gh + 1, 0);
    std::vector<int> ci(n);
    for (int i = 0; i < n; ++i) {
        int cx = (int)((X[i] - minx) * g.invCell);
        int cy = (int)((Y[i] - miny) * g.invCell);
        cx = std::min(std::max(cx, 0), gw - 1);
        cy = std::min(std::max(cy, 0), gh - 1);
        ci[i] = cy * gw + cx;
        cnt[ci[i] + 1]++;
    }
    for (size_t c = 1; c < cnt.size(); ++c) cnt[c] += cnt[c - 1];
    g.items.resize(n);
    std::vector<int> fill = cnt;
    for (int i = 0; i < n; ++i) g.items[fill[ci[i]]++] = i;
    g.cellStart = std::move(cnt);
    return g;
}

// k nearest neighbours of every city (flat array n*k, -1 padded)
static std::vector<int> buildKNN(int n, int k, const Grid& g) {
    std::vector<int> nbr((size_t)n * k, -1);
    double cellSize = 1.0 / g.invCell;

    auto work = [&](int lo, int hi) {
        std::vector<int> candIdx(k);
        std::vector<double> candD(k);
        for (int i = lo; i < hi; ++i) {
            int cnt = 0;
            double worst = 1e300;
            int cx = (int)((X[i] - g.minx) * g.invCell);
            int cy = (int)((Y[i] - g.miny) * g.invCell);
            cx = std::min(std::max(cx, 0), g.gw - 1);
            cy = std::min(std::max(cy, 0), g.gh - 1);
            int maxRing = std::max(g.gw, g.gh);
            for (int r = 0; r <= maxRing; ++r) {
                // stop when the ring cannot contain anything closer
                if (cnt >= k) {
                    double ringMin = (r - 1) * cellSize;
                    if (ringMin > worst) break;
                }
                int x0 = cx - r, x1 = cx + r, y0 = cy - r, y1 = cy + r;
                for (int yy = y0; yy <= y1; ++yy) {
                    if (yy < 0 || yy >= g.gh) continue;
                    bool edgeRow = (yy == y0 || yy == y1);
                    int step = edgeRow ? 1 : (x1 - x0 == 0 ? 1 : x1 - x0);
                    for (int xx = x0; xx <= x1; xx += step) {
                        if (xx < 0 || xx >= g.gw) continue;
                        int c = yy * g.gw + xx;
                        for (int q = g.cellStart[c]; q < g.cellStart[c + 1]; ++q) {
                            int j = g.items[q];
                            if (j == i) continue;
                            double d = dist(i, j);
                            if (cnt < k) {
                                // insertion sort into candidate list
                                int p = cnt++;
                                while (p > 0 && candD[p - 1] > d) {
                                    candD[p] = candD[p - 1]; candIdx[p] = candIdx[p - 1]; --p;
                                }
                                candD[p] = d; candIdx[p] = j;
                                worst = candD[cnt - 1];
                            } else if (d < worst) {
                                int p = k - 1;
                                while (p > 0 && candD[p - 1] > d) {
                                    candD[p] = candD[p - 1]; candIdx[p] = candIdx[p - 1]; --p;
                                }
                                candD[p] = d; candIdx[p] = j;
                                worst = candD[k - 1];
                            }
                        }
                        if (step == 0) break;
                    }
                }
            }
            for (int q = 0; q < cnt; ++q) nbr[(size_t)i * k + q] = candIdx[q];
        }
    };

    int hw = (int)std::thread::hardware_concurrency();
    if (hw < 1) hw = 1;
    int nthreads = (n > 20000) ? std::min(hw, 48) : 1;
    if (nthreads == 1) work(0, n);
    else {
        std::vector<std::thread> th;
        int chunk = (n + nthreads - 1) / nthreads;
        for (int t = 0; t < nthreads; ++t) {
            int lo = t * chunk, hi = std::min(n, lo + chunk);
            if (lo < hi) th.emplace_back(work, lo, hi);
        }
        for (auto& t : th) t.join();
    }
    return nbr;
}

// -------------------------------------------------------------- 2-opt ------

static void twoOpt(std::vector<int>& tour, const std::vector<int>& nbr, int k,
                   int maxRev, double deadline) {
    const int n = (int)tour.size();
    std::vector<int> pos(n);
    for (int i = 0; i < n; ++i) pos[tour[i]] = i;

    // cyclic segment reversal over positions [from..to], choosing given ends
    auto reverseSeg = [&](int from, int to) {
        int len = to - from; if (len < 0) len += n; ++len;
        int a = from, b = to;
        for (int s = 0; s < len / 2; ++s) {
            int ta = tour[a], tb = tour[b];
            tour[a] = tb; tour[b] = ta;
            pos[tb] = a; pos[ta] = b;
            if (++a >= n) a = 0;
            if (--b < 0) b = n - 1;
        }
    };

    std::vector<uint8_t> inQ(n, 1);
    std::vector<int> queue(n);
    for (int i = 0; i < n; ++i) queue[i] = tour[i];
    size_t qhead = 0;
    auto push = [&](int c) { if (!inQ[c]) { inQ[c] = 1; queue.push_back(c); } };

    long long checkCtr = 0;
    while (qhead < queue.size()) {
        if (((++checkCtr) & 1023) == 0 && elapsedSec() > deadline) break;
        int a = queue[qhead++];
        inQ[a] = 0;
        bool improvedAny = false;

        for (int dir = 0; dir < 2; ++dir) {
            // dir 0: edge (a, succ(a)); dir 1: edge (pred(a), a) treated as
            // (a, succ(a)) on the reversed orientation — handled symmetrically.
            int pa = pos[a];
            int c1, c2; // edge endpoints, c2 = tour-successor side
            if (dir == 0) { c1 = a; c2 = tour[(pa + 1 == n) ? 0 : pa + 1]; }
            else          { c1 = a; c2 = tour[(pa == 0) ? n - 1 : pa - 1]; }
            double d12 = dist(c1, c2);

            const int* nb = &nbr[(size_t)a * k];
            for (int q = 0; q < k; ++q) {
                int b = nb[q];
                if (b < 0) break;
                double dab = dist(a, b);
                if (dab >= d12) break; // sorted neighbours: no gain possible
                int pb = pos[b];
                int b2; // same-direction successor of b
                if (dir == 0) b2 = tour[(pb + 1 == n) ? 0 : pb + 1];
                else          b2 = tour[(pb == 0) ? n - 1 : pb - 1];
                if (b2 == a || b == c2) continue;
                double gain = d12 + dist(b, b2) - dab - dist(c2, b2);
                if (gain <= 1e-9 * (d12 + 1.0)) continue;

                // reverse segment between c2..b (in tour direction `dir`)
                int from, to;
                if (dir == 0) { from = (pa + 1 == n) ? 0 : pa + 1; to = pb; }
                else          { from = pb; to = (pa == 0) ? n - 1 : pa - 1; }
                int len = to - from; if (len < 0) len += n; ++len;
                if (len * 2 > n) {
                    // reverse the complementary segment instead (equivalent tour)
                    int nf = (to + 1 == n) ? 0 : to + 1;
                    int nt = (from == 0) ? n - 1 : from - 1;
                    from = nf; to = nt;
                    len = n - len;
                }
                if (len > maxRev) continue;
                reverseSeg(from, to);
                push(a); push(b); push(c2); push(b2);
                improvedAny = true;
                break;
            }
            if (improvedAny) break;
        }
        if (improvedAny) push(a);

        // keep queue memory bounded
        if (qhead > (size_t)4 * n && qhead * 2 > queue.size()) {
            queue.erase(queue.begin(), queue.begin() + qhead);
            qhead = 0;
        }
    }
}

// -------------------------------------------------------------- Or-opt -----

// Segment relocation (lengths 1..3, both orientations) on a doubly linked
// tour. O(1) per applied move; returns true if anything improved.
static bool orOpt(std::vector<int>& tour, const std::vector<int>& nbr, int k,
                  double deadline) {
    const int n = (int)tour.size();
    if (n < 5) return false;
    std::vector<int> nxt(n), prv(n);
    for (int i = 0; i < n; ++i) {
        int a = tour[i], b = tour[(i + 1 == n) ? 0 : i + 1];
        nxt[a] = b; prv[b] = a;
    }
    std::vector<uint8_t> inQ(n, 1);
    std::vector<int> queue(tour);
    size_t qhead = 0;
    auto push = [&](int c) { if (!inQ[c]) { inQ[c] = 1; queue.push_back(c); } };

    bool any = false;
    long long ctr = 0;
    int seg[3];
    while (qhead < queue.size()) {
        if (((++ctr) & 1023) == 0 && elapsedSec() > deadline) break;
        int a = queue[qhead++];
        inQ[a] = 0;

        bool moved = false;
        for (int L = 1; L <= 3 && !moved; ++L) {
            seg[0] = a;
            bool ok = true;
            for (int l = 1; l < L; ++l) {
                seg[l] = nxt[seg[l - 1]];
                if (seg[l] == a) { ok = false; break; }
            }
            if (!ok) break;
            int s0 = seg[0], s1 = seg[L - 1];
            int p = prv[s0], q = nxt[s1];
            if (p == s1 || q == s0) break; // segment is (almost) whole tour
            double removal = dist(p, s0) + dist(s1, q) - dist(p, q);
            if (removal <= 1e-12) continue;

            const int* nb = &nbr[(size_t)a * k];
            for (int t = 0; t < k; ++t) {
                int b = nb[t];
                if (b < 0) break;
                if (b == p) continue;
                bool inSeg = false;
                for (int l = 0; l < L; ++l) if (seg[l] == b) { inSeg = true; break; }
                if (inSeg) continue;
                int c = nxt[b];
                for (int l = 0; l < L; ++l) if (seg[l] == c) { inSeg = true; break; }
                if (inSeg) continue;
                double base = dist(b, c);
                double addF = dist(b, s0) + dist(s1, c) - base;
                double addR = dist(b, s1) + dist(s0, c) - base;
                bool rev = addR < addF;
                double add = rev ? addR : addF;
                double gain = removal - add;
                if (gain <= 1e-9 * (removal + base + 1.0)) continue;

                // unlink segment
                nxt[p] = q; prv[q] = p;
                // relink between b and c, possibly reversed
                int h = rev ? s1 : s0, tl = rev ? s0 : s1;
                nxt[b] = h; prv[h] = b;
                nxt[tl] = c; prv[c] = tl;
                if (rev) {
                    for (int l = L - 1; l > 0; --l) {
                        nxt[seg[l]] = seg[l - 1];
                        prv[seg[l - 1]] = seg[l];
                    }
                } else if (L > 1) {
                    for (int l = 0; l + 1 < L; ++l) {
                        nxt[seg[l]] = seg[l + 1];
                        prv[seg[l + 1]] = seg[l];
                    }
                }
                push(p); push(q); push(b); push(c); push(s0); push(s1);
                any = true; moved = true;
                break;
            }
        }
        if (moved) push(a);

        if (qhead > (size_t)4 * n && qhead * 2 > queue.size()) {
            queue.erase(queue.begin(), queue.begin() + qhead);
            qhead = 0;
        }
    }
    // rebuild tour array from links
    int cur = 0;
    for (int i = 0; i < n; ++i) { tour[i] = cur; cur = nxt[cur]; }
    return any;
}

// ---------------------------------------------------------------- main -----

int main() {
    g_start = Clock::now();
    std::vector<char> in = readAllStdin();
    const char* p = in.data();
    long long nl = parseInt(p);
    if (nl <= 0) { printf("0\n"); return 0; }
    int n = (int)nl;

    X.resize(n); Y.resize(n);
    for (int i = 0; i < n; ++i) {
        X[i] = (float)parseFloat(p);
        Y[i] = (float)parseFloat(p);
    }
    in.clear(); in.shrink_to_fit();

    std::vector<int> tour;

    if (n == 1) { tour = {0}; writeTour(tour); return 0; }
    if (n <= 3) {
        tour.resize(n);
        for (int i = 0; i < n; ++i) tour[i] = i;
        writeTour(tour);
        return 0;
    }

    if (n <= 13) {
        tour = heldKarp(n); // optimal
    } else {
        // construction: Hilbert space-filling curve
        tour = hilbertTour(n);

        // improvement: neighbour-list 2-opt, scaled to problem size
        if (n <= 20000000) {
            int k = (n <= 10000) ? 12 : (n <= 1000000) ? 10 : 6;
            int maxRev = (n <= 200000) ? n
                       : (n <= 2000000) ? 30000
                       : 3000;
            double budget = std::max(5.0, std::min(240.0, n / 40000.0 + 5.0));
            double deadline = std::min(270.0, elapsedSec() + budget);

            Grid g = buildGrid(n);
            std::vector<int> nbr = buildKNN(n, k, g);
            g.cellStart.clear(); g.items.clear();

            // alternate 2-opt and Or-opt until neither improves or time is up
            for (int round = 0; round < 6; ++round) {
                twoOpt(tour, nbr, k, maxRev, deadline);
                if (elapsedSec() > deadline) break;
                bool improved = orOpt(tour, nbr, k, deadline);
                if (!improved || elapsedSec() > deadline) break;
            }
        }
        // else: Hilbert order alone — near-linear, completes at any scale
    }

    // rotate so the tour starts at city 0
    int z = 0;
    for (int i = 0; i < n; ++i) if (tour[i] == 0) { z = i; break; }
    std::rotate(tour.begin(), tour.begin() + z, tour.end());

    writeTour(tour);
    return 0;
}
