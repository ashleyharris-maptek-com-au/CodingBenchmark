```glsl
#version 450

layout(local_size_x = 256) in;

// Each body is stored as two consecutive vec4s:
//   vec4(position.xyz, mass)
//   vec4(velocity.xyz, 0.0)
// So body i starts at index i*2 in the buffer.

layout(set = 0, binding = 0) readonly buffer InputBodies {
    vec4 inData[];   // length = N * 2
};

layout(set = 0, binding = 1) buffer OutputBodies {
    vec4 outData[];  // length = N * 2
};

layout(set = 0, binding = 2) uniform Params {
    uvec4 counts;    // counts.x = N (number of bodies)
    vec4 physics;    // physics.x = dt, physics.y = softening, physics.z = G
};

void main() {
    uint i = gl_GlobalInvocationID.x;
    uint N = counts.x;
    if (i >= N) {
        return;
    }

    uint index_i = i * 2;
    vec4 pos_mass_i = inData[index_i];
    vec3 pos_i = pos_mass_i.xyz;
    float mass_i = pos_mass_i.w;
    
    vec3 vel_i = inData[index_i + 1].xyz;

    vec3 acc = vec3(0.0);
    float softening = physics.y;
    float softening_sq = softening * softening;
    float G = physics.z;

    for (uint j = 0; j < N; j++) {
        if (i == j) {
            continue;
        }
        uint index_j = j * 2;
        vec4 pos_mass_j = inData[index_j];
        vec3 pos_j = pos_mass_j.xyz;
        float mass_j = pos_mass_j.w;

        vec3 diff = pos_j - pos_i;
        float dist_sq = dot(diff, diff) + softening_sq;
        float inv_dist = inversesqrt(dist_sq);
        float inv_dist3 = inv_dist * inv_dist * inv_dist;

        acc += (G * mass_j * inv_dist3) * diff;
    }

    float dt = physics.x;
    vec3 vel_new = vel_i + acc * dt;
    vec3 pos_new = pos_i + vel_new * dt;

    outData[index_i] = vec4(pos_new, mass_i);
    outData[index_i + 1] = vec4(vel_new, 0.0);
}
```
