using System;
using System.IO;

public static class CppSolver
{
    // Growable multigraph as arc lists. Edge e -> arcs 2e, 2e+1.
    static int[] head;          // head[v] = first arc index or -1
    static int[] nxt;           // nxt[arc]
    static int[] to;            // to[arc]
    static int[] eid;           // eid[arc] = edge id
    static long[] ew;           // ew[edge] = weight
    static int arcCount = 0;
    static int edgeCount = 0;
    static int N;

    static void EnsureArcCap(int need)
    {
        if (need <= nxt.Length) return;
        int cap = nxt.Length;
        while (cap < need) cap <<= 1;
        Array.Resize(ref nxt, cap);
        Array.Resize(ref to, cap);
        Array.Resize(ref eid, cap);
    }

    static void EnsureEdgeCap(int need)
    {
        if (need <= ew.Length) return;
        int cap = ew.Length;
        while (cap < need) cap <<= 1;
        Array.Resize(ref ew, cap);
    }

    static void AddEdge(int u, int v, long w)
    {
        EnsureEdgeCap(edgeCount + 1);
        EnsureArcCap(arcCount + 2);
        ew[edgeCount] = w;
        to[arcCount] = v; eid[arcCount] = edgeCount; nxt[arcCount] = head[u]; head[u] = arcCount; arcCount++;
        to[arcCount] = u; eid[arcCount] = edgeCount; nxt[arcCount] = head[v]; head[v] = arcCount; arcCount++;
        edgeCount++;
    }

    // Dijkstra state (versioned to avoid clearing)
    static long[] dist;
    static int[] parentArc;
    static int[] version;
    static int curVersion = 0;
    static long[] heapKey;
    static int[] heapNode;
    static int heapSize;

    static void HeapPush(long k, int v)
    {
        if (heapSize == heapKey.Length)
        {
            Array.Resize(ref heapKey, heapSize << 1);
            Array.Resize(ref heapNode, heapSize << 1);
        }
        int i = heapSize++;
        heapKey[i] = k; heapNode[i] = v;
        while (i > 0)
        {
            int p = (i - 1) >> 1;
            if (heapKey[p] <= heapKey[i]) break;
            long tk = heapKey[p]; heapKey[p] = heapKey[i]; heapKey[i] = tk;
            int tn = heapNode[p]; heapNode[p] = heapNode[i]; heapNode[i] = tn;
            i = p;
        }
    }

    static void HeapPop()
    {
        heapSize--;
        heapKey[0] = heapKey[heapSize]; heapNode[0] = heapNode[heapSize];
        int i = 0;
        while (true)
        {
            int l = 2 * i + 1, r = l + 1, m = i;
            if (l < heapSize && heapKey[l] < heapKey[m]) m = l;
            if (r < heapSize && heapKey[r] < heapKey[m]) m = r;
            if (m == i) break;
            long tk = heapKey[m]; heapKey[m] = heapKey[i]; heapKey[i] = tk;
            int tn = heapNode[m]; heapNode[m] = heapNode[i]; heapNode[i] = tn;
            i = m;
        }
    }

    static long GetDist(int v) { return version[v] == curVersion ? dist[v] : long.MaxValue; }
    static void SetDist(int v, long d, int pa)
    {
        if (version[v] != curVersion) { version[v] = curVersion; }
        dist[v] = d; parentArc[v] = pa;
    }

    // Dijkstra from src; if stopAtOdd, terminate when settling an unmatched odd vertex != src.
    // Returns that vertex or -1. If full, runs to completion.
    static int Dijkstra(int src, bool[] isTargetUnmatched)
    {
        curVersion++;
        heapSize = 0;
        SetDist(src, 0, -1);
        HeapPush(0, src);
        while (heapSize > 0)
        {
            long d = heapKey[0]; int v = heapNode[0]; HeapPop();
            if (d > GetDist(v)) continue;
            if (isTargetUnmatched != null && v != src && isTargetUnmatched[v]) return v;
            for (int a = head[v]; a != -1; a = nxt[a])
            {
                int u = to[a];
                long nd = d + ew[eid[a]];
                if (nd < GetDist(u))
                {
                    SetDist(u, nd, a);
                    HeapPush(nd, u);
                }
            }
        }
        return -1;
    }

    // Duplicate edges along shortest path back from t (using parentArc of most recent Dijkstra
    // when parents given as arrays paDist/pa). Uses current parentArc/version state.
    static void DuplicatePathFromCurrent(int t)
    {
        int v = t;
        while (true)
        {
            int a = parentArc[v];
            if (a == -1) break;
            int e = eid[a];
            int u = to[a ^ 1]; // arc a goes u->v, twin's 'to' is u
            AddEdge(u, v, ew[e]);
            v = u;
        }
    }

    public static void Main()
    {
        // ---- fast input ----
        var input = Console.OpenStandardInput();
        var buf = new byte[1 << 16];
        int bufLen = 0, bufPos = 0;
        Func<int> readByte = () =>
        {
            if (bufPos >= bufLen) { bufLen = input.Read(buf, 0, buf.Length); bufPos = 0; if (bufLen <= 0) return -1; }
            return buf[bufPos++];
        };
        Func<long> readLong = () =>
        {
            int c; do { c = readByte(); } while (c != '-' && (c < '0' || c > '9') && c != -1);
            bool neg = false;
            if (c == '-') { neg = true; c = readByte(); }
            long x = 0;
            while (c >= '0' && c <= '9') { x = x * 10 + (c - '0'); c = readByte(); }
            return neg ? -x : x;
        };

        N = (int)readLong();
        int M = (int)readLong();

        head = new int[N];
        for (int i = 0; i < N; i++) head[i] = -1;
        int arcCap = Math.Max(16, 4 * M + 16);
        nxt = new int[arcCap]; to = new int[arcCap]; eid = new int[arcCap];
        ew = new long[Math.Max(16, 2 * M + 16)];

        for (int i = 0; i < M; i++)
        {
            int u = (int)readLong();
            int v = (int)readLong();
            long w = readLong();
            AddEdge(u, v, w);
        }

        var stdout = new StreamWriter(Console.OpenStandardOutput(), System.Text.Encoding.ASCII, 1 << 20);

        if (M == 0)
        {
            stdout.Write("0\n");
            stdout.Flush();
            return;
        }

        dist = new long[N];
        parentArc = new int[N];
        version = new int[N];
        heapKey = new long[1024];
        heapNode = new int[1024];

        // ---- find odd-degree vertices ----
        var deg = new int[N];
        for (int e = 0; e < M; e++)
        {
            deg[to[2 * e]]++;
            deg[to[2 * e + 1]]++;
        }
        int oddCount = 0;
        for (int v = 0; v < N; v++) if ((deg[v] & 1) != 0) oddCount++;
        var odd = new int[oddCount];
        {
            int k = 0;
            for (int v = 0; v < N; v++) if ((deg[v] & 1) != 0) odd[k++] = v;
        }

        if (oddCount > 0)
        {
            if (oddCount <= 12)
            {
                // Exact: all-pairs shortest paths among odd vertices + bitmask DP matching.
                int K = oddCount;
                var dmat = new long[K, K];
                var parents = new int[K][]; // parentArc snapshot per source
                for (int i = 0; i < K; i++)
                {
                    Dijkstra(odd[i], null);
                    var pa = new int[N];
                    for (int v = 0; v < N; v++) pa[v] = version[v] == curVersion ? parentArc[v] : -1;
                    parents[i] = pa;
                    for (int j = 0; j < K; j++) dmat[i, j] = GetDist(odd[j]);
                }
                int full = 1 << K;
                var dp = new long[full];
                var choice = new int[full]; // packed (i<<8)|j pair chosen
                for (int m = 1; m < full; m++) dp[m] = long.MaxValue;
                for (int m = 0; m < full; m++)
                {
                    if (dp[m] == long.MaxValue) continue;
                    int i = 0;
                    while (i < K && (m & (1 << i)) != 0) i++;
                    if (i >= K) continue;
                    for (int j = i + 1; j < K; j++)
                    {
                        if ((m & (1 << j)) != 0) continue;
                        long nd = dp[m] + dmat[i, j];
                        int nm = m | (1 << i) | (1 << j);
                        if (nd < dp[nm]) { dp[nm] = nd; choice[nm] = (i << 8) | j; }
                    }
                }
                // reconstruct
                int mask = full - 1;
                while (mask != 0)
                {
                    int c = choice[mask];
                    int i = c >> 8, j = c & 255;
                    // duplicate path from odd[j] back to odd[i] using parents[i]
                    int v = odd[j];
                    var pa = parents[i];
                    while (pa[v] != -1)
                    {
                        int a = pa[v];
                        AddEdge(to[a ^ 1], v, ew[eid[a]]);
                        v = to[a ^ 1];
                    }
                    mask &= ~((1 << i) | (1 << j));
                }
            }
            else
            {
                // Round-based multi-source Dijkstra (Voronoi) matching:
                // grow regions from all unmatched odd vertices at once; boundary edges
                // give candidate pairs (regionA, regionB) with cost distU + w + distV;
                // match greedily by cost. Each round roughly halves the unmatched set.
                var owner = new int[N];
                var unmatched = new bool[N];
                var cur = new int[oddCount];
                Array.Copy(odd, cur, oddCount);
                int curCount = oddCount;
                foreach (int v in odd) unmatched[v] = true;
                int rounds = 0;
                while (curCount > 1 && rounds++ < 100)
                {
                    // multi-source Dijkstra
                    curVersion++;
                    heapSize = 0;
                    for (int i = 0; i < curCount; i++)
                    {
                        int s = cur[i];
                        SetDist(s, 0, -1);
                        owner[s] = s;
                        HeapPush(0, s);
                    }
                    while (heapSize > 0)
                    {
                        long d = heapKey[0]; int v = heapNode[0]; HeapPop();
                        if (d > GetDist(v)) continue;
                        int ov = owner[v];
                        for (int a = head[v]; a != -1; a = nxt[a])
                        {
                            int u = to[a];
                            long nd = d + ew[eid[a]];
                            if (nd < GetDist(u))
                            {
                                SetDist(u, nd, a);
                                owner[u] = ov;
                                HeapPush(nd, u);
                            }
                        }
                    }
                    // collect boundary-edge candidates
                    int snapEdges = edgeCount;
                    var candKey = new long[snapEdges];
                    var candEdge = new int[snapEdges];
                    int cc = 0;
                    for (int e = 0; e < snapEdges; e++)
                    {
                        int v = to[2 * e], u = to[2 * e + 1];
                        if (version[u] != curVersion || version[v] != curVersion) continue;
                        if (owner[u] == owner[v]) continue;
                        candKey[cc] = dist[u] + dist[v] + ew[e];
                        candEdge[cc] = e;
                        cc++;
                    }
                    Array.Sort(candKey, candEdge, 0, cc);
                    int matchedInRound = 0;
                    for (int i = 0; i < cc; i++)
                    {
                        int e = candEdge[i];
                        int v = to[2 * e], u = to[2 * e + 1];
                        int a = owner[u], b = owner[v];
                        if (!unmatched[a] || !unmatched[b]) continue;
                        unmatched[a] = false; unmatched[b] = false;
                        matchedInRound += 2;
                        // duplicate the path: a ..parents.. u -(e)- v ..parents.. b
                        AddEdge(u, v, ew[e]);
                        int x = u;
                        while (parentArc[x] != -1) { int pa = parentArc[x]; AddEdge(to[pa ^ 1], x, ew[eid[pa]]); x = to[pa ^ 1]; }
                        x = v;
                        while (parentArc[x] != -1) { int pa = parentArc[x]; AddEdge(to[pa ^ 1], x, ew[eid[pa]]); x = to[pa ^ 1]; }
                    }
                    if (matchedInRound == 0) break; // remaining odds mutually unreachable
                    int k = 0;
                    for (int i = 0; i < curCount; i++) if (unmatched[cur[i]]) cur[k++] = cur[i];
                    curCount = k;
                }
                // fallback for any stragglers (rare): sequential early-stop Dijkstra
                for (int i = 0; i < curCount; i++)
                {
                    int s = cur[i];
                    if (!unmatched[s]) continue;
                    unmatched[s] = false;
                    int t = Dijkstra(s, unmatched);
                    if (t == -1) { continue; }
                    unmatched[t] = false;
                    DuplicatePathFromCurrent(t);
                }
            }
        }

        // ---- Hierholzer's algorithm (iterative) ----
        int totalEdges = edgeCount;
        var used = new bool[totalEdges];
        var ptr = new int[N];
        for (int v = 0; v < N; v++) ptr[v] = head[v];
        var stack = new int[totalEdges + 2];
        var circuit = new int[totalEdges + 2];
        int sp = 0, cp = 0;
        stack[sp++] = 0;
        while (sp > 0)
        {
            int v = stack[sp - 1];
            int a = ptr[v];
            while (a != -1 && used[eid[a]]) a = nxt[a];
            ptr[v] = a;
            if (a == -1)
            {
                circuit[cp++] = v;
                sp--;
            }
            else
            {
                used[eid[a]] = true;
                stack[sp++] = to[a];
            }
        }

        // ---- output ----
        stdout.Write(cp);
        stdout.Write('\n');
        var sb = new System.Text.StringBuilder(1 << 16);
        for (int i = cp - 1; i >= 0; i--)
        {
            sb.Append(circuit[i]);
            sb.Append(i > 0 ? ' ' : '\n');
            if (sb.Length > (1 << 15)) { stdout.Write(sb.ToString()); sb.Clear(); }
        }
        stdout.Write(sb.ToString());
        stdout.Flush();
    }
}
