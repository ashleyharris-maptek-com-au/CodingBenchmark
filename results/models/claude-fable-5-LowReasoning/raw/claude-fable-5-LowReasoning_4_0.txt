// Tetrahedron packing into a polyhedron (convex hull of given vertices).
// Strategy: pack the circumspheres of the tetrahedra on an HCP lattice.
// Since a regular tetrahedron (any orientation, centered at its centroid)
// fits inside its circumsphere of radius R = e*sqrt(3/8), non-overlapping
// spheres guarantee non-overlapping tetrahedra regardless of the canonical
// orientation used by the checker. All Euler angles are 0.
//
// Containment: a tetrahedron whose circumsphere lies inside the container
// is itself inside. We test the sphere center against every hull plane.
//
// Scalability: grid iteration is capped by wall-clock time and a maximum
// output count, so ludicrous containers still finish within the limit.

#include <cstdio>
#include <cstdlib>
#include <cmath>
#include <vector>
#include <chrono>
#include <algorithm>

struct Vec3 { double x, y, z; };
static inline Vec3 sub(const Vec3&a,const Vec3&b){return {a.x-b.x,a.y-b.y,a.z-b.z};}
static inline Vec3 cross(const Vec3&a,const Vec3&b){return {a.y*b.z-a.z*b.y,a.z*b.x-a.x*b.z,a.x*b.y-a.y*b.x};}
static inline double dot(const Vec3&a,const Vec3&b){return a.x*b.x+a.y*b.y+a.z*b.z;}

struct Plane { double nx, ny, nz, off; }; // inside: n.p - off <= 0

int main() {
    int V;
    double e;
    if (scanf("%d %lf", &V, &e) != 2 || V < 4 || e <= 0.0) { printf("0\n"); return 0; }
    std::vector<Vec3> P(V);
    for (int i = 0; i < V; ++i)
        if (scanf("%lf %lf %lf", &P[i].x, &P[i].y, &P[i].z) != 3) { printf("0\n"); return 0; }

    auto t0 = std::chrono::steady_clock::now();
    auto elapsed = [&]{ return std::chrono::duration<double>(std::chrono::steady_clock::now()-t0).count(); };

    // Bounding box
    Vec3 lo = P[0], hi = P[0];
    for (auto&p : P) {
        lo.x = std::min(lo.x,p.x); lo.y = std::min(lo.y,p.y); lo.z = std::min(lo.z,p.z);
        hi.x = std::max(hi.x,p.x); hi.y = std::max(hi.y,p.y); hi.z = std::max(hi.z,p.z);
    }
    double scale = std::sqrt((hi.x-lo.x)*(hi.x-lo.x)+(hi.y-lo.y)*(hi.y-lo.y)+(hi.z-lo.z)*(hi.z-lo.z));
    if (scale <= 0.0) { printf("0\n"); return 0; }

    // Convex hull planes by brute-force triple enumeration (V expected modest).
    // Dedupe near-identical planes.
    std::vector<Plane> planes;
    const double sideTol = 1e-9 * scale;
    // Limit work for very large V: sample triples deterministically if needed.
    long long tripleBudget = 40000000LL;
    long long triplesTotal = (long long)V*(V-1)*(V-2)/6;
    int stride = 1;
    while (triplesTotal / ((long long)stride*stride*stride) > tripleBudget) stride++;
    for (int i = 0; i < V; i += stride)
    for (int j = i+1; j < V; j += stride)
    for (int k = j+1; k < V; k += stride) {
        Vec3 n = cross(sub(P[j],P[i]), sub(P[k],P[i]));
        double len = std::sqrt(dot(n,n));
        if (len < 1e-12 * scale * scale) continue;
        n.x/=len; n.y/=len; n.z/=len;
        double off = dot(n, P[i]);
        bool pos=false, neg=false;
        for (int m = 0; m < V; ++m) {
            double s = dot(n, P[m]) - off;
            if (s > sideTol) pos = true; else if (s < -sideTol) neg = true;
            if (pos && neg) break;
        }
        if (pos && neg) continue;
        if (pos) { n.x=-n.x; n.y=-n.y; n.z=-n.z; off=-off; }
        bool dup = false;
        for (auto&pl : planes)
            if (std::fabs(pl.nx-n.x)<1e-7 && std::fabs(pl.ny-n.y)<1e-7 &&
                std::fabs(pl.nz-n.z)<1e-7 && std::fabs(pl.off-off)<1e-6*scale) { dup = true; break; }
        if (!dup) planes.push_back({n.x,n.y,n.z,off});
    }
    if (planes.empty()) { printf("0\n"); return 0; }

    // Circumsphere radius with safety margin.
    const double R = e * std::sqrt(3.0/8.0);
    const double Rneed = R * (1.0 + 1e-7) + 1e-9 * scale;
    const double d  = 2.0 * R * (1.0 + 1e-6);   // sphere center spacing
    const double dy = d * std::sqrt(3.0) / 2.0; // row spacing
    const double dz = d * std::sqrt(2.0/3.0);   // layer spacing

    const size_t MAX_OUT = 2000000;
    const double TIME_LIMIT = 24.0;
    std::vector<Vec3> out;
    out.reserve(1 << 16);

    long long nz = (long long)std::floor((hi.z - lo.z) / dz) + 1;
    long long ny = (long long)std::floor((hi.y - lo.y) / dy) + 1;
    long long nx = (long long)std::floor((hi.x - lo.x) / d)  + 1;

    bool stopped = false;
    for (long long iz = 0; iz < nz && !stopped; ++iz) {
        double cz = lo.z + Rneed + iz * dz;
        if (cz > hi.z) break;
        int layer = (int)(iz & 1);
        double yoff = layer ? d * std::sqrt(3.0)/6.0 : 0.0;
        double xoffL = layer ? d * 0.5 : 0.0;
        for (long long iy = 0; iy < ny && !stopped; ++iy) {
            double cy = lo.y + Rneed + yoff + iy * dy;
            if (cy > hi.y) break;
            double xoff = xoffL + ((iy & 1) ? d * 0.5 : 0.0);
            if (elapsed() > TIME_LIMIT) { stopped = true; break; }
            for (long long ix = 0; ix < nx; ++ix) {
                double cx = lo.x + Rneed + xoff + ix * d;
                if (cx > hi.x) break;
                bool ok = true;
                for (auto&pl : planes) {
                    if (pl.nx*cx + pl.ny*cy + pl.nz*cz - pl.off > -Rneed) { ok = false; break; }
                }
                if (ok) {
                    out.push_back({cx, cy, cz});
                    if (out.size() >= MAX_OUT) { stopped = true; break; }
                }
            }
        }
    }

    // Output (buffered)
    std::vector<char> buf;
    buf.reserve(out.size() * 64 + 64);
    char line[128];
    int n = snprintf(line, sizeof(line), "%zu\n", out.size());
    buf.insert(buf.end(), line, line + n);
    for (auto&c : out) {
        n = snprintf(line, sizeof(line), "%.9g %.9g %.9g 0 0 0\n", c.x, c.y, c.z);
        buf.insert(buf.end(), line, line + n);
    }
    fwrite(buf.data(), 1, buf.size(), stdout);
    return 0;
}
