// Hamiltonian path / cycle solver on 2D grids with obstacles.
//
// Strategy (adaptive):
//  - Obstacle-free grids: O(N) direct serpentine construction (path) or
//    reserved-column snake (cycle) -> instant at any scale.
//  - Static feasibility checks: bipartite parity counts, connectivity of the
//    free cells from (0,0), degree-1 dead-end counting.
//  - With obstacles: Posa rotation-extension heuristic (randomized, with
//    restarts) — extremely effective on grid graphs, handles 2500-cell
//    grids in well under the time limit.
//  - Exact fallback: DFS backtracking with heavy pruning (connectivity
//    flood-fill, parity, forced-endpoint analysis) to prove impossibility
//    on small / tightly constrained instances.

use std::io::{self, Read, Write};
use std::time::Instant;

const TOTAL_BUDGET: f64 = 280.0; // seconds, hard cap well under 5 min

struct Rng(u64);
impl Rng {
    fn next(&mut self) -> u64 {
        let mut x = self.0;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.0 = x;
        x
    }
    fn below(&mut self, n: usize) -> usize {
        (self.next() % n as u64) as usize
    }
}

struct Grid {
    w: i64,
    h: i64,
    blocked: Vec<bool>,
    n_free: usize,
    start: usize,
}

impl Grid {
    fn neighbors(&self, idx: usize, out: &mut [usize; 4]) -> usize {
        let x = (idx as i64) % self.w;
        let y = (idx as i64) / self.w;
        let mut n = 0;
        if x > 0 {
            out[n] = idx - 1;
            n += 1;
        }
        if x + 1 < self.w {
            out[n] = idx + 1;
            n += 1;
        }
        if y > 0 {
            out[n] = idx - self.w as usize;
            n += 1;
        }
        if y + 1 < self.h {
            out[n] = idx + self.w as usize;
            n += 1;
        }
        n
    }

    fn free_neighbors(&self, idx: usize, out: &mut [usize; 4]) -> usize {
        let mut nb = [0usize; 4];
        let cnt = self.neighbors(idx, &mut nb);
        let mut m = 0;
        for i in 0..cnt {
            if !self.blocked[nb[i]] {
                out[m] = nb[i];
                m += 1;
            }
        }
        m
    }

    fn adjacent(&self, a: usize, b: usize) -> bool {
        let ax = (a as i64) % self.w;
        let ay = (a as i64) / self.w;
        let bx = (b as i64) % self.w;
        let by = (b as i64) / self.w;
        (ax - bx).abs() + (ay - by).abs() == 1
    }

    fn color(&self, idx: usize) -> usize {
        let x = (idx as i64) % self.w;
        let y = (idx as i64) / self.w;
        ((x + y) & 1) as usize
    }
}

// ---------------- Posa rotation-extension heuristic ----------------
// Maintains a simple path with the start pinned at (0,0). At the free end:
// extend into an unvisited neighbor when possible, otherwise rotate (pick a
// visited neighbor u at position i; reverse the suffix after i, making
// path[i+1] the new free end). Restart when stuck / stagnant.
fn posa(g: &Grid, cycle: bool, rng: &mut Rng, t0: &Instant, deadline: f64) -> Option<Vec<usize>> {
    let cells = g.blocked.len();
    let n = g.n_free;
    let mut pos: Vec<usize> = vec![usize::MAX; cells];
    let mut path: Vec<usize> = Vec::with_capacity(n);
    let stagnation_limit = 200 * n as u64 + 20_000;

    loop {
        // (re)start
        for p in pos.iter_mut() {
            *p = usize::MAX;
        }
        path.clear();
        path.push(g.start);
        pos[g.start] = 0;
        let mut best = 1usize;
        let mut stagnant: u64 = 0;
        let mut ticks: u64 = 0;

        loop {
            ticks += 1;
            if ticks % 4096 == 0 && t0.elapsed().as_secs_f64() > deadline {
                return None;
            }

            let e = *path.last().unwrap();

            if path.len() == n {
                if !cycle || g.adjacent(e, g.start) {
                    return Some(path);
                }
                // Hamiltonian path found but must close the cycle: keep
                // rotating (fall through; no extension is possible).
            } else {
                // try to extend
                let mut nb = [0usize; 4];
                let m = g.free_neighbors(e, &mut nb);
                let mut ext = [0usize; 4];
                let mut k = 0;
                for i in 0..m {
                    if pos[nb[i]] == usize::MAX {
                        ext[k] = nb[i];
                        k += 1;
                    }
                }
                if k > 0 {
                    // Warnsdorff-ish bias: prefer the extension with the
                    // fewest unvisited neighbors, random tie-break.
                    let mut bestd = 5usize;
                    let mut choice = ext[0];
                    let mut ties = 0usize;
                    for i in 0..k {
                        let v = ext[i];
                        let mut vb = [0usize; 4];
                        let vm = g.free_neighbors(v, &mut vb);
                        let mut d = 0;
                        for j in 0..vm {
                            if pos[vb[j]] == usize::MAX {
                                d += 1;
                            }
                        }
                        if d < bestd {
                            bestd = d;
                            choice = v;
                            ties = 1;
                        } else if d == bestd {
                            ties += 1;
                            if rng.below(ties) == 0 {
                                choice = v;
                            }
                        }
                    }
                    pos[choice] = path.len();
                    path.push(choice);
                    if path.len() > best {
                        best = path.len();
                        stagnant = 0;
                    }
                    continue;
                }
            }

            // rotation: pick a visited neighbor u of e (not the predecessor)
            let mut nb = [0usize; 4];
            let m = g.free_neighbors(e, &mut nb);
            let pred = if path.len() >= 2 {
                path[path.len() - 2]
            } else {
                usize::MAX
            };
            let mut rot = [0usize; 4];
            let mut k = 0;
            for i in 0..m {
                let u = nb[i];
                if u != pred && pos[u] != usize::MAX {
                    rot[k] = u;
                    k += 1;
                }
            }
            if k == 0 {
                break; // dead end at the tip with nothing to rotate: restart
            }
            let u = rot[rng.below(k)];
            let i = pos[u];
            // reverse path[i+1..]
            let (mut a, mut b) = (i + 1, path.len() - 1);
            while a < b {
                path.swap(a, b);
                pos[path[a]] = a;
                pos[path[b]] = b;
                a += 1;
                b -= 1;
            }
            if a == b {
                pos[path[a]] = a;
            }
            stagnant += 1;
            if stagnant > stagnation_limit {
                break; // restart
            }
        }

        if t0.elapsed().as_secs_f64() > deadline {
            return None;
        }
    }
}

// ---------------- Exact DFS with pruning ----------------
struct Dfs<'a> {
    g: &'a Grid,
    visited: Vec<bool>,
    path: Vec<usize>,
    cycle: bool,
    rem: [i64; 2],
    mark: Vec<u32>,
    stamp: u32,
    stack: Vec<usize>,
    t0: &'a Instant,
    deadline: f64,
    nodes: u64,
    timed_out: bool,
}

impl<'a> Dfs<'a> {
    fn rdeg(&self, idx: usize) -> usize {
        let mut nb = [0usize; 4];
        let cnt = self.g.free_neighbors(idx, &mut nb);
        let mut d = 0;
        for i in 0..cnt {
            if !self.visited[nb[i]] {
                d += 1;
            }
        }
        d
    }

    fn prune_ok(&mut self, c: usize) -> bool {
        let r = (self.g.n_free - self.path.len()) as i64;
        let s = 1 - self.g.color(c);
        if self.rem[s] != (r + 1) / 2 || self.rem[1 - s] != r / 2 {
            return false;
        }
        if self.cycle {
            let mut nb = [0usize; 4];
            let cnt = self.g.free_neighbors(self.g.start, &mut nb);
            let mut ok = false;
            for i in 0..cnt {
                if !self.visited[nb[i]] {
                    ok = true;
                    break;
                }
            }
            if !ok {
                return false;
            }
        }
        // connectivity of remaining cells from c
        self.stamp += 1;
        let stamp = self.stamp;
        self.stack.clear();
        let mut nb = [0usize; 4];
        let cnt = self.g.free_neighbors(c, &mut nb);
        let mut reached: i64 = 0;
        for i in 0..cnt {
            let m = nb[i];
            if !self.visited[m] && self.mark[m] != stamp {
                self.mark[m] = stamp;
                self.stack.push(m);
                reached += 1;
            }
        }
        if reached == 0 {
            return false;
        }
        while let Some(u) = self.stack.pop() {
            let ucnt = self.g.free_neighbors(u, &mut nb);
            for i in 0..ucnt {
                let m = nb[i];
                if !self.visited[m] && self.mark[m] != stamp {
                    self.mark[m] = stamp;
                    self.stack.push(m);
                    reached += 1;
                }
            }
        }
        if reached != r {
            return false;
        }
        // forced endpoint analysis
        let mut forced = 0i64;
        for idx in 0..self.g.blocked.len() {
            if self.g.blocked[idx] || self.visited[idx] {
                continue;
            }
            let d = self.rdeg(idx);
            let adj_c = self.g.adjacent(idx, c);
            if d == 0 {
                if !(adj_c && r == 1) {
                    return false;
                }
            } else if d == 1 && !adj_c {
                forced += 1;
                if forced > 1 {
                    return false;
                }
                if self.cycle && !self.g.adjacent(idx, self.g.start) {
                    return false;
                }
            }
        }
        true
    }

    // Returns Some(true) found, Some(false) exhausted (proven none from
    // here), None if timed out (unknown).
    fn dfs(&mut self, c: usize) -> Option<bool> {
        self.nodes += 1;
        if self.nodes % 1024 == 0 && self.t0.elapsed().as_secs_f64() > self.deadline {
            self.timed_out = true;
        }
        if self.timed_out {
            return None;
        }
        self.path.push(c);
        self.visited[c] = true;
        self.rem[self.g.color(c)] -= 1;

        let mut result = Some(false);
        if self.path.len() == self.g.n_free {
            if !self.cycle || self.g.n_free == 1 || self.g.adjacent(c, self.g.start) {
                return Some(true);
            }
        } else if self.prune_ok(c) {
            let mut nb = [0usize; 4];
            let cnt = self.g.free_neighbors(c, &mut nb);
            let mut cand: [(usize, usize); 4] = [(0, 0); 4];
            let mut m = 0;
            for i in 0..cnt {
                let v = nb[i];
                if !self.visited[v] {
                    cand[m] = (self.rdeg(v), v);
                    m += 1;
                }
            }
            cand[..m].sort_unstable();
            for i in 0..m {
                match self.dfs(cand[i].1) {
                    Some(true) => return Some(true),
                    Some(false) => {}
                    None => {
                        result = None;
                        break;
                    }
                }
            }
        }

        self.rem[self.g.color(c)] += 1;
        self.visited[c] = false;
        self.path.pop();
        result
    }
}

// ---------------- direct constructions for full grids ----------------
fn serpentine_path(w: i64, h: i64, out: &mut Vec<(i64, i64)>) {
    for y in 0..h {
        if y % 2 == 0 {
            for x in 0..w {
                out.push((x, y));
            }
        } else {
            for x in (0..w).rev() {
                out.push((x, y));
            }
        }
    }
}

fn full_cycle_h_even(w: i64, h: i64, out: &mut Vec<(i64, i64)>) {
    for x in 0..w {
        out.push((x, 0));
    }
    for y in 1..h {
        if y % 2 == 1 {
            for x in (1..w).rev() {
                out.push((x, y));
            }
        } else {
            for x in 1..w {
                out.push((x, y));
            }
        }
    }
    for y in (1..h).rev() {
        out.push((0, y));
    }
}

fn main() {
    let t0 = Instant::now();
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input
        .split_ascii_whitespace()
        .map(|t| t.parse::<i64>().unwrap());
    let w = it.next().unwrap();
    let h = it.next().unwrap();
    let k = it.next().unwrap();
    let cycle = it.next().unwrap() != 0;

    let stdout = io::stdout();
    let mut out = io::BufWriter::new(stdout.lock());

    if w <= 0 || h <= 0 {
        writeln!(out, "0").unwrap();
        return;
    }

    let cells = (w * h) as usize;
    let mut blocked = vec![false; cells];
    let mut n_blocked = 0usize;
    for _ in 0..k {
        let x = it.next().unwrap();
        let y = it.next().unwrap();
        if x >= 0 && x < w && y >= 0 && y < h {
            let idx = (y * w + x) as usize;
            if !blocked[idx] {
                blocked[idx] = true;
                n_blocked += 1;
            }
        }
    }
    let n_free = cells - n_blocked;

    if blocked[0] {
        writeln!(out, "0").unwrap();
        return;
    }

    if n_free == 1 {
        if cycle {
            writeln!(out, "0").unwrap(); // a cell is not adjacent to itself
        } else {
            writeln!(out, "1\n0 0").unwrap();
        }
        return;
    }

    let mut result: Vec<(i64, i64)> = Vec::new();

    if n_blocked == 0 {
        if !cycle {
            serpentine_path(w, h, &mut result);
        } else if w == 1 || h == 1 || (w * h) % 2 != 0 {
            writeln!(out, "0").unwrap();
            return;
        } else if h % 2 == 0 {
            full_cycle_h_even(w, h, &mut result);
        } else {
            let mut t = Vec::with_capacity(cells);
            full_cycle_h_even(h, w, &mut t);
            for (x, y) in t {
                result.push((y, x));
            }
        }
    } else {
        let g = Grid {
            w,
            h,
            blocked,
            n_free,
            start: 0,
        };

        // ---- static feasibility checks ----
        // parity
        let mut cnt = [0i64; 2];
        for idx in 0..cells {
            if !g.blocked[idx] {
                cnt[g.color(idx)] += 1;
            }
        }
        let n = n_free as i64;
        let feasible = if cycle {
            cnt[0] == cnt[1] && n % 2 == 0
        } else {
            // start (0,0) has color 0
            cnt[0] == (n + 1) / 2 && cnt[1] == n / 2
        };
        if !feasible {
            writeln!(out, "0").unwrap();
            return;
        }
        // connectivity of free cells from start
        {
            let mut seen = vec![false; cells];
            let mut st = vec![g.start];
            seen[g.start] = true;
            let mut c = 1usize;
            let mut nb = [0usize; 4];
            while let Some(u) = st.pop() {
                let m = g.free_neighbors(u, &mut nb);
                for i in 0..m {
                    if !seen[nb[i]] {
                        seen[nb[i]] = true;
                        st.push(nb[i]);
                        c += 1;
                    }
                }
            }
            if c != n_free {
                writeln!(out, "0").unwrap();
                return;
            }
        }
        // degree-1 dead ends
        {
            let mut nb = [0usize; 4];
            let mut deg1_non_start = 0i64;
            for idx in 0..cells {
                if g.blocked[idx] {
                    continue;
                }
                let d = g.free_neighbors(idx, &mut nb);
                if d <= 1 {
                    if cycle {
                        writeln!(out, "0").unwrap();
                        return;
                    }
                    if idx != g.start {
                        deg1_non_start += 1;
                    }
                }
            }
            if !cycle && deg1_non_start > 1 {
                writeln!(out, "0").unwrap();
                return;
            }
        }

        // ---- solve ----
        let mut rng = Rng(0x9E3779B97F4A7C15);
        let mut solved: Option<Vec<usize>> = None;
        let mut proven_none = false;

        if n_free <= 120 {
            // small: exact DFS first (fast with pruning, proves infeasibility)
            let mut d = Dfs {
                g: &g,
                visited: vec![false; cells],
                path: Vec::with_capacity(n_free),
                cycle,
                rem: cnt,
                mark: vec![0u32; cells],
                stamp: 0,
                stack: Vec::with_capacity(cells),
                t0: &t0,
                deadline: 60.0,
                nodes: 0,
                timed_out: false,
            };
            match d.dfs(0) {
                Some(true) => solved = Some(d.path),
                Some(false) => proven_none = true,
                None => {}
            }
        }

        if solved.is_none() && !proven_none {
            solved = posa(&g, cycle, &mut rng, &t0, TOTAL_BUDGET - 25.0);
        }

        if solved.is_none() && !proven_none && n_free > 120 {
            // last-resort exact attempt with remaining time
            let mut d = Dfs {
                g: &g,
                visited: vec![false; cells],
                path: Vec::with_capacity(n_free),
                cycle,
                rem: cnt,
                mark: vec![0u32; cells],
                stamp: 0,
                stack: Vec::with_capacity(cells),
                t0: &t0,
                deadline: TOTAL_BUDGET,
                nodes: 0,
                timed_out: false,
            };
            if let Some(true) = d.dfs(0) {
                solved = Some(d.path);
            }
        }

        match solved {
            Some(p) => {
                for idx in p {
                    result.push((idx as i64 % w, idx as i64 / w));
                }
            }
            None => {
                writeln!(out, "0").unwrap();
                return;
            }
        }
    }

    writeln!(out, "{}", result.len()).unwrap();
    let mut buf = String::with_capacity(result.len() * 8);
    for (x, y) in result {
        buf.push_str(&x.to_string());
        buf.push(' ');
        buf.push_str(&y.to_string());
        buf.push('\n');
    }
    out.write_all(buf.as_bytes()).unwrap();
}
