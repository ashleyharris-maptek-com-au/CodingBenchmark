// Adaptive TSP solver: scales from 10 to 10M+ cities.
// Strategy:
//   - Read input with a fast custom parser.
//   - Build an initial tour via Hilbert space-filling curve sort (O(N log N)).
//   - For small/medium N, refine with 2-opt + Or-opt using k-nearest-neighbor
//     candidate lists (built from a uniform grid), under a wall-clock budget.
//   - Rotate tour so it starts at city 0.
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <cstdint>
#include <cmath>
#include <vector>
#include <algorithm>
#include <chrono>
#include <thread>
#include <execution>

#ifdef _WIN32
#include <io.h>
#include <fcntl.h>
#endif

using namespace std;
using Clock = chrono::steady_clock;
static Clock::time_point g_start;
static double elapsed() {
    return chrono::duration<double>(Clock::now() - g_start).count();
}

static const double TIME_LIMIT = 250.0; // safety margin under 300s

// ---------- fast input ----------
static char* readAll(size_t& len) {
#ifdef _WIN32
    _setmode(_fileno(stdin), _O_BINARY);
#endif
    size_t cap = 1 << 24, n = 0;
    char* buf = (char*)malloc(cap);
    for (;;) {
        if (n + (1 << 22) > cap) { cap <<= 1; buf = (char*)realloc(buf, cap); }
        size_t r = fread(buf + n, 1, 1 << 22, stdin);
        n += r;
        if (r == 0) break;
    }
    len = n;
    return buf;
}

static inline double parseDouble(const char*& p, const char* end) {
    while (p < end && (*p == ' ' || *p == '\n' || *p == '\r' || *p == '\t' || *p == ',')) ++p;
    bool neg = false;
    if (p < end && (*p == '-' || *p == '+')) { neg = (*p == '-'); ++p; }
    double v = 0;
    while (p < end && *p >= '0' && *p <= '9') { v = v * 10 + (*p - '0'); ++p; }
    if (p < end && *p == '.') {
        ++p;
        double f = 0.1;
        while (p < end && *p >= '0' && *p <= '9') { v += (*p - '0') * f; f *= 0.1; ++p; }
    }
    if (p < end && (*p == 'e' || *p == 'E')) {
        ++p;
        bool en = false;
        if (p < end && (*p == '-' || *p == '+')) { en = (*p == '-'); ++p; }
        int ex = 0;
        while (p < end && *p >= '0' && *p <= '9') { ex = ex * 10 + (*p - '0'); ++p; }
        v *= pow(10.0, en ? -ex : ex);
    }
    return neg ? -v : v;
}

// ---------- geometry ----------
static vector<float> X, Y;
static inline double dist(int a, int b) {
    double dx = (double)X[a] - X[b], dy = (double)Y[a] - Y[b];
    return sqrt(dx * dx + dy * dy);
}

// ---------- Hilbert curve ----------
static inline uint64_t hilbert_d(uint32_t x, uint32_t y, int order) {
    uint64_t rx, ry, d = 0;
    for (uint32_t s = 1u << (order - 1); s > 0; s >>= 1) {
        rx = (x & s) > 0;
        ry = (y & s) > 0;
        d += (uint64_t)s * s * ((3 * rx) ^ ry);
        // rotate
        if (ry == 0) {
            if (rx == 1) { x = s - 1 - x; y = s - 1 - y; }
            uint32_t t = x; x = y; y = t;
        }
    }
    return d;
}

int main() {
    g_start = Clock::now();
    size_t len;
    char* buf = readAll(len);
    const char* p = buf;
    const char* end = buf + len;

    long long Nll = (long long)parseDouble(p, end);
    int N = (int)Nll;
    if (N <= 0) { printf("0\n"); return 0; }

    X.resize(N); Y.resize(N);
    double minx = 1e300, maxx = -1e300, miny = 1e300, maxy = -1e300;
    for (int i = 0; i < N; ++i) {
        double x = parseDouble(p, end);
        double y = parseDouble(p, end);
        X[i] = (float)x; Y[i] = (float)y;
        if (x < minx) minx = x; if (x > maxx) maxx = x;
        if (y < miny) miny = y; if (y > maxy) maxy = y;
    }
    free(buf);

    vector<int> tour(N);
    if (N == 1) { printf("0\n"); return 0; }
    if (N == 2) { printf("0 1\n"); return 0; }

    // ---- initial tour: Hilbert sort ----
    {
        double sx = (maxx > minx) ? (double)((1u << 16) - 1) / (maxx - minx) : 0.0;
        double sy = (maxy > miny) ? (double)((1u << 16) - 1) / (maxy - miny) : 0.0;
        vector<uint64_t> key(N);
        auto keyRange = [&](int lo, int hi) {
            for (int i = lo; i < hi; ++i) {
                uint32_t hx = (uint32_t)((X[i] - minx) * sx);
                uint32_t hy = (uint32_t)((Y[i] - miny) * sy);
                key[i] = hilbert_d(hx, hy, 16);
            }
        };
        unsigned nt = max(1u, thread::hardware_concurrency());
        if (N > 100000 && nt > 1) {
            vector<thread> th;
            int chunk = (N + (int)nt - 1) / (int)nt;
            for (unsigned t = 0; t < nt; ++t) {
                int lo = (int)t * chunk, hi = min(N, lo + chunk);
                if (lo < hi) th.emplace_back(keyRange, lo, hi);
            }
            for (auto& t : th) t.join();
        } else {
            keyRange(0, N);
        }
        for (int i = 0; i < N; ++i) tour[i] = i;
        auto cmp = [&](int a, int b) {
            if (key[a] != key[b]) return key[a] < key[b];
            return a < b;
        };
        if (N > 100000) sort(execution::par_unseq, tour.begin(), tour.end(), cmp);
        else sort(tour.begin(), tour.end(), cmp);
    }

    // ---- local improvement for feasible sizes ----
    // Build k-NN candidate lists via a uniform grid, then 2-opt + Or-opt.
    bool doOpt = (N <= 12000000);
    int K = (N <= 2000) ? 16 : (N <= 100000) ? 10 : (N <= 500000) ? 8 : 6;
    // caps so no single move degenerates to O(N) on huge instances
    int maxRev = (N > 2000000) ? 100000 : N;
    int maxShift = (N > 2000000) ? 2000 : N;

    if (doOpt && elapsed() < TIME_LIMIT * 0.5) {
        // grid
        int gridDim = (int)ceil(sqrt((double)N / 2.0));
        if (gridDim < 1) gridDim = 1;
        if (gridDim > 4096) gridDim = 4096;
        double cw = (maxx > minx) ? (maxx - minx) / gridDim : 1.0;
        double ch = (maxy > miny) ? (maxy - miny) / gridDim : 1.0;
        if (cw <= 0) cw = 1.0;
        if (ch <= 0) ch = 1.0;
        auto cellOf = [&](int i, int& cx, int& cy) {
            cx = (int)((X[i] - minx) / cw); if (cx >= gridDim) cx = gridDim - 1; if (cx < 0) cx = 0;
            cy = (int)((Y[i] - miny) / ch); if (cy >= gridDim) cy = gridDim - 1; if (cy < 0) cy = 0;
        };
        int nc = gridDim * gridDim;
        vector<int> cnt(nc + 1, 0), cellPts(N);
        for (int i = 0; i < N; ++i) { int cx, cy; cellOf(i, cx, cy); cnt[cy * gridDim + cx + 1]++; }
        for (int i = 0; i < nc; ++i) cnt[i + 1] += cnt[i];
        {
            vector<int> fill(cnt.begin(), cnt.end() - 1);
            for (int i = 0; i < N; ++i) {
                int cx, cy; cellOf(i, cx, cy);
                cellPts[fill[cy * gridDim + cx]++] = i;
            }
        }

        // k nearest neighbors via ring search around each point's cell
        vector<int> nbr((size_t)N * K);
        auto knnRange = [&](int ilo, int ihi) {
        vector<pair<float, int>> cand;
        cand.reserve(256);
        for (int i = ilo; i < ihi; ++i) {
            int cx, cy; cellOf(i, cx, cy);
            cand.clear();
            int ring = 0;
            while ((int)cand.size() < K + 1 && ring <= gridDim) {
                int x0 = cx - ring, x1 = cx + ring, y0 = cy - ring, y1 = cy + ring;
                for (int gy = y0; gy <= y1; ++gy) {
                    if (gy < 0 || gy >= gridDim) continue;
                    for (int gx = x0; gx <= x1; ++gx) {
                        if (gx < 0 || gx >= gridDim) continue;
                        if (ring > 0 && gx != x0 && gx != x1 && gy != y0 && gy != y1) continue;
                        int c = gy * gridDim + gx;
                        for (int t = cnt[c]; t < cnt[c + 1]; ++t) {
                            int j = cellPts[t];
                            if (j == i) continue;
                            cand.push_back({ (float)dist(i, j), j });
                        }
                    }
                }
                ++ring;
            }
            // one extra ring for correctness of nearness
            {
                int x0 = cx - ring, x1 = cx + ring, y0 = cy - ring, y1 = cy + ring;
                for (int gy = y0; gy <= y1; ++gy) {
                    if (gy < 0 || gy >= gridDim) continue;
                    for (int gx = x0; gx <= x1; ++gx) {
                        if (gx < 0 || gx >= gridDim) continue;
                        if (gx != x0 && gx != x1 && gy != y0 && gy != y1) continue;
                        int c = gy * gridDim + gx;
                        for (int t = cnt[c]; t < cnt[c + 1]; ++t) {
                            int j = cellPts[t];
                            if (j == i) continue;
                            cand.push_back({ (float)dist(i, j), j });
                        }
                    }
                }
            }
            int kk = (int)min((size_t)K, cand.size());
            partial_sort(cand.begin(), cand.begin() + kk, cand.end());
            for (int t = 0; t < K; ++t)
                nbr[(size_t)i * K + t] = (t < kk) ? cand[t].second : -1;
        }
        };
        {
            unsigned nt = max(1u, thread::hardware_concurrency());
            if (N > 20000 && nt > 1) {
                vector<thread> th;
                int chunk = (N + (int)nt - 1) / (int)nt;
                for (unsigned t = 0; t < nt; ++t) {
                    int lo = (int)t * chunk, hi = min(N, lo + chunk);
                    if (lo < hi) th.emplace_back(knnRange, lo, hi);
                }
                for (auto& t : th) t.join();
            } else {
                knnRange(0, N);
            }
        }

        // position array
        vector<int> pos(N);
        for (int i = 0; i < N; ++i) pos[tour[i]] = i;

        auto tnext = [&](int i) { return tour[(pos[i] + 1) % N]; };
        auto tprev = [&](int i) { return tour[(pos[i] + N - 1) % N]; };

        // 2-opt with neighbor lists + don't-look bits; segment reversal picks shorter side
        vector<uint8_t> dontlook(N, 0);
        double budget = min(TIME_LIMIT, (N <= 100000) ? TIME_LIMIT : 200.0);
        bool improvedAny = true;
        int rounds = 0;
        while (improvedAny && elapsed() < budget) {
            improvedAny = false;
            ++rounds;
            for (int a = 0; a < N; ++a) {
                if (dontlook[a]) continue;
                if ((a & 1023) == 0 && elapsed() > budget) break;
                bool improved = false;

                // 2-opt: try both tour edges at a
                for (int dir = 0; dir < 2 && !improved; ++dir) {
                    int b = dir == 0 ? tnext(a) : tprev(a);
                    double dab = dist(a, b);
                    for (int t = 0; t < K; ++t) {
                        int c = nbr[(size_t)a * K + t];
                        if (c < 0) break;
                        double dac = dist(a, c);
                        if (dac >= dab) break; // sorted; no gain possible
                        int d = dir == 0 ? tnext(c) : tprev(c);
                        if (d == a || c == b) continue;
                        double delta = dac + dist(b, d) - dab - dist(c, d);
                        if (delta < -1e-9) {
                            // reverse segment between b..c (in tour direction dir==0)
                            int i, j;
                            if (dir == 0) { i = pos[b]; j = pos[c]; }
                            else { i = pos[c]; j = pos[b]; }
                            int L = (j - i + N) % N + 1;
                            if (min(L, N - L) > maxRev) continue;
                            if (L > N / 2) {
                                // reverse the complementary segment instead
                                i = (j + 1) % N; j = (i + (N - L) - 1 + N) % N;
                                // recompute complement bounds: complement of [i..j] is [(j+1)..(i-1)]
                            }
                            // do reversal of tour[i..j] cyclically
                            int li = i, lj = j;
                            int steps = (((lj - li + N) % N) + 1) / 2;
                            for (int s = 0; s < steps; ++s) {
                                int u = tour[li], v = tour[lj];
                                tour[li] = v; tour[lj] = u;
                                pos[v] = li; pos[u] = lj;
                                li = (li + 1) % N; lj = (lj + N - 1) % N;
                            }
                            dontlook[a] = dontlook[b] = dontlook[c] = dontlook[d] = 0;
                            improved = true;
                            improvedAny = true;
                            break;
                        }
                    }
                }

                // Or-opt: move city a between two near neighbors
                if (!improved) {
                    int pa = tprev(a), na = tnext(a);
                    double removeGain = dist(pa, a) + dist(a, na) - dist(pa, na);
                    if (removeGain > 1e-9) {
                        for (int t = 0; t < K; ++t) {
                            int c = nbr[(size_t)a * K + t];
                            if (c < 0) break;
                            if (c == pa || c == na || c == a) continue;
                            int d = tnext(c);
                            if (d == a) continue;
                            double insCost = dist(c, a) + dist(a, d) - dist(c, d);
                            if (insCost < removeGain - 1e-9) {
                                // remove a, insert between c and d.
                                // Implement via rotation of the small span.
                                int ia = pos[a];
                                // shift elements to close the gap, then insert
                                // simple approach: erase-like shifting along shorter direction
                                int ic = pos[c];
                                // distance from a to c forward
                                int fwd = (ic - ia + N) % N;
                                int bwd = (ia - ic + N) % N;
                                if (min(fwd, bwd) > maxShift) continue;
                                if (fwd <= bwd) {
                                    // shift a..c left by one: a takes c's old spot region
                                    int cur = ia;
                                    for (int s = 0; s < fwd; ++s) {
                                        int nxt = (cur + 1) % N;
                                        tour[cur] = tour[nxt];
                                        pos[tour[cur]] = cur;
                                        cur = nxt;
                                    }
                                    tour[cur] = a; pos[a] = cur;
                                } else {
                                    // shift d..a right: move a backwards to just after c
                                    int idd = (ic + 1) % N;
                                    int cur = ia;
                                    for (int s = 0; s < bwd - 1; ++s) {
                                        int prv = (cur + N - 1) % N;
                                        tour[cur] = tour[prv];
                                        pos[tour[cur]] = cur;
                                        cur = prv;
                                    }
                                    (void)idd;
                                    tour[cur] = a; pos[a] = cur;
                                }
                                dontlook[pa] = dontlook[na] = dontlook[c] = dontlook[d] = dontlook[a] = 0;
                                improved = true;
                                improvedAny = true;
                                break;
                            }
                        }
                    }
                }

                if (!improved) dontlook[a] = 1;
            }
            if (rounds > 60) break;
        }
    }

    // ---- rotate so tour starts at city 0 ----
    {
        int z = 0;
        for (int i = 0; i < N; ++i) if (tour[i] == 0) { z = i; break; }
        rotate(tour.begin(), tour.begin() + z, tour.end());
    }

    // ---- fast output ----
    {
        size_t cap = (size_t)N * 9 + 16;
        char* out = (char*)malloc(cap);
        size_t o = 0;
        char tmp[16];
        for (int i = 0; i < N; ++i) {
            int v = tour[i];
            int t = 0;
            if (v == 0) tmp[t++] = '0';
            while (v > 0) { tmp[t++] = char('0' + v % 10); v /= 10; }
            while (t > 0) out[o++] = tmp[--t];
            out[o++] = (i + 1 < N) ? ' ' : '\n';
        }
#ifdef _WIN32
        _setmode(_fileno(stdout), _O_BINARY);
#endif
        fwrite(out, 1, o, stdout);
        free(out);
    }
    return 0;
}
