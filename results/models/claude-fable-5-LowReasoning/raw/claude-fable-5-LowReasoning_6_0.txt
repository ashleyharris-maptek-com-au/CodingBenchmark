using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Globalization;
using System.Text;

// Orbital Travelling Salesman Problem solver.
//
// Strategy:
//  1. Parse start state and N station states (position + velocity).
//  2. Convert each state vector to classical orbital elements.
//  3. Build a cost matrix of impulsive delta-V estimates between orbits:
//     Hohmann-like two-burn transfer between mean radii, with the plane
//     change optimally split between the two burns, plus a penalty for
//     eccentricity mismatch.
//  4. Solve the open-path TSP from the start orbit:
//       N <= 16 : exact Held-Karp dynamic programming.
//       N  > 16 : nearest-neighbour seed + 2-opt / Or-opt local search,
//                 then double-bridge perturbation restarts within a time
//                 budget (well under the 5-minute limit for N = 100).

static class Program
{
    static double Mu = 398600.4418;

    struct Vec3
    {
        public double X, Y, Z;
        public Vec3(double x, double y, double z) { X = x; Y = y; Z = z; }
        public static Vec3 operator -(Vec3 a, Vec3 b) => new Vec3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
        public static Vec3 operator *(double s, Vec3 a) => new Vec3(s * a.X, s * a.Y, s * a.Z);
        public double Norm() => Math.Sqrt(X * X + Y * Y + Z * Z);
        public static double Dot(Vec3 a, Vec3 b) => a.X * b.X + a.Y * b.Y + a.Z * b.Z;
        public static Vec3 Cross(Vec3 a, Vec3 b) =>
            new Vec3(a.Y * b.Z - a.Z * b.Y, a.Z * b.X - a.X * b.Z, a.X * b.Y - a.Y * b.X);
    }

    struct Orbit
    {
        public double Sma;   // semi-major axis (km)
        public double Ecc;   // eccentricity
        public Vec3 HHat;    // unit angular momentum (orbit plane normal)
    }

    static Orbit ToOrbit(Vec3 r, Vec3 v)
    {
        double rn = r.Norm();
        if (rn < 1e-9) rn = 1e-9;
        double energy = Vec3.Dot(v, v) / 2.0 - Mu / rn;
        double a = (energy < -1e-12) ? -Mu / (2.0 * energy) : rn; // guard hyperbolic/degenerate
        Vec3 h = Vec3.Cross(r, v);
        double hn = h.Norm();
        Vec3 hhat = hn > 1e-12 ? (1.0 / hn) * h : new Vec3(0, 0, 1);
        // eccentricity vector: e = (v x h)/mu - r/|r|
        Vec3 vxh = Vec3.Cross(v, h);
        Vec3 ev = new Vec3(vxh.X / Mu - r.X / rn, vxh.Y / Mu - r.Y / rn, vxh.Z / Mu - r.Z / rn);
        Orbit o;
        o.Sma = a;
        o.Ecc = Math.Min(ev.Norm(), 0.999);
        o.HHat = hhat;
        return o;
    }

    static double VisViva(double r, double a) => Math.Sqrt(Math.Max(1e-9, Mu * (2.0 / r - 1.0 / a)));

    // Impulsive delta-V estimate from orbit o1 to orbit o2.
    static double TransferDV(Orbit o1, Orbit o2)
    {
        double r1 = o1.Sma, r2 = o2.Sma;

        double c = Vec3.Dot(o1.HHat, o2.HHat);
        c = Math.Max(-1.0, Math.Min(1.0, c));
        double di = Math.Acos(c);

        double at = (r1 + r2) / 2.0;
        double v1 = VisViva(r1, o1.Sma);   // speed on current orbit near r1
        double vt1 = VisViva(r1, at);      // transfer-orbit speed at r1
        double vt2 = VisViva(r2, at);      // transfer-orbit speed at r2
        double v2 = VisViva(r2, o2.Sma);   // target-orbit speed at r2

        // Split the plane change between the two burns; take the cheapest split.
        double dvBest = double.MaxValue;
        for (int k = 0; k <= 8; k++)
        {
            double f = k / 8.0;
            double a1 = f * di, a2 = (1 - f) * di;
            double b1 = Math.Sqrt(Math.Max(0, v1 * v1 + vt1 * vt1 - 2 * v1 * vt1 * Math.Cos(a1)));
            double b2 = Math.Sqrt(Math.Max(0, vt2 * vt2 + v2 * v2 - 2 * vt2 * v2 * Math.Cos(a2)));
            double dv = b1 + b2;
            if (dv < dvBest) dvBest = dv;
        }

        // Penalty for eccentricity mismatch (rough shape-correction cost).
        double eccPen = 0.5 * Math.Abs(o1.Ecc - o2.Ecc) * Math.Sqrt(Mu / r1);

        return dvBest + eccPen;
    }

    // Cost[i, j]: node 0 is the start orbit, nodes 1..N are stations 0..N-1.
    static double[,] Cost;
    static int N;

    static double PathCost(int[] order)
    {
        double t = Cost[0, order[0] + 1];
        for (int i = 0; i + 1 < N; i++) t += Cost[order[i] + 1, order[i + 1] + 1];
        return t;
    }

    static void Main()
    {
        var tok = Console.In.ReadToEnd()
            .Split(new[] { ' ', '\t', '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries);
        int p = 0;
        Func<double> next = () => double.Parse(tok[p++], CultureInfo.InvariantCulture);

        N = (int)next();
        double mu = next();
        if (mu > 0) Mu = mu;

        var orbits = new Orbit[N + 1];
        for (int i = 0; i <= N; i++)
        {
            var r = new Vec3(next(), next(), next());
            var v = new Vec3(next(), next(), next());
            orbits[i] = ToOrbit(r, v);
        }

        Cost = new double[N + 1, N + 1];
        for (int i = 0; i <= N; i++)
            for (int j = 0; j <= N; j++)
                Cost[i, j] = (i == j) ? 0.0 : TransferDV(orbits[i], orbits[j]);

        int[] best;
        if (N <= 0) best = new int[0];
        else if (N == 1) best = new[] { 0 };
        else if (N <= 16) best = HeldKarp();
        else best = Heuristic();

        var sb = new StringBuilder();
        for (int i = 0; i < best.Length; i++)
        {
            if (i > 0) sb.Append(' ');
            sb.Append(best[i]);
        }
        Console.WriteLine(sb.ToString());
    }

    // Exact open-path TSP via Held-Karp DP (feasible up to N = 16).
    static int[] HeldKarp()
    {
        int full = 1 << N;
        var dp = new double[full, N];
        var par = new int[full, N];
        for (int m = 0; m < full; m++)
            for (int j = 0; j < N; j++)
                dp[m, j] = double.MaxValue;
        for (int j = 0; j < N; j++)
            dp[1 << j, j] = Cost[0, j + 1];

        for (int m = 1; m < full; m++)
        {
            for (int j = 0; j < N; j++)
            {
                if ((m & (1 << j)) == 0) continue;
                double d = dp[m, j];
                if (d == double.MaxValue) continue;
                for (int k = 0; k < N; k++)
                {
                    if ((m & (1 << k)) != 0) continue;
                    int nm = m | (1 << k);
                    double nd = d + Cost[j + 1, k + 1];
                    if (nd < dp[nm, k]) { dp[nm, k] = nd; par[nm, k] = j; }
                }
            }
        }

        int bestEnd = 0;
        double bestCost = double.MaxValue;
        for (int j = 0; j < N; j++)
            if (dp[full - 1, j] < bestCost) { bestCost = dp[full - 1, j]; bestEnd = j; }

        var order = new int[N];
        int mask = full - 1, cur = bestEnd;
        for (int i = N - 1; i >= 0; i--)
        {
            order[i] = cur;
            int prev = par[mask, cur];
            mask &= ~(1 << cur);
            cur = prev;
        }
        return order;
    }

    static int[] Heuristic()
    {
        var sw = Stopwatch.StartNew();
        long budgetMs = 45000; // comfortably inside the 5-minute limit

        int[] best = NearestNeighbour();
        LocalSearch(best);
        double bestCost = PathCost(best);

        var rng = new Random(12345);
        while (sw.ElapsedMilliseconds < budgetMs)
        {
            var cand = (int[])best.Clone();
            DoubleBridge(cand, rng);
            LocalSearch(cand);
            double cc = PathCost(cand);
            if (cc < bestCost - 1e-12) { bestCost = cc; best = cand; }
        }
        return best;
    }

    static int[] NearestNeighbour()
    {
        var order = new int[N];
        var used = new bool[N];
        int cur = 0; // node index in the cost matrix (0 = start)
        for (int i = 0; i < N; i++)
        {
            int bj = -1;
            double bd = double.MaxValue;
            for (int j = 0; j < N; j++)
            {
                if (used[j]) continue;
                double d = Cost[cur, j + 1];
                if (d < bd) { bd = d; bj = j; }
            }
            order[i] = bj;
            used[bj] = true;
            cur = bj + 1;
        }
        return order;
    }

    // 2-opt + Or-opt (segment lengths 1..3) to a local optimum on the open path.
    static void LocalSearch(int[] o)
    {
        bool improved = true;
        while (improved)
        {
            improved = false;

            // 2-opt: reverse segment o[i..j].
            for (int i = 0; i < N - 1; i++)
            {
                int a = (i == 0) ? 0 : o[i - 1] + 1;
                for (int j = i + 1; j < N; j++)
                {
                    int b = o[i] + 1, cN = o[j] + 1;
                    double before = Cost[a, b];
                    double after = Cost[a, cN];
                    if (j < N - 1)
                    {
                        int d = o[j + 1] + 1;
                        before += Cost[cN, d];
                        after += Cost[b, d];
                    }
                    if (after < before - 1e-12)
                    {
                        Array.Reverse(o, i, j - i + 1);
                        improved = true;
                    }
                }
            }

            // Or-opt: relocate a segment of length L elsewhere in the path.
            for (int L = 1; L <= 3 && L < N; L++)
            {
                for (int i = 0; i + L <= N; i++)
                {
                    int a = (i == 0) ? 0 : o[i - 1] + 1;
                    int b = o[i] + 1;             // first node of segment
                    int c = o[i + L - 1] + 1;     // last node of segment
                    bool segAtEnd = (i + L == N);
                    int d = segAtEnd ? -1 : o[i + L] + 1;

                    double removeGain = Cost[a, b]
                                      + (segAtEnd ? 0 : Cost[c, d])
                                      - (segAtEnd ? 0 : Cost[a, d]);
                    if (removeGain <= 1e-12) continue;

                    // Try inserting between consecutive remaining nodes.
                    // Positions are described by (pre, post) node pairs in the
                    // path with the segment removed.
                    int prev = 0; // start node
                    bool moved = false;
                    for (int t = 0; t <= N; t++)
                    {
                        if (t >= i && t < i + L) continue; // inside removed segment
                        int post = (t == N) ? -1 : o[t] + 1;
                        // skip the original location (prev == a means same place)
                        if (!(prev == a && ((segAtEnd && post == -1) || post == d)))
                        {
                            double addCost = Cost[prev, b]
                                           + (post < 0 ? 0 : Cost[c, post])
                                           - (post < 0 ? 0 : Cost[prev, post]);
                            if (addCost < removeGain - 1e-12)
                            {
                                MoveSegment(o, i, L, t);
                                improved = true;
                                moved = true;
                                break;
                            }
                        }
                        if (t < N) prev = o[t] + 1;
                    }
                    if (moved) break;
                }
                if (improved) break;
            }
        }
    }

    // Move segment o[i..i+L-1] so it ends up immediately before the node that
    // was at original index t (t == N means move to the end).
    static void MoveSegment(int[] o, int i, int L, int t)
    {
        var seg = new int[L];
        Array.Copy(o, i, seg, 0, L);
        var rest = new List<int>(N - L);
        var restIdx = new int[N + 1]; // original index -> index in rest
        for (int q = 0; q < N; q++)
        {
            if (q >= i && q < i + L) continue;
            restIdx[q] = rest.Count;
            rest.Add(o[q]);
        }
        int target = (t == N) ? rest.Count : restIdx[t];
        rest.InsertRange(target, seg);
        for (int q = 0; q < N; q++) o[q] = rest[q];
    }

    static void DoubleBridge(int[] o, Random rng)
    {
        if (N < 8)
        {
            int x = rng.Next(N), y = rng.Next(N);
            int tmp = o[x]; o[x] = o[y]; o[y] = tmp;
            return;
        }
        int a = 1 + rng.Next(N - 3);
        int b = a + 1 + rng.Next(N - a - 2);
        int c = b + 1 + rng.Next(N - b - 1);
        var res = new int[N];
        int idx = 0;
        for (int t = 0; t < a; t++) res[idx++] = o[t];
        for (int t = b; t < c; t++) res[idx++] = o[t];
        for (int t = a; t < b; t++) res[idx++] = o[t];
        for (int t = c; t < N; t++) res[idx++] = o[t];
        Array.Copy(res, o, N);
    }
}
