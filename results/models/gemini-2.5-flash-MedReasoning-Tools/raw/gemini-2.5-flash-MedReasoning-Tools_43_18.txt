// Basic HLSL Pixel Shader for Vulkan

// Input structure (if any, for now, just a dummy)
struct PSInput
{
    float4 Position : SV_Position; // Pixel position in screen space
};

// Output structure
struct PSOutput
{
    float4 Color : SV_Target0; // Output color to render target 0
};

PSOutput main(PSInput input)
{
    PSOutput output;
    output.Color = float4(1.0f, 0.0f, 0.0f, 1.0f); // Red color
    return output;
}
