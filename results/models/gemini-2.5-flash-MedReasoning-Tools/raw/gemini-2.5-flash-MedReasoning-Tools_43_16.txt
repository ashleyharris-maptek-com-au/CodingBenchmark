struct PSInput {
    [[vk::location(0)]] float3 worldPos : TEXCOORD0;
    [[vk::location(1)]] float3 normal   : TEXCOORD1;
    [[vk::location(2)]] float2 uv       : TEXCOORD2;
    [[vk::location(3)]] float3 tangent  : TEXCOORD3;
    [[vk::location(4)]] float3 color    : TEXCOORD4;
};

[[vk::binding(0, 0)]]
cbuffer UBO : register(b0) {
    float4x4 model;
    float4x4 view;
    float4x4 projection;
    float4 lightPos;
    float4 cameraPos;
    float4 params;
};

float4 main(PSInput input) : SV_Target0 {
    // Map UVs to complex number c
    float2 c = float2(input.uv.x * 3.0 - 2.0, input.uv.y * 2.0 - 1.0);

    float2 z = float2(0.0, 0.0);
    int iterations = 0;
    const int maxIterations = 20;

    for (int i = 0; i < maxIterations; ++i) {
        // z = z*z + c
        float zx_new = z.x * z.x - z.y * z.y + c.x;
        float zy_new = 2.0 * z.x * z.y + c.y;
        z = float2(zx_new, zy_new);

        // Escape condition: |z|^2 > 4
        if (dot(z, z) > 4.0) {
            iterations = i;
            break;
        }
        iterations = i + 1; // If it didn't escape, count this iteration
    }

    float t = saturate((float)iterations / (float)maxIterations);
    float3 finalColor = float3(t, t * 0.5, 1.0 - t);

    // If it never escaped, make it black (part of the set)
    if (iterations == maxIterations) {
        finalColor = float3(0.0, 0.0, 0.0);
    }
    
    return float4(finalColor, 1.0);
}
