// Pixel shader for Vulkan
// This shader samples a texture and outputs its color.

// Input structure from the vertex shader
struct PSInput
{
    float4 position : SV_POSITION;
    float2 texCoord : TEXCOORD0;
};

// Output structure for the pixel shader
struct PSOutput
{
    float4 color : SV_TARGET0;
};

// Texture and Sampler declarations
Texture2D    g_texture : register(t0);
SamplerState g_sampler : register(s0);

PSOutput main(PSInput input)
{
    PSOutput output;
    output.color = g_texture.Sample(g_sampler, input.texCoord);
    return output;
}
