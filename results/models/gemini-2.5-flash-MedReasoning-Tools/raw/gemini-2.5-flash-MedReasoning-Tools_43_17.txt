// Vulkan-compatible HLSL Pixel Shader

// Input structure for the pixel shader
// This structure should match the output of the vertex shader
struct PSInput
{
    float4 Position : SV_POSITION;
    float2 TexCoord : TEXCOORD0; // Example: Texture coordinates
    float3 Normal   : NORMAL0;   // Example: Normal vector
    float4 Color    : COLOR0;    // Example: Vertex color
};

// Output structure for the pixel shader
struct PSOutput
{
    float4 Color : SV_TARGET0; // Output color to render target 0
};

// Example texture and sampler
Texture2D    g_texture : register(t0);
SamplerState g_sampler : register(s0);

// Main pixel shader function
PSOutput main(PSInput input)
{
    PSOutput output;

    // Example: Sample a texture using the provided texture coordinates
    float4 texColor = g_texture.Sample(g_sampler, input.TexCoord);

    // Example: Apply vertex color
    float4 finalColor = texColor * input.Color;

    // Example: Simple lighting based on normal (for demonstration, not full PBR)
    // Assuming a light direction (e.g., coming from the camera or a fixed direction)
    float3 lightDir = normalize(float3(0.5f, 0.5f, -1.0f)); // Example light direction
    float  diffuse  = saturate(dot(input.Normal, lightDir));
    finalColor.rgb *= diffuse + 0.2f; // Add some ambient light

    output.Color = finalColor;

    return output;
}
