OpCapability Shader
               OpCapability Geometry
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in_Position %_gl_Position
               OpExecutionMode %main InputTriangles
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 3
               OpSource GLSL 450
               OpDecorate %gl_in_Position BuiltIn Position
               OpDecorate %_gl_Position BuiltIn Position

; Types
       %void = OpTypeVoid
      %float = OpTypeFloat 32
    %v4float = OpTypeVector %float 4
      %uint = OpTypeInt 32 0
    %uint_0 = OpConstant %uint 0
    %uint_1 = OpConstant %uint 1
    %uint_2 = OpConstant %uint 2
    %uint_3 = OpConstant %uint 3

%array_v4float_3 = OpTypeArray %v4float %uint_3 ; Array of 3 v4floats for input
%ptr_Input_array_v4float_3 = OpTypePointer Input %array_v4float_3
%ptr_Input_v4float = OpTypePointer Input %v4float ; Pointer to a single v4float in Input storage class
%ptr_Output_v4float = OpTypePointer Output %v4float ; Pointer to a single v4float in Output storage class

; Function Type
          %2 = OpTypeFunction %void

; Global Variables
%gl_in_Position = OpVariable %ptr_Input_array_v4float_3 Input ; Input Position array
%_gl_Position = OpVariable %ptr_Output_v4float Output ; Output Position

; Main Function
       %main = OpFunction %void None %2
      %entry = OpLabel
               ; Loop for 3 vertices
               %_in_pos_ptr0 = OpAccessChain %ptr_Input_v4float %gl_in_Position %uint_0
               %_in_pos_val0 = OpLoad %v4float %_in_pos_ptr0
               OpStore %_gl_Position %_in_pos_val0
               OpEmitVertex

               %_in_pos_ptr1 = OpAccessChain %ptr_Input_v4float %gl_in_Position %uint_1
               %_in_pos_val1 = OpLoad %v4float %_in_pos_ptr1
               OpStore %_gl_Position %_in_pos_val1
               OpEmitVertex

               %_in_pos_ptr2 = OpAccessChain %ptr_Input_v4float %gl_in_Position %uint_2
               %_in_pos_val2 = OpLoad %v4float %_in_pos_ptr2
               OpStore %_gl_Position %_in_pos_val2
               OpEmitVertex

               OpEndPrimitive
               OpReturn
               OpFunctionEnd
