const double PI = 3.14159265358979323846;

struct Vector3 {
    double x, y, z;
    Vector3 operator+(const Vector3& o) const { return {x + o.x, y + o.y, z + o.z}; }
    Vector3 operator-(const Vector3& o) const { return {x - o.x, y - o.y, z - o.z}; }
    Vector3 operator*(double s) const { return {x * s, y * s, z * s}; }
    double dot(const Vector3& o) const { return x * o.x + y * o.y + z * o.z; }
    Vector3 cross(const Vector3& o) const {
        return {
            y * o.z - z * o.y,
            z * o.x - x * o.z,
            x * o.y - y * o.x
        };
    }
    double norm() const { return std::sqrt(x*x + y*y + z*z); }
    Vector3 normalize() const {
        double n = norm();
        if (n < 1e-9) return {0, 0, 0};
        return {x / n, y / n, z / n};
    }
};

double dist(const Vector3& a, const Vector3& b) {
    return (a - b).norm();
}

double det3d(const Vector3& A, const Vector3& B, const Vector3& C, const Vector3& D) {
    double ux = B.x - A.x, uy = B.y - A.y, uz = B.z - A.z;
    double vx = C.x - A.x, vy = C.y - A.y, vz = C.z - A.z;
    double wx = D.x - A.x, wy = D.y - A.y, wz = D.z - A.z;
    return ux * (vy * wz - vz * wy) - uy * (vx * wz - vz * wx) + uz * (vx * wy - vy * wx);
}

struct Face {
    int a, b, c;
    bool operator<(const Face& o) const {
        return std::tie(a, b, c) < std::tie(o.a, o.b, o.c);
    }
};

// Robust 3D Convex Hull Algorithm using Gift Wrapping
std::vector<Face> compute_convex_hull(const std::vector<Vector3>& pts) {
    int n = pts.size();
    if (n < 4) return {};

    // 1. Find P0 (min x, then min y, then min z)
    int p0 = 0;
    for (int i = 1; i < n; ++i) {
        if (pts[i].x < pts[p0].x) {
            p0 = i;
        } else if (pts[i].x == pts[p0].x) {
            if (pts[i].y < pts[p0].y) {
                p0 = i;
            } else if (pts[i].y == pts[p0].y) {
                if (pts[i].z < pts[p0].z) {
                    p0 = i;
                }
            }
        }
    }

    // 2. Find P1 on the hull by minimizing the angle with positive x-axis
    int p1 = -1;
    double min_cos = 2.0;
    double max_d = -1.0;
    for (int i = 0; i < n; ++i) {
        if (i == p0) continue;
        double d = dist(pts[i], pts[p0]);
        if (d < 1e-9) continue;
        double cos_val = (pts[i].x - pts[p0].x) / d;
        if (cos_val < min_cos - 1e-9) {
            min_cos = cos_val;
            p1 = i;
            max_d = d;
        } else if (std::abs(cos_val - min_cos) < 1e-9) {
            if (d > max_d) {
                max_d = d;
                p1 = i;
            }
        }
    }

    if (p1 == -1) return {};

    // 3. Find P2 to form the first face
    int p2 = -1;
    for (int i = 0; i < n; ++i) {
        if (i == p0 || i == p1) continue;
        if (p2 == -1) {
            p2 = i;
            continue;
        }
        double val = det3d(pts[p0], pts[p1], pts[p2], pts[i]);
        if (val > 1e-9) {
            p2 = i;
        } else if (std::abs(val) < 1e-9) {
            if (dist(pts[p0], pts[i]) > dist(pts[p0], pts[p2])) {
                p2 = i;
            }
        }
    }

    if (p2 == -1) return {};

    // Ensure correct orientation (det3d(p0, p1, p2, P) <= 0 for all P)
    bool swap_needed = false;
    for (int i = 0; i < n; ++i) {
        if (i == p0 || i == p1 || i == p2) continue;
        double val = det3d(pts[p0], pts[p1], pts[p2], pts[i]);
        if (val > 1e-9) {
            swap_needed = true;
            break;
        } else if (val < -1e-9) {
            break;
        }
    }
    if (swap_needed) {
        std::swap(p1, p2);
    }

    std::vector<Face> faces;
    std::set<std::pair<int, int>> active_edges;
    std::set<Face> face_set;

    auto add_face = [&](int a, int b, int c) {
        Face f{a, b, c};
        int arr[3] = {a, b, c};
        std::sort(arr, arr + 3);
        Face sorted_f{arr[0], arr[1], arr[2]};
        if (face_set.count(sorted_f)) return;
        face_set.insert(sorted_f);
        faces.push_back(f);

        auto handle_edge = [&](int u, int v) {
            if (active_edges.count({v, u})) {
                active_edges.erase({v, u});
            } else {
                active_edges.insert({u, v});
            }
        };
        handle_edge(a, b);
        handle_edge(b, c);
        handle_edge(c, a);
    };

    add_face(p0, p1, p2);

    int iter = 0;
    while (!active_edges.empty() && iter < 200000) {
        iter++;
        auto it = active_edges.begin();
        int u = it->first;
        int v = it->second;
        active_edges.erase(it);

        int w = -1;
        for (int i = 0; i < n; ++i) {
            if (i == u || i == v) continue;
            if (w == -1) {
                w = i;
                continue;
            }
            double val = det3d(pts[v], pts[u], pts[w], pts[i]);
            if (val > 1e-9) {
                w = i;
            } else if (std::abs(val) < 1e-9) {
                if (dist(pts[v], pts[i]) > dist(pts[v], pts[w])) {
                    w = i;
                }
            }
        }

        if (w != -1) {
            add_face(v, u, w);
        }
    }

    return faces;
}

struct Plane {
    Vector3 n;
    double d;
};

struct PlacedTetrahedron {
    double cx, cy, cz;
    double rx, ry, rz;
};

int main() {
    // Fast I/O
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    auto start_time = std::chrono::steady_clock::now();

    int V;
    double edge_length;
    if (!(std::cin >> V >> edge_length)) {
        std::cout << 0 << "\n";
        return 0;
    }

    std::vector<Vector3> original_vertices(V);
    for (int i = 0; i < V; ++i) {
        std::cin >> original_vertices[i].x >> original_vertices[i].y >> original_vertices[i].z;
    }

    if (V < 4) {
        std::cout << 0 << "\n";
        return 0;
    }

    // Apply tiny perturbation to handle coplanarities robustly
    std::vector<Vector3> vertices = original_vertices;
    std::mt19937 rng(1337);
    std::uniform_real_distribution<double> dist_noise(-1e-8, 1e-8);
    for (int i = 0; i < V; ++i) {
        vertices[i].x += dist_noise(rng);
        vertices[i].y += dist_noise(rng);
        vertices[i].z += dist_noise(rng);
    }

    // Compute convex hull
    auto hull_faces = compute_convex_hull(vertices);
    if (hull_faces.empty()) {
        std::cout << 0 << "\n";
        return 0;
    }

    // Convert hull faces to planes
    std::vector<Plane> planes;
    for (const auto& f : hull_faces) {
        Vector3 A = original_vertices[f.a];
        Vector3 B = original_vertices[f.b];
        Vector3 C = original_vertices[f.c];
        Vector3 normal = (B - A).cross(C - A).normalize();
        double d = -A.dot(normal);
        planes.push_back({normal, d});
    }

    // Bounding box of the container
    double x_min = original_vertices[0].x, x_max = x_min;
    double y_min = original_vertices[0].y, y_max = y_min;
    double z_min = original_vertices[0].z, z_max = z_min;
    for (const auto& v : original_vertices) {
        x_min = std::min(x_min, v.x); x_max = std::max(x_max, v.x);
        y_min = std::min(y_min, v.y); y_max = std::max(y_max, v.y);
        z_min = std::min(z_min, v.z); z_max = std::max(z_max, v.z);
    }

    // Grid spacings with safety margin (to prevent precision overlap errors)
    double dx = 0.5105 * edge_length;
    double dy = 0.5105 * edge_length;
    double dz = 0.7105 * edge_length;

    // Canonical vertices scaled by edge_length
    Vector3 base_v[2][4];
    // Rotation 0
    base_v[0][0] = Vector3{-0.5, 0, -0.25 * std::sqrt(2.0)} * edge_length;
    base_v[0][1] = Vector3{0.5, 0, -0.25 * std::sqrt(2.0)} * edge_length;
    base_v[0][2] = Vector3{0, 0.5, 0.25 * std::sqrt(2.0)} * edge_length;
    base_v[0][3] = Vector3{0, -0.5, 0.25 * std::sqrt(2.0)} * edge_length;

    // Rotation 90 degrees around Z
    auto rotate_z = [](const Vector3& p, double angle) -> Vector3 {
        double s = std::sin(angle);
        double c = std::cos(angle);
        return {p.x * c - p.y * s, p.x * s + p.y * c, p.z};
    };
    for (int i = 0; i < 4; ++i) {
        base_v[1][i] = rotate_z(base_v[0][i], PI / 2.0);
    }

    // Precompute R_max for each plane and each rotation
    std::vector<std::vector<double>> r_max(planes.size(), std::vector<double>(2, 0.0));
    for (size_t i = 0; i < planes.size(); ++i) {
        for (int r = 0; r < 2; ++r) {
            double mx = base_v[r][0].dot(planes[i].n);
            for (int j = 1; j < 4; ++j) {
                mx = std::max(mx, base_v[r][j].dot(planes[i].n));
            }
            r_max[i][r] = mx;
        }
    }

    // Bounding box of grid indices
    int i_min = std::floor((x_min - edge_length) / dx);
    int i_max = std::ceil((x_max + edge_length) / dx);
    int j_min = std::floor((y_min - edge_length) / dy);
    int j_max = std::ceil((y_max + edge_length) / dy);
    int k_min = std::floor((z_min - edge_length) / dz);
    int k_max = std::ceil((z_max + edge_length) / dz);

    long long total_cells = (long long)(i_max - i_min + 1) * (j_max - j_min + 1) * (k_max - k_min + 1);

    // Adapt the number of random offset searches based on total grid cells (Scalability)
    int num_offsets = 25;
    if (total_cells > 100000) num_offsets = 10;
    if (total_cells > 500000) num_offsets = 5;
    if (total_cells > 2000000) num_offsets = 2;
    if (total_cells > 5000000) num_offsets = 1;

    std::vector<PlacedTetrahedron> best_placement;
    std::mt19937 offset_rng(1337);
    std::uniform_real_distribution<double> dist_x(0, dx);
    std::uniform_real_distribution<double> dist_y(0, dy);
    std::uniform_real_distribution<double> dist_z(0, dz);

    bool global_timeout = false;

    for (int offset_idx = 0; offset_idx < num_offsets; ++offset_idx) {
        double ox = (offset_idx == 0) ? 0.0 : dist_x(offset_rng);
        double oy = (offset_idx == 0) ? 0.0 : dist_y(offset_rng);
        double oz = (offset_idx == 0) ? 0.0 : dist_z(offset_rng);

        std::vector<PlacedTetrahedron> current_placement;

        long long iterations = 0;
        for (int i = i_min; i <= i_max; ++i) {
            for (int j = j_min; j <= j_max; ++j) {
                for (int k = k_min; k <= k_max; ++k) {
                    // Periodic timeout check to guarantee completion within 30 seconds
                    if (++iterations >= 100000) {
                        iterations = 0;
                        auto now = std::chrono::steady_clock::now();
                        double elapsed = std::chrono::duration<double>(now - start_time).count();
                        if (elapsed > 28.0) {
                            global_timeout = true;
                            break;
                        }
                    }

                    double cx = i * dx + ox;
                    double cy = j * dy + oy;
                    double cz = k * dz + oz;

                    int sum = i + j + k;
                    if (sum < 0) sum = -sum;
                    int r = sum % 2;

                    // Fast inside check
                    bool inside = true;
                    for (size_t p = 0; p < planes.size(); ++p) {
                        if (cx * planes[p].n.x + cy * planes[p].n.y + cz * planes[p].n.z + r_max[p][r] + planes[p].d > 1e-7) {
                            inside = false;
                            break;
                        }
                    }

                    if (inside) {
                        current_placement.push_back({cx, cy, cz, 0.0, 0.0, r * (PI / 2.0)});
                    }
                }
                if (global_timeout) break;
            }
            if (global_timeout) break;
        }

        if (current_placement.size() > best_placement.size()) {
            best_placement = std::move(current_placement);
        }
        if (global_timeout) break;
    }

    // Output results
    std::cout << best_placement.size() << "\n";
    std::cout << std::fixed << std::setprecision(10);
    for (const auto& pt : best_placement) {
        std::cout << pt.cx << " " << pt.cy << " " << pt.cz << " "
                  << pt.rx << " " << pt.ry << " " << pt.rz << "\n";
    }

    return 0;
}
