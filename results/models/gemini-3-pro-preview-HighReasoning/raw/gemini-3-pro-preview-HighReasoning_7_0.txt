struct Vector3 {
    double x, y, z;
    Vector3() : x(0), y(0), z(0) {}
    Vector3(double x, double y, double z) : x(x), y(y), z(z) {}
    
    Vector3 operator+(const Vector3& o) const { return {x + o.x, y + o.y, z + o.z}; }
    Vector3 operator-(const Vector3& o) const { return {x - o.x, y - o.y, z - o.z}; }
    Vector3 operator*(double s) const { return {x * s, y * s, z * s}; }
    Vector3 operator/(double s) const { return {x / s, y / s, z / s}; }
};

inline double dot(const Vector3& a, const Vector3& b) {
    return a.x * b.x + a.y * b.y + a.z * b.z;
}

inline Vector3 cross(const Vector3& a, const Vector3& b) {
    return {
        a.y * b.z - a.z * b.y,
        a.z * b.x - a.x * b.z,
        a.x * b.y - a.y * b.x
    };
}

inline double length_sq(const Vector3& v) { return dot(v, v); }
inline double length(const Vector3& v) { return sqrt(length_sq(v)); }
inline Vector3 normalize(const Vector3& v) {
    double len = length(v);
    if (len < 1e-12) return {0, 0, 0};
    return v / len;
}

struct Vector2 {
    double x, y;
    Vector2() : x(0), y(0) {}
    Vector2(double x, double y) : x(x), y(y) {}
    Vector2 operator+(const Vector2& o) const { return {x + o.x, y + o.y}; }
    Vector2 operator-(const Vector2& o) const { return {x - o.x, y - o.y}; }
    Vector2 operator*(double s) const { return {x * s, y * s}; }
};

inline double dot2d(const Vector2& a, const Vector2& b) { return a.x * b.x + a.y * b.y; }
inline double cross2d(const Vector2& a, const Vector2& b) { return a.x * b.y - a.y * b.x; }
inline double length2d(const Vector2& v) { return sqrt(dot2d(v, v)); }

inline Vector3 operator*(double s, const Vector3& v) { return v * s; }
inline Vector2 operator*(double s, const Vector2& v) { return v * s; }

struct Triangle3D {
    Vector3 v[3];
};

struct Segment3D {
    Vector3 p1, p2;
    int g1 = -1;
    int g2 = -1;
};

struct AABBNode {
    Vector3 box_min;
    Vector3 box_max;
    int left = -1;
    int right = -1;
    int triangle_idx = -1;
};

bool aabb_overlap(const Vector3& min_a, const Vector3& max_a, const Vector3& min_b, const Vector3& max_b) {
    return (min_a.x <= max_b.x && max_a.x >= min_b.x) &&
           (min_a.y <= max_b.y && max_a.y >= min_b.y) &&
           (min_a.z <= max_b.z && max_a.z >= min_b.z);
}

int build_aabb_tree(vector<int>& indices, int start, int end, vector<AABBNode>& nodes, const vector<Triangle3D>& triangles) {
    int node_idx = nodes.size();
    nodes.emplace_back();
    
    Vector3 box_min(1e18, 1e18, 1e18);
    Vector3 box_max(-1e18, -1e18, -1e18);
    for (int i = start; i < end; ++i) {
        const auto& T = triangles[indices[i]];
        for (int j = 0; j < 3; ++j) {
            box_min.x = min(box_min.x, T.v[j].x);
            box_min.y = min(box_min.y, T.v[j].y);
            box_min.z = min(box_min.z, T.v[j].z);
            box_max.x = max(box_max.x, T.v[j].x);
            box_max.y = max(box_max.y, T.v[j].y);
            box_max.z = max(box_max.z, T.v[j].z);
        }
    }
    
    nodes[node_idx].box_min = box_min;
    nodes[node_idx].box_max = box_max;
    
    if (end - start == 1) {
        nodes[node_idx].triangle_idx = indices[start];
        return node_idx;
    }
    
    Vector3 size = box_max - box_min;
    int axis = 0;
    if (size.y > size.x) axis = 1;
    if (size.z > size.x && size.z > size.y) axis = 2;
    
    auto get_centroid = [&](int tri_idx) {
        const auto& T = triangles[tri_idx];
        return (T.v[0] + T.v[1] + T.v[2]) / 3.0;
    };
    
    int mid = start + (end - start) / 2;
    std::nth_element(indices.begin() + start, indices.begin() + mid, indices.begin() + end,
                     [&](int a, int b) {
                         Vector3 ca = get_centroid(a);
                         Vector3 cb = get_centroid(b);
                         if (axis == 0) return ca.x < cb.x;
                         if (axis == 1) return ca.y < cb.y;
                         return ca.z < cb.z;
                     });
                     
    int left_child = build_aabb_tree(indices, start, mid, nodes, triangles);
    int right_child = build_aabb_tree(indices, mid, end, nodes, triangles);
    
    nodes[node_idx].left = left_child;
    nodes[node_idx].right = right_child;
    return node_idx;
}

vector<Vector3> intersect_triangle_plane(const Triangle3D& T, const Vector3& n, double d) {
    double h[3];
    for (int i = 0; i < 3; ++i) {
        h[i] = dot(n, T.v[i]) + d;
        if (abs(h[i]) < 1e-11) h[i] = 0.0;
    }
    
    vector<Vector3> pts;
    for (int i = 0; i < 3; ++i) {
        int j = (i + 1) % 3;
        double hi = h[i];
        double hj = h[j];
        if (hi == 0.0) {
            pts.push_back(T.v[i]);
        }
        if (hi * hj < 0.0) {
            double t = hi / (hi - hj);
            pts.push_back(T.v[i] + t * (T.v[j] - T.v[i]));
        }
    }
    vector<Vector3> unique_pts;
    for (const auto& p : pts) {
        bool dup = false;
        for (const auto& up : unique_pts) {
            if (length(p - up) < 1e-9) {
                dup = true;
                break;
            }
        }
        if (!dup) unique_pts.push_back(p);
    }
    return unique_pts;
}

bool intersect_triangles(const Triangle3D& T1, const Triangle3D& T2, Segment3D& out_seg) {
    Vector3 n1 = cross(T1.v[1] - T1.v[0], T1.v[2] - T1.v[0]);
    double len1 = length(n1);
    if (len1 < 1e-12) return false;
    n1 = n1 / len1;
    double d1 = -dot(n1, T1.v[0]);
    
    Vector3 n2 = cross(T2.v[1] - T2.v[0], T2.v[2] - T2.v[0]);
    double len2 = length(n2);
    if (len2 < 1e-12) return false;
    n2 = n2 / len2;
    double d2 = -dot(n2, T2.v[0]);
    
    Vector3 D = cross(n1, n2);
    double lenD = length(D);
    if (lenD < 1e-7) return false; // Parallel
    D = D / lenD;
    
    auto pts2 = intersect_triangle_plane(T2, n1, d1);
    if (pts2.size() != 2) return false;
    auto pts1 = intersect_triangle_plane(T1, n2, d2);
    if (pts1.size() != 2) return false;
    
    double t1_min = dot(pts1[0], D);
    double t1_max = dot(pts1[1], D);
    if (t1_min > t1_max) swap(t1_min, t1_max);
    
    double t2_min = dot(pts2[0], D);
    double t2_max = dot(pts2[1], D);
    if (t2_min > t2_max) swap(t2_min, t2_max);
    
    double t_start = max(t1_min, t2_min);
    double t_end = min(t1_max, t2_max);
    
    if (t_start < t_end - 1e-8) {
        auto get_pt = [&](double t) {
            double d_p1_0 = abs(dot(pts1[0], D) - t);
            double d_p1_1 = abs(dot(pts1[1], D) - t);
            double d_p2_0 = abs(dot(pts2[0], D) - t);
            double d_p2_1 = abs(dot(pts2[1], D) - t);
            double min_d = min({d_p1_0, d_p1_1, d_p2_0, d_p2_1});
            if (min_d == d_p1_0) return pts1[0];
            if (min_d == d_p1_1) return pts1[1];
            if (min_d == d_p2_0) return pts2[0];
            return pts2[1];
        };
        out_seg.p1 = get_pt(t_start);
        out_seg.p2 = get_pt(t_end);
        return true;
    }
    return false;
}

void find_intersecting_pairs(int node_a, int node_b,
                             const vector<AABBNode>& nodes_a, const vector<AABBNode>& nodes_b,
                             const vector<Triangle3D>& triangles_a, const vector<Triangle3D>& triangles_b,
                             vector<pair<int, int>>& results) {
    const auto& na = nodes_a[node_a];
    const auto& nb = nodes_b[node_b];
    
    if (!aabb_overlap(na.box_min, na.box_max, nb.box_min, nb.box_max)) {
        return;
    }
    
    if (na.left == -1 && nb.left == -1) {
        Segment3D out_seg;
        if (intersect_triangles(triangles_a[na.triangle_idx], triangles_b[nb.triangle_idx], out_seg)) {
            results.push_back({na.triangle_idx, nb.triangle_idx});
        }
        return;
    }
    
    if (na.left == -1) {
        find_intersecting_pairs(node_a, nb.left, nodes_a, nodes_b, triangles_a, triangles_b, results);
        find_intersecting_pairs(node_a, nb.right, nodes_a, nodes_b, triangles_a, triangles_b, results);
    } else if (nb.left == -1) {
        find_intersecting_pairs(na.left, node_b, nodes_a, nodes_b, triangles_a, triangles_b, results);
        find_intersecting_pairs(na.right, node_b, nodes_a, nodes_b, triangles_a, triangles_b, results);
    } else {
        double vol_a = (na.box_max.x - na.box_min.x) * (na.box_max.y - na.box_min.y) * (na.box_max.z - na.box_min.z);
        double vol_b = (nb.box_max.x - nb.box_min.x) * (nb.box_max.y - nb.box_min.y) * (nb.box_max.z - nb.box_min.z);
        if (vol_a > vol_b) {
            find_intersecting_pairs(na.left, node_b, nodes_a, nodes_b, triangles_a, triangles_b, results);
            find_intersecting_pairs(na.right, node_b, nodes_a, nodes_b, triangles_a, triangles_b, results);
        } else {
            find_intersecting_pairs(node_a, nb.left, nodes_a, nodes_b, triangles_a, triangles_b, results);
            find_intersecting_pairs(node_a, nb.right, nodes_a, nodes_b, triangles_a, triangles_b, results);
        }
    }
}

bool ray_box_intersect(const Vector3& orig, const Vector3& dir, const Vector3& box_min, const Vector3& box_max, double& t_enter) {
    double tmin = 0.0;
    double tmax = 1e18;
    
    double invDx = 1.0 / (abs(dir.x) < 1e-15 ? (dir.x < 0 ? -1e-15 : 1e-15) : dir.x);
    double tx0 = (box_min.x - orig.x) * invDx;
    double tx1 = (box_max.x - orig.x) * invDx;
    if (invDx < 0.0) swap(tx0, tx1);
    tmin = max(tmin, tx0);
    tmax = min(tmax, tx1);
    
    double invDy = 1.0 / (abs(dir.y) < 1e-15 ? (dir.y < 0 ? -1e-15 : 1e-15) : dir.y);
    double ty0 = (box_min.y - orig.y) * invDy;
    double ty1 = (box_max.y - orig.y) * invDy;
    if (invDy < 0.0) swap(ty0, ty1);
    tmin = max(tmin, ty0);
    tmax = min(tmax, ty1);
    
    double invDz = 1.0 / (abs(dir.z) < 1e-15 ? (dir.z < 0 ? -1e-15 : 1e-15) : dir.z);
    double tz0 = (box_min.z - orig.z) * invDz;
    double tz1 = (box_max.z - orig.z) * invDz;
    if (invDz < 0.0) swap(tz0, tz1);
    tmin = max(tmin, tz0);
    tmax = min(tmax, tz1);
    
    t_enter = tmin;
    return tmin <= tmax && tmax >= 0.0;
}

bool ray_triangle_intersect(const Vector3& orig, const Vector3& dir, const Vector3& A, const Vector3& B, const Vector3& C, double& t_out, double& u_out, double& v_out) {
    Vector3 edge1 = B - A;
    Vector3 edge2 = C - A;
    Vector3 pvec = cross(dir, edge2);
    double det = dot(edge1, pvec);
    if (abs(det) < 1e-12) return false;
    double invDet = 1.0 / det;
    Vector3 tvec = orig - A;
    double u = dot(tvec, pvec) * invDet;
    if (u < 0.0 || u > 1.0) return false;
    Vector3 qvec = cross(tvec, edge1);
    double v = dot(dir, qvec) * invDet;
    if (v < 0.0 || u + v > 1.0) return false;
    double t = dot(edge2, qvec) * invDet;
    if (t < 0.0) return false;
    t_out = t;
    u_out = u;
    v_out = v;
    return true;
}

void ray_cast_traverse(int node_idx, const vector<AABBNode>& nodes, const vector<Triangle3D>& triangles, const Vector3& orig, const Vector3& dir, int& count, bool& retry) {
    if (retry) return;
    double t_enter;
    if (!ray_box_intersect(orig, dir, nodes[node_idx].box_min, nodes[node_idx].box_max, t_enter)) {
        return;
    }
    if (nodes[node_idx].left == -1) {
        double t, u, v;
        const auto& T = triangles[nodes[node_idx].triangle_idx];
        if (ray_triangle_intersect(orig, dir, T.v[0], T.v[1], T.v[2], t, u, v)) {
            if (u < 1e-5 || v < 1e-5 || (1.0 - u - v) < 1e-5) {
                retry = true;
                return;
            }
            count++;
        }
        return;
    }
    ray_cast_traverse(nodes[node_idx].left, nodes, triangles, orig, dir, count, retry);
    ray_cast_traverse(nodes[node_idx].right, nodes, triangles, orig, dir, count, retry);
}

double hash_double(int id, int coord) {
    unsigned int x = id * 73856093 ^ coord * 19349663;
    x = ((x >> 16) ^ x) * 0x45d9f3b;
    x = ((x >> 16) ^ x) * 0x45d9f3b;
    x = (x >> 16) ^ x;
    return (double)(x % 20001) / 10000.0 - 1.0;
}

bool is_point_inside(const Vector3& p, const vector<AABBNode>& nodes, const vector<Triangle3D>& triangles) {
    if (nodes.empty()) return false;
    int attempt = 0;
    while (attempt < 20) {
        double theta = (hash_double(attempt + 12345, (int)(p.x * 100) + attempt) + 1.0) * 0.5 * M_PI;
        double phi = (hash_double(attempt + 67890, (int)(p.y * 100) + attempt) + 1.0) * M_PI;
        Vector3 dir(sin(theta)*cos(phi), sin(theta)*sin(phi), cos(theta));
        
        int count = 0;
        bool retry = false;
        ray_cast_traverse(0, nodes, triangles, p, dir, count, retry);
        if (!retry) {
            return (count % 2) != 0;
        }
        attempt++;
    }
    return false;
}

bool point_on_segment_3d(Vector3 p, Vector3 a, Vector3 b) {
    Vector3 ab = b - a;
    double len_ab = length(ab);
    if (len_ab < 1e-12) return false;
    Vector3 ap = p - a;
    double d = length(cross(ap, ab)) / len_ab;
    if (d > 1e-7 * len_ab) return false;
    double dp = dot(ap, b - p);
    return dp >= -1e-7 * len_ab * len_ab;
}

bool point_on_segment_2d(Vector2 p, Vector2 a, Vector2 b) {
    double cp = cross2d(p - a, b - a);
    if (abs(cp) > 1e-7) return false;
    double dp = dot2d(p - a, b - p);
    return dp >= -1e-7;
}

bool segments_intersect_open(Vector2 p1, Vector2 q1, Vector2 p2, Vector2 q2) {
    double c1 = cross2d(q1 - p1, p2 - p1);
    double c2 = cross2d(q1 - p1, q2 - p1);
    double c3 = cross2d(q2 - p2, p1 - p2);
    double c4 = cross2d(q2 - p2, q1 - p2);
    return (c1 * c2 < -1e-13) && (c3 * c4 < -1e-13);
}

struct TriangulationSolver {
    vector<Vector2> V2;
    vector<pair<int, int>> constraints;
    vector<pair<int, int>> boundary_constraints;
    vector<Triangle2D> valid_triangles;
    vector<Triangle2D> selected;
    vector<double> suffix_areas;
    double parent_area;
    bool flipped;
    int step_count = 0;
    
    bool interiors_overlap(const Triangle2D& t1, const Triangle2D& t2) {
        int idxs1[3] = {t1.idxs[0], t1.idxs[1], t1.idxs[2]};
        int idxs2[3] = {t2.idxs[0], t2.idxs[1], t2.idxs[2]};
        
        auto is_separating = [&](int u, int v, int w, int q1, int q2, int q3) {
            Vector2 U = V2[u], V = V2[v], W = V2[w];
            Vector2 Q1 = V2[q1], Q2 = V2[q2], Q3 = V2[q3];
            double cp_w = cross2d(V - U, W - U);
            double cp_q1 = cross2d(V - U, Q1 - U);
            double cp_q2 = cross2d(V - U, Q2 - U);
            double cp_q3 = cross2d(V - U, Q3 - U);
            if (cp_w > 0) {
                return cp_q1 <= 1e-13 && cp_q2 <= 1e-13 && cp_q3 <= 1e-13;
            } else {
                return cp_q1 >= -1e-13 && cp_q2 >= -1e-13 && cp_q3 >= -1e-13;
            }
        };
        
        if (is_separating(idxs1[0], idxs1[1], idxs1[2], idxs2[0], idxs2[1], idxs2[2])) return false;
        if (is_separating(idxs1[1], idxs1[2], idxs1[0], idxs2[0], idxs2[1], idxs2[2])) return false;
        if (is_separating(idxs1[2], idxs1[0], idxs1[1], idxs2[0], idxs2[1], idxs2[2])) return false;
        
        if (is_separating(idxs2[0], idxs2[1], idxs2[2], idxs1[0], idxs1[1], idxs1[2])) return false;
        if (is_separating(idxs2[1], idxs2[2], idxs2[0], idxs1[0], idxs1[1], idxs1[2])) return false;
        if (is_separating(idxs2[2], idxs2[0], idxs2[1], idxs1[0], idxs1[1], idxs1[2])) return false;
        
        return true;
    }
    
    bool solve(int idx, double current_area, double target_area) {
        step_count++;
        if (step_count > 10000) return false;
        
        if (idx < suffix_areas.size() && current_area + suffix_areas[idx] < 0.95 * target_area) {
            return false;
        }
        
        if (abs(current_area - target_area) < 0.05 * target_area) {
            map<pair<int, int>, int> edge_counts;
            for (const auto& t : selected) {
                edge_counts[{min(t.idxs[0], t.idxs[1]), max(t.idxs[0], t.idxs[1])}]++;
                edge_counts[{min(t.idxs[1], t.idxs[2]), max(t.idxs[1], t.idxs[2])}]++;
                edge_counts[{min(t.idxs[2], t.idxs[0]), max(t.idxs[2], t.idxs[0])}]++;
            }
            set<pair<int, int>> boundary_set;
            for (const auto& bc : boundary_constraints) {
                boundary_set.insert({min(bc.first, bc.second), max(bc.first, bc.second)});
            }
            bool ok = true;
            for (const auto& pair : edge_counts) {
                const auto& edge = pair.first;
                int count = pair.second;
                if (boundary_set.count(edge)) {
                    if (count != 1) { ok = false; break; }
                } else {
                    if (count != 2) { ok = false; break; }
                }
            }
            if (ok) {
                for (const auto& edge : boundary_set) {
                    if (!edge_counts.count(edge)) { ok = false; break; }
                }
            }
            if (ok) {
                for (const auto& cons : constraints) {
                    if (!edge_counts.count({min(cons.first, cons.second), max(cons.first, cons.second)})) {
                        ok = false;
                        break;
                    }
                }
            }
            if (ok) return true;
        }
        if (current_area > target_area + 1e-6 * target_area) {
            return false;
        }
        
        for (int i = idx; i < valid_triangles.size(); ++i) {
            bool overlaps = false;
            for (const auto& t : selected) {
                if (interiors_overlap(t, valid_triangles[i])) {
                    overlaps = true;
                    break;
                }
            }
            if (!overlaps) {
                selected.push_back(valid_triangles[i]);
                if (solve(i + 1, current_area + valid_triangles[i].area, target_area)) {
                    return true;
                }
                selected.pop_back();
            }
        }
        return false;
    }
};

double tri_area2d(Vector2 a, Vector2 b, Vector2 c) {
    return 0.5 * cross2d(b - a, c - a);
}

void find_valid_triangles(TriangulationSolver& solver) {
    int n = solver.V2.size();
    for (int a = 0; a < n; ++a) {
        for (int b = a + 1; b < n; ++b) {
            for (int c = b + 1; c < n; ++c) {
                double area = tri_area2d(solver.V2[a], solver.V2[b], solver.V2[c]);
                if (abs(area) < 1e-18) continue;
                
                int idxs[3] = {a, b, c};
                if (solver.flipped) {
                    if (area > 0) swap(idxs[1], idxs[2]);
                } else {
                    if (area < 0) swap(idxs[1], idxs[2]);
                }
                
                bool vertex_inside = false;
                for (int i = 0; i < n; ++i) {
                    if (i == a || i == b || i == c) continue;
                    Vector2 P = solver.V2[i];
                    double cp1 = cross2d(solver.V2[idxs[1]] - solver.V2[idxs[0]], P - solver.V2[idxs[0]]);
                    double cp2 = cross2d(solver.V2[idxs[2]] - solver.V2[idxs[1]], P - solver.V2[idxs[1]]);
                    double cp3 = cross2d(solver.V2[idxs[0]] - solver.V2[idxs[2]], P - solver.V2[idxs[2]]);
                    
                    if (solver.flipped) {
                        if (cp1 < -1e-9 && cp2 < -1e-9 && cp3 < -1e-9) {
                            vertex_inside = true;
                            break;
                        }
                    } else {
                        if (cp1 > 1e-9 && cp2 > 1e-9 && cp3 > 1e-9) {
                            vertex_inside = true;
                            break;
                        }
                    }
                }
                if (vertex_inside) continue;
                
                bool crosses = false;
                for (const auto& cons : solver.constraints) {
                    int u = cons.first;
                    int v = cons.second;
                    
                    if ((u == idxs[0] && v == idxs[1]) || (u == idxs[1] && v == idxs[0]) ||
                        (u == idxs[1] && v == idxs[2]) || (u == idxs[2] && v == idxs[1]) ||
                        (u == idxs[2] && v == idxs[0]) || (u == idxs[0] && v == idxs[2])) {
                        continue;
                    }
                    
                    if (segments_intersect_open(solver.V2[u], solver.V2[v], solver.V2[idxs[0]], solver.V2[idxs[1]]) ||
                        segments_intersect_open(solver.V2[u], solver.V2[v], solver.V2[idxs[1]], solver.V2[idxs[2]]) ||
                        segments_intersect_open(solver.V2[u], solver.V2[v], solver.V2[idxs[2]], solver.V2[idxs[0]])) {
                        crosses = true;
                        break;
                    }
                    
                    auto check_on_any_edge = [&](int pt_idx) {
                        return point_on_segment_2d(solver.V2[pt_idx], solver.V2[idxs[0]], solver.V2[idxs[1]]) ||
                               point_on_segment_2d(solver.V2[pt_idx], solver.V2[idxs[1]], solver.V2[idxs[2]]) ||
                               point_on_segment_2d(solver.V2[pt_idx], solver.V2[idxs[2]], solver.V2[idxs[0]]);
                    };
                    
                    if (check_on_any_edge(u) && check_on_any_edge(v)) {
                        bool same_edge = false;
                        if (point_on_segment_2d(solver.V2[u], solver.V2[idxs[0]], solver.V2[idxs[1]]) &&
                            point_on_segment_2d(solver.V2[v], solver.V2[idxs[0]], solver.V2[idxs[1]])) same_edge = true;
                        if (point_on_segment_2d(solver.V2[u], solver.V2[idxs[1]], solver.V2[idxs[2]]) &&
                            point_on_segment_2d(solver.V2[v], solver.V2[idxs[1]], solver.V2[idxs[2]])) same_edge = true;
                        if (point_on_segment_2d(solver.V2[u], solver.V2[idxs[2]], solver.V2[idxs[0]]) &&
                            point_on_segment_2d(solver.V2[v], solver.V2[idxs[2]], solver.V2[idxs[0]])) same_edge = true;
                            
                        if (!same_edge) {
                            crosses = true;
                            break;
                        }
                    }
                }
                if (crosses) continue;
                
                solver.valid_triangles.push_back({
                    {idxs[0], idxs[1], idxs[2]},
                    abs(area)
                });
            }
        }
    }
    
    sort(solver.valid_triangles.begin(), solver.valid_triangles.end(),
         [](const Triangle2D& t1, const Triangle2D& t2) {
             return t1.area > t2.area;
         });
}

struct Face {
    int v[3];
};

struct FaceGlobal {
    int v[3];
};

void triangulate_mesh_faces(
    bool is_mesh_A,
    int num_original_faces,
    const vector<Face>& original_faces,
    const vector<Triangle3D>& original_triangles,
    int start_global_v_idx,
    const map<int, vector<int>>& face_to_segment_indices,
    const vector<Segment3D>& all_intersection_segments,
    vector<FaceGlobal>& faces_prime
) {
    for (int i = 0; i < num_original_faces; ++i) {
        auto it = face_to_segment_indices.find(i);
        if (it == face_to_segment_indices.end()) {
            faces_prime.push_back({
                start_global_v_idx + original_faces[i].v[0],
                start_global_v_idx + original_faces[i].v[1],
                start_global_v_idx + original_faces[i].v[2]
            });
            continue;
        }
        
        const auto& T = original_triangles[i];
        int g0 = start_global_v_idx + original_faces[i].v[0];
        int g1 = start_global_v_idx + original_faces[i].v[1];
        int g2 = start_global_v_idx + original_faces[i].v[2];
        
        vector<LocalVertex> V;
        V.push_back({T.v[0], g0});
        V.push_back({T.v[1], g1});
        V.push_back({T.v[2], g2});
        
        for (int seg_idx : it->second) {
            const auto& seg = all_intersection_segments[seg_idx];
            V.push_back({seg.p1, seg.g1});
            V.push_back({seg.p2, seg.g2});
        }
        
        vector<LocalVertex> unique_V;
        for (const auto& lv : V) {
            bool dup = false;
            for (const auto& ulv : unique_V) {
                if (lv.global_idx == ulv.global_idx) {
                    dup = true;
                    break;
                }
            }
            if (!dup) unique_V.push_back(lv);
        }
        
        Vector3 normal = normalize(cross(T.v[1] - T.v[0], T.v[2] - T.v[0]));
        int drop_axis = 0;
        if (abs(normal.y) > abs(normal.x)) drop_axis = 1;
        if (abs(normal.z) > abs(normal.y) && abs(normal.z) > abs(normal.x)) drop_axis = 2;
        
        auto project2d = [&](const Vector3& p) -> Vector2 {
            if (drop_axis == 0) return {p.y, p.z};
            if (drop_axis == 1) return {p.x, p.z};
            return {p.x, p.y};
        };
        
        vector<Vector2> V2(unique_V.size());
        for (size_t k = 0; k < unique_V.size(); ++k) {
            V2[k] = project2d(unique_V[k].p);
        }
        
        double min_x = 1e18, max_x = -1e18, min_y = 1e18, max_y = -1e18;
        for (const auto& p : V2) {
            min_x = min(min_x, p.x);
            max_x = max(max_x, p.x);
            min_y = min(min_y, p.y);
            max_y = max(max_y, p.y);
        }
        double dx = max_x - min_x;
        double dy = max_y - min_y;
        double scale = max({dx, dy, 1e-12});
        for (auto& p : V2) {
            p.x = (p.x - min_x) / scale;
            p.y = (p.y - min_y) / scale;
        }
        
        double area2d = tri_area2d(V2[0], V2[1], V2[2]);
        bool flipped = (area2d < 0.0);
        double parent_area = abs(area2d);
        
        vector<pair<int, int>> constraints;
        vector<pair<int, int>> boundary_constraints;
        int orig_local_idx[3] = {0, 1, 2};
        for (int k = 0; k < 3; ++k) {
            int target_g = (k == 0 ? g0 : (k == 1 ? g1 : g2));
            for (size_t m = 0; m < unique_V.size(); ++m) {
                if (unique_V[m].global_idx == target_g) {
                    orig_local_idx[k] = m;
                    break;
                }
            }
        }
        
        for (int k = 0; k < 3; ++k) {
            Vector3 A3 = (k == 0 ? T.v[0] : (k == 1 ? T.v[1] : T.v[2]));
            Vector3 B3 = (k == 0 ? T.v[1] : (k == 1 ? T.v[2] : T.v[0]));
            
            vector<pair<double, int>> pts_on_edge;
            for (size_t m = 0; m < unique_V.size(); ++m) {
                if (point_on_segment_3d(unique_V[m].p, A3, B3)) {
                    double dist = dot(unique_V[m].p - A3, B3 - A3);
                    pts_on_edge.push_back({dist, (int)m});
                }
            }
            sort(pts_on_edge.begin(), pts_on_edge.end());
            for (size_t m = 0; m + 1 < pts_on_edge.size(); ++m) {
                int c1 = pts_on_edge[m].second;
                int c2 = pts_on_edge[m+1].second;
                constraints.push_back({min(c1, c2), max(c1, c2)});
                boundary_constraints.push_back({min(c1, c2), max(c1, c2)});
            }
        }
        
        for (int seg_idx : it->second) {
            const auto& seg = all_intersection_segments[seg_idx];
            int u_local = -1, v_local = -1;
            for (size_t m = 0; m < unique_V.size(); ++m) {
                if (unique_V[m].global_idx == seg.g1) u_local = m;
                if (unique_V[m].global_idx == seg.g2) v_local = m;
            }
            if (u_local != -1 && v_local != -1) {
                constraints.push_back({min(u_local, v_local), max(u_local, v_local)});
            }
        }
        
        sort(constraints.begin(), constraints.end());
        constraints.erase(unique(constraints.begin(), constraints.end()), constraints.end());
        
        TriangulationSolver solver;
        solver.V2 = V2;
        solver.constraints = constraints;
        solver.boundary_constraints = boundary_constraints;
        solver.parent_area = parent_area;
        solver.flipped = flipped;
        
        find_valid_triangles(solver);
        
        solver.suffix_areas.resize(solver.valid_triangles.size() + 1, 0.0);
        for (int j = (int)solver.valid_triangles.size() - 1; j >= 0; --j) {
            solver.suffix_areas[j] = solver.suffix_areas[j + 1] + solver.valid_triangles[j].area;
        }
        
        bool success = solver.solve(0, 0.0, parent_area);
        
        if (success) {
            for (const auto& t : solver.selected) {
                faces_prime.push_back({
                    unique_V[t.idxs[0]].global_idx,
                    unique_V[t.idxs[1]].global_idx,
                    unique_V[t.idxs[2]].global_idx
                });
            }
        } else {
            faces_prime.push_back({g0, g1, g2});
        }
    }
}

struct EdgeToTriangle {
    int u, v;
    int tri_idx;
    int edge_num;
};

void build_neighbors(int num_triangles, const vector<FaceGlobal>& faces, vector<vector<int>>& neighbors, vector<vector<pair<int, int>>>& edge_vertices) {
    neighbors.assign(num_triangles, vector<int>(3, -1));
    edge_vertices.assign(num_triangles, vector<pair<int, int>>(3));
    
    vector<EdgeToTriangle> ett;
    ett.reserve(num_triangles * 3);
    
    for (int i = 0; i < num_triangles; ++i) {
        int a = faces[i].v[0];
        int b = faces[i].v[1];
        int c = faces[i].v[2];
        
        ett.push_back({min(a, b), max(a, b), i, 0});
        ett.push_back({min(b, c), max(b, c), i, 1});
        ett.push_back({min(c, a), max(c, a), i, 2});
        
        edge_vertices[i][0] = {a, b};
        edge_vertices[i][1] = {b, c};
        edge_vertices[i][2] = {c, a};
    }
    
    sort(ett.begin(), ett.end(), [](const EdgeToTriangle& x, const EdgeToTriangle& y) {
        if (x.u != y.u) return x.u < y.u;
        return x.v < y.v;
    });
    
    for (size_t i = 0; i + 1 < ett.size(); ) {
        if (ett[i].u == ett[i+1].u && ett[i].v == ett[i+1].v) {
            int tri1 = ett[i].tri_idx;
            int edge1 = ett[i].edge_num;
            int tri2 = ett[i+1].tri_idx;
            int edge2 = ett[i+1].edge_num;
            
            neighbors[tri1][edge1] = tri2;
            neighbors[tri2][edge2] = tri1;
            
            i += 2;
        } else {
            i += 1;
        }
    }
}

struct MergePoint {
    Vector3 p;
    int original_idx;
};

bool compare_merge_points(const MergePoint& a, const MergePoint& b) {
    if (a.p.x != b.p.x) return a.p.x < b.p.x;
    if (a.p.y != b.p.y) return a.p.y < b.p.y;
    return a.p.z < b.p.z;
}

        triangles_B[i].v[2] = perturbed_vertices_B[faces_B[i].v[2]];

            if (length(merge_points[i].p - merge_points[j].p) < 1e-8) {
                match_idx = parent_map[merge_points[j].original_idx];
                break;
            }
        }
        if (match_idx == -1) {

            unique_intersection_vertices.push_back(merge_points[i].p);
        }
        parent_map[merge_points[i].original_idx] = match_idx;
    }
    
    int num_unique_int_v = unique_intersection_vertices.size();
    int start_idx_int_v = VA + VB;
    for (size_t i = 0; i < all_intersection_segments.size(); ++i) {
        int g1_local = parent_map[2 * i];
        int g2_local = parent_map[2 * i + 1];
        all_intersection_segments[i].g1 = start_idx_int_v + g1_local;
        all_intersection_segments[i].g2 = start_idx_int_v + g2_local;
    }
    
    vector<Vector3> global_vertices;
    global_vertices.reserve(VA + VB + num_unique_int_v);
    for (int i = 0; i < VA; ++i) global_vertices.push_back(vertices_A[i]);
    for (int i = 0; i < VB; ++i) global_vertices.push_back(perturbed_vertices_B[i]);
    for (int i = 0; i < num_unique_int_v; ++i) global_vertices.push_back(unique_intersection_vertices[i]);
    
    vector<FaceGlobal> faces_A_prime;

    triangulate_mesh_faces(

    vector<vector<int>> neighbors_B_prime;
    vector<vector<pair<int, int>>> edge_vertices_B_prime;
    build_neighbors(faces_B_prime.size(), faces_B_prime, neighbors_B_prime, edge_vertices_B_prime);
    
    int unpaired_A = 0;
    for (size_t i = 0; i < faces_A_prime.size(); ++i) {
        for (int j = 0; j < 3; ++j) {
            if (neighbors_A_prime[i][j] == -1) {
                unpaired_A++;
                int u = edge_vertices_A_prime[i][j].first;
                int v = edge_vertices_A_prime[i][j].second;
                std::cerr << "  Unpaired edge: (" << u << ", " << v << ") on triangle " << i 
                          << " with vertices (" << faces_A_prime[i].v[0] << ", " 
                          << faces_A_prime[i].v[1] << ", " << faces_A_prime[i].v[2] << ")" << std::endl;
            }

    for (const auto& seg : all_intersection_segments) {
        intersection_edges_set.insert({min(seg.g1, seg.g2), max(seg.g1, seg.g2)});
    }
    
    std::cerr << "DEBUG: intersection_edges_set contents:" << std::endl;
    for (const auto& edge : intersection_edges_set) {
        std::cerr << "  (" << edge.first << ", " << edge.second << ")" << std::endl;
    }
    
    int num_triangles_A_prime = faces_A_prime.size();
    vector<int> component_id_A(num_triangles_A_prime, -1);
    int num_components_A = 0;
    
    for (int i = 0; i < num_triangles_A_prime; ++i) {
        if (component_id_A[i] != -1) continue;
        
        vector<int> q = {i};
        component_id_A[i] = num_components_A;
        int head = 0;
        while (head < q.size()) {
            int curr = q[head++];
            for (int j = 0; j < 3; ++j) {
                int neighbor = neighbors_A_prime[curr][j];
                if (neighbor == -1) continue;
                if (component_id_A[neighbor] != -1) continue;
                
                int u = edge_vertices_A_prime[curr][j].first;
                int v = edge_vertices_A_prime[curr][j].second;
                pair<int, int> edge_key = {min(u, v), max(u, v)};
                if (intersection_edges_set.count(edge_key)) {
                    std::cerr << "DEBUG: Found intersection edge in BFS: (" << edge_key.first << ", " << edge_key.second << ")" << std::endl;
                    continue;
                }
                
                component_id_A[neighbor] = num_components_A;
                q.push_back(neighbor);
            }
        }
        num_components_A++;
    }
    

    int num_triangles_B_prime = faces_B_prime.size();
    vector<int> component_id_B(num_triangles_B_prime, -1);
    int num_components_B = 0;
    
    for (int i = 0; i < num_triangles_B_prime; ++i) {
        if (component_id_B[i] != -1) continue;
        
        vector<int> q = {i};
        component_id_B[i] = num_components_B;
        int head = 0;
        while (head < q.size()) {
            int curr = q[head++];
            for (int j = 0; j < 3; ++j) {
                int neighbor = neighbors_B_prime[curr][j];
                if (neighbor == -1) continue;
                if (component_id_B[neighbor] != -1) continue;
                
                int u = edge_vertices_B_prime[curr][j].first;
                int v = edge_vertices_B_prime[curr][j].second;
                if (intersection_edges_set.count({min(u, v), max(u, v)})) {
                    continue;
                }
                
                component_id_B[neighbor] = num_components_B;
                q.push_back(neighbor);
            }
        }
        num_components_B++;
    }
    
    vector<bool> component_outside_B(num_components_A, true);
    vector<double> area_A_prime(num_triangles_A_prime);
    for (int i = 0; i < num_triangles_A_prime; ++i) {
        Vector3 a = global_vertices[faces_A_prime[i].v[0]];
        Vector3 b = global_vertices[faces_A_prime[i].v[1]];
        Vector3 c = global_vertices[faces_A_prime[i].v[2]];
        area_A_prime[i] = length(cross(b - a, c - a)) * 0.5;
    }
    
    vector<vector<int>> component_to_triangles_A(num_components_A);
    for (int i = 0; i < num_triangles_A_prime; ++i) {
        component_to_triangles_A[component_id_A[i]].push_back(i);
    }
    
    std::cerr << "DEBUG: num_components_A = " << num_components_A << std::endl;
    for (int c = 0; c < num_components_A; ++c) {
        std::cerr << "  Component " << c << " has " << component_to_triangles_A[c].size() << " faces." << std::endl;
    }
    
    vector<Triangle3D> triangles_B_prime(num_triangles_B_prime);
    for (int i = 0; i < num_triangles_B_prime; ++i) {
        triangles_B_prime[i].v[0] = global_vertices[faces_B_prime[i].v[0]];
        triangles_B_prime[i].v[1] = global_vertices[faces_B_prime[i].v[1]];
        triangles_B_prime[i].v[2] = global_vertices[faces_B_prime[i].v[2]];
    }
    
    vector<int> indices_B_prime(num_triangles_B_prime);
    for (int i = 0; i < num_triangles_B_prime; ++i) indices_B_prime[i] = i;
    vector<AABBNode> nodes_B_prime;
    int root_B_prime = -1;
    if (num_triangles_B_prime > 0) {
        root_B_prime = build_aabb_tree(indices_B_prime, 0, num_triangles_B_prime, nodes_B_prime, triangles_B_prime);
    }
    
    for (int c = 0; c < num_components_A; ++c) {
        int rep_idx = component_to_triangles_A[c][0];
        double max_area = area_A_prime[rep_idx];
        for (int tri_idx : component_to_triangles_A[c]) {
            if (area_A_prime[tri_idx] > max_area) {
                max_area = area_A_prime[tri_idx];
                rep_idx = tri_idx;
            }
        }
        
        Vector3 a = global_vertices[faces_A_prime[rep_idx].v[0]];
        Vector3 b = global_vertices[faces_A_prime[rep_idx].v[1]];
        Vector3 c_pt = global_vertices[faces_A_prime[rep_idx].v[2]];
        Vector3 centroid = (a + b + c_pt) / 3.0;
        
        bool inside = is_point_inside(centroid, nodes_B_prime, triangles_B_prime);
        component_outside_B[c] = !inside;
    }
    
    vector<bool> component_outside_A(num_components_B, true);
    vector<double> area_B_prime(num_triangles_B_prime);
    for (int i = 0; i < num_triangles_B_prime; ++i) {
        Vector3 a = global_vertices[faces_B_prime[i].v[0]];
        Vector3 b = global_vertices[faces_B_prime[i].v[1]];
        Vector3 c = global_vertices[faces_B_prime[i].v[2]];
        area_B_prime[i] = length(cross(b - a, c - a)) * 0.5;
    }
    
    vector<vector<int>> component_to_triangles_B(num_components_B);
    for (int i = 0; i < num_triangles_B_prime; ++i) {
        component_to_triangles_B[component_id_B[i]].push_back(i);
    }
    
    std::cerr << "DEBUG: num_components_B = " << num_components_B << std::endl;
    for (int c = 0; c < num_components_B; ++c) {
        std::cerr << "  Component " << c << " has " << component_to_triangles_B[c].size() << " faces." << std::endl;
    }
    
    vector<Triangle3D> triangles_A_prime(num_triangles_A_prime);
    for (int i = 0; i < num_triangles_A_prime; ++i) {
        triangles_A_prime[i].v[0] = global_vertices[faces_A_prime[i].v[0]];
        triangles_A_prime[i].v[1] = global_vertices[faces_A_prime[i].v[1]];
        triangles_A_prime[i].v[2] = global_vertices[faces_A_prime[i].v[2]];
    }
    
    vector<int> indices_A_prime(num_triangles_A_prime);
    for (int i = 0; i < num_triangles_A_prime; ++i) indices_A_prime[i] = i;
    vector<AABBNode> nodes_A_prime;
    int root_A_prime = -1;
    if (num_triangles_A_prime > 0) {
        root_A_prime = build_aabb_tree(indices_A_prime, 0, num_triangles_A_prime, nodes_A_prime, triangles_A_prime);
    }
    
    for (int c = 0; c < num_components_B; ++c) {
        int rep_idx = component_to_triangles_B[c][0];
        double max_area = area_B_prime[rep_idx];
        for (int tri_idx : component_to_triangles_B[c]) {
            if (area_B_prime[tri_idx] > max_area) {
                max_area = area_B_prime[tri_idx];
                rep_idx = tri_idx;
            }
        }
        
        Vector3 a = global_vertices[faces_B_prime[rep_idx].v[0]];
        Vector3 b = global_vertices[faces_B_prime[rep_idx].v[1]];
        Vector3 c_pt = global_vertices[faces_B_prime[rep_idx].v[2]];
        Vector3 centroid = (a + b + c_pt) / 3.0;
        
        bool inside = is_point_inside(centroid, nodes_A_prime, triangles_A_prime);
        component_outside_A[c] = !inside;
    }
    
    vector<FaceGlobal> union_faces;
    for (int i = 0; i < num_triangles_A_prime; ++i) {
        if (component_outside_B[component_id_A[i]]) {
            union_faces.push_back(faces_A_prime[i]);
        }
    }
    for (int i = 0; i < num_triangles_B_prime; ++i) {
        if (component_outside_A[component_id_B[i]]) {
            union_faces.push_back(faces_B_prime[i]);
        }
    }
    
    vector<int> vertex_map(global_vertices.size(), -1);
    vector<Vector3> final_vertices;
    for (auto& f : union_faces) {
        for (int j = 0; j < 3; ++j) {
            int v_idx = f.v[j];
            if (vertex_map[v_idx] == -1) {
                vertex_map[v_idx] = final_vertices.size();
                final_vertices.push_back(global_vertices[v_idx]);
            }
            f.v[j] = vertex_map[v_idx];
        }
    }
    
    cout << final_vertices.size() << " " << union_faces.size() << "\n";
    cout << fixed << setprecision(10);
    for (const auto& v : final_vertices) {
        cout << v.x << " " << v.y << " " << v.z << "\n";
    }
    for (const auto& f : union_faces) {
        cout << f.v[0] << " " << f.v[1] << " " << f.v[2] << "\n";
    }
    
    return 0;
}
