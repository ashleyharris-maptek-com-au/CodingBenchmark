using System;
using System.Collections.Generic;
using System.Linq;

namespace QAPSolver
{
    public struct FastRandom
    {
        private uint x, y, z, w;

        public FastRandom(uint seed)
        {
            x = seed;
            y = x ^ 123456789;
            z = y ^ 987654321;
            w = z ^ 43219876;
        }

        public uint Next()
        {
            uint t = x ^ (x << 11);
            x = y; y = z; z = w;
            return w = w ^ (w >> 19) ^ (t ^ (t >> 8));
        }

        public int Next(int max)
        {
            // Lemire's reduction to map to [0, max-1] without division/modulo
            return (int)(((ulong)Next() * (ulong)max) >> 32);
        }

        public double NextDouble()
        {
            return (double)Next() / uint.MaxValue;
        }
    }

    public class Tokenizer
    {
        private readonly System.IO.TextReader reader;
        private string[] currentTokens = new string[0];
        private int currentIdx = 0;

        public Tokenizer(System.IO.TextReader reader)
        {
            this.reader = reader;
        }

        public string NextToken()
        {
            while (currentIdx >= currentTokens.Length)
            {
                string line = reader.ReadLine();
                if (line == null) return null;
                currentTokens = line.Split(new[] { ' ', '\t', '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries);
                currentIdx = 0;
            }
            return currentTokens[currentIdx++];
        }

        public int NextInt()
        {
            string token = NextToken();
            if (token == null) throw new System.IO.EndOfStreamException();
            return int.Parse(token);
        }
    }

    public class Program
    {
        public static void Main(string[] args)
        {
            // Optimize Console I/O
            Console.OutputEncoding = System.Text.Encoding.UTF8;

            Tokenizer tokenizer = new Tokenizer(Console.In);
            string nToken = tokenizer.NextToken();
            if (nToken == null) return;
            int N = int.Parse(nToken);

            int[] flow_row = new int[N * N];
            int[] flow_col = new int[N * N];
            for (int i = 0; i < N; i++)
            {
                for (int j = 0; j < N; j++)
                {
                    int val = tokenizer.NextInt();
                    flow_row[i * N + j] = val;
                    flow_col[j * N + i] = val;
                }
            }

            int[] dist_row = new int[N * N];
            int[] dist_col = new int[N * N];
            for (int i = 0; i < N; i++)
            {
                for (int j = 0; j < N; j++)
                {
                    int val = tokenizer.NextInt();
                    dist_row[i * N + j] = val;
                    dist_col[j * N + i] = val;
                }
            }

            if (N == 1)
            {
                long cost = (long)flow_row[0] * dist_row[0];
                Console.WriteLine(cost);
                Console.WriteLine("0");
                return;
            }

            // Check if all elements are non-negative
            bool all_non_negative = true;
            for (int i = 0; i < N * N; i++)
            {
                if (flow_row[i] < 0 || dist_row[i] < 0)
                {
                    all_non_negative = false;
                    break;
                }
            }

            // Solve using exact search if N is small
            if (N <= 10 || (N <= 14 && all_non_negative))
            {
                SolveExact(N, flow_row, dist_row, all_non_negative);
                return;
            }

            // Otherwise, use metaheuristic (ILS with Simulated Annealing)
            System.Diagnostics.Stopwatch sw = System.Diagnostics.Stopwatch.StartNew();
            long timeLimitMs = 295000; // 295 seconds (5-minute limit is 300s)

            FastRandom rand = new FastRandom(1337);

            // Initial solution: greedy constructive
            int[] best_p = GetGreedySolution(N, flow_row, dist_row);
            long best_cost = ComputeCost(N, best_p, flow_row, dist_row);

            // First run of SA is long and comprehensive
            double T_start = CalibrateTStart(N, best_p, flow_row, flow_col, dist_row, dist_col);
            double T_end = 0.001;
            int M_first = Math.Max(1000000, 5000 * N);

            RunSA(N, flow_row, flow_col, dist_row, dist_col,
                  best_p, best_cost, T_start, T_end, M_first,
                  sw, timeLimitMs, ref best_p, ref best_cost, rand);

            // Subsequent runs (Iterated Local Search style)
            int M_sub = Math.Max(100000, 500 * N);
            
            // Perturbation size (number of swaps)
            int K = Math.Max(2, Math.Min(50, N / 50));

            int[] curr_p = new int[N];

            while (sw.ElapsedMilliseconds < timeLimitMs)
            {
                // Perturb the current best solution to get a new starting solution
                Array.Copy(best_p, curr_p, N);
                for (int k = 0; k < K; k++)
                {
                    int u = rand.Next(N);
                    int v = rand.Next(N);
                    while (u == v) v = rand.Next(N);
                    int tmp = curr_p[u];
                    curr_p[u] = curr_p[v];
                    curr_p[v] = tmp;
                }

                long curr_cost = ComputeCost(N, curr_p, flow_row, dist_row);

                // Calibrate T_start for the perturbed solution, use slightly lower temperature
                double T_start_sub = CalibrateTStart(N, curr_p, flow_row, flow_col, dist_row, dist_col) * 0.5;

                RunSA(N, flow_row, flow_col, dist_row, dist_col,
                      curr_p, curr_cost, T_start_sub, T_end, M_sub,
                      sw, timeLimitMs, ref best_p, ref best_cost, rand);
            }

            // Verify and output the best result
            long final_cost = ComputeCost(N, best_p, flow_row, dist_row);
            Console.WriteLine(final_cost);
            Console.WriteLine(string.Join(" ", best_p));
        }

        private static void SolveExact(int N, int[] flow_row, int[] dist_row, bool all_non_negative)
        {
            int[] current_p = new int[N];
            bool[] used = new bool[N];
            
            // Initialize with greedy solution to prune aggressively
            int[] best_p = GetGreedySolution(N, flow_row, dist_row);
            long best_cost = ComputeCost(N, best_p, flow_row, dist_row);
            
            Backtrack(0, 0, N, flow_row, dist_row, current_p, used, ref best_cost, best_p, all_non_negative);
            
            Console.WriteLine(best_cost);
            Console.WriteLine(string.Join(" ", best_p));
        }

        private static void Backtrack(int fac, long current_cost, int N, int[] flow_row, int[] dist_row, 
                                      int[] current_p, bool[] used, ref long best_cost, int[] best_p, bool all_non_negative)
        {
            if (fac == N)
            {
                if (current_cost < best_cost)
                {
                    best_cost = current_cost;
                    Array.Copy(current_p, best_p, N);
                }
                return;
            }

            for (int loc = 0; loc < N; loc++)
            {
                if (!used[loc])
                {
                    long added_cost = 0;
                    int offset_fac = fac * N;
                    int offset_loc = loc * N;
                    for (int j = 0; j < fac; j++)
                    {
                        int pj = current_p[j];
                        added_cost += (long)flow_row[offset_fac + j] * dist_row[offset_loc + pj] +
                                      (long)flow_row[j * N + fac] * dist_row[pj * N + loc];
                    }
                    added_cost += (long)flow_row[offset_fac + fac] * dist_row[offset_loc + loc];

                    long new_cost = current_cost + added_cost;
                    if (all_non_negative && new_cost >= best_cost)
                    {
                        continue; // Prune
                    }

                    used[loc] = true;
                    current_p[fac] = loc;
                    Backtrack(fac + 1, new_cost, N, flow_row, dist_row, current_p, used, ref best_cost, best_p, all_non_negative);
                    used[loc] = false;
                }
            }
        }

        private static int[] GetGreedySolution(int N, int[] flow_row, int[] dist_row)
        {
            long[] f_sum = new long[N];
            long[] d_sum = new long[N];
            for (int i = 0; i < N; i++)
            {
                for (int j = 0; j < N; j++)
                {
                    f_sum[i] += flow_row[i * N + j] + flow_row[j * N + i];
                    d_sum[i] += dist_row[i * N + j] + dist_row[j * N + i];
                }
            }
            
            var f_indices = Enumerable.Range(0, N).OrderByDescending(i => f_sum[i]).ToArray();
            var d_indices = Enumerable.Range(0, N).OrderBy(i => d_sum[i]).ToArray();
            
            int[] p = new int[N];
            for (int k = 0; k < N; k++)
            {
                p[f_indices[k]] = d_indices[k];
            }
            return p;
        }

        private static long ComputeCost(int N, int[] p, int[] flow_row, int[] dist_row)
        {
            long cost = 0;
            for (int i = 0; i < N; i++)
            {
                int pi = p[i];
                int offset_f = i * N;
                int offset_d = pi * N;
                for (int j = 0; j < N; j++)
                {
                    cost += (long)flow_row[offset_f + j] * dist_row[offset_d + p[j]];
                }
            }
            return cost;
        }

        private static double CalibrateTStart(int N, int[] p, int[] flow_row, int[] flow_col, int[] dist_row, int[] dist_col)
        {
            FastRandom rand = new FastRandom(42);
            List<long> positiveDeltas = new List<long>();
            int samples = Math.Min(1000, N * N);
            for (int i = 0; i < samples; i++)
            {
                int u = rand.Next(N);
                int v = rand.Next(N);
                while (u == v) v = rand.Next(N);
                
                long delta = ComputeDelta(N, p, u, v, flow_row, flow_col, dist_row, dist_col);
                if (delta > 0)
                {
                    positiveDeltas.Add(delta);
                }
            }
            if (positiveDeltas.Count == 0) return 1.0;
            double avgDelta = positiveDeltas.Average();
            return -avgDelta / Math.Log(0.7); // 70% acceptance probability for average bad swap
        }

        private static void RunSA(int N, int[] flow_row, int[] flow_col, int[] dist_row, int[] dist_col,
                                  int[] start_p, long start_cost, double T_start, double T_end, int M,
                                  System.Diagnostics.Stopwatch sw, long timeLimitMs,
                                  ref int[] best_p, ref long best_cost, FastRandom rand)
        {
            int[] p = new int[N];
            Array.Copy(start_p, p, N);
            long current_cost = start_cost;
            
            double T = T_start;
            double alpha = Math.Pow(T_end / T_start, 1.0 / M);

            for (int iter = 0; iter < M; iter++)
            {
                if ((iter & 65535) == 0)
                {
                    if (sw.ElapsedMilliseconds > timeLimitMs)
                    {
                        break;
                    }
                }

                int u = rand.Next(N);
                int v = rand.Next(N);
                while (u == v)
                {
                    v = rand.Next(N);
                }

                long delta = ComputeDelta(N, p, u, v, flow_row, flow_col, dist_row, dist_col);

                bool accept = false;
                if (delta < 0)
                {
                    accept = true;
                }
                else
                {
                    double prob = Math.Exp(-delta / T);
                    if (rand.NextDouble() < prob)
                    {
                        accept = true;
                    }
                }

                if (accept)
                {
                    int tmp = p[u];
                    p[u] = p[v];
                    p[v] = tmp;
                    
                    current_cost += delta;

                    if (current_cost < best_cost)
                    {
                        best_cost = current_cost;
                        Array.Copy(p, best_p, N);
                    }
                }

                T *= alpha;
            }
        }

        private static long ComputeDelta(int N, int[] p, int u, int v, int[] flow_row, int[] flow_col, int[] dist_row, int[] dist_col)
        {
            int pu = p[u];
            int pv = p[v];
            
            long sum = 0;
            int offset_flow_u = u * N;
            int offset_flow_v = v * N;
            int offset_dist_pv = pv * N;
            int offset_dist_pu = pu * N;

            for (int k = 0; k < N; k++)
            {
                int pk = p[k];
                long f_diff_row = flow_row[offset_flow_u + k] - flow_row[offset_flow_v + k];
                long d_diff_row = dist_row[offset_dist_pv + pk] - dist_row[offset_dist_pu + pk];
                long f_diff_col = flow_col[offset_flow_u + k] - flow_col[offset_flow_v + k];
                long d_diff_col = dist_col[offset_dist_pv + pk] - dist_col[offset_dist_pu + pk];
                sum += f_diff_row * d_diff_row + f_diff_col * d_diff_col;
            }

            long term_u = GetLoopTerm(u, u, v, pu, pv, N, p, flow_row, flow_col, dist_row, dist_col);
            long term_v = GetLoopTerm(v, u, v, pu, pv, N, p, flow_row, flow_col, dist_row, dist_col);

            sum -= (term_u + term_v);

            long self_cross = (long)(flow_row[u * N + u] - flow_row[v * N + v]) * (dist_row[pv * N + pv] - dist_row[pu * N + pu]) +
                              (long)(flow_row[u * N + v] - flow_row[v * N + u]) * (dist_row[pv * N + pu] - dist_row[pu * N + pv]);
            sum += self_cross;

            return sum;
        }

        private static long GetLoopTerm(int k, int u, int v, int pu, int pv, int N, int[] p, int[] flow_row, int[] flow_col, int[] dist_row, int[] dist_col)
        {
            int pk = p[k];
            long f_diff_row = flow_row[u * N + k] - flow_row[v * N + k];
            long d_diff_row = dist_row[pv * N + pk] - dist_row[pu * N + pk];
            long f_diff_col = flow_col[u * N + k] - flow_col[v * N + k];
            long d_diff_col = dist_col[pv * N + pk] - dist_col[pu * N + pk];
            return f_diff_row * d_diff_row + f_diff_col * d_diff_col;
        }
    }
}
