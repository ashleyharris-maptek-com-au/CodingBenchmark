    public struct Vector3
    {
        public double X, Y, Z;
        public Vector3(double x, double y, double z) { X = x; Y = y; Z = z; }
        public static Vector3 Add(Vector3 a, Vector3 b) => new Vector3(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
        public static Vector3 Subtract(Vector3 a, Vector3 b) => new Vector3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
        public static Vector3 Multiply(Vector3 a, double scale) => new Vector3(a.X * scale, a.Y * scale, a.Z * scale);
        public static double Dot(Vector3 a, Vector3 b) => a.X * b.X + a.Y * b.Y + a.Z * b.Z;
        public static Vector3 Cross(Vector3 a, Vector3 b) => new Vector3(
            a.Y * b.Z - a.Z * b.Y,
            a.Z * b.X - a.X * b.Z,
            a.X * b.Y - a.Y * b.X
        );
        public double Length() => Math.Sqrt(X * X + Y * Y + Z * Z);
        public Vector3 Normalize()
        {
            double len = Length();
            return len > 1e-9 ? new Vector3(X / len, Y / len, Z / len) : this;
        }
    }

    public struct Quaternion
    {
        public double W, X, Y, Z;
        public Quaternion(double w, double x, double y, double z) { W = w; X = x; Y = y; Z = z; }

        public static Quaternion Multiply(Quaternion a, Quaternion b)
        {
            return new Quaternion(
                a.W * b.W - a.X * b.X - a.Y * b.Y - a.Z * b.Z,
                a.W * b.X + a.X * b.W + a.Y * b.Z - a.Z * b.Y,
                a.W * b.Y - a.X * b.Z + a.Y * b.W + a.Z * b.X,
                a.W * b.Z + a.X * b.Y - a.Y * b.X + a.Z * b.W
            );
        }

        public Quaternion Conjugate() => new Quaternion(W, -X, -Y, -Z);

        public Vector3 Rotate(Vector3 v)
        {
            Vector3 u = new Vector3(X, Y, Z);
            double dotUV = Vector3.Dot(u, v);
            double dotUU = Vector3.Dot(u, u);
            Vector3 crossUV = Vector3.Cross(u, v);

            Vector3 term1 = Vector3.Multiply(u, 2.0 * dotUV);
            Vector3 term2 = Vector3.Multiply(v, W * W - dotUU);
            Vector3 term3 = Vector3.Multiply(crossUV, 2.0 * W);

            return Vector3.Add(Vector3.Add(term1, term2), term3);
        }
    }

    public struct Face
    {
        public int A, B, C;
        public Face(int a, int b, int c) { A = a; B = b; C = c; }
    }

    public class PlacedItem
    {
        public Quaternion Q;
        public double Tx, Ty, Tz;
    }

    public class VoxelTemplate
    {
        public Quaternion Q;
        public int W, H, D;
        public int XStart, YStart, ZStart;
        public List<(int X, int Y, int Z)> ActiveVoxels = new List<(int X, int Y, int Z)>();
        public int[] VoxelOffsets;

        public void ComputeOffsets(int containerH, int containerD)
        {
            VoxelOffsets = new int[ActiveVoxels.Count];
            for (int i = 0; i < ActiveVoxels.Count; i++)
            {
                var v = ActiveVoxels[i];
                VoxelOffsets[i] = v.X * (containerH * containerD) + v.Y * containerD + v.Z;
            }
        }
    }

    public class AABBTemplate
    {
        public Quaternion Q;
        public double MinX, MinY, MinZ;
        public double W, H, D;
    }

    public class BitGrid3D
    {
        private readonly ulong[] _data;
        private readonly int _width;
        private readonly int _height;
        private readonly int _depth;
        private readonly int _strideY;

        public BitGrid3D(int width, int height, int depth)
        {
            _width = width;
            _height = height;
            _depth = depth;
            _strideY = depth;
            long totalBits = (long)width * height * depth;
            _data = new ulong[(totalBits + 63) / 64];
        }

        public bool IsSetFlat(int index)
        {
            return (_data[index >> 6] & (1UL << (index & 63))) != 0;
        }

        public void SetFlat(int index)
        {
            _data[index >> 6] |= (1UL << (index & 63));
        }
    }

    public class Program
    {
        public static List<Quaternion> GenerateOrthotropicRotations()
        {
            var rotations = new List<Quaternion>();
            double r = Math.Sqrt(0.5);
            var qx = new Quaternion(r, r, 0, 0);
            var qy = new Quaternion(r, 0, r, 0);
            var qz = new Quaternion(r, 0, 0, r);

            var queue = new Queue<Quaternion>();
            queue.Enqueue(new Quaternion(1, 0, 0, 0));
            rotations.Add(queue.Peek());

            while (queue.Count > 0)
            {
                var curr = queue.Dequeue();
                var nextRotations = new[] {
                    Quaternion.Multiply(curr, qx),
                    Quaternion.Multiply(curr, qy),
                    Quaternion.Multiply(curr, qz)
                };
                foreach (var next in nextRotations)
                {
                    double len = Math.Sqrt(next.W * next.W + next.X * next.X + next.Y * next.Y + next.Z * next.Z);
                    var norm = new Quaternion(next.W / len, next.X / len, next.Y / len, next.Z / len);

                    bool duplicate = false;
                    foreach (var existing in rotations)
                    {
                        double dot = existing.W * norm.W + existing.X * norm.X + existing.Y * norm.Y + existing.Z * norm.Z;
                        if (Math.Abs(dot) > 0.999)
                        {
                            duplicate = true;
                            break;
                        }
                    }
                    if (!duplicate)
                    {
                        rotations.Add(norm);
                        queue.Enqueue(norm);
                    }
                }
            }
            return rotations;
        }

        public static List<VoxelTemplate> GenerateVoxelTemplates(Vector3[] vertices, Face[] faces, double S, int W_s, int H_s, int D_s)
        {
            var rotations = GenerateOrthotropicRotations();
            var allTemplates = new List<VoxelTemplate>();

            // Compute polyhedron centroid once
            Vector3 baseCentroid = new Vector3(0, 0, 0);
            foreach (var v in vertices)
            {
                baseCentroid = Vector3.Add(baseCentroid, v);
            }
            baseCentroid = Vector3.Multiply(baseCentroid, 1.0 / vertices.Length);

            foreach (var q in rotations)
            {
                var rotatedVertices = new Vector3[vertices.Length];
                for (int i = 0; i < vertices.Length; i++)
                {
                    rotatedVertices[i] = q.Rotate(vertices[i]);
                }

                // Scale rotated vertices
                var scaledVertices = new Vector3[vertices.Length];
                for (int i = 0; i < vertices.Length; i++)
                {
                    scaledVertices[i] = Vector3.Multiply(rotatedVertices[i], S);
                }

                Vector3 centroid = q.Rotate(baseCentroid);
                centroid = Vector3.Multiply(centroid, S);

                double minX = double.MaxValue, maxX = double.MinValue;
                double minY = double.MaxValue, maxY = double.MinValue;
                double minZ = double.MaxValue, maxZ = double.MinValue;

                foreach (var v in scaledVertices)
                {
                    if (v.X < minX) minX = v.X;
                    if (v.X > maxX) maxX = v.X;
                    if (v.Y < minY) minY = v.Y;
                    if (v.Y > maxY) maxY = v.Y;
                    if (v.Z < minZ) minZ = v.Z;
                    if (v.Z > maxZ) maxZ = v.Z;
                }

                int xStart = (int)Math.Floor(minX);
                int xEnd = (int)Math.Ceiling(maxX);
                int yStart = (int)Math.Floor(minY);
                int yEnd = (int)Math.Ceiling(maxY);
                int zStart = (int)Math.Floor(minZ);
                int zEnd = (int)Math.Ceiling(maxZ);

                int w = Math.Max(1, xEnd - xStart);
                int h = Math.Max(1, yEnd - yStart);
                int d = Math.Max(1, zEnd - zStart);

                // If this orientation doesn't fit in the container, discard it
                if (w > W_s || h > H_s || d > D_s)
                    continue;

                var temp = new VoxelTemplate
                {
                    Q = q,
                    W = w,
                    H = h,
                    D = d,
                    XStart = xStart,
                    YStart = yStart,
                    ZStart = zStart
                };

                long voxelCount = (long)w * h * d;
                if (voxelCount > 50000)
                {
                    // Safeguard: too complex/large voxelization, represent as solid AABB
                    for (int x = 0; x < w; x++)
                    for (int y = 0; y < h; y++)
                    for (int z = 0; z < d; z++)
                    {
                        temp.ActiveVoxels.Add((x, y, z));
                    }
                }
                else
                {
                    // Compute plane equations for each face
                    var planes = new List<(Vector3 Normal, double Dist)>();
                    foreach (var face in faces)
                    {
                        Vector3 a = scaledVertices[face.A];
                        Vector3 b = scaledVertices[face.B];
                        Vector3 c = scaledVertices[face.C];
                        Vector3 n = Vector3.Cross(Vector3.Subtract(b, a), Vector3.Subtract(c, a));
                        double len = n.Length();
                        if (len > 1e-9)
                        {
                            n = Vector3.Multiply(n, 1.0 / len);
                            if (Vector3.Dot(Vector3.Subtract(a, centroid), n) < 0)
                            {
                                n = Vector3.Multiply(n, -1.0);
                            }
                            double dist = Vector3.Dot(a, n);
                            planes.Add((n, dist));
                        }
                    }

                    // Scan voxels
                    for (int x = 0; x < w; x++)
                    {
                        for (int y = 0; y < h; y++)
                        {
                            for (int z = 0; z < d; z++)
                            {
                                Vector3 P = new Vector3(xStart + x + 0.5, yStart + y + 0.5, zStart + z + 0.5);
                                bool inside = true;
                                foreach (var plane in planes)
                                {
                                    if (Vector3.Dot(P, plane.Normal) > plane.Dist + 1e-3)
                                    {
                                        inside = false;
                                        break;
                                    }
                                }
                                if (inside)
                                {
                                    temp.ActiveVoxels.Add((x, y, z));
                                }
                            }
                        }
                    }

                    // Always add vertex voxels to guarantee no empty templates
                    foreach (var v in scaledVertices)
                    {
                        int vx = Math.Max(0, Math.Min(w - 1, (int)Math.Floor(v.X) - xStart));
                        int vy = Math.Max(0, Math.Min(h - 1, (int)Math.Floor(v.Y) - yStart));
                        int vz = Math.Max(0, Math.Min(d - 1, (int)Math.Floor(v.Z) - zStart));
                        bool exists = false;
                        foreach (var av in temp.ActiveVoxels)
                        {
                            if (av.X == vx && av.Y == vy && av.Z == vz)
                            {
                                exists = true;
                                break;
                            }
                        }
                        if (!exists)
                        {
                            temp.ActiveVoxels.Add((vx, vy, vz));
                        }
                    }

                    // Fallback to centroid if still empty
                    if (temp.ActiveVoxels.Count == 0)
                    {
                        int cx = Math.Max(0, Math.Min(w - 1, (int)Math.Floor(centroid.X) - xStart));
                        int cy = Math.Max(0, Math.Min(h - 1, (int)Math.Floor(centroid.Y) - yStart));
                        int cz = Math.Max(0, Math.Min(d - 1, (int)Math.Floor(centroid.Z) - zStart));
                        temp.ActiveVoxels.Add((cx, cy, cz));
                    }
                }

                // Sort active voxels for deterministic ordering/deduplication
                temp.ActiveVoxels.Sort((a, b) => a.X != b.X ? a.X.CompareTo(b.X) : (a.Y != b.Y ? a.Y.CompareTo(b.Y) : a.Z.CompareTo(b.Z)));
                allTemplates.Add(temp);
            }

            // Deduplicate templates based on active voxels
            var uniqueTemplates = new List<VoxelTemplate>();
            foreach (var temp in allTemplates)
            {
                bool isDup = false;
                foreach (var ut in uniqueTemplates)
                {
                    if (temp.W == ut.W && temp.H == ut.H && temp.D == ut.D && temp.ActiveVoxels.Count == ut.ActiveVoxels.Count)
                    {
                        bool match = true;
                        for (int i = 0; i < temp.ActiveVoxels.Count; i++)
                        {
                            if (temp.ActiveVoxels[i] != ut.ActiveVoxels[i])
                            {
                                match = false;
                                break;
                            }
                        }
                        if (match)
                        {
                            isDup = true;
                            break;
                        }
                    }
                }
                if (!isDup)
                {
                    temp.ComputeOffsets(H_s, D_s);
                    uniqueTemplates.Add(temp);
                }
            }

            return uniqueTemplates;
        }

        public struct CandidatePoint
        {
            public int X, Y, Z;
            public CandidatePoint(int x, int y, int z) { X = x; Y = y; Z = z; }
        }

        public static List<PlacedItem> PackVoxelGrid(int W_s, int H_s, int D_s, List<VoxelTemplate> templates, int orderType, double S)
        {
            var container = new BitGrid3D(W_s, H_s, D_s);
            var placements = new List<PlacedItem>();
            var candidates = new List<CandidatePoint> { new CandidatePoint(0, 0, 0) };
            var candidateSet = new HashSet<string> { "0,0,0" };

            int strideH = H_s * D_s;
            int strideD = D_s;

            while (candidates.Count > 0)
            {
                if (orderType == 0)
                {
                    // Z, Y, X ascending (bottom-up, back-to-front, left-to-right)
                    candidates.Sort((a, b) => a.Z != b.Z ? a.Z.CompareTo(b.Z) : (a.Y != b.Y ? a.Y.CompareTo(b.Y) : a.X.CompareTo(b.X)));
                }
                else if (orderType == 1)
                {
                    // Y, Z, X ascending
                    candidates.Sort((a, b) => a.Y != b.Y ? a.Y.CompareTo(b.Y) : (a.Z != b.Z ? a.Z.CompareTo(b.Z) : a.X.CompareTo(b.X)));
                }
                else if (orderType == 2)
                {
                    // X, Y, Z ascending
                    candidates.Sort((a, b) => a.X != b.X ? a.X.CompareTo(b.X) : (a.Y != b.Y ? a.Y.CompareTo(b.Y) : a.Z.CompareTo(b.Z)));
                }
                else
                {
                    // Z, X, Y ascending
                    candidates.Sort((a, b) => a.Z != b.Z ? a.Z.CompareTo(b.Z) : (a.X != b.X ? a.X.CompareTo(b.X) : a.Y.CompareTo(b.Y)));
                }

                bool placedAny = false;
                for (int i = 0; i < candidates.Count; i++)
                {
                    var c = candidates[i];

                    VoxelTemplate bestTemp = null;
                    foreach (var temp in templates)
                    {
                        if (c.X + temp.W <= W_s && c.Y + temp.H <= H_s && c.Z + temp.D <= D_s)
                        {
                            int t_offset = c.X * strideH + c.Y * strideD + c.Z;
                            bool overlap = false;
                            foreach (int offset in temp.VoxelOffsets)
                            {
                                if (container.IsSetFlat(t_offset + offset))
                                {
                                    overlap = true;
                                    break;
                                }
                            }
                            if (!overlap)
                            {
                                bestTemp = temp;
                                break;
                            }
                        }
                    }

                    if (bestTemp != null)
                    {
                        int t_offset = c.X * strideH + c.Y * strideD + c.Z;
                        foreach (int offset in bestTemp.VoxelOffsets)
                        {
                            container.SetFlat(t_offset + offset);
                        }

                        placements.Add(new PlacedItem
                        {
                            Q = bestTemp.Q,
                            Tx = (c.X - bestTemp.XStart) / S,
                            Ty = (c.Y - bestTemp.YStart) / S,
                            Tz = (c.Z - bestTemp.ZStart) / S
                        });

                        candidates.RemoveAt(i);
                        candidateSet.Remove($"{c.X},{c.Y},{c.Z}");

                        // Filter candidates inside placed bounding box
                        for (int j = candidates.Count - 1; j >= 0; j--)
                        {
                            var oc = candidates[j];
                            if (oc.X >= c.X && oc.X < c.X + bestTemp.W &&
                                oc.Y >= c.Y && oc.Y < c.Y + bestTemp.H &&
                                oc.Z >= c.Z && oc.Z < c.Z + bestTemp.D)
                            {
                                candidates.RemoveAt(j);
                                candidateSet.Remove($"{oc.X},{oc.Y},{oc.Z}");
                            }
                        }

                        // Add new candidates
                        var newPoints = new[] {
                            new CandidatePoint(c.X + bestTemp.W, c.Y, c.Z),
                            new CandidatePoint(c.X, c.Y + bestTemp.H, c.Z),
                            new CandidatePoint(c.X, c.Y, c.Z + bestTemp.D)
                        };

                        foreach (var np in newPoints)
                        {
                            if (np.X < W_s && np.Y < H_s && np.Z < D_s)
                            {
                                string key = $"{np.X},{np.Y},{np.Z}";
                                if (!candidateSet.Contains(key))
                                {
                                    candidates.Add(np);
                                    candidateSet.Add(key);
                                }
                            }
                        }

                        placedAny = true;
                        break;
                    }
                }

                if (!placedAny)
                {
                    break;
                }
            }

            return placements;
        }

        public static List<PlacedItem> RunBlockPacker(double W, double H, double D, Vector3[] vertices, Face[] faces)
        {
            var rotations = GenerateOrthotropicRotations();
            var templates = new List<AABBTemplate>();
            foreach (var q in rotations)
            {
                double minX = double.MaxValue, maxX = double.MinValue;
                double minY = double.MaxValue, maxY = double.MinValue;
                double minZ = double.MaxValue, maxZ = double.MinValue;
                foreach (var v in vertices)
                {
                    var rv = q.Rotate(v);
                    if (rv.X < minX) minX = rv.X;
                    if (rv.X > maxX) maxX = rv.X;
                    if (rv.Y < minY) minY = rv.Y;
                    if (rv.Y > maxY) maxY = rv.Y;
                    if (rv.Z < minZ) minZ = rv.Z;
                    if (rv.Z > maxZ) maxZ = rv.Z;
                }
                templates.Add(new AABBTemplate
                {
                    Q = q,
                    MinX = minX,
                    MinY = minY,
                    MinZ = minZ,
                    W = maxX - minX,
                    H = maxY - minY,
                    D = maxZ - minZ
                });
            }

            List<PlacedItem> bestResult = new List<PlacedItem>();

            for (int perm = 0; perm < 6; perm++)
            {
                var currentResult = new List<PlacedItem>();
                PackBlockRecursive(0, 0, 0, W, H, D, templates, perm, currentResult, 0);
                if (currentResult.Count > bestResult.Count)
                {
                    bestResult = currentResult;
                }
            }

            return bestResult;
        }

        public static void PackBlockRecursive(double x0, double y0, double z0, double w_box, double h_box, double d_box, List<AABBTemplate> templates, int perm, List<PlacedItem> results, int depth)
        {
            if (depth > 25) return;

            AABBTemplate bestTemp = null;
            int bestCount = 0;
            int bestNx = 0, bestNy = 0, bestNz = 0;

            foreach (var temp in templates)
            {
                if (temp.W <= w_box + 1e-9 && temp.H <= h_box + 1e-9 && temp.D <= d_box + 1e-9)
                {
                    int nx = (int)Math.Floor((w_box + 1e-9) / temp.W);
                    int ny = (int)Math.Floor((h_box + 1e-9) / temp.H);
                    int nz = (int)Math.Floor((d_box + 1e-9) / temp.D);
                    int count = nx * ny * nz;
                    if (count > bestCount)
                    {
                        bestCount = count;
                        bestTemp = temp;
                        bestNx = nx;
                        bestNy = ny;
                        bestNz = nz;
                    }
                }
            }

            if (bestTemp == null || bestCount == 0) return;

            double fit_x = bestNx * bestTemp.W;
            double fit_y = bestNy * bestTemp.H;
            double fit_z = bestNz * bestTemp.D;

            for (int ix = 0; ix < bestNx; ix++)
            {
                for (int iy = 0; iy < bestNy; iy++)
                {
                    for (int iz = 0; iz < bestNz; iz++)
                    {
                        double Tx = x0 + ix * bestTemp.W;
                        double Ty = y0 + iy * bestTemp.H;
                        double Tz = z0 + iz * bestTemp.D;
                        results.Add(new PlacedItem
                        {
                            Q = bestTemp.Q,
                            Tx = Tx - bestTemp.MinX,
                            Ty = Ty - bestTemp.MinY,
                            Tz = Tz - bestTemp.MinZ
                        });
                    }
                }
            }

            double rem_x = w_box - fit_x;
            double rem_y = h_box - fit_y;
            double rem_z = d_box - fit_z;

            double x1 = 0, y1 = 0, z1 = 0, w1 = 0, h1 = 0, d1 = 0;
            double x2 = 0, y2 = 0, z2 = 0, w2 = 0, h2 = 0, d2 = 0;
            double x3 = 0, y3 = 0, z3 = 0, w3 = 0, h3 = 0, d3 = 0;

            if (perm == 0) // XYZ
            {
                x1 = x0 + fit_x; y1 = y0; z1 = z0; w1 = rem_x; h1 = h_box; d1 = d_box;
                x2 = x0; y2 = y0 + fit_y; z2 = z0; w2 = fit_x; h2 = rem_y; d2 = d_box;
                x3 = x0; y3 = y0; z3 = z0 + fit_z; w3 = fit_x; h3 = fit_y; d3 = rem_z;
            }
            else if (perm == 1) // XZY
            {
                x1 = x0 + fit_x; y1 = y0; z1 = z0; w1 = rem_x; h1 = h_box; d1 = d_box;
                x2 = x0; y2 = y0; z2 = z0 + fit_z; w2 = fit_x; h2 = h_box; d2 = rem_z;
                x3 = x0; y3 = y0 + fit_y; z3 = z0; w3 = fit_x; h3 = rem_y; d3 = fit_z;
            }
            else if (perm == 2) // YXZ
            {
                x1 = x0; y1 = y0 + fit_y; z1 = z0; w1 = w_box; h1 = rem_y; d1 = d_box;
                x2 = x0 + fit_x; y2 = y0; z2 = z0; w2 = rem_x; h2 = fit_y; d2 = d_box;
                x3 = x0; y3 = y0; z3 = z0 + fit_z; w3 = fit_x; h3 = fit_y; d3 = rem_z;
            }
            else if (perm == 3) // YZX
            {
                x1 = x0; y1 = y0 + fit_y; z1 = z0; w1 = w_box; h1 = rem_y; d1 = d_box;
                x2 = x0; y2 = y0; z2 = z0 + fit_z; w2 = w_box; h2 = fit_y; d2 = rem_z;
                x3 = x0 + fit_x; y3 = y0; z3 = z0; w3 = rem_x; h3 = fit_y; d3 = fit_z;
            }
            else if (perm == 4) // ZXY
            {
                x1 = x0; y1 = y0; z1 = z0 + fit_z; w1 = w_box; h1 = h_box; d1 = rem_z;
                x2 = x0 + fit_x; y2 = y0; z2 = z0; w2 = rem_x; h2 = h_box; d2 = fit_z;
                x3 = x0; y3 = y0 + fit_y; z3 = z0; w3 = fit_x; h3 = rem_y; d3 = fit_z;
            }
            else // ZYX
            {
                x1 = x0; y1 = y0; z1 = z0 + fit_z; w1 = w_box; h1 = h_box; d1 = rem_z;
                x2 = x0; y2 = y0 + fit_y; z2 = z0; w2 = w_box; h2 = rem_y; d2 = fit_z;
                x3 = x0 + fit_x; y3 = y0; z3 = z0; w3 = rem_x; h3 = fit_y; d3 = fit_z;
            }

            if (w1 > 1e-5 && h1 > 1e-5 && d1 > 1e-5) PackBlockRecursive(x1, y1, z1, w1, h1, d1, templates, perm, results, depth + 1);
            if (w2 > 1e-5 && h2 > 1e-5 && d2 > 1e-5) PackBlockRecursive(x2, y2, z2, w2, h2, d2, templates, perm, results, depth + 1);
            if (w3 > 1e-5 && h3 > 1e-5 && d3 > 1e-5) PackBlockRecursive(x3, y3, z3, w3, h3, d3, templates, perm, results, depth + 1);
        }

        public static string ReadNonEmptyLine()
        {
            string line;
            while ((line = Console.ReadLine()) != null)
            {
                line = line.Trim();
                if (line.Length > 0) return line;
            }
            return null;
        }

        public static void Main(string[] args)
        {
            var firstLine = ReadNonEmptyLine();
            if (firstLine == null) return;
            var parts = firstLine.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
            int V = int.Parse(parts[0]);
            int F = int.Parse(parts[1]);

            if (V < 3 || F < 1)
            {
                Console.WriteLine(0);
                return;
            }

            var vertices = new Vector3[V];
            for (int i = 0; i < V; i++)
            {
                var line = ReadNonEmptyLine();
                var vParts = line.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
                vertices[i] = new Vector3(
                    double.Parse(vParts[0], CultureInfo.InvariantCulture),
                    double.Parse(vParts[1], CultureInfo.InvariantCulture),
                    double.Parse(vParts[2], CultureInfo.InvariantCulture)
                );
            }

            var faces = new Face[F];
            for (int i = 0; i < F; i++)
            {
                var line = ReadNonEmptyLine();
                var fParts = line.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
                faces[i] = new Face(
                    int.Parse(fParts[0]),
                    int.Parse(fParts[1]),
                    int.Parse(fParts[2])
                );
            }

            var lastLine = ReadNonEmptyLine();
            if (lastLine == null) return;
            var cParts = lastLine.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
            double W = double.Parse(cParts[0], CultureInfo.InvariantCulture);
            double H = double.Parse(cParts[1], CultureInfo.InvariantCulture);
            double D = double.Parse(cParts[2], CultureInfo.InvariantCulture);

            // Shift faces if 1-indexed
            int minIndex = int.MaxValue;
            int maxIndex = int.MinValue;
            for (int i = 0; i < F; i++)
            {
                if (faces[i].A < minIndex) minIndex = faces[i].A;
                if (faces[i].B < minIndex) minIndex = faces[i].B;
                if (faces[i].C < minIndex) minIndex = faces[i].C;
                if (faces[i].A > maxIndex) maxIndex = faces[i].A;
                if (faces[i].B > maxIndex) maxIndex = faces[i].B;
                if (faces[i].C > maxIndex) maxIndex = faces[i].C;
            }
            if (minIndex == 1 && maxIndex == V)
            {
                for (int i = 0; i < F; i++)
                {
                    faces[i].A--;
                    faces[i].B--;
                    faces[i].C--;
                }
            }

            List<PlacedItem> bestResult = null;

            // Adaptive scale selection
            double volume = W * H * D;
            if (volume <= 300000)
            {
                // Run voxel packing
                double maxDim = Math.Max(W, Math.Max(H, D));
                double S = Math.Max(1.0, Math.Min(20.0, 100.0 / maxDim));

                int W_s = (int)Math.Floor(W * S);
                int H_s = (int)Math.Floor(H * S);
                int D_s = (int)Math.Floor(D * S);

                var voxelTemplates = GenerateVoxelTemplates(vertices, faces, S, W_s, H_s, D_s);

                int bestCount = -1;
                for (int order = 0; order < 4; order++)
                {
                    var result = PackVoxelGrid(W_s, H_s, D_s, voxelTemplates, order, S);
                    if (result.Count > bestCount)
                    {
                        bestCount = result.Count;
                        bestResult = result;
                    }
                }
            }

            // Run recursive AABB block packing
            var blockResult = RunBlockPacker(W, H, D, vertices, faces);

            if (bestResult == null || blockResult.Count > bestResult.Count)
            {
                bestResult = blockResult;
            }

            // Write output using buffered writer for extreme scales
            using (var writer = new StreamWriter(Console.OpenStandardOutput(), System.Text.Encoding.UTF8, 65536))
            {
                writer.WriteLine(bestResult.Count);
                foreach (var item in bestResult)
                {
                    writer.WriteLine(string.Format(CultureInfo.InvariantCulture,
                        "{0:F6} {1:F6} {2:F6} {3:F6} {4:F6} {5:F6} {6:F6}",
                        item.Tx, item.Ty, item.Tz,
                        item.Q.W, item.Q.X, item.Q.Y, item.Q.Z));
                }
            }
        }
    }
