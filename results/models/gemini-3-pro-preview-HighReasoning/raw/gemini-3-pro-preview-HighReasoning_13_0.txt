#include <vector>
#include <string>
#include <algorithm>
#include <iostream>
#include <chrono>
#include <random>
#include <thread>
#include <atomic>
#include <cstdint>
#include <limits>
#include <string_view>

struct HashEntry {
    uint64_t h;
    int idx;
};

struct TableEntry {
    uint64_t h;
    int candidate_idx;
};

class HashTable {
public:
    std::vector<TableEntry> table;
    uint64_t mask;
    
    void build(const std::vector<HashEntry>& candidates) {
        size_t size = candidates.size();
        size_t cap = 1;
        while (cap < size * 2) cap *= 2;
        table.assign(cap, TableEntry{0, -1});
        mask = cap - 1;
        for (int i = 0; i < (int)size; ++i) {
            uint64_t h = candidates[i].h;
            if (h == 0) h = 1;
            uint64_t idx = h & mask;
            while (table[idx].candidate_idx != -1) {
                idx = (idx + 1) & mask;
            }
            table[idx] = TableEntry{h, i};
        }
    }
    
    int find(uint64_t h) const {
        if (h == 0) h = 1;
        uint64_t idx = h & mask;
        while (table[idx].candidate_idx != -1) {
            if (table[idx].h == h) {
                return table[idx].candidate_idx;
            }
            idx = (idx + 1) & mask;
        }
        return -1;
    }
};

void process_string(const std::string& S, int K, uint64_t B1, uint64_t B1_K,
                    const HashTable& ht, const std::vector<HashEntry>& candidates,
                    const std::string& S0, std::vector<uint8_t>& matched) {
    int L = S.length();
    if (K > L) return;
    uint64_t h1 = 0;
    for (int i = 0; i < K; ++i) {
        h1 = h1 * B1 + static_cast<uint8_t>(S[i]);
    }
    int idx = ht.find(h1);
    if (idx != -1) {
        if (std::string_view(S0).substr(candidates[idx].idx, K) == std::string_view(S).substr(0, K)) {
            matched[idx] = 1;
        }
    }
    for (int i = K; i < L; ++i) {
        h1 = h1 * B1 + static_cast<uint8_t>(S[i]) - static_cast<uint8_t>(S[i - K]) * B1_K;
        idx = ht.find(h1);
        if (idx != -1) {
            if (std::string_view(S0).substr(candidates[idx].idx, K) == std::string_view(S).substr(i - K + 1, K)) {
                matched[idx] = 1;
            }
        }
    }
}

int check(int K, const std::vector<std::string>& strings, uint64_t B1,
          const std::vector<uint64_t>& B1_powers, int L_max) {
    const std::string& S0 = strings[0];
    int L0 = S0.length();
    if (K > L0) return -1;
    
    std::vector<HashEntry> candidates;
    candidates.reserve(L0 - K + 1);
    
    uint64_t B1_K = B1_powers[K];
    uint64_t h1 = 0;
    for (int i = 0; i < K; ++i) {
        h1 = h1 * B1 + static_cast<uint8_t>(S0[i]);
    }
    candidates.push_back({h1, 0});
    for (int i = K; i < L0; ++i) {
        h1 = h1 * B1 + static_cast<uint8_t>(S0[i]) - static_cast<uint8_t>(S0[i - K]) * B1_K;
        candidates.push_back({h1, i - K + 1});
    }
    
    std::sort(candidates.begin(), candidates.end(), [](const HashEntry& a, const HashEntry& b) {
        return a.h < b.h;
    });
    
    std::vector<HashEntry> unique_candidates;
    unique_candidates.reserve(candidates.size());
    for (const auto& entry : candidates) {
        if (unique_candidates.empty() || unique_candidates.back().h != entry.h) {
            unique_candidates.push_back(entry);
        }
    }
    candidates = std::move(unique_candidates);
    
    if (candidates.empty()) return -1;
    
    HashTable ht;
    ht.build(candidates);
    
    std::vector<uint8_t> matched(candidates.size(), 0);
    process_string(strings[1], K, B1, B1_K, ht, candidates, S0, matched);
    
    std::vector<HashEntry> next_candidates;
    next_candidates.reserve(candidates.size());
    for (size_t i = 0; i < candidates.size(); ++i) {
        if (matched[i]) {
            next_candidates.push_back(candidates[i]);
        }
    }
    candidates = std::move(next_candidates);
    if (candidates.empty()) return -1;
    
    if (strings.size() > 2) {
        ht.build(candidates);
        matched.assign(candidates.size(), 0);
        process_string(strings[2], K, B1, B1_K, ht, candidates, S0, matched);
        
        next_candidates.clear();
        for (size_t i = 0; i < candidates.size(); ++i) {
            if (matched[i]) {
                next_candidates.push_back(candidates[i]);
            }
        }
        candidates = std::move(next_candidates);
        if (candidates.empty()) return -1;
    }
    
    int N = strings.size();
    if (N > 3) {
        bool use_parallel = (candidates.size() > 5000) && (L_max > 10000);
        if (use_parallel) {
            ht.build(candidates);
            int num_candidates = candidates.size();
            int num_threads = std::min((int)std::thread::hardware_concurrency(), N - 3);
            if (num_threads < 1) num_threads = 1;
            
            std::vector<std::vector<uint8_t>> present(N - 3, std::vector<uint8_t>(num_candidates, 0));
            std::vector<std::thread> threads;
            threads.reserve(num_threads);
            
            std::atomic<int> next_str_idx(3);
            auto worker = [&](int thread_id) {
                while (true) {
                    int s_idx = next_str_idx.fetch_add(1);
                    if (s_idx >= N) break;
                    process_string(strings[s_idx], K, B1, B1_K, ht, candidates, S0, present[s_idx - 3]);
                }
            };
            
            for (int t = 0; t < num_threads; ++t) {
                threads.emplace_back(worker, t);
            }
            for (auto& t : threads) {
                t.join();
            }
            
            for (int j = 0; j < num_candidates; ++j) {
                bool all_present = true;
                for (int i = 0; i < N - 3; ++i) {
                    if (!present[i][j]) {
                        all_present = false;
                        break;
                    }
                }
                if (all_present) {
                    return candidates[j].idx;
                }
            }
            return -1;
        } else {
            for (int s_idx = 3; s_idx < N; ++s_idx) {
                ht.build(candidates);
                matched.assign(candidates.size(), 0);
                process_string(strings[s_idx], K, B1, B1_K, ht, candidates, S0, matched);
                
                next_candidates.clear();
                for (size_t i = 0; i < candidates.size(); ++i) {
                    if (matched[i]) {
                        next_candidates.push_back(candidates[i]);
                    }
                }
                candidates = std::move(next_candidates);
                if (candidates.empty()) return -1;
            }
            return candidates[0].idx;
        }
    } else {
        return candidates[0].idx;
    }
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    int N;
    if (!(std::cin >> N)) return 0;
    std::cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n');
    
    std::vector<std::string> strings;
    strings.reserve(N);
    for (int i = 0; i < N; ++i) {
        std::string S;
        if (std::getline(std::cin, S)) {
            strings.push_back(std::move(S));
        }
    }
    
    if (strings.empty()) {
        std::cout << "\n";
        return 0;
    }
    
    std::sort(strings.begin(), strings.end(), [](const std::string& a, const std::string& b) {
        return a.length() < b.length();
    });
    
    if (strings[0].empty()) {
        std::cout << "\n";
        return 0;
    }
    
    if (strings.size() == 1) {
        std::cout << strings[0] << "\n";
        return 0;
    }
    
    std::random_device rd;
    std::mt19937_64 gen(rd());
    std::uniform_int_distribution<uint64_t> dis(1ULL << 30, (1ULL << 62) - 1);
    uint64_t B1 = dis(gen) | 1;
    
    int L_max = strings.back().length();
    std::vector<uint64_t> B1_powers(L_max + 1);
    B1_powers[0] = 1;
    for (int i = 1; i <= L_max; ++i) {
        B1_powers[i] = B1_powers[i - 1] * B1;
    }
    
    int low = 1, high = strings[0].length();
    int ans_len = 0;
    int ans_idx = -1;
    
    while (low <= high) {
        int mid = low + (high - low) / 2;
        int idx = check(mid, strings, B1, B1_powers, L_max);
        if (idx != -1) {
            ans_len = mid;
            ans_idx = idx;
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }
    
    if (ans_len > 0 && ans_idx != -1) {
        std::cout << strings[0].substr(ans_idx, ans_len) << "\n";
    } else {
        std::cout << "\n";
    }
    
    return 0;
}
