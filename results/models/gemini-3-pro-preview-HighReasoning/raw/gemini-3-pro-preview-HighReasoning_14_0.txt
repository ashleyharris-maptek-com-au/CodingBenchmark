using System;
using System.IO;
using System.Collections.Generic;

public class MinesweeperSolver
{
    private struct Constraint
    {
        public int CellIdx;
        public int Target;
        public int VarCount;
        public int V0, V1, V2, V3, V4, V5, V6, V7;
        
        public int GetVar(int i)
        {
            if (i == 0) return V0;
            if (i == 1) return V1;
            if (i == 2) return V2;
            if (i == 3) return V3;
            if (i == 4) return V4;
            if (i == 5) return V5;
            if (i == 6) return V6;
            return V7;
        }
        
        public void SetVar(int i, int val)
        {
            if (i == 0) V0 = val;
            else if (i == 1) V1 = val;
            else if (i == 2) V2 = val;
            else if (i == 3) V3 = val;
            else if (i == 4) V4 = val;
            else if (i == 5) V5 = val;
            else if (i == 6) V6 = val;
            else V7 = val;
        }
    }

    private struct VarConstraintList
    {
        public int Count;
        public int C0, C1, C2, C3, C4, C5, C6, C7;
        
        public void Add(int cIdx)
        {
            if (Count == 0) C0 = cIdx;
            else if (Count == 1) C1 = cIdx;
            else if (Count == 2) C2 = cIdx;
            else if (Count == 3) C3 = cIdx;
            else if (Count == 4) C4 = cIdx;
            else if (Count == 5) C5 = cIdx;
            else if (Count == 6) C6 = cIdx;
            else C7 = cIdx;
            Count++;
        }
        
        public int Get(int i)
        {
            if (i == 0) return C0;
            if (i == 1) return C1;
            if (i == 2) return C2;
            if (i == 3) return C3;
            if (i == 4) return C4;
            if (i == 5) return C5;
            if (i == 6) return C6;
            return C7;
        }
    }

    // Solver state
    private static int H;
    private static int W;
    private static byte[] board;
    private static byte[] remainingMines;
    private static byte[] unknownNeighborCount;
    private static int[] queue;
    private static int head;
    private static int tail;
    private static bool[] inQueue;
    private static bool[] visited;
    private static int[] bfsQueue;

    // Preallocated data structures for Tier 2
    private static int[] ua = new int[8];
    private static int[] ub = new int[8];
    private static int[] candidates = new int[64];

    // Preallocated data structures for Tier 3
    private static List<int> compRevealed = new List<int>(100);
    private static List<int> compUnknown = new List<int>(100);
    private static Dictionary<int, int> varMap = new Dictionary<int, int>(100);
    private static Constraint[] constraints = new Constraint[1000];
    private static VarConstraintList[] varConstraints = new VarConstraintList[100];
    private static int[] assignment = new int[100];
    private static int[] assignedMines = new int[1000];
    private static int[] unassignedNeighbors = new int[1000];
    private static bool[] canBeSafe = new bool[100];
    private static bool[] canBeMine = new bool[100];

    // Output formatting buffer
    private static char[] intBuf = new char[11];

    public static void Main(string[] args)
    {
        // Set standard input/output encoding and buffer sizes
        Console.OutputEncoding = System.Text.Encoding.ASCII;
        
        string firstLine = Console.ReadLine();
        if (string.IsNullOrEmpty(firstLine)) return;
        string[] parts = firstLine.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        if (parts.Length < 2) return;
        H = int.Parse(parts[0]);
        W = int.Parse(parts[1]);

        int totalCells = H * W;
        board = new byte[totalCells];
        remainingMines = new byte[totalCells];
        unknownNeighborCount = new byte[totalCells];
        queue = new int[totalCells];
        inQueue = new bool[totalCells];
        visited = new bool[totalCells];
        bfsQueue = new int[totalCells];

        // Initialize varConstraints array
        for (int i = 0; i < varConstraints.Length; i++)
        {
            varConstraints[i] = new VarConstraintList();
        }

        for (int r = 0; r < H; r++)
        {
            string line = Console.ReadLine();
            if (line == null) break;
            int len = Math.Min(W, line.Length);
            for (int c = 0; c < len; c++)
            {
                char ch = line[c];
                byte val;
                if (ch == ' ') val = 9;
                else if (ch == '*') val = 10;
                else val = (byte)(ch - '0');
                board[r * W + c] = val;
            }
        }

        // Initialize state and queue
        InitializeSolver();

        // Main solve loop
        Solve();

        // Output results
        OutputResults();
    }

    private static void InitializeSolver()
    {
        head = 0;
        tail = 0;
        for (int r = 0; r < H; r++)
        {
            int rowOffset = r * W;
            for (int c = 0; c < W; c++)
            {
                int idx = rowOffset + c;
                byte val = board[idx];
                if (val <= 8) // revealed cell
                {
                    int unknownCount = 0;
                    int flaggedCount = 0;

                    int rMin = r > 0 ? r - 1 : 0;
                    int rMax = r < H - 1 ? r + 1 : H - 1;
                    int cMin = c > 0 ? c - 1 : 0;
                    int cMax = c < W - 1 ? c + 1 : W - 1;
                    for (int nr = rMin; nr <= rMax; nr++)
                    {
                        int nRowOffset = nr * W;
                        for (int nc = cMin; nc <= cMax; nc++)
                        {
                            if (nr == r && nc == c) continue;
                            int nIdx = nRowOffset + nc;
                            byte nVal = board[nIdx];
                            if (nVal == 9) unknownCount++;
                            else if (nVal == 10) flaggedCount++;
                        }
                    }

                    remainingMines[idx] = (byte)Math.Max(0, val - flaggedCount);
                    unknownNeighborCount[idx] = (byte)unknownCount;

                    if (unknownCount > 0 && (remainingMines[idx] == 0 || unknownNeighborCount[idx] == remainingMines[idx]))
                    {
                        queue[tail++] = idx;
                        inQueue[idx] = true;
                    }
                }
            }
        }
    }

    private static void Solve()
    {
        while (true)
        {
            // Tier 1: Single-cell propagation
            RunTier1();

            // Tier 2: Double-cell overlap
            bool p2 = RunTier2();
            if (p2) continue; // If Tier 2 made progress, run Tier 1/2 again

            // Tier 3: Component-based backtracking CSP
            if (ShouldRunTier3())
            {
                bool p3 = RunTier3();
                if (p3) continue; // If Tier 3 made progress, run Tier 1/2/3 again
            }

            break; // No further progress can be made
        }
    }

    private static void RunTier1()
    {
        while (head < tail)
        {
            int idx = queue[head++];
            inQueue[idx] = false;

            int r = idx / W;
            int c = idx % W;

            int rem = remainingMines[idx];
            int unk = unknownNeighborCount[idx];

            if (unk == 0) continue;

            if (rem == 0)
            {
                // All unknown neighbors are safe
                int rMin = r > 0 ? r - 1 : 0;
                int rMax = r < H - 1 ? r + 1 : H - 1;
                int cMin = c > 0 ? c - 1 : 0;
                int cMax = c < W - 1 ? c + 1 : W - 1;
                for (int nr = rMin; nr <= rMax; nr++)
                {
                    int rowOffset = nr * W;
                    for (int nc = cMin; nc <= cMax; nc++)
                    {
                        if (nr == r && nc == c) continue;
                        int nIdx = rowOffset + nc;
                        if (board[nIdx] == 9)
                        {
                            MarkSafe(nIdx);
                        }
                    }
                }
                unknownNeighborCount[idx] = 0;
            }
            else if (unk == rem)
            {
                // All unknown neighbors are mines
                int rMin = r > 0 ? r - 1 : 0;
                int rMax = r < H - 1 ? r + 1 : H - 1;
                int cMin = c > 0 ? c - 1 : 0;
                int cMax = c < W - 1 ? c + 1 : W - 1;
                for (int nr = rMin; nr <= rMax; nr++)
                {
                    int rowOffset = nr * W;
                    for (int nc = cMin; nc <= cMax; nc++)
                    {
                        if (nr == r && nc == c) continue;
                        int nIdx = rowOffset + nc;
                        if (board[nIdx] == 9)
                        {
                            MarkMine(nIdx);
                        }
                    }
                }
                unknownNeighborCount[idx] = 0;
            }
        }
    }

    private static void MarkSafe(int nIdx)
    {
        board[nIdx] = 11; // proven safe
        int nr = nIdx / W;
        int nc = nIdx % W;
        int rMin = nr > 0 ? nr - 1 : 0;
        int rMax = nr < H - 1 ? nr + 1 : H - 1;
        int cMin = nc > 0 ? nc - 1 : 0;
        int cMax = nc < W - 1 ? nc + 1 : W - 1;
        for (int nnr = rMin; nnr <= rMax; nnr++)
        {
            int rowOffset = nnr * W;
            for (int nnc = cMin; nnc <= cMax; nnc++)
            {
                if (nnr == nr && nnc == nc) continue;
                int rnIdx = rowOffset + nnc;
                if (board[rnIdx] <= 8)
                {
                    unknownNeighborCount[rnIdx]--;
                    if (unknownNeighborCount[rnIdx] > 0 && !inQueue[rnIdx])
                    {
                        if (remainingMines[rnIdx] == 0 || unknownNeighborCount[rnIdx] == remainingMines[rnIdx])
                        {
                            queue[tail++] = rnIdx;
                            inQueue[rnIdx] = true;
                        }
                    }
                }
            }
        }
    }

    private static void MarkMine(int nIdx)
    {
        board[nIdx] = 12; // proven mine
        int nr = nIdx / W;
        int nc = nIdx % W;
        int rMin = nr > 0 ? nr - 1 : 0;
        int rMax = nr < H - 1 ? nr + 1 : H - 1;
        int cMin = nc > 0 ? nc - 1 : 0;
        int cMax = nc < W - 1 ? nc + 1 : W - 1;
        for (int nnr = rMin; nnr <= rMax; nnr++)
        {
            int rowOffset = nnr * W;
            for (int nnc = cMin; nnc <= cMax; nnc++)
            {
                if (nnr == nr && nnc == nc) continue;
                int rnIdx = rowOffset + nnc;
                if (board[rnIdx] <= 8)
                {
                    unknownNeighborCount[rnIdx]--;
                    remainingMines[rnIdx]--;
                    if (unknownNeighborCount[rnIdx] > 0 && !inQueue[rnIdx])
                    {
                        if (remainingMines[rnIdx] == 0 || unknownNeighborCount[rnIdx] == remainingMines[rnIdx])
                        {
                            queue[tail++] = rnIdx;
                            inQueue[rnIdx] = true;
                        }
                    }
                }
            }
        }
    }

    private static bool RunTier2()
    {
        bool globalProgress = false;

        for (int idxA = 0; idxA < H * W; idxA++)
        {
            if (board[idxA] > 8 || unknownNeighborCount[idxA] == 0) continue;

            // Get unknown neighbors of A
            int rA = idxA / W;
            int cA = idxA % W;
            int uaCount = 0;
            int rMinA = rA > 0 ? rA - 1 : 0;
            int rMaxA = rA < H - 1 ? rA + 1 : H - 1;
            int cMinA = cA > 0 ? cA - 1 : 0;
            int cMaxA = cA < W - 1 ? cA + 1 : W - 1;
            for (int nr = rMinA; nr <= rMaxA; nr++)
            {
                int rowOffset = nr * W;
                for (int nc = cMinA; nc <= cMaxA; nc++)
                {
                    if (nr == rA && nc == cA) continue;
                    int nIdx = rowOffset + nc;
                    if (board[nIdx] == 9)
                    {
                        ua[uaCount++] = nIdx;
                    }
                }
            }
            if (uaCount == 0) continue;

            // Find candidates for B
            int candCount = 0;
            for (int i = 0; i < uaCount; i++)
            {
                int uIdx = ua[i];
                int ru = uIdx / W;
                int cu = uIdx % W;
                int rMinU = ru > 0 ? ru - 1 : 0;
                int rMaxU = ru < H - 1 ? ru + 1 : H - 1;
                int cMinU = cu > 0 ? cu - 1 : 0;
                int cMaxU = cu < W - 1 ? cu + 1 : W - 1;
                for (int nr = rMinU; nr <= rMaxU; nr++)
                {
                    int rowOffset = nr * W;
                    for (int nc = cMinU; nc <= cMaxU; nc++)
                    {
                        if (nr == ru && nc == cu) continue;
                        int idxB = rowOffset + nc;
                        if (idxB > idxA && board[idxB] <= 8 && unknownNeighborCount[idxB] > 0)
                        {
                            bool found = false;
                            for (int k = 0; k < candCount; k++)
                            {
                                if (candidates[k] == idxB)
                                {
                                    found = true;
                                    break;
                                }
                            }
                            if (!found)
                            {
                                candidates[candCount++] = idxB;
                            }
                        }
                    }
                }
            }

            // For each candidate B, compare A and B
            for (int candIdx = 0; candIdx < candCount; candIdx++)
            {
                int idxB = candidates[candIdx];
                int rB = idxB / W;
                int cB = idxB % W;
                int ubCount = 0;
                int rMinB = rB > 0 ? rB - 1 : 0;
                int rMaxB = rB < H - 1 ? rB + 1 : H - 1;
                int cMinB = cB > 0 ? cB - 1 : 0;
                int cMaxB = cB < W - 1 ? cB + 1 : W - 1;
                for (int nr = rMinB; nr <= rMaxB; nr++)
                {
                    int rowOffset = nr * W;
                    for (int nc = cMinB; nc <= cMaxB; nc++)
                    {
                        if (nr == rB && nc == cB) continue;
                        int nIdx = rowOffset + nc;
                        if (board[nIdx] == 9)
                        {
                            ub[ubCount++] = nIdx;
                        }
                    }
                }
                if (ubCount == 0) continue;

                // Compute overlap
                int intersectionCount = 0;
                int onlyACount = 0;
                for (int i = 0; i < uaCount; i++)
                {
                    int uVal = ua[i];
                    bool inB = false;
                    for (int j = 0; j < ubCount; j++)
                    {
                        if (ub[j] == uVal)
                        {
                            inB = true;
                            break;
                        }
                    }
                    if (inB) intersectionCount++;
                    else onlyACount++;
                }
                int onlyBCount = ubCount - intersectionCount;

                int R_A = remainingMines[idxA];
                int R_B = remainingMines[idxB];

                int L = Math.Max(0, Math.Max(R_A - onlyACount, R_B - onlyBCount));
                int U = Math.Min(intersectionCount, Math.Min(R_A, R_B));

                bool localProgress = false;

                // Check OnlyA deductions
                if (onlyACount > 0)
                {
                    if (R_A - U == onlyACount)
                    {
                        for (int i = 0; i < uaCount; i++)
                        {
                            int uVal = ua[i];
                            bool inB = false;
                            for (int j = 0; j < ubCount; j++)
                            {
                                if (ub[j] == uVal) { inB = true; break; }
                            }
                            if (!inB && board[uVal] == 9)
                            {
                                MarkMine(uVal);
                                localProgress = true;
                            }
                        }
                    }
                    else if (R_A - L == 0)
                    {
                        for (int i = 0; i < uaCount; i++)
                        {
                            int uVal = ua[i];
                            bool inB = false;
                            for (int j = 0; j < ubCount; j++)
                            {
                                if (ub[j] == uVal) { inB = true; break; }
                            }
                            if (!inB && board[uVal] == 9)
                            {
                                MarkSafe(uVal);
                                localProgress = true;
                            }
                        }
                    }
                }

                // Check OnlyB deductions
                if (onlyBCount > 0)
                {
                    if (R_B - U == onlyBCount)
                    {
                        for (int j = 0; j < ubCount; j++)
                        {
                            int uVal = ub[j];
                            bool inA = false;
                            for (int i = 0; i < uaCount; i++)
                            {
                                if (ua[i] == uVal) { inA = true; break; }
                            }
                            if (!inA && board[uVal] == 9)
                            {
                                MarkMine(uVal);
                                localProgress = true;
                            }
                        }
                    }
                    else if (R_B - L == 0)
                    {
                        for (int j = 0; j < ubCount; j++)
                        {
                            int uVal = ub[j];
                            bool inA = false;
                            for (int i = 0; i < uaCount; i++)
                            {
                                if (ua[i] == uVal) { inA = true; break; }
                            }
                            if (!inA && board[uVal] == 9)
                            {
                                MarkSafe(uVal);
                                localProgress = true;
                            }
                        }
                    }
                }

                if (localProgress)
                {
                    globalProgress = true;
                }
            }
        }

        return globalProgress;
    }

    private static bool ShouldRunTier3()
    {
        if (H * W <= 1000000) return true;

        int activeCount = 0;
        for (int i = 0; i < H * W; i++)
        {
            if (board[i] <= 8 && unknownNeighborCount[i] > 0)
            {
                activeCount++;
                if (activeCount >= 5000) return false;
            }
        }
        return true;
    }

    private static int stepCount;
    private static int validConfigCount;
    private static int backtrackN;
    private static int backtrackM;

    private static bool Backtrack(int varIdx)
    {
        stepCount++;
        if (stepCount > 100000) return false;

        if (varIdx == backtrackN)
        {
            validConfigCount++;
            for (int i = 0; i < backtrackN; i++)
            {
                if (assignment[i] == 0) canBeSafe[i] = true;
                else canBeMine[i] = true;
            }
            return true;
        }

        for (int val = 0; val <= 1; val++)
        {
            assignment[varIdx] = val;
            bool ok = true;
            int numAffected = varConstraints[varIdx].Count;
            for (int i = 0; i < numAffected; i++)
            {
                int cIdx = varConstraints[varIdx].Get(i);
                assignedMines[cIdx] += val;
                unassignedNeighbors[cIdx]--;

                if (assignedMines[cIdx] > constraints[cIdx].Target ||
                    assignedMines[cIdx] + unassignedNeighbors[cIdx] < constraints[cIdx].Target)
                {
                    ok = false;
                }
            }

            if (ok)
            {
                if (!Backtrack(varIdx + 1)) return false;
            }

            for (int i = 0; i < numAffected; i++)
            {
                int cIdx = varConstraints[varIdx].Get(i);
                assignedMines[cIdx] -= val;
                unassignedNeighbors[cIdx]++;
            }
        }

        assignment[varIdx] = -1;
        return true;
    }

    private static bool RunTier3()
    {
        Array.Clear(visited, 0, visited.Length);
        bool globalProgress = false;

        for (int startIdx = 0; startIdx < H * W; startIdx++)
        {
            if (visited[startIdx]) continue;
            if (board[startIdx] > 8 || unknownNeighborCount[startIdx] == 0) continue;

            compRevealed.Clear();
            compUnknown.Clear();

            int bfsHead = 0;
            int bfsTail = 0;
            bfsQueue[bfsTail++] = startIdx;
            visited[startIdx] = true;

            while (bfsHead < bfsTail)
            {
                int curr = bfsQueue[bfsHead++];
                int r = curr / W;
                int c = curr % W;

                int rMin = r > 0 ? r - 1 : 0;
                int rMax = r < H - 1 ? r + 1 : H - 1;
                int cMin = c > 0 ? c - 1 : 0;
                int cMax = c < W - 1 ? c + 1 : W - 1;

                if (board[curr] <= 8)
                {
                    compRevealed.Add(curr);
                    for (int nr = rMin; nr <= rMax; nr++)
                    {
                        int rowOffset = nr * W;
                        for (int nc = cMin; nc <= cMax; nc++)
                        {
                            if (nr == r && nc == c) continue;
                            int nIdx = rowOffset + nc;
                            if (board[nIdx] == 9 && !visited[nIdx])
                            {
                                visited[nIdx] = true;
                                bfsQueue[bfsTail++] = nIdx;
                            }
                        }
                    }
                }
                else if (board[curr] == 9)
                {
                    compUnknown.Add(curr);
                    for (int nr = rMin; nr <= rMax; nr++)
                    {
                        int rowOffset = nr * W;
                        for (int nc = cMin; nc <= cMax; nc++)
                        {
                            if (nr == r && nc == c) continue;
                            int nIdx = rowOffset + nc;
                            if (board[nIdx] <= 8 && unknownNeighborCount[nIdx] > 0 && !visited[nIdx])
                            {
                                visited[nIdx] = true;
                                bfsQueue[bfsTail++] = nIdx;
                            }
                        }
                    }
                }
            }

            int N = compUnknown.Count;
            if (N == 0 || N > 40) continue;

            varMap.Clear();
            for (int i = 0; i < N; i++)
            {
                varMap[compUnknown[i]] = i;
            }

            int M = compRevealed.Count;
            if (M > constraints.Length) continue;

            for (int i = 0; i < M; i++)
            {
                int rIdx = compRevealed[i];
                constraints[i].CellIdx = rIdx;
                constraints[i].Target = remainingMines[rIdx];

                int r = rIdx / W;
                int c = rIdx % W;
                int rMin = r > 0 ? r - 1 : 0;
                int rMax = r < H - 1 ? r + 1 : H - 1;
                int cMin = c > 0 ? c - 1 : 0;
                int cMax = c < W - 1 ? c + 1 : W - 1;

                int varCount = 0;
                for (int nr = rMin; nr <= rMax; nr++)
                {
                    int rowOffset = nr * W;
                    for (int nc = cMin; nc <= cMax; nc++)
                    {
                        if (nr == r && nc == c) continue;
                        int nIdx = rowOffset + nc;
                        if (board[nIdx] == 9 && varMap.TryGetValue(nIdx, out int varIdx))
                        {
                            constraints[i].SetVar(varCount++, varIdx);
                        }
                    }
                }
                constraints[i].VarCount = varCount;
            }

            for (int i = 0; i < N; i++)
            {
                varConstraints[i].Count = 0;
            }

            for (int i = 0; i < M; i++)
            {
                assignedMines[i] = 0;
                unassignedNeighbors[i] = constraints[i].VarCount;
                for (int j = 0; j < constraints[i].VarCount; j++)
                {
                    int vIdx = constraints[i].GetVar(j);
                    varConstraints[vIdx].Add(i);
                }
            }

            backtrackN = N;
            backtrackM = M;
            stepCount = 0;
            validConfigCount = 0;

            for (int i = 0; i < N; i++)
            {
                canBeSafe[i] = false;
                canBeMine[i] = false;
                assignment[i] = -1;
            }

            bool finished = Backtrack(0);

            if (finished && validConfigCount > 0)
            {
                for (int i = 0; i < N; i++)
                {
                    int uIdx = compUnknown[i];
                    if (canBeSafe[i] && !canBeMine[i])
                    {
                        if (board[uIdx] == 9)
                        {
                            MarkSafe(uIdx);
                            globalProgress = true;
                        }
                    }
                    else if (!canBeSafe[i] && canBeMine[i])
                    {
                        if (board[uIdx] == 9)
                        {
                            MarkMine(uIdx);
                            globalProgress = true;
                        }
                    }
                }
            }
        }

        return globalProgress;
    }

    private static void OutputResults()
    {
        int safeCount = 0;
        for (int i = 0; i < H * W; i++)
        {
            if (board[i] == 11)
            {
                safeCount++;
            }
        }

        using (var sw = new StreamWriter(Console.OpenStandardOutput(), System.Text.Encoding.ASCII, 65536))
        {
            WriteInt(sw, safeCount);
            sw.Write('\n');
            for (int i = 0; i < H * W; i++)
            {
                if (board[i] == 11)
                {
                    int x = i % W;
                    int y = i / W;
                    WriteInt(sw, x);
                    sw.Write(' ');
                    WriteInt(sw, y);
                    sw.Write('\n');
                }
            }
        }
    }

    private static void WriteInt(StreamWriter sw, int val)
    {
        if (val == 0)
        {
            sw.Write('0');
            return;
        }
        int pos = 11;
        while (val > 0)
        {
            intBuf[--pos] = (char)('0' + (val % 10));
            val /= 10;
        }
        sw.Write(intBuf, pos, 11 - pos);
    }
