using System;
using System.IO;
using System.Text;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;

public struct Box
{
    public int Index;
    public int W, H, D;
    public double Volume;
}

public struct FreeSpace
{
    public int X, Y, Z;
    public int W, H, D;
    public double Volume;

    public FreeSpace(int x, int y, int z, int w, int h, int d)
    {
        X = x; Y = y; Z = z;
        W = w; H = h; D = d;
        Volume = (double)w * h * d;
    }
}

public struct PackedBox
{
    public int Index;
    public int X, Y, Z;
    public int W, H, D;
}

public enum BoxSortOrder
{
    VolumeAsc,
    VolumeAscShortestDimFirst,
    VolumeAscLongestDimFirst,
    AreaAsc,
    PerimeterAsc,
    MaxDimAsc,
    MinDimAsc,
}

public enum FreeSpaceSelection
{
    BestVolumeFit,
    BottomLeft,
    BottomLeftHeight,
    BottomLeftWidth,
    ShortSideFit,
    LongSideFit,
}

public class PackConfig
{
    public BoxSortOrder SortOrder;
    public FreeSpaceSelection Rule;
    public int MaxFreeSpaces;
}

public struct PlacementScore : IComparable<PlacementScore>
{
    public double Key1;
    public double Key2;
    public double Key3;
    public double Key4;
    public double Key5;

    public PlacementScore(double k1, double k2, double k3, double k4, double k5)
    {
        Key1 = k1;
        Key2 = k2;
        Key3 = k3;
        Key4 = k4;
        Key5 = k5;
    }

    public int CompareTo(PlacementScore other)
    {
        int c = Key1.CompareTo(other.Key1);
        if (c != 0) return c;
        c = Key2.CompareTo(other.Key2);
        if (c != 0) return c;
        c = Key3.CompareTo(other.Key3);
        if (c != 0) return c;
        c = Key4.CompareTo(other.Key4);
        if (c != 0) return c;
        return Key5.CompareTo(other.Key5);
    }
}

public static class FastScanner
{
    private static readonly byte[] Buffer = new byte[65536];
    private static int _head;
    private static int _tail;
    private static Stream _stream;

    public static void Init(Stream stream)
    {
        _stream = stream;
        _head = 0;
        _tail = 0;
    }

    private static byte Read()
    {
        if (_head >= _tail)
        {
            _head = 0;
            _tail = _stream.Read(Buffer, 0, Buffer.Length);
            if (_tail <= 0) return 0;
        }
        return Buffer[_head++];
    }

    public static int NextInt()
    {
        int res = 0;
        byte b = Read();
        while (b > 0 && b <= 32)
        {
            b = Read();
        }
        if (b == 0) return -1;
        while (b > 32)
        {
            if (b >= '0' && b <= '9')
            {
                res = res * 10 + (b - '0');
            }
            b = Read();
        }
        return res;
    }
}

public class Program
{
    public static void Main()
    {
        using (var stdin = Console.OpenStandardInput())
        {
            FastScanner.Init(stdin);
            int N = FastScanner.NextInt();
            if (N <= 0) return;
            int W = FastScanner.NextInt();
            int H = FastScanner.NextInt();
            int D = FastScanner.NextInt();

            Box[] boxes = new Box[N];
            for (int i = 0; i < N; i++)
            {
                int w = FastScanner.NextInt();
                int h = FastScanner.NextInt();
                int d = FastScanner.NextInt();
                boxes[i] = new Box
                {
                    Index = i,
                    W = w,
                    H = h,
                    D = d,
                    Volume = (double)w * h * d
                };
            }

            var bestPacked = Solve(boxes, W, H, D);

            using (var sw = new StreamWriter(Console.OpenStandardOutput(), new UTF8Encoding(false), 65536))
            {
                sw.WriteLine(bestPacked.Count);
                for (int i = 0; i < bestPacked.Count; i++)
                {
                    var pb = bestPacked[i];
                    sw.Write(pb.Index);
                    sw.Write(' ');
                    sw.Write(pb.X);
                    sw.Write(' ');
                    sw.Write(pb.Y);
                    sw.Write(' ');
                    sw.WriteLine(pb.Z);
                }
            }
        }
    }

    public static List<PackedBox> Solve(Box[] boxes, int containerW, int containerH, int containerD)
    {
        int N = boxes.Length;
        int maxFreeSpaces = Math.Max(100, Math.Min(5000, 1500000 / N));

        var sortOrders = new[]
        {
            BoxSortOrder.VolumeAsc,
            BoxSortOrder.VolumeAscShortestDimFirst,
            BoxSortOrder.VolumeAscLongestDimFirst,
            BoxSortOrder.PerimeterAsc,
            BoxSortOrder.MaxDimAsc,
            BoxSortOrder.MinDimAsc,
            BoxSortOrder.AreaAsc
        };

        var rules = new[]
        {
            FreeSpaceSelection.BestVolumeFit,
            FreeSpaceSelection.BottomLeft,
            FreeSpaceSelection.BottomLeftHeight,
            FreeSpaceSelection.BottomLeftWidth,
            FreeSpaceSelection.ShortSideFit,
            FreeSpaceSelection.LongSideFit
        };

        var configs = new List<PackConfig>();
        foreach (var order in sortOrders)
        {
            foreach (var rule in rules)
            {
                configs.Add(new PackConfig { SortOrder = order, Rule = rule, MaxFreeSpaces = maxFreeSpaces });
            }
        }

        if (maxFreeSpaces > 100)
        {
            int alternateSpaces = (int)(maxFreeSpaces * 1.5);
            configs.Add(new PackConfig { SortOrder = BoxSortOrder.VolumeAsc, Rule = FreeSpaceSelection.BestVolumeFit, MaxFreeSpaces = alternateSpaces });
            configs.Add(new PackConfig { SortOrder = BoxSortOrder.VolumeAsc, Rule = FreeSpaceSelection.BottomLeftHeight, MaxFreeSpaces = alternateSpaces });
            configs.Add(new PackConfig { SortOrder = BoxSortOrder.VolumeAscShortestDimFirst, Rule = FreeSpaceSelection.BestVolumeFit, MaxFreeSpaces = alternateSpaces });
            configs.Add(new PackConfig { SortOrder = BoxSortOrder.VolumeAscShortestDimFirst, Rule = FreeSpaceSelection.BottomLeftHeight, MaxFreeSpaces = alternateSpaces });
        }

        object lockObj = new object();
        List<PackedBox> bestResult = null;
        int maxPackedCount = -1;

        using (var cts = new CancellationTokenSource())
        {
            var timer = new System.Timers.Timer(28000);
            timer.Elapsed += (s, e) => { cts.Cancel(); };
            timer.Start();

            try
            {
                Parallel.ForEach(configs, new ParallelOptions { CancellationToken = cts.Token, MaxDegreeOfParallelism = Environment.ProcessorCount }, config =>
                {
                    if (cts.Token.IsCancellationRequested)
                        return;

                    var result = Pack(boxes, containerW, containerH, containerD, config, cts.Token);

                    lock (lockObj)
                    {
                        if (result != null && result.Count > maxPackedCount)
                        {
                            maxPackedCount = result.Count;
                            bestResult = result;

                            if (maxPackedCount == boxes.Length)
                            {
                                cts.Cancel();
                            }
                        }
                    }
                });
            }
            catch (OperationCanceledException)
            {
            }
            finally
            {
                timer.Stop();
                timer.Dispose();
            }
        }

        if (bestResult == null)
        {
            var fastConfig = new PackConfig { SortOrder = BoxSortOrder.VolumeAsc, Rule = FreeSpaceSelection.BestVolumeFit, MaxFreeSpaces = 100 };
            bestResult = Pack(boxes, containerW, containerH, containerD, fastConfig, CancellationToken.None);
        }

        return bestResult;
    }

    public static List<PackedBox> Pack(Box[] boxes, int containerW, int containerH, int containerD, PackConfig config, CancellationToken token)
    {
        var packed = new List<PackedBox>();
        var spaces = new List<FreeSpace>
        {
            new FreeSpace(0, 0, 0, containerW, containerH, containerD)
        };

        Box[] sortedBoxes = SortBoxes(boxes, config.SortOrder);

        var newSpaces = new List<FreeSpace>(6);
        var filteredNewSpaces = new List<FreeSpace>(6);

        for (int bIdx = 0; bIdx < sortedBoxes.Length; bIdx++)
        {
            if (token.IsCancellationRequested)
                break;

            var box = sortedBoxes[bIdx];
            var orientations = GetOrientations(box.W, box.H, box.D);

            int bestSpaceIdx = -1;
            int bestW = 0, bestH = 0, bestD = 0;
            PlacementScore bestScore = new PlacementScore(double.MaxValue, double.MaxValue, double.MaxValue, double.MaxValue, double.MaxValue);

            for (int i = 0; i < spaces.Count; i++)
            {
                var S = spaces[i];
                int spaceMax = Math.Max(S.W, Math.Max(S.H, S.D));
                int boxMin = Math.Min(box.W, Math.Min(box.H, box.D));
                if (boxMin > spaceMax) continue;

                for (int o = 0; o < orientations.Count; o++)
                {
                    int bw = orientations[o].w;
                    int bh = orientations[o].h;
                    int bd = orientations[o].d;

                    if (bw <= S.W && bh <= S.H && bd <= S.D)
                    {
                        var score = GetScore(config.Rule, S, bw, bh, bd);
                        if (score.CompareTo(bestScore) < 0)
                        {
                            bestScore = score;
                            bestSpaceIdx = i;
                            bestW = bw;
                            bestH = bh;
                            bestD = bd;
                        }
                    }
                }
            }

            if (bestSpaceIdx != -1)
            {
                var S = spaces[bestSpaceIdx];
                int bx = S.X;
                int by = S.Y;
                int bz = S.Z;

                packed.Add(new PackedBox
                {
                    Index = box.Index,
                    X = bx,
                    Y = by,
                    Z = bz,
                    W = bestW,
                    H = bestH,
                    D = bestD
                });

                newSpaces.Clear();
                for (int i = spaces.Count - 1; i >= 0; i--)
                {
                    var currSpace = spaces[i];
                    if (Overlaps(currSpace, bx, by, bz, bestW, bestH, bestD))
                    {
                        spaces.RemoveAt(i);

                        if (bx > currSpace.X)
                            newSpaces.Add(new FreeSpace(currSpace.X, currSpace.Y, currSpace.Z, bx - currSpace.X, currSpace.H, currSpace.D));
                        if (bx + bestW < currSpace.X + currSpace.W)
                            newSpaces.Add(new FreeSpace(bx + bestW, currSpace.Y, currSpace.Z, currSpace.X + currSpace.W - (bx + bestW), currSpace.H, currSpace.D));
                        if (by > currSpace.Y)
                            newSpaces.Add(new FreeSpace(currSpace.X, currSpace.Y, currSpace.Z, currSpace.W, by - currSpace.Y, currSpace.D));
                        if (by + bestH < currSpace.Y + currSpace.H)
                            newSpaces.Add(new FreeSpace(currSpace.X, by + bestH, currSpace.Z, currSpace.W, currSpace.Y + currSpace.H - (by + bestH), currSpace.D));
                        if (bz > currSpace.Z)
                            newSpaces.Add(new FreeSpace(currSpace.X, currSpace.Y, currSpace.Z, currSpace.W, currSpace.H, bz - currSpace.Z));
                        if (bz + bestD < currSpace.Z + currSpace.D)
                            newSpaces.Add(new FreeSpace(currSpace.X, currSpace.Y, bz + bestD, currSpace.W, currSpace.H, currSpace.Z + currSpace.D - (bz + bestD)));
                    }
                }

                filteredNewSpaces.Clear();
                for (int j = 0; j < newSpaces.Count; j++)
                {
                    var ns = newSpaces[j];
                    bool containedInExisting = false;
                    for (int i = 0; i < spaces.Count; i++)
                    {
                        if (IsContained(ns, spaces[i]))
                        {
                            containedInExisting = true;
                            break;
                        }
                    }
                    if (containedInExisting) continue;

                    bool containedInNew = false;
                    for (int i = filteredNewSpaces.Count - 1; i >= 0; i--)
                    {
                        if (IsContained(ns, filteredNewSpaces[i]))
                        {
                            containedInNew = true;
                            break;
                        }
                        if (IsContained(filteredNewSpaces[i], ns))
                        {
                            filteredNewSpaces.RemoveAt(i);
                        }
                    }
                    if (!containedInNew)
                    {
                        filteredNewSpaces.Add(ns);
                    }
                }

                spaces.AddRange(filteredNewSpaces);

                if (spaces.Count > config.MaxFreeSpaces)
                {
                    spaces.Sort((a, b) => b.Volume.CompareTo(a.Volume));
                    spaces.RemoveRange(config.MaxFreeSpaces, spaces.Count - config.MaxFreeSpaces);
                }
            }
        }

        return packed;
    }

    public static Box[] SortBoxes(Box[] boxes, BoxSortOrder order)
    {
        var list = new List<Box>(boxes);
        switch (order)
        {
            case BoxSortOrder.VolumeAsc:
                list.Sort((a, b) => {
                    int c = a.Volume.CompareTo(b.Volume);
                    if (c != 0) return c;
                    return (a.W + a.H + a.D).CompareTo(b.W + b.H + b.D);
                });
                break;
            case BoxSortOrder.VolumeAscShortestDimFirst:
                list.Sort((a, b) => {
                    int c = a.Volume.CompareTo(b.Volume);
                    if (c != 0) return c;
                    int minA = Math.Min(a.W, Math.Min(a.H, a.D));
                    int minB = Math.Min(b.W, Math.Min(b.H, b.D));
                    return minA.CompareTo(minB);
                });
                break;
            case BoxSortOrder.VolumeAscLongestDimFirst:
                list.Sort((a, b) => {
                    int c = a.Volume.CompareTo(b.Volume);
                    if (c != 0) return c;
                    int maxA = Math.Max(a.W, Math.Max(a.H, a.D));
                    int maxB = Math.Max(b.W, Math.Max(b.H, b.D));
                    return maxB.CompareTo(maxA);
                });
                break;
            case BoxSortOrder.AreaAsc:
                list.Sort((a, b) => {
                    double areaA = (double)a.W * a.H;
                    double areaB = (double)b.W * b.H;
                    return areaA.CompareTo(areaB);
                });
                break;
            case BoxSortOrder.PerimeterAsc:
                list.Sort((a, b) => (a.W + a.H + a.D).CompareTo(b.W + b.H + b.D));
                break;
            case BoxSortOrder.MaxDimAsc:
                list.Sort((a, b) => {
                    int maxA = Math.Max(a.W, Math.Max(a.H, a.D));
                    int maxB = Math.Max(b.W, Math.Max(b.H, b.D));
                    return maxA.CompareTo(maxB);
                });
                break;
            case BoxSortOrder.MinDimAsc:
                list.Sort((a, b) => {
                    int minA = Math.Min(a.W, Math.Min(a.H, a.D));
                    int minB = Math.Min(b.W, Math.Min(b.H, b.D));
                    return minA.CompareTo(minB);
                });
                break;
        }
        return list.ToArray();
    }

    public static List<(int w, int h, int d)> GetOrientations(int w, int h, int d)
    {
        var list = new List<(int, int, int)>();
        AddUnique(list, w, h, d);
        AddUnique(list, w, d, h);
        AddUnique(list, h, w, d);
        AddUnique(list, h, d, w);
        AddUnique(list, d, w, h);
        AddUnique(list, d, h, w);
        return list;
    }

    private static void AddUnique(List<(int, int, int)> list, int w, int h, int d)
    {
        foreach (var item in list)
        {
            if (item.Item1 == w && item.Item2 == h && item.Item3 == d)
                return;
        }
        list.Add((w, h, d));
    }

    public static PlacementScore GetScore(FreeSpaceSelection rule, FreeSpace S, int w, int h, int d)
    {
        int rw = S.W - w;
        int rh = S.H - h;
        int rd = S.D - d;

        switch (rule)
        {
            case FreeSpaceSelection.BestVolumeFit:
                return new PlacementScore(S.Volume, (double)rw * rh * rd, S.Z, S.Y, S.X);

            case FreeSpaceSelection.BottomLeft:
                return new PlacementScore((double)S.X + S.Y + S.Z, S.Z, S.Y, S.X, d);

            case FreeSpaceSelection.BottomLeftHeight:
                return new PlacementScore(S.Z, S.Y, S.X, d, h);

            case FreeSpaceSelection.BottomLeftWidth:
                return new PlacementScore(S.X, S.Y, S.Z, w, h);

            case FreeSpaceSelection.ShortSideFit:
                {
                    int minRemainder = Math.Min(rw, Math.Min(rh, rd));
                    int maxRemainder = Math.Max(rw, Math.Max(rh, rd));
                    return new PlacementScore(minRemainder, maxRemainder, S.Z, S.Y, S.X);
                }

            case FreeSpaceSelection.LongSideFit:
                {
                    int minRemainder = Math.Min(rw, Math.Min(rh, rd));
                    int maxRemainder = Math.Max(rw, Math.Max(rh, rd));
                    return new PlacementScore(maxRemainder, minRemainder, S.Z, S.Y, S.X);
                }

            default:
                return new PlacementScore(S.Volume, 0, 0, 0, 0);
        }
    }

    public static bool Overlaps(FreeSpace a, int bx, int by, int bz, int bw, int bh, int bd)
    {
        return a.X < bx + bw && a.X + a.W > bx &&
               a.Y < by + bh && a.Y + a.H > by &&
               a.Z < bz + bd && a.Z + a.D > bz;
    }

    public static bool IsContained(FreeSpace a, FreeSpace b)
    {
        return a.X >= b.X && a.X + a.W <= b.X + b.W &&
               a.Y >= b.Y && a.Y + a.H <= b.Y + b.H &&
               a.Z >= b.Z && a.Z + a.D <= b.Z + b.D;
    }
}
