use std::io::{self, Read, Write, BufWriter};


#[derive(Clone, Copy, Debug)]
struct Point {
    x: f64,
    y: f64,
}

// Deterministic RNG (Linear Congruential Generator)
struct SimpleRng {
    state: u64,
}

impl SimpleRng {
    fn new(seed: u64) -> Self {
        Self { state: seed }
    }
    fn next_u32(&mut self) -> u32 {
        self.state = self.state.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
        (self.state >> 32) as u32
    }
    fn next_f64(&mut self) -> f64 {
        (self.next_u32() as f64) / (u32::MAX as f64)
    }
}

// BFS workspace to avoid frequent vector allocations
struct BfsWorkspace {
    queue: Vec<usize>,
    dist: Vec<i32>,
}

impl BfsWorkspace {
    fn new(n: usize) -> Self {
        Self {
            queue: vec![0; n],
            dist: vec![-1; n],
        }
    }
    fn run(&mut self, start: usize, adj: &[Vec<usize>]) {
        let n = adj.len();
        for d in self.dist[0..n].iter_mut() {
            *d = -1;
        }
        let mut head = 0;
        let mut tail = 0;
        self.dist[start] = 0;
        self.queue[tail] = start;
        tail += 1;
        while head < tail {
            let u = self.queue[head];
            head += 1;
            let d = self.dist[u];
            for &v in &adj[u] {
                if self.dist[v] == -1 {
                    self.dist[v] = d + 1;
                    self.queue[tail] = v;
                    tail += 1;
                }
            }
        }
    }
}

// Fast buffered input reader
struct FastBufReader {
    buffer: Vec<u8>,
    pos: usize,
}

impl FastBufReader {
    fn new() -> Self {
        let mut buffer = Vec::new();
        io::stdin().read_to_end(&mut buffer).unwrap();
        Self { buffer, pos: 0 }
    }
    #[inline]
    fn next_u32(&mut self) -> Option<u32> {
        while self.pos < self.buffer.len() && self.buffer[self.pos] <= b' ' {
            self.pos += 1;
        }
        if self.pos >= self.buffer.len() {
            return None;
        }
        let mut val = 0u32;
        while self.pos < self.buffer.len() && self.buffer[self.pos] > b' ' {
            val = val * 10 + (self.buffer[self.pos] - b'0') as u32;
            self.pos += 1;
        }
        Some(val)
    }
}

// Line segment intersection helpers
fn ccw(px: f64, py: f64, qx: f64, qy: f64, rx: f64, ry: f64) -> i32 {
    let val = (qy - py) * (rx - qx) - (qx - px) * (ry - qy);
    if val.abs() < 1e-9 {
        0
    } else if val > 0.0 {
        1
    } else {
        -1
    }
}

fn on_segment(px: f64, py: f64, qx: f64, qy: f64, rx: f64, ry: f64) -> bool {
    qx >= px.min(rx) && qx <= px.max(rx) && qy >= py.min(ry) && qy <= py.max(ry)
}

fn intersect(
    ax: f64, ay: f64, bx: f64, by: f64,
    cx: f64, cy: f64, dx: f64, dy: f64,
) -> bool {
    let o1 = ccw(ax, ay, bx, by, cx, cy);
    let o2 = ccw(ax, ay, bx, by, dx, dy);
    let o3 = ccw(cx, cy, dx, dy, ax, ay);
    let o4 = ccw(cx, cy, dx, dy, bx, by);
    
    if o1 != o2 && o3 != o4 {
        return true;
    }
    if o1 == 0 && on_segment(ax, ay, cx, cy, bx, by) { return true; }
    if o2 == 0 && on_segment(ax, ay, dx, dy, bx, by) { return true; }
    if o3 == 0 && on_segment(cx, cy, ax, ay, dx, dy) { return true; }
    if o4 == 0 && on_segment(cx, cy, bx, by, dx, dy) { return true; }
    false
}

fn count_crossings(edges: &[(usize, usize)], pos: &[Point]) -> usize {
    let mut count = 0;
    let m = edges.len();
    for i in 0..m {
        let (u1, v1) = edges[i];
        let ax = pos[u1].x;
        let ay = pos[u1].y;
        let bx = pos[v1].x;
        let by = pos[v1].y;
        for j in (i+1)..m {
            let (u2, v2) = edges[j];
            if u1 == u2 || u1 == v2 || v1 == u2 || v1 == v2 {
                continue;
            }
            let cx = pos[u2].x;
            let cy = pos[u2].y;
            let dx = pos[v2].x;
            let dy = pos[v2].y;
            if intersect(ax, ay, bx, by, cx, cy, dx, dy) {
                count += 1;
            }
        }
    }
    count
}

// Jacobi Eigenvalue solver for symmetric matrices
fn jacobi_eigen(
    a_in: &[f64],
    k: usize,
    eigenvalues: &mut [f64],
    eigenvectors: &mut [f64],
) {
    let mut a = a_in.to_vec();
    for i in 0..k {
        for j in 0..k {
            eigenvectors[i * k + j] = if i == j { 1.0 } else { 0.0 };
        }
    }
    let max_iterations = 100 * k * k;
    let eps = 1e-9;
    for _iter in 0..max_iterations {
        let mut p = 0;
        let mut q = 1;
        let mut max_val = a[0 * k + 1].abs();
        for i in 0..k {
            for j in (i+1)..k {
                let val = a[i * k + j].abs();
                if val > max_val {
                    max_val = val;
                    p = i;
                    q = j;
                }
            }
        }
        if max_val < eps {
            break;
        }
        let app = a[p * k + p];
        let aqq = a[q * k + q];
        let apq = a[p * k + q];
        let tau = (aqq - app) / (2.0 * apq);
        let t = if tau >= 0.0 {
            1.0 / (tau + (1.0 + tau * tau).sqrt())
        } else {
            -1.0 / (-tau + (1.0 + tau * tau).sqrt())
        };
        let c = 1.0 / (1.0 + t * t).sqrt();
        let s = c * t;
        
        a[p * k + p] = app - t * apq;
        a[q * k + q] = aqq + t * apq;
        a[p * k + q] = 0.0;
        a[q * k + p] = 0.0;
        
        for r in 0..k {
            if r != p && r != q {
                let arp = a[r * k + p];
                let arq = a[r * k + q];
                a[r * k + p] = c * arp - s * arq;
                a[p * k + r] = a[r * k + p];
                a[r * k + q] = c * arq + s * arp;
                a[q * k + r] = a[r * k + q];
            }
        }
        for r in 0..k {
            let vrp = eigenvectors[r * k + p];
            let vrq = eigenvectors[r * k + q];
            eigenvectors[r * k + p] = c * vrp - s * vrq;
            eigenvectors[r * k + q] = c * vrq + s * vrp;
        }
    }
    for i in 0..k {
        eigenvalues[i] = a[i * k + i];
    }
}

// Choose pivots using Farthest Point Sampling / k-centers
fn choose_pivots(
    comp_nodes: &[usize],
    adj: &[Vec<usize>],
    k: usize,
    workspace: &mut BfsWorkspace,
) -> Vec<usize> {
    let n_c = comp_nodes.len();
    if n_c <= k {
        return comp_nodes.to_vec();
    }
    let mut pivots = Vec::with_capacity(k);
    let mut max_deg = 0;
    let mut first_pivot = comp_nodes[0];
    for &u in comp_nodes {
        if adj[u].len() > max_deg {
            max_deg = adj[u].len();
            first_pivot = u;
        }
    }
    pivots.push(first_pivot);
    
    let mut min_dist = vec![i32::MAX; adj.len()];
    for i in 0..k {
        let last_pivot = pivots[i];
        workspace.run(last_pivot, adj);
        if i == k - 1 {
            break;
        }
        let mut next_pivot = last_pivot;
        let mut max_val = -1;
        for &u in comp_nodes {
            let d = workspace.dist[u];
            if d != -1 {
                if d < min_dist[u] {
                    min_dist[u] = d;
                }
                if min_dist[u] > max_val {
                    max_val = min_dist[u];
                    next_pivot = u;
                }
            }
        }
        pivots.push(next_pivot);
    }
    pivots
}

// Run standard O(N_c^2) Fruchterman-Reingold layout
fn run_standard_fr(
    comp_nodes: &[usize],
    adj: &[Vec<usize>],
    pos: &mut [Point],
    iterations: usize,
    initial_temp: f64,
) {
    let n_c = comp_nodes.len();
    if n_c <= 1 { return; }
    let k_dist = 15.0;
    let k2 = k_dist * k_dist;
    let mut temp = initial_temp;
    let cooling = 0.95;
    let n = pos.len();
    let mut disp = vec![Point { x: 0.0, y: 0.0 }; n];
    
    for _iter in 0..iterations {
        for &u in comp_nodes {
            disp[u].x = 0.0;
            disp[u].y = 0.0;
        }
        // Repulsive forces
        for i in 0..n_c {
            let u = comp_nodes[i];
            let px = pos[u].x;
            let py = pos[u].y;
            for j in (i+1)..n_c {
                let v = comp_nodes[j];
                let dx = px - pos[v].x;
                let dy = py - pos[v].y;
                let dist_sq = dx*dx + dy*dy;
                if dist_sq > 1e-6 {
                    let dist = dist_sq.sqrt();
                    let f = k2 / dist;
                    let fx = (dx / dist) * f;
                    let fy = (dy / dist) * f;
                    disp[u].x += fx;
                    disp[u].y += fy;
                    disp[v].x -= fx;
                    disp[v].y -= fy;
                } else {
                    disp[u].x += 0.1;
                    disp[u].y += 0.1;
                    disp[v].x -= 0.1;
                    disp[v].y -= 0.1;
                }
            }
        }
        // Attractive forces
        for &u in comp_nodes {
            let px = pos[u].x;
            let py = pos[u].y;
            for &v in &adj[u] {
                if u < v {
                    let dx = px - pos[v].x;
                    let dy = py - pos[v].y;
                    let dist_sq = dx*dx + dy*dy;
                    if dist_sq > 1e-6 {
                        let dist = dist_sq.sqrt();
                        let f = dist_sq / k_dist;
                        let fx = (dx / dist) * f;
                        let fy = (dy / dist) * f;
                        disp[u].x -= fx;
                        disp[u].y -= fy;
                        disp[v].x += fx;
                        disp[v].y += fy;
                    }
                }
            }
        }
        // Gravity force (weak attraction to center)
        let mut cx = 0.0;
        let mut cy = 0.0;
        for &u in comp_nodes {
            cx += pos[u].x;
            cy += pos[u].y;
        }
        cx /= n_c as f64;
        cy /= n_c as f64;
        let gravity = 0.01;
        for &u in comp_nodes {
            let dx = pos[u].x - cx;
            let dy = pos[u].y - cy;
            disp[u].x -= gravity * dx;
            disp[u].y -= gravity * dy;
        }
        // Update positions
        for &u in comp_nodes {
            let dx = disp[u].x;
            let dy = disp[u].y;
            let dist_sq = dx*dx + dy*dy;
            if dist_sq > 1e-6 {
                let dist = dist_sq.sqrt();
                let limit = dist.min(temp);
                pos[u].x += (dx / dist) * limit;
                pos[u].y += (dy / dist) * limit;
            }
        }
        temp *= cooling;
    }
}

// Run grid-based Fruchterman-Reingold layout using spatial hashing and flat arrays
fn run_grid_fr(
    comp_nodes: &[usize],
    adj: &[Vec<usize>],
    pos: &mut [Point],
    iterations: usize,
    initial_temp: f64,
) {
    let n_c = comp_nodes.len();
    if n_c <= 1 { return; }
    let k_dist = 15.0;
    let k2 = k_dist * k_dist;
    let cell_size = 2.0 * k_dist;
    let mut temp = initial_temp;
    let cooling = 0.95;
    let n = pos.len();
    let mut disp = vec![Point { x: 0.0, y: 0.0 }; n];
    
    let b_size = (2 * n_c).next_power_of_two();
    let mut head = vec![-1i32; b_size];
    let mut next = vec![-1i32; n];
    let mut cell_coords = vec![(0, 0); n];
    
    for _iter in 0..iterations {
        for &u in comp_nodes {
            disp[u].x = 0.0;
            disp[u].y = 0.0;
        }
        
        head.fill(-1);
        
        for &u in comp_nodes {
            let cx = (pos[u].x / cell_size).floor() as i32;
            let cy = (pos[u].y / cell_size).floor() as i32;
            cell_coords[u] = (cx, cy);
            
            // Spatial hashing
            let h = ((cx.wrapping_mul(73856093)) ^ (cy.wrapping_mul(19349663))) & ((b_size - 1) as i32);
            let bucket = h as usize;
            next[u] = head[bucket];
            head[bucket] = u as i32;
        }
        
        // Repulsive forces
        for &u in comp_nodes {
            let px = pos[u].x;
            let py = pos[u].y;
            let (cx, cy) = cell_coords[u];
            let mut neighbors_checked = 0;
            const MAX_NEIGHBORS: usize = 100;
            
            'outer: for dx in -1..=1 {
                for dy in -1..=1 {
                    let ncx = cx + dx;
                    let ncy = cy + dy;
                    let h = ((ncx.wrapping_mul(73856093)) ^ (ncy.wrapping_mul(19349663))) & ((b_size - 1) as i32);
                    let bucket = h as usize;
                    
                    let mut curr = head[bucket];
                    while curr != -1 {
                        let v = curr as usize;
                        if u != v {
                            let (v_cx, v_cy) = cell_coords[v];
                            if v_cx == ncx && v_cy == ncy {
                                let dx = px - pos[v].x;
                                let dy = py - pos[v].y;
                                let dist_sq = dx*dx + dy*dy;
                                if dist_sq > 1e-6 {
                                    let dist = dist_sq.sqrt();
                                    if dist < 3.0 * k_dist {
                                        let f = k2 / dist;
                                        disp[u].x += (dx / dist) * f;
                                        disp[u].y += (dy / dist) * f;
                                    }
                                } else {
                                    disp[u].x += 0.1;
                                    disp[u].y += 0.1;
                                }
                                neighbors_checked += 1;
                                if neighbors_checked >= MAX_NEIGHBORS {
                                    break 'outer;
                                }
                            }
                        }
                        curr = next[v];
                    }
                }
            }
        }
        // Attractive forces
        for &u in comp_nodes {
            let px = pos[u].x;
            let py = pos[u].y;
            for &v in &adj[u] {
                if u < v {
                    let dx = px - pos[v].x;
                    let dy = py - pos[v].y;
                    let dist_sq = dx*dx + dy*dy;
                    if dist_sq > 1e-6 {
                        let dist = dist_sq.sqrt();
                        let f = dist_sq / k_dist;
                        let fx = (dx / dist) * f;
                        let fy = (dy / dist) * f;
                        disp[u].x -= fx;
                        disp[u].y -= fy;
                        disp[v].x += fx;
                        disp[v].y += fy;
                    }
                }
            }
        }
        // Gravity force
        let mut cx = 0.0;
        let mut cy = 0.0;
        for &u in comp_nodes {
            cx += pos[u].x;
            cy += pos[u].y;
        }
        cx /= n_c as f64;
        cy /= n_c as f64;
        let gravity = 0.01;
        for &u in comp_nodes {
            let dx = pos[u].x - cx;
            let dy = pos[u].y - cy;
            disp[u].x -= gravity * dx;
            disp[u].y -= gravity * dy;
        }
        // Update positions
        for &u in comp_nodes {
            let dx = disp[u].x;
            let dy = disp[u].y;
            let dist_sq = dx*dx + dy*dy;
            if dist_sq > 1e-6 {
                let dist = dist_sq.sqrt();
                let limit = dist.min(temp);
                pos[u].x += (dx / dist) * limit;
                pos[u].y += (dy / dist) * limit;
            }
        }
        temp *= cooling;
    }
}

// Find all connected components
fn find_components(n: usize, adj: &[Vec<usize>]) -> Vec<Vec<usize>> {
    let mut visited = vec![false; n];
    let mut components = Vec::new();
    let mut queue = Vec::new();
    for i in 0..n {
        if !visited[i] {
            let mut comp = Vec::new();
            queue.clear();
            queue.push(i);
            visited[i] = true;
            let mut head = 0;
            while head < queue.len() {
                let u = queue[head];
                head += 1;
                comp.push(u);
                for &v in &adj[u] {
                    if !visited[v] {
                        visited[v] = true;
                        queue.push(v);
                    }
                }
            }
            components.push(comp);
        }
    }
    components
}

fn main() {
    let mut reader = FastBufReader::new();
    let n = match reader.next_u32() {
        Some(val) => val as usize,
        None => return,
    };
    let m = match reader.next_u32() {
        Some(val) => val as usize,
        None => return,
    };
    
    let mut adj = vec![Vec::new(); n];
    for _ in 0..m {
        if let (Some(u), Some(v)) = (reader.next_u32(), reader.next_u32()) {
            let u = u as usize;
            let v = v as usize;
            if u < n && v < n {
                if u != v {
                    adj[u].push(v);
                    adj[v].push(u);
                }
            }
        }
    }
    
    // Remove duplicate/parallel edges
    for u in 0..n {
        adj[u].sort_unstable();
        adj[u].dedup();
    }
    
    // Find components
    let components = find_components(n, &adj);
    
    let mut pos = vec![Point { x: 0.0, y: 0.0 }; n];
    let mut large_components = Vec::new();
    let mut small_components = Vec::new();
    
    for comp in components {
        if comp.len() > 3 {
            large_components.push(comp);
        } else {
            small_components.push(comp);
        }
    }
    
    // Workspace for BFS
    let mut bfs_workspace = BfsWorkspace::new(n);
    let mut packed_large_circles = Vec::new(); // (cx, cy, radius, comp_idx)
    
    for (comp_idx, comp) in large_components.iter().enumerate() {
        let n_c = comp.len();
        
        // Find local edges in component
        let mut comp_edges = Vec::new();
        for &u in comp {
            for &v in &adj[u] {
                if u < v {
                    comp_edges.push((u, v));
                }
            }
        }
        
        if n_c <= 15 {
            // Trivial scale: run multiple random restarts to find 0 crossings
            let mut best_crossings = usize::MAX;
            let mut best_pos = vec![Point { x: 0.0, y: 0.0 }; n_c];
            
            // Up to 100 restarts
            let mut rng = SimpleRng::new(1337);
            for _restart in 0..100 {
                // Initialize randomly in a circle
                let mut local_pos = vec![Point { x: 0.0, y: 0.0 }; n];
                for &u in comp {
                    let r = 20.0 * rng.next_f64().sqrt();
                    let theta = rng.next_f64() * 2.0 * std::f64::consts::PI;
                    local_pos[u] = Point { x: r * theta.cos(), y: r * theta.sin() };
                }
                
                run_standard_fr(comp, &adj, &mut local_pos, 200, 10.0 * (n_c as f64).sqrt().max(10.0));
                let crossings = count_crossings(&comp_edges, &local_pos);
                if crossings < best_crossings {
                    best_crossings = crossings;
                    for (i, &u) in comp.iter().enumerate() {
                        best_pos[i] = local_pos[u];
                    }
                    if crossings == 0 {
                        break;
                    }
                }
            }
            
            // Assign best coordinates
            for (i, &u) in comp.iter().enumerate() {
                pos[u] = best_pos[i];
            }
            
        } else if n_c <= 100 {
            // Medium scale: 10 restarts
            let mut best_crossings = usize::MAX;
            let mut best_pos = vec![Point { x: 0.0, y: 0.0 }; n_c];
            let mut rng = SimpleRng::new(42);
            for _restart in 0..10 {
                let mut local_pos = vec![Point { x: 0.0, y: 0.0 }; n];
                for &u in comp {
                    let r = 20.0 * rng.next_f64().sqrt();
                    let theta = rng.next_f64() * 2.0 * std::f64::consts::PI;
                    local_pos[u] = Point { x: r * theta.cos(), y: r * theta.sin() };
                }
                run_standard_fr(comp, &adj, &mut local_pos, 250, 10.0 * (n_c as f64).sqrt().max(10.0));
                let crossings = count_crossings(&comp_edges, &local_pos);
                if crossings < best_crossings {
                    best_crossings = crossings;
                    for (i, &u) in comp.iter().enumerate() {
                        best_pos[i] = local_pos[u];
                    }
                    if crossings == 0 {
                        break;
                    }
                }
            }
            for (i, &u) in comp.iter().enumerate() {
                pos[u] = best_pos[i];
            }
            
        } else {
            // Large & Extreme scale: HDE + FR
            let k_pivots = 15.min(n_c);
            let pivots = choose_pivots(comp, &adj, k_pivots, &mut bfs_workspace);
            
            let mut coords = vec![0.0; n_c * k_pivots];
            for j in 0..k_pivots {
                bfs_workspace.run(pivots[j], &adj);
                for (v_idx, &u) in comp.iter().enumerate() {
                    let d = bfs_workspace.dist[u];
                    coords[v_idx * k_pivots + j] = if d == -1 { 1e9 } else { d as f64 };
                }
            }
            
            // Center columns
            let mut means = vec![0.0; k_pivots];
            for j in 0..k_pivots {
                let mut sum = 0.0;
                for v_idx in 0..n_c {
                    sum += coords[v_idx * k_pivots + j];
                }
                means[j] = sum / (n_c as f64);
            }
            for v_idx in 0..n_c {
                for j in 0..k_pivots {
                    coords[v_idx * k_pivots + j] -= means[j];
                }
            }
            
            // Covariance matrix
            let mut cov = vec![0.0; k_pivots * k_pivots];
            for i in 0..k_pivots {
                for j in i..k_pivots {
                    let mut sum = 0.0;
                    for v_idx in 0..n_c {
                        sum += coords[v_idx * k_pivots + i] * coords[v_idx * k_pivots + j];
                    }
                    let val = sum / (n_c as f64);
                    cov[i * k_pivots + j] = val;
                    cov[j * k_pivots + i] = val;
                }
            }
            
            // Jacobi solver
            let mut eigenvalues = vec![0.0; k_pivots];
            let mut eigenvectors = vec![0.0; k_pivots * k_pivots];
            jacobi_eigen(&cov, k_pivots, &mut eigenvalues, &mut eigenvectors);
            
            let mut eigen_indices: Vec<usize> = (0..k_pivots).collect();
            eigen_indices.sort_by(|&i, &j| eigenvalues[j].partial_cmp(&eigenvalues[i]).unwrap_or(std::cmp::Ordering::Equal));
            let idx1 = eigen_indices[0];
            let idx2 = eigen_indices[1];
            
            // Project
            for (v_idx, &u) in comp.iter().enumerate() {
                let mut x = 0.0;
                let mut y = 0.0;
                for j in 0..k_pivots {
                    let c = coords[v_idx * k_pivots + j];
                    x += c * eigenvectors[j * k_pivots + idx1];
                    y += c * eigenvectors[j * k_pivots + idx2];
                }
                pos[u] = Point { x, y };
            }
            
            // Fallback check & add tiny noise to break collinearity
            let mut rng = SimpleRng::new(57);
            for &u in comp {
                pos[u].x += (rng.next_f64() - 0.5) * 0.1;
                pos[u].y += (rng.next_f64() - 0.5) * 0.1;
            }
            
            // Center and scale initial HDE layout
            let mut cx = 0.0;
            let mut cy = 0.0;
            for &u in comp {
                cx += pos[u].x;
                cy += pos[u].y;
            }
            cx /= n_c as f64;
            cy /= n_c as f64;
            
            let mut total_dist = 0.0;
            for &u in comp {
                let dx = pos[u].x - cx;
                let dy = pos[u].y - cy;
                total_dist += (dx*dx + dy*dy).sqrt();
            }
            let avg_dist = total_dist / (n_c as f64);
            let target_avg = 10.0 * (n_c as f64).sqrt().max(10.0);
            let scale = if avg_dist > 1e-6 { target_avg / avg_dist } else { 1.0 };
            for &u in comp {
                pos[u].x = (pos[u].x - cx) * scale;
                pos[u].y = (pos[u].y - cy) * scale;
            }
            
            // Refine with FR
            let cost_standard = n_c * n_c + comp_edges.len();
            let cost_grid = n_c * 100 + comp_edges.len();
            
            if n_c <= 2500 {
                let iterations = (500_000_000 / cost_standard).clamp(5, 150);
                run_standard_fr(comp, &adj, &mut pos, iterations, 30.0);
            } else {
                let iterations = (500_000_000 / cost_grid).clamp(5, 50);
                run_grid_fr(comp, &adj, &mut pos, iterations, 30.0);
            }
        }
        
        // Final centering for the component and calculate radius
        let mut cx = 0.0;
        let mut cy = 0.0;
        for &u in comp {
            cx += pos[u].x;
            cy += pos[u].y;
        }
        cx /= n_c as f64;
        cy /= n_c as f64;
        
        let mut max_dist = 0.0;
        for &u in comp {
            let dx = pos[u].x - cx;
            let dy = pos[u].y - cy;
            let d = (dx*dx + dy*dy).sqrt();
            if d > max_dist {
                max_dist = d;
            }
            pos[u].x = dx;
            pos[u].y = dy;
        }
        
        let radius = max_dist.max(5.0);
        packed_large_circles.push((0.0, 0.0, radius, comp_idx));
    }
    
    // Sort large components by radius descending
    packed_large_circles.sort_by(|a, b| b.2.partial_cmp(&a.2).unwrap_or(std::cmp::Ordering::Equal));
    
    // Pack large components using spiral search
    let mut placed_circles: Vec<(f64, f64, f64)> = Vec::new(); // (x, y, radius)
    let mut final_large_centers = vec![(0.0, 0.0); large_components.len()];
    
    for &(_cx_init, _cy_init, radius, comp_idx) in &packed_large_circles {
        if placed_circles.is_empty() {
            placed_circles.push((0.0, 0.0, radius));
            final_large_centers[comp_idx] = (0.0, 0.0);
            continue;
        }
        
        let mut theta: f64 = 0.0;
        let step: f64 = 0.05;
        let mut r_spiral = 0.05 * radius;
        let mut best_pos = (0.0, 0.0);
        let mut found = false;
        
        while !found {
            let x = r_spiral * theta.cos();
            let y = r_spiral * theta.sin();
            let mut overlap = false;
            for &(px, py, pr) in &placed_circles {
                let dx = x - px;
                let dy = y - py;
                let dist = (dx*dx + dy*dy).sqrt();
                if dist < radius + pr + 10.0 {
                    overlap = true;
                    break;
                }
            }
            if !overlap {
                best_pos = (x, y);
                found = true;
            } else {
                theta += step;
                r_spiral += 0.02 * radius;
            }
        }
        
        placed_circles.push((best_pos.0, best_pos.1, radius));
        final_large_centers[comp_idx] = (best_pos.0, best_pos.1);
    }
    
    // Apply shift to large component nodes
    for (comp_idx, comp) in large_components.iter().enumerate() {
        let (tx, ty) = final_large_centers[comp_idx];
        for &u in comp {
            pos[u].x += tx;
            pos[u].y += ty;
        }
    }
    
    // Bounding box of packed large components
    let mut _x_min = -10.0;
    let mut x_max = 10.0;
    let mut y_min = -10.0;
    let mut y_max = 10.0;
    
    if !placed_circles.is_empty() {
        _x_min = placed_circles.iter().map(|c| c.0 - c.2).fold(f64::INFINITY, f64::min);
        x_max = placed_circles.iter().map(|c| c.0 + c.2).fold(f64::NEG_INFINITY, f64::max);
        y_min = placed_circles.iter().map(|c| c.1 - c.2).fold(f64::INFINITY, f64::min);
        y_max = placed_circles.iter().map(|c| c.1 + c.2).fold(f64::NEG_INFINITY, f64::max);
    }
    
    // Place small components on a square grid next to the large components
    if !small_components.is_empty() {
        let mut cur_x = x_max + 20.0;
        let cur_y_start = y_min;
        let grid_height = (y_max - y_min).max(15.0 * (small_components.len() as f64).sqrt());
        let mut cur_y = cur_y_start;
        
        for comp in &small_components {
            let n_c = comp.len();
            if n_c == 1 {
                let u = comp[0];
                pos[u] = Point { x: cur_x, y: cur_y };
            } else if n_c == 2 {
                let u0 = comp[0];
                let u1 = comp[1];
                pos[u0] = Point { x: cur_x - 3.0, y: cur_y };
                pos[u1] = Point { x: cur_x + 3.0, y: cur_y };
            } else if n_c == 3 {
                let u0 = comp[0];
                let u1 = comp[1];
                let u2 = comp[2];
                pos[u0] = Point { x: cur_x, y: cur_y + 4.0 };
                pos[u1] = Point { x: cur_x - 3.5, y: cur_y - 2.0 };
                pos[u2] = Point { x: cur_x + 3.5, y: cur_y - 2.0 };
            }
            
            cur_y += 15.0;
            if cur_y > cur_y_start + grid_height {
                cur_y = cur_y_start;
                cur_x += 15.0;
            }
        }
    }
    
    // Fast output writing
    let stdout = io::stdout();
    let mut writer = BufWriter::new(stdout.lock());
    for i in 0..n {
        writeln!(writer, "{:.6} {:.6}", pos[i].x, pos[i].y).unwrap();
    }
}
