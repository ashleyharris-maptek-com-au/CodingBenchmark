const EPS: f64 = 1e-9;
const BB_EPS: f64 = 1e-9;
const DUP_EPS: f64 = 1e-5;

#[derive(Clone, Copy, Debug)]
struct Point {
    x: f64,
    y: f64,
}

struct Piece {
    id: usize,
    vertices: Vec<Point>,
    area: f64,
    internal_point: Point,
}

    piece_id: usize,
    vertices: Vec<Point>,
    bb: (Point, Point),
    internal_point: Point,
}

    rotation: f64,
    translation: Point,
    vertices: Vec<Point>,
    bb: (Point, Point),
    internal_point: Point,
}

#[derive(Clone)]
struct PlacedPieceSolution {
    piece_id: usize,
    stock_index: usize,
    translation: Point,
    rotation: f64,
}

    num_stocks: usize,
    solutions: Vec<PlacedPieceSolution>,
}

    Large,
}

struct Rng {
    state: u64,
}

impl Rng {
    fn new(seed: u64) -> Self {
        Self {
            state: if seed == 0 { 1 } else { seed },
        }
    }

    fn next_u64(&mut self) -> u64 {
        let mut x = self.state;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.state = x;
        x
    }

    fn next_f64(&mut self) -> f64 {
        (self.next_u64() & 0xffffffffffffff) as f64 / 9007199254740992.0
    }
}

fn ccw(a: Point, b: Point, c: Point) -> f64 {
    (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x)
}

fn segments_cross_properly(a: Point, b: Point, c: Point, d: Point) -> bool {
    let cp1 = ccw(a, b, c);
    let cp2 = ccw(a, b, d);
    let cp3 = ccw(c, d, a);
    let cp4 = ccw(c, d, b);

    let cross_ab = (cp1 > EPS && cp2 < -EPS) || (cp1 < -EPS && cp2 > EPS);
    let cross_cd = (cp3 > EPS && cp4 < -EPS) || (cp3 < -EPS && cp4 > EPS);

    cross_ab && cross_cd
}

fn point_on_segment(p: Point, a: Point, b: Point) -> bool {
    let cross = (p.y - a.y) * (b.x - a.x) - (p.x - a.x) * (b.y - a.y);
    if cross.abs() > EPS {
        return false;
    }
    p.x >= a.x.min(b.x) - EPS
        && p.x <= a.x.max(b.x) + EPS
        && p.y >= a.y.min(b.y) - EPS
        && p.y <= a.y.max(b.y) + EPS
}

fn point_on_polygon_boundary(p: Point, poly: &[Point]) -> bool {
    let n = poly.len();
    for i in 0..n {
        let next = (i + 1) % n;
        if point_on_segment(p, poly[i], poly[next]) {
            return true;
        }
    }
    false
}

fn point_in_polygon(p: Point, poly: &[Point]) -> bool {
    let mut inside = false;
    let n = poly.len();
    let mut j = n - 1;
    for i in 0..n {
        let vi = poly[i];
        let vj = poly[j];
        if (vi.y > p.y) != (vj.y > p.y) {
            let dy = vj.y - vi.y;
            let val1 = (p.x - vi.x) * dy;
            let val2 = (vj.x - vi.x) * (p.y - vi.y);
            if dy > 0.0 {
                if val1 < val2 {
                    inside = !inside;
                }
            } else if val1 > val2 {
                inside = !inside;
            }
        }
        j = i;
    }
    inside
}

fn is_point_strictly_inside(p: Point, poly: &[Point]) -> bool {
    if point_on_polygon_boundary(p, poly) {
        return false;
    }
    point_in_polygon(p, poly)
}

fn polygon_area(poly: &[Point]) -> f64 {
    let n = poly.len();
    let mut area = 0.0;
    let mut j = n - 1;
    for i in 0..n {
        area += (poly[j].x + poly[i].x) * (poly[j].y - poly[i].y);
        j = i;
    }
    (area / 2.0).abs()
}

fn get_bounding_box(poly: &[Point]) -> (Point, Point) {
    let mut min_x = f64::INFINITY;
    let mut min_y = f64::INFINITY;
    let mut max_x = f64::NEG_INFINITY;
    let mut max_y = f64::NEG_INFINITY;
    for &p in poly {
        if p.x < min_x {
            min_x = p.x;
        }
        if p.x > max_x {
            max_x = p.x;
        }
        if p.y < min_y {
            min_y = p.y;
        }
        if p.y > max_y {
            max_y = p.y;
        }
    }
    (
        Point { x: min_x, y: min_y },
        Point { x: max_x, y: max_y },
    )
}

fn find_internal_point(poly: &[Point]) -> Point {
    let n = poly.len();
    if n == 3 {
        return Point {
            x: (poly[0].x + poly[1].x + poly[2].x) / 3.0,
            y: (poly[0].y + poly[1].y + poly[2].y) / 3.0,
        };
    }
    for i in 0..n {
        let prev = poly[(i + n - 1) % n];
        let curr = poly[i];
        let next = poly[(i + 1) % n];
        let c = Point {
            x: (prev.x + curr.x + next.x) / 3.0,
            y: (prev.y + curr.y + next.y) / 3.0,
        };
        if is_point_strictly_inside(c, poly) {
            return c;
        }
    }
    let mut sx = 0.0;
    let mut sy = 0.0;
    for p in poly {
        sx += p.x;
        sy += p.y;
    }
    Point {
        x: sx / n as f64,
        y: sy / n as f64,
    }
}

fn rotate_point(p: Point, theta: f64) -> Point {
    let cos = theta.cos();
    let sin = theta.sin();
    Point {
        x: p.x * cos - p.y * sin,
        y: p.x * sin + p.y * cos,
    }
}

fn is_polygon_inside_stock(
    poly: &[Point],
    poly_bb: (Point, Point),
    poly_internal: Point,
    stock: &[Point],
    stock_bb: (Point, Point),
) -> bool {
    if poly_bb.0.x < stock_bb.0.x - BB_EPS
        || poly_bb.1.x > stock_bb.1.x + BB_EPS
        || poly_bb.0.y < stock_bb.0.y - BB_EPS
        || poly_bb.1.y > stock_bb.1.y + BB_EPS
    {
        return false;
    }

    if !point_in_polygon(poly_internal, stock) && !point_on_polygon_boundary(poly_internal, stock) {
        return false;
    }

    let n_p = poly.len();
    for i in 0..n_p {
        let p = poly[i];
        if !point_in_polygon(p, stock) && !point_on_polygon_boundary(p, stock) {
            return false;
        }

        let next_i = (i + 1) % n_p;
        let mid = Point {
            x: (p.x + poly[next_i].x) * 0.5,
            y: (p.y + poly[next_i].y) * 0.5,
        };
        if !point_in_polygon(mid, stock) && !point_on_polygon_boundary(mid, stock) {
            return false;
        }
    }

    let n_s = stock.len();
    for i in 0..n_p {
        let next_i = (i + 1) % n_p;
        let p1 = poly[i];
        let p2 = poly[next_i];
        for j in 0..n_s {
            let next_j = (j + 1) % n_s;
            let s1 = stock[j];
            let s2 = stock[next_j];
            if segments_cross_properly(p1, p2, s1, s2) {
                return false;
            }
        }
    }

    true
}

fn polygons_overlap(
    p_vertices: &[Point],
    p_bb: (Point, Point),
    p_internal: Point,
    q_vertices: &[Point],
    q_bb: (Point, Point),
    q_internal: Point,
) -> bool {
    if p_bb.1.x < q_bb.0.x - BB_EPS
        || p_bb.0.x > q_bb.1.x + BB_EPS
        || p_bb.1.y < q_bb.0.y - BB_EPS
        || p_bb.0.y > q_bb.1.y + BB_EPS
    {
        return false;
    }

    let n_p = p_vertices.len();
    let n_q = q_vertices.len();
    for i in 0..n_p {
        let next_i = (i + 1) % n_p;
        let p1 = p_vertices[i];
        let p2 = p_vertices[next_i];
        for j in 0..n_q {
            let next_j = (j + 1) % n_q;
            let q1 = q_vertices[j];
            let q2 = q_vertices[next_j];
            if segments_cross_properly(p1, p2, q1, q2) {
                return true;
            }
        }
    }

    if is_point_strictly_inside(p_internal, q_vertices) {
        return true;
    }
    if is_point_strictly_inside(q_internal, p_vertices) {
        return true;
    }

    for i in 0..n_p {
        let next_i = (i + 1) % n_p;
        let mid = Point {
            x: (p_vertices[i].x + p_vertices[next_i].x) * 0.5,
            y: (p_vertices[i].y + p_vertices[next_i].y) * 0.5,
        };
        if is_point_strictly_inside(mid, q_vertices) {
            return true;
        }
    }

    for j in 0..n_q {
        let next_j = (j + 1) % n_q;
        let mid = Point {
            x: (q_vertices[j].x + q_vertices[next_j].x) * 0.5,
            y: (q_vertices[j].y + q_vertices[next_j].y) * 0.5,
        };
        if is_point_strictly_inside(mid, p_vertices) {
            return true;
        }
    }

    false
}

#[derive(Clone)]
struct SpatialGrid {
    inv_cell_width: f64,
    inv_cell_height: f64,
    cols: usize,
    rows: usize,
    min_x: f64,
    min_y: f64,
    cells: Vec<Vec<usize>>,
}

impl SpatialGrid {
    fn new(min_x: f64, min_y: f64, max_x: f64, max_y: f64, cols: usize, rows: usize) -> Self {
        let cell_width = (max_x - min_x) / cols as f64;
        let cell_height = (max_y - min_y) / rows as f64;
        Self {
            inv_cell_width: 1.0 / cell_width,
            inv_cell_height: 1.0 / cell_height,
            cols,
            rows,
            min_x,
            min_y,
            cells: vec![Vec::new(); cols * rows],
        }
    }

    fn get_cell_range(
        &self,
        min_x: f64,
        min_y: f64,
        max_x: f64,
        max_y: f64,
    ) -> (usize, usize, usize, usize) {
        let c_min = ((min_x - self.min_x) * self.inv_cell_width)
            .floor()
            .max(0.0) as usize;
        let c_max = ((max_x - self.min_x) * self.inv_cell_width)
            .floor()
            .max(0.0) as usize;
        let r_min = ((min_y - self.min_y) * self.inv_cell_height)
            .floor()
            .max(0.0) as usize;
        let r_max = ((max_y - self.min_y) * self.inv_cell_height)
            .floor()
            .max(0.0) as usize;

        (
            c_min.min(self.cols - 1),
            c_max.min(self.cols - 1),
            r_min.min(self.rows - 1),
            r_max.min(self.rows - 1),
        )
    }

    fn insert(&mut self, idx: usize, min_x: f64, min_y: f64, max_x: f64, max_y: f64) {
        let (c_min, c_max, r_min, r_max) = self.get_cell_range(min_x, min_y, max_x, max_y);
        for r in r_min..=r_max {
            for c in c_min..=c_max {
                self.cells[r * self.cols + c].push(idx);
            }
        }
    }

    fn query(&self, min_x: f64, min_y: f64, max_x: f64, max_y: f64, candidates: &mut Vec<usize>) {
        candidates.clear();
        let (c_min, c_max, r_min, r_max) = self.get_cell_range(min_x, min_y, max_x, max_y);
        for r in r_min..=r_max {
            for c in c_min..=c_max {
                candidates.extend_from_slice(&self.cells[r * self.cols + c]);
            }
        }
        candidates.sort_unstable();
        candidates.dedup();
    }
}

#[derive(Clone)]
struct StockState {
    placed: Vec<PlacedPiece>,
    grid: SpatialGrid,
    used_area: f64,
}

fn generate_candidates(
    p_rotated: &[Point],
    stock: &[Point],
    stock_bb: (Point, Point),
    placed_pieces: &[PlacedPiece],
    k1: usize,
    k2: usize,
    candidates: &mut Vec<Point>,
) {
    candidates.clear();

    let p_bb = get_bounding_box(p_rotated);

    candidates.push(Point {
        x: stock_bb.0.x - p_bb.0.x,
        y: stock_bb.0.y - p_bb.0.y,
    });

    for &vs in stock {
        for &vp in p_rotated {
            candidates.push(Point {
                x: vs.x - vp.x,
                y: vs.y - vp.y,
            });
        }
    }

    if !placed_pieces.is_empty() {
        let mut selected = Vec::with_capacity(k1 + k2);
        let start = placed_pieces.len().saturating_sub(k1);
        for i in start..placed_pieces.len() {
            selected.push(i);
        }

        if placed_pieces.len() > k1 {
            let mut top: Vec<(usize, f64)> = Vec::with_capacity(k2 + 1);
            for (idx, p) in placed_pieces.iter().enumerate() {
                let val = p.bb.1.y;
                let pos = match top.binary_search_by(|probe| val.partial_cmp(&probe.1).unwrap()) {
                    Ok(p) => p,
                    Err(p) => p,
                };
                if pos < k2 {
                    top.insert(pos, (idx, val));
                    if top.len() > k2 {
                        top.pop();
                    }
                }
            }
            for &(idx, _) in &top {
                selected.push(idx);
            }
        }
        selected.sort_unstable();
        selected.dedup();

        for &idx in &selected {
            let q = &placed_pieces[idx];
            for &vq in &q.vertices {
                for &vp in p_rotated {
                    candidates.push(Point {
                        x: vq.x - vp.x,
                        y: vq.y - vp.y,
                    });
                }
            }
        }
    }

    candidates.retain(|&t| {
        let tx = t.x;
        let ty = t.y;
        let px1 = p_bb.0.x + tx;
        let py1 = p_bb.0.y + ty;
        let px2 = p_bb.1.x + tx;
        let py2 = p_bb.1.y + ty;

        px1 >= stock_bb.0.x - BB_EPS
            && px2 <= stock_bb.1.x + BB_EPS
            && py1 >= stock_bb.0.y - BB_EPS
            && py2 <= stock_bb.1.y + BB_EPS
    });

    candidates.sort_unstable_by(|a, b| match a.y.partial_cmp(&b.y).unwrap() {
        Ordering::Equal => a.x.partial_cmp(&b.x).unwrap(),
        other => other,
    });

    candidates.dedup_by(|a, b| (a.x - b.x).abs() < DUP_EPS && (a.y - b.y).abs() < DUP_EPS);
}

fn is_valid_placement(
    p_placed_vertices: &[Point],
    p_placed_bb: (Point, Point),
    p_placed_internal: Point,
    stock: &[Point],
    stock_bb: (Point, Point),
    stock_state: &StockState,
    candidates_buf: &mut Vec<usize>,
) -> bool {
    if !is_polygon_inside_stock(p_placed_vertices, p_placed_bb, p_placed_internal, stock, stock_bb) {
        return false;
    }

    stock_state
        .grid
        .query(p_placed_bb.0.x, p_placed_bb.0.y, p_placed_bb.1.x, p_placed_bb.1.y, candidates_buf);

    for &idx in candidates_buf.iter() {
        let q = &stock_state.placed[idx];
        if polygons_overlap(
            p_placed_vertices,
            p_placed_bb,
            p_placed_internal,
            &q.vertices,
            q.bb,
            q.internal_point,
        ) {
            return false;
        }
    }

    true
}

fn find_best_placement(
    piece: &Piece,
    stock: &[Point],
    stock_bb: (Point, Point),
    stock_state: &StockState,
    rotations_to_try: &[f64],
    k1: usize,
    k2: usize,
    candidates_buf: &mut Vec<usize>,
    candidates_pts_buf: &mut Vec<Point>,
    p_rotated_buf: &mut Vec<Point>,
    p_placed_buf: &mut Vec<Point>,
) -> Option<Placement> {
    let mut best_placement: Option<Placement> = None;
    let mut best_score = f64::INFINITY;

    for &theta in rotations_to_try {
        p_rotated_buf.clear();
        for &p in &piece.vertices {
            p_rotated_buf.push(rotate_point(p, theta));
        }
        let p_bb = get_bounding_box(p_rotated_buf);
        let p_internal = rotate_point(piece.internal_point, theta);

        generate_candidates(
            p_rotated_buf,
            stock,
            stock_bb,
            &stock_state.placed,
            k1,
            k2,
            candidates_pts_buf,
        );

        for i in 0..candidates_pts_buf.len() {
            let t = candidates_pts_buf[i];
            let score = p_bb.0.y + t.y;
            if score >= best_score {
                break;
            }

            p_placed_buf.clear();
            for &p in p_rotated_buf.iter() {
                p_placed_buf.push(Point {
                    x: p.x + t.x,
                    y: p.y + t.y,
                });
            }
            let p_placed_bb = (
                Point {
                    x: p_bb.0.x + t.x,
                    y: p_bb.0.y + t.y,
                },
                Point {
                    x: p_bb.1.x + t.x,
                    y: p_bb.1.y + t.y,
                },
            );
            let p_placed_internal = Point {
                x: p_internal.x + t.x,
                y: p_internal.y + t.y,
            };

            if is_valid_placement(
                p_placed_buf,
                p_placed_bb,
                p_placed_internal,
                stock,
                stock_bb,
                stock_state,
                candidates_buf,
            ) {
                best_score = score;
                best_placement = Some(Placement {
                    rotation: theta,
                    translation: t,
                    vertices: p_placed_buf.clone(),
                    bb: p_placed_bb,
                    internal_point: p_placed_internal,
                });
                break;
            }
        }
    }

    best_placement
}

fn run_packing_trial(
    pieces: &[Piece],
    stock: &[Point],
    stock_bb: (Point, Point),
    stock_area: f64,
    rotations_to_try: &[f64],
    rng: &mut Rng,
    perturbation: f64,
    k1: usize,
    k2: usize,
) -> Vec<PlacedPieceSolution> {
    let mut ordered_pieces: Vec<&Piece> = pieces.iter().collect();
    if perturbation > 0.0 {
        ordered_pieces.sort_by(|a, b| {
            let p_a = a.area * (1.0 + perturbation * (2.0 * rng.next_f64() - 1.0));
            let p_b = b.area * (1.0 + perturbation * (2.0 * rng.next_f64() - 1.0));
            p_b.partial_cmp(&p_a).unwrap()
        });
    } else {
        ordered_pieces.sort_by(|a, b| b.area.partial_cmp(&a.area).unwrap());
    }

    let mut stocks: Vec<StockState> = Vec::new();
    let mut solutions = Vec::with_capacity(pieces.len());

    let mut candidates_buf = Vec::new();
    let mut candidates_pts_buf = Vec::new();
    let mut p_rotated_buf = Vec::new();
    let mut p_placed_buf = Vec::new();

    let grid_size = (pieces.len() as f64).sqrt().round() as usize;
    let grid_size = grid_size.clamp(10, 80);

    for piece in ordered_pieces {
        let mut placed = false;

        for (stock_idx, stock_state) in stocks.iter_mut().enumerate() {
            if stock_area - stock_state.used_area < piece.area {
                continue;
            }

            if let Some(placement) = find_best_placement(
                piece,
                stock,
                stock_bb,
                stock_state,
                rotations_to_try,
                k1,
                k2,
                &mut candidates_buf,
                &mut candidates_pts_buf,
                &mut p_rotated_buf,
                &mut p_placed_buf,
            ) {
                let next_idx = stock_state.placed.len();
                stock_state.grid.insert(
                    next_idx,
                    placement.bb.0.x,
                    placement.bb.0.y,
                    placement.bb.1.x,
                    placement.bb.1.y,
                );

                stock_state.used_area += piece.area;

                let placed_piece = PlacedPiece {
                    piece_id: piece.id,
                    vertices: placement.vertices,
                    bb: placement.bb,
                    internal_point: placement.internal_point,
                };
                stock_state.placed.push(placed_piece);

                solutions.push(PlacedPieceSolution {
                    piece_id: piece.id,
                    stock_index: stock_idx,
                    translation: placement.translation,
                    rotation: placement.rotation,
                });

                placed = true;
                break;
            }
        }

        if !placed {
            let mut new_stock_state = StockState {
                placed: Vec::new(),
                grid: SpatialGrid::new(
                    stock_bb.0.x,
                    stock_bb.0.y,
                    stock_bb.1.x,
                    stock_bb.1.y,
                    grid_size,
                    grid_size,
                ),
                used_area: 0.0,
            };

            if let Some(placement) = find_best_placement(
                piece,
                stock,
                stock_bb,
                &new_stock_state,
                rotations_to_try,
                k1,
                k2,
                &mut candidates_buf,
                &mut candidates_pts_buf,
                &mut p_rotated_buf,
                &mut p_placed_buf,
            ) {
                let next_idx = new_stock_state.placed.len();
                new_stock_state.grid.insert(
                    next_idx,
                    placement.bb.0.x,
                    placement.bb.0.y,
                    placement.bb.1.x,
                    placement.bb.1.y,
                );

                new_stock_state.used_area += piece.area;

                let stock_idx = stocks.len();

                let placed_piece = PlacedPiece {
                    piece_id: piece.id,
                    vertices: placement.vertices,
                    bb: placement.bb,
                    internal_point: placement.internal_point,
                };
                new_stock_state.placed.push(placed_piece);
                stocks.push(new_stock_state);

                solutions.push(PlacedPieceSolution {
                    piece_id: piece.id,
                    stock_index: stock_idx,
                    translation: placement.translation,
                    rotation: placement.rotation,
                });
            } else {
                let stock_idx = stocks.len();
                solutions.push(PlacedPieceSolution {
                    piece_id: piece.id,
                    stock_index: stock_idx,
                    translation: Point { x: 0.0, y: 0.0 },
                    rotation: 0.0,
                });

                let mut dummy_stock = StockState {
                    placed: Vec::new(),
                    grid: SpatialGrid::new(
                        stock_bb.0.x,
                        stock_bb.0.y,
                        stock_bb.1.x,
                        stock_bb.1.y,
                        grid_size,
                        grid_size,
                    ),
                    used_area: piece.area,
                };
                dummy_stock.placed.push(PlacedPiece {
                    piece_id: piece.id,
                    vertices: piece.vertices.clone(),
                    bb: get_bounding_box(&piece.vertices),
                    internal_point: piece.internal_point,
                });
                stocks.push(dummy_stock);
            }
        }
    }

    solutions
}

fn get_scale(np: usize) -> Scale {
    if np <= 50 {
        Scale::Small
    } else if np <= 1000 {
        Scale::Medium
    } else {
        Scale::Large
    }
}

fn get_k_params(scale: &Scale) -> (usize, usize) {
    match scale {
        Scale::Small => (30, 30),
        Scale::Medium => (15, 15),
        Scale::Large => (5, 5),
    }
}

fn get_rotations(thread_idx: usize, num_threads: usize, scale: &Scale) -> Vec<f64> {
    let num_rotations = match scale {
        Scale::Small => 16,
        Scale::Medium => 8,
        Scale::Large => 4,
    };

    let mut rotations = Vec::with_capacity(num_rotations);

    // Always include critical orthogonal rotations
    rotations.push(0.0);
    if num_rotations > 1 {
        rotations.push(PI * 0.5);
    }
    if num_rotations > 2 {
        rotations.push(PI);
    }
    if num_rotations > 3 {
        rotations.push(PI * 1.5);
    }

    let start_idx = rotations.len();
    if start_idx < num_rotations {
        let remaining = num_rotations - start_idx;
        let shift = (thread_idx as f64) * (2.0 * PI) / (remaining as f64 * num_threads as f64);
        for i in 0..remaining {
            let angle = shift + (i as f64) * (2.0 * PI) / (remaining as f64);
            rotations.push(angle % (2.0 * PI));
        }
    }

    rotations
}

fn parse_input_robust() -> io::Result<(Vec<Point>, Vec<Piece>)> {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines().map(|l| l.unwrap());

    let mut next_line = || {
        while let Some(line) = lines.next() {
            let trimmed = line.trim();
            if !trimmed.is_empty() {
                return Some(trimmed.to_string());
            }
        }
        None
    };

    let first_line = match next_line() {
        Some(l) => l,
        None => return Err(io::Error::new(io::ErrorKind::InvalidInput, "Empty input")),
    };

    let mut parts = first_line.split_whitespace();
    let sv: usize = parts
        .next()
        .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidInput, "Invalid first line"))?
        .parse()
        .map_err(|e| io::Error::new(io::ErrorKind::InvalidInput, e))?;
    let np: usize = parts
        .next()
        .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidInput, "Invalid first line"))?
        .parse()
        .map_err(|e| io::Error::new(io::ErrorKind::InvalidInput, e))?;

    let mut stock = Vec::with_capacity(sv);
    for _ in 0..sv {
        let line = next_line().ok_or_else(|| {
            io::Error::new(io::ErrorKind::InvalidInput, "Missing stock vertex")
        })?;
        let mut p_parts = line.split_whitespace();
        let x: f64 = p_parts
            .next()
            .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidInput, "Missing x"))?
            .parse()
            .map_err(|e| io::Error::new(io::ErrorKind::InvalidInput, e))?;
        let y: f64 = p_parts
            .next()
            .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidInput, "Missing y"))?
            .parse()
            .map_err(|e| io::Error::new(io::ErrorKind::InvalidInput, e))?;
        stock.push(Point { x, y });
    }

    let mut pieces = Vec::with_capacity(np);
    for i in 0..np {
        let pv_line = next_line().ok_or_else(|| {
            io::Error::new(io::ErrorKind::InvalidInput, "Missing piece vertex count")
        })?;
        let pv: usize = pv_line
            .parse()
            .map_err(|e| io::Error::new(io::ErrorKind::InvalidInput, e))?;
        let mut vertices = Vec::with_capacity(pv);
        for _ in 0..pv {
            let line = next_line().ok_or_else(|| {
                io::Error::new(io::ErrorKind::InvalidInput, "Missing piece vertex")
            })?;
            let mut p_parts = line.split_whitespace();
            let x: f64 = p_parts
                .next()
                .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidInput, "Missing x"))?
                .parse()
                .map_err(|e| io::Error::new(io::ErrorKind::InvalidInput, e))?;
            let y: f64 = p_parts
                .next()
                .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidInput, "Missing y"))?
                .parse()
                .map_err(|e| io::Error::new(io::ErrorKind::InvalidInput, e))?;
            vertices.push(Point { x, y });
        }

        let area = polygon_area(&vertices);
        let internal_point = find_internal_point(&vertices);

        pieces.push(Piece {
            id: i,
            vertices,
            area,
            internal_point,
        });
    }

    Ok((stock, pieces))
}

fn main() -> io::Result<()> {
    let start_time = Instant::now();

    let (stock, pieces) = parse_input_robust()?;
    let np = pieces.len();

    let stock_bb = get_bounding_box(&stock);
    let stock_area = polygon_area(&stock);
    let scale = get_scale(np);

    let total_piece_area: f64 = pieces.iter().map(|p| p.area).sum();
    let lower_bound = ((total_piece_area / stock_area - 1e-9).ceil() as usize).max(1);

    let num_threads = thread::available_parallelism()
        .map(|p| p.get())
        .unwrap_or(64);

    let pieces = Arc::new(pieces);
    let stock = Arc::new(stock);

    let best_solution = Arc::new(Mutex::new(BestSolution {
        num_stocks: usize::MAX,
        solutions: Vec::new(),
    }));

    let mut handles = Vec::with_capacity(num_threads);

    for thread_idx in 0..num_threads {
        let pieces = Arc::clone(&pieces);
        let stock = Arc::clone(&stock);
        let best_solution = Arc::clone(&best_solution);
        let scale = scale.clone();

        let seed = (thread_idx as u64 + 1) * 123456789;

        let handle = thread::spawn(move || {
            let mut rng = Rng::new(seed);
            let (k1, k2) = get_k_params(&scale);

            let perturbation = if thread_idx == 0 {
                0.0
            } else {
                0.01 + rng.next_f64() * 0.19
            };

            let rotations_to_try = get_rotations(thread_idx, num_threads, &scale);

            let max_iterations = match scale {
                Scale::Small => 200,
                Scale::Medium => 50,
                Scale::Large => usize::MAX,
            };

            let mut iteration = 0;

            loop {
                if iteration >= max_iterations {
                    break;
                }

                if start_time.elapsed().as_secs_f64() > 285.0 {
                    break;
                }

                {
                    let best = best_solution.lock().unwrap();
                    if best.num_stocks <= lower_bound {
                        break;
                    }
                }

                let current_rotations = if iteration == 0 {
                    rotations_to_try.clone()
                } else {
                    let num_rotations = match scale {
                        Scale::Small => 16,
                        Scale::Medium => 8,
                        Scale::Large => 4,
                    };
                    let mut rots = Vec::with_capacity(num_rotations);
                    rots.push(0.0);
                    if num_rotations > 1 {
                        rots.push(PI * 0.5);
                    }
                    if num_rotations > 2 {
                        rots.push(PI);
                    }
                    if num_rotations > 3 {
                        rots.push(PI * 1.5);
                    }
                    while rots.len() < num_rotations {
                        rots.push(rng.next_f64() * 2.0 * PI);
                    }
                    rots
                };

                let current_perturbation = if iteration == 0 {
                    perturbation
                } else {
                    0.01 + rng.next_f64() * 0.19
                };

                let solutions = run_packing_trial(
                    &pieces,
                    &stock,
                    stock_bb,
                    stock_area,
                    &current_rotations,
                    &mut rng,
                    current_perturbation,
                    k1,
                    k2,
                );

                let max_stock_idx = solutions.iter().map(|s| s.stock_index).max().unwrap_or(0);
                let num_stocks = max_stock_idx + 1;

                {
                    let mut best = best_solution.lock().unwrap();
                    if num_stocks < best.num_stocks {
                        best.num_stocks = num_stocks;
                        best.solutions = solutions;
                    }
                }

                iteration += 1;
            }
        });

        handles.push(handle);
    }

    for handle in handles {
        let _ = handle.join();
    }

    let mut best = best_solution.lock().unwrap();
    
    // Sort solutions back to original piece order
    best.solutions.sort_unstable_by_key(|s| s.piece_id);

    println!("{}", best.num_stocks);
    for sol in &best.solutions {
        println!(
            "{} {:.9} {:.9} {:.9}",
            sol.stock_index, sol.translation.x, sol.translation.y, sol.rotation
        );
    }

    Ok(())
}
