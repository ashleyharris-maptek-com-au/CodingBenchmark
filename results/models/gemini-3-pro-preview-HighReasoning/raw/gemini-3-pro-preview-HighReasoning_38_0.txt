import sys

# Increase recursion depth for safety on large instances
sys.setrecursionlimit(20000)

def main():
    # Read all lines from standard input
    input_data = sys.stdin.read().splitlines()
    
    # Find the line containing U and S
    line_idx = 0
    U, S = 0, 0
    while line_idx < len(input_data):
        line = input_data[line_idx].strip()
        if line:
            parts = line.split()
            if len(parts) >= 2:
                try:
                    U = int(parts[0])
                    S = int(parts[1])
                    line_idx += 1
                    break
                except ValueError:
                    pass
        line_idx += 1
    else:
        # No valid header found
        print("NO SOLUTION")
        return

    sets = []
    # Read exactly S lines for the sets
    for _ in range(S):
        if line_idx < len(input_data):
            line = input_data[line_idx].strip()
            if line:
                try:
                    sets.append(list(map(int, line.split())))
                except ValueError:
                    sets.append([])
            else:
                sets.append([])
            line_idx += 1
        else:
            sets.append([])

    # sets_containing[e] is the list of set indices that contain element e
    sets_containing = [[] for _ in range(U)]
    cleaned_sets = []
    for i, s in enumerate(sets):
        # Clean set to contain unique elements within range [0, U-1]
        s_clean = list({e for e in s if 0 <= e < U})
        cleaned_sets.append(s_clean)
        for e in s_clean:
            sets_containing[e].append(i)

    eliminated = [False] * S
    covered = [False] * U
    solution = []

    def backtrack(cov_count):
        if cov_count == U:
            return True

        best_e = -1
        min_active = S + 1

        for e in range(U):
            if not covered[e]:
                cnt = 0
                for s_idx in sets_containing[e]:
                    if not eliminated[s_idx]:
                        cnt += 1
                if cnt == 0:
                    return False
                if cnt < min_active:
                    min_active = cnt
                    best_e = e
                    if cnt == 1:
                        break

        if best_e == -1:
            return False

        # Get active sets for best_e
        best_active_sets = [s_idx for s_idx in sets_containing[best_e] if not eliminated[s_idx]]

        for s_idx in best_active_sets:
            solution.append(s_idx)
            newly_eliminated = []
            newly_covered = []

            for e in cleaned_sets[s_idx]:
                if not covered[e]:
                    covered[e] = True
                    newly_covered.append(e)

                for other_s in sets_containing[e]:
                    if not eliminated[other_s]:
                        eliminated[other_s] = True
                        newly_eliminated.append(other_s)

            if backtrack(cov_count + len(newly_covered)):
                return True

            for other_s in newly_eliminated:
                eliminated[other_s] = False
            for e in newly_covered:
                covered[e] = False
            solution.pop()

        return False

    if backtrack(0):
        print("SOLUTION")
        print(" ".join(map(str, sorted(solution))))
    else:
        print("NO SOLUTION")

if __name__ == '__main__':
    main()
