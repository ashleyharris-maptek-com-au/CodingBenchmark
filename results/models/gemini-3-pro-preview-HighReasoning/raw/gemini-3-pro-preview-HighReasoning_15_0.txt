use std::f64;
use std::io::{self, BufRead};

#[derive(Debug, Clone, Copy)]
struct Point2D {
    x: f64,
    y: f64,
}

struct EdgeIndex {
    min_u: f64,
    min_v: f64,
    cell_w: f64,
    cell_h: f64,
    rows: usize,
    cols: usize,
    buckets: Vec<Vec<usize>>,
}

impl EdgeIndex {
    fn new(poly: &[Point2D], rows: usize, cols: usize) -> Self {
        let mut min_u = f64::INFINITY;
        let mut min_v = f64::INFINITY;
        let mut max_u = f64::NEG_INFINITY;
        let mut max_v = f64::NEG_INFINITY;
        for p in poly {
            if p.x < min_u { min_u = p.x; }
            if p.y < min_v { min_v = p.y; }
            if p.x > max_u { max_u = p.x; }
            if p.y > max_v { max_v = p.y; }
        }
        let du = max_u - min_u;
        let dv = max_v - min_v;
        let cell_w = if du > 0.0 { du / (cols as f64) } else { 1.0 };
        let cell_h = if dv > 0.0 { dv / (rows as f64) } else { 1.0 };

        let mut buckets = vec![Vec::new(); rows * cols];
        let n = poly.len();
        for i in 0..n {
            let p1 = poly[i];
            let p2 = poly[(i + 1) % n];

            let b_min_u = p1.x.min(p2.x);
            let b_max_u = p1.x.max(p2.x);
            let b_min_v = p1.y.min(p2.y);
            let b_max_v = p1.y.max(p2.y);

            let c1 = (((b_min_u - min_u) / cell_w).floor() as usize).min(cols - 1);
            let c2 = (((b_max_u - min_u) / cell_w).floor() as usize).min(cols - 1);
            let r1 = (((b_min_v - min_v) / cell_h).floor() as usize).min(rows - 1);
            let r2 = (((b_max_v - min_v) / cell_h).floor() as usize).min(rows - 1);

            for r in r1..=r2 {
                for c in c1..=c2 {
                    buckets[r * cols + c].push(i);
                }
            }
        }

        EdgeIndex {
            min_u,
            min_v,
            cell_w,
            cell_h,
            rows,
            cols,
            buckets,
        }
    }

    fn get_overlapping_edges(&self, rect_min: Point2D, rect_max: Point2D, visitor: &mut Vec<usize>) {
        visitor.clear();
        if rect_max.x < self.min_u || rect_min.x > self.min_u + (self.cols as f64) * self.cell_w ||
           rect_max.y < self.min_v || rect_min.y > self.min_v + (self.rows as f64) * self.cell_h {
            return;
        }

        let c1 = (((rect_min.x - self.min_u) / self.cell_w).floor() as isize).max(0) as usize;
        let c2 = (((rect_max.x - self.min_u) / self.cell_w).floor() as isize).min(self.cols as isize - 1) as usize;
        let r1 = (((rect_min.y - self.min_v) / self.cell_h).floor() as isize).max(0) as usize;
        let r2 = (((rect_max.y - self.min_v) / self.cell_h).floor() as isize).min(self.rows as isize - 1) as usize;

        for r in r1..=r2 {
            for c in c1..=c2 {
                visitor.extend_from_slice(&self.buckets[r * self.cols + c]);
            }
        }
        visitor.sort_unstable();
        visitor.dedup();
    }
}

fn ccw(a: Point2D, b: Point2D, c: Point2D) -> f64 {
    (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x)
}

fn segments_intersect(a1: Point2D, a2: Point2D, b1: Point2D, b2: Point2D) -> bool {
    let cp1 = ccw(a1, a2, b1);
    let cp2 = ccw(a1, a2, b2);
    let cp3 = ccw(b1, b2, a1);
    let cp4 = ccw(b1, b2, a2);

    if ((cp1 > 1e-9 && cp2 < -1e-9) || (cp1 < -1e-9 && cp2 > 1e-9)) &&
       ((cp3 > 1e-9 && cp4 < -1e-9) || (cp3 < -1e-9 && cp4 > 1e-9)) {
        return true;
    }
    false
}

fn point_in_polygon_opt(p: Point2D, poly: &[Point2D], index: &EdgeIndex, visitor: &mut Vec<usize>) -> bool {
    let max_u = index.min_u + (index.cols as f64) * index.cell_w;
    let max_v = index.min_v + (index.rows as f64) * index.cell_h;
    if p.x < index.min_u || p.x > max_u || p.y < index.min_v || p.y > max_v {
        return false;
    }

    let r_idx = ((p.y - index.min_v) / index.cell_h).floor() as isize;
    if r_idx < 0 || r_idx >= index.rows as isize {
        return false;
    }
    let r = r_idx as usize;

    let c_start_idx = ((p.x - index.min_u) / index.cell_w).floor() as isize;
    let c_start = c_start_idx.max(0) as usize;
    if c_start >= index.cols {
        return false;
    }

    visitor.clear();
    for c in c_start..index.cols {
        visitor.extend_from_slice(&index.buckets[r * index.cols + c]);
    }
    visitor.sort_unstable();
    visitor.dedup();

    let mut inside = false;
    let n = poly.len();
    for &edge_idx in visitor.iter() {
        let p1 = poly[edge_idx];
        let p2 = poly[(edge_idx + 1) % n];

        if ((p1.y > p.y) != (p2.y > p.y)) &&
           (p.x < (p2.x - p1.x) * (p.y - p1.y) / (p2.y - p1.y + 1e-18) + p1.x) {
            inside = !inside;
        }
    }
    inside
}

fn rect_intersects_polygon_opt(
    rect_min: Point2D,
    rect_max: Point2D,
    poly: &[Point2D],
    index: &EdgeIndex,
    edge_visitor: &mut Vec<usize>,
    pip_visitor: &mut Vec<usize>,
) -> bool {
    let poly_max_u = index.min_u + (index.cols as f64) * index.cell_w;
    let poly_max_v = index.min_v + (index.rows as f64) * index.cell_h;
    if rect_max.x < index.min_u || rect_min.x > poly_max_u ||
       rect_max.y < index.min_v || rect_min.y > poly_max_v {
        return false;
    }

    let corners = [
        Point2D { x: rect_min.x, y: rect_min.y },
        Point2D { x: rect_max.x, y: rect_min.y },
        Point2D { x: rect_max.x, y: rect_max.y },
        Point2D { x: rect_min.x, y: rect_max.y },
    ];
    for &c in &corners {
        if point_in_polygon_opt(c, poly, index, pip_visitor) {
            return true;
        }
    }

    index.get_overlapping_edges(rect_min, rect_max, edge_visitor);

    let n = poly.len();
    for &edge_idx in edge_visitor.iter() {
        let p1 = poly[edge_idx];
        if p1.x >= rect_min.x && p1.x <= rect_max.x && p1.y >= rect_min.y && p1.y <= rect_max.y {
            return true;
        }
        let p2 = poly[(edge_idx + 1) % n];
        if p2.x >= rect_min.x && p2.x <= rect_max.x && p2.y >= rect_min.y && p2.y <= rect_max.y {
            return true;
        }
    }

    let rect_edges = [
        (corners[0], corners[1]),
        (corners[1], corners[2]),
        (corners[2], corners[3]),
        (corners[3], corners[0]),
    ];
    for &edge_idx in edge_visitor.iter() {
        let p1 = poly[edge_idx];
        let p2 = poly[(edge_idx + 1) % n];
        for &re in &rect_edges {
            if segments_intersect(p1, p2, re.0, re.1) {
                return true;
            }
        }
    }

    false
}

fn matrix_to_quaternion(r: [[f64; 3]; 3]) -> [f64; 4] {
    let trace = r[0][0] + r[1][1] + r[2][2];
    let mut q = [0.0; 4];
    if trace > 0.0 {
        let s = (trace + 1.0).sqrt() * 2.0;
        q[0] = 0.25 * s;
        q[1] = (r[2][1] - r[1][2]) / s;
        q[2] = (r[0][2] - r[2][0]) / s;
        q[3] = (r[1][0] - r[0][1]) / s;
    } else if (r[0][0] > r[1][1]) && (r[0][0] > r[2][2]) {
        let s = (1.0 + r[0][0] - r[1][1] - r[2][2]).sqrt() * 2.0;
        q[0] = (r[2][1] - r[1][2]) / s;
        q[1] = 0.25 * s;
        q[2] = (r[0][1] + r[1][0]) / s;
        q[3] = (r[0][2] + r[2][0]) / s;
    } else if r[1][1] > r[2][2] {
        let s = (1.0 + r[1][1] - r[0][0] - r[2][2]).sqrt() * 2.0;
        q[0] = (r[0][2] - r[2][0]) / s;
        q[1] = (r[0][1] + r[1][0]) / s;
        q[2] = 0.25 * s;
        q[3] = (r[1][2] + r[2][1]) / s;
    } else {
        let s = (1.0 + r[2][2] - r[0][0] - r[1][1]).sqrt() * 2.0;
        q[0] = (r[1][0] - r[0][1]) / s;
        q[1] = (r[0][2] + r[2][0]) / s;
        q[2] = (r[1][2] + r[2][1]) / s;
        q[3] = 0.25 * s;
    }
    let len = (q[0]*q[0] + q[1]*q[1] + q[2]*q[2] + q[3]*q[3]).sqrt();
    [q[0]/len, q[1]/len, q[2]/len, q[3]/len]
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let nv_line = match lines.next() {
        Some(Ok(l)) => l.trim().to_string(),
        _ => return,
    };
    if nv_line.is_empty() {
        return;
    }
    let nv: usize = match nv_line.parse() {
        Ok(v) => v,
        Err(_) => return,
    };

    let mut poly = Vec::with_capacity(nv);
    for _ in 0..nv {
        let line = match lines.next() {
            Some(Ok(l)) => l,
            _ => return,
        };
        let parts: Vec<&str> = line.split_whitespace().collect();
        if parts.len() < 2 {
            return;
        }
        let x: f64 = parts[0].parse().unwrap();
        let y: f64 = parts[1].parse().unwrap();
        poly.push(Point2D { x, y });
    }

    let sun_line = match lines.next() {
        Some(Ok(l)) => l,
        _ => return,
    };
    let sun_parts: Vec<&str> = sun_line.split_whitespace().collect();
    if sun_parts.len() < 3 {
        return;
    }
    let mut sx: f64 = sun_parts[0].parse().unwrap();
    let mut sy: f64 = sun_parts[1].parse().unwrap();
    let mut sz: f64 = sun_parts[2].parse().unwrap();

    if sz.abs() < 1e-9 {
        sz = -1e-9;
    }
    if sz > 0.0 {
        sx = -sx;
        sy = -sy;
        sz = -sz;
    }

    let sz_val = sz;
    let a = -sx / sz_val;
    let b = -sy / sz_val;

    let len_s = (sx*sx + sy*sy + sz*sz).sqrt();
    let u3 = [sx / len_s, sy / len_s, sz / len_s];

    let e1 = if u3[0].abs() > 1e-6 || u3[1].abs() > 1e-6 {
        let len_perp = (u3[0]*u3[0] + u3[1]*u3[1]).sqrt();
        [-u3[1] / len_perp, u3[0] / len_perp, 0.0]
    } else {
        [1.0, 0.0, 0.0]
    };
    let e2 = [
        u3[1] * e1[2] - u3[2] * e1[1],
        u3[2] * e1[0] - u3[0] * e1[2],
        u3[0] * e1[1] - u3[1] * e1[0],
    ];

    let f1 = [e1[0] + a * e1[2], e1[1] + b * e1[2]];
    let f2 = [e2[0] + a * e2[2], e2[1] + b * e2[2]];

    let f1_len2 = f1[0]*f1[0] + f1[1]*f1[1];
    let f2_len2 = f2[0]*f2[0] + f2[1]*f2[1];
    let f1_dot_f2 = f1[0]*f2[0] + f1[1]*f2[1];

    let x_val = 2.0 * f1_dot_f2;
    let y_val = f2_len2 - f1_len2;

    let theta = 0.5 * y_val.atan2(x_val);

    let cos_t = theta.cos();
    let sin_t = theta.sin();

    let u1 = [
        cos_t * e1[0] + sin_t * e2[0],
        cos_t * e1[1] + sin_t * e2[1],
        cos_t * e1[2] + sin_t * e2[2],
    ];
    let u2 = [
        -sin_t * e1[0] + cos_t * e2[0],
        -sin_t * e1[1] + cos_t * e2[1],
        -sin_t * e1[2] + cos_t * e2[2],
    ];

    let r = [
        [u1[0], u2[0], u3[0]],
        [u1[1], u2[1], u3[1]],
        [u1[2], u2[2], u3[2]],
    ];
    let q = matrix_to_quaternion(r);

    let proj_u1 = [u1[0] + a * u1[2], u1[1] + b * u1[2]];
    let proj_u2 = [u2[0] + a * u2[2], u2[1] + b * u2[2]];

    let w_vec = [proj_u2[0] - proj_u1[0], proj_u2[1] - proj_u1[1]];
    let h_vec = [proj_u1[0] + proj_u2[0], proj_u1[1] + proj_u2[1]];

    let w = (w_vec[0]*w_vec[0] + w_vec[1]*w_vec[1]).sqrt();
    let h = (h_vec[0]*h_vec[0] + h_vec[1]*h_vec[1]).sqrt();

    let v_width = [w_vec[0] / w, w_vec[1] / w];
    let v_height = [h_vec[0] / h, h_vec[1] / h];

    let s: f64 = 10.0;
    let cell_w_size = s * w;
    let cell_h_size = s * h;

    let mut poly_transformed = Vec::with_capacity(nv);
    for p in &poly {
        let tu = p.x * v_width[0] + p.y * v_width[1];
        let tv = p.x * v_height[0] + p.y * v_height[1];
        poly_transformed.push(Point2D { x: tu, y: tv });
    }

    let mut u_min = f64::INFINITY;
    let mut u_max = f64::NEG_INFINITY;
    let mut v_min = f64::INFINITY;
    let mut v_max = f64::NEG_INFINITY;
    for p in &poly_transformed {
        if p.x < u_min { u_min = p.x; }
        if p.x > u_max { u_max = p.x; }
        if p.y < v_min { v_min = p.y; }
        if p.y > v_max { v_max = p.y; }
    }

    let bb_w = u_max - u_min;
    let bb_h = v_max - v_min;
    let approx_cells = (bb_w / cell_w_size).ceil() * (bb_h / cell_h_size).ceil();
    let n_shifts = if approx_cells < 5.0 {
        12
    } else if approx_cells < 25.0 {
        8
    } else if approx_cells < 100.0 {
        5
    } else if approx_cells < 400.0 {
        3
    } else {
        1
    };

    let grid_size = ((nv as f64).sqrt() as usize).clamp(4, 100);
    let edge_index = EdgeIndex::new(&poly_transformed, grid_size, grid_size);

    let mut best_count = usize::MAX;
    let mut best_offset = (0.0, 0.0);
    let mut best_cells = Vec::new();

    let mut edge_visitor = Vec::new();
    let mut pip_visitor = Vec::new();

    for ku in 0..n_shifts {
        let u_start = u_min - (ku as f64) * (cell_w_size / (n_shifts as f64));
        for kv in 0..n_shifts {
            let v_start = v_min - (kv as f64) * (cell_h_size / (n_shifts as f64));

            let i_min = ((u_min - u_start) / cell_w_size).floor() as isize;
            let i_max = ((u_max - u_start) / cell_w_size).floor() as isize;
            let j_min = ((v_min - v_start) / cell_h_size).floor() as isize;
            let j_max = ((v_max - v_start) / cell_h_size).floor() as isize;

            let mut current_cells = Vec::new();
            for i in i_min..=i_max {
                for j in j_min..=j_max {
                    let r_min = Point2D {
                        x: u_start + (i as f64) * cell_w_size,
                        y: v_start + (j as f64) * cell_h_size,
                    };
                    let r_max = Point2D {
                        x: r_min.x + cell_w_size,
                        y: r_min.y + cell_h_size,
                    };

                    if rect_intersects_polygon_opt(r_min, r_max, &poly_transformed, &edge_index, &mut edge_visitor, &mut pip_visitor) {
                        current_cells.push((i, j));
                    }
                }
            }

            if current_cells.len() < best_count {
                best_count = current_cells.len();
                best_offset = (u_start, v_start);
                best_cells = current_cells;
            }
        }
    }

    println!("{}", best_cells.len());

    let u_start = best_offset.0;
    let v_start = best_offset.1;
    for (idx, &(i, j)) in best_cells.iter().enumerate() {
        let uc = u_start + (i as f64 + 0.5) * cell_w_size;
        let vc = v_start + (j as f64 + 0.5) * cell_h_size;

        let ux = uc * v_width[0] + vc * v_height[0];
        let uy = uc * v_width[1] + vc * v_height[1];

        let pz = (idx as f64) * 30.0;
        let px = ux + pz * sx / sz_val;
        let py = uy + pz * sy / sz_val;

        println!("{} {} {} {} {} {} {} {}",
            px, py, pz, q[0], q[1], q[2], q[3], s);
    }
}
