using System;
using System.Collections.Generic;
using System.Linq;

class TetrisPlayer
{
    struct Point
    {
        public int X, Y;
        public Point(int x, int y) { X = x; Y = y; }
    }

    class Piece
    {
        public string Type;
        public int Rotation;
        public int Width;
        public int Height;
        public Point[] Blocks;

        public Piece(string type, int rotation, Point[] blocks)
        {
            Type = type;
            Rotation = rotation;
            Blocks = blocks;
            Width = blocks.Max(p => p.X) + 1;
            Height = blocks.Max(p => p.Y) + 1;
        }
    }

    class BoardState
    {
        public int Width;
        public int Height;
        public bool[,] Grid;
        public int[] ColumnHeights;

        public BoardState(int width, int height)
        {
            Width = width;
            Height = height;
            Grid = new bool[width, height];
            ColumnHeights = new int[width];
        }

        public void CopyFrom(BoardState source)
        {
            Array.Copy(source.Grid, Grid, Grid.Length);
            Array.Copy(source.ColumnHeights, ColumnHeights, Width);
        }

        public bool PlacePiece(Piece piece, int col, out int linesCleared, out int erodedCells)
        {
            linesCleared = 0;
            erodedCells = 0;

            // Compute landing height
            int landingY = 0;
            for (int i = 0; i < piece.Blocks.Length; i++)
            {
                int x = col + piece.Blocks[i].X;
                int y = ColumnHeights[x] - piece.Blocks[i].Y;
                if (y > landingY) landingY = y;
            }

            // Check if piece overflows board height (game over)
            for (int i = 0; i < piece.Blocks.Length; i++)
            {
                int y = landingY + piece.Blocks[i].Y;
                if (y >= Height)
                {
                    return false;
                }
            }

            // Place blocks
            for (int i = 0; i < piece.Blocks.Length; i++)
            {
                int x = col + piece.Blocks[i].X;
                int y = landingY + piece.Blocks[i].Y;
                Grid[x, y] = true;
            }

            // Update column heights for placed piece columns
            for (int x = col; x < col + piece.Width; x++)
            {
                ColumnHeights[x] = GetColumnHeight(x);
            }

            // Detect and clear completed lines
            List<int> linesToClear = new List<int>();
            int minY = landingY;
            int maxY = landingY + piece.Height - 1;
            for (int y = minY; y <= maxY; y++)
            {
                bool full = true;
                for (int x = 0; x < Width; x++)
                {
                    if (!Grid[x, y])
                    {
                        full = false;
                        break;
                    }
                }
                if (full)
                {
                    linesToClear.Add(y);
                }
            }

            linesCleared = linesToClear.Count;
            if (linesCleared > 0)
            {
                // Count how many blocks of the placed piece are in the cleared lines (eroded cells)
                for (int i = 0; i < piece.Blocks.Length; i++)
                {
                    int y = landingY + piece.Blocks[i].Y;
                    if (linesToClear.Contains(y))
                    {
                        erodedCells++;
                    }
                }

                // Shift rows down to clear lines
                linesToClear.Sort((a, b) => b.CompareTo(a));
                foreach (int row in linesToClear)
                {
                    for (int y = row; y < Height - 1; y++)
                    {
                        for (int x = 0; x < Width; x++)
                        {
                            Grid[x, y] = Grid[x, y + 1];
                        }
                    }
                    for (int x = 0; x < Width; x++)
                    {
                        Grid[x, Height - 1] = false;
                    }
                }

                // Recompute all column heights after shifting
                for (int x = 0; x < Width; x++)
                {
                    ColumnHeights[x] = GetColumnHeight(x);
                }
            }

            return true;
        }

        private int GetColumnHeight(int x)
        {
            for (int y = Height - 1; y >= 0; y--)
            {
                if (Grid[x, y]) return y + 1;
            }
            return 0;
        }

        public int GetMaxHeight()
        {
            int max = 0;
            for (int x = 0; x < Width; x++)
            {
                if (ColumnHeights[x] > max) max = ColumnHeights[x];
            }
            return max;
        }

        // Evaluate features of the board state that are persistent
        public double EvaluateStaticOnly(double[] weights)
        {
            int maxH = GetMaxHeight();

            // Row Transitions
            int rowTransitions = 0;
            for (int y = 0; y < maxH; y++)
            {
                bool prevOccupied = true; // left boundary wall
                for (int x = 0; x < Width; x++)
                {
                    bool occupied = Grid[x, y];
                    if (occupied != prevOccupied) rowTransitions++;
                    prevOccupied = occupied;
                }
                if (!prevOccupied) rowTransitions++; // right boundary wall
            }

            // Column Transitions
            int colTransitions = 0;
            for (int x = 0; x < Width; x++)
            {
                bool prevOccupied = true; // bottom floor
                for (int y = 0; y < maxH; y++)
                {
                    bool occupied = Grid[x, y];
                    if (occupied != prevOccupied) colTransitions++;
                    prevOccupied = occupied;
                }
                if (prevOccupied) colTransitions++; // top transition to empty space
            }

            // Holes
            int holes = 0;
            for (int x = 0; x < Width; x++)
            {
                bool blockAbove = false;
                for (int y = ColumnHeights[x] - 1; y >= 0; y--)
                {
                    if (Grid[x, y])
                    {
                        blockAbove = true;
                    }
                    else if (blockAbove)
                    {
                        holes++;
                    }
                }
            }

            // Well Sums
            int wellSum = 0;
            for (int x = 0; x < Width; x++)
            {
                int hMid = ColumnHeights[x];
                int hLeft = (x - 1 >= 0) ? ColumnHeights[x - 1] : Height;
                int hRight = (x + 1 < Width) ? ColumnHeights[x + 1] : Height;
                int depth = Math.Min(hLeft, hRight) - hMid;
                if (depth > 0)
                {
                    wellSum += depth * (depth + 1) / 2;
                }
            }

            return weights[2] * rowTransitions
                 + weights[3] * colTransitions
                 + weights[4] * holes
                 + weights[5] * wellSum;
        }

        // Full evaluation including transient placement metrics
        public double Evaluate(double[] weights, double landingHeight, int linesCleared, int erodedCells)
        {
            double erodedValue = linesCleared * erodedCells;
            return weights[0] * landingHeight
                 + weights[1] * erodedValue
                 + EvaluateStaticOnly(weights);
        }
    }

    static Dictionary<string, List<Piece>> PieceTemplates = new Dictionary<string, List<Piece>>();

    // Set up pieces and rotation permutations using standard SRS rules
    static void InitializePieces()
    {
        var basePieces = new Dictionary<string, (int N, Point[] points)>()
        {
            { "I", (4, new Point[] { new Point(0, 2), new Point(1, 2), new Point(2, 2), new Point(3, 2) }) },
            { "O", (2, new Point[] { new Point(0, 0), new Point(1, 0), new Point(0, 1), new Point(1, 1) }) },
            { "T", (3, new Point[] { new Point(1, 1), new Point(0, 0), new Point(1, 0), new Point(2, 0) }) },
            { "S", (3, new Point[] { new Point(0, 0), new Point(1, 0), new Point(1, 1), new Point(2, 1) }) },
            { "Z", (3, new Point[] { new Point(0, 1), new Point(1, 1), new Point(1, 0), new Point(2, 0) }) },
            { "J", (3, new Point[] { new Point(0, 1), new Point(0, 0), new Point(1, 0), new Point(2, 0) }) },
            { "L", (3, new Point[] { new Point(2, 1), new Point(0, 0), new Point(1, 0), new Point(2, 0) }) }
        };

        foreach (var kvp in basePieces)
        {
            string type = kvp.Key;
            int N = kvp.Value.N;
            Point[] pts = kvp.Value.points;

            Piece[] rotations = new Piece[4];
            for (int rot = 0; rot < 4; rot++)
            {
                Point[] current = new Point[pts.Length];
                for (int i = 0; i < pts.Length; i++)
                {
                    Point p = pts[i];
                    for (int r = 0; r < rot; r++)
                    {
                        int nx = p.Y;
                        int ny = N - 1 - p.X;
                        p = new Point(nx, ny);
                    }
                    current[i] = p;
                }

                // Normalize rotation so leftmost column is 0 and bottommost row is 0
                int minX = current.Min(p => p.X);
                int minY = current.Min(p => p.Y);
                for (int i = 0; i < current.Length; i++)
                {
                    current[i] = new Point(current[i].X - minX, current[i].Y - minY);
                }

                rotations[rot] = new Piece(type, rot, current);
            }

            // Prune duplicate rotations to save computation time
            List<Piece> uniqueRotations = new List<Piece>();
            foreach (var rPiece in rotations)
            {
                bool isDuplicate = false;
                foreach (var unique in uniqueRotations)
                {
                    if (AreShapesEqual(unique.Blocks, rPiece.Blocks))
                    {
                        isDuplicate = true;
                        break;
                    }
                }
                if (!isDuplicate)
                {
                    uniqueRotations.Add(rPiece);
                }
            }

            PieceTemplates[type] = uniqueRotations;
        }
    }

    static bool AreShapesEqual(Point[] a, Point[] b)
    {
        if (a.Length != b.Length) return false;
        var setA = new HashSet<(int, int)>(a.Select(p => (p.X, p.Y)));
        return b.All(p => setA.Contains((p.X, p.Y)));
    }

    // Adaptive weights based on current stack height and board width
    static double[] GetWeights(int width, int height, int maxHeight)
    {
        // El-Tetris (PSO-tuned Dellacherie) baseline weights
        double[] w = new double[] {
            -4.500158825082766,  // Landing Height
             3.4181268101392694, // Eroded Piece Cells Value
            -3.2178882868487753, // Row Transitions
            -9.348695305445199,  // Column Transitions
            -7.899265427351652,  // Holes
            -3.3855972247263626  // Well Sums
        };

        // If the board is narrow, holes and wells are significantly more critical
        if (width <= 8)
        {
            w[4] = -12.0; // Higher hole penalty
            w[5] = -5.0;  // Higher well penalty
        }

        // If stack is high, prioritize survival: clear lines and keep the board flat
        double fillRatio = (double)maxHeight / height;
        if (fillRatio > 0.6)
        {
            w[0] = -12.0; // Punish landing high severely
            w[1] = 8.0;   // High reward for line clears
            w[4] = -20.0; // Extreme penalty for holes
            w[5] = -6.0;  // Extra penalty for wells
        }

        return w;
    }

    // Robust parser to read tokens separated by any whitespace (spaces, tabs, newlines)
    class TokenReader
    {
        private Queue<string> tokens = new Queue<string>();

        public string NextToken()
        {
            while (tokens.Count == 0)
            {
                string line = Console.ReadLine();
                if (line == null) return null;
                string[] parts = line.Split(new char[] { ' ', '\t', '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries);
                foreach (var part in parts)
                {
                    tokens.Enqueue(part);
                }
            }
            return tokens.Dequeue();
        }
    }

    static void Main(string[] args)
    {
        InitializePieces();

        TokenReader reader = new TokenReader();

        string wToken = reader.NextToken();
        if (wToken == null) return;

        int width = int.Parse(wToken);
        int height = int.Parse(reader.NextToken());
        int numPieces = int.Parse(reader.NextToken());

        BoardState board = new BoardState(width, height);

        // Preallocate scratch board states to eliminate heap allocations inside search loops
        BoardState nextB = new BoardState(width, height);
        BoardState leafB = new BoardState(width, height);

        for (int p = 0; p < numPieces; p++)
        {
            string pieceType = reader.NextToken();
            if (pieceType == null) break;

            double bestScore = double.MinValue;
            Piece bestPiece = null;
            int bestCol = -1;

            int currentMaxHeight = board.GetMaxHeight();
            double[] weights = GetWeights(width, height, currentMaxHeight);

            // Dynamic depth selection based on board width to guarantee < 5s run times
            int depth = (width <= 20) ? 2 : 1;

            foreach (var piece in PieceTemplates[pieceType])
            {
                for (int col = 0; col <= width - piece.Width; col++)
                {
                    nextB.CopyFrom(board);
                    int linesCleared, erodedCells;
                    if (!nextB.PlacePiece(piece, col, out linesCleared, out erodedCells))
                    {
                        continue; // Invalid placement (triggers game over)
                    }

                    double score;
                    if (depth == 1)
                    {
                        int landingY = 0;
                        for (int i = 0; i < piece.Blocks.Length; i++)
                        {
                            int x = col + piece.Blocks[i].X;
                            int y = board.ColumnHeights[x] - piece.Blocks[i].Y;
                            if (y > landingY) landingY = y;
                        }
                        double landingHeightVal = landingY + 1.0 + (piece.Height - 1) / 2.0;
                        score = nextB.Evaluate(weights, landingHeightVal, linesCleared, erodedCells);
                    }
                    else
                    {
                        // Depth 2: Expectimax over all possible 7 next piece types
                        int landingY = 0;
                        for (int i = 0; i < piece.Blocks.Length; i++)
                        {
                            int x = col + piece.Blocks[i].X;
                            int y = board.ColumnHeights[x] - piece.Blocks[i].Y;
                            if (y > landingY) landingY = y;
                        }
                        double landingHeightVal = landingY + 1.0 + (piece.Height - 1) / 2.0;
                        double erodedValue = linesCleared * erodedCells;
                        double immediateScore = weights[0] * landingHeightVal + weights[1] * erodedValue;

                        double expectedFuture = 0;
                        foreach (var nextType in PieceTemplates.Keys)
                        {
                            double bestNextScore = double.MinValue;
                            foreach (var nextPiece in PieceTemplates[nextType])
                            {
                                for (int nextCol = 0; nextCol <= width - nextPiece.Width; nextCol++)
                                {
                                    leafB.CopyFrom(nextB);
                                    int nextLinesCleared, nextErodedCells;
                                    if (!leafB.PlacePiece(nextPiece, nextCol, out nextLinesCleared, out nextErodedCells))
                                    {
                                        continue;
                                    }

                                    int nextLandingY = 0;
                                    for (int i = 0; i < nextPiece.Blocks.Length; i++)
                                    {
                                        int x = nextCol + nextPiece.Blocks[i].X;
                                        int y = nextB.ColumnHeights[x] - nextPiece.Blocks[i].Y;
                                        if (y > nextLandingY) nextLandingY = y;
                                    }
                                    double nextLandingHeightVal = nextLandingY + 1.0 + (nextPiece.Height - 1) / 2.0;
                                    double nextErodedValue = nextLinesCleared * nextErodedCells;

                                    double nextScore = weights[0] * nextLandingHeightVal 
                                                     + weights[1] * nextErodedValue 
                                                     + leafB.EvaluateStaticOnly(weights);

                                    if (nextScore > bestNextScore)
                                    {
                                        bestNextScore = nextScore;
                                    }
                                }
                            }
                            if (bestNextScore == double.MinValue)
                            {
                                // No valid placements for this piece type results in gameover penalty
                                bestNextScore = -1000000.0;
                            }
                            expectedFuture += bestNextScore;
                        }
                        expectedFuture /= 7.0;

                        score = immediateScore + expectedFuture;
                    }

                    if (score > bestScore)
                    {
                        bestScore = score;
                        bestPiece = piece;
                        bestCol = col;
                    }
                }
            }

            // Fallback in case of unavoidable gameover
            if (bestPiece == null)
            {
                bestPiece = PieceTemplates[pieceType][0];
                bestCol = 0;
            }

            // Output optimal placement
            Console.WriteLine($"{bestPiece.Rotation} {bestCol}");
            Console.Out.Flush();

            // Receive feedback from evaluator
            string result = reader.NextToken();
            if (result == null || result == "gameover")
            {
                break;
            }

            // Synchronize local board state with the evaluator's state
            int localLines, localEroded;
            board.PlacePiece(bestPiece, bestCol, out localLines, out localEroded);
        }
    }
}
