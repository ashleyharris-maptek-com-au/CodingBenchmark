use std::collections::HashMap;
use std::io::{self, Read, Write};
use std::sync::atomic::{AtomicBool, Ordering};
use std::sync::{Arc, Mutex};
use std::thread;

// Custom fast parser for byte slices
struct FastParser<'a> {
    bytes: &'a [u8],
    pos: usize,
}

impl<'a> FastParser<'a> {
    fn new(bytes: &'a [u8]) -> Self {
        Self { bytes, pos: 0 }
    }

    #[inline(always)]
    fn next_u64(&mut self) -> Option<u64> {
        while self.pos < self.bytes.len() && self.bytes[self.pos] <= b' ' {
            self.pos += 1;
        }
        if self.pos >= self.bytes.len() {
            return None;
        }
        let mut val = 0;
        while self.pos < self.bytes.len() && self.bytes[self.pos] > b' ' {
            let b = self.bytes[self.pos];
            if b >= b'0' && b <= b'9' {
                val = val * 10 + (b - b'0') as u64;
            }
            self.pos += 1;
        }
        Some(val)
    }

    #[inline(always)]
    fn next_i64(&mut self) -> Option<i64> {
        while self.pos < self.bytes.len() && self.bytes[self.pos] <= b' ' {
            self.pos += 1;
        }
        if self.pos >= self.bytes.len() {
            return None;
        }
        let mut neg = false;
        if self.bytes[self.pos] == b'-' {
            neg = true;
            self.pos += 1;
        } else if self.bytes[self.pos] == b'+' {
            self.pos += 1;
        }
        let mut val = 0i64;
        while self.pos < self.bytes.len() && self.bytes[self.pos] > b' ' {
            let b = self.bytes[self.pos];
            if b >= b'0' && b <= b'9' {
                val = val * 10 + (b - b'0') as i64;
            }
            self.pos += 1;
        }
        if neg {
            Some(-val)
        } else {
            Some(val)
        }
    }
}

fn parse_input(bytes: &[u8]) -> (usize, i64, Vec<i64>) {
    let mut seq_parser = FastParser::new(bytes);
    let num_numbers = seq_parser.next_u64().unwrap_or(0) as usize;
    let target = seq_parser.next_i64().unwrap_or(0);
    let start_pos = seq_parser.pos;
    
    let numbers_bytes = &bytes[start_pos..];
    
    if numbers_bytes.len() < 1_000_000 {
        let mut parser = FastParser::new(numbers_bytes);
        let mut numbers = Vec::with_capacity(num_numbers);
        while let Some(val) = parser.next_i64() {
            numbers.push(val);
        }
        return (num_numbers, target, numbers);
    }
    
    let num_threads = thread::available_parallelism().map(|n| n.get()).unwrap_or(8).max(1);
    let chunk_size = numbers_bytes.len() / num_threads;
    
    let mut chunk_boundaries = vec![0; num_threads + 1];
    chunk_boundaries[num_threads] = numbers_bytes.len();
    
    for i in 1..num_threads {
        let mut pos = i * chunk_size;
        while pos < numbers_bytes.len() && numbers_bytes[pos] > b' ' {
            pos += 1;
        }
        while pos < numbers_bytes.len() && numbers_bytes[pos] <= b' ' {
            pos += 1;
        }
        chunk_boundaries[i] = pos.min(numbers_bytes.len());
    }
    
    let mut results = vec![Vec::new(); num_threads];
    
    thread::scope(|s| {
        let mut threads = Vec::with_capacity(num_threads);
        for (i, res) in results.iter_mut().enumerate() {
            let start = chunk_boundaries[i];
            let end = chunk_boundaries[i + 1];
            if start >= end {
                continue;
            }
            let slice = &numbers_bytes[start..end];
            threads.push(s.spawn(move || {
                let mut parser = FastParser::new(slice);
                let mut chunk_nums = Vec::new();
                while let Some(val) = parser.next_i64() {
                    chunk_nums.push(val);
                }
                *res = chunk_nums;
            }));
        }
        for t in threads {
            let _ = t.join();
        }
    });
    
    let mut numbers = Vec::with_capacity(num_numbers);
    for mut chunk_nums in results {
        numbers.append(&mut chunk_nums);
    }
    
    (num_numbers, target, numbers)
}

#[derive(Clone, Debug)]
struct BElement {
    id: usize,
    val: i64,
    was_negative: bool,
}

#[derive(Clone, Debug)]
struct GroupedElement {
    val: i64,
    constituents: Vec<BElement>,
}

struct Lcg {
    state: u64,
}

impl Lcg {
    fn new(seed: u64) -> Self {
        Self { state: seed }
    }

    fn next_u64(&mut self) -> u64 {
        self.state = self.state.wrapping_mul(6364136223846793005).wrapping_add(1);
        self.state
    }

    fn next_usize(&mut self, limit: usize) -> usize {
        (self.next_u64() as usize) % limit
    }

    fn next_bool(&mut self, p: f64) -> bool {
        let val = (self.next_u64() & 0xFFFF_FFFF) as f64 / 4294967296.0;
        val < p
    }

    fn shuffle<T>(&mut self, slice: &mut [T]) {
        for i in (1..slice.len()).rev() {
            let j = self.next_usize(i + 1);
            slice.swap(i, j);
        }
    }
}

#[derive(Clone)]
struct BitSet {
    words: Vec<u64>,
}

impl BitSet {
    fn new(size: usize) -> Self {
        let num_words = (size + 63) / 64;
        Self {
            words: vec![0; num_words],
        }
    }

    fn set(&mut self, idx: usize) {
        let w = idx / 64;
        let b = idx % 64;
        if w < self.words.len() {
            self.words[w] |= 1 << b;
        }
    }

    fn get(&self, idx: usize) -> bool {
        let w = idx / 64;
        let b = idx % 64;
        if w < self.words.len() {
            (self.words[w] & (1 << b)) != 0
        } else {
            false
        }
    }

    fn or_shl(&mut self, x: usize) {
        let word_shift = x / 64;
        let bit_shift = x % 64;
        let w_len = self.words.len();
        if word_shift >= w_len {
            return;
        }
        if bit_shift == 0 {
            for i in (word_shift..w_len).rev() {
                self.words[i] |= self.words[i - word_shift];
            }
        } else {
            let shift_right = 64 - bit_shift;
            for i in (word_shift..w_len).rev() {
                let j = i - word_shift;
                let mut val = self.words[j] << bit_shift;
                if j > 0 {
                    val |= self.words[j - 1] >> shift_right;
                }
                self.words[i] |= val;
            }
        }
    }
}

fn generate_subsets_grouped(half: &[GroupedElement]) -> Vec<(i128, u64)> {
    let n = half.len();
    let num_subsets = 1 << n;
    let mut subsets = Vec::with_capacity(num_subsets);
    subsets.push((0i128, 0u64));
    for i in 0..n {
        let val = half[i].val as i128;
        let len = subsets.len();
        for j in 0..len {
            let (s, mask) = subsets[j];
            subsets.push((s + val, mask | (1 << i)));
        }
    }
    subsets
}

fn solve_mitm_grouped(a: &[GroupedElement], target: i128) -> Option<Vec<usize>> {
    let n = a.len();
    if n == 0 {
        if target == 0 {
            return Some(vec![]);
        }
        return None;
    }
    let mid = n / 2;
    let (a1, a2) = a.split_at(mid);

    let mut subsets1 = generate_subsets_grouped(a1);
    let subsets2 = generate_subsets_grouped(a2);

    subsets1.sort_unstable_by_key(|x| x.0);
    subsets1.dedup_by(|x, y| x.0 == y.0);

    for (s2, mask2) in subsets2 {
        let needed = target - s2;
        if let Ok(idx) = subsets1.binary_search_by_key(&needed, |x| x.0) {
            let mask1 = subsets1[idx].1;
            let mut res = Vec::new();
            for i in 0..a1.len() {
                if (mask1 & (1 << i)) != 0 {
                    res.push(i);
                }
            }
            for i in 0..a2.len() {
                if (mask2 & (1 << i)) != 0 {
                    res.push(mid + i);
                }
            }
            return Some(res);
        }
    }
    None
}

fn solve_dp_grouped(a: &[GroupedElement], target: usize) -> Option<Vec<usize>> {
    if target == 0 {
        return Some(vec![]);
    }
    if a.is_empty() {
        return None;
    }
    if a.len() == 1 {
        if a[0].val as usize == target {
            return Some(vec![0]);
        } else {
            return None;
        }
    }

    let mid = a.len() / 2;
    let (a1, a2) = a.split_at(mid);

    let sum1 = a1.iter().map(|x| x.val as usize).sum::<usize>();
    let sum2 = a2.iter().map(|x| x.val as usize).sum::<usize>();

    let limit1 = sum1.min(target);
    let limit2 = sum2.min(target);

    let mut dp1 = BitSet::new(limit1 + 1);
    dp1.set(0);
    for elem in a1 {
        dp1.or_shl(elem.val as usize);
    }

    let mut dp2 = BitSet::new(limit2 + 1);
    dp2.set(0);
    for elem in a2 {
        dp2.or_shl(elem.val as usize);
    }

    let start = if target > limit2 { target - limit2 } else { 0 };
    let end = limit1.min(target);

    for s1 in start..=end {
        if dp1.get(s1) && dp2.get(target - s1) {
            if let Some(mut res1) = solve_dp_grouped(a1, s1) {
                if let Some(mut res2) = solve_dp_grouped(a2, target - s1) {
                    for idx in &mut res2 {
                        *idx += mid;
                    }
                    res1.append(&mut res2);
                    return Some(res1);
                }
            }
        }
    }
    None
}

fn solve_rg_mitm_grouped_parallel(a: &[GroupedElement], target: i128, total_iterations: usize) -> Option<Vec<usize>> {
    let num_threads = thread::available_parallelism().map(|n| n.get()).unwrap_or(8).max(1);
    let iterations_per_thread = (total_iterations + num_threads - 1) / num_threads;
    
    let found = Arc::new(AtomicBool::new(false));
    let result = Arc::new(Mutex::new(None));
    let mut handles = Vec::with_capacity(num_threads);
    
    let a_shared = Arc::new(a.to_vec());
    
    for t_idx in 0..num_threads {
        let found = Arc::clone(&found);
        let result = Arc::clone(&result);
        let a_shared = Arc::clone(&a_shared);
        let seed = 42 + t_idx as u64 * 1000;
        
        let handle = thread::spawn(move || {
            let mut rng = Lcg::new(seed);
            let mut indexed_a: Vec<(usize, &GroupedElement)> = a_shared.iter().enumerate().collect();
            indexed_a.sort_unstable_by_key(|x| std::cmp::Reverse(x.1.val));
            
            for _iter in 0..iterations_per_thread {
                if found.load(Ordering::Relaxed) {
                    return;
                }
                
                let mut rem_target = target;
                let mut greedy_subset = Vec::new();
                let mut unused = Vec::new();

                for &item in &indexed_a {
                    let val = item.1.val as i128;
                    if val <= rem_target && rng.next_bool(0.9) {
                        rem_target -= val;
                        greedy_subset.push(item.0);
                    } else {
                        unused.push(item);
                    }
                }

                if rem_target == 0 {
                    let mut res_lock = result.lock().unwrap();
                    if res_lock.is_none() {
                        *res_lock = Some(greedy_subset);
                        found.store(true, Ordering::Relaxed);
                    }
                    return;
                }

                let mut candidates: Vec<(usize, &GroupedElement)> = unused.into_iter().filter(|x| x.1.val as i128 <= rem_target).collect();
                let sum_candidates: i128 = candidates.iter().map(|x| x.1.val as i128).sum();

                if sum_candidates < rem_target {
                    continue;
                }

                let m = 30.min(candidates.len());
                rng.shuffle(&mut candidates);
                let mitm_candidates: Vec<GroupedElement> = candidates[0..m].iter().map(|x| (*x.1).clone()).collect();

                if let Some(mitm_indices) = solve_mitm_grouped(&mitm_candidates, rem_target) {
                    for idx in mitm_indices {
                        greedy_subset.push(candidates[idx].0);
                    }
                    let mut res_lock = result.lock().unwrap();
                    if res_lock.is_none() {
                        *res_lock = Some(greedy_subset);
                        found.store(true, Ordering::Relaxed);
                    }
                    return;
                }
            }
        });
        handles.push(handle);
    }
    
    for handle in handles {
        let _ = handle.join();
    }
    
    let res = result.lock().unwrap().clone();
    res
}

fn solve_subset_sum(a: &[i64], target: i64) -> Option<Vec<usize>> {
    let mut sum_neg = 0i128;
    for &val in a {
        if val < 0 {
            sum_neg += val as i128;
        }
    }
    let target_prime = target as i128 - sum_neg;
    if target_prime < 0 {
        return None;
    }

    let mut b_elements = Vec::with_capacity(a.len());
    for (i, &val) in a.iter().enumerate() {
        let (b_val, was_neg) = if val < 0 {
            (-val, true)
        } else {
            (val, false)
        };
        b_elements.push(BElement {
            id: i,
            val: b_val,
            was_negative: was_neg,
        });
    }

    let sum_b: i128 = b_elements.iter().map(|x| x.val as i128).sum();
    if target_prime > sum_b {
        return None;
    }

    if target_prime == 0 {
        let mut final_subset = Vec::new();
        for elem in &b_elements {
            if elem.was_negative {
                final_subset.push(elem.id);
            }
        }
        return Some(final_subset);
    }

    if target_prime == sum_b {
        let mut final_subset = Vec::new();
        for elem in &b_elements {
            if !elem.was_negative {
                final_subset.push(elem.id);
            }
        }
        return Some(final_subset);
    }

    let filtered_b: Vec<BElement> = b_elements
        .into_iter()
        .filter(|x| !(!x.was_negative && x.val as i128 > target_prime))
        .collect();

    let n_orig = a.len();

    // Group and decompose duplicates
    let grouped = {
        let mut map: HashMap<i64, Vec<BElement>> = HashMap::new();
        for elem in filtered_b {
            map.entry(elem.val).or_default().push(elem);
        }

        let mut result = Vec::new();
        for (val, elems) in map {
            let mut count = elems.len();
            let mut elem_iter = elems.into_iter();
            
            let mut power = 1;
            while count > 0 {
                let take = power.min(count);
                let mut taken_elems = Vec::with_capacity(take);
                for _ in 0..take {
                    if let Some(elem) = elem_iter.next() {
                        taken_elems.push(elem);
                    }
                }
                result.push(GroupedElement {
                    val: val.checked_mul(take as i64).unwrap(),
                    constituents: taken_elems,
                });
                count -= take;
                power *= 2;
            }
        }
        result
    };

    let chosen_grouped_indices = if grouped.len() <= 40 {
        solve_mitm_grouped(&grouped, target_prime)
    } else if target_prime <= 10_000_000 {
        solve_dp_grouped(&grouped, target_prime as usize)
    } else {
        if let Some(res) = solve_rg_mitm_grouped_parallel(&grouped, target_prime, 2000) {
            Some(res)
        } else if target_prime <= 100_000_000 && grouped.len() <= 20000 {
            solve_dp_grouped(&grouped, target_prime as usize)
        } else {
            solve_rg_mitm_grouped_parallel(&grouped, target_prime, 10000)
        }
    };

    let chosen_indices = chosen_grouped_indices?;

    let mut b_chosen = vec![false; n_orig];
    for &g_idx in &chosen_indices {
        for elem in &grouped[g_idx].constituents {
            b_chosen[elem.id] = true;
        }
    }

    let mut final_subset = Vec::new();
    for i in 0..n_orig {
        let val = a[i];
        let was_neg = val < 0;
        let chosen_in_b = b_chosen[i];
        if (!was_neg && chosen_in_b) || (was_neg && !chosen_in_b) {
            final_subset.push(i);
        }
    }

    Some(final_subset)
}

fn print_result(indices: &[usize]) {
    let stdout = io::stdout();
    let mut writer = io::BufWriter::new(stdout.lock());
    writeln!(writer, "YES").unwrap();
    write!(writer, "[").unwrap();
    for (i, &idx) in indices.iter().enumerate() {
        if i > 0 {
            write!(writer, " ").unwrap();
        }
        write!(writer, "{}", idx).unwrap();
    }
    writeln!(writer, "]").unwrap();
}

fn main() {
    let mut input_bytes = Vec::new();
    io::stdin().read_to_end(&mut input_bytes).unwrap();
    
    if input_bytes.is_empty() {
        println!("NO");
        return;
    }
    
    let (_, target, numbers) = parse_input(&input_bytes);
    
    if let Some(indices) = solve_subset_sum(&numbers, target) {
        print_result(&indices);
    } else {
        println!("NO");
    }
}
