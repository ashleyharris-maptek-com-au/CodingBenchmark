use std::io::{self, BufRead, Read, BufWriter, Write};

// Look-up table for cell categories:
// 0: Wall or invalid (like #, \r, \n)
// 1: Open path (' ')
// 2: Visited by forward search ('A', '^', 'v', '<', '>')
// 3: Visited by backward search ('B', 'U', 'D', 'L', 'R')
const LUT: [u8; 256] = {
    let mut lut = [0u8; 256];
    lut[b' ' as usize] = 1;
    
    lut[b'A' as usize] = 2;
    lut[b'^' as usize] = 2;
    lut[b'v' as usize] = 2;
    lut[b'<' as usize] = 2;
    lut[b'>' as usize] = 2;
    
    lut[b'B' as usize] = 3;
    lut[b'U' as usize] = 3;
    lut[b'D' as usize] = 3;
    lut[b'L' as usize] = 3;
    lut[b'R' as usize] = 3;
    
    lut
};

#[inline(always)]
fn get_category(val: u8) -> u8 {
    LUT[val as usize]
}

struct SearchState {
    level: Vec<(u32, u32)>,
    next_level: Vec<(u32, u32)>,
    parent_codes: [u8; 4],
    is_forward: bool,
}

fn main() -> io::Result<()> {
    // Set up buffered stdin reader
    let mut reader = io::BufReader::with_capacity(256 * 1024, io::stdin().lock());
    let mut first_line = String::new();
    reader.read_line(&mut first_line)?;
    
    let mut parts = first_line.split_whitespace();
    let h: usize = match parts.next().and_then(|s| s.parse().ok()) {
        Some(val) => val,
        None => return Ok(()),
    };
    let w: usize = match parts.next().and_then(|s| s.parse().ok()) {
        Some(val) => val,
        None => return Ok(()),
    };
    
    if h == 0 || w == 0 {
        println!("0");
        return Ok(());
    }
    
    // Read the rest of stdin into a flat buffer
    let mut buffer = Vec::with_capacity(h * (w + 2));
    reader.read_to_end(&mut buffer)?;
    
    // Detect newline length and stride
    let mut newline_len = 0;
    if w < buffer.len() {
        if buffer[w] == b'\n' {
            newline_len = 1;
        } else if buffer[w] == b'\r' {
            newline_len = 2;
        }
    }
    let stride = w + newline_len;
    
    // Ensure buffer size is at least h * stride to guarantee safe indexing
    let expected_len = h * stride;
    if buffer.len() < expected_len {
        buffer.resize(expected_len, b'#');
    }
    
    // Find start (A) and end (B) positions
    let mut start_pos = None;
    let mut end_pos = None;
    for y in 0..h {
        let offset = y * stride;
        for x in 0..w {
            let val = buffer[offset + x];
            if val == b'A' && start_pos.is_none() {
                start_pos = Some((x as u32, y as u32));
            } else if val == b'B' && end_pos.is_none() {
                end_pos = Some((x as u32, y as u32));
            }
        }
    }
    
    let start_pos = match start_pos {
        Some(pos) => pos,
        None => {
            println!("0");
            return Ok(());
        }
    };
    let end_pos = match end_pos {
        Some(pos) => pos,
        None => {
            println!("0");
            return Ok(());
        }
    };
    
    // If start is already at the end, the path has 1 step
    if start_pos == end_pos {
        println!("1\n{} {}", start_pos.0, start_pos.1);
        return Ok(());
    }
    
    // Initialize search states for bidirectional BFS
    let mut fw = SearchState {
        level: Vec::with_capacity(1024),
        next_level: Vec::with_capacity(1024),
        parent_codes: [b'^', b'v', b'<', b'>'], // Up, Down, Left, Right
        is_forward: true,
    };
    fw.level.push(start_pos);
    
    let mut bw = SearchState {
        level: Vec::with_capacity(1024),
        next_level: Vec::with_capacity(1024),
        parent_codes: [b'U', b'D', b'L', b'R'], // Up, Down, Left, Right
        is_forward: false,
    };
    bw.level.push(end_pos);
    
    let mut meeting_point = None;
    
    'outer: while !fw.level.is_empty() && !bw.level.is_empty() {
        // Expand the smaller level queue
        let active = if fw.level.len() <= bw.level.len() {
            &mut fw
        } else {
            &mut bw
        };
        
        let is_forward = active.is_forward;
        let parent_codes = active.parent_codes;
        
        for &(cx, cy) in &active.level {
            let cx_usize = cx as usize;
            let cy_offset = (cy as usize) * stride;
            
            // 1. Up Neighbor
            if cy > 0 {
                let ny = cy - 1;
                let offset = cy_offset - stride + cx_usize;
                let val = buffer[offset];
                let cat = get_category(val);
                if cat == 1 {
                    buffer[offset] = parent_codes[1]; // parent is Down
                    active.next_level.push((cx, ny));
                } else if is_forward {
                    if cat == 3 {
                        meeting_point = Some(((cx, cy), (cx, ny)));
                        break 'outer;
                    }
                } else {
                    if cat == 2 {
                        meeting_point = Some(((cx, ny), (cx, cy)));
                        break 'outer;
                    }
                }
            }
            // 2. Down Neighbor
            if (cy as usize) + 1 < h {
                let ny = cy + 1;
                let offset = cy_offset + stride + cx_usize;
                let val = buffer[offset];
                let cat = get_category(val);
                if cat == 1 {
                    buffer[offset] = parent_codes[0]; // parent is Up
                    active.next_level.push((cx, ny));
                } else if is_forward {
                    if cat == 3 {
                        meeting_point = Some(((cx, cy), (cx, ny)));
                        break 'outer;
                    }
                } else {
                    if cat == 2 {
                        meeting_point = Some(((cx, ny), (cx, cy)));
                        break 'outer;
                    }
                }
            }
            // 3. Left Neighbor
            if cx > 0 {
                let nx = cx - 1;
                let offset = cy_offset + (nx as usize);
                let val = buffer[offset];
                let cat = get_category(val);
                if cat == 1 {
                    buffer[offset] = parent_codes[3]; // parent is Right
                    active.next_level.push((nx, cy));
                } else if is_forward {
                    if cat == 3 {
                        meeting_point = Some(((cx, cy), (nx, cy)));
                        break 'outer;
                    }
                } else {
                    if cat == 2 {
                        meeting_point = Some(((nx, cy), (cx, cy)));
                        break 'outer;
                    }
                }
            }
            // 4. Right Neighbor
            if (cx as usize) + 1 < w {
                let nx = cx + 1;
                let offset = cy_offset + (nx as usize);
                let val = buffer[offset];
                let cat = get_category(val);
                if cat == 1 {
                    buffer[offset] = parent_codes[2]; // parent is Left
                    active.next_level.push((nx, cy));
                } else if is_forward {
                    if cat == 3 {
                        meeting_point = Some(((cx, cy), (nx, cy)));
                        break 'outer;
                    }
                } else {
                    if cat == 2 {
                        meeting_point = Some(((nx, cy), (cx, cy)));
                        break 'outer;
                    }
                }
            }
        }
        
        active.level.clear();
        std::mem::swap(&mut active.level, &mut active.next_level);
    }
    
    // If no meeting point was found, no path exists
    let ((fw_x, fw_y), (bw_x, bw_y)) = match meeting_point {
        Some(pt) => pt,
        None => {
            println!("0");
            return Ok(());
        }
    };
    
    // Trace forward path from fw meeting node back to start
    let mut fw_path = Vec::new();
    let (mut px, mut py) = (fw_x, fw_y);
    fw_path.push((px, py));
    while (px, py) != (start_pos.0, start_pos.1) {
        let offset = (py as usize) * stride + (px as usize);
        let parent_dir = buffer[offset];
        match parent_dir {
            b'^' => py -= 1,
            b'v' => py += 1,
            b'<' => px -= 1,
            b'>' => px += 1,
            _ => break,
        }
        fw_path.push((px, py));
    }
    fw_path.reverse();
    
    // Trace backward path from bw meeting node back to end
    let mut bw_path = Vec::new();
    let (mut px, mut py) = (bw_x, bw_y);
    bw_path.push((px, py));
    while (px, py) != (end_pos.0, end_pos.1) {
        let offset = (py as usize) * stride + (px as usize);
        let parent_dir = buffer[offset];
        match parent_dir {
            b'U' => py -= 1,
            b'D' => py += 1,
            b'L' => px -= 1,
            b'R' => px += 1,
            _ => break,
        }
        bw_path.push((px, py));
    }
    
    // Combine forward and backward paths
    let mut path = fw_path;
    path.extend(bw_path);
    
    // Write output to stdout using a large buffer to maximize I/O throughput
    let mut stdout = BufWriter::with_capacity(256 * 1024, io::stdout().lock());
    writeln!(stdout, "{}", path.len())?;
    for (x, y) in path {
        writeln!(stdout, "{} {}", x, y)?;
    }
    
    Ok(())
}
