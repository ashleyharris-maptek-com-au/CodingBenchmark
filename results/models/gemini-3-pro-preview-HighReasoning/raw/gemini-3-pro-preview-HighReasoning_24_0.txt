#include <iostream>
#include <string>
#include <cmath>
#include <algorithm>
#include <iomanip>

const double PI = 3.14159265358979323846;

double wrap_angle(double a) {
    while (a > PI) a -= 2.0 * PI;
    while (a < -PI) a += 2.0 * PI;
    return a;
}

double clamp(double v, double lo, double hi) {
    return std::max(lo, std::min(hi, v));
}

int main() {
    // Speed up standard C++ I/O operations for real-time performance
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    std::cout << std::fixed << std::setprecision(6);

    double width, depth, height, gravity, max_thrust, fuel;
    double start_x, start_y, start_z, target_x, target_y, target_z;
    std::string terrain_file_path;
    std::string state_word;

    // Parse the initial header
    if (!(std::cin >> width >> depth >> height >> gravity >> max_thrust >> fuel)) return 0;
    if (!(std::cin >> start_x >> start_y >> start_z >> target_x >> target_y >> target_z)) return 0;

    std::string dummy;
    std::getline(std::cin, dummy); // Consume the newline after target_z
    if (!std::getline(std::cin, terrain_file_path)) return 0;
    if (!(std::cin >> state_word)) return 0;

    double pad_r = std::max(5.0, std::min(width, depth) * 0.02);
    std::string phase = "CRUISE";

    double x, y, z, vx, vy, vz, pitch, yaw, roll, pitch_rate, yaw_rate, roll_rate, fuel_remaining;
    
    // Continuous real-time control loop
    while (std::cin >> x >> y >> z >> vx >> vy >> vz >> pitch >> yaw >> roll >> pitch_rate >> yaw_rate >> roll_rate >> fuel_remaining) {
        double dx = target_x - x;
        double dy = target_y - y;
        double dz = target_z - z;
        double dh = std::hypot(dx, dy);

        // Transition from CRUISE phase to vertical DESCENT phase when we are close to the target pad
        if (phase == "CRUISE" && dh < pad_r * 0.5) {
            phase = "DESCENT";
        }

        double a_h_safe = 0.15 * max_thrust;
        double dvx = 0.0;
        double dvy = 0.0;
        double danger_x = 0.0;
        double danger_y = 0.0;

        if (phase == "CRUISE") {
            // Horizontal speed target using safe deceleration limits
            double safe_speed = std::sqrt(2.0 * a_h_safe * dh);
            double V_cruise_max = std::min(150.0, 0.12 * std::max(width, depth));
            double speed_target = std::min(V_cruise_max, safe_speed);
            if (dh < 10.0) {
                speed_target = std::min(speed_target, dh * 0.4);
            }
            if (dh > 0.1) {
                dvx = (dx / dh) * speed_target;
                dvy = (dy / dh) * speed_target;
            }
        } else {
            // Precise positioning target when directly descending over the pad
            dvx = clamp(dx * 0.8, -1.5, 1.5);
            dvy = clamp(dy * 0.8, -1.5, 1.5);
        }

        // Limit target velocity by boundary distance based on actual velocity direction
        double boundary_margin = 8.0;
        if (vx > 0) {
            double safe_vx = std::sqrt(2.0 * a_h_safe * std::max(0.1, width - boundary_margin - x));
            dvx = std::min(dvx, safe_vx);
            danger_x = (safe_vx > 0.01) ? (vx / safe_vx) : 0.0;
        } else {
            double safe_vx = std::sqrt(2.0 * a_h_safe * std::max(0.1, x - boundary_margin));
            dvx = std::max(dvx, -safe_vx);
            danger_x = (safe_vx > 0.01) ? (-vx / safe_vx) : 0.0;
        }

        if (vy > 0) {
            double safe_vy = std::sqrt(2.0 * a_h_safe * std::max(0.1, depth - boundary_margin - y));
            dvy = std::min(dvy, safe_vy);
            danger_y = (safe_vy > 0.01) ? (vy / safe_vy) : 0.0;
        } else {
            double safe_vy = std::sqrt(2.0 * a_h_safe * std::max(0.1, y - boundary_margin));
            dvy = std::max(dvy, -safe_vy);
            danger_y = (safe_vy > 0.01) ? (-vy / safe_vy) : 0.0;
        }

        // Vertical guidance
        double dvz = 0.0;
        if (phase == "CRUISE") {
            // Cruise altitude is set to 72% of world height, which is guaranteed to clear overhangs/obstacles
            double z_target = height * 0.72;
            double dz_eff = z_target - z;
            dvz = clamp(dz_eff * 0.5, -20.0, 20.0);
        } else {
            // Smooth vertical landing descent above target pad
            double d_z = std::max(0.0, z - target_z);
            double a_z_safe = 0.35 * (max_thrust - gravity);
            double safe_down_speed = std::sqrt(2.0 * a_z_safe * d_z);
            double V_desc_max = 20.0;
            
            double speed_down = std::min(V_desc_max, safe_down_speed);
            if (d_z > 0.1) {
                speed_down = std::min(speed_down, d_z * 0.25);
            }
            double vz_target = -speed_down;
            if (d_z < 3.0) {
                vz_target = -1.2;
            }
            if (d_z < 1.0) {
                vz_target = -0.8;
            }
            dvz = vz_target;
        }

        // Acceleration tracking
        double K_v = 1.8;
        double K_vz = 1.8;
        double ax_des = K_v * (dvx - vx);
        double ay_des = K_v * (dvy - vy);

        // Danger-based axis prioritization: prioritize the axis closer to its boundary
        if (danger_x > 1.0 || danger_y > 1.0) {
            if (danger_x > danger_y) {
                ay_des *= std::max(0.05, danger_y / danger_x);
            } else {
                ax_des *= std::max(0.05, danger_x / danger_y);
            }
        }

        double az_des = K_vz * (dvz - vz) + gravity;
        az_des = std::max(0.0, az_des);

        // Compute desired thrust vector with dynamic tilt angle limit to prevent altitude climbing when braking
        double theta_max = std::acos(clamp(az_des / max_thrust, 0.0, 1.0));
        theta_max = std::min(theta_max, 88.0 * PI / 180.0);

        double ah = std::hypot(ax_des, ay_des);
        double T_total = std::sqrt(ax_des*ax_des + ay_des*ay_des + az_des*az_des);

        double T_mag = std::min(max_thrust, T_total);
        double T_x = 0.0;
        double T_y = 0.0;
        double T_z = 0.0;

        double theta = std::atan2(ah, az_des);
        if (theta > theta_max) {
            T_z = T_mag * std::cos(theta_max);
            double T_h = T_mag * std::sin(theta_max);
            T_x = (ah > 0.01) ? (T_h * (ax_des / ah)) : 0.0;
            T_y = (ah > 0.01) ? (T_h * (ay_des / ah)) : 0.0;
        } else {
            if (T_total > 0.01) {
                T_x = ax_des * (T_mag / T_total);
                T_y = ay_des * (T_mag / T_total);
                T_z = az_des * (T_mag / T_total);
            } else {
                T_x = 0.0;
                T_y = 0.0;
                T_z = T_mag;
            }
        }

        // Map thrust vector to control commands
        double T_total_actual = std::sqrt(T_x*T_x + T_y*T_y + T_z*T_z);
        double dx_unit = 0.0, dy_unit = 0.0, dz_unit = 1.0;
        if (T_total_actual > 0.01) {
            dx_unit = T_x / T_total_actual;
            dy_unit = T_y / T_total_actual;
            dz_unit = T_z / T_total_actual;
        }

        // Get actual orientation unit vector
        double ax_unit = -std::sin(pitch) * std::cos(yaw);
        double ay_unit = -std::sin(pitch) * std::sin(yaw);
        double az_unit = std::cos(pitch);

        // Attitude alignment scaling: scale down thrust when not aligned with desired thrust direction
        double dot = ax_unit * dx_unit + ay_unit * dy_unit + az_unit * dz_unit;
        double align_factor = std::max(0.0, dot);
        align_factor = align_factor * align_factor;

        double thrust = (T_total_actual * align_factor) / max_thrust;
        double pitch_target = (T_total_actual > 0.01) ? std::acos(clamp(T_z / T_total_actual, 0.0, 1.0)) : 0.0;
        double yaw_target = (T_total_actual > 0.01) ? std::atan2(-T_y, -T_x) : yaw;

        // Yaw and pitch direction optimization (comparing positive vs negative pitch and yaw+pi)
        double p_opt1 = pitch_target;
        double y_opt1 = yaw_target;
        double p_opt2 = -pitch_target;
        double y_opt2 = std::atan2(T_y, T_x);

        double diff1 = std::abs(wrap_angle(y_opt1 - yaw));
        double diff2 = std::abs(wrap_angle(y_opt2 - yaw));

        if (diff1 <= diff2) {
            pitch_target = p_opt1;
            yaw_target = y_opt1;
        } else {
            pitch_target = p_opt2;
            yaw_target = y_opt2;
        }

        // PD control for attitude commands
        double Kp = 9.0;
        double Kd = 3.0;

        double pitch_cmd = Kp * (pitch_target - pitch) - Kd * pitch_rate;
        double yaw_cmd = Kp * wrap_angle(yaw_target - yaw) - Kd * yaw_rate;
        double roll_cmd = -Kd * roll_rate;

        pitch_cmd = clamp(pitch_cmd, -1.0, 1.0);
        yaw_cmd = clamp(yaw_cmd, -1.0, 1.0);
        roll_cmd = clamp(roll_cmd, -1.0, 1.0);

        // Output command in the real-time game's expected format
        std::cout << thrust << " " << pitch_cmd << " " << yaw_cmd << " " << roll_cmd << "\n";
        std::cout.flush();
    }

    return 0;
}
