use std::io::{self, BufRead};
use std::thread;

const INVALID_INDEX: u32 = u32::MAX;
const K_NEIGHBORS: usize = 15;
const OUTLIER_THRESHOLD: f64 = 5.0; // Z-score threshold for outliers
const EPSILON: f64 = 1e-5; // to avoid division by zero in distance weighting

#[allow(dead_code)]
struct Hole {
    id: String,
    start: [f64; 3],
    dir: [f64; 3],
    length: f64,
    num_samples: usize,
    samples_start_idx: usize,
}

struct KdNode {
    point: [f64; 3],
    idx: u32,
    left: u32,
    right: u32,
}

struct KNearest {
    k: usize,
    dists: Vec<f64>,
    indices: Vec<u32>,
}

impl KNearest {
    fn new(k: usize) -> Self {
        Self {
            k,
            dists: Vec::with_capacity(k),
            indices: Vec::with_capacity(k),
        }
    }

    fn clear(&mut self) {
        self.dists.clear();
        self.indices.clear();
    }

    fn insert(&mut self, dist_sq: f64, idx: u32) {
        if self.dists.len() < self.k {
            let pos = match self.dists.binary_search_by(|val| val.partial_cmp(&dist_sq).unwrap_or(std::cmp::Ordering::Equal)) {
                Ok(p) => p,
                Err(p) => p,
            };
            self.dists.insert(pos, dist_sq);
            self.indices.insert(pos, idx);
        } else {
            let max_dist = self.dists[self.k - 1];
            if dist_sq < max_dist {
                self.dists.pop();
                self.indices.pop();
                let pos = match self.dists.binary_search_by(|val| val.partial_cmp(&dist_sq).unwrap_or(std::cmp::Ordering::Equal)) {
                    Ok(p) => p,
                    Err(p) => p,
                };
                self.dists.insert(pos, dist_sq);
                self.indices.insert(pos, idx);
            }
        }
    }

    fn max_dist(&self) -> f64 {
        if self.dists.len() < self.k {
            f64::INFINITY
        } else {
            self.dists[self.k - 1]
        }
    }
}

fn build_kd_tree(
    nodes: &mut Vec<KdNode>,
    indices: &mut [u32],
    samples: &[[f64; 3]],
    depth: usize,
) -> u32 {
    if indices.is_empty() {
        return INVALID_INDEX;
    }

    let axis = depth % 3;
    let median_idx = indices.len() / 2;

    indices.select_nth_unstable_by(median_idx, |&a, &b| {
        let val_a = samples[a as usize][axis];
        let val_b = samples[b as usize][axis];
        val_a.partial_cmp(&val_b).unwrap_or(std::cmp::Ordering::Equal)
    });

    let root_sample_idx = indices[median_idx];
    let node_idx = nodes.len() as u32;

    nodes.push(KdNode {
        point: samples[root_sample_idx as usize],
        idx: root_sample_idx,
        left: INVALID_INDEX,
        right: INVALID_INDEX,
    });

    let (left_indices, right_indices) = indices.split_at_mut(median_idx);
    let right_indices = &mut right_indices[1..];

    let left_child = build_kd_tree(nodes, left_indices, samples, depth + 1);
    let right_child = build_kd_tree(nodes, right_indices, samples, depth + 1);

    nodes[node_idx as usize].left = left_child;
    nodes[node_idx as usize].right = right_child;

    node_idx
}

fn query_kd_tree(
    nodes: &[KdNode],
    node_idx: u32,
    target: [f64; 3],
    depth: usize,
    k_nearest: &mut KNearest,
    target_idx: u32,
) {
    if node_idx == INVALID_INDEX {
        return;
    }

    let node = &nodes[node_idx as usize];
    let axis = depth % 3;
    let dist_sq = sq_dist(node.point, target);

    if node.idx != target_idx {
        k_nearest.insert(dist_sq, node.idx);
    }

    let target_val = target[axis];
    let node_val = node.point[axis];

    let (first, second) = if target_val < node_val {
        (node.left, node.right)
    } else {
        (node.right, node.left)
    };

    query_kd_tree(nodes, first, target, depth + 1, k_nearest, target_idx);

    let axis_dist = (target_val - node_val).abs();
    if axis_dist * axis_dist < k_nearest.max_dist() {
        query_kd_tree(nodes, second, target, depth + 1, k_nearest, target_idx);
    }
}

#[inline(always)]
fn sq_dist(a: [f64; 3], b: [f64; 3]) -> f64 {
    let dx = a[0] - b[0];
    let dy = a[1] - b[1];
    let dz = a[2] - b[2];
    dx * dx + dy * dy + dz * dz
}

fn parse_property(s: &str) -> Option<f64> {
    if s.is_empty() || s == "-" || s == "*" || s.eq_ignore_ascii_case("nan") || s.eq_ignore_ascii_case("na") || s.eq_ignore_ascii_case("null") {
        None
    } else if let Ok(val) = s.parse::<f64>() {
        if (val - -999.0).abs() < 1e-3 || (val - -9999.0).abs() < 1e-3 || (val - -99.0).abs() < 1e-3 {
            None
        } else {
            Some(val)
        }
    } else {
        None
    }
}

fn calculate_median(values: &mut [f64]) -> f64 {
    if values.is_empty() {
        return 0.0;
    }
    let mid = values.len() / 2;
    values.select_nth_unstable_by(mid, |a, b| {
        a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal)
    });
    values[mid]
}

fn calculate_global_stats(values: &[f64]) -> (f64, f64) {
    let mut valid_vals: Vec<f64> = values.iter().copied().filter(|v| !v.is_nan()).collect();
    if valid_vals.is_empty() {
        return (0.0, 0.0);
    }

    let median = calculate_median(&mut valid_vals);
    let mut diffs: Vec<f64> = valid_vals.iter().map(|&v| (v - median).abs()).collect();
    let mad = calculate_median(&mut diffs);

    (median, mad)
}

struct Suspect {
    hole_id: String,
    sample_idx: u32,
    property_name: String,
    confidence: f64,
}

fn main() -> io::Result<()> {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // 1. Read first line
    let first_line = match lines.next() {
        Some(Ok(l)) => l,
        _ => return Err(io::Error::new(io::ErrorKind::InvalidInput, "Missing first line")),
    };
    let mut parts = first_line.split_whitespace();
    let num_holes: usize = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0);
    let num_properties: usize = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0);

    // 2. Read property names
    let second_line = match lines.next() {
        Some(Ok(l)) => l,
        _ => return Err(io::Error::new(io::ErrorKind::InvalidInput, "Missing second line")),
    };
    let property_names: Vec<String> = second_line.split_whitespace().map(|s| s.to_string()).collect();
    if property_names.len() != num_properties {
        return Err(io::Error::new(io::ErrorKind::InvalidInput, "Property name count mismatch"));
    }

    let mut holes = Vec::with_capacity(num_holes);
    let mut sample_positions = Vec::new();
    let mut sample_hole_indices = Vec::new();
    let mut sample_indices = Vec::new();
    let mut property_values = vec![Vec::new(); num_properties];

    let mut hole_idx = 0;

    while let Some(line_res) = lines.next() {
        let line = line_res?;
        let trimmed = line.trim();
        if trimmed.is_empty() {
            continue;
        }

        let mut hole_parts = trimmed.split_whitespace();
        let hole_id = match hole_parts.next() {
            Some(id) => id.to_string(),
            None => break,
        };

        let start_x: f64 = hole_parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);
        let start_y: f64 = hole_parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);
        let start_z: f64 = hole_parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);

        let dir_x: f64 = hole_parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);
        let dir_y: f64 = hole_parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);
        let dir_z: f64 = hole_parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);

        let length: f64 = hole_parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);
        let num_samples: usize = hole_parts.next().and_then(|s| s.parse().ok()).unwrap_or(0);

        let mut dir = [dir_x, dir_y, dir_z];
        let norm = (dir[0] * dir[0] + dir[1] * dir[1] + dir[2] * dir[2]).sqrt();
        if norm > 1e-7 {
            dir[0] /= norm;
            dir[1] /= norm;
            dir[2] /= norm;
        } else {
            dir = [0.0, 0.0, -1.0];
        }

        let samples_start_idx = sample_positions.len();
        let mut read_samples = 0;

        while read_samples < num_samples {
            if let Some(sample_line_res) = lines.next() {
                let sample_line = sample_line_res?;
                let s_trimmed = sample_line.trim();
                if s_trimmed.is_empty() {
                    continue;
                }

                let mut s_parts = s_trimmed.split_whitespace();
                let depth: f64 = match s_parts.next() {
                    Some(d) => d.parse().unwrap_or(0.0),
                    None => continue,
                };

                let px = start_x + depth * dir[0];
                let py = start_y + depth * dir[1];
                let pz = start_z + depth * dir[2];

                sample_positions.push([px, py, pz]);
                sample_hole_indices.push(hole_idx as u32);
                sample_indices.push(read_samples as u32);

                for p in 0..num_properties {
                    let val_str = s_parts.next().unwrap_or("");
                    let val = parse_property(val_str).unwrap_or(f64::NAN);
                    property_values[p].push(val);
                }

                read_samples += 1;
            } else {
                break;
            }
        }

        holes.push(Hole {
            id: hole_id,
            start: [start_x, start_y, start_z],
            dir,
            length,
            num_samples: read_samples,
            samples_start_idx,
        });

        hole_idx += 1;
    }

    let n_samples = sample_positions.len();
    if n_samples == 0 {
        println!("0");
        return Ok(());
    }

    // 3. Compute global stats per property
    let mut global_medians = Vec::with_capacity(num_properties);
    let mut global_mads = Vec::with_capacity(num_properties);
    for p in 0..num_properties {
        let (median, mad) = calculate_global_stats(&property_values[p]);
        global_medians.push(median);
        global_mads.push(mad);
    }

    // 4. Build K-D Tree
    let mut kd_nodes = Vec::with_capacity(n_samples);
    let mut indices: Vec<u32> = (0..n_samples as u32).collect();
    let root = build_kd_tree(&mut kd_nodes, &mut indices, &sample_positions, 0);

    // 5. Determine Scale-Based Strategy
    // For large datasets, we use a 1D check filter to prune definitely-normal samples.
    let run_3d_for_all = n_samples <= 1_000;

    // Helper to determine if a sample needs a 3D query
    let needs_3d = |i: usize| -> bool {
        if run_3d_for_all {
            return true;
        }

        let hole_idx = sample_hole_indices[i];
        for p in 0..num_properties {
            let val = property_values[p][i];
            if val.is_nan() {
                continue;
            }

            let mut neighbors = Vec::new();
            if i > 0 && sample_hole_indices[i - 1] == hole_idx {
                let nval = property_values[p][i - 1];
                if !nval.is_nan() {
                    neighbors.push(nval);
                }
            }
            if i + 1 < sample_hole_indices.len() && sample_hole_indices[i + 1] == hole_idx {
                let nval = property_values[p][i + 1];
                if !nval.is_nan() {
                    neighbors.push(nval);
                }
            }

            if neighbors.is_empty() {
                return true;
            }

            let sum: f64 = neighbors.iter().sum();
            let avg = sum / neighbors.len() as f64;
            let diff = (val - avg).abs();

            // Safe threshold to skip 3D lookup if it's very close to same-hole neighbors
            let limit = 0.4 * global_mads[p] + 1e-5;
            if diff > limit {
                return true;
            }
        }
        false
    };

    // 6. Run queries in parallel using scoped threads
    let num_threads = thread::available_parallelism()
        .map(|n| n.get())
        .unwrap_or(1);
    
    let chunk_size = (n_samples + num_threads - 1) / num_threads;

    let mut suspects = Vec::new();

    thread::scope(|s| {
        let mut handles = Vec::new();

        // Share immutable references across threads
        let sample_positions = &sample_positions;
        let sample_hole_indices = &sample_hole_indices;
        let sample_indices = &sample_indices;
        let property_values = &property_values;
        let property_names = &property_names;
        let global_mads = &global_mads;
        let holes = &holes;
        let kd_nodes = &kd_nodes;

        for chunk_idx in 0..num_threads {
            let start = chunk_idx * chunk_size;
            if start >= n_samples {
                break;
            }
            let end = (start + chunk_size).min(n_samples);

            let handle = s.spawn(move || {
                let mut local_suspects = Vec::new();
                let mut k_nearest = KNearest::new(K_NEIGHBORS);

                for i in start..end {
                    if !needs_3d(i) {
                        continue;
                    }

                    k_nearest.clear();
                    query_kd_tree(kd_nodes, root, sample_positions[i], 0, &mut k_nearest, i as u32);

                    let neighbors = &k_nearest.indices;
                    if neighbors.is_empty() {
                        continue;
                    }

                    let target_pos = sample_positions[i];

                    for p in 0..num_properties {
                        let val_i = property_values[p][i];
                        if val_i.is_nan() {
                            continue;
                        }

                        // Filter neighbors that have valid values
                        let mut valid_neighbors = Vec::new();
                        for &n_idx in neighbors {
                            let val_n = property_values[p][n_idx as usize];
                            if !val_n.is_nan() {
                                valid_neighbors.push((n_idx as usize, val_n));
                            }
                        }

                        if valid_neighbors.len() < 2 {
                            continue;
                        }

                        // Compute IDW prediction
                        let mut sum_w = 0.0;
                        let mut sum_w_val = 0.0;
                        for &(n_idx, val_n) in &valid_neighbors {
                            let d2 = sq_dist(target_pos, sample_positions[n_idx]);
                            let w = 1.0 / (d2 + EPSILON);
                            sum_w += w;
                            sum_w_val += w * val_n;
                        }

                        let expected = sum_w_val / sum_w;

                        // Compute weighted local standard deviation
                        let mut sum_w_diff_sq = 0.0;
                        for &(n_idx, val_n) in &valid_neighbors {
                            let d2 = sq_dist(target_pos, sample_positions[n_idx]);
                            let w = 1.0 / (d2 + EPSILON);
                            let diff = val_n - expected;
                            sum_w_diff_sq += w * diff * diff;
                        }

                        let local_var = sum_w_diff_sq / sum_w;
                        let local_std = local_var.sqrt();

                        // Local scale factor
                        let scale_baseline = 0.1 * global_mads[p] + 1e-6;
                        let local_scale = local_std + scale_baseline;

                        let diff = (val_i - expected).abs();
                        let score = diff / local_scale;

                        // Check minimum difference to avoid noise at detection limits
                        let min_diff = 1e-4_f64.max(0.01 * global_mads[p]);

                        if score > OUTLIER_THRESHOLD && diff > min_diff {
                            let hole_id = holes[sample_hole_indices[i] as usize].id.clone();
                            let s_idx = sample_indices[i];
                            local_suspects.push(Suspect {
                                hole_id,
                                sample_idx: s_idx,
                                property_name: property_names[p].clone(),
                                confidence: score,
                            });
                        }
                    }
                }
                local_suspects
            });
            handles.push(handle);
        }

        for h in handles {
            if let Ok(res) = h.join() {
                suspects.extend(res);
            }
        }
    });

    // 7. Sort suspects by confidence desc
    suspects.sort_by(|a, b| b.confidence.partial_cmp(&a.confidence).unwrap_or(std::cmp::Ordering::Equal));

    // 8. Write output
    println!("{}", suspects.len());
    for s in &suspects {
        println!("{} {} {} {:.4}", s.hole_id, s.sample_idx, s.property_name, s.confidence);
    }

    Ok(())
}
