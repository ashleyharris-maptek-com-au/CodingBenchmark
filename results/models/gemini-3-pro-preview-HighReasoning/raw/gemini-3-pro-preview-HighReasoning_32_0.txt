const long long INF = 1e18;

struct Edge {
    int to;
    long long weight;
    double perturbed_weight;
    int id;
};

struct Transition {
    uint32_t data = 0;
    void set(int type, int val) {
        data = ((uint32_t)type << 30) | (val & 0x3FFFFFFF);
    }
    int getType() const {
        return data >> 30;
    }
    int getVal() const {
        return data & 0x3FFFFFFF;
    }
};

struct SimpleEdge {
    int u, v;
    long long weight;
};

struct DSU {
    std::vector<int> parent;
    DSU(int n) {
        parent.resize(n);
        for (int i = 0; i < n; ++i) parent[i] = i;
    }
    int find(int i) {
        if (parent[i] == i)
            return i;
        return parent[i] = find(parent[i]);
    }
    bool unite(int i, int j) {
        int root_i = find(i);
        int root_j = find(j);
        if (root_i != root_j) {
            parent[root_i] = root_j;
            return true;
        }
        return false;
    }
};

struct InputEdge {
    int u, v;
    long long weight;
};

// Function declarations
void get_edges(int u, int mask, std::vector<std::pair<int, int>>& out_edges, 
               const std::vector<Transition>& trans, int K, int V, const std::vector<int>& term_to_mask);

std::pair<long long, std::vector<std::pair<int, int>>> refine(
    int num_vertices,
    const std::vector<std::pair<int, int>>& tree_edges,
    const std::vector<int>& terminals,
    const std::vector<std::vector<Edge>>& adj
);

// Dreyfus-Wagner solver (exact DP)
std::pair<long long, std::vector<std::pair<int, int>>> solve_exact(
    int V,
    const std::vector<std::vector<Edge>>& adj,
    const std::vector<int>& terminals
) {
    int K = terminals.size();
    if (K <= 1) return {0, {}};
    
    int num_masks = 1 << K;
    std::vector<long long> dp(num_masks * V, INF);
    std::vector<Transition> trans(num_masks * V);
    
    std::vector<int> term_to_mask(V, 0);
    for (int i = 0; i < K; ++i) {
        term_to_mask[terminals[i]] = 1 << i;
    }
    
    // Base cases: single terminal masks
    for (int i = 0; i < K; ++i) {
        int t = terminals[i];
        int mask = 1 << i;
        dp[mask * V + t] = 0;
        
        using State = std::pair<long long, int>;
        std::priority_queue<State, std::vector<State>, std::greater<State>> pq;
        pq.push({0, t});
        
        while (!pq.empty()) {
            auto [d, u] = pq.top();
            pq.pop();
            
            if (d > dp[mask * V + u]) continue;
            
            for (const auto& edge : adj[u]) {
                int v = edge.to;
                long long w = edge.weight;
                if (dp[mask * V + u] + w < dp[mask * V + v]) {
                    dp[mask * V + v] = dp[mask * V + u] + w;
                    trans[mask * V + v].set(2, u);
                    pq.push({dp[mask * V + v], v});
                }
            }
        }
    }
    
    // DP for larger masks
    for (int mask = 1; mask < num_masks; ++mask) {
        if ((mask & (mask - 1)) == 0) continue; // already solved (single terminal)
        
        int lsb_mask = mask & -mask;
        int mask_offset = mask * V;
        
        // Transition 1: Split
        for (int s1 = (mask - 1) & mask; s1 > 0; s1 = (s1 - 1) & mask) {
            if (s1 & lsb_mask) {
                int s2 = mask ^ s1;
                int s1_offset = s1 * V;
                int s2_offset = s2 * V;
                for (int i = 0; i < V; ++i) {
                    long long val = dp[s1_offset + i] + dp[s2_offset + i];
                    if (val < dp[mask_offset + i]) {
                        dp[mask_offset + i] = val;
                        trans[mask_offset + i].set(1, s1);
                    }
                }
            }
        }
        
        // Transition 2: Dijkstra
        using State = std::pair<long long, int>;
        std::priority_queue<State, std::vector<State>, std::greater<State>> pq;
        for (int i = 0; i < V; ++i) {
            if (dp[mask_offset + i] < INF) {
                pq.push({dp[mask_offset + i], i});
            }
        }
        
        while (!pq.empty()) {
            auto [d, u] = pq.top();
            pq.pop();
            
            if (d > dp[mask_offset + u]) continue;
            
            for (const auto& edge : adj[u]) {
                int v = edge.to;
                long long w = edge.weight;
                if (dp[mask_offset + u] + w < dp[mask_offset + v]) {
                    dp[mask_offset + v] = dp[mask_offset + u] + w;
                    trans[mask_offset + v].set(2, u);
                    pq.push({dp[mask_offset + v], v});
                }
            }
        }
    }
    
    // Find best root
    long long best_weight = INF;
    int best_root = -1;
    int all_mask = num_masks - 1;
    for (int i = 0; i < V; ++i) {
        if (dp[all_mask * V + i] < best_weight) {
            best_weight = dp[all_mask * V + i];
            best_root = i;
        }
    }
    
    if (best_root == -1 || best_weight >= INF) return {-1, {}};
    
    std::vector<std::pair<int, int>> edges;
    get_edges(best_root, all_mask, edges, trans, K, V, term_to_mask);
    
    return refine(V, edges, terminals, adj);
}

void get_edges(int u, int mask, std::vector<std::pair<int, int>>& out_edges, 
               const std::vector<Transition>& trans, int K, int V, const std::vector<int>& term_to_mask) {
    if (mask == 0) return;
    
    if ((mask & (mask - 1)) == 0) {
        if (term_to_mask[u] == mask) {
            return;
        }
    }
    
    int idx = mask * V + u;
    int type = trans[idx].getType();
    int val = trans[idx].getVal();
    
    if (type == 1) {
        int s1 = val;
        int s2 = mask ^ s1;
        get_edges(u, s1, out_edges, trans, K, V, term_to_mask);
        get_edges(u, s2, out_edges, trans, K, V, term_to_mask);
    } else if (type == 2) {
        int parent = val;
        out_edges.push_back({u, parent});
        get_edges(parent, mask, out_edges, trans, K, V, term_to_mask);
    }
}

// SPH implementation
std::pair<long long, std::vector<std::pair<int, int>>> run_sph(
    int num_vertices,
    const std::vector<std::vector<Edge>>& adj,
    const std::vector<int>& terminals,
    int start_term_idx
) {
    int K = terminals.size();
    if (K <= 1) return {0, {}};
    
    std::vector<bool> is_terminal(num_vertices, false);
    for (int t : terminals) is_terminal[t] = true;
    
    std::vector<bool> in_tree(num_vertices, false);
    std::vector<double> dist(num_vertices, INF);
    std::vector<int> parent(num_vertices, -1);
    
    std::vector<std::pair<int, int>> tree_edges;
    
    int start_node = terminals[start_term_idx];
    in_tree[start_node] = true;
    dist[start_node] = 0.0;
    parent[start_node] = start_node;
    
    struct State {
        double d;
        int u;
        bool operator>(const State& o) const {
            return d > o.d;
        }
    };
    
    std::priority_queue<State, std::vector<State>, std::greater<State>> pq;
    pq.push({0.0, start_node});
    
    int terminals_connected = 1;
    
    while (!pq.empty() && terminals_connected < K) {
        auto [d, u] = pq.top();
        pq.pop();
        
        if (d > dist[u]) continue;
        
        if (is_terminal[u] && !in_tree[u]) {
            int curr = u;
            std::vector<int> path;
            while (!in_tree[curr]) {
                path.push_back(curr);
                curr = parent[curr];
            }
            int prev = curr;
            for (int i = (int)path.size() - 1; i >= 0; --i) {
                int node = path[i];
                tree_edges.push_back({prev, node});
                in_tree[node] = true;
                dist[node] = 0.0;
                parent[node] = node;
                pq.push({0.0, node});
                prev = node;
            }
            terminals_connected++;
            continue;
        }
        
        for (const auto& edge : adj[u]) {
            int v = edge.to;
            double w = edge.perturbed_weight;
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                parent[v] = u;
                pq.push({dist[v], v});
            }
        }
    }
    
    if (terminals_connected < K) {
        return {-1, {}};
    }
    
    return refine(num_vertices, tree_edges, terminals, adj);
}

std::pair<long long, std::vector<std::pair<int, int>>> refine(
    int num_vertices,
    const std::vector<std::pair<int, int>>& tree_edges,
    const std::vector<int>& terminals,
    const std::vector<std::vector<Edge>>& adj
) {
    if (tree_edges.empty()) return {0, {}};
    
    std::vector<bool> in_sub(num_vertices, false);
    for (const auto& p : tree_edges) {
        in_sub[p.first] = true;
        in_sub[p.second] = true;
    }
    
    std::vector<SimpleEdge> candidates;
    for (int u = 0; u < num_vertices; ++u) {
        if (!in_sub[u]) continue;
        for (const auto& edge : adj[u]) {
            int v = edge.to;
            if (in_sub[v] && u < v) {
                candidates.push_back({u, v, edge.weight});
            }
        }
    }
    
    std::sort(candidates.begin(), candidates.end(), [](const SimpleEdge& a, const SimpleEdge& b) {
        return a.weight < b.weight;
    });
    
    DSU dsu(num_vertices);
    std::vector<SimpleEdge> mst_edges;
    for (const auto& e : candidates) {
        if (dsu.unite(e.u, e.v)) {
            mst_edges.push_back(e);
        }
    }
    
    std::vector<int> degree(num_vertices, 0);
    std::vector<std::unordered_set<int>> active_neighbors(num_vertices);
    for (const auto& e : mst_edges) {
        active_neighbors[e.u].insert(e.v);
        active_neighbors[e.v].insert(e.u);
        degree[e.u]++;
        degree[e.v]++;
    }
    
    std::vector<bool> is_terminal(num_vertices, false);
    for (int t : terminals) is_terminal[t] = true;
    
    std::vector<int> leaf_queue;
    for (int i = 0; i < num_vertices; ++i) {
        if (in_sub[i] && degree[i] == 1 && !is_terminal[i]) {
            leaf_queue.push_back(i);
        }
    }
    
    int head = 0;
    while (head < (int)leaf_queue.size()) {
        int u = leaf_queue[head++];
        if (degree[u] != 1 || is_terminal[u]) continue;
        
        if (active_neighbors[u].empty()) continue;
        int v = *active_neighbors[u].begin();
        
        active_neighbors[u].erase(v);
        active_neighbors[v].erase(u);
        degree[u]--;
        degree[v]--;
        
        if (degree[v] == 1 && !is_terminal[v]) {
            leaf_queue.push_back(v);
        }
    }
    
    long long total_weight = 0;
    std::vector<std::pair<int, int>> final_edges;
    
    auto get_original_weight = [&](int u, int v) -> long long {
        for (const auto& edge : adj[u]) {
            if (edge.to == v) return edge.weight;
        }
        return 0;
    };
    
    for (int u = 0; u < num_vertices; ++u) {
        for (int v : active_neighbors[u]) {
            if (u < v) {
                long long w = get_original_weight(u, v);
                total_weight += w;
                final_edges.push_back({u, v});
            }
        }
    }
    
    return {total_weight, final_edges};
}

long long power3(int k) {
    long long res = 1;
    for (int i = 0; i < k; ++i) res *= 3;
    return res;
}

int main() {
    // Fast I/O
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    auto start_time = std::chrono::high_resolution_clock::now();
    
    int num_vertices, num_edges, num_terminals;
    if (!(std::cin >> num_vertices >> num_edges >> num_terminals)) {
        return 0;
    }
    
    std::vector<InputEdge> input_edges(num_edges);
    for (int i = 0; i < num_edges; ++i) {
        std::cin >> input_edges[i].u >> input_edges[i].v >> input_edges[i].weight;
    }
    
    std::vector<int> terminals(num_terminals);
    for (int i = 0; i < num_terminals; ++i) {
        std::cin >> terminals[i];
    }
    
    if (num_terminals <= 1) {
        std::cout << "0 0\n";
        return 0;
    }
    
    // Sort and deduplicate input edges:
    // First, normalize all edges so that u < v
    for (auto& e : input_edges) {
        if (e.u > e.v) std::swap(e.u, e.v);
    }
    // Filter self-loops
    std::vector<InputEdge> filtered;
    filtered.reserve(num_edges);
    for (const auto& e : input_edges) {
        if (e.u != e.v) {
            filtered.push_back(e);
        }
    }
    // Sort filtered edges by u, then v, then weight
    std::sort(filtered.begin(), filtered.end(), [](const InputEdge& a, const InputEdge& b) {
        if (a.u != b.u) return a.u < b.u;
        if (a.v != b.v) return a.v < b.v;
        return a.weight < b.weight;
    });
    
    // Group by (u, v) and keep only the minimum weight edge
    std::vector<std::vector<Edge>> adj(num_vertices);
    int edge_id_counter = 0;
    for (size_t i = 0; i < filtered.size(); ) {
        size_t j = i;
        while (j < filtered.size() && filtered[j].u == filtered[i].u && filtered[j].v == filtered[i].v) {
            j++;
        }
        int u = filtered[i].u;
        int v = filtered[i].v;
        long long w = filtered[i].weight;
        int id = edge_id_counter++;
        adj[u].push_back({v, w, (double)w, id});
        adj[v].push_back({u, w, (double)w, id});
        i = j;
    }
    int num_unique_edges = edge_id_counter;
    
    // Decide whether to use Exact DP (Dreyfus-Wagner)
    bool use_dp = false;
    if (num_terminals <= 15) {
        long long mem_needed = (1LL << num_terminals) * num_vertices * 12LL;
        long long time_needed = (power3(num_terminals) / 2LL) * num_vertices;
        if (mem_needed <= 250000000LL && time_needed <= 150000000LL) {
            use_dp = true;
        }
    }
    
    long long best_weight = INF;
    std::vector<std::pair<int, int>> best_edges;
    
    if (use_dp) {
        auto [exact_w, exact_edges] = solve_exact(num_vertices, adj, terminals);
        if (exact_w != -1) {
            best_weight = exact_w;
            best_edges = exact_edges;
        }
    }
    
    // Fallback or heuristic enhancement (SPH)
    if (best_weight == INF || !use_dp) {
        // Run SPH from multiple start terminals
        int max_starts = std::min(num_terminals, 50);
        for (int i = 0; i < max_starts; ++i) {
            auto [w, edges] = run_sph(num_vertices, adj, terminals, i);
            if (w != -1 && w < best_weight) {
                best_weight = w;
                best_edges = edges;
            }
        }
        
        // GRASP / Perturbed SPH
        std::mt19937 rng(1337);
        std::uniform_real_distribution<double> perturb_dist(-0.15, 0.15);
        std::uniform_int_distribution<int> start_dist(0, num_terminals - 1);
        
        while (true) {
            auto now = std::chrono::high_resolution_clock::now();
            double elapsed = std::chrono::duration<double>(now - start_time).count();
            if (elapsed > 0.85) break;
            
            // Generate perturbed weights
            std::vector<double> factors(num_unique_edges);
            for (int i = 0; i < num_unique_edges; ++i) {
                factors[i] = 1.0 + perturb_dist(rng);
            }
            for (int u = 0; u < num_vertices; ++u) {
                for (auto& edge : adj[u]) {
                    edge.perturbed_weight = edge.weight * factors[edge.id];
                }
            }
            
            int r_start = start_dist(rng);
            auto [w, edges] = run_sph(num_vertices, adj, terminals, r_start);
            if (w != -1 && w < best_weight) {
                best_weight = w;
                best_edges = edges;
            }
        }
    }
    
    if (best_weight == INF) {
        // Failed to connect, output 0 0
        std::cout << "0 0\n";
    } else {
        std::cout << best_weight << " " << best_edges.size() << "\n";
        for (const auto& edge : best_edges) {
            std::cout << edge.first << " " << edge.second << "\n";
        }
    }
    
    return 0;
}
