using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

namespace DFVSSolver
{
    public class Graph
    {
        public int NumVertices { get; private set; }
        public int MaxVertexId => NumVertices - 1;

        private HashSet<int>[] adj;
        private HashSet<int>[] revAdj;
        private bool[] active;
        private bool[] selfLoop;
        private int activeCount;

        public Graph(int numVertices)
        {
            NumVertices = numVertices;
            adj = new HashSet<int>[numVertices];
            revAdj = new HashSet<int>[numVertices];
            active = new bool[numVertices];
            selfLoop = new bool[numVertices];
            for (int i = 0; i < numVertices; i++)
            {
                adj[i] = new HashSet<int>();
                revAdj[i] = new HashSet<int>();
                active[i] = true;
            }
            activeCount = numVertices;
        }

        public bool IsActive(int v) => active[v];
        public int ActiveCount => activeCount;

        public void AddEdge(int u, int v)
        {
            if (u == v)
            {
                selfLoop[u] = true;
                return;
            }
            adj[u].Add(v);
            revAdj[v].Add(u);
        }

        public void AddSelfLoop(int u)
        {
            selfLoop[u] = true;
        }

        public bool HasSelfLoop(int u) => selfLoop[u];

        public int InDegree(int v) => revAdj[v].Count;
        public int OutDegree(int v) => adj[v].Count;

        public int GetUniquePredecessor(int v)
        {
            foreach (int u in revAdj[v]) return u;
            throw new Exception("No predecessor");
        }

        public int GetUniqueSuccessor(int v)
        {
            foreach (int w in adj[v]) return w;
            throw new Exception("No successor");
        }

        public List<int> GetSuccessors(int v) => new List<int>(adj[v]);
        public List<int> GetPredecessors(int v) => new List<int>(revAdj[v]);

        public List<int> GetNeighbors(int v)
        {
            List<int> neighbors = new List<int>();
            foreach (int u in adj[v]) neighbors.Add(u);
            foreach (int u in revAdj[v])
            {
                if (!adj[v].Contains(u)) neighbors.Add(u);
            }
            return neighbors;
        }

        public void RemoveEdge(int u, int v)
        {
            adj[u].Remove(v);
            revAdj[v].Remove(u);
        }

        public void DeleteVertex(int v)
        {
            if (!active[v]) return;
            active[v] = false;
            activeCount--;
            selfLoop[v] = false;

            foreach (int u in adj[v])
            {
                revAdj[u].Remove(v);
            }
            adj[v].Clear();

            foreach (int u in revAdj[v])
            {
                adj[u].Remove(v);
            }
            revAdj[v].Clear();
        }

        public IEnumerable<int> ActiveVertices
        {
            get
            {
                for (int i = 0; i < NumVertices; i++)
                {
                    if (active[i]) yield return i;
                }
            }
        }

        public Graph Clone()
        {
            Graph g = new Graph(NumVertices);
            g.activeCount = this.activeCount;
            g.selfLoop = (bool[])this.selfLoop.Clone();
            g.active = (bool[])this.active.Clone();
            for (int i = 0; i < NumVertices; i++)
            {
                g.adj[i] = new HashSet<int>(this.adj[i]);
                g.revAdj[i] = new HashSet<int>(this.revAdj[i]);
            }
            return g;
        }
    }

    public class Program
    {
        public static void Main(string[] args)
        {
            // Set input reader with large buffer
            using (var reader = new StreamReader(Console.OpenStandardInput(), System.Text.Encoding.UTF8, true, 1024 * 1024))
            {
                string line = reader.ReadLine();
                if (string.IsNullOrEmpty(line)) return;
                string[] parts = line.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
                if (parts.Length < 2) return;
                int n = int.Parse(parts[0]);
                int m = int.Parse(parts[1]);

                Graph g = new Graph(n);
                for (int i = 0; i < m; i++)
                {
                    line = reader.ReadLine();
                    if (string.IsNullOrEmpty(line)) continue;
                    parts = line.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
                    if (parts.Length < 2) continue;
                    int u = int.Parse(parts[0]);
                    int v = int.Parse(parts[1]);
                    g.AddEdge(u, v);
                }

                List<int> fvs = SolveDFVS(g);

                Console.WriteLine(fvs.Count);
                Console.WriteLine(string.Join(" ", fvs));
            }
        }

        public static List<int> SolveDFVS(Graph g)
        {
            List<int> globalFvs = new List<int>();

            bool progress = true;
            while (progress)
            {
                progress = false;

                // 1. Run reductions
                int beforeActive = g.ActiveCount;
                int beforeFvs = globalFvs.Count;
                Reduce(g, globalFvs);
                if (g.ActiveCount < beforeActive || globalFvs.Count > beforeFvs)
                {
                    progress = true;
                }

                if (g.ActiveCount == 0) break;

                // 2. Compute SCCs
                List<List<int>> sccs = GetSCCsIterative(g);

                // 3. Remove cross-SCC edges
                int[] sccId = new int[g.NumVertices];
                for (int i = 0; i < sccs.Count; i++)
                {
                    foreach (int v in sccs[i])
                    {
                        sccId[v] = i;
                    }
                }

                bool edgesRemoved = false;
                for (int u = 0; u < g.NumVertices; u++)
                {
                    if (!g.IsActive(u)) continue;
                    List<int> successors = g.GetSuccessors(u);
                    foreach (int v in successors)
                    {
                        if (sccId[u] != sccId[v])
                        {
                            g.RemoveEdge(u, v);
                            edgesRemoved = true;
                        }
                    }
                }

                if (edgesRemoved)
                {
                    progress = true;
                }
            }

            if (g.ActiveCount == 0)
            {
                return globalFvs;
            }

            List<List<int>> finalSccs = GetSCCsIterative(g);

            foreach (var scc in finalSccs)
            {
                if (scc.Count <= 1) continue;

                Graph shrunk = ShrinkSCC(g, scc, out int[] originalId);
                List<int> sccSol = SolveSCC(shrunk);

                foreach (int v in sccSol)
                {
                    globalFvs.Add(originalId[v]);
                }
            }

            return globalFvs;
        }

        public static void Reduce(Graph g, List<int> localFvs)
        {
            Queue<int> q = new Queue<int>();
            bool[] inQueue = new bool[g.NumVertices];
            for (int i = 0; i < g.NumVertices; i++)
            {
                if (g.IsActive(i))
                {
                    q.Enqueue(i);
                    inQueue[i] = true;
                }
            }

            while (q.Count > 0)
            {
                int v = q.Dequeue();
                inQueue[v] = false;
                if (!g.IsActive(v)) continue;

                // 1. Self-loop
                if (g.HasSelfLoop(v))
                {
                    localFvs.Add(v);
                    List<int> neighbors = g.GetNeighbors(v);
                    g.DeleteVertex(v);
                    foreach (int u in neighbors)
                    {
                        if (g.IsActive(u) && !inQueue[u])
                        {
                            q.Enqueue(u);
                            inQueue[u] = true;
                        }
                    }
                    continue;
                }

                int inDeg = g.InDegree(v);
                int outDeg = g.OutDegree(v);

                // 2. Degree 0
                if (inDeg == 0 || outDeg == 0)
                {
                    List<int> neighbors = g.GetNeighbors(v);
                    g.DeleteVertex(v);
                    foreach (int u in neighbors)
                    {
                        if (g.IsActive(u) && !inQueue[u])
                        {
                            q.Enqueue(u);
                            inQueue[u] = true;
                        }
                    }
                    continue;
                }

                // 3. In-degree 1
                if (inDeg == 1)
                {
                    int u = g.GetUniquePredecessor(v);
                    List<int> successors = g.GetSuccessors(v);
                    foreach (int w in successors)
                    {
                        if (u != w)
                        {
                            g.AddEdge(u, w);
                        }
                        else
                        {
                            g.AddSelfLoop(u);
                        }
                        if (!inQueue[u]) { q.Enqueue(u); inQueue[u] = true; }
                        if (!inQueue[w]) { q.Enqueue(w); inQueue[w] = true; }
                    }
                    g.DeleteVertex(v);
                    if (!inQueue[u]) { q.Enqueue(u); inQueue[u] = true; }
                    continue;
                }

                // 4. Out-degree 1
                if (outDeg == 1)
                {
                    int w = g.GetUniqueSuccessor(v);
                    List<int> predecessors = g.GetPredecessors(v);
                    foreach (int u in predecessors)
                    {
                        if (u != w)
                        {
                            g.AddEdge(u, w);
                        }
                        else
                        {
                            g.AddSelfLoop(w);
                        }
                        if (!inQueue[u]) { q.Enqueue(u); inQueue[u] = true; }
                        if (!inQueue[w]) { q.Enqueue(w); inQueue[w] = true; }
                    }
                    g.DeleteVertex(v);
                    if (!inQueue[w]) { q.Enqueue(w); inQueue[w] = true; }
                    continue;
                }
            }
        }

        public static int[] GetPostOrder(Graph g)
        {
            int n = g.NumVertices;
            bool[] visited = new bool[n];
            int[] postOrder = new int[n];
            int postOrderCount = 0;

            int[] stack = new int[n];
            int stackPtr = 0;

            List<int>[] adjLists = new List<int>[n];
            int[] nextNeighborIdx = new int[n];

            for (int i = 0; i < n; i++)
            {
                if (g.IsActive(i))
                {
                    adjLists[i] = g.GetSuccessors(i);
                }
            }

            for (int start = 0; start < n; start++)
            {
                if (!g.IsActive(start) || visited[start]) continue;

                stack[stackPtr++] = start;
                visited[start] = true;

                while (stackPtr > 0)
                {
                    int u = stack[stackPtr - 1];
                    List<int> neighbors = adjLists[u];
                    int idx = nextNeighborIdx[u];

                    if (idx < neighbors.Count)
                    {
                        int v = neighbors[idx];
                        nextNeighborIdx[u]++;
                        if (!visited[v])
                        {
                            visited[v] = true;
                            stack[stackPtr++] = v;
                        }
                    }
                    else
                    {
                        stackPtr--;
                        postOrder[postOrderCount++] = u;
                    }
                }
            }

            Array.Resize(ref postOrder, postOrderCount);
            return postOrder;
        }

        public static List<List<int>> GetSCCsIterative(Graph g)
        {
            int n = g.NumVertices;
            int[] postOrder = GetPostOrder(g);

            bool[] visited = new bool[n];
            List<List<int>> sccs = new List<List<int>>();

            List<int>[] revAdjLists = new List<int>[n];
            int[] nextNeighborIdx = new int[n];
            for (int i = 0; i < n; i++)
            {
                if (g.IsActive(i))
                {
                    revAdjLists[i] = g.GetPredecessors(i);
                }
            }

            int[] stack = new int[n];

            for (int i = postOrder.Length - 1; i >= 0; i--)
            {
                int start = postOrder[i];
                if (visited[start]) continue;

                List<int> scc = new List<int>();
                int stackPtr = 0;

                stack[stackPtr++] = start;
                visited[start] = true;

                while (stackPtr > 0)
                {
                    int u = stack[stackPtr - 1];
                    List<int> neighbors = revAdjLists[u];
                    int idx = nextNeighborIdx[u];

                    if (idx < neighbors.Count)
                    {
                        int v = neighbors[idx];
                        nextNeighborIdx[u]++;
                        if (!visited[v])
                        {
                            visited[v] = true;
                            stack[stackPtr++] = v;
                        }
                    }
                    else
                    {
                        stackPtr--;
                        scc.Add(u);
                    }
                }
                sccs.Add(scc);
            }

            return sccs;
        }

        public static Graph ShrinkSCC(Graph globalGraph, List<int> sccVertices, out int[] originalId)
        {
            int K = sccVertices.Count;
            originalId = sccVertices.ToArray();
            Dictionary<int, int> map = new Dictionary<int, int>();
            for (int i = 0; i < K; i++)
            {
                map[sccVertices[i]] = i;
            }

            Graph g = new Graph(K);
            for (int i = 0; i < K; i++)
            {
                int u = sccVertices[i];
                foreach (int v in globalGraph.GetSuccessors(u))
                {
                    if (map.TryGetValue(v, out int newV))
                    {
                        g.AddEdge(i, newV);
                    }
                }
            }
            return g;
        }

        public static List<int> SolveSCC(Graph g)
        {
            List<int> bestFvs = null;
            int n = g.NumVertices;
            int[] dist = new int[n];
            int[] parent = new int[n];
            int[] visited = new int[n];
            int visitToken = 0;
            bool[] matched = new bool[n];
            Search(g, new List<int>(), ref bestFvs, dist, parent, visited, ref visitToken, matched);
            return bestFvs;
        }

        private static void Search(Graph g, List<int> currentFvs, ref List<int> bestFvs, int[] dist, int[] parent, int[] visited, ref int visitToken, bool[] matched)
        {
            if (bestFvs != null && currentFvs.Count >= bestFvs.Count)
            {
                return;
            }

            Graph gCopy = g.Clone();
            List<int> localFvs = new List<int>();

            Reduce(gCopy, localFvs);

            List<int> newFvs = new List<int>(currentFvs);
            newFvs.AddRange(localFvs);

            if (bestFvs != null && newFvs.Count >= bestFvs.Count)
            {
                return;
            }

            if (gCopy.ActiveCount == 0)
            {
                if (bestFvs == null || newFvs.Count < bestFvs.Count)
                {
                    bestFvs = newFvs;
                }
                return;
            }

            int lb = GetLowerBound(gCopy, matched);
            if (bestFvs != null && newFvs.Count + lb >= bestFvs.Count)
            {
                return;
            }

            List<int> cycle = FindShortestCycle(gCopy, dist, parent, visited, ref visitToken);
            if (cycle == null)
            {
                if (bestFvs == null || newFvs.Count < bestFvs.Count)
                {
                    bestFvs = newFvs;
                }
                return;
            }

            var sortedCycle = new List<int>(cycle);
            sortedCycle.Sort((x, y) =>
            {
                int degX = gCopy.InDegree(x) + gCopy.OutDegree(x);
                int degY = gCopy.InDegree(y) + gCopy.OutDegree(y);
                return degY.CompareTo(degX);
            });

            foreach (int v in sortedCycle)
            {
                Graph nextG = gCopy.Clone();
                nextG.DeleteVertex(v);

                List<int> nextFvs = new List<int>(newFvs);
                nextFvs.Add(v);

                Search(nextG, nextFvs, ref bestFvs, dist, parent, visited, ref visitToken, matched);
            }
        }

        public static int GetLowerBound(Graph g, bool[] matched)
        {
            int lb = 0;
            int n = g.NumVertices;
            Array.Clear(matched, 0, n);

            for (int u = 0; u < n; u++)
            {
                if (!g.IsActive(u) || matched[u]) continue;

                foreach (int v in g.GetSuccessors(u))
                {
                    if (matched[v]) continue;
                    if (g.GetSuccessors(v).Contains(u))
                    {
                        matched[u] = true;
                        matched[v] = true;
                        lb++;
                        break;
                    }
                }
            }

            for (int u = 0; u < n; u++)
            {
                if (!g.IsActive(u) || matched[u]) continue;

                bool found3Cycle = false;
                foreach (int v in g.GetSuccessors(u))
                {
                    if (matched[v]) continue;

                    foreach (int w in g.GetSuccessors(v))
                    {
                        if (matched[w]) continue;

                        if (g.GetSuccessors(w).Contains(u))
                        {
                            matched[u] = true;
                            matched[v] = true;
                            matched[w] = true;
                            lb++;
                            found3Cycle = true;
                            break;
                        }
                    }
                    if (found3Cycle) break;
                }
            }

            return lb;
        }

        public static List<int> FindShortestCycle(Graph g, int[] dist, int[] parent, int[] visited, ref int visitToken)
        {
            List<int> bestCycle = null;
            int minLen = int.MaxValue;

            foreach (int start in g.ActiveVertices)
            {
                visitToken++;
                Queue<int> q = new Queue<int>();

                foreach (int next in g.GetSuccessors(start))
                {
                    dist[next] = 1;
                    parent[next] = start;
                    visited[next] = visitToken;
                    q.Enqueue(next);
                }

                bool found = false;
                while (q.Count > 0)
                {
                    int u = q.Dequeue();
                    int d = dist[u];
                    if (d >= minLen - 1) break;

                    foreach (int v in g.GetSuccessors(u))
                    {
                        if (v == start)
                        {
                            minLen = d + 1;
                            List<int> cycle = new List<int>();
                            int curr = u;
                            while (curr != start)
                            {
                                cycle.Add(curr);
                                curr = parent[curr];
                            }
                            cycle.Add(start);
                            cycle.Reverse();
                            bestCycle = cycle;
                            found = true;
                            break;
                        }
                        if (visited[v] != visitToken)
                        {
                            visited[v] = visitToken;
                            dist[v] = d + 1;
                            parent[v] = u;
                            q.Enqueue(v);
                        }
                    }
                    if (found) break;
                }
                if (minLen == 2) break;
            }
            return bestCycle;
        }
    }
