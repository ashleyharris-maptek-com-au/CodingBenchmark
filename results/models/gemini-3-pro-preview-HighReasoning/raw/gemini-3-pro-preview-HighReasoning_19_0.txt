using System;
using System.IO;
using System.Text;
using System.Collections.Generic;
using System.Diagnostics;

namespace MiningProblem
{
    public struct ValuableVoxel
    {
        public int X;
        public int Y;
        public int Z;
        public byte Value;
        public byte Mass;
    }

    public struct HeapNode
    {
        public int CellIdx;
        public int FScore;
        public int GScore;
    }

    public class MinHeap
    {
        private HeapNode[] nodes;
        private int count;

        public MinHeap(int capacity)
        {
            nodes = new HeapNode[capacity];
            count = 0;
        }

        public int Count => count;

        public void Clear()
        {
            count = 0;
        }

        public void Push(int cellIdx, int fScore, int gScore)
        {
            if (count == nodes.Length)
            {
                Array.Resize(ref nodes, nodes.Length * 2);
            }
            nodes[count] = new HeapNode { CellIdx = cellIdx, FScore = fScore, GScore = gScore };
            SiftUp(count);
            count++;
        }

        public HeapNode Pop()
        {
            HeapNode result = nodes[0];
            count--;
            if (count > 0)
            {
                nodes[0] = nodes[count];
                SiftDown(0);
            }
            return result;
        }

        private void SiftUp(int idx)
        {
            while (idx > 0)
            {
                int parentIdx = (idx - 1) / 2;
                if (nodes[idx].FScore >= nodes[parentIdx].FScore)
                    break;
                HeapNode temp = nodes[idx];
                nodes[idx] = nodes[parentIdx];
                nodes[parentIdx] = temp;
                idx = parentIdx;
            }
        }

        private void SiftDown(int idx)
        {
            while (true)
            {
                int left = 2 * idx + 1;
                int right = 2 * idx + 2;
                int smallest = idx;
                if (left < count && nodes[left].FScore < nodes[smallest].FScore)
                    smallest = left;
                if (right < count && nodes[right].FScore < nodes[smallest].FScore)
                    smallest = right;
                if (smallest == idx)
                    break;
                HeapNode temp = nodes[idx];
                nodes[idx] = nodes[smallest];
                nodes[smallest] = temp;
                idx = smallest;
            }
        }
    }

    public struct ColumnTarget
    {
        public int X;
        public int Y;
        public int OptimalZ;
        public long EstProfit;
    }

    public struct HeightMod
    {
        public int CellIdx;
        public int OldHeight;
    }

    public class PlannedOp
    {
        public int DigX;
        public int DigY;
        public int DigZ;
        public List<int> Path;
        public bool IsDeliver;
        public int DumpX;
        public int DumpY;
        public byte VoxelByte;
    }

    public class MiningSolver
    {
        private int XVoxelCount;
        private int YVoxelCount;
        private int ZVoxelCount;

        private byte[][] slices;
        private int[] H;
        private bool[] hasValuableOre;

        private int[] dist;
        private int[] distParent;

        private int[] visited;
        private int[] gScore;
        private int[] parent;
        private int[] nodeHeight;
        private int[] depth;
        private int[] bfsQueue;

        private int currentRunId = 0;
        private MinHeap minHeap;

        private int[] dx = { 1, -1, 0, 0 };
        private int[] dy = { 0, 0, 1, -1 };

        private int maxDumpBfsDepth;

        private List<ValuableVoxel> valuableVoxels = new List<ValuableVoxel>();
        private List<string> operations = new List<string>();

        public MiningSolver(int xCount, int yCount, int zCount)
        {
            XVoxelCount = xCount;
            YVoxelCount = yCount;
            ZVoxelCount = zCount;

            slices = new byte[XVoxelCount][];
            int numCells = XVoxelCount * YVoxelCount;
            H = new int[numCells];
            hasValuableOre = new bool[numCells];

            dist = new int[numCells];
            distParent = new int[numCells];
            for (int i = 0; i < distParent.Length; i++) distParent[i] = -1;

            visited = new int[numCells];
            gScore = new int[numCells];
            parent = new int[numCells];
            nodeHeight = new int[numCells];
            depth = new int[numCells];
            bfsQueue = new int[numCells];

            minHeap = new MinHeap(1024);

            long volume = (long)XVoxelCount * YVoxelCount * ZVoxelCount;
            if (volume < 1000000)
            {
                maxDumpBfsDepth = 100;
            }
            else
            {
                maxDumpBfsDepth = 30;
            }
        }

        public void SetVoxel(int x, int y, int z, byte d)
        {
            if (slices[x] == null)
            {
                if (d == 0) return;
                slices[x] = new byte[YVoxelCount * ZVoxelCount];
            }
            slices[x][y * ZVoxelCount + z] = d;
        }

        public byte GetVoxel(int x, int y, int z)
        {
            if (slices[x] == null) return 0;
            return slices[x][y * ZVoxelCount + z];
        }

        public static void Solve()
        {
            Stream stdin = Console.OpenStandardInput();
            string header = ReadHeaderLine(stdin);
            if (string.IsNullOrEmpty(header)) return;

            string[] parts = header.Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
            int xCount = int.Parse(parts[0]);
            int yCount = int.Parse(parts[1]);
            int zCount = int.Parse(parts[2]);

            MiningSolver solver = new MiningSolver(xCount, yCount, zCount);
            solver.ReadGrid(stdin);
            solver.ProcessMining();
            solver.WriteOutput();
        }

        private static string ReadHeaderLine(Stream stream)
        {
            List<byte> bytes = new List<byte>();
            while (true)
            {
                int b = stream.ReadByte();
                if (b == -1) break;
                if (b == '\n') break;
                if (b != '\r')
                {
                    bytes.Add((byte)b);
                }
            }
            return Encoding.ASCII.GetString(bytes.ToArray());
        }

        private static void ReadExactly(Stream stream, byte[] buffer, int count)
        {
            int offset = 0;
            while (offset < count)
            {
                int read = stream.Read(buffer, offset, count - offset);
                if (read <= 0)
                    throw new EndOfStreamException();
                offset += read;
            }
        }

        private void ReadGrid(Stream stdin)
        {
            byte[] sliceBuffer = new byte[YVoxelCount * ZVoxelCount];
            for (int x = 0; x < XVoxelCount; x++)
            {
                ReadExactly(stdin, sliceBuffer, sliceBuffer.Length);

                bool isEmpty = true;
                for (int i = 0; i < sliceBuffer.Length; i++)
                {
                    if (sliceBuffer[i] > 0)
                    {
                        isEmpty = false;
                        break;
                    }
                }

                if (!isEmpty)
                {
                    slices[x] = new byte[YVoxelCount * ZVoxelCount];
                    Array.Copy(sliceBuffer, slices[x], sliceBuffer.Length);

                    for (int y = 0; y < YVoxelCount; y++)
                    {
                        int colIdx = x * YVoxelCount + y;
                        H[colIdx] = -1;
                        for (int z = ZVoxelCount - 1; z >= 0; z--)
                        {
                            int idx = y * ZVoxelCount + z;
                            byte d = sliceBuffer[idx];
                            if (d > 0)
                            {
                                H[colIdx] = z;
                                break;
                            }
                        }

                        // Don't mine column (0,0)
                        if (x == 0 && y == 0) continue;

                        for (int z = 0; z < ZVoxelCount; z++)
                        {
                            int idx = y * ZVoxelCount + z;
                            byte d = sliceBuffer[idx];
                            if (d > 0)
                            {
                                int val = d / 16;
                                if (val > 0)
                                {
                                    valuableVoxels.Add(new ValuableVoxel
                                    {
                                        X = x,
                                        Y = y,
                                        Z = z,
                                        Value = (byte)val,
                                        Mass = (byte)(d % 16)
                                    });
                                    hasValuableOre[colIdx] = true;
                                }
                            }
                        }
                    }
                }
                else
                {
                    for (int y = 0; y < YVoxelCount; y++)
                    {
                        H[x * YVoxelCount + y] = -1;
                    }
                }
            }
        }

        private void RunBackwardDijkstra()
        {
            for (int i = 0; i < dist.Length; i++) dist[i] = int.MaxValue;
            minHeap.Clear();

            dist[0] = 0;
            minHeap.Push(0, 0, 0);

            while (minHeap.Count > 0)
            {
                var node = minHeap.Pop();
                int currIdx = node.CellIdx;
                int currD = node.FScore;

                if (currD > dist[currIdx])
                    continue;

                int currX = currIdx / YVoxelCount;
                int currY = currIdx % YVoxelCount;
                int currH = H[currIdx];

                for (int i = 0; i < 4; i++)
                {
                    int nx = currX + dx[i];
                    int ny = currY + dy[i];
                    if (nx >= 0 && nx < XVoxelCount && ny >= 0 && ny < YVoxelCount)
                    {
                        int nIdx = nx * YVoxelCount + ny;
                        int nH = H[nIdx];
                        if (Math.Abs(nH - currH) <= 1)
                        {
                            int edgeCost = 1;
                            if (currH - nH == 1) edgeCost = 5;
                            else if (currH - nH == -1) edgeCost = 0;

                            int nextD = currD + edgeCost;
                            if (nextD < dist[nIdx])
                            {
                                dist[nIdx] = nextD;
                                distParent[nIdx] = currIdx;
                                minHeap.Push(nIdx, nextD, nextD);
                            }
                        }
                    }
                }
            }
        }

        private double GetHeuristic(int x1, int y1, int h1, int x2, int y2, int h2)
        {
            int dx = Math.Abs(x1 - x2);
            int dy = Math.Abs(y1 - y2);
            int D = dx + dy;
            int dz = h2 - h1;
            if (dz >= 0)
            {
                return D > dz ? D + 4 * dz : 5 * dz;
            }
            else
            {
                return Math.Max(0, D + dz);
            }
        }

        private bool VerifyDijkstraPath(int startIdx, out List<int> path, out int cost)
        {
            path = new List<int>();
            cost = 0;

            int curr = startIdx;
            path.Add(curr);

            while (curr != 0)
            {
                int next = distParent[curr];
                if (next == -1)
                    return false;

                int currH = H[curr];
                int nextH = H[next];
                if (Math.Abs(nextH - currH) > 1)
                    return false;

                int stepCost = 1;
                if (nextH - currH == 1) stepCost = 5;
                else if (nextH - currH == -1) stepCost = 0;

                cost += stepCost;
                path.Add(next);
                curr = next;
            }

            return true;
        }

        private bool FindPathToPlantFast(int startX, int startY, int startH, out List<int> path, out int cost)
        {
            path = null;
            cost = int.MaxValue;

            int bestCost = int.MaxValue;
            List<int> bestPath = null;

            int startIdx = startX * YVoxelCount + startY;

            for (int i = 0; i < 4; i++)
            {
                int nx = startX + dx[i];
                int ny = startY + dy[i];
                if (nx >= 0 && nx < XVoxelCount && ny >= 0 && ny < YVoxelCount)
                {
                    int nIdx = nx * YVoxelCount + ny;
                    int nH = H[nIdx];
                    if (Math.Abs(nH - startH) <= 1)
                    {
                        int firstStepCost = 1;
                        if (nH - startH == 1) firstStepCost = 5;
                        else if (nH - startH == -1) firstStepCost = 0;

                        if (VerifyDijkstraPath(nIdx, out List<int> restPath, out int restCost))
                        {
                            int totalCost = firstStepCost + restCost;
                            if (totalCost < bestCost)
                            {
                                bestCost = totalCost;
                                bestPath = new List<int> { startIdx };
                                bestPath.AddRange(restPath);
                            }
                        }
                    }
                }
            }

            if (bestPath != null)
            {
                path = bestPath;
                cost = bestCost;
                return true;
            }

            return FindPathToPlantAStar(startX, startY, startH, out path, out cost);
        }

        private bool FindPathToPlantAStar(int startX, int startY, int startH, out List<int> path, out int cost)
        {
            path = null;
            cost = int.MaxValue;

            currentRunId++;

            int startIdx = startX * YVoxelCount + startY;
            visited[startIdx] = currentRunId;
            gScore[startIdx] = 0;
            parent[startIdx] = -1;

            minHeap.Clear();
            int startF = (int)GetHeuristic(startX, startY, startH, 0, 0, H[0]);
            minHeap.Push(startIdx, startF, 0);

            nodeHeight[startIdx] = startH;

            bool found = false;
            while (minHeap.Count > 0)
            {
                var node = minHeap.Pop();
                int currIdx = node.CellIdx;
                if (node.GScore > gScore[currIdx])
                    continue;

                if (currIdx == 0)
                {
                    found = true;
                    break;
                }

                int currG = gScore[currIdx];
                int currX = currIdx / YVoxelCount;
                int currY = currIdx % YVoxelCount;
                int currH = nodeHeight[currIdx];

                for (int i = 0; i < 4; i++)
                {
                    int nx = currX + dx[i];
                    int ny = currY + dy[i];
                    if (nx >= 0 && nx < XVoxelCount && ny >= 0 && ny < YVoxelCount)
                    {
                        int nIdx = nx * YVoxelCount + ny;
                        int nH = H[nIdx];
                        if (Math.Abs(nH - currH) <= 1)
                        {
                            int edgeCost = 1;
                            if (nH - currH == 1) edgeCost = 5;
                            else if (nH - currH == -1) edgeCost = 0;

                            int nextG = currG + edgeCost;
                            if (visited[nIdx] != currentRunId || nextG < gScore[nIdx])
                            {
                                visited[nIdx] = currentRunId;
                                gScore[nIdx] = nextG;
                                nodeHeight[nIdx] = nH;
                                parent[nIdx] = currIdx;
                                int nextF = nextG + (int)GetHeuristic(nx, ny, nH, 0, 0, H[0]);
                                minHeap.Push(nIdx, nextF, nextG);
                            }
                        }
                    }
                }
            }

            if (found)
            {
                path = new List<int>();
                int curr = 0;
                while (curr != -1)
                {
                    path.Add(curr);
                    curr = parent[curr];
                }
                path.Reverse();
                cost = gScore[0];
                return true;
            }

            return false;
        }

        private int FindDumpSite(int startX, int startY, int startH, out List<int> path)
        {
            path = null;
            currentRunId++;

            int startIdx = startX * YVoxelCount + startY;
            visited[startIdx] = currentRunId;
            gScore[startIdx] = startH;
            parent[startIdx] = -1;
            depth[startIdx] = 0;

            int qHead = 0;
            int qTail = 0;
            bfsQueue[qTail++] = startIdx;

            int foundIdx = -1;

            while (qHead < qTail)
            {
                int currIdx = bfsQueue[qHead++];
                int currX = currIdx / YVoxelCount;
                int currY = currIdx % YVoxelCount;
                int currH = gScore[currIdx];

                // Valid dump site check
                if (currIdx != 0 && !hasValuableOre[currIdx] && H[currIdx] + 1 < ZVoxelCount)
                {
                    foundIdx = currIdx;
                    break;
                }

                int currDepth = depth[currIdx];
                if (currDepth >= maxDumpBfsDepth)
                    continue;

                for (int i = 0; i < 4; i++)
                {
                    int nx = currX + dx[i];
                    int ny = currY + dy[i];
                    if (nx >= 0 && nx < XVoxelCount && ny >= 0 && ny < YVoxelCount)
                    {
                        int nIdx = nx * YVoxelCount + ny;
                        if (visited[nIdx] != currentRunId)
                        {
                            int nH = H[nIdx];
                            if (Math.Abs(nH - currH) <= 1)
                            {
                                visited[nIdx] = currentRunId;
                                gScore[nIdx] = nH;
                                parent[nIdx] = currIdx;
                                depth[nIdx] = currDepth + 1;
                                bfsQueue[qTail++] = nIdx;

                                if (qTail >= bfsQueue.Length)
                                    break;
                            }
                        }
                    }
                }

                if (qTail >= bfsQueue.Length)
                    break;
            }

            if (foundIdx != -1)
            {
                path = new List<int>();
                int curr = foundIdx;
                while (curr != -1)
                {
                    path.Add(curr);
                    curr = parent[curr];
                }
                path.Reverse();
                return foundIdx;
            }

            return -1;
        }

        private void ProcessMining()
        {
            var stopwatch = Stopwatch.StartNew();

            RunBackwardDijkstra();

            List<ColumnTarget> targets = new List<ColumnTarget>();
            for (int x = 0; x < XVoxelCount; x++)
            {
                for (int y = 0; y < YVoxelCount; y++)
                {
                    int colIdx = x * YVoxelCount + y;
                    if (!hasValuableOre[colIdx]) continue;

                    int dCost = dist[colIdx];
                    if (dCost == int.MaxValue) continue;

                    long maxProfit = 0;
                    int optimalZ = -1;
                    long currentProfit = 0;

                    for (int z = H[colIdx]; z >= 0; z--)
                    {
                        byte d = GetVoxel(x, y, z);
                        if (d > 0)
                        {
                            int mass = d % 16;
                            int val = d / 16;
                            if (val > 0)
                            {
                                currentProfit += (long)mass * (1000 * val - dCost);
                            }
                            else
                            {
                                currentProfit -= (long)mass * 2;
                            }
                        }

                        if (currentProfit > maxProfit)
                        {
                            maxProfit = currentProfit;
                            optimalZ = z;
                        }
                    }

                    if (maxProfit > 0)
                    {
                        targets.Add(new ColumnTarget
                        {
                            X = x,
                            Y = y,
                            OptimalZ = optimalZ,
                            EstProfit = maxProfit
                        });
                    }
                }
            }

            // Sort targets by estimated profit in descending order
            targets.Sort((a, b) => b.EstProfit.CompareTo(a.EstProfit));

            List<HeightMod> heightMods = new List<HeightMod>();
            List<PlannedOp> colOps = new List<PlannedOp>();

            for (int i = 0; i < targets.Count; i++)
            {
                // Timeout check: leave 30 seconds safety margin (4.5 minutes limit)
                if (stopwatch.ElapsedMilliseconds > 270000)
                {
                    break;
                }

                var target = targets[i];
                int colIdx = target.X * YVoxelCount + target.Y;

                if (H[colIdx] < target.OptimalZ)
                    continue;

                bool planSuccess = true;
                int startH = H[colIdx];
                heightMods.Clear();
                colOps.Clear();

                for (int z_curr = startH; z_curr >= target.OptimalZ; z_curr--)
                {
                    byte d = GetVoxel(target.X, target.Y, z_curr);
                    if (d == 0) continue;

                    int mass = d % 16;
                    int val = d / 16;

                    int newColH = z_curr - 1;
                    heightMods.Add(new HeightMod { CellIdx = colIdx, OldHeight = H[colIdx] });
                    H[colIdx] = newColH;

                    if (val > 0)
                    {
                        if (FindPathToPlantFast(target.X, target.Y, newColH, out List<int> path, out int pathCost))
                        {
                            colOps.Add(new PlannedOp
                            {
                                DigX = target.X,
                                DigY = target.Y,
                                DigZ = z_curr,
                                Path = path,
                                IsDeliver = true,
                                VoxelByte = d
                            });
                        }
                        else
                        {
                            planSuccess = false;
                            break;
                        }
                    }
                    else
                    {
                        int dumpIdx = FindDumpSite(target.X, target.Y, newColH, out List<int> path);
                        if (dumpIdx != -1)
                        {
                            int dx = dumpIdx / YVoxelCount;
                            int dy = dumpIdx % YVoxelCount;

                            heightMods.Add(new HeightMod { CellIdx = dumpIdx, OldHeight = H[dumpIdx] });
                            H[dumpIdx] = H[dumpIdx] + 1;

                            colOps.Add(new PlannedOp
                            {
                                DigX = target.X,
                                DigY = target.Y,
                                DigZ = z_curr,
                                Path = path,
                                IsDeliver = false,
                                DumpX = dx,
                                DumpY = dy,
                                VoxelByte = d
                            });
                        }
                        else
                        {
                            planSuccess = false;
                            break;
                        }
                    }
                }

                if (planSuccess)
                {
                    // Commit
                    foreach (var op in colOps)
                    {
                        SetVoxel(op.DigX, op.DigY, op.DigZ, 0);

                        if (!op.IsDeliver)
                        {
                            int dumpIdx = op.DumpX * YVoxelCount + op.DumpY;
                            int dumpH = H[dumpIdx];
                            SetVoxel(op.DumpX, op.DumpY, dumpH, op.VoxelByte);
                        }

                        StringBuilder sb = new StringBuilder();
                        sb.Append(op.Path.Count - 1);
                        sb.Append(" ");
                        sb.Append(op.DigX);
                        sb.Append(" ");
                        sb.Append(op.DigY);
                        sb.Append(" ");
                        sb.Append(op.DigZ);
                        for (int k = 1; k < op.Path.Count; k++)
                        {
                            int idx = op.Path[k];
                            int x = idx / YVoxelCount;
                            int y = idx % YVoxelCount;
                            sb.Append(" ");
                            sb.Append(x);
                            sb.Append(" ");
                            sb.Append(y);
                        }
                        operations.Add(sb.ToString());
                    }
                    heightMods.Clear();

                    bool hasVal = false;
                    for (int z = H[colIdx]; z >= 0; z--)
                    {
                        if (GetVoxel(target.X, target.Y, z) / 16 > 0)
                        {
                            hasVal = true;
                            break;
                        }
                    }
                    hasValuableOre[colIdx] = hasVal;
                }
                else
                {
                    // Rollback
                    for (int k = heightMods.Count - 1; k >= 0; k--)
                    {
                        H[heightMods[k].CellIdx] = heightMods[k].OldHeight;
                    }
                    heightMods.Clear();
                }
            }
        }

        private void WriteOutput()
        {
            using (var writer = new StreamWriter(Console.OpenStandardOutput(), Encoding.ASCII, 65536))
            {
                writer.WriteLine(operations.Count);
                foreach (var op in operations)
                {
                    writer.WriteLine(op);
                }
            }
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            MiningSolver.Solve();
        }
    }
