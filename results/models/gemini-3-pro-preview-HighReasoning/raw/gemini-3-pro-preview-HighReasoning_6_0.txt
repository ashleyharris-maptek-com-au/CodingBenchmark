using System;
using System.Text.RegularExpressions;
using System.Collections.Generic;
using System.Globalization;

class Program
{
    const double MU_DEFAULT = 398600.4418;

    public struct OrbitState
    {
        public double x, y, z;
        public double vx, vy, vz;
    }

    public struct OrbitElements
    {
        public double a;
        public double ex, ey, ez;
        public double hx, hy, hz;
        public double h_mag;
    }

    static double Clamp(double value, double min, double max)
    {
        if (value < min) return min;
        if (value > max) return max;
        return value;
    }

    static double[] ParseDoubles(string line)
    {
        var matches = Regex.Matches(line, @"[-+]?[0-9]*\.?[0-9]+(?:[eE][-+]?[0-9]+)?");
        List<double> list = new List<double>();
        foreach (Match m in matches)
        {
            double val;
            if (double.TryParse(m.Value, NumberStyles.Float, CultureInfo.InvariantCulture, out val))
            {
                list.Add(val);
            }
        }
        return list.ToArray();
    }

    static OrbitElements GetElements(OrbitState state, double mu)
    {
        double r_mag = Math.Sqrt(state.x * state.x + state.y * state.y + state.z * state.z);
        double v_mag = Math.Sqrt(state.vx * state.vx + state.vy * state.vy + state.vz * state.vz);

        r_mag = Math.Max(r_mag, 1e-6);

        double inv_a = 2.0 / r_mag - v_mag * v_mag / mu;
        double a = inv_a > 1e-12 ? 1.0 / inv_a : 1e12;

        double hx = state.y * state.vz - state.z * state.vy;
        double hy = state.z * state.vx - state.x * state.vz;
        double hz = state.x * state.vy - state.y * state.vx;
        double h_mag = Math.Sqrt(hx * hx + hy * hy + hz * hz);
        h_mag = Math.Max(h_mag, 1e-12);

        double rxv = state.x * state.vx + state.y * state.vy + state.z * state.vz;
        double ex = ((v_mag * v_mag - mu / r_mag) * state.x - rxv * state.vx) / mu;
        double ey = ((v_mag * v_mag - mu / r_mag) * state.y - rxv * state.vy) / mu;
        double ez = ((v_mag * v_mag - mu / r_mag) * state.z - rxv * state.vz) / mu;

        return new OrbitElements
        {
            a = a,
            ex = ex, ey = ey, ez = ez,
            hx = hx, hy = hy, hz = hz,
            h_mag = h_mag
        };
    }

    static double GetTransferCost(OrbitElements o1, OrbitElements o2, double mu)
    {
        double cos_theta = (o1.hx * o2.hx + o1.hy * o2.hy + o1.hz * o2.hz) / (o1.h_mag * o2.h_mag);
        double theta = Math.Acos(Clamp(cos_theta, -1.0, 1.0));
        double v_avg = 0.5 * (Math.Sqrt(mu / Math.Max(o1.a, 1.0)) + Math.Sqrt(mu / Math.Max(o2.a, 1.0)));
        double dv_plane = 2.0 * v_avg * Math.Sin(theta / 2.0);

        double v_c1 = Math.Sqrt(mu / Math.Max(o1.a, 1.0));
        double v_c2 = Math.Sqrt(mu / Math.Max(o2.a, 1.0));
        double a_trans = 0.5 * (o1.a + o2.a);
        double v_trans1 = Math.Sqrt(mu * Math.Max(2.0 / o1.a - 1.0 / a_trans, 0.0));
        double v_trans2 = Math.Sqrt(mu * Math.Max(2.0 / o2.a - 1.0 / a_trans, 0.0));
        double dv_hohmann = Math.Abs(v_trans1 - v_c1) + Math.Abs(v_c2 - v_trans2);

        double dex = o1.ex - o2.ex;
        double dey = o1.ey - o2.ey;
        double dez = o1.ez - o2.ez;
        double de_mag = Math.Sqrt(dex * dex + dey * dey + dez * dez);
        double dv_ecc = v_avg * de_mag;

        double dv_in_plane = dv_hohmann + dv_ecc;
        return Math.Sqrt(dv_in_plane * dv_in_plane + dv_plane * dv_plane);
    }

    static int[] SolveDP(double[,] costMatrix, int N)
    {
        int numStates = 1 << N;
        double[,] dp = new double[numStates, N];
        int[,] parent = new int[numStates, N];

        for (int i = 0; i < numStates; i++)
        {
            for (int j = 0; j < N; j++)
            {
                dp[i, j] = double.MaxValue;
                parent[i, j] = -1;
            }
        }

        for (int j = 0; j < N; j++)
        {
            dp[1 << j, j] = costMatrix[N, j];
        }

        for (int mask = 1; mask < numStates; mask++)
        {
            for (int last = 0; last < N; last++)
            {
                double currentVal = dp[mask, last];
                if (currentVal == double.MaxValue) continue;

                for (int next = 0; next < N; next++)
                {
                    if ((mask & (1 << next)) != 0) continue;
                    int nextMask = mask | (1 << next);
                    double nextCost = currentVal + costMatrix[last, next];
                    if (nextCost < dp[nextMask, next])
                    {
                        dp[nextMask, next] = nextCost;
                        parent[nextMask, next] = last;
                    }
                }
            }
        }

        double minTotalCost = double.MaxValue;
        int bestLast = -1;
        int finalMask = numStates - 1;
        for (int j = 0; j < N; j++)
        {
            if (dp[finalMask, j] < minTotalCost)
            {
                minTotalCost = dp[finalMask, j];
                bestLast = j;
            }
        }

        int[] path = new int[N];
        int currMask = finalMask;
        int currNode = bestLast;
        for (int i = N - 1; i >= 0; i--)
        {
            path[i] = currNode;
            int prevNode = parent[currMask, currNode];
            currMask &= ~(1 << currNode);
            currNode = prevNode;
        }

        return path;
    }

    static int[] SolveSA(double[,] costMatrix, int N, int seed)
    {
        int[] currentTour = new int[N];
        bool[] visited = new bool[N];
        int curr = N;
        for (int i = 0; i < N; i++)
        {
            double minCost = double.MaxValue;
            int bestNext = -1;
            for (int j = 0; j < N; j++)
            {
                if (!visited[j] && costMatrix[curr, j] < minCost)
                {
                    minCost = costMatrix[curr, j];
                    bestNext = j;
                }
            }
            currentTour[i] = bestNext;
            visited[bestNext] = true;
            curr = bestNext;
        }

        double currentCost = GetTourCost(currentTour, costMatrix, N);

        int[] bestTour = (int[])currentTour.Clone();
        double bestCost = currentCost;

        Random rand = new Random(seed);

        double T = 100.0;
        double T_min = 1e-6;
        double alpha = 0.9995;
        int stepsPerTemp = 200;

        while (T > T_min)
        {
            for (int step = 0; step < stepsPerTemp; step++)
            {
                int moveType = rand.Next(10);
                if (moveType < 8)
                {
                    int i = rand.Next(N);
                    int j = rand.Next(N);
                    if (i > j) { int tmp = i; i = j; j = tmp; }
                    if (i == j) continue;

                    double oldEdges = 0;
                    double newEdges = 0;

                    int prev = (i == 0) ? N : currentTour[i - 1];
                    oldEdges += costMatrix[prev, currentTour[i]];
                    newEdges += costMatrix[prev, currentTour[j]];

                    if (j < N - 1)
                    {
                        int next = currentTour[j + 1];
                        oldEdges += costMatrix[currentTour[j], next];
                        newEdges += costMatrix[currentTour[i], next];
                    }

                    double delta = newEdges - oldEdges;

                    if (delta < 0 || Math.Exp(-delta / T) > rand.NextDouble())
                    {
                        Array.Reverse(currentTour, i, j - i + 1);
                        currentCost += delta;

                        if (currentCost < bestCost)
                        {
                            bestCost = currentCost;
                            Array.Copy(currentTour, bestTour, N);
                        }
                    }
                }
                else
                {
                    int i = rand.Next(N);
                    int j = rand.Next(N);
                    if (i == j) continue;
                    if (i > j) { int tmp = i; i = j; j = tmp; }

                    double oldEdges = 0;
                    double newEdges = 0;

                    if (j == i + 1)
                    {
                        int prev = (i == 0) ? N : currentTour[i - 1];
                        oldEdges += costMatrix[prev, currentTour[i]] + costMatrix[currentTour[i], currentTour[j]];
                        newEdges += costMatrix[prev, currentTour[j]] + costMatrix[currentTour[j], currentTour[i]];

                        if (j < N - 1)
                        {
                            int next = currentTour[j + 1];
                            oldEdges += costMatrix[currentTour[j], next];
                            newEdges += costMatrix[currentTour[i], next];
                        }
                    }
                    else
                    {
                        int prevI = (i == 0) ? N : currentTour[i - 1];
                        int nextI = currentTour[i + 1];
                        int prevJ = currentTour[j - 1];
                        int nextJ = (j == N - 1) ? -1 : currentTour[j + 1];

                        oldEdges += costMatrix[prevI, currentTour[i]] + costMatrix[currentTour[i], nextI];
                        newEdges += costMatrix[prevI, currentTour[j]] + costMatrix[currentTour[j], nextI];

                        oldEdges += costMatrix[prevJ, currentTour[j]];
                        newEdges += costMatrix[prevJ, currentTour[i]];

                        if (nextJ != -1)
                        {
                            oldEdges += costMatrix[currentTour[j], nextJ];
                            newEdges += costMatrix[currentTour[i], nextJ];
                        }
                    }

                    double delta = newEdges - oldEdges;

                    if (delta < 0 || Math.Exp(-delta / T) > rand.NextDouble())
                    {
                        int tmp = currentTour[i];
                        currentTour[i] = currentTour[j];
                        currentTour[j] = tmp;
                        currentCost += delta;

                        if (currentCost < bestCost)
                        {
                            bestCost = currentCost;
                            Array.Copy(currentTour, bestTour, N);
                        }
                    }
                }
            }
            T *= alpha;
        }

        return bestTour;
    }

    static double GetTourCost(int[] tour, double[,] costMatrix, int N)
    {
        double cost = costMatrix[N, tour[0]];
        for (int i = 0; i < N - 1; i++)
        {
            cost += costMatrix[tour[i], tour[i+1]];
        }
        return cost;
    }

    static void Main(string[] args)
    {
        List<double[]> dataLines = new List<double[]>();
        string line;
        while ((line = Console.ReadLine()) != null)
        {
            double[] vals = ParseDoubles(line);
            if (vals.Length > 0)
            {
                dataLines.Add(vals);
            }
        }

        if (dataLines.Count < 2)
        {
            return;
        }

        double[] line1 = dataLines[0];
        int N = (int)line1[0];
        double mu = line1.Length > 1 ? line1[1] : MU_DEFAULT;

        double[] startLine = dataLines[1];
        OrbitState startState = new OrbitState
        {
            x = startLine[0], y = startLine[1], z = startLine[2],
            vx = startLine[3], vy = startLine[4], vz = startLine[5]
        };

        OrbitElements startElem = GetElements(startState, mu);
        OrbitElements[] elements = new OrbitElements[N + 1];
        elements[N] = startElem;

        for (int i = 0; i < N; i++)
        {
            if (i + 2 < dataLines.Count)
            {
                double[] sLine = dataLines[i + 2];
                OrbitState sState = new OrbitState
                {
                    x = sLine[0], y = sLine[1], z = sLine[2],
                    vx = sLine[3], vy = sLine[4], vz = sLine[5]
                };
                elements[i] = GetElements(sState, mu);
            }
            else
            {
                elements[i] = startElem;
            }
        }

        double[,] costMatrix = new double[N + 1, N + 1];
        for (int i = 0; i <= N; i++)
        {
            for (int j = 0; j <= N; j++)
            {
                if (i == j) costMatrix[i, j] = 0;
                else costMatrix[i, j] = GetTransferCost(elements[i], elements[j], mu);
            }
        }

        int[] bestTour = null;

        if (N <= 20)
        {
            bestTour = SolveDP(costMatrix, N);
        }
        else
        {
            double bestCostAcrossRestarts = double.MaxValue;
            int numRestarts = 10;
            for (int r = 0; r < numRestarts; r++)
            {
                int[] tour = SolveSA(costMatrix, N, 42 + r * 100);
                double cost = GetTourCost(tour, costMatrix, N);
                if (cost < bestCostAcrossRestarts)
                {
                    bestCostAcrossRestarts = cost;
                    bestTour = tour;
                }
            }
        }

        if (bestTour != null)
        {
            Console.WriteLine(string.Join(" ", bestTour));
        }
    }
}
