using System;
using System.IO;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;
using System.Diagnostics;

public class Program {
    private static int[][] adj;
    private static int maxCliqueSize = 0;
    private static int[] bestClique = new int[0];
    private static readonly object lockObj = new object();
    
    private static Stopwatch stopwatch;
    private static volatile bool stopSearch = false;
    
    private const int ColoringThreshold = 1528;
    
    [ThreadStatic]
    private static int callCount;

    [ThreadStatic]
    private static int threadColorVersion;

    private class ThreadBuffers {
        public int[] NextColors;
        public int[] ColorUsed;
        public int[] SortedCandidates;
        public int[] SortedColors;
        public int[] NextCandidates;
        public int[] BucketCount;
        public int[] Pos;

        public ThreadBuffers(int capacity) {
            NextColors = new int[capacity];
            ColorUsed = new int[capacity + 2];
            SortedCandidates = new int[capacity];
            SortedColors = new int[capacity];
            NextCandidates = new int[capacity];
            BucketCount = new int[capacity + 2];
            Pos = new int[capacity + 2];
        }
    }

    [ThreadStatic]
    private static List<ThreadBuffers> threadBuffers;

    private static ThreadBuffers GetBuffers(int depth, int capacity) {
        if (threadBuffers == null) {
            threadBuffers = new List<ThreadBuffers>();
        }
        while (threadBuffers.Count <= depth) {
            threadBuffers.Add(new ThreadBuffers(Math.Max(capacity, 1024)));
        }
        var buf = threadBuffers[depth];
        if (buf.NextColors.Length < capacity) {
            buf = new ThreadBuffers(capacity);
            threadBuffers[depth] = buf;
        }
        return buf;
    }

    public static void Main(string[] args) {
        stopwatch = Stopwatch.StartNew();
        
        // Use fast I/O
        var scanner = new FastScanner(Console.OpenStandardInput());
        int numVertices = scanner.NextInt();
        if (numVertices == -1) return;
        int numEdges = scanner.NextInt();
        if (numEdges == -1) numEdges = 0;

        if (numVertices == 0) {
            Console.WriteLine(0);
            Console.WriteLine();
            return;
        }
        if (numVertices == 1) {
            Console.WriteLine(1);
            Console.WriteLine(0);
            return;
        }

        // Read edges and count degrees to build adjacency list
        int[] degrees = new int[numVertices];
        int[] uEdges = new int[numEdges];
        int[] vEdges = new int[numEdges];
        int actualEdges = 0;
        for (int i = 0; i < numEdges; i++) {
            int u = scanner.NextInt();
            if (u == -1) break;
            int v = scanner.NextInt();
            if (v == -1) break;
            if (u < 0 || u >= numVertices || v < 0 || v >= numVertices) continue;
            uEdges[actualEdges] = u;
            vEdges[actualEdges] = v;
            degrees[u]++;
            degrees[v]++;
            actualEdges++;
        }

        adj = new int[numVertices][];
        for (int i = 0; i < numVertices; i++) {
            adj[i] = new int[degrees[i]];
        }

        int[] nextIdx = new int[numVertices];
        for (int i = 0; i < actualEdges; i++) {
            int u = uEdges[i];
            int v = vEdges[i];
            if (u == v) continue; // skip self loops
            adj[u][nextIdx[u]++] = v;
            adj[v][nextIdx[v]++] = u;
        }

        // Clean adjacency lists (remove self-loops, duplicate edges, and sort)
        for (int i = 0; i < numVertices; i++) {
            int[] list = adj[i];
            int validCount = nextIdx[i];
            if (validCount <= 1) {
                if (validCount == 0) {
                    adj[i] = new int[0];
                } else {
                    if (list[0] == i) {
                        adj[i] = new int[0];
                    } else {
                        adj[i] = new[] { list[0] };
                    }
                }
                continue;
            }
            
            Array.Sort(list, 0, validCount);
            int uniqueCount = 0;
            int last = -1;
            for (int j = 0; j < validCount; j++) {
                int val = list[j];
                if (val != i && val != last) {
                    list[uniqueCount++] = val;
                    last = val;
                }
            }
            if (uniqueCount < list.Length) {
                int[] newList = new int[uniqueCount];
                Array.Copy(list, newList, uniqueCount);
                adj[i] = newList;
            }
        }

        // Step 1: Run greedy heuristic to get a high lower bound
        int[] greedyClique = RunGreedyHeuristic(numVertices);
        maxCliqueSize = greedyClique.Length;
        bestClique = greedyClique;

        // Step 2: Compute degeneracy ordering
        var degInfo = GetDegeneracyOrdering(numVertices);
        
        // If the degeneracy bound proves that the greedy clique is maximum, we are done!
        if (degInfo.Degeneracy + 1 <= maxCliqueSize) {
            PrintResult();
            return;
        }

        // Build BitSet graph for small graphs to allow O(1) edge checks in coloring
        BitSetGraph bitsetGraph = null;
        if (numVertices <= 50000) {
            bitsetGraph = new BitSetGraph(numVertices);
            for (int u = 0; u < numVertices; u++) {
                int[] neighbors = adj[u];
                for (int j = 0; j < neighbors.Length; j++) {
                    int v = neighbors[j];
                    if (u < v) {
                        bitsetGraph.AddEdge(u, v);
                    }
                }
            }
        }

        // Step 3: Run exact search in parallel using degeneracy ordering
        int[] order = degInfo.Order;
        int[] pos = degInfo.Pos;

        Parallel.For(0, numVertices, new ParallelOptions { MaxDegreeOfParallelism = Environment.ProcessorCount }, (i, state) => {
            if (stopSearch) {
                state.Stop();
                return;
            }

            int idx = numVertices - 1 - i;
            int currentMax = Volatile.Read(ref maxCliqueSize);
            if (idx + 1 <= currentMax) {
                state.Stop();
                return;
            }

            int v = order[idx];
            int[] neighbors = adj[v];
            
            // Candidates are neighbors that appear before v in the degeneracy ordering (pos[u] < idx)
            int[] candidates = new int[neighbors.Length];
            int candCount = 0;
            for (int j = 0; j < neighbors.Length; j++) {
                int u = neighbors[j];
                if (pos[u] < idx) {
                    candidates[candCount++] = u;
                }
            }

            if (1 + candCount > currentMax) {
                var clique = new List<int> { v };
                Search(clique, candidates, candCount, bitsetGraph, 0);
            }
        });

        PrintResult();
    }

    private static void PrintResult() {
        Console.WriteLine(bestClique.Length);
        Console.WriteLine(string.Join(" ", bestClique));
    }

    private static int[] RunGreedyHeuristic(int numVertices) {
        int[] verticesByDegree = new int[numVertices];
        for (int i = 0; i < numVertices; i++) verticesByDegree[i] = i;
        Array.Sort(verticesByDegree, (a, b) => adj[b].Length.CompareTo(adj[a].Length));

        int[] localBestClique = new int[0];
        int bestSize = 0;
        int limit = Math.Min(numVertices, numVertices > 2000 ? 100 : numVertices);

        for (int i = 0; i < limit; i++) {
            int start = verticesByDegree[i];
            var clique = new List<int> { start };
            int[] candidates = adj[start];
            int candidatesCount = candidates.Length;

            if (candidatesCount == 0) {
                if (clique.Count > bestSize) {
                    bestSize = clique.Count;
                    localBestClique = clique.ToArray();
                }
                continue;
            }

            int[] candBuf = new int[candidatesCount];
            Array.Copy(candidates, candBuf, candidatesCount);
            int[] nextBuf = new int[candidatesCount];

            while (candidatesCount > 0) {
                int bestCand = -1;
                int maxDeg = -1;
                for (int j = 0; j < candidatesCount; j++) {
                    int c = candBuf[j];
                    int deg = adj[c].Length;
                    if (deg > maxDeg) {
                        maxDeg = deg;
                        bestCand = c;
                    }
                }

                clique.Add(bestCand);
                int nextCount = Intersect(candBuf, candidatesCount, adj[bestCand], nextBuf);
                Array.Copy(nextBuf, candBuf, nextCount);
                candidatesCount = nextCount;
            }

            if (clique.Count > bestSize) {
                bestSize = clique.Count;
                localBestClique = clique.ToArray();
            }
        }

        return localBestClique;
    }

    private static void Search(List<int> clique, int[] candidates, int candidatesCount, BitSetGraph bitsetGraph, int depth) {
        if (stopSearch) return;

        if (++callCount >= 8192) {
            callCount = 0;
            if (stopwatch.ElapsedMilliseconds > 280000) {
                stopSearch = true;
                return;
            }
        }

        int currentMax = Volatile.Read(ref maxCliqueSize);
        if (clique.Count + candidatesCount <= currentMax) {
            return;
        }

        if (candidatesCount == 0) {
            lock (lockObj) {
                if (clique.Count > maxCliqueSize) {
                    maxCliqueSize = clique.Count;
                    bestClique = clique.ToArray();
                }
            }
            return;
        }

        if (candidatesCount <= ColoringThreshold) {
            var buf = GetBuffers(depth, candidatesCount);
            int[] nextColors = buf.NextColors;
            int[] colorUsed = buf.ColorUsed;
            int[] sortedCandidates = buf.SortedCandidates;
            int[] sortedColors = buf.SortedColors;
            int[] nextCandidates = buf.NextCandidates;

            ColorCandidates(candidates, candidatesCount, nextColors, colorUsed, bitsetGraph);

            SortByColor(candidates, nextColors, candidatesCount, sortedCandidates, sortedColors, buf.BucketCount, buf.Pos);

            for (int i = candidatesCount - 1; i >= 0; i--) {
                int currentMaxInner = Volatile.Read(ref maxCliqueSize);
                if (clique.Count + sortedColors[i] <= currentMaxInner) {
                    break; // Prune all remaining candidates since their colors are even smaller
                }

                int v = sortedCandidates[i];
                int nextCount = Intersect(sortedCandidates, i, adj[v], nextCandidates);

                clique.Add(v);
                Search(clique, nextCandidates, nextCount, bitsetGraph, depth + 1);
                clique.RemoveAt(clique.Count - 1);
            }
        } else {
            var buf = GetBuffers(depth, candidatesCount);
            int[] nextCandidates = buf.NextCandidates;

            for (int i = candidatesCount - 1; i >= 0; i--) {
                int currentMaxInner = Volatile.Read(ref maxCliqueSize);
                if (clique.Count + i + 1 <= currentMaxInner) {
                    break; // Prune
                }

                int v = candidates[i];
                int nextCount = Intersect(candidates, i, adj[v], nextCandidates);

                clique.Add(v);
                Search(clique, nextCandidates, nextCount, bitsetGraph, depth + 1);
                clique.RemoveAt(clique.Count - 1);
            }
        }
    }

    private static void ColorCandidates(int[] candidates, int count, int[] colors, int[] colorUsed, BitSetGraph bitsetGraph) {
        if (bitsetGraph != null) {
            for (int i = 0; i < count; i++) {
                int v = candidates[i];
                threadColorVersion++;
                for (int j = 0; j < i; j++) {
                    int u = candidates[j];
                    if (bitsetGraph.HasEdge(v, u)) {
                        int col = colors[j];
                        colorUsed[col] = threadColorVersion;
                    }
                }
                int c = 1;
                while (colorUsed[c] == threadColorVersion) {
                    c++;
                }
                colors[i] = c;
            }
        } else {
            for (int i = 0; i < count; i++) {
                int v = candidates[i];
                threadColorVersion++;
                int[] neighbors = adj[v];
                int ptrCand = 0;
                int ptrNeigh = 0;
                int neighLen = neighbors.Length;
                while (ptrCand < i && ptrNeigh < neighLen) {
                    int candVal = candidates[ptrCand];
                    int neighVal = neighbors[ptrNeigh];
                    if (candVal == neighVal) {
                        int col = colors[ptrCand];
                        colorUsed[col] = threadColorVersion;
                        ptrCand++;
                        ptrNeigh++;
                    } else if (candVal < neighVal) {
                        ptrCand++;
                    } else {
                        ptrNeigh++;
                    }
                }
                int c = 1;
                while (colorUsed[c] == threadColorVersion) {
                    c++;
                }
                colors[i] = c;
            }
        }
    }

    private static void SortByColor(int[] candidates, int[] colors, int count, int[] sortedCandidates, int[] sortedColors, int[] bucketCount, int[] pos) {
        int maxColor = 0;
        for (int i = 0; i < count; i++) {
            if (colors[i] > maxColor) maxColor = colors[i];
        }

        Array.Clear(bucketCount, 0, maxColor + 2);
        for (int i = 0; i < count; i++) {
            bucketCount[colors[i]]++;
        }

        pos[0] = 0;
        for (int c = 1; c <= maxColor + 1; c++) {
            pos[c] = pos[c - 1] + bucketCount[c - 1];
        }

        for (int i = 0; i < count; i++) {
            int col = colors[i];
            int destIdx = pos[col]++;
            sortedCandidates[destIdx] = candidates[i];
            sortedColors[destIdx] = col;
        }
    }

    private static int Intersect(int[] a, int aCount, int[] b, int[] result) {
        int i = 0, j = 0, k = 0;
        int bLen = b.Length;
        while (i < aCount && j < bLen) {
            int valA = a[i];
            int valB = b[j];
            if (valA == valB) {
                result[k++] = valA;
                i++;
                j++;
            } else if (valA < valB) {
                i++;
            } else {
                j++;
            }
        }
        return k;
    }

    private struct DegInfo {
        public int[] Order;
        public int[] Pos;
        public int Degeneracy;
    }

    private static DegInfo GetDegeneracyOrdering(int numVertices) {
        int[] degrees = new int[numVertices];
        int maxDeg = 0;
        for (int i = 0; i < numVertices; i++) {
            degrees[i] = adj[i].Length;
            if (degrees[i] > maxDeg) maxDeg = degrees[i];
        }

        int[] head = new int[maxDeg + 1];
        for (int i = 0; i < head.Length; i++) head[i] = -1;
        int[] prev = new int[numVertices];
        int[] next = new int[numVertices];

        for (int i = 0; i < numVertices; i++) {
            int d = degrees[i];
            next[i] = head[d];
            prev[i] = -1;
            if (head[d] != -1) {
                prev[head[d]] = i;
            }
            head[d] = i;
        }

        int[] order = new int[numVertices];
        int[] pos = new int[numVertices];
        bool[] removed = new bool[numVertices];
        int minDeg = 0;
        int degeneracy = 0;

        for (int step = 0; step < numVertices; step++) {
            while (minDeg <= maxDeg && head[minDeg] == -1) {
                minDeg++;
            }

            int v = head[minDeg];
            head[minDeg] = next[v];
            if (next[v] != -1) {
                prev[next[v]] = -1;
            }

            removed[v] = true;
            order[step] = v;
            pos[v] = step;

            if (minDeg > degeneracy) {
                degeneracy = minDeg;
            }

            int[] neighbors = adj[v];
            for (int j = 0; j < neighbors.Length; j++) {
                int u = neighbors[j];
                if (!removed[u]) {
                    int du = degrees[u];
                    int pu = prev[u];
                    int nu = next[u];

                    if (pu != -1) {
                        next[pu] = nu;
                    } else {
                        head[du] = nu;
                    }
                    if (nu != -1) {
                        prev[nu] = pu;
                    }

                    degrees[u]--;
                    du--;

                    next[u] = head[du];
                    prev[u] = -1;
                    if (head[du] != -1) {
                        prev[head[du]] = u;
                    }
                    head[du] = u;

                    if (du < minDeg) {
                        minDeg = du;
                    }
                }
            }
        }

        return new DegInfo { Order = order, Pos = pos, Degeneracy = degeneracy };
    }
}

public class BitSetGraph {
    public ulong[][] Bitsets;
    public int Words;

    public BitSetGraph(int V) {
        Words = (V + 63) / 64;
        Bitsets = new ulong[V][];
        for (int i = 0; i < V; i++) {
            Bitsets[i] = new ulong[Words];
        }
    }

    public void AddEdge(int u, int v) {
        Bitsets[u][v >> 6] |= (1UL << (v & 63));
        Bitsets[v][u >> 6] |= (1UL << (u & 63));
    }

    public bool HasEdge(int u, int v) {
        return (Bitsets[u][v >> 6] & (1UL << (v & 63))) != 0;
    }
}

public class FastScanner {
    private readonly Stream stream;
    private readonly byte[] buffer = new byte[65536];
    private int bufferLeft = 0;
    private int bufferRead = 0;

    public FastScanner(Stream stream) {
        this.stream = stream;
    }

    private int ReadByte() {
        if (bufferLeft >= bufferRead) {
            bufferRead = stream.Read(buffer, 0, buffer.Length);
            bufferLeft = 0;
            if (bufferRead == 0) return -1;
        }
        return buffer[bufferLeft++];
    }

    public int NextInt() {
        int val = 0;
        int b = ReadByte();
        while (b != -1 && b <= 32) {
            b = ReadByte();
        }
        if (b == -1) return -1;
        bool neg = false;
        if (b == '-') {
            neg = true;
            b = ReadByte();
        }
        while (b >= '0' && b <= '9') {
            val = val * 10 + (b - '0');
            b = ReadByte();
        }
        return neg ? -val : val;
    }
}
