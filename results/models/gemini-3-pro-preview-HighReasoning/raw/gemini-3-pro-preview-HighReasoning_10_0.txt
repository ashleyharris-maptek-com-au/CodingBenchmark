struct Rectangle {
    int id;
    int w, h;
};

struct Placement {
    int rect_id;
    int board_idx;
    int x, y;
    int rotated;
};

struct Segment {
    int x, y, w;
};

struct FreeRect {
    int x, y, w, h;
};

inline bool IsPotentiallyUseful(const FreeRect& F, int min_S, int min_L) {
    int f_min = std::min(F.w, F.h);
    int f_max = std::max(F.w, F.h);
    return (f_min >= min_S && f_max >= min_L);
}

struct SkylineBoard {
    int W, H;
    std::vector<Segment> segments;
    std::vector<FreeRect> waste_map;

    SkylineBoard(int w, int h) : W(w), H(h) {
        segments.push_back({0, 0, w});
    }

    bool TryPack(int rw, int rh, int& px, int& py, bool& rotated, int min_S, int min_L, bool prefer_waste_fit) {
        // Prune waste map
        waste_map.erase(
            std::remove_if(waste_map.begin(), waste_map.end(), [&](const FreeRect& F) {
                return !IsPotentiallyUseful(F, min_S, min_L);
            }),
            waste_map.end()
        );

        // 1. Try Waste Map
        int best_waste_idx = -1;
        int best_waste_score = 2000000000;
        bool waste_rot = false;

        for (size_t i = 0; i < waste_map.size(); ++i) {
            const auto& R = waste_map[i];
            if (rw <= R.w && rh <= R.h) {
                int score = std::min(R.w - rw, R.h - rh);
                if (score < best_waste_score) {
                    best_waste_score = score;
                    best_waste_idx = i;
                    waste_rot = false;
                }
            }
            if (rh <= R.w && rw <= R.h) {
                int score = std::min(R.w - rh, R.h - rw);
                if (score < best_waste_score) {
                    best_waste_score = score;
                    best_waste_idx = i;
                    waste_rot = true;
                }
            }
        }

        if (best_waste_idx != -1) {
            FreeRect R = waste_map[best_waste_idx];
            int pw = waste_rot ? rh : rw;
            int ph = waste_rot ? rw : rh;
            px = R.x;
            py = R.y;
            rotated = waste_rot;

            waste_map.erase(waste_map.begin() + best_waste_idx);
            int dw = R.w - pw;
            int dh = R.h - ph;
            
            if (dw > 0 || dh > 0) {
                bool split_horizontal = (dw > dh);
                if (split_horizontal) {
                    FreeRect r1 = {R.x + pw, R.y, dw, R.h};
                    FreeRect r2 = {R.x, R.y + ph, pw, dh};
                    if (IsPotentiallyUseful(r1, min_S, min_L)) waste_map.push_back(r1);
                    if (IsPotentiallyUseful(r2, min_S, min_L)) waste_map.push_back(r2);
                } else {
                    FreeRect r1 = {R.x + pw, R.y, dw, ph};
                    FreeRect r2 = {R.x, R.y + ph, R.w, dh};
                    if (IsPotentiallyUseful(r1, min_S, min_L)) waste_map.push_back(r1);
                    if (IsPotentiallyUseful(r2, min_S, min_L)) waste_map.push_back(r2);
                }
            }
            return true;
        }

        // 2. Try Skyline
        int best_x = -1, best_y = H + 1;
        int best_waste = 2000000000;
        int best_seg_idx = -1;
        bool sky_rot = false;

        for (int rot = 0; rot < 2; ++rot) {
            if (rot == 1 && rw == rh) continue; // Skip redundant check for squares
            int target_w = (rot == 0) ? rw : rh;
            int target_h = (rot == 0) ? rh : rw;
            if (target_w > W || target_h > H) continue;

            for (size_t i = 0; i < segments.size(); ++i) {
                int x = segments[i].x;
                if (x + target_w > W) break;

                int y_new = 0;
                for (size_t j = i; j < segments.size(); ++j) {
                    if (segments[j].x >= x + target_w) break;
                    y_new = std::max(y_new, segments[j].y);
                }
                if (y_new + target_h > H) continue;

                int waste_area = 0;
                for (size_t j = i; j < segments.size(); ++j) {
                    int seg_x = segments[j].x;
                    if (seg_x >= x + target_w) break;
                    int seg_r = std::min(x + target_w, seg_x + segments[j].w);
                    waste_area += (y_new - segments[j].y) * (seg_r - seg_x);
                }

                bool is_better = false;
                if (prefer_waste_fit) {
                    if (waste_area < best_waste) {
                        is_better = true;
                    } else if (waste_area == best_waste) {
                        is_better = (y_new < best_y);
                    }
                } else {
                    if (y_new < best_y) {
                        is_better = true;
                    } else if (y_new == best_y) {
                        is_better = (waste_area < best_waste);
                    }
                }

                if (is_better) {
                    best_y = y_new;
                    best_waste = waste_area;
                    best_x = x;
                    best_seg_idx = i;
                    sky_rot = (rot == 1);
                }
            }
        }

        if (best_x != -1) {
            px = best_x;
            py = best_y;
            rotated = sky_rot;
            int pw = sky_rot ? rh : rw;
            int ph = sky_rot ? rw : rh;

            // Collect waste space under the placement
            for (size_t j = best_seg_idx; j < segments.size(); ++j) {
                int seg_x = segments[j].x;
                if (seg_x >= best_x + pw) break;
                int seg_r = std::min(best_x + pw, seg_x + segments[j].w);
                int overlap_w = seg_r - std::max(best_x, seg_x);
                int overlap_h = best_y - segments[j].y;
                FreeRect F = {std::max(best_x, seg_x), segments[j].y, overlap_w, overlap_h};
                if (IsPotentiallyUseful(F, min_S, min_L)) {
                    waste_map.push_back(F);
                }
            }

            // Update skyline
            std::vector<Segment> next_segs;
            bool added = false;
            for (const auto& seg : segments) {
                int seg_l = seg.x;
                int seg_r = seg.x + seg.w;
                if (seg_r <= best_x) {
                    next_segs.push_back(seg);
                } else if (seg_l >= best_x + pw) {
                    if (!added) {
                        next_segs.push_back({best_x, best_y + ph, pw});
                        added = true;
                    }
                    next_segs.push_back(seg);
                } else {
                    if (seg_l < best_x) {
                        next_segs.push_back({seg_l, seg.y, best_x - seg_l});
                    }
                    if (seg_r > best_x + pw) {
                        if (!added) {
                            next_segs.push_back({best_x, best_y + ph, pw});
                            added = true;
                        }
                        next_segs.push_back({best_x + pw, seg.y, seg_r - (best_x + pw)});
                    }
                }
            }
            if (!added) {
                next_segs.push_back({best_x, best_y + ph, pw});
            }

            segments.clear();
            for (const auto& seg : next_segs) {
                if (segments.empty()) {
                    segments.push_back(seg);
                } else {
                    if (segments.back().y == seg.y) {
                        segments.back().w += seg.w;
                    } else {
                        segments.push_back(seg);
                    }
                }
            }
            return true;
        }

        return false;
    }
};

struct GuillotineBoard {
    int W, H;
    std::vector<FreeRect> free_rects;

    GuillotineBoard(int w, int h) : W(w), H(h) {
        free_rects.push_back({0, 0, w, h});
    }

    bool TryPack(int rw, int rh, int& px, int& py, bool& rotated, int min_S, int min_L, int split_rule) {
        // Prune free rects
        free_rects.erase(
            std::remove_if(free_rects.begin(), free_rects.end(), [&](const FreeRect& F) {
                return !IsPotentiallyUseful(F, min_S, min_L);
            }),
            free_rects.end()
        );

        int best_idx = -1;
        int best_score = 2000000000;
        bool best_rot = false;

        for (size_t i = 0; i < free_rects.size(); ++i) {
            const auto& R = free_rects[i];
            if (rw <= R.w && rh <= R.h) {
                int score = std::min(R.w - rw, R.h - rh);
                if (score < best_score) {
                    best_score = score;
                    best_idx = i;
                    best_rot = false;
                }
            }
            if (rh <= R.w && rw <= R.h) {
                int score = std::min(R.w - rh, R.h - rw);
                if (score < best_score) {
                    best_score = score;
                    best_idx = i;
                    best_rot = true;
                }
            }
        }

        if (best_idx != -1) {
            FreeRect R = free_rects[best_idx];
            int pw = best_rot ? rh : rw;
            int ph = best_rot ? rw : rh;
            px = R.x;
            py = R.y;
            rotated = best_rot;

            free_rects.erase(free_rects.begin() + best_idx);

            int dw = R.w - pw;
            int dh = R.h - ph;

            if (dw > 0 || dh > 0) {
                bool split_horizontal = false;
                if (split_rule == 0) {
                    split_horizontal = (dw > dh);
                } else if (split_rule == 1) {
                    split_horizontal = (dw <= dh);
                } else if (split_rule == 2) {
                    split_horizontal = true;
                } else {
                    split_horizontal = false;
                }

                if (split_horizontal) {
                    FreeRect r1 = {R.x + pw, R.y, dw, R.h};
                    FreeRect r2 = {R.x, R.y + ph, pw, dh};
                    if (IsPotentiallyUseful(r1, min_S, min_L)) free_rects.push_back(r1);
                    if (IsPotentiallyUseful(r2, min_S, min_L)) free_rects.push_back(r2);
                } else {
                    FreeRect r1 = {R.x + pw, R.y, dw, ph};
                    FreeRect r2 = {R.x, R.y + ph, R.w, dh};
                    if (IsPotentiallyUseful(r1, min_S, min_L)) free_rects.push_back(r1);
                    if (IsPotentiallyUseful(r2, min_S, min_L)) free_rects.push_back(r2);
                }
            }
            return true;
        }
        return false;
    }
};

struct PackingResult {
    int num_boards;
    std::vector<Placement> placements;
    std::string config_name;
};

PackingResult RunConfig(const std::vector<Rectangle>& input_rects, int W, int H,
                         int pre_orient, // 0: none, 1: w >= h, 2: h >= w
                         int sort_type,  // 0: area, 1: height, 2: width, 3: max_dim, 4: sum_dim
                         int K_active,
                         int algo,       // 0: Skyline BL, 1: Skyline MW, 2: Guillotine SplitShorter, 3: Guillotine SplitLonger
                         std::string name) {
    std::vector<Rectangle> rects = input_rects;
    if (pre_orient == 1) {
        for (auto& r : rects) {
            if (r.w < r.h) std::swap(r.w, r.h);
        }
    } else if (pre_orient == 2) {
        for (auto& r : rects) {
            if (r.h < r.w) std::swap(r.w, r.h);
        }
    }

    if (sort_type == 0) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            long long area_a = (long long)a.w * a.h;
            long long area_b = (long long)b.w * b.h;
            if (area_a != area_b) return area_a > area_b;
            return std::max(a.w, a.h) > std::max(b.w, b.h);
        });
    } else if (sort_type == 1) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            if (a.h != b.h) return a.h > b.h;
            return a.w > b.w;
        });
    } else if (sort_type == 2) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            if (a.w != b.w) return a.w > b.w;
            return a.h > b.h;
        });
    } else if (sort_type == 3) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            int max_a = std::max(a.w, a.h);
            int max_b = std::max(b.w, b.h);
            if (max_a != max_b) return max_a > max_b;
            return (long long)a.w * a.h > (long long)b.w * b.h;
        });
    } else if (sort_type == 4) {
        std::sort(rects.begin(), rects.end(), [](const Rectangle& a, const Rectangle& b) {
            int sum_a = a.w + a.h;
            int sum_b = b.w + b.h;
            if (sum_a != sum_b) return sum_a > sum_b;
            return (long long)a.w * a.h > (long long)b.w * b.h;
        });
    }

    int N = rects.size();
    std::vector<int> suffix_min_S(N);
    std::vector<int> suffix_min_L(N);
    if (N > 0) {
        suffix_min_S[N - 1] = std::min(rects[N - 1].w, rects[N - 1].h);
        suffix_min_L[N - 1] = std::max(rects[N - 1].w, rects[N - 1].h);
        for (int i = N - 2; i >= 0; --i) {
            suffix_min_S[i] = std::min(suffix_min_S[i + 1], std::min(rects[i].w, rects[i].h));
            suffix_min_L[i] = std::min(suffix_min_L[i + 1], std::max(rects[i].w, rects[i].h));
        }
    }

    std::vector<Placement> placements(rects.size());

    int act_K = K_active;
    if (N <= 500) act_K = N;
    else if (N <= 5000) act_K = std::min(K_active, 30);
    else if (N <= 15000) act_K = std::min(K_active, 15);
    else act_K = std::min(K_active, 8);

    if (algo == 0 || algo == 1) {
        std::vector<SkylineBoard> boards;
        bool prefer_waste_fit = (algo == 1);
        for (size_t i = 0; i < rects.size(); ++i) {
            bool packed = false;
            int px, py;
            bool rotated;
            int min_S = suffix_min_S[i];
            int min_L = suffix_min_L[i];
            int start_b = std::max(0, (int)boards.size() - act_K);
            for (int b = start_b; b < (int)boards.size(); ++b) {
                if (boards[b].TryPack(rects[i].w, rects[i].h, px, py, rotated, min_S, min_L, prefer_waste_fit)) {
                    placements[i] = {rects[i].id, b, px, py, rotated ? 1 : 0};
                    packed = true;
                    break;
                }
            }
            if (!packed) {
                boards.push_back(SkylineBoard(W, H));
                int b = (int)boards.size() - 1;
                bool ok = boards[b].TryPack(rects[i].w, rects[i].h, px, py, rotated, min_S, min_L, prefer_waste_fit);
                if (!ok) return {2000000000, {}, name};
                placements[i] = {rects[i].id, b, px, py, rotated ? 1 : 0};
            }
        }
        std::vector<Placement> final_placements(rects.size());
        for (const auto& p : placements) final_placements[p.rect_id] = p;
        return {(int)boards.size(), final_placements, name};
    } else {
        std::vector<GuillotineBoard> boards;
        int split_rule = (algo == 2) ? 0 : 1;
        for (size_t i = 0; i < rects.size(); ++i) {
            bool packed = false;
            int px, py;
            bool rotated;
            int min_S = suffix_min_S[i];
            int min_L = suffix_min_L[i];
            int start_b = std::max(0, (int)boards.size() - act_K);
            for (int b = start_b; b < (int)boards.size(); ++b) {
                if (boards[b].TryPack(rects[i].w, rects[i].h, px, py, rotated, min_S, min_L, split_rule)) {
                    placements[i] = {rects[i].id, b, px, py, rotated ? 1 : 0};
                    packed = true;
                    break;
                }
            }
            if (!packed) {
                boards.push_back(GuillotineBoard(W, H));
                int b = (int)boards.size() - 1;
                bool ok = boards[b].TryPack(rects[i].w, rects[i].h, px, py, rotated, min_S, min_L, split_rule);
                if (!ok) return {2000000000, {}, name};
                placements[i] = {rects[i].id, b, px, py, rotated ? 1 : 0};
            }
        }
        std::vector<Placement> final_placements(rects.size());
        for (const auto& p : placements) final_placements[p.rect_id] = p;
        return {(int)boards.size(), final_placements, name};
    }
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    int N, W, H;
    if (!(std::cin >> N >> W >> H)) return 0;

    std::vector<Rectangle> rects(N);
    for (int i = 0; i < N; ++i) {
        rects[i].id = i;
        std::cin >> rects[i].w >> rects[i].h;
    }

    if (N == 0) {
        std::cout << 0 << "\n";
        return 0;
    }

    struct Config {
        int pre_orient;
        int sort_type;
        int K_active;
        int algo;
        std::string name;
    } configs[] = {
        // Skyline BL
        {1, 1, 15, 0, "Skyline_Orient(w>=h)_Sort(Height)_K(15)_BL"},
        {2, 1, 15, 0, "Skyline_Orient(h>=w)_Sort(Height)_K(15)_BL"},
        {1, 2, 15, 0, "Skyline_Orient(w>=h)_Sort(Width)_K(15)_BL"},
        {2, 2, 15, 0, "Skyline_Orient(h>=w)_Sort(Width)_K(15)_BL"},
        {1, 0, 15, 0, "Skyline_Orient(w>=h)_Sort(Area)_K(15)_BL"},
        {2, 0, 15, 0, "Skyline_Orient(h>=w)_Sort(Area)_K(15)_BL"},
        {1, 3, 15, 0, "Skyline_Orient(w>=h)_Sort(MaxDim)_K(15)_BL"},
        {2, 3, 15, 0, "Skyline_Orient(h>=w)_Sort(MaxDim)_K(15)_BL"},

        // Skyline MW
        {1, 1, 15, 1, "Skyline_Orient(w>=h)_Sort(Height)_K(15)_MW"},
        {2, 1, 15, 1, "Skyline_Orient(h>=w)_Sort(Height)_K(15)_MW"},
        {1, 2, 15, 1, "Skyline_Orient(w>=h)_Sort(Width)_K(15)_MW"},
        {2, 2, 15, 1, "Skyline_Orient(h>=w)_Sort(Width)_K(15)_MW"},

        // Guillotine Shorter
        {1, 1, 15, 2, "Guillotine_Orient(w>=h)_Sort(Height)_K(15)_Shorter"},
        {2, 1, 15, 2, "Guillotine_Orient(h>=w)_Sort(Height)_K(15)_Shorter"},
        {1, 2, 15, 2, "Guillotine_Orient(w>=h)_Sort(Width)_K(15)_Shorter"},
        {2, 2, 15, 2, "Guillotine_Orient(h>=w)_Sort(Width)_K(15)_Shorter"},
        {1, 0, 15, 2, "Guillotine_Orient(w>=h)_Sort(Area)_K(15)_Shorter"},
        {2, 0, 15, 2, "Guillotine_Orient(h>=w)_Sort(Area)_K(15)_Shorter"},

        // Guillotine Longer
        {1, 1, 15, 3, "Guillotine_Orient(w>=h)_Sort(Height)_K(15)_Longer"},
        {2, 1, 15, 3, "Guillotine_Orient(h>=w)_Sort(Height)_K(15)_Longer"},
        {1, 2, 15, 3, "Guillotine_Orient(w>=h)_Sort(Width)_K(15)_Longer"},
        {2, 2, 15, 3, "Guillotine_Orient(h>=w)_Sort(Width)_K(15)_Longer"},
        {1, 4, 15, 3, "Guillotine_Orient(w>=h)_Sort(SumDim)_K(15)_Longer"},
        {2, 4, 15, 3, "Guillotine_Orient(h>=w)_Sort(SumDim)_K(15)_Longer"},
    };

    std::vector<std::future<PackingResult>> futures;
    for (const auto& c : configs) {
        futures.push_back(std::async(std::launch::async, [&, c]() {
            return RunConfig(rects, W, H, c.pre_orient, c.sort_type, c.K_active, c.algo, c.name);
        }));
    }

    int best_boards = 2000000000;
    std::vector<Placement> best_placements;

    for (auto& f : futures) {
        auto res = f.get();
        if (res.num_boards < best_boards) {
            best_boards = res.num_boards;
            best_placements = std::move(res.placements);
        }
    }

    std::cout << best_boards << "\n";
    for (const auto& p : best_placements) {
        std::cout << p.board_idx << " " << p.x << " " << p.y << " " << p.rotated << "\n";
    }

    return 0;
}
