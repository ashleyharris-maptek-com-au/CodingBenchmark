use std::io::{self, Read, Write, BufReader, BufWriter, BufRead};

struct SimpleRng {
    state: u64,
}

impl SimpleRng {
    fn new(seed: u64) -> Self {
        SimpleRng {
            state: if seed == 0 { 1 } else { seed },
        }
    }

    fn next_u64(&mut self) -> u64 {
        let mut x = self.state;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.state = x;
        x
    }

    fn gen_range(&mut self, min: usize, max: usize) -> usize {
        assert!(min < max);
        let range = (max - min) as u64;
        let val = self.next_u64() % range;
        min + val as usize
    }

    fn gen_f32(&mut self) -> f32 {
        let val = self.next_u64() & 0xFFFFFF; // 24 bits
        (val as f32) / (16777216.0f32)
    }
}

struct KdNode {
    point: [f32; 3],
    id: u32,
    axis: u8,
    left: i32,
    right: i32,
}

#[inline(always)]
fn dist_sq(a: &[f32; 3], b: &[f32; 3]) -> f32 {
    let dx = a[0] - b[0];
    let dy = a[1] - b[1];
    let dz = a[2] - b[2];
    dx * dx + dy * dy + dz * dz
}

fn build_kd_tree(
    points: &[[f32; 3]],
    ids: &[u32],
    indices: &mut [usize],
    nodes: &mut Vec<KdNode>,
    depth: usize,
) -> i32 {
    if indices.is_empty() {
        return -1;
    }
    let axis = (depth % 3) as u8;

    indices.sort_unstable_by(|&a, &b| {
        points[a][axis as usize]
            .partial_cmp(&points[b][axis as usize])
            .unwrap_or(std::cmp::Ordering::Equal)
    });

    let mid = indices.len() / 2;
    let median_idx = indices[mid];

    let node_idx = nodes.len() as i32;
    nodes.push(KdNode {
        point: points[median_idx],
        id: ids[median_idx],
        axis,
        left: -1,
        right: -1,
    });

    let left_idx = build_kd_tree(points, ids, &mut indices[..mid], nodes, depth + 1);
    let right_idx = build_kd_tree(points, ids, &mut indices[mid + 1..], nodes, depth + 1);

    nodes[node_idx as usize].left = left_idx;
    nodes[node_idx as usize].right = right_idx;

    node_idx
}

fn rebuild_tree(centroids: &[[f32; 3]], nodes: &mut Vec<KdNode>, indices: &mut Vec<usize>) -> i32 {
    nodes.clear();
    indices.clear();
    indices.extend(0..centroids.len());
    let ids: Vec<u32> = (0..centroids.len() as u32).collect();
    build_kd_tree(centroids, &ids, indices, nodes, 0)
}

fn nearest_neighbor(
    nodes: &[KdNode],
    root: i32,
    target: &[f32; 3],
    best_id: &mut u32,
    best_dist_sq: &mut f32,
) {
    if root == -1 {
        return;
    }
    let node = &nodes[root as usize];
    let d_sq = dist_sq(&node.point, target);
    if d_sq < *best_dist_sq {
        *best_dist_sq = d_sq;
        *best_id = node.id;
    }

    let axis = node.axis as usize;
    let diff = target[axis] - node.point[axis];

    let (near, far) = if diff <= 0.0 {
        (node.left, node.right)
    } else {
        (node.right, node.left)
    };

    nearest_neighbor(nodes, near, target, best_id, best_dist_sq);

    if diff * diff < *best_dist_sq {
        nearest_neighbor(nodes, far, target, best_id, best_dist_sq);
    }
}

fn find_closest_kd(nodes: &[KdNode], root: i32, p: &[f32; 3]) -> usize {
    let mut best_id = 0u32;
    let mut best_dist_sq = f32::INFINITY;
    nearest_neighbor(nodes, root, p, &mut best_id, &mut best_dist_sq);
    best_id as usize
}

fn find_closest_linear(centroids: &[[f32; 3]], p: &[f32; 3]) -> usize {
    let mut best_idx = 0;
    let mut best_dist_sq = dist_sq(&centroids[0], p);
    for i in 1..centroids.len() {
        let d = dist_sq(&centroids[i], p);
        if d < best_dist_sq {
            best_dist_sq = d;
            best_idx = i;
        }
    }
    best_idx
}

fn compute_data_scale(points: &[[f32; 3]]) -> f32 {
    if points.is_empty() {
        return 1.0;
    }
    let mut sum_sq = [0.0; 3];
    let mut sum = [0.0; 3];
    for p in points {
        for d in 0..3 {
            sum[d] += p[d];
            sum_sq[d] += p[d] * p[d];
        }
    }
    let n = points.len() as f32;
    let mut variance = 0.0;
    for d in 0..3 {
        let mean = sum[d] / n;
        let var = (sum_sq[d] / n) - (mean * mean);
        variance += var;
    }
    let std_dev = (variance / 3.0).sqrt();
    if std_dev > 0.0 {
        std_dev
    } else {
        1.0
    }
}

fn has_converged(old_centroids: &[[f32; 3]], new_centroids: &[[f32; 3]], threshold: f32) -> bool {
    let mut max_shift_sq = 0.0f32;
    for i in 0..old_centroids.len() {
        let shift_sq = dist_sq(&old_centroids[i], &new_centroids[i]);
        if shift_sq > max_shift_sq {
            max_shift_sq = shift_sq;
        }
    }
    max_shift_sq.sqrt() < threshold
}

fn fix_empty_clusters(
    centroids: &mut [[f32; 3]],
    points: &[[f32; 3]],
    v: &mut [f32],
    nodes: &mut Vec<KdNode>,
) {
    let mut empty_centroids: Vec<usize> = Vec::new();
    for i in 0..centroids.len() {
        if v[i] == 0.0 {
            empty_centroids.push(i);
        }
    }

    if empty_centroids.is_empty() {
        return;
    }

    let non_empty_centroids: Vec<[f32; 3]> = centroids
        .iter()
        .enumerate()
        .filter(|&(i, _)| v[i] > 0.0)
        .map(|(_, &c)| c)
        .collect();

    let non_empty_ids: Vec<u32> = centroids
        .iter()
        .enumerate()
        .filter(|&(i, _)| v[i] > 0.0)
        .map(|(i, _)| i as u32)
        .collect();

    if non_empty_centroids.is_empty() {
        return;
    }

    nodes.clear();
    let mut temp_indices: Vec<usize> = (0..non_empty_centroids.len()).collect();
    let root = build_kd_tree(&non_empty_centroids, &non_empty_ids, &mut temp_indices, nodes, 0);

    let mut distances: Vec<(f32, usize)> = Vec::with_capacity(points.len());
    for (i, p) in points.iter().enumerate() {
        let mut best_id = 0u32;
        let mut best_dist_sq = f32::INFINITY;
        if non_empty_centroids.len() > 32 {
            nearest_neighbor(nodes, root, p, &mut best_id, &mut best_dist_sq);
            distances.push((best_dist_sq, i));
        } else {
            let mut local_best_id = non_empty_ids[0] as usize;
            for (j, &c) in non_empty_centroids.iter().enumerate() {
                let d = dist_sq(&c, p);
                if d < best_dist_sq {
                    best_dist_sq = d;
                    local_best_id = non_empty_ids[j] as usize;
                }
            }
            let _ = local_best_id;
            distances.push((best_dist_sq, i));
        }
    }

    distances.sort_unstable_by(|a, b| b.0.partial_cmp(&a.0).unwrap_or(std::cmp::Ordering::Equal));

    let num_to_fix = std::cmp::min(empty_centroids.len(), distances.len());
    for idx in 0..num_to_fix {
        let empty_c = empty_centroids[idx];
        let pt_idx = distances[idx].1;
        centroids[empty_c] = points[pt_idx];
        v[empty_c] = 1.0;
    }
}

fn train_centroids(
    points: Vec<[f32; 3]>,
    k: usize,
    rng: &mut SimpleRng,
) -> Vec<[f32; 3]> {
    let mut centroids = vec![[0.0f32; 3]; k];

    if points.is_empty() {
        return centroids;
    }

    // Initialize centroids
    if k <= 2000 {
        let mut min_dist_sq = vec![f32::INFINITY; points.len()];
        centroids[0] = points[rng.gen_range(0, points.len())];
        let init_k = std::cmp::min(k, points.len());
        for c in 1..init_k {
            let prev_centroid = &centroids[c - 1];
            let mut sum_dist_sq = 0.0;
            for (i, p) in points.iter().enumerate() {
                let d = dist_sq(p, prev_centroid);
                if d < min_dist_sq[i] {
                    min_dist_sq[i] = d;
                }
                sum_dist_sq += min_dist_sq[i];
            }

            let target = rng.gen_f32() * sum_dist_sq;
            let mut current_sum = 0.0;
            let mut selected_idx = points.len() - 1;
            for (i, &d) in min_dist_sq.iter().enumerate() {
                current_sum += d;
                if current_sum >= target {
                    selected_idx = i;
                    break;
                }
            }
            centroids[c] = points[selected_idx];
        }
        for c in init_k..k {
            centroids[c] = points[c % points.len()];
        }
    } else {
        let init_k = std::cmp::min(k, points.len());
        let mut indices: Vec<usize> = (0..points.len()).collect();
        for i in 0..init_k {
            let j = rng.gen_range(i, points.len());
            indices.swap(i, j);
            centroids[i] = points[indices[i]];
        }
        for c in init_k..k {
            centroids[c] = points[c % points.len()];
        }
    }

    let data_scale = compute_data_scale(&points);
    let conv_threshold = 1e-4 * data_scale;

    let mut old_centroids = centroids.clone();
    let mut v = vec![1.0f32; k];

    let batch_size = 2048;
    let max_epochs = 10;

    let mut nodes = Vec::new();
    let mut indices = Vec::new();

    for _epoch in 0..max_epochs {
        let mut shuffle_indices: Vec<usize> = (0..points.len()).collect();
        for i in (1..points.len()).rev() {
            let j = rng.gen_range(0, i + 1);
            shuffle_indices.swap(i, j);
        }

        for chunk in shuffle_indices.chunks(batch_size) {
            let root = if k > 32 {
                rebuild_tree(&centroids, &mut nodes, &mut indices)
            } else {
                -1
            };

            for &idx in chunk {
                let p = &points[idx];
                let c = if k > 32 {
                    find_closest_kd(&nodes, root, p)
                } else {
                    find_closest_linear(&centroids, p)
                };

                v[c] += 1.0;
                if v[c] > 5000.0 {
                    v[c] = 5000.0;
                }
                let eta = 1.0 / v[c];
                centroids[c][0] += eta * (p[0] - centroids[c][0]);
                centroids[c][1] += eta * (p[1] - centroids[c][1]);
                centroids[c][2] += eta * (p[2] - centroids[c][2]);
            }
        }

        fix_empty_clusters(&mut centroids, &points, &mut v, &mut nodes);

        if has_converged(&old_centroids, &centroids, conv_threshold) {
            break;
        }
        old_centroids.copy_from_slice(&centroids);
    }

    centroids
}

fn run_streaming<R: Read>(
    mut stdin: R,
    _num_points: usize,
    num_clusters: usize,
    temp_path: &std::path::Path,
    rng: &mut SimpleRng,
    sample_limit: usize,
) -> io::Result<()> {
    let temp_file = std::fs::File::create(temp_path)?;
    let mut temp_writer = BufWriter::with_capacity(1024 * 1024, temp_file);

    let mut chunk = vec![0u8; 65536 * 12];
    let mut total_points_read = 0;
    let mut reservoir = Vec::with_capacity(sample_limit);

    loop {
        let mut total_read = 0;
        while total_read < chunk.len() {
            match stdin.read(&mut chunk[total_read..]) {
                Ok(0) => break,
                Ok(n) => total_read += n,
                Err(ref e) if e.kind() == io::ErrorKind::Interrupted => {}
                Err(e) => return Err(e),
            }
        }
        if total_read == 0 {
            break;
        }
        let actual_bytes = (total_read / 12) * 12;
        if actual_bytes == 0 {
            break;
        }

        temp_writer.write_all(&chunk[..actual_bytes])?;

        let num_points_in_chunk = actual_bytes / 12;
        for i in 0..num_points_in_chunk {
            let pos = i * 12;
            let x = f32::from_le_bytes([chunk[pos], chunk[pos + 1], chunk[pos + 2], chunk[pos + 3]]);
            let y = f32::from_le_bytes([chunk[pos + 4], chunk[pos + 5], chunk[pos + 6], chunk[pos + 7]]);
            let z = f32::from_le_bytes([chunk[pos + 8], chunk[pos + 9], chunk[pos + 10], chunk[pos + 11]]);
            let p = [x, y, z];

            let global_idx = total_points_read + i;
            if global_idx < sample_limit {
                reservoir.push(p);
            } else {
                let r = rng.gen_range(0, global_idx + 1);
                if r < sample_limit {
                    reservoir[r] = p;
                }
            }
        }
        total_points_read += num_points_in_chunk;
    }
    temp_writer.flush()?;

    let centroids = train_centroids(reservoir, num_clusters, rng);

    let temp_file = std::fs::File::open(temp_path)?;
    let mut temp_reader = BufReader::with_capacity(1024 * 1024, temp_file);
    let mut stdout = BufWriter::with_capacity(1024 * 1024, io::stdout());

    let mut out_chunk = vec![0u8; 65536 * 4];

    let mut nodes = Vec::new();
    let mut indices = Vec::new();
    let root = if num_clusters > 32 {
        rebuild_tree(&centroids, &mut nodes, &mut indices)
    } else {
        -1
    };

    loop {
        let mut total_read = 0;
        while total_read < chunk.len() {
            match temp_reader.read(&mut chunk[total_read..]) {
                Ok(0) => break,
                Ok(n) => total_read += n,
                Err(ref e) if e.kind() == io::ErrorKind::Interrupted => {}
                Err(e) => return Err(e),
            }
        }
        if total_read == 0 {
            break;
        }
        let actual_bytes = (total_read / 12) * 12;
        if actual_bytes == 0 {
            break;
        }

        let num_points_in_chunk = actual_bytes / 12;
        for i in 0..num_points_in_chunk {
            let pos = i * 12;
            let x = f32::from_le_bytes([chunk[pos], chunk[pos + 1], chunk[pos + 2], chunk[pos + 3]]);
            let y = f32::from_le_bytes([chunk[pos + 4], chunk[pos + 5], chunk[pos + 6], chunk[pos + 7]]);
            let z = f32::from_le_bytes([chunk[pos + 8], chunk[pos + 9], chunk[pos + 10], chunk[pos + 11]]);
            let p = [x, y, z];

            let c = if num_clusters > 32 {
                find_closest_kd(&nodes, root, &p)
            } else {
                find_closest_linear(&centroids, &p)
            };

            let out_pos = i * 4;
            out_chunk[out_pos..out_pos + 4].copy_from_slice(&(c as u32).to_le_bytes());
        }
        stdout.write_all(&out_chunk[..num_points_in_chunk * 4])?;
    }
    stdout.flush()?;

    Ok(())
}

fn run() -> io::Result<()> {
    let mut stdin = BufReader::new(io::stdin());

    let mut line = String::new();
    while let Ok(n) = stdin.read_line(&mut line) {
        if n == 0 {
            break;
        }
        if !line.trim().is_empty() {
            break;
        }
        line.clear();
    }

    let parts: Vec<&str> = line.split_whitespace().collect();
    if parts.len() < 2 {
        return Err(io::Error::new(io::ErrorKind::InvalidData, "Invalid header"));
    }
    let num_points: usize = parts[0].parse().unwrap();
    let num_clusters: usize = parts[1].parse().unwrap();

    if num_clusters == 0 {
        return Err(io::Error::new(io::ErrorKind::InvalidData, "Num clusters cannot be 0"));
    }

    let memory_threshold = 5_000_000;

    let mut rng = SimpleRng::new(1337);
    let sample_limit = std::cmp::min(num_points, 500_000);

    if num_points <= memory_threshold {
        let mut points = Vec::with_capacity(num_points);
        let mut reservoir = Vec::with_capacity(sample_limit);

        let mut chunk = vec![0u8; 65536 * 12];
        let mut total_points_read = 0;

        loop {
            let mut total_read = 0;
            while total_read < chunk.len() {
                match stdin.read(&mut chunk[total_read..]) {
                    Ok(0) => break,
                    Ok(n) => total_read += n,
                    Err(ref e) if e.kind() == io::ErrorKind::Interrupted => {}
                    Err(e) => return Err(e),
                }
            }
            if total_read == 0 {
                break;
            }
            let actual_bytes = (total_read / 12) * 12;
            if actual_bytes == 0 {
                break;
            }

            let num_points_in_chunk = actual_bytes / 12;
            for i in 0..num_points_in_chunk {
                let pos = i * 12;
                let x = f32::from_le_bytes([chunk[pos], chunk[pos + 1], chunk[pos + 2], chunk[pos + 3]]);
                let y = f32::from_le_bytes([chunk[pos + 4], chunk[pos + 5], chunk[pos + 6], chunk[pos + 7]]);
                let z = f32::from_le_bytes([chunk[pos + 8], chunk[pos + 9], chunk[pos + 10], chunk[pos + 11]]);
                let p = [x, y, z];
                points.push(p);

                let global_idx = total_points_read + i;
                if global_idx < sample_limit {
                    reservoir.push(p);
                } else {
                    let r = rng.gen_range(0, global_idx + 1);
                    if r < sample_limit {
                        reservoir[r] = p;
                    }
                }
            }
            total_points_read += num_points_in_chunk;
        }

        let centroids = train_centroids(reservoir, num_clusters, &mut rng);

        let mut stdout = BufWriter::with_capacity(1024 * 1024, io::stdout());
        let mut out_chunk = vec![0u8; 65536 * 4];

        let mut nodes = Vec::new();
        let mut indices = Vec::new();
        let root = if num_clusters > 32 {
            rebuild_tree(&centroids, &mut nodes, &mut indices)
        } else {
            -1
        };

        for chunk in points.chunks(65536) {
            for (i, p) in chunk.iter().enumerate() {
                let c = if num_clusters > 32 {
                    find_closest_kd(&nodes, root, p)
                } else {
                    find_closest_linear(&centroids, p)
                };
                let out_pos = i * 4;
                out_chunk[out_pos..out_pos + 4].copy_from_slice(&(c as u32).to_le_bytes());
            }
            stdout.write_all(&out_chunk[..chunk.len() * 4])?;
        }
        stdout.flush()?;
    } else {
        let pid = std::process::id();
        let temp_dir = std::env::temp_dir();
        let temp_path = temp_dir.join(format!("clustering_temp_{}.bin", pid));

        let res = run_streaming(stdin, num_points, num_clusters, &temp_path, &mut rng, sample_limit);
        let _ = std::fs::remove_file(&temp_path);
        res?;
    }
    Ok(())
}

fn main() {
    if let Err(e) = run() {
        eprintln!("Error: {:?}", e);
        std::process::exit(1);
    }
}
