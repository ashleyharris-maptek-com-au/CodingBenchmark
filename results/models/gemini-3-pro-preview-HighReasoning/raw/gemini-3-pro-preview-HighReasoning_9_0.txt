using System;
using System.IO;
using System.Collections.Generic;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;
using System.Diagnostics;

public class Program {
    public class FastScanner {
        private const int BufferSize = 1048576;
        private readonly Stream stream;
        private readonly byte[] buffer = new byte[BufferSize];
        private int head = 0;
        private int tail = 0;

        public FastScanner() {
            stream = Console.OpenStandardInput();
        }

        private byte Read() {
            if (head >= tail) {
                head = 0;
                tail = stream.Read(buffer, 0, BufferSize);
                if (tail <= 0) return 0;
            }
            return buffer[head++];
        }

        public int NextInt() {
            int val = 0;
            byte b = Read();
            while (b > 0 && b <= 32) {
                b = Read();
            }
            if (b == 0) return -1;
            while (b > 32) {
                if (b >= '0' && b <= '9') {
                    val = val * 10 + (b - '0');
                }
                b = Read();
            }
            return val;
        }
    }

    public class FfdSegmentTree {
        public int[] tree;
        public int n;

        public FfdSegmentTree(int size, int initialCapacity) {
            n = 1;
            while (n < size) n <<= 1;
            tree = new int[2 * n];
            for (int i = 0; i < size; i++) {
                tree[n + i] = initialCapacity;
            }
            for (int i = n - 1; i > 0; i--) {
                tree[i] = Math.Max(tree[2 * i], tree[2 * i + 1]);
            }
        }

        public int FindFirstFit(int weight) {
            if (tree[1] < weight) return -1;
            int idx = 1;
            while (idx < n) {
                if (tree[2 * idx] >= weight) {
                    idx = 2 * idx;
                } else {
                    idx = 2 * idx + 1;
                }
            }
            return idx - n;
        }

        public void UpdateCapacity(int binIdx, int newCapacity) {
            int idx = n + binIdx;
            tree[idx] = newCapacity;
            while (idx > 1) {
                idx >>= 1;
                tree[idx] = Math.Max(tree[2 * idx], tree[2 * idx + 1]);
            }
        }
    }

    public class PackingResult {
        public int NumBins;
        public int[] BinHead;
        public int[] CutNext;
        public int[] BinCapacities;
        public int NumBinsUsed;
    }

    public static PackingResult SolveSingle(int[] cuts, int stockLength, double perturbationScale, int seed, int maxRefineSteps) {
        int n = cuts.Length;
        Random rand = new Random(seed);

        double[] keys = new double[n];
        for (int i = 0; i < n; i++) {
            keys[i] = -(cuts[i] + (rand.NextDouble() - 0.5) * perturbationScale);
        }

        int[] indices = new int[n];
        for (int i = 0; i < n; i++) indices[i] = i;
        Array.Sort(keys, indices);

        FfdSegmentTree tree = new FfdSegmentTree(n, stockLength);
        int[] binHead = new int[n];
        int[] cutNext = new int[n];
        for (int i = 0; i < n; i++) {
            binHead[i] = -1;
            cutNext[i] = -1;
        }
        int[] binCapacities = new int[n];
        for (int i = 0; i < n; i++) binCapacities[i] = stockLength;

        int numBinsUsed = 0;

        for (int i = 0; i < n; i++) {
            int cutIdx = indices[i];
            int w = cuts[cutIdx];
            int dest = tree.FindFirstFit(w);
            if (dest > numBinsUsed) {
                numBinsUsed = dest;
            }
            cutNext[cutIdx] = binHead[dest];
            binHead[dest] = cutIdx;

            binCapacities[dest] -= w;
            tree.UpdateCapacity(dest, binCapacities[dest]);
        }
        numBinsUsed++;

        // Refinement: try to eliminate emptiest bins
        int[] activeBins = new int[numBinsUsed];
        int activeCount = 0;
        for (int i = 0; i < numBinsUsed; i++) {
            if (binHead[i] != -1) {
                activeBins[activeCount++] = i;
            }
        }

        Array.Sort(activeBins, 0, activeCount, Comparer<int>.Create((a, b) => binCapacities[b].CompareTo(binCapacities[a])));

        List<int> undoBinIdx = new List<int>();
        List<int> undoOriginalCap = new List<int>();
        List<int> tempPlacedBins = new List<int>();
        int[] tempCuts = new int[1000];

        int eliminated = 0;
        int steps = Math.Min(activeCount, maxRefineSteps);

        for (int step = 0; step < steps; step++) {
            int targetBin = activeBins[step];
            if (binHead[targetBin] == -1) continue;

            int curr = binHead[targetBin];
            int cutCount = 0;
            while (curr != -1) {
                if (cutCount >= tempCuts.Length) {
                    Array.Resize(ref tempCuts, tempCuts.Length * 2);
                }
                tempCuts[cutCount++] = curr;
                curr = cutNext[curr];
            }

            undoBinIdx.Clear();
            undoOriginalCap.Clear();
            tempPlacedBins.Clear();

            int originalTargetCap = binCapacities[targetBin];
            tree.UpdateCapacity(targetBin, 0);
            undoBinIdx.Add(targetBin);
            undoOriginalCap.Add(originalTargetCap);

            bool success = true;
            for (int i = 0; i < cutCount; i++) {
                int cutIdx = tempCuts[i];
                int w = cuts[cutIdx];
                int dest = tree.FindFirstFit(w);
                if (dest == -1 || dest >= numBinsUsed) {
                    success = false;
                    break;
                }

                undoBinIdx.Add(dest);
                undoOriginalCap.Add(w);
                tempPlacedBins.Add(dest);

                binCapacities[dest] -= w;
                tree.UpdateCapacity(dest, binCapacities[dest]);
            }

            if (success) {
                for (int i = 0; i < cutCount; i++) {
                    int cutIdx = tempCuts[i];
                    int dest = tempPlacedBins[i];
                    cutNext[cutIdx] = binHead[dest];
                    binHead[dest] = cutIdx;
                }
                binHead[targetBin] = -1;
                binCapacities[targetBin] = 0;
                eliminated++;
            } else {
                binCapacities[targetBin] = originalTargetCap;
                tree.UpdateCapacity(targetBin, originalTargetCap);
                for (int j = undoBinIdx.Count - 1; j >= 1; j--) {
                    int b = undoBinIdx[j];
                    int w = undoOriginalCap[j];
                    binCapacities[b] += w;
                    tree.UpdateCapacity(b, binCapacities[b]);
                }
            }
        }

        int finalBinsUsed = 0;
        for (int i = 0; i < numBinsUsed; i++) {
            if (binHead[i] != -1) finalBinsUsed++;
        }

        return new PackingResult {
            NumBins = finalBinsUsed,
            BinHead = binHead,
            CutNext = cutNext,
            BinCapacities = binCapacities,
            NumBinsUsed = numBinsUsed
        };
    }

    public static void Main(string[] args) {
        FastScanner scanner = new FastScanner();
        int n = scanner.NextInt();
        if (n <= 0) return;
        int stockLength = scanner.NextInt();

        int[] cuts = new int[n];
        long sumOfCuts = 0;
        for (int i = 0; i < n; i++) {
            cuts[i] = scanner.NextInt();
            sumOfCuts += cuts[i];
        }

        int lowerBound = (int)((sumOfCuts + stockLength - 1) / stockLength);

        int numThreads = 1;
        int maxRefineSteps = 1000;
        if (n <= 10000) {
            numThreads = Math.Min(64, Environment.ProcessorCount);
            maxRefineSteps = 5000;
        } else if (n <= 100000) {
            numThreads = Math.Min(16, Environment.ProcessorCount);
            maxRefineSteps = 2000;
        } else if (n <= 1000000) {
            numThreads = Math.Min(8, Environment.ProcessorCount);
            maxRefineSteps = 1000;
        } else {
            numThreads = Math.Min(4, Environment.ProcessorCount);
            maxRefineSteps = 500;
        }

        Stopwatch globalSw = Stopwatch.StartNew();
        PackingResult bestOverall = null;
        int iteration = 0;

        while (true) {
            long elapsed = globalSw.ElapsedMilliseconds;
            if (iteration > 0 && elapsed >= 8000) {
                break;
            }

            PackingResult[] results = new PackingResult[numThreads];
            int currentIteration = iteration;

            Parallel.For(0, numThreads, i => {
                double scale = 0;
                int seed = 1337 + currentIteration * 100 + i;
                if (i > 0 || currentIteration > 0) {
                    Random threadRand = new Random(seed);
                    scale = 0.1 + threadRand.NextDouble() * 9.9;
                }
                results[i] = SolveSingle(cuts, stockLength, scale, seed, maxRefineSteps);
            });

            for (int i = 0; i < numThreads; i++) {
                if (bestOverall == null || results[i].NumBins < bestOverall.NumBins) {
                    bestOverall = results[i];
                }
            }

            if (bestOverall.NumBins == lowerBound) {
                break;
            }

            iteration++;
        }

        var best = bestOverall;

        var stdout = new StreamWriter(Console.OpenStandardOutput(), System.Text.Encoding.ASCII, 1048576);
        stdout.WriteLine(best.NumBins);
        int[] printBuf = new int[n];
        for (int i = 0; i < best.NumBinsUsed; i++) {
            int curr = best.BinHead[i];
            if (curr == -1) continue;
            int cnt = 0;
            while (curr != -1) {
                printBuf[cnt++] = curr;
                curr = best.CutNext[curr];
            }
            Array.Sort(printBuf, 0, cnt);
            for (int j = 0; j < cnt; j++) {
                if (j > 0) stdout.Write(' ');
                stdout.Write(printBuf[j]);
            }
            stdout.WriteLine();
        }
        stdout.Flush();
    }
}
