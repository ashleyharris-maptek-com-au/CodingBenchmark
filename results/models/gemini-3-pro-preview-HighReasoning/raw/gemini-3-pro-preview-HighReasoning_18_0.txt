// Fast I/O
struct FastInput {
    static const int BUF_SIZE = 1 << 20;
    char buf[BUF_SIZE];
    int buf_ptr = 0;
    int buf_len = 0;

    inline int read_char() {
        if (buf_ptr >= buf_len) {
            buf_len = fread(buf, 1, BUF_SIZE, stdin);
            buf_ptr = 0;
            if (buf_len == 0) return EOF;
        }
        return (unsigned char)buf[buf_ptr++];
    }

    inline int read_int() {
        int x = 0;
        int c = read_char();
        while (c <= ' ' && c != EOF) {
            c = read_char();
        }
        if (c == EOF) return -1;
        bool neg = false;
        if (c == '-') {
            neg = true;
            c = read_char();
        }
        while (c >= '0' && c <= '9') {
            x = x * 10 + (c - '0');
            c = read_char();
        }
        return neg ? -x : x;
    }
};

// CSR representation
int N, M;
vector<int> head;
vector<int> csr_edges;
vector<int> rev;
vector<int> initial_cap;

// Bridge finder variables
vector<int> tin;
vector<int> low;
int timer_dfs = 0;
int bridge_u = -1, bridge_v = -1;

void dfs_bridge(int u, int p_edge_idx) {
    tin[u] = low[u] = timer_dfs++;
    for (int i = head[u]; i < head[u+1]; ++i) {
        if (i == p_edge_idx) continue;
        int v = csr_edges[i];
        if (tin[v] != -1) {
            low[u] = min(low[u], tin[v]);
        } else {
            dfs_bridge(v, rev[i]);
            low[u] = min(low[u], low[v]);
            if (low[v] > tin[u]) {
                bridge_u = u;
                bridge_v = v;
            }
        }
    }
}

// Stoer-Wagner structs
struct SW_Edge {
    int to;
    int weight;
    int rev;
};

void run_stoer_wagner() {
    vector<vector<SW_Edge>> adj(N);
    for (int u = 0; u < N; ++u) {
        vector<int> neighbors;
        for (int i = head[u]; i < head[u+1]; ++i) {
            int v = csr_edges[i];
            if (v != u) neighbors.push_back(v);
        }
        sort(neighbors.begin(), neighbors.end());
        int last = -1;
        for (int v : neighbors) {
            if (v == last) {
                adj[u].back().weight++;
            } else {
                adj[u].push_back({v, 1, -1});
                last = v;
            }
        }
    }
    for (int u = 0; u < N; ++u) {
        for (int i = 0; i < (int)adj[u].size(); ++i) {
            if (adj[u][i].rev == -1) {
                int v = adj[u][i].to;
                for (int j = 0; j < (int)adj[v].size(); ++j) {
                    if (adj[v][j].to == u) {
                        adj[u][i].rev = j;
                        adj[v][j].rev = i;
                        break;
                    }
                }
            }
        }
    }

    vector<bool> active(N, true);
    vector<vector<int>> components(N);
    for (int i = 0; i < N; ++i) {
        components[i] = {i};
    }

    int min_cut = 1e9;
    vector<int> best_partition;

    for (int phase = 0; phase < N - 1; ++phase) {
        vector<int> active_verts;
        for (int i = 0; i < N; ++i) {
            if (active[i]) active_verts.push_back(i);
        }

        int n_act = active_verts.size();
        vector<int> w_sum(N, 0);
        vector<bool> in_A(N, false);

        int prev = -1;
        int curr = active_verts[0];

        in_A[curr] = true;
        for (auto& edge : adj[curr]) {
            if (active[edge.to] && !in_A[edge.to]) {
                w_sum[edge.to] += edge.weight;
            }
        }

        for (int step = 1; step < n_act; ++step) {
            int next_v = -1;
            int max_w = -1;
            for (int v : active_verts) {
                if (!in_A[v]) {
                    if (w_sum[v] > max_w) {
                        max_w = w_sum[v];
                        next_v = v;
                    }
                }
            }

            prev = curr;
            curr = next_v;

            in_A[curr] = true;
            for (auto& edge : adj[curr]) {
                if (active[edge.to] && !in_A[edge.to]) {
                    w_sum[edge.to] += edge.weight;
                }
            }
        }

        int s = prev;
        int t = curr;
        int cut_val = w_sum[t];

        if (cut_val < min_cut) {
            min_cut = cut_val;
            best_partition = components[t];
        }

        // Merge t into s
        vector<int> pos(N, -1);
        for (int i = 0; i < (int)adj[s].size(); ++i) {
            if (active[adj[s][i].to]) {
                pos[adj[s][i].to] = i;
            }
        }

        for (auto& edge : adj[t]) {
            int v = edge.to;
            if (v == s || !active[v]) continue;
            int w = edge.weight;
            if (pos[v] != -1) {
                adj[s][pos[v]].weight += w;
                adj[v][adj[s][pos[v]].rev].weight += w;
            } else {
                adj[s].push_back({v, w, (int)adj[v].size()});
                adj[v].push_back({s, w, (int)adj[s].size() - 1});
                pos[v] = adj[s].size() - 1;
            }
        }

        components[s].insert(components[s].end(), components[t].begin(), components[t].end());
        active[t] = false;
    }

    vector<bool> in_S(N, false);
    for (int v : best_partition) {
        in_S[v] = true;
    }

    int cut_count = 0;
    for (int u = 0; u < N; ++u) {
        if (in_S[u]) {
            for (int i = head[u]; i < head[u+1]; ++i) {
                int v = csr_edges[i];
                if (!in_S[v]) {
                    cut_count++;
                }
            }
        }
    }

    cout << cut_count << "\n";
    for (int u = 0; u < N; ++u) {
        if (in_S[u]) {
            for (int i = head[u]; i < head[u+1]; ++i) {
                int v = csr_edges[i];
                if (!in_S[v]) {
                    cout << u << " " << v << "\n";
                }
            }
        }
    }
}

// Parallel Dinic's structs
struct ThreadWorker {
    vector<int> cap;
    vector<int> level;
    vector<int> q_arr;
    vector<int> ptr;
    vector<int> changed_edges;

    ThreadWorker(int N, int M, const vector<int>& init_cap) {
        cap = init_cap;
        level.resize(N);
        q_arr.resize(N);
        ptr.resize(N);
    }

    bool bfs(int s, int t, const vector<int>& head, const vector<int>& csr_edges) {
        fill(level.begin(), level.end(), -1);
        level[s] = 0;
        int q_head = 0, q_tail = 0;
        q_arr[q_tail++] = s;
        while (q_head < q_tail) {
            int u = q_arr[q_head++];
            for (int i = head[u]; i < head[u+1]; ++i) {
                int v = csr_edges[i];
                if (cap[i] > 0 && level[v] == -1) {
                    level[v] = level[u] + 1;
                    q_arr[q_tail++] = v;
                }
            }
        }
        return level[t] != -1;
    }

    int dfs(int u, int t, int push, const vector<int>& head, const vector<int>& csr_edges, const vector<int>& rev) {
        if (push == 0) return 0;
        if (u == t) return push;
        for (int& cid = ptr[u]; cid < head[u+1]; ++cid) {
            int v = csr_edges[cid];
            int tr = rev[cid];
            if (level[u] + 1 != level[v] || cap[cid] == 0) continue;
            int pushed = dfs(v, t, min(push, cap[cid]), head, csr_edges, rev);
            if (pushed == 0) continue;
            cap[cid] -= pushed;
            cap[tr] += pushed;
            changed_edges.push_back(cid);
            changed_edges.push_back(tr);
            return pushed;
        }
        return 0;
    }

    int max_flow_limit(int s, int t, int limit, const vector<int>& head, const vector<int>& csr_edges, const vector<int>& rev) {
        int flow = 0;
        while (flow < limit && bfs(s, t, head, csr_edges)) {
            for (int i = 0; i < (int)level.size(); ++i) ptr[i] = head[i];
            while (flow < limit) {
                int pushed = dfs(s, t, 1, head, csr_edges, rev);
                if (pushed == 0) break;
                flow += pushed;
            }
        }
        return flow;
    }
};

struct Task {
    int s;
    int t;
};

mutex queue_mutex;
vector<Task> tasks;
int next_task_idx = 0;

atomic<int> best_cut_val;
mutex cut_mutex;
vector<bool> best_in_S;

void worker_thread_fn(int N, int M, const vector<int>& head, const vector<int>& csr_edges, const vector<int>& rev) {
    ThreadWorker worker(N, M, initial_cap);
    while (true) {
        Task task;
        {
            lock_guard<mutex> lock(queue_mutex);
            if (next_task_idx >= (int)tasks.size()) {
                break;
            }
            task = tasks[next_task_idx++];
        }

        int current_limit = best_cut_val.load(memory_order_relaxed);
        int flow = worker.max_flow_limit(task.s, task.t, current_limit, head, csr_edges, rev);

        if (flow < current_limit) {
            lock_guard<mutex> lock(cut_mutex);
            if (flow < best_cut_val.load(memory_order_relaxed)) {
                best_cut_val.store(flow, memory_order_relaxed);
                // Reconstruct reachable component
                fill(worker.level.begin(), worker.level.end(), -1);
                worker.level[task.s] = 0;
                int q_head = 0, q_tail = 0;
                worker.q_arr[q_tail++] = task.s;
                while (q_head < q_tail) {
                    int u = worker.q_arr[q_head++];
                    for (int i = head[u]; i < head[u+1]; ++i) {
                        int v = csr_edges[i];
                        if (worker.cap[i] > 0 && worker.level[v] == -1) {
                            worker.level[v] = 0;
                            worker.q_arr[q_tail++] = v;
                        }
                    }
                }
                for (int i = 0; i < N; ++i) {
                    best_in_S[i] = (worker.level[i] != -1);
                }
            }
        }

        for (int idx : worker.changed_edges) {
            worker.cap[idx] = 1;
        }
        worker.changed_edges.clear();
    }
}

void solve_large_graph_parallel() {
    int s1 = -1, s2 = -1, s3 = -1;
    int d1 = 1e9, d2 = 1e9, d3 = 1e9;
    for (int i = 0; i < N; ++i) {
        int d = head[i+1] - head[i];
        if (d < d1) {
            d3 = d2; s3 = s2;
            d2 = d1; s2 = s1;
            d1 = d;  s1 = i;
        } else if (d < d2) {
            d3 = d2; s3 = s2;
            d2 = d;  s2 = i;
        } else if (d < d3) {
            d3 = d;  s3 = i;
        }
    }

    vector<int> sources;
    if (s1 != -1) sources.push_back(s1);
    if (s2 != -1 && s2 != s1) sources.push_back(s2);
    if (s3 != -1 && s3 != s1 && s3 != s2) sources.push_back(s3);

    best_cut_val.store(d1, memory_order_relaxed);
    best_in_S.assign(N, false);
    best_in_S[s1] = true;

    vector<pair<int, int>> low_degs;
    for (int i = 0; i < N; ++i) {
        int d = head[i+1] - head[i];
        if (low_degs.size() < 15) {
            low_degs.push_back({d, i});
            push_heap(low_degs.begin(), low_degs.end());
        } else if (d < low_degs.front().first) {
            pop_heap(low_degs.begin(), low_degs.end());
            low_degs.back() = {d, i};
            push_heap(low_degs.begin(), low_degs.end());
        }
    }

    mt19937 rng(1337);
    vector<int> dist(N, -1);
    vector<int> q_dist(N);

    tasks.clear();
    next_task_idx = 0;

    for (int s : sources) {
        fill(dist.begin(), dist.end(), -1);
        int q_h = 0, q_t = 0;
        dist[s] = 0;
        q_dist[q_t++] = s;
        while (q_h < q_t) {
            int u = q_dist[q_h++];
            int d = dist[u];
            for (int i = head[u]; i < head[u+1]; ++i) {
                int v = csr_edges[i];
                if (dist[v] == -1) {
                    dist[v] = d + 1;
                    q_dist[q_t++] = v;
                }
            }
        }

        vector<int> candidates;
        for (int i = 0; i < 15; ++i) {
            int idx = N - 1 - i;
            if (idx >= 0) candidates.push_back(q_dist[idx]);
        }
        for (int i = 0; i < 15; ++i) {
            candidates.push_back(rng() % N);
        }
        for (auto& p : low_degs) {
            candidates.push_back(p.second);
        }

        sort(candidates.begin(), candidates.end());
        candidates.erase(unique(candidates.begin(), candidates.end()), candidates.end());
        candidates.erase(remove(candidates.begin(), candidates.end(), s), candidates.end());

        for (int t : candidates) {
            tasks.push_back({s, t});
        }
    }

    int num_threads = min(8, (int)thread::hardware_concurrency());
    if (num_threads < 1) num_threads = 1;

    vector<thread> pool;
    for (int i = 0; i < num_threads; ++i) {
        pool.push_back(thread(worker_thread_fn, N, M, ref(head), ref(csr_edges), ref(rev)));
    }

    for (auto& th : pool) {
        th.join();
    }

    int cut_count = 0;
    for (int u = 0; u < N; ++u) {
        if (best_in_S[u]) {
            for (int i = head[u]; i < head[u+1]; ++i) {
                int v = csr_edges[i];
                if (!best_in_S[v]) {
                    cut_count++;
                }
            }
        }
    }

    cout << cut_count << "\n";
    for (int u = 0; u < N; ++u) {
        if (best_in_S[u]) {
            for (int i = head[u]; i < head[u+1]; ++i) {
                int v = csr_edges[i];
                if (!best_in_S[v]) {
                    cout << u << " " << v << "\n";
                }
            }
        }
    }
}

int main() {
    static FastInput in; // static to prevent stack overflow!
    N = in.read_int();
    M = in.read_int();
    if (N <= 0) return 0;

    struct RawEdge {
        int u, v;
    };
    vector<RawEdge> raw_edges;
    raw_edges.reserve(M);
    for (int i = 0; i < M; ++i) {
        int u = in.read_int();
        int v = in.read_int();
        raw_edges.push_back({u, v});
    }

    head.assign(N + 1, 0);
    for (const auto& e : raw_edges) {
        head[e.u]++;
        head[e.v]++;
    }

    int current_sum = 0;
    for (int i = 0; i < N; ++i) {
        int deg = head[i];
        head[i] = current_sum;
        current_sum += deg;
    }
    head[N] = current_sum;

    csr_edges.resize(2 * M);
    rev.resize(2 * M);
    initial_cap.assign(2 * M, 1);

    vector<int> current_pos = head;
    for (int i = 0; i < M; ++i) {
        int u = raw_edges[i].u;
        int v = raw_edges[i].v;
        int idx_uv = current_pos[u]++;
        int idx_vu = current_pos[v]++;
        csr_edges[idx_uv] = v;
        csr_edges[idx_vu] = u;
        rev[idx_uv] = idx_vu;
        rev[idx_vu] = idx_uv;
    }

    raw_edges.clear();
    raw_edges.shrink_to_fit();

    int s_min = -1;
    int d_min = 1e9;
    for (int i = 0; i < N; ++i) {
        int d = head[i+1] - head[i];
        if (d < d_min) {
            d_min = d;
            s_min = i;
        }
    }

    if (d_min == 1) {
        cout << 1 << "\n";
        cout << s_min << " " << csr_edges[head[s_min]] << "\n";
        return 0;
    }

    tin.assign(N, -1);
    low.assign(N, -1);
    timer_dfs = 0;
    bridge_u = -1;
    bridge_v = -1;
    dfs_bridge(s_min, -1);

    if (bridge_u != -1) {
        cout << 1 << "\n";
        cout << bridge_u << " " << bridge_v << "\n";
        return 0;
    }

    if (d_min == 2) {
        cout << 2 << "\n";
        cout << s_min << " " << csr_edges[head[s_min]] << "\n";
        cout << s_min << " " << csr_edges[head[s_min] + 1] << "\n";
        return 0;
    }

    if (N <= 2000) {
        run_stoer_wagner();
    } else {
        solve_large_graph_parallel();
    }

    return 0;
}
