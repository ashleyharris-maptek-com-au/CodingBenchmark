declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type TaskInput = {
  id: string | number;
  duration: number;
  deps?: Array<string | number>;
  deadline?: number;
  weight?: number;
};

type Task = {
  id: string | number;
  duration: number;
  deps: number[];
  children: number[];
  deadline: number;
  weight: number;
  indegree: number;
  remainingDeps: number;
  est: number;
  tail: number;
  level: number;
  avgWeight: number;
  descendantWeight: number;
};

type QueueItem = {
  idx: number;
  version: number;
};

class BinaryHeap<T> {
  private readonly data: T[] = [];
  private readonly compare: (a: T, b: T) => number;

  constructor(compare: (a: T, b: T) => number) {
    this.compare = compare;
  }

  push(value: T): void {
    const arr = this.data;
    arr.push(value);
    let i = arr.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (this.compare(arr[i], arr[p]) >= 0) {
        break;
      }
      [arr[i], arr[p]] = [arr[p], arr[i]];
      i = p;
    }
  }

  pop(): T | undefined {
    const arr = this.data;
    if (arr.length === 0) {
      return undefined;
    }
    const top = arr[0];
    const last = arr.pop()!;
    if (arr.length > 0) {
      arr[0] = last;
      let i = 0;
      for (;;) {
        const l = i * 2 + 1;
        const r = l + 1;
        let best = i;
        if (l < arr.length && this.compare(arr[l], arr[best]) < 0) {
          best = l;
        }
        if (r < arr.length && this.compare(arr[r], arr[best]) < 0) {
          best = r;
        }
        if (best === i) {
          break;
        }
        [arr[i], arr[best]] = [arr[best], arr[i]];
        i = best;
      }
    }
    return top;
  }

  get size(): number {
    return this.data.length;
  }
}

const machineCount = Math.max(1, Number(data.machineCount) | 0);
const inputTasks: TaskInput[] = Array.isArray(data.tasks) ? data.tasks : [];
const n = inputTasks.length;

const idToIndex = new Map<string | number, number>();
for (let i = 0; i < n; i++) {
  idToIndex.set(inputTasks[i].id, i);
}

const tasks: Task[] = new Array(n);
for (let i = 0; i < n; i++) {
  const src = inputTasks[i];
  const depsRaw = Array.isArray(src.deps) ? src.deps : [];
  const deps: number[] = [];
  for (let j = 0; j < depsRaw.length; j++) {
    const depIndex = idToIndex.get(depsRaw[j]);
    if (depIndex !== undefined && depIndex !== i) {
      deps.push(depIndex);
    }
  }
  tasks[i] = {
    id: src.id,
    duration: Math.max(0, Number(src.duration) | 0),
    deps,
    children: [],
    deadline: Number.isFinite(src.deadline) ? Number(src.deadline) : Number.MAX_SAFE_INTEGER / 4,
    weight: Math.max(0, Number(src.weight) || 0),
    indegree: 0,
    remainingDeps: 0,
    est: 0,
    tail: 0,
    level: 0,
    avgWeight: 0,
    descendantWeight: 0,
  };
}

for (let i = 0; i < n; i++) {
  const task = tasks[i];
  const seen = new Set<number>();
  for (let j = 0; j < task.deps.length; j++) {
    const dep = task.deps[j];
    if (seen.has(dep)) {
      continue;
    }
    seen.add(dep);
    tasks[dep].children.push(i);
    task.indegree++;
  }
  task.remainingDeps = task.indegree;
}

const topo: number[] = [];
const topoQueue = new Array<number>(n);
let qHead = 0;
let qTail = 0;
for (let i = 0; i < n; i++) {
  if (tasks[i].indegree === 0) {
    topoQueue[qTail++] = i;
  }
}

while (qHead < qTail) {
  const u = topoQueue[qHead++];
  topo.push(u);
  const task = tasks[u];
  for (let i = 0; i < task.children.length; i++) {
    const v = task.children[i];
    tasks[v].remainingDeps--;
    if (tasks[v].remainingDeps === 0) {
      topoQueue[qTail++] = v;
    }
  }
}

if (topo.length !== n) {
  const order = inputTasks.map((task) => task.id);
  process.stdout.write(JSON.stringify({ order }));
  process.exit(0);
}

for (let i = 0; i < n; i++) {
  tasks[i].remainingDeps = tasks[i].indegree;
}

for (let k = 0; k < topo.length; k++) {
  const u = topo[k];
  const task = tasks[u];
  for (let i = 0; i < task.children.length; i++) {
    const v = task.children[i];
    const candidate = task.est + task.duration;
    if (candidate > tasks[v].est) {
      tasks[v].est = candidate;
    }
  }
}

for (let k = topo.length - 1; k >= 0; k--) {
  const u = topo[k];
  const task = tasks[u];
  let bestTail = task.duration;
  let bestLevel = 0;
  let sumWeight = task.weight;
  let sumCount = 1;
  let sumDescWeight = task.weight;
  for (let i = 0; i < task.children.length; i++) {
    const child = tasks[task.children[i]];
    const childTail = task.duration + child.tail;
    if (childTail > bestTail) {
      bestTail = childTail;
    }
    if (child.level + 1 > bestLevel) {
      bestLevel = child.level + 1;
    }
    sumWeight += child.avgWeight;
    sumCount++;
    sumDescWeight += child.descendantWeight;
  }
  task.tail = bestTail;
  task.level = bestLevel;
  task.avgWeight = sumWeight / sumCount;
  task.descendantWeight = sumDescWeight;
}

const versions = new Int32Array(n);
const finishTimes = new Int32Array(n);
const readyTime = new Int32Array(n);
const startTimes = new Int32Array(n);
const assignedMachine = new Int32Array(n);
const machineFree = new Int32Array(machineCount);

for (let i = 0; i < n; i++) {
  assignedMachine[i] = -1;
}

function score(idx: number, now: number): number {
  const task = tasks[idx];
  const predictedFinish = Math.max(now, readyTime[idx]) + task.duration;
  const lateness = predictedFinish - task.deadline;
  const urgency = lateness > 0 ? lateness * (task.weight + 1) : lateness * 0.25;
  return (
    urgency * 1e9 -
    task.tail * 1e6 -
    task.descendantWeight * 1e3 -
    task.weight * 100 -
    task.duration
  );
}

const readyHeap = new BinaryHeap<QueueItem>((a, b) => {
  const sa = score(a.idx, currentTime);
  const sb = score(b.idx, currentTime);
  if (sa !== sb) {
    return sa < sb ? -1 : 1;
  }
  const ta = tasks[a.idx];
  const tb = tasks[b.idx];
  if (ta.deadline !== tb.deadline) {
    return ta.deadline - tb.deadline;
  }
  if (tb.weight !== ta.weight) {
    return tb.weight - ta.weight;
  }
  return a.idx - b.idx;
});

const machineHeap = new BinaryHeap<number>((a, b) => {
  const da = machineFree[a];
  const db = machineFree[b];
  if (da !== db) {
    return da - db;
  }
  return a - b;
});

let currentTime = 0;
let minMachineFree = 0;
for (let i = 0; i < machineCount; i++) {
  machineHeap.push(i);
}

for (let i = 0; i < n; i++) {
  if (tasks[i].indegree === 0) {
    readyHeap.push({ idx: i, version: 0 });
  }
}

let scheduled = 0;
while (scheduled < n) {
  while (readyHeap.size > 0) {
    const machine = machineHeap.pop()!;
    const item = readyHeap.pop()!;
    if (item.version !== versions[item.idx]) {
      machineHeap.push(machine);
      continue;
    }
    const task = tasks[item.idx];
    const start = Math.max(machineFree[machine], readyTime[item.idx]);
    startTimes[item.idx] = start;
    assignedMachine[item.idx] = machine;
    const finish = start + task.duration;
    finishTimes[item.idx] = finish;
    machineFree[machine] = finish;
    machineHeap.push(machine);
    scheduled++;
    for (let i = 0; i < task.children.length; i++) {
      const childIdx = task.children[i];
      if (finish > readyTime[childIdx]) {
        readyTime[childIdx] = finish;
        versions[childIdx]++;
      }
      tasks[childIdx].remainingDeps--;
      if (tasks[childIdx].remainingDeps === 0) {
        readyHeap.push({ idx: childIdx, version: versions[childIdx] });
      }
    }
    minMachineFree = machineFree[0];
    for (let i = 1; i < machineCount; i++) {
      if (machineFree[i] < minMachineFree) {
        minMachineFree = machineFree[i];
      }
    }
    currentTime = minMachineFree;
  }
}

const schedule = new Array(n);
for (let i = 0; i < n; i++) {
  schedule[i] = {
    id: tasks[i].id,
    start: startTimes[i],
    machine: assignedMachine[i],
  };
}

schedule.sort((a, b) => {
  if (a.start !== b.start) {
    return a.start - b.start;
  }
  if (a.machine !== b.machine) {
    return a.machine - b.machine;
  }
  const as = String(a.id);
  const bs = String(b.id);
  return as < bs ? -1 : as > bs ? 1 : 0;
});

process.stdout.write(JSON.stringify({ schedule }));
