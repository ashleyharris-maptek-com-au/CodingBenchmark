declare const require: any;
declare const process: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type Point = [number, number];

interface Town {
  id: string;
  x: number;
  y: number;
  value: number;
}

interface InputData {
  width: number;
  height: number;
  grid: string[];
  ignitions: Point[];
  towns: Town[];
  cutBudget: number;
}

const parsed = data as InputData;
const width = parsed.width | 0;
const height = parsed.height | 0;
const size = width * height;
const cutBudget = Math.max(0, parsed.cutBudget | 0);

const grid = parsed.grid;
const burnable = new Uint8Array(size);
for (let y = 0; y < height; y++) {
  const row = grid[y] || '';
  for (let x = 0; x < width; x++) {
    if (row.charCodeAt(x) === 46) burnable[y * width + x] = 1; // '.'
  }
}

const dirs = [-1, 0, 1, 0, -1];

function inBounds(x: number, y: number): boolean {
  return x >= 0 && x < width && y >= 0 && y < height;
}

function idx(x: number, y: number): number {
  return y * width + x;
}

function xy(index: number): Point {
  return [index % width, (index / width) | 0];
}

function bfsFromSources(blocked: Uint8Array): Uint8Array {
  const seen = new Uint8Array(size);
  const queue = new Int32Array(size);
  let head = 0;
  let tail = 0;

  for (const [x, y] of parsed.ignitions) {
    if (!inBounds(x, y)) continue;
    const start = idx(x, y);
    if (!burnable[start] || blocked[start] || seen[start]) continue;
    seen[start] = 1;
    queue[tail++] = start;
  }

  while (head < tail) {
    const cur = queue[head++];
    const cx = cur % width;
    const cy = (cur / width) | 0;
    for (let k = 0; k < 4; k++) {
      const nx = cx + dirs[k];
      const ny = cy + dirs[k + 1];
      if (!inBounds(nx, ny)) continue;
      const ni = idx(nx, ny);
      if (!burnable[ni] || blocked[ni] || seen[ni]) continue;
      seen[ni] = 1;
      queue[tail++] = ni;
    }
  }
  return seen;
}

const emptyBlocked = new Uint8Array(size);
const threatened = bfsFromSources(emptyBlocked);

const townValueAt = new Float64Array(size);
let totalThreatenedTownValue = 0;
for (const town of parsed.towns) {
  if (!inBounds(town.x, town.y)) continue;
  const t = idx(town.x, town.y);
  if (!burnable[t]) continue;
  townValueAt[t] += town.value;
  if (threatened[t]) totalThreatenedTownValue += town.value;
}

if (cutBudget <= 0 || totalThreatenedTownValue <= 0) {
  process.stdout.write(JSON.stringify({ firebreaks: [] }));
  process.exit(0);
}

const dist = new Int32Array(size);
dist.fill(-1);
{
  const queue = new Int32Array(size);
  let head = 0;
  let tail = 0;
  for (const [x, y] of parsed.ignitions) {
    if (!inBounds(x, y)) continue;
    const start = idx(x, y);
    if (!burnable[start] || dist[start] !== -1) continue;
    dist[start] = 0;
    queue[tail++] = start;
  }
  while (head < tail) {
    const cur = queue[head++];
    const base = dist[cur] + 1;
    const cx = cur % width;
    const cy = (cur / width) | 0;
    for (let k = 0; k < 4; k++) {
      const nx = cx + dirs[k];
      const ny = cy + dirs[k + 1];
      if (!inBounds(nx, ny)) continue;
      const ni = idx(nx, ny);
      if (!burnable[ni] || dist[ni] !== -1) continue;
      dist[ni] = base;
      queue[tail++] = ni;
    }
  }
}

interface Candidate {
  cell: number;
  score: number;
}

function evaluateCandidate(cell: number, blocked: Uint8Array): number {
  if (!threatened[cell] || !burnable[cell] || blocked[cell]) return -1;

  const seen = new Uint8Array(size);
  const queue = new Int32Array(size);
  let head = 0;
  let tail = 0;
  seen[cell] = 1;
  queue[tail++] = cell;
  let enclosedTownValue = 0;
  let touchesBoundary = false;
  let touchesIgnitionSide = false;
  let area = 0;

  while (head < tail) {
    const cur = queue[head++];
    area++;
    enclosedTownValue += townValueAt[cur];
    const cx = cur % width;
    const cy = (cur / width) | 0;
    if (cx === 0 || cy === 0 || cx === width - 1 || cy === height - 1) {
      touchesBoundary = true;
    }
    for (let k = 0; k < 4; k++) {
      const nx = cx + dirs[k];
      const ny = cy + dirs[k + 1];
      if (!inBounds(nx, ny)) continue;
      const ni = idx(nx, ny);
      if (!burnable[ni] || blocked[ni]) continue;
      if (ni === cell) continue;
      if (threatened[ni] && dist[ni] >= 0 && dist[ni] < dist[cell]) {
        touchesIgnitionSide = true;
        continue;
      }
      if (seen[ni]) continue;
      if (dist[ni] >= 0 && dist[ni] < dist[cell]) continue;
      seen[ni] = 1;
      queue[tail++] = ni;
    }
  }

  if (!touchesIgnitionSide) return -1;
  let score = enclosedTownValue;
  if (touchesBoundary) score *= 0.35;
  if (area > cutBudget * 8) score *= 0.7;
  return score;
}

const candidates: Candidate[] = [];
for (let i = 0; i < size; i++) {
  if (!threatened[i] || !burnable[i]) continue;
  if (townValueAt[i] > 0) continue;
  let frontier = false;
  const cx = i % width;
  const cy = (i / width) | 0;
  for (let k = 0; k < 4; k++) {
    const nx = cx + dirs[k];
    const ny = cy + dirs[k + 1];
    if (!inBounds(nx, ny)) continue;
    const ni = idx(nx, ny);
    if (!burnable[ni]) continue;
    if (dist[ni] > dist[i]) {
      frontier = true;
      break;
    }
  }
  if (frontier) candidates.push({ cell: i, score: 0 });
}

const blocked = new Uint8Array(size);
const firebreaks: Point[] = [];
let budgetLeft = cutBudget;

const initialSample = Math.min(candidates.length, Math.max(256, cutBudget * 64));
for (let i = 0; i < initialSample; i++) {
  const c = candidates[i];
  c.score = evaluateCandidate(c.cell, blocked);
}
candidates.sort((a, b) => b.score - a.score || dist[b.cell] - dist[a.cell]);

while (budgetLeft > 0) {
  const reevaluateCount = Math.min(candidates.length, Math.max(32, cutBudget * 8));
  for (let i = 0; i < reevaluateCount; i++) {
    candidates[i].score = evaluateCandidate(candidates[i].cell, blocked);
  }
  candidates.sort((a, b) => b.score - a.score || dist[b.cell] - dist[a.cell]);

  const best = candidates[0];
  if (!best || best.score <= 0) break;

  blocked[best.cell] = 1;
  firebreaks.push(xy(best.cell));
  budgetLeft--;

  candidates.shift();
}

const reachableAfterCuts = bfsFromSources(blocked);
const filtered: Point[] = [];
for (const p of firebreaks) {
  const cell = idx(p[0], p[1]);
  let adjacentToReachable = false;
  for (let k = 0; k < 4; k++) {
    const nx = p[0] + dirs[k];
    const ny = p[1] + dirs[k + 1];
    if (!inBounds(nx, ny)) continue;
    const ni = idx(nx, ny);
    if (reachableAfterCuts[ni]) {
      adjacentToReachable = true;
      break;
    }
  }
  if (adjacentToReachable) filtered.push(p);
}

process.stdout.write(JSON.stringify({ firebreaks: filtered }));
