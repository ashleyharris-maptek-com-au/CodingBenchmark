declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type Robot = { id: number; x: number; y: number };
type Job = { id: number; x: number; y: number; priority: number };

const grid: string[] = Array.isArray(data.grid) ? data.grid : [];
const robots: Robot[] = Array.isArray(data.robots) ? data.robots : [];
const jobs: Job[] = Array.isArray(data.jobs) ? data.jobs : [];

const height = grid.length;
const width = height > 0 ? grid[0].length : 0;
const cellCount = width * height;

function cellIndex(x: number, y: number): number {
  return y * width + x;
}

const open = new Uint8Array(cellCount);
for (let y = 0; y < height; y++) {
  const row = grid[y];
  for (let x = 0; x < width; x++) {
    if (row.charCodeAt(x) === 46) {
      open[cellIndex(x, y)] = 1;
    }
  }
}

const neighbors: number[][] = new Array(cellCount);
for (let y = 0; y < height; y++) {
  for (let x = 0; x < width; x++) {
    const idx = cellIndex(x, y);
    if (open[idx] === 0) {
      neighbors[idx] = [];
      continue;
    }
    const list: number[] = [];
    if (x > 0) {
      const n = idx - 1;
      if (open[n]) list.push(n);
    }
    if (x + 1 < width) {
      const n = idx + 1;
      if (open[n]) list.push(n);
    }
    if (y > 0) {
      const n = idx - width;
      if (open[n]) list.push(n);
    }
    if (y + 1 < height) {
      const n = idx + width;
      if (open[n]) list.push(n);
    }
    neighbors[idx] = list;
  }
}

let maxRobotId = -1;
for (const robot of robots) {
  if (robot.id > maxRobotId) maxRobotId = robot.id;
}

const assignments: number[][] = Array.from({ length: maxRobotId + 1 }, () => []);

if (jobs.length === 0 || robots.length === 0 || width === 0 || height === 0) {
  process.stdout.write(JSON.stringify({ assignments }));
} else {
  const robotPositions = new Int32Array(robots.length);
  for (let i = 0; i < robots.length; i++) {
    robotPositions[i] = cellIndex(robots[i].x, robots[i].y);
  }

  const jobsSorted = jobs.slice().sort((a, b) => {
    if (b.priority !== a.priority) return b.priority - a.priority;
    if (a.id !== b.id) return a.id - b.id;
    if (a.y !== b.y) return a.y - b.y;
    return a.x - b.x;
  });

  const dist = new Int32Array(cellCount);
  const queue = new Int32Array(cellCount);

  function bfsToTargets(start: number, targets: Set<number>): Map<number, number> {
    dist.fill(-1);
    let head = 0;
    let tail = 0;
    queue[tail++] = start;
    dist[start] = 0;
    const found = new Map<number, number>();
    while (head < tail && found.size < targets.size) {
      const cur = queue[head++];
      const curDist = dist[cur];
      if (targets.has(cur)) {
        found.set(cur, curDist);
      }
      const nextDist = curDist + 1;
      const list = neighbors[cur];
      for (let i = 0; i < list.length; i++) {
        const nxt = list[i];
        if (dist[nxt] === -1) {
          dist[nxt] = nextDist;
          queue[tail++] = nxt;
        }
      }
    }
    return found;
  }

  const distinctJobCells = new Set<number>();
  for (const job of jobsSorted) {
    distinctJobCells.add(cellIndex(job.x, job.y));
  }

  const initialDistances = new Map<number, Map<number, number>>();
  for (let i = 0; i < robots.length; i++) {
    initialDistances.set(i, bfsToTargets(robotPositions[i], distinctJobCells));
  }

  const unreachablePenalty = cellCount + width + height + jobs.length + 1;

  for (const job of jobsSorted) {
    const jobCell = cellIndex(job.x, job.y);
    let bestRobot = 0;
    let bestScore = Number.POSITIVE_INFINITY;
    let bestDistance = Number.POSITIVE_INFINITY;

    for (let i = 0; i < robots.length; i++) {
      const baseDistances = initialDistances.get(i)!;
      let distance: number;
      if (robotPositions[i] === jobCell) {
        distance = 0;
      } else {
        const known = baseDistances.get(jobCell);
        if (known !== undefined) {
          distance = known;
        } else {
          distance = unreachablePenalty;
        }
      }
      const score = distance * 1_000_000 + assignments[robots[i].id].length;
      if (
        score < bestScore ||
        (score === bestScore &&
          (distance < bestDistance ||
            (distance === bestDistance && robots[i].id < robots[bestRobot].id)))
      ) {
        bestScore = score;
        bestDistance = distance;
        bestRobot = i;
      }
    }

    assignments[robots[bestRobot].id].push(job.id);
  }

  process.stdout.write(JSON.stringify({ assignments }));
}
