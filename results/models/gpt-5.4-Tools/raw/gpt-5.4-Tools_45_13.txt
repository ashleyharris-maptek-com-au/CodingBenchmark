OpCapability Shader
%1 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %worldPos %outColor
OpExecutionMode %main OriginUpperLeft
OpDecorate %worldPos Location 0
OpDecorate %outColor Location 0
%void = OpTypeVoid
%3 = OpTypeFunction %void
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%_ptr_Input_v3float = OpTypePointer Input %v3float
%worldPos = OpVariable %_ptr_Input_v3float Input
%_ptr_Output_v4float = OpTypePointer Output %v4float
%outColor = OpVariable %_ptr_Output_v4float Output
%float_0 = OpConstant %float 0
%float_0_5 = OpConstant %float 0.5
%float_1 = OpConstant %float 1
%float_0_1 = OpConstant %float 0.100000001
%float_0_4 = OpConstant %float 0.400000006
%bottom = OpConstantComposite %v3float %float_0_1 %float_0_4 %float_0_1
%top = OpConstantComposite %v3float %float_1 %float_1 %float_1
%main = OpFunction %void None %3
%5 = OpLabel
%wp = OpLoad %v3float %worldPos
%y = OpCompositeExtract %float %wp 1
%mul = OpFMul %float %y %float_0_5
%add = OpFAdd %float %mul %float_0_5
%t = OpExtInst %float %1 FClamp %add %float_0 %float_1
%one_minus_t = OpFSub %float %float_1 %t
%bot_scaled = OpVectorTimesScalar %v3float %bottom %one_minus_t
%top_scaled = OpVectorTimesScalar %v3float %top %t
%rgb = OpFAdd %v3float %bot_scaled %top_scaled
%color = OpCompositeConstruct %v4float %rgb %float_1
OpStore %outColor %color
OpReturn
OpFunctionEnd
