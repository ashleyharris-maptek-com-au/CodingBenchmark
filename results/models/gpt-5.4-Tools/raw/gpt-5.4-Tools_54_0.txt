from typing import Dict, List, Tuple


_memory = {
    "time": 0.0,
    "phase": "intercept",
    "go_around": False,
    "flare": False,
    "rollout": False,
    "trim": 0.0,
    "last_loc": 0.0,
    "last_gs": 0.0,
    "loc_i": 0.0,
    "gs_i": 0.0,
    "spd_i": 0.0,
    "hdg_i": 0.0,
    "ga_timer": 0.0,
}


def _clamp(x: float, lo: float, hi: float) -> float:
    return lo if x < lo else hi if x > hi else x


def _median(values: List[float]) -> float:
    if not values:
        return 0.0
    values = sorted(values)
    n = len(values)
    mid = n // 2
    if n % 2:
        return values[mid]
    return 0.5 * (values[mid - 1] + values[mid])


def _vote_triplet(state: Dict, base: str, suffix: str = "", valid_suffix: str = "_valid") -> Tuple[float, int]:
    vals = []
    for i in (1, 2, 3):
        valid_key = f"{base}_{i}{valid_suffix}"
        value_key = f"{base}_{i}{suffix}"
        if state.get(valid_key, True):
            value = state.get(value_key)
            if isinstance(value, (int, float)):
                vals.append(float(value))
    return _median(vals), len(vals)


def _fused_ils(state: Dict, base: str) -> Tuple[float, int]:
    vals = []
    for i in (1, 2, 3):
        if state.get(f"{base}_{i}_valid", False):
            v = state.get(f"{base}_{i}_dots")
            if isinstance(v, (int, float)):
                vals.append(float(v))
    if not vals:
        return 0.0, 0
    if len(vals) == 3:
        a, b, c = vals
        spreads = [
            (abs(a - b), 0.5 * (a + b)),
            (abs(a - c), 0.5 * (a + c)),
            (abs(b - c), 0.5 * (b + c)),
        ]
        best = min(spreads, key=lambda x: x[0])
        if best[0] < 1.2:
            return best[1], 2
    return _median(vals), len(vals)


def _pick_altitude_agl(state: Dict) -> float:
    ra = state.get("radio_alt_ft")
    if isinstance(ra, (int, float)) and ra > 0:
        return float(ra)
    baro, n = _vote_triplet(state, "baro_alt", "_ft")
    if n:
        target = state.get("target_alt_ft", 0.0)
        if isinstance(target, (int, float)):
            return max(0.0, float(baro) - max(0.0, float(target) - 50.0))
        return max(0.0, float(baro))
    gps, n = _vote_triplet(state, "gps", "_alt_ft")
    return max(0.0, gps) if n else 0.0


def _pick_speed(state: Dict) -> float:
    ias, n = _vote_triplet(state, "ias", "_kt")
    if n:
        return ias
    gs = state.get("groundspeed_kt")
    if isinstance(gs, (int, float)) and gs > 0:
        return float(gs)
    irs, n = _vote_triplet(state, "irs", "_gs_kt")
    if n:
        return irs
    gps, n = _vote_triplet(state, "gps", "_gs_kt")
    return gps if n else float(state.get("target_vref_kt", 140.0))


def _pick_vs(state: Dict) -> float:
    vals = []
    for i in (1, 2, 3):
        if state.get(f"irs_{i}_valid", False):
            v = state.get(f"irs_{i}_vs_fpm")
            if isinstance(v, (int, float)):
                vals.append(float(v))
    return _median(vals) if vals else float(state.get("target_vs_fpm", -700.0))


def _pick_pitch(state: Dict) -> float:
    vals = []
    for i in (1, 2, 3):
        if state.get(f"irs_{i}_valid", False):
            v = state.get(f"irs_{i}_pitch_deg")
            if isinstance(v, (int, float)):
                vals.append(float(v))
    return _median(vals) if vals else 0.0


def _pick_roll(state: Dict) -> float:
    vals = []
    for i in (1, 2, 3):
        if state.get(f"irs_{i}_valid", False):
            v = state.get(f"irs_{i}_roll_deg")
            if isinstance(v, (int, float)):
                vals.append(float(v))
    return _median(vals) if vals else 0.0


def _pick_heading(state: Dict) -> float:
    vals = []
    for i in (1, 2, 3):
        if state.get(f"irs_{i}_valid", False):
            v = state.get(f"irs_{i}_heading_deg")
            if isinstance(v, (int, float)):
                vals.append(float(v))
    return _median(vals) if vals else float(state.get("target_hdg_deg", 0.0))


def _pick_track(state: Dict) -> float:
    vals = []
    for i in (1, 2, 3):
        if state.get(f"irs_{i}_valid", False):
            v = state.get(f"irs_{i}_track_deg")
            if isinstance(v, (int, float)):
                vals.append(float(v))
    if vals:
        return _median(vals)
    for i in (1, 2, 3):
        if state.get(f"gps_{i}_valid", False):
            v = state.get(f"gps_{i}_track_deg")
            if isinstance(v, (int, float)):
                vals.append(float(v))
    return _median(vals) if vals else _pick_heading(state)


def _ang_err(target: float, actual: float) -> float:
    return ((target - actual + 180.0) % 360.0) - 180.0


def _engines_running(state: Dict) -> Tuple[bool, bool]:
    left = bool(state.get("engine_1_running", True)) and not bool(state.get("engine_1_fire", False))
    right = bool(state.get("engine_2_running", True)) and not bool(state.get("engine_2_fire", False))
    return left, right


def autopilot_step(state: dict, dt: float) -> dict:
    dt = _clamp(float(dt) if isinstance(dt, (int, float)) else 0.1, 0.01, 0.2)
    _memory["time"] += dt

    loc, loc_count = _fused_ils(state, "localizer")
    gs, gs_count = _fused_ils(state, "glideslope")
    alt = _pick_altitude_agl(state)
    speed = _pick_speed(state)
    vs = _pick_vs(state)
    pitch = _pick_pitch(state)
    roll = _pick_roll(state)
    heading = _pick_heading(state)
    track = _pick_track(state)
    target_hdg = float(state.get("target_hdg_deg", heading))
    vref = float(state.get("target_vref_kt", state.get("target_ias_kt", 137.0)))
    dme_vals = [state.get("dme_1_nm"), state.get("dme_2_nm")]
    dme = _median([float(v) for v in dme_vals if isinstance(v, (int, float)) and v > 0]) if any(
        isinstance(v, (int, float)) and v > 0 for v in dme_vals
    ) else 99.0
    left_ok, right_ok = _engines_running(state)
    engine_count = int(left_ok) + int(right_ok)
    on_ground = bool(state.get("on_ground", False) or state.get("weight_on_wheels", False))
    below_dh = bool(state.get("below_decision_height", alt < 200.0))

    if on_ground:
        _memory["phase"] = "rollout"
    elif _memory["go_around"]:
        _memory["phase"] = "go_around"
    elif alt < 50.0:
        _memory["phase"] = "flare"
    elif alt < 1000.0:
        _memory["phase"] = "approach"
    else:
        _memory["phase"] = "intercept"

    unstable = False
    if not on_ground and not _memory["go_around"]:
        if alt < 1000.0:
            unstable = (
                abs(loc) > 2.5
                or (gs_count > 0 and abs(gs) > 2.0)
                or speed > vref + 25.0
                or speed < max(95.0, vref - 20.0)
                or abs(roll) > 25.0
                or vs < -1600.0
            )
        if below_dh and (loc_count == 0 or abs(loc) > 1.0 or (gs_count > 0 and abs(gs) > 1.0)):
            unstable = True
        if alt < 120.0 and vs < -1100.0:
            unstable = True
        if alt < 80.0 and speed > vref + 20.0:
            unstable = True
        if alt > 80.0 and engine_count == 0 and dme < 1.5:
            unstable = False

    if unstable:
        _memory["go_around"] = True
        _memory["ga_timer"] = 0.0
    if _memory["go_around"]:
        _memory["ga_timer"] += dt
        if alt > 1400.0 and vs > 200.0:
            _memory["go_around"] = False

    _memory["loc_i"] = _clamp(_memory["loc_i"] + loc * dt, -2.0, 2.0)
    _memory["gs_i"] = _clamp(_memory["gs_i"] + gs * dt, -2.0, 2.0)
    _memory["spd_i"] = _clamp(_memory["spd_i"] + (vref - speed) * dt * 0.03, -1.0, 1.0)
    hdg_err = _ang_err(target_hdg, heading)
    trk_err = _ang_err(target_hdg, track)
    _memory["hdg_i"] = _clamp(_memory["hdg_i"] + hdg_err * dt * 0.03, -1.0, 1.0)

    target_speed = vref + 8.0
    if alt > 1500.0:
        target_speed = max(target_speed + 15.0, float(state.get("target_ias_kt", target_speed)))
    elif alt > 500.0:
        target_speed = vref + 5.0
    elif alt > 50.0:
        target_speed = vref + 2.0
    else:
        target_speed = max(vref - 5.0, 105.0)

    flaps = 15.0 if alt > 2000.0 else 20.0 if alt > 1000.0 else 30.0 if alt > 300.0 else 35.0
    if engine_count <= 1:
        flaps = min(flaps, 30.0)
        target_speed += 5.0
    if speed > vref + 35.0 and alt > 500.0:
        flaps = min(flaps, 20.0)

    gear = 1.0 if alt < 2500.0 or dme < 10.0 else 0.0
    speedbrake = 0.0
    spoiler = 0.0
    autobrake = 0.0
    reverse_1 = 0.0
    reverse_2 = 0.0

    if _memory["go_around"]:
        pitch_target = 10.0 if engine_count > 0 else 7.0
        elevator = _clamp(0.08 * (pitch_target - pitch) - 0.0005 * vs, -1.0, 1.0)
        loc_term = _clamp(loc * 0.18 + _memory["loc_i"] * 0.06 + trk_err * 0.02, -1.0, 1.0)
        aileron = _clamp(loc_term - 0.05 * roll, -1.0, 1.0)
        rudder = _clamp(0.5 * aileron + 0.03 * trk_err, -1.0, 1.0)
        throttle_base = 1.0 if engine_count > 0 else 0.0
        throttle_1 = throttle_base if left_ok else 0.0
        throttle_2 = throttle_base if right_ok else 0.0
        _memory["trim"] = _clamp(_memory["trim"] + 0.08 * elevator * dt, -1.0, 1.0)
        return {
            "elevator": elevator,
            "aileron": aileron,
            "rudder": rudder,
            "throttle_1": throttle_1,
            "throttle_2": throttle_2,
            "elevator_trim": _memory["trim"],
            "flaps": 20.0,
            "gear": 1.0,
            "speedbrake": 0.0,
            "spoiler": 0.0,
            "autobrake": 0.0,
            "reverse_1": 0.0,
            "reverse_2": 0.0,
        }

    if on_ground:
        groundspeed = float(state.get("groundspeed_kt", speed))
        spoiler = 1.0
        speedbrake = 1.0
        autobrake = 1.0 if groundspeed > 80.0 else 0.6 if groundspeed > 40.0 else 0.25
        reverse_1 = 0.8 if left_ok and groundspeed > 60.0 else 0.0
        reverse_2 = 0.8 if right_ok and groundspeed > 60.0 else 0.0
        throttle_1 = 0.0
        throttle_2 = 0.0
        aileron = _clamp(-0.04 * roll, -0.3, 0.3)
        rudder = _clamp(0.03 * _ang_err(target_hdg, heading), -0.6, 0.6)
        elevator = 0.15
        return {
            "elevator": elevator,
            "aileron": aileron,
            "rudder": rudder,
            "throttle_1": throttle_1,
            "throttle_2": throttle_2,
            "elevator_trim": _memory["trim"],
            "flaps": 35.0,
            "gear": 1.0,
            "speedbrake": speedbrake,
            "spoiler": spoiler,
            "autobrake": autobrake,
            "reverse_1": reverse_1,
            "reverse_2": reverse_2,
        }

    pitch_target = 2.5
    if gs_count > 0:
        pitch_target += -1.7 * gs
    else:
        desired_vs = -max(350.0, min(900.0, speed * 5.0))
        pitch_target += 0.003 * (desired_vs - vs)
    if alt < 100.0:
        pitch_target = 1.0
    if alt < 50.0:
        pitch_target = 3.0 + 0.02 * max(0.0, 50.0 - alt)
    if alt < 20.0:
        pitch_target = 5.0
    if engine_count == 0:
        pitch_target -= 1.0
    elevator = _clamp(0.09 * (pitch_target - pitch) - 0.0007 * vs, -1.0, 1.0)

    localizer_gain = 0.20 if alt > 200.0 else 0.14
    bank_cmd = localizer_gain * loc + 0.015 * trk_err + 0.006 * hdg_err + 0.04 * _memory["loc_i"]
    if alt < 100.0:
        bank_cmd = _clamp(bank_cmd, -5.0, 5.0)
    else:
        bank_cmd = _clamp(bank_cmd, -12.0, 12.0)
    aileron = _clamp(0.09 * (bank_cmd - roll), -1.0, 1.0)

    crosswind = _ang_err(track, heading)
    rudder = _clamp(0.45 * aileron + 0.015 * crosswind, -1.0, 1.0)
    if alt < 80.0:
        rudder = _clamp(rudder + 0.02 * loc, -1.0, 1.0)

    speed_err = target_speed - speed
    throttle_bias = 0.52 if engine_count == 2 else 0.62 if engine_count == 1 else 0.0
    gs_comp = 0.025 * max(0.0, -gs) if gs_count > 0 else 0.0
    sink_comp = 0.00018 * max(0.0, -(vs + 650.0))
    throttle_cmd = _clamp(throttle_bias + 0.018 * speed_err + gs_comp + sink_comp + _memory["spd_i"] * 0.08, 0.0, 1.0)
    if alt < 35.0:
        throttle_cmd = 0.05 if engine_count > 0 else 0.0
    elif alt < 15.0:
        throttle_cmd = 0.0

    throttle_1 = throttle_cmd if left_ok else 0.0
    throttle_2 = throttle_cmd if right_ok else 0.0
    if left_ok and not right_ok:
        rudder = _clamp(rudder + 0.25, -1.0, 1.0)
        aileron = _clamp(aileron + 0.06, -1.0, 1.0)
        throttle_1 = min(1.0, throttle_cmd + 0.05)
    elif right_ok and not left_ok:
        rudder = _clamp(rudder - 0.25, -1.0, 1.0)
        aileron = _clamp(aileron - 0.06, -1.0, 1.0)
        throttle_2 = min(1.0, throttle_cmd + 0.05)

    if alt < 50.0:
        speedbrake = 0.0

    _memory["trim"] = _clamp(_memory["trim"] + 0.10 * elevator * dt, -1.0, 1.0)

    return {
        "elevator": elevator,
        "aileron": aileron,
        "rudder": rudder,
        "throttle_1": throttle_1,
        "throttle_2": throttle_2,
        "elevator_trim": _memory["trim"],
        "flaps": flaps,
        "gear": gear,
        "speedbrake": speedbrake,
        "spoiler": spoiler,
        "autobrake": autobrake,
        "reverse_1": reverse_1,
        "reverse_2": reverse_2,
    }
