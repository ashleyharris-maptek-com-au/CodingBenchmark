#version 450

layout(location = 2) in vec2 uv;
layout(location = 0) out vec4 outColor;

void main() {
    vec2 c = vec2(uv.x * 3.0 - 2.0, uv.y * 2.0 - 1.0);
    vec2 z = vec2(0.0);
    int iterations = 0;

    for (int i = 0; i < 20; ++i) {
        float x = z.x;
        float y = z.y;
        float x2 = x * x;
        float y2 = y * y;
        if (x2 + y2 > 4.0) {
            break;
        }
        z = vec2(x2 - y2 + c.x, 2.0 * x * y + c.y);
        iterations++;
    }

    float t = float(iterations) / 20.0;
    outColor = vec4(t, t * 0.5, 1.0 - t, 1.0);
}
