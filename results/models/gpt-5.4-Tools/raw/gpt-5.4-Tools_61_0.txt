type Point = {
  x: number;
  y: number;
};

type Package = Point & {
  id: string | number;
  index: number;
  angle: number;
  radius: number;
};

type InputData = {
  depot: [number, number];
  truckCount: number;
  packages: Array<{ id: string | number; x: number; y: number }>;
};

declare const require: any;
declare const process: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input) as InputData;

function dist(a: Point, b: Point): number {
  const dx = a.x - b.x;
  const dy = a.y - b.y;
  return Math.hypot(dx, dy);
}

function routeDistance(route: Package[], depot: Point): number {
  if (route.length === 0) return 0;
  let total = dist(depot, route[0]);
  for (let i = 1; i < route.length; i++) {
    total += dist(route[i - 1], route[i]);
  }
  total += dist(route[route.length - 1], depot);
  return total;
}

function insertionCost(route: Package[], pkg: Package, pos: number, depot: Point): number {
  const prev = pos === 0 ? depot : route[pos - 1];
  const next = pos === route.length ? depot : route[pos];
  return dist(prev, pkg) + dist(pkg, next) - dist(prev, next);
}

function bestInsertion(route: Package[], pkg: Package, depot: Point): { cost: number; pos: number } {
  let bestCost = Number.POSITIVE_INFINITY;
  let bestPos = 0;
  for (let pos = 0; pos <= route.length; pos++) {
    const cost = insertionCost(route, pkg, pos, depot);
    if (cost < bestCost - 1e-12 || (Math.abs(cost - bestCost) <= 1e-12 && pos < bestPos)) {
      bestCost = cost;
      bestPos = pos;
    }
  }
  return { cost: bestCost, pos: bestPos };
}

function removeCost(route: Package[], idx: number, depot: Point): number {
  const prev = idx === 0 ? depot : route[idx - 1];
  const curr = route[idx];
  const next = idx === route.length - 1 ? depot : route[idx + 1];
  return dist(prev, curr) + dist(curr, next) - dist(prev, next);
}

function twoOpt(route: Package[], depot: Point): void {
  const n = route.length;
  if (n < 4) return;
  let improved = true;
  let passes = 0;
  while (improved && passes < 6) {
    improved = false;
    passes++;
    for (let i = 0; i < n - 2; i++) {
      const a = i === 0 ? depot : route[i - 1];
      const b = route[i];
      for (let k = i + 1; k < n - 1; k++) {
        const c = route[k];
        const d = route[k + 1];
        const delta = dist(a, c) + dist(b, d) - dist(a, b) - dist(c, d);
        if (delta < -1e-12) {
          let left = i;
          let right = k;
          while (left < right) {
            const tmp = route[left];
            route[left] = route[right];
            route[right] = tmp;
            left++;
            right--;
          }
          improved = true;
        }
      }
      const c = route[n - 1];
      const d = depot;
      const deltaEnd = dist(a, c) + dist(b, d) - dist(a, b) - dist(c, d);
      if (deltaEnd < -1e-12) {
        let left = i;
        let right = n - 1;
        while (left < right) {
          const tmp = route[left];
          route[left] = route[right];
          route[right] = tmp;
          left++;
          right--;
        }
        improved = true;
      }
    }
  }
}

function relocate(routes: Package[][], depot: Point, truckLimit: number): void {
  if (routes.length <= 1) return;
  let passes = 0;
  let improved = true;
  while (improved && passes < 8) {
    improved = false;
    passes++;
    for (let from = 0; from < routes.length; from++) {
      const source = routes[from];
      if (source.length === 0) continue;
      for (let idx = 0; idx < source.length; idx++) {
        const pkg = source[idx];
        const removalGain = removeCost(source, idx, depot);
        let bestRoute = -1;
        let bestPos = 0;
        let bestDelta = 0;
        for (let to = 0; to < routes.length; to++) {
          if (to === from) continue;
          const target = routes[to];
          const ins = bestInsertion(target, pkg, depot);
          const delta = ins.cost - removalGain;
          if (
            delta < bestDelta - 1e-12 ||
            (Math.abs(delta - bestDelta) <= 1e-12 &&
              (bestRoute === -1 || target.length < routes[bestRoute].length))
          ) {
            bestDelta = delta;
            bestRoute = to;
            bestPos = ins.pos;
          }
        }
        if (bestRoute !== -1) {
          source.splice(idx, 1);
          routes[bestRoute].splice(bestPos, 0, pkg);
          if (source.length === 0 && routes.length > 1 && routes.length > truckLimit) {
            routes.splice(from, 1);
          }
          improved = true;
          twoOpt(routes[bestRoute === from ? from : bestRoute > from ? bestRoute - 1 : bestRoute], depot);
          if (from < routes.length) twoOpt(routes[from], depot);
          break;
        }
      }
      if (improved) break;
    }
  }
}

function buildRoutes(packages: Package[], depot: Point, truckCount: number): Package[][] {
  if (packages.length === 0) return [];
  const routeCount = Math.max(1, Math.min(truckCount, packages.length));
  const ordered = packages.slice().sort((a, b) => {
    if (a.angle !== b.angle) return a.angle - b.angle;
    if (a.radius !== b.radius) return a.radius - b.radius;
    return a.index - b.index;
  });

  const routes: Package[][] = Array.from({ length: routeCount }, () => []);
  for (let i = 0; i < ordered.length; i++) {
    const routeIndex = Math.floor((i * routeCount) / ordered.length);
    routes[routeIndex].push(ordered[i]);
  }

  for (const route of routes) {
    route.sort((a, b) => a.radius - b.radius || a.angle - b.angle || a.index - b.index);
    twoOpt(route, depot);
  }

  const leftovers: Package[] = [];
  for (let i = routes.length - 1; i >= 0; i--) {
    if (routes[i].length === 0) {
      routes.splice(i, 1);
    }
  }

  if (leftovers.length > 0) {
    for (const pkg of leftovers) {
      let bestRoute = 0;
      let bestPos = 0;
      let bestCost = Number.POSITIVE_INFINITY;
      for (let i = 0; i < routes.length; i++) {
        const ins = bestInsertion(routes[i], pkg, depot);
        if (ins.cost < bestCost - 1e-12) {
          bestCost = ins.cost;
          bestRoute = i;
          bestPos = ins.pos;
        }
      }
      routes[bestRoute].splice(bestPos, 0, pkg);
    }
  }

  relocate(routes, depot, routeCount);

  for (const route of routes) {
    twoOpt(route, depot);
  }

  routes.sort((a, b) => {
    const aFirst = a.length === 0 ? Number.POSITIVE_INFINITY : a[0].angle;
    const bFirst = b.length === 0 ? Number.POSITIVE_INFINITY : b[0].angle;
    if (aFirst !== bFirst) return aFirst - bFirst;
    return routeDistance(a, depot) - routeDistance(b, depot);
  });

  return routes;
}

const depot: Point = { x: data.depot[0], y: data.depot[1] };
const packages: Package[] = data.packages.map((pkg, index) => {
  const dx = pkg.x - depot.x;
  const dy = pkg.y - depot.y;
  return {
    id: pkg.id,
    x: pkg.x,
    y: pkg.y,
    index,
    angle: Math.atan2(dy, dx),
    radius: Math.hypot(dx, dy),
  };
});

const routes = buildRoutes(packages, depot, Math.max(0, Math.trunc(data.truckCount))).map((route) =>
  route.map((pkg) => pkg.id),
);

process.stdout.write(JSON.stringify({ routes }));
