const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const grid = Array.isArray(data.probability) ? data.probability : [];
const drones = Array.isArray(data.drones) ? data.drones : [];
const maxStepsPerDrone = Math.max(0, data.maxStepsPerDrone | 0);

const height = grid.length;
const width = height > 0 && Array.isArray(grid[0]) ? grid[0].length : 0;
const cellCount = width * height;

function indexOf(x, y) {
  return y * width + x;
}

function coordsOf(idx) {
  return [idx % width, Math.floor(idx / width)];
}

function inBounds(x, y) {
  return x >= 0 && y >= 0 && x < width && y < height;
}

const values = new Float64Array(cellCount);
const validCell = new Uint8Array(cellCount);

for (let y = 0; y < height; y++) {
  const row = Array.isArray(grid[y]) ? grid[y] : [];
  for (let x = 0; x < width; x++) {
    const idx = indexOf(x, y);
    const v = Number(row[x]);
    if (!Number.isFinite(v) || v < 0) {
      continue;
    }
    validCell[idx] = 1;
    values[idx] = v > 0 ? v : 0;
  }
}

const neighbors = Array.from({ length: cellCount }, () => []);
for (let y = 0; y < height; y++) {
  for (let x = 0; x < width; x++) {
    const idx = indexOf(x, y);
    if (!validCell[idx]) continue;
    neighbors[idx].push(idx);
    if (x > 0) {
      const n = idx - 1;
      if (validCell[n]) neighbors[idx].push(n);
    }
    if (x + 1 < width) {
      const n = idx + 1;
      if (validCell[n]) neighbors[idx].push(n);
    }
    if (y > 0) {
      const n = idx - width;
      if (validCell[n]) neighbors[idx].push(n);
    }
    if (y + 1 < height) {
      const n = idx + width;
      if (validCell[n]) neighbors[idx].push(n);
    }
  }
}

function sanitizeStart(drone) {
  let x = drone && Number.isFinite(drone.x) ? drone.x | 0 : 0;
  let y = drone && Number.isFinite(drone.y) ? drone.y | 0 : 0;
  if (!inBounds(x, y) || !validCell[indexOf(x, y)]) {
    for (let idx = 0; idx < cellCount; idx++) {
      if (validCell[idx]) return idx;
    }
    return -1;
  }
  return indexOf(x, y);
}

const covered = new Uint8Array(cellCount);
const bestVisitStep = new Int32Array(cellCount);
bestVisitStep.fill(1 << 30);

function coverCell(idx, step) {
  if (idx < 0 || idx >= cellCount) return;
  covered[idx] = 1;
  if (step < bestVisitStep[idx]) bestVisitStep[idx] = step;
}

const queue = new Int32Array(cellCount);
const seenStamp = new Int32Array(cellCount);
const distArr = new Int32Array(cellCount);
const parentArr = new Int32Array(cellCount);
let bfsStamp = 1;

function reconstructPath(targetIdx) {
  const path = [];
  let cur = targetIdx;
  while (cur !== -1) {
    path.push(cur);
    cur = parentArr[cur];
  }
  path.reverse();
  return path;
}

function bfsBestTarget(startIdx, maxDist, scoringFn) {
  bfsStamp++;
  if (bfsStamp === 0x7fffffff) {
    seenStamp.fill(0);
    bfsStamp = 1;
  }
  let head = 0;
  let tail = 0;
  queue[tail++] = startIdx;
  seenStamp[startIdx] = bfsStamp;
  distArr[startIdx] = 0;
  parentArr[startIdx] = -1;

  let bestIdx = startIdx;
  let bestScore = scoringFn(startIdx, 0);

  while (head < tail) {
    const cur = queue[head++];
    const dist = distArr[cur];
    const score = scoringFn(cur, dist);
    if (score > bestScore) {
      bestScore = score;
      bestIdx = cur;
    }
    if (dist >= maxDist) continue;
    const adj = neighbors[cur];
    for (let i = 0; i < adj.length; i++) {
      const nxt = adj[i];
      if (seenStamp[nxt] === bfsStamp) continue;
      seenStamp[nxt] = bfsStamp;
      distArr[nxt] = dist + 1;
      parentArr[nxt] = cur;
      queue[tail++] = nxt;
    }
  }

  return { target: bestIdx, score: bestScore, path: reconstructPath(bestIdx) };
}

function scoreCell(idx, arrivalStep, horizonFactor) {
  const base = covered[idx] ? 0 : values[idx];
  if (base <= 0) return -arrivalStep * 0.0001;

  let local = base;
  const adj = neighbors[idx];
  for (let i = 0; i < adj.length; i++) {
    const n = adj[i];
    if (n === idx || covered[n]) continue;
    local += values[n] * horizonFactor;
  }

  const revisitPenalty = bestVisitStep[idx] < (1 << 29) ? 0.08 / (1 + bestVisitStep[idx]) : 0;
  return local / (arrivalStep + 1.25) - revisitPenalty;
}

function greedyExpandFrom(startIdx, stepsRemaining, stepOffset) {
  const path = [];
  let current = startIdx;
  for (let s = 0; s < stepsRemaining; s++) {
    const adj = neighbors[current];
    let best = current;
    let bestScore = -Infinity;
    for (let i = 0; i < adj.length; i++) {
      const nxt = adj[i];
      const arrival = stepOffset + s + 1;
      const score = scoreCell(nxt, arrival, 0.18) + (nxt === current ? -0.02 : 0);
      if (score > bestScore) {
        bestScore = score;
        best = nxt;
      }
    }
    current = best;
    path.push(current);
    coverCell(current, stepOffset + s + 1);
  }
  return path;
}

function planDrone(startIdx) {
  if (startIdx < 0) return Array.from({ length: maxStepsPerDrone + 1 }, () => [0, 0]);
  const path = [coordsOf(startIdx)];
  coverCell(startIdx, 0);
  let current = startIdx;
  let usedSteps = 0;

  const searchRadius = Math.max(6, Math.min(24, Math.floor(Math.sqrt(cellCount)) + 4));

  while (usedSteps < maxStepsPerDrone) {
    const remaining = maxStepsPerDrone - usedSteps;
    const targetInfo = bfsBestTarget(
      current,
      Math.min(searchRadius, remaining),
      (idx, dist) => scoreCell(idx, usedSteps + dist, 0.22)
    );

    if (targetInfo.path.length <= 1) {
      const tail = greedyExpandFrom(current, remaining, usedSteps);
      for (let i = 0; i < tail.length; i++) path.push(coordsOf(tail[i]));
      break;
    }

    const pathToTarget = targetInfo.path;
    let advanced = false;
    for (let i = 1; i < pathToTarget.length && usedSteps < maxStepsPerDrone; i++) {
      current = pathToTarget[i];
      usedSteps++;
      coverCell(current, usedSteps);
      path.push(coordsOf(current));
      advanced = true;
    }

    if (!advanced) {
      const tail = greedyExpandFrom(current, remaining, usedSteps);
      for (let i = 0; i < tail.length; i++) path.push(coordsOf(tail[i]));
      break;
    }
  }

  while (path.length < maxStepsPerDrone + 1) {
    path.push(path[path.length - 1]);
  }

  return path;
}

const droneOrder = drones.map((drone, idx) => {
  const startIdx = sanitizeStart(drone);
  const startValue = startIdx >= 0 ? values[startIdx] : 0;
  return { idx, startIdx, startValue };
});

droneOrder.sort((a, b) => b.startValue - a.startValue);
const resultPaths = new Array(drones.length);

for (let i = 0; i < droneOrder.length; i++) {
  const item = droneOrder[i];
  resultPaths[item.idx] = planDrone(item.startIdx);
}

process.stdout.write(JSON.stringify({ paths: resultPaths }));
