declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type Cell = [number, number];

interface DroneInput {
  id: string | number;
  x: number;
  y: number;
}

interface InputData {
  probability: number[][];
  drones: DroneInput[];
  maxStepsPerDrone: number;
}

interface DroneState {
  index: number;
  startX: number;
  startY: number;
  x: number;
  y: number;
  stepsUsed: number;
  path: Cell[];
  component: number;
}

interface Candidate {
  idx: number;
  score: number;
  dist: number;
}

const parsed = data as InputData;
const grid = parsed.probability;
const drones = Array.isArray(parsed.drones) ? parsed.drones : [];
const maxSteps = Math.max(0, parsed.maxStepsPerDrone | 0);

const height = grid.length;
const width = height > 0 ? grid[0].length : 0;
const cellCount = width * height;

function inBounds(x: number, y: number): boolean {
  return x >= 0 && y >= 0 && x < width && y < height;
}

function idxOf(x: number, y: number): number {
  return y * width + x;
}

function xOf(idx: number): number {
  return idx % width;
}

function yOf(idx: number): number {
  return Math.floor(idx / width);
}

const dirs: Cell[] = [
  [0, -1],
  [1, 0],
  [0, 1],
  [-1, 0],
];

if (height === 0 || width === 0 || drones.length === 0) {
  const emptyPaths = drones.map(() => [] as Cell[]);
  process.stdout.write(JSON.stringify({ paths: emptyPaths }));
  process.exit(0);
}

const traversable = new Uint8Array(cellCount);
const values = new Float64Array(cellCount);
for (let y = 0; y < height; y++) {
  const row = grid[y];
  for (let x = 0; x < width; x++) {
    const v = row[x];
    const idx = idxOf(x, y);
    if (v !== -1) {
      traversable[idx] = 1;
      values[idx] = typeof v === 'number' && v > 0 ? v : 0;
    }
  }
}

const component = new Int32Array(cellCount);
component.fill(-1);
const componentCells: number[][] = [];
const componentValueSums: number[] = [];
const componentTopCells: number[][] = [];
const queue = new Int32Array(cellCount);

function buildComponents(): void {
  let compId = 0;
  for (let start = 0; start < cellCount; start++) {
    if (!traversable[start] || component[start] !== -1) continue;
    let qh = 0;
    let qt = 0;
    queue[qt++] = start;
    component[start] = compId;
    const cells: number[] = [];
    let valueSum = 0;
    while (qh < qt) {
      const cur = queue[qh++];
      cells.push(cur);
      valueSum += values[cur];
      const cx = xOf(cur);
      const cy = yOf(cur);
      for (let d = 0; d < 4; d++) {
        const nx = cx + dirs[d][0];
        const ny = cy + dirs[d][1];
        if (!inBounds(nx, ny)) continue;
        const ni = idxOf(nx, ny);
        if (!traversable[ni] || component[ni] !== -1) continue;
        component[ni] = compId;
        queue[qt++] = ni;
      }
    }
    cells.sort((a, b) => values[b] - values[a]);
    componentCells[compId] = cells;
    componentValueSums[compId] = valueSum;
    componentTopCells[compId] = cells.slice(0, Math.min(cells.length, 96));
    compId++;
  }
}

buildComponents();

const covered = new Uint8Array(cellCount);
const distances = new Int32Array(cellCount);
const parents = new Int32Array(cellCount);
const visitMark = new Int32Array(cellCount);
let bfsToken = 1;

function runBfs(startIdx: number): void {
  bfsToken++;
  if (bfsToken === 2147483647) {
    visitMark.fill(0);
    bfsToken = 1;
  }
  let qh = 0;
  let qt = 0;
  queue[qt++] = startIdx;
  visitMark[startIdx] = bfsToken;
  distances[startIdx] = 0;
  parents[startIdx] = -1;
  while (qh < qt) {
    const cur = queue[qh++];
    const cx = xOf(cur);
    const cy = yOf(cur);
    const baseDist = distances[cur] + 1;
    for (let d = 0; d < 4; d++) {
      const nx = cx + dirs[d][0];
      const ny = cy + dirs[d][1];
      if (!inBounds(nx, ny)) continue;
      const ni = idxOf(nx, ny);
      if (!traversable[ni] || visitMark[ni] === bfsToken) continue;
      visitMark[ni] = bfsToken;
      distances[ni] = baseDist;
      parents[ni] = cur;
      queue[qt++] = ni;
    }
  }
}

function reconstructPath(targetIdx: number): number[] {
  const path: number[] = [];
  let cur = targetIdx;
  while (cur !== -1) {
    path.push(cur);
    cur = parents[cur];
  }
  path.reverse();
  return path;
}

const droneStates: DroneState[] = drones.map((drone, index) => {
  const sx = Math.max(0, Math.min(width - 1, drone.x | 0));
  const sy = Math.max(0, Math.min(height - 1, drone.y | 0));
  const startIdx = idxOf(sx, sy);
  const validStart = traversable[startIdx] ? startIdx : -1;
  const comp = validStart >= 0 ? component[validStart] : -1;
  if (validStart >= 0) {
    covered[validStart] = 1;
  }
  return {
    index,
    startX: sx,
    startY: sy,
    x: sx,
    y: sy,
    stepsUsed: 0,
    path: [[sx, sy]],
    component: comp,
  };
});

function appendPath(drone: DroneState, route: number[]): void {
  for (let i = 1; i < route.length; i++) {
    const idx = route[i];
    drone.x = xOf(idx);
    drone.y = yOf(idx);
    drone.path.push([drone.x, drone.y]);
    drone.stepsUsed++;
    covered[idx] = 1;
    if (drone.stepsUsed >= maxSteps) break;
  }
}

function chooseCandidate(drone: DroneState): Candidate | null {
  if (drone.component < 0) return null;
  const remaining = maxSteps - drone.stepsUsed;
  if (remaining <= 0) return null;

  const top = componentTopCells[drone.component];
  const startIdx = idxOf(drone.x, drone.y);
  runBfs(startIdx);

  let best: Candidate | null = null;
  const sampleLimit = Math.min(top.length, remaining > 24 ? top.length : 48);
  for (let i = 0; i < sampleLimit; i++) {
    const idx = top[i];
    if (covered[idx]) continue;
    if (visitMark[idx] !== bfsToken) continue;
    const dist = distances[idx];
    if (dist <= 0 || dist > remaining) continue;
    const value = values[idx];
    let neighborGain = 0;
    const cx = xOf(idx);
    const cy = yOf(idx);
    for (let d = 0; d < 4; d++) {
      const nx = cx + dirs[d][0];
      const ny = cy + dirs[d][1];
      if (!inBounds(nx, ny)) continue;
      const ni = idxOf(nx, ny);
      if (traversable[ni] && !covered[ni]) {
        neighborGain += values[ni] * 0.2;
      }
    }
    const score = (value + neighborGain) / dist;
    if (best === null || score > best.score || (score === best.score && dist < best.dist)) {
      best = { idx, score, dist };
    }
  }

  if (best !== null) return best;

  let fallback: Candidate | null = null;
  const cells = componentCells[drone.component];
  const scanLimit = Math.min(cells.length, 512);
  for (let i = 0; i < scanLimit; i++) {
    const idx = cells[i];
    if (visitMark[idx] !== bfsToken) continue;
    const dist = distances[idx];
    if (dist <= 0 || dist > remaining) continue;
    const score = (covered[idx] ? 0.01 : values[idx] + 0.01) / dist;
    if (fallback === null || score > fallback.score || (score === fallback.score && dist < fallback.dist)) {
      fallback = { idx, score, dist };
    }
  }
  return fallback;
}

function localStep(drone: DroneState): boolean {
  if (drone.stepsUsed >= maxSteps) return false;
  const cx = drone.x;
  const cy = drone.y;
  let bestIdx = idxOf(cx, cy);
  let bestScore = covered[bestIdx] ? -0.05 : values[bestIdx] + 0.05;
  for (let d = 0; d < 4; d++) {
    const nx = cx + dirs[d][0];
    const ny = cy + dirs[d][1];
    if (!inBounds(nx, ny)) continue;
    const ni = idxOf(nx, ny);
    if (!traversable[ni]) continue;
    let score = covered[ni] ? 0 : values[ni] + 0.1;
    for (let k = 0; k < 4; k++) {
      const tx = nx + dirs[k][0];
      const ty = ny + dirs[k][1];
      if (!inBounds(tx, ty)) continue;
      const ti = idxOf(tx, ty);
      if (traversable[ti] && !covered[ti]) {
        score += values[ti] * 0.15;
      }
    }
    if (score > bestScore) {
      bestScore = score;
      bestIdx = ni;
    }
  }
  drone.x = xOf(bestIdx);
  drone.y = yOf(bestIdx);
  drone.path.push([drone.x, drone.y]);
  drone.stepsUsed++;
  covered[bestIdx] = 1;
  return true;
}

const activeByComponent = new Map<number, DroneState[]>();
for (const drone of droneStates) {
  if (drone.component < 0) continue;
  const list = activeByComponent.get(drone.component);
  if (list) list.push(drone);
  else activeByComponent.set(drone.component, [drone]);
}

for (const [, group] of activeByComponent) {
  group.sort((a, b) => a.index - b.index);
  let progress = true;
  while (progress) {
    progress = false;
    for (const drone of group) {
      if (drone.stepsUsed >= maxSteps) continue;
      const candidate = chooseCandidate(drone);
      if (candidate !== null) {
        const route = reconstructPath(candidate.idx);
        appendPath(drone, route);
        progress = true;
      } else if (localStep(drone)) {
        progress = true;
      }
    }
  }
}

for (const drone of droneStates) {
  while (drone.stepsUsed < maxSteps) {
    if (!localStep(drone)) break;
  }
}

const paths: Cell[][] = droneStates.map((drone) => drone.path);
process.stdout.write(JSON.stringify({ paths }));
