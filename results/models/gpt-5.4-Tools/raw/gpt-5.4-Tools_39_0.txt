use std::io::{self, Read};

#[derive(Clone)]
struct Solver {
    n: usize,
    half: usize,
    adj_mask: Vec<u64>,
    adj_list: Vec<Vec<usize>>,
    order: Vec<usize>,
    side: Vec<i8>, // -1 unassigned, 0 in A, 1 in B
    a_count: usize,
    b_count: usize,
    assigned_mask: u64,
    cut_so_far: i32,
    best_cut: i32,
    best_mask_a: u64,
}

impl Solver {
    fn new(n: usize, edges: &[(usize, usize)]) -> Self {
        let mut adj_mask = vec![0u64; n];
        let mut adj_list = vec![Vec::new(); n];
        for &(u, v) in edges {
            if u == v {
                continue;
            }
            let bu = 1u64 << u;
            let bv = 1u64 << v;
            if adj_mask[u] & bv == 0 {
                adj_mask[u] |= bv;
                adj_mask[v] |= bu;
                adj_list[u].push(v);
                adj_list[v].push(u);
            }
        }

        let mut order: Vec<usize> = (0..n).collect();
        order.sort_by_key(|&v| std::cmp::Reverse(adj_list[v].len()));

        Self {
            n,
            half: n / 2,
            adj_mask,
            adj_list,
            order,
            side: vec![-1; n],
            a_count: 0,
            b_count: 0,
            assigned_mask: 0,
            cut_so_far: 0,
            best_cut: i32::MAX,
            best_mask_a: 0,
        }
    }

    fn popcount(mask: u64) -> usize {
        mask.count_ones() as usize
    }

    fn assign(&mut self, v: usize, s: i8) -> i32 {
        let opposite_mask = if s == 0 {
            self.mask_of_side(1)
        } else {
            self.mask_of_side(0)
        };
        let added_cut = Self::popcount(self.adj_mask[v] & opposite_mask) as i32;
        self.side[v] = s;
        self.assigned_mask |= 1u64 << v;
        if s == 0 {
            self.a_count += 1;
        } else {
            self.b_count += 1;
        }
        self.cut_so_far += added_cut;
        added_cut
    }

    fn unassign(&mut self, v: usize, s: i8, added_cut: i32) {
        self.cut_so_far -= added_cut;
        if s == 0 {
            self.a_count -= 1;
        } else {
            self.b_count -= 1;
        }
        self.assigned_mask &= !(1u64 << v);
        self.side[v] = -1;
    }

    fn mask_of_side(&self, side_value: i8) -> u64 {
        let mut mask = 0u64;
        for v in 0..self.n {
            if self.side[v] == side_value {
                mask |= 1u64 << v;
            }
        }
        mask
    }

    fn lower_bound(&self) -> i32 {
        let rem_a = self.half - self.a_count;
        let mask_a = self.mask_of_side(0);
        let mask_b = self.mask_of_side(1);

        let mut forced = 0i32;
        let mut deltas = Vec::with_capacity(self.n);

        for v in 0..self.n {
            if self.side[v] != -1 {
                continue;
            }
            let to_a = Self::popcount(self.adj_mask[v] & mask_b) as i32;
            let to_b = Self::popcount(self.adj_mask[v] & mask_a) as i32;
            forced += to_a.min(to_b);
            deltas.push(to_a - to_b);
        }

        deltas.sort_unstable();
        let choose_a = rem_a;
        let mut balance_adjust = 0i32;
        for &delta in deltas.iter().take(choose_a) {
            balance_adjust += delta;
        }

        self.cut_so_far + forced + balance_adjust
    }

    fn current_a_mask(&self) -> u64 {
        self.mask_of_side(0)
    }

    fn choose_vertex(&self) -> Option<usize> {
        let mask_a = self.mask_of_side(0);
        let mask_b = self.mask_of_side(1);
        let mut best_v = None;
        let mut best_score = -1isize;

        for &v in &self.order {
            if self.side[v] != -1 {
                continue;
            }
            let score = Self::popcount(self.adj_mask[v] & (mask_a | mask_b)) as isize * 2
                + self.adj_list[v].len() as isize;
            if score > best_score {
                best_score = score;
                best_v = Some(v);
            }
        }
        best_v
    }

    fn dfs(&mut self) {
        if self.a_count > self.half || self.b_count > self.half {
            return;
        }
        if self.cut_so_far >= self.best_cut {
            return;
        }
        let lb = self.lower_bound();
        if lb >= self.best_cut {
            return;
        }

        if self.a_count == self.half && self.b_count == self.half {
            self.best_cut = self.cut_so_far;
            self.best_mask_a = self.current_a_mask();
            return;
        }

        let v = match self.choose_vertex() {
            Some(v) => v,
            None => return,
        };

        let mask_a = self.mask_of_side(0);
        let mask_b = self.mask_of_side(1);
        let cost_a = Self::popcount(self.adj_mask[v] & mask_b) as i32;
        let cost_b = Self::popcount(self.adj_mask[v] & mask_a) as i32;

        let mut branches = [(0i8, cost_a), (1i8, cost_b)];
        if cost_b < cost_a {
            branches.swap(0, 1);
        }

        for &(s, _) in &branches {
            if s == 0 && self.a_count == self.half {
                continue;
            }
            if s == 1 && self.b_count == self.half {
                continue;
            }
            let added = self.assign(v, s);
            self.dfs();
            self.unassign(v, s, added);
        }
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();
    let m: usize = it.next().unwrap().parse().unwrap();

    assert!(n % 2 == 0, "n must be even");
    assert!(n <= 64, "this exact solver supports n <= 64");

    let mut edges = Vec::with_capacity(m);
    for _ in 0..m {
        let u: usize = it.next().unwrap().parse().unwrap();
        let v: usize = it.next().unwrap().parse().unwrap();
        edges.push((u, v));
    }

    let mut solver = Solver::new(n, &edges);

    if n > 0 {
        solver.assign(0, 0);
    }
    solver.dfs();

    let mut part_a = Vec::new();
    for v in 0..n {
        if (solver.best_mask_a >> v) & 1 == 1 {
            part_a.push(v);
        }
    }
    part_a.sort_unstable();

    println!("{}", solver.best_cut);
    for (i, v) in part_a.iter().enumerate() {
        if i > 0 {
            print!(" ");
        }
        print!("{}", v);
    }
    println!();
}
