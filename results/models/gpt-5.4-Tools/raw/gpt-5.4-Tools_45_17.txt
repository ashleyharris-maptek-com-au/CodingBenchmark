#version 450

layout(location = 0) in vec3 worldPos;
layout(location = 1) in vec3 normal;
layout(location = 2) in vec2 uv;
layout(location = 3) in vec3 tangent;
layout(location = 4) in vec3 color;

layout(set = 0, binding = 0) uniform UBO {
    mat4 model;
    mat4 view;
    mat4 projection;
    vec4 lightPos;
    vec4 cameraPos;
    vec4 params;
} ubo;

layout(location = 0) out vec4 outColor;

void main() {
    vec2 p = uv * vec2(8.0, 16.0);
    float row = floor(p.y);
    float rowParity = mod(row, 2.0);
    p.x += rowParity * 0.5;
    vec2 cell = fract(p);

    float inMortarX = step(cell.x, 0.05) + step(0.95, cell.x);
    float inMortarY = step(cell.y, 0.1) + step(0.9, cell.y);
    float mortarMask = clamp(inMortarX + inMortarY, 0.0, 1.0);

    vec3 brickColor = vec3(0.7, 0.2, 0.1);
    vec3 mortarColor = vec3(0.7, 0.7, 0.7);
    vec3 baseColor = mix(brickColor, mortarColor, mortarMask);

    vec3 N = normalize(normal);
    vec3 L = normalize(ubo.lightPos.xyz - worldPos);
    float lambert = max(dot(N, L), 0.0);
    vec3 lit = baseColor * lambert;

    outColor = vec4(lit, 1.0);
}
