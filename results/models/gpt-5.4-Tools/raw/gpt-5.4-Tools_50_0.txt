#version 450

layout(local_size_x = 256) in;

// Each boid stored as two vec4s: [pos.xyz, 0], [vel.xyz, 0]
// Boid i is at index i*2 and i*2+1
layout(set = 0, binding = 0) readonly buffer InputBoids {
    vec4 inData[];
};

layout(set = 0, binding = 1) buffer OutputBoids {
    vec4 outData[];
};

layout(set = 0, binding = 2) uniform Params {
    uvec4 counts;   // counts.x = N
    vec4 radii;     // x=dt, y=sepRadius, z=alignRadius, w=cohRadius
    vec4 weights;   // x=sepWeight, y=alignWeight, z=cohWeight, w=maxSpeed
    vec4 bounds;    // x=boundSize, y=boundForce
};

void main() {
    uint i = gl_GlobalInvocationID.x;
    uint boidCount = counts.x;

    if (i >= boidCount) {
        return;
    }

    uint baseIndex = i * 2u;
    vec3 pos = inData[baseIndex].xyz;
    vec3 vel = inData[baseIndex + 1u].xyz;

    float dt = radii.x;
    float sepRadius = radii.y;
    float alignRadius = radii.z;
    float cohRadius = radii.w;

    float sepWeight = weights.x;
    float alignWeight = weights.y;
    float cohWeight = weights.z;
    float maxSpeed = weights.w;

    float boundSize = bounds.x;
    float boundForce = bounds.y;

    float sepRadiusSq = sepRadius * sepRadius;
    float alignRadiusSq = alignRadius * alignRadius;
    float cohRadiusSq = cohRadius * cohRadius;

    vec3 sepAccum = vec3(0.0);
    vec3 alignAccum = vec3(0.0);
    vec3 cohAccum = vec3(0.0);

    uint sepCount = 0u;
    uint alignCount = 0u;
    uint cohCount = 0u;

    for (uint j = 0u; j < boidCount; ++j) {
        if (j == i) {
            continue;
        }

        uint otherIndex = j * 2u;
        vec3 otherPos = inData[otherIndex].xyz;
        vec3 delta = pos - otherPos;
        float distSq = dot(delta, delta);

        if (distSq <= 0.0) {
            continue;
        }

        if (distSq < sepRadiusSq) {
            float dist = sqrt(distSq);
            sepAccum += delta / dist;
            sepCount++;
        }

        if (distSq < alignRadiusSq) {
            vec3 otherVel = inData[otherIndex + 1u].xyz;
            alignAccum += otherVel;
            alignCount++;
        }

        if (distSq < cohRadiusSq) {
            cohAccum += otherPos;
            cohCount++;
        }
    }

    vec3 accel = vec3(0.0);

    if (sepCount > 0u) {
        vec3 separation = sepAccum / float(sepCount);
        accel += separation * sepWeight;
    }

    if (alignCount > 0u) {
        vec3 avgVel = alignAccum / float(alignCount);
        vec3 alignment = avgVel - vel;
        accel += alignment * alignWeight;
    }

    if (cohCount > 0u) {
        vec3 center = cohAccum / float(cohCount);
        vec3 cohesion = center - pos;
        accel += cohesion * cohWeight;
    }

    float boundThreshold = boundSize * 0.4;
    float boundScale = boundSize * 0.5;
    vec3 boundaryAccel = vec3(0.0);

    if (pos.x > boundThreshold) {
        boundaryAccel.x -= boundForce * ((pos.x - boundThreshold) / boundScale);
    } else if (pos.x < -boundThreshold) {
        boundaryAccel.x += boundForce * ((-boundThreshold - pos.x) / boundScale);
    }

    if (pos.y > boundThreshold) {
        boundaryAccel.y -= boundForce * ((pos.y - boundThreshold) / boundScale);
    } else if (pos.y < -boundThreshold) {
        boundaryAccel.y += boundForce * ((-boundThreshold - pos.y) / boundScale);
    }

    if (pos.z > boundThreshold) {
        boundaryAccel.z -= boundForce * ((pos.z - boundThreshold) / boundScale);
    } else if (pos.z < -boundThreshold) {
        boundaryAccel.z += boundForce * ((-boundThreshold - pos.z) / boundScale);
    }

    accel += boundaryAccel;

    vel += accel * dt;

    float speed = length(vel);
    if (speed > maxSpeed && speed > 0.0) {
        vel = vel / speed * maxSpeed;
    }

    pos += vel * dt;

    outData[baseIndex] = vec4(pos, 0.0);
    outData[baseIndex + 1u] = vec4(vel, 0.0);
}
