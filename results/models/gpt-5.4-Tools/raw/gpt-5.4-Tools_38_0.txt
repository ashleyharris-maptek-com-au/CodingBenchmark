import sys


def parse_input() -> tuple[int, list[set[int]]] | None:
    lines = sys.stdin.read().splitlines()
    if not lines:
        return None

    header = lines[0].split()
    if len(header) != 2:
        return None

    try:
        universe_size = int(header[0])
        set_count = int(header[1])
    except ValueError:
        return None

    if len(lines) < 1 + set_count:
        return None

    sets: list[set[int]] = []
    for i in range(set_count):
        raw = lines[1 + i].split()
        current: set[int] = set()
        for token in raw:
            try:
                value = int(token)
            except ValueError:
                return None
            if value < 0 or value >= universe_size:
                return None
            current.add(value)
        sets.append(current)

    return universe_size, sets


def solve_exact_cover(universe_size: int, sets: list[set[int]]) -> list[int] | None:
    element_to_sets: list[list[int]] = [[] for _ in range(universe_size)]
    for idx, current in enumerate(sets):
        for element in current:
            element_to_sets[element].append(idx)

    if universe_size == 0:
        return []

    for candidates in element_to_sets:
        if not candidates:
            return None

    chosen: list[int] = []
    covered = [False] * universe_size
    used = [False] * len(sets)

    def search(covered_count: int) -> bool:
        if covered_count == universe_size:
            return True

        target = -1
        target_candidates: list[int] | None = None
        best_count = None

        for element in range(universe_size):
            if covered[element]:
                continue

            candidates: list[int] = []
            for set_idx in element_to_sets[element]:
                if used[set_idx]:
                    continue
                current = sets[set_idx]
                valid = True
                for member in current:
                    if covered[member]:
                        valid = False
                        break
                if valid:
                    candidates.append(set_idx)

            if not candidates:
                return False

            count = len(candidates)
            if best_count is None or count < best_count:
                best_count = count
                target = element
                target_candidates = candidates
                if count == 1:
                    break

        assert target != -1 and target_candidates is not None

        # Deterministic ordering: smaller sets first, then lower index.
        target_candidates.sort(key=lambda idx: (len(sets[idx]), idx))

        for set_idx in target_candidates:
            used[set_idx] = True
            chosen.append(set_idx)

            newly_covered: list[int] = []
            for element in sets[set_idx]:
                covered[element] = True
                newly_covered.append(element)

            if search(covered_count + len(newly_covered)):
                return True

            for element in newly_covered:
                covered[element] = False
            chosen.pop()
            used[set_idx] = False

        return False

    if not search(0):
        return None

    chosen.sort()
    return chosen


def main() -> None:
    parsed = parse_input()
    if parsed is None:
        print("NO SOLUTION")
        return

    universe_size, sets = parsed
    solution = solve_exact_cover(universe_size, sets)

    if solution is None:
        print("NO SOLUTION")
        return

    print("SOLUTION")
    print(" ".join(map(str, solution)))


if __name__ == "__main__":
    main()
