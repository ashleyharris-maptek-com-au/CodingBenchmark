@group(0) @binding(0) var<storage, read> baseData : array<u32, 16>;
@group(0) @binding(1) var<storage, read_write> result : array<u32, 4>;
@group(0) @binding(2) var<uniform> params : vec4<u32>;

fn rotl32(x : u32, shift : u32) -> u32 {
    return (x << shift) | (x >> (32u - shift));
}

fn ctz32(x : u32) -> u32 {
    if (x == 0u) {
        return 32u;
    }

    var n : u32 = 0u;
    var v : u32 = x;
    loop {
        if ((v & 1u) != 0u) {
            break;
        }
        n = n + 1u;
        v = v >> 1u;
    }
    return n;
}

fn custom_hash(nonce : u32) -> vec2<u32> {
    var state : array<u32, 16>;

    state[0] = baseData[0] ^ (nonce + 0u);
    state[1] = baseData[1] ^ (nonce + 1u);
    state[2] = baseData[2] ^ (nonce + 2u);
    state[3] = baseData[3] ^ (nonce + 3u);
    state[4] = baseData[4] ^ (nonce + 4u);
    state[5] = baseData[5] ^ (nonce + 5u);
    state[6] = baseData[6] ^ (nonce + 6u);
    state[7] = baseData[7] ^ (nonce + 7u);
    state[8] = baseData[8] ^ (nonce + 8u);
    state[9] = baseData[9] ^ (nonce + 9u);
    state[10] = baseData[10] ^ (nonce + 10u);
    state[11] = baseData[11] ^ (nonce + 11u);
    state[12] = baseData[12] ^ (nonce + 12u);
    state[13] = baseData[13] ^ (nonce + 13u);
    state[14] = baseData[14] ^ (nonce + 14u);
    state[15] = baseData[15] ^ (nonce + 15u);

    for (var round : u32 = 0u; round < 8u; round = round + 1u) {
        var a : u32;
        var b : u32;
        var c : u32;
        var d : u32;

        a = state[0];
        b = state[1];
        c = state[2];
        d = state[3];
        a = a + b; d = d ^ a; d = rotl32(d, 16u);
        c = c + d; b = b ^ c; b = rotl32(b, 12u);
        a = a + b; d = d ^ a; d = rotl32(d, 8u);
        c = c + d; b = b ^ c; b = rotl32(b, 7u);
        state[0] = a;
        state[1] = b;
        state[2] = c;
        state[3] = d;

        a = state[4];
        b = state[5];
        c = state[6];
        d = state[7];
        a = a + b; d = d ^ a; d = rotl32(d, 16u);
        c = c + d; b = b ^ c; b = rotl32(b, 12u);
        a = a + b; d = d ^ a; d = rotl32(d, 8u);
        c = c + d; b = b ^ c; b = rotl32(b, 7u);
        state[4] = a;
        state[5] = b;
        state[6] = c;
        state[7] = d;

        a = state[8];
        b = state[9];
        c = state[10];
        d = state[11];
        a = a + b; d = d ^ a; d = rotl32(d, 16u);
        c = c + d; b = b ^ c; b = rotl32(b, 12u);
        a = a + b; d = d ^ a; d = rotl32(d, 8u);
        c = c + d; b = b ^ c; b = rotl32(b, 7u);
        state[8] = a;
        state[9] = b;
        state[10] = c;
        state[11] = d;

        a = state[12];
        b = state[13];
        c = state[14];
        d = state[15];
        a = a + b; d = d ^ a; d = rotl32(d, 16u);
        c = c + d; b = b ^ c; b = rotl32(b, 12u);
        a = a + b; d = d ^ a; d = rotl32(d, 8u);
        c = c + d; b = b ^ c; b = rotl32(b, 7u);
        state[12] = a;
        state[13] = b;
        state[14] = c;
        state[15] = d;
    }

    let h0 : u32 = state[0] ^ state[2] ^ state[4] ^ state[6] ^ state[8] ^ state[10] ^ state[12] ^ state[14];
    let h1 : u32 = state[1] ^ state[3] ^ state[5] ^ state[7] ^ state[9] ^ state[11] ^ state[13] ^ state[15];
    return vec2<u32>(h0, h1);
}

fn trailing_zeros_64(h0 : u32, h1 : u32) -> u32 {
    if (h0 != 0u) {
        return ctz32(h0);
    }
    return 32u + ctz32(h1);
}

@compute @workgroup_size(256)
fn main(@builtin(global_invocation_id) gid : vec3<u32>) {
    if (gid.x >= params.z) {
        return;
    }

    let nonce : u32 = params.y + gid.x;
    let hash : vec2<u32> = custom_hash(nonce);
    let h0 : u32 = hash.x;
    let h1 : u32 = hash.y;
    let zeros : u32 = trailing_zeros_64(h0, h1);

    if (zeros >= params.x) {
        result[0] = 1u;
        result[1] = nonce;
        result[2] = h0;
        result[3] = h1;
    }
}
