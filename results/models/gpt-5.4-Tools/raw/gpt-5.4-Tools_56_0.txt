const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

function solve(instance) {
  const depot = instance.depot;
  const packages = Array.isArray(instance.packages) ? instance.packages : [];
  const n = packages.length;
  if (n === 0) {
    return { routes: [] };
  }

  const truckCount = Math.max(1, Math.min(Number.isFinite(instance.truckCount) ? Math.floor(instance.truckCount) : 1, n));
  const depotX = +depot[0];
  const depotY = +depot[1];

  const pts = new Array(n);
  let sumX = 0;
  let sumY = 0;
  for (let i = 0; i < n; i++) {
    const p = packages[i];
    const x = +p.x;
    const y = +p.y;
    const dx = x - depotX;
    const dy = y - depotY;
    const radius = Math.hypot(dx, dy);
    const angle = Math.atan2(dy, dx);
    pts[i] = {
      idx: i,
      id: p.id,
      x,
      y,
      dx,
      dy,
      radius,
      angle
    };
    sumX += x;
    sumY += y;
  }

  const centroidX = sumX / n;
  const centroidY = sumY / n;
  const orient = Math.atan2(centroidY - depotY, centroidX - depotX);
  for (let i = 0; i < n; i++) {
    let rel = pts[i].angle - orient;
    while (rel <= -Math.PI) rel += Math.PI * 2;
    while (rel > Math.PI) rel -= Math.PI * 2;
    pts[i].relAngle = rel;
  }

  pts.sort((a, b) => {
    if (a.relAngle !== b.relAngle) return a.relAngle - b.relAngle;
    if (a.radius !== b.radius) return a.radius - b.radius;
    return a.idx - b.idx;
  });

  const targetSizes = balancedSizes(n, truckCount);
  let routes = buildInitialRoutes(pts, targetSizes);
  routes = optimizeRouteOrder(routes, depotX, depotY);
  routes = improveBoundaries(routes, depotX, depotY);
  routes = optimizeRouteOrder(routes, depotX, depotY);

  return {
    routes: routes.filter(route => route.length > 0).map(route => route.map(p => p.id))
  };
}

function balancedSizes(total, groups) {
  const base = Math.floor(total / groups);
  const extra = total % groups;
  const out = new Array(groups);
  for (let i = 0; i < groups; i++) {
    out[i] = base + (i < extra ? 1 : 0);
  }
  return out;
}

function buildInitialRoutes(sortedPts, targetSizes) {
  const routes = [];
  let cursor = 0;
  for (let i = 0; i < targetSizes.length; i++) {
    const size = targetSizes[i];
    const route = new Array(size);
    for (let j = 0; j < size; j++) {
      route[j] = sortedPts[cursor++];
    }
    routes.push(route);
  }
  return routes;
}

function optimizeRouteOrder(routes, depotX, depotY) {
  const out = new Array(routes.length);
  for (let i = 0; i < routes.length; i++) {
    out[i] = optimizeSingleRoute(routes[i], depotX, depotY);
  }
  return out;
}

function optimizeSingleRoute(route, depotX, depotY) {
  const m = route.length;
  if (m <= 2) return route.slice();

  let best = route.slice();
  let improved = true;
  while (improved) {
    improved = false;
    for (let i = 0; i < m - 1; i++) {
      const prev = i === 0 ? null : best[i - 1];
      const a = best[i];
      for (let k = i + 1; k < m; k++) {
        const b = best[k];
        const next = k === m - 1 ? null : best[k + 1];
        const before =
          distNode(prev, a, depotX, depotY) +
          distNode(b, next, depotX, depotY);
        const after =
          distNode(prev, b, depotX, depotY) +
          distNode(a, next, depotX, depotY);
        if (after + 1e-9 < before) {
          reverseSegment(best, i, k);
          improved = true;
        }
      }
    }
  }

  let bestCost = routeCost(best, depotX, depotY);
  for (let shift = 1; shift < m; shift++) {
    const rotated = rotateRoute(best, shift);
    const rotatedCost = routeCost(rotated, depotX, depotY);
    const reversed = rotated.slice().reverse();
    const reversedCost = routeCost(reversed, depotX, depotY);
    if (rotatedCost + 1e-9 < bestCost) {
      best = rotated;
      bestCost = rotatedCost;
    }
    if (reversedCost + 1e-9 < bestCost) {
      best = reversed;
      bestCost = reversedCost;
    }
  }
  return best;
}

function improveBoundaries(routes, depotX, depotY) {
  const result = routes.map(route => route.slice());
  let changed = true;
  while (changed) {
    changed = false;
    for (let i = 0; i + 1 < result.length; i++) {
      const left = result[i];
      const right = result[i + 1];
      const best = bestBoundaryAdjustment(left, right, depotX, depotY);
      if (best.changed) {
        result[i] = optimizeSingleRoute(best.left, depotX, depotY);
        result[i + 1] = optimizeSingleRoute(best.right, depotX, depotY);
        changed = true;
      }
    }
  }
  return result;
}

function bestBoundaryAdjustment(left, right, depotX, depotY) {
  const currentCost = routeCost(left, depotX, depotY) + routeCost(right, depotX, depotY);
  let bestCost = currentCost;
  let bestLeft = left;
  let bestRight = right;

  for (let takeLeft = 0; takeLeft <= Math.min(2, left.length); takeLeft++) {
    for (let takeRight = 0; takeRight <= Math.min(2, right.length); takeRight++) {
      if (takeLeft === 0 && takeRight === 0) continue;
      if (takeLeft === left.length && takeRight === right.length) continue;

      const leftKeep = left.slice(0, left.length - takeLeft);
      const leftMove = left.slice(left.length - takeLeft);
      const rightMove = right.slice(0, takeRight);
      const rightKeep = right.slice(takeRight);
      const newLeft = leftKeep.concat(rightMove);
      const newRight = leftMove.concat(rightKeep);

      if (newLeft.length === 0 || newRight.length === 0) continue;

      const optLeft = optimizeSingleRoute(newLeft, depotX, depotY);
      const optRight = optimizeSingleRoute(newRight, depotX, depotY);
      const cost = routeCost(optLeft, depotX, depotY) + routeCost(optRight, depotX, depotY);
      if (cost + 1e-9 < bestCost) {
        bestCost = cost;
        bestLeft = optLeft;
        bestRight = optRight;
      }
    }
  }

  return {
    changed: bestCost + 1e-9 < currentCost,
    left: bestLeft,
    right: bestRight
  };
}

function rotateRoute(route, shift) {
  return route.slice(shift).concat(route.slice(0, shift));
}

function reverseSegment(arr, start, end) {
  while (start < end) {
    const tmp = arr[start];
    arr[start] = arr[end];
    arr[end] = tmp;
    start++;
    end--;
  }
}

function routeCost(route, depotX, depotY) {
  if (route.length === 0) return 0;
  let total = Math.hypot(route[0].x - depotX, route[0].y - depotY);
  for (let i = 1; i < route.length; i++) {
    total += Math.hypot(route[i].x - route[i - 1].x, route[i].y - route[i - 1].y);
  }
  total += Math.hypot(route[route.length - 1].x - depotX, route[route.length - 1].y - depotY);
  return total;
}

function distNode(a, b, depotX, depotY) {
  if (a === null && b === null) return 0;
  if (a === null) return Math.hypot(b.x - depotX, b.y - depotY);
  if (b === null) return Math.hypot(a.x - depotX, a.y - depotY);
  return Math.hypot(a.x - b.x, a.y - b.y);
}

const result = solve(data);
process.stdout.write(JSON.stringify(result));
