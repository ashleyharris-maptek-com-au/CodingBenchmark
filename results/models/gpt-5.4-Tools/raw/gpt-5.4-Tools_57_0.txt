const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const width = data.width | 0;
const height = data.height | 0;
const grid = data.grid || [];
const ignitions = data.ignitions || [];
const towns = data.towns || [];
const cutBudget = Math.max(0, data.cutBudget | 0);

const n = width * height;
const dirs = [1, 0, -1, 0, 1];

function idx(x, y) {
  return y * width + x;
}

function inBounds(x, y) {
  return x >= 0 && x < width && y >= 0 && y < height;
}

const burnable = new Uint8Array(n);
for (let y = 0; y < height; y++) {
  const row = grid[y] || '';
  for (let x = 0; x < width; x++) {
    if (row.charCodeAt(x) === 46) burnable[idx(x, y)] = 1;
  }
}

const ignited = new Uint8Array(n);
const ignitionList = [];
for (const p of ignitions) {
  if (!Array.isArray(p) || p.length < 2) continue;
  const x = p[0] | 0;
  const y = p[1] | 0;
  if (!inBounds(x, y)) continue;
  const id = idx(x, y);
  if (!burnable[id] || ignited[id]) continue;
  ignited[id] = 1;
  ignitionList.push(id);
}

const townValueAt = new Float64Array(n);
for (const town of towns) {
  if (!town) continue;
  const x = town.x | 0;
  const y = town.y | 0;
  if (!inBounds(x, y)) continue;
  const id = idx(x, y);
  if (!burnable[id]) continue;
  townValueAt[id] += Number(town.value) || 0;
}

function bfsReachable(blocked) {
  const reached = new Uint8Array(n);
  const q = new Int32Array(n);
  let head = 0;
  let tail = 0;
  for (let i = 0; i < ignitionList.length; i++) {
    const s = ignitionList[i];
    if (blocked[s] || reached[s]) continue;
    reached[s] = 1;
    q[tail++] = s;
  }
  while (head < tail) {
    const v = q[head++];
    const x = v % width;
    const y = (v / width) | 0;
    for (let k = 0; k < 4; k++) {
      const nx = x + dirs[k];
      const ny = y + dirs[k + 1];
      if (!inBounds(nx, ny)) continue;
      const ni = idx(nx, ny);
      if (!burnable[ni] || blocked[ni] || reached[ni]) continue;
      reached[ni] = 1;
      q[tail++] = ni;
    }
  }
  return reached;
}

const emptyBlocked = new Uint8Array(n);
const baseFire = bfsReachable(emptyBlocked);

let baseBurnedTownValue = 0;
for (let i = 0; i < n; i++) {
  if (baseFire[i]) baseBurnedTownValue += townValueAt[i];
}

if (cutBudget === 0 || baseBurnedTownValue <= 0) {
  process.stdout.write(JSON.stringify({ firebreaks: [] }));
  process.exit(0);
}

const safeComponent = new Int32Array(n);
safeComponent.fill(-1);
const compTownValue = [];
const compCells = [];
const qComp = new Int32Array(n);
let compCount = 0;

for (let start = 0; start < n; start++) {
  if (!burnable[start] || baseFire[start] || safeComponent[start] !== -1) continue;
  let head = 0;
  let tail = 0;
  qComp[tail++] = start;
  safeComponent[start] = compCount;
  let townSum = 0;
  const cells = [];
  while (head < tail) {
    const v = qComp[head++];
    cells.push(v);
    townSum += townValueAt[v];
    const x = v % width;
    const y = (v / width) | 0;
    for (let k = 0; k < 4; k++) {
      const nx = x + dirs[k];
      const ny = y + dirs[k + 1];
      if (!inBounds(nx, ny)) continue;
      const ni = idx(nx, ny);
      if (!burnable[ni] || baseFire[ni] || safeComponent[ni] !== -1) continue;
      safeComponent[ni] = compCount;
      qComp[tail++] = ni;
    }
  }
  compTownValue.push(townSum);
  compCells.push(cells);
  compCount++;
}

if (compCount === 0) {
  process.stdout.write(JSON.stringify({ firebreaks: [] }));
  process.exit(0);
}

const frontierSeen = new Uint8Array(n);
const frontierCells = [];
for (let v = 0; v < n; v++) {
  if (!burnable[v] || !baseFire[v]) continue;
  const x = v % width;
  const y = (v / width) | 0;
  let touchesSafe = false;
  for (let k = 0; k < 4; k++) {
    const nx = x + dirs[k];
    const ny = y + dirs[k + 1];
    if (!inBounds(nx, ny)) continue;
    const ni = idx(nx, ny);
    if (burnable[ni] && !baseFire[ni]) {
      touchesSafe = true;
      break;
    }
  }
  if (touchesSafe) {
    frontierSeen[v] = 1;
    frontierCells.push(v);
  }
}

if (frontierCells.length === 0) {
  process.stdout.write(JSON.stringify({ firebreaks: [] }));
  process.exit(0);
}

const candidates = [];
const candidateCellSets = [];
const candidateCompSets = [];
const dedup = new Map();

function addCandidate(cellList, compList) {
  if (!cellList.length || cellList.length > cutBudget) return;
  const uniqCells = Array.from(new Set(cellList)).sort((a, b) => a - b);
  const key = uniqCells.join(',');
  if (!key) return;
  const uniqComps = Array.from(new Set(compList));
  let value = 0;
  for (let i = 0; i < uniqComps.length; i++) value += compTownValue[uniqComps[i]];
  if (value <= 0) return;
  const score = value / uniqCells.length;
  const prev = dedup.get(key);
  if (prev !== undefined) {
    if (candidates[prev].value >= value) return;
    candidates[prev] = { cells: uniqCells, comps: uniqComps, value, score };
    return;
  }
  dedup.set(key, candidates.length);
  candidates.push({ cells: uniqCells, comps: uniqComps, value, score });
}

for (let c = 0; c < compCount; c++) {
  const boundary = [];
  const cells = compCells[c];
  for (let i = 0; i < cells.length; i++) {
    const v = cells[i];
    const x = v % width;
    const y = (v / width) | 0;
    let exposed = false;
    for (let k = 0; k < 4; k++) {
      const nx = x + dirs[k];
      const ny = y + dirs[k + 1];
      if (!inBounds(nx, ny)) continue;
      const ni = idx(nx, ny);
      if (burnable[ni] && baseFire[ni]) {
        exposed = true;
        break;
      }
    }
    if (exposed) boundary.push(v);
  }
  addCandidate(boundary, [c]);
}

if (cutBudget > 0) {
  const townCells = [];
  for (let v = 0; v < n; v++) {
    if (townValueAt[v] > 0 && burnable[v]) townCells.push(v);
  }
  townCells.sort((a, b) => townValueAt[b] - townValueAt[a]);
  const limitTowns = Math.min(townCells.length, Math.max(8, cutBudget * 4));

  for (let ti = 0; ti < limitTowns; ti++) {
    const center = townCells[ti];
    if (!baseFire[center]) continue;
    const cx = center % width;
    const cy = (center / width) | 0;
    const maxRadius = Math.min(Math.max(2, cutBudget), 12);
    for (let r = 1; r <= maxRadius; r++) {
      const ring = [];
      for (let dx = -r; dx <= r; dx++) {
        const dy = r - Math.abs(dx);
        const x1 = cx + dx;
        const y1 = cy + dy;
        const x2 = cx + dx;
        const y2 = cy - dy;
        if (inBounds(x1, y1)) {
          const v1 = idx(x1, y1);
          if (burnable[v1]) ring.push(v1);
        }
        if (dy !== 0 && inBounds(x2, y2)) {
          const v2 = idx(x2, y2);
          if (burnable[v2]) ring.push(v2);
        }
      }
      addCandidate(ring, []);
      if (ring.length > cutBudget) break;
    }
  }
}

const maxSeeds = Math.min(frontierCells.length, Math.max(32, cutBudget * 6));
for (let s = 0; s < maxSeeds; s++) {
  const start = frontierCells[(s * 1103515245 + 12345) % frontierCells.length];
  const localSeen = new Uint8Array(n);
  const q = new Int32Array(Math.min(n, cutBudget * 16 + 1024));
  let head = 0;
  let tail = 0;
  q[tail++] = start;
  localSeen[start] = 1;
  const firePatch = [];
  const compMark = new Int32Array(Math.min(compCount, cutBudget * 32 + 2048));
  compMark.fill(-1);
  const touchedComps = [];
  while (head < tail && firePatch.length <= cutBudget * 4) {
    const v = q[head++];
    firePatch.push(v);
    const x = v % width;
    const y = (v / width) | 0;
    for (let k = 0; k < 4; k++) {
      const nx = x + dirs[k];
      const ny = y + dirs[k + 1];
      if (!inBounds(nx, ny)) continue;
      const ni = idx(nx, ny);
      if (!burnable[ni]) continue;
      if (baseFire[ni]) {
        if (!frontierSeen[ni] || localSeen[ni]) continue;
        localSeen[ni] = 1;
        if (tail < q.length) q[tail++] = ni;
      } else {
        const c = safeComponent[ni];
        if (c >= 0) touchedComps.push(c);
      }
    }
  }
  if (firePatch.length === 0) continue;
  const cellSet = [];
  const compSet = [];
  const seenCell = new Uint8Array(n);
  const seenComp = new Uint8Array(compCount);
  for (let i = 0; i < firePatch.length; i++) {
    const v = firePatch[i];
    const x = v % width;
    const y = (v / width) | 0;
    let isCut = false;
    for (let k = 0; k < 4; k++) {
      const nx = x + dirs[k];
      const ny = y + dirs[k + 1];
      if (!inBounds(nx, ny)) continue;
      const ni = idx(nx, ny);
      if (burnable[ni] && !baseFire[ni]) {
        isCut = true;
        if (!seenComp[safeComponent[ni]]) {
          seenComp[safeComponent[ni]] = 1;
          compSet.push(safeComponent[ni]);
        }
      }
    }
    if (isCut && !seenCell[v]) {
      seenCell[v] = 1;
      cellSet.push(v);
    }
  }
  addCandidate(cellSet, compSet);
}

candidates.sort((a, b) => {
  if (b.score !== a.score) return b.score - a.score;
  if (b.value !== a.value) return b.value - a.value;
  return a.cells.length - b.cells.length;
});

const blocked = new Uint8Array(n);
const chosen = [];
let remaining = cutBudget;

for (let i = 0; i < candidates.length; i++) {
  const cand = candidates[i];
  if (cand.cells.length > remaining) continue;
  let fresh = 0;
  for (let j = 0; j < cand.cells.length; j++) {
    if (!blocked[cand.cells[j]]) fresh++;
  }
  if (fresh === 0 || fresh > remaining) continue;
  let marginal = 0;
  for (let j = 0; j < cand.comps.length; j++) {
    const c = cand.comps[j];
    marginal += compTownValue[c];
  }
  if (cand.comps.length === 0) {
    const trialBlocked = blocked.slice();
    for (let j = 0; j < cand.cells.length; j++) trialBlocked[cand.cells[j]] = 1;
    const fire = bfsReachable(trialBlocked);
    for (let j = 0; j < cand.cells.length; j++) {
      const v = cand.cells[j];
      if (baseFire[v] && !fire[v]) marginal += townValueAt[v];
    }
    for (let v = 0; v < n; v++) {
      if (townValueAt[v] > 0 && baseFire[v] && !fire[v]) marginal += townValueAt[v];
    }
  }
  if (marginal <= 0) continue;
  for (let j = 0; j < cand.cells.length; j++) {
    const v = cand.cells[j];
    if (!blocked[v]) {
      blocked[v] = 1;
      chosen.push(v);
      remaining--;
    }
  }
  if (remaining === 0) break;
}

if (chosen.length < cutBudget) {
  const dist = new Int32Array(n);
  dist.fill(-1);
  const q = new Int32Array(n);
  let head = 0;
  let tail = 0;
  for (let i = 0; i < ignitionList.length; i++) {
    const s = ignitionList[i];
    if (blocked[s]) continue;
    dist[s] = 0;
    q[tail++] = s;
  }
  while (head < tail) {
    const v = q[head++];
    const x = v % width;
    const y = (v / width) | 0;
    for (let k = 0; k < 4; k++) {
      const nx = x + dirs[k];
      const ny = y + dirs[k + 1];
      if (!inBounds(nx, ny)) continue;
      const ni = idx(nx, ny);
      if (!burnable[ni] || blocked[ni] || dist[ni] !== -1) continue;
      dist[ni] = dist[v] + 1;
      q[tail++] = ni;
    }
  }

  const fallback = [];
  for (let v = 0; v < n; v++) {
    if (!burnable[v] || blocked[v] || dist[v] < 0) continue;
    const x = v % width;
    const y = (v / width) | 0;
    let nearbyTownValue = townValueAt[v];
    let touchesWaterOrEdge = 0;
    for (let k = 0; k < 4; k++) {
      const nx = x + dirs[k];
      const ny = y + dirs[k + 1];
      if (!inBounds(nx, ny)) {
        touchesWaterOrEdge++;
        continue;
      }
      const ni = idx(nx, ny);
      if (!burnable[ni]) touchesWaterOrEdge++;
      nearbyTownValue += townValueAt[ni];
    }
    const heuristic = nearbyTownValue * 10 + touchesWaterOrEdge * 2 - dist[v] * 0.01;
    fallback.push([heuristic, v]);
  }
  fallback.sort((a, b) => b[0] - a[0]);
  for (let i = 0; i < fallback.length && chosen.length < cutBudget; i++) {
    const v = fallback[i][1];
    if (blocked[v]) continue;
    blocked[v] = 1;
    chosen.push(v);
  }
}

const finalBlocked = new Uint8Array(n);
for (let i = 0; i < chosen.length; i++) finalBlocked[chosen[i]] = 1;
let bestFire = bfsReachable(finalBlocked);
let bestSaved = 0;
for (let i = 0; i < n; i++) {
  if (baseFire[i] && !bestFire[i]) bestSaved += townValueAt[i];
}

if (chosen.length > 1) {
  const order = chosen.slice().sort((a, b) => {
    const va = townValueAt[a];
    const vb = townValueAt[b];
    if (vb !== va) return vb - va;
    return a - b;
  });
  const keep = new Uint8Array(n);
  const trialBlocked = new Uint8Array(n);
  for (let i = 0; i < order.length; i++) {
    keep[order[i]] = 1;
    trialBlocked[order[i]] = 1;
  }
  for (let i = order.length - 1; i >= 0; i--) {
    const v = order[i];
    trialBlocked[v] = 0;
    const fire = bfsReachable(trialBlocked);
    let saved = 0;
    for (let j = 0; j < n; j++) {
      if (baseFire[j] && !fire[j]) saved += townValueAt[j];
    }
    if (saved >= bestSaved) {
      bestSaved = saved;
      keep[v] = 0;
    } else {
      trialBlocked[v] = 1;
    }
  }
  chosen.length = 0;
  for (let v = 0; v < n; v++) {
    if (keep[v]) chosen.push(v);
  }
}

const firebreaks = chosen.slice(0, cutBudget).map((v) => [v % width, (v / width) | 0]);
process.stdout.write(JSON.stringify({ firebreaks }));
