#include <algorithm>
#include <cstdint>
#include <iostream>
#include <numeric>
#include <vector>

using namespace std;

struct Solver {
    int n;
    vector<long long> weight;
    vector<vector<int>> adj;

    long long best_value = 0;
    vector<int> best_solution;

    vector<int> color;
    vector<long long> color_weight;

    explicit Solver(int n_) : n(n_), weight(n_), adj(n_), best_solution(n_, 0), color(n_, -1) {}

    long long greedy_coloring_bound(const vector<int>& vertices) {
        int used_colors = 0;
        fill(color.begin(), color.end(), -1);
        color_weight.assign(vertices.size(), 0);

        vector<int> order = vertices;
        sort(order.begin(), order.end(), [&](int a, int b) {
            if (adj[a].size() != adj[b].size()) return adj[a].size() > adj[b].size();
            return weight[a] > weight[b];
        });

        for (int v : order) {
            int chosen = 0;
            while (true) {
                bool ok = true;
                for (int to : adj[v]) {
                    if (color[to] == chosen) {
                        ok = false;
                        break;
                    }
                }
                if (ok) break;
                ++chosen;
            }
            color[v] = chosen;
            if (chosen == used_colors) {
                ++used_colors;
            }
            color_weight[chosen] = max(color_weight[chosen], weight[v]);
        }

        long long bound = 0;
        for (int i = 0; i < used_colors; ++i) bound += color_weight[i];
        return bound;
    }

    void branch(vector<int>& candidates, vector<int>& current, long long current_value) {
        if (candidates.empty()) {
            if (current_value > best_value) {
                best_value = current_value;
                best_solution = current;
            }
            return;
        }

        long long upper_bound = current_value + greedy_coloring_bound(candidates);
        if (upper_bound <= best_value) return;

        int pick = candidates[0];
        for (int v : candidates) {
            if (weight[v] > weight[pick]) pick = v;
        }

        vector<int> next_with;
        next_with.reserve(candidates.size());
        vector<char> blocked(n, 0);
        blocked[pick] = 1;
        for (int to : adj[pick]) blocked[to] = 1;
        for (int v : candidates) {
            if (!blocked[v]) next_with.push_back(v);
        }

        current[pick] = 1;
        branch(next_with, current, current_value + weight[pick]);
        current[pick] = 0;

        vector<int> next_without;
        next_without.reserve(candidates.size() - 1);
        for (int v : candidates) {
            if (v != pick) next_without.push_back(v);
        }
        branch(next_without, current, current_value);
    }

    vector<int> solve() {
        vector<int> candidates(n);
        iota(candidates.begin(), candidates.end(), 0);
        vector<int> current(n, 0);
        branch(candidates, current, 0);
        return best_solution;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    if (!(cin >> n >> m)) return 0;

    Solver solver(n);
    for (int i = 0; i < n; ++i) {
        cin >> solver.weight[i];
    }

    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        if (u == v) continue;
        solver.adj[u].push_back(v);
        solver.adj[v].push_back(u);
    }

    vector<int> ans = solver.solve();
    long long objective = 0;
    for (int i = 0; i < n; ++i) {
        if (ans[i]) objective += solver.weight[i];
    }

    cout << objective << '\n';
    for (int i = 0; i < n; ++i) {
        if (i) cout << ' ';
        cout << ans[i];
    }
    cout << '\n';
    return 0;
}
