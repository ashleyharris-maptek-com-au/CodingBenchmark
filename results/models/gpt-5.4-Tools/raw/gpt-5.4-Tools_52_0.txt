import math
from statistics import median


_MEM = {
    "speed_kmh": 0.0,
    "yaw_rate_dps": 0.0,
    "lane_offset_m": 0.0,
    "curvature_1pm": 0.0,
    "speed_limit_kmh": 130.0,
    "front_obstacle_dist_m": 250.0,
    "front_rel_speed_kmh": 0.0,
    "left_blocked": False,
    "right_blocked": False,
    "target_lane_hold": 1,
    "lane_change_cooldown_s": 0.0,
    "steer_i": 0.0,
    "speed_i": 0.0,
    "last_steering": 0.0,
    "last_throttle": 0.0,
    "last_brake": 0.0,
    "stuck_throttle_timer_s": 0.0,
    "brake_failure_timer_s": 0.0,
}


def _clamp(value, lo, hi):
    return max(lo, min(hi, value))


def _get(state, key, default=0.0):
    value = state.get(key, default)
    return default if value is None else value


def _is_finite_number(value):
    return isinstance(value, (int, float)) and math.isfinite(value)


def _robust_pick(readings, default=None, tolerance=None, plausible=None):
    candidates = []
    for value, claimed_valid in readings:
        if not _is_finite_number(value):
            continue
        if plausible and not plausible(value):
            continue
        weight = 1.0 if claimed_valid else 0.45
        candidates.append((float(value), weight))
    if not candidates:
        return default
    values = [v for v, _ in candidates]
    med = median(values)
    if tolerance is not None:
        filtered = [v for v, _ in candidates if abs(v - med) <= tolerance]
        if filtered:
            return float(sum(filtered) / len(filtered))
    return float(med)


def _smooth(name, new_value, dt, tau, default=None):
    if new_value is None:
        return _MEM.get(name, default if default is not None else 0.0)
    prev = _MEM.get(name, new_value if default is None else default)
    alpha = _clamp(dt / max(tau, dt), 0.0, 1.0)
    filt = prev + alpha * (new_value - prev)
    _MEM[name] = filt
    return filt


def _lane_index(value):
    try:
        return int(round(float(value)))
    except Exception:
        return _MEM.get("target_lane_hold", 1)


def autopilot_step(state: dict, dt: float) -> dict:
    dt = float(dt if _is_finite_number(dt) and dt > 1e-4 else _get(state, "dt", 0.1))
    dt = _clamp(dt, 0.02, 0.25)

    wheel_readings = [
        (_get(state, "wheel_speed_fl_kmh"), bool(state.get("wheel_speed_fl_valid", True))),
        (_get(state, "wheel_speed_fr_kmh"), bool(state.get("wheel_speed_fr_valid", True))),
        (_get(state, "wheel_speed_rl_kmh"), bool(state.get("wheel_speed_rl_valid", True))),
        (_get(state, "wheel_speed_rr_kmh"), bool(state.get("wheel_speed_rr_valid", True))),
        (_get(state, "vehicle_speed_1_kmh"), True),
        (_get(state, "vehicle_speed_2_kmh"), True),
        (_get(state, "gps_1_speed_kmh"), bool(state.get("gps_1_valid", True))),
        (_get(state, "gps_2_speed_kmh"), bool(state.get("gps_2_valid", True))),
    ]
    speed_kmh_raw = _robust_pick(
        wheel_readings,
        default=_MEM["speed_kmh"],
        tolerance=12.0,
        plausible=lambda v: -1.0 <= v <= 300.0,
    )
    speed_kmh = _smooth("speed_kmh", _clamp(speed_kmh_raw, 0.0, 250.0), dt, 0.35, 0.0)
    speed_ms = speed_kmh / 3.6

    yaw_rate = _robust_pick(
        [
            (_get(state, "gyro_1_yaw_rate_dps"), bool(state.get("gyro_1_valid", True))),
            (_get(state, "gyro_2_yaw_rate_dps"), bool(state.get("gyro_2_valid", True))),
            (_get(state, "gyro_3_yaw_rate_dps"), bool(state.get("gyro_3_valid", True))),
        ],
        default=_MEM["yaw_rate_dps"],
        tolerance=6.0,
        plausible=lambda v: -120.0 <= v <= 120.0,
    )
    yaw_rate = _smooth("yaw_rate_dps", yaw_rate, dt, 0.25, 0.0)

    lat_accel = _robust_pick(
        [
            (_get(state, "accel_1_lateral_ms2"), bool(state.get("accel_1_valid", True))),
            (_get(state, "accel_2_lateral_ms2"), bool(state.get("accel_2_valid", True))),
            (_get(state, "accel_3_lateral_ms2"), bool(state.get("accel_3_valid", True))),
        ],
        default=0.0,
        tolerance=3.0,
        plausible=lambda v: -20.0 <= v <= 20.0,
    )
    lon_accel = _robust_pick(
        [
            (_get(state, "accel_1_longitudinal_ms2"), bool(state.get("accel_1_valid", True))),
            (_get(state, "accel_2_longitudinal_ms2"), bool(state.get("accel_2_valid", True))),
            (_get(state, "accel_3_longitudinal_ms2"), bool(state.get("accel_3_valid", True))),
        ],
        default=0.0,
        tolerance=3.0,
        plausible=lambda v: -20.0 <= v <= 20.0,
    )

    steering_angle_deg = _robust_pick(
        [
            (_get(state, "steering_angle_1_deg"), bool(state.get("steering_angle_1_valid", True))),
            (_get(state, "steering_angle_2_deg"), bool(state.get("steering_angle_2_valid", True))),
        ],
        default=0.0,
        tolerance=10.0,
        plausible=lambda v: -720.0 <= v <= 720.0,
    )

    mu = _clamp(float(_get(state, "estimated_mu", 0.9)), 0.1, 1.2)
    rain = float(_get(state, "rain_sensor", 0.0))
    temperature_c = float(_get(state, "temperature_c", 20.0))

    lane_offsets = []
    curvatures = []
    speed_limits = []
    obstacle_distances = []
    obstacle_rel_speeds = []
    left_solid_votes = 0
    right_solid_votes = 0
    lane_sources = 0

    for idx in (1, 2, 3):
        left_valid = bool(state.get(f"vision_{idx}_left_line_valid", False))
        right_valid = bool(state.get(f"vision_{idx}_right_line_valid", False))
        left_dist = _get(state, f"vision_{idx}_left_line_dist_m")
        right_dist = _get(state, f"vision_{idx}_right_line_dist_m")
        lane_offset = _get(state, f"vision_{idx}_lane_offset_m")
        if _is_finite_number(lane_offset) and abs(lane_offset) < 4.0:
            weight = 1.0
            if left_valid or right_valid:
                weight += 0.5
            lane_offsets.append((float(lane_offset), weight))
        elif left_valid and right_valid and _is_finite_number(left_dist) and _is_finite_number(right_dist):
            if 0.0 < left_dist < 6.0 and 0.0 < right_dist < 6.0:
                lane_offsets.append(((float(left_dist) - float(right_dist)) * 0.5, 1.2))

        if left_valid or right_valid:
            lane_sources += 1

        curvature = _get(state, f"vision_{idx}_road_curvature_1pm")
        if _is_finite_number(curvature) and abs(curvature) < 0.05:
            curvatures.append(float(curvature))

        speed_limit = _get(state, f"vision_{idx}_speed_limit_kmh")
        if bool(state.get(f"vision_{idx}_speed_limit_valid", False)) and _is_finite_number(speed_limit):
            if 20.0 <= speed_limit <= 160.0:
                speed_limits.append(float(speed_limit))

        if str(state.get(f"vision_{idx}_left_line_type", "")).lower() == "solid":
            left_solid_votes += 1
        if str(state.get(f"vision_{idx}_right_line_type", "")).lower() == "solid":
            right_solid_votes += 1

        if bool(state.get(f"vision_{idx}_obstacle_detected", False)) and bool(state.get(f"vision_{idx}_obstacle_valid", True)):
            dist = _get(state, f"vision_{idx}_obstacle_dist_m")
            rel_speed = _get(state, f"vision_{idx}_obstacle_rel_speed_kmh")
            if _is_finite_number(dist) and 0.0 < dist < 250.0:
                obstacle_distances.append(float(dist))
                if _is_finite_number(rel_speed) and -200.0 < rel_speed < 200.0:
                    obstacle_rel_speeds.append(float(rel_speed))

    lane_offset = None
    if lane_offsets:
        lane_offset = sum(v * w for v, w in lane_offsets) / sum(w for _, w in lane_offsets)
    lane_offset = _smooth("lane_offset_m", lane_offset, dt, 0.30, 0.0)

    curvature = median(curvatures) if curvatures else None
    curvature = _smooth("curvature_1pm", curvature, dt, 0.40, 0.0)

    speed_limit_kmh = median(speed_limits) if speed_limits else None
    speed_limit_kmh = _smooth("speed_limit_kmh", speed_limit_kmh, dt, 1.00, 130.0)

    lidar_angles = [-90, -60, -45, -30, -15, -5, 5, 15, 30, 45, 60, 90]
    lidar = {}
    for ang in lidar_angles:
        key = f"lidar_{ang:+04d}_deg_range_m"
        valid_key = f"lidar_{ang:+04d}_deg_valid"
        dist = _get(state, key, None)
        if _is_finite_number(dist) and 0.0 < dist < 300.0:
            lidar[ang] = float(dist)
        elif valid_key in state and bool(state.get(valid_key, False)):
            lidar[ang] = None

    front_lidar_candidates = [lidar[a] for a in (-15, -5, 5, 15) if lidar.get(a) is not None]
    front_side_left = [lidar[a] for a in (30, 45, 60) if lidar.get(a) is not None]
    front_side_right = [lidar[a] for a in (-30, -45, -60) if lidar.get(a) is not None]

    front_lidar = min(front_lidar_candidates) if front_lidar_candidates else None
    left_gap = min(front_side_left) if front_side_left else None
    right_gap = min(front_side_right) if front_side_right else None

    if front_lidar is not None:
        obstacle_distances.append(front_lidar)
    front_obstacle_dist = min(obstacle_distances) if obstacle_distances else None
    front_obstacle_dist = _smooth("front_obstacle_dist_m", front_obstacle_dist, dt, 0.18, 250.0)

    rel_speed_kmh = median(obstacle_rel_speeds) if obstacle_rel_speeds else None
    if rel_speed_kmh is None and front_obstacle_dist < 200.0:
        closing_est = max(0.0, _MEM["front_obstacle_dist_m"] - front_obstacle_dist) / max(dt, 1e-3) * 3.6
        rel_speed_kmh = -closing_est
    rel_speed_kmh = _smooth("front_rel_speed_kmh", rel_speed_kmh, dt, 0.25, 0.0)

    left_blocked = (left_gap is not None and left_gap < max(10.0, speed_ms * 0.7 + 8.0)) or left_solid_votes >= 2
    right_blocked = (right_gap is not None and right_gap < max(10.0, speed_ms * 0.7 + 8.0)) or right_solid_votes >= 2
    _MEM["left_blocked"] = left_blocked
    _MEM["right_blocked"] = right_blocked

    target_lane = _clamp(_lane_index(state.get("target_lane", _MEM["target_lane_hold"])), 0, 2)
    current_lane = _clamp(_lane_index(state.get("current_lane", target_lane)), 0, 2)
    _MEM["lane_change_cooldown_s"] = max(0.0, _MEM["lane_change_cooldown_s"] - dt)

    desired_speed_kmh = _clamp(float(_get(state, "target_speed_kmh", 100.0)), 0.0, 160.0)
    desired_speed_kmh = min(desired_speed_kmh, speed_limit_kmh + 2.0)

    mu_speed_scale = _clamp(0.45 + 0.55 * mu, 0.35, 1.0)
    if rain > 0.5:
        mu_speed_scale *= 0.92
    if temperature_c < 1.0:
        mu_speed_scale *= 0.9
    desired_speed_kmh *= mu_speed_scale

    curve_mag = abs(curvature)
    curve_speed_cap = 140.0
    if curve_mag > 1e-4:
        curve_speed_cap = math.sqrt(max(6.0, mu * 2.7 / curve_mag)) * 3.6
    desired_speed_kmh = min(desired_speed_kmh, _clamp(curve_speed_cap, 30.0, 140.0))

    ttc = 999.0
    if front_obstacle_dist < 250.0:
        closing_speed_ms = max(0.0, (speed_kmh - max(0.0, speed_kmh + rel_speed_kmh)) / 3.6)
        if obstacle_rel_speeds:
            closing_speed_ms = max(0.0, -rel_speed_kmh / 3.6)
        if closing_speed_ms > 0.1:
            ttc = front_obstacle_dist / closing_speed_ms
        else:
            ttc = 999.0 if front_obstacle_dist > 35.0 else 4.0

    safety_buffer = max(10.0, speed_ms * 1.6)
    emergency_buffer = max(7.0, speed_ms * 0.8)
    obstacle_ahead = front_obstacle_dist < safety_buffer or ttc < 4.0
    emergency = front_obstacle_dist < emergency_buffer or ttc < 1.8

    chosen_lane = target_lane
    if obstacle_ahead and _MEM["lane_change_cooldown_s"] <= 0.0:
        left_lane_ok = current_lane > 0 and not left_blocked
        right_lane_ok = current_lane < 2 and not right_blocked
        prefer_left = target_lane < current_lane
        if left_lane_ok and (prefer_left or not right_lane_ok):
            chosen_lane = current_lane - 1
        elif right_lane_ok:
            chosen_lane = current_lane + 1
        elif left_lane_ok:
            chosen_lane = current_lane - 1
        else:
            chosen_lane = current_lane
        if chosen_lane != current_lane:
            _MEM["lane_change_cooldown_s"] = 2.0
    elif _MEM["lane_change_cooldown_s"] > 0.0:
        chosen_lane = _MEM["target_lane_hold"]

    if current_lane > chosen_lane and left_blocked:
        chosen_lane = current_lane
    if current_lane < chosen_lane and right_blocked:
        chosen_lane = current_lane
    _MEM["target_lane_hold"] = chosen_lane

    lane_change_error = float(chosen_lane - current_lane)
    desired_lane_offset = _clamp(lane_change_error * 1.1, -1.2, 1.2)
    offset_error = _clamp(lane_offset - desired_lane_offset, -3.0, 3.0)

    steer_limit = 0.9 if mu > 0.4 else 0.55
    yaw_damp = yaw_rate * (0.018 + 0.010 * (1.0 - mu_speed_scale))
    crosswind_term = _clamp(lat_accel * 0.035, -0.20, 0.20)
    curvature_ff = _clamp(curvature * speed_ms * speed_ms * 1.2, -0.35, 0.35)

    kp = 0.36 if lane_sources >= 1 else 0.18
    ki = 0.08 if lane_sources >= 1 else 0.02
    _MEM["steer_i"] = _clamp(_MEM["steer_i"] + (-offset_error) * ki * dt, -0.25, 0.25)

    steering_cmd = -kp * offset_error + curvature_ff - yaw_damp - crosswind_term + _MEM["steer_i"]
    if lane_sources == 0:
        steering_cmd += _clamp(-math.radians(yaw_rate) * 0.8, -0.15, 0.15)

    steering_error = abs(steering_cmd - _MEM["last_steering"])
    jam_limit = 0.18 if abs(steering_angle_deg) > 180.0 else 0.28
    max_steer_rate = 1.4 * mu_speed_scale
    steering_cmd = _clamp(
        steering_cmd,
        _MEM["last_steering"] - max_steer_rate * dt,
        _MEM["last_steering"] + max_steer_rate * dt,
    )
    steering_cmd = _clamp(steering_cmd, -min(steer_limit, jam_limit + steering_error), min(steer_limit, jam_limit + steering_error))

    follow_speed_cap = desired_speed_kmh
    if obstacle_ahead:
        desired_gap = max(10.0, speed_ms * (2.2 if mu < 0.45 else 1.8))
        gap_ratio = _clamp((front_obstacle_dist - 4.0) / max(desired_gap, 1.0), 0.0, 1.2)
        follow_speed_cap = min(follow_speed_cap, speed_kmh * gap_ratio + max(0.0, rel_speed_kmh))
        follow_speed_cap = max(0.0, follow_speed_cap)

    desired_speed_kmh = min(desired_speed_kmh, follow_speed_cap if obstacle_ahead else desired_speed_kmh)
    speed_error = desired_speed_kmh - speed_kmh
    accel_target = _clamp(0.22 * speed_error, -6.0 * mu_speed_scale, 2.5 * mu_speed_scale)

    if emergency:
        accel_target = -8.5 * mu_speed_scale - 1.5
    elif ttc < 2.8:
        accel_target = min(accel_target, -4.5 * mu_speed_scale - 0.5)

    _MEM["speed_i"] = _clamp(_MEM["speed_i"] + speed_error * 0.02 * dt, -0.30, 0.30)

    throttle = 0.0
    brake = 0.0
    if accel_target >= 0.0:
        throttle = _clamp(0.12 + 0.16 * accel_target + _MEM["speed_i"], 0.0, 0.75 * mu_speed_scale + 0.15)
        if abs(steering_cmd) > 0.35 or mu < 0.35:
            throttle *= 0.65
    else:
        brake = _clamp((-accel_target) / max(2.5 + 6.5 * mu_speed_scale, 2.5), 0.0, 1.0)
        if mu < 0.35:
            brake = min(brake, 0.55)
        throttle = 0.0

    reported_brake = _robust_pick(
        [
            (_get(state, "brake_pressure_1_pct"), bool(state.get("brake_pressure_1_valid", True))),
            (_get(state, "brake_pressure_2_pct"), bool(state.get("brake_pressure_2_valid", True))),
        ],
        default=0.0,
        tolerance=25.0,
        plausible=lambda v: 0.0 <= v <= 100.0,
    )
    if brake > 0.35 and reported_brake < brake * 45.0 and lon_accel > -0.5:
        _MEM["brake_failure_timer_s"] += dt
    else:
        _MEM["brake_failure_timer_s"] = max(0.0, _MEM["brake_failure_timer_s"] - 2.0 * dt)

    if throttle < 0.05 and speed_error < -8.0 and lon_accel > -0.2:
        _MEM["stuck_throttle_timer_s"] += dt
    else:
        _MEM["stuck_throttle_timer_s"] = max(0.0, _MEM["stuck_throttle_timer_s"] - dt)

    if _MEM["brake_failure_timer_s"] > 0.6:
        throttle = 0.0
        steering_cmd = _clamp(steering_cmd, -0.20, 0.20)
        brake = min(max(brake, 0.2), 1.0)

    if _MEM["stuck_throttle_timer_s"] > 0.7:
        throttle = 0.0
        brake = max(brake, 0.35 if not emergency else 0.7)

    if lane_sources == 0 and front_lidar is None:
        desired_speed_kmh = min(desired_speed_kmh, 60.0 * mu_speed_scale)
        throttle = min(throttle, 0.18)
        brake = max(brake, 0.08 if speed_kmh > desired_speed_kmh + 5.0 else 0.0)

    if front_lidar is None and obstacle_distances:
        brake = max(brake, 0.15 if front_obstacle_dist < 18.0 else brake)

    if emergency and mu < 0.3:
        steering_cmd = _clamp(steering_cmd, -0.22, 0.22)
        brake = max(brake, 0.55)

    _MEM["last_steering"] = steering_cmd
    _MEM["last_throttle"] = throttle
    _MEM["last_brake"] = brake

    return {
        "steering": _clamp(steering_cmd, -1.0, 1.0),
        "throttle": _clamp(throttle, 0.0, 1.0),
        "brake": _clamp(brake, 0.0, 1.0),
    }
