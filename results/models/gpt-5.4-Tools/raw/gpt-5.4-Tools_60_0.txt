const fs = require('fs');

const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

class MaxHeap {
  constructor(compare) {
    this.compare = compare;
    this.arr = [];
  }

  size() {
    return this.arr.length;
  }

  push(value) {
    const arr = this.arr;
    arr.push(value);
    let idx = arr.length - 1;
    while (idx > 0) {
      const parent = (idx - 1) >> 1;
      if (this.compare(arr[idx], arr[parent]) <= 0) break;
      [arr[idx], arr[parent]] = [arr[parent], arr[idx]];
      idx = parent;
    }
  }

  pop() {
    const arr = this.arr;
    if (arr.length === 0) return null;
    const top = arr[0];
    const last = arr.pop();
    if (arr.length > 0) {
      arr[0] = last;
      let idx = 0;
      for (;;) {
        let best = idx;
        const left = idx * 2 + 1;
        const right = left + 1;
        if (left < arr.length && this.compare(arr[left], arr[best]) > 0) best = left;
        if (right < arr.length && this.compare(arr[right], arr[best]) > 0) best = right;
        if (best === idx) break;
        [arr[idx], arr[best]] = [arr[best], arr[idx]];
        idx = best;
      }
    }
    return top;
  }
}

function solve(machineCount, tasksInput) {
  const n = tasksInput.length;
  if (n === 0) return { schedule: [] };

  const idToIndex = new Map();
  for (let i = 0; i < n; i++) idToIndex.set(tasksInput[i].id, i);

  const tasks = new Array(n);
  const succ = Array.from({ length: n }, () => []);
  const indegree = new Int32Array(n);

  for (let i = 0; i < n; i++) {
    const src = tasksInput[i];
    const deps = Array.isArray(src.deps) ? src.deps : [];
    const depIdx = new Int32Array(deps.length);
    for (let j = 0; j < deps.length; j++) {
      const p = idToIndex.get(deps[j]);
      if (p === undefined) continue;
      depIdx[j] = p;
      succ[p].push(i);
    }
    indegree[i] = deps.length;
    tasks[i] = {
      id: src.id,
      duration: Math.max(0, src.duration | 0),
      deadline: Number.isFinite(src.deadline) ? src.deadline : 0,
      weight: Number.isFinite(src.weight) ? src.weight : 1,
      deps: depIdx
    };
  }

  const topo = new Int32Array(n);
  let head = 0;
  let tail = 0;
  for (let i = 0; i < n; i++) {
    if (indegree[i] === 0) topo[tail++] = i;
  }
  while (head < tail) {
    const u = topo[head++];
    const nexts = succ[u];
    for (let i = 0; i < nexts.length; i++) {
      const v = nexts[i];
      indegree[v]--;
      if (indegree[v] === 0) topo[tail++] = v;
    }
  }

  if (tail !== n) {
    return { order: tasksInput.map((t) => t.id) };
  }

  const revTopo = new Int32Array(n);
  for (let i = 0; i < n; i++) revTopo[i] = topo[n - 1 - i];

  const bottomLen = new Float64Array(n);
  const succWeight = new Float64Array(n);
  for (let k = 0; k < n; k++) {
    const u = revTopo[k];
    let bestChild = 0;
    let weightSum = 0;
    const nexts = succ[u];
    for (let i = 0; i < nexts.length; i++) {
      const v = nexts[i];
      if (bottomLen[v] > bestChild) bestChild = bottomLen[v];
      weightSum += tasks[v].weight;
    }
    bottomLen[u] = tasks[u].duration + bestChild;
    succWeight[u] = weightSum;
  }

  const depRemain = new Int32Array(n);
  const release = new Float64Array(n);
  for (let i = 0; i < n; i++) depRemain[i] = tasks[i].deps.length;

  const machineAvail = new Float64Array(machineCount);
  const finish = new Float64Array(n);
  const schedule = [];

  function priority(idx, nowEstimate) {
    const task = tasks[idx];
    const slack = task.deadline - (nowEstimate + bottomLen[idx]);
    const urgency = slack < 0 ? -slack : 0;
    return (
      urgency * (task.weight + 1) * 1e9 +
      bottomLen[idx] * 1e6 +
      task.weight * 1e4 +
      succ[idx].length * 1e2 +
      succWeight[idx]
    );
  }

  const ready = new MaxHeap((a, b) => {
    if (a.score !== b.score) return a.score - b.score;
    if (a.release !== b.release) return b.release - a.release;
    return String(b.id).localeCompare(String(a.id));
  });

  for (let i = 0; i < n; i++) {
    if (depRemain[i] === 0) {
      ready.push({
        idx: i,
        score: priority(i, 0),
        release: 0,
        id: tasks[i].id
      });
    }
  }

  let scheduled = 0;
  while (ready.size() > 0) {
    let bestTask = null;
    let bestMachine = 0;
    let bestStart = 0;
    let bestFinish = 0;
    let bestValue = -Infinity;

    const candidates = [];
    const inspect = Math.min(ready.size(), Math.max(8, machineCount * 2));
    for (let i = 0; i < inspect; i++) {
      const item = ready.pop();
      if (!item) break;
      candidates.push(item);
    }

    for (let ci = 0; ci < candidates.length; ci++) {
      const item = candidates[ci];
      const idx = item.idx;
      const task = tasks[idx];
      for (let m = 0; m < machineCount; m++) {
        const start = Math.max(release[idx], machineAvail[m]);
        const end = start + task.duration;
        const late = Math.max(0, end - task.deadline);
        const value =
          item.score -
          late * (task.weight + 1) * 1e7 -
          start * 10 -
          machineAvail[m] * 0.01;
        if (value > bestValue) {
          bestValue = value;
          bestTask = item;
          bestMachine = m;
          bestStart = start;
          bestFinish = end;
        }
      }
    }

    for (let i = 0; i < candidates.length; i++) {
      const item = candidates[i];
      if (item !== bestTask) ready.push(item);
    }

    if (!bestTask) break;

    const idx = bestTask.idx;
    machineAvail[bestMachine] = bestFinish;
    finish[idx] = bestFinish;
    schedule.push({
      id: tasks[idx].id,
      start: bestStart | 0,
      machine: bestMachine
    });
    scheduled++;

    const nexts = succ[idx];
    for (let i = 0; i < nexts.length; i++) {
      const v = nexts[i];
      if (finish[idx] > release[v]) release[v] = finish[idx];
      depRemain[v]--;
      if (depRemain[v] === 0) {
        ready.push({
          idx: v,
          score: priority(v, release[v]),
          release: release[v],
          id: tasks[v].id
        });
      }
    }
  }

  if (scheduled !== n) {
    return { order: Array.from(topo, (idx) => tasks[idx].id) };
  }

  schedule.sort((a, b) => {
    if (a.start !== b.start) return a.start - b.start;
    if (a.machine !== b.machine) return a.machine - b.machine;
    return String(a.id).localeCompare(String(b.id));
  });

  return { schedule };
}

const result = solve(
  Math.max(1, data.machineCount | 0),
  Array.isArray(data.tasks) ? data.tasks : []
);

process.stdout.write(JSON.stringify(result));
