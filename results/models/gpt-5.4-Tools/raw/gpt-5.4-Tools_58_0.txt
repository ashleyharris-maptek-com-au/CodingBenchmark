const fs = require('fs');

const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const grid = Array.isArray(data.grid) ? data.grid : [];
const robots = Array.isArray(data.robots) ? data.robots : [];
const jobs = Array.isArray(data.jobs) ? data.jobs : [];

const height = grid.length;
const width = height > 0 ? grid[0].length : 0;
const cellCount = width * height;

function toIndex(x, y) {
  return y * width + x;
}

function isOpen(x, y) {
  return y >= 0 && y < height && x >= 0 && x < width && grid[y][x] === '.';
}

const open = new Uint8Array(cellCount);
for (let y = 0; y < height; y++) {
  const row = grid[y];
  for (let x = 0; x < width; x++) {
    if (row[x] === '.') {
      open[toIndex(x, y)] = 1;
    }
  }
}

const dirs = [1, 0, -1, 0, 1];
const queue = new Int32Array(Math.max(1, cellCount));
const dist = new Int32Array(Math.max(1, cellCount));
const seen = new Int32Array(Math.max(1, cellCount));
let bfsToken = 1;

function shortestDistance(sx, sy, tx, ty) {
  if (sx === tx && sy === ty) return 0;
  if (!isOpen(sx, sy) || !isOpen(tx, ty)) return Infinity;

  const start = toIndex(sx, sy);
  const target = toIndex(tx, ty);
  const token = bfsToken++;
  let head = 0;
  let tail = 0;

  queue[tail++] = start;
  seen[start] = token;
  dist[start] = 0;

  while (head < tail) {
    const cur = queue[head++];
    const curDist = dist[cur] + 1;
    const cx = cur % width;
    const cy = (cur / width) | 0;

    for (let k = 0; k < 4; k++) {
      const nx = cx + dirs[k];
      const ny = cy + dirs[k + 1];
      if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;
      const ni = toIndex(nx, ny);
      if (!open[ni] || seen[ni] === token) continue;
      if (ni === target) return curDist;
      seen[ni] = token;
      dist[ni] = curDist;
      queue[tail++] = ni;
    }
  }

  return Infinity;
}

const robotIds = robots.map((r) => r.id);
let maxRobotId = -1;
for (let i = 0; i < robotIds.length; i++) {
  if (robotIds[i] > maxRobotId) maxRobotId = robotIds[i];
}

const assignments = Array.from({ length: maxRobotId + 1 }, () => []);

const robotStates = robots
  .map((robot, index) => ({
    id: robot.id,
    x: robot.x,
    y: robot.y,
    order: index,
  }))
  .sort((a, b) => a.id - b.id || a.order - b.order);

const remaining = jobs
  .map((job, index) => ({
    id: job.id,
    x: job.x,
    y: job.y,
    priority: job.priority,
    order: index,
  }))
  .sort((a, b) => b.priority - a.priority || a.id - b.id || a.order - b.order);

while (remaining.length > 0) {
  let bestRobot = -1;
  let bestJob = -1;
  let bestPriority = -Infinity;
  let bestDistance = Infinity;
  let bestRobotId = Infinity;
  let bestJobId = Infinity;

  for (let r = 0; r < robotStates.length; r++) {
    const robot = robotStates[r];
    for (let j = 0; j < remaining.length; j++) {
      const job = remaining[j];
      const d = shortestDistance(robot.x, robot.y, job.x, job.y);

      if (
        job.priority > bestPriority ||
        (job.priority === bestPriority && d < bestDistance) ||
        (job.priority === bestPriority && d === bestDistance && robot.id < bestRobotId) ||
        (
          job.priority === bestPriority &&
          d === bestDistance &&
          robot.id === bestRobotId &&
          job.id < bestJobId
        )
      ) {
        bestPriority = job.priority;
        bestDistance = d;
        bestRobotId = robot.id;
        bestJobId = job.id;
        bestRobot = r;
        bestJob = j;
      }
    }
  }

  const robot = robotStates[bestRobot];
  const job = remaining[bestJob];
  assignments[robot.id].push(job.id);
  robot.x = job.x;
  robot.y = job.y;
  remaining.splice(bestJob, 1);
}

process.stdout.write(JSON.stringify({ assignments }));
