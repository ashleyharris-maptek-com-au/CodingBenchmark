using System;
using System.Collections.Generic;
using System.Linq;

public class Program
{
    public static void Main(string[] args)
    {
        // Read input
        int n = int.Parse(Console.ReadLine());

        int[][] flow = ReadMatrix(n);
        int[][] dist = ReadMatrix(n);

        // Initialize a random permutation
        int[] bestPermutation = Enumerable.Range(0, n).ToArray();
        Shuffle(bestPermutation); // Random initial permutation

        long bestCost = CalculateCost(bestPermutation, flow, dist);

        // --- Simulated Annealing Algorithm ---
        // Parameters (can be tuned)
        double temperature = 100000.0;
        double coolingRate = 0.999; // Adjust based on N
        int maxIterations = 1000000; // Adjust based on N

        Random rand = new Random();

        for (int i = 0; i < maxIterations; i++)
        {
            if (temperature <= 0.1) break; // Stop if temperature is too low

            // Generate a neighbor solution by swapping two elements
            int[] currentPermutation = (int[])bestPermutation.Clone();
            int idx1 = rand.Next(n);
            int idx2 = rand.Next(n);
            while (idx1 == idx2)
            {
                idx2 = rand.Next(n);
            }
            Swap(currentPermutation, idx1, idx2);

            long currentCost = CalculateCost(currentPermutation, flow, dist);

            // Decide whether to accept the new solution
            if (currentCost < bestCost)
            {
                bestCost = currentCost;
                bestPermutation = currentPermutation;
            }
            else if (AcceptanceProbability(bestCost, currentCost, temperature, rand) > rand.NextDouble())
            {
                bestCost = currentCost;
                bestPermutation = currentPermutation;
            }

            temperature *= coolingRate; // Cool down
        }

        // Output result
        Console.WriteLine(bestCost);
        Console.WriteLine(string.Join(" ", bestPermutation));
    }

    // Reads an N x N matrix from console input
    private static int[][] ReadMatrix(int n)
    {
        int[][] matrix = new int[n][];
        for (int i = 0; i < n; i++)
        {
            matrix[i] = Console.ReadLine().Split(' ').Select(int.Parse).ToArray();
        }
        return matrix;
    }

    // Calculates the cost of a given permutation
    private static long CalculateCost(int[] permutation, int[][] flow, int[][] dist)
    {
        long cost = 0;
        int n = permutation.Length;
        for (int i = 0; i < n; i++)
        {
            for (int j = 0; j < n; j++)
            {
                cost += (long)flow[i][j] * dist[permutation[i]][permutation[j]];
            }
        }
        return cost;
    }

    // Shuffles an array (Fisher-Yates)
    private static void Shuffle<T>(T[] array)
    {
        Random rand = new Random();
        int n = array.Length;
        for (int i = n - 1; i > 0; i--)
        {
            int j = rand.Next(i + 1);
            Swap(array, i, j);
        }
    }

    // Swaps two elements in an array
    private static void Swap<T>(T[] array, int i, int j)
    {
        T temp = array[i];
        array[i] = array[j];
        array[j] = temp;
    }

    // Calculates the acceptance probability for Simulated Annealing
    private static double AcceptanceProbability(long energy, long newEnergy, double temperature, Random rand)
    {
        if (newEnergy < energy)
        {
            return 1.0;
        }
        return Math.Exp((energy - newEnergy) / temperature);
    }
}
