; Main function
%main = OpFunction %void None %fn_void_void
%label_main = OpLabel

; Load input gl_Position for p0 and p1
%p0_gl_in_access = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%p0 = OpLoad %v4float %p0_gl_in_access

%p1_gl_in_access = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%p1 = OpLoad %v4float %p1_gl_in_access

; Load input color for c0 and c1
%c0_color_access = OpAccessChain %ptr_input_v3 %_color %int_0
%c0 = OpLoad %v3float %c0_color_access

%c1_color_access = OpAccessChain %ptr_input_v3 %_color %int_1
%c1 = OpLoad %v3float %c1_color_access

; Compute line direction: dir = p1.xy - p0.xy
%p0_xy = OpVectorShuffle %v2float %p0 %p0 %int_0 %int_1
%p1_xy = OpVectorShuffle %v2float %p1 %p1 %int_0 %int_1
%dir_xy = OpFSub %v2float %p1_xy %p0_xy

; Compute perpendicular: perp = normalize(-dir.y, dir.x) * thickness
%dir_y_val = OpVectorExtractDynamic %float %dir_xy %int_1
%neg_dir_y = OpFNegate %float %dir_y_val
%dir_x_val = OpVectorExtractDynamic %float %dir_xy %int_0

%perp_unnorm = OpCompositeConstruct %v2float %neg_dir_y %dir_x_val

; Calculate length of perp_unnorm for normalization
%dot_perp = OpDot %float %perp_unnorm %perp_unnorm
%sqrt_len_perp = OpExtInst %float %_glsl_ext Sqrt %dot_perp

; Check for zero length to avoid division by zero
%is_zero_len = OpFOrdLessThanEqual %bool %sqrt_len_perp %float_epsilon

%splat_sqrt_len_perp = OpVectorSplat %v2float %sqrt_len_perp
%normalized_perp_div = OpFDiv %v2float %perp_unnorm %splat_sqrt_len_perp
%normalized_perp = OpSelect %v2float %is_zero_len %perp_unnorm %normalized_perp_div

; Apply thickness
%perp_scaled = OpFMul %v2float %normalized_perp %float_0_03

; Convert perp_scaled back to v4float for addition with position
%perp_x = OpVectorExtractDynamic %float %perp_scaled %int_0
%perp_y = OpVectorExtractDynamic %float %perp_scaled %int_1
%perp_v4 = OpCompositeConstruct %v4float %perp_x %perp_y %float_0_0 %float_1_0 ; x, y, 0, 1 for position transformation

; Emit 4 vertices
; Vertex 0: p0 + perp_v4
%p0_plus_perp = OpFAdd %v4float %p0 %perp_v4
%gl_out_pos_ptr_v0 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_out_pos_ptr_v0 %p0_plus_perp

; Color for p0 + perp is c0
%out_color_ptr_v0 = OpAccessChain %ptr_output_v3 %_outColor %int_0
OpStore %out_color_ptr_v0 %c0
OpEmitVertex

; Vertex 1: p0 - perp_v4
%p0_minus_perp = OpFSub %v4float %p0 %perp_v4
%gl_out_pos_ptr_v1 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_out_pos_ptr_v1 %p0_minus_perp

; Color for p0 - perp is c0
%out_color_ptr_v1 = OpAccessChain %ptr_output_v3 %_outColor %int_0
OpStore %out_color_ptr_v1 %c0
OpEmitVertex

; Vertex 2: p1 + perp_v4
%p1_plus_perp = OpFAdd %v4float %p1 %perp_v4
%gl_out_pos_ptr_v2 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_out_pos_ptr_v2 %p1_plus_perp

; Color for p1 + perp is c1
%out_color_ptr_v2 = OpAccessChain %ptr_output_v3 %_outColor %int_0
OpStore %out_color_ptr_v2 %c1
OpEmitVertex

; Vertex 3: p1 - perp_v4
%p1_minus_perp = OpFSub %v4float %p1 %perp_v4
%gl_out_pos_ptr_v3 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %gl_out_pos_ptr_v3 %p1_minus_perp

; Color for p1 - perp is c1
%out_color_ptr_v3 = OpAccessChain %ptr_output_v3 %_outColor %int_0
OpStore %out_color_ptr_v3 %c1
OpEmitVertex

OpEndPrimitive
