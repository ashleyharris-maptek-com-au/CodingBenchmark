#include <iostream>
#include <vector>
#include <algorithm>

// Function to find the maximum independent set using backtracking
std::vector<int> findMaxIndependentSet(int numNodes, const std::vector<std::vector<int>>& adj) {
    std::vector<int> maxIndependentSet;
    std::vector<int> currentIndependentSet;
    std::vector<bool> isIndependent(numNodes, true);

    // Backtracking function
    std::function<void(int)> backtrack = [&](int k) {
        if (k == numNodes) {
            if (currentIndependentSet.size() > maxIndependentSet.size()) {
                maxIndependentSet = currentIndependentSet;
            }
            return;
        }

        // Option 1: Include node k in the current independent set
        if (isIndependent[k]) {
            currentIndependentSet.push_back(k);
            std::vector<bool> tempIsIndependent = isIndependent; // Save state

            // Mark k and its neighbors as not independent
            isIndependent[k] = false;
            for (int neighbor : adj[k]) {
                isIndependent[neighbor] = false;
            }

            backtrack(k + 1);

            // Restore state
            currentIndependentSet.pop_back();
            isIndependent = tempIsIndependent;
        }

        // Option 2: Exclude node k from the current independent set
        backtrack(k + 1);
    };

    backtrack(0);
    return maxIndependentSet;
}

int main() {
    // Example Usage:
    // Graph: 0 -- 1
    //        |    |
    //        3 -- 2
    // Nodes: 4
    // Edges: (0,1), (0,3), (1,2), (2,3)

    int numNodes = 4;
    std::vector<std::vector<int>> adj(numNodes);

    // Add edges
    adj[0].push_back(1);
    adj[1].push_back(0);

    adj[0].push_back(3);
    adj[3].push_back(0);

    adj[1].push_back(2);
    adj[2].push_back(1);

    adj[2].push_back(3);
    adj[3].push_back(2);

    std::vector<int> maxIS = findMaxIndependentSet(numNodes, adj);

    std::cout << "Maximum Independent Set: { ";
    for (int node : maxIS) {
        std::cout << node << " ";
    }
    std::cout << "}" << std::endl;

    // Expected output for this graph: {0,2} or {1,3} (size 2)

    return 0;
}
