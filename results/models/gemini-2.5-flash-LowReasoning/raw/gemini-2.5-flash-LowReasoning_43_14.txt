// Basic HLSL Pixel Shader for Vulkan

struct PSInput
{
    [[vk::location(0)]] float4 Color : COLOR;
};

// Define a uniform buffer for external data
[[vk::binding(0, 0)]] ConstantBuffer<ObjectData> ObjectUniforms : register(b0)
{
    float4 MultiplierColor;
};

float4 main(PSInput input) : SV_TARGET
{
    return input.Color * ObjectUniforms.MultiplierColor;
}
