import math
import sys

class Customer:
    def __init__(self, id, x, y, demand):
        self.id = id
        self.x = x
        self.y = y
        self.demand = demand

class Depot:
    def __init__(self, x, y):
        self.x = x
        self.y = y

def calculate_distance(p1_x, p1_y, p2_x, p2_y):
    return math.sqrt((p1_x - p2_x)**2 + (p1_y - p2_y)**2)

def parse_input():
    lines = sys.stdin.readlines()
    
    # N K CAPACITY
    n, k, capacity = map(int, lines[0].split())

    # depot_x depot_y
    depot_x, depot_y = map(int, lines[1].split())
    depot = Depot(depot_x, depot_y)

    customers = []
    for i in range(2, 2 + n):
        parts = list(map(int, lines[i].split()))
        customers.append(Customer(i - 2, parts[0], parts[1], parts[2])) # 0-based index

    return n, k, capacity, depot, customers

def solve_cvrp(n, k, capacity, depot, customers):
    unvisited_customers = set(customer.id for customer in customers)
    routes = []
    total_overall_distance = 0.0

    customer_map = {customer.id: customer for customer in customers}

    while unvisited_customers:
        if len(routes) >= k: # If we exceed the maximum number of routes, we might have an issue
            # This case indicates that even with k vehicles, we can't serve all customers
            # With Nearest Neighbor, this usually means capacity or k is too restrictive.
            # For now, we'll proceed, but a more robust solution might handle this differently.
            pass

        current_route = []
        current_load = 0
        current_x, current_y = depot.x, depot.y
        route_distance = 0.0

        # Start a new route
        while True:
            nearest_customer_id = -1
            min_distance = float('inf')

            for customer_id in unvisited_customers:
                customer = customer_map[customer_id]
                if current_load + customer.demand <= capacity:
                    dist = calculate_distance(current_x, current_y, customer.x, customer.y)
                    if dist < min_distance:
                        min_distance = dist
                        nearest_customer_id = customer_id
            
            if nearest_customer_id != -1:
                customer = customer_map[nearest_customer_id]
                current_route.append(nearest_customer_id)
                unvisited_customers.remove(nearest_customer_id)
                current_load += customer.demand
                route_distance += min_distance
                current_x, current_y = customer.x, customer.y
            else:
                # No more customers can be added to this route (either all visited or capacity reached)
                break
        
        if not current_route and unvisited_customers: # This can happen if a single customer's demand exceeds capacity
            # Or if k is too small to cover all customers
            # For this simple Nearest Neighbor, it means a customer cannot be served by any vehicle due to capacity.
            # A more advanced algorithm would handle this (e.g., by relaxing capacity or assigning large demand customers specially).
            # For now, we assume valid input where this won't be an issue based on problem statement.
            break # Exit if no customers could be added but some are still unvisited

        if current_route: # If the route is not empty, add return trip to depot
            route_distance += calculate_distance(current_x, current_y, depot.x, depot.y)
            routes.append(current_route)
            total_overall_distance += route_distance

        if not unvisited_customers:
            break # All customers visited

    return total_overall_distance, len(routes), routes

def main():
    n, k, capacity, depot, customers = parse_input()
    total_distance, num_routes, routes = solve_cvrp(n, k, capacity, depot, customers)

    print(f"{total_distance}")
    print(f"{num_routes}")
    for route in routes:
        print(" ".join(map(str, route)))

if __name__ == "__main__":
    main()
