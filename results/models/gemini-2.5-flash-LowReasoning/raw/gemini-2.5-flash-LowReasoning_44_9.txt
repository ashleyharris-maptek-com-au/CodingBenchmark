#version 450

layout(location = 0) in vec3 worldPos;
layout(location = 1) in vec3 normal;
layout(location = 2) in vec2 uv;

layout(set = 0, binding = 0) uniform UBO {
    mat4 model;
    mat4 view;
    mat4 projection;
    vec4 lightPos;
    vec4 cameraPos;
    vec4 params;
} ubo;

layout(location = 0) out vec4 outColor;

void main() {
    // Checkerboard pattern
    vec2 scaled_uv = uv * 8.0;
    vec2 checkered_uv_int = floor(scaled_uv);
    float parity = mod(checkered_uv_int.x + checkered_uv_int.y, 2.0);
    float checker_base_color = mix(0.2, 0.9, parity); // 0.2 for even, 0.9 for odd

    // Lambertian shading
    vec3 N = normalize(normal);
    vec3 L = normalize(ubo.lightPos.xyz - worldPos);
    float diffuse = max(0.0, dot(N, L));
    float lambertian_factor = 0.5 + 0.5 * diffuse;

    // Final color calculation
    outColor = vec4(vec3(checker_base_color) * lambertian_factor, 1.0);
}
