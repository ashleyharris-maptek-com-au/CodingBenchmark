use crossterm::{
    cursor::MoveTo,
    style::{Print, ResetColor, SetForegroundColor, Color},
    terminal::{Clear, ClearType},
    ExecutableCommand,
};
use std::{
    io::{self, stdout, Write},
};

// Game constants
const GRAVITY: f64 = 0.002; // pixels/frame^2
const THRUST_POWER: f64 = 0.005; // pixels/frame^2
const FUEL_CONSUMPTION: f64 = 0.1; // units/frame
const ROTATION_SPEED: f64 = 0.05; // radians/frame
const LANDER_WIDTH: u16 = 3;
const LANDER_HEIGHT: u16 = 2;
const LANDING_PAD_WIDTH: u16 = 10;
const INITIAL_FUEL: f64 = 1000.0;
const INITIAL_X_POS: f64 = 50.0;
const INITIAL_Y_POS: f64 = 10.0;
const INITIAL_X_VEL: f64 = 0.05;
const INITIAL_Y_VEL: f64 = 0.0;

#[derive(Debug, Clone, Copy)]
pub struct Lander {
    pub position: (f64, f64), // (x, y)
    pub velocity: (f64, f64), // (vx, vy)
    pub fuel: f64,
    pub angle: f64, // radians
}

impl Lander {
    pub fn new() -> Self {
        Lander {
            position: (INITIAL_X_POS, INITIAL_Y_POS),
            velocity: (INITIAL_X_VEL, INITIAL_Y_VEL),
            fuel: INITIAL_FUEL,
            angle: 0.0, // facing straight up
        }
    }

    pub fn update(&mut self, thrusting: bool, rotate_left: bool, rotate_right: bool, dt: f64) {
        // Apply gravity
        self.velocity.1 += GRAVITY * dt;

        // Apply thrust
        if thrusting && self.fuel > 0.0 {
            let thrust_x = THRUST_POWER * self.angle.sin() * dt;
            let thrust_y = -THRUST_POWER * self.angle.cos() * dt;
            self.velocity.0 += thrust_x;
            self.velocity.1 += thrust_y;
            self.fuel -= FUEL_CONSUMPTION * dt;
            if self.fuel < 0.0 {
                self.fuel = 0.0;
            }
        }

        // Apply rotation
        if rotate_left {
            self.angle -= ROTATION_SPEED * dt;
        }
        if rotate_right {
            self.angle += ROTATION_SPEED * dt;
        }

        // Update position
        self.position.0 += self.velocity.0 * dt;
        self.position.1 += self.velocity.1 * dt;
    }
}

pub enum GameStatus {
    Playing,
    Landed,
    Crashed,
    OutOfFuel,
}

pub struct GameState {
    pub lander: Lander,
    pub status: GameStatus,
    pub score: u64,
    pub landing_pad: (u16, u16, u16), // (x, y, width)
    pub terminal_size: (u16, u16),
}

impl GameState {
    pub fn new(terminal_size: (u16, u16)) -> io::Result<Self> {
        let (cols, rows) = terminal_size;
        let landing_pad_x = cols / 2 - LANDING_PAD_WIDTH / 2;
        let landing_pad_y = rows - 2;

        Ok(GameState {
            lander: Lander::new(),
            status: GameStatus::Playing,
            score: 0,
            landing_pad: (landing_pad_x, landing_pad_y, LANDING_PAD_WIDTH),
            terminal_size,
        })
    }

    pub fn update(&mut self, thrusting: bool, rotate_left: bool, rotate_right: bool, dt: f64) {
        if matches!(self.status, GameStatus::Playing) {
            self.lander.update(thrusting, rotate_left, rotate_right, dt);
            self.check_collisions();
            if self.lander.fuel <= 0.0 && thrusting {
                self.status = GameStatus::OutOfFuel;
            }
        }
    }

    fn check_collisions(&mut self) {
        let (_cols, rows) = self.terminal_size;
        let lander_bottom = self.lander.position.1 + LANDER_HEIGHT as f64;

        // Check collision with ground
        if lander_bottom >= (rows - 1) as f64 {
            // Check if on landing pad
            let lander_center_x = self.lander.position.0 + (LANDER_WIDTH as f64 / 2.0);
            if lander_center_x >= self.landing_pad.0 as f64
                && lander_center_x <= (self.landing_pad.0 + self.landing_pad.2) as f64
            {
                // Check landing speed and angle
                // Tolerances for a soft landing
                const MAX_VERTICAL_VELOCITY: f64 = 0.1;
                const MAX_HORIZONTAL_VELOCITY: f64 = 0.1;
                const MAX_ANGLE_DEVIATION: f64 = 0.1; // radians, ~5.7 degrees

                if self.lander.velocity.1 <= MAX_VERTICAL_VELOCITY
                    && self.lander.velocity.0.abs() <= MAX_HORIZONTAL_VELOCITY
                    && self.lander.angle.abs() <= MAX_ANGLE_DEVIATION
                {
                    self.status = GameStatus::Landed;
                    self.score = (self.lander.fuel * 10.0) as u64 + 100; // Example score
                } else {
                    self.status = GameStatus::Crashed;
                }
            } else {
                self.status = GameStatus::Crashed;
            }
        }
        // Prevent lander from going off screen left/right
        if self.lander.position.0 < 0.0 {
            self.lander.position.0 = 0.0;
            self.lander.velocity.0 = 0.0; // Stop horizontal movement
        } else if self.lander.position.0 + LANDER_WIDTH as f64 >= self.terminal_size.0 as f64 {
            self.lander.position.0 = self.terminal_size.0 as f64 - LANDER_WIDTH as f64;
            self.lander.velocity.0 = 0.0; // Stop horizontal movement
        }

        // Prevent lander from going off screen top (though gravity should prevent this usually)
        if self.lander.position.1 < 0.0 {
            self.lander.position.1 = 0.0;
            self.lander.velocity.1 = 0.0;
        }
    }
}

pub fn draw_game(game_state: &GameState) -> io::Result<()> {
    let mut stdout = stdout();
    let _ = stdout.execute(Clear(ClearType::All))?;

    // Draw landing pad
    let _ = stdout.execute(SetForegroundColor(Color::Green))?;
    for i in 0..game_state.landing_pad.2 {
        let _ = stdout
            .execute(MoveTo(game_state.landing_pad.0 + i, game_state.landing_pad.1))?
            .execute(Print("="));
    }
    let _ = stdout.execute(ResetColor)?;

    // Draw lander
    let lander_x = game_state.lander.position.0 as u16;
    let lander_y = game_state.lander.position.1 as u16;

    let _ = stdout.execute(SetForegroundColor(Color::Yellow))?;
    if lander_y < game_state.terminal_size.1 && lander_x < game_state.terminal_size.0 {
        // Simple lander representation
        let _ = stdout
            .execute(MoveTo(lander_x + 1, lander_y))?
            .execute(Print("^"))?; // Top
        let _ = stdout
            .execute(MoveTo(lander_x, lander_y + 1))?
            .execute(Print("/_\\"))?; // Body
    }
    let _ = stdout.execute(ResetColor)?;

    // Draw info text
    let _ = stdout.execute(MoveTo(0, 0))?;
    let _ = stdout.execute(Print(format!(
        "Fuel: {:.1} | Altitude: {:.1} | Speed: {:.2} (Vx: {:.2}, Vy: {:.2}) | Angle: {:.2}",
        game_state.lander.fuel,
        game_state.terminal_size.1 as f64 - game_state.lander.position.1 - LANDER_HEIGHT as f64,
        (game_state.lander.velocity.0.powi(2) + game_state.lander.velocity.1.powi(2)).sqrt(),
        game_state.lander.velocity.0,
        game_state.lander.velocity.1,
        game_state.lander.angle.to_degrees()
    )))?;

    // Draw game status
    let _ = stdout.execute(MoveTo(0, 1))?;
    match game_state.status {
        GameStatus::Playing => { let _ = stdout.execute(Print("Status: In Air"))?; },
        GameStatus::Landed => { let _ = stdout.execute(Print(format!(
            "Status: Landed! Score: {}",
            game_state.score
        )))?; },
        GameStatus::Crashed => { let _ = stdout.execute(Print("Status: CRASHED!"))?; },
        GameStatus::OutOfFuel => { let _ = stdout.execute(Print("Status: Out of Fuel!"))?; },
    };

    let _ = stdout.flush()?;
    Ok(())
}
