const fs = require('fs');

try {
    // Read input from stdin
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);

    const machineCount = data.machineCount; // Not directly used in generating topological order, but part of input
    const tasks = data.tasks;

    const tasksMap = new Map(); // taskId -> task object
    const inDegree = new Map(); // taskId -> count of unmet dependencies
    const outNeighbors = new Map(); // taskId -> array of taskIds that depend on it
    const readyQueue = []; // Tasks with no unmet dependencies, ready to be processed
    const orderedTasks = []; // Final topological order of task IDs

    // Pre-process tasks to populate maps and initial readyQueue
    for (const task of tasks) {
        tasksMap.set(task.id, task);
        inDegree.set(task.id, task.deps.length);

        // If a task has no dependencies, it's initially ready
        if (task.deps.length === 0) {
            readyQueue.push(task);
        }

        // Build outNeighbors map (adjacency list for dependents)
        for (const depId of task.deps) {
            if (!outNeighbors.has(depId)) {
                outNeighbors.set(depId, []);
            }
            outNeighbors.get(depId).push(task.id);
        }
    }

    // Comparison function for sorting tasks in the readyQueue
    // Priority: Earliest Deadline -> Highest Weight -> Shortest Duration -> Alphabetical ID
    function compareTasks(taskA, taskB) {
        // 1. Earliest Deadline First (ascending)
        if (taskA.deadline !== taskB.deadline) {
            return taskA.deadline - taskB.deadline;
        }
        // 2. Highest Weight First (descending)
        if (taskA.weight !== taskB.weight) {
            return taskB.weight - taskA.weight;
        }
        // 3. Shortest Duration First (ascending)
        if (taskA.duration !== taskB.duration) {
            return taskA.duration - taskB.duration;
        }
        // 4. Tie-break by ID for deterministic results (ascending)
        return taskA.id.localeCompare(taskB.id);
    }

    // Main scheduling loop to generate topological order
    // Continues as long as there are tasks ready to be scheduled
    while (readyQueue.length > 0) {
        // Sort the readyQueue by priority
        readyQueue.sort(compareTasks);

        // Dequeue the highest priority task (first element after sort)
        const currentTask = readyQueue.shift();
        orderedTasks.push(currentTask.id);

        // Get tasks that depend on the currentTask
        const dependents = outNeighbors.get(currentTask.id);
        if (dependents) {
            // For each dependent task, decrement its unmet dependency count
            for (const dependentId of dependents) {
                const newInDegree = inDegree.get(dependentId) - 1;
                inDegree.set(dependentId, newInDegree);

                // If a dependent task now has no unmet dependencies, add it to the readyQueue
                if (newInDegree === 0) {
                    readyQueue.push(tasksMap.get(dependentId));
                }
            }
        }
    }

    // Output the topological order as a JSON object to stdout
    console.log(JSON.stringify({ order: orderedTasks }));

} catch (error) {
    // Basic error handling to catch issues like malformed JSON input
    console.error(JSON.stringify({ error: error.message }));
}
