const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);

    const grid = data.probability;
    const drones = data.drones;
    const maxSteps = data.maxStepsPerDrone;

    if (!grid || grid.length === 0 || grid[0].length === 0) {
        console.log(JSON.stringify({ paths: [] }));
        return;
    }

    const numRows = grid.length;
    const numCols = grid[0].length;

    const allPaths = [];
    // Stores "x,y" strings of cells visited by *any* drone across all paths generated so far.
    const visitedCells = new Set(); 

    // Define possible moves: stay, up, down, left, right (dx, dy)
    // The order matters for deterministic tie-breaking if scores are equal.
    const moves = [[0, 0], [0, -1], [0, 1], [-1, 0], [1, 0]]; 

    for (let i = 0; i < drones.length; i++) {
        const drone = drones[i];
        let currentX = drone.x;
        let currentY = drone.y;
        const dronePath = [];

        for (let step = 0; step < maxSteps; step++) {
            // Add current position to the drone's path for this step
            dronePath.push([currentX, currentY]);
            // Mark the current cell as visited by any drone
            visitedCells.add(`${currentX},${currentY}`);

            let bestNextMove = [currentX, currentY]; // Default: stay in place
            let maxScore = -Infinity; 

            for (const [dx, dy] of moves) {
                const nextX = currentX + dx;
                const nextY = currentY + dy;

                // Check bounds
                if (nextX < 0 || nextX >= numCols || nextY < 0 || nextY >= numRows) {
                    continue;
                }

                // Check for obstacles
                if (grid[nextY][nextX] === -1) {
                    continue;
                }

                const cellValue = grid[nextY][nextX];
                const cellKey = `${nextX},${nextY}`;
                let currentScore = cellValue;

                // Penalize revisiting already covered cells by any drone
                // This encourages exploration of new, high-probability areas.
                if (visitedCells.has(cellKey)) {
                    currentScore *= 0.1; // Penalty factor
                }

                // Update best move if a higher score is found
                if (currentScore > maxScore) {
                    maxScore = currentScore;
                    bestNextMove = [nextX, nextY];
                }
                // Tie-breaking: If scores are equal, the move order in `moves` array acts as a deterministic tie-breaker.
                // For example, if multiple moves have the same highest score, the one that appears
                // earlier in the `moves` array (e.g., stay, then up, then down, etc.) will be chosen.
            }

            // Move the drone to the chosen best position for the next step
            currentX = bestNextMove[0];
            currentY = bestNextMove[1];
        }
        allPaths.push(dronePath);
    }

    console.log(JSON.stringify({ paths: allPaths }));
}

solve();
