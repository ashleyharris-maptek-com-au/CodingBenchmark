import sys
import random
import time

def solve_3sat():
    start_time = time.time()

    # Read num_variables and num_clauses
    line = sys.stdin.readline().split()
    num_variables = int(line[0])
    num_clauses = int(line[1])

    clauses = []
    for _ in range(num_clauses):
        clause_literals = list(map(int, sys.stdin.readline().split()))
        clauses.append(clause_literals)

    # Initialize a random assignment
    # Using a list where index i corresponds to variable i+1
    # value is 0 for False, 1 for True
    current_assignment = [random.randint(0, 1) for _ in range(num_variables)]

    # --- Local Search Algorithm ---
    max_flips = 10 * num_variables  # Heuristic for max iterations
    max_tries = 5 # Number of random restarts

    for _try in range(max_tries):
        # Reset assignment for each try
        current_assignment = [random.randint(0, 1) for _ in range(num_variables)]

        for _flip in range(max_flips):
            # Check if current assignment satisfies all clauses
            satisfied_all = True
            unsatisfied_clauses_indices = []

            for i, clause in enumerate(clauses):
                clause_satisfied = False
                for literal in clause:
                    var_index = abs(literal) - 1
                    var_value = current_assignment[var_index]
                    
                    if (literal > 0 and var_value == 1) or 
                       (literal < 0 and var_value == 0):
                        clause_satisfied = True
                        break
                
                if not clause_satisfied:
                    satisfied_all = False
                    unsatisfied_clauses_indices.append(i)
            
            if satisfied_all:
                print("SAT")
                print(*(1 if val else 0 for val in current_assignment))
                return

            # Pick a random unsatisfied clause
            if not unsatisfied_clauses_indices: # Should not happen if not satisfied_all is True
                continue

            target_clause_index = random.choice(unsatisfied_clauses_indices)
            target_clause = clauses[target_clause_index]

            # Decide which literal's variable to flip
            # Simplified approach: randomly pick one of the literals in the clause
            literal_to_flip = random.choice(target_clause)
            var_to_flip_index = abs(literal_to_flip) - 1
            
            current_assignment[var_to_flip_index] = 1 - current_assignment[var_to_flip_index] # Flip the variable

    # If after all tries and flips, no solution is found (shouldn't happen based on problem statement)
    sys.stderr.write("No solution found within limits (problem guarantees satisfiability).
")

if __name__ == "__main__":
    solve_3sat()
