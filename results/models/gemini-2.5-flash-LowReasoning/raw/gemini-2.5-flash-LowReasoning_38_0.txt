# exact_cover.py
# Implementation of the Exact Cover problem using Algorithm X

def solve_exact_cover(universe, subsets):
    """
    Solves the exact cover problem using Donald Knuth's Algorithm X.

    Args:
        universe: A set of items that need to be covered.
        subsets: A list of sets, where each set is a subset of the universe.

    Returns:
        A list of subsets that form an exact cover, or None if no exact cover exists.
    """
    solutions = []
    _solve(universe, subsets, [], solutions)
    return solutions[0] if solutions else None

def _solve(universe, subsets, current_solution, solutions):
    if not universe:
        solutions.append(list(current_solution))
        return

    # Choose an item from the universe (e.g., the one that appears in the fewest subsets)
    # This is a simple heuristic to potentially reduce branching.
    item_counts = {item: 0 for item in universe}
    for s in subsets:
        for item in s:
            if item in universe: # Only count items still in the universe
                item_counts[item] += 1
    
    # Select the item with the minimum count (greedy approach)
    min_count = float('inf')
    chosen_item = None
    for item, count in item_counts.items():
        if count < min_count:
            min_count = count
            chosen_item = item
    
    if chosen_item is None: # No item left to cover, but universe not empty, implies no solution
        return

    # Iterate through subsets that contain the chosen item
    # Make a copy of subsets to iterate over, as we'll modify it within the loop
    for s in list(subsets): # Convert to list to avoid RuntimeError if subsets is a set
        if chosen_item in s:
            current_solution.append(s)

            # --- Simulate "Cover" operation ---
            # Items to remove from the universe are all items in the chosen subset 's'
            covered_items = s

            # Subsets that would be 'removed' in Algorithm X (temporarily)
            # These are subsets that contain any of the covered_items
            subsets_to_consider = [
                sub for sub in subsets if not any(item in sub for item in covered_items)
            ]
            
            # Create a new universe by removing covered items
            new_universe = universe.difference(covered_items)

            _solve(new_universe, subsets_to_consider, current_solution, solutions)
            
            # --- Simulate "Uncover" operation (backtrack) ---
            current_solution.pop()

if __name__ == "__main__":
    # Example Usage:
    # Universe of elements to be covered
    U = {1, 2, 3, 4, 5, 6, 7}

    # Collection of subsets
    S = [
        {1, 4},
        {2, 5},
        {3, 6},
        {1, 2, 3},
        {4, 5, 6},
        {7}
    ]

    print(f"Universe: {U}")
    print(f"Subsets: {S}")

    solution = solve_exact_cover(U, S)

    if solution:
        print("
Exact Cover found:")
        for s_set in solution:
            print(s_set)
    else:
        print("
No exact cover found.")
