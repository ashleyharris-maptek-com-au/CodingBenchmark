#include "steiner_tree.h"
#include <algorithm> // For std::min
#include <map>

SteinerTreeSolver::SteinerTreeSolver(int num_vertices)
    : num_vertices_(num_vertices),
      adj_(num_vertices),
      dist_(num_vertices, std::vector<int>(num_vertices, INF)) {
    for (int i = 0; i < num_vertices_; ++i) {
        dist_[i][i] = 0;
    }
}

void SteinerTreeSolver::add_edge(int u, int v, int weight) {
    adj_[u].push_back({v, weight});
    adj_[v].push_back({u, weight});
    dist_[u][v] = std::min(dist_[u][v], weight);
    dist_[v][u] = std::min(dist_[v][u], weight);
}

void SteinerTreeSolver::set_terminals(const std::vector<int>& terminals) {
    terminals_ = terminals;
    terminal_to_idx_.clear();
    for (int i = 0; i < terminals_.size(); ++i) {
        terminal_to_idx_[terminals_[i]] = i;
    }
}

void SteinerTreeSolver::compute_all_pairs_shortest_paths() {
    // Initialize dist_ with direct edge weights and INF for no direct edge
    // This is already done in add_edge for direct connections and constructor for self-loops.
    // For non-existent direct edges, dist_ is already INF.

    // Floyd-Warshall algorithm
    for (int k = 0; k < num_vertices_; ++k) {
        for (int i = 0; i < num_vertices_; ++i) {
            for (int j = 0; j < num_vertices_; ++j) {
                if (dist_[i][k] != INF && dist_[k][j] != INF) {
                    dist_[i][j] = std::min(dist_[i][j], dist_[i][k] + dist_[k][j]);
                }
            }
        }
    }
}

int SteinerTreeSolver::solve() {
    // This will be implemented in subsequent steps.
    // For now, call APSP to ensure it's computed.
    compute_all_pairs_shortest_paths();
    return calculate_steiner_tree_cost(); // Placeholder
}

int SteinerTreeSolver::calculate_steiner_tree_cost() {
    int num_terminals = terminals_.size();
    if (num_terminals == 0) {
        return 0; // No terminals, cost is 0
    }
    if (num_vertices_ == 0) {
        return INF; // No vertices, cannot form a tree
    }

    // dp[mask][v] = minimum cost of a Steiner tree connecting terminals in mask,
    // where vertex v is part of the tree.
    // The mask represents a subset of terminals.
    std::vector<std::vector<int>> dp(1 << num_terminals, std::vector<int>(num_vertices_, INF));

    // Initialization: Base cases for DP
    // For each terminal, a tree containing only itself has cost 0.
    for (int i = 0; i < num_terminals; ++i) {
        dp[1 << i][terminals_[i]] = 0;
    }

    // DP Iteration
    // Iterate over increasing mask values (subsets of terminals)
    for (int mask = 1; mask < (1 << num_terminals); ++mask) {
        // Case 1: Merging Subtrees (connecting two smaller Steiner trees)
        // Try all possible splits of 'mask' into 'submask' and 'mask ^ submask'
        for (int submask = (mask - 1) & mask; submask > 0; submask = (submask - 1) & mask) {
            int remaining_mask = mask ^ submask;
            for (int v = 0; v < num_vertices_; ++v) {
                if (dp[submask][v] != INF && dp[remaining_mask][v] != INF) {
                    dp[mask][v] = std::min(dp[mask][v], dp[submask][v] + dp[remaining_mask][v]);
                }
            }
        }

        // Case 2: Shortest Path Connection (extending a subtree from u to v)
        // This is like running Dijkstra for each mask.
        // Relax edges using all-pairs shortest paths (dist_)
        // We iterate k times to simulate shortest path finding for current mask
        // This is effectively a Bellman-Ford like relaxation for the current mask.
        // A more efficient way would be to use a priority queue for each mask,
        // but for simplicity and within limits of this problem size, this can work.
        // For each mask, ensure that the minimum cost to reach 'v' from any 'u'
        // that is already part of the subtree 'mask' is considered.
        // This part means that we try to improve dp[mask][v] by connecting it to dp[mask][u]
        // via shortest path dist[u][v].
        bool changed;
        for (int iter = 0; iter < num_vertices_; ++iter) { // iterate V times for Bellman-Ford style relaxation
            changed = false;
            for (int v = 0; v < num_vertices_; ++v) {
                for (int u = 0; u < num_vertices_; ++u) {
                    if (dp[mask][u] != INF && dist_[u][v] != INF) {
                        if (dp[mask][v] > dp[mask][u] + dist_[u][v]) {
                            dp[mask][v] = dp[mask][u] + dist_[u][v];
                            changed = true;
                        }
                    }
                }
            }
            if (!changed) {
                break;
            }
        }
    }

    // Final Result: Find the minimum cost for the tree connecting all terminals.
    int min_steiner_tree_cost = INF;
    int all_terminals_mask = (1 << num_terminals) - 1;
    for (int v = 0; v < num_vertices_; ++v) {
        min_steiner_tree_cost = std::min(min_steiner_tree_cost, dp[all_terminals_mask][v]);
    }

    return min_steiner_tree_cost;
}
