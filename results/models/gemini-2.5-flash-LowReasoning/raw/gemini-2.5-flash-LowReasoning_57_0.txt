const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);

    const { width, height, grid, ignitions, towns, cutBudget } = data;

    // Helper to convert coordinates to a string key for Sets/Maps
    const coordsToKey = (x, y) => `${x},${y}`;

    // Helper to check if a cell is within bounds and is burnable ('.')
    const isValid = (x, y, currentGrid) => {
        return x >= 0 && x < height && y >= 0 && y < width && currentGrid[x][y] === '.';
    };

    // Helper to run a multi-source BFS for fire spread
    // Returns a Set of keys of burned cells
    const calculateFireSpread = (startCells, currentGridChars, currentBlockedSet) => {
        const burnedCells = new Set();
        const queue = [];

        // Add ignitions to queue if they are valid and not blocked
        for (const { x, y } of startCells) {
            if (isValid(x, y, currentGridChars) && !currentBlockedSet.has(coordsToKey(x, y))) {
                burnedCells.add(coordsToKey(x, y));
                queue.push({ x, y });
            }
        }

        let head = 0;
        while (head < queue.length) {
            const { x, y } = queue[head++];

            // Directions: up, down, left, right
            const directions = [[0, 1], [0, -1], [1, 0], [-1, 0]];

            for (const [dx, dy] of directions) {
                const nx = x + dx;
                const ny = y + dy;
                const nKey = coordsToKey(nx, ny);

                if (isValid(nx, ny, currentGridChars) &&
                    !currentBlockedSet.has(nKey) &&
                    !burnedCells.has(nKey)) {
                    burnedCells.add(nKey);
                    queue.push({ x: nx, y: ny });
                }
            }
        }
        return burnedCells;
    };

    // --- Main Logic ---

    // 1. Initialize blocked cells with water/rock (~)
    const initialBlockedCells = new Set();
    for (let r = 0; r < height; r++) {
        for (let c = 0; c < width; c++) {
            if (grid[r][c] === '~') {
                initialBlockedCells.add(coordsToKey(r, c));
            }
        }
    }

    // 2. Calculate initial fire spread without any firebreaks
    const burnedWithoutFirebreaks = calculateFireSpread(ignitions, grid, initialBlockedCells);

    // 3. Identify threatened towns and their total unprotected value
    let totalBurnedUnprotectedValue = 0;
    const threatenedTowns = [];
    for (const town of towns) {
        if (burnedWithoutFirebreaks.has(coordsToKey(town.x, town.y))) {
            totalBurnedUnprotectedValue += town.value;
            threatenedTowns.push(town);
        }
    }

    // 4. Calculate impact score for each potential firebreak cell
    //    A cell's score is the accumulated value of threatened towns that
    //    could potentially be saved if fire is blocked through that cell.
    const cellImpactScores = new Map(); // Key: "x,y", Value: accumulated town value

    const ignitionKeys = new Set(ignitions.map(ign => coordsToKey(ign.x, ign.y)));

    for (const town of threatenedTowns) {
        const queue = [];
        const visitedForThisTown = new Set(); // Cells visited for tracing paths for *this specific town*

        // Start reverse BFS from the threatened town
        queue.push({ x: town.x, y: town.y });
        visitedForThisTown.add(coordsToKey(town.x, town.y));

        let head = 0;
        while (head < queue.length) {
            const { x: cx, y: cy } = queue[head++];
            const cellKey = coordsToKey(cx, cy);

            // If this cell is burnable, not initially blocked, and not an ignition point
            // (Ignition points cannot be chosen as firebreaks)
            if (isValid(cx, cy, grid) &&
                !initialBlockedCells.has(cellKey) &&
                !ignitionKeys.has(cellKey)) {
                
                const currentScore = cellImpactScores.get(cellKey) || 0;
                cellImpactScores.set(cellKey, currentScore + town.value);
            }

            // Explore neighbors (backwards towards ignitions)
            const directions = [[0, 1], [0, -1], [1, 0], [-1, 0]];
            for (const [dx, dy] of directions) {
                const nx = cx + dx;
                const ny = cy + dy;
                const nKey = coordsToKey(nx, ny);

                if (isValid(nx, ny, grid) &&
                    !initialBlockedCells.has(nKey) &&
                    burnedWithoutFirebreaks.has(nKey) && // Only traverse through cells that actually burned
                    !visitedForThisTown.has(nKey)) {
                    
                    visitedForThisTown.add(nKey);
                    queue.push({ x: nx, y: ny });
                }
            }
        }
    }

    // 5. Sort candidate firebreak cells by their impact score (descending)
    const sortedCandidateFirebreaks = Array.from(cellImpactScores.entries())
        .map(([key, score]) => {
            const [x, y] = key.split(',').map(Number);
            return { x, y, score };
        })
        .sort((a, b) => b.score - a.score);

    // 6. Select the top 'cutBudget' firebreaks
    const finalFirebreaks = [];
    for (let i = 0; i < Math.min(cutBudget, sortedCandidateFirebreaks.length); i++) {
        finalFirebreaks.push({
            x: sortedCandidateFirebreaks[i].x,
            y: sortedCandidateFirebreaks[i].y,
        });
    }

    // 7. Output the result in the required JSON format
    process.stdout.write(JSON.stringify({ firebreaks: finalFirebreaks }));
}

solve();
