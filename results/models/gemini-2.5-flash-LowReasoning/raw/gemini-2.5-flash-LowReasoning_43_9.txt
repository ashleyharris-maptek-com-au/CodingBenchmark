// A simple HLSL pixel shader for Vulkan

// Input from the vertex shader (e.g., position, color, UVs)
// These typically match the output of the vertex shader
struct PSInput
{
    float4 position : SV_POSITION;
    float3 color : COLOR0; // Example: assuming vertex shader passes a color
};

// Main function for the pixel shader
float4 main(PSInput input) : SV_TARGET
{
    // Output the color received from the vertex shader
    return float4(input.color, 1.0f);
}
