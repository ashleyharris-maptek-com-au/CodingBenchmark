// src/main.rs

use std::collections::{HashSet, HashMap};

#[derive(Debug, PartialEq, Eq, Hash, Clone)]
struct Vertex(usize);

#[derive(Debug)]
struct Graph {
    num_vertices: usize,
    adj: HashMap<Vertex, HashSet<Vertex>>,
}

impl Graph {
    fn new(num_vertices: usize) -> Self {
        let mut adj = HashMap::new();
        for i in 0..num_vertices {
            adj.insert(Vertex(i), HashSet::new());
        }
        Graph { num_vertices, adj }
    }

    fn add_edge(&mut self, u: usize, v: usize) {
        if u >= self.num_vertices || v >= self.num_vertices {
            // Handle error or panic, or resize the graph
            eprintln!("Error: Vertex out of bounds.");
            return;
        }
        self.adj.get_mut(&Vertex(u)).unwrap().insert(Vertex(v));
        self.adj.get_mut(&Vertex(v)).unwrap().insert(Vertex(u));
    }


    fn min_vertex_cover(&self) -> HashSet<Vertex> {
        let mut min_cover = HashSet::new();
        let mut edges_to_cover: HashSet<(Vertex, Vertex)> = HashSet::new();

        // Collect all edges that need to be covered
        for (u, neighbors) in &self.adj {
            for v in neighbors {
                // Add edge only once (e.g., (u,v) but not (v,u))
                if u.0 < v.0 {
                    edges_to_cover.insert((u.clone(), v.clone()));
                }
            }
        }

        self.find_min_vertex_cover_recursive(
            edges_to_cover,
            &mut HashSet::new(), // current_cover
            &mut min_cover,      // best_cover_found
        );

        min_cover
    }

    fn find_min_vertex_cover_recursive(
        &self,
        mut edges_to_cover: HashSet<(Vertex, Vertex)>,
        current_cover: &mut HashSet<Vertex>,
        min_cover: &mut HashSet<Vertex>,
    ) {
        // Base case: If no edges are left, this `current_cover` is a valid vertex cover.
        // Check if it's smaller than the `min_cover` found so far.
        if edges_to_cover.is_empty() {
            if min_cover.is_empty() || current_cover.len() < min_cover.len() {
                *min_cover = current_cover.clone();
            }
            return;
        }

        // Optimization: If the current cover is already larger than or equal to the best found so far,
        // there's no need to continue this path.
        if !min_cover.is_empty() && current_cover.len() >= min_cover.len() {
            return;
        }

        // Pick an arbitrary uncovered edge
        let (u, v) = edges_to_cover.iter().next().unwrap().clone();

        // Option 1: Include u in the vertex cover
        current_cover.insert(u.clone());
        let mut next_edges_to_cover_u = edges_to_cover.clone();
        self.remove_covered_edges(&u, &mut next_edges_to_cover_u);
        self.find_min_vertex_cover_recursive(
            next_edges_to_cover_u,
            current_cover,
            min_cover,
        );
        current_cover.remove(&u); // backtrack

        // Option 2: Include v in the vertex cover
        current_cover.insert(v.clone());
        let mut next_edges_to_cover_v = edges_to_cover.clone();
        self.remove_covered_edges(&v, &mut next_edges_to_cover_v);
        self.find_min_vertex_cover_recursive(
            next_edges_to_cover_v,
            current_cover,
            min_cover,
        );
        current_cover.remove(&v); // backtrack
    }

    fn remove_covered_edges(&self, vertex: &Vertex, edges: &mut HashSet<(Vertex, Vertex)>) {
        // Remove all edges incident to 'vertex' from the set of 'edges'.
        edges.retain(|(u, v)| u != vertex && v != vertex);
    }
}

fn main() {
    println!("Minimum Vertex Cover Solver");

    // Example 1: Simple Path Graph 0--1--2
    let mut graph1 = Graph::new(3);
    graph1.add_edge(0, 1);
    graph1.add_edge(1, 2);
    let cover1 = graph1.min_vertex_cover();
    println!("Graph 1 (0--1--2) MVC: {:?}", cover1);
    println!("Size: {}", cover1.len()); // Expected size: 1, e.g., {Vertex(1)}

    println!("\n---------------------\n");

    // Example 2: Triangle Graph 0--1, 1--2, 2--0
    let mut graph2 = Graph::new(3);
    graph2.add_edge(0, 1);
    graph2.add_edge(1, 2);
    graph2.add_edge(2, 0);
    let cover2 = graph2.min_vertex_cover();
    println!("Graph 2 (Triangle) MVC: {:?}", cover2);
    println!("Size: {}", cover2.len()); // Expected size: 2, e.g., {Vertex(0), Vertex(1)}

    println!("\n---------------------\n");

    // Example 3: Star Graph (center 0, connected to 1, 2, 3)
    let mut graph3 = Graph::new(4);
    graph3.add_edge(0, 1);
    graph3.add_edge(0, 2);
    graph3.add_edge(0, 3);
    let cover3 = graph3.min_vertex_cover();
    println!("Graph 3 (Star) MVC: {:?}", cover3);
    println!("Size: {}", cover3.len()); // Expected size: 1, e.g., {Vertex(0)}
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_graph_creation() {
        let graph = Graph::new(3);
        assert_eq!(graph.num_vertices, 3);
        assert_eq!(graph.adj.len(), 3);
        assert!(graph.adj.contains_key(&Vertex(0)));
        assert!(graph.adj.contains_key(&Vertex(1)));
        assert!(graph.adj.contains_key(&Vertex(2)));
    }

    #[test]
    fn test_add_edge() {
        let mut graph = Graph::new(3);
        graph.add_edge(0, 1);
        assert!(graph.adj[&Vertex(0)].contains(&Vertex(1)));
        assert!(graph.adj[&Vertex(1)].contains(&Vertex(0)));
        assert!(!graph.adj[&Vertex(0)].contains(&Vertex(2)));
    }

    #[test]
    fn test_min_vertex_cover_empty_graph() {
        let graph = Graph::new(0);
        let cover = graph.min_vertex_cover();
        assert!(cover.is_empty());
    }

    #[test]
    fn test_min_vertex_cover_no_edges() {
        let graph = Graph::new(3); // 3 vertices, no edges
        let cover = graph.min_vertex_cover();
        assert!(cover.is_empty());
    }

    #[test]
    fn test_min_vertex_cover_single_edge() {
        let mut graph = Graph::new(2);
        graph.add_edge(0, 1);
        let cover = graph.min_vertex_cover();
        assert_eq!(cover.len(), 1);
        assert!(cover.contains(&Vertex(0)) || cover.contains(&Vertex(1)));
    }

    #[test]
    fn test_min_vertex_cover_path_graph_3_vertices() {
        // Graph: 0--1--2
        let mut graph = Graph::new(3);
        graph.add_edge(0, 1);
        graph.add_edge(1, 2);
        let cover = graph.min_vertex_cover();
        assert_eq!(cover.len(), 1);
        assert!(cover.contains(&Vertex(1)));
    }

    #[test]
    fn test_min_vertex_cover_triangle_graph() {
        // Graph: 0--1, 1--2, 2--0
        let mut graph = Graph::new(3);
        graph.add_edge(0, 1);
        graph.add_edge(1, 2);
        graph.add_edge(2, 0);
        let cover = graph.min_vertex_cover();
        assert_eq!(cover.len(), 2);
        // Possible covers: {0,1}, {1,2}, {0,2}
        let expected1: HashSet<Vertex> = [Vertex(0), Vertex(1)].iter().cloned().collect();
        let expected2: HashSet<Vertex> = [Vertex(1), Vertex(2)].iter().cloned().collect();
        let expected3: HashSet<Vertex> = [Vertex(0), Vertex(2)].iter().cloned().collect();
        assert!(cover == expected1 || cover == expected2 || cover == expected3);
    }

    #[test]
    fn test_min_vertex_cover_star_graph() {
        // Star graph: 0 is center, connected to 1, 2, 3
        let mut graph = Graph::new(4);
        graph.add_edge(0, 1);
        graph.add_edge(0, 2);
        graph.add_edge(0, 3);
        let cover = graph.min_vertex_cover();
        assert_eq!(cover.len(), 1);
        assert!(cover.contains(&Vertex(0)));
    }

    // A more complex graph, often used for testing MVC
    // 0--1--2
    // |  |  |
    // 3--4--5
    #[test]
    fn test_min_vertex_cover_grid_graph() {
        let mut graph = Graph::new(6);
        graph.add_edge(0, 1);
        graph.add_edge(1, 2);
        graph.add_edge(0, 3);
        graph.add_edge(1, 4);
        graph.add_edge(2, 5);
        graph.add_edge(3, 4);
        graph.add_edge(4, 5);

        let cover = graph.min_vertex_cover();
        // The minimum vertex cover for this graph is 3.
        // Example covers: {1,3,5}, {0,2,4}, {0,1,5}, {0,2,3}, etc.
        assert_eq!(cover.len(), 3);
        // We cannot assert specific vertices easily because multiple valid MVCs exist.
        // Instead, we verify that the size is correct and that it actually is a vertex cover.
        assert!(is_vertex_cover(&graph, &cover));
    }

    // Helper function to check if a given set of vertices is a valid vertex cover
    fn is_vertex_cover(graph: &Graph, cover: &HashSet<Vertex>) -> bool {
        for u in 0..graph.num_vertices {
            for v in graph.adj[&Vertex(u)].iter() {
                if u < v.0 { // Consider each edge only once
                    if !cover.contains(&Vertex(u)) && !cover.contains(v) {
                        return false; // Edge (u,v) is not covered
                    }
                }
            }
        }
        true
    }
}
