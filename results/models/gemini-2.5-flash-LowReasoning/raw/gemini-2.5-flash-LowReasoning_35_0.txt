use std::collections::{HashSet, VecDeque};
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Read num_vertices and num_edges
    let first_line = lines.next().expect("Expected graph dimensions").unwrap();
    let parts: Vec<usize> = first_line
        .split_whitespace()
        .map(|s| s.parse().expect("Invalid number"))
        .collect();
    let num_vertices = parts[0];
    let num_edges = parts[1];

    // Adjacency list representation
    let mut graph: Vec<Vec<usize>> = vec![vec![]; num_vertices];
    for _ in 0..num_edges {
        let line = lines.next().expect("Expected edge").unwrap();
        let edge_parts: Vec<usize> = line
            .split_whitespace()
            .map(|s| s.parse().expect("Invalid edge node"))
            .collect();
        let u = edge_parts[0];
        let v = edge_parts[1];
        graph[u].push(v);
        graph[v].push(u); // Undirected graph
    }

    // Initialize dominating set and undominated vertices
    let mut dominating_set: HashSet<usize> = HashSet::new();
    let mut undominated_vertices: HashSet<usize> = (0..num_vertices).collect();

    // Greedy algorithm
    while !undominated_vertices.is_empty() {
        let mut best_vertex: Option<usize> = None;
        let mut max_dominated_count = 0;

        // Iterate through all vertices to find the one that dominates the most undominated vertices
        // This is inefficient. We should iterate only on the undominated vertices.
        // A more efficient approach would be to calculate the "score" for each vertex once
        // and update scores only for affected vertices when a vertex is added to D.
        // However, for simplicity and within the 5-minute constraint for potentially large graphs,
        // this full scan per iteration might be too slow.
        // Let's refine the greedy choice to be more efficient:
        // Pick the vertex `v` that, if added to the dominating set, would dominate
        // the maximum number of *currently undominated* vertices (including `v` itself if `v` is undominated).

        // Optimization: only consider vertices that are currently undominated or
        // neighbors of undominated vertices.
        // A simpler heuristic for "best_vertex" to reduce iteration:
        // Iterate through all vertices `i` from `0` to `num_vertices - 1`.
        // Calculate the score for each `i`:
        //  score = (1 if i is undominated else 0)
        //  for each neighbor `j` of `i`:
        //      if `j` is undominated:
        //          score += 1
        //  Select `i` with max score.

        for i in 0..num_vertices {
            let mut current_dominated_count = 0;
            let mut considered_for_domination: HashSet<usize> = HashSet::new();

            // If vertex `i` itself is undominated
            if undominated_vertices.contains(&i) {
                current_dominated_count += 1;
                considered_for_domination.insert(i);
            }

            // Check neighbors of `i`
            for &neighbor in &graph[i] {
                if undominated_vertices.contains(&neighbor) && !considered_for_domination.contains(&neighbor) {
                    current_dominated_count += 1;
                    considered_for_domination.insert(neighbor);
                }
            }

            if current_dominated_count > max_dominated_count {
                max_dominated_count = current_dominated_count;
                best_vertex = Some(i);
            }
        }

        let chosen_vertex = best_vertex.expect("Should always find a vertex to dominate");
        dominating_set.insert(chosen_vertex);

        // Remove chosen_vertex and its neighbors from undominated_vertices
        undominated_vertices.remove(&chosen_vertex);
        for &neighbor in &graph[chosen_vertex] {
            undominated_vertices.remove(&neighbor);
        }
    }

    // Output the result
    println!("{}", dominating_set.len());
    let mut sorted_dominating_set: Vec<usize> = dominating_set.into_iter().collect();
    sorted_dominating_set.sort_unstable();
    let output_line: Vec<String> = sorted_dominating_set.iter().map(|v| v.to_string()).collect();
    println!("{}", output_line.join(" "));
}
