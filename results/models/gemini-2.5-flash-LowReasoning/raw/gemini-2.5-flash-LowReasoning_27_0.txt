#include <iostream>
#include <vector>
#include <numeric> // For std::iota
#include <algorithm> // For std::sort

// Function to check if it's safe to assign color 'c' to vertex 'v'
bool is_safe(int v, int c, int num_vertices, const std::vector<std::vector<int>>& adj, const std::vector<int>& colors) {
    for (int neighbor : adj[v]) {
        if (colors[neighbor] == c) {
            return false;
        }
    }
    return true;
}

// Recursive utility function to solve graph coloring problem
bool solve_coloring(int v_idx, int num_vertices, int num_colors, const std::vector<std::vector<int>>& adj, std::vector<int>& colors) {
    // Base case: If all vertices are colored, then a solution exists
    if (v_idx == num_vertices) {
        return true;
    }

    // Consider current vertex 'v_idx' and try different colors
    for (int c = 0; c < num_colors; ++c) {
        if (is_safe(v_idx, c, num_vertices, adj, colors)) {
            colors[v_idx] = c;

            // Recur for the next vertex
            if (solve_coloring(v_idx + 1, num_vertices, num_colors, adj, colors)) {
                return true;
            }

            // If assigning color 'c' doesn't lead to a solution, then backtrack
            colors[v_idx] = -1; // Unassign color
        }
    }

    // If no color can be assigned to this vertex, return false
    return false;
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    int num_vertices, num_edges, num_colors;
    std::cin >> num_vertices >> num_edges >> num_colors;

    std::vector<std::vector<int>> adj(num_vertices);
    for (int i = 0; i < num_edges; ++i) {
        int u, v;
        std::cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    std::vector<int> colors(num_vertices, -1); // Initialize all vertices as uncolored (-1)

    if (solve_coloring(0, num_vertices, num_colors, adj, colors)) {
        for (int i = 0; i < num_vertices; ++i) {
            std::cout << colors[i] << (i == num_vertices - 1 ? "" : " ");
        }
        std::cout << std::endl;
    } else {
        // According to the problem statement, if no solution is found,
        // we might not need to print anything or print an error.
        // For now, an empty line or specific error message could be used
        // but the prompt implies outputting colors on success.
        // Let's print "No solution found" for clarity during development.
        std::cerr << "No solution found with " << num_colors << " colors." << std::endl;
        return 1; // Indicate failure
    }

    return 0;
}
