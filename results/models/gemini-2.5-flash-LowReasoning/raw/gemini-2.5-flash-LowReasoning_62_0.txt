/**
 * Simulates fire spread on the grid and returns the list of towns that burned.
 * @param grid A 2D number array where 0=forest, 1=water/rock, 2=cut firebreak.
 * @param ignitions Starting fire cells.
 * @param allTowns List of all towns.
 * @param currentFirebreaks The firebreak cells currently applied to the grid (used to mark grid cells as 2)
 * @returns An array of towns that burned.
 */
function simulateFire(baseGrid: number[][], ignitions: [number, number][], allTowns: Town[], currentFirebreaks: [number, number][]): Town[] {
    const height = baseGrid.length;
    const width = baseGrid[0].length;

    // Create a mutable grid for simulation, applying firebreaks
    const simulationGrid: number[][] = JSON.parse(JSON.stringify(baseGrid)); // Deep copy
    currentFirebreaks.forEach(([x, y]) => {
        if (y >= 0 && y < height && x >= 0 && x < width) {
            simulationGrid[y][x] = 2; // Mark as cut
        }
    });

    const fireQueue: [number, number][] = [];
    const burned: boolean[][] = Array.from({ length: height }, () =>
        Array.from({ length: width }, () => false)
    );

    // Initialize ignitions
    for (const [startX, startY] of ignitions) {
        if (startY >= 0 && startY < height && startX >= 0 && startX < width && simulationGrid[startY][startX] === 0 && !burned[startY][startX]) {
            fireQueue.push([startX, startY]);
            burned[startY][startX] = true;
        }
    }

    const dx = [-1, 1, 0, 0]; // Left, Right
    const dy = [0, 0, -1, 1]; // Up, Down

    let head = 0;
    while (head < fireQueue.length) {
        const [x, y] = fireQueue[head++];

        for (let i = 0; i < 4; i++) {
            const nx = x + dx[i];
            const ny = y + dy[i];

            // Check bounds and if it's a burnable, unburned forest cell
            if (
                ny >= 0 && ny < height &&
                nx >= 0 && nx < width &&
                simulationGrid[ny][nx] === 0 && // Is forest
                !burned[ny][nx]
            ) {
                burned[ny][nx] = true;
                fireQueue.push([nx, ny]);
            }
        }
    }

    const burnedTowns: Town[] = [];
    for (const town of allTowns) {
        if (town.y >= 0 && town.y < height && town.x >= 0 && town.x < width && burned[town.y][town.x]) {
            burnedTowns.push(town);
        }
    }

    return burnedTowns;
}
