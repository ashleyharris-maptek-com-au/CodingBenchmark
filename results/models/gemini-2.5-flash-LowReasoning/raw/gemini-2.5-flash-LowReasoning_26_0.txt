use linfa::prelude::*;
use linfa_clustering::KMeans;
use ndarray::{array, Array2};

fn main() {
    println!("Starting 3D point clustering...");

    // 1. Define sample 3D points
    let N_CLUSTERS = 3; // The desired number of clusters

    let observations = array![
        // Cluster 0
        [1.0, 1.0, 1.0],
        [1.1, 1.2, 1.0],
        [1.0, 1.1, 1.2],
        // Cluster 1
        [10.0, 10.0, 10.0],
        [10.1, 10.2, 10.0],
        [10.0, 10.1, 10.2],
        // Cluster 2
        [20.0, 20.0, 20.0],
        [20.1, 20.2, 20.0],
        [20.0, 20.1, 20.2],
    ];

    // 2. Create a dataset from the observations
    let dataset = Dataset::from(observations);

    // 3. Configure and fit the KMeans model
    let model = KMeans::params(N_CLUSTERS)
        .max_n_iterations(100) // Maximum number of iterations for the algorithm
        .tolerance(1e-4)      // Convergence tolerance
        .fit(&dataset)
        .expect("KMeans fitting failed");

    // 4. Predict cluster assignments
    let predictions = model.predict(&dataset);

    // 5. Print the cluster assignments
    println!("Cluster assignments:");
    for (i, point) in dataset.records().outer_iter().enumerate() {
        println!("Point {:?} assigned to Cluster {}", point.to_vec(), predictions[i]);
    }
}
