#version 450

layout(local_size_x = 256, local_size_y = 1, local_size_z = 1) in;

struct Particle {
    vec4 position; // xyz for position, w for mass
    vec4 velocity; // xyz for velocity, w unused
};

layout(std430, binding = 0) buffer ParticlesIn {
    Particle particles_in[];
};

layout(std430, binding = 1) buffer ParticlesOut {
    Particle particles_out[];
};

uniform float deltaTime;
uniform float G; // Gravitational constant
uniform uint numParticles;

void main() {
    uint i = gl_GlobalInvocationID.x;

    if (i >= numParticles) {
        return;
    }

    Particle p = particles_in[i];
    vec3 current_pos = p.position.xyz;
    float mass_i = p.position.w;
    vec3 current_vel = p.velocity.xyz;

    vec3 acceleration = vec3(0.0);

    for (uint j = 0; j < numParticles; ++j) {
        if (i == j) {
            continue;
        }

        Particle other_p = particles_in[j];
        vec3 other_pos = other_p.position.xyz;
        float mass_j = other_p.position.w;

        vec3 direction = other_pos - current_pos;
        float distSq = dot(direction, direction);
        float dist = sqrt(distSq);

        // Avoid division by zero and excessively large forces at very close distances
        // A small epsilon can be added to distSq or dist to soften the interaction
        float softening = 0.1; // Adjust as needed
        distSq = distSq + softening * softening;
        dist = sqrt(distSq);

        // F = G * (m1 * m2) / r^2
        // a = F / m1 = G * m2 / r^2
        acceleration += (G * mass_j / (distSq * dist)) * direction;
    }

    // Update velocity
    vec3 new_vel = current_vel + acceleration * deltaTime;

    // Update position (using new velocity for more accurate integration, e.g., Leapfrog-like)
    vec3 new_pos = current_pos + new_vel * deltaTime;

    particles_out[i].position = vec4(new_pos, mass_i);
    particles_out[i].velocity = vec4(new_vel, 0.0);
}
