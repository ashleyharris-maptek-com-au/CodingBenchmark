interface Task {
    id: string;
    duration: number; // Time in minutes
    deps: string[]; // Array of task IDs that must complete before this task can start
    deadline: number; // Time in minutes from start of day
    weight: number; // Importance of the task
}

interface InputData {
    machineCount: number;
    tasks: Task[];
}

interface OutputOrder {
    order: string[]; // Array of task IDs in topological order
}

declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data: InputData = JSON.parse(input);

function topologicalSort(tasks: Task[]): string[] {
    const taskMap = new Map<string, Task>();
    for (const task of tasks) {
        taskMap.set(task.id, task);
    }

    const inDegree = new Map<string, number>();
    const adjList = new Map<string, string[]>(); // Graph where key depends on value

    for (const task of tasks) {
        inDegree.set(task.id, 0);
        adjList.set(task.id, []);
    }

    // Build graph and calculate in-degrees
    for (const task of tasks) {
        for (const depId of task.deps) {
            // depId -> task.id means task.id depends on depId
            // So, add task.id to the adjacency list of depId
            if (!adjList.has(depId)) {
                // This case handles a dependency that might not be in the tasks list
                // Although problem statement implies all deps are valid task IDs,
                // better to be safe or assume valid input
                // For now, let's assume all deps are valid task IDs present in 'tasks'
                // If not, this might cause issues or indicate malformed input.
                // For a benchmark, we assume valid inputs for simplicity.
                adjList.set(depId, []);
            }
            adjList.get(depId)?.push(task.id);
            inDegree.set(task.id, (inDegree.get(task.id) || 0) + 1);
        }
    }

    const queue: string[] = [];
    for (const task of tasks) {
        if (inDegree.get(task.id) === 0) {
            queue.push(task.id);
        }
    }

    const result: string[] = [];
    let head = 0; // Manual queue head for performance in Node.js array

    while (head < queue.length) {
        const uId = queue[head++];
        result.push(uId);

        for (const vId of adjList.get(uId) || []) {
            inDegree.set(vId, (inDegree.get(vId) || 0) - 1);
            if (inDegree.get(vId) === 0) {
                queue.push(vId);
            }
        }
    }

    // If there's a cycle, result.length will be less than tasks.length
    // The problem statement implies valid inputs, so we don't handle cycles explicitly.
    // If a cycle were present, this would be an invalid schedule.
    // For a benchmark, a valid topological sort is expected for valid inputs.
    return result;
}

const orderedTaskIds = topologicalSort(data.tasks);

const output: OutputOrder = {
    order: orderedTaskIds
};

console.log(JSON.stringify(output));
