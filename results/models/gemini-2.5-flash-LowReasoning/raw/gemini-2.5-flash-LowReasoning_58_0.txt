const fs = require('fs');

/**
 * Parses the grid strings into a 2D array for pathfinding.
 * Converts '#' (blocked) to 1 and '.' (open) to 0.
 *
 * @param {string[]} gridStrings - Array of strings representing the grid.
 * @returns {{gridData: number[][], width: number, height: number}} - Parsed grid data and dimensions.
 */
function parseGrid(gridStrings) {
    const height = gridStrings.length;
    if (height === 0) {
        return { gridData: [], width: 0, height: 0 };
    }
    const width = gridStrings[0].length;
    const gridData = Array(height).fill(0).map(() => Array(width).fill(0));

    for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
            gridData[y][x] = gridStrings[y][x] === '#' ? 1 : 0; // 1 for blocked, 0 for open
        }
    }
    return { gridData, width, height };
}

/**
 * Performs a Breadth-First Search (BFS) to find the shortest path distance
 * between a start point and a target point on the grid.
 *
 * @param {{x: number, y: number}} startCoords - Starting coordinates.
 * @param {{x: number, y: number}} targetCoords - Target coordinates.
 * @param {number[][]} grid - 2D array representation of the grid (0 for open, 1 for blocked).
 * @param {number} width - Width of the grid.
 * @param {number} height - Height of the grid.
 * @returns {number} - The shortest distance, or Infinity if no path is found.
 */
function bfs(startCoords, targetCoords, grid, width, height) {
    if (startCoords.x === targetCoords.x && startCoords.y === targetCoords.y) {
        return 0;
    }

    // Check if start or end is out of bounds or blocked
    if (startCoords.x < 0 || startCoords.x >= width || startCoords.y < 0 || startCoords.y >= height ||
        targetCoords.x < 0 || targetCoords.x >= width || targetCoords.y < 0 || targetCoords.y >= height ||
        grid[startCoords.y][startCoords.x] === 1 || grid[targetCoords.y][targetCoords.x] === 1) {
        return Infinity;
    }

    const queue = [{ x: startCoords.x, y: startCoords.y, dist: 0 }];
    const visited = Array(height).fill(false).map(() => Array(width).fill(false));
    visited[startCoords.y][startCoords.x] = true;

    const dx = [0, 0, 1, -1]; // Possible moves: up, down, right, left
    const dy = [1, -1, 0, 0];

    let head = 0;
    while (head < queue.length) {
        const { x, y, dist } = queue[head++];

        for (let i = 0; i < 4; i++) {
            const nx = x + dx[i];
            const ny = y + dy[i];

            if (nx >= 0 && nx < width && ny >= 0 && ny < height && grid[ny][nx] === 0 && !visited[ny][nx]) {
                if (nx === targetCoords.x && ny === targetCoords.y) {
                    return dist + 1;
                }
                visited[ny][nx] = true;
                queue.push({ x: nx, y: ny, dist: dist + 1 });
            }
        }
    }

    return Infinity; // No path found
}

/**
 * Main function to solve the robot job assignment problem.
 * Reads input from stdin, processes it, and prints JSON output to stdout.
 */
function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);

    const { grid: gridStrings, robots, jobs } = data;
    const { gridData, width, height } = parseGrid(gridStrings);

    // Map robot and job IDs to their details for easy lookup
    const robotInitialPositions = new Map(robots.map(r => [r.id, { x: r.x, y: r.y }]));
    const jobMap = new Map(jobs.map(j => [j.id, { x: j.x, y: j.y, priority: j.priority }]));

    // Initialize data structures for assignments
    // robotAssignments: Map<robotId, jobId[]>
    // currentRobotPositions: Map<robotId, {x, y}> - tracks the last assigned job's location or initial robot position
    const robotAssignments = new Map();
    const currentRobotPositions = new Map();

    robots.forEach(r => {
        robotAssignments.set(r.id, []);
        currentRobotPositions.set(r.id, { x: r.x, y: r.y });
    });

    const unassignedJobIds = new Set(jobs.map(j => j.id));

    // Greedy assignment loop: In each iteration, assign the "best" (robot, job) pair
    while (unassignedJobIds.size > 0) {
        let minCost = Infinity;
        let bestRobotId = null;
        let bestJobId = null;
        let bestJobCoords = null;

        for (const robot of robots) {
            const robotId = robot.id;
            const robotCurrentPos = currentRobotPositions.get(robotId);

            for (const jobId of unassignedJobIds) {
                const job = jobMap.get(jobId);
                const dist = bfs(robotCurrentPos, { x: job.x, y: job.y }, gridData, width, height);

                // For the "deterministic greedy reference", we simply pick the assignment
                // that results in the shortest immediate travel distance.
                // Priority is ignored as per the problem description's lack of guidance on its use.
                if (dist !== Infinity && dist < minCost) {
                    minCost = dist;
                    bestRobotId = robotId;
                    bestJobId = jobId;
                    bestJobCoords = { x: job.x, y: job.y };
                }
            }
        }

        // If no reachable jobs were found, something is wrong with the input or grid.
        // Or if the loop finishes due to all jobs being assigned, bestRobotId will remain null.
        if (bestRobotId === null) {
            // This scenario should ideally not happen if all jobs are reachable.
            // If it does, it implies an unsolvable state or an error in logic.
            // For robust error handling, one might throw an error or log a warning.
            // Per instructions: "Do not print logs or explanatory text."
            break;
        }

        // Assign the job and update robot's current position
        robotAssignments.get(bestRobotId).push(bestJobId);
        currentRobotPositions.set(bestRobotId, bestJobCoords);
        unassignedJobIds.delete(bestJobId);
    }

    // Convert the Map of assignments to the required array format
    // The problem states: "The array index is the robot id."
    // This implies robot IDs are 0-indexed and sequential.
    // We'll find the maximum robot ID to correctly size the output array.
    const maxRobotId = robots.length > 0 ? Math.max(...robots.map(r => r.id)) : -1;
    const finalAssignmentsArray = Array(maxRobotId + 1).fill(null).map(() => []);

    for (const [robotId, assignments] of robotAssignments.entries()) {
        if (robotId >= 0 && robotId <= maxRobotId) { // Ensure robotId is a valid array index
            finalAssignmentsArray[robotId] = assignments;
        }
    }

    // Prepare and print the final JSON output
    const output = {
        assignments: finalAssignmentsArray
    };

    console.log(JSON.stringify(output));
}

// Execute the main solve function
solve();
