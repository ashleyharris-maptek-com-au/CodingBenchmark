#include "NVector.h"
#include <stdexcept> // For std::out_of_range
#include <algorithm> // For std::transform, std::clamp

// Constructor with dimensions and initial value
NVector::NVector(int dimensions, int initial_value) {
    if (dimensions <= 0) {
        throw std::invalid_argument("Dimensions must be positive.");
    }
    coords_.resize(dimensions, initial_value);
}

// Constructor with a vector of coordinates
NVector::NVector(const std::vector<int>& coords) : coords_(coords) {
    if (coords.empty()) {
        throw std::invalid_argument("Coordinates vector cannot be empty.");
    }
}

// Returns the number of dimensions
int NVector::getDimension() const {
    return coords_.size();
}

// Returns the coordinate at a specific dimension
int NVector::getCoord(int dim) const {
    if (dim < 0 || dim >= coords_.size()) {
        throw std::out_of_range("Dimension index out of range.");
    }
    return coords_[dim];
}

// Returns the underlying coordinates vector
const std::vector<int>& NVector::getCoords() const {
    return coords_;
}

// Equality operator
bool NVector::operator==(const NVector& other) const {
    if (coords_.size() != other.coords_.size()) {
        return false;
    }
    return coords_ == other.coords_;
}

// Inequality operator
bool NVector::operator!=(const NVector& other) const {
    return !(*this == other);
}

// Vector addition
NVector NVector::operator+(const NVector& other) const {
    if (coords_.size() != other.coords_.size()) {
        throw std::invalid_argument("Vectors must have the same number of dimensions for addition.");
    }
    NVector result(coords_.size());
    for (size_t i = 0; i < coords_.size(); ++i) {
        result.coords_[i] = coords_[i] + other.coords_[i];
    }
    return result;
}

// Vector subtraction
NVector NVector::operator-(const NVector& other) const {
    if (coords_.size() != other.coords_.size()) {
        throw std::invalid_argument("Vectors must have the same number of dimensions for subtraction.");
    }
    NVector result(coords_.size());
    for (size_t i = 0; i < coords_.size(); ++i) {
        result.coords_[i] = coords_[i] - other.coords_[i];
    }
    return result;
}

// Clamps the vector's coordinates within specified min and max bounds
NVector NVector::clamp(const NVector& min_coords, const NVector& max_coords) const {
    if (coords_.size() != min_coords.coords_.size() || coords_.size() != max_coords.coords_.size()) {
        throw std::invalid_argument("Clamp vectors must have the same number of dimensions.");
    }
    NVector result(coords_.size());
    for (size_t i = 0; i < coords_.size(); ++i) {
        result.coords_[i] = std::clamp(coords_[i], min_coords.coords_[i], max_coords.coords_[i]);
    }
    return result;
}

// Converts the vector to a string representation (e.g., "(1, 2, 3)")
std::string NVector::toString() const {
    std::string s = "(";
    for (size_t i = 0; i < coords_.size(); ++i) {
        s += std::to_string(coords_[i]);
        if (i < coords_.size() - 1) {
            s += ", ";
        }
    }
    s += ")";
    return s;
}

// Friend function for ostream output
std::ostream& operator<<(std::ostream& os, const NVector& vec) {
    os << vec.toString();
    return os;
}
