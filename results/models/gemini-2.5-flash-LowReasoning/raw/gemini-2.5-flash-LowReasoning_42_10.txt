; SPIR-V
; Version: 1.0
; Generator: LLVM-SPIRV; 0
; Capabilities: Geometry Shader
; Memory Model: GLSL450 DLT
; Entry Point: %main
; Execution Mode: %main InputPoints
; Execution Mode: %main OutputTriangleStrip
; Execution Mode: %main OutputVertices 4

OpCapability Geometry
OpCapability ShaderViewportIndex
OpCapability ShaderLayer

%1 = OpExtInstImport "GLSL.std.450"

OpMemoryModel Logical GLSL450

OpEntryPoint Geometry %main "main" %gl_in_Position %gl_Position %gl_PointSize %gl_PrimitiveIDIn %gl_InvocationID %gl_Layer %gl_ViewportIndex

OpSource GLSL 450

; All types and constants
%void = OpTypeVoid
%void_fn = OpTypeFunction %void
%float = OpTypeFloat 32
%v4float = OpTypeVector %float 4
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0

%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1

%uint_1 = OpConstant %uint 1 ; For array size for gl_in

%_ptr_Output_v4float = OpTypePointer Output %v4float
%_ptr_Output_float = OpTypePointer Output %float
%_ptr_Output_int = OpTypePointer Output %int

; Input variables - Geometry shader inputs are arrays
%_arr_v4float_uint_1 = OpTypeArray %v4float %uint_1
%_ptr_Input_arr_v4float_uint_1 = OpTypePointer Input %_arr_v4float_uint_1
%_ptr_Input_int = OpTypePointer Input %int

%gl_in_Position = OpVariable %_ptr_Input_arr_v4float_uint_1 Input ; Input position for a point
%gl_PrimitiveIDIn = OpVariable %_ptr_Input_int Input
%gl_InvocationID = OpVariable %_ptr_Input_int Input

; Output variables
%gl_Position = OpVariable %_ptr_Output_v4float Output
%gl_PointSize = OpVariable %_ptr_Output_float Output
%gl_Layer = OpVariable %_ptr_Output_int Output
%gl_ViewportIndex = OpVariable %_ptr_Output_int Output

; Built-in decorations for input
OpDecorate %gl_in_Position BuiltIn Position
OpDecorate %gl_PrimitiveIDIn BuiltIn PrimitiveID
OpDecorate %gl_InvocationID BuiltIn InvocationID

; Built-in decorations for output
OpDecorate %gl_Position BuiltIn Position
OpDecorate %gl_PointSize BuiltIn PointSize
OpDecorate %gl_Layer BuiltIn Layer
OpDecorate %gl_ViewportIndex BuiltIn ViewportIndex

; Main function
%main = OpFunction %void None %void_fn
%entry = OpLabel
    %in_pos_0_ptr = OpAccessChain %_ptr_Input_v4float %gl_in_Position %int_0
    %pos_val = OpLoad %v4float %in_pos_0_ptr

    ; For demonstration, we'll just output the input position for 4 vertices to make a quad.
    ; In a real scenario, these positions would be transformed to form the quad.
    ; This is a very basic passthrough/template.
    OpStore %gl_Position %pos_val
    OpEmitVertex

    OpStore %gl_Position %pos_val ; Example for a second vertex
    OpEmitVertex

    OpStore %gl_Position %pos_val ; Example for a third vertex
    OpEmitVertex

    OpStore %gl_Position %pos_val ; Example for a fourth vertex
    OpEmitVertex

OpEndPrimitive
OpReturn
OpFunctionEnd
