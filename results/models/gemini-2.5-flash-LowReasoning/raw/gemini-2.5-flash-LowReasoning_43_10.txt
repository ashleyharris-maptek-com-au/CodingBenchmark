// Basic HLSL pixel shader for Vulkan
// This shader will output a red color.

// Struct for vertex shader output (pixel shader input)
struct PSInput
{
    float4 Position : SV_POSITION;
    float2 TexCoord : TEXCOORD0; // Example: texture coordinates
};

// Main pixel shader function
float4 main(PSInput input) : SV_TARGET
{
    // Output a solid red color
    return float4(1.0f, 0.0f, 0.0f, 1.0f);
}
