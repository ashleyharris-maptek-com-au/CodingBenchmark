use rand::seq::SliceRandom;
use rand::thread_rng;

// Example: Adjacency list representation
pub type AdjacencyList = Vec<Vec<usize>>;

/// Computes the number of edges between `set_a` and `set_b`.
fn calculate_cut_size(adj_list: &AdjacencyList, set_a: &[usize], _set_b: &[usize]) -> usize {
    let mut cut_size = 0;
    let set_a_nodes: std::collections::HashSet<usize> = set_a.iter().cloned().collect();

    for &u in set_a.iter() {
        if let Some(neighbors) = adj_list.get(u) {
            for &v in neighbors.iter() {
                // If v is in set_b, then this edge contributes to the cut
                if !set_a_nodes.contains(&v) {
                    cut_size += 1;
                }
            }
        }
    }
    cut_size
}



pub fn graph_bisection(adj_list: &AdjacencyList) -> (Vec<usize>, Vec<usize>, usize) {
    let n = adj_list.len();
    if n == 0 {
        return (vec![], vec![], 0);
    }

    // Step 1: Initialize two sets, A and B, by randomly distributing vertices.
    let mut vertices: Vec<usize> = (0..n).collect();
    vertices.shuffle(&mut thread_rng()); // Randomly shuffle vertices

    let mut set_a: Vec<usize> = Vec::with_capacity(n / 2 + (n % 2));
    let mut set_b: Vec<usize> = Vec::with_capacity(n / 2 + (n % 2));

    // Distribute vertices to maintain balance as much as possible
    for (i, &vertex) in vertices.iter().enumerate() {
        if i < n / 2 {
            set_a.push(vertex);
        } else {
            set_b.push(vertex);
        }
    }

    // Ensure initial balance (for odd number of vertices, one set will have one more)
    let target_size = n / 2;
    let current_a_len = set_a.len();
    let current_b_len = set_b.len();

    // Adjust if necessary to meet strict balance for initial random distribution
    if current_a_len > target_size + (n % 2) {
        // Move from A to B
        while set_a.len() > target_size + (n % 2) {
            if let Some(v) = set_a.pop() {
                set_b.push(v);
            }
        }
    } else if current_b_len > target_size + (n % 2) {
        // Move from B to A
        while set_b.len() > target_size + (n % 2) {
            if let Some(v) = set_b.pop() {
                set_a.push(v);
            }
        }
    }


    // Step 2 & 3: Iteratively move vertices between A and B if it improves the cut size and maintains the balance constraint.
    // Repeat until no further improvement can be made.
    let mut current_cut_size = calculate_cut_size(adj_list, &set_a, &set_b);
    let mut improved = true;

    while improved {
        improved = false;
        let mut best_gain = 0;
        let mut best_swap: Option<(usize, usize)> = None; // (vertex_in_A, vertex_in_B)

        // Try swapping one vertex from A with one from B
        for &u_idx in set_a.iter() {
            for &v_idx in set_b.iter() {
                // Simulate swap: u_idx moves to B, v_idx moves to A
                // Current sets: set_a, set_b
                // Proposed sets: (set_a - {u_idx}) U {v_idx}, (set_b - {v_idx}) U {u_idx}
                // The balance is maintained by definition of a swap.

                let mut temp_set_a: Vec<usize> = set_a.iter().filter(|&&x| x != u_idx).cloned().collect();
                temp_set_a.push(v_idx);
                let mut temp_set_b: Vec<usize> = set_b.iter().filter(|&&x| x != v_idx).cloned().collect();
                temp_set_b.push(u_idx);

                let new_cut_size = calculate_cut_size(adj_list, &temp_set_a, &temp_set_b);
                let gain = (current_cut_size as isize) - (new_cut_size as isize);

                if gain > best_gain {
                    best_gain = gain;
                    best_swap = Some((u_idx, v_idx));
                }
            }
        }

        if let Some((u_to_b, v_to_a)) = best_swap {
            if best_gain > 0 {
                // Perform the swap
                let mut new_set_a = Vec::with_capacity(set_a.len());
                let mut new_set_b = Vec::with_capacity(set_b.len());

                for &node in set_a.iter() {
                    if node != u_to_b {
                        new_set_a.push(node);
                    }
                }
                new_set_a.push(v_to_a);

                for &node in set_b.iter() {
                    if node != v_to_a {
                        new_set_b.push(node);
                    }
                }
                new_set_b.push(u_to_b);

                set_a = new_set_a;
                set_b = new_set_b;
                current_cut_size -= best_gain as usize;
                improved = true;
            }
        }
    }

    set_a.sort_unstable();
    set_b.sort_unstable();

    (set_a, set_b, current_cut_size)
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_calculate_cut_size_simple() {
        // Graph: 0--1, 0--2, 1--3, 2--3
        let adj_list = vec![
            vec![1, 2],
            vec![0, 3],
            vec![0, 3],
            vec![1, 2],
        ];

        // Sets: A={0,1}, B={2,3}. Expected cut: (0-2), (1-3) => 2
        let set_a = vec![0, 1];
        let set_b = vec![2, 3];
        assert_eq!(calculate_cut_size(&adj_list, &set_a, &set_b), 2);

        // Sets: A={0,3}, B={1,2}. Expected cut: (0-1), (0-2), (3-1), (3-2) => 4 (Incorrect bisection example)
        let set_a_bad = vec![0, 3];
        let set_b_bad = vec![1, 2];
        assert_eq!(calculate_cut_size(&adj_list, &set_a_bad, &set_b_bad), 4);
    }



    #[test]
    fn test_graph_bisection_empty_graph() {
        let adj_list: AdjacencyList = vec![];
        let (_set_a, _set_b, cut_size) = graph_bisection(&adj_list);
        assert!(_set_a.is_empty());
        assert!(_set_b.is_empty());
        assert_eq!(cut_size, 0);
    }

    #[test]
    fn test_graph_bisection_two_nodes_no_edge() {
        let adj_list: AdjacencyList = vec![vec![], vec![]]; // Nodes 0 and 1, no edges
        let (_set_a, _set_b, cut_size) = graph_bisection(&adj_list);
        assert!((_set_a == vec![0] && _set_b == vec![1]) || (_set_a == vec![1] && _set_b == vec![0]));
        assert_eq!(cut_size, 0);
    }

    #[test]
    fn test_graph_bisection_two_nodes_with_edge() {
        let adj_list: AdjacencyList = vec![vec![1], vec![0]]; // Nodes 0 and 1, edge between them
        let (_set_a, _set_b, cut_size) = graph_bisection(&adj_list);
        assert!((_set_a == vec![0] && _set_b == vec![1]) || (_set_a == vec![1] && _set_b == vec![0]));
        assert_eq!(cut_size, 1);
    }

    #[test]
    fn test_graph_bisection_simple_path_graph() {
        // Graph: 0--1--2--3 (4 nodes)
        let adj_list = vec![
            vec![1],          // 0
            vec![0, 2],       // 1
            vec![1, 3],       // 2
            vec![2],          // 3
        ];
        // Optimal bisection: {0,1}, {2,3} cut_size = 1 (edge 1-2)
        let (_set_a, _set_b, cut_size) = graph_bisection(&adj_list);
        assert_eq!(cut_size, 1);
        // assert!(is_balanced(&set_a, &set_b, adj_list.len()));
        assert!((_set_a == vec![0, 1] && _set_b == vec![2, 3]) || (_set_a == vec![2, 3] && _set_b == vec![0, 1]));
    }

    #[test]
    fn test_graph_bisection_disconnected_graph() {
        // Graph: 0--1  2--3 (4 nodes, two disconnected components)
        let adj_list = vec![
            vec![1],    // 0
            vec![0],    // 1
            vec![3],    // 2
            vec![2],    // 3
        ];
        // Optimal bisection: {0,2}, {1,3} cut_size = 0 (or {0,3}, {1,2})
        let (_set_a, _set_b, cut_size) = graph_bisection(&adj_list);
        assert_eq!(cut_size, 0);
        // assert!(is_balanced(&set_a, &set_b, adj_list.len()));
    }

    #[test]
    fn test_graph_bisection_star_graph() {
        // Star graph: 0 is center, connected to 1, 2, 3 (4 nodes)
        let adj_list = vec![
            vec![1, 2, 3], // 0
            vec![0],       // 1
            vec![0],       // 2
            vec![0],       // 3
        ];
        // Bisection: {0, X}, {Y, Z} where X,Y,Z are leaves
        // e.g., {0,1}, {2,3} -> cut_size = 2 (edges 0-2, 0-3)
        // This greedy approach might not find optimal for all cases, but should be reasonable.
        let (_set_a, _set_b, cut_size) = graph_bisection(&adj_list);
        assert_eq!(cut_size, 2); // Optimal bisection should be 2 for a star graph with 4 nodes
        // assert!(is_balanced(&set_a, &set_b, adj_list.len()));
    }
}
