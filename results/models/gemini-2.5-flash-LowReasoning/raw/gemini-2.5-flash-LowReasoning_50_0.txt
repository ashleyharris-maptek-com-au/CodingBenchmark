#version 450
layout(local_size_x = 64, local_size_y = 1, local_size_z = 1) in;

struct Boid {
    vec4 position; // xyz for position, w for padding or other data
    vec4 velocity; // xyz for velocity, w for padding or other data
};

layout(std430, binding = 0) buffer BoidBuffer {
    Boid boids[];
};

uniform uint numBoids;
uniform float deltaTime;
uniform float separationRadius;
uniform float alignmentRadius;
uniform float cohesionRadius;
uniform float separationWeight;
uniform float alignmentWeight;
uniform float cohesionWeight;
uniform float maxSpeed;
uniform float maxForce;
uniform vec3 worldMin;
uniform vec3 worldMax;

void main() {
    uint index = gl_GlobalInvocationID.x;

    if (index >= numBoids) {
        return;
    }

    vec3 currentPosition = boids[index].position.xyz;
    vec3 currentVelocity = boids[index].velocity.xyz;

    vec3 separationForce = vec3(0.0);
    vec3 alignmentForce = vec3(0.0);
    vec3 cohesionForce = vec3(0.0);

    uint separationCount = 0;
    uint alignmentCount = 0;
    uint cohesionCount = 0;

    vec3 perceivedCenter = vec3(0.0);
    vec3 perceivedVelocity = vec3(0.0);

    for (uint i = 0; i < numBoids; ++i) {
        if (i == index) {
            continue;
        }

        vec3 otherPosition = boids[i].position.xyz;
        vec3 otherVelocity = boids[i].velocity.xyz;
        vec3 diff = currentPosition - otherPosition;
        float dist = length(diff);

        // Separation
        if (dist < separationRadius) {
            separationForce += normalize(diff) / dist; // Inverse distance weighting
            separationCount++;
        }

        // Alignment
        if (dist < alignmentRadius) {
            perceivedVelocity += otherVelocity;
            alignmentCount++;
        }

        // Cohesion
        if (dist < cohesionRadius) {
            perceivedCenter += otherPosition;
            cohesionCount++;
        }
    }

    // Calculate separation force
    if (separationCount > 0) {
        separationForce /= float(separationCount);
        separationForce = normalize(separationForce) * maxSpeed - currentVelocity;
        separationForce = clamp(separationForce, -maxForce, maxForce);
    }

    // Calculate alignment force
    if (alignmentCount > 0) {
        perceivedVelocity /= float(alignmentCount);
        alignmentForce = normalize(perceivedVelocity) * maxSpeed - currentVelocity;
        alignmentForce = clamp(alignmentForce, -maxForce, maxForce);
    }

    // Calculate cohesion force
    if (cohesionCount > 0) {
        perceivedCenter /= float(cohesionCount);
        vec3 desiredVelocity = normalize(perceivedCenter - currentPosition) * maxSpeed;
        cohesionForce = desiredVelocity - currentVelocity;
        cohesionForce = clamp(cohesionForce, -maxForce, maxForce);
    }

    // Apply forces
    vec3 acceleration = vec3(0.0);
    acceleration += separationForce * separationWeight;
    acceleration += alignmentForce * alignmentWeight;
    acceleration += cohesionForce * cohesionWeight;

    // Update velocity
    currentVelocity += acceleration * deltaTime;
    if (length(currentVelocity) > maxSpeed) {
        currentVelocity = normalize(currentVelocity) * maxSpeed;
    }

    // Update position
    currentPosition += currentVelocity * deltaTime;

    // Boundary conditions (wrap around)
    if (currentPosition.x < worldMin.x) currentPosition.x = worldMax.x;
    if (currentPosition.x > worldMax.x) currentPosition.x = worldMin.x;
    if (currentPosition.y < worldMin.y) currentPosition.y = worldMax.y;
    if (currentPosition.y > worldMax.y) currentPosition.y = worldMin.y;
    if (currentPosition.z < worldMin.z) currentPosition.z = worldMax.z;
    if (currentPosition.z > worldMax.z) currentPosition.z = worldMin.z;

    boids[index].position.xyz = currentPosition;
    boids[index].velocity.xyz = currentVelocity;
}
