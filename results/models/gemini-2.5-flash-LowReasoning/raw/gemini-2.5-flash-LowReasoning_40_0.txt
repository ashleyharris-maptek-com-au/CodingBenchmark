#include <iostream>
#include <vector>
#include <string>

#include "ortools/linear_solver/linear_solver.h"

namespace operations_research {

void SolveKnapsackProblem() {
    // 1. Create the solver.
    // Using "CP_SAT" for integer problems is generally recommended.
    std::unique_ptr<MPSolver> solver(MPSolver::CreateSolver("CP_SAT"));
    if (!solver) {
        std::cerr << "Could not create solver CP_SAT" << std::endl;
        return;
    }

    // 2. Define the problem data.
    const std::vector<double> weights = {10, 20, 30};
    const std::vector<double> values = {60, 100, 120};
    const double knapsack_capacity = 50;
    const int num_items = weights.size();

    // 3. Create the variables.
    // x[i] is a binary variable, 1 if item i is taken, 0 otherwise.
    std::vector<MPVariable*> x(num_items);
    for (int i = 0; i < num_items; ++i) {
        x[i] = solver->MakeBoolVar("x_" + std::to_string(i));
    }

    // 4. Define the constraints.
    // The total weight of selected items must not exceed knapsack_capacity.
    MPConstraint* capacity_constraint = solver->MakeRowConstraint(0.0, knapsack_capacity, "capacity_constraint");
    for (int i = 0; i < num_items; ++i) {
        capacity_constraint->SetCoefficient(x[i], weights[i]);
    }

    // 5. Define the objective function.
    // Maximize the total value of selected items.
    MPObjective* objective = solver->MutableObjective();
    for (int i = 0; i < num_items; ++i) {
        objective->SetCoefficient(x[i], values[i]);
    }
    objective->SetMaximization();

    // 6. Solve the problem.
    const MPSolver::ResultStatus status = solver->Solve();

    // 7. Process the results.
    if (status == MPSolver::OPTIMAL || status == MPSolver::FEASIBLE) {
        std::cout << "Solution:" << std::endl;
        std::cout << "Optimal objective value = " << objective->Value() << std::endl;
        double total_weight = 0;
        std::cout << "Items selected:" << std::endl;
        for (int i = 0; i < num_items; ++i) {
            if (x[i]->solution_value() > 0.5) { // Using 0.5 threshold for binary variables
                std::cout << "  Item " << i << " (Weight: " << weights[i] << ", Value: " << values[i] << ")" << std::endl;
                total_weight += weights[i];
            }
        }
        std::cout << "Total weight = " << total_weight << std::endl;
    } else {
        std::cout << "The problem does not have an optimal or feasible solution." << std::endl;
    }
}

}  // namespace operations_research

int main(int argc, char** argv) {
    operations_research::SolveKnapsackProblem();
    return 0;
}
