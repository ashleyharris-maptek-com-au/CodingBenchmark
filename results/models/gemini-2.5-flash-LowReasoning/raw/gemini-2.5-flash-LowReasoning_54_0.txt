import numpy as np
from utils import PIDController
from ils_sensor import ILSSensor
from aircraft_model import AircraftModel
from environment import Environment

class Autopilot:
    """
    Manages the autoland process, including ILS tracking, flare, and rollout.
    Uses PID controllers for guidance.
    """
    def __init__(self, aircraft: AircraftModel, ils_sensor: ILSSensor, environment: Environment):
        self.aircraft = aircraft
        self.ils_sensor = ils_sensor
        self.env = environment

        # PID controllers
        # Lateral control (Localizer) -> Roll command
        # Tuned to provide reasonable response. These values would be determined through extensive simulation and flight testing.
        self.loc_pid = PIDController(kp=0.5, ki=0.01, kd=0.1, setpoint=0.0) # setpoint 0.0 for centerline

        # Vertical control (Glide Slope) -> Pitch command
        self.gs_pid = PIDController(kp=0.8, ki=0.02, kd=0.15, setpoint=0.0) # setpoint 0.0 for on-glide-slope

        # Airspeed control (Auto-throttle) -> Thrust command
        self.airspeed_pid = PIDController(kp=0.1, ki=0.005, kd=0.02, setpoint=70.0) # Desired approach speed in m/s

        self.autoland_state = "APPROACH"
        self.last_time = 0.0

        print(f"Autopilot initialized. Initial state: {self.autoland_state}")
        self.last_autopilot_data = {
            "state": self.autoland_state,
            "loc_dev": 0.0,
            "gs_dev": 0.0,
            "hag": self.ils_sensor.get_radio_altimeter_reading(self.aircraft.position),
            "airspeed": self.aircraft.airspeed,
            "pitch_cmd": 0.0,
            "roll_cmd": 0.0,
            "thrust_cmd": 0.0
        }


    def update(self, current_time):
        dt = current_time - self.last_time
        if dt <= 0:
            return self.last_autopilot_data

        loc_dev = self.ils_sensor.get_localizer_deviation(self.aircraft.position)
        gs_dev = self.ils_sensor.get_glideslope_deviation(self.aircraft.position)
        hag = self.ils_sensor.get_radio_altimeter_reading(self.aircraft.position)
        dme = self.ils_sensor.get_dme_distance(self.aircraft.position)
        airspeed = self.aircraft.airspeed


        pitch_cmd = None
        roll_cmd = None
        thrust_cmd = None

        if self.autoland_state == "APPROACH":
            # Autopilot captures LOC and GS
            # For simplicity, we assume capture is successful and directly transition to tracking.
            # In a real system, there would be a dedicated capture logic.
            self.autoland_state = "ILS_TRACKING"
            print(f"Time: {current_time:.2f}s - Autopilot state: {self.autoland_state}")

        if self.autoland_state == "ILS_TRACKING":
            roll_cmd = self.loc_pid.update(loc_dev, current_time)
            pitch_cmd = self.gs_pid.update(gs_dev, current_time)
            thrust_cmd = self.airspeed_pid.update(airspeed, current_time)

            # Check for flare condition
            if hag <= 50.0 and self.aircraft.velocity[2] < 0: # Below 50ft and descending
                self.autoland_state = "FLARE"
                self.loc_pid.reset() # Reset PID for rollout
                self.gs_pid.reset()  # Reset GS PID, flare logic takes over
                self.airspeed_pid.reset() # Reset airspeed PID, throttles to idle
                print(f"Time: {current_time:.2f}s - Autopilot state: {self.autoland_state} (HAG: {hag:.2f}m)")

        elif self.autoland_state == "FLARE":
            # Flare logic: reduce vertical speed for smooth touchdown
            # This is a critical phase. Pitch up slightly to reduce descent rate.
            # Very simplified: target a specific vertical speed at touchdown or a pitch angle.
            
            # Reduce thrust to idle
            thrust_cmd = 0.0 
            
            # Simple flare: pitch up proportional to remaining height
            # Target a small pitch-up to smoothly reduce vertical velocity
            # A more advanced flare would use radio altimeter rate.
            
            # Current vertical speed (negative for descent)
            vertical_speed = self.aircraft.velocity[2]

            # Example: target vertical speed of -0.5 m/s at touchdown
            # Proportionally increase pitch command as HAG decreases
            target_pitch_increase = (50.0 - hag) / 50.0 * np.deg2rad(2.0) # Max 2 deg pitch up
            pitch_cmd = np.deg2rad(self.env.glideslope_angle_deg) + target_pitch_increase # Maintain glideslope pitch initially, then add flare

            # Maintain centerline using localizer (even during flare)
            roll_cmd = self.loc_pid.update(loc_dev, current_time) # Continue localizer tracking

            if hag <= 0.5: # Touched down (or very close)
                self.autoland_state = "ROLLOUT"
                self.airspeed_pid.reset() # Ensure thrust is at idle
                print(f"Time: {current_time:.2f}s - Autopilot state: {self.autoland_state} (Touchdown HAG: {hag:.2f}m)")

        elif self.autoland_state == "ROLLOUT":
            # Rollout logic: maintain centerline and brake
            roll_cmd = self.loc_pid.update(loc_dev, current_time) # Use localizer for nose wheel steering
            thrust_cmd = -0.5 # Apply reverse thrust or brakes (negative for deceleration)
            
            # If aircraft speed is very low, landing is complete
            if self.aircraft.airspeed < 5.0: # m/s
                self.autoland_state = "COMPLETE"
                print(f"Time: {current_time:.2f}s - Autopilot state: {self.autoland_state} (Airspeed: {airspeed:.2f} m/s)")
        
        elif self.autoland_state == "COMPLETE":
            roll_cmd = 0.0
            pitch_cmd = 0.0
            thrust_cmd = 0.0
            # Autopilot disengages or hands over to ground control

        # Apply commands to aircraft
        self.aircraft.set_control_commands(pitch_cmd=pitch_cmd, roll_cmd=roll_cmd, thrust_cmd=thrust_cmd)
        self.last_time = current_time
        
        # Return current state and commands for logging/debugging
        return {
            "state": self.autoland_state,
            "loc_dev": loc_dev,
            "gs_dev": gs_dev,
            "hag": hag,
            "airspeed": airspeed,
            "pitch_cmd": pitch_cmd,
            "roll_cmd": roll_cmd,
            "thrust_cmd": thrust_cmd
        }
