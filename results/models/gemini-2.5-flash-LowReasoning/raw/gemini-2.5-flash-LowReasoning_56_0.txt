const fs = require('fs');

// Function to calculate Euclidean distance between two points
function calculateDistance(p1, p2) {
    return Math.sqrt(Math.pow(p1[0] - p2[0], 2) + Math.pow(p1[1] - p2[1], 2));
}

// 2-opt optimization for a single route
function twoOpt(route, coordinates, depot) {
    let bestRoute = [...route];
    let improved = true;

    while (improved) {
        improved = false;
        for (let i = 1; i < bestRoute.length - 1; i++) {
            for (let j = i + 1; j < bestRoute.length; j++) {
                // Generate a new route by swapping two segments
                const newRoute = [...bestRoute];
                const segment = newRoute.splice(i, j - i + 1);
                newRoute.splice(i, 0, ...segment.reverse());

                if (calculateTotalDistance(newRoute, coordinates, depot) < calculateTotalDistance(bestRoute, coordinates, depot)) {
                    bestRoute = newRoute;
                    improved = true;
                }
            }
        }
    }
    return bestRoute;
}

// Calculate total distance for a route
function calculateTotalDistance(route, coordinates, depot) {
    if (route.length === 0) {
        return 0;
    }

    let totalDistance = 0;
    let prevPoint = depot;

    for (const packageId of route) {
        const currentPoint = coordinates[packageId];
        totalDistance += calculateDistance(prevPoint, currentPoint);
        prevPoint = currentPoint;
    }

    totalDistance += calculateDistance(prevPoint, depot); // Return to depot
    return totalDistance;
}

function calculateIncrease(currentRoute, pkgId, insertPos, packageCoordinates, depot) {
    const pkgCoords = packageCoordinates[pkgId];

    let prevNodeCoords;
    if (insertPos === 0) {
        prevNodeCoords = depot;
    } else {
        prevNodeCoords = packageCoordinates[currentRoute[insertPos - 1]];
    }

    let nextNodeCoords;
    if (insertPos === currentRoute.length) { // inserting at the end
        nextNodeCoords = depot;
    } else {
        nextNodeCoords = packageCoordinates[currentRoute[insertPos]];
    }

    // If currentRoute is empty, then prevNode is depot and nextNode is also depot
    // This case is handled correctly by the above logic (insertPos 0 and currentRoute.length 0 means both prev/next are depot)
    // The "old distance" is the direct distance between prevNode and nextNode
    const oldDistance = calculateDistance(prevNodeCoords, nextNodeCoords);
    // The "new distance" is prevNode -> pkg -> nextNode
    const newDistance = calculateDistance(prevNodeCoords, pkgCoords) + calculateDistance(pkgCoords, nextNodeCoords);

    return newDistance - oldDistance;
}


function solveVRP(data) {
    const depot = data.depot;
    const truckCount = data.truckCount;
    const packages = data.packages;

    const packageCoordinates = {};
    for (const pkg of packages) {
        packageCoordinates[pkg.id] = [pkg.x, pkg.y];
    }

    let routes = Array.from({ length: truckCount }, () => []);

    let unassignedPackages = [...packages];

    while (unassignedPackages.length > 0) {
        let bestPackageIndex = -1;
        let bestRouteIndex = -1;
        let bestInsertPos = -1;
        let minIncrease = Infinity;

        for (let pIdx = 0; pIdx < unassignedPackages.length; pIdx++) {
            const pkg = unassignedPackages[pIdx];

            for (let rIdx = 0; rIdx < truckCount; rIdx++) {
                const currentRoute = routes[rIdx];

                for (let insertPos = 0; insertPos <= currentRoute.length; insertPos++) {
                    const increase = calculateIncrease(currentRoute, pkg.id, insertPos, packageCoordinates, depot);

                    if (increase < minIncrease) {
                        minIncrease = increase;
                        bestPackageIndex = pIdx;
                        bestRouteIndex = rIdx;
                        bestInsertPos = insertPos;
                    }
                }
            }
        }

        if (bestPackageIndex !== -1) {
            const assignedPackage = unassignedPackages.splice(bestPackageIndex, 1)[0];
            routes[bestRouteIndex].splice(bestInsertPos, 0, assignedPackage.id);
        } else {
            console.error("Could not assign a package. This should not happen.");
            break;
        }
    }

    // Optimize each route using 2-opt
    for (let rIdx = 0; rIdx < routes.length; rIdx++) {
        routes[rIdx] = twoOpt(routes[rIdx], packageCoordinates, depot);
    }

    return { "routes": routes };
}


// Read input from stdin
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

// Solve the VRP
const result = solveVRP(data);
