#version 450 core

layout(local_size_x = 256) in; // Common workgroup size

layout(std430, binding = 0) buffer InputBuffer {
    uint data_in[];
};

layout(std430, binding = 1) buffer OutputBuffer {
    uint data_out[];
};

// Shared memory for workgroup-local data
// Using a size of 256, matching local_size_x
shared uint temp_shared_mem[256];

void main() {
    uint local_id = gl_LocalInvocationID.x;
    uint global_id = gl_GlobalInvocationID.x;
    uint group_size = gl_WorkGroupSize.x;

    // Load data from global memory to shared memory
    // Only load if within bounds of the input data
    if (global_id < data_in.length()) {
        temp_shared_mem[local_id] = data_in[global_id];
    } else {
        temp_shared_mem[local_id] = 0; // Initialize out-of-bounds to 0
    }
    
    barrier(); // Synchronize threads within the workgroup

    // Hillis-Steele Parallel Inclusive Scan
    for (uint stride = 1; stride < group_size; stride <<= 1) {
        // Ensure that local_id - stride is not negative (unsigned int)
        // and that local_id is within the valid range for the stride.
        if (local_id >= stride) {
            temp_shared_mem[local_id] += temp_shared_mem[local_id - stride];
        }
        barrier(); // Synchronize after each step
    }

    // Write the final inclusive sum from shared memory to global output
    if (global_id < data_in.length()) { // Only write if within original input bounds
        data_out[global_id] = temp_shared_mem[local_id];
    }
}
