fn subset_sum(nums: &[i32], target: i32) -> bool {
    let n = nums.len();
    let mut dp = vec![vec![false; (target + 1) as usize]; n + 1];

    // Base case: if target is 0, an empty subset sums to 0
    for i in 0..=n {
        dp[i][0] = true;
    }

    // Fill the dp table
    for i in 1..=n {
        for j in 1..=target {
            // If the current number is greater than the current sum,
            // we cannot include it. So, the result is the same as
            // without the current number.
            if nums[i - 1] > j {
                dp[i][j] = dp[i - 1][j];
            } else {
                // If we include the current number, check if the remaining
                // sum (j - nums[i-1]) can be formed by previous numbers.
                // Or, if we don't include the current number, check if
                // the current sum (j) can be formed by previous numbers.
                dp[i][j] = dp[i - 1][j] || dp[i - 1][(j - nums[i - 1]) as usize];
            }
        }
    }

    dp[n][target as usize]
}

fn main() {
    // Test cases
    let nums1 = vec![3, 34, 4, 12, 5, 2];
    let target1 = 9;
    println!("Can subset_sum({:?}, {}) be formed? {}", nums1, target1, subset_sum(&nums1, target1)); // Expected: true (4 + 5)

    let nums2 = vec![3, 34, 4, 12, 5, 2];
    let target2 = 30;
    println!("Can subset_sum({:?}, {}) be formed? {}", nums2, target2, subset_sum(&nums2, target2)); // Expected: false

    let nums3 = vec![1, 2, 3];
    let target3 = 6;
    println!("Can subset_sum({:?}, {}) be formed? {}", nums3, target3, subset_sum(&nums3, target3)); // Expected: true (1 + 2 + 3)

    let nums4 = vec![1, 2, 3];
    let target4 = 7;
    println!("Can subset_sum({:?}, {}) be formed? {}", nums4, target4, subset_sum(&nums4, target4)); // Expected: false

    let nums5 = vec![];
    let target5 = 0;
    println!("Can subset_sum({:?}, {}) be formed? {}", nums5, target5, subset_sum(&nums5, target5)); // Expected: true

    let nums6 = vec![10];
    let target6 = 10;
    println!("Can subset_sum({:?}, {}) be formed? {}", nums6, target6, subset_sum(&nums6, target6)); // Expected: true

    let nums7 = vec![10];
    let target7 = 5;
    println!("Can subset_sum({:?}, {}) be formed? {}", nums7, target7, subset_sum(&nums7, target7)); // Expected: false
}
