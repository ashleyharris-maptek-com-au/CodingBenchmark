// Castling rights bitmasks
const uint8_t WHITE_KING_SIDE_CASTLE = 1;
const uint8_t WHITE_QUEEN_SIDE_CASTLE = 2;
const uint8_t BLACK_KING_SIDE_CASTLE = 4;
const uint8_t BLACK_QUEEN_SIDE_CASTLE = 8;

// Helper function to get rank from square
int getRank(Square sq) {
    return sq / 8;
}

// Helper function to get file from square
int getFile(Square sq) {
    return sq % 8;
}

// Helper function to check if a square is on the board
bool isValidSquare(Square sq) {
    return sq >= A1 && sq <= H8;
}

// Helper to determine color of piece at a square
Color getPieceColor(Piece p) {
    return p.color;
}

Board::Board() : sideToMove(Color::WHITE), castlingRights(0), enPassantSquare(NO_SQUARE), halfMoveClock(0), fullMoveNumber(1) {
    // Initialize with empty board
    for (int i = 0; i < 64; ++i) {
        squares[i] = {PieceType::NONE, Color::NONE};
    }
    setupFromFen("rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1");
}

Board::Board(const std::string& fen) : sideToMove(Color::WHITE), castlingRights(0), enPassantSquare(NO_SQUARE), halfMoveClock(0), fullMoveNumber(1) {
    // Initialize with empty board
    for (int i = 0; i < 64; ++i) {
        squares[i] = {PieceType::NONE, Color::NONE};
    }

Piece Board::getPiece(Square sq) const {
    if (sq < 0 || sq >= 64) {
        // Handle invalid square, maybe throw an exception or return a default invalid piece
        return {PieceType::NONE, Color::NONE};
    }
    return squares[sq];
}

void Board::setPiece(Square sq, Piece p) {
    if (sq >= 0 && sq < 64) {
        squares[sq] = p;
    }
}

void Board::setupFromFen(const std::string& fen) {
    std::istringstream iss(fen);
    std::string boardFenStr, activeColorStr, castlingRightsStr, enPassantSquareStr, halfMoveClockStr, fullMoveNumberStr;

    iss >> boardFenStr >> activeColorStr >> castlingRightsStr >> enPassantSquareStr >> halfMoveClockStr >> fullMoveNumberStr;

    // Parse board position
    int file = 0;
    int rank = 7; // FEN starts from rank 8 (index 7)
    for (char c : boardFenStr) {
        if (c == '/') {
            rank--;
            file = 0;
        } else if (isdigit(c)) {
            file += (c - '0');
        } else {
            Color color = isupper(c) ? Color::WHITE : Color::BLACK;
            PieceType type = pieceTypeFromChar.at(tolower(c));
            squares[rank * 8 + file] = {type, color};
            file++;
        }
    }

    // Parse active color
    sideToMove = (activeColorStr == "w") ? Color::WHITE : Color::BLACK;


    // Parse castling rights
    this->castlingRights = 0;
    for (char c : castlingRightsStr) {
        if (c == 'K') this->castlingRights |= WHITE_KING_SIDE_CASTLE;
        else if (c == 'Q') this->castlingRights |= WHITE_QUEEN_SIDE_CASTLE;
        else if (c == 'k') this->castlingRights |= BLACK_KING_SIDE_CASTLE;
        else if (c == 'q') this->castlingRights |= BLACK_QUEEN_SIDE_CASTLE;
    }

    // Parse en passant square
    if (enPassantSquareStr != "-") {
        this->enPassantSquare = algebraicToSquare.at(enPassantSquareStr);
    } else {
        this->enPassantSquare = NO_SQUARE;
    }

    // Parse halfmove clock and fullmove number
    this->halfMoveClock = std::stoi(halfMoveClockStr);
    this->fullMoveNumber = std::stoi(fullMoveNumberStr);

}

std::vector<Move> Board::generateLegalMoves() const {
    std::vector<Move> pseudoLegalMoves;
    // Iterate over all squares to find pieces of the current player
    for (int i = 0; i < 64; ++i) {
        Square fromSq = static_cast<Square>(i);
        Piece piece = getPiece(fromSq);

        if (piece.color == sideToMove) {
            switch (piece.type) {
                case PieceType::PAWN: {
                    int direction = (sideToMove == Color::WHITE) ? 8 : -8; // +8 for white (up), -8 for black (down)
                    int startRank = (sideToMove == Color::WHITE) ? 1 : 6;
                    int promotionRank = (sideToMove == Color::WHITE) ? 7 : 0;

                    // Single pawn push
                    Square targetSq = static_cast<Square>(fromSq + direction);
                    if (isValidSquare(targetSq) && getPiece(targetSq).type == PieceType::NONE) {
                        if (getRank(targetSq) == promotionRank) {
                            pseudoLegalMoves.emplace_back(fromSq, targetSq, PieceType::QUEEN);
                            pseudoLegalMoves.emplace_back(fromSq, targetSq, PieceType::ROOK);
                            pseudoLegalMoves.emplace_back(fromSq, targetSq, PieceType::BISHOP);
                            pseudoLegalMoves.emplace_back(fromSq, targetSq, PieceType::KNIGHT);
                        } else {
                            pseudoLegalMoves.emplace_back(fromSq, targetSq);
                        }

                        // Double pawn push
                        if (getRank(fromSq) == startRank) {
                            Square doubleTargetSq = static_cast<Square>(fromSq + 2 * direction);
                            if (isValidSquare(doubleTargetSq) && getPiece(doubleTargetSq).type == PieceType::NONE) {
                                pseudoLegalMoves.emplace_back(fromSq, doubleTargetSq);
                            }
                        }
                    }

                    // Pawn captures
                    int captureLeftOffset = (sideToMove == Color::WHITE) ? 7 : -9; // White: -1 file, +1 rank. Black: -1 file, -1 rank
                    int captureRightOffset = (sideToMove == Color::WHITE) ? 9 : -7; // White: +1 file, +1 rank. Black: +1 file, -1 rank

                    // Check left capture
                    targetSq = static_cast<Square>(fromSq + captureLeftOffset);
                    // Ensure the target square is on the board AND that the file has not wrapped around
                    if (isValidSquare(targetSq) && getFile(targetSq) != getFile(fromSq) &&
                        getPiece(targetSq).type != PieceType::NONE && getPieceColor(getPiece(targetSq)) != sideToMove) {
                        if (getRank(targetSq) == promotionRank) {
                            pseudoLegalMoves.emplace_back(fromSq, targetSq, PieceType::QUEEN);
                            pseudoLegalMoves.emplace_back(fromSq, targetSq, PieceType::ROOK);
                            pseudoLegalMoves.emplace_back(fromSq, targetSq, PieceType::BISHOP);
                            pseudoLegalMoves.emplace_back(fromSq, targetSq, PieceType::KNIGHT);
                        } else {
                            pseudoLegalMoves.emplace_back(fromSq, targetSq);
                        }
                    }

                    // Check right capture
                    targetSq = static_cast<Square>(fromSq + captureRightOffset);
                    // Ensure the target square is on the board AND that the file has not wrapped around
                    if (isValidSquare(targetSq) && getFile(targetSq) != getFile(fromSq) &&
                        getPiece(targetSq).type != PieceType::NONE && getPieceColor(getPiece(targetSq)) != sideToMove) {
                        if (getRank(targetSq) == promotionRank) {
                            pseudoLegalMoves.emplace_back(fromSq, targetSq, PieceType::QUEEN);
                            pseudoLegalMoves.emplace_back(fromSq, targetSq, PieceType::ROOK);
                            pseudoLegalMoves.emplace_back(fromSq, targetSq, PieceType::BISHOP);
                            pseudoLegalMoves.emplace_back(fromSq, targetSq, PieceType::KNIGHT);
                        } else {
                            pseudoLegalMoves.emplace_back(fromSq, targetSq);
                        }
                    }
                    
                    // En passant
                    if (enPassantSquare != NO_SQUARE) {
                        // Check if current pawn is on the correct rank for en passant (rank 4 for white, rank 3 for black)
                        int enPassantRank = (sideToMove == Color::WHITE) ? 4 : 3;
                        if (getRank(fromSq) == enPassantRank) {
                            // Check if en passant target square is diagonally adjacent to the pawn's current square
                            if ((static_cast<int>(fromSq) + captureLeftOffset == static_cast<int>(enPassantSquare)) ||
                                (static_cast<int>(fromSq) + captureRightOffset == static_cast<int>(enPassantSquare))) {
                                pseudoLegalMoves.emplace_back(fromSq, enPassantSquare);
                            }
                        }
                    }

                    break;
                }
                case PieceType::KNIGHT: {
                    // Knight move offsets
                    const int knightMoves[] = {-17, -15, -10, -6, 6, 10, 15, 17};
                    for (int move : knightMoves) {
                        Square targetSq = static_cast<Square>(fromSq + move);
                        // Ensure target square is on the board
                        if (!isValidSquare(targetSq)) continue;

                        // Check for wrapping around the board edges (file difference should be 1 or 2)
                        int fileDiff = std::abs(getFile(fromSq) - getFile(targetSq));
                        if (fileDiff != 1 && fileDiff != 2) continue;

                        Piece targetPiece = getPiece(targetSq);
                        // If square is empty or occupied by an opponent's piece
                        if (targetPiece.type == PieceType::NONE || getPieceColor(targetPiece) != sideToMove) {
                            pseudoLegalMoves.emplace_back(fromSq, targetSq);
                        }
                    }
                    break;
                }
                // TODO: Implement other piece types
                case PieceType::BISHOP:
                case PieceType::ROOK:
                case PieceType::QUEEN:
                case PieceType::KING:
                case PieceType::NONE:
                    break;
            }
        }
    }
    return pseudoLegalMoves;
}
