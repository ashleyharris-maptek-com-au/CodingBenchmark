using System;
using System.Collections.Generic;
using System.Linq;

namespace MaximumCliqueSolver
{
    // Represents a graph using an adjacency list
    public class Graph
    {
        private readonly int _numVertices;
        private readonly HashSet<int>[] _adj;

        public Graph(int numVertices)
        {
            _numVertices = numVertices;
            _adj = new HashSet<int>[numVertices];
            for (int i = 0; i < numVertices; i++)
            {
                _adj[i] = new HashSet<int>();
            }
        }

        // Add an edge between two vertices
        public void AddEdge(int u, int v)
        {
            _adj[u].Add(v);
            _adj[v].Add(u); // Assuming undirected graph
        }

        // Get neighbors of a vertex
        public HashSet<int> GetNeighbors(int v)
        {
            return _adj[v];
        }

        // Check if two vertices are adjacent
        public bool AreAdjacent(int u, int v)
        {
            return _adj[u].Contains(v);
        }

        public int NumVertices => _numVertices;
    }

    // Solves the Maximum Clique problem using a backtracking approach (Bron-Kerbosch-like)
    public class MaximumCliqueSolver
    {
        private Graph _graph;
        private List<int> _maxClique;

        public MaximumCliqueSolver(Graph graph)
        {
            _graph = graph;
            _maxClique = new List<int>();
        }

        public List<int> FindMaximumClique()
        {
            _maxClique.Clear();
            List<int> currentClique = new List<int>();
            HashSet<int> candidates = new HashSet<int>(Enumerable.Range(0, _graph.NumVertices));

            BronKerbosch(currentClique, candidates, new HashSet<int>());

            return _maxClique;
        }

        private void BronKerbosch(List<int> R, HashSet<int> P, HashSet<int> X)
        {
            if (P.Count == 0 && X.Count == 0)
            {
                if (R.Count > _maxClique.Count)
                {
                    _maxClique = new List<int>(R);
                }
                return;
            }

            int pivot = -1;
            int maxDegree = -1;
            // Combine P and X for pivot selection
            var P_union_X = P.Union(X); 

            foreach (int v in P_union_X)
            {
                int degree = _graph.GetNeighbors(v).Intersect(P).Count(); 
                if (degree > maxDegree)
                {
                    maxDegree = degree;
                    pivot = v;
                }
            }

            // P_minus_N_pivot = P \ N(pivot)
            // Iterate over a copy to allow modification of P inside the loop
            var P_minus_N_pivot_list = P.Except(_graph.GetNeighbors(pivot)).ToList();

            foreach (int v in P_minus_N_pivot_list)
            {
                R.Add(v);
                var N_v = _graph.GetNeighbors(v);
                
                // Create new HashSets for the recursive call to maintain state for backtracking
                BronKerbosch(R, new HashSet<int>(P.Intersect(N_v)), new HashSet<int>(X.Union(new HashSet<int> { v }).Intersect(N_v)));
                
                R.Remove(v);
                P.Remove(v); // Remove v from candidates as well for the current level of recursion
                X.Add(v); // Add v to the excluded set for the current level of recursion
            }
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            string line = Console.ReadLine();
            string[] parts = line.Split(' ');
            int numVertices = int.Parse(parts[0]);
            int numEdges = int.Parse(parts[1]);

            Graph graph = new Graph(numVertices);

            for (int i = 0; i < numEdges; i++)
            {
                line = Console.ReadLine();
                parts = line.Split(' ');
                int u = int.Parse(parts[0]);
                int v = int.Parse(parts[1]);
                graph.AddEdge(u, v);
            }

            MaximumCliqueSolver solver = new MaximumCliqueSolver(graph);
            List<int> maxClique = solver.FindMaximumClique();

            Console.WriteLine(maxClique.Count);
            Console.WriteLine(string.Join(" ", maxClique));
        }
    }
}
