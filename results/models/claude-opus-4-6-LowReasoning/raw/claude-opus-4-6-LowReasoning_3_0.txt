use std::io::{self, Read, Write, BufWriter};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let stdout = io::stdout();
    let mut out = BufWriter::new(stdout.lock());

    let mut iter = input.split_ascii_whitespace();
    let n: usize = iter.next().unwrap().parse().unwrap();
    let m: usize = iter.next().unwrap().parse().unwrap();

    let mut adj: Vec<Vec<usize>> = vec![vec![]; n];
    let mut edges: Vec<(usize, usize)> = Vec::with_capacity(m);

    for _ in 0..m {
        let u: usize = iter.next().unwrap().parse().unwrap();
        let v: usize = iter.next().unwrap().parse().unwrap();
        if u == v || u >= n || v >= n { continue; }
        adj[u].push(v);
        adj[v].push(u);
        edges.push((u, v));
    }

    // Deduplicate adjacency lists
    for i in 0..n {
        adj[i].sort_unstable();
        adj[i].dedup();
    }

    let mut x = vec![0.0f64; n];
    let mut y = vec![0.0f64; n];

    if n <= 1 {
        for i in 0..n {
            writeln!(out, "{:.6} {:.6}", i as f64, 0.0).unwrap();
        }
        return;
    }

    // Find connected components
    let mut comp = vec![usize::MAX; n];
    let mut comp_id = 0usize;
    let mut comp_sizes: Vec<usize> = Vec::new();
    let mut stack = Vec::new();
    for start in 0..n {
        if comp[start] != usize::MAX { continue; }
        let mut size = 0usize;
        stack.push(start);
        comp[start] = comp_id;
        while let Some(u) = stack.pop() {
            size += 1;
            for &v in &adj[u] {
                if comp[v] == usize::MAX {
                    comp[v] = comp_id;
                    stack.push(v);
                }
            }
        }
        comp_sizes.push(size);
        comp_id += 1;
    }

    // Layout each component separately, then pack them
    let num_comps = comp_sizes.len();
    let mut comp_nodes: Vec<Vec<usize>> = vec![vec![]; num_comps];
    for i in 0..n {
        comp_nodes[comp[i]].push(i);
    }

    for c in 0..num_comps {
        let nodes = &comp_nodes[c];
        let cn = nodes.len();
        if cn == 1 {
            x[nodes[0]] = 0.0;
            y[nodes[0]] = 0.0;
            continue;
        }

        // Map global node ids to local ids
        let mut local_id = vec![0usize; n];
        for (li, &gi) in nodes.iter().enumerate() {
            local_id[gi] = li;
        }

        // Build local adjacency
        let mut ladj: Vec<Vec<usize>> = vec![vec![]; cn];
        for &gi in nodes {
            let li = local_id[gi];
            for &gv in &adj[gi] {
                if comp[gv] == c {
                    ladj[li].push(local_id[gv]);
                }
            }
        }

        let (lx, ly) = layout_component(cn, &ladj);
        for (li, &gi) in nodes.iter().enumerate() {
            x[gi] = lx[li];
            y[gi] = ly[li];
        }
    }

    // Pack components: arrange bounding boxes in rows
    if num_comps > 1 {
        // Sort components by size descending for better packing
        let mut comp_order: Vec<usize> = (0..num_comps).collect();
        comp_order.sort_unstable_by(|&a, &b| comp_sizes[b].cmp(&comp_sizes[a]));

        let mut offset_x = 0.0f64;
        let mut offset_y = 0.0f64;
        let mut row_height = 0.0f64;
        let max_row_width = (n as f64).sqrt() * 3.0;

        for &c in &comp_order {
            let nodes = &comp_nodes[c];
            if nodes.is_empty() { continue; }
            let mut minx = f64::MAX; let mut maxx = f64::MIN;
            let mut miny = f64::MAX; let mut maxy = f64::MIN;
            for &gi in nodes {
                minx = minx.min(x[gi]); maxx = maxx.max(x[gi]);
                miny = miny.min(y[gi]); maxy = maxy.max(y[gi]);
            }
            let w = maxx - minx + 2.0;
            let h = maxy - miny + 2.0;

            if offset_x + w > max_row_width && offset_x > 0.0 {
                offset_x = 0.0;
                offset_y += row_height + 2.0;
                row_height = 0.0;
            }

            let dx = offset_x - minx;
            let dy = offset_y - miny;
            for &gi in nodes {
                x[gi] += dx;
                y[gi] += dy;
            }
            offset_x += w + 2.0;
            row_height = row_height.max(h);
        }
    }

    writeln!(out, "{}", n).unwrap();
    for i in 0..n {
        writeln!(out, "{:.6} {:.6}", x[i], y[i]).unwrap();
    }
}

fn layout_component(n: usize, adj: &[Vec<usize>]) -> (Vec<f64>, Vec<f64>) {
    if n <= 3 {
        // Place on a circle
        let mut lx = vec![0.0; n];
        let mut ly = vec![0.0; n];
        for i in 0..n {
            let angle = 2.0 * std::f64::consts::PI * i as f64 / n as f64;
            lx[i] = angle.cos();
            ly[i] = angle.sin();
        }
        return (lx, ly);
    }

    if n > 500 {
        return multilevel_layout(n, adj);
    }

    // Standard force-directed for small-medium graphs
    force_directed(n, adj, None, None)
}

fn force_directed(
    n: usize,
    adj: &[Vec<usize>],
    init_x: Option<&[f64]>,
    init_y: Option<&[f64]>,
) -> (Vec<f64>, Vec<f64>) {
    let mut x = vec![0.0f64; n];
    let mut y = vec![0.0f64; n];

    let area = (n as f64) * 100.0;
    let k = (area / n as f64).sqrt();
    let k2 = k * k;

    // Initialize positions
    match (init_x, init_y) {
        (Some(ix), Some(iy)) => {
            x.copy_from_slice(ix);
            y.copy_from_slice(iy);
        }
        _ => {
            // Use spectral-like initialization for better starting positions
            // BFS from highest-degree node for x, perpendicular for y
            let max_deg_node = (0..n).max_by_key(|&i| adj[i].len()).unwrap_or(0);
            let bfs_dist = bfs(n, adj, max_deg_node);
            let max_dist = *bfs_dist.iter().max().unwrap_or(&1) as f64;
            for i in 0..n {
                x[i] = if max_dist > 0.0 { bfs_dist[i] as f64 / max_dist * k * (n as f64).sqrt() } else { 0.0 };
            }
            // Find farthest node from max_deg_node, do BFS from there for y
            let far = (0..n).max_by_key(|&i| bfs_dist[i]).unwrap_or(0);
            let bfs2 = bfs(n, adj, far);
            let max_dist2 = *bfs2.iter().max().unwrap_or(&1) as f64;
            for i in 0..n {
                y[i] = if max_dist2 > 0.0 { bfs2[i] as f64 / max_dist2 * k * (n as f64).sqrt() } else { 0.0 };
            }
            // Add small perturbation
            for i in 0..n {
                let h = ((i * 2654435761) & 0xFFFF) as f64 / 65536.0;
                x[i] += (h - 0.5) * k * 0.1;
                let h2 = ((i * 1013904223 + 1) & 0xFFFF) as f64 / 65536.0;
                y[i] += (h2 - 0.5) * k * 0.1;
            }
        }
    }

    let iterations = if n < 50 { 300 } else if n < 200 { 200 } else { 100 };
    let mut temp = k * (n as f64).sqrt() * 0.1;
    let cooling = temp / (iterations as f64 + 1.0);

    let mut dx = vec![0.0f64; n];
    let mut dy = vec![0.0f64; n];

    let use_barnes_hut = n > 100;

    for _iter in 0..iterations {
        for i in 0..n { dx[i] = 0.0; dy[i] = 0.0; }

        if use_barnes_hut {
            // Barnes-Hut quad tree for repulsive forces
            let tree = QuadTree::build(&x, &y, n);
            for i in 0..n {
                let (fx, fy) = tree.repulsive_force(i, x[i], y[i], k2, 1.2);
                dx[i] += fx;
                dy[i] += fy;
            }
        } else {
            // Direct O(n^2) repulsive forces
            for i in 0..n {
                for j in (i+1)..n {
                    let mut ddx = x[i] - x[j];
                    let mut ddy = y[i] - y[j];
                    let dist2 = ddx * ddx + ddy * ddy;
                    if dist2 < 1e-4 {
                        let h = ((i * 2654435761 + j) & 0xFF) as f64 / 256.0;
                        ddx = (h - 0.5) * 0.1;
                        ddy = ((h * 2.0).fract() - 0.5) * 0.1;
                    }
                    let dist = dist2.max(1e-4).sqrt();
                    let force = k2 / dist;
                    let fx = ddx / dist * force;
                    let fy = ddy / dist * force;
                    dx[i] += fx; dy[i] += fy;
                    dx[j] -= fx; dy[j] -= fy;
                }
            }
        }

        // Attractive forces
        for i in 0..n {
            for &j in &adj[i] {
                if j <= i { continue; }
                let ddx = x[i] - x[j];
                let ddy = y[i] - y[j];
                let dist = (ddx * ddx + ddy * ddy).max(1e-8).sqrt();
                let force = dist / k;
                let fx = ddx / dist * force;
                let fy = ddy / dist * force;
                dx[i] -= fx; dy[i] -= fy;
                dx[j] += fx; dy[j] += fy;
            }
        }

        // Apply forces with temperature limiting
        for i in 0..n {
            let disp = (dx[i] * dx[i] + dy[i] * dy[i]).max(1e-8).sqrt();
            let scale = temp.min(disp) / disp;
            x[i] += dx[i] * scale;
            y[i] += dy[i] * scale;
        }

        temp -= cooling;
        if temp < 0.0 { temp = 0.0; }
    }

    (x, y)
}

fn bfs(n: usize, adj: &[Vec<usize>], start: usize) -> Vec<usize> {
    let mut dist = vec![usize::MAX; n];
    dist[start] = 0;
    let mut queue = std::collections::VecDeque::new();
    queue.push_back(start);
    while let Some(u) = queue.pop_front() {
        for &v in &adj[u] {
            if dist[v] == usize::MAX {
                dist[v] = dist[u] + 1;
                queue.push_back(v);
            }
        }
    }
    // Set unreachable nodes to 0
    for d in dist.iter_mut() {
        if *d == usize::MAX { *d = 0; }
    }
    dist
}

fn multilevel_layout(n: usize, adj: &[Vec<usize>]) -> (Vec<f64>, Vec<f64>) {
    // Coarsen the graph
    let levels = coarsen(n, adj);

    // Layout the coarsest level
    let coarsest = levels.last().unwrap();
    let cn = coarsest.num_nodes;
    let (mut cx, mut cy) = if cn <= 500 {
        force_directed(cn, &coarsest.adj, None, None)
    } else {
        // Circle layout for very coarse
        let mut lx = vec![0.0; cn];
        let mut ly = vec![0.0; cn];
        let r = (cn as f64).sqrt() * 5.0;
        for i in 0..cn {
            let angle = 2.0 * std::f64::consts::PI * i as f64 / cn as f64;
            lx[i] = r * angle.cos();
            ly[i] = r * angle.sin();
        }
        (lx, ly)
    };

    // Uncoarsen and refine
    for level in levels.iter().rev() {
        let fn_n = level.parent_n;
        let mut fx = vec![0.0f64; fn_n];
        let mut fy = vec![0.0f64; fn_n];

        // Interpolate positions from coarse to fine
        for i in 0..fn_n {
            let ci = level.mapping[i];
            fx[i] = cx[ci];
            fy[i] = cy[ci];
        }

        // Add small perturbation to nodes mapped to the same coarse node
        for i in 0..fn_n {
            let h = ((i * 2654435761) & 0xFFFF) as f64 / 65536.0;
            fx[i] += (h - 0.5) * 0.5;
            let h2 = ((i * 1013904223 + 1) & 0xFFFF) as f64 / 65536.0;
            fy[i] += (h2 - 0.5) * 0.5;
        }

        // Refine with limited force-directed iterations
        let iters = if fn_n < 1000 { 50 } else { 20 };
        let (rx, ry) = force_directed_limited(fn_n, &level.parent_adj, &fx, &fy, iters);
        cx = rx;
        cy = ry;
    }

    (cx, cy)
}

struct CoarseLevel {
    num_nodes: usize,
    adj: Vec<Vec<usize>>,
    mapping: Vec<usize>,   // fine node -> coarse node
    parent_n: usize,
    parent_adj: Vec<Vec<usize>>,
}

fn coarsen(n: usize, adj: &[Vec<usize>]) -> Vec<CoarseLevel> {
    let mut levels = Vec::new();
    let mut cur_n = n;
    let mut cur_adj = adj.to_vec();

    loop {
        if cur_n <= 500 { break; }

        // Heavy-edge matching
        let mut matched = vec![false; cur_n];
        let mut mapping = vec![usize::MAX; cur_n];
        let mut coarse_id = 0usize;

        // Process nodes in order of increasing degree for better matching
        let mut order: Vec<usize> = (0..cur_n).collect();
        order.sort_unstable_by_key(|&i| cur_adj[i].len());

        for &u in &order {
            if matched[u] { continue; }
            // Find unmatched neighbor with heaviest edge (highest degree overlap)
            let mut best = None;
            let mut best_score = usize::MAX; // prefer low-degree neighbors
            for &v in &cur_adj[u] {
                if !matched[v] && v != u {
                    let score = cur_adj[v].len();
                    if best.is_none() || score < best_score {
                        best = Some(v);
                        best_score = score;
                    }
                }
            }
            matched[u] = true;
            mapping[u] = coarse_id;
            if let Some(v) = best {
                matched[v] = true;
                mapping[v] = coarse_id;
            }
            coarse_id += 1;
        }

        // Build coarse adjacency
        let cn = coarse_id;
        if cn >= cur_n * 3 / 4 { break; } // not enough coarsening

        let mut cadj: Vec<Vec<usize>> = vec![vec![]; cn];
        for u in 0..cur_n {
            let cu = mapping[u];
            for &v in &cur_adj[u] {
                let cv = mapping[v];
                if cu != cv {
                    cadj[cu].push(cv);
                }
            }
        }
        for i in 0..cn {
            cadj[i].sort_unstable();
            cadj[i].dedup();
        }

        levels.push(CoarseLevel {
            num_nodes: cn,
            adj: cadj.clone(),
            mapping: mapping,
            parent_n: cur_n,
            parent_adj: cur_adj.clone(),
        });

        cur_n = cn;
        cur_adj = cadj;
    }

    if levels.is_empty() {
        // No coarsening happened, just create a dummy level
        levels.push(CoarseLevel {
            num_nodes: n,
            adj: adj.to_vec(),
            mapping: (0..n).collect(),
            parent_n: n,
            parent_adj: adj.to_vec(),
        });
    }

    levels
}

fn force_directed_limited(
    n: usize,
    adj: &[Vec<usize>],
    init_x: &[f64],
    init_y: &[f64],
    iterations: usize,
) -> (Vec<f64>, Vec<f64>) {
    let mut x = init_x.to_vec();
    let mut y = init_y.to_vec();

    let area = (n as f64) * 100.0;
    let k = (area / n as f64).sqrt();
    let k2 = k * k;

    let mut temp = k * 0.5;
    let cooling = temp / (iterations as f64 + 1.0);

    let mut dx = vec![0.0f64; n];
    let mut dy = vec![0.0f64; n];

    for _iter in 0..iterations {
        for i in 0..n { dx[i] = 0.0; dy[i] = 0.0; }

        // Barnes-Hut repulsive forces
        let tree = QuadTree::build(&x, &y, n);
        for i in 0..n {
            let (fx, fy) = tree.repulsive_force(i, x[i], y[i], k2, 1.5);
            dx[i] += fx;
            dy[i] += fy;
        }

        // Attractive forces
        for i in 0..n {
            for &j in &adj[i] {
                if j <= i { continue; }
                let ddx = x[i] - x[j];
                let ddy = y[i] - y[j];
                let dist = (ddx * ddx + ddy * ddy).max(1e-8).sqrt();
                let force = dist / k;
                let fx = ddx / dist * force;
                let fy = ddy / dist * force;
                dx[i] -= fx; dy[i] -= fy;
                dx[j] += fx; dy[j] += fy;
            }
        }

        for i in 0..n {
            let disp = (dx[i] * dx[i] + dy[i] * dy[i]).max(1e-8).sqrt();
            let scale = temp.min(disp) / disp;
            x[i] += dx[i] * scale;
            y[i] += dy[i] * scale;
        }

        temp -= cooling;
        if temp < 0.0 { temp = 0.0; }
    }

    (x, y)
}

// Barnes-Hut Quad Tree
struct QuadTree {
    nodes: Vec<QTNode>,
}

struct QTNode {
    cx: f64,
    cy: f64,
    mass: f64,
    is_leaf: bool,
    point_idx: usize, // only valid for leaf with single point
    children: [usize; 4], // NW, NE, SW, SE; 0 = empty
    x_min: f64, x_max: f64,
    y_min: f64, y_max: f64,
}

impl QuadTree {
    fn build(x: &[f64], y: &[f64], n: usize) -> Self {
        if n == 0 {
            return QuadTree { nodes: vec![] };
        }
        let mut xmin = x[0]; let mut xmax = x[0];
        let mut ymin = y[0]; let mut ymax = y[0];
        for i in 1..n {
            xmin = xmin.min(x[i]); xmax = xmax.max(x[i]);
            ymin = ymin.min(y[i]); ymax = ymax.max(y[i]);
        }
        let pad = 1.0;
        xmin -= pad; xmax += pad; ymin -= pad; ymax += pad;
        // Make square
        let size = (xmax - xmin).max(ymax - ymin);
        xmax = xmin + size;
        ymax = ymin + size;

        let mut qt = QuadTree {
            nodes: Vec::with_capacity(n * 4),
        };
        let root = qt.new_node(xmin, xmax, ymin, ymax);
        let _ = root;

        for i in 0..n {
            qt.insert(0, i, x[i], y[i], 0);
        }

        qt
    }

    fn new_node(&mut self, x_min: f64, x_max: f64, y_min: f64, y_max: f64) -> usize {
        let idx = self.nodes.len();
        self.nodes.push(QTNode {
            cx: 0.0, cy: 0.0, mass: 0.0,
            is_leaf: true, point_idx: usize::MAX,
            children: [0; 4],
            x_min, x_max, y_min, y_max,
        });
        idx
    }

    fn insert(&mut self, node: usize, idx: usize, px: f64, py: f64, depth: usize) {
        if depth > 60 { return; }

        let mass = self.nodes[node].mass;
        if mass == 0.0 && self.nodes[node].is_leaf {
            // Empty leaf, just store
            self.nodes[node].cx = px;
            self.nodes[node].cy = py;
            self.nodes[node].mass = 1.0;
            self.nodes[node].point_idx = idx;
            return;
        }

        if self.nodes[node].is_leaf {
            // Split: re-insert existing point
            self.nodes[node].is_leaf = false;
            let old_idx = self.nodes[node].point_idx;
            let old_cx = self.nodes[node].cx;
            let old_cy = self.nodes[node].cy;
            self.nodes[node].point_idx = usize::MAX;
            // Reset mass/center, will be recalculated
            self.nodes[node].mass = 0.0;
            self.nodes[node].cx = 0.0;
            self.nodes[node].cy = 0.0;

            self.insert_into_child(node, old_idx, old_cx, old_cy, depth);
            self.insert_into_child(node, idx, px, py, depth);
        } else {
            self.insert_into_child(node, idx, px, py, depth);
        }

        // Update center of mass
        let m = self.nodes[node].mass;
        // Recalculate from scratch is expensive; do incremental
        // Actually let's just do it incrementally during insert_into_child
    }

    fn insert_into_child(&mut self, node: usize, idx: usize, px: f64, py: f64, depth: usize) {
        let xmid = (self.nodes[node].x_min + self.nodes[node].x_max) * 0.5;
        let ymid = (self.nodes[node].y_min + self.nodes[node].y_max) * 0.5;

        let quadrant = if px <= xmid {
            if py <= ymid { 0 } else { 2 }
        } else {
            if py <= ymid { 1 } else { 3 }
        };

        let child = self.nodes[node].children[quadrant];
        if child == 0 {
            // Create child node
            let (cxmin, cxmax, cymin, cymax) = match quadrant {
                0 => (self.nodes[node].x_min, xmid, self.nodes[node].y_min, ymid),
                1 => (xmid, self.nodes[node].x_max, self.nodes[node].y_min, ymid),
                2 => (self.nodes[node].x_min, xmid, ymid, self.nodes[node].y_max),
                _ => (xmid, self.nodes[node].x_max, ymid, self.nodes[node].y_max),
            };
            let new_child = self.new_node(cxmin, cxmax, cymin, cymax);
            self.nodes[node].children[quadrant] = new_child;
            self.insert(new_child, idx, px, py, depth + 1);
        } else {
            self.insert(child, idx, px, py, depth + 1);
        }

        // Update center of mass for this node
        let old_mass = self.nodes[node].mass;
        let new_mass = old_mass + 1.0;
        self.nodes[node].cx = (self.nodes[node].cx * old_mass + px) / new_mass;
        self.nodes[node].cy = (self.nodes[node].cy * old_mass + py) / new_mass;
        self.nodes[node].mass = new_mass;
    }

    fn repulsive_force(&self, idx: usize, px: f64, py: f64, k2: f64, theta: f64) -> (f64, f64) {
        if self.nodes.is_empty() { return (0.0, 0.0); }
        self.repulsive_recursive(0, idx, px, py, k2, theta)
    }

    fn repulsive_recursive(&self, node: usize, idx: usize, px: f64, py: f64, k2: f64, theta: f64) -> (f64, f64) {
        let nd = &self.nodes[node];
        if nd.mass == 0.0 { return (0.0, 0.0); }

        let ddx = px - nd.cx;
        let ddy = py - nd.cy;
        let dist2 = ddx * ddx + ddy * ddy;

        if nd.is_leaf {
            if nd.point_idx == idx { return (0.0, 0.0); }
            if dist2 < 1e-4 { return (0.0, 0.0); }
            let dist = dist2.sqrt();
            let force = k2 / dist;
            return (ddx / dist * force, ddy / dist * force);
        }

        let size = nd.x_max - nd.x_min;
        if size * size / dist2 < theta * theta {
            // Use approximation
            if dist2 < 1e-4 { return (0.0, 0.0); }
            let dist = dist2.sqrt();
            let force = k2 * nd.mass / dist;
            return (ddx / dist * force, ddy / dist * force);
        }

        // Recurse into children
        let mut fx = 0.0;
        let mut fy = 0.0;
        for q in 0..4 {
            let child = nd.children[q];
            if child != 0 {
                let (cfx, cfy) = self.repulsive_recursive(child, idx, px, py, k2, theta);
                fx += cfx;
                fy += cfy;
            }
        }
        (fx, fy)
    }
}
