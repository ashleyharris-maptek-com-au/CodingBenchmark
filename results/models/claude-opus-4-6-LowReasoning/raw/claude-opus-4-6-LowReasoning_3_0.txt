use std::io::{self, Read, Write, BufWriter};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let stdout = io::stdout();
    let mut out = BufWriter::new(stdout.lock());

    let mut iter = input.split_ascii_whitespace();
    let n: usize = iter.next().unwrap().parse().unwrap();
    let m: usize = iter.next().unwrap().parse().unwrap();

    let mut adj: Vec<Vec<usize>> = vec![vec![]; n];
    for _ in 0..m {
        let u: usize = iter.next().unwrap().parse().unwrap();
        let v: usize = iter.next().unwrap().parse().unwrap();
        if u == v || u >= n || v >= n { continue; }
        adj[u].push(v);
        adj[v].push(u);
    }

    // Deduplicate adjacency lists
    for i in 0..n {
        adj[i].sort_unstable();
        adj[i].dedup();
    }

    let mut x = vec![0.0f64; n];
    let mut y = vec![0.0f64; n];

    // Find connected components via BFS
    let mut comp = vec![usize::MAX; n];
    let mut components: Vec<Vec<usize>> = Vec::new();
    let mut queue = Vec::new();
    for start in 0..n {
        if comp[start] != usize::MAX { continue; }
        let c = components.len();
        let mut nodes = Vec::new();
        comp[start] = c;
        queue.push(start);
        while let Some(u) = queue.pop() {
            nodes.push(u);
            for &v in &adj[u] {
                if comp[v] == usize::MAX {
                    comp[v] = c;
                    queue.push(v);
                }
            }
        }
        components.push(nodes);
    }

    // Layout each component, then pack them
    let mut comp_positions: Vec<(Vec<usize>, Vec<f64>, Vec<f64>)> = Vec::new();

    for nodes in &components {
        let cn = nodes.len();
        if cn == 1 {
            comp_positions.push((nodes.clone(), vec![0.0], vec![0.0]));
            continue;
        }

        // Map global -> local
        let mut local_id = vec![0usize; n];
        for (i, &node) in nodes.iter().enumerate() {
            local_id[node] = i;
        }
        let mut local_adj: Vec<Vec<usize>> = vec![vec![]; cn];
        for &node in nodes {
            for &nb in &adj[node] {
                local_adj[local_id[node]].push(local_id[nb]);
            }
        }

        let (cx, cy) = layout_component(cn, &local_adj);
        comp_positions.push((nodes.clone(), cx, cy));
    }

    // Pack components in a row
    let mut offset_x = 0.0f64;
    for (nodes, cx, cy) in &comp_positions {
        let cn = nodes.len();
        let min_x = cx.iter().cloned().fold(f64::INFINITY, f64::min);
        let max_x = cx.iter().cloned().fold(f64::NEG_INFINITY, f64::max);
        for i in 0..cn {
            let node = nodes[i];
            x[node] = cx[i] - min_x + offset_x;
            y[node] = cy[i];
        }
        offset_x += (max_x - min_x) + 10.0;
    }

    writeln!(out, "{}", n).unwrap();
    for i in 0..n {
        writeln!(out, "{:.6} {:.6}", x[i], y[i]).unwrap();
    }
}

fn layout_component(n: usize, adj: &[Vec<usize>]) -> (Vec<f64>, Vec<f64>) {
    if n <= 2 {
        let mut cx = vec![0.0; n];
        let mut cy = vec![0.0; n];
        if n == 2 { cx[1] = 1.0; }
        return (cx, cy);
    }

    // Use multi-level force-directed (Fruchterman-Reingold style)
    // For very large graphs, use spectral initialization + limited FR iterations

    let mut cx = vec![0.0f64; n];
    let mut cy = vec![0.0f64; n];

    // Spectral-like initialization using BFS layering from a pseudo-peripheral node
    let start = find_pseudo_peripheral(n, adj);
    let dist = bfs_dist(n, adj, start);
    let max_dist = *dist.iter().max().unwrap() as f64;

    // Find another pseudo-peripheral from the far end
    let far = dist.iter().enumerate().max_by_key(|&(_, &d)| d).unwrap().0;
    let dist2 = bfs_dist(n, adj, far);

    // Use dist and dist2 as two coordinates (like spectral embedding)
    for i in 0..n {
        cx[i] = dist[i] as f64;
        cy[i] = dist2[i] as f64;
    }

    // Add small perturbation to break ties
    let mut rng: u64 = 12345;
    for i in 0..n {
        rng = rng.wrapping_mul(6364136223846793005).wrapping_add(1);
        let r1 = ((rng >> 33) as f64) / (1u64 << 31) as f64 - 0.5;
        rng = rng.wrapping_mul(6364136223846793005).wrapping_add(1);
        let r2 = ((rng >> 33) as f64) / (1u64 << 31) as f64 - 0.5;
        cx[i] += r1 * 0.1;
        cy[i] += r2 * 0.1;
    }

    // Scale to reasonable area
    let area = (n as f64).sqrt();
    let k = area / (max_dist.max(1.0));
    for i in 0..n {
        cx[i] *= k;
        cy[i] *= k;
    }

    // Fruchterman-Reingold iterations
    let ideal_len = (1000.0 * 1000.0 / n as f64).sqrt();
    let iterations = if n <= 50 {
        300
    } else if n <= 500 {
        150
    } else if n <= 5000 {
        80
    } else if n <= 50000 {
        30
    } else {
        15
    };

    // For large graphs, use grid-based repulsion approximation
    let use_grid = n > 500;

    let mut temp = ideal_len * 2.0;
    let cool = temp / (iterations as f64 + 1.0);

    let mut dx = vec![0.0f64; n];
    let mut dy = vec![0.0f64; n];

    for iter in 0..iterations {
        for i in 0..n { dx[i] = 0.0; dy[i] = 0.0; }

        // Repulsive forces
        if use_grid {
            // Grid-based approximation
            grid_repulsion(n, &cx, &cy, ideal_len, &mut dx, &mut dy);
        } else {
            // O(n^2) exact
            for i in 0..n {
                for j in (i+1)..n {
                    let ddx = cx[i] - cx[j];
                    let ddy = cy[i] - cy[j];
                    let dist_sq = ddx * ddx + ddy * ddy;
                    let dist = dist_sq.sqrt().max(0.01);
                    let force = ideal_len * ideal_len / dist;
                    let fx = ddx / dist * force;
                    let fy = ddy / dist * force;
                    dx[i] += fx;
                    dy[i] += fy;
                    dx[j] -= fx;
                    dy[j] -= fy;
                }
            }
        }

        // Attractive forces (along edges)
        for u in 0..n {
            for &v in &adj[u] {
                if v > u { continue; }
                let ddx = cx[u] - cx[v];
                let ddy = cy[u] - cy[v];
                let dist = (ddx * ddx + ddy * ddy).sqrt().max(0.01);
                let force = dist * dist / ideal_len;
                let fx = ddx / dist * force;
                let fy = ddy / dist * force;
                dx[u] -= fx;
                dy[u] -= fy;
                dx[v] += fx;
                dy[v] += fy;
            }
        }

        // Apply with temperature limiting
        for i in 0..n {
            let disp = (dx[i] * dx[i] + dy[i] * dy[i]).sqrt().max(0.01);
            let scale = temp.min(disp) / disp;
            cx[i] += dx[i] * scale;
            cy[i] += dy[i] * scale;
        }

        temp -= cool;
        if temp < 0.01 { temp = 0.01; }

        // Early exit check for large graphs (time budget)
        if n > 10000 && iter > 10 { break; }
    }

    (cx, cy)
}

fn grid_repulsion(n: usize, cx: &[f64], cy: &[f64], k: f64, dx: &mut [f64], dy: &mut [f64]) {
    // Find bounds
    let mut min_x = f64::INFINITY;
    let mut max_x = f64::NEG_INFINITY;
    let mut min_y = f64::INFINITY;
    let mut max_y = f64::NEG_INFINITY;
    for i in 0..n {
        min_x = min_x.min(cx[i]);
        max_x = max_x.max(cx[i]);
        min_y = min_y.min(cy[i]);
        max_y = max_y.max(cy[i]);
    }

    let range = 2.0 * k; // interaction range
    let cell_size = range;
    if cell_size <= 0.0 { return; }

    let cols = ((max_x - min_x) / cell_size) as usize + 1;
    let rows = ((max_y - min_y) / cell_size) as usize + 1;
    let grid_size = cols.checked_mul(rows).unwrap_or(0);

    // If grid is too large or degenerate, fall back to sampling
    if grid_size > 10_000_000 || grid_size == 0 || cols == 0 || rows == 0 {
        // Sample-based repulsion
        sample_repulsion(n, cx, cy, k, dx, dy);
        return;
    }

    let mut grid: Vec<Vec<usize>> = vec![vec![]; grid_size];
    for i in 0..n {
        let col = ((cx[i] - min_x) / cell_size) as usize;
        let row = ((cy[i] - min_y) / cell_size) as usize;
        let col = col.min(cols - 1);
        let row = row.min(rows - 1);
        grid[row * cols + col].push(i);
    }

    for row in 0..rows {
        for col in 0..cols {
            let idx = row * cols + col;
            if grid[idx].is_empty() { continue; }

            // Check 3x3 neighborhood
            let r_start = row.saturating_sub(1);
            let r_end = (row + 2).min(rows);
            let c_start = col.saturating_sub(1);
            let c_end = (col + 2).min(cols);

            for &i in &grid[idx] {
                for nr in r_start..r_end {
                    for nc in c_start..c_end {
                        let nidx = nr * cols + nc;
                        for &j in &grid[nidx] {
                            if j <= i { continue; }
                            let ddx = cx[i] - cx[j];
                            let ddy = cy[i] - cy[j];
                            let dist_sq = ddx * ddx + ddy * ddy;
                            if dist_sq > range * range { continue; }
                            let dist = dist_sq.sqrt().max(0.01);
                            let force = k * k / dist;
                            let fx = ddx / dist * force;
                            let fy = ddy / dist * force;
                            dx[i] += fx;
                            dy[i] += fy;
                            dx[j] -= fx;
                            dy[j] -= fy;
                        }
                    }
                }
            }
        }
    }
}

fn sample_repulsion(n: usize, cx: &[f64], cy: &[f64], k: f64, dx: &mut [f64], dy: &mut [f64]) {
    // Each node repels a random sample of others
    let samples = 50.min(n - 1);
    let scale = n as f64 / samples as f64;
    let mut rng: u64 = 54321;
    for i in 0..n {
        for _ in 0..samples {
            rng = rng.wrapping_mul(6364136223846793005).wrapping_add(1);
            let j = ((rng >> 33) as usize) % n;
            if j == i { continue; }
            let ddx = cx[i] - cx[j];
            let ddy = cy[i] - cy[j];
            let dist = (ddx * ddx + ddy * ddy).sqrt().max(0.01);
            let force = k * k / dist * scale;
            let fx = ddx / dist * force;
            let fy = ddy / dist * force;
            dx[i] += fx;
            dy[i] += fy;
        }
    }
}

fn find_pseudo_peripheral(n: usize, adj: &[Vec<usize>]) -> usize {
    let mut node = 0;
    for _ in 0..3 {
        let dist = bfs_dist(n, adj, node);
        node = dist.iter().enumerate().max_by_key(|&(_, &d)| d).unwrap().0;
    }
    node
}

fn bfs_dist(n: usize, adj: &[Vec<usize>], start: usize) -> Vec<usize> {
    let mut dist = vec![usize::MAX; n];
    dist[start] = 0;
    let mut queue = std::collections::VecDeque::new();
    queue.push_back(start);
    while let Some(u) = queue.pop_front() {
        for &v in &adj[u] {
            if dist[v] == usize::MAX {
                dist[v] = dist[u] + 1;
                queue.push_back(v);
            }
        }
    }
    // Handle disconnected nodes (shouldn't happen within a component but be safe)
    for d in &mut dist {
        if *d == usize::MAX { *d = 0; }
    }
    dist
}
