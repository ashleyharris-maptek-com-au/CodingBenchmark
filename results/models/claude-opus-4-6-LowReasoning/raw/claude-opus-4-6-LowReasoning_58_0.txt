const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { grid, robots, jobs } = data;
const rows = grid.length;
const cols = grid[0].length;

// BFS shortest distance from a source point to all reachable cells
function bfsFrom(sx, sy) {
  const dist = new Int32Array(rows * cols).fill(-1);
  dist[sy * cols + sx] = 0;
  const q = [sx, sy];
  let head = 0;
  const dx = [1, -1, 0, 0];
  const dy = [0, 0, 1, -1];
  while (head < q.length) {
    const cx = q[head++];
    const cy = q[head++];
    const cd = dist[cy * cols + cx];
    for (let d = 0; d < 4; d++) {
      const nx = cx + dx[d];
      const ny = cy + dy[d];
      if (nx >= 0 && nx < cols && ny >= 0 && ny < rows && grid[ny][nx] === '.' && dist[ny * cols + nx] === -1) {
        dist[ny * cols + nx] = cd + 1;
        q.push(nx, ny);
      }
    }
  }
  return dist;
}

const nR = robots.length;
const nJ = jobs.length;

// Compute BFS from each robot and each job location
const robotDist = robots.map(r => bfsFrom(r.x, r.y));
const jobDist = jobs.map(j => bfsFrom(j.x, j.y));

// Distance between robot r and job j
function distRJ(r, j) {
  const job = jobs[j];
  return robotDist[r][job.y * cols + job.x];
}

// Distance between job a and job b
function distJJ(a, b) {
  const jb = jobs[b];
  return jobDist[a][jb.y * cols + jb.x];
}

// Sort jobs by priority descending
const sortedJobs = jobs.map((j, i) => i).sort((a, b) => jobs[b].priority - jobs[a].priority);

// Greedy assignment: for each job (highest priority first), assign to the robot
// that can reach it with minimum additional travel (from its current position)
const assignments = Array.from({ length: nR }, () => []);
const robotPos = robots.map((r, i) => ({ x: r.x, y: r.y, distMap: robotDist[i] }));
// Track current position as last assigned job (or starting position)
const robotLastJob = new Array(nR).fill(-1);
const robotTravelSoFar = new Float64Array(nR);

for (const ji of sortedJobs) {
  let bestRobot = 0;
  let bestCost = Infinity;
  for (let r = 0; r < nR; r++) {
    let d;
    if (robotLastJob[r] === -1) {
      d = distRJ(r, ji);
    } else {
      d = distJJ(robotLastJob[r], ji);
    }
    if (d < 0) d = 1e9; // unreachable
    // Favor robots with less total travel so far
    const cost = d + robotTravelSoFar[r] * 0.01;
    if (cost < bestCost) {
      bestCost = cost;
      bestRobot = r;
    }
  }
  let d;
  if (robotLastJob[bestRobot] === -1) {
    d = distRJ(bestRobot, ji);
  } else {
    d = distJJ(robotLastJob[bestRobot], ji);
  }
  if (d < 0) d = 1e9;
  robotTravelSoFar[bestRobot] += d;
  robotLastJob[bestRobot] = ji;
  assignments[bestRobot].push(jobs[ji].id);
}
