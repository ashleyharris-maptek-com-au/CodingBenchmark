declare const require: any; const fs = require('fs'); const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const grid: string[] = data.grid;
const robots: {id: number; x: number; y: number}[] = data.robots;
const jobs: {id: number; x: number; y: number; priority: number}[] = data.jobs;
const rows = grid.length;
const cols = grid[0].length;

function bfs(sx: number, sy: number): Int32Array {
  const dist = new Int32Array(rows * cols).fill(-1);
  dist[sy * cols + sx] = 0;
  const q: number[] = [sx, sy];
  let head = 0;
  const dx = [1, -1, 0, 0];
  const dy = [0, 0, 1, -1];
  while (head < q.length) {
    const cx = q[head++];
    const cy = q[head++];
    const cd = dist[cy * cols + cx];
    for (let d = 0; d < 4; d++) {
      const nx = cx + dx[d];
      const ny = cy + dy[d];
      if (nx >= 0 && nx < cols && ny >= 0 && ny < rows && grid[ny][nx] === '.' && dist[ny * cols + nx] === -1) {
        dist[ny * cols + nx] = cd + 1;
        q.push(nx, ny);
      }
    }
  }
  return dist;
}

// Precompute BFS from each robot and each job position
const robotDists: Int32Array[] = robots.map(r => bfs(r.x, r.y));
const jobDists: Int32Array[] = jobs.map(j => bfs(j.x, j.y));

function getDist(distMap: Int32Array, x: number, y: number): number {
  const d = distMap[y * cols + x];
  return d === -1 ? 1e9 : d;
}

// Greedy assignment: repeatedly pick the best (robot, job) pair considering priority-weighted cost
// Each robot tracks current position (last assigned job's position)
const nRobots = robots.length;
const nJobs = jobs.length;

const assignments: number[][] = new Array(nRobots);
for (let i = 0; i < nRobots; i++) assignments[i] = [];

const assigned = new Uint8Array(nJobs);
// Track current position of each robot as index into jobs array (-1 = starting pos)
const robotCurJob: number[] = new Array(nRobots).fill(-1);

// Sort jobs by priority descending for greedy
const jobOrder = jobs.map((_, i) => i).sort((a, b) => jobs[b].priority - jobs[a].priority);

for (const ji of jobOrder) {
  if (assigned[ji]) continue;
  let bestRobot = 0;
  let bestDist = Infinity;
  for (let ri = 0; ri < nRobots; ri++) {
    let d: number;
    const prev = robotCurJob[ri];
    if (prev === -1) {
      d = getDist(robotDists[ri], jobs[ji].x, jobs[ji].y);
    } else {
      d = getDist(jobDists[prev], jobs[ji].x, jobs[ji].y);
    }
    // Bias toward robots with fewer jobs to balance load
    const load = assignments[ri].length;
    const score = d + load * 2;
    if (score < bestDist) {
      bestDist = score;
      bestRobot = ri;
    }
  }
  assignments[bestRobot].push(jobs[ji].id);
  assigned[ji] = 1;
  robotCurJob[bestRobot] = ji;
}

// Build output indexed by robot id
const maxId = robots.reduce((m, r) => Math.max(m, r.id), 0);
const result: number[][] = new Array(maxId + 1);
for (let i = 0; i <= maxId; i++) result[i] = [];
for (let ri = 0; ri < nRobots; ri++) {
  result[robots[ri].id] = assignments[ri];
}

console.log(JSON.stringify({assignments: result}));
