use std::io::{self, BufRead, Write, BufWriter};
use std::collections::HashMap;

struct Sample {
    hole_idx: usize,
    sample_idx: usize,
    x: f64, y: f64, z: f64,
    values: Vec<f64>,
}

fn main() {
    let stdin = io::stdin();
    let stdout = io::stdout();
    let mut out = BufWriter::new(stdout.lock());
    let mut lines = stdin.lock().lines();

    let first = lines.next().unwrap().unwrap();
    let mut it = first.split_whitespace();
    let num_holes: usize = it.next().unwrap().parse().unwrap();
    let num_props: usize = it.next().unwrap().parse().unwrap();

    let prop_line = lines.next().unwrap().unwrap();
    let prop_names: Vec<String> = prop_line.split_whitespace().map(|s| s.to_string()).collect();

    let mut holes: Vec<String> = Vec::with_capacity(num_holes);
    let mut all_samples: Vec<Sample> = Vec::new();

    for h in 0..num_holes {
        let line = match lines.next() { Some(Ok(l)) => l, _ => break };
        let parts: Vec<&str> = line.split_whitespace().collect();
        if parts.len() < 9 { continue; }
        let hole_id = parts[0].to_string();
        let sx: f64 = parts[1].parse().unwrap_or(0.0);
        let sy: f64 = parts[2].parse().unwrap_or(0.0);
        let sz: f64 = parts[3].parse().unwrap_or(0.0);
        let dx: f64 = parts[4].parse().unwrap_or(0.0);
        let dy: f64 = parts[5].parse().unwrap_or(0.0);
        let dz: f64 = parts[6].parse().unwrap_or(0.0);
        let _length: f64 = parts[7].parse().unwrap_or(0.0);
        let ns: usize = parts[8].parse().unwrap_or(0);

        let dir_len = (dx*dx + dy*dy + dz*dz).sqrt().max(1e-12);
        let ndx = dx / dir_len;
        let ndy = dy / dir_len;
        let ndz = dz / dir_len;

        holes.push(hole_id);

        for s in 0..ns {
            let sline = match lines.next() { Some(Ok(l)) => l, _ => break };
            let sp: Vec<&str> = sline.split_whitespace().collect();
            if sp.is_empty() { continue; }
            let depth: f64 = sp[0].parse().unwrap_or(0.0);
            let mut values = Vec::with_capacity(num_props);
            for p in 0..num_props {
                let v = if p + 1 < sp.len() { sp[p+1].parse().unwrap_or(f64::NAN) } else { f64::NAN };
                values.push(v);
            }
            all_samples.push(Sample {
                hole_idx: h,
                sample_idx: s,
                x: sx + ndx * depth,
                y: sy + ndy * depth,
                z: sz + ndz * depth,
                values,
            });
        }
    }

    let n = all_samples.len();
    if n == 0 || num_props == 0 {
        writeln!(out, "0").unwrap();
        return;
    }

    // Compute per-property stats (mean, std) ignoring NaN
    let mut means = vec![0.0f64; num_props];
    let mut stds = vec![0.0f64; num_props];
    let mut counts = vec![0usize; num_props];
    for s in &all_samples {
        for p in 0..num_props {
            let v = s.values[p];
            if v.is_finite() {
                means[p] += v;
                counts[p] += 1;
            }
        }
    }
    for p in 0..num_props {
        if counts[p] > 0 { means[p] /= counts[p] as f64; }
    }
    for s in &all_samples {
        for p in 0..num_props {
            let v = s.values[p];
            if v.is_finite() {
                let d = v - means[p];
                stds[p] += d * d;
            }
        }
    }
    for p in 0..num_props {
        if counts[p] > 1 { stds[p] = (stds[p] / (counts[p] - 1) as f64).sqrt(); }
        else { stds[p] = 1.0; }
        if stds[p] < 1e-15 { stds[p] = 1.0; }
    }

    // Compute robust stats (median, MAD) for better outlier detection
    let mut medians = vec![0.0f64; num_props];
    let mut mads = vec![1.0f64; num_props];
    for p in 0..num_props {
        let mut vals: Vec<f64> = all_samples.iter().map(|s| s.values[p]).filter(|v| v.is_finite()).collect();
        if vals.is_empty() { continue; }
        vals.sort_by(|a,b| a.partial_cmp(b).unwrap());
        let med = vals[vals.len()/2];
        medians[p] = med;
        let mut abs_devs: Vec<f64> = vals.iter().map(|v| (v - med).abs()).collect();
        abs_devs.sort_by(|a,b| a.partial_cmp(b).unwrap());
        let mad = abs_devs[abs_devs.len()/2] * 1.4826;
        mads[p] = if mad < 1e-15 { stds[p].max(1e-15) } else { mad };
    }

    // Spatial neighbor search: use grid-based approach
    // Determine k neighbors based on scale
    let k = if n < 100 { n.min(8) } else if n < 10000 { 15 } else { 20 };

    // Build spatial grid
    let mut min_x = f64::MAX; let mut max_x = f64::MIN;
    let mut min_y = f64::MAX; let mut max_y = f64::MIN;
    let mut min_z = f64::MAX; let mut max_z = f64::MIN;
    for s in &all_samples {
        if s.x < min_x { min_x = s.x; } if s.x > max_x { max_x = s.x; }
        if s.y < min_y { min_y = s.y; } if s.y > max_y { max_y = s.y; }
        if s.z < min_z { min_z = s.z; } if s.z > max_z { max_z = s.z; }
    }

    // Adaptive grid size
    let grid_dim = if n < 1000 { 10usize } else if n < 100000 { 50 } else { 100 };
    let cell_x = ((max_x - min_x) / grid_dim as f64).max(1e-10);
    let cell_y = ((max_y - min_y) / grid_dim as f64).max(1e-10);
    let cell_z = ((max_z - min_z) / grid_dim as f64).max(1e-10);

    let mut grid: HashMap<(usize,usize,usize), Vec<usize>> = HashMap::new();
    for (i, s) in all_samples.iter().enumerate() {
        let gx = ((s.x - min_x) / cell_x).min((grid_dim-1) as f64) as usize;
        let gy = ((s.y - min_y) / cell_y).min((grid_dim-1) as f64) as usize;
        let gz = ((s.z - min_z) / cell_z).min((grid_dim-1) as f64) as usize;
        grid.entry((gx,gy,gz)).or_default().push(i);
    }

    // For each sample, find k nearest neighbors and compute local outlier score
    struct Suspect {
        hole_idx: usize,
        sample_idx: usize,
        prop_idx: usize,
        confidence: f64,
    }
    let mut suspects: Vec<Suspect> = Vec::new();

    for i in 0..n {
        let s = &all_samples[i];
        let gx = ((s.x - min_x) / cell_x).min((grid_dim-1) as f64) as usize;
        let gy = ((s.y - min_y) / cell_y).min((grid_dim-1) as f64) as usize;
        let gz = ((s.z - min_z) / cell_z).min((grid_dim-1) as f64) as usize;

        // Search expanding radius until we have enough neighbors
        let mut neighbors: Vec<(f64, usize)> = Vec::new();
        let mut radius = 1usize;
        loop {
            neighbors.clear();
            let r = radius;
            let gx_lo = gx.saturating_sub(r);
            let gy_lo = gy.saturating_sub(r);
            let gz_lo = gz.saturating_sub(r);
            let gx_hi = (gx + r).min(grid_dim - 1);
            let gy_hi = (gy + r).min(grid_dim - 1);
            let gz_hi = (gz + r).min(grid_dim - 1);
            for cx in gx_lo..=gx_hi {
                for cy in gy_lo..=gy_hi {
                    for cz in gz_lo..=gz_hi {
                        if let Some(indices) = grid.get(&(cx,cy,cz)) {
                            for &j in indices {
                                if j == i { continue; }
                                let o = &all_samples[j];
                                let dx = s.x - o.x;
                                let dy = s.y - o.y;
                                let dz = s.z - o.z;
                                let dist2 = dx*dx + dy*dy + dz*dz;
                                neighbors.push((dist2, j));
                            }
                        }
                    }
                }
            }
            if neighbors.len() >= k || radius >= grid_dim { break; }
            radius += 1;
        }

        neighbors.sort_by(|a,b| a.0.partial_cmp(&b.0).unwrap());
        let use_k = neighbors.len().min(k);
        if use_k < 3 { continue; }

        // IDW weights
        let mut weights: Vec<f64> = Vec::with_capacity(use_k);
        for ni in 0..use_k {
            let d = neighbors[ni].0.sqrt().max(1e-10);
            weights.push(1.0 / d);
        }
        let wsum: f64 = weights.iter().sum();

        for p in 0..num_props {
            let v = s.values[p];
            if !v.is_finite() { continue; }

            // Weighted local mean
            let mut local_mean = 0.0f64;
            let mut local_var = 0.0f64;
            let mut valid_w = 0.0f64;
            let mut valid_count = 0usize;
            for ni in 0..use_k {
                let nv = all_samples[neighbors[ni].1].values[p];
                if !nv.is_finite() { continue; }
                local_mean += weights[ni] * nv;
                valid_w += weights[ni];
                valid_count += 1;
            }
            if valid_count < 3 { continue; }
            local_mean /= valid_w;

            for ni in 0..use_k {
                let nv = all_samples[neighbors[ni].1].values[p];
                if !nv.is_finite() { continue; }
                let d = nv - local_mean;
                local_var += weights[ni] * d * d;
            }
            local_var /= valid_w;
            let local_std = local_var.sqrt().max(mads[p] * 0.1);

            let local_z = (v - local_mean).abs() / local_std;
            let global_z = (v - medians[p]).abs() / mads[p];

            // Combined score: spatial outlier that is also globally unusual
            let score = local_z * 0.7 + global_z * 0.3;

            if score > 3.5 {
                suspects.push(Suspect {
                    hole_idx: s.hole_idx,
                    sample_idx: s.sample_idx,
                    prop_idx: p,
                    confidence: (score * 100.0).round() / 100.0,
                });
            }
        }
    }

    suspects.sort_by(|a,b| b.confidence.partial_cmp(&a.confidence).unwrap());

    writeln!(out, "{}", suspects.len()).unwrap();
    for s in &suspects {
        writeln!(out, "{} {} {} {:.2}", holes[s.hole_idx], s.sample_idx, prop_names[s.prop_idx], s.confidence).unwrap();
    }
}
