using System;
using System.Collections.Generic;
using System.Globalization;
using System.Linq;

class OrbitalTSP
{
    const double MU = 398600.4418;
    const double PI = Math.PI;

    struct Vec3
    {
        public double X, Y, Z;
        public Vec3(double x, double y, double z) { X = x; Y = y; Z = z; }
        public static Vec3 operator +(Vec3 a, Vec3 b) => new Vec3(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
        public static Vec3 operator -(Vec3 a, Vec3 b) => new Vec3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
        public static Vec3 operator *(double s, Vec3 a) => new Vec3(s * a.X, s * a.Y, s * a.Z);
        public double Mag() => Math.Sqrt(X * X + Y * Y + Z * Z);
        public Vec3 Norm() { double m = Mag(); return m > 0 ? new Vec3(X / m, Y / m, Z / m) : this; }
        public static double Dot(Vec3 a, Vec3 b) => a.X * b.X + a.Y * b.Y + a.Z * b.Z;
        public static Vec3 Cross(Vec3 a, Vec3 b) => new Vec3(
            a.Y * b.Z - a.Z * b.Y, a.Z * b.X - a.X * b.Z, a.X * b.Y - a.Y * b.X);
    }

    struct Orbit
    {
        public double A, E, Inc, Raan, ArgP, Nu;
        public Vec3 R, V;
    }

    static Orbit StateToOrbit(Vec3 r, Vec3 v, double mu)
    {
        var o = new Orbit { R = r, V = v };
        double rm = r.Mag(), vm = v.Mag();
        var h = Vec3.Cross(r, v);
        double hm = h.Mag();
        var n = Vec3.Cross(new Vec3(0, 0, 1), h);
        double nm = n.Mag();
        var ev = (1.0 / mu) * ((vm * vm - mu / rm) * 1.0 * r - Vec3.Dot(r, v) * 1.0 * v);
        double e = new Vec3(ev.X, ev.Y, ev.Z).Mag();
        double energy = vm * vm / 2.0 - mu / rm;
        double a = e < 1.0 - 1e-10 ? -mu / (2.0 * energy) : rm;
        double inc = Math.Acos(Math.Max(-1, Math.Min(1, h.Z / hm)));
        double raan = 0, argp = 0, nu = 0;
        if (nm > 1e-10)
        {
            raan = Math.Acos(Math.Max(-1, Math.Min(1, n.X / nm)));
            if (n.Y < 0) raan = 2 * PI - raan;
        }
        if (e > 1e-10 && nm > 1e-10)
        {
            argp = Math.Acos(Math.Max(-1, Math.Min(1, Vec3.Dot(n, ev) / (nm * e))));
            if (ev.Z < 0) argp = 2 * PI - argp;
            nu = Math.Acos(Math.Max(-1, Math.Min(1, Vec3.Dot(ev, r) / (e * rm))));
            if (Vec3.Dot(r, v) < 0) nu = 2 * PI - nu;
        }
        else if (e > 1e-10)
        {
            argp = Math.Atan2(ev.Y, ev.X);
            if (inc > PI / 2) argp = -argp;
            nu = Math.Acos(Math.Max(-1, Math.Min(1, Vec3.Dot(ev, r) / (e * rm))));
            if (Vec3.Dot(r, v) < 0) nu = 2 * PI - nu;
        }
        o.A = a; o.E = e; o.Inc = inc; o.Raan = raan; o.ArgP = argp; o.Nu = nu;
        return o;
    }

    static double EstimateDeltaV(Orbit from, Orbit to, double mu)
    {
        double a1 = from.A, a2 = to.A;
        if (a1 <= 0 || a2 <= 0) a1 = from.R.Mag(); // fallback
        if (a2 <= 0) a2 = to.R.Mag();

        // Hohmann-like transfer cost
        double r1 = a1 * (1 - from.E * from.E) / (1 + from.E);
        double r2 = a2 * (1 - to.E * to.E) / (1 + to.E);
        if (r1 <= 0) r1 = from.R.Mag();
        if (r2 <= 0) r2 = to.R.Mag();

        double at = (r1 + r2) / 2.0;
        double dv1 = Math.Abs(Math.Sqrt(2 * mu / r1 - mu / at) - Math.Sqrt(mu / r1));
        double dv2 = Math.Abs(Math.Sqrt(mu / r2) - Math.Sqrt(2 * mu / r2 - mu / at));
        if (double.IsNaN(dv1)) dv1 = 0;
        if (double.IsNaN(dv2)) dv2 = 0;
        double hohmann = dv1 + dv2;

        // Plane change cost
        double di = Math.Abs(from.Inc - to.Inc);
        double dRaan = Math.Abs(from.Raan - to.Raan);
        if (dRaan > PI) dRaan = 2 * PI - dRaan;
        double cosAngle = Math.Cos(from.Inc) * Math.Cos(to.Inc) +
                          Math.Sin(from.Inc) * Math.Sin(to.Inc) * Math.Cos(dRaan);
        cosAngle = Math.Max(-1, Math.Min(1, cosAngle));
        double planeAngle = Math.Acos(cosAngle);
        double vAvg = Math.Sqrt(mu / Math.Max(r1, r2));
        double planeChange = 2 * vAvg * Math.Sin(planeAngle / 2);

        return Math.Sqrt(hohmann * hohmann + planeChange * planeChange);
    }

    static int[] NearestNeighbor(double[,] cost, int n, int start)
    {
        var visited = new bool[n];
        var route = new int[n];
        route[0] = start;
        visited[start] = true;
        for (int i = 1; i < n; i++)
        {
            int best = -1; double bestCost = double.MaxValue;
            for (int j = 0; j < n; j++)
            {
                if (!visited[j] && cost[route[i - 1], j] < bestCost)
                { bestCost = cost[route[i - 1], j]; best = j; }
            }
            route[i] = best; visited[best] = true;
        }
        return route;
    }

    static double RouteCost(int[] route, double[,] cost, int startIdx, int n)
    {
        double total = cost[startIdx, route[0]];
        for (int i = 0; i < n - 1; i++) total += cost[route[i], route[i + 1]];
        return total;
    }

    static void TwoOpt(int[] route, double[,] cost, int startIdx, int n, int maxIter)
    {
        for (int iter = 0; iter < maxIter; iter++)
        {
            bool improved = false;
            for (int i = 0; i < n - 1; i++)
            {
                for (int j = i + 1; j < n; j++)
                {
                    double prevI = i == 0 ? cost[startIdx, route[0]] : cost[route[i - 1], route[i]];
                    double prevJ = j < n - 1 ? cost[route[j], route[j + 1]] : 0;
                    double newI = i == 0 ? cost[startIdx, route[j]] : cost[route[i - 1], route[j]];
                    double newJ = j < n - 1 ? cost[route[i], route[j + 1]] : 0;
                    if (newI + newJ < prevI + prevJ - 1e-10)
                    {
                        Array.Reverse(route, i, j - i + 1);
                        improved = true;
                    }
                }
            }
            if (!improved) break;
        }
    }

    static void OrOpt(int[] route, double[,] cost, int startIdx, int n)
    {
        bool improved = true;
        while (improved)
        {
            improved = false;
            for (int segLen = 1; segLen <= 3; segLen++)
            {
                for (int i = 0; i < n - segLen; i++)
                {
                    double removeCostBefore = i == 0 ? cost[startIdx, route[i]] : cost[route[i - 1], route[i]];
                    double removeCostAfter = i + segLen < n ? cost[route[i + segLen - 1], route[i + segLen]] : 0;
                    double removeBridge = i == 0 ? (i + segLen < n ? cost[startIdx, route[i + segLen]] : 0)
                        : (i + segLen < n ? cost[route[i - 1], route[i + segLen]] : 0);
                    double removeGain = removeCostBefore + removeCostAfter - removeBridge;

                    for (int j = 0; j < n - segLen; j++)
                    {
                        if (j >= i && j < i + segLen) continue;
                        int aj = j >= i + segLen ? j - segLen + (i == 0 ? 0 : 0) : j;
                        // simplified: just try inserting segment [i..i+segLen-1] after position j
                        // Too complex to get indices right with removal; use simpler approach
                    }
                    // Skip or-opt for simplicity, 2-opt is sufficient
                    break;
                }
                break;
            }
        }
    }

    static void Main(string[] args)
    {
        var line1 = Console.ReadLine().Trim().Split();
        int N = int.Parse(line1[0]);
        double mu = double.Parse(line1[1], CultureInfo.InvariantCulture);

        var startParts = Console.ReadLine().Trim().Split();
        var startR = new Vec3(double.Parse(startParts[0], CultureInfo.InvariantCulture),
                              double.Parse(startParts[1], CultureInfo.InvariantCulture),
                              double.Parse(startParts[2], CultureInfo.InvariantCulture));
        var startV = new Vec3(double.Parse(startParts[3], CultureInfo.InvariantCulture),
                              double.Parse(startParts[4], CultureInfo.InvariantCulture),
                              double.Parse(startParts[5], CultureInfo.InvariantCulture));
        var startOrbit = StateToOrbit(startR, startV, mu);

        var stations = new Orbit[N];
        for (int i = 0; i < N; i++)
        {
            var p = Console.ReadLine().Trim().Split();
            var r = new Vec3(double.Parse(p[0], CultureInfo.InvariantCulture),
                             double.Parse(p[1], CultureInfo.InvariantCulture),
                             double.Parse(p[2], CultureInfo.InvariantCulture));
            var v = new Vec3(double.Parse(p[3], CultureInfo.InvariantCulture),
                             double.Parse(p[4], CultureInfo.InvariantCulture),
                             double.Parse(p[5], CultureInfo.InvariantCulture));
            stations[i] = StateToOrbit(r, v, mu);
        }

        // Build cost matrix (N+1 nodes: index 0=start, 1..N=stations)
        int total = N + 1;
        var cost = new double[total, total];
        var allOrbits = new Orbit[total];
        allOrbits[0] = startOrbit;
        for (int i = 0; i < N; i++) allOrbits[i + 1] = stations[i];

        for (int i = 0; i < total; i++)
            for (int j = 0; j < total; j++)
                cost[i, j] = i == j ? 0 : EstimateDeltaV(allOrbits[i], allOrbits[j], mu);

        if (N <= 1)
        {
            Console.WriteLine(N == 1 ? "0" : "");
            return;
        }

        if (N <= 8)
        {
            // Brute force for small N
            var bestRoute = Enumerable.Range(1, N).ToArray();
            double bestCost = double.MaxValue;
            var perm = Enumerable.Range(1, N).ToArray();
            do
            {
                double c = cost[0, perm[0]];
                for (int i = 0; i < N - 1; i++) c += cost[perm[i], perm[i + 1]];
                if (c < bestCost) { bestCost = c; bestRoute = (int[])perm.Clone(); }
            } while (NextPermutation(perm));
            Console.WriteLine(string.Join(" ", bestRoute.Select(x => x - 1)));
        }
        else
        {
            // Multi-start nearest neighbor + 2-opt
            var stationIndices = Enumerable.Range(1, N).ToArray();
            int[] bestRoute = null;
            double bestCost = double.MaxValue;

            // Try starting from each station via NN
            int tries = Math.Min(N, 20);
            for (int t = 0; t < tries; t++)
            {
                var route = NearestNeighbor(cost, total, 0);
                // Remove start node, keep order
                var stRoute = route.Where(x => x > 0).ToArray();
                TwoOpt(stRoute, cost, 0, N, 100);
                double c = RouteCost(stRoute, cost, 0, N);
                if (c < bestCost) { bestCost = c; bestRoute = stRoute; }
            }

            // Also try NN from different "first stations"
            for (int s = 0; s < Math.Min(N, 30); s++)
            {
                var visited = new bool[N + 1];
                var route = new List<int>();
                int cur = s + 1; // start with station s as first
                route.Add(cur); visited[cur] = true;
                for (int i = 1; i < N; i++)
                {
                    int best = -1; double bc = double.MaxValue;
                    for (int j = 1; j <= N; j++)
                    {
                        if (!visited[j] && cost[cur, j] < bc) { bc = cost[cur, j]; best = j; }
                    }
                    route.Add(best); visited[best] = true; cur = best;
                }
                var arr = route.ToArray();
                TwoOpt(arr, cost, 0, N, 100);
                double c = RouteCost(arr, cost, 0, N);
                if (c < bestCost) { bestCost = c; bestRoute = arr; }
            }

            Console.WriteLine(string.Join(" ", bestRoute.Select(x => x - 1)));
        }
    }

    static bool NextPermutation(int[] arr)
    {
        int n = arr.Length, i = n - 2;
        while (i >= 0 && arr[i] >= arr[i + 1]) i--;
        if (i < 0) return false;
        int j = n - 1;
        while (arr[j] <= arr[i]) j--;
        (arr[i], arr[j]) = (arr[j], arr[i]);
        Array.Reverse(arr, i + 1, n - i - 1);
        return true;
    }
}
