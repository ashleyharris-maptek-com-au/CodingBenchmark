using System;
using System.Collections.Generic;
using System.Linq;

class Program
{
    const double MU = 398600.4418;

    static double[] Cross(double[] a, double[] b) => new double[] {
        a[1]*b[2]-a[2]*b[1], a[2]*b[0]-a[0]*b[2], a[0]*b[1]-a[1]*b[0]
    };
    static double Dot(double[] a, double[] b) => a[0]*b[0]+a[1]*b[1]+a[2]*b[2];
    static double Mag(double[] a) => Math.Sqrt(Dot(a,a));

    struct Orbit { public double a, e, i, raan, argp; }

    static Orbit StateToOrbit(double[] r, double[] v, double mu)
    {
        double rmag = Mag(r), vmag = Mag(v);
        double[] h = Cross(r, v);
        double hmag = Mag(h);
        double[] n = new double[] { -h[1], h[0], 0 };
        double nmag = Mag(n);

        double energy = vmag * vmag / 2.0 - mu / rmag;
        double a = -mu / (2.0 * energy);
        if (a < 0) a = rmag;

        double[] ev = new double[3];
        for (int j = 0; j < 3; j++)
            ev[j] = ((vmag * vmag - mu / rmag) * r[j] - Dot(r, v) * v[j]) / mu;
        double e = Mag(ev);

        double inc = Math.Acos(Math.Max(-1, Math.Min(1, h[2] / hmag)));
        double raan = 0;
        if (nmag > 1e-10) {
            raan = Math.Acos(Math.Max(-1, Math.Min(1, n[0] / nmag)));
            if (n[1] < 0) raan = 2 * Math.PI - raan;
        }
        double argp = 0;
        if (nmag > 1e-10 && e > 1e-10) {
            argp = Math.Acos(Math.Max(-1, Math.Min(1, Dot(n, ev) / (nmag * e))));
            if (ev[2] < 0) argp = 2 * Math.PI - argp;
        }

        return new Orbit { a = a, e = e, i = inc, raan = raan, argp = argp };
    }

    static double TransferCost(Orbit from, Orbit to)
    {
        double a1 = from.a, a2 = to.a;
        if (a1 < 100 || a2 < 100) return 1e9;

        // Hohmann transfer delta-V
        double r1 = a1 * (1 - from.e * from.e) / (1 + from.e);
        double r2 = a2 * (1 - to.e * to.e) / (1 + to.e);
        if (r1 < 100) r1 = a1;
        if (r2 < 100) r2 = a2;

        double at = (r1 + r2) / 2.0;
        double v1c = Math.Sqrt(MU / r1);
        double v1t = Math.Sqrt(MU * (2.0 / r1 - 1.0 / at));
        double v2t = Math.Sqrt(MU * (2.0 / r2 - 1.0 / at));
        double v2c = Math.Sqrt(MU / r2);

        double dv_hohmann = Math.Abs(v1t - v1c) + Math.Abs(v2c - v2t);

        // Plane change cost
        double di = Math.Abs(from.i - to.i);
        double dO = Math.Abs(from.raan - to.raan);
        if (dO > Math.PI) dO = 2 * Math.PI - dO;
        double totalAngle = Math.Sqrt(di * di + Math.Pow(dO * Math.Sin((from.i + to.i) / 2), 2));
        double vavg = (v1c + v2c) / 2.0;
        double dv_plane = 2 * vavg * Math.Sin(totalAngle / 2);

        // Eccentricity change cost
        double de = Math.Abs(from.e - to.e);
        double dv_ecc = de * vavg * 0.5;

        return dv_hohmann + dv_plane + dv_ecc;
    }

    static void Main(string[] args)
    {
        string line1 = Console.ReadLine();
        var parts1 = line1.Trim().Split(new[]{' ','\t'}, StringSplitOptions.RemoveEmptyEntries);
        int N = int.Parse(parts1[0]);
        double mu = double.Parse(parts1[1]);

        double[][] states = new double[N + 1][];
        for (int i = 0; i <= N; i++)
        {
            var parts = Console.ReadLine().Trim().Split(new[]{' ','\t'}, StringSplitOptions.RemoveEmptyEntries);
            states[i] = parts.Select(double.Parse).ToArray();
        }

        Orbit[] orbits = new Orbit[N + 1];
        for (int i = 0; i <= N; i++)
        {
            double[] r = { states[i][0], states[i][1], states[i][2] };
            double[] v = { states[i][3], states[i][4], states[i][5] };
            orbits[i] = StateToOrbit(r, v, mu);
        }

        // Cost matrix: index 0 = start, 1..N = stations
        double[,] cost = new double[N + 1, N + 1];
        for (int i = 0; i <= N; i++)
            for (int j = 0; j <= N; j++)
                cost[i, j] = (i == j) ? 0 : TransferCost(orbits[i], orbits[j]);

        int[] bestOrder;

        if (N <= 8)
        {
            // Brute force
            bestOrder = BruteForce(cost, N);
        }
        else
        {
            bestOrder = NearestNeighbor(cost, N);
            bestOrder = TwoOpt(bestOrder, cost, N);
            // Try multiple NN starts and keep best
            for (int start = 0; start < Math.Min(N, 20); start++)
            {
                int[] trial = NearestNeighborFrom(cost, N, start);
                trial = TwoOpt(trial, cost, N);
                if (TourCost(trial, cost) < TourCost(bestOrder, cost))
                    bestOrder = trial;
            }
        }

        Console.WriteLine(string.Join(" ", bestOrder));
    }

    static double TourCost(int[] order, double[,] cost)
    {
        double total = cost[0, order[0] + 1];
        for (int i = 0; i < order.Length - 1; i++)
            total += cost[order[i] + 1, order[i + 1] + 1];
        return total;
    }

    static int[] BruteForce(double[,] cost, int N)
    {
        int[] best = null;
        double bestCost = double.MaxValue;
        int[] perm = Enumerable.Range(0, N).ToArray();
        Permute(perm, 0, cost, ref best, ref bestCost);
        return best;
    }

    static void Permute(int[] arr, int k, double[,] cost, ref int[] best, ref double bestCost)
    {
        if (k == arr.Length)
        {
            double c = TourCost(arr, cost);
            if (c < bestCost) { bestCost = c; best = (int[])arr.Clone(); }
            return;
        }
        for (int i = k; i < arr.Length; i++)
        {
            int tmp = arr[k]; arr[k] = arr[i]; arr[i] = tmp;
            Permute(arr, k + 1, cost, ref best, ref bestCost);
            tmp = arr[k]; arr[k] = arr[i]; arr[i] = tmp;
        }
    }

    static int[] NearestNeighbor(double[,] cost, int N)
    {
        return NearestNeighborFrom(cost, N, -1);
    }

    static int[] NearestNeighborFrom(double[,] cost, int N, int firstStation)
    {
        bool[] visited = new bool[N];
        List<int> order = new List<int>();

        if (firstStation >= 0 && firstStation < N)
        {
            order.Add(firstStation);
            visited[firstStation] = true;
        }

        while (order.Count < N)
        {
            int current = order.Count == 0 ? -1 : order[order.Count - 1];
            int currentIdx = current == -1 ? 0 : current + 1;
            double bestD = double.MaxValue;
            int bestJ = -1;
            for (int j = 0; j < N; j++)
            {
                if (visited[j]) continue;
                double d = cost[currentIdx, j + 1];
                if (d < bestD) { bestD = d; bestJ = j; }
            }
            visited[bestJ] = true;
            order.Add(bestJ);
        }
        return order.ToArray();
    }

    static int[] TwoOpt(int[] route, double[,] cost, int N)
    {
        int[] best = (int[])route.Clone();
        bool improved = true;
        int maxIter = N > 50 ? 200 : 5000;
        int iter = 0;
        while (improved && iter++ < maxIter)
        {
            improved = false;
            for (int i = 0; i < N - 1; i++)
            {
                for (int j = i + 1; j < N; j++)
                {
                    int prevI = i == 0 ? 0 : best[i - 1] + 1;
                    int nextJ = j == N - 1 ? -1 : best[j + 1] + 1;

                    double oldCost = cost[prevI, best[i] + 1];
                    double newCost = cost[prevI, best[j] + 1];

                    if (nextJ >= 0)
                    {
                        oldCost += cost[best[j] + 1, nextJ];
                        newCost += cost[best[i] + 1, nextJ];
                    }

                    if (newCost < oldCost - 1e-10)
                    {
                        Array.Reverse(best, i, j - i + 1);
                        improved = true;
                    }
                }
            }
        }
        return best;
    }
}
