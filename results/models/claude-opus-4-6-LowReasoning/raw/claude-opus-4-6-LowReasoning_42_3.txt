; SPIR-V Geometry Shader: Line to thick ribbon (quad)
               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %in_color %out_color
               OpExecutionMode %main InputLines
               OpExecutionMode %main Invocations 1
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 4

               ; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

               OpDecorate %in_color Location 2
               OpDecorate %out_color Location 0

               ; Types
      %void  = OpTypeVoid
   %voidfn   = OpTypeFunction %void
     %float  = OpTypeFloat 32
   %v3float  = OpTypeVector %float 3
   %v4float  = OpTypeVector %float 4
      %uint  = OpTypeInt 32 0
       %int  = OpTypeInt 32 1

     %int_0  = OpConstant %int 0
     %int_1  = OpConstant %int 1
    %uint_2  = OpConstant %uint 2
  %float_0   = OpConstant %float 0
  %float_1   = OpConstant %float 1
  %thickness = OpConstant %float 0.03

               ; Input gl_PerVertex
%gl_PerVertex_in = OpTypeStruct %v4float
  %arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr  = OpTypePointer Input %arr_pv_in
    %gl_in   = OpVariable %ptr_in_arr Input
%ptr_in_v4   = OpTypePointer Input %v4float

               ; Input color (location 2, array of 2 vec3)
%arr_v3_2    = OpTypeArray %v3float %uint_2
%ptr_in_arr3 = OpTypePointer Input %arr_v3_2
  %in_color  = OpVariable %ptr_in_arr3 Input
%ptr_in_v3   = OpTypePointer Input %v3float

               ; Output gl_PerVertex
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv  = OpTypePointer Output %gl_PerVertex_out
   %gl_out   = OpVariable %ptr_out_pv Output
%ptr_out_v4  = OpTypePointer Output %v4float

               ; Output color
%ptr_out_v3  = OpTypePointer Output %v3float
 %out_color  = OpVariable %ptr_out_v3 Output

               ; Function
     %main   = OpFunction %void None %voidfn
    %entry   = OpLabel

               ; Load gl_in[0].gl_Position
   %ac_p0    = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
       %p0   = OpLoad %v4float %ac_p0

               ; Load gl_in[1].gl_Position
   %ac_p1    = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
       %p1   = OpLoad %v4float %ac_p1

               ; Load color[0] and color[1]
   %ac_c0    = OpAccessChain %ptr_in_v3 %in_color %int_0
       %c0   = OpLoad %v3float %ac_c0
   %ac_c1    = OpAccessChain %ptr_in_v3 %in_color %int_1
       %c1   = OpLoad %v3float %ac_c1

               ; dir = p1 - p0
      %dir   = OpFSub %v4float %p1 %p0

               ; Extract dir.x and dir.y
    %dir_x   = OpCompositeExtract %float %dir 0
    %dir_y   = OpCompositeExtract %float %dir 1

               ; perp_raw = (-dir.y, dir.x, 0, 0)
  %neg_dy    = OpFNegate %float %dir_y
  %perp_raw  = OpCompositeConstruct %v4float %neg_dy %dir_x %float_0 %float_0

               ; length = sqrt(perp_raw.x^2 + perp_raw.y^2)
  %perp_x    = OpCompositeExtract %float %perp_raw 0
  %perp_y    = OpCompositeExtract %float %perp_raw 1
  %px2       = OpFMul %float %perp_x %perp_x
  %py2       = OpFMul %float %perp_y %perp_y
  %sum_sq    = OpFAdd %float %px2 %py2
  %len       = OpExtInst %float %1 Sqrt %sum_sq

               ; normalize and scale by thickness
  %inv_len   = OpFDiv %float %thickness %len
  %perp_vec  = OpVectorTimesScalar %v4float %perp_raw %inv_len

               ; Compute 4 corner positions
  %v0        = OpFAdd %v4float %p0 %perp_vec
  %v1        = OpFSub %v4float %p0 %perp_vec
  %v2        = OpFAdd %v4float %p1 %perp_vec
  %v3        = OpFSub %v4float %p1 %perp_vec

               ; Emit vertex 0: p0 + perp, color c0
  %out_pos0  = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %out_pos0 %v0
               OpStore %out_color %c0
               OpEmitVertex

               ; Emit vertex 1: p0 - perp, color c0
               OpStore %out_pos0 %v1
               OpStore %out_color %c0
               OpEmitVertex

               ; Emit vertex 2: p1 + perp, color c1
               OpStore %out_pos0 %v2
               OpStore %out_color %c1
               OpEmitVertex

               ; Emit vertex 3: p1 - perp, color c1
               OpStore %out_pos0 %v3
               OpStore %out_color %c1
               OpEmitVertex

               OpEndPrimitive
               OpReturn
               OpFunctionEnd
