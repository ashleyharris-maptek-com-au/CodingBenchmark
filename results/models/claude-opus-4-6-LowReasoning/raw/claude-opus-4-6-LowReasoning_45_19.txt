import struct

def encode_string(s):
    b = s.encode('utf-8') + b'\x00'
    while len(b) % 4 != 0:
        b += b'\x00'
    result = []
    for i in range(0, len(b), 4):
        result.append(struct.unpack('<I', b[i:i+4])[0])
    return result

def f2u(f):
    return struct.unpack('<I', struct.pack('<f', f))[0]

def op(opcode, *operands):
    wc = 1 + len(operands)
    words.append((wc << 16) | opcode)
    words.extend(operands)

def op_str(opcode, *before_str, s="", after=None):
    sw = encode_string(s)
    if after is None: after = []
    wc = 1 + len(before_str) + len(sw) + len(after)
    words.append((wc << 16) | opcode)
    words.extend(before_str)
    words.extend(sw)
    words.extend(after)
