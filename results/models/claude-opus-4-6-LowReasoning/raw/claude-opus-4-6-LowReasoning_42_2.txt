; SPIR-V Geometry Shader: Expand points into hexagons
               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %inColor %outColor
               OpExecutionMode %main InputPoints
               OpExecutionMode %main Invocations 1
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 18

               ; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %inColor Location 2
               OpDecorate %outColor Location 0

               ; Types
      %void = OpTypeVoid
  %void_fn = OpTypeFunction %void
     %float = OpTypeFloat 32
   %v3float = OpTypeVector %float 3
   %v4float = OpTypeVector %float 4
       %int = OpTypeInt 32 1
      %uint = OpTypeInt 32 0

    %int_0 = OpConstant %int 0
    %int_1 = OpConstant %int 1
    %int_6 = OpConstant %int 6
   %uint_1 = OpConstant %uint 1

  %float_0 = OpConstant %float 0.0
  %float_1 = OpConstant %float 1.0
%float_0_04 = OpConstant %float 0.04
%float_0_7 = OpConstant %float 0.7
%float_0_5 = OpConstant %float 0.5
%float_0_866 = OpConstant %float 0.866025
%float_n0_5 = OpConstant %float -0.5
%float_n0_866 = OpConstant %float -0.866025

      %bool = OpTypeBool

               ; Input gl_PerVertex block (array size 1 for InputPoints)
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in_1 = OpTypeArray %gl_PerVertex_in %uint_1
%ptr_in_arr = OpTypePointer Input %arr_pv_in_1
    %gl_in = OpVariable %ptr_in_arr Input
%ptr_in_v4 = OpTypePointer Input %v4float

               ; Input color (array size 1)
%arr_v3_1 = OpTypeArray %v3float %uint_1
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_1
   %inColor = OpVariable %ptr_in_arr_v3 Input
%ptr_in_v3 = OpTypePointer Input %v3float

               ; Output gl_PerVertex block
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
   %gl_out = OpVariable %ptr_out_pv Output
%ptr_out_v4 = OpTypePointer Output %v4float

               ; Output color
%ptr_out_v3 = OpTypePointer Output %v3float
  %outColor = OpVariable %ptr_out_v3 Output

               ; Cosines and sines for 0,60,120,180,240,300,360 degrees
               ; cos: 1, 0.5, -0.5, -1, -0.5, 0.5, 1
               ; sin: 0, 0.866, 0.866, 0, -0.866, -0.866, 0
%float_n1 = OpConstant %float -1.0

               ; Array type for edge offsets (7 pairs of cos,sin)
   %uint_7 = OpConstant %uint 7
%arr_float_7 = OpTypeArray %float %uint_7
%ptr_fn_float = OpTypePointer Function %float
%ptr_fn_arr7 = OpTypePointer Function %arr_float_7

      %main = OpFunction %void None %void_fn
 %entry = OpLabel

               ; Load input position (gl_in[0].gl_Position)
  %pos_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
  %pos = OpLoad %v4float %pos_ptr

               ; Load input color
  %col_ptr = OpAccessChain %ptr_in_v3 %inColor %int_0
  %col = OpLoad %v3float %col_ptr

               ; Darker color for edges
  %dark_col = OpVectorTimesScalar %v3float %col %float_0_7

               ; Extract position x,y,z,w
  %px = OpCompositeExtract %float %pos 0
  %py = OpCompositeExtract %float %pos 1
  %pz = OpCompositeExtract %float %pos 2
  %pw = OpCompositeExtract %float %pos 3

               ; Precompute edge positions (7 edges: 0..6 corresponding to 0,60,...,360 deg)
               ; edge_x[i] = px + r * cos(i*60), edge_y[i] = py + r * sin(i*60)

               ; Edge 0: angle=0, cos=1, sin=0
  %ex0 = OpFAdd %float %px %float_0_04
  %ey0 = OpCopy %float %py

               ; Edge 1: angle=60, cos=0.5, sin=0.866
  %dx1 = OpFMul %float %float_0_04 %float_0_5
  %dy1 = OpFMul %float %float_0_04 %float_0_866
  %ex1 = OpFAdd %float %px %dx1
  %ey1 = OpFAdd %float %py %dy1

               ; Edge 2: angle=120, cos=-0.5, sin=0.866
  %dx2 = OpFMul %float %float_0_04 %float_n0_5
  %dy2 = OpFMul %float %float_0_04 %float_0_866
  %ex2 = OpFAdd %float %px %dx2
  %ey2 = OpFAdd %float %py %dy2

               ; Edge 3: angle=180, cos=-1, sin=0
  %ex3 = OpFSub %float %px %float_0_04
  %ey3 = OpCopy %float %py

               ; Edge 4: angle=240, cos=-0.5, sin=-0.866
  %dx4 = OpFMul %float %float_0_04 %float_n0_5
  %dy4 = OpFMul %float %float_0_04 %float_n0_866
  %ex4 = OpFAdd %float %px %dx4
  %ey4 = OpFAdd %float %py %dy4

               ; Edge 5: angle=300, cos=0.5, sin=-0.866
  %dx5 = OpFMul %float %float_0_04 %float_0_5
  %dy5 = OpFMul %float %float_0_04 %float_n0_866
  %ex5 = OpFAdd %float %px %dx5
  %ey5 = OpFAdd %float %py %dy5

               ; === Triangle 0: center, e0, e1 ===
  %e0_pos = OpCompositeConstruct %v4float %ex0 %ey0 %pz %pw
  %e1_pos = OpCompositeConstruct %v4float %ex1 %ey1 %pz %pw

  %out_ptr_0 = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %out_ptr_0 %pos
               OpStore %outColor %col
               OpEmitVertex
               OpStore %out_ptr_0 %e0_pos
               OpStore %outColor %dark_col
               OpEmitVertex
               OpStore %out_ptr_0 %e1_pos
               OpStore %outColor %dark_col
               OpEmitVertex
               OpEndPrimitive

               ; === Triangle 1: center, e1, e2 ===
  %e2_pos = OpCompositeConstruct %v4float %ex2 %ey2 %pz %pw

               OpStore %out_ptr_0 %pos
               OpStore %outColor %col
               OpEmitVertex
               OpStore %out_ptr_0 %e1_pos
               OpStore %outColor %dark_col
               OpEmitVertex
               OpStore %out_ptr_0 %e2_pos
               OpStore %outColor %dark_col
               OpEmitVertex
               OpEndPrimitive

               ; === Triangle 2: center, e2, e3 ===
  %e3_pos = OpCompositeConstruct %v4float %ex3 %ey3 %pz %pw

               OpStore %out_ptr_0 %pos
               OpStore %outColor %col
               OpEmitVertex
               OpStore %out_ptr_0 %e2_pos
               OpStore %outColor %dark_col
               OpEmitVertex
               OpStore %out_ptr_0 %e3_pos
               OpStore %outColor %dark_col
               OpEmitVertex
               OpEndPrimitive

               ; === Triangle 3: center, e3, e4 ===
  %e4_pos = OpCompositeConstruct %v4float %ex4 %ey4 %pz %pw

               OpStore %out_ptr_0 %pos
               OpStore %outColor %col
               OpEmitVertex
               OpStore %out_ptr_0 %e3_pos
               OpStore %outColor %dark_col
               OpEmitVertex
               OpStore %out_ptr_0 %e4_pos
               OpStore %outColor %dark_col
               OpEmitVertex
               OpEndPrimitive

               ; === Triangle 4: center, e4, e5 ===
  %e5_pos = OpCompositeConstruct %v4float %ex5 %ey5 %pz %pw

               OpStore %out_ptr_0 %pos
               OpStore %outColor %col
               OpEmitVertex
               OpStore %out_ptr_0 %e4_pos
               OpStore %outColor %dark_col
               OpEmitVertex
               OpStore %out_ptr_0 %e5_pos
               OpStore %outColor %dark_col
               OpEmitVertex
               OpEndPrimitive

               ; === Triangle 5: center, e5, e0 ===
               OpStore %out_ptr_0 %pos
               OpStore %outColor %col
               OpEmitVertex
               OpStore %out_ptr_0 %e5_pos
               OpStore %outColor %dark_col
               OpEmitVertex
               OpStore %out_ptr_0 %e0_pos
               OpStore %outColor %dark_col
               OpEmitVertex
               OpEndPrimitive

               OpReturn
               OpFunctionEnd
