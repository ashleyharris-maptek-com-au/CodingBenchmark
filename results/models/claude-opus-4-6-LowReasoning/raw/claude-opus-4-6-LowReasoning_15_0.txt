use std::io::{self, Read, Write, BufWriter};

fn cross2d(o: [f64; 2], a: [f64; 2], b: [f64; 2]) -> f64 {
    (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
}

fn convex_hull_2d(pts: &[[f64; 2]]) -> Vec<[f64; 2]> {
    let mut pts = pts.to_vec();
    pts.sort_by(|a, b| a[0].partial_cmp(&b[0]).unwrap().then(a[1].partial_cmp(&b[1]).unwrap()));
    let n = pts.len();
    if n <= 1 {
        return pts;
    }
    let mut hull: Vec<[f64; 2]> = Vec::with_capacity(2 * n);
    for &p in &pts {
        while hull.len() >= 2 && cross2d(hull[hull.len() - 2], hull[hull.len() - 1], p) <= 0.0 {
            hull.pop();
        }
        hull.push(p);
    }
    let lower_len = hull.len() + 1;
    for i in (0..n - 1).rev() {
        let p = pts[i];
        while hull.len() >= lower_len && cross2d(hull[hull.len() - 2], hull[hull.len() - 1], p) <= 0.0 {
            hull.pop();
        }
        hull.push(p);
    }
    hull.pop();
    hull
}

fn inscribed_radius(hull: &[[f64; 2]]) -> f64 {
    let n = hull.len();
    let mut min_dist = f64::MAX;
    for i in 0..n {
        let j = (i + 1) % n;
        let dx = hull[j][0] - hull[i][0];
        let dy = hull[j][1] - hull[i][1];
        let len = (dx * dx + dy * dy).sqrt();
        if len < 1e-15 {
            continue;
        }
        let dist = (hull[i][0] * dy - hull[i][1] * dx) / len;
        min_dist = min_dist.min(dist.abs());
    }
    min_dist
}

fn signed_area(poly: &[[f64; 2]]) -> f64 {
    let n = poly.len();
    let mut area = 0.0;
    for i in 0..n {
        let j = (i + 1) % n;
        area += poly[i][0] * poly[j][1] - poly[j][0] * poly[i][1];
    }
    area * 0.5
}

fn point_in_triangle(p: [f64; 2], a: [f64; 2], b: [f64; 2], c: [f64; 2]) -> bool {
    let d1 = cross2d(a, b, p);
    let d2 = cross2d(b, c, p);
    let d3 = cross2d(c, a, p);
    let has_neg = (d1 < 0.0) || (d2 < 0.0) || (d3 < 0.0);
    let has_pos = (d1 > 0.0) || (d2 > 0.0) || (d3 > 0.0);
    !(has_neg && has_pos)
}

fn ear_clip(poly: &[[f64; 2]]) -> Vec<(usize, usize, usize)> {
    let n = poly.len();
    if n < 3 {
        return vec![];
    }
    if n == 3 {
        return vec![(0, 1, 2)];
    }
    let mut indices: Vec<usize> = (0..n).collect();
    let mut result = Vec::new();
    let mut iterations = 0;
    let max_iterations = n * n * 2;

    while indices.len() > 3 {
        iterations += 1;
        if iterations > max_iterations {
            break;
        }
        let m = indices.len();
        let mut found = false;
        for i in 0..m {
            let pi = (i + m - 1) % m;
            let ni = (i + 1) % m;
            let prev = indices[pi];
            let curr = indices[i];
            let next = indices[ni];

            if cross2d(poly[prev], poly[curr], poly[next]) <= 1e-12 {
                continue;
            }

            let mut ear = true;
            for j in 0..m {
                let idx = indices[j];
                if idx == prev || idx == curr || idx == next {
                    continue;
                }
                if point_in_triangle(poly[idx], poly[prev], poly[curr], poly[next]) {
                    ear = false;
                    break;
                }
            }

            if ear {
                result.push((prev, curr, next));
                indices.remove(i);
                found = true;
                break;
            }
        }
        if !found {
            let prev = indices[0];
            let curr = indices[1];
            let next = indices[2];
            result.push((prev, curr, next));
            indices.remove(1);
        }
    }

    if indices.len() == 3 {
        result.push((indices[0], indices[1], indices[2]));
    }
    result
}

fn subdivide_and_place(
    pts: &[[f64; 2]; 3],
    r_insc: f64,
    max_scale: f64,
    results: &mut Vec<[f64; 8]>,
    rx: f64,
    ry: f64,
    depth: u32,
) {
    let cx = (pts[0][0] + pts[1][0] + pts[2][0]) / 3.0;
    let cy = (pts[0][1] + pts[1][1] + pts[2][1]) / 3.0;

    let mut max_dist = 0.0_f64;
    for p in pts {
        let d = ((p[0] - cx).powi(2) + (p[1] - cy).powi(2)).sqrt();
        max_dist = max_dist.max(d);
    }

    if max_dist < 1e-12 {
        return;
    }

    let scale = max_dist / r_insc * 1.15;

    if scale <= max_scale || depth >= 20 {
        let s = scale.min(max_scale).max(0.001);
        let idx = results.len();
        let pz = 100.0 + idx as f64 * 16.0;
        let px = cx + pz * rx;
        let py = cy + pz * ry;
        results.push([px, py, pz, 1.0, 0.0, 0.0, 0.0, s]);
    } else {
        let m01 = [
            (pts[0][0] + pts[1][0]) * 0.5,
            (pts[0][1] + pts[1][1]) * 0.5,
        ];
        let m12 = [
            (pts[1][0] + pts[2][0]) * 0.5,
            (pts[1][1] + pts[2][1]) * 0.5,
        ];
        let m20 = [
            (pts[2][0] + pts[0][0]) * 0.5,
            (pts[2][1] + pts[0][1]) * 0.5,
        ];
        subdivide_and_place(&[pts[0], m01, m20], r_insc, max_scale, results, rx, ry, depth + 1);
        subdivide_and_place(&[pts[1], m12, m01], r_insc, max_scale, results, rx, ry, depth + 1);
        subdivide_and_place(&[pts[2], m20, m12], r_insc, max_scale, results, rx, ry, depth + 1);
        subdivide_and_place(&[m01, m12, m20], r_insc, max_scale, results, rx, ry, depth + 1);
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let stdout = io::stdout();
    let mut out = BufWriter::new(stdout.lock());

    let mut iter = input.split_whitespace();
    let nv: usize = iter.next().unwrap().parse().unwrap();

    let mut poly: Vec<[f64; 2]> = Vec::with_capacity(nv);
    for _ in 0..nv {
        let x: f64 = iter.next().unwrap().parse().unwrap();
        let y: f64 = iter.next().unwrap().parse().unwrap();
        poly.push([x, y]);
    }

    let sx: f64 = iter.next().unwrap().parse().unwrap();
    let sy: f64 = iter.next().unwrap().parse().unwrap();
    let sz: f64 = iter.next().unwrap().parse().unwrap();

    if nv < 3 {
        writeln!(out, "0").unwrap();
        return;
    }

    let rx = sx / sz;
    let ry = sy / sz;

    let base = [
        [1.0_f64, 0.0, -0.707],
        [-1.0, 0.0, -0.707],
        [0.0, 1.0, 0.707],
        [0.0, -1.0, 0.707],
    ];

    let offsets: Vec<[f64; 2]> = base
        .iter()
        .map(|v| [v[0] - v[2] * rx, v[1] - v[2] * ry])
        .collect();

    let hull = convex_hull_2d(&offsets);
    let r_insc = inscribed_radius(&hull);

    if r_insc < 1e-15 {
        writeln!(out, "0").unwrap();
        return;
    }

    // Ensure CCW
    if signed_area(&poly) < 0.0 {
        poly.reverse();
    }

    let tris = ear_clip(&poly);

    let mut placements: Vec<[f64; 8]> = Vec::new();
    let max_scale = 9.5;

    for (i0, i1, i2) in &tris {
        let pts = [poly[*i0], poly[*i1], poly[*i2]];
        subdivide_and_place(&pts, r_insc, max_scale, &mut placements, rx, ry, 0);
    }

    writeln!(out, "{}", placements.len()).unwrap();
    for p in &placements {
        writeln!(
            out,
            "{} {} {} {} {} {} {} {}",
            p[0], p[1], p[2], p[3], p[4], p[5], p[6], p[7]
        )
        .unwrap();
    }
}
