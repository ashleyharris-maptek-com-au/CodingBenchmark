import struct

def f2i(f):
    return struct.unpack('<I', struct.pack('<f', f))[0]

def encode_string(s):
    b = s.encode('utf-8') + b'\x00'
    while len(b) % 4 != 0:
        b += b'\x00'
    words = []
    for i in range(0, len(b), 4):
        words.append(struct.unpack('<I', b[i:i+4])[0])
    return words

class SPIRVBuilder:
    def __init__(self):
        self.capabilities = []
        self.extensions = []
        self.ext_imports = []
        self.memory_model = []
        self.entry_points = []
        self.exec_modes = []
        self.decorations = []
        self.type_decls = []
        self.constants = []
        self.globals = []
        self.functions = []

    def inst(self, opcode, operands):
        wc = 1 + len(operands)
        return [(wc << 16) | opcode] + operands

    def inst_str(self, opcode, before, s, after=[]):
        sw = encode_string(s)
        ops = list(before) + sw + list(after)
        wc = 1 + len(ops)
        return [(wc << 16) | opcode] + ops

    def build(self, bound):
        all_words = []
        # Header
        all_words += [0x07230203, 0x00010000, 0, bound, 0]
        for section in [self.capabilities, self.extensions, self.ext_imports,
                       self.memory_model, self.entry_points, self.exec_modes,
                       self.decorations, self.type_decls, self.constants,
                       self.globals, self.functions]:
            for inst in section:
                all_words += inst
        return b''.join(struct.pack('<I', w) for w in all_words)
