import struct

words = []

def w(v):
    words.append(v)

def le(v):
    return v  # already storing as int, will pack as little-endian

def op(wordcount, opcode):
    return (wordcount << 16) | opcode

def str_words(s):
    b = s.encode('utf-8') + b'\x00'
    while len(b) % 4 != 0:
        b += b'\x00'
    result = []
    for i in range(0, len(b), 4):
        result.append(struct.unpack('<I', b[i:i+4])[0])
    return result

def f32(v):
    return struct.unpack('<I', struct.pack('<f', v))[0]

# Header
w(0x07230203)  # magic
w(0x00010000)  # version

w(54)          # bound

# OpCapability Shader
w(op(2,17)); w(1)

# OpExtInstImport %1 "GLSL.std.450"
s = str_words("GLSL.std.450")
w(op(2+len(s),11)); w(1)
for x in s: w(x)

# OpMemoryModel Logical GLSL450
w(op(3,14)); w(0); w(1)

# OpEntryPoint Fragment %29 "main" %15 %16 %17 %18

for x in nm: w(x)
w(15); w(16); w(17); w(18)

# OpExecutionMode %29 OriginUpperLeft
w(op(3,16)); w(29); w(7)

# Decorations
def decorate(target, dec, *vals):
    w(op(3+len(vals),71)); w(target); w(dec)
    for v in vals: w(v)

def member_decorate(struct_id, member, dec, *vals):
    w(op(4+len(vals),72)); w(struct_id); w(member); w(dec)
    for v in vals: w(v)

decorate(15, 30, 0)   # Location 0
decorate(16, 30, 1)   # Location 1
decorate(17, 30, 4)   # Location 4
decorate(18, 30, 0)   # Location 0
decorate(19, 34, 0)   # DescriptorSet 0
decorate(19, 33, 0)   # Binding 0
decorate(13, 2)       # Block

for i, off in enumerate([0, 64, 128, 192, 208, 224]):
    member_decorate(13, i, 35, off)  # Offset

for i in range(3):
    member_decorate(13, i, 5)        # ColMajor
    member_decorate(13, i, 7, 16)    # MatrixStride 16
