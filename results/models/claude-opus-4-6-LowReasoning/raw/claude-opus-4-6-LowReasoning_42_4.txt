; SPIR-V Geometry Shader: Tapered ribbon from InputLines
               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %inWorldPos %inNormal %inColor %gl_out %outColor
               OpExecutionMode %main InputLines
               OpExecutionMode %main Invocations 1
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 4

               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %inWorldPos Location 0
               OpDecorate %inNormal Location 1
               OpDecorate %inColor Location 2
               OpDecorate %outColor Location 0

      %void  = OpTypeVoid
  %voidfunc  = OpTypeFunction %void
     %float  = OpTypeFloat 32
   %v3float  = OpTypeVector %float 3
   %v4float  = OpTypeVector %float 4
      %uint  = OpTypeInt 32 0
       %int  = OpTypeInt 32 1

    %uint_2  = OpConstant %uint 2
     %int_0  = OpConstant %int 0
     %int_1  = OpConstant %int 1
   %float_1  = OpConstant %float 1
%float_0_04  = OpConstant %float 0.04
%float_0_005 = OpConstant %float 0.005

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in   = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr  = OpTypePointer Input %arr_pv_in
     %gl_in  = OpVariable %ptr_in_arr Input
%ptr_in_v4   = OpTypePointer Input %v4float

%arr_v3_2    = OpTypeArray %v3float %uint_2
%ptr_in_arr3 = OpTypePointer Input %arr_v3_2
%ptr_in_v3   = OpTypePointer Input %v3float
 %inWorldPos = OpVariable %ptr_in_arr3 Input
   %inNormal = OpVariable %ptr_in_arr3 Input
    %inColor = OpVariable %ptr_in_arr3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv  = OpTypePointer Output %gl_PerVertex_out
    %gl_out  = OpVariable %ptr_out_pv Output
%ptr_out_v4  = OpTypePointer Output %v4float
%ptr_out_v3  = OpTypePointer Output %v3float
  %outColor  = OpVariable %ptr_out_v3 Output

      %main  = OpFunction %void None %voidfunc
    %entry   = OpLabel

      %ap0   = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
     %pos0   = OpLoad %v4float %ap0
      %ap1   = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
     %pos1   = OpLoad %v4float %ap1

     %ac0    = OpAccessChain %ptr_in_v3 %inColor %int_0
     %col0   = OpLoad %v3float %ac0
     %ac1    = OpAccessChain %ptr_in_v3 %inColor %int_1
     %col1   = OpLoad %v3float %ac1

     %p0x    = OpCompositeExtract %float %pos0 0
     %p0y    = OpCompositeExtract %float %pos0 1
     %p0z    = OpCompositeExtract %float %pos0 2
     %p0w    = OpCompositeExtract %float %pos0 3
     %p1x    = OpCompositeExtract %float %pos1 0
     %p1y    = OpCompositeExtract %float %pos1 1
     %p1z    = OpCompositeExtract %float %pos1 2
     %p1w    = OpCompositeExtract %float %pos1 3

     %dx     = OpFSub %float %p1x %p0x
     %dy     = OpFSub %float %p1y %p0y

     %ndy    = OpFNegate %float %dy
     %ndy2   = OpFMul %float %ndy %ndy
     %dx2    = OpFMul %float %dx %dx
     %lsq    = OpFAdd %float %ndy2 %dx2
     %len    = OpExtInst %float %1 Sqrt %lsq
     %invl   = OpFDiv %float %float_1 %len
     %px     = OpFMul %float %ndy %invl
     %py     = OpFMul %float %dx %invl

     %osx    = OpFMul %float %px %float_0_04
     %osy    = OpFMul %float %py %float_0_04
     %oex    = OpFMul %float %px %float_0_005
     %oey    = OpFMul %float %py %float_0_005

     %out_p  = OpAccessChain %ptr_out_v4 %gl_out %int_0

     %v0x    = OpFAdd %float %p0x %osx
     %v0y    = OpFAdd %float %p0y %osy
     %v0     = OpCompositeConstruct %v4float %v0x %v0y %p0z %p0w
               OpStore %out_p %v0
               OpStore %outColor %col0
               OpEmitVertex

     %v1x    = OpFSub %float %p0x %osx
     %v1y    = OpFSub %float %p0y %osy
     %v1     = OpCompositeConstruct %v4float %v1x %v1y %p0z %p0w
               OpStore %out_p %v1
               OpStore %outColor %col0
               OpEmitVertex

     %v2x    = OpFAdd %float %p1x %oex
     %v2y    = OpFAdd %float %p1y %oey
     %v2     = OpCompositeConstruct %v4float %v2x %v2y %p1z %p1w
               OpStore %out_p %v2
               OpStore %outColor %col1
               OpEmitVertex

     %v3x    = OpFSub %float %p1x %oex
     %v3y    = OpFSub %float %p1y %oey
     %v3     = OpCompositeConstruct %v4float %v3x %v3y %p1z %p1w
               OpStore %out_p %v3
               OpStore %outColor %col1
               OpEmitVertex

               OpEndPrimitive
               OpReturn
               OpFunctionEnd
