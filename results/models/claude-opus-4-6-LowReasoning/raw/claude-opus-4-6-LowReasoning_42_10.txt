; Geometry shader: thick smooth curve from lines with adjacency
; Catmull-Rom style interpolation, triangle strip output

               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %inColor %outColor
               OpExecutionMode %main InputLinesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 10
               OpExecutionMode %main Invocations 1

               ; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

               OpDecorate %inColor Location 2
               OpDecorate %outColor Location 0

               ; Types
      %void  = OpTypeVoid
  %voidfunc  = OpTypeFunction %void
     %float  = OpTypeFloat 32
   %v3float  = OpTypeVector %float 3
   %v4float  = OpTypeVector %float 4
      %uint  = OpTypeInt 32 0
       %int  = OpTypeInt 32 1
      %bool  = OpTypeBool

   %uint_4   = OpConstant %uint 4
   %int_0    = OpConstant %int 0
   %int_1    = OpConstant %int 1
   %int_2    = OpConstant %int 2
   %int_3    = OpConstant %int 3

   %float_0  = OpConstant %float 0.0
   %float_1  = OpConstant %float 1.0
  %float_025 = OpConstant %float 0.25
  %float_05  = OpConstant %float 0.5
  %float_075 = OpConstant %float 0.75
 %float_002  = OpConstant %float 0.02
%float_n002  = OpConstant %float -0.02

; gl_PerVertex input (array of 4)
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in  = OpTypeArray %gl_PerVertex_in %uint_4
%ptr_in_arr = OpTypePointer Input %arr_pv_in
%ptr_in_v4  = OpTypePointer Input %v4float
    %gl_in  = OpVariable %ptr_in_arr Input

; gl_PerVertex output
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
   %gl_out  = OpVariable %ptr_out_pv Output

; Color input array (location 2, vec3, array of 4)
%arr_v3_4   = OpTypeArray %v3float %uint_4
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_4
%ptr_in_v3  = OpTypePointer Input %v3float
  %inColor  = OpVariable %ptr_in_arr_v3 Input

; Color output (location 0, vec3)
%ptr_out_v3 = OpTypePointer Output %v3float
 %outColor  = OpVariable %ptr_out_v3 Output

; Function pointer types
%ptr_func_v4 = OpTypePointer Function %v4float
%ptr_func_v3 = OpTypePointer Function %v3float

      %main  = OpFunction %void None %voidfunc
 %entry     = OpLabel

; Load the 4 input positions
 %p0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
 %p1_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
 %p2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
 %p3_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
 %P0 = OpLoad %v4float %p0_ptr
 %P1 = OpLoad %v4float %p1_ptr
 %P2 = OpLoad %v4float %p2_ptr
 %P3 = OpLoad %v4float %p3_ptr

; Load colors at index 1 and 2
 %c1_ptr = OpAccessChain %ptr_in_v3 %inColor %int_1
 %c2_ptr = OpAccessChain %ptr_in_v3 %inColor %int_2
 %C1 = OpLoad %v3float %c1_ptr
 %C2 = OpLoad %v3float %c2_ptr

; Tangent at P1: (P2 - P0) * 0.5, tangent at P2: (P3 - P1) * 0.5
; We use tangents for perpendicular direction
 %t_start_full = OpFSub %v4float %P2 %P0
 %t_end_full   = OpFSub %v4float %P3 %P1
 %half_vec4    = OpCompositeConstruct %v4float %float_05 %float_05 %float_05 %float_05

 %t_start = OpFMul %v4float %t_start_full %half_vec4
 %t_end   = OpFMul %v4float %t_end_full %half_vec4

; === Point 0: t=0.0 (at P1) ===
 %pos0 = OpCopyObject %v4float %P1
 %col0 = OpCopyObject %v3float %C1
; tangent at t=0 is t_start; perpendicular = (-ty, tx, 0, 0) normalized * thickness
 %t0x = OpCompositeExtract %float %t_start 0
 %t0y = OpCompositeExtract %float %t_start 1
 %t0nx = OpFNegate %float %t0y
 %t0len = OpExtInst %float %1 Length %t_start
 %t0invlen = OpFDiv %float %float_1 %t0len
 %perp0x = OpFMul %float %t0nx %t0invlen
 %perp0y = OpFMul %float %t0x %t0invlen
 %off0x_p = OpFMul %float %perp0x %float_002
 %off0y_p = OpFMul %float %perp0y %float_002
 %off0x_n = OpFMul %float %perp0x %float_n002
 %off0y_n = OpFMul %float %perp0y %float_n002
; pos + perp
 %p0_px = OpCompositeExtract %float %pos0 0
 %p0_py = OpCompositeExtract %float %pos0 1
 %p0_pz = OpCompositeExtract %float %pos0 2
 %p0_pw = OpCompositeExtract %float %pos0 3
 %e0_ax = OpFAdd %float %p0_px %off0x_p
 %e0_ay = OpFAdd %float %p0_py %off0y_p
 %emit0a = OpCompositeConstruct %v4float %e0_ax %e0_ay %p0_pz %p0_pw
 %out0a_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %out0a_ptr %emit0a
               OpStore %outColor %col0
               OpEmitVertex
; pos - perp
 %e0_bx = OpFAdd %float %p0_px %off0x_n
 %e0_by = OpFAdd %float %p0_py %off0y_n
 %emit0b = OpCompositeConstruct %v4float %e0_bx %e0_by %p0_pz %p0_pw
               OpStore %out0a_ptr %emit0b
               OpStore %outColor %col0
               OpEmitVertex

; === Point 1: t=0.25 ===
 %t1_vec = OpCompositeConstruct %v4float %float_025 %float_025 %float_025 %float_025
 %t1_inv = OpCompositeConstruct %v4float %float_075 %float_075 %float_075 %float_075
 %pos1_a = OpFMul %v4float %P1 %t1_inv
 %pos1_b = OpFMul %v4float %P2 %t1_vec
 %pos1   = OpFAdd %v4float %pos1_a %pos1_b
 %col1_a = OpVectorTimesScalar %v3float %C1 %float_075
 %col1_b = OpVectorTimesScalar %v3float %C2 %float_025
 %col1   = OpFAdd %v3float %col1_a %col1_b
; tangent: lerp(t_start, t_end, 0.25)
 %tan1_a = OpFMul %v4float %t_start %t1_inv
 %tan1_b = OpFMul %v4float %t_end %t1_vec
 %tan1   = OpFAdd %v4float %tan1_a %tan1_b
 %t1x = OpCompositeExtract %float %tan1 0
 %t1y = OpCompositeExtract %float %tan1 1
 %t1nx = OpFNegate %float %t1y
 %t1len = OpExtInst %float %1 Length %tan1
 %t1invlen = OpFDiv %float %float_1 %t1len
 %perp1x = OpFMul %float %t1nx %t1invlen
 %perp1y = OpFMul %float %t1x %t1invlen
 %off1x_p = OpFMul %float %perp1x %float_002
 %off1y_p = OpFMul %float %perp1y %float_002
 %off1x_n = OpFMul %float %perp1x %float_n002
 %off1y_n = OpFMul %float %perp1y %float_n002
 %p1_px = OpCompositeExtract %float %pos1 0
 %p1_py = OpCompositeExtract %float %pos1 1
 %p1_pz = OpCompositeExtract %float %pos1 2
 %p1_pw = OpCompositeExtract %float %pos1 3
 %e1_ax = OpFAdd %float %p1_px %off1x_p
 %e1_ay = OpFAdd %float %p1_py %off1y_p
 %emit1a = OpCompositeConstruct %v4float %e1_ax %e1_ay %p1_pz %p1_pw
               OpStore %out0a_ptr %emit1a
               OpStore %outColor %col1
               OpEmitVertex
 %e1_bx = OpFAdd %float %p1_px %off1x_n
 %e1_by = OpFAdd %float %p1_py %off1y_n
 %emit1b = OpCompositeConstruct %v4float %e1_bx %e1_by %p1_pz %p1_pw
               OpStore %out0a_ptr %emit1b
               OpStore %outColor %col1
               OpEmitVertex

; === Point 2: t=0.5 ===
 %half4 = OpCompositeConstruct %v4float %float_05 %float_05 %float_05 %float_05
 %pos2_a = OpFMul %v4float %P1 %half4
 %pos2_b = OpFMul %v4float %P2 %half4
 %pos2   = OpFAdd %v4float %pos2_a %pos2_b
 %col2_a = OpVectorTimesScalar %v3float %C1 %float_05
 %col2_b = OpVectorTimesScalar %v3float %C2 %float_05
 %col2   = OpFAdd %v3float %col2_a %col2_b
 %tan2_a = OpFMul %v4float %t_start %half4
 %tan2_b = OpFMul %v4float %t_end %half4
 %tan2   = OpFAdd %v4float %tan2_a %tan2_b
 %t2x = OpCompositeExtract %float %tan2 0
 %t2y = OpCompositeExtract %float %tan2 1
 %t2nx = OpFNegate %float %t2y
 %t2len = OpExtInst %float %1 Length %tan2
 %t2invlen = OpFDiv %float %float_1 %t2len
 %perp2x = OpFMul %float %t2nx %t2invlen
 %perp2y = OpFMul %float %t2x %t2invlen
 %off2x_p = OpFMul %float %perp2x %float_002
 %off2y_p = OpFMul %float %perp2y %float_002
 %off2x_n = OpFMul %float %perp2x %float_n002
 %off2y_n = OpFMul %float %perp2y %float_n002
 %p2_px = OpCompositeExtract %float %pos2 0
 %p2_py = OpCompositeExtract %float %pos2 1
 %p2_pz = OpCompositeExtract %float %pos2 2
 %p2_pw = OpCompositeExtract %float %pos2 3
 %e2_ax = OpFAdd %float %p2_px %off2x_p
 %e2_ay = OpFAdd %float %p2_py %off2y_p
 %emit2a = OpCompositeConstruct %v4float %e2_ax %e2_ay %p2_pz %p2_pw
               OpStore %out0a_ptr %emit2a
               OpStore %outColor %col2
               OpEmitVertex
 %e2_bx = OpFAdd %float %p2_px %off2x_n
 %e2_by = OpFAdd %float %p2_py %off2y_n
 %emit2b = OpCompositeConstruct %v4float %e2_bx %e2_by %p2_pz %p2_pw
               OpStore %out0a_ptr %emit2b
               OpStore %outColor %col2
               OpEmitVertex

; === Point 3: t=0.75 ===
 %pos3_a = OpFMul %v4float %P1 %t1_vec
 %pos3_b = OpFMul %v4float %P2 %t1_inv
 %pos3   = OpFAdd %v4float %pos3_a %pos3_b
 %col3_a = OpVectorTimesScalar %v3float %C1 %float_025
 %col3_b = OpVectorTimesScalar %v3float %C2 %float_075
 %col3   = OpFAdd %v3float %col3_a %col3_b
 %tan3_a = OpFMul %v4float %t_start %t1_vec
 %tan3_b = OpFMul %v4float %t_end %t1_inv
 %tan3   = OpFAdd %v4float %tan3_a %tan3_b
 %t3x = OpCompositeExtract %float %tan3 0
 %t3y = OpCompositeExtract %float %tan3 1
 %t3nx = OpFNegate %float %t3y
 %t3len = OpExtInst %float %1 Length %tan3
 %t3invlen = OpFDiv %float %float_1 %t3len
 %perp3x = OpFMul %float %t3nx %t3invlen
 %perp3y = OpFMul %float %t3x %t3invlen
 %off3x_p = OpFMul %float %perp3x %float_002
 %off3y_p = OpFMul %float %perp3y %float_002
 %off3x_n = OpFMul %float %perp3x %float_n002
 %off3y_n = OpFMul %float %perp3y %float_n002
 %p3_px = OpCompositeExtract %float %pos3 0
 %p3_py = OpCompositeExtract %float %pos3 1
 %p3_pz = OpCompositeExtract %float %pos3 2
 %p3_pw = OpCompositeExtract %float %pos3 3
 %e3_ax = OpFAdd %float %p3_px %off3x_p
 %e3_ay = OpFAdd %float %p3_py %off3y_p
 %emit3a = OpCompositeConstruct %v4float %e3_ax %e3_ay %p3_pz %p3_pw
               OpStore %out0a_ptr %emit3a
               OpStore %outColor %col3
               OpEmitVertex
 %e3_bx = OpFAdd %float %p3_px %off3x_n
 %e3_by = OpFAdd %float %p3_py %off3y_n
 %emit3b = OpCompositeConstruct %v4float %e3_bx %e3_by %p3_pz %p3_pw
               OpStore %out0a_ptr %emit3b
               OpStore %outColor %col3
               OpEmitVertex

; === Point 4: t=1.0 (at P2) ===
 %pos4 = OpCopyObject %v4float %P2
 %col4 = OpCopyObject %v3float %C2
 %t4x = OpCompositeExtract %float %t_end 0
 %t4y = OpCompositeExtract %float %t_end 1
 %t4nx = OpFNegate %float %t4y
 %t4len = OpExtInst %float %1 Length %t_end
 %t4invlen = OpFDiv %float %float_1 %t4len
 %perp4x = OpFMul %float %t4nx %t4invlen
 %perp4y = OpFMul %float %t4x %t4invlen
 %off4x_p = OpFMul %float %perp4x %float_002
 %off4y_p = OpFMul %float %perp4y %float_002
 %off4x_n = OpFMul %float %perp4x %float_n002
 %off4y_n = OpFMul %float %perp4y %float_n002
 %p4_px = OpCompositeExtract %float %pos4 0
 %p4_py = OpCompositeExtract %float %pos4 1
 %p4_pz = OpCompositeExtract %float %pos4 2
 %p4_pw = OpCompositeExtract %float %pos4 3
 %e4_ax = OpFAdd %float %p4_px %off4x_p
 %e4_ay = OpFAdd %float %p4_py %off4y_p
 %emit4a = OpCompositeConstruct %v4float %e4_ax %e4_ay %p4_pz %p4_pw
               OpStore %out0a_ptr %emit4a
               OpStore %outColor %col4
               OpEmitVertex
 %e4_bx = OpFAdd %float %p4_px %off4x_n
 %e4_by = OpFAdd %float %p4_py %off4y_n
 %emit4b = OpCompositeConstruct %v4float %e4_bx %e4_by %p4_pz %p4_pw
               OpStore %out0a_ptr %emit4b
               OpStore %outColor %col4
               OpEmitVertex

               OpEndPrimitive
               OpReturn
               OpFunctionEnd
