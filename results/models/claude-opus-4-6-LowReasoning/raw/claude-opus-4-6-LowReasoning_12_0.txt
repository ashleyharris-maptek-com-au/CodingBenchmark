using System;
using System.Collections.Generic;
using System.Globalization;
using System.Linq;

class Program
{
    struct Vec3
    {
        public double X, Y, Z;
        public Vec3(double x, double y, double z) { X = x; Y = y; Z = z; }
        public static Vec3 operator +(Vec3 a, Vec3 b) => new Vec3(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
        public static Vec3 operator -(Vec3 a, Vec3 b) => new Vec3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
    }

    struct Quat
    {
        public double W, X, Y, Z;
        public Quat(double w, double x, double y, double z) { W = w; X = x; Y = y; Z = z; }

        public Vec3 Rotate(Vec3 v)
        {
            double tx = 2 * (Y * v.Z - Z * v.Y);
            double ty = 2 * (Z * v.X - X * v.Z);
            double tz = 2 * (X * v.Y - Y * v.X);
            return new Vec3(
                v.X + W * tx + Y * tz - Z * ty,
                v.Y + W * ty + Z * tx - X * tz,
                v.Z + W * tz + X * ty - Y * tx);
        }
    }

    static Quat[] Get24Rotations()
    {
        var rots = new List<Quat>();
        double s = Math.Sqrt(0.5);
        Quat[] bases = {
            new Quat(1,0,0,0),
            new Quat(s,s,0,0), new Quat(0,1,0,0), new Quat(s,-s,0,0),
            new Quat(s,0,s,0), new Quat(s,0,-s,0),
            new Quat(s,0,0,s), new Quat(0,0,0,1), new Quat(s,0,0,-s),
            new Quat(s,0,s,0),
            new Quat(0,0,1,0),
            new Quat(s,0,-s,0),
            new Quat(0,s,0,0),
            new Quat(0,0,s,0),
            new Quat(0,0,0,s),
        };

        // Generate all 24 by combining face rotations with z-spins
        Quat[] faceRots = {
            new Quat(1,0,0,0),
            new Quat(s,s,0,0),
            new Quat(0,1,0,0),
            new Quat(s,-s,0,0),
            new Quat(s,0,s,0),
            new Quat(s,0,-s,0),
        };
        Quat[] zSpins = {
            new Quat(1,0,0,0),
            new Quat(s,0,0,s),
            new Quat(0,0,0,1),
            new Quat(s,0,0,-s),
        };

        foreach (var f in faceRots)
            foreach (var z in zSpins)
                rots.Add(QMul(f, z));

        return rots.ToArray();
    }

    static Quat QMul(Quat a, Quat b)
    {
        return new Quat(
            a.W*b.W - a.X*b.X - a.Y*b.Y - a.Z*b.Z,
            a.W*b.X + a.X*b.W + a.Y*b.Z - a.Z*b.Y,
            a.W*b.Y - a.X*b.Z + a.Y*b.W + a.Z*b.X,
            a.W*b.Z + a.X*b.Y - a.Y*b.X + a.Z*b.W);
    }

    static void GetAABB(Vec3[] verts, Quat q, out double sx, out double sy, out double sz,
                        out double minX, out double minY, out double minZ)
    {
        double mnx = double.MaxValue, mny = double.MaxValue, mnz = double.MaxValue;
        double mxx = double.MinValue, mxy = double.MinValue, mxz = double.MinValue;
        foreach (var v in verts)
        {
            var r = q.Rotate(v);
            if (r.X < mnx) mnx = r.X; if (r.X > mxx) mxx = r.X;
            if (r.Y < mny) mny = r.Y; if (r.Y > mxy) mxy = r.Y;
            if (r.Z < mnz) mnz = r.Z; if (r.Z > mxz) mxz = r.Z;
        }
        sx = mxx - mnx; sy = mxy - mny; sz = mxz - mnz;
        minX = mnx; minY = mny; minZ = mnz;
    }

    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        var line = Console.ReadLine().Trim().Split();
        int V = int.Parse(line[0]), F = int.Parse(line[1]);
        var verts = new Vec3[V];
        for (int i = 0; i < V; i++)
        {
            var p = Console.ReadLine().Trim().Split();
            verts[i] = new Vec3(double.Parse(p[0]), double.Parse(p[1]), double.Parse(p[2]));
        }
        for (int i = 0; i < F; i++) Console.ReadLine(); // skip faces

        var cline = Console.ReadLine().Trim().Split();
        double W = double.Parse(cline[0]), H = double.Parse(cline[1]), D = double.Parse(cline[2]);

        var rotations = Get24Rotations();
        double eps = 1e-9;

        // For each rotation, compute AABB and how many fit in grid packing
        // Try all 24 rotations × 6 axis permutations (swap W/H/D assignment)
        int bestCount = 0;
        Quat bestQ = new Quat(1, 0, 0, 0);
        double bestSx = 0, bestSy = 0, bestSz = 0;
        double bestMinX = 0, bestMinY = 0, bestMinZ = 0;
        int bestNx = 0, bestNy = 0, bestNz = 0;

        foreach (var q in rotations)
        {
            GetAABB(verts, q, out double sx, out double sy, out double sz,
                    out double mx, out double my, out double mz);
            if (sx < eps || sy < eps || sz < eps) continue;

            int nx = (int)((W + eps) / sx);
            int ny = (int)((H + eps) / sy);
            int nz = (int)((D + eps) / sz);
            int count = nx * ny * nz;

            if (count > bestCount)
            {
                bestCount = count;
                bestQ = q; bestSx = sx; bestSy = sy; bestSz = sz;
                bestMinX = mx; bestMinY = my; bestMinZ = mz;
                bestNx = nx; bestNy = ny; bestNz = nz;
            }
        }

        // Also try with slight offsets for non-axis-aligned beneficial rotations
        // For cubelike shapes, try 45-degree rotations
        double s45 = Math.Sin(Math.PI / 8), c45 = Math.Cos(Math.PI / 8);
        Quat[] extraRots = {
            new Quat(Math.Cos(Math.PI/8), Math.Sin(Math.PI/8), 0, 0),
            new Quat(Math.Cos(Math.PI/8), 0, Math.Sin(Math.PI/8), 0),
            new Quat(Math.Cos(Math.PI/8), 0, 0, Math.Sin(Math.PI/8)),
        };

        foreach (var q in extraRots)
        {
            foreach (var q2 in rotations)
            {
                var qq = QMul(q, q2);
                GetAABB(verts, qq, out double sx, out double sy, out double sz,
                        out double mx, out double my, out double mz);
                if (sx < eps || sy < eps || sz < eps) continue;
                int nx = (int)((W + eps) / sx);
                int ny = (int)((H + eps) / sy);
                int nz = (int)((D + eps) / sz);
                int count = nx * ny * nz;
                if (count > bestCount)
                {
                    bestCount = count;
                    bestQ = qq; bestSx = sx; bestSy = sy; bestSz = sz;
                    bestMinX = mx; bestMinY = my; bestMinZ = mz;
                    bestNx = nx; bestNy = ny; bestNz = nz;
                }
            }
        }

        var results = new List<string>(bestCount + 1);
        results.Add(bestCount.ToString());

        // Offset so that rotated AABB min corner is at grid position
        double offX = -bestMinX;
        double offY = -bestMinY;
        double offZ = -bestMinZ;

        for (int ix = 0; ix < bestNx; ix++)
            for (int iy = 0; iy < bestNy; iy++)
                for (int iz = 0; iz < bestNz; iz++)
                {
                    double tx = offX + ix * bestSx;
                    double ty = offY + iy * bestSy;
                    double tz = offZ + iz * bestSz;
                    results.Add($"{tx:G15} {ty:G15} {tz:G15} {bestQ.W:G15} {bestQ.X:G15} {bestQ.Y:G15} {bestQ.Z:G15}");
                }

        Console.Write(string.Join("\n", results));
    }
}
