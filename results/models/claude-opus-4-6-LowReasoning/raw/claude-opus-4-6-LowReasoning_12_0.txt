using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;

class Program
{
    static void Main(string[] args)
    {
        CultureInfo.DefaultThreadCurrentCulture = CultureInfo.InvariantCulture;
        var sw = new StreamWriter(Console.OpenStandardOutput(), System.Text.Encoding.ASCII, 1 << 16);

        var line1 = Console.ReadLine().Trim().Split();
        int V = int.Parse(line1[0]);
        int F = int.Parse(line1[1]);

        double[][] vertices = new double[V][];
        for (int i = 0; i < V; i++)
        {
            var parts = Console.ReadLine().Trim().Split();
            vertices[i] = new double[] {
                double.Parse(parts[0], CultureInfo.InvariantCulture),
                double.Parse(parts[1], CultureInfo.InvariantCulture),
                double.Parse(parts[2], CultureInfo.InvariantCulture)
            };
        }

        for (int i = 0; i < F; i++)
            Console.ReadLine();

        var lastLine = Console.ReadLine().Trim().Split();
        double W = double.Parse(lastLine[0], CultureInfo.InvariantCulture);
        double H = double.Parse(lastLine[1], CultureInfo.InvariantCulture);
        double D = double.Parse(lastLine[2], CultureInfo.InvariantCulture);

        var rotations = Generate24Rotations();

        long bestCount = 0;
        double[] bestQuat = null;
        double bestOffX = 0, bestOffY = 0, bestOffZ = 0;
        double bestSX = 0, bestSY = 0, bestSZ = 0;
        long bestNX = 0, bestNY = 0, bestNZ = 0;

        foreach (var q in rotations)
        {
            double minX = double.MaxValue, minY = double.MaxValue, minZ = double.MaxValue;
            double maxX = double.MinValue, maxY = double.MinValue, maxZ = double.MinValue;

            for (int i = 0; i < V; i++)
            {
                var r = RotatePoint(vertices[i], q);
                if (r[0] < minX) minX = r[0]; if (r[0] > maxX) maxX = r[0];
                if (r[1] < minY) minY = r[1]; if (r[1] > maxY) maxY = r[1];
                if (r[2] < minZ) minZ = r[2]; if (r[2] > maxZ) maxZ = r[2];
            }

            double sx = maxX - minX;
            double sy = maxY - minY;
            double sz = maxZ - minZ;

            if (sx < 1e-12) sx = 1e-12;
            if (sy < 1e-12) sy = 1e-12;
            if (sz < 1e-12) sz = 1e-12;

            long nx = (long)(W / sx + 1e-9);
            long ny = (long)(H / sy + 1e-9);
            long nz = (long)(D / sz + 1e-9);
            if (nx < 0) nx = 0;
            if (ny < 0) ny = 0;
            if (nz < 0) nz = 0;

            long count = nx * ny * nz;
            if (count > bestCount)
            {
                bestCount = count;
                bestQuat = q;
                bestOffX = -minX;
                bestOffY = -minY;
                bestOffZ = -minZ;
                bestSX = sx;
                bestSY = sy;
                bestSZ = sz;
                bestNX = nx;
                bestNY = ny;
                bestNZ = nz;
            }
        }

        sw.WriteLine(bestCount);

        if (bestQuat != null)
        {
            string qStr = string.Format(CultureInfo.InvariantCulture, "{0} {1} {2} {3}",
                bestQuat[0], bestQuat[1], bestQuat[2], bestQuat[3]);

            for (long ix = 0; ix < bestNX; ix++)
            {
                double tx = bestOffX + ix * bestSX;
                for (long iy = 0; iy < bestNY; iy++)
                {
                    double ty = bestOffY + iy * bestSY;
                    for (long iz = 0; iz < bestNZ; iz++)
                    {
                        double tz = bestOffZ + iz * bestSZ;
                        sw.Write(tx.ToString(CultureInfo.InvariantCulture));
                        sw.Write(' ');
                        sw.Write(ty.ToString(CultureInfo.InvariantCulture));
                        sw.Write(' ');
                        sw.Write(tz.ToString(CultureInfo.InvariantCulture));
                        sw.Write(' ');
                        sw.WriteLine(qStr);
                    }
                }
            }
        }

        sw.Flush();
    }

    static double[] RotatePoint(double[] p, double[] q)
    {
        double qw = q[0], qx = q[1], qy = q[2], qz = q[3];
        double px = p[0], py = p[1], pz = p[2];
        double t0 = qw * px + qy * pz - qz * py;
        double t1 = qw * py + qz * px - qx * pz;
        double t2 = qw * pz + qx * py - qy * px;
        double t3 = -qx * px - qy * py - qz * pz;
        return new double[] {
            t0*qw - t1*qz + t2*qy - t3*qx,
            t1*qw - t2*qx + t0*qz - t3*qy,
            t2*qw - t0*qy + t1*qx - t3*qz
        };
    }

    static List<double[]> Generate24Rotations()
    {
        var result = new List<double[]>();
        double s = Math.Sqrt(0.5);
        double h = 0.5;
        result.Add(new double[]{1, 0, 0, 0});
        result.Add(new double[]{s, s, 0, 0});
        result.Add(new double[]{0, 1, 0, 0});
        result.Add(new double[]{s, -s, 0, 0});
        result.Add(new double[]{s, 0, s, 0});
        result.Add(new double[]{0, 0, 1, 0});
        result.Add(new double[]{s, 0, -s, 0});
        result.Add(new double[]{s, 0, 0, s});
        result.Add(new double[]{0, 0, 0, 1});
        result.Add(new double[]{s, 0, 0, -s});
        result.Add(new double[]{h, h, h, h});
        result.Add(new double[]{h, -h, -h, -h});
        result.Add(new double[]{h, h, -h, h});
        result.Add(new double[]{h, -h, h, -h});
        result.Add(new double[]{h, -h, h, h});
        result.Add(new double[]{h, h, -h, -h});
        result.Add(new double[]{h, h, h, -h});
        result.Add(new double[]{h, -h, -h, h});
        result.Add(new double[]{0, s, s, 0});
        result.Add(new double[]{0, s, -s, 0});
        result.Add(new double[]{0, s, 0, s});
        result.Add(new double[]{0, s, 0, -s});
        result.Add(new double[]{0, 0, s, s});
        result.Add(new double[]{0, 0, s, -s});
        return result;
    }
}
