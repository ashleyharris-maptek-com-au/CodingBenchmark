const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { depot, truckCount, packages } = data;
const n = packages.length;

if (n === 0) {
  console.log(JSON.stringify({ routes: [] }));
  process.exit(0);
}

const dx = depot[0], dy = depot[1];

function dist(ax, ay, bx, by) {
  const ex = ax - bx, ey = ay - by;
  return Math.sqrt(ex * ex + ey * ey);
}

// Precompute distances from depot and between packages
const depotDist = new Float64Array(n);
for (let i = 0; i < n; i++) {
  depotDist[i] = dist(dx, dy, packages[i].x, packages[i].y);
}

const pairDist = new Float64Array(n * n);
for (let i = 0; i < n; i++) {
  for (let j = i + 1; j < n; j++) {
    const d = dist(packages[i].x, packages[i].y, packages[j].x, packages[j].y);
    pairDist[i * n + j] = d;
    pairDist[j * n + i] = d;
  }
}

function d(i, j) { return pairDist[i * n + j]; }

// Build initial routes using sweep algorithm
const angles = new Float64Array(n);
const order = new Int32Array(n);
for (let i = 0; i < n; i++) {
  angles[i] = Math.atan2(packages[i].y - dy, packages[i].x - dx);
  order[i] = i;
}
order.sort((a, b) => angles[a] - angles[b]);

const k = Math.min(truckCount, n);
const routeSize = Math.ceil(n / k);
const routes = [];
for (let r = 0; r < k; r++) {
  const start = r * routeSize;
  if (start >= n) break;
  const end = Math.min(start + routeSize, n);
  const route = [];
  for (let i = start; i < end; i++) route.push(order[i]);
  routes.push(route);
}

// 2-opt improvement for a single route
function twoOpt(route) {
  const len = route.length;
  if (len < 2) return;
  let improved = true;
  while (improved) {
    improved = false;
    for (let i = 0; i < len - 1; i++) {
      for (let j = i + 1; j < len; j++) {
        const pi = i === 0 ? -1 : route[i - 1];
        const pj = j === len - 1 ? -1 : route[j + 1];
        const ci = route[i], cj = route[j];
        const oldD = (pi === -1 ? depotDist[ci] : d(pi, ci)) +
                     (pj === -1 ? depotDist[cj] : d(cj, pj));
        const newD = (pi === -1 ? depotDist[cj] : d(pi, cj)) +
                     (pj === -1 ? depotDist[ci] : d(ci, pj));
        if (newD < oldD - 1e-10) {
          // Reverse segment [i..j]
          let lo = i, hi = j;
          while (lo < hi) {
            const tmp = route[lo]; route[lo] = route[hi]; route[hi] = tmp;
            lo++; hi--;
          }
          improved = true;
        }
      }
    }
  }
}

// Or-opt: move segments between routes
function routeCost(route) {
  if (route.length === 0) return 0;
  let c = depotDist[route[0]] + depotDist[route[route.length - 1]];
  for (let i = 1; i < route.length; i++) c += d(route[i - 1], route[i]);
  return c;
}

// Apply 2-opt to all routes
for (const route of routes) twoOpt(route);

// Cross-route relocate improvement
let globalImproved = true;
let iterations = 0;
const maxIter = Math.min(20, Math.max(3, Math.floor(5000 / n)));
while (globalImproved && iterations < maxIter) {
  globalImproved = false;
  iterations++;
  for (let ri = 0; ri < routes.length; ri++) {
    for (let rj = 0; rj < routes.length; rj++) {
      if (ri === rj) continue;
      const ra = routes[ri], rb = routes[rj];
      for (let i = 0; i < ra.length; i++) {
        const node = ra[i];
        const prevA = i === 0 ? -1 : ra[i - 1];
        const nextA = i === ra.length - 1 ? -1 : ra[i + 1];
        const removeGain = (prevA === -1 ? depotDist[node] : d(prevA, node)) +
                           (nextA === -1 ? depotDist[node] : d(node, nextA)) -
                           (prevA === -1 && nextA === -1 ? 0 :
                            prevA === -1 ? depotDist[nextA] :
                            nextA === -1 ? depotDist[prevA] : d(prevA, nextA));

        let bestInsertCost = Infinity, bestPos = -1;
        for (let j = 0; j <= rb.length; j++) {
          const prev = j === 0 ? -1 : rb[j - 1];
          const next = j === rb.length ? -1 : rb[j];
          const insertCost = (prev === -1 ? depotDist[node] : d(prev, node)) +
                             (next === -1 ? depotDist[node] : d(node, next)) -
                             (prev === -1 && next === -1 ? 0 :
                              prev === -1 ? depotDist[next] :
                              next === -1 ? depotDist[prev] : d(prev, next));
          if (insertCost < bestInsertCost) {
            bestInsertCost = insertCost;
            bestPos = j;
          }
        }

        if (bestInsertCost < removeGain - 1e-10) {
          ra.splice(i, 1);
          rb.splice(bestPos > i && ri === rj ? bestPos - 1 : bestPos, 0, node);
          globalImproved = true;
          i--;
        }
      }
    }
  }
  for (const route of routes) twoOpt(route);
}

// Build output
const result = { routes: [] };
for (const route of routes) {
  if (route.length > 0) {
    result.routes.push(route.map(i => packages[i].id));
  }
}

console.log(JSON.stringify(result));
