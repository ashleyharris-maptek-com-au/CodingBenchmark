declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const prob: number[][] = data.probability;
const drones: {id: number; x: number; y: number}[] = data.drones;
const maxSteps: number = data.maxStepsPerDrone;

const rows = prob.length;
const cols = prob[0].length;

const dirs = [[0,0],[0,1],[0,-1],[1,0],[-1,0]];

// Track which cells have been visited by any drone
const visited = new Set<number>();

function key(x: number, y: number): number {
  return y * cols + x;
}

function isValid(x: number, y: number): boolean {
  return x >= 0 && x < cols && y >= 0 && y < rows && prob[y][x] !== -1;
}

// For each drone, greedily pick the best path step by step
// Use a lookahead greedy: at each step, pick the neighbor that leads to highest
// cumulative value over a short horizon

const paths: number[][][] = [];

// Sort drones by index to maintain order
const droneOrder = drones.map((d, i) => ({...d, idx: i}));

// Greedy with lookahead for each drone
for (const drone of droneOrder) {
  const path: number[][] = [];
  let cx = drone.x;
  let cy = drone.y;

  // Mark starting cell
  path.push([cx, cy]);
  if (isValid(cx, cy) && prob[cy][cx] > 0) {
    visited.add(key(cx, cy));
  }

  for (let step = 0; step < maxSteps; step++) {
    let bestDir = 0;
    let bestScore = -1;

    // Lookahead depth scales down for large grids
    const lookahead = Math.min(8, maxSteps - step);

    for (let d = 0; d < dirs.length; d++) {
      const nx = cx + dirs[d][0];
      const ny = cy + dirs[d][1];
      if (!isValid(nx, ny)) continue;

      // Evaluate this direction with BFS-like lookahead
      const score = evaluateMove(nx, ny, lookahead - 1, visited);
      if (score > bestScore) {
        bestScore = score;
        bestDir = d;
      }
    }

    cx += dirs[bestDir][0];
    cy += dirs[bestDir][1];
    path.push([cx, cy]);
    visited.add(key(cx, cy));
  }

  paths[drone.idx] = path;
}

// Greedy beam evaluation: from a position, greedily walk for `depth` steps
// collecting unvisited value
function evaluateMove(sx: number, sy: number, depth: number, globalVisited: Set<number>): number {
  const localVisited = new Set<number>(globalVisited);
  let total = 0;
  let cx = sx;
  let cy = sy;

  const k = key(cx, cy);
  if (!localVisited.has(k) && prob[cy][cx] > 0) {
    total += prob[cy][cx];
    localVisited.add(k);
  }

  for (let s = 0; s < depth; s++) {
    let bestVal = -1;
    let bx = cx, by = cy;

    for (const [dx, dy] of dirs) {
      const nx = cx + dx;
      const ny = cy + dy;
      if (!isValid(nx, ny)) continue;
      const nk = key(nx, ny);
      const val = localVisited.has(nk) ? 0 : prob[ny][nx];
      if (val > bestVal) {
        bestVal = val;
        bx = nx;
        by = ny;
      }
    }

    cx = bx;
    cy = by;
    const ck = key(cx, cy);
    if (!localVisited.has(ck) && prob[cy][cx] > 0) {
      total += prob[cy][cx];
      localVisited.add(ck);
    }
  }

  return total;
}
