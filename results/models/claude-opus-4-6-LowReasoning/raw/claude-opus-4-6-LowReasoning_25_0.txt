using System;
using System.Collections.Generic;
using System.Globalization;

class Program
{
    const double G_ = 6.67430e-11;
    const double DAY_ = 86400.0;
    static CultureInfo IC = CultureInfo.InvariantCulture;

    static int N;
    static string[] nm;
    static double[] ms;
    static double[][] p0, v0;
    static int iE, iA, iS;
    static double wD, mSC, dvB, mA;

    static int eN;
    static double eDs;
    static double[][][] eP, eV;

    static double[] VS(double[] a, double[] b) { return new[] { a[0] - b[0], a[1] - b[1], a[2] - b[2] }; }
    static double[] VA(double[] a, double[] b) { return new[] { a[0] + b[0], a[1] + b[1], a[2] + b[2] }; }
    static double[] VM(double s, double[] a) { return new[] { s * a[0], s * a[1], s * a[2] }; }
    static double VD(double[] a, double[] b) { return a[0] * b[0] + a[1] * b[1] + a[2] * b[2]; }
    static double VN(double[] a) { return Math.Sqrt(a[0] * a[0] + a[1] * a[1] + a[2] * a[2]); }

    static void Parse()
    {
        var L = new List<string>();
        string s;
        while ((s = Console.ReadLine()) != null)
        {
            s = s.Trim();
            if (s.Length > 0) L.Add(s);
        }
        var h = L[0].Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        N = int.Parse(h[0]);
        wD = double.Parse(h[1], IC);
        mSC = double.Parse(h[2], IC);
        dvB = double.Parse(h[3], IC);
        mA = double.Parse(h[4], IC);

        nm = new string[N]; ms = new double[N];
        p0 = new double[N][]; v0 = new double[N][];
        iS = 0;
        for (int i = 0; i < N; i++)
        {
            var t = L[1 + i].Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
            nm[i] = t[0];
            ms[i] = double.Parse(t[1], IC);
            p0[i] = new[] { double.Parse(t[2], IC), double.Parse(t[3], IC), double.Parse(t[4], IC) };
            v0[i] = new[] { double.Parse(t[5], IC), double.Parse(t[6], IC), double.Parse(t[7], IC) };
            if (nm[i] == "Earth") iE = i;
            else if (nm[i] == "Asteroid") iA = i;
            if (ms[i] > ms[iS]) iS = i;
        }
    }

    static void Accel(double[][] p, double[][] a)
    {
        for (int i = 0; i < N; i++) { a[i][0] = 0; a[i][1] = 0; a[i][2] = 0; }
        for (int i = 0; i < N; i++)
            for (int j = i + 1; j < N; j++)
            {
                double dx = p[j][0] - p[i][0], dy = p[j][1] - p[i][1], dz = p[j][2] - p[i][2];
                double r2 = dx * dx + dy * dy + dz * dz, r = Math.Sqrt(r2), r3 = r * r2;
                double ai = G_ * ms[j] / r3, aj = G_ * ms[i] / r3;
                a[i][0] += ai * dx; a[i][1] += ai * dy; a[i][2] += ai * dz;
                a[j][0] -= aj * dx; a[j][1] -= aj * dy; a[j][2] -= aj * dz;
            }
    }

    static void BuildEph(double totD, double dtD)
    {
        eDs = dtD * DAY_;
        eN = (int)(totD / dtD) + 2;
        var p = new double[N][];
        var v = new double[N][];
        var a = new double[N][];
        for (int i = 0; i < N; i++) { p[i] = (double[])p0[i].Clone(); v[i] = (double[])v0[i].Clone(); a[i] = new double[3]; }
        eP = new double[N][][]; eV = new double[N][][];
        for (int i = 0; i < N; i++)
        {
            eP[i] = new double[eN][];
            eV[i] = new double[eN][];
            eP[i][0] = (double[])p[i].Clone();
            eV[i][0] = (double[])v[i].Clone();
        }
        for (int s = 1; s < eN; s++)
        {
            Accel(p, a);
            for (int i = 0; i < N; i++) for (int k = 0; k < 3; k++) v[i][k] += 0.5 * eDs * a[i][k];
            for (int i = 0; i < N; i++) for (int k = 0; k < 3; k++) p[i][k] += eDs * v[i][k];
            Accel(p, a);
            for (int i = 0; i < N; i++) for (int k = 0; k < 3; k++) v[i][k] += 0.5 * eDs * a[i][k];
            for (int i = 0; i < N; i++)
            {
                eP[i][s] = (double[])p[i].Clone();
                eV[i][s] = (double[])v[i].Clone();
            }
        }
    }

    static void GS(int b, double td, out double[] p, out double[] v)
    {
        double fi = td * DAY_ / eDs;
        int i0 = (int)fi;
        if (i0 < 0) i0 = 0;
        if (i0 >= eN - 1) i0 = eN - 2;
        double f = fi - i0;
        p = new double[3]; v = new double[3];
        for (int k = 0; k < 3; k++)
        {
            p[k] = eP[b][i0][k] * (1 - f) + eP[b][i0 + 1][k] * f;
            v[k] = eV[b][i0][k] * (1 - f) + eV[b][i0 + 1][k] * f;
        }
    }

    static double SC2(double z)
    {
        if (Math.Abs(z) < 1e-10) return 0.5;
        if (z > 0) return (1 - Math.Cos(Math.Sqrt(z))) / z;
        return (Math.Cosh(Math.Sqrt(-z)) - 1) / (-z);
    }

    static double SS2(double z)
    {
        if (Math.Abs(z) < 1e-10) return 1.0 / 6.0;
        if (z > 0) { double q = Math.Sqrt(z); return (q - Math.Sin(q)) / (z * q); }
        double q2 = Math.Sqrt(-z); return (Math.Sinh(q2) - q2) / ((-z) * q2);
    }

    static double ETOF(double z, double r1, double r2, double A, double mu)
    {
        double c = SC2(z), s = SS2(z);
        double sc = Math.Sqrt(Math.Max(c, 1e-30));
        double y = r1 + r2 + A * (z * s - 1) / sc;
        if (y < 0) return double.NaN;
        double chi = Math.Sqrt(y / Math.Max(c, 1e-30));
        return (chi * chi * chi * s + A * Math.Sqrt(y)) / Math.Sqrt(mu);
    }

    static bool Lamb(double[] r1v, double[] r2v, double tof, double mu, bool lw, out double[] v1, out double[] v2)
    {
        v1 = null; v2 = null;
        if (tof <= 0) return false;
        double r1 = VN(r1v), r2 = VN(r2v);
        if (r1 < 1e3 || r2 < 1e3) return false;
        double cda = VD(r1v, r2v) / (r1 * r2);
        cda = Math.Max(-1, Math.Min(1, cda));
        double sda = Math.Sqrt(Math.Max(0, 1 - cda * cda));
        double cz = r1v[0] * r2v[1] - r1v[1] * r2v[0];
        if (cz < 0) sda = -sda;
        if (lw) sda = -sda;
        if (Math.Abs(sda) < 1e-14 && cda > 0.99) return false;

        double denom = 1 - cda;
        if (Math.Abs(denom) < 1e-14) return false;
        double A = sda * Math.Sqrt(r1 * r2 / denom);
        if (Math.Abs(A) < 1e-20 || double.IsNaN(A)) return false;

        double zlo = -500, zhi = 4 * Math.PI * Math.PI * 0.9;
        for (int i = 0; i < 300; i++)
        {
            double c = SC2(zlo), s = SS2(zlo);
            double sc = Math.Sqrt(Math.Max(c, 1e-30));
            double y = r1 + r2 + A * (zlo * s - 1) / sc;
            if (y > 0) break;
            zlo /= 2;
        }

        double tlo = ETOF(zlo, r1, r2, A, mu);
        double thi = ETOF(zhi, r1, r2, A, mu);
        if (double.IsNaN(tlo)) return false;
        if (double.IsNaN(thi))
        {
            zhi = 2 * Math.PI * Math.PI;
            thi = ETOF(zhi, r1, r2, A, mu);
            if (double.IsNaN(thi)) return false;
        }

        bool inc = thi >= tlo;
        if (inc) { if (tof < tlo * 0.99 || tof > thi * 1.01) return false; }
        else { if (tof > tlo * 1.01 || tof < thi * 0.99) return false; }

        for (int it = 0; it < 120; it++)
        {
            double z = (zlo + zhi) / 2;
            double t = ETOF(z, r1, r2, A, mu);
            if (double.IsNaN(t)) { if (inc) zhi = z; else zlo = z; continue; }
            if (inc) { if (t < tof) zlo = z; else zhi = z; }
            else { if (t > tof) zlo = z; else zhi = z; }
            if (Math.Abs(zhi - zlo) < 1e-13) break;
        }

        double zf = (zlo + zhi) / 2;
        double cf = SC2(zf), sf = SS2(zf);
        double scf = Math.Sqrt(Math.Max(cf, 1e-30));
        double yf = r1 + r2 + A * (zf * sf - 1) / scf;
        if (yf < 0) return false;
        double ff = 1 - yf / r1;
        double gf = A * Math.Sqrt(yf / mu);
        double gdf = 1 - yf / r2;
        if (Math.Abs(gf) < 1e-20) return false;

        v1 = new double[3]; v2 = new double[3];
        for (int k = 0; k < 3; k++)
        {
            v1[k] = (r2v[k] - ff * r1v[k]) / gf;
            v2[k] = (gdf * r2v[k] - r1v[k]) / gf;
        }
        return true;
    }

    static double[] ScAcc(double[] sp, double td)
    {
        double[] a = new double[3];
        for (int i = 0; i < N; i++)
        {
            double[] bp, bv; GS(i, td, out bp, out bv);
            double dx = bp[0] - sp[0], dy = bp[1] - sp[1], dz = bp[2] - sp[2];
            double r2 = dx * dx + dy * dy + dz * dz + 1e10;
            double r = Math.Sqrt(r2);
            double f = G_ * ms[i] / (r * r2);
            a[0] += f * dx; a[1] += f * dy; a[2] += f * dz;
        }
        return a;
    }

    static void PropSC(double[] sp, double[] sv, double t0d, double t1d)
    {
        double dt = 1800;
        double t = t0d * DAY_;
        double te = t1d * DAY_;
        while (t < te - 0.5)
        {
            double h = Math.Min(dt, te - t);
            double td = t / DAY_;
            double[] a1 = ScAcc(sp, td);
            for (int k = 0; k < 3; k++) sv[k] += 0.5 * h * a1[k];
            for (int k = 0; k < 3; k++) sp[k] += h * sv[k];
            double[] a2 = ScAcc(sp, (t + h) / DAY_);
            for (int k = 0; k < 3; k++) sv[k] += 0.5 * h * a2[k];
            t += h;
        }
    }

    static string F(double x) { return x.ToString("R", IC); }

    static void Main()
    {
        Parse();

        double totD = wD * 1.5 + 500;
        double dtD = 0.25;
        if (totD / dtD > 400000) dtD = totD / 400000;
        BuildEph(totD, dtD);

        double mu = G_ * ms[iS];

        double impD = wD;
        double minDist = double.MaxValue;
        for (double t = 0; t < totD; t += dtD)
        {
            double[] ep, ev, ap, av;
            GS(iE, t, out ep, out ev);
            GS(iA, t, out ap, out av);
            double d = VN(VS(ap, ep));
            if (d < minDist) { minDist = d; impD = t; }
        }

        double maxArr = impD - 1;
        if (maxArr < 20) maxArr = impD;
        double minFl = Math.Max(5, Math.Min(60, maxArr * 0.01));

        double bestDv = double.MaxValue;
        double bestTL = 0, bestTA = maxArr * 0.5;
        double[] bestV1 = null, bestV2 = null;
        double bestScore = -1;
        double bestScoreTL = 0, bestScoreTA = 0;
        double[] bestScoreV1 = null;

        int nL = Math.Min(200, Math.Max(50, (int)(wD / 5)));
        int nA = Math.Min(200, Math.Max(50, (int)(maxArr / 5)));
        double sL = Math.Max(0.5, wD / nL);
        double sA = Math.Max(0.5, (maxArr - minFl) / nA);

        for (int lw = 0; lw < 2; lw++)
        {
            for (double tL = 0; tL <= wD; tL += sL)
            {
                double[] ep, ev; GS(iE, tL, out ep, out ev);
                for (double tA = tL + minFl; tA <= maxArr; tA += sA)
                {
                    double[] ap, av; GS(iA, tA, out ap, out av);
                    double[] lv1, lv2;
                    if (!Lamb(ep, ap, (tA - tL) * DAY_, mu, lw == 1, out lv1, out lv2)) continue;
                    double dv = VN(VS(lv1, ev));
                    if (dv < bestDv)
                    {
                        bestDv = dv; bestTL = tL; bestTA = tA;
                        bestV1 = (double[])lv1.Clone(); bestV2 = (double[])lv2.Clone();
                    }
                    if (dv <= dvB * 0.85)
                    {
                        double vrel = VN(VS(lv2, av));
                        double score = vrel * (impD - tA);
                        if (score > bestScore)
                        {
                            bestScore = score; bestScoreTL = tL; bestScoreTA = tA;
                            bestScoreV1 = (double[])lv1.Clone();
                        }
                    }
                }
            }
        }

        if (bestScore > 0)
        {
            bestTL = bestScoreTL; bestTA = bestScoreTA; bestV1 = bestScoreV1;
            double[] ep2, ev2; GS(iE, bestTL, out ep2, out ev2);
            bestDv = VN(VS(bestV1, ev2));
        }

        for (int round = 0; round < 3 && bestV1 != null; round++)
        {
            double rng = (round == 0) ? Math.Max(sL, sA) * 1.5 : (round == 1) ? sL * 0.2 : sL * 0.05;
            double fs = Math.Max(0.005, rng / 30);
            double tl0 = bestTL, ta0 = bestTA;
            for (int lw = 0; lw < 2; lw++)
                for (double tL = Math.Max(0, tl0 - rng); tL <= Math.Min(wD, tl0 + rng); tL += fs)
                {
                    double[] ep, ev; GS(iE, tL, out ep, out ev);
                    for (double tA = Math.Max(tL + minFl, ta0 - rng); tA <= Math.Min(maxArr, ta0 + rng); tA += fs)
                    {
                        double[] ap, av; GS(iA, tA, out ap, out av);
                        double[] lv1, lv2;
                        if (!Lamb(ep, ap, (tA - tL) * DAY_, mu, lw == 1, out lv1, out lv2)) continue;
                        double dv = VN(VS(lv1, ev));

                        if (bestScore > 0)
                        {
                            if (dv <= dvB * 0.85)
                            {
                                double vrel = VN(VS(lv2, av));
                                double sc = vrel * (impD - tA);
                                if (sc > bestScore)
                                {
                                    bestScore = sc; bestTL = tL; bestTA = tA;
                                    bestV1 = (double[])lv1.Clone(); bestDv = dv;
                                }
                            }
                        }
                        else if (dv < bestDv)
                        {
                            bestDv = dv; bestTL = tL; bestTA = tA;
                            bestV1 = (double[])lv1.Clone();
                        }
                    }
                }
            sL = fs; sA = fs;
        }

        if (bestV1 == null)
        {
            double[] ep, ev, ap, av;
            GS(iE, 0, out ep, out ev);
            GS(iA, impD, out ap, out av);
            double[] dir = VS(ap, ep);
            double d = VN(dir);
            if (d < 1) d = 1;
            dir = VM(dvB * 0.4 / d, dir);
            Console.WriteLine("2");
            Console.WriteLine(F(0) + " " + F(dir[0]) + " " + F(dir[1]) + " " + F(dir[2]));
            Console.WriteLine(F(impD) + " " + F(0) + " " + F(0) + " " + F(0));
            return;
        }

        double[] ep0, ev0; GS(iE, bestTL, out ep0, out ev0);
        double[] launchBurn = VS(bestV1, ev0);
        double usedDv = VN(launchBurn);

        double flight = bestTA - bestTL;
        double[] sp = (double[])ep0.Clone();
        double[] sv = VA(ev0, launchBurn);

        var burns = new List<double[]>();
        burns.Add(new[] { bestTL, launchBurn[0], launchBurn[1], launchBurn[2] });

        double[] corrTimes;
        if (flight > 60)
            corrTimes = new[] { bestTL + flight * 0.3, bestTL + flight * 0.7, bestTL + flight * 0.92 };
        else if (flight > 20)
            corrTimes = new[] { bestTL + flight * 0.5, bestTL + flight * 0.85 };
        else
            corrTimes = new[] { bestTL + flight * 0.5 };

        double prevT = bestTL;
        double[] astPA, astVA; GS(iA, bestTA, out astPA, out astVA);

        for (int ci = 0; ci < corrTimes.Length; ci++)
        {
            double ct = corrTimes[ci];
            PropSC(sp, sv, prevT, ct);
            prevT = ct;

            double remTof = (bestTA - ct) * DAY_;
            if (remTof < DAY_ * 2) continue;

            double[] cv1, cv2;
            if (!Lamb(sp, astPA, remTof, mu, false, out cv1, out cv2))
            {
                if (!Lamb(sp, astPA, remTof, mu, true, out cv1, out cv2))
                    continue;
            }

            double[] cb = VS(cv1, sv);
            double cdv = VN(cb);
            double avail = dvB - usedDv;
            if (avail <= 0) continue;
            double allocFrac = (ci == corrTimes.Length - 1) ? 0.95 : 0.6;
            if (cdv > avail * allocFrac)
                cb = VM(avail * allocFrac / cdv, cb);

            double cdvActual = VN(cb);
            if (cdvActual > 0.001)
            {
                usedDv += cdvActual;
                sv = VA(sv, cb);
                burns.Add(new[] { ct, cb[0], cb[1], cb[2] });
            }
        }

        burns.Add(new[] { bestTA, 0.0, 0.0, 0.0 });

        Console.WriteLine(burns.Count);
        foreach (var b in burns)
            Console.WriteLine(F(b[0]) + " " + F(b[1]) + " " + F(b[2]) + " " + F(b[3]));
    }
}
