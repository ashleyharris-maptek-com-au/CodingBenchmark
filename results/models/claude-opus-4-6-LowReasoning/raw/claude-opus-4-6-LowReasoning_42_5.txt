; SPIR-V Geometry Shader - Arrow lines
; Input: Lines (2 verts), Output: TriangleStrip, max 7 verts

               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %in_color %out_color
               OpExecutionMode %main InputLines
               OpExecutionMode %main Invocations 1
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 7

               ; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

               OpDecorate %in_color Location 2
               OpDecorate %out_color Location 0

               ; Types
      %void  = OpTypeVoid
   %voidfn   = OpTypeFunction %void
     %float  = OpTypeFloat 32
   %v2float  = OpTypeVector %float 2
   %v3float  = OpTypeVector %float 3
   %v4float  = OpTypeVector %float 4
      %int   = OpTypeInt 32 1
     %uint   = OpTypeInt 32 0

    %int_0   = OpConstant %int 0
    %int_1   = OpConstant %int 1
   %uint_2   = OpConstant %uint 2

  %float_0   = OpConstant %float 0.0
  %float_1   = OpConstant %float 1.0
%float_0p75  = OpConstant %float 0.75
%float_0p015 = OpConstant %float 0.015
%float_0p05  = OpConstant %float 0.05
%float_n1    = OpConstant %float -1.0

               ; Input gl_PerVertex
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in   = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr  = OpTypePointer Input %arr_pv_in
%ptr_in_v4   = OpTypePointer Input %v4float
    %gl_in   = OpVariable %ptr_in_arr Input

               ; Input color (location 2, array of 2)
%arr_v3_2    = OpTypeArray %v3float %uint_2
%ptr_in_arr3 = OpTypePointer Input %arr_v3_2
%ptr_in_v3   = OpTypePointer Input %v3float
  %in_color  = OpVariable %ptr_in_arr3 Input

               ; Output gl_PerVertex
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv  = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4  = OpTypePointer Output %v4float
   %gl_out   = OpVariable %ptr_out_pv Output

               ; Output color (location 0)
%ptr_out_v3  = OpTypePointer Output %v3float
 %out_color  = OpVariable %ptr_out_v3 Output

               ; Function
     %main   = OpFunction %void None %voidfn
    %entry   = OpLabel

               ; Load p0, p1 (gl_Position)
  %p0_ptr    = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
      %p0_4  = OpLoad %v4float %p0_ptr
  %p1_ptr    = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
      %p1_4  = OpLoad %v4float %p1_ptr

               ; Load colors
  %c0_ptr    = OpAccessChain %ptr_in_v3 %in_color %int_0
      %c0    = OpLoad %v3float %c0_ptr
  %c1_ptr    = OpAccessChain %ptr_in_v3 %in_color %int_1
      %c1    = OpLoad %v3float %c1_ptr

               ; Extract xy from p0, p1
   %p0x      = OpCompositeExtract %float %p0_4 0
   %p0y      = OpCompositeExtract %float %p0_4 1
   %p1x      = OpCompositeExtract %float %p1_4 0
   %p1y      = OpCompositeExtract %float %p1_4 1

               ; dir = p1 - p0
   %dx       = OpFSub %float %p1x %p0x
   %dy       = OpFSub %float %p1y %p0y

               ; perp = normalize(-dir.y, dir.x)
   %neg_dy   = OpFNegate %float %dy
   %perp_raw = OpCompositeConstruct %v2float %neg_dy %dx
   %perp     = OpExtInst %v2float %1 Normalize %perp_raw
   %perp_x   = OpCompositeExtract %float %perp 0
   %perp_y   = OpCompositeExtract %float %perp 1

               ; mid = p0 + 0.75 * (p1 - p0)
   %mx       = OpFMul %float %dx %float_0p75
   %my       = OpFMul %float %dy %float_0p75
   %midx     = OpFAdd %float %p0x %mx
   %midy     = OpFAdd %float %p0y %my

               ; Shaft offsets: perp * 0.015
   %sx       = OpFMul %float %perp_x %float_0p015
   %sy       = OpFMul %float %perp_y %float_0p015

               ; Output pointer for gl_Position
   %opos_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0

               ; === SHAFT (4 verts triangle strip) ===
               ; Vert 0: p0 + perp*0.015
   %v0x      = OpFAdd %float %p0x %sx
   %v0y      = OpFAdd %float %p0y %sy
   %v0       = OpCompositeConstruct %v4float %v0x %v0y %float_0 %float_1
               OpStore %opos_ptr %v0
               OpStore %out_color %c0
               OpEmitVertex

               ; Vert 1: p0 - perp*0.015
   %v1x      = OpFSub %float %p0x %sx
   %v1y      = OpFSub %float %p0y %sy
   %v1       = OpCompositeConstruct %v4float %v1x %v1y %float_0 %float_1
               OpStore %opos_ptr %v1
               OpStore %out_color %c0
               OpEmitVertex

               ; Vert 2: mid + perp*0.015
   %v2x      = OpFAdd %float %midx %sx
   %v2y      = OpFAdd %float %midy %sy
   %v2       = OpCompositeConstruct %v4float %v2x %v2y %float_0 %float_1
               OpStore %opos_ptr %v2
               OpStore %out_color %c0
               OpEmitVertex

               ; Vert 3: mid - perp*0.015
   %v3x      = OpFSub %float %midx %sx
   %v3y      = OpFSub %float %midy %sy
   %v3       = OpCompositeConstruct %v4float %v3x %v3y %float_0 %float_1
               OpStore %opos_ptr %v3
               OpStore %out_color %c0
               OpEmitVertex

               OpEndPrimitive

               ; === ARROWHEAD (3 verts triangle strip) ===
               ; Arrow offsets: perp * 0.05
   %ax       = OpFMul %float %perp_x %float_0p05
   %ay       = OpFMul %float %perp_y %float_0p05

               ; Vert 4: mid + perp*0.05
   %v4x      = OpFAdd %float %midx %ax
   %v4y      = OpFAdd %float %midy %ay
   %v4       = OpCompositeConstruct %v4float %v4x %v4y %float_0 %float_1
               OpStore %opos_ptr %v4
               OpStore %out_color %c1
               OpEmitVertex

               ; Vert 5: mid - perp*0.05
   %v5x      = OpFSub %float %midx %ax
   %v5y      = OpFSub %float %midy %ay
   %v5       = OpCompositeConstruct %v4float %v5x %v5y %float_0 %float_1
               OpStore %opos_ptr %v5
               OpStore %out_color %c1
               OpEmitVertex

               ; Vert 6: p1 (tip)
               OpStore %opos_ptr %p1_4
               OpStore %out_color %c1
               OpEmitVertex

               OpEndPrimitive

               OpReturn
               OpFunctionEnd
