const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { machineCount, tasks } = data;
const n = tasks.length;

const byId = new Map();
for (const t of tasks) {
  byId.set(t.id, t);
}

// Build adjacency and in-degree
const children = new Map();
const inDeg = new Map();
for (const t of tasks) {
  inDeg.set(t.id, (t.deps && t.deps.length) || 0);
  children.set(t.id, []);
}
for (const t of tasks) {
  if (t.deps) {
    for (const d of t.deps) {
      children.get(d).push(t.id);
    }
  }
}

// Compute bottom-level (longest path to sink) for each task
const blevel = new Map();
function computeBlevel(id) {
  if (blevel.has(id)) return blevel.get(id);
  const t = byId.get(id);
  let maxChild = 0;
  for (const c of children.get(id)) {
    maxChild = Math.max(maxChild, computeBlevel(c));
  }
  const val = t.duration + maxChild;
  blevel.set(id, val);
  return val;
}
for (const t of tasks) computeBlevel(t.id);

// Schedule using list scheduling with priority = weighted combo of deadline urgency and blevel
// Simulate: track machine availability and task finish times

const finishTime = new Map();
const schedule = [];

// Priority queue using sorted ready list
const inDegree = new Map();
for (const t of tasks) {
  inDegree.set(t.id, (t.deps && t.deps.length) || 0);
}

const ready = [];
for (const t of tasks) {
  if (inDegree.get(t.id) === 0) ready.push(t.id);
}

// Machine availability times
const machineAvail = new Array(machineCount).fill(0);

function getPriority(id) {
  const t = byId.get(id);
  // Primary: deadline urgency (earlier deadline first), secondary: higher blevel, tertiary: higher weight
  const dl = t.deadline != null ? t.deadline : 1e15;
  // Weighted slack: deadline - earliest_start - duration; lower slack = more urgent
  let earliest = 0;
  if (t.deps) {
    for (const d of t.deps) {
      const ft = finishTime.get(d);
      if (ft !== undefined && ft > earliest) earliest = ft;
    }
  }
  const slack = dl - earliest - t.duration;
  return { slack, bl: -blevel.get(id), w: -(t.weight || 0) };
}

function comparePriority(a, b) {
  const pa = getPriority(a);
  const pb = getPriority(b);
  // Lower slack first
  if (pa.slack !== pb.slack) return pa.slack - pb.slack;
  // Higher blevel first (stored negative)
  if (pa.bl !== pb.bl) return pa.bl - pb.bl;
  // Higher weight first
  if (pa.w !== pb.w) return pa.w - pb.w;
  return 0;
}

while (ready.length > 0) {
  // Sort ready list by priority
  ready.sort(comparePriority);

  const id = ready.shift();
  const t = byId.get(id);

  // Earliest start based on deps
  let earliest = 0;
  if (t.deps) {
    for (const d of t.deps) {
      const ft = finishTime.get(d);
      if (ft !== undefined && ft > earliest) earliest = ft;
    }
  }

  // Find machine that allows earliest start
  let bestMachine = 0;
  let bestStart = Math.max(earliest, machineAvail[0]);
  for (let m = 1; m < machineCount; m++) {
    const start = Math.max(earliest, machineAvail[m]);
    if (start < bestStart) {
      bestStart = start;
      bestMachine = m;
    }
  }

  schedule.push({ id: t.id, start: bestStart, machine: bestMachine });
  const ft = bestStart + t.duration;
  finishTime.set(t.id, ft);
  machineAvail[bestMachine] = ft;

  // Update children
  for (const c of children.get(t.id)) {
    inDegree.set(c, inDegree.get(c) - 1);
    if (inDegree.get(c) === 0) {
      ready.push(c);
    }
  }
}

console.log(JSON.stringify({ schedule }));
