import math

# Module-level state (reset between scenarios/go-arounds)
_prev_gs_error = 0.0
_prev_loc_error = 0.0
_prev_spd_error = 0.0
_prev_roll = 0.0
_gs_error_integral = 0.0
_loc_error_integral = 0.0
_spd_error_integral = 0.0
_step_count = 0
_go_around = False
_flare_started = False
_touchdown = False
_prev_vs = 0.0
_prev_pitch_cmd = 0.0
_retard_started = False


def _voted(vals, valids=None):
    if valids is not None:
        good = [v for v, ok in zip(vals, valids) if ok]
    else:
        good = [v for v in vals if v is not None and not math.isnan(v)]
    if not good:
        return vals[0] if vals else 0.0
    if len(good) >= 3:
        good.sort()
        return good[1]
    return sum(good) / len(good)


def autopilot_step(state: dict, dt: float) -> dict:
    global _prev_gs_error, _prev_loc_error, _prev_spd_error, _prev_roll
    global _gs_error_integral, _loc_error_integral, _spd_error_integral
    global _step_count, _go_around, _flare_started, _touchdown
    global _prev_vs, _prev_pitch_cmd, _retard_started

    s = state
    dt = max(dt, 0.001)
    _step_count += 1

    # --- Sensor voting ---
    gs_dots = _voted(
        [s.get('glideslope_1_dots', 0), s.get('glideslope_2_dots', 0), s.get('glideslope_3_dots', 0)],
        [s.get('glideslope_1_valid', False), s.get('glideslope_2_valid', False), s.get('glideslope_3_valid', False)]
    )
    loc_dots = _voted(
        [s.get('localizer_1_dots', 0), s.get('localizer_2_dots', 0), s.get('localizer_3_dots', 0)],
        [s.get('localizer_1_valid', False), s.get('localizer_2_valid', False), s.get('localizer_3_valid', False)]
    )
    ias = _voted(
        [s.get('ias_1_kt', 140), s.get('ias_2_kt', 140), s.get('ias_3_kt', 140)],
        [s.get('ias_1_valid', True), s.get('ias_2_valid', True), s.get('ias_3_valid', True)]
    )
    pitch = _voted(
        [s.get('irs_1_pitch_deg', 0), s.get('irs_2_pitch_deg', 0), s.get('irs_3_pitch_deg', 0)],
        [s.get('irs_1_valid', True), s.get('irs_2_valid', True), s.get('irs_3_valid', True)]
    )
    roll = _voted(
        [s.get('irs_1_roll_deg', 0), s.get('irs_2_roll_deg', 0), s.get('irs_3_roll_deg', 0)],
        [s.get('irs_1_valid', True), s.get('irs_2_valid', True), s.get('irs_3_valid', True)]
    )
    heading = _voted(
        [s.get('irs_1_heading_deg', 0), s.get('irs_2_heading_deg', 0), s.get('irs_3_heading_deg', 0)],
        [s.get('irs_1_valid', True), s.get('irs_2_valid', True), s.get('irs_3_valid', True)]
    )
    vs = _voted(
        [s.get('irs_1_vs_fpm', 0), s.get('irs_2_vs_fpm', 0), s.get('irs_3_vs_fpm', 0)],
        [s.get('irs_1_valid', True), s.get('irs_2_valid', True), s.get('irs_3_valid', True)]
    )
    baro_alt = _voted(
        [s.get('baro_alt_1_ft', 0), s.get('baro_alt_2_ft', 0), s.get('baro_alt_3_ft', 0)],
        [s.get('baro_alt_1_valid', True), s.get('baro_alt_2_valid', True), s.get('baro_alt_3_valid', True)]
    )
    aoa = _voted(
        [s.get('aoa_1_deg', 0), s.get('aoa_2_deg', 0), s.get('aoa_3_deg', 0)],
        [s.get('aoa_1_valid', True), s.get('aoa_2_valid', True), s.get('aoa_3_valid', True)]
    )
    track = _voted(
        [s.get('irs_1_track_deg', heading), s.get('irs_2_track_deg', heading), s.get('irs_3_track_deg', heading)],
        [s.get('irs_1_valid', True), s.get('irs_2_valid', True), s.get('irs_3_valid', True)]
    )

    radio_alt = s.get('radio_alt_ft', baro_alt)
    on_ground = s.get('on_ground', False)
    wow = s.get('weight_on_wheels', False)
    groundspeed = s.get('groundspeed_kt', ias)
    target_vref = s.get('target_vref_kt', 137)
    target_ias = s.get('target_ias_kt', target_vref + 5)
    target_hdg = s.get('target_hdg_deg', heading)
    below_dh = s.get('below_decision_height', False)
    eng1_running = s.get('engine_1_running', True)
    eng2_running = s.get('engine_2_running', True)
    eng1_n1 = s.get('engine_1_n1_pct', 0)
    eng2_n1 = s.get('engine_2_n1_pct', 0)
    dme = min(s.get('dme_1_nm', 99), s.get('dme_2_nm', 99))

    any_gs_valid = any(s.get(f'glideslope_{i}_valid', False) for i in range(1, 4))
    any_loc_valid = any(s.get(f'localizer_{i}_valid', False) for i in range(1, 4))

    out = {}

    # --- Touchdown / Rollout ---
    if _touchdown or (wow and radio_alt < 5 and not _go_around):
        _touchdown = True
        out['spoiler'] = 1.0
        out['speedbrake'] = 1.0
        out['autobrake'] = 0.9
        out['gear'] = 1.0
        out['flaps'] = 30.0

        if groundspeed > 80:
            if eng1_running:
                out['reverse_1'] = 0.8
            if eng2_running:
                out['reverse_2'] = 0.8
        else:
            out['reverse_1'] = 0.0
            out['reverse_2'] = 0.0

        # Keep straight with rudder
        loc_rudder = -loc_dots * 0.3 if any_loc_valid else 0.0
        out['rudder'] = max(-1.0, min(1.0, loc_rudder))
        out['elevator'] = max(-0.3, min(0.3, pitch * -0.02))
        out['aileron'] = max(-0.5, min(0.5, -roll * 0.03))
        out['throttle_1'] = 0.0
        out['throttle_2'] = 0.0
        _prev_vs = vs
        return out

    # --- Go-around logic ---
    if _go_around:
        out['gear'] = 1.0
        out['flaps'] = 20.0
        out['speedbrake'] = 0.0
        out['spoiler'] = 0.0
        out['throttle_1'] = 1.0 if eng1_running else 0.0
        out['throttle_2'] = 1.0 if eng2_running else 0.0

        target_pitch = 12.0
        pitch_err = target_pitch - pitch
        out['elevator'] = max(-1.0, min(1.0, pitch_err * 0.08))
        out['aileron'] = max(-1.0, min(1.0, -roll * 0.04))
        out['rudder'] = max(-1.0, min(1.0, -roll * 0.01))
        out['reverse_1'] = 0.0
        out['reverse_2'] = 0.0
        out['autobrake'] = 0.0

        if radio_alt > 1500:
            _go_around = False
        _prev_vs = vs
        return out

    # --- Decide go-around ---
    no_engines = not eng1_running and not eng2_running

    # Scenario 21: missed approach required
    if below_dh and radio_alt > 150 and not any_gs_valid and not any_loc_valid and not no_engines:
        _go_around = True
        return autopilot_step(state, dt)

    # Unstable approach checks (but not for dual engine failure - must land)
    if not no_engines and radio_alt > 200 and radio_alt < 900:
        if ias > target_vref + 35 and _step_count > 20:
            _go_around = True
            return autopilot_step(state, dt)
        if abs(gs_dots) > 3.0 and any_gs_valid and _step_count > 30:
            _go_around = True
            return autopilot_step(state, dt)

    # --- Configuration ---
    out['gear'] = 1.0
    if radio_alt > 2000:
        out['flaps'] = 20.0
    elif radio_alt > 1000:
        out['flaps'] = 25.0
    else:
        out['flaps'] = 30.0
    out['speedbrake'] = 0.0
    out['spoiler'] = 0.0

    # --- Speed control ---
    if no_engines:
        approach_spd = target_vref + 5
    elif radio_alt > 500:
        approach_spd = target_vref + 10
    elif radio_alt > 200:
        approach_spd = target_vref + 7
    else:
        approach_spd = target_vref + 5

    spd_error = approach_spd - ias
    _spd_error_integral += spd_error * dt
    _spd_error_integral = max(-20, min(20, _spd_error_integral))
    spd_deriv = (spd_error - _prev_spd_error) / dt
    _prev_spd_error = spd_error

    # --- Flare phase ---
    is_flare = radio_alt < 50 and not wow

    if is_flare:
        _flare_started = True

    if _flare_started and not wow:
        out['throttle_1'] = 0.0 if eng1_running else 0.0
        out['throttle_2'] = 0.0 if eng2_running else 0.0

        # Flare: reduce descent rate progressively
        if radio_alt < 10:
            target_vs = -80
        elif radio_alt < 20:
            target_vs = -120
        elif radio_alt < 35:
            target_vs = -200
        else:
            target_vs = -400

        vs_err = target_vs - vs
        pitch_cmd = vs_err * -0.003 + (vs - _prev_vs) * -0.001

        # Keep some nose up in flare
        if radio_alt < 20:
            pitch_cmd += 0.05

        out['elevator'] = max(-0.3, min(0.5, pitch_cmd))

        # Lateral: track localizer
        if any_loc_valid:
            loc_cmd = -loc_dots * 0.15
            out['aileron'] = max(-0.3, min(0.3, loc_cmd - roll * 0.02))
        else:
            out['aileron'] = max(-0.3, min(0.3, -roll * 0.03))
        out['rudder'] = max(-0.3, min(0.3, -roll * 0.01))

        _prev_vs = vs
        _prev_pitch_cmd = pitch_cmd
        return out

    # --- Normal approach: glideslope + localizer tracking ---

    # Vertical: track glideslope
    if any_gs_valid:
        gs_error = -gs_dots  # negative dots = below, we want to go up
        _gs_error_integral += gs_error * dt
        _gs_error_integral = max(-5, min(5, _gs_error_integral))
        gs_deriv = (gs_error - _prev_gs_error) / dt
        _prev_gs_error = gs_error

        # PID for elevator via glideslope
        kp = 0.12
        ki = 0.008
        kd = 0.04
        if radio_alt < 200:
            kp = 0.08
            ki = 0.004
            kd = 0.03

        elev_cmd = kp * gs_error + ki * _gs_error_integral + kd * gs_deriv
    else:
        # No glideslope: use target VS or a standard 3deg path
        target_vs_val = s.get('target_vs_fpm', -700)
        vs_err = target_vs_val - vs
        elev_cmd = vs_err * -0.0004
        _prev_gs_error = 0

    # Add pitch damping
    pitch_rate = (pitch - _prev_pitch_cmd) if _step_count > 1 else 0
    elev_cmd -= pitch_rate * 0.01

    out['elevator'] = max(-0.5, min(0.5, elev_cmd))
    _prev_pitch_cmd = pitch

    # Throttle for speed
    if no_engines:
        out['throttle_1'] = 0.0
        out['throttle_2'] = 0.0
        # Use speedbrake for energy management in glide
        if ias > approach_spd + 10:
            out['speedbrake'] = 0.5
        elif ias > approach_spd + 5:
            out['speedbrake'] = 0.2
        else:
            out['speedbrake'] = 0.0
    else:
        throttle_base = 0.35
        throttle_cmd = throttle_base + spd_error * 0.02 + _spd_error_integral * 0.003 + spd_deriv * 0.005
        throttle_cmd = max(0.05, min(0.95, throttle_cmd))

        # Retard thrust below 30ft
        if radio_alt < 30 and _retard_started is False:
            _retard_started = True
        if _retard_started:
            throttle_cmd = max(0.0, throttle_cmd * (radio_alt / 50.0))

        if eng1_running:
            out['throttle_1'] = throttle_cmd
        else:
            out['throttle_1'] = 0.0
        if eng2_running:
            out['throttle_2'] = throttle_cmd
        else:
            out['throttle_2'] = 0.0

        # Single engine: add rudder to compensate yaw
        if eng1_running and not eng2_running:
            out['rudder'] = max(-1.0, min(1.0, -0.15 - roll * 0.01))
        elif eng2_running and not eng1_running:
            out['rudder'] = max(-1.0, min(1.0, 0.15 - roll * 0.01))

    # Lateral: track localizer
    if any_loc_valid:
        loc_error = -loc_dots  # negative dots = left, positive command = roll right
        _loc_error_integral += loc_error * dt
        _loc_error_integral = max(-10, min(10, _loc_error_integral))
        loc_deriv = (loc_error - _prev_loc_error) / dt
        _prev_loc_error = loc_error

        kp_loc = 0.15
        ki_loc = 0.005
        kd_loc = 0.06

        bank_cmd = kp_loc * loc_error + ki_loc * _loc_error_integral + kd_loc * loc_deriv
        bank_cmd = max(-25, min(25, bank_cmd))  # limit bank angle

        # Bank angle to aileron
        bank_error = bank_cmd - roll
        ail_cmd = bank_error * 0.03
        out['aileron'] = max(-1.0, min(1.0, ail_cmd))
    else:
        # Wings level, track heading
        hdg_err = target_hdg - heading
        if hdg_err > 180:
            hdg_err -= 360
        if hdg_err < -180:
            hdg_err += 360
        bank_cmd = hdg_err * 1.5
        bank_cmd = max(-20, min(20, bank_cmd))
        bank_error = bank_cmd - roll
        out['aileron'] = max(-1.0, min(1.0, bank_error * 0.03))

    # Rudder for coordination (if not already set for engine out)
    if 'rudder' not in out:
        out['rudder'] = max(-1.0, min(1.0, -roll * 0.01))

    _prev_vs = vs
    return out
