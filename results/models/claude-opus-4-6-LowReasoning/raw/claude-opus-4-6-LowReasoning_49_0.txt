import struct

def op(opcode, *args):
    wc = 1 + len(args)
    return [(wc << 16) | opcode] + list(args)

def spirv_str(s):
    b = s.encode() + b'\x00'
    b += b'\x00' * ((-len(b)) % 4)
    return [int.from_bytes(b[i:i+4], 'little') for i in range(0, len(b), 4)]

# ID assignments
VOID, UINT, BOOL = 1, 2, 3
UVEC3, UVEC4, FTYPE = 4, 5, 6
PI_UVEC3, PI_UINT = 7, 8
RTA, S_BUF, PU_BUF = 9, 10, 11
S_UNI, PU_UNI, PU_UINT = 12, 13, 14
C0, C1 = 15, 16
GID_VAR, IN_VAR, OUT_VAR, UNI_VAR = 17, 18, 19, 20
MAIN = 21
LENTRY, LBODY, LRET, LMERGE = 22, 23, 24, 25
LLOOPH, LLOOPB, LLOOPC, LLOOPM = 26, 27, 28, 29
vGP, vGD, vNP, vN, vCOND = 30, 31, 32, 33, 34
vSUM, vI, vLC = 35, 36, 37
vEP, vEL, vSN, vIN, vOP = 38, 39, 40, 41, 42
BOUND = 43

w = []
# Header: magic, version 1.0, generator, bound, schema
w += [0x07230203, 0x00010000, 0, BOUND, 0]

# Capability Shader
w += op(17, 1)
# MemoryModel Logical GLSL450
w += op(14, 0, 1)
# EntryPoint GLCompute %MAIN "main" %GID_VAR
w += op(15, 5, MAIN, *spirv_str("main"), GID_VAR)
# ExecutionMode LocalSize 256 1 1
w += op(16, MAIN, 17, 256, 1, 1)

# Decorations
w += op(71, GID_VAR, 11, 28)   # BuiltIn GlobalInvocationId
w += op(71, S_BUF, 3)          # BufferBlock (SSBO in SPV 1.0)
w += op(71, S_UNI, 2)          # Block (UBO)
w += op(71, IN_VAR, 34, 0)     # DescriptorSet 0
w += op(71, IN_VAR, 33, 0)     # Binding 0
w += op(71, OUT_VAR, 34, 0)
w += op(71, OUT_VAR, 33, 1)
w += op(71, UNI_VAR, 34, 0)
w += op(71, UNI_VAR, 33, 2)
w += op(71, RTA, 6, 4)         # ArrayStride 4
w += op(72, S_BUF, 0, 35, 0)   # MemberDecorate Offset 0
w += op(72, S_UNI, 0, 35, 0)

# Types
w += op(19, VOID)               # OpTypeVoid
w += op(21, UINT, 32, 0)        # OpTypeInt 32 unsigned
w += op(20, BOOL)                # OpTypeBool
w += op(23, UVEC3, UINT, 3)     # OpTypeVector uvec3
w += op(23, UVEC4, UINT, 4)     # OpTypeVector uvec4
w += op(33, FTYPE, VOID)        # OpTypeFunction void()
w += op(32, PI_UVEC3, 1, UVEC3) # ptr Input uvec3
w += op(32, PI_UINT, 1, UINT)   # ptr Input uint
w += op(29, RTA, UINT)          # OpTypeRuntimeArray uint
w += op(30, S_BUF, RTA)         # OpTypeStruct { uint[] }
w += op(32, PU_BUF, 2, S_BUF)   # ptr Uniform struct_ssbo
w += op(30, S_UNI, UVEC4)       # OpTypeStruct { uvec4 }
w += op(32, PU_UNI, 2, S_UNI)   # ptr Uniform struct_ubo
w += op(32, PU_UINT, 2, UINT)   # ptr Uniform uint

# Constants
w += op(43, UINT, C0, 0)        # const 0
w += op(43, UINT, C1, 1)        # const 1

# Variables
w += op(59, PI_UVEC3, GID_VAR, 1)  # gl_GlobalInvocationID (Input)
w += op(59, PU_BUF, IN_VAR, 2)     # input buffer (Uniform)
w += op(59, PU_BUF, OUT_VAR, 2)    # output buffer (Uniform)
w += op(59, PU_UNI, UNI_VAR, 2)    # uniform buffer (Uniform)

# Function main
w += op(54, VOID, MAIN, 0, FTYPE)  # OpFunction
w += op(248, LENTRY)                # OpLabel entry

# gid = gl_GlobalInvocationID.x
w += op(65, PI_UINT, vGP, GID_VAR, C0)   # AccessChain
w += op(61, UINT, vGD, vGP)               # Load

# N = uniform.params.x
w += op(65, PU_UINT, vNP, UNI_VAR, C0, C0)  # AccessChain
w += op(61, UINT, vN, vNP)                    # Load

# if (gid >= N) return
w += op(174, BOOL, vCOND, vGD, vN)   # UGreaterThanEqual
w += op(247, LMERGE, 0)               # SelectionMerge
w += op(250, vCOND, LRET, LBODY)      # BranchConditional

w += op(248, LRET)                     # early return path
w += op(249, LMERGE)

w += op(248, LBODY)                    # main body
w += op(249, LLOOPH)

# Loop header
w += op(248, LLOOPH)
w += op(245, UINT, vSUM, C0, LBODY, vSN, LLOOPC)   # phi sum
w += op(245, UINT, vI, C0, LBODY, vIN, LLOOPC)      # phi i
w += op(178, BOOL, vLC, vI, vGD)                      # i <= gid
w += op(246, LLOOPM, LLOOPC, 0)                        # LoopMerge
w += op(250, vLC, LLOOPB, LLOOPM)                      # BranchConditional

# Loop body
w += op(248, LLOOPB)
w += op(65, PU_UINT, vEP, IN_VAR, C0, vI)   # AccessChain input[i]
w += op(61, UINT, vEL, vEP)                  # Load
w += op(128, UINT, vSN, vSUM, vEL)           # sum += elem
w += op(128, UINT, vIN, vI, C1)              # i++
w += op(249, LLOOPC)

# Loop continue
w += op(248, LLOOPC)
w += op(249, LLOOPH)

# Loop merge - store result
w += op(248, LLOOPM)
w += op(65, PU_UINT, vOP, OUT_VAR, C0, vGD)  # AccessChain output[gid]
w += op(62, vOP, vSUM)                         # Store
w += op(249, LMERGE)

# Selection merge - return
w += op(248, LMERGE)
w += op(253)   # OpReturn
w += op(56)    # OpFunctionEnd

w[3] = BOUND
print(''.join(struct.pack('<I', v).hex() for v in w))
