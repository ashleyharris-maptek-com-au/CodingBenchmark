#include <cstdio>
#include <cmath>
#include <vector>
#include <array>
#include <algorithm>
#include <unordered_map>

struct Vec3 {
    double x, y, z;
    Vec3() : x(0), y(0), z(0) {}
    Vec3(double x_, double y_, double z_) : x(x_), y(y_), z(z_) {}
    Vec3 operator+(const Vec3& o) const { return {x+o.x, y+o.y, z+o.z}; }
    Vec3 operator-(const Vec3& o) const { return {x-o.x, y-o.y, z-o.z}; }
    Vec3 operator*(double s) const { return {x*s, y*s, z*s}; }
    double dot(const Vec3& o) const { return x*o.x + y*o.y + z*o.z; }
    Vec3 cross(const Vec3& o) const {
        return {y*o.z - z*o.y, z*o.x - x*o.z, x*o.y - y*o.x};
    }
    double norm2() const { return x*x + y*y + z*z; }
    double norm() const { return std::sqrt(norm2()); }
};

struct Face {
    Vec3 normal;
    double d;
};

std::vector<Face> hullFaces;

void buildConvexHull(const std::vector<Vec3>& pts) {
    hullFaces.clear();
    int n = (int)pts.size();
    if (n < 4) return;

    Vec3 centroid{0,0,0};
    for (auto& p : pts) { centroid.x += p.x; centroid.y += p.y; centroid.z += p.z; }
    centroid = centroid * (1.0 / n);

    if (n <= 200) {
        for (int i = 0; i < n; i++)
        for (int j = i+1; j < n; j++)
        for (int k = j+1; k < n; k++) {
            Vec3 ab = pts[j] - pts[i];
            Vec3 ac = pts[k] - pts[i];
            Vec3 norm = ab.cross(ac);
            double len = norm.norm();
            if (len < 1e-12) continue;
            norm = norm * (1.0/len);
            double d0 = norm.dot(pts[i]);

            int side = 0;
            bool valid = true;
            for (int m = 0; m < n; m++) {
                if (m == i || m == j || m == k) continue;
                double val = norm.dot(pts[m]) - d0;
                if (val > 1e-9) {
                    if (side < 0) { valid = false; break; }
                    side = 1;
                } else if (val < -1e-9) {
                    if (side > 0) { valid = false; break; }
                    side = -1;
                }
            }
            if (!valid) continue;

            Face f;
            f.normal = norm;
            f.d = d0;
            // Ensure centroid is on the <= side
            if (f.normal.dot(centroid) > f.d) {
                f.normal = f.normal * (-1.0);
                f.d = -f.d;
            }
            hullFaces.push_back(f);
        }
    } else {
        Vec3 mn = pts[0], mx = pts[0];
        for (auto& p : pts) {
            mn.x = std::min(mn.x, p.x); mn.y = std::min(mn.y, p.y); mn.z = std::min(mn.z, p.z);
            mx.x = std::max(mx.x, p.x); mx.y = std::max(mx.y, p.y); mx.z = std::max(mx.z, p.z);
        }
        hullFaces.push_back({{1,0,0}, mx.x});
        hullFaces.push_back({{-1,0,0}, -mn.x});
        hullFaces.push_back({{0,1,0}, mx.y});
        hullFaces.push_back({{0,-1,0}, -mn.y});
        hullFaces.push_back({{0,0,1}, mx.z});
        hullFaces.push_back({{0,0,-1}, -mn.z});
    }
}

bool pointInHull(const Vec3& p, double margin = 0.0) {
    for (auto& f : hullFaces) {
        if (f.normal.dot(p) > f.d - margin) return false;
    }
    return true;
}

void getTetraVerts(double e, const Vec3& center, double rx, double ry, double rz, Vec3 out[4]) {
    double a = e / (2.0 * std::sqrt(2.0));
    Vec3 v[4] = {{a,a,a},{a,-a,-a},{-a,a,-a},{-a,-a,a}};

    double cx_ = std::cos(rx), sx_ = std::sin(rx);
    double cy_ = std::cos(ry), sy_ = std::sin(ry);
    double cz_ = std::cos(rz), sz_ = std::sin(rz);

    double R[3][3] = {
        {cz_*cy_, cz_*sy_*sx_ - sz_*cx_, cz_*sy_*cx_ + sz_*sx_},
        {sz_*cy_, sz_*sy_*sx_ + cz_*cx_, sz_*sy_*cx_ - cz_*sx_},
        {-sy_,    cy_*sx_,               cy_*cx_}
    };

    for (int i = 0; i < 4; i++) {
        out[i].x = R[0][0]*v[i].x + R[0][1]*v[i].y + R[0][2]*v[i].z + center.x;
        out[i].y = R[1][0]*v[i].x + R[1][1]*v[i].y + R[1][2]*v[i].z + center.y;
        out[i].z = R[2][0]*v[i].x + R[2][1]*v[i].y + R[2][2]*v[i].z + center.z;
    }
}

bool tetraInside(double e, const Vec3& center, double rx, double ry, double rz) {
    Vec3 v[4];
    getTetraVerts(e, center, rx, ry, rz, v);
    for (int i = 0; i < 4; i++) {
        if (!pointInHull(v[i])) return false;
    }
    return true;
}

bool projectSeparated(const Vec3 a[4], const Vec3 b[4], const Vec3& axis) {
    if (axis.norm2() < 1e-20) return false;
    double amin = 1e30, amax = -1e30, bmin = 1e30, bmax = -1e30;
    for (int i = 0; i < 4; i++) {
        double pa = axis.dot(a[i]);
        double pb = axis.dot(b[i]);
        if (pa < amin) amin = pa; if (pa > amax) amax = pa;
        if (pb < bmin) bmin = pb; if (pb > bmax) bmax = pb;
    }
    return (amax < bmin + 1e-9 || bmax < amin + 1e-9);
}

bool tetraOverlap(const Vec3 a[4], const Vec3 b[4]) {
    static const int faces[][3] = {{0,1,2},{0,1,3},{0,2,3},{1,2,3}};
    static const int edges[][2] = {{0,1},{0,2},{0,3},{1,2},{1,3},{2,3}};

    for (int i = 0; i < 4; i++) {
        Vec3 n = (a[faces[i][1]]-a[faces[i][0]]).cross(a[faces[i][2]]-a[faces[i][0]]);
        if (projectSeparated(a, b, n)) return false;
    }
    for (int i = 0; i < 4; i++) {
        Vec3 n = (b[faces[i][1]]-b[faces[i][0]]).cross(b[faces[i][2]]-b[faces[i][0]]);
        if (projectSeparated(a, b, n)) return false;
    }
    for (int i = 0; i < 6; i++) {
        Vec3 ea = a[edges[i][1]] - a[edges[i][0]];
        for (int j = 0; j < 6; j++) {
            Vec3 eb = b[edges[j][1]] - b[edges[j][0]];
            Vec3 axis = ea.cross(eb);
            if (projectSeparated(a, b, axis)) return false;
        }
    }
    return true;
}

struct Placement {
    Vec3 center;
    double rx, ry, rz;
    Vec3 verts[4];
};

// Spatial hash for fast neighbor lookup
struct SpatialHash {
    double cellSize;
    std::unordered_map<long long, std::vector<int>> grid;

    void init(double cs) { cellSize = cs; grid.clear(); }

    long long key(int gx, int gy, int gz) const {
        long long a = gx + 500000LL;
        long long b = gy + 500000LL;
        long long c = gz + 500000LL;
        return a * 1000001LL * 1000001LL + b * 1000001LL + c;
    }

    void insert(int idx, const Vec3& pos) {
        int gx = (int)std::floor(pos.x / cellSize);
        int gy = (int)std::floor(pos.y / cellSize);
        int gz = (int)std::floor(pos.z / cellSize);
        grid[key(gx, gy, gz)].push_back(idx);
    }

    void queryNeighbors(const Vec3& pos, std::vector<int>& result) const {
        result.clear();
        int gx = (int)std::floor(pos.x / cellSize);
        int gy = (int)std::floor(pos.y / cellSize);
        int gz = (int)std::floor(pos.z / cellSize);
        for (int dx = -1; dx <= 1; dx++)
        for (int dy = -1; dy <= 1; dy++)
        for (int dz = -1; dz <= 1; dz++) {
            auto it = grid.find(key(gx+dx, gy+dy, gz+dz));
            if (it != grid.end()) {
                for (int i : it->second) result.push_back(i);
            }
        }
    }
};

int main() {
    int V;
    double edgeLen;
    if (scanf("%d %lf", &V, &edgeLen) != 2) return 1;

    std::vector<Vec3> containerVerts(V);
    for (int i = 0; i < V; i++) {
        scanf("%lf %lf %lf", &containerVerts[i].x, &containerVerts[i].y, &containerVerts[i].z);
    }

    buildConvexHull(containerVerts);

    Vec3 mn = containerVerts[0], mx = containerVerts[0];
    for (auto& p : containerVerts) {
        mn.x = std::min(mn.x, p.x); mn.y = std::min(mn.y, p.y); mn.z = std::min(mn.z, p.z);
        mx.x = std::max(mx.x, p.x); mx.y = std::max(mx.y, p.y); mx.z = std::max(mx.z, p.z);
    }

    double e = edgeLen;
    double circumR = e * std::sqrt(3.0/8.0);
    double a = e / (2.0 * std::sqrt(2.0));
    double step = 2.0 * a; // e/sqrt(2) - bounding box side of a single tet

    // Estimate grid size and adjust if needed
    double vol = (mx.x-mn.x)*(mx.y-mn.y)*(mx.z-mn.z);
    long long estCells = (long long)std::ceil((mx.x-mn.x)/step) *
                         (long long)std::ceil((mx.y-mn.y)/step) *
                         (long long)std::ceil((mx.z-mn.z)/step);

    double actualStep = step;
    if (estCells > 3000000LL) {
        double scale = std::cbrt((double)estCells / 3000000.0);
        actualStep = step * scale;
    }

    int nx = std::max(1, (int)std::ceil((mx.x - mn.x) / actualStep));
    int ny = std::max(1, (int)std::ceil((mx.y - mn.y) / actualStep));
    int nz = std::max(1, (int)std::ceil((mx.z - mn.z) / actualStep));

    std::vector<Placement> placed;
    SpatialHash spatial;
    spatial.init(2.0 * circumR + 0.01);

    std::vector<int> neighbors;

    // Generate grid candidates and try to place
    // Two passes: orientation 0 on main grid, orientation pi on offset grid
    auto tryPlace = [&](const Vec3& c, double rx, double ry, double rz) {
        if (!pointInHull(c, -circumR * 0.5)) return;
        if (!tetraInside(e, c, rx, ry, rz)) return;

        Vec3 verts[4];
        getTetraVerts(e, c, rx, ry, rz, verts);

        spatial.queryNeighbors(c, neighbors);
        for (int idx : neighbors) {
            Vec3 diff = c - placed[idx].center;
            if (diff.norm2() > (2.0*circumR+0.01)*(2.0*circumR+0.01)) continue;
            if (tetraOverlap(verts, placed[idx].verts)) return;
        }

        int pidx = (int)placed.size();
        Placement pl;
        pl.center = c;
        pl.rx = rx; pl.ry = ry; pl.rz = rz;
        for (int i = 0; i < 4; i++) pl.verts[i] = verts[i];
        placed.push_back(pl);
        spatial.insert(pidx, c);
    };

    // Main grid - orientation 0
    for (int iz = 0; iz < nz; iz++)
    for (int iy = 0; iy < ny; iy++)
    for (int ix = 0; ix < nx; ix++) {
        Vec3 c;
        c.x = mn.x + (ix + 0.5) * actualStep;
        c.y = mn.y + (iy + 0.5) * actualStep;
        c.z = mn.z + (iz + 0.5) * actualStep;
        tryPlace(c, 0, 0, 0);
    }

    // Interleaved grid - inverted orientation
    const double PI = 3.14159265358979323846;
    for (int iz = 0; iz < nz; iz++)
    for (int iy = 0; iy < ny; iy++)
    for (int ix = 0; ix < nx; ix++) {
        Vec3 c;
        c.x = mn.x + (ix + 1.0) * actualStep;
        c.y = mn.y + (iy + 1.0) * actualStep;
        c.z = mn.z + (iz + 1.0) * actualStep;
        tryPlace(c, PI, 0, 0);
    }

    printf("%d\n", (int)placed.size());
    for (auto& p : placed) {
        printf("%.9f %.9f %.9f %.9f %.9f %.9f\n",
            p.center.x, p.center.y, p.center.z,
            p.rx, p.ry, p.rz);
    }

    return 0;
}
