using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;

class MaxClique
{
    static int N;
    static List<int>[] adj;
    static HashSet<int>[] adjSet;
    static int[] bestClique;
    static int bestSize;
    static object bestLock = new object();
    static Stopwatch sw = new Stopwatch();
    static long timeLimitMs = 280000;
    static bool timeUp = false;

    static void Main()
    {
        sw.Start();
        var input = Console.In.ReadToEnd().Split(new[]{' ','\t','\r','\n'}, StringSplitOptions.RemoveEmptyEntries);
        int idx = 0;
        N = int.Parse(input[idx++]);
        int M = int.Parse(input[idx++]);

        adj = new List<int>[N];
        adjSet = new HashSet<int>[N];
        for (int i = 0; i < N; i++)
        {
            adj[i] = new List<int>();
            adjSet[i] = new HashSet<int>();
        }

        for (int i = 0; i < M; i++)
        {
            int u = int.Parse(input[idx++]);
            int v = int.Parse(input[idx++]);
            adj[u].Add(v);
            adj[v].Add(u);
            adjSet[u].Add(v);
            adjSet[v].Add(u);
        }

        int[] degOrder = Enumerable.Range(0, N).ToArray();
        int[] deg = new int[N];
        for (int i = 0; i < N; i++) deg[i] = adj[i].Count;

        bestClique = new int[0];
        bestSize = 0;

        // Run greedy first for a baseline
        GreedyClique();

        if (N <= 50000 && (long)M * 2 < (long)N * N)
        {
            // For manageable sizes, try exact BK with ordering
            ExactBK();
        }
        else
        {
            // Large graph: parallel greedy + local search
            ParallelLocalSearch();
        }

        Console.WriteLine(bestSize);
        Console.WriteLine(string.Join(" ", bestClique));
    }

    static void UpdateBest(List<int> clique)
    {
        lock (bestLock)
        {
            if (clique.Count > bestSize)
            {
                bestSize = clique.Count;
                bestClique = clique.ToArray();
            }
        }
    }

    static void GreedyClique()
    {
        int[] order = Enumerable.Range(0, N).OrderByDescending(v => adj[v].Count).ToArray();
        var clique = new List<int>();
        var cliqueSet = new HashSet<int>();

        foreach (int v in order)
        {
            bool canAdd = true;
            foreach (int c in clique)
            {
                if (!adjSet[v].Contains(c)) { canAdd = false; break; }
            }
            if (canAdd) { clique.Add(v); cliqueSet.Add(v); }
        }
        UpdateBest(clique);
    }

    static void ExactBK()
    {
        // Degeneracy ordering
        int[] ordNum = new int[N];
        int[] ordVert = new int[N];
        int[] tempDeg = new int[N];
        bool[] removed = new bool[N];
        for (int i = 0; i < N; i++) tempDeg[i] = adj[i].Count;

        var buckets = new LinkedList<int>[N];
        var nodes = new LinkedListNode<int>[N];
        for (int i = 0; i < N; i++) buckets[i] = new LinkedList<int>();
        for (int i = 0; i < N; i++)
        {
            int d = Math.Min(tempDeg[i], N - 1);
            nodes[i] = buckets[d].AddLast(i);
        }

        for (int i = 0; i < N; i++)
        {
            int minD = 0;
            while (buckets[minD].Count == 0) minD++;
            int v = buckets[minD].First.Value;
            buckets[minD].RemoveFirst();
            removed[v] = true;
            ordVert[i] = v;
            ordNum[v] = i;

            foreach (int u in adj[v])
            {
                if (!removed[u])
                {
                    int od = tempDeg[u];
                    buckets[od].Remove(nodes[u]);
                    tempDeg[u]--;
                    nodes[u] = buckets[tempDeg[u]].AddLast(u);
                }
            }
        }

        // Process vertices in reverse degeneracy order with parallel branching
        int totalVerts = N;
        int processed = 0;

        // For parallel BK: process each vertex in reverse order
        var tasks = new List<Task>();
        int parallelism = Math.Min(Environment.ProcessorCount, 32);
        var semaphore = new SemaphoreSlim(parallelism);

        for (int ii = totalVerts - 1; ii >= 0; ii--)
        {
            if (timeUp || sw.ElapsedMilliseconds > timeLimitMs) break;

            int v = ordVert[ii];
            // P = neighbors of v with higher order
            var P = new List<int>();
            foreach (int u in adj[v])
            {
                if (ordNum[u] > ordNum[v]) P.Add(u);
            }

            if (P.Count + 1 <= bestSize) continue;

            semaphore.Wait();
            if (timeUp || sw.ElapsedMilliseconds > timeLimitMs) { semaphore.Release(); break; }

            int capturedV = v;
            var capturedP = P;
            var t = Task.Run(() =>
            {
                try
                {
                    var curClique = new List<int> { capturedV };
                    var candidates = new List<int>(capturedP);
                    var excluded = new List<int>();
                    BronKerbosch(curClique, candidates, excluded, 0);
                }
                finally { semaphore.Release(); }
            });
            tasks.Add(t);
        }

        Task.WaitAll(tasks.ToArray());
    }

    static void BronKerbosch(List<int> clique, List<int> P, List<int> X, int depth)
    {
        if (sw.ElapsedMilliseconds > timeLimitMs) { timeUp = true; return; }

        if (P.Count == 0)
        {
            if (X.Count == 0) UpdateBest(new List<int>(clique));
            return;
        }

        if (clique.Count + P.Count <= bestSize) return;

        // Choose pivot with max connections to P
        int pivot = -1, pivotConn = -1;
        foreach (int u in P)
        {
            int conn = 0;
            foreach (int w in P) if (adjSet[u].Contains(w)) conn++;
            if (conn > pivotConn) { pivotConn = conn; pivot = u; }
        }
        foreach (int u in X)
        {
            int conn = 0;
            foreach (int w in P) if (adjSet[u].Contains(w)) conn++;
            if (conn > pivotConn) { pivotConn = conn; pivot = u; }
        }

        var branching = new List<int>();
        foreach (int v in P)
        {
            if (!adjSet[pivot].Contains(v)) branching.Add(v);
        }

        foreach (int v in branching)
        {
            if (timeUp) return;

            var newP = new List<int>();
            var newX = new List<int>();
            foreach (int u in P) if (adjSet[v].Contains(u)) newP.Add(u);
            foreach (int u in X) if (adjSet[v].Contains(u)) newX.Add(u);

            if (clique.Count + 1 + newP.Count <= bestSize) { P.Remove(v); X.Add(v); continue; }

            clique.Add(v);
            BronKerbosch(clique, newP, newX, depth + 1);
            clique.RemoveAt(clique.Count - 1);
            P.Remove(v);
            X.Add(v);
        }
    }

    static void ParallelLocalSearch()
    {
        int threads = Math.Min(Environment.ProcessorCount, 64);
        var tasks = new Task[threads];
        for (int t = 0; t < threads; t++)
        {
            int seed = t;
            tasks[t] = Task.Run(() => LocalSearchWorker(seed));
        }
        Task.WaitAll(tasks);
    }

    static void LocalSearchWorker(int seed)
    {
        var rng = new Random(seed * 31337 + 42);
        int[] perm = Enumerable.Range(0, N).ToArray();

        while (sw.ElapsedMilliseconds < timeLimitMs)
        {
            // Shuffle with bias toward high-degree vertices
            for (int i = 0; i < Math.Min(N, 5000); i++)
            {
                int j = rng.Next(i, Math.Min(N, 5000));
                int tmp = perm[i]; perm[i] = perm[j]; perm[j] = tmp;
            }

            // Sort first chunk by degree descending with some randomness
            Array.Sort(perm, 0, Math.Min(N, 5000), Comparer<int>.Create((a, b) =>
            {
                int da = adj[a].Count + rng.Next(0, Math.Max(1, adj[a].Count / 4));
                int db = adj[b].Count + rng.Next(0, Math.Max(1, adj[b].Count / 4));
                return db.CompareTo(da);
            }));

            var clique = new List<int>();
            var cliqueSet = new HashSet<int>();
            int limit = Math.Min(N, 10000);

            for (int i = 0; i < limit; i++)
            {
                int v = perm[i];
                bool canAdd = true;
                foreach (int c in clique)
                {
                    if (!adjSet[v].Contains(c)) { canAdd = false; break; }
                }
                if (canAdd) { clique.Add(v); cliqueSet.Add(v); }
            }

            // Try to expand by looking at neighbors
            var candidates = new HashSet<int>();
            foreach (int v in clique)
                foreach (int u in adj[v])
                    if (!cliqueSet.Contains(u)) candidates.Add(u);

            foreach (int v in candidates)
            {
                bool canAdd = true;
                foreach (int c in clique)
                {
                    if (!adjSet[v].Contains(c)) { canAdd = false; break; }
                }
                if (canAdd) { clique.Add(v); cliqueSet.Add(v); }
            }

            // Swap-based improvement
            for (int iter = 0; iter < 100 && sw.ElapsedMilliseconds < timeLimitMs; iter++)
            {
                bool improved = false;
                // Try removing one vertex and adding two
                for (int ci = clique.Count - 1; ci >= 0 && !improved; ci--)
                {
                    int rem = clique[ci];
                    // Find vertices connected to all clique members except rem
                    var potentials = new List<int>();
                    foreach (int u in adj[rem])
                    {
                        if (cliqueSet.Contains(u)) continue;
                        bool ok = true;
                        foreach (int c in clique)
                        {
                            if (c == rem) continue;
                            if (!adjSet[u].Contains(c)) { ok = false; break; }
                        }
                        if (ok) potentials.Add(u);
                    }

                    // Also check non-neighbor potentials
                    if (potentials.Count < 2)
                    {
                        // Check some random vertices
                        for (int trial = 0; trial < 50 && potentials.Count < 10; trial++)
                        {
                            int u = rng.Next(N);
                            if (cliqueSet.Contains(u) || potentials.Contains(u)) continue;
                            bool ok = true;
                            foreach (int c in clique)
                            {
                                if (c == rem) continue;
                                if (!adjSet[u].Contains(c)) { ok = false; break; }
                            }
                            if (ok) potentials.Add(u);
                        }
                    }

                    // Check pairs of potentials
                    for (int pi = 0; pi < potentials.Count && !improved; pi++)
                    {
                        for (int pj = pi + 1; pj < potentials.Count && !improved; pj++)
                        {
                            if (adjSet[potentials[pi]].Contains(potentials[pj]))
                            {
                                clique.RemoveAt(ci);
                                cliqueSet.Remove(rem);
                                clique.Add(potentials[pi]);
                                cliqueSet.Add(potentials[pi]);
                                clique.Add(potentials[pj]);
                                cliqueSet.Add(potentials[pj]);
                                improved = true;
                            }
                        }
                    }
                }
                if (!improved) break;
            }

            UpdateBest(clique);
        }
    }
}
