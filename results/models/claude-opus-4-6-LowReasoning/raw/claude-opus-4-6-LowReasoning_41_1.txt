; SPIR-V
; Version: 1.0
               OpCapability Shader
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Fragment %main "main" %inNormal %outColor
               OpExecutionMode %main OriginUpperLeft

               OpDecorate %inNormal Location 1
               OpDecorate %outColor Location 0

       %void = OpTypeVoid
   %funcType = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v3float = OpTypeVector %float 3
    %v4float = OpTypeVector %float 4
 %ptr_in_v3  = OpTypePointer Input %v3float
%ptr_out_v4  = OpTypePointer Output %v4float

  %float_0_5 = OpConstant %float 0.5
    %float_1 = OpConstant %float 1.0

   %inNormal = OpVariable %ptr_in_v3 Input
   %outColor = OpVariable %ptr_out_v4 Output

       %main = OpFunction %void None %funcType
      %entry = OpLabel
          %n = OpLoad %v3float %inNormal
       %norm = OpExtInst %v3float %1 Normalize %n
       %half = OpCompositeConstruct %v3float %float_0_5 %float_0_5 %float_0_5
       %scaled = OpVectorTimesScalar %v3float %norm %float_0_5
       %mapped = OpFAdd %v3float %scaled %half
         %rx = OpCompositeExtract %float %mapped 0
         %ry = OpCompositeExtract %float %mapped 1
         %rz = OpCompositeExtract %float %mapped 2
     %result = OpCompositeConstruct %v4float %rx %ry %rz %float_1
               OpStore %outColor %result
               OpReturn
               OpFunctionEnd
