; SPIR-V Geometry Shader: Expand points into colored quads
               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %inColor %outColor
               OpExecutionMode %main InputPoints
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 4
               OpExecutionMode %main Invocations 1

               ; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %inColor Location 2
               OpDecorate %outColor Location 0

               ; Types
      %void = OpTypeVoid
  %void_fn = OpTypeFunction %void
     %float = OpTypeFloat 32
   %v3float = OpTypeVector %float 3
   %v4float = OpTypeVector %float 4
      %uint = OpTypeInt 32 0
       %int = OpTypeInt 32 1

     %int_0 = OpConstant %int 0
    %uint_1 = OpConstant %uint 1
     %float_0 = OpConstant %float 0
     %float_1 = OpConstant %float 1
     %float_h = OpConstant %float 0.04
   %float_nh = OpConstant %float -0.04

; Input gl_PerVertex block (array of 1 for InputPoints)
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_1
%ptr_in_arr = OpTypePointer Input %arr_pv_in
%ptr_in_v4 = OpTypePointer Input %v4float
     %gl_in = OpVariable %ptr_in_arr Input

; Input color (array of 1)
%arr_v3_1 = OpTypeArray %v3float %uint_1
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_1
%ptr_in_v3 = OpTypePointer Input %v3float
   %inColor = OpVariable %ptr_in_arr_v3 Input

; Output gl_PerVertex block
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
    %gl_out = OpVariable %ptr_out_pv Output

; Output color
%ptr_out_v3 = OpTypePointer Output %v3float
  %outColor = OpVariable %ptr_out_v3 Output

      %main = OpFunction %void None %void_fn
     %entry = OpLabel

               ; Load input position
     %pos_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
       %pos = OpLoad %v4float %pos_ptr
         %x = OpCompositeExtract %float %pos 0
         %y = OpCompositeExtract %float %pos 1

               ; Load input color
   %col_ptr = OpAccessChain %ptr_in_v3 %inColor %int_0
      %color = OpLoad %v3float %col_ptr

               ; Compute corner positions
       %xmh = OpFAdd %float %x %float_nh
       %xph = OpFAdd %float %x %float_h
       %ymh = OpFAdd %float %y %float_nh
       %yph = OpFAdd %float %y %float_h

               ; Vertex 0: (x-h, y-h, 0, 1)
       %p0 = OpCompositeConstruct %v4float %xmh %ymh %float_0 %float_1
    %out0_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %out0_ptr %p0
               OpStore %outColor %color
               OpEmitVertex

               ; Vertex 1: (x+h, y-h, 0, 1)
       %p1 = OpCompositeConstruct %v4float %xph %ymh %float_0 %float_1
    %out1_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %out1_ptr %p1
               OpStore %outColor %color
               OpEmitVertex

               ; Vertex 2: (x-h, y+h, 0, 1)
       %p2 = OpCompositeConstruct %v4float %xmh %yph %float_0 %float_1
    %out2_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %out2_ptr %p2
               OpStore %outColor %color
               OpEmitVertex

               ; Vertex 3: (x+h, y+h, 0, 1)
       %p3 = OpCompositeConstruct %v4float %xph %yph %float_0 %float_1
    %out3_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %out3_ptr %p3
               OpStore %outColor %color
               OpEmitVertex

               OpEndPrimitive
               OpReturn
               OpFunctionEnd
