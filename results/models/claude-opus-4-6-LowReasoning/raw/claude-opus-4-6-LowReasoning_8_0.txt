use std::io::{self, Read, Write, BufWriter};
use std::collections::VecDeque;

fn main() {
    let mut input = Vec::new();
    io::stdin().read_to_end(&mut input).unwrap();
    let stdout = io::stdout();
    let mut out = BufWriter::new(stdout.lock());

    let mut pos = 0;

    // Parse H W
    while pos < input.len() && input[pos] == b' ' { pos += 1; }
    let mut h: usize = 0;
    while pos < input.len() && input[pos] >= b'0' && input[pos] <= b'9' {
        h = h * 10 + (input[pos] - b'0') as usize;
        pos += 1;
    }
    while pos < input.len() && (input[pos] == b' ' || input[pos] == b'\t') { pos += 1; }
    let mut w: usize = 0;
    while pos < input.len() && input[pos] >= b'0' && input[pos] <= b'9' {
        w = w * 10 + (input[pos] - b'0') as usize;
        pos += 1;
    }
    // Skip to end of line
    while pos < input.len() && input[pos] != b'\n' { pos += 1; }
    if pos < input.len() { pos += 1; } // skip \n

    let total = h * w;
    // Pack maze into bit vector: 1 = walkable
    let mut walkable = vec![0u64; (total + 63) / 64];
    let mut start = 0usize;
    let mut end = 0usize;

    for r in 0..h {
        let row_start = pos;
        // Find end of this line
        let mut line_end = pos;
        while line_end < input.len() && input[line_end] != b'\n' && input[line_end] != b'\r' {
            line_end += 1;
        }
        for c in 0..w {
            let byte_pos = row_start + c;
            let ch = if byte_pos < line_end { input[byte_pos] } else { b' ' };
            let idx = r * w + c;
            if ch != b'#' {
                walkable[idx / 64] |= 1u64 << (idx % 64);
            }
            if ch == b'A' { start = idx; }
            if ch == b'B' { end = idx; }
        }
        pos = line_end;
        if pos < input.len() && input[pos] == b'\r' { pos += 1; }
        if pos < input.len() && input[pos] == b'\n' { pos += 1; }
    }

    // BFS with packed visited bits
    let mut visited = vec![0u64; (total + 63) / 64];
    // Parent stored as u32 indices (enough for 50000*50000 = 2.5B... need u32? No, 2.5B > u32 max)
    // Use a flat vec of u32 "from direction" to reconstruct path (saves memory vs storing full parent)
    // Direction: 0=none(start), 1=from up, 2=from down, 3=from left, 4=from right
    let mut from_dir = vec![0u8; total];

    visited[start / 64] |= 1u64 << (start % 64);

    let mut queue = VecDeque::with_capacity(1024.min(total));
    queue.push_back(start as u32);

    let found;
    let dirs: [(i64, i64, u8); 4] = [
        (-(w as i64), 0, 1),   // up -> came from down
        (w as i64, 0, 2),      // down -> came from up
        (0, -1, 3),            // left -> came from right
        (0, 1, 4),             // right -> came from left
    ];

    'bfs: loop {
        let cur = match queue.pop_front() {
            Some(c) => c as usize,
            None => { found = false; break; }
        };
        if cur == end { found = true; break 'bfs; }

        let r = cur / w;
        let c = cur % w;

        for &(row_off, col_off, dir_code) in &dirs {
            let nr = r as i64 + if row_off != 0 { if row_off < 0 { -1 } else { 1 } } else { 0 };
            let nc = c as i64 + col_off;
            if nr < 0 || nr >= h as i64 || nc < 0 || nc >= w as i64 { continue; }
            let ni = nr as usize * w + nc as usize;
            if walkable[ni / 64] & (1u64 << (ni % 64)) == 0 { continue; }
            if visited[ni / 64] & (1u64 << (ni % 64)) != 0 { continue; }
            visited[ni / 64] |= 1u64 << (ni % 64);
            from_dir[ni] = dir_code;
            queue.push_back(ni as u32);
            if ni == end { found = true; break 'bfs; }
        }
    }

    if !found {
        let _ = out.write_all(b"0\n");
        return;
    }

    // Reconstruct path
    let mut path = Vec::new();
    let mut cur = end;
    loop {
        path.push(cur);
        if cur == start { break; }
        cur = match from_dir[cur] {
            1 => cur + w,   // came from down (we went up to get here)
            2 => cur - w,   // came from up
            3 => cur + 1,   // came from right
            4 => cur - 1,   // came from left
            _ => break,
        };
    }
    path.reverse();

    // Output
    let mut buf = Vec::with_capacity(path.len() * 12 + 20);
    write_usize(&mut buf, path.len());
    buf.push(b'\n');
    for &idx in &path {
        let r = idx / w;
        let c = idx % w;
        write_usize(&mut buf, r);
        buf.push(b' ');
        write_usize(&mut buf, c);
        buf.push(b'\n');
    }
    let _ = out.write_all(&buf);
}

fn write_usize(buf: &mut Vec<u8>, mut n: usize) {
    if n == 0 { buf.push(b'0'); return; }
    let start = buf.len();
    while n > 0 {
        buf.push(b'0' + (n % 10) as u8);
        n /= 10;
    }
    buf[start..].reverse();
}
