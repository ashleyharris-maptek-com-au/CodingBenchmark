use std::io::{self, Read, Write, BufWriter};
use std::collections::VecDeque;

fn main() {
    let mut input = Vec::new();
    io::stdin().read_to_end(&mut input).unwrap();
    let stdout = io::stdout();
    let mut out = BufWriter::new(stdout.lock());

    let mut pos = 0;

    // Parse H W
    while pos < input.len() && input[pos] == b' ' { pos += 1; }
    let mut h: usize = 0;
    while pos < input.len() && input[pos] >= b'0' && input[pos] <= b'9' {
        h = h * 10 + (input[pos] - b'0') as usize;
        pos += 1;
    }
    while pos < input.len() && (input[pos] == b' ' || input[pos] == b'\t') { pos += 1; }
    let mut w: usize = 0;
    while pos < input.len() && input[pos] >= b'0' && input[pos] <= b'9' {
        w = w * 10 + (input[pos] - b'0') as usize;
        pos += 1;
    }
    // Skip to end of line
    while pos < input.len() && input[pos] != b'\n' { pos += 1; }
    if pos < input.len() { pos += 1; } // skip \n

    // Read maze rows
    let mut grid = vec![0u8; h * w];
    let mut start = 0usize;
    let mut end = 0usize;

    for r in 0..h {
        let line_start = pos;
        // Find end of line
        let mut line_end = pos;
        while line_end < input.len() && input[line_end] != b'\n' && input[line_end] != b'\r' {
            line_end += 1;
        }
        for c in 0..w {
            let ch = if line_start + c < line_end { input[line_start + c] } else { b' ' };
            let idx = r * w + c;
            grid[idx] = ch;
            if ch == b'A' { start = idx; }
            if ch == b'B' { end = idx; }
        }
        pos = line_end;
        if pos < input.len() && input[pos] == b'\r' { pos += 1; }
        if pos < input.len() && input[pos] == b'\n' { pos += 1; }
    }

    // BFS
    // parent stores the index we came from; usize::MAX = unvisited
    let total = h * w;
    let mut parent = vec![u32::MAX; total];
    parent[start] = start as u32;

    let mut queue = VecDeque::new();
    queue.push_back(start as u32);

    let w32 = w as u32;
    let found = 'bfs: {
        while let Some(cur32) = queue.pop_front() {
            let cur = cur32 as usize;
            if cur == end { break 'bfs true; }
            let r = (cur / w) as u32;
            let c = (cur % w) as u32;
            // 4 neighbors
            let neighbors: [(u32, u32); 4] = [
                (r.wrapping_sub(1), c),
                (r + 1, c),
                (r, c.wrapping_sub(1)),
                (r, c + 1),
            ];
            for (nr, nc) in neighbors {
                if nr < h as u32 && nc < w32 {
                    let ni = nr as usize * w + nc as usize;
                    if parent[ni] == u32::MAX && grid[ni] != b'#' {
                        parent[ni] = cur32;
                        if ni == end { break 'bfs true; }
                        queue.push_back(ni as u32);
                    }
                }
            }
        }
        false
    };

    if !found {
        let _ = out.write_all(b"0\n");
        return;
    }

    // Trace path
    let mut path = Vec::new();
    let mut cur = end;
    loop {
        path.push(cur);
        if cur == start { break; }
        cur = parent[cur] as usize;
    }
    path.reverse();

    let _ = write!(out, "{}\n", path.len());
    for &idx in &path {
        let r = idx / w;
        let c = idx % w;
        let _ = write!(out, "{} {}\n", r, c);
    }
}
