import sys, math

def dist(a, b):
    return math.hypot(a[0]-b[0], a[1]-b[1])

def solve():
    data = sys.stdin.read().split()
    idx = 0
    def nxt(): nonlocal idx; v = data[idx]; idx += 1; return v
    N, K, CAP = int(nxt()), int(nxt()), int(nxt())
    depot = (float(nxt()), float(nxt()))
    customers = []
    for i in range(N):
        x, y, d = float(nxt()), float(nxt()), float(nxt())
        customers.append((x, y, d))

    coords = [depot] + [(c[0], c[1]) for c in customers]
    demands = [0] + [c[2] for c in customers]

    # Precompute distance matrix
    n = N + 1
    D = [[0.0]*n for _ in range(n)]
    for i in range(n):
        for j in range(i+1, n):
            d = dist(coords[i], coords[j])
            D[i][j] = D[j][i] = d

    # Clarke-Wright savings heuristic
    savings = []
    for i in range(1, n):
        for j in range(i+1, n):
            s = D[0][i] + D[0][j] - D[i][j]
            savings.append((s, i, j))
    savings.sort(reverse=True)

    route_of = [0] * n  # route id for each customer node
    routes = {}
    for i in range(1, n):
        routes[i] = [i]
        route_of[i] = i

    route_demand = {i: demands[i] for i in range(1, n)}

    for s, i, j in savings:
        ri, rj = route_of[i], route_of[j]
        if ri == rj:
            continue
        if route_demand[ri] + route_demand[rj] > CAP:
            continue
        r_i = routes[ri]
        r_j = routes[rj]
        # Can only merge if i and j are at ends of their routes
        if not (r_i[0] == i or r_i[-1] == i):
            continue
        if not (r_j[0] == j or r_j[-1] == j):
            continue
        # Orient so i is at end of r_i and j is at start of r_j
        if r_i[0] == i:
            r_i.reverse()
        if r_j[-1] == j:
            r_j.reverse()
        merged = r_i + r_j
        new_demand = route_demand[ri] + route_demand[rj]
        new_id = ri
        routes[new_id] = merged
        route_demand[new_id] = new_demand
        del routes[rj]
        del route_demand[rj]
        for node in merged:
            route_of[node] = new_id

    # If too many routes, force-merge smallest pairs
    route_list = list(routes.values())
    while len(route_list) > K:
        # Find two routes whose merge is feasible and cheapest
        best = None
        best_cost = float('inf')
        for a in range(len(route_list)):
            for b in range(a+1, len(route_list)):
                da = sum(demands[x] for x in route_list[a])
                db = sum(demands[x] for x in route_list[b])
                if da + db > CAP:
                    continue
                cost = D[route_list[a][-1]][route_list[b][0]]
                if cost < best_cost:
                    best_cost = cost
                    best = (a, b)
        if best is None:
            break
        a, b = best
        route_list[a] = route_list[a] + route_list[b]
        route_list.pop(b)

    # 2-opt improvement per route
    def route_dist(route):
        d = D[0][route[0]]
        for i in range(len(route)-1):
            d += D[route[i]][route[i+1]]
        d += D[route[-1]][0]
        return d

    def two_opt(route):
        improved = True
        while improved:
            improved = False
            for i in range(len(route)-1):
                for j in range(i+1, len(route)):
                    # Cost of current edges
                    pi = 0 if i == 0 else route[i-1]
                    if i == 0: pi = 0
                    else: pi = route[i-1]
                    nj = 0 if j == len(route)-1 else route[j+1]
                    old = D[pi if i > 0 else 0][route[i]] + D[route[j]][nj if j < len(route)-1 else 0]
                    new = D[pi if i > 0 else 0][route[j]] + D[route[i]][nj if j < len(route)-1 else 0]
                    if new < old - 1e-10:
                        route[i:j+1] = route[i:j+1][::-1]
                        improved = True
        return route

    for i in range(len(route_list)):
        route_list[i] = two_opt(route_list[i])

    # Or-opt: move single customers between routes
    improved = True
    while improved:
        improved = False
        for ri in range(len(route_list)):
            for ci in range(len(route_list[ri])):
                cust = route_list[ri][ci]
                d_cust = demands[cust]
                # Cost of removing cust from ri
                prev_r = 0 if ci == 0 else route_list[ri][ci-1]
                next_r = 0 if ci == len(route_list[ri])-1 else route_list[ri][ci+1]
                remove_save = D[prev_r][cust] + D[cust][next_r] - D[prev_r][next_r]
                for rj in range(len(route_list)):
                    if ri == rj:
                        continue
                    rj_demand = sum(demands[x] for x in route_list[rj])
                    if rj_demand + d_cust > CAP:
                        continue
                    # Best insertion in rj
                    best_ins_cost = float('inf')
                    best_pos = 0
                    for pos in range(len(route_list[rj])+1):
                        p = 0 if pos == 0 else route_list[rj][pos-1]
                        nx = 0 if pos == len(route_list[rj]) else route_list[rj][pos]
                        ins_cost = D[p][cust] + D[cust][nx] - D[p][nx]
                        if ins_cost < best_ins_cost:
                            best_ins_cost = ins_cost
                            best_pos = pos
                    if best_ins_cost < remove_save - 1e-10:
                        route_list[ri].pop(ci)
                        route_list[rj].insert(best_pos, cust)
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        # Clean empty routes
        route_list = [r for r in route_list if r]

    # Final 2-opt
    for i in range(len(route_list)):
        route_list[i] = two_opt(route_list[i])

    total = sum(route_dist(r) for r in route_list)
    print(f"{total:.6f}")
    print(len(route_list))
    for r in route_list:
        print(" ".join(str(x-1) for x in r))

solve()
