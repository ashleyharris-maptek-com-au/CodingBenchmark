#include <iostream>
#include <vector>
#include <deque>
#include <set>
#include <map>
#include <queue>
#include <algorithm>
#include <cstdint>
#include <functional>

using namespace std;

using Pos = vector<int>;

struct PosHash {
    size_t operator()(const Pos& p) const {
        size_t h = 0;
        for (int v : p) {
            h ^= hash<int>()(v) + 0x9e3779b9 + (h << 6) + (h >> 2);
        }
        return h;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int D, max_turns;
    cin >> D >> max_turns;

    vector<int> bounds(D);
    long long total_cells = 1;
    for (int i = 0; i < D; i++) {
        cin >> bounds[i];
        if (total_cells < 2000000LL) total_cells *= bounds[i];
    }

    int snake_len;
    cin >> snake_len;
    Pos head(D);
    for (int i = 0; i < D; i++) cin >> head[i];

    // Build initial snake body - just head repeated (length=snake_len, all at head pos?
    // Actually the problem says snake occupies connected grid points. Initial snake:
    // we only know head position and length. Assume body trails behind somehow.
    // Safest: treat initial body as just the head, track length for growth.
    // Actually re-reading: "snake_length start_pos" - the snake starts with given length at head.
    // For a fresh game the body would trail. We don't know direction, so assume all at head.
    // We'll just track the head positions as we move - the deque grows to snake_len.

    deque<Pos> snake;
    snake.push_back(head);
    int target_len = snake_len;

    int F;
    cin >> F;
    set<Pos> food_set;
    vector<Pos> food_list;
    for (int i = 0; i < F; i++) {
        Pos p(D);
        for (int j = 0; j < D; j++) cin >> p[j];
        food_set.insert(p);
        food_list.push_back(p);
    }

    int O;
    cin >> O;
    set<Pos> obstacles;
    for (int i = 0; i < O; i++) {
        Pos p(D);
        for (int j = 0; j < D; j++) cin >> p[j];
        obstacles.insert(p);
    }

    set<Pos> body_set(snake.begin(), snake.end());

    auto is_valid = [&](const Pos& p) -> bool {
        for (int i = 0; i < D; i++)
            if (p[i] < 0 || p[i] >= bounds[i]) return false;
        if (obstacles.count(p)) return false;
        return true;
    };

    auto is_safe = [&](const Pos& p) -> bool {
        if (!is_valid(p)) return false;
        if (body_set.count(p)) return false;
        return true;
    };

    // BFS from current head to nearest food - returns sequence of (axis, dir) moves
    auto bfs_to_food = [&]() -> vector<pair<int,int>> {
        if (food_set.empty()) return {};

        map<Pos, pair<Pos, pair<int,int>>> parent;
        queue<Pos> q;
        q.push(head);
        parent[head] = {head, {-1, 0}};

        Pos found_target;
        bool found = false;
        int visited = 0;
        int visit_limit = min((long long)500000, total_cells);

        while (!q.empty() && !found && visited < visit_limit) {
            Pos cur = q.front(); q.pop();
            visited++;

            for (int axis = 0; axis < D; axis++) {
                for (int dir : {-1, 1}) {
                    Pos next = cur;
                    next[axis] += dir;
                    if (!is_valid(next)) continue;
                    if (parent.count(next)) continue;
                    // Allow moving through own body for path planning
                    // (body will have moved by the time we get there, mostly)
                    // But avoid immediate collision
                    parent[next] = {cur, {axis, dir}};
                    if (food_set.count(next)) {
                        found_target = next;
                        found = true;
                        break;
                    }
                    q.push(next);
                }
                if (found) break;
            }
        }

        if (!found) return {};

        vector<pair<int,int>> path;
        Pos cur = found_target;
        while (cur != head) {
            auto& [prev, move] = parent[cur];
            path.push_back(move);
            cur = prev;
        }
        reverse(path.begin(), path.end());
        return path;
    };

    // Greedy: move toward nearest food avoiding immediate death
    auto greedy_move = [&]() -> pair<int,int> {
        // Find nearest food by Manhattan distance
        Pos* nearest = nullptr;
        int best_dist = INT_MAX;
        for (auto& f : food_list) {
            if (!food_set.count(f)) continue;
            int dist = 0;
            for (int i = 0; i < D; i++) dist += abs(f[i] - head[i]);
            if (dist < best_dist) { best_dist = dist; nearest = &f; }
        }

        // Try moves toward food first, then any safe move
        vector<pair<int,int>> moves;
        if (nearest) {
            // Prioritize axes with largest difference
            vector<pair<int,int>> ranked;
            for (int i = 0; i < D; i++) {
                int diff = (*nearest)[i] - head[i];
                if (diff != 0) ranked.push_back({abs(diff), i});
            }
            sort(ranked.rbegin(), ranked.rend());
            for (auto& [_, axis] : ranked) {
                int dir = ((*nearest)[axis] > head[axis]) ? 1 : -1;
                moves.push_back({axis, dir});
            }
        }
        // Add all other moves as fallback
        for (int axis = 0; axis < D; axis++) {
            for (int dir : {1, -1}) {
                moves.push_back({axis, dir});
            }
        }

        // Pick first safe move
        // The tail will be removed unless we eat food, so check if tail pos frees up
        Pos tail_pos = snake.back();
        bool tail_leaving = ((int)snake.size() >= target_len);

        for (auto& [axis, dir] : moves) {
            Pos next = head;
            next[axis] += dir;
            if (!is_valid(next)) continue;
            if (body_set.count(next)) {
                // OK if it's the tail and tail is leaving
                if (tail_leaving && next == tail_pos && snake.size() > 1) {
                    return {axis, dir};
                }
                continue;
            }
            return {axis, dir};
        }
        // No safe move, just do something
        return {0, 1};
    };

    vector<pair<int,int>> planned_path;
    int path_idx = 0;

    for (int turn = 0; turn < max_turns; turn++) {
        if (food_set.empty()) break;

        pair<int,int> move;
        bool used_plan = false;

        // Try to use planned path
        if (path_idx < (int)planned_path.size()) {
            auto [axis, dir] = planned_path[path_idx];
            Pos next = head;
            next[axis] += dir;
            bool tail_leaving = ((int)snake.size() >= target_len);
            Pos tail_pos = snake.back();
            bool safe = is_valid(next) && (!body_set.count(next) ||
                         (tail_leaving && next == tail_pos && snake.size() > 1));
            if (safe) {
                move = {axis, dir};
                path_idx++;
                used_plan = true;
            }
        }

        if (!used_plan) {
            // Replan
            if (total_cells < 2000000LL) {
                planned_path = bfs_to_food();
                path_idx = 0;
                if (!planned_path.empty()) {
                    move = planned_path[0];
                    path_idx = 1;
                } else {
                    move = greedy_move();
                }
            } else {
                move = greedy_move();
            }
        }

        cout << move.first << " " << move.second << "\n";
        cout.flush();

        // Update state
        Pos new_head = head;
        new_head[move.first] += move.second;
        head = new_head;
        snake.push_front(head);
        body_set.insert(head);

        if (food_set.count(head)) {
            food_set.erase(head);
            target_len++;
        }

        // Remove tail if at target length
        while ((int)snake.size() > target_len) {
            Pos tail = snake.back();
            snake.pop_back();
            // Only remove from body_set if not duplicated
            bool still_in = false;
            for (auto& p : snake) { if (p == tail) { still_in = true; break; } }
            if (!still_in) body_set.erase(tail);
        }
    }

    return 0;
}
