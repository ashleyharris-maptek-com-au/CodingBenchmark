const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { width, height, grid, ignitions, towns, cutBudget } = data;

const idx = (x, y) => y * width + x;
const DX = [1, -1, 0, 0];
const DY = [0, 0, 1, -1];

// Build blocked array
const blocked = new Uint8Array(width * height);
for (let y = 0; y < height; y++) {
  const row = grid[y];
  for (let x = 0; x < width; x++) {
    if (row[x] !== '.') blocked[idx(x, y)] = 1;
  }
}

// Mark town positions as unblockable
const townAt = new Map();
for (const t of towns) townAt.set(idx(t.x, t.y), t);

const cuts = new Uint8Array(width * height);

function bfsFromSources(sources, curCuts) {
  const dist = new Int32Array(width * height).fill(-1);
  const queue = [];
  let head = 0;
  for (const [x, y] of sources) {
    const i = idx(x, y);
    if (blocked[i] || curCuts[i]) continue;
    if (dist[i] >= 0) continue;
    dist[i] = 0;
    queue.push(i);
  }
  while (head < queue.length) {
    const ci = queue[head++];
    const cx = ci % width, cy = (ci - cx) / width;
    const nd = dist[ci] + 1;
    for (let d = 0; d < 4; d++) {
      const nx = cx + DX[d], ny = cy + DY[d];
      if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;
      const ni = idx(nx, ny);
      if (blocked[ni] || curCuts[ni] || dist[ni] >= 0) continue;
      dist[ni] = nd;
      queue.push(ni);
    }
  }
  return dist;
}

function bfsFrom(sx, sy, curCuts) {
  const dist = new Int32Array(width * height).fill(-1);
  const i0 = idx(sx, sy);
  if (blocked[i0] || curCuts[i0]) return dist;
  dist[i0] = 0;
  const queue = [i0];
  let head = 0;
  while (head < queue.length) {
    const ci = queue[head++];
    const cx = ci % width, cy = (ci - cx) / width;
    const nd = dist[ci] + 1;
    for (let d = 0; d < 4; d++) {
      const nx = cx + DX[d], ny = cy + DY[d];
      if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;
      const ni = idx(nx, ny);
      if (blocked[ni] || curCuts[ni] || dist[ni] >= 0) continue;
      dist[ni] = nd;
      queue.push(ni);
    }
  }
  return dist;
}

function getThreatenedTowns(curCuts) {
  const fireDist = bfsFromSources(ignitions, curCuts);
  const threatened = [];
  for (const t of towns) {
    if (fireDist[idx(t.x, t.y)] >= 0) threatened.push(t);
  }
  return { fireDist, threatened };
}

if (cutBudget === 0 || towns.length === 0) {
  console.log(JSON.stringify({ firebreaks: [] }));
  process.exit(0);
}

// Initial fire analysis
let { fireDist, threatened } = getThreatenedTowns(cuts);

if (threatened.length === 0) {
  console.log(JSON.stringify({ firebreaks: [] }));
  process.exit(0);
}

// Score cells by importance: for each threatened town, find cells on shortest
// paths from fire to town, weight by town value
const cellScore = new Float64Array(width * height);

// Sort threatened towns by value descending to prioritize
threatened.sort((a, b) => b.value - a.value);

for (const t of threatened) {
  const ti = idx(t.x, t.y);
  const fd = fireDist[ti];
  if (fd < 0) continue;

  // BFS from town
  const townDist = bfsFrom(t.x, t.y, cuts);

  // Cells on shortest paths: fireDist[c] + townDist[c] == fd
  // Weight inversely by number of such cells (narrow bottlenecks score higher)
  let pathCellCount = 0;
  for (let i = 0; i < width * height; i++) {
    if (fireDist[i] >= 0 && townDist[i] >= 0 && fireDist[i] + townDist[i] === fd) {
      // Don't count town cell itself or ignition cells
      if (!townAt.has(i)) pathCellCount++;
    }
  }
  if (pathCellCount === 0) continue;

  const weight = t.value / Math.max(1, Math.sqrt(pathCellCount));
  for (let i = 0; i < width * height; i++) {
    if (fireDist[i] >= 0 && townDist[i] >= 0 && fireDist[i] + townDist[i] === fd) {
      if (!townAt.has(i)) {
        // Prefer cells closer to the fire (smaller fireDist) for tighter cuts
        // and cells at bottlenecks
        cellScore[i] += weight * (1 + 1 / (1 + fireDist[i]));
      }
    }
  }
}

// Collect and sort candidate cells
const candidates = [];
for (let i = 0; i < width * height; i++) {
  if (cellScore[i] > 0 && !blocked[i]) {
    candidates.push(i);
  }
}
candidates.sort((a, b) => cellScore[b] - cellScore[a]);

// Greedy: cut top scored cells
let budgetLeft = cutBudget;
for (const ci of candidates) {
  if (budgetLeft <= 0) break;
  if (cuts[ci]) continue;
  cuts[ci] = 1;
  budgetLeft--;
}

// If budget remains, do iterative refinement: recompute fire and find more cuts
let iterations = 0;
while (budgetLeft > 0 && iterations < 5) {
  iterations++;
  const result = getThreatenedTowns(cuts);
  if (result.threatened.length === 0) break;

  const fDist = result.fireDist;
  const newScore = new Float64Array(width * height);

  for (const t of result.threatened) {
    const ti = idx(t.x, t.y);
    const fd = fDist[ti];
    if (fd < 0) continue;

    const tDist = bfsFrom(t.x, t.y, cuts);
    let pathCount = 0;
    for (let i = 0; i < width * height; i++) {
      if (fDist[i] >= 0 && tDist[i] >= 0 && fDist[i] + tDist[i] === fd && !townAt.has(i)) {
        pathCount++;
      }
    }
    if (pathCount === 0) continue;

    const w = t.value / Math.max(1, Math.sqrt(pathCount));
    for (let i = 0; i < width * height; i++) {
      if (fDist[i] >= 0 && tDist[i] >= 0 && fDist[i] + tDist[i] === fd && !townAt.has(i) && !cuts[i]) {
        newScore[i] += w * (1 + 1 / (1 + fDist[i]));
      }
    }
  }

  const newCands = [];
  for (let i = 0; i < width * height; i++) {
    if (newScore[i] > 0 && !blocked[i] && !cuts[i]) newCands.push(i);
  }
  if (newCands.length === 0) break;
  newCands.sort((a, b) => newScore[b] - newScore[a]);

  let placed = 0;
  for (const ci of newCands) {
    if (budgetLeft <= 0) break;
    cuts[ci] = 1;
    budgetLeft--;
    placed++;
  }
  if (placed === 0) break;
}

// Collect result
const firebreaks = [];
for (let i = 0; i < width * height; i++) {
  if (cuts[i]) {
    firebreaks.push([i % width, (i - i % width) / width]);
  }
}

console.log(JSON.stringify({ firebreaks }));
