#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <random>
#include <chrono>
#include <cstring>
#include <queue>
#include <functional>

using namespace std;
using Clock = chrono::steady_clock;

struct Task {
    int job, idx, machine, duration;
};

struct Solution {
    vector<vector<int>> machineOrder; // machineOrder[m] = sequence of task global ids on machine m
    vector<int> start;               // start[globalId]
    int makespan;
};

int numJobs, numMachines;
vector<vector<int>> jobTasks; // jobTasks[j] = list of global task ids
vector<Task> tasks;
int totalTasks;

void evaluate(Solution& sol) {
    auto& st = sol.start;
    st.assign(totalTasks, 0);

    // For each machine, build predecessor map
    vector<int> machPrev(totalTasks, -1);
    for (int m = 0; m < numMachines; m++) {
        auto& seq = sol.machineOrder[m];
        for (int i = 1; i < (int)seq.size(); i++) {
            machPrev[seq[i]] = seq[i-1];
        }
    }

    // Topological order: process tasks respecting both job order and machine order
    vector<int> inDeg(totalTasks, 0);
    for (int g = 0; g < totalTasks; g++) {
        auto& t = tasks[g];
        if (t.idx > 0) inDeg[g]++;
        if (machPrev[g] >= 0) inDeg[g]++;
    }

    queue<int> q;
    for (int g = 0; g < totalTasks; g++)
        if (inDeg[g] == 0) q.push(g);

    // successors
    vector<vector<int>> succ(totalTasks);
    for (int g = 0; g < totalTasks; g++) {
        auto& t = tasks[g];
        if (t.idx + 1 < (int)jobTasks[t.job].size())
            succ[g].push_back(jobTasks[t.job][t.idx + 1]);
    }
    for (int m = 0; m < numMachines; m++) {
        auto& seq = sol.machineOrder[m];
        for (int i = 0; i + 1 < (int)seq.size(); i++)
            succ[seq[i]].push_back(seq[i+1]);
    }

    sol.makespan = 0;
    while (!q.empty()) {
        int g = q.front(); q.pop();
        auto& t = tasks[g];
        int earliest = 0;
        if (t.idx > 0) {
            int prev = jobTasks[t.job][t.idx - 1];
            earliest = max(earliest, st[prev] + tasks[prev].duration);
        }
        if (machPrev[g] >= 0) {
            int prev = machPrev[g];
            earliest = max(earliest, st[prev] + tasks[prev].duration);
        }
        st[g] = earliest;
        sol.makespan = max(sol.makespan, earliest + t.duration);
        for (int s : succ[g]) {
            if (--inDeg[s] == 0) q.push(s);
        }
    }
}

// Build solution using priority dispatch
Solution buildGreedy(int rule, mt19937& rng) {
    Solution sol;
    sol.machineOrder.resize(numMachines);
    sol.start.assign(totalTasks, 0);

    vector<int> jobNext(numJobs, 0); // next task index for each job
    vector<int> jobAvail(numJobs, 0);
    vector<int> machAvail(numMachines, 0);

    // Precompute job remaining work
    vector<int> jobRemaining(numJobs, 0);
    for (int j = 0; j < numJobs; j++)
        for (int g : jobTasks[j])
            jobRemaining[j] += tasks[g].duration;

    int scheduled = 0;
    while (scheduled < totalTasks) {
        // Collect schedulable tasks
        vector<int> ready;
        for (int j = 0; j < numJobs; j++) {
            if (jobNext[j] < (int)jobTasks[j].size())
                ready.push_back(j);
        }

        // Pick based on rule
        int bestJob = -1;
        double bestPri = 1e18;
        for (int j : ready) {
            int g = jobTasks[j][jobNext[j]];
            auto& t = tasks[g];
            int startTime = max(jobAvail[j], machAvail[t.machine]);
            double pri;
            switch (rule) {
                case 0: pri = startTime + t.duration; break; // earliest completion
                case 1: pri = t.duration; break;             // SPT
                case 2: pri = -t.duration; break;            // LPT
                case 3: pri = -(double)jobRemaining[j]; break; // most work remaining
                case 4: pri = startTime; break;              // earliest start
                case 5: { // random
                    pri = (double)(rng() % 10000);
                    break;
                }
                default: pri = startTime + t.duration; break;
            }
            if (bestJob < 0 || pri < bestPri) {
                bestPri = pri;
                bestJob = j;
            }
        }

        int g = jobTasks[bestJob][jobNext[bestJob]];
        auto& t = tasks[g];
        int startTime = max(jobAvail[bestJob], machAvail[t.machine]);
        sol.start[g] = startTime;
        sol.machineOrder[t.machine].push_back(g);
        machAvail[t.machine] = startTime + t.duration;
        jobAvail[bestJob] = startTime + t.duration;
        jobRemaining[bestJob] -= t.duration;
        jobNext[bestJob]++;
        scheduled++;
    }

    sol.makespan = *max_element(machAvail.begin(), machAvail.end());
    return sol;
}

// Find critical path and return tasks on it
vector<int> findCriticalPath(Solution& sol) {
    // Find a task that ends at makespan
    int last = -1;
    for (int g = 0; g < totalTasks; g++) {
        if (sol.start[g] + tasks[g].duration == sol.makespan) {
            last = g;
            break;
        }
    }

    // Build machine predecessor map
    vector<int> machPrev(totalTasks, -1);
    for (int m = 0; m < numMachines; m++) {
        auto& seq = sol.machineOrder[m];
        for (int i = 1; i < (int)seq.size(); i++)
            machPrev[seq[i]] = seq[i-1];
    }

    vector<int> path;
    int cur = last;
    while (cur >= 0) {
        path.push_back(cur);
        auto& t = tasks[cur];
        int jobPred = (t.idx > 0) ? jobTasks[t.job][t.idx - 1] : -1;
        int mPred = machPrev[cur];

        int best = -1;
        int bestEnd = -1;
        if (jobPred >= 0) {
            int e = sol.start[jobPred] + tasks[jobPred].duration;
            if (e > bestEnd) { bestEnd = e; best = jobPred; }
        }
        if (mPred >= 0) {
            int e = sol.start[mPred] + tasks[mPred].duration;
            if (e > bestEnd) { bestEnd = e; best = mPred; }
        }
        cur = best;
    }
    reverse(path.begin(), path.end());
    return path;
}

// Try swapping adjacent tasks on a machine within critical blocks
bool trySwapImprove(Solution& sol, mt19937& rng) {
    auto path = findCriticalPath(sol);
    if (path.size() < 2) return false;

    // Find critical blocks (consecutive tasks on same machine in the path)
    // Try swapping first/last elements of blocks with adjacent block boundaries
    vector<pair<int,int>> swapCandidates;

    for (int i = 0; i + 1 < (int)path.size(); i++) {
        int g1 = path[i], g2 = path[i+1];
        if (tasks[g1].machine == tasks[g2].machine) {
            swapCandidates.push_back({g1, g2});
        }
    }

    if (swapCandidates.empty()) return false;

    // Try each swap candidate
    shuffle(swapCandidates.begin(), swapCandidates.end(), rng);

    for (auto [g1, g2] : swapCandidates) {
        int m = tasks[g1].machine;
        auto& seq = sol.machineOrder[m];

        // Find positions
        int pos1 = -1, pos2 = -1;
        for (int i = 0; i < (int)seq.size(); i++) {
            if (seq[i] == g1) pos1 = i;
            if (seq[i] == g2) pos2 = i;
        }
        if (pos1 < 0 || pos2 < 0 || abs(pos1 - pos2) != 1) continue;

        // Swap and evaluate
        swap(seq[pos1], seq[pos2]);
        int oldMakespan = sol.makespan;
        evaluate(sol);
        if (sol.makespan < oldMakespan) {
            return true;
        }
        // Undo
        swap(seq[pos1], seq[pos2]);
        evaluate(sol);
    }
    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    auto startTime = Clock::now();

    cin >> numJobs >> numMachines;
    jobTasks.resize(numJobs);

    int gid = 0;
    for (int j = 0; j < numJobs; j++) {
        int T; cin >> T;
        for (int t = 0; t < T; t++) {
            int m, d; cin >> m >> d;
            tasks.push_back({j, t, m, d});
            jobTasks[j].push_back(gid++);
        }
    }
    totalTasks = gid;

    mt19937 rng(42);

    // Determine scale
    bool isSmall = (totalTasks <= 500);
    bool isMedium = (totalTasks <= 5000);

    int numRules = isSmall ? 6 : (isMedium ? 4 : 2);
    int numRandom = isSmall ? 20 : (isMedium ? 5 : 0);

    Solution best;
    best.makespan = INT_MAX;

    // Try different dispatch rules
    for (int r = 0; r < numRules; r++) {
        auto sol = buildGreedy(r, rng);
        if (sol.makespan < best.makespan) best = sol;
    }

    // Try random constructions
    for (int i = 0; i < numRandom; i++) {
        auto sol = buildGreedy(5, rng);
        if (sol.makespan < best.makespan) best = sol;
    }

    // Local search improvement
    auto elapsed = [&]() {
        return chrono::duration_cast<chrono::milliseconds>(Clock::now() - startTime).count();
    };

    long long timeLimit = isSmall ? 5000 : (isMedium ? 8000 : 10000);
    // Cap at reasonable time
    if (totalTasks > 20000) timeLimit = 15000;

    int improvements = 0;
    while (elapsed() < timeLimit) {
        bool improved = trySwapImprove(best, rng);
        if (improved) {
            improvements++;
        } else {
            // Try a perturbation: random swap on a random machine
            if (numMachines == 0) break;
            int m = rng() % numMachines;
            auto& seq = best.machineOrder[m];
            if (seq.size() < 2) continue;
            int i = rng() % ((int)seq.size() - 1);
            swap(seq[i], seq[i+1]);
            int oldMs = best.makespan;
            evaluate(best);
            if (best.makespan <= oldMs) {
                // Accept equal or better
            } else {
                // Revert
                swap(seq[i], seq[i+1]);
                evaluate(best);
            }
        }

        if (elapsed() > timeLimit) break;
    }

    // Output
    for (int j = 0; j < numJobs; j++) {
        for (int i = 0; i < (int)jobTasks[j].size(); i++) {
            int g = jobTasks[j][i];
            if (i > 0) cout << ' ';
            cout << best.start[g] << ' ' << tasks[g].machine;
        }
        cout << '\n';
    }

    return 0;
}
