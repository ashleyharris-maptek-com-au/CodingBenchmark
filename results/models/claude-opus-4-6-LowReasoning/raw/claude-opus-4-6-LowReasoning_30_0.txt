import sys
import random

def solve():
    input_data = sys.stdin.buffer.read().decode()
    tokens = input_data.split()
    idx = 0
    n = int(tokens[idx]); idx += 1
    m = int(tokens[idx]); idx += 1

    # Build clause list and occurrence lists
    clauses = []
    # For each variable, track which clauses it appears in positively/negatively
    pos_occ = [[] for _ in range(n + 1)]  # clauses where var appears positive
    neg_occ = [[] for _ in range(n + 1)]  # clauses where var appears negative

    for i in range(m):
        a = int(tokens[idx]); idx += 1
        b = int(tokens[idx]); idx += 1
        c = int(tokens[idx]); idx += 1
        clauses.append((a, b, c))
        for lit in (a, b, c):
            v = abs(lit)
            if lit > 0:
                pos_occ[v].append(i)
            else:
                neg_occ[v].append(i)

    # WalkSAT / GSAT hybrid
    assign = bytearray(n + 1)  # 0 or 1 for each variable 1..n

    # Initial assignment: set each var to satisfy more clauses
    for v in range(1, n + 1):
        if len(pos_occ[v]) >= len(neg_occ[v]):
            assign[v] = 1
        else:
            assign[v] = 0

    # Track which clauses are unsatisfied
    sat_count = [0] * m  # number of true literals per clause
    for i in range(m):
        a, b, c = clauses[i]
        cnt = 0
        for lit in (a, b, c):
            v = abs(lit)
            if (lit > 0 and assign[v] == 1) or (lit < 0 and assign[v] == 0):
                cnt += 1
        sat_count[i] = cnt

    unsat = set()
    for i in range(m):
        if sat_count[i] == 0:
            unsat.add(i)

    if not unsat:
        sys.stdout.write("SAT\n")
        sys.stdout.write(" ".join(str(assign[v]) for v in range(1, n + 1)) + "\n")
        return

    max_flips = min(m * 20, 200_000_000)
    noise = 0.57  # walksat noise parameter

    randbelow = random.randrange
    rand = random.random
    unsat_list = list(unsat)

    for flip in range(max_flips):
        if not unsat:
            break

        # Pick a random unsatisfied clause
        # Maintain unsat_list loosely - pick random, check membership
        while True:
            ci = unsat_list[randbelow(len(unsat_list))]
            if ci in unsat:
                break
            # Swap remove stale entry
            unsat_list[unsat_list.index(ci)] = unsat_list[-1]
            unsat_list.pop()
            if not unsat_list:
                unsat_list = list(unsat)

        a, b, c = clauses[ci]
        lits = (a, b, c)

        # With probability noise, flip a random variable in the clause
        if rand() < noise:
            lit = lits[randbelow(3)]
            var = abs(lit)
        else:
            # Pick the variable whose flip causes minimum break count
            best_var = 0
            best_break = m + 1
            for lit in lits:
                var = abs(lit)
                # If we flip var, how many currently satisfied clauses become unsatisfied?
                # A clause breaks if it has sat_count == 1 and var is the sole satisfier
                cur = assign[var]
                # Flipping: clauses where var appears matching current assignment lose a sat
                # Those are pos_occ if cur==1, neg_occ if cur==0
                if cur == 1:
                    break_count = sum(1 for ci2 in pos_occ[var] if sat_count[ci2] == 1)
                else:
                    break_count = sum(1 for ci2 in neg_occ[var] if sat_count[ci2] == 1)
                if break_count < best_break:
                    best_break = break_count
                    best_var = var
                    if best_break == 0:
                        break
            var = best_var

        # Flip the variable
        cur = assign[var]
        assign[var] = 1 - cur

        # Update sat_count: clauses where var appears positively
        if cur == 1:
            # Was true, now false: positive occurrences lose a sat, negative gain
            for ci2 in pos_occ[var]:
                sat_count[ci2] -= 1
                if sat_count[ci2] == 0:
                    unsat.add(ci2)
                    unsat_list.append(ci2)
            for ci2 in neg_occ[var]:
                sat_count[ci2] += 1
                if sat_count[ci2] == 1:
                    unsat.discard(ci2)
        else:
            # Was false, now true: positive occurrences gain, negative lose
            for ci2 in pos_occ[var]:
                sat_count[ci2] += 1
                if sat_count[ci2] == 1:
                    unsat.discard(ci2)
            for ci2 in neg_occ[var]:
                sat_count[ci2] -= 1
                if sat_count[ci2] == 0:
                    unsat.add(ci2)
                    unsat_list.append(ci2)

    sys.stdout.write("SAT\n")
    sys.stdout.write(" ".join(str(assign[v]) for v in range(1, n + 1)) + "\n")

solve()
