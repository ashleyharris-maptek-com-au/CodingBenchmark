use std::io::{self, BufRead, Write, BufWriter};
use std::f64::consts::PI;

#[derive(Clone, Debug)]
struct Poly {
    verts: Vec<(f64, f64)>,
}

impl Poly {
    fn bbox(&self) -> (f64, f64, f64, f64) {
        let mut minx = f64::MAX; let mut miny = f64::MAX;
        let mut maxx = f64::MIN; let mut maxy = f64::MIN;
        for &(x,y) in &self.verts {
            if x < minx { minx = x; }
            if y < miny { miny = y; }
            if x > maxx { maxx = x; }
            if y > maxy { maxy = y; }
        }
        (minx, miny, maxx, maxy)
    }

    fn area(&self) -> f64 {
        let n = self.verts.len();
        let mut a = 0.0;
        for i in 0..n {
            let j = (i+1) % n;
            a += self.verts[i].0 * self.verts[j].1;
            a -= self.verts[j].0 * self.verts[i].1;
        }
        a.abs() * 0.5
    }

    fn centroid(&self) -> (f64, f64) {
        let n = self.verts.len();
        let mut cx = 0.0; let mut cy = 0.0;
        for &(x,y) in &self.verts { cx += x; cy += y; }
        (cx / n as f64, cy / n as f64)
    }

    fn translate(&self, dx: f64, dy: f64) -> Poly {
        Poly { verts: self.verts.iter().map(|&(x,y)| (x+dx, y+dy)).collect() }
    }

    fn rotate(&self, angle: f64) -> Poly {
        let (cx, cy) = self.centroid();
        let cos = angle.cos();
        let sin = angle.sin();
        Poly {
            verts: self.verts.iter().map(|&(x,y)| {
                let dx = x - cx;
                let dy = y - cy;
                (cx + dx*cos - dy*sin, cy + dx*sin + dy*cos)
            }).collect()
        }
    }

    fn center_at_origin(&self) -> (Poly, f64, f64) {
        let (cx, cy) = self.centroid();
        (self.translate(-cx, -cy), cx, cy)
    }
}

fn point_in_convex(poly: &[(f64,f64)], px: f64, py: f64) -> bool {
    let n = poly.len();
    for i in 0..n {
        let j = (i+1) % n;
        let ex = poly[j].0 - poly[i].0;
        let ey = poly[j].1 - poly[i].1;
        let cross = ex * (py - poly[i].1) - ey * (px - poly[i].0);
        if cross < -1e-7 { return false; }
    }
    true
}

fn ensure_ccw(verts: &mut Vec<(f64,f64)>) {
    let n = verts.len();
    let mut a = 0.0;
    for i in 0..n {
        let j = (i+1) % n;
        a += verts[i].0 * verts[j].1;
        a -= verts[j].0 * verts[i].1;
    }
    if a < 0.0 { verts.reverse(); }
}

fn is_convex(verts: &[(f64,f64)]) -> bool {
    let n = verts.len();
    for i in 0..n {
        let j = (i+1) % n;
        let k = (i+2) % n;
        let cross = (verts[j].0 - verts[i].0) * (verts[k].1 - verts[j].1)
                  - (verts[j].1 - verts[i].1) * (verts[k].0 - verts[j].0);
        if cross < -1e-9 { return false; }
    }
    true
}

fn point_in_polygon(poly: &[(f64,f64)], px: f64, py: f64) -> bool {
    let n = poly.len();
    let mut inside = false;
    let mut j = n - 1;
    for i in 0..n {
        let (xi, yi) = poly[i];
        let (xj, yj) = poly[j];
        if ((yi > py) != (yj > py)) && (px < (xj - xi) * (py - yi) / (yj - yi) + xi) {
            inside = !inside;
        }
        j = i;
    }
    inside
}

fn point_in_polygon_or_edge(poly: &[(f64,f64)], px: f64, py: f64) -> bool {
    let n = poly.len();
    for i in 0..n {
        let j = (i+1) % n;
        let (x1,y1) = poly[i];
        let (x2,y2) = poly[j];
        let dx = x2-x1; let dy = y2-y1;
        let t = if dx*dx+dy*dy > 1e-15 { ((px-x1)*dx+(py-y1)*dy)/(dx*dx+dy*dy) } else { 0.0 };
        if t >= -1e-7 && t <= 1.0+1e-7 {
            let cx = x1 + t.clamp(0.0,1.0)*dx;
            let cy = y1 + t.clamp(0.0,1.0)*dy;
            let d2 = (px-cx)*(px-cx)+(py-cy)*(py-cy);
            if d2 < 1e-10 { return true; }
        }
    }
    point_in_polygon(poly, px, py)
}

fn poly_inside_stock(piece: &[(f64,f64)], stock: &[(f64,f64)], stock_convex: bool) -> bool {
    if stock_convex {
        piece.iter().all(|&(x,y)| point_in_convex(stock, x, y))
    } else {
        piece.iter().all(|&(x,y)| point_in_polygon_or_edge(stock, x, y))
    }
}

fn segments_intersect_proper(a1: (f64,f64), a2: (f64,f64), b1: (f64,f64), b2: (f64,f64)) -> bool {
    let d1 = cross2d(b1, b2, a1);
    let d2 = cross2d(b1, b2, a2);
    let d3 = cross2d(a1, a2, b1);
    let d4 = cross2d(a1, a2, b2);
    if ((d1 > 1e-9 && d2 < -1e-9) || (d1 < -1e-9 && d2 > 1e-9))
        && ((d3 > 1e-9 && d4 < -1e-9) || (d3 < -1e-9 && d4 > 1e-9)) {
        return true;
    }
    false
}

fn cross2d(o: (f64,f64), a: (f64,f64), b: (f64,f64)) -> f64 {
    (a.0-o.0)*(b.1-o.1) - (a.1-o.1)*(b.0-o.0)
}

fn polys_overlap(a: &[(f64,f64)], b: &[(f64,f64)]) -> bool {
    let na = a.len(); let nb = b.len();
    for i in 0..na {
        let i2 = (i+1)%na;
        for j in 0..nb {
            let j2 = (j+1)%nb;
            if segments_intersect_proper(a[i], a[i2], b[j], b[j2]) {
                return true;
            }
        }
    }
    let ac = centroid_of(a);
    if point_in_polygon(b, ac.0, ac.1) { return true; }
    let bc = centroid_of(b);
    if point_in_polygon(a, bc.0, bc.1) { return true; }
    false
}

fn centroid_of(v: &[(f64,f64)]) -> (f64,f64) {
    let n = v.len() as f64;
    let mut cx = 0.0; let mut cy = 0.0;
    for &(x,y) in v { cx+=x; cy+=y; }
    (cx/n, cy/n)
}

fn bbox_overlap(a: (f64,f64,f64,f64), b: (f64,f64,f64,f64)) -> bool {
    a.0 < b.2 + 1e-9 && a.2 > b.0 - 1e-9 && a.1 < b.3 + 1e-9 && a.3 > b.1 - 1e-9
}

fn poly_bbox(v: &[(f64,f64)]) -> (f64,f64,f64,f64) {
    let mut r = (f64::MAX, f64::MAX, f64::MIN, f64::MIN);
    for &(x,y) in v {
        if x < r.0 { r.0 = x; }
        if y < r.1 { r.1 = y; }
        if x > r.2 { r.2 = x; }
        if y > r.3 { r.3 = y; }
    }
    r
}

struct Stock {
    placed: Vec<(Vec<(f64,f64)>, (f64,f64,f64,f64))>,
    stock_bb: (f64,f64,f64,f64),
}

impl Stock {
    fn new(sbb: (f64,f64,f64,f64)) -> Self {
        Stock { placed: Vec::new(), stock_bb: sbb }
    }

    fn can_place(&self, piece_verts: &[(f64,f64)], stock_verts: &[(f64,f64)], stock_convex: bool) -> bool {
        if !poly_inside_stock(piece_verts, stock_verts, stock_convex) {
            return false;
        }
        let pbb = poly_bbox(piece_verts);
        for &(ref pv, ref bb) in &self.placed {
            if bbox_overlap(pbb, *bb) {
                if polys_overlap(piece_verts, pv) {
                    return false;
                }
            }
        }
        true
    }

    fn place(&mut self, piece_verts: Vec<(f64,f64)>) {
        let bb = poly_bbox(&piece_verts);
        self.placed.push((piece_verts, bb));
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = Vec::new();
    for line in stdin.lock().lines() {
        let l = line.unwrap();
        let trimmed = l.trim().to_string();
        if !trimmed.is_empty() {
            lines.push(trimmed);
        }
    }

    let mut idx = 0;
    let first: Vec<usize> = lines[idx].split_whitespace().map(|s| s.parse().unwrap()).collect();
    let sv = first[0];
    let np = first[1];
    idx += 1;

    let mut stock_verts: Vec<(f64,f64)> = Vec::with_capacity(sv);
    for i in 0..sv {
        let parts: Vec<f64> = lines[idx+i].split_whitespace().map(|s| s.parse().unwrap()).collect();
        stock_verts.push((parts[0], parts[1]));
    }
    idx += sv;
    ensure_ccw(&mut stock_verts);
    let stock_convex = is_convex(&stock_verts);
    let stock_bb = poly_bbox(&stock_verts);
    let stock_w = stock_bb.2 - stock_bb.0;
    let stock_h = stock_bb.3 - stock_bb.1;

    let mut pieces: Vec<Poly> = Vec::with_capacity(np);
    for _ in 0..np {
        let pv: usize = lines[idx].parse().unwrap();
        idx += 1;
        let mut verts = Vec::with_capacity(pv);
        for i in 0..pv {
            let parts: Vec<f64> = lines[idx+i].split_whitespace().map(|s| s.parse().unwrap()).collect();
            verts.push((parts[0], parts[1]));
        }
        idx += pv;
        ensure_ccw(&mut verts);
        pieces.push(Poly { verts });
    }

    let mut piece_indices: Vec<usize> = (0..np).collect();
    piece_indices.sort_by(|&a, &b| {
        let area_a = pieces[a].area();
        let area_b = pieces[b].area();
        area_b.partial_cmp(&area_a).unwrap()
    });

    let num_rotations = if np > 2000 { 4 } else if np > 500 { 8 } else { 16 };
    let grid_step = if np > 2000 {
        (stock_w.max(stock_h)) / 15.0
    } else if np > 500 {
        (stock_w.max(stock_h)) / 25.0
    } else {
        (stock_w.max(stock_h)) / 40.0
    };

    let mut stocks: Vec<Stock> = Vec::new();
    let mut results: Vec<(usize, f64, f64, f64)> = vec![(0, 0.0, 0.0, 0.0); np];

    for &pi in &piece_indices {
        let piece = &pieces[pi];
        let (centered, ocx, ocy) = piece.center_at_origin();

        let mut placed = false;

        'try_stocks: for si in 0..stocks.len() {
            for ri in 0..num_rotations {
                let angle = (ri as f64) * 2.0 * PI / (num_rotations as f64);
                let rotated = centered.rotate(angle);
                let rbb = Poly { verts: rotated.verts.clone() }.bbox();
                let rw = rbb.2 - rbb.0;
                let rh = rbb.3 - rbb.1;

                if rw > stock_w + 1e-6 || rh > stock_h + 1e-6 {
                    continue;
                }

                let sx = stock_bb.0 - rbb.0;
                let sy = stock_bb.1 - rbb.1;
                let ex = stock_bb.2 - rbb.2;
                let ey = stock_bb.3 - rbb.3;

                let mut best_y = f64::MAX;
                let mut best_x = f64::MAX;
                let mut best_tx = 0.0;
                let mut best_ty = 0.0;
                let mut found = false;

                let mut ty = sy;
                while ty <= ey + 1e-9 {
                    let mut tx = sx;
                    while tx <= ex + 1e-9 {
                        let candidate: Vec<(f64,f64)> = rotated.verts.iter().map(|&(x,y)| (x+tx, y+ty)).collect();
                        if stocks[si].can_place(&candidate, &stock_verts, stock_convex) {
                            if ty < best_y - 1e-9 || (ty < best_y + 1e-9 && tx < best_x - 1e-9) {
                                best_y = ty;
                                best_x = tx;
                                best_tx = tx;
                                best_ty = ty;
                                found = true;
                            }
                        }
                        tx += grid_step;
                    }
                    ty += grid_step;
                }

                if found {
                    let final_verts: Vec<(f64,f64)> = rotated.verts.iter().map(|&(x,y)| (x+best_tx, y+best_ty)).collect();
                    stocks[si].place(final_verts);
                    let final_dx = best_tx - ocx;
                    let final_dy = best_ty - ocy;
                    results[pi] = (si, final_dx, final_dy, angle);
                    placed = true;
                    break 'try_stocks;
                }
            }
        }

        if !placed {
            let new_si = stocks.len();
            stocks.push(Stock::new(stock_bb));

            let mut done = false;
            for ri in 0..num_rotations {
                let angle = (ri as f64) * 2.0 * PI / (num_rotations as f64);
                let rotated = centered.rotate(angle);
                let rbb = Poly { verts: rotated.verts.clone() }.bbox();
                let rw = rbb.2 - rbb.0;
                let rh = rbb.3 - rbb.1;

                if rw > stock_w + 1e-6 || rh > stock_h + 1e-6 {
                    continue;
                }

                let sx = stock_bb.0 - rbb.0;
                let sy = stock_bb.1 - rbb.1;
                let ex = stock_bb.2 - rbb.2;
                let ey = stock_bb.3 - rbb.3;

                let mut best_y = f64::MAX;
                let mut best_x = f64::MAX;
                let mut best_tx = 0.0;
                let mut best_ty = 0.0;
                let mut found = false;

                let mut ty = sy;
                while ty <= ey + 1e-9 {
                    let mut tx = sx;
                    while tx <= ex + 1e-9 {
                        let candidate: Vec<(f64,f64)> = rotated.verts.iter().map(|&(x,y)| (x+tx, y+ty)).collect();
                        if stocks[new_si].can_place(&candidate, &stock_verts, stock_convex) {
                            if ty < best_y - 1e-9 || (ty < best_y + 1e-9 && tx < best_x - 1e-9) {
                                best_y = ty;
                                best_x = tx;
                                best_tx = tx;
                                best_ty = ty;
                                found = true;
                            }
                        }
                        tx += grid_step;
                    }
                    ty += grid_step;
                }

                if found {
                    let final_verts: Vec<(f64,f64)> = rotated.verts.iter().map(|&(x,y)| (x+best_tx, y+best_ty)).collect();
                    stocks[new_si].place(final_verts);
                    let final_dx = best_tx - ocx;
                    let final_dy = best_ty - ocy;
                    results[pi] = (new_si, final_dx, final_dy, angle);
                    done = true;
                    break;
                }
            }

            if !done {
                let tx = stock_bb.0 + stock_w * 0.5;
                let ty = stock_bb.1 + stock_h * 0.5;
                let final_dx = tx - ocx;
                let final_dy = ty - ocy;
                results[pi] = (new_si, final_dx, final_dy, 0.0);
            }
        }
    }

    let stdout = io::stdout();
    let mut out = BufWriter::new(stdout.lock());
    writeln!(out, "{}", stocks.len().max(1)).unwrap();
    for i in 0..np {
        let (si, dx, dy, rot) = results[i];
        writeln!(out, "{} {} {} {}", si, dx, dy, rot).unwrap();
    }
}
