use std::io::{self, Read, Write, BufWriter};

type Pt = (f64, f64);
type BB = (f64, f64, f64, f64);

struct Scanner<'a> { it: std::str::SplitAsciiWhitespace<'a> }
impl<'a> Scanner<'a> {
    fn new(s: &'a str) -> Self { Self { it: s.split_ascii_whitespace() } }
    fn f(&mut self) -> f64 { self.it.next().unwrap().parse().unwrap() }
    fn u(&mut self) -> usize { self.it.next().unwrap().parse().unwrap() }
}

struct Placed { verts: Vec<Pt>, bb: BB }

fn bbox(p: &[Pt]) -> BB {
    p.iter().fold((f64::MAX, f64::MAX, f64::MIN, f64::MIN),
        |(x0, y0, x1, y1), &(x, y)| (x0.min(x), y0.min(y), x1.max(x), y1.max(y)))
}

fn poly_area(p: &[Pt]) -> f64 {
    (0..p.len()).map(|i| { let j = (i + 1) % p.len(); p[i].0 * p[j].1 - p[j].0 * p[i].1 }).sum::<f64>() * 0.5
}

fn pip(pt: Pt, poly: &[Pt]) -> bool {
    let n = poly.len();
    let mut inside = false;
    let mut j = n - 1;
    for i in 0..n {
        let (xi, yi) = poly[i];
        let (xj, yj) = poly[j];
        if ((yi > pt.1) != (yj > pt.1)) && (pt.0 < (xj - xi) * (pt.1 - yi) / (yj - yi) + xi) {
            inside = !inside;
        }
        j = i;
    }
    inside
}

fn cross2(o: Pt, a: Pt, b: Pt) -> f64 {
    (a.0 - o.0) * (b.1 - o.1) - (a.1 - o.1) * (b.0 - o.0)
}

fn seg_cross(a1: Pt, a2: Pt, b1: Pt, b2: Pt) -> bool {
    let d1 = cross2(b1, b2, a1);
    let d2 = cross2(b1, b2, a2);
    let d3 = cross2(a1, a2, b1);
    let d4 = cross2(a1, a2, b2);
    let e = 1e-9;
    ((d1 > e && d2 < -e) || (d1 < -e && d2 > e)) && ((d3 > e && d4 < -e) || (d3 < -e && d4 > e))
}

fn contained(piece: &[Pt], stock: &[Pt]) -> bool {
    for &v in piece {
        if !pip(v, stock) { return false; }
    }
    let np = piece.len();
    let ns = stock.len();
    for i in 0..np {
        for j in 0..ns {
            if seg_cross(piece[i], piece[(i + 1) % np], stock[j], stock[(j + 1) % ns]) {
                return false;
            }
        }
    }
    true
}

fn polys_overlap(a: &[Pt], b: &[Pt]) -> bool {
    let na = a.len();
    let nb = b.len();
    for i in 0..na {
        for j in 0..nb {
            if seg_cross(a[i], a[(i + 1) % na], b[j], b[(j + 1) % nb]) { return true; }
        }
    }
    if pip(a[0], b) || pip(b[0], a) { return true; }
    false
}

fn try_place(
    rv: &[Pt], rw: f64, rh: f64, rot: f64,
    piece: &[Pt], stock: &[Pt], sb: BB, sw: f64, sh: f64, nsteps: usize,
    placed: &[Placed],
) -> Option<(Vec<Pt>, BB, f64, f64, f64)> {
    let max_tx = sw - rw;
    let max_ty = sh - rh;
    if max_tx < -1e-9 || max_ty < -1e-9 { return None; }

    let nx = if max_tx < 1e-12 { 0 } else { nsteps };
    let ny = if max_ty < 1e-12 { 0 } else { nsteps };
    let sx = if nx == 0 { 0.0 } else { max_tx / nx as f64 };
    let sy = if ny == 0 { 0.0 } else { max_ty / ny as f64 };

    for gy in 0..=ny {
        let ty = sb.1 + gy as f64 * sy;
        for gx in 0..=nx {
            let tx = sb.0 + gx as f64 * sx;
            let tb: BB = (tx, ty, tx + rw, ty + rh);
            let tv: Vec<Pt> = rv.iter().map(|&(x, y)| (x + tx, y + ty)).collect();

            if !contained(&tv, stock) { continue; }

            let mut ok = true;
            for pp in placed {
                if tb.0 >= pp.bb.2 - 1e-9 || tb.2 <= pp.bb.0 + 1e-9 ||
                   tb.1 >= pp.bb.3 - 1e-9 || tb.3 <= pp.bb.1 + 1e-9 { continue; }
                if polys_overlap(&tv, &pp.verts) { ok = false; break; }
            }
            if !ok { continue; }

            let (s, c) = rot.sin_cos();
            let rp0 = (piece[0].0 * c - piece[0].1 * s, piece[0].0 * s + piece[0].1 * c);
            let tx_out = tv[0].0 - rp0.0;
            let ty_out = tv[0].1 - rp0.1;
            return Some((tv, tb, tx_out, ty_out, rot));
        }
    }
    None
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let out = io::stdout();
    let mut out = BufWriter::new(out.lock());

    let mut sc = Scanner::new(&input);
    let sv = sc.u();
    let np = sc.u();
    let stock: Vec<Pt> = (0..sv).map(|_| (sc.f(), sc.f())).collect();
    let pieces: Vec<Vec<Pt>> = (0..np).map(|_| {
        let pv = sc.u();
        (0..pv).map(|_| (sc.f(), sc.f())).collect()
    }).collect();

    if np == 0 {
        writeln!(out, "0").unwrap();
        return;
    }

    let sb = bbox(&stock);
    let sw = sb.2 - sb.0;
    let sh = sb.3 - sb.1;
    let stock_area = poly_area(&stock).abs();

    let mut order: Vec<(usize, f64)> = pieces.iter().enumerate()
        .map(|(i, p)| (i, poly_area(p).abs()))
        .collect();
    order.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap());

    let nrot: usize = if np <= 50 { 24 } else if np <= 500 { 12 } else { 6 };
    let nsteps: usize = if np <= 50 { 30 } else if np <= 500 { 20 } else { 12 };
    let max_stocks_try: usize = if np <= 200 { usize::MAX } else { 20 };

    let rots: Vec<f64> = (0..nrot).map(|i| i as f64 * std::f64::consts::TAU / nrot as f64).collect();

    let mut stocks: Vec<(Vec<Placed>, f64)> = Vec::new();
    let mut results = vec![(0usize, 0.0f64, 0.0f64, 0.0f64); np];

    for &(oi, piece_area) in &order {
        let piece = &pieces[oi];
        let pb = bbox(piece);
        let norm: Vec<Pt> = piece.iter().map(|&(x, y)| (x - pb.0, y - pb.1)).collect();

        let mut rot_versions: Vec<(Vec<Pt>, f64, f64, f64)> = Vec::new();
        for &r in &rots {
            let (s, c) = r.sin_cos();
            let rv: Vec<Pt> = norm.iter().map(|&(x, y)| (x * c - y * s, x * s + y * c)).collect();
            let rb = bbox(&rv);
            let rw = rb.2 - rb.0;
            let rh = rb.3 - rb.1;
            if rw > sw + 1e-9 || rh > sh + 1e-9 { continue; }
            let rv: Vec<Pt> = rv.iter().map(|&(x, y)| (x - rb.0, y - rb.1)).collect();
            rot_versions.push((rv, rw, rh, r));
        }

        let mut done = false;

        let start = stocks.len().saturating_sub(max_stocks_try);
        'found: for si in start..stocks.len() {
            if piece_area > stock_area - stocks[si].1 + 1e-6 { continue; }
            for (rv, rw, rh, rot) in &rot_versions {
                if let Some((tv, tb, tx, ty, r)) = try_place(
                    rv, *rw, *rh, *rot, piece, &stock, sb, sw, sh, nsteps, &stocks[si].0,
                ) {
                    results[oi] = (si, tx, ty, r);
                    stocks[si].0.push(Placed { verts: tv, bb: tb });
                    stocks[si].1 += piece_area;
                    done = true;
                    break 'found;
                }
            }
        }

        if !done {
            let si = stocks.len();
            stocks.push((Vec::new(), 0.0));
            for (rv, rw, rh, rot) in &rot_versions {
                if let Some((tv, tb, tx, ty, r)) = try_place(
                    rv, *rw, *rh, *rot, piece, &stock, sb, sw, sh, nsteps, &[],
                ) {
                    results[oi] = (si, tx, ty, r);
                    stocks[si].0.push(Placed { verts: tv, bb: tb });
                    stocks[si].1 += piece_area;
                    done = true;
                    break;
                }
            }
            if !done {
                results[oi] = (si, 0.0, 0.0, 0.0);
            }
        }
    }

    writeln!(out, "{}", stocks.len()).unwrap();
    for &(si, tx, ty, r) in &results {
        writeln!(out, "{} {} {} {}", si, tx, ty, r).unwrap();
    }
}
