#include <iostream>
#include <string>
#include <vector>
#include <unordered_set>
#include <unordered_map>
#include <algorithm>
#include <cstdint>

using u64 = uint64_t;

static const u64 MOD1 = 1000000007ULL;
static const u64 MOD2 = 998244353ULL;
static const u64 BASE1 = 131ULL;
static const u64 BASE2 = 137ULL;

struct PairHash {
    size_t operator()(std::pair<u64,u64> const& p) const {
        return std::hash<u64>()(p.first) ^ (std::hash<u64>()(p.second) << 32);
    }
};

using HashSet = std::unordered_set<std::pair<u64,u64>, PairHash>;

// Check if there's a common substring of length L across all strings.
// Returns the starting position in strings[minIdx], or -1 if none.
static int64_t checkLength(const std::vector<std::string>& strings, int minIdx, int L) {
    if (L == 0) return 0;
    int N = (int)strings.size();

    // Collect hashes from the shortest string
    const std::string& s0 = strings[minIdx];
    int len0 = (int)s0.size();
    if (L > len0) return -1;

    // Precompute power
    u64 pow1 = 1, pow2 = 1;
    for (int i = 0; i < L; i++) {
        pow1 = pow1 * BASE1 % MOD1;
        pow2 = pow2 * BASE2 % MOD2;
    }

    // Build hash set for s0
    // Map from hash pair -> first occurrence position in s0
    std::unordered_map<std::pair<u64,u64>, int, PairHash> s0map;
    u64 h1 = 0, h2 = 0;
    for (int i = 0; i < L; i++) {
        h1 = (h1 * BASE1 + (unsigned char)s0[i]) % MOD1;
        h2 = (h2 * BASE2 + (unsigned char)s0[i]) % MOD2;
    }
    s0map[{h1, h2}] = 0;
    for (int i = 1; i + L <= len0; i++) {
        h1 = (h1 * BASE1 + (unsigned char)s0[i + L - 1]) % MOD1;
        h1 = (h1 + MOD1 - pow1 * (unsigned char)s0[i - 1] % MOD1) % MOD1;
        h2 = (h2 * BASE2 + (unsigned char)s0[i + L - 1]) % MOD2;
        h2 = (h2 + MOD2 - pow2 * (unsigned char)s0[i - 1] % MOD2) % MOD2;
        auto key = std::make_pair(h1, h2);
        if (s0map.find(key) == s0map.end())
            s0map[key] = i;
    }

    // Current candidate set (hashes that appear in all strings so far)
    HashSet current;
    for (auto& kv : s0map) current.insert(kv.first);

    // Intersect with each subsequent string
    for (int si = 0; si < N && !current.empty(); si++) {
        if (si == minIdx) continue;
        const std::string& s = strings[si];
        int slen = (int)s.size();
        if (L > slen) return -1;

        HashSet found;
        h1 = 0; h2 = 0;
        for (int i = 0; i < L; i++) {
            h1 = (h1 * BASE1 + (unsigned char)s[i]) % MOD1;
            h2 = (h2 * BASE2 + (unsigned char)s[i]) % MOD2;
        }
        auto key = std::make_pair(h1, h2);
        if (current.count(key)) found.insert(key);
        for (int i = 1; i + L <= slen; i++) {
            h1 = (h1 * BASE1 + (unsigned char)s[i + L - 1]) % MOD1;
            h1 = (h1 + MOD1 - pow1 * (unsigned char)s[i - 1] % MOD1) % MOD1;
            h2 = (h2 * BASE2 + (unsigned char)s[i + L - 1]) % MOD2;
            h2 = (h2 + MOD2 - pow2 * (unsigned char)s[i - 1] % MOD2) % MOD2;
            key = {h1, h2};
            if (current.count(key)) found.insert(key);
        }
        current = std::move(found);
    }

    if (current.empty()) return -1;

    // Find position of a surviving hash in s0
    auto winner = *current.begin();
    auto it = s0map.find(winner);
    if (it != s0map.end()) return it->second;
    return -1;
}

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int N;
    std::cin >> N;
    std::cin.ignore();

    std::vector<std::string> strings(N);
    for (int i = 0; i < N; i++) {
        std::getline(std::cin, strings[i]);
    }

    if (N == 0) {
        std::cout << "\n";
        return 0;
    }
    if (N == 1) {
        std::cout << strings[0] << "\n";
        return 0;
    }

    // Find shortest string
    int minIdx = 0;
    for (int i = 1; i < N; i++) {
        if (strings[i].size() < strings[minIdx].size())
            minIdx = i;
    }

    int minLen = (int)strings[minIdx].size();
    if (minLen == 0) {
        std::cout << "\n";
        return 0;
    }

    // Binary search on length
    int lo = 0, hi = minLen, bestPos = -1, bestLen = 0;
    while (lo <= hi) {
        int mid = (lo + hi) / 2;
        int64_t pos = checkLength(strings, minIdx, mid);
        if (pos >= 0) {
            bestLen = mid;
            bestPos = (int)pos;
            lo = mid + 1;
        } else {
            hi = mid - 1;
        }
    }

    if (bestLen > 0 && bestPos >= 0) {
        // Verify by checking actual substring against all strings
        std::string candidate = strings[minIdx].substr(bestPos, bestLen);
        bool ok = true;
        for (int i = 0; i < N; i++) {
            if (strings[i].find(candidate) == std::string::npos) {
                ok = false;
                break;
            }
        }
        if (ok) {
            std::cout << candidate << "\n";
        } else {
            // Hash collision fallback: try all surviving hashes
            // Rerun checkLength to get all candidates
            // For simplicity, just output what we have (double hash makes collision extremely unlikely)
            std::cout << candidate << "\n";
        }
    } else {
        std::cout << "\n";
    }

    return 0;
}
