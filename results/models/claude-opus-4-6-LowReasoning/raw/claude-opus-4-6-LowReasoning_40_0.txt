#include <bits/stdc++.h>
using namespace std;

int n, m;
vector<int> c;
vector<vector<int>> adj;
vector<int> best_sol;
int best_obj;

// Maximum Weight Independent Set via branch and bound
void branch_and_bound(vector<int>& sol, vector<int>& status, int cur_obj, int ub) {
    if (cur_obj + ub <= best_obj) return;

    // Find next undecided variable (pick highest weight)
    int pick = -1;
    for (int i = 0; i < n; i++) {
        if (status[i] == 0 && (pick == -1 || c[i] > c[pick])) pick = i;
    }

    if (pick == -1) {
        if (cur_obj > best_obj) {
            best_obj = cur_obj;
            best_sol = sol;
        }
        return;
    }

    // Try including pick (x=1)
    {
        bool can = true;
        for (int v : adj[pick]) {
            if (sol[v] == 1) { can = false; break; }
        }
        if (can) {
            sol[pick] = 1;
            status[pick] = 1;
            int new_ub = ub - c[pick];
            // Exclude neighbors
            vector<int> excluded;
            for (int v : adj[pick]) {
                if (status[v] == 0) {
                    status[v] = -1;
                    new_ub -= c[v];
                    excluded.push_back(v);
                }
            }
            branch_and_bound(sol, status, cur_obj + c[pick], new_ub);
            for (int v : excluded) status[v] = 0;
            sol[pick] = 0;
            status[pick] = 0;
        }
    }

    // Try excluding pick (x=0)
    {
        status[pick] = -1;
        branch_and_bound(sol, status, cur_obj, ub - c[pick]);
        status[pick] = 0;
    }
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    c.resize(n);
    adj.resize(n);
    for (int i = 0; i < n; i++) cin >> c[i];
    for (int i = 0; i < m; i++) {
        int u, v; cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    best_obj = -1;
    best_sol.assign(n, 0);
    vector<int> sol(n, 0);
    vector<int> status(n, 0); // 0=undecided, 1=included, -1=excluded

    int total = 0;
    for (int i = 0; i < n; i++) total += c[i];

    branch_and_bound(sol, status, 0, total);

    cout << best_obj << "\n";
    for (int i = 0; i < n; i++) {
        if (i) cout << " ";
        cout << best_sol[i];
    }
    cout << "\n";
}
