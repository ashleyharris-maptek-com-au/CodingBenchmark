import struct

def w(val):
    """Encode a 32-bit word as little-endian hex."""
    return struct.pack('<I', val & 0xFFFFFFFF).hex()

def op(wordcount, opcode):
    return (wordcount << 16) | opcode

def string_words(s):
    """Encode a string as SPIR-V words (null-terminated, padded to 4 bytes)."""
    b = s.encode('utf-8') + b'\x00'
    while len(b) % 4 != 0:
        b += b'\x00'
    words = []
    for i in range(0, len(b), 4):
        words.append(struct.unpack('<I', b[i:i+4])[0])
    return words

# IDs

T_LOAD_N = 24

BOUND = 35

hex_str = ''

# Header
hex_str += w(0x07230203)  # magic
hex_str += w(0x00010000)  # version 1.0
hex_str += w(0)           # generator
hex_str += w(BOUND)
hex_str += w(0)           # reserved

# OpCapability Shader
hex_str += w(op(2, 17)) + w(1)

# OpExtInstImport %1 "GLSL.std.450"
name_w = string_words("GLSL.std.450")
hex_str += w(op(2 + len(name_w), 11)) + w(EXT)
for nw in name_w:
    hex_str += w(nw)

# OpMemoryModel Logical GLSL450
hex_str += w(op(3, 14)) + w(0) + w(1)

# OpEntryPoint Fragment %22 "main" %14 %15 %16
name_w = string_words("main")
iface = [VAR_WORLDPOS, VAR_NORMAL, VAR_OUTCOLOR]
wc = 3 + len(name_w) + len(iface)
hex_str += w(op(wc, 15)) + w(4) + w(MAIN)
for nw in name_w:
    hex_str += w(nw)
for iv in iface:
    hex_str += w(iv)

# OpExecutionMode %22 OriginUpperLeft
hex_str += w(op(3, 16)) + w(MAIN) + w(7)

# Decorations
def decorate(target, dec, val):
    return w(op(4, 71)) + w(target) + w(dec) + w(val)

def decorate_no_val(target, dec):
    return w(op(3, 71)) + w(target) + w(dec)

def member_decorate(struct, member, dec, val=None):
    if val is not None:
        return w(op(5, 72)) + w(struct) + w(member) + w(dec) + w(val)
    else:
        return w(op(4, 72)) + w(struct) + w(member) + w(dec)

hex_str += decorate(VAR_WORLDPOS, 30, 0)  # Location 0
hex_str += decorate(VAR_NORMAL, 30, 1)    # Location 1
hex_str += decorate(VAR_OUTCOLOR, 30, 0)  # Location 0
hex_str += decorate(VAR_UBO, 34, 0)       # DescriptorSet 0
hex_str += decorate(VAR_UBO, 33, 0)       # Binding 0
hex_str += decorate_no_val(UBO_STRUCT, 2) # Block

# Member offsets
hex_str += member_decorate(UBO_STRUCT, 0, 35, 0)    # Offset 0
hex_str += member_decorate(UBO_STRUCT, 1, 35, 64)   # Offset 64
hex_str += member_decorate(UBO_STRUCT, 2, 35, 128)  # Offset 128
hex_str += member_decorate(UBO_STRUCT, 3, 35, 192)  # Offset 192
hex_str += member_decorate(UBO_STRUCT, 4, 35, 208)  # Offset 208
hex_str += member_decorate(UBO_STRUCT, 5, 35, 224)  # Offset 224

# ColMajor and MatrixStride for mat4 members (0,1,2)
for i in range(3):
    hex_str += member_decorate(UBO_STRUCT, i, 5)      # ColMajor
    hex_str += member_decorate(UBO_STRUCT, i, 7, 16)  # MatrixStride 16

# Types
hex_str += w(op(2, 19)) + w(VOID)                        # OpTypeVoid
hex_str += w(op(3, 33)) + w(FUNC_TYPE) + w(VOID)         # OpTypeFunction void()
hex_str += w(op(3, 22)) + w(FLOAT) + w(32)               # OpTypeFloat 32
hex_str += w(op(4, 23)) + w(VEC3) + w(FLOAT) + w(3)      # OpTypeVector vec3
hex_str += w(op(4, 23)) + w(VEC4) + w(FLOAT) + w(4)      # OpTypeVector vec4
hex_str += w(op(4, 24)) + w(MAT4) + w(VEC4) + w(4)       # OpTypeMatrix mat4
hex_str += w(op(4, 21)) + w(INT) + w(32) + w(0)          # OpTypeInt 32 unsigned
hex_str += w(op(8, 30)) + w(UBO_STRUCT) + w(MAT4) + w(MAT4) + w(MAT4) + w(VEC4) + w(VEC4) + w(VEC4)
hex_str += w(op(4, 32)) + w(PTR_IN_VEC3) + w(1) + w(VEC3)    # ptr Input vec3
hex_str += w(op(4, 32)) + w(PTR_OUT_VEC4) + w(3) + w(VEC4)   # ptr Output vec4
hex_str += w(op(4, 32)) + w(PTR_UNI_STRUCT) + w(2) + w(UBO_STRUCT)  # ptr Uniform struct
hex_str += w(op(4, 32)) + w(PTR_UNI_VEC4) + w(2) + w(VEC4)   # ptr Uniform vec4

# Variables
hex_str += w(op(4, 59)) + w(PTR_IN_VEC3) + w(VAR_WORLDPOS) + w(1)
hex_str += w(op(4, 59)) + w(PTR_IN_VEC3) + w(VAR_NORMAL) + w(1)
hex_str += w(op(4, 59)) + w(PTR_OUT_VEC4) + w(VAR_OUTCOLOR) + w(3)
hex_str += w(op(4, 59)) + w(PTR_UNI_STRUCT) + w(VAR_UBO) + w(2)

# Constants
hex_str += w(op(4, 43)) + w(INT) + w(CONST_INT0) + w(0)
hex_str += w(op(4, 43)) + w(INT) + w(CONST_INT3) + w(3)
hex_str += w(op(4, 43)) + w(FLOAT) + w(CONST_F0) + w(0x00000000)
hex_str += w(op(4, 43)) + w(FLOAT) + w(CONST_F1) + w(0x3F800000)

# Function
hex_str += w(op(5, 54)) + w(VOID) + w(MAIN) + w(0) + w(FUNC_TYPE)  # OpFunction
hex_str += w(op(2, 248)) + w(LABEL)  # OpLabel

# Load normal
hex_str += w(op(4, 61)) + w(VEC3) + w(T_LOAD_N) + w(VAR_NORMAL)
# Normalize normal
hex_str += w(op(6, 12)) + w(VEC3) + w(T_NORM_N) + w(EXT) + w(69) + w(T_LOAD_N)
# Load worldPos
hex_str += w(op(4, 61)) + w(VEC3) + w(T_LOAD_WP) + w(VAR_WORLDPOS)
# Access chain to lightPos (member 3)
hex_str += w(op(5, 65)) + w(PTR_UNI_VEC4) + w(T_AC_LP) + w(VAR_UBO) + w(CONST_INT3)
# Load lightPos
hex_str += w(op(4, 61)) + w(VEC4) + w(T_LOAD_LP) + w(T_AC_LP)
# VectorShuffle lightPos.xyz
hex_str += w(op(8, 79)) + w(VEC3) + w(T_LP_XYZ) + w(T_LOAD_LP) + w(T_LOAD_LP) + w(0) + w(1) + w(2)
# FSub lightPos.xyz - worldPos
hex_str += w(op(5, 131)) + w(VEC3) + w(T_SUB) + w(T_LP_XYZ) + w(T_LOAD_WP)
# Normalize L
hex_str += w(op(6, 12)) + w(VEC3) + w(T_NORM_L) + w(EXT) + w(69) + w(T_SUB)
# Dot(N, L)
hex_str += w(op(5, 148)) + w(FLOAT) + w(T_DOT) + w(T_NORM_N) + w(T_NORM_L)
# FMax(dot, 0.0)
hex_str += w(op(7, 12)) + w(FLOAT) + w(T_MAX) + w(EXT) + w(40) + w(T_DOT) + w(CONST_F0)
# CompositeConstruct vec4(d, d, d, 1.0)
hex_str += w(op(7, 80)) + w(VEC4) + w(T_VEC4) + w(T_MAX) + w(T_MAX) + w(T_MAX) + w(CONST_F1)
# Store outColor
hex_str += w(op(3, 62)) + w(VAR_OUTCOLOR) + w(T_VEC4)
# Return
hex_str += w(op(1, 253))
# FunctionEnd
hex_str += w(op(1, 56))

print(hex_str)
