; Version: 1.0
; Generator: Hand-written
               OpCapability Shader
               OpMemoryModel Logical GLSL450
               OpEntryPoint Fragment %main "main" %outColor
               OpExecutionMode %main OriginUpperLeft

               ; Decorations
               OpDecorate %outColor Location 0

               ; Types
       %void = OpTypeVoid
   %funcType = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v4float = OpTypeVector %float 4
%ptr_out_v4f = OpTypePointer Output %v4float

               ; Constants
    %float_0 = OpConstant %float 0.0
    %float_1 = OpConstant %float 1.0
        %red = OpConstantComposite %v4float %float_1 %float_0 %float_0 %float_1

               ; Variables
   %outColor = OpVariable %ptr_out_v4f Output

               ; Main function
       %main = OpFunction %void None %funcType
      %entry = OpLabel
               OpStore %outColor %red
               OpReturn
               OpFunctionEnd
