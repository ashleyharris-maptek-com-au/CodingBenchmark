import struct

words = []

def w(val):
    words.append(val)

def op(opcode, *operands):
    wc = 1 + len(operands)
    w((wc << 16) | opcode)
    for o in operands:
        w(o)

def op_str(opcode, *operands_before, name, operands_after=()):
    # Encode name as null-terminated, padded to 4-byte boundary
    encoded = name.encode('utf-8') + b'\x00'
    while len(encoded) % 4 != 0:
        encoded += b'\x00'
    name_words = [struct.unpack('<I', encoded[i:i+4])[0] for i in range(0, len(encoded), 4)]
    wc = 1 + len(operands_before) + len(name_words) + len(operands_after)
    w((wc << 16) | opcode)
    for o in operands_before:
        w(o)
    for nw in name_words:
        w(nw)
    for o in operands_after:
        w(o)

def f32(val):
    return struct.unpack('<I', struct.pack('<f', val))[0]

# Header placeholder
header_pos = len(words)

w(0)           # generator
w(0)           # bound - fill later

# OpCapability Shader
op(17, 1)

# OpExtInstImport %1 "GLSL.std.450"
op_str(11, GLSL, name="GLSL.std.450")

# OpMemoryModel Logical GLSL450
op(14, 0, 1)

# OpEntryPoint Fragment %27 "main" %15 %16 %17
op_str(15, 4, MAIN, name="main", operands_after=(WORLDPOS_VAR, NORMAL_VAR, OUTCOLOR_VAR))

# OpExecutionMode %27 OriginUpperLeft
op(16, MAIN, 7)

# Decorations
# Location decorations
op(71, WORLDPOS_VAR, 30, 0)   # worldPos location 0
op(71, NORMAL_VAR, 30, 1)     # normal location 1
op(71, OUTCOLOR_VAR, 30, 0)   # outColor location 0

# UBO decorations
op(71, UBO_VAR, 34, 0)    # DescriptorSet 0
op(71, UBO_VAR, 33, 0)    # Binding 0
op(71, UBO_STRUCT, 2)     # Block

# MemberDecorate for offsets (opcode 72)
for member, offset in [(0,0),(1,64),(2,128),(3,192),(4,208),(5,224)]:
    op(72, UBO_STRUCT, member, 35, offset)  # Offset

# ColMajor (5) and MatrixStride (7) for mat4 members 0,1,2
for member in [0, 1, 2]:
    op(72, UBO_STRUCT, member, 5)      # ColMajor (word_count=4)
    op(72, UBO_STRUCT, member, 7, 16)  # MatrixStride 16

# Type declarations
op(19, VOID)                    # OpTypeVoid
op(33, FUNC_TYPE, VOID)         # OpTypeFunction void
op(22, FLOAT, 32)               # OpTypeFloat 32
op(23, VEC3, FLOAT, 3)          # OpTypeVector float 3
op(23, VEC4, FLOAT, 4)          # OpTypeVector float 4
op(24, MAT4, VEC4, 4)           # OpTypeMatrix vec4 4
# OpTypeStruct
w((8 << 16) | 30)  # 8 words: opcode + result + 6 members, OpTypeStruct=30
w(UBO_STRUCT)
w(MAT4); w(MAT4); w(MAT4); w(VEC4); w(VEC4); w(VEC4)

op(32, PTR_UNI_STRUCT, 2, UBO_STRUCT)  # Uniform=2
op(32, PTR_UNI_VEC4, 2, VEC4)
op(32, PTR_IN_VEC3, 1, VEC3)          # Input=1
op(32, PTR_OUT_VEC4, 3, VEC4)         # Output=3

# Fix: OpTypeInt is opcode 21
words.pop(); words.pop(); words.pop()  # remove wrong entry
op(21, INT32, 32, 1)  # OpTypeInt %13 32 signed

# Variables
op(59, PTR_UNI_STRUCT, UBO_VAR, 2)    # Uniform storage class = 2
op(59, PTR_IN_VEC3, WORLDPOS_VAR, 1)  # Input = 1
op(59, PTR_IN_VEC3, NORMAL_VAR, 1)
op(59, PTR_OUT_VEC4, OUTCOLOR_VAR, 3) # Output = 3

# Constants
op(43, INT32, CONST_INT_3, 3)
op(43, INT32, CONST_INT_4, 4)
op(43, FLOAT, CONST_F_0, f32(0.0))

op(43, FLOAT, CONST_F_03, f32(0.3))

# Function
op(54, VOID, MAIN, 0, FUNC_TYPE)  # OpFunction None
op(248, LABEL)

# Load inputs
op(61, VEC3, LOAD_WP, WORLDPOS_VAR)   # OpLoad
op(61, VEC3, LOAD_N, NORMAL_VAR)

# Access chain and load lightPos (member 3)
op(65, PTR_UNI_VEC4, AC_LIGHT, UBO_VAR, CONST_INT_3)  # OpAccessChain=65
op(61, VEC4, LOAD_LIGHT, AC_LIGHT)
# VectorShuffle to get xyz
op(79, VEC3, LIGHT_XYZ, LOAD_LIGHT, LOAD_LIGHT, 0, 1, 2)  # OpVectorShuffle

# Access chain and load cameraPos (member 4)
op(65, PTR_UNI_VEC4, AC_CAM, UBO_VAR, CONST_INT_4)
op(61, VEC4, LOAD_CAM, AC_CAM)
op(79, VEC3, CAM_XYZ, LOAD_CAM, LOAD_CAM, 0, 1, 2)

# N = normalize(normal)
# OpExtInst %result %type %extset instruction_number %operand
# GLSL Normalize = 69
w((6 << 16) | 12); w(VEC3); w(NORM_N); w(GLSL); w(69); w(LOAD_N)

# L = normalize(lightPos.xyz - worldPos)
op(131, VEC3, L_UNNORM, LIGHT_XYZ, LOAD_WP)  # OpFSub
w((6 << 16) | 12); w(VEC3); w(NORM_L); w(GLSL); w(69); w(L_UNNORM)

# V = normalize(cameraPos.xyz - worldPos)
op(131, VEC3, V_UNNORM, CAM_XYZ, LOAD_WP)
w((6 << 16) | 12); w(VEC3); w(NORM_V); w(GLSL); w(69); w(V_UNNORM)

# diffuse = max(dot(N, L), 0)
op(148, FLOAT, DOT_NL, NORM_N, NORM_L)  # OpDot
# FMax = GLSL 40
w((7 << 16) | 12); w(FLOAT); w(DIFFUSE); w(GLSL); w(40); w(DOT_NL); w(CONST_F_0)

# dot(N, V) and max
op(148, FLOAT, DOT_NV, NORM_N, NORM_V)
w((7 << 16) | 12); w(FLOAT); w(MAX_NV); w(GLSL); w(40); w(DOT_NV); w(CONST_F_0)

# 1 - max(dot(N,V), 0)
op(131, FLOAT, ONE_MINUS, CONST_F_1, MAX_NV)

# rim = pow(1 - max(dot(N,V),0), 3)
# Pow = GLSL 26
w((7 << 16) | 12); w(FLOAT); w(RIM); w(GLSL); w(26); w(ONE_MINUS); w(CONST_F_3)

# 0.5 * diffuse
op(133, FLOAT, HALF_DIFF, CONST_F_05, DIFFUSE)
# 0.3 + 0.5*diffuse
op(129, FLOAT, FACTOR, CONST_F_03, HALF_DIFF)

# vec3(0.2, 0.3, 0.8)
op(80, VEC3, BASE_VEC, CONST_F_02, CONST_F_03, CONST_F_08)

# base * factor: OpVectorTimesScalar = 142
op(142, VEC3, BASE_COL, BASE_VEC, FACTOR)

# vec3(1.0, 0.8, 0.5)
op(80, VEC3, RIM_VEC, CONST_F_1, CONST_F_08, CONST_F_05)

# rimTint * rim
op(142, VEC3, RIM_COL, RIM_VEC, RIM)

# final = base + rim
op(129, VEC3, FINAL, BASE_COL, RIM_COL)

# Extract and construct vec4
op(81, FLOAT, EX_X, FINAL, 0)
op(81, FLOAT, EX_Y, FINAL, 1)
op(81, FLOAT, EX_Z, FINAL, 2)
op(80, VEC4, OUT_VEC, EX_X, EX_Y, EX_Z, CONST_F_1)

# Store
op(62, OUTCOLOR_VAR, OUT_VEC)

# Return
op(253)

# FunctionEnd
op(56)
