using System;
using System.Collections.Generic;
using System.IO;

class Program
{
    static int N;
    static HashSet<int>[] Succ, Pred;
    static bool[] Dead;

    static void Main()
    {
        var sr = new StreamReader(Console.OpenStandardInput());
        var tok = sr.ReadLine().Split();
        N = int.Parse(tok[0]);
        int M = int.Parse(tok[1]);

        Succ = new HashSet<int>[N];
        Pred = new HashSet<int>[N];
        Dead = new bool[N];
        for (int i = 0; i < N; i++) { Succ[i] = new HashSet<int>(); Pred[i] = new HashSet<int>(); }

        bool[] hasSelf = new bool[N];
        for (int i = 0; i < M; i++)
        {
            tok = sr.ReadLine().Split();
            int u = int.Parse(tok[0]), v = int.Parse(tok[1]);
            if (u == v) hasSelf[u] = true;
            else { Succ[u].Add(v); Pred[v].Add(u); }
        }

        var result = new List<int>();
        for (int i = 0; i < N; i++)
            if (hasSelf[i]) { result.Add(i); Kill(i); }

        result.AddRange(Reduce());

        var active = new HashSet<int>();
        for (int i = 0; i < N; i++) if (!Dead[i]) active.Add(i);
        var sccs = FindSCCs(active);
        foreach (var scc in sccs)
            if (scc.Count > 1) result.AddRange(SolveSCC(scc));

        var sw = new StreamWriter(Console.OpenStandardOutput());
        sw.WriteLine(result.Count);
        sw.WriteLine(result.Count > 0 ? string.Join(" ", result) : "");
        sw.Flush();
    }

    static void Kill(int v)
    {
        Dead[v] = true;
        foreach (var w in Succ[v]) Pred[w].Remove(v);
        foreach (var u in Pred[v]) Succ[u].Remove(v);
        Succ[v].Clear(); Pred[v].Clear();
    }

    static int First(HashSet<int> s) { foreach (var x in s) return x; return -1; }

    static List<int> Reduce()
    {
        var forced = new List<int>();
        var q = new Queue<int>();
        var inQ = new bool[N];
        for (int i = 0; i < N; i++) if (!Dead[i]) { q.Enqueue(i); inQ[i] = true; }

        while (q.Count > 0)
        {
            int v = q.Dequeue(); inQ[v] = false;
            if (Dead[v]) continue;

            if (Succ[v].Contains(v))
            {
                forced.Add(v);
                foreach (var w in Succ[v]) if (!Dead[w] && w != v && !inQ[w]) { q.Enqueue(w); inQ[w] = true; }
                foreach (var u in Pred[v]) if (!Dead[u] && u != v && !inQ[u]) { q.Enqueue(u); inQ[u] = true; }
                Kill(v); continue;
            }

            if (Succ[v].Count == 0 || Pred[v].Count == 0)
            {
                foreach (var w in Succ[v]) if (!Dead[w] && !inQ[w]) { q.Enqueue(w); inQ[w] = true; }
                foreach (var u in Pred[v]) if (!Dead[u] && !inQ[u]) { q.Enqueue(u); inQ[u] = true; }
                Kill(v); continue;
            }

            if (Pred[v].Count == 1)
            {
                int u = First(Pred[v]);
                if (!inQ[u] && !Dead[u]) { q.Enqueue(u); inQ[u] = true; }
                var succCopy = new List<int>(Succ[v]);
                foreach (var w in succCopy)
                {
                    Pred[w].Remove(v);
                    if (w != u) { Succ[u].Add(w); Pred[w].Add(u); if (!inQ[w] && !Dead[w]) { q.Enqueue(w); inQ[w] = true; } }
                    else { Succ[u].Add(u); Pred[u].Add(u); }
                }
                Succ[u].Remove(v);
                Succ[v].Clear(); Pred[v].Clear(); Dead[v] = true;
                continue;
            }

            if (Succ[v].Count == 1)
            {
                int w = First(Succ[v]);
                if (!inQ[w] && !Dead[w]) { q.Enqueue(w); inQ[w] = true; }
                var predCopy = new List<int>(Pred[v]);
                foreach (var u in predCopy)
                {
                    Succ[u].Remove(v);
                    if (u != w) { Pred[w].Add(u); Succ[u].Add(w); if (!inQ[u] && !Dead[u]) { q.Enqueue(u); inQ[u] = true; } }
                    else { Succ[w].Add(w); Pred[w].Add(w); }
                }
                Pred[w].Remove(v);
                Succ[v].Clear(); Pred[v].Clear(); Dead[v] = true;
                continue;
            }
        }
        return forced;
    }

    static List<List<int>> FindSCCs(HashSet<int> active)
    {
        var visited = new HashSet<int>();
        var order = new List<int>();
        foreach (var v in active)
            if (!visited.Contains(v)) DFS1(v, active, visited, order);
        visited.Clear();
        var sccs = new List<List<int>>();
        for (int i = order.Count - 1; i >= 0; i--)
            if (!visited.Contains(order[i]))
            {
                var scc = new List<int>();
                DFS2(order[i], active, visited, scc);
                sccs.Add(scc);
            }
        return sccs;
    }

    static void DFS1(int s, HashSet<int> active, HashSet<int> vis, List<int> order)
    {
        var stk = new Stack<(int, bool)>();
        stk.Push((s, false));
        while (stk.Count > 0)
        {
            var (v, done) = stk.Pop();
            if (done) { order.Add(v); continue; }
            if (vis.Contains(v)) continue;
            vis.Add(v);
            stk.Push((v, true));
            foreach (var w in Succ[v]) if (active.Contains(w) && !vis.Contains(w)) stk.Push((w, false));
        }
    }

    static void DFS2(int s, HashSet<int> active, HashSet<int> vis, List<int> scc)
    {
        var stk = new Stack<int>();
        stk.Push(s);
        while (stk.Count > 0)
        {
            int v = stk.Pop();
            if (vis.Contains(v)) continue;
            vis.Add(v); scc.Add(v);
            foreach (var u in Pred[v]) if (active.Contains(u) && !vis.Contains(u)) stk.Push(u);
        }
    }

    static List<int> SolveSCC(List<int> scc)
    {
        var active = new HashSet<int>(scc);
        var greedy = GreedyFVS(new HashSet<int>(active));
        int bestCost = greedy.Count;
        var bestSol = greedy;
        var cur = new List<int>();
        BnB(active, cur, ref bestCost, ref bestSol);
        return bestSol;
    }

    static List<int> GreedyFVS(HashSet<int> work)
    {
        var res = new List<int>();
        while (true)
        {
            bool changed = true;
            while (changed)
            {
                changed = false;
                var rm = new List<int>();
                foreach (var v in work)
                {
                    bool hi = false, ho = false;
                    foreach (var u in Pred[v]) if (work.Contains(u)) { hi = true; break; }
                    if (!hi) { rm.Add(v); continue; }
                    foreach (var w in Succ[v]) if (work.Contains(w)) { ho = true; break; }
                    if (!ho) rm.Add(v);
                }
                foreach (var v in rm) { work.Remove(v); changed = true; }
            }
            if (work.Count == 0) break;
            int best = -1, bd = -1;
            foreach (var v in work)
            {
                int d = 0;
                foreach (var u in Pred[v]) if (work.Contains(u)) d++;
                foreach (var w in Succ[v]) if (work.Contains(w)) d++;
                if (d > bd) { bd = d; best = v; }
            }
            res.Add(best); work.Remove(best);
        }
        return res;
    }

    static void BnB(HashSet<int> active, List<int> cur, ref int bestCost, ref List<int> bestSol)
    {
        if (cur.Count >= bestCost) return;

        var sccs = FindSCCs(active);
        int lb = 0;
        List<int> target = null;
        foreach (var scc in sccs)
        {
            if (scc.Count > 1)
            {
                int slb = PackCycles(scc);
                lb += slb;
                if (target == null) target = scc;
            }
        }

        if (lb == 0) { if (cur.Count < bestCost) { bestCost = cur.Count; bestSol = new List<int>(cur); } return; }
        if (cur.Count + lb >= bestCost) return;

        var cycle = ShortestCycle(target, active);
        if (cycle == null) return;

        foreach (var v in cycle)
        {
            active.Remove(v);
            cur.Add(v);
            BnB(active, cur, ref bestCost, ref bestSol);
            cur.RemoveAt(cur.Count - 1);
            active.Add(v);
        }
    }

    static int PackCycles(List<int> scc)
    {
        var work = new HashSet<int>(scc);
        int cnt = 0;
        while (work.Count > 1)
        {
            var cy = AnyCycle(work);
            if (cy == null) break;
            cnt++;
            foreach (var v in cy) work.Remove(v);
        }
        return cnt;
    }

    static List<int> AnyCycle(HashSet<int> verts)
    {
        var color = new Dictionary<int, int>();
        foreach (var v in verts) color[v] = 0;

        foreach (var start in verts)
        {
            if (color[start] != 0) continue;
            var path = new List<int>();
            var onPath = new HashSet<int>();
            var stk = new Stack<(int v, List<int> succs, int idx)>();

            color[start] = 1;
            path.Add(start); onPath.Add(start);
            stk.Push((start, Filtered(start, verts), 0));

            while (stk.Count > 0)
            {
                var (v, succs, idx) = stk.Pop();
                bool pushed = false;
                for (int i = idx; i < succs.Count; i++)
                {
                    int w = succs[i];
                    if (onPath.Contains(w))
                    {
                        int ci = path.IndexOf(w);
                        return path.GetRange(ci, path.Count - ci);
                    }
                    if (color.ContainsKey(w) && color[w] == 0)
                    {
                        stk.Push((v, succs, i + 1));
                        color[w] = 1;
                        path.Add(w); onPath.Add(w);
                        stk.Push((w, Filtered(w, verts), 0));
                        pushed = true;
                        break;
                    }
                }
                if (!pushed) { color[v] = 2; onPath.Remove(v); path.RemoveAt(path.Count - 1); }
            }
        }
        return null;
    }

    static List<int> Filtered(int v, HashSet<int> verts)
    {
        var r = new List<int>();
        foreach (var w in Succ[v]) if (verts.Contains(w)) r.Add(w);
        return r;
    }

    static List<int> ShortestCycle(List<int> scc, HashSet<int> active)
    {
        var sccSet = new HashSet<int>(scc);
        List<int> best = null;

        foreach (var start in scc)
        {
            var prev = new Dictionary<int, int>();
            var bq = new Queue<int>();
            foreach (var w in Succ[start])
            {
                if (!sccSet.Contains(w)) continue;
                if (w == start) return new List<int> { start };
                if (!prev.ContainsKey(w)) { prev[w] = start; bq.Enqueue(w); }
            }
            bool found = false;
            while (bq.Count > 0 && !found)
            {
                int v = bq.Dequeue();
                foreach (var w in Succ[v])
                {
                    if (!sccSet.Contains(w)) continue;
                    if (w == start)
                    {
                        var cy = new List<int>();
                        int c = v; while (c != start) { cy.Add(c); c = prev[c]; }
                        cy.Add(start); cy.Reverse();
                        if (best == null || cy.Count < best.Count) best = cy;
                        found = true; break;
                    }
                    if (!prev.ContainsKey(w)) { prev[w] = v; bq.Enqueue(w); }
                }
            }
            if (best != null && best.Count <= 2) return best;
        }
        return best;
    }
}
