import struct

def word(val):
    return struct.pack('<I', val & 0xFFFFFFFF)

def encode_string(s):
    b = s.encode('utf-8') + b'\x00'
    while len(b) % 4 != 0:
        b += b'\x00'
    words = []
    for i in range(0, len(b), 4):
        words.append(struct.unpack('<I', b[i:i+4])[0])
    return words

def op(opcode, *operands):
    wc = 1 + len(operands)
    return [((wc) << 16) | opcode] + list(operands)

def float_to_uint(f):
    return struct.unpack('<I', struct.pack('<f', f))[0]
