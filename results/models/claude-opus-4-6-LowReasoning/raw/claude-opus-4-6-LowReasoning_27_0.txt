#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <random>
#include <chrono>
#include <cstring>

using namespace std;

struct Graph {
    int n, m, k;
    vector<vector<int>> adj;
};

// Greedy coloring with DSatur (degree of saturation) for small-medium graphs
vector<int> dsatur(const Graph& g) {
    int n = g.n;
    vector<int> color(n, -1);
    vector<int> degree(n);
    vector<vector<bool>> adjColors(n, vector<bool>(g.k, false));
    vector<int> saturation(n, 0);

    for (int i = 0; i < n; i++) degree[i] = (int)g.adj[i].size();

    for (int step = 0; step < n; step++) {
        // Pick vertex with highest saturation, break ties by degree
        int best = -1;
        for (int i = 0; i < n; i++) {
            if (color[i] != -1) continue;
            if (best == -1 || saturation[i] > saturation[best] ||
                (saturation[i] == saturation[best] && degree[i] > degree[best]))
                best = i;
        }

        // Assign smallest available color
        for (int c = 0; c < g.k; c++) {
            if (!adjColors[best][c]) {
                color[best] = c;
                for (int nb : g.adj[best]) {
                    if (!adjColors[nb][c]) {
                        adjColors[nb][c] = true;
                        saturation[nb]++;
                    }
                }
                break;
            }
        }
        if (color[best] == -1) return {}; // failed
    }
    return color;
}

// Fast greedy for large graphs: order by degree desc, assign first available
vector<int> greedyLargest(const Graph& g) {
    int n = g.n;
    vector<int> order(n);
    iota(order.begin(), order.end(), 0);
    sort(order.begin(), order.end(), [&](int a, int b) {
        return g.adj[a].size() > g.adj[b].size();
    });

    vector<int> color(n, -1);
    vector<bool> used(g.k, false);

    for (int v : order) {
        fill(used.begin(), used.end(), false);
        for (int nb : g.adj[v]) {
            if (color[nb] != -1 && color[nb] < g.k) used[color[nb]] = true;
        }
        for (int c = 0; c < g.k; c++) {
            if (!used[c]) { color[v] = c; break; }
        }
        if (color[v] == -1) return {};
    }
    return color;
}

// Local search: try to fix conflicts via TabuCol-like approach
vector<int> tabuSearch(const Graph& g, vector<int> color, int maxIter) {
    int n = g.n;
    int k = g.k;
    mt19937 rng(42);

    // Count conflicts
    auto countConflicts = [&]() {
        int c = 0;
        for (int i = 0; i < n; i++)
            for (int j : g.adj[i])
                if (j > i && color[i] == color[j]) c++;
        return c;
    };

    // Build conflict count per vertex
    vector<int> conflicts(n, 0);
    for (int i = 0; i < n; i++)
        for (int j : g.adj[i])
            if (color[i] == color[j]) conflicts[i]++;

    int totalConflicts = 0;
    for (int i = 0; i < n; i++) totalConflicts += conflicts[i];
    totalConflicts /= 2;

    if (totalConflicts == 0) return color;

    // Tabu tenure
    int tenure = 10 + n / 1000;
    vector<vector<int>> tabu(n, vector<int>(k, 0));

    for (int iter = 0; iter < maxIter && totalConflicts > 0; iter++) {
        // Find best non-tabu move among conflicting vertices
        int bestV = -1, bestC = -1, bestDelta = n + 1;

        // Sample conflicting vertices
        vector<int> conflicting;
        conflicting.reserve(n);
        for (int i = 0; i < n; i++)
            if (conflicts[i] > 0) conflicting.push_back(i);

        if (conflicting.empty()) break;

        // Sample a subset for large graphs
        int sampleSize = min((int)conflicting.size(), max(50, n / 100));
        if ((int)conflicting.size() > sampleSize)
            shuffle(conflicting.begin(), conflicting.end(), rng);

        for (int idx = 0; idx < sampleSize; idx++) {
            int v = conflicting[idx];
            int oldC = color[v];
            for (int c = 0; c < k; c++) {
                if (c == oldC) continue;
                // Delta = new conflicts - old conflicts for v
                int delta = 0;
                for (int nb : g.adj[v]) {
                    if (color[nb] == oldC) delta--;
                    if (color[nb] == c) delta++;
                }
                if (delta < bestDelta && (tabu[v][c] <= iter || (totalConflicts + delta == 0))) {
                    bestDelta = delta;
                    bestV = v;
                    bestC = c;
                }
            }
        }

        if (bestV == -1) continue;

        int oldC = color[bestV];
        // Update conflicts
        for (int nb : g.adj[bestV]) {
            if (color[nb] == oldC) { conflicts[nb]--; conflicts[bestV]--; totalConflicts--; }
            if (color[nb] == bestC) { conflicts[nb]++; conflicts[bestV]++; totalConflicts++; }
        }
        color[bestV] = bestC;
        tabu[bestV][oldC] = iter + tenure + (rng() % 10);
    }

    if (totalConflicts == 0) return color;
    return {};
}

bool verify(const Graph& g, const vector<int>& color) {
    if ((int)color.size() != g.n) return false;
    for (int i = 0; i < g.n; i++) {
        if (color[i] < 0 || color[i] >= g.k) return false;
        for (int j : g.adj[i])
            if (color[i] == color[j]) return false;
    }
    return true;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    Graph g;
    cin >> g.n >> g.m >> g.k;
    g.adj.resize(g.n);

    for (int i = 0; i < g.m; i++) {
        int u, v;
        cin >> u >> v;
        g.adj[u].push_back(v);
        g.adj[v].push_back(u);
    }

    auto startTime = chrono::steady_clock::now();
    auto elapsed = [&]() {
        return chrono::duration_cast<chrono::seconds>(chrono::steady_clock::now() - startTime).count();
    };

    vector<int> result;

    // Strategy 1: For small graphs or many colors, DSatur works well
    if (g.n <= 50000) {
        result = dsatur(g);
        if (!result.empty() && verify(g, result)) {
            for (int i = 0; i < g.n; i++) cout << result[i] << " \n"[i == g.n - 1];
            return 0;
        }
    }

    // Strategy 2: Greedy largest-first
    result = greedyLargest(g);
    if (!result.empty() && verify(g, result)) {
        for (int i = 0; i < g.n; i++) cout << result[i] << " \n"[i == g.n - 1];
        return 0;
    }

    // Strategy 3: Random greedy + tabu search
    mt19937 rng(123);
    for (int attempt = 0; attempt < 100 && elapsed() < 240; attempt++) {
        // Random initial coloring
        vector<int> color(g.n);

        // Start with greedy but randomized order
        vector<int> order(g.n);
        iota(order.begin(), order.end(), 0);
        shuffle(order.begin(), order.end(), rng);

        // Sort partially by degree
        stable_sort(order.begin(), order.end(), [&](int a, int b) {
            return g.adj[a].size() > g.adj[b].size();
        });

        fill(color.begin(), color.end(), -1);
        vector<bool> used(g.k, false);
        bool ok = true;
        for (int v : order) {
            fill(used.begin(), used.end(), false);
            for (int nb : g.adj[v])
                if (color[nb] != -1) used[color[nb]] = true;
            int chosen = -1;
            for (int c = 0; c < g.k; c++)
                if (!used[c]) { chosen = c; break; }
            if (chosen == -1) {
                // Assign random color, let tabu fix it
                chosen = rng() % g.k;
            }
            color[v] = chosen;
        }

        int maxTabuIter = min(1000000, max(100000, g.n * 10));
        result = tabuSearch(g, color, maxTabuIter);
        if (!result.empty() && verify(g, result)) {
            for (int i = 0; i < g.n; i++) cout << result[i] << " \n"[i == g.n - 1];
            return 0;
        }
    }

    // Fallback: just output greedy even if it might have issues
    // (shouldn't reach here for valid inputs with enough colors)
    result.resize(g.n);
    for (int i = 0; i < g.n; i++) result[i] = i % g.k;
    for (int i = 0; i < g.n; i++) cout << result[i] << " \n"[i == g.n - 1];

    return 0;
}
