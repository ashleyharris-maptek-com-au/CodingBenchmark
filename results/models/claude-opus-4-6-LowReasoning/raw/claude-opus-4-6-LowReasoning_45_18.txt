import struct

words = []

def w(v):
    words.append(v)

def le(v):
    return struct.pack('<I', v & 0xFFFFFFFF)

def encode_string(s):
    b = s.encode('utf-8') + b'\x00'
    while len(b) % 4 != 0:
        b += b'\x00'
    result = []
    for i in range(0, len(b), 4):
        result.append(struct.unpack('<I', b[i:i+4])[0])
    return result

def op(wordcount, opcode, *operands):
    words.append((wordcount << 16) | opcode)
    for o in operands:
        if isinstance(o, list):
            words.extend(o)
        else:
            words.append(o)

def float_bits(f):
    return struct.unpack('<I', struct.pack('<f', f))[0]

# Header
w(0x07230203)  # magic
w(0x00010000)  # version 1.0
w(0)           # generator
w(54)          # bound
w(0)           # reserved

# IDs:
# 1=GLSL ext, 2=void, 3=func_type, 4=float, 5=vec3, 6=vec4, 7=vec2
# 8=mat4, 9=UBO struct, 10=ptr_in_vec3, 11=ptr_in_vec2, 12=ptr_out_vec4
# 13=ptr_uniform_struct, 14=ptr_uniform_vec4, 15=int, 16=bool
# 17=worldPos, 18=normal, 19=uv, 20=tangent, 21=color, 22=outColor, 23=ubo
# 24=0.0, 25=1.0, 26=0.5, 27=2.0, 28=int(4)
# 29=main, 30=label
# 31-53=SSA

# OpCapability Shader
op(2, 17, 1)

# OpExtInstImport %1 "GLSL.std.450"
s = encode_string("GLSL.std.450")
op(2 + len(s), 11, 1, s)

# OpMemoryModel Logical GLSL450
op(3, 14, 0, 1)

# OpEntryPoint Fragment %29 "main" %17 %18 %19 %20 %21 %22
name = encode_string("main")
op(3 + len(name) + 6, 15, 4, 29, name, 17, 18, 19, 20, 21, 22)

# OpExecutionMode %29 OriginUpperLeft
op(3, 16, 29, 7)

# Decorations
# Input locations
for var_id, loc in [(17,0),(18,1),(19,2),(20,3),(21,4)]:
    op(4, 71, var_id, 30, loc)  # Location
# Output location
op(4, 71, 22, 30, 0)
# UBO decorations
op(4, 71, 23, 34, 0)  # DescriptorSet 0
op(4, 71, 23, 33, 0)  # Binding 0
op(3, 71, 9, 2)       # Block

# Member decorations for UBO struct
# Members 0,1,2 are mat4 (need Offset, ColMajor, MatrixStride)
for member, offset in [(0,0),(1,64),(2,128)]:
    op(5, 72, 9, member, 35, offset)   # Offset
    op(4, 72, 9, member, 5)            # ColMajor
    op(5, 72, 9, member, 7, 16)        # MatrixStride 16
# Members 3,4,5 are vec4 (just Offset)
for member, offset in [(3,192),(4,208),(5,224)]:
    op(5, 72, 9, member, 35, offset)

# Types
op(2, 19, 2)              # %2 = OpTypeVoid
op(3, 22, 4, 32)          # %4 = OpTypeFloat 32
op(4, 23, 5, 4, 3)        # %5 = OpTypeVector %4 3
op(4, 23, 6, 4, 4)        # %6 = OpTypeVector %4 4
op(4, 23, 7, 4, 2)        # %7 = OpTypeVector %4 2
op(4, 24, 8, 6, 4)        # %8 = OpTypeMatrix %6 4
op(8, 30, 9, 8, 8, 8, 6, 6, 6)  # %9 = OpTypeStruct
op(3, 33, 3, 2)           # %3 = OpTypeFunction %2
op(4, 21, 15, 32, 1)      # %15 = OpTypeInt 32 1
op(2, 20, 16)             # %16 = OpTypeBool
op(4, 32, 10, 1, 5)       # %10 = ptr Input vec3
op(4, 32, 11, 1, 7)       # %11 = ptr Input vec2
op(4, 32, 12, 3, 6)       # %12 = ptr Output vec4
op(4, 32, 13, 2, 9)       # %13 = ptr Uniform struct
op(4, 32, 14, 2, 6)       # %14 = ptr Uniform vec4

# Constants
op(4, 43, 24, 4, float_bits(0.0))  # %24 = 0.0
op(4, 43, 25, 4, float_bits(1.0))  # %25 = 1.0
op(4, 43, 26, 4, float_bits(0.5))  # %26 = 0.5
op(4, 43, 27, 4, float_bits(2.0))  # %27 = 2.0
op(4, 43, 28, 15, 4)               # %28 = int 4

# Variables
op(4, 59, 10, 17, 1)   # worldPos Input
op(4, 59, 10, 18, 1)   # normal Input
op(4, 59, 11, 19, 1)   # uv Input
op(4, 59, 10, 20, 1)   # tangent Input
op(4, 59, 10, 21, 1)   # color Input
op(4, 59, 12, 22, 3)   # outColor Output
op(4, 59, 13, 23, 2)   # ubo Uniform

# Function
op(5, 54, 2, 29, 0, 3)    # OpFunction
op(2, 248, 30)             # OpLabel

# Load normal and worldPos
op(4, 61, 5, 31, 18)      # %31 = load normal
op(4, 61, 5, 32, 17)      # %32 = load worldPos

# Access cameraPos from UBO (member 4)
op(5, 65, 14, 33, 23, 28) # %33 = OpAccessChain
op(4, 61, 6, 34, 33)      # %34 = load cameraPos

# Extract xyz
op(5, 81, 4, 35, 34, 0)   # %35 = cameraPos.x
op(5, 81, 4, 36, 34, 1)   # %36 = cameraPos.y
op(5, 81, 4, 37, 34, 2)   # %37 = cameraPos.z

# Construct vec3 and compute V
op(6, 80, 5, 38, 35, 36, 37)  # %38 = vec3(cam.xyz)
op(5, 131, 5, 39, 38, 32)     # %39 = cam.xyz - worldPos
op(6, 12, 5, 40, 1, 69, 39)   # %40 = normalize(V)
op(6, 12, 5, 41, 1, 69, 31)   # %41 = normalize(N)

# NdotV
op(5, 148, 4, 42, 41, 40)     # %42 = dot(N, V)
op(7, 12, 4, 43, 1, 40, 42, 24)  # %43 = max(dot, 0.0)  FMax=40

# Compute color components
op(5, 133, 4, 44, 27, 43)     # %44 = 2*NdotV
op(5, 131, 4, 45, 43, 26)     # %45 = NdotV - 0.5
op(5, 133, 4, 46, 27, 45)     # %46 = (NdotV-0.5)*2
op(5, 131, 4, 47, 25, 46)     # %47 = 1 - (NdotV-0.5)*2

# Comparison
op(5, 184, 16, 48, 43, 26)    # %48 = NdotV < 0.5

# Select
op(6, 169, 4, 49, 48, 25, 47) # r: <0.5 ? 1.0 : 1-(NdotV-0.5)*2
op(6, 169, 4, 50, 48, 44, 47) # g: <0.5 ? 2*NdotV : 1-(NdotV-0.5)*2
op(6, 169, 4, 51, 48, 24, 46) # b: <0.5 ? 0.0 : (NdotV-0.5)*2

# Construct output and store
op(7, 80, 6, 52, 49, 50, 51, 25)  # vec4(r,g,b,1)
op(3, 62, 22, 52)                   # store outColor

# Return and end
op(1, 253)       # OpReturn
op(1, 56)        # OpFunctionEnd

# Output as hex
result = b''
for word in words:
    result += struct.pack('<I', word & 0xFFFFFFFF)
print(result.hex())
