declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

interface Task {
  id: number;
  duration: number;
  deps: number[];
  deadline: number;
  weight: number;
}

const machineCount: number = data.machineCount;
const tasks: Task[] = data.tasks;
const n = tasks.length;

const taskMap = new Map<number, Task>();
for (const t of tasks) taskMap.set(t.id, t);

// Build adjacency for dependents and in-degree
const dependents = new Map<number, number[]>();
const inDeg = new Map<number, number>();
for (const t of tasks) {
  inDeg.set(t.id, t.deps.length);
  dependents.set(t.id, []);
}
for (const t of tasks) {
  for (const d of t.deps) {
    dependents.get(d)!.push(t.id);
  }
}

// Compute earliest start times (EST) via topological BFS
const est = new Map<number, number>();
const queue: number[] = [];
for (const t of tasks) {
  if (t.deps.length === 0) {
    est.set(t.id, 0);
    queue.push(t.id);
  }
}

let qi = 0;
while (qi < queue.length) {
  const uid = queue[qi++];
  const uEnd = est.get(uid)! + taskMap.get(uid)!.duration;
  for (const vid of dependents.get(uid)!) {
    const cur = est.get(vid);
    if (cur === undefined || uEnd > cur) est.set(vid, uEnd);
    inDeg.set(vid, inDeg.get(vid)! - 1);
    if (inDeg.get(vid) === 0) queue.push(vid);
  }
}

// Compute latest start times (for slack) - need max makespan first
// Use a simple heuristic: schedule greedily with priority based on
// weighted deadline urgency and critical path length

// Compute longest path from each task to any sink (critical path tail)
const lpCache = new Map<number, number>();
function longestPath(id: number): number {
  if (lpCache.has(id)) return lpCache.get(id)!;
  const t = taskMap.get(id)!;
  let maxChild = 0;
  for (const cid of dependents.get(id)!) {
    maxChild = Math.max(maxChild, longestPath(cid));
  }
  const val = t.duration + maxChild;
  lpCache.set(id, val);
  return val;
}
for (const t of tasks) longestPath(t.id);

// Direct scheduling using priority queue approach
// Priority: higher weight/deadline urgency first, then critical path

// We'll do list scheduling with a priority that combines:
// 1. Deadline slack (deadline - est - duration): lower = more urgent
// 2. Critical path length: longer = more urgent
// 3. Weight: higher = more urgent

const finishTime = new Map<number, number>();
const scheduled = new Map<number, { start: number; machine: number }>();
const remaining = new Map<number, number>(); // remaining deps count

for (const t of tasks) {
  remaining.set(t.id, t.deps.length);
}

// Machine availability times
const machineAvail = new Float64Array(machineCount); // all 0

// Ready set with priority
const ready: number[] = [];
for (const t of tasks) {
  if (t.deps.length === 0) ready.push(t.id);
}

function taskPriority(id: number): number {
  const t = taskMap.get(id)!;
  // Primary: deadline slack (lower = schedule first)
  const slack = t.deadline - est.get(id)! - t.duration;
  // Use composite: prioritize by slack, then by -weight, then by -longestPath
  // Lower score = higher priority
  return slack - t.weight * 0.01 - longestPath(id) * 0.001;
}

function sortReady() {
  ready.sort((a, b) => taskPriority(a) - taskPriority(b));
}

// Event-driven simulation
while (ready.length > 0) {
  sortReady();

  // Find earliest available machine
  // Try to schedule as many ready tasks as possible
  const toSchedule: number[] = [];
  const usedMachines = new Set<number>();

  for (const tid of ready) {
    const t = taskMap.get(tid)!;
    // Earliest this task can start (all deps finished)
    let earliest = est.get(tid)!;
    // But we need actual finish times of deps
    for (const d of t.deps) {
      const ft = finishTime.get(d);
      if (ft !== undefined && ft > earliest) earliest = ft;
    }

    // Find best machine: earliest available that is >= earliest
    let bestMachine = -1;
    let bestStart = Infinity;
    for (let m = 0; m < machineCount; m++) {
      if (usedMachines.has(m)) continue;
      const start = Math.max(machineAvail[m], earliest);
      if (start < bestStart) {
        bestStart = start;
        bestMachine = m;
      }
    }

    if (bestMachine >= 0) {
      scheduled.set(tid, { start: bestStart, machine: bestMachine });
      finishTime.set(tid, bestStart + t.duration);
      machineAvail[bestMachine] = bestStart + t.duration;
      usedMachines.add(bestMachine);
      toSchedule.push(tid);
    }

    if (usedMachines.size >= machineCount) break;
  }

  if (toSchedule.length === 0) {
    // All ready tasks are blocked on machine availability
    // Schedule the first ready task on the earliest-available machine
    const tid = ready[0];
    const t = taskMap.get(tid)!;
    let earliest = 0;
    for (const d of t.deps) {
      const ft = finishTime.get(d);
      if (ft !== undefined && ft > earliest) earliest = ft;
    }
    let bestMachine = 0;
    let bestStart = Math.max(machineAvail[0], earliest);
    for (let m = 1; m < machineCount; m++) {
      const start = Math.max(machineAvail[m], earliest);
      if (start < bestStart) {
        bestStart = start;
        bestMachine = m;
      }
    }
    scheduled.set(tid, { start: bestStart, machine: bestMachine });
    finishTime.set(tid, bestStart + t.duration);
    machineAvail[bestMachine] = bestStart + t.duration;
    toSchedule.push(tid);
  }

  // Remove scheduled from ready, add newly ready
  const scheduledSet = new Set(toSchedule);
  const newReady: number[] = [];
  for (const r of ready) {
    if (!scheduledSet.has(r)) newReady.push(r);
  }

  for (const tid of toSchedule) {
    for (const cid of dependents.get(tid)!) {
      remaining.set(cid, remaining.get(cid)! - 1);
      if (remaining.get(cid) === 0) {
        // Compute actual EST based on finish times
        const ct = taskMap.get(cid)!;
        let e = 0;
        for (const d of ct.deps) {
          const ft = finishTime.get(d)!;
          if (ft > e) e = ft;
        }
        est.set(cid, e);
        newReady.push(cid);
      }
    }
  }

  ready.length = 0;
  ready.push(...newReady);
}

const result: { id: number; start: number; machine: number }[] = [];
for (const t of tasks) {
  const s = scheduled.get(t.id)!;
  result.push({ id: t.id, start: s.start, machine: s.machine });
}

console.log(JSON.stringify({ schedule: result }));
