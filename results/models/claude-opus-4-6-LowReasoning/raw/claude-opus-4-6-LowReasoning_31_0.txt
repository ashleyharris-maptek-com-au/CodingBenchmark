use std::io::{self, Read, Write, BufWriter};
use std::collections::HashMap;

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let stdout = io::stdout();
    let mut out = BufWriter::new(stdout.lock());

    let mut iter = input.split_ascii_whitespace();
    let n: usize = iter.next().unwrap().parse().unwrap();
    let target: i64 = iter.next().unwrap().parse().unwrap();

    let nums: Vec<i64> = (0..n).map(|_| iter.next().unwrap().parse().unwrap()).collect();

    // Edge case: target is 0 -> empty subset
    if target == 0 {
        writeln!(out, "YES").unwrap();
        writeln!(out, "[]").unwrap();
        return;
    }

    if n == 0 {
        writeln!(out, "NO").unwrap();
        return;
    }

    // Check if any single element equals target
    for (i, &v) in nums.iter().enumerate() {
        if v == target {
            writeln!(out, "YES").unwrap();
            writeln!(out, "[{}]", i).unwrap();
            return;
        }
    }

    // Check if sum of all equals target
    let total: i64 = nums.iter().sum();
    if total == target {
        writeln!(out, "YES").unwrap();
        let indices: Vec<String> = (0..n).map(|i| i.to_string()).collect();
        writeln!(out, "[{}]", indices.join(" ")).unwrap();
        return;
    }

    // Separate positives and negatives, filter zeros
    let mut pos_indices: Vec<usize> = Vec::new();
    let mut neg_indices: Vec<usize> = Vec::new();
    let mut zero_indices: Vec<usize> = Vec::new();

    for (i, &v) in nums.iter().enumerate() {
        if v > 0 {
            pos_indices.push(i);
        } else if v < 0 {
            neg_indices.push(i);
        } else {
            zero_indices.push(i);
        }
    }

    // If target > 0, we need positives; if target < 0, we need negatives
    // Try greedy approaches first for large sets

    if n > 5000 {
        // Large set: try greedy/heuristic approaches
        if let Some(result) = greedy_subset_sum(&nums, target, &pos_indices, &neg_indices) {
            writeln!(out, "YES").unwrap();
            let indices: Vec<String> = result.iter().map(|i| i.to_string()).collect();
            writeln!(out, "[{}]", indices.join(" ")).unwrap();
            return;
        }

        // Try random restarts with greedy
        if let Some(result) = random_greedy(&nums, target, 1000) {
            writeln!(out, "YES").unwrap();
            let indices: Vec<String> = result.iter().map(|i| i.to_string()).collect();
            writeln!(out, "[{}]", indices.join(" ")).unwrap();
            return;
        }

        writeln!(out, "NO").unwrap();
        return;
    }

    // For small-medium sets, try meet-in-the-middle if n <= 40
    if n <= 40 {
        if let Some(result) = meet_in_middle(&nums, target) {
            writeln!(out, "YES").unwrap();
            let mut indices: Vec<usize> = result;
            indices.sort();
            let indices: Vec<String> = indices.iter().map(|i| i.to_string()).collect();
            writeln!(out, "[{}]", indices.join(" ")).unwrap();
            return;
        }
        writeln!(out, "NO").unwrap();
        return;
    }

    // Medium sets: DP with hash map if range is manageable
    // Check if all values are non-negative and target is positive
    let all_non_neg = nums.iter().all(|&v| v >= 0);
    if all_non_neg && target > 0 {
        let sum_pos: i64 = nums.iter().filter(|&&v| v > 0).sum();
        if target > sum_pos {
            writeln!(out, "NO").unwrap();
            return;
        }

        // If target fits in reasonable DP range, use bitset DP
        if target <= 50_000_000 {
            if let Some(result) = bitset_dp(&nums, target) {
                writeln!(out, "YES").unwrap();
                let indices: Vec<String> = result.iter().map(|i| i.to_string()).collect();
                writeln!(out, "[{}]", indices.join(" ")).unwrap();
                return;
            }
            writeln!(out, "NO").unwrap();
            return;
        }
    }

    // Hashmap DP for medium sets with arbitrary values
    if let Some(result) = hashmap_dp(&nums, target) {
        writeln!(out, "YES").unwrap();
        let indices: Vec<String> = result.iter().map(|i| i.to_string()).collect();
        writeln!(out, "[{}]", indices.join(" ")).unwrap();
        return;
    }

    writeln!(out, "NO").unwrap();
}

fn meet_in_middle(nums: &[i64], target: i64) -> Option<Vec<usize>> {
    let n = nums.len();
    let half = n / 2;

    let mut left: HashMap<i64, Vec<usize>> = HashMap::new();
    let left_n = half;
    for mask in 0u64..(1u64 << left_n) {
        let mut sum: i64 = 0;
        let mut indices = Vec::new();
        for i in 0..left_n {
            if mask & (1u64 << i) != 0 {
                sum += nums[i];
                indices.push(i);
            }
        }
        left.entry(sum).or_insert(indices);
    }

    let right_n = n - half;
    for mask in 0u64..(1u64 << right_n) {
        let mut sum: i64 = 0;
        let mut indices = Vec::new();
        for i in 0..right_n {
            if mask & (1u64 << i) != 0 {
                sum += nums[half + i];
                indices.push(half + i);
            }
        }
        let need = target - sum;
        if let Some(left_indices) = left.get(&need) {
            if left_indices.is_empty() && indices.is_empty() {
                continue; // skip empty subset (target=0 handled earlier)
            }
            let mut result = left_indices.clone();
            result.extend(indices);
            return Some(result);
        }
    }
    None
}

fn bitset_dp(nums: &[i64], target: i64) -> Option<Vec<usize>> {
    let t = target as usize;
    let words = (t + 64) / 64;
    let mut dp = vec![0u64; words];
    dp[0] = 1; // sum 0 is reachable

    // Track which item achieved each sum
    let mut parent: Vec<Vec<i32>> = vec![vec![-1i32; 64]; words];

    for (idx, &val) in nums.iter().enumerate() {
        if val <= 0 || val as usize > t {
            continue;
        }
        let v = val as usize;
        let word_shift = v / 64;
        let bit_shift = v % 64;

        // Process from high to low to avoid using same element twice
        if bit_shift == 0 {
            for w in (word_shift..words).rev() {
                let prev = dp[w - word_shift];
                let new_bits = prev & !dp[w];
                if new_bits != 0 {
                    dp[w] |= new_bits;
                    for b in 0..64 {
                        if new_bits & (1u64 << b) != 0 {
                            parent[w][b] = idx as i32;
                        }
                    }
                }
            }
        } else {
            for w in (0..words).rev() {
                let mut prev = 0u64;
                if w >= word_shift + 1 {
                    prev = dp[w - word_shift - 1] >> (64 - bit_shift);
                }
                if w >= word_shift {
                    prev |= dp[w - word_shift] << bit_shift;
                }
                let new_bits = prev & !dp[w];
                if new_bits != 0 {
                    dp[w] |= new_bits;
                    for b in 0..64 {
                        if new_bits & (1u64 << b) != 0 {
                            parent[w][b] = idx as i32;
                        }
                    }
                }
            }
        }

        // Early exit
        let tw = t / 64;
        let tb = t % 64;
        if tw < words && dp[tw] & (1u64 << tb) != 0 {
            // Reconstruct
            return Some(reconstruct_bitset(&nums, &parent, t));
        }
    }

    let tw = t / 64;
    let tb = t % 64;
    if tw < words && dp[tw] & (1u64 << tb) != 0 {
        return Some(reconstruct_bitset(&nums, &parent, t));
    }
    None
}

fn reconstruct_bitset(nums: &[i64], parent: &[Vec<i32>], mut remaining: usize) -> Vec<usize> {
    let mut result = Vec::new();
    while remaining > 0 {
        let w = remaining / 64;
        let b = remaining % 64;
        let idx = parent[w][b];
        if idx < 0 {
            break;
        }
        result.push(idx as usize);
        remaining -= nums[idx as usize] as usize;
    }
    result.sort();
    result
}

fn hashmap_dp(nums: &[i64], target: i64) -> Option<Vec<usize>> {
    // DP with HashMap, limit size
    let max_states = 2_000_000usize;
    let mut dp: HashMap<i64, Vec<usize>> = HashMap::new();
    dp.insert(0, Vec::new());

    for (idx, &val) in nums.iter().enumerate() {
        if val == 0 {
            continue;
        }
        let entries: Vec<(i64, Vec<usize>)> = dp.iter()
            .map(|(k, v)| (*k, v.clone()))
            .collect();

        for (sum, indices) in entries {
            let new_sum = sum + val;
            if new_sum == target {
                let mut result = indices;
                result.push(idx);
                result.sort();
                return Some(result);
            }
            if !dp.contains_key(&new_sum) && dp.len() < max_states {
                let mut new_indices = indices;
                new_indices.push(idx);
                dp.insert(new_sum, new_indices);
            }
        }
    }
    None
}

fn greedy_subset_sum(nums: &[i64], target: i64, pos_indices: &[usize], neg_indices: &[usize]) -> Option<Vec<usize>> {
    // Sort by value descending (for positive target) or ascending (for negative)
    if target > 0 {
        let mut sorted: Vec<(i64, usize)> = pos_indices.iter().map(|&i| (nums[i], i)).collect();
        sorted.sort_by(|a, b| b.0.cmp(&a.0));

        let mut sum = 0i64;
        let mut result = Vec::new();
        for (val, idx) in &sorted {
            if sum + val <= target {
                sum += val;
                result.push(*idx);
                if sum == target {
                    result.sort();
                    return Some(result);
                }
            }
        }

        // Try also including negatives to adjust
        let mut neg_sorted: Vec<(i64, usize)> = neg_indices.iter().map(|&i| (nums[i], i)).collect();
        neg_sorted.sort_by(|a, b| a.0.cmp(&b.0));
        for (val, idx) in &neg_sorted {
            if sum + val >= target {
                sum += val;
                result.push(*idx);
                if sum == target {
                    result.sort();
                    return Some(result);
                }
            }
        }
    } else {
        let mut sorted: Vec<(i64, usize)> = neg_indices.iter().map(|&i| (nums[i], i)).collect();
        sorted.sort_by(|a, b| a.0.cmp(&b.0));

        let mut sum = 0i64;
        let mut result = Vec::new();
        for (val, idx) in &sorted {
            if sum + val >= target {
                sum += val;
                result.push(*idx);
                if sum == target {
                    result.sort();
                    return Some(result);
                }
            }
        }

        let mut pos_sorted: Vec<(i64, usize)> = pos_indices.iter().map(|&i| (nums[i], i)).collect();
        pos_sorted.sort_by(|a, b| b.0.cmp(&a.0));
        for (val, idx) in &pos_sorted {
            if sum + val <= target {
                sum += val;
                result.push(*idx);
                if sum == target {
                    result.sort();
                    return Some(result);
                }
            }
        }
    }
    None
}

fn random_greedy(nums: &[i64], target: i64, iterations: usize) -> Option<Vec<usize>> {
    let n = nums.len();
    if n == 0 {
        return None;
    }

    // Simple LCG PRNG
    let mut seed: u64 = 12345678901u64;
    let lcg_next = |s: &mut u64| -> u64 {
        *s = s.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
        *s
    };

    for _ in 0..iterations {
        // Create a random permutation (Fisher-Yates with our PRNG)
        let mut order: Vec<usize> = (0..n).collect();
        for i in (1..n).rev() {
            let j = (lcg_next(&mut seed) as usize) % (i + 1);
            order.swap(i, j);
        }

        let mut sum = 0i64;
        let mut result = Vec::new();

        for &idx in &order {
            let val = nums[idx];
            let new_sum = sum + val;
            // Include if it moves us closer to target or hits it
            if target > 0 {
                if val > 0 && new_sum <= target {
                    sum = new_sum;
                    result.push(idx);
                } else if val < 0 && sum > target {
                    sum = new_sum;
                    result.push(idx);
                }
            } else {
                if val < 0 && new_sum >= target {
                    sum = new_sum;
                    result.push(idx);
                } else if val > 0 && sum < target {
                    sum = new_sum;
                    result.push(idx);
                }
            }
            if sum == target {
                result.sort();
                return Some(result);
            }
        }
    }
    None
}
