using System;
using System.Collections.Generic;
using System.IO;
using System.Text;

class Program
{
    static void Main()
    {
        var header = Console.ReadLine().Split(' ');
        int n = int.Parse(header[0]);
        int stockLength = int.Parse(header[1]);

        if (n == 0)
        {
            Console.WriteLine("0");
            return;
        }

        var parts = Console.ReadLine().Split(' ');
        int[] cuts = new int[n];
        int[] sortedIndices = new int[n];
        for (int i = 0; i < n; i++)
        {
            cuts[i] = int.Parse(parts[i]);
            sortedIndices[i] = i;
        }

        // Sort indices by cut length descending (First Fit Decreasing)
        Array.Sort(sortedIndices, (a, b) => cuts[b].CompareTo(cuts[a]));

        // FFD bin packing
        var stocks = new List<(int remaining, List<int> indices)>();

        for (int i = 0; i < n; i++)
        {
            int idx = sortedIndices[i];
            int cutLen = cuts[idx];

            // Find best fit (smallest remaining that still fits)
            int bestJ = -1;
            int bestRem = int.MaxValue;
            for (int j = 0; j < stocks.Count; j++)
            {
                int rem = stocks[j].remaining;
                if (rem >= cutLen && rem < bestRem)
                {
                    bestRem = rem;
                    bestJ = j;
                    if (rem == cutLen) break; // perfect fit
                }
            }

            if (bestJ >= 0)
            {
                var s = stocks[bestJ];
                s.indices.Add(idx);
                s.remaining -= cutLen;
                stocks[bestJ] = s;
            }
            else
            {
                var list = new List<int> { idx };
                stocks.Add((stockLength - cutLen, list));
            }
        }

        var sb = new StringBuilder();
        sb.AppendLine(stocks.Count.ToString());
        for (int i = 0; i < stocks.Count; i++)
        {
            var indices = stocks[i].indices;
            for (int j = 0; j < indices.Count; j++)
            {
                if (j > 0) sb.Append(' ');
                sb.Append(indices[j]);
            }
            sb.AppendLine();
        }
        Console.Write(sb);
    }
}
