#pragma comment(linker, "/STACK:1000000000")
#include <cstdio>
#include <vector>
#include <queue>
#include <algorithm>
#include <random>
#include <climits>
#include <functional>

using namespace std;

static char ibuf[1 << 22];
static int ipos = 0, ilen = 0;
inline int rc() {
    if (ipos == ilen) { ilen = (int)fread(ibuf, 1, sizeof(ibuf), stdin); ipos = 0; }
    return ipos < ilen ? (unsigned char)ibuf[ipos++] : -1;
}
inline int ri() {
    int c, x = 0;
    while ((c = rc()) <= ' ') if (c < 0) return -1;
    while (c >= '0' && c <= '9') { x = x * 10 + c - '0'; c = rc(); }
    return x;
}

static int N, M;
static vector<pair<int,int>> edges;
static vector<vector<pair<int,int>>> adj;
static vector<int> deg;

struct Edge { int to, rev, cap; };
static vector<vector<Edge>> g;
static vector<int> lev, it;

void build_flow() {
    g.assign(N, {});
    for (auto& [u, v] : edges) {
        int xu = (int)g[u].size(), xv = (int)g[v].size();
        g[u].push_back({v, xv, 1});
        g[v].push_back({u, xu, 1});
    }
}

void reset_flow() {
    for (auto& a : g) for (auto& e : a) e.cap = 1;
}

bool bfs(int s, int t) {
    fill(lev.begin(), lev.end(), -1);
    queue<int> q;
    lev[s] = 0; q.push(s);
    while (!q.empty()) {
        int v = q.front(); q.pop();
        for (auto& e : g[v])
            if (e.cap > 0 && lev[e.to] < 0) { lev[e.to] = lev[v]+1; q.push(e.to); }
    }
    return lev[t] >= 0;
}

int dfs(int v, int t, int f) {
    if (v == t) return f;
    for (int& i = it[v]; i < (int)g[v].size(); i++) {
        auto& e = g[v][i];
        if (e.cap > 0 && lev[v] < lev[e.to]) {
            int d = dfs(e.to, t, min(f, e.cap));
            if (d > 0) { e.cap -= d; g[e.to][e.rev].cap += d; return d; }
        }
    }
    return 0;
}

int maxflow(int s, int t, int limit) {
    int flow = 0;
    while (flow < limit && bfs(s, t)) {
        fill(it.begin(), it.end(), 0);
        int d;
        while (flow < limit && (d = dfs(s, t, limit - flow)) > 0) flow += d;
    }
    return flow;
}

vector<bool> cut_side(int s) {
    vector<bool> vis(N, false);
    queue<int> q;
    vis[s] = true; q.push(s);
    while (!q.empty()) {
        int v = q.front(); q.pop();
        for (auto& e : g[v])
            if (e.cap > 0 && !vis[e.to]) { vis[e.to] = true; q.push(e.to); }
    }
    return vis;
}

int main() {
    N = ri(); M = ri();
    if (N <= 1) { puts("0"); return 0; }
    if (M == 0) { puts("0"); return 0; }

    edges.resize(M);
    adj.resize(N);
    deg.assign(N, 0);
    for (int i = 0; i < M; i++) {
        int u = ri(), v = ri();
        edges[i] = {u, v};
        adj[u].push_back({v, i});
        adj[v].push_back({u, i});
        deg[u]++; deg[v]++;
    }

    // Bridge detection (iterative Tarjan's)
    {
        vector<int> disc(N, -1), low(N);
        int tmr = 0;
        struct F { int u, pe, ci; };
        vector<F> stk;
        stk.reserve(N);
        disc[0] = low[0] = tmr++;
        stk.push_back({0, -1, 0});
        int br_u = -1, br_v = -1;

        while (!stk.empty()) {
            auto& f = stk.back();
            if (f.ci < (int)adj[f.u].size()) {
                auto [v, eidx] = adj[f.u][f.ci++];
                if (eidx == f.pe) continue;
                if (disc[v] == -1) {
                    disc[v] = low[v] = tmr++;
                    stk.push_back({v, eidx, 0});
                } else {
                    low[f.u] = min(low[f.u], disc[v]);
                }
            } else {
                int cu = f.u;
                stk.pop_back();
                if (!stk.empty()) {
                    low[stk.back().u] = min(low[stk.back().u], low[cu]);
                    if (low[cu] > disc[stk.back().u]) {
                        br_u = stk.back().u;
                        br_v = cu;
                    }
                }
            }
        }

        if (br_u != -1) {
            printf("1\n%d %d\n", br_u, br_v);
            return 0;
        }
    }

    // Find min-degree vertex
    int s = 0;
    for (int i = 1; i < N; i++)
        if (deg[i] < deg[s]) s = i;
    int best = deg[s];
    vector<bool> best_side;

    build_flow();
    lev.resize(N);
    it.resize(N);

    // Build target list: neighbors of s first, then random others
    vector<int> nbrs, others;
    {
        vector<bool> is_nbr(N, false);
        for (auto& [v, _] : adj[s]) is_nbr[v] = true;
        for (int i = 0; i < N; i++) {
            if (i == s) continue;
            if (is_nbr[i]) nbrs.push_back(i);
            else others.push_back(i);
        }
    }

    int max_targets;
    long long nm = (long long)N * M;
    if (nm <= 200000000LL) max_targets = N;
    else if (nm <= 2000000000LL) max_targets = 3000;
    else max_targets = 800;

    vector<int> targets = nbrs;
    mt19937 rng(42);
    shuffle(others.begin(), others.end(), rng);
    int extra = min((int)others.size(), max(max_targets - (int)nbrs.size(), 0));
    for (int i = 0; i < extra; i++) targets.push_back(others[i]);

    for (int t : targets) {
        if (best <= 1) break;
        reset_flow();
        int f = maxflow(s, t, best);
        if (f < best) {
            best = f;
            best_side = cut_side(s);
        }
    }

    if (best_side.empty()) {
        best_side.assign(N, true);
        best_side[s] = false;
    }

    vector<pair<int,int>> cut_edges;
    for (auto& [u, v] : edges)
        if (best_side[u] != best_side[v])
            cut_edges.push_back({u, v});

    printf("%d\n", (int)cut_edges.size());
    for (auto& [u, v] : cut_edges)
        printf("%d %d\n", u, v);

    return 0;
}
