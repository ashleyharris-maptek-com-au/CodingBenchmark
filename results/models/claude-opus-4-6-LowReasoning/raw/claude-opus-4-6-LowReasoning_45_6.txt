import struct

words = []

def w(val):
    words.append(val)

def op(opcode, *operands):
    wc = 1 + len(operands)
    words.append((wc << 16) | opcode)
    for o in operands:
        words.append(o)

def op_str(opcode, *operands_before_str, string, operands_after=None):
    if operands_after is None:
        operands_after = []
    encoded = string.encode('utf-8') + b'\x00'
    while len(encoded) % 4 != 0:
        encoded += b'\x00'
    str_words = len(encoded) // 4
    wc = 1 + len(operands_before_str) + str_words + len(operands_after)
    words.append((wc << 16) | opcode)
    for o in operands_before_str:
        words.append(o)
    for i in range(str_words):
        words.append(struct.unpack('<I', encoded[i*4:i*4+4])[0])
    for o in operands_after:
        words.append(o)

# IDs
ID = 0
def nid():
    global ID
    ID += 1
    return ID

id_glsl = nid()       # 1
id_void = nid()       # 2
id_func_type = nid()  # 3
id_float = nid()      # 4
id_vec2 = nid()       # 5
id_vec3 = nid()       # 6
id_vec4 = nid()       # 7
id_mat4 = nid()       # 8

id_ptr_input_vec3 = nid()   # 9
id_ptr_input_vec2 = nid()   # 10
id_ptr_output_vec4 = nid()  # 11

id_in_worldPos = nid()  # 12
id_in_normal = nid()    # 13
id_in_uv = nid()        # 14
id_in_tangent = nid()   # 15
id_in_color = nid()     # 16
id_out_color = nid()    # 17

# UBO struct type and variable
id_ubo_struct = nid()      # 18
id_ptr_uniform_ubo = nid() # 19
id_ubo_var = nid()         # 20

id_ptr_uniform_vec4 = nid() # 21
id_ptr_uniform_mat4 = nid() # 22

id_main = nid()         # 23
id_label = nid()        # 24

# Constants
id_const_0 = nid()      # 25  float 0.0
id_const_1 = nid()      # 26  float 1.0
id_const_01 = nid()     # 27  float 0.1
id_const_06 = nid()     # 28  float 0.6
id_const_08 = nid()     # 29  float 0.8
id_const_64 = nid()     # 30  float 64.0

id_int = nid()           # 31
id_const_int_4 = nid()  # 32  int 4 (index for lightPos)
id_const_int_5 = nid()  # 33  int 5 (index for cameraPos - wait, need member indices)

# Actually for AccessChain into UBO struct, we need int constants for member indices
# UBO struct members: mat4 model(0), mat4 view(1), mat4 projection(2), vec4 lightPos(3), vec4 cameraPos(4), vec4 params(5)
id_const_int_0 = nid()  # 34
id_const_int_1 = nid()  # 35
id_const_int_2 = nid()  # 36
id_const_int_3 = nid()  # 37
id_const_int_4v = nid() # 38
id_const_int_5v = nid() # 39

# SSA IDs for operations
id_load_worldPos = nid()  # 40
id_load_normal = nid()    # 41
id_ac_lightPos = nid()    # 42
id_load_lightPos = nid()  # 43
id_ac_cameraPos = nid()   # 44
id_load_cameraPos = nid() # 45

id_lightXYZ = nid()     # 46
id_camXYZ = nid()       # 47
id_L_unnorm = nid()     # 48
id_V_unnorm = nid()     # 49
id_N = nid()            # 50
id_L = nid()            # 51
id_V = nid()            # 52
id_H_unnorm = nid()     # 53
id_H = nid()            # 54

id_dot_NL = nid()       # 55
id_diffuse = nid()      # 56
id_dot_NH = nid()       # 57
id_spec_base = nid()    # 58
id_specular = nid()     # 59

id_diff_scaled = nid()  # 60
id_spec_scaled = nid()  # 61
id_sum1 = nid()         # 62
id_intensity = nid()    # 63
id_result = nid()       # 64

BOUND = 65

# Header
w(0x07230203)  # magic
w(0x00010000)  # version 1.0
w(0)           # generator
w(BOUND)       # bound
w(0)           # reserved

# OpCapability Shader
op(17, 1)

# OpExtInstImport %1 "GLSL.std.450"
op_str(11, id_glsl, string="GLSL.std.450")

# OpMemoryModel Logical GLSL450
op(14, 0, 1)

# OpEntryPoint Fragment %main "main" %in_worldPos %in_normal %in_uv %in_tangent %in_color %out_color
op_str(15, 4, id_main, string="main", operands_after=[id_in_worldPos, id_in_normal, id_in_uv, id_in_tangent, id_in_color, id_out_color])

# OpExecutionMode %main OriginUpperLeft
op(16, id_main, 7)

# Decorations for inputs
# Location decorations
for var_id, loc in [(id_in_worldPos, 0), (id_in_normal, 1), (id_in_uv, 2), (id_in_tangent, 3), (id_in_color, 4)]:
    op(71, var_id, 30, loc)  # OpDecorate target Location loc

# Output decoration
op(71, id_out_color, 30, 0)  # Location 0

# UBO decorations
op(71, id_ubo_var, 34, 0)  # DescriptorSet 0
op(71, id_ubo_var, 33, 0)  # Binding 0
op(71, id_ubo_struct, 2)   # Block (word_count=3)

# Member decorations for UBO struct (OpMemberDecorate = opcode 72)
# mat4 model offset 0, mat4 view offset 64, mat4 proj offset 128, vec4 lightPos offset 192, vec4 cameraPos offset 208, vec4 params offset 224
for member, offset in [(0, 0), (1, 64), (2, 128), (3, 192), (4, 208), (5, 224)]:
    # OpMemberDecorate struct member Offset offset
    op(72, id_ubo_struct, member, 35, offset)

# ColMajor and MatrixStride for mat4 members
for member in [0, 1, 2]:
    op(72, id_ubo_struct, member, 5)  # ColMajor (decoration 5), wc=4
    op(72, id_ubo_struct, member, 7, 16)  # MatrixStride 16 (decoration 7)

# Type declarations
op(19, id_void)                          # OpTypeVoid
op(22, id_float, 32)                     # OpTypeFloat 32
op(23, id_vec2, id_float, 2)             # OpTypeVector float 2
op(23, id_vec3, id_float, 3)             # OpTypeVector float 3
op(23, id_vec4, id_float, 4)             # OpTypeVector float 4
op(24, id_mat4, id_vec4, 4)              # OpTypeMatrix vec4 4

op(33, id_func_type, id_void)            # OpTypeFunction void

# UBO struct: { mat4, mat4, mat4, vec4, vec4, vec4 }
op(30, id_ubo_struct, id_mat4, id_mat4, id_mat4, id_vec4, id_vec4, id_vec4)

# Pointer types
op(32, id_ptr_input_vec3, 1, id_vec3)    # Input
op(32, id_ptr_input_vec2, 1, id_vec2)    # Input
op(32, id_ptr_output_vec4, 3, id_vec4)   # Output
op(32, id_ptr_uniform_vec4, 2, id_vec4)  # Uniform
op(32, id_ptr_uniform_mat4, 2, id_mat4)  # Uniform
op(32, id_ptr_uniform_ubo, 2, id_ubo_struct) # Uniform

# Variables
# Storage class: Input=1, Output=3, Uniform=2, UniformConstant=0
op(59, id_ptr_input_vec3, id_in_worldPos, 1)   # Input vec3
op(59, id_ptr_input_vec3, id_in_normal, 1)
op(59, id_ptr_input_vec2, id_in_uv, 1)
op(59, id_ptr_input_vec3, id_in_tangent, 1)
op(59, id_ptr_input_vec3, id_in_color, 1)
op(59, id_ptr_output_vec4, id_out_color, 3)    # Output vec4
op(59, id_ptr_uniform_ubo, id_ubo_var, 2)     # Uniform UBO

# Int type for indices
op(21, id_int, 32, 1)  # OpTypeInt 32 signed

# Constants
def float_to_uint(f):
    return struct.unpack('<I', struct.pack('<f', f))[0]

op(43, id_float, id_const_0, float_to_uint(0.0))
op(43, id_float, id_const_1, float_to_uint(1.0))
op(43, id_float, id_const_01, float_to_uint(0.1))
op(43, id_float, id_const_06, float_to_uint(0.6))
op(43, id_float, id_const_08, float_to_uint(0.8))
op(43, id_float, id_const_64, float_to_uint(64.0))

op(43, id_int, id_const_int_0, 0)
op(43, id_int, id_const_int_1, 1)
op(43, id_int, id_const_int_2, 2)
op(43, id_int, id_const_int_3, 3)
op(43, id_int, id_const_int_4v, 4)
op(43, id_int, id_const_int_5v, 5)

# Function
op(54, id_void, id_main, 0, id_func_type)  # OpFunction void None func_type
# FunctionControl = 0 (None)

op(248, id_label)  # OpLabel

# Load inputs
op(61, id_vec3, id_load_worldPos, id_in_worldPos)
op(61, id_vec3, id_load_normal, id_in_normal)

# Access chain for lightPos (member 3) and cameraPos (member 4)
# OpAccessChain opcode=65
op(65, id_ptr_uniform_vec4, id_ac_lightPos, id_ubo_var, id_const_int_3)
op(61, id_vec4, id_load_lightPos, id_ac_lightPos)

op(65, id_ptr_uniform_vec4, id_ac_cameraPos, id_ubo_var, id_const_int_4v)
op(61, id_vec4, id_load_cameraPos, id_ac_cameraPos)

# Extract xyz from vec4 lightPos and cameraPos -> vec3
# OpVectorShuffle opcode=79: result_type result_id vec1 vec2 components...
op(79, id_vec3, id_lightXYZ, id_load_lightPos, id_load_lightPos, 0, 1, 2)
op(79, id_vec3, id_camXYZ, id_load_cameraPos, id_load_cameraPos, 0, 1, 2)

# L_unnorm = lightXYZ - worldPos
op(131, id_vec3, id_L_unnorm, id_lightXYZ, id_load_worldPos)
# V_unnorm = camXYZ - worldPos
op(131, id_vec3, id_V_unnorm, id_camXYZ, id_load_worldPos)

# Normalize N, L, V using GLSL.std.450 Normalize (ext inst 69)
op(12, id_vec3, id_N, id_glsl, 69, id_load_normal)
op(12, id_vec3, id_L, id_glsl, 69, id_L_unnorm)
op(12, id_vec3, id_V, id_glsl, 69, id_V_unnorm)

# H_unnorm = L + V
op(129, id_vec3, id_H_unnorm, id_L, id_V)
# H = normalize(H_unnorm)
op(12, id_vec3, id_H, id_glsl, 69, id_H_unnorm)

# dot(N, L)
op(148, id_float, id_dot_NL, id_N, id_L)
# diffuse = max(dot_NL, 0.0) using GLSL FMax (ext inst 40)
op(12, id_float, id_diffuse, id_glsl, 40, id_dot_NL, id_const_0)

# dot(N, H)
op(148, id_float, id_dot_NH, id_N, id_H)
# spec_base = max(dot_NH, 0.0)
op(12, id_float, id_spec_base, id_glsl, 40, id_dot_NH, id_const_0)
# specular = pow(spec_base, 64.0) using GLSL Pow (ext inst 26)
op(12, id_float, id_specular, id_glsl, 26, id_spec_base, id_const_64)

# intensity = 0.1 + 0.6*diffuse + 0.8*specular
op(133, id_float, id_diff_scaled, id_const_06, id_diffuse)
op(133, id_float, id_spec_scaled, id_const_08, id_specular)
op(129, id_float, id_sum1, id_const_01, id_diff_scaled)
op(129, id_float, id_intensity, id_sum1, id_spec_scaled)

# result = vec4(intensity, intensity, intensity, 1.0)
op(80, id_vec4, id_result, id_intensity, id_intensity, id_intensity, id_const_1)

# Store result
op(62, id_out_color, id_result)

# Return
op(253)

# OpFunctionEnd
op(56)

# Output
data = b''
for word in words:
    data += struct.pack('<I', word)

print(data.hex())
