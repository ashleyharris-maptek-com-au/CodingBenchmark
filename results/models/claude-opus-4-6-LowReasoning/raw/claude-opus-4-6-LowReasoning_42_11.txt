; SPIR-V Geometry Shader: Dashed thick curve with adjacency
               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %inColor %outColor
               OpExecutionMode %main InputLinesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 8
               OpExecutionMode %main Invocations 1

               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %inColor Location 2
               OpDecorate %outColor Location 0

      %void  = OpTypeVoid
    %voidfn  = OpTypeFunction %void
     %float  = OpTypeFloat 32
   %v2float  = OpTypeVector %float 2
   %v3float  = OpTypeVector %float 3
   %v4float  = OpTypeVector %float 4
       %int  = OpTypeInt 32 1
      %uint  = OpTypeInt 32 0

     %int_0  = OpConstant %int 0
     %int_1  = OpConstant %int 1
     %int_2  = OpConstant %int 2
     %int_3  = OpConstant %int 3
    %uint_4  = OpConstant %uint 4
   %float_0  = OpConstant %float 0.0
   %float_1  = OpConstant %float 1.0
 %float_1o3  = OpConstant %float 0.333333343
 %float_2o3  = OpConstant %float 0.666666687
%float_thick = OpConstant %float 0.02

%gl_PerVertex_in = OpTypeStruct %v4float
  %arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
 %ptr_in_arr = OpTypePointer Input %arr_pv_in
  %ptr_in_v4 = OpTypePointer Input %v4float
      %gl_in = OpVariable %ptr_in_arr Input

   %arr_v3_4 = OpTypeArray %v3float %uint_4
%ptr_in_arr3 = OpTypePointer Input %arr_v3_4
  %ptr_in_v3 = OpTypePointer Input %v3float
    %inColor = OpVariable %ptr_in_arr3 Input

%gl_PerVertex_out = OpTypeStruct %v4float
 %ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
 %ptr_out_v4 = OpTypePointer Output %v4float
     %gl_out = OpVariable %ptr_out_pv Output

 %ptr_out_v3 = OpTypePointer Output %v3float
   %outColor = OpVariable %ptr_out_v3 Output

      %main  = OpFunction %void None %voidfn
     %entry  = OpLabel

; Load gl_in positions
    %ac_p0   = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
       %p0   = OpLoad %v4float %ac_p0
    %ac_p1   = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
       %p1   = OpLoad %v4float %ac_p1
    %ac_p2   = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
       %p2   = OpLoad %v4float %ac_p2
    %ac_p3   = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
       %p3   = OpLoad %v4float %ac_p3

; Load colors
   %ac_c1    = OpAccessChain %ptr_in_v3 %inColor %int_1
      %c1    = OpLoad %v3float %ac_c1
   %ac_c2    = OpAccessChain %ptr_in_v3 %inColor %int_2
      %c2    = OpLoad %v3float %ac_c2

; Extract xy
     %p0x    = OpCompositeExtract %float %p0 0
     %p0y    = OpCompositeExtract %float %p0 1
     %p1x    = OpCompositeExtract %float %p1 0
     %p1y    = OpCompositeExtract %float %p1 1
     %p1z    = OpCompositeExtract %float %p1 2
     %p1w    = OpCompositeExtract %float %p1 3
     %p2x    = OpCompositeExtract %float %p2 0
     %p2y    = OpCompositeExtract %float %p2 1
     %p2z    = OpCompositeExtract %float %p2 2
     %p2w    = OpCompositeExtract %float %p2 3
     %p3x    = OpCompositeExtract %float %p3 0
     %p3y    = OpCompositeExtract %float %p3 1

; Tangent at p1: p2 - p0
   %t1x_raw  = OpFSub %float %p2x %p0x
   %t1y_raw  = OpFSub %float %p2y %p0y
    %t1_v2   = OpCompositeConstruct %v2float %t1x_raw %t1y_raw
   %t1_len   = OpExtInst %float %1 Length %t1_v2
    %t1x     = OpFDiv %float %t1x_raw %t1_len
    %t1y     = OpFDiv %float %t1y_raw %t1_len

; Tangent at p2: p3 - p1
   %t2x_raw  = OpFSub %float %p3x %p1x
   %t2y_raw  = OpFSub %float %p3y %p1y
    %t2_v2   = OpCompositeConstruct %v2float %t2x_raw %t2y_raw
   %t2_len   = OpExtInst %float %1 Length %t2_v2
    %t2x     = OpFDiv %float %t2x_raw %t2_len
    %t2y     = OpFDiv %float %t2y_raw %t2_len

; Perpendiculars (rotated 90): perp = (-ty, tx) * thickness
  %n_t1y     = OpFNegate %float %t1y
  %perp1x    = OpFMul %float %n_t1y %float_thick
  %perp1y    = OpFMul %float %t1x %float_thick

  %n_t2y     = OpFNegate %float %t2y
  %perp2x    = OpFMul %float %n_t2y %float_thick
  %perp2y    = OpFMul %float %t2x %float_thick

; Segment direction
  %segx      = OpFSub %float %p2x %p1x
  %segy      = OpFSub %float %p2y %p1y
  %segz      = OpFSub %float %p2z %p1z
  %segw      = OpFSub %float %p2w %p1w

; Color difference
  %cdiff     = OpFSub %v3float %c2 %c1

; === Interpolation points along segment ===
; t=0 (p1), t=1/3, t=2/3, t=1 (p2)
; Dash 1: t=0 to t=1/3
; Dash 2: t=2/3 to t=1

; Point at t=1/3: lerp(p1, p2, 1/3)
  %ax_13     = OpFMul %float %segx %float_1o3
  %ay_13     = OpFMul %float %segy %float_1o3
  %az_13     = OpFMul %float %segz %float_1o3
  %aw_13     = OpFMul %float %segw %float_1o3
  %px_13     = OpFAdd %float %p1x %ax_13
  %py_13     = OpFAdd %float %p1y %ay_13
  %pz_13     = OpFAdd %float %p1z %az_13
  %pw_13     = OpFAdd %float %p1w %aw_13

; Perp at t=1/3: lerp(perp1, perp2, 1/3)
  %pdx       = OpFSub %float %perp2x %perp1x
  %pdy       = OpFSub %float %perp2y %perp1y
  %ppx_13    = OpFMul %float %pdx %float_1o3
  %ppy_13    = OpFMul %float %pdy %float_1o3
  %perpx_13  = OpFAdd %float %perp1x %ppx_13
  %perpy_13  = OpFAdd %float %perp1y %ppy_13

; Color at t=1/3
  %cd_13     = OpVectorTimesScalar %v3float %cdiff %float_1o3
  %col_13    = OpFAdd %v3float %c1 %cd_13

; Point at t=2/3
  %ax_23     = OpFMul %float %segx %float_2o3
  %ay_23     = OpFMul %float %segy %float_2o3
  %az_23     = OpFMul %float %segz %float_2o3
  %aw_23     = OpFMul %float %segw %float_2o3
  %px_23     = OpFAdd %float %p1x %ax_23
  %py_23     = OpFAdd %float %p1y %ay_23
  %pz_23     = OpFAdd %float %p1z %az_23
  %pw_23     = OpFAdd %float %p1w %aw_23

; Perp at t=2/3
  %ppx_23    = OpFMul %float %pdx %float_2o3
  %ppy_23    = OpFMul %float %pdy %float_2o3
  %perpx_23  = OpFAdd %float %perp1x %ppx_23
  %perpy_23  = OpFAdd %float %perp1y %ppy_23

; Color at t=2/3
  %cd_23     = OpVectorTimesScalar %v3float %cdiff %float_2o3
  %col_23    = OpFAdd %v3float %c1 %cd_23

; ============ DASH 1: t=0 to t=1/3 ============
; Vertex 0: p1 + perp1
  %v0x       = OpFAdd %float %p1x %perp1x
  %v0y       = OpFAdd %float %p1y %perp1y
  %v0pos     = OpCompositeConstruct %v4float %v0x %v0y %p1z %p1w
  %ac_out0   = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %ac_out0 %v0pos
               OpStore %outColor %c1
               OpEmitVertex

; Vertex 1: p1 - perp1
  %v1x       = OpFSub %float %p1x %perp1x
  %v1y       = OpFSub %float %p1y %perp1y
  %v1pos     = OpCompositeConstruct %v4float %v1x %v1y %p1z %p1w
               OpStore %ac_out0 %v1pos
               OpStore %outColor %c1
               OpEmitVertex

; Vertex 2: p(1/3) + perp(1/3)
  %v2x       = OpFAdd %float %px_13 %perpx_13
  %v2y       = OpFAdd %float %py_13 %perpy_13
  %v2pos     = OpCompositeConstruct %v4float %v2x %v2y %pz_13 %pw_13
               OpStore %ac_out0 %v2pos
               OpStore %outColor %col_13
               OpEmitVertex

; Vertex 3: p(1/3) - perp(1/3)
  %v3x       = OpFSub %float %px_13 %perpx_13
  %v3y       = OpFSub %float %py_13 %perpy_13
  %v3pos     = OpCompositeConstruct %v4float %v3x %v3y %pz_13 %pw_13
               OpStore %ac_out0 %v3pos
               OpStore %outColor %col_13
               OpEmitVertex

               OpEndPrimitive

; ============ DASH 2: t=2/3 to t=1 ============
; Vertex 4: p(2/3) + perp(2/3)
  %v4x       = OpFAdd %float %px_23 %perpx_23
  %v4y       = OpFAdd %float %py_23 %perpy_23
  %v4pos     = OpCompositeConstruct %v4float %v4x %v4y %pz_23 %pw_23
               OpStore %ac_out0 %v4pos
               OpStore %outColor %col_23
               OpEmitVertex

; Vertex 5: p(2/3) - perp(2/3)
  %v5x       = OpFSub %float %px_23 %perpx_23
  %v5y       = OpFSub %float %py_23 %perpy_23
  %v5pos     = OpCompositeConstruct %v4float %v5x %v5y %pz_23 %pw_23
               OpStore %ac_out0 %v5pos
               OpStore %outColor %col_23
               OpEmitVertex

; Vertex 6: p2 + perp2
  %v6x       = OpFAdd %float %p2x %perp2x
  %v6y       = OpFAdd %float %p2y %perp2y
  %v6pos     = OpCompositeConstruct %v4float %v6x %v6y %p2z %p2w
               OpStore %ac_out0 %v6pos
               OpStore %outColor %c2
               OpEmitVertex

; Vertex 7: p2 - perp2
  %v7x       = OpFSub %float %p2x %perp2x
  %v7y       = OpFSub %float %p2y %perp2y
  %v7pos     = OpCompositeConstruct %v4float %v7x %v7y %p2z %p2w
               OpStore %ac_out0 %v7pos
               OpStore %outColor %c2
               OpEmitVertex

               OpEndPrimitive

               OpReturn
               OpFunctionEnd
