#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>
#include <numeric>
#include <cstring>

using namespace std;

static const long long INF = 1e18;

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, M, K;
    cin >> N >> M >> K;

    struct Edge { int to; long long w; };
    vector<vector<Edge>> adj(N);
    for(int i=0;i<M;i++){
        int u,v; long long w;
        cin >> u >> v >> w;
        adj[u].push_back({v,w});
        adj[v].push_back({u,w});
    }
    vector<int> terms(K);
    for(int i=0;i<K;i++) cin >> terms[i];

    if(K <= 1){
        cout << "0 0\n";
        return 0;
    }

    // Dijkstra from source
    auto dijkstra = [&](int src) -> pair<vector<long long>, vector<int>> {
        vector<long long> dist(N, INF);
        vector<int> par(N, -1);
        priority_queue<pair<long long,int>, vector<pair<long long,int>>, greater<>> pq;
        dist[src] = 0;
        pq.push({0, src});
        while(!pq.empty()){
            auto [d,u] = pq.top(); pq.pop();
            if(d > dist[u]) continue;
            for(auto &e : adj[u]){
                if(dist[u]+e.w < dist[e.to]){
                    dist[e.to] = dist[u]+e.w;
                    par[e.to] = u;
                    pq.push({dist[e.to], e.to});
                }
            }
        }
        return {dist, par};
    };

    // For small K, use Dreyfus-Wagner DP
    if(K <= 15){
        // dp[S][v] = min cost to connect terminal subset S with v as root
        int full = (1 << K) - 1;
        vector<vector<long long>> dp(1<<K, vector<long long>(N, INF));
        // parent tracking: for each (S,v), store how we got there
        // type 0: split into (S1,v) and (S2,v)
        // type 1: relax from (S,u) via edge u->v
        struct Trace { int type; int s1; int u; };
        vector<vector<Trace>> tr(1<<K, vector<Trace>(N, {-1,0,0}));

        // Base: each terminal alone
        for(int i=0;i<K;i++){
            dp[1<<i][terms[i]] = 0;
        }

        for(int S=1; S<=full; S++){
            // Merge subsets
            for(int sub = (S-1)&S; sub > 0; sub = (sub-1)&S){
                int comp = S ^ sub;
                if(sub < comp) continue; // avoid double counting
                for(int v=0;v<N;v++){
                    if(dp[sub][v] == INF || dp[comp][v] == INF) continue;
                    long long val = dp[sub][v] + dp[comp][v];
                    if(val < dp[S][v]){
                        dp[S][v] = val;
                        tr[S][v] = {0, sub, 0};
                    }
                }
            }
            // Dijkstra-like relaxation on graph
            priority_queue<pair<long long,int>, vector<pair<long long,int>>, greater<>> pq;
            for(int v=0;v<N;v++){
                if(dp[S][v] < INF) pq.push({dp[S][v], v});
            }
            while(!pq.empty()){
                auto [d,u] = pq.top(); pq.pop();
                if(d > dp[S][u]) continue;
                for(auto &e : adj[u]){
                    if(dp[S][u] + e.w < dp[S][e.to]){
                        dp[S][e.to] = dp[S][u] + e.w;
                        tr[S][e.to] = {1, S, u};
                        pq.push({dp[S][e.to], e.to});
                    }
                }
            }
        }

        // Find best root for full set
        int bestV = 0;
        for(int v=1;v<N;v++){
            if(dp[full][v] < dp[full][bestV]) bestV = v;
        }

        // Reconstruct edges
        vector<pair<int,int>> edges;
        // DFS on trace
        struct StackItem { int S; int v; };
        vector<StackItem> stk;
        stk.push_back({full, bestV});
        while(!stk.empty()){
            auto [S,v] = stk.back(); stk.pop_back();
            if(__popcnt(S) <= 1 && dp[S][v] == 0) continue;
            auto &t = tr[S][v];
            if(t.type == 0){
                // split into (sub, v) and (S^sub, v)
                int sub = t.s1;
                int comp = S ^ sub;
                stk.push_back({sub, v});
                stk.push_back({comp, v});
            } else {
                // came from (S, u) via edge u->v
                edges.push_back({t.u, v});
                stk.push_back({S, t.u});
            }
        }

        cout << dp[full][bestV] << " " << edges.size() << "\n";
        for(auto &[u,v] : edges) cout << u << " " << v << "\n";
    }
    else {
        // Large K: MST-based heuristic
        // 1. Compute shortest paths between all terminals
        // 2. Build complete graph on terminals with shortest path weights
        // 3. Find MST of complete graph
        // 4. Expand MST edges to actual paths, remove duplicate edges, find MST again

        vector<vector<long long>> tdist(K);
        vector<vector<int>> tpar(K);
        for(int i=0;i<K;i++){
            auto [d,p] = dijkstra(terms[i]);
            tdist[i] = move(d);
            tpar[i] = move(p);
        }

        // Prim's MST on complete terminal graph
        vector<bool> inMST(K, false);
        vector<long long> key(K, INF);
        vector<int> from(K, -1);
        key[0] = 0;
        vector<pair<int,int>> mstEdges; // pairs of terminal indices

        for(int iter=0; iter<K; iter++){
            int u = -1;
            for(int i=0;i<K;i++){
                if(!inMST[i] && (u==-1 || key[i]<key[u])) u=i;
            }
            inMST[u] = true;
            if(from[u] != -1) mstEdges.push_back({from[u], u});
            for(int i=0;i<K;i++){
                if(!inMST[i] && tdist[u][terms[i]] < key[i]){
                    key[i] = tdist[u][terms[i]];
                    from[i] = u;
                }
            }
        }

        // Expand MST edges to actual graph edges
        // Collect all edges used
        vector<pair<int,int>> allEdges;
        vector<bool> used(N, false);

        for(auto &[ti, tj] : mstEdges){
            // Path from terms[ti] to terms[tj] using tpar[ti]
            int cur = terms[tj];
            while(cur != terms[ti]){
                int p = tpar[ti][cur];
                allEdges.push_back({min(p,cur), max(p,cur)});
                cur = p;
            }
        }

        // Remove duplicates
        sort(allEdges.begin(), allEdges.end());
        allEdges.erase(unique(allEdges.begin(), allEdges.end()), allEdges.end());

        // Build subgraph and find MST to remove cycles
        // Collect vertices
        vector<int> verts;
        for(auto &[u,v] : allEdges){ verts.push_back(u); verts.push_back(v); }
        sort(verts.begin(), verts.end());
        verts.erase(unique(verts.begin(), verts.end()), verts.end());

        // Find edge weights
        struct WEdge { int u,v; long long w; };
        vector<WEdge> wedges;
        for(auto &[eu, ev] : allEdges){
            // find weight
            long long w = INF;
            for(auto &e : adj[eu]){
                if(e.to == ev && e.w < w) w = e.w;
            }
            wedges.push_back({eu, ev, w});
        }
        sort(wedges.begin(), wedges.end(), [](const WEdge&a, const WEdge&b){ return a.w < b.w; });

        // Kruskal
        vector<int> parent(N);
        iota(parent.begin(), parent.end(), 0);
        function<int(int)> find = [&](int x)->int{
            while(parent[x]!=x) x=parent[x]=parent[parent[x]];
            return x;
        };

        vector<pair<int,int>> result;
        long long totalW = 0;
        for(auto &e : wedges){
            int pu=find(e.u), pv=find(e.v);
            if(pu!=pv){
                parent[pu]=pv;
                result.push_back({e.u, e.v});
                totalW += e.w;
            }
        }

        // Prune leaves that are not terminals
        vector<bool> isTerm(N, false);
        for(int t : terms) isTerm[t] = true;

        // Build adjacency from result
        bool changed = true;
        while(changed){
            changed = false;
            vector<vector<pair<int,int>>> tadj(N);
            for(int i=0;i<(int)result.size();i++){
                auto [u,v] = result[i];
                tadj[u].push_back({v,i});
                tadj[v].push_back({u,i});
            }
            vector<bool> removed(result.size(), false);
            for(auto &vert : verts){
                if(isTerm[vert]) continue;
                int deg = 0; int eidx = -1;
                for(auto &[nb, idx] : tadj[vert]){
                    if(!removed[idx]){ deg++; eidx=idx; }
                }
                if(deg == 1){
                    removed[eidx] = true;
                    // recompute weight
                    for(auto &e : wedges){
                        if((e.u==result[eidx].first && e.v==result[eidx].second) ||
                           (e.v==result[eidx].first && e.u==result[eidx].second)){
                            totalW -= e.w;
                            break;
                        }
                    }
                    changed = true;
                }
            }
            if(changed){
                vector<pair<int,int>> nr;
                for(int i=0;i<(int)result.size();i++){
                    if(!removed[i]) nr.push_back(result[i]);
                }
                result = nr;
                verts.clear();
                for(auto &[u,v] : result){ verts.push_back(u); verts.push_back(v); }
                sort(verts.begin(), verts.end());
                verts.erase(unique(verts.begin(), verts.end()), verts.end());
            }
        }

        cout << totalW << " " << result.size() << "\n";
        for(auto &[u,v] : result) cout << u << " " << v << "\n";
    }

    return 0;
}
