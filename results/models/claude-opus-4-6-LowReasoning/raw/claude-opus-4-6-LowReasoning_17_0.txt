using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

class Program
{
    static int CW, CH, CD;
    static int cellSize;
    static int gx, gy, gz;
    static List<int>[][][] grid;

    static List<int[]> placed = new List<int[]>();

    static void Main()
    {
        var br = new StreamReader(Console.OpenStandardInput());
        var first = br.ReadLine().Split(' ');
        int N = int.Parse(first[0]);
        CW = int.Parse(first[1]);
        CH = int.Parse(first[2]);
        CD = int.Parse(first[3]);

        int[][] boxes = new int[N][];
        for (int i = 0; i < N; i++)
        {
            var p = br.ReadLine().Split(' ');
            boxes[i] = new int[] { int.Parse(p[0]), int.Parse(p[1]), int.Parse(p[2]) };
        }

        int[] order = Enumerable.Range(0, N).ToArray();
        Array.Sort(order, (a, b) =>
        {
            long va = (long)boxes[a][0] * boxes[a][1] * boxes[a][2];
            long vb = (long)boxes[b][0] * boxes[b][1] * boxes[b][2];
            if (vb != va) return vb.CompareTo(va);
            int ma = Math.Max(boxes[a][0], Math.Max(boxes[a][1], boxes[a][2]));
            int mb = Math.Max(boxes[b][0], Math.Max(boxes[b][1], boxes[b][2]));
            return mb.CompareTo(ma);
        });

        // Determine grid cell size for spatial hashing
        // Aim for ~100-200 cells per dimension max
        int maxGridCells = 200;
        cellSize = Math.Max(1, Math.Max(CW, Math.Max(CH, CD)) / maxGridCells);
        // But also consider average box size
        if (N > 0)
        {
            long sumDim = 0;
            for (int i = 0; i < N; i++)
                sumDim += boxes[i][0] + boxes[i][1] + boxes[i][2];
            int avgDim = (int)(sumDim / (3L * N));
            if (avgDim > cellSize) cellSize = avgDim;
        }
        cellSize = Math.Max(1, cellSize);

        gx = Math.Min(maxGridCells, (CW + cellSize - 1) / cellSize);
        gy = Math.Min(maxGridCells, (CH + cellSize - 1) / cellSize);
        gz = Math.Min(maxGridCells, (CD + cellSize - 1) / cellSize);
        gx = Math.Max(1, gx);
        gy = Math.Max(1, gy);
        gz = Math.Max(1, gz);

        grid = new List<int>[gx][][];
        for (int i = 0; i < gx; i++)
        {
            grid[i] = new List<int>[gy][];
            for (int j = 0; j < gy; j++)
                grid[i][j] = new List<int>[gz];
        }

        // Extreme points sorted by (y, z, x) via SortedSet encoding
        var eps = new SortedSet<long>();
        eps.Add(EncodeEP(0, 0, 0));

        var result = new List<int[]>();

        foreach (int idx in order)
        {
            int[] box = boxes[idx];
            var orients = GetOrientations(box);

            long bestScore = long.MaxValue;
            int bestX = -1, bestY = -1, bestZ = -1, bestW = 0, bestH = 0, bestD = 0;

            // For large N, limit EP search
            int epLimit = N > 1000 ? 200 : (N > 100 ? 500 : int.MaxValue);
            int epChecked = 0;

            foreach (long epCode in eps)
            {
                if (epChecked++ >= epLimit) break;

                int ex, ey, ez;
                DecodeEP(epCode, out ex, out ey, out ez);

                foreach (var o in orients)
                {
                    int ow = o[0], oh = o[1], od = o[2];
                    if (ex + ow > CW || ey + oh > CH || ez + od > CD) continue;

                    long score = (long)ey * 1000001L * 1000001L + (long)ez * 1000001L + ex;
                    if (score >= bestScore) continue;

                    if (!Overlaps(ex, ey, ez, ow, oh, od))
                    {
                        bestScore = score;
                        bestX = ex; bestY = ey; bestZ = ez;
                        bestW = ow; bestH = oh; bestD = od;
                    }
                }
            }

            if (bestX >= 0)
            {
                result.Add(new int[] { idx, bestX, bestY, bestZ });
                int pi = placed.Count;
                placed.Add(new int[] { bestX, bestY, bestZ, bestW, bestH, bestD });
                AddToGrid(pi, bestX, bestY, bestZ, bestW, bestH, bestD);

                int nx = bestX + bestW, ny = bestY + bestH, nz = bestZ + bestD;
                if (nx < CW) eps.Add(EncodeEP(nx, bestY, bestZ));
                if (ny < CH) eps.Add(EncodeEP(bestX, ny, bestZ));
                if (nz < CD) eps.Add(EncodeEP(bestX, bestY, nz));
                eps.Remove(EncodeEP(bestX, bestY, bestZ));
            }
        }

        var sw = new StreamWriter(Console.OpenStandardOutput());
        sw.WriteLine(result.Count);
        foreach (var r in result)
        {
            sw.Write(r[0]); sw.Write(' ');
            sw.Write(r[1]); sw.Write(' ');
            sw.Write(r[2]); sw.Write(' ');
            sw.WriteLine(r[3]);
        }
        sw.Flush();
    }

    static void AddToGrid(int pi, int x, int y, int z, int w, int h, int d)
    {
        int x1 = x / cellSize, x2 = Math.Min(gx - 1, (x + w - 1) / cellSize);
        int y1 = y / cellSize, y2 = Math.Min(gy - 1, (y + h - 1) / cellSize);
        int z1 = z / cellSize, z2 = Math.Min(gz - 1, (z + d - 1) / cellSize);
        for (int ix = x1; ix <= x2; ix++)
            for (int iy = y1; iy <= y2; iy++)
                for (int iz = z1; iz <= z2; iz++)
                {
                    var cell = grid[ix][iy][iz];
                    if (cell == null) { cell = new List<int>(); grid[ix][iy][iz] = cell; }
                    cell.Add(pi);
                }
    }

    static bool Overlaps(int x, int y, int z, int w, int h, int d)
    {
        int x2 = x + w, y2 = y + h, z2 = z + d;
        int gx1 = x / cellSize, gx2 = Math.Min(gx - 1, (x2 - 1) / cellSize);
        int gy1 = y / cellSize, gy2 = Math.Min(gy - 1, (y2 - 1) / cellSize);
        int gz1 = z / cellSize, gz2 = Math.Min(gz - 1, (z2 - 1) / cellSize);

        var seen = new HashSet<int>();
        for (int ix = gx1; ix <= gx2; ix++)
            for (int iy = gy1; iy <= gy2; iy++)
                for (int iz = gz1; iz <= gz2; iz++)
                {
                    var cell = grid[ix][iy][iz];
                    if (cell == null) continue;
                    for (int ci = 0; ci < cell.Count; ci++)
                    {
                        int pi = cell[ci];
                        if (!seen.Add(pi)) continue;
                        var p = placed[pi];
                        if (x < p[0] + p[3] && x2 > p[0] &&
                            y < p[1] + p[4] && y2 > p[1] &&
                            z < p[2] + p[5] && z2 > p[2])
                            return true;
                    }
                }
        return false;
    }

    static int[][] GetOrientations(int[] b)
    {
        var set = new HashSet<long>();
        var list = new List<int[]>();
        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++)
                for (int k = 0; k < 3; k++)
                {
                    if (i == j || i == k || j == k) continue;
                    long key = (long)b[i] * 1000001L * 1000001L + (long)b[j] * 1000001L + b[k];
                    if (set.Add(key))
                        list.Add(new int[] { b[i], b[j], b[k] });
                }
        return list.ToArray();
    }

    static long EncodeEP(int x, int y, int z)
    {
        return (long)y * 1000001L * 1000001L + (long)z * 1000001L + x;
    }

    static void DecodeEP(long code, out int x, out int y, out int z)
    {
        x = (int)(code % 1000001L);
        code /= 1000001L;
        z = (int)(code % 1000001L);
        y = (int)(code / 1000001L);
    }
}
