; SPIR-V Geometry Shader - Flat shading with InputTrianglesAdjacency
               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %inColor %outColor
               OpExecutionMode %main InputTrianglesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 3
               OpExecutionMode %main Invocations 1

               ; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position

               OpDecorate %inColor Location 2
               OpDecorate %outColor Location 0

               ; Types
      %void  = OpTypeVoid
   %voidfn   = OpTypeFunction %void
     %float  = OpTypeFloat 32
   %v3float  = OpTypeVector %float 3
   %v4float  = OpTypeVector %float 4
      %uint  = OpTypeInt 32 0
       %int  = OpTypeInt 32 1

     %int_0  = OpConstant %int 0
     %int_1  = OpConstant %int 1
     %int_2  = OpConstant %int 2
     %int_4  = OpConstant %int 4
    %uint_6  = OpConstant %uint 6

   %float_0  = OpConstant %float 0.0
 %float_0_2  = OpConstant %float 0.2
 %float_0_8  = OpConstant %float 0.8
   %float_1  = OpConstant %float 1.0
 %float_1_3  = OpConstant %float 0.333333343
%light_dir_val = OpConstantComposite %v3float %float_1 %float_1 %float_1

; Input gl_PerVertex block (array of 6)
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in  = OpTypeArray %gl_PerVertex_in %uint_6
%ptr_in_arr = OpTypePointer Input %arr_pv_in
%ptr_in_v4  = OpTypePointer Input %v4float

; Input color array (location 2)
%arr_v3_6   = OpTypeArray %v3float %uint_6
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_6
%ptr_in_v3  = OpTypePointer Input %v3float

; Output gl_PerVertex block
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float

; Output color
%ptr_out_v3 = OpTypePointer Output %v3float

; Variables
     %gl_in  = OpVariable %ptr_in_arr Input
    %gl_out  = OpVariable %ptr_out_pv Output
   %inColor  = OpVariable %ptr_in_arr_v3 Input
  %outColor  = OpVariable %ptr_out_v3 Output

; Main function
      %main  = OpFunction %void None %voidfn
     %entry  = OpLabel

; Load positions of vertices 0, 2, 4
   %p0_ptr  = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
       %p0  = OpLoad %v4float %p0_ptr
   %p2_ptr  = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
       %p2  = OpLoad %v4float %p2_ptr
   %p4_ptr  = OpAccessChain %ptr_in_v4 %gl_in %int_4 %int_0
       %p4  = OpLoad %v4float %p4_ptr

; Load colors at 0, 2, 4
   %c0_ptr  = OpAccessChain %ptr_in_v3 %inColor %int_0
       %c0  = OpLoad %v3float %c0_ptr
   %c2_ptr  = OpAccessChain %ptr_in_v3 %inColor %int_2
       %c2  = OpLoad %v3float %c2_ptr
   %c4_ptr  = OpAccessChain %ptr_in_v3 %inColor %int_4
       %c4  = OpLoad %v3float %c4_ptr

; Compute edge vectors (using xyz of positions)
   %p0_xyz  = OpVectorShuffle %v3float %p0 %p0 0 1 2
   %p2_xyz  = OpVectorShuffle %v3float %p2 %p2 0 1 2
   %p4_xyz  = OpVectorShuffle %v3float %p4 %p4 0 1 2

    %edge1  = OpFSub %v3float %p2_xyz %p0_xyz
    %edge2  = OpFSub %v3float %p4_xyz %p0_xyz

; Cross product: edge1 x edge2
   %e1_x   = OpCompositeExtract %float %edge1 0
   %e1_y   = OpCompositeExtract %float %edge1 1
   %e1_z   = OpCompositeExtract %float %edge1 2
   %e2_x   = OpCompositeExtract %float %edge2 0
   %e2_y   = OpCompositeExtract %float %edge2 1
   %e2_z   = OpCompositeExtract %float %edge2 2

  %cx_a    = OpFMul %float %e1_y %e2_z
  %cx_b    = OpFMul %float %e1_z %e2_y
  %cx      = OpFSub %float %cx_a %cx_b

  %cy_a    = OpFMul %float %e1_z %e2_x
  %cy_b    = OpFMul %float %e1_x %e2_z
  %cy      = OpFSub %float %cy_a %cy_b

  %cz_a    = OpFMul %float %e1_x %e2_y
  %cz_b    = OpFMul %float %e1_y %e2_x
  %cz      = OpFSub %float %cz_a %cz_b

  %cross   = OpCompositeConstruct %v3float %cx %cy %cz

; Normalize face normal
  %normal  = OpExtInst %v3float %1 Normalize %cross

; Normalize light direction
  %light   = OpExtInst %v3float %1 Normalize %light_dir_val

; Diffuse = max(dot(N, L), 0.0)
  %ndotl   = OpDot %float %normal %light
  %diffuse = OpExtInst %float %1 FMax %ndotl %float_0

; Base color = average of color[0], color[2], color[4]
  %c_sum1  = OpFAdd %v3float %c0 %c2
  %c_sum2  = OpFAdd %v3float %c_sum1 %c4
  %sc_1_3  = OpCompositeConstruct %v3float %float_1_3 %float_1_3 %float_1_3
  %base    = OpFMul %v3float %c_sum2 %sc_1_3

; Final color = base * (0.2 + 0.8 * diffuse)
  %scaled  = OpFMul %float %float_0_8 %diffuse
  %factor  = OpFAdd %float %float_0_2 %scaled
  %fac_v   = OpCompositeConstruct %v3float %factor %factor %factor
  %final   = OpFMul %v3float %base %fac_v

; Emit vertex 0
  %out0_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %out0_ptr %p0
               OpStore %outColor %final
               OpEmitVertex

; Emit vertex 2
               OpStore %out0_ptr %p2
               OpStore %outColor %final
               OpEmitVertex

; Emit vertex 4
               OpStore %out0_ptr %p4
               OpStore %outColor %final
               OpEmitVertex

               OpEndPrimitive
               OpReturn
               OpFunctionEnd
