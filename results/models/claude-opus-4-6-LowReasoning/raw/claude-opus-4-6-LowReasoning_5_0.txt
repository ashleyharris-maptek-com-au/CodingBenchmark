use std::io::{self, Read, Write, BufWriter};
use std::time::Instant;
use std::collections::VecDeque;
use std::sync::atomic::{AtomicBool, Ordering};
use std::sync::{Arc, Mutex};
use std::thread;

const DX: [i32; 4] = [1, -1, 0, 0];
const DY: [i32; 4] = [0, 0, 1, -1];

fn are_adj(a: (usize, usize), b: (usize, usize)) -> bool {
    (a.0 as i32 - b.0 as i32).abs() + (a.1 as i32 - b.1 as i32).abs() == 1
}

struct Solver {
    w: usize, h: usize, total: usize,
    blocked: Vec<bool>, visited: Vec<bool>,
    path: Vec<(usize, usize)>,
    require_cycle: bool, found: bool, calls: u64,
    stop: Arc<AtomicBool>,
}

impl Solver {
    fn udeg(&self, x: usize, y: usize) -> i32 {
        let mut d = 0;
        for dir in 0..4 {
            let nx = x as i32 + DX[dir]; let ny = y as i32 + DY[dir];
            if nx >= 0 && nx < self.w as i32 && ny >= 0 && ny < self.h as i32 {
                let ni = ny as usize * self.w + nx as usize;
                if !self.blocked[ni] && !self.visited[ni] { d += 1; }
            }
        }
        d
    }

    fn is_connected(&self) -> bool {
        let rem = self.total - self.path.len();
        if rem <= 1 { return true; }
        let n = self.w * self.h;
        let mut si = n;
        for i in 0..n { if !self.blocked[i] && !self.visited[i] { si = i; break; } }
        if si == n { return true; }
        let mut seen = vec![false; n];
        seen[si] = true;
        let mut stk = vec![(si % self.w, si / self.w)];
        let mut c = 1usize;
        while let Some((qx, qy)) = stk.pop() {
            for dir in 0..4 {
                let nx = qx as i32 + DX[dir]; let ny = qy as i32 + DY[dir];
                if nx >= 0 && nx < self.w as i32 && ny >= 0 && ny < self.h as i32 {
                    let ni = ny as usize * self.w + nx as usize;
                    if !self.blocked[ni] && !self.visited[ni] && !seen[ni] {
                        seen[ni] = true; c += 1; stk.push((nx as usize, ny as usize));
                    }
                }
            }
        }
        c == rem
    }

    fn solve(&mut self) {
        if self.blocked[0] || self.total == 0 { return; }
        if self.total == 1 { if !self.require_cycle { self.found = true; } return; }
        self.bt();
    }

    fn bt(&mut self) {
        if self.found { return; }
        self.calls += 1;
        if self.calls & 0x3FFFF == 0 && self.stop.load(Ordering::Relaxed) { return; }

        let len = self.path.len();
        if len == self.total {
            if self.require_cycle {
                if are_adj(self.path[len - 1], (0, 0)) { self.found = true; }
            } else { self.found = true; }
            return;
        }

        let (cx, cy) = self.path[len - 1];
        let rem = self.total - len;

        let mut nbrs = [(0usize, 0usize, 0i32); 4];
        let mut nc = 0;
        for dir in 0..4 {
            let nx = cx as i32 + DX[dir]; let ny = cy as i32 + DY[dir];
            if nx >= 0 && nx < self.w as i32 && ny >= 0 && ny < self.h as i32 {
                let (ux, uy) = (nx as usize, ny as usize);
                let ni = uy * self.w + ux;
                if !self.blocked[ni] && !self.visited[ni] {
                    nbrs[nc] = (ux, uy, self.udeg(ux, uy)); nc += 1;
                }
            }
        }
        if nc == 0 { return; }
        if rem > 1 { for i in 0..nc { if nbrs[i].2 == 0 { return; } } }
        if rem > 2 && !self.is_connected() { return; }

        nbrs[..nc].sort_by_key(|&(_, _, d)| d);
        for i in 0..nc {
            let (nx, ny, _) = nbrs[i];
            let ni = ny * self.w + nx;
            self.visited[ni] = true;
            self.path.push((nx, ny));
            self.bt();
            if self.found { return; }
            self.path.pop();
            self.visited[ni] = false;
        }
    }
}

fn construct_snake(w: usize, h: usize, cycle: bool) -> Option<Vec<(usize, usize)>> {
    let mut p = Vec::with_capacity(w * h);
    if !cycle {
        for y in 0..h {
            if y % 2 == 0 { for x in 0..w { p.push((x, y)); } }
            else { for x in (0..w).rev() { p.push((x, y)); } }
        }
        return Some(p);
    }
    if (w * h) % 2 != 0 || w < 2 || h < 2 { return None; }
    if w % 2 == 0 {
        for x in 0..w { p.push((x, 0)); }
        for y in 1..h { p.push((w - 1, y)); }
        for x in (1..w - 1).rev() {
            if (w - 1 - x) % 2 == 1 { for y in 1..h { p.push((x, y)); } }
            else { for y in (1..h).rev() { p.push((x, y)); } }
        }
        for y in (1..h).rev() { p.push((0, y)); }
        if p.len() == w * h && *p.last().unwrap() == (0, 1) { return Some(p); }
        p.clear();
    }
    if h % 2 == 0 {
        for y in 0..h { p.push((0, y)); }
        for x in 1..w { p.push((x, h - 1)); }
        for y in (1..h - 1).rev() {
            if (h - 1 - y) % 2 == 1 { for x in 1..w { p.push((x, y)); } }
            else { for x in (1..w).rev() { p.push((x, y)); } }
        }
        for x in (1..w).rev() { p.push((x, 0)); }
        if p.len() == w * h && *p.last().unwrap() == (1, 0) { return Some(p); }
    }
    None
}

// Parallel Warnsdorff with rotation for large grids
fn solve_large_parallel(w: usize, h: usize, blocked: &[bool], total: usize, cycle: bool, t0: Instant, deadline_secs: u64) -> Option<Vec<(usize, usize)>> {
    let n = w * h;
    let blocked = Arc::new(blocked.to_vec());
    let found = Arc::new(AtomicBool::new(false));
    let result: Arc<Mutex<Option<Vec<(usize, usize)>>>> = Arc::new(Mutex::new(None));

    let num_threads = 32usize.min(total / 10 + 1);
    let handles: Vec<_> = (0..num_threads).map(|tid| {
        let blocked = Arc::clone(&blocked);
        let found = Arc::clone(&found);
        let result = Arc::clone(&result);

        thread::spawn(move || {
            let mut s0: u64 = 0x123456789ABCDEF0u64.wrapping_add(tid as u64 * 0x9E3779B97F4A7C15);
            let mut s1: u64 = 0xFEDCBA9876543210u64.wrapping_add(tid as u64 * 0x6C62272E07BB0142);

            let xsh = |s0: &mut u64, s1: &mut u64| -> u64 {
                let mut t = *s0; let s = *s1; *s0 = s;
                t ^= t << 23; t ^= t >> 17; t ^= s ^ (s >> 26); *s1 = t;
                s.wrapping_add(t)
            };

            loop {
                if found.load(Ordering::Relaxed) { return; }
                if t0.elapsed().as_secs() > deadline_secs { return; }

                xsh(&mut s0, &mut s1);

                let mut vis = vec![false; n];
                vis[0] = true;
                let mut path: Vec<(usize, usize)> = vec![(0, 0)];

                // Warnsdorff
                while path.len() < total {
                    if found.load(Ordering::Relaxed) { return; }
                    let (cx, cy) = *path.last().unwrap();
                    let mut cands = [(0usize, 0usize, 0i32); 4];
                    let mut cc = 0;
                    for dir in 0..4 {
                        let nx = cx as i32 + DX[dir]; let ny = cy as i32 + DY[dir];
                        if nx < 0 || nx >= w as i32 || ny < 0 || ny >= h as i32 { continue; }
                        let (ux, uy) = (nx as usize, ny as usize);
                        let ni = uy * w + ux;
                        if blocked[ni] || vis[ni] { continue; }
                        let mut deg = 0i32;
                        for d2 in 0..4 {
                            let mx = ux as i32 + DX[d2]; let my = uy as i32 + DY[d2];
                            if mx >= 0 && mx < w as i32 && my >= 0 && my < h as i32 {
                                let mi = my as usize * w + mx as usize;
                                if !blocked[mi] && !vis[mi] { deg += 1; }
                            }
                        }
                        cands[cc] = (ux, uy, deg); cc += 1;
                    }
                    if cc == 0 { break; }
                    cands[..cc].sort_by_key(|c| c.2);
                    let md = cands[0].2;
                    let te = cands[..cc].iter().position(|c| c.2 != md).unwrap_or(cc);
                    let pick = if te > 1 { xsh(&mut s0, &mut s1) as usize % te } else { 0 };
                    let (nx, ny, _) = cands[pick];
                    vis[ny * w + nx] = true;
                    path.push((nx, ny));
                }

                if path.len() == total {
                    if !cycle || are_adj(*path.last().unwrap(), (0, 0)) {
                        found.store(true, Ordering::Relaxed);
                        *result.lock().unwrap() = Some(path);
                        return;
                    }
                }

                // Rotation: keep rotating and trying to extend
                let mut pos = vec![usize::MAX; n];
                for (i, &(px, py)) in path.iter().enumerate() { pos[py * w + px] = i; }

                for _rot in 0..total * 500 {
                    if found.load(Ordering::Relaxed) { return; }
                    if t0.elapsed().as_secs() > deadline_secs { return; }

                    let len = path.len();
                    if len == 0 { break; }
                    let (ex, ey) = path[len - 1];

                    // Try extend
                    {
                        let mut ecands = [(0usize, 0usize, 0i32); 4];
                        let mut ec = 0;
                        for dir in 0..4 {
                            let nx = ex as i32 + DX[dir]; let ny = ey as i32 + DY[dir];
                            if nx < 0 || nx >= w as i32 || ny < 0 || ny >= h as i32 { continue; }
                            let ni = ny as usize * w + nx as usize;
                            if !blocked[ni] && !vis[ni] {
                                let mut deg = 0i32;
                                for d2 in 0..4 {
                                    let mx = nx + DX[d2]; let my = ny + DY[d2];
                                    if mx >= 0 && mx < w as i32 && my >= 0 && my < h as i32 {
                                        let mi = my as usize * w + mx as usize;
                                        if !blocked[mi] && !vis[mi] { deg += 1; }
                                    }
                                }
                                ecands[ec] = (nx as usize, ny as usize, deg); ec += 1;
                            }
                        }
                        if ec > 0 {
                            ecands[..ec].sort_by_key(|c| c.2);
                            let md = ecands[0].2;
                            let te = ecands[..ec].iter().position(|c| c.2 != md).unwrap_or(ec);
                            let pick = if te > 1 { xsh(&mut s0, &mut s1) as usize % te } else { 0 };
                            let (nx, ny, _) = ecands[pick];
                            vis[ny * w + nx] = true;
                            path.push((nx, ny));
                            pos[ny * w + nx] = path.len() - 1;

                            // Continue Warnsdorff
                            while path.len() < total {
                                let (cx, cy) = *path.last().unwrap();
                                let mut cs = [(0usize, 0usize, 0i32); 4];
                                let mut cc2 = 0;
                                for dir in 0..4 {
                                    let nx2 = cx as i32 + DX[dir]; let ny2 = cy as i32 + DY[dir];
                                    if nx2 < 0 || nx2 >= w as i32 || ny2 < 0 || ny2 >= h as i32 { continue; }
                                    let ni = ny2 as usize * w + nx2 as usize;
                                    if blocked[ni] || vis[ni] { continue; }
                                    let mut deg = 0i32;
                                    for d2 in 0..4 {
                                        let mx = nx2 + DX[d2]; let my = ny2 + DY[d2];
                                        if mx >= 0 && mx < w as i32 && my >= 0 && my < h as i32 {
                                            let mi = my as usize * w + mx as usize;
                                            if !blocked[mi] && !vis[mi] { deg += 1; }
                                        }
                                    }
                                    cs[cc2] = (nx2 as usize, ny2 as usize, deg); cc2 += 1;
                                }
                                if cc2 == 0 { break; }
                                cs[..cc2].sort_by_key(|c| c.2);
                                let md = cs[0].2;
                                let te = cs[..cc2].iter().position(|c| c.2 != md).unwrap_or(cc2);
                                let pick = if te > 1 { xsh(&mut s0, &mut s1) as usize % te } else { 0 };
                                let (nx2, ny2, _) = cs[pick];
                                vis[ny2 * w + nx2] = true;
                                path.push((nx2, ny2));
                                pos[ny2 * w + nx2] = path.len() - 1;
                            }

                            if path.len() == total {
                                if !cycle || are_adj(*path.last().unwrap(), (0, 0)) {
                                    found.store(true, Ordering::Relaxed);
                                    *result.lock().unwrap() = Some(path);
                                    return;
                                }
                                break;
                            }
                            continue;
                        }
                    }

                    // No extension, rotate
                    let mut targets = [usize::MAX; 4];
                    let mut tc = 0;
                    for dir in 0..4 {
                        let nx = ex as i32 + DX[dir]; let ny = ey as i32 + DY[dir];
                        if nx < 0 || nx >= w as i32 || ny < 0 || ny >= h as i32 { continue; }
                        let ni = ny as usize * w + nx as usize;
                        let k = pos[ni];
                        if k != usize::MAX && k + 2 < len {
                            targets[tc] = k; tc += 1;
                        }
                    }
                    if tc == 0 { break; }
                    let k = targets[xsh(&mut s0, &mut s1) as usize % tc];
                    path[k + 1..].reverse();
                    for i in k + 1..path.len() {
                        let (px, py) = path[i];
                        pos[py * w + px] = i;
                    }
                }
            }
        })
    }).collect();

    for h in handles { h.join().unwrap(); }
    let r = result.lock().unwrap().clone();
    r
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let stdout = io::stdout();
    let mut out = BufWriter::new(stdout.lock());

    let mut iter = input.split_whitespace();
    let w: usize = iter.next().unwrap().parse().unwrap();
    let h: usize = iter.next().unwrap().parse().unwrap();
    let obs_count: usize = iter.next().unwrap().parse().unwrap();
    let require_cycle: u8 = iter.next().unwrap().parse().unwrap();

    let n = w * h;
    let mut blocked = vec![false; n];
    for _ in 0..obs_count {
        let x: usize = iter.next().unwrap().parse().unwrap();
        let y: usize = iter.next().unwrap().parse().unwrap();
        blocked[y * w + x] = true;
    }

    let total = n - obs_count;
    if total == 0 || w == 0 || h == 0 || blocked[0] {
        writeln!(out, "0").unwrap(); return;
    }
    if require_cycle != 0 && total % 2 != 0 {
        writeln!(out, "0").unwrap(); return;
    }

    // Connectivity
    {
        let mut seen = vec![false; n];
        seen[0] = true;
        let mut q = VecDeque::new();
        q.push_back((0usize, 0usize));
        let mut c = 1usize;
        while let Some((qx, qy)) = q.pop_front() {
            for dir in 0..4 {
                let nx = qx as i32 + DX[dir]; let ny = qy as i32 + DY[dir];
                if nx >= 0 && nx < w as i32 && ny >= 0 && ny < h as i32 {
                    let ni = ny as usize * w + nx as usize;
                    if !blocked[ni] && !seen[ni] { seen[ni] = true; c += 1; q.push_back((nx as usize, ny as usize)); }
                }
            }
        }
        if c != total { writeln!(out, "0").unwrap(); return; }
    }

    let t0 = Instant::now();

    // Obstacle-free: snake
    if obs_count == 0 {
        if let Some(p) = construct_snake(w, h, require_cycle != 0) {
            writeln!(out, "{}", p.len()).unwrap();
            for (x, y) in &p { writeln!(out, "{} {}", x, y).unwrap(); }
            return;
        }
    }

    // Large grids: parallel Warnsdorff + rotation (spend most time here)
    if total > 100 {
        if let Some(p) = solve_large_parallel(w, h, &blocked, total, require_cycle != 0, t0, 250) {
            writeln!(out, "{}", p.len()).unwrap();
            for &(x, y) in &p { writeln!(out, "{} {}", x, y).unwrap(); }
            return;
        }
    }

    // Small: backtracking with time limit
    let stop = Arc::new(AtomicBool::new(false));
    let stop2 = Arc::clone(&stop);
    let t0_2 = t0;
    thread::spawn(move || {
        loop {
            thread::sleep(std::time::Duration::from_secs(1));
            if t0_2.elapsed().as_secs() > 280 { stop2.store(true, Ordering::Relaxed); return; }
        }
    });

    let mut solver = Solver {
        w, h, total, blocked: blocked.to_vec(),
        visited: { let mut v = vec![false; n]; v[0] = true; v },
        path: vec![(0, 0)], require_cycle: require_cycle != 0,
        found: false, calls: 0, stop,
    };
    solver.solve();

    if solver.found {
        writeln!(out, "{}", solver.path.len()).unwrap();
        for &(x, y) in &solver.path { writeln!(out, "{} {}", x, y).unwrap(); }
    } else {
        writeln!(out, "0").unwrap();
    }
}
