use std::io::{self, BufRead, Write, BufWriter};
use std::collections::VecDeque;

const DX: [i32; 4] = [1, -1, 0, 0];
const DY: [i32; 4] = [0, 0, 1, -1];

struct Grid {
    w: usize,
    h: usize,
    blocked: Vec<bool>,
    total_free: usize,
    require_cycle: bool,
}

impl Grid {
    fn idx(&self, x: usize, y: usize) -> usize { y * self.w + x }
    fn is_free(&self, x: usize, y: usize) -> bool { !self.blocked[self.idx(x, y)] }

    fn neighbors(&self, x: usize, y: usize) -> Vec<(usize, usize)> {
        let mut res = Vec::new();
        for i in 0..4 {
            let nx = x as i32 + DX[i];
            let ny = y as i32 + DY[i];
            if nx >= 0 && ny >= 0 && (nx as usize) < self.w && (ny as usize) < self.h {
                let (ux, uy) = (nx as usize, ny as usize);
                if self.is_free(ux, uy) {
                    res.push((ux, uy));
                }
            }
        }
        res
    }
}

fn is_connected(grid: &Grid, visited: &[bool], exclude_idx: usize) -> bool {
    let n = grid.w * grid.h;
    let mut start = None;
    let mut free_count = 0u32;
    for i in 0..n {
        if !grid.blocked[i] && !visited[i] && i != exclude_idx {
            free_count += 1;
            if start.is_none() { start = Some(i); }
        }
    }
    if free_count == 0 { return true; }
    let start = match start { Some(s) => s, None => return true };

    let mut seen = vec![false; n];
    seen[start] = true;
    let mut queue = VecDeque::new();
    queue.push_back(start);
    let mut count = 1u32;

    while let Some(idx) = queue.pop_front() {
        let x = idx % grid.w;
        let y = idx / grid.w;
        for i in 0..4 {
            let nx = x as i32 + DX[i];
            let ny = y as i32 + DY[i];
            if nx >= 0 && ny >= 0 && (nx as usize) < grid.w && (ny as usize) < grid.h {
                let ni = ny as usize * grid.w + nx as usize;
                if !grid.blocked[ni] && !visited[ni] && ni != exclude_idx && !seen[ni] {
                    seen[ni] = true;
                    count += 1;
                    queue.push_back(ni);
                }
            }
        }
    }
    count == free_count
}

fn degree(grid: &Grid, visited: &[bool], x: usize, y: usize) -> u32 {
    let mut d = 0;
    for i in 0..4 {
        let nx = x as i32 + DX[i];
        let ny = y as i32 + DY[i];
        if nx >= 0 && ny >= 0 && (nx as usize) < grid.w && (ny as usize) < grid.h {
            let ni = ny as usize * grid.w + nx as usize;
            if !grid.blocked[ni] && !visited[ni] { d += 1; }
        }
    }
    d
}

fn solve_backtrack(
    grid: &Grid,
    visited: &mut Vec<bool>,
    path: &mut Vec<(usize, usize)>,
    x: usize, y: usize,
    remaining: usize,
    depth_limit: usize,
) -> bool {
    if remaining == 0 {
        if grid.require_cycle {
            let (sx, sy) = (0usize, 0usize);
            let dx = (x as i32 - sx as i32).abs();
            let dy = (y as i32 - sy as i32).abs();
            return dx + dy == 1;
        }
        return true;
    }

    // Collect neighbors sorted by Warnsdorff degree
    let mut nbrs: Vec<(u32, usize, usize)> = Vec::new();
    for i in 0..4 {
        let nx = x as i32 + DX[i];
        let ny = y as i32 + DY[i];
        if nx >= 0 && ny >= 0 && (nx as usize) < grid.w && (ny as usize) < grid.h {
            let (ux, uy) = (nx as usize, ny as usize);
            let ni = uy * grid.w + ux;
            if !grid.blocked[ni] && !visited[ni] {
                let d = degree(grid, visited, ux, uy);
                nbrs.push((d, ux, uy));
            }
        }
    }
    nbrs.sort_unstable_by_key(|&(d, _, _)| d);

    // Pruning: if any unvisited neighbor has degree 0 but remaining > 1, dead end
    // (unless it's the only one left)
    if remaining > 1 {
        for &(d, _, _) in &nbrs {
            if d == 1 && nbrs.len() > 1 {
                // This neighbor will become isolated if we don't visit it next
                // Only valid if we visit it now - but there might be multiple such neighbors
                let dead_ends: usize = nbrs.iter().filter(|&&(dd, _, _)| dd == 1).count();
                if dead_ends > 2 { return false; }
            }
        }
    }

    // Connectivity check (expensive, do sparingly)
    let do_connectivity = remaining > 1 && (depth_limit > 0 && path.len() % 8 == 0 && grid.total_free <= 200);

    for &(_, nx, ny) in &nbrs {
        let ni = ny * grid.w + nx;
        visited[ni] = true;
        path.push((nx, ny));

        if do_connectivity && !is_connected(grid, visited, usize::MAX) {
            path.pop();
            visited[ni] = false;
            continue;
        }

        if solve_backtrack(grid, visited, path, nx, ny, remaining - 1, depth_limit) {
            return true;
        }

        path.pop();
        visited[ni] = false;
    }
    false
}

fn try_snake_pattern(grid: &Grid) -> Option<Vec<(usize, usize)>> {
    // For obstacle-free rectangular grids, construct path directly
    // Check parity: for cycle requirement on even-total grids
    let w = grid.w;
    let h = grid.h;

    // Simple snake: row by row, alternating direction
    let mut path = Vec::with_capacity(w * h);
    for row in 0..h {
        if row % 2 == 0 {
            for col in 0..w {
                if grid.is_free(col, row) {
                    path.push((col, row));
                }
            }
        } else {
            for col in (0..w).rev() {
                if grid.is_free(col, row) {
                    path.push((col, row));
                }
            }
        }
    }

    // Verify it's a valid Hamiltonian path
    if path.len() != grid.total_free { return None; }
    if path[0] != (0, 0) { return None; }

    // Check adjacency
    for i in 1..path.len() {
        let (x1, y1) = path[i-1];
        let (x2, y2) = path[i];
        let dx = (x1 as i32 - x2 as i32).abs();
        let dy = (y1 as i32 - y2 as i32).abs();
        if dx + dy != 1 { return None; }
    }

    if grid.require_cycle {
        let (lx, ly) = path[path.len()-1];
        let dx = lx as i32;
        let dy = ly as i32;
        if dx + dy != 1 { return None; }
    }

    Some(path)
}

// For grids with no obstacles, use structured approach
fn solve_no_obstacles(grid: &Grid) -> Option<Vec<(usize, usize)>> {
    let w = grid.w;
    let h = grid.h;

    if w == 1 && h == 1 {
        if grid.require_cycle { return Some(vec![(0,0)]); }
        return Some(vec![(0, 0)]);
    }

    if w == 1 {
        let path: Vec<(usize,usize)> = (0..h).map(|y| (0, y)).collect();
        if grid.require_cycle {
            // cycle on 1xN impossible for N>1
            return None;
        }
        return Some(path);
    }
    if h == 1 {
        let path: Vec<(usize,usize)> = (0..w).map(|x| (x, 0)).collect();
        if grid.require_cycle {
            return None;
        }
        return Some(path);
    }

    // For cycle: need even number of cells
    if grid.require_cycle && (w * h) % 2 != 0 {
        return None;
    }

    // Snake pattern works for all obstacle-free grids >= 2x2
    try_snake_pattern(grid)
}

fn parity_check(grid: &Grid) -> bool {
    // On a grid, cells can be colored like a checkerboard.
    // A Hamiltonian path alternates colors. If total_free is even,
    // start and end must be different colors. If odd, same color.
    // Start is (0,0) which is color (0+0)%2 = 0 (black).
    let mut black = 0u32;
    let mut white = 0u32;
    for y in 0..grid.h {
        for x in 0..grid.w {
            if grid.is_free(x, y) {
                if (x + y) % 2 == 0 { black += 1; } else { white += 1; }
            }
        }
    }

    // Path of length N visits cells alternating B,W,B,W,...
    // Starting from black (0,0):
    // If N is odd: (N+1)/2 black, N/2 white → need black = white+1
    // If N is even: N/2 black, N/2 white → need black = white
    // Actually more precisely: path has total_free cells.
    // Cells along path: color of cell i = (start_color + i) % 2
    // So # black cells in path = ceil(total/2) if start is black
    // # white cells in path = floor(total/2)
    // So we need: black = ceil(total/2), white = floor(total/2)
    // i.e. black - white = total_free % 2 (0 or 1)
    let total = grid.total_free as i32;
    let diff = black as i32 - white as i32;

    // For a Hamiltonian path starting at black:
    // black = (total+1)/2, white = total/2
    // diff = ((total+1)/2) - (total/2) = 1 if odd, 0 if even...
    // Actually: if total is odd, diff should be 1. If even, diff should be 0.
    // But also could be -1 if total odd (more white than black) — impossible
    // since we start at black.
    // Wait: we need exactly the right count. If diff != total%2, impossible.
    // But also diff must be 0 or 1 (can't be negative since start is black).
    if diff < 0 || diff > 1 { return false; }
    if diff != total % 2 { return false; }

    // For cycle: last cell must be adjacent to (0,0), so last cell is white.
    // Path: B,W,B,W,...,W → total must be even, and diff must be 0.
    if grid.require_cycle {
        if total % 2 != 0 { return false; }
        // Also last cell is white (adjacent to black start), which is consistent
        // with even length path ending on white.
    }

    true
}

fn main() {
    let stdin = io::stdin();
    let stdout = io::stdout();
    let mut out = BufWriter::new(stdout.lock());

    let mut lines = stdin.lock().lines();
    let first_line = lines.next().unwrap().unwrap();
    let parts: Vec<usize> = first_line.split_whitespace().map(|s| s.parse().unwrap()).collect();
    let (w, h, obs_count, req_cycle) = (parts[0], parts[1], parts[2], parts[3] != 0);

    let n = w * h;
    let mut blocked = vec![false; n];
    for _ in 0..obs_count {
        let line = lines.next().unwrap().unwrap();
        let p: Vec<usize> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
        blocked[p[1] * w + p[0]] = true;
    }

    let total_free = blocked.iter().filter(|&&b| !b).count();

    if blocked[0] || total_free == 0 {
        writeln!(out, "0").unwrap();
        return;
    }

    let grid = Grid { w, h, blocked, total_free, require_cycle: req_cycle };

    // Parity check
    if !parity_check(&grid) {
        writeln!(out, "0").unwrap();
        return;
    }

    // Check connectivity of free cells
    {
        let visited = vec![false; n];
        if !is_connected(&grid, &visited, usize::MAX) {
            writeln!(out, "0").unwrap();
            return;
        }
    }

    // Single cell
    if total_free == 1 {
        writeln!(out, "1").unwrap();
        writeln!(out, "0 0").unwrap();
        return;
    }

    // Try structured approach for no obstacles
    if grid.blocked.iter().all(|&b| !b) {
        if let Some(path) = solve_no_obstacles(&grid) {
            writeln!(out, "{}", path.len()).unwrap();
            for (x, y) in &path {
                writeln!(out, "{} {}", x, y).unwrap();
            }
            return;
        }
        // If structured approach failed (e.g. cycle on odd grid), report impossible
        // But parity check should have caught this... fall through to backtrack
    }

    // Try snake pattern even with obstacles (might work)
    if let Some(path) = try_snake_pattern(&grid) {
        writeln!(out, "{}", path.len()).unwrap();
        for (x, y) in &path {
            writeln!(out, "{} {}", x, y).unwrap();
        }
        return;
    }

    // Backtracking with Warnsdorff heuristic
    let mut visited = vec![false; n];
    visited[0] = true;
    let mut path = vec![(0usize, 0usize)];

    if solve_backtrack(&grid, &mut visited, &mut path, 0, 0, total_free - 1, 1000) {
        writeln!(out, "{}", path.len()).unwrap();
        for (x, y) in &path {
            writeln!(out, "{} {}", x, y).unwrap();
        }
    } else {
        writeln!(out, "0").unwrap();
    }
}
