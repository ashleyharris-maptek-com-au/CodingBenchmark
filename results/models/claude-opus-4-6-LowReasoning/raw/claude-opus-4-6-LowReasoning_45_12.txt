import struct

words = []

def w(val):
    words.append(val)

def op(wordcount, opcode):
    w((wordcount << 16) | opcode)

def encode_string(s):
    """Encode string as 32-bit words (null-terminated, padded to 4 bytes)"""
    b = s.encode('utf-8') + b'\x00'
    while len(b) % 4 != 0:
        b += b'\x00'
    result = []
    for i in range(0, len(b), 4):
        result.append(struct.unpack('<I', b[i:i+4])[0])
    return result

def float_to_uint(f):
    return struct.unpack('<I', struct.pack('<f', f))[0]

# IDs
EXT = 1
VOID = 2
FUNC_TYPE = 3
FLOAT = 4
VEC3 = 5
VEC4 = 6
PTR_IN_V3 = 7
PTR_OUT_V4 = 8
NORMAL_VAR = 9
OUTCOLOR_VAR = 10
MAIN = 11
LABEL = 12
C_0_0 = 13
C_0_5 = 14
C_0_3 = 15
C_0_4 = 16
C_0_6 = 17
C_1_0 = 18
C_0_15 = 19
C_0_05 = 20
C_0_577 = 21
# temps start at 30

BOUND = 55

# Header
w(0x07230203)  # magic
w(0x00010000)  # version 1.0
w(0)           # generator
w(BOUND)       # bound
w(0)           # reserved

# OpCapability Shader (1)
op(2, 17); w(1)

# OpExtInstImport %1 "GLSL.std.450"
name_words = encode_string("GLSL.std.450")
op(2 + len(name_words), 11); w(EXT)
for nw in name_words:
    w(nw)

# OpMemoryModel Logical(0) GLSL450(1)
op(3, 14); w(0); w(1)

# OpEntryPoint Fragment(4) %main "main" %normal_var %outcolor_var
name_main = encode_string("main")
op(3 + len(name_main) + 2, 15); w(4); w(MAIN)
for nw in name_main:
    w(nw)
w(NORMAL_VAR); w(OUTCOLOR_VAR)

# OpExecutionMode %main OriginUpperLeft(7)
op(3, 16); w(MAIN); w(7)

# Decorations
# OpDecorate %normal_var Location 1
op(4, 71); w(NORMAL_VAR); w(30); w(1)
# OpDecorate %outcolor_var Location 0
op(4, 71); w(OUTCOLOR_VAR); w(30); w(0)

# Types
# OpTypeVoid %void
op(2, 19); w(VOID)
# OpTypeFunction %func_type %void
op(3, 33); w(FUNC_TYPE); w(VOID)
# OpTypeFloat %float 32
op(3, 22); w(FLOAT); w(32)
# OpTypeVector %vec3 %float 3
op(4, 23); w(VEC3); w(FLOAT); w(3)
# OpTypeVector %vec4 %float 4
op(4, 23); w(VEC4); w(FLOAT); w(4)
# OpTypePointer %ptr_in_v3 Input(1) %vec3
op(4, 32); w(PTR_IN_V3); w(1); w(VEC3)
# OpTypePointer %ptr_out_v4 Output(3) %vec4
op(4, 32); w(PTR_OUT_V4); w(3); w(VEC4)

# Constants
def const_float(id, val):
    op(4, 43); w(FLOAT); w(id); w(float_to_uint(val))

const_float(C_0_0, 0.0)
const_float(C_0_5, 0.5)
const_float(C_0_3, 0.3)
const_float(C_0_4, 0.4)
const_float(C_0_6, 0.6)
const_float(C_1_0, 1.0)
const_float(C_0_15, 0.15)
const_float(C_0_05, 0.05)
const_float(C_0_577, 0.57735)

# Variables (must be in correct section - before function)
# OpVariable %ptr_in_v3 %normal_var Input(1)
op(4, 59); w(PTR_IN_V3); w(NORMAL_VAR); w(1)
# OpVariable %ptr_out_v4 %outcolor_var Output(3)
op(4, 59); w(PTR_OUT_V4); w(OUTCOLOR_VAR); w(3)

# Function
# OpFunction %void %main None(0) %func_type
op(5, 54); w(VOID); w(MAIN); w(0); w(FUNC_TYPE)
# OpLabel %label
op(2, 248); w(LABEL)

T = 30  # temp counter

# %30 = OpLoad %vec3 %normal_var
op(4, 61); w(VEC3); w(T); w(NORMAL_VAR)  # T=30: raw normal
T += 1

# %31 = OpExtInst %vec3 %1 69(Normalize) %30
op(6, 12); w(VEC3); w(T); w(EXT); w(69); w(30)  # T=31: normalized normal
T += 1

# %32 = OpCompositeExtract %float %31 1  (N.y)
op(5, 81); w(FLOAT); w(T); w(31); w(1)  # T=32: N.y
T += 1

# %33 = OpFMul %float %32 %c_0_5
op(5, 133); w(FLOAT); w(T); w(32); w(C_0_5)  # T=33: N.y*0.5
T += 1

# %34 = OpFAdd %float %33 %c_0_5
op(5, 129); w(FLOAT); w(T); w(33); w(C_0_5)  # T=34: factor
T += 1

# sky = vec3(0.4, 0.6, 1.0)
# %35 = OpCompositeConstruct %vec3 %c_0_4 %c_0_6 %c_1_0
op(6, 80); w(VEC3); w(T); w(C_0_4); w(C_0_6); w(C_1_0)  # T=35: sky
T += 1

# ground = vec3(0.3, 0.15, 0.05)
# %36 = OpCompositeConstruct %vec3 %c_0_3 %c_0_15 %c_0_05
op(6, 80); w(VEC3); w(T); w(C_0_3); w(C_0_15); w(C_0_05)  # T=36: ground
T += 1

# diff = sky - ground
# %37 = OpFSub %vec3 %35 %36
op(5, 131); w(VEC3); w(T); w(35); w(36)  # T=37: diff
T += 1

# scaled = diff * factor  (VectorTimesScalar, opcode 142)
# %38 = OpVectorTimesScalar %vec3 %37 %34
op(5, 142); w(VEC3); w(T); w(37); w(34)  # T=38: scaled
T += 1

# ambient = ground + scaled
# %39 = OpFAdd %vec3 %36 %38
op(5, 129); w(VEC3); w(T); w(36); w(38)  # T=39: ambient
T += 1

# Light direction = normalize(1,1,1) ≈ (0.577, 0.577, 0.577)
# %40 = OpCompositeConstruct %vec3 %c_0_577 %c_0_577 %c_0_577
op(6, 80); w(VEC3); w(T); w(C_0_577); w(C_0_577); w(C_0_577)  # T=40: L
T += 1

# d = dot(N, L)
# %41 = OpDot %float %31 %40
op(5, 148); w(FLOAT); w(T); w(31); w(40)  # T=41: dot
T += 1

# max(0, d)
# %42 = OpExtInst %float %1 40(FMax) %c_0_0 %41
op(7, 12); w(FLOAT); w(T); w(EXT); w(40); w(C_0_0); w(41)  # T=42: max
T += 1

# 0.3 * d
# %43 = OpFMul %float %c_0_3 %42
op(5, 133); w(FLOAT); w(T); w(C_0_3); w(42)  # T=43: 0.3*d
T += 1

# vec3(0.3*d, 0.3*d, 0.3*d)
# %44 = OpCompositeConstruct %vec3 %43 %43 %43
op(6, 80); w(VEC3); w(T); w(43); w(43); w(43)  # T=44
T += 1

# color = ambient + diffuse
# %45 = OpFAdd %vec3 %39 %44
op(5, 129); w(VEC3); w(T); w(39); w(44)  # T=45: final color
T += 1

# Extract components for vec4 construction
# %46 = OpCompositeExtract %float %45 0
op(5, 81); w(FLOAT); w(T); w(45); w(0)  # T=46: r
T += 1
# %47 = OpCompositeExtract %float %45 1
op(5, 81); w(FLOAT); w(T); w(45); w(1)  # T=47: g
T += 1
# %48 = OpCompositeExtract %float %45 2
op(5, 81); w(FLOAT); w(T); w(45); w(2)  # T=48: b
T += 1

# %49 = OpCompositeConstruct %vec4 %46 %47 %48 %c_1_0
op(7, 80); w(VEC4); w(T); w(46); w(47); w(48); w(C_1_0)  # T=49: outColor
T += 1

# OpStore %outcolor_var %49
op(3, 62); w(OUTCOLOR_VAR); w(49)

# OpReturn
op(1, 253)

# OpFunctionEnd
op(1, 56)

# Convert to hex
data = b''
for word in words:
    data += struct.pack('<I', word)

print(data.hex())
