using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

class MinesweeperSolver
{
    static int H, W;
    static char[,] board;
    static int[] dx = {-1,-1,-1,0,0,1,1,1};
    static int[] dy = {-1,0,1,-1,1,-1,0,1};

    static bool InBounds(int r, int c) => r >= 0 && r < H && c >= 0 && c < W;

    static void Main()
    {
        var firstLine = Console.ReadLine().Trim().Split();
        H = int.Parse(firstLine[0]);
        W = int.Parse(firstLine[1]);
        board = new char[H, W];

        for (int r = 0; r < H; r++)
        {
            string line = Console.ReadLine();
            if (line.Length < W) line = line.PadRight(W);
            for (int c = 0; c < W; c++)
                board[r, c] = line[c];
        }

        var safe = new HashSet<long>();
        bool changed = true;

        while (changed)
        {
            changed = false;

            for (int r = 0; r < H; r++)
            for (int c = 0; c < W; c++)
            {
                if (board[r, c] < '0' || board[r, c] > '8') continue;
                int num = board[r, c] - '0';

                int mineCount = 0, unknownCount = 0;
                var unknowns = new List<(int r, int c)>();

                for (int d = 0; d < 8; d++)
                {
                    int nr = r + dy[d], nc = c + dx[d];
                    if (!InBounds(nr, nc)) continue;
                    if (board[nr, nc] == '*') mineCount++;
                    else if (board[nr, nc] == ' ')
                    {
                        long key = (long)nr * W + nc;
                        if (safe.Contains(key))
                            continue;
                        unknownCount++;
                        unknowns.Add((nr, nc));
                    }
                }

                if (unknownCount == 0) continue;

                // All mines accounted for -> remaining unknowns are safe
                if (mineCount == num)
                {
                    foreach (var (ur, uc) in unknowns)
                    {
                        long key = (long)ur * W + uc;
                        if (safe.Add(key))
                            changed = true;
                    }
                }
                // All unknowns are mines -> flag them
                else if (mineCount + unknownCount == num)
                {
                    foreach (var (ur, uc) in unknowns)
                    {
                        board[ur, uc] = '*';
                        changed = true;
                    }
                }
            }
        }

        // CSP for frontier cells
        SolveCSP(safe);

        var results = new List<(int x, int y)>();
        foreach (long key in safe)
        {
            int r = (int)(key / W);
            int c = (int)(key % W);
            if (board[r, c] == ' ')
                results.Add((c, r));
        }

        var sb = new System.Text.StringBuilder();
        sb.AppendLine(results.Count.ToString());
        foreach (var (x, y) in results)
            sb.Append(x).Append(' ').AppendLine(y.ToString());
        Console.Write(sb);
    }

    static void SolveCSP(HashSet<long> safe)
    {
        // Build constraints from numbered cells that still have unknowns
        var frontierUnknowns = new HashSet<long>();
        var constraints = new List<(int remaining, List<long> cells)>();

        for (int r = 0; r < H; r++)
        for (int c = 0; c < W; c++)
        {
            if (board[r, c] < '0' || board[r, c] > '8') continue;
            int num = board[r, c] - '0';
            int mineCount = 0;
            var unknowns = new List<long>();

            for (int d = 0; d < 8; d++)
            {
                int nr = r + dy[d], nc = c + dx[d];
                if (!InBounds(nr, nc)) continue;
                if (board[nr, nc] == '*') mineCount++;
                else if (board[nr, nc] == ' ' && !safe.Contains((long)nr * W + nc))
                    unknowns.Add((long)nr * W + nc);
            }

            if (unknowns.Count > 0)
            {
                constraints.Add((num - mineCount, unknowns));
                foreach (var u in unknowns) frontierUnknowns.Add(u);
            }
        }

        if (frontierUnknowns.Count == 0 || frontierUnknowns.Count > 200) return;

        // Subset/superset constraint propagation
        bool progressed = true;
        while (progressed)
        {
            progressed = false;
            for (int i = 0; i < constraints.Count; i++)
            {
                var (ri, ci) = constraints[i];
                if (ci.Count == 0) continue;
                for (int j = 0; j < constraints.Count; j++)
                {
                    if (i == j) continue;
                    var (rj, cj) = constraints[j];
                    if (cj.Count == 0) continue;

                    // Check if ci is subset of cj
                    if (ci.Count < cj.Count && ci.All(x => cj.Contains(x)))
                    {
                        var diff = cj.Where(x => !ci.Contains(x)).ToList();
                        int newRemaining = rj - ri;
                        if (newRemaining == 0)
                        {
                            foreach (var cell in diff)
                                safe.Add(cell);
                            constraints[j] = (0, new List<long>());
                            progressed = true;
                        }
                        else if (newRemaining == diff.Count)
                        {
                            foreach (var cell in diff)
                            {
                                int cr = (int)(cell / W), cc = (int)(cell % W);
                                board[cr, cc] = '*';
                            }
                            constraints[j] = (0, new List<long>());
                            progressed = true;
                        }
                        else if (diff.Count < cj.Count)
                        {
                            constraints[j] = (newRemaining, diff);
                            progressed = true;
                        }
                    }
                }
            }

            // Remove resolved cells from constraints
            for (int i = 0; i < constraints.Count; i++)
            {
                var (rem, cells) = constraints[i];
                var filtered = cells.Where(c => board[(int)(c / W), (int)(c % W)] == ' ' && !safe.Contains(c)).ToList();
                int minesRemoved = cells.Count - filtered.Count - cells.Count(c => safe.Contains(c));
                // Recalc: mines flagged among removed
                int flagged = 0;
                foreach (var c in cells)
                {
                    int cr = (int)(c / W), cc = (int)(c % W);
                    if (board[cr, cc] == '*') flagged++;
                }
                int newRem = rem - flagged;
                if (filtered.Count != cells.Count || newRem != rem)
                {
                    constraints[i] = (newRem, filtered);
                    if (filtered.Count > 0)
                    {
                        if (newRem == 0)
                        {
                            foreach (var cell in filtered) safe.Add(cell);
                            constraints[i] = (0, new List<long>());
                            progressed = true;
                        }
                        else if (newRem == filtered.Count)
                        {
                            foreach (var cell in filtered)
                            {
                                int cr = (int)(cell / W), cc = (int)(cell % W);
                                board[cr, cc] = '*';
                            }
                            constraints[i] = (0, new List<long>());
                            progressed = true;
                        }
                    }
                }
            }
        }

        // Rebuild frontier after propagation
        frontierUnknowns.RemoveWhere(u => safe.Contains(u) || board[(int)(u / W), (int)(u % W)] != ' ');
        if (frontierUnknowns.Count == 0 || frontierUnknowns.Count > 60) return;

        // Backtracking CSP: try all assignments
        var activeConstraints = constraints.Where(c => c.cells.Count > 0).ToList();
        if (activeConstraints.Count == 0) return;

        var cells_list = frontierUnknowns.ToList();
        int n = cells_list.Count;
        var cellIndex = new Dictionary<long, int>();
        for (int i = 0; i < n; i++) cellIndex[cells_list[i]] = i;

        // For each constraint, store which cell indices it references
        var cRefs = new List<(int remaining, int[] indices)>();
        foreach (var (rem, cls) in activeConstraints)
        {
            var idx = cls.Where(c => cellIndex.ContainsKey(c)).Select(c => cellIndex[c]).ToArray();
            if (idx.Length > 0)
                cRefs.Add((rem, idx));
        }

        int[] canBeMine = new int[n]; // count of solutions where cell is mine
        int[] canBeSafe = new int[n]; // count of solutions where cell is safe
        int[] assignment = new int[n]; // -1=unset, 0=safe, 1=mine
        for (int i = 0; i < n; i++) assignment[i] = -1;

        int solutionCount = 0;
        int maxSolutions = 100000;

        void Backtrack(int idx)
        {
            if (solutionCount >= maxSolutions) return;
            if (idx == n)
            {
                // Verify all constraints
                foreach (var (rem, indices) in cRefs)
                {
                    int mines = 0;
                    foreach (int i in indices) mines += assignment[i];
                    if (mines != rem) return;
                }
                solutionCount++;
                for (int i = 0; i < n; i++)
                {
                    if (assignment[i] == 1) canBeMine[i]++;
                    else canBeSafe[i]++;
                }
                return;
            }

            for (int val = 0; val <= 1; val++)
            {
                assignment[idx] = val;
                // Prune: check constraints where all referenced cells assigned
                bool ok = true;
                foreach (var (rem, indices) in cRefs)
                {
                    int mines = 0, unset = 0;
                    foreach (int i in indices)
                    {
                        if (assignment[i] == -1) unset++;
                        else mines += assignment[i];
                    }
                    if (mines > rem) { ok = false; break; }
                    if (mines + unset < rem) { ok = false; break; }
                }
                if (ok) Backtrack(idx + 1);
                if (solutionCount >= maxSolutions) { assignment[idx] = -1; return; }
            }
            assignment[idx] = -1;
        }

        Backtrack(0);

        if (solutionCount > 0 && solutionCount < maxSolutions)
        {
            for (int i = 0; i < n; i++)
            {
                if (canBeMine[i] == 0 && canBeSafe[i] > 0)
                    safe.Add(cells_list[i]);
            }
        }
    }
}
