; SPIR-V Geometry Shader - Silhouette/Boundary Edge Detection
; Input: TrianglesAdjacency (6 verts), Output: TriangleStrip, max 15 verts
               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %inColor %outColor
               OpExecutionMode %main InputTrianglesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 15
               OpExecutionMode %main Invocations 1

               ; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %inColor Location 2
               OpDecorate %outColor Location 0

               ; Types
      %void  = OpTypeVoid
   %voidfn   = OpTypeFunction %void
     %float  = OpTypeFloat 32
   %v3float  = OpTypeVector %float 3
   %v4float  = OpTypeVector %float 4
      %int   = OpTypeInt 32 1
     %uint   = OpTypeInt 32 0
      %bool  = OpTypeBool

               ; Constants
    %int_0   = OpConstant %int 0
    %int_1   = OpConstant %int 1
    %int_2   = OpConstant %int 2
    %int_3   = OpConstant %int 3
    %int_4   = OpConstant %int 4
    %int_5   = OpConstant %int 5
   %uint_6   = OpConstant %uint 6
  %float_0   = OpConstant %float 0.0
  %float_1   = OpConstant %float 1.0
 %float_03   = OpConstant %float 0.3
%float_thick = OpConstant %float 0.015
%float_nthick = OpConstant %float -0.015

               ; Composite constants
  %white     = OpConstantComposite %v3float %float_1 %float_1 %float_1

               ; Input struct and arrays
%gl_PerVertex_in = OpTypeStruct %v4float
 %arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_6
%ptr_in_arr = OpTypePointer Input %arr_pv_in
 %ptr_in_v4 = OpTypePointer Input %v4float
     %gl_in = OpVariable %ptr_in_arr Input

               ; Input color array
%arr_v3_6   = OpTypeArray %v3float %uint_6
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_6
%ptr_in_v3  = OpTypePointer Input %v3float
   %inColor = OpVariable %ptr_in_arr_v3 Input

               ; Output struct
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
    %gl_out = OpVariable %ptr_out_pv Output

               ; Output color
%ptr_out_v3 = OpTypePointer Output %v3float
  %outColor = OpVariable %ptr_out_v3 Output

               ; Function
     %main  = OpFunction %void None %voidfn
    %entry  = OpLabel

               ; Load all 6 positions
   %p0_ptr  = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
       %p0  = OpLoad %v4float %p0_ptr
   %p1_ptr  = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
       %p1  = OpLoad %v4float %p1_ptr
   %p2_ptr  = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
       %p2  = OpLoad %v4float %p2_ptr
   %p3_ptr  = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
       %p3  = OpLoad %v4float %p3_ptr
   %p4_ptr  = OpAccessChain %ptr_in_v4 %gl_in %int_4 %int_0
       %p4  = OpLoad %v4float %p4_ptr
   %p5_ptr  = OpAccessChain %ptr_in_v4 %gl_in %int_5 %int_0
       %p5  = OpLoad %v4float %p5_ptr

               ; Load colors for main triangle vertices (0, 2, 4)
  %c0_ptr   = OpAccessChain %ptr_in_v3 %inColor %int_0
      %c0   = OpLoad %v3float %c0_ptr
  %c2_ptr   = OpAccessChain %ptr_in_v3 %inColor %int_2
      %c2   = OpLoad %v3float %c2_ptr
  %c4_ptr   = OpAccessChain %ptr_in_v3 %inColor %int_4
      %c4   = OpLoad %v3float %c4_ptr

               ; === Emit the filled triangle (dimmed) ===
               ; Vertex 0
  %dc0      = OpVectorTimesScalar %v3float %c0 %float_03
               OpStore %outColor %dc0
  %gout0    = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %gout0 %p0
               OpEmitVertex

               ; Vertex 2
  %dc2      = OpVectorTimesScalar %v3float %c2 %float_03
               OpStore %outColor %dc2
               OpStore %gout0 %p2
               OpEmitVertex

               ; Vertex 4
  %dc4      = OpVectorTimesScalar %v3float %c4 %float_03
               OpStore %outColor %dc4
               OpStore %gout0 %p4
               OpEmitVertex
               OpEndPrimitive

               ; === Edge 0->2 (adjacent vertex 1) ===
               ; Check boundary: if p1 == p0 or p1 == p2 or p1 == p4
  %diff1_0  = OpFSub %v4float %p1 %p0
 %dot1_0    = OpDot %float %diff1_0 %diff1_0
  %diff1_2  = OpFSub %v4float %p1 %p2
 %dot1_2    = OpDot %float %diff1_2 %diff1_2
  %diff1_4  = OpFSub %v4float %p1 %p4
 %dot1_4    = OpDot %float %diff1_4 %diff1_4
 %eps       = OpConstant %float 0.0001
 %b1_0      = OpFOrdLessThan %bool %dot1_0 %eps
 %b1_2      = OpFOrdLessThan %bool %dot1_2 %eps
 %b1_4      = OpFOrdLessThan %bool %dot1_4 %eps
 %b1_02     = OpLogicalOr %bool %b1_0 %b1_2
 %bound1    = OpLogicalOr %bool %b1_02 %b1_4
               OpSelectionMerge %edge1_end None
               OpBranchConditional %bound1 %edge1_emit %edge1_end

 %edge1_emit = OpLabel
               ; Compute perpendicular offset for edge 0->2
  %e1_dir4  = OpFSub %v4float %p2 %p0
  %e1_dx    = OpCompositeExtract %float %e1_dir4 0
  %e1_dy    = OpCompositeExtract %float %e1_dir4 1
               ; perp = normalize(-dy, dx) * thickness
  %e1_neg_dy = OpFNegate %float %e1_dy
  %e1_len   = OpExtInst %float %1 Length %e1_dir4
  %e1_inv   = OpFDiv %float %float_thick %e1_len
  %e1_ninv  = OpFNegate %float %e1_inv
  %e1_ox    = OpFMul %float %e1_neg_dy %e1_inv
  %e1_oy    = OpFMul %float %e1_dx %e1_inv
  %e1_nox   = OpFMul %float %e1_neg_dy %e1_ninv
  %e1_noy   = OpFMul %float %e1_dx %e1_ninv

               ; 4 vertices of the quad
  %p0x      = OpCompositeExtract %float %p0 0
  %p0y      = OpCompositeExtract %float %p0 1
  %p0z      = OpCompositeExtract %float %p0 2
  %p0w      = OpCompositeExtract %float %p0 3
  %p2x      = OpCompositeExtract %float %p2 0
  %p2y      = OpCompositeExtract %float %p2 1
  %p2z      = OpCompositeExtract %float %p2 2
  %p2w      = OpCompositeExtract %float %p2 3

  %q1x      = OpFAdd %float %p0x %e1_ox
  %q1y      = OpFAdd %float %p0y %e1_oy
  %q1       = OpCompositeConstruct %v4float %q1x %q1y %p0z %p0w
  %q2x      = OpFAdd %float %p0x %e1_nox
  %q2y      = OpFAdd %float %p0y %e1_noy
  %q2       = OpCompositeConstruct %v4float %q2x %q2y %p0z %p0w
  %q3x      = OpFAdd %float %p2x %e1_ox
  %q3y      = OpFAdd %float %p2y %e1_oy
  %q3       = OpCompositeConstruct %v4float %q3x %q3y %p2z %p2w
  %q4x      = OpFAdd %float %p2x %e1_nox
  %q4y      = OpFAdd %float %p2y %e1_noy
  %q4       = OpCompositeConstruct %v4float %q4x %q4y %p2z %p2w

               OpStore %outColor %white
  %go1      = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %go1 %q1
               OpEmitVertex
               OpStore %go1 %q2
               OpEmitVertex
               OpStore %go1 %q3
               OpEmitVertex
               OpStore %go1 %q4
               OpEmitVertex
               OpEndPrimitive
               OpBranch %edge1_end
 %edge1_end = OpLabel

               ; === Edge 2->4 (adjacent vertex 3) ===
  %diff3_0  = OpFSub %v4float %p3 %p0
 %dot3_0    = OpDot %float %diff3_0 %diff3_0
  %diff3_2  = OpFSub %v4float %p3 %p2
 %dot3_2    = OpDot %float %diff3_2 %diff3_2
  %diff3_4  = OpFSub %v4float %p3 %p4
 %dot3_4    = OpDot %float %diff3_4 %diff3_4
 %b3_0      = OpFOrdLessThan %bool %dot3_0 %eps
 %b3_2      = OpFOrdLessThan %bool %dot3_2 %eps
 %b3_4      = OpFOrdLessThan %bool %dot3_4 %eps
 %b3_02     = OpLogicalOr %bool %b3_0 %b3_2
 %bound3    = OpLogicalOr %bool %b3_02 %b3_4
               OpSelectionMerge %edge2_end None
               OpBranchConditional %bound3 %edge2_emit %edge2_end

 %edge2_emit = OpLabel
  %e2_dir4  = OpFSub %v4float %p4 %p2
  %e2_dx    = OpCompositeExtract %float %e2_dir4 0
  %e2_dy    = OpCompositeExtract %float %e2_dir4 1
  %e2_neg_dy = OpFNegate %float %e2_dy
  %e2_len   = OpExtInst %float %1 Length %e2_dir4
  %e2_inv   = OpFDiv %float %float_thick %e2_len
  %e2_ninv  = OpFNegate %float %e2_inv
  %e2_ox    = OpFMul %float %e2_neg_dy %e2_inv
  %e2_oy    = OpFMul %float %e2_dx %e2_inv
  %e2_nox   = OpFMul %float %e2_neg_dy %e2_ninv
  %e2_noy   = OpFMul %float %e2_dx %e2_ninv

  %p2x2     = OpCompositeExtract %float %p2 0
  %p2y2     = OpCompositeExtract %float %p2 1
  %p2z2     = OpCompositeExtract %float %p2 2
  %p2w2     = OpCompositeExtract %float %p2 3
  %p4x      = OpCompositeExtract %float %p4 0
  %p4y      = OpCompositeExtract %float %p4 1
  %p4z      = OpCompositeExtract %float %p4 2
  %p4w      = OpCompositeExtract %float %p4 3

  %r1x      = OpFAdd %float %p2x2 %e2_ox
  %r1y      = OpFAdd %float %p2y2 %e2_oy
  %r1       = OpCompositeConstruct %v4float %r1x %r1y %p2z2 %p2w2
  %r2x      = OpFAdd %float %p2x2 %e2_nox
  %r2y      = OpFAdd %float %p2y2 %e2_noy
  %r2       = OpCompositeConstruct %v4float %r2x %r2y %p2z2 %p2w2
  %r3x      = OpFAdd %float %p4x %e2_ox
  %r3y      = OpFAdd %float %p4y %e2_oy
  %r3       = OpCompositeConstruct %v4float %r3x %r3y %p4z %p4w
  %r4x      = OpFAdd %float %p4x %e2_nox
  %r4y      = OpFAdd %float %p4y %e2_noy
  %r4       = OpCompositeConstruct %v4float %r4x %r4y %p4z %p4w

               OpStore %outColor %white
  %go2      = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %go2 %r1
               OpEmitVertex
               OpStore %go2 %r2
               OpEmitVertex
               OpStore %go2 %r3
               OpEmitVertex
               OpStore %go2 %r4
               OpEmitVertex
               OpEndPrimitive
               OpBranch %edge2_end
 %edge2_end = OpLabel

               ; === Edge 4->0 (adjacent vertex 5) ===
  %diff5_0  = OpFSub %v4float %p5 %p0
 %dot5_0    = OpDot %float %diff5_0 %diff5_0
  %diff5_2  = OpFSub %v4float %p5 %p2
 %dot5_2    = OpDot %float %diff5_2 %diff5_2
  %diff5_4  = OpFSub %v4float %p5 %p4
 %dot5_4    = OpDot %float %diff5_4 %diff5_4
 %b5_0      = OpFOrdLessThan %bool %dot5_0 %eps
 %b5_2      = OpFOrdLessThan %bool %dot5_2 %eps
 %b5_4      = OpFOrdLessThan %bool %dot5_4 %eps
 %b5_02     = OpLogicalOr %bool %b5_0 %b5_2
 %bound5    = OpLogicalOr %bool %b5_02 %b5_4
               OpSelectionMerge %edge3_end None
               OpBranchConditional %bound5 %edge3_emit %edge3_end

 %edge3_emit = OpLabel
  %e3_dir4  = OpFSub %v4float %p0 %p4
  %e3_dx    = OpCompositeExtract %float %e3_dir4 0
  %e3_dy    = OpCompositeExtract %float %e3_dir4 1
  %e3_neg_dy = OpFNegate %float %e3_dy
  %e3_len   = OpExtInst %float %1 Length %e3_dir4
  %e3_inv   = OpFDiv %float %float_thick %e3_len
  %e3_ninv  = OpFNegate %float %e3_inv
  %e3_ox    = OpFMul %float %e3_neg_dy %e3_inv
  %e3_oy    = OpFMul %float %e3_dx %e3_inv
  %e3_nox   = OpFMul %float %e3_neg_dy %e3_ninv
  %e3_noy   = OpFMul %float %e3_dx %e3_ninv

  %p4x2     = OpCompositeExtract %float %p4 0
  %p4y2     = OpCompositeExtract %float %p4 1
  %p4z2     = OpCompositeExtract %float %p4 2
  %p4w2     = OpCompositeExtract %float %p4 3
  %p0x2     = OpCompositeExtract %float %p0 0
  %p0y2     = OpCompositeExtract %float %p0 1
  %p0z2     = OpCompositeExtract %float %p0 2
  %p0w2     = OpCompositeExtract %float %p0 3

  %s1x      = OpFAdd %float %p4x2 %e3_ox
  %s1y      = OpFAdd %float %p4y2 %e3_oy
  %s1       = OpCompositeConstruct %v4float %s1x %s1y %p4z2 %p4w2
  %s2x      = OpFAdd %float %p4x2 %e3_nox
  %s2y      = OpFAdd %float %p4y2 %e3_noy
  %s2       = OpCompositeConstruct %v4float %s2x %s2y %p4z2 %p4w2
  %s3x      = OpFAdd %float %p0x2 %e3_ox
  %s3y      = OpFAdd %float %p0y2 %e3_oy
  %s3       = OpCompositeConstruct %v4float %s3x %s3y %p0z2 %p0w2
  %s4x      = OpFAdd %float %p0x2 %e3_nox
  %s4y      = OpFAdd %float %p0y2 %e3_noy
  %s4       = OpCompositeConstruct %v4float %s4x %s4y %p0z2 %p0w2

               OpStore %outColor %white
  %go3      = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %go3 %s1
               OpEmitVertex
               OpStore %go3 %s2
               OpEmitVertex
               OpStore %go3 %s3
               OpEmitVertex
               OpStore %go3 %s4
               OpEmitVertex
               OpEndPrimitive
               OpBranch %edge3_end
 %edge3_end = OpLabel

               OpReturn
               OpFunctionEnd
