import math

_prev_pos = None
_prev_vel = None
_time = 0.0

def _median_valid(values_and_valids):
    valid = [v for v, ok in values_and_valids if ok]
    if not valid:
        return None
    valid.sort()
    n = len(valid)
    return valid[n // 2]

def _get_rel_pos(state):
    px = _median_valid([
        (state['radar_rel_pos_x_m'], state['radar_rel_x_valid']),
        (state['lidar_rel_pos_x_m'], state['lidar_rel_x_valid']),
    ])
    py = _median_valid([
        (state['radar_rel_pos_y_m'], state['radar_rel_y_valid']),
        (state['lidar_rel_pos_y_m'], state['lidar_rel_y_valid']),
    ])
    pz = _median_valid([
        (state['radar_rel_pos_z_m'], state['radar_rel_z_valid']),
        (state['lidar_rel_pos_z_m'], state['lidar_rel_z_valid']),
    ])
    return px, py, pz

def _get_rel_vel(state):
    vx = _median_valid([
        (state['radar_rel_vel_x_ms'], state['radar_relv_x_valid']),
        (state['lidar_rel_vel_x_ms'], state['lidar_relv_x_valid']),
    ])
    vy = _median_valid([
        (state['radar_rel_vel_y_ms'], state['radar_relv_y_valid']),
        (state['lidar_rel_vel_y_ms'], state['lidar_relv_y_valid']),
    ])
    vz = _median_valid([
        (state['radar_rel_vel_z_ms'], state['radar_relv_z_valid']),
        (state['lidar_rel_vel_z_ms'], state['lidar_relv_z_valid']),
    ])
    return vx, vy, vz

def _clamp(v, lo, hi):
    return max(lo, min(hi, v))

def autopilot_step(state: dict, dt: float) -> dict:
    global _prev_pos, _prev_vel, _time
    _time += dt

    MU = 3.986004418e14
    R_EARTH = 6371000.0
    DEFAULT_ALT = 400000.0

    alt = None
    if state.get('ground_altitude_km_valid'):
        alt = state['ground_altitude_km'] * 1000.0
    if alt is None:
        alt = DEFAULT_ALT

    sma = None
    if state.get('ground_semi_major_axis_m_valid'):
        sma = state['ground_semi_major_axis_m']
    if sma is None:
        sma = R_EARTH + alt

    n = math.sqrt(MU / sma**3)

    px, py, pz = _get_rel_pos(state)
    vx, vy, vz = _get_rel_vel(state)

    if px is None or py is None or pz is None:
        if _prev_pos is not None:
            px, py, pz = _prev_pos
        else:
            px, py, pz = 0.0, -200.0, 0.0

    if vx is None or vy is None or vz is None:
        if _prev_vel is not None:
            vx, vy, vz = _prev_vel
        else:
            vx, vy, vz = 0.0, 0.0, 0.0

    _prev_pos = (px, py, pz)
    _prev_vel = (vx, vy, vz)

    rng = math.sqrt(px**2 + py**2 + pz**2)
    speed = math.sqrt(vx**2 + vy**2 + vz**2)

    fuel_pct = state.get('fuel_remaining_pct', 100)
    low_fuel = fuel_pct < 15

    thrust_x = 0.0
    thrust_y = 0.0
    thrust_z = 0.0

    if rng > 200:
        # Far range: CW-inspired approach
        # Along-track: to close negative y (behind), thrust retrograde (negative y) to lower orbit
        # This is counter-intuitive but correct for orbital mechanics

        # Desired closing velocity scales with distance
        max_close_speed = 2.0 if not low_fuel else 1.0
        desired_vy = _clamp(-py * 0.003, -max_close_speed, max_close_speed)

        # CW coupling: radial position causes along-track drift
        # To handle x (radial) offset: use CW steady-state
        # For y-closing: we want vy ~ desired_vy
        # CW: to move in -y direction, thrust +y (retrograde if behind) is WRONG
        # Actually: thrust_y negative (retrograde) lowers orbit -> speeds up -> closes gap

        vy_err = desired_vy - vy
        thrust_y = _clamp(vy_err * 0.5, -1.0, 1.0)

        # CW correction for radial: want px -> 0
        # In CW, x oscillates. Use x and vx to compute correction.
        # Desired: x=0, vx=0
        # CW x-accel = 3n²x + 2n*vy + ax  -> for hover at x=0: ax = -3n²x - 2n*vy
        # But we also want to damp vx
        desired_vx = -px * 0.002
        vx_err = desired_vx - vx
        # CW coupling term
        cw_x_accel = 3 * n * n * px + 2 * n * vy
        thrust_x = _clamp(vx_err * 0.3 - cw_x_accel * 0.01, -0.5, 0.5)

        # Cross-track: simple spring-damper (decoupled in CW)
        desired_vz = -pz * 0.002
        vz_err = desired_vz - vz
        thrust_z = _clamp(vz_err * 0.3, -0.5, 0.5)

    elif rng > 50:
        # Medium range: gentler approach, V-bar preferred
        max_close_speed = 0.8 if not low_fuel else 0.4

        desired_vy = _clamp(-py * 0.005, -max_close_speed, max_close_speed)
        vy_err = desired_vy - vy
        thrust_y = _clamp(vy_err * 0.4, -0.5, 0.5)

        desired_vx = -px * 0.005
        vx_err = desired_vx - vx
        cw_x_accel = 3 * n * n * px + 2 * n * vy
        thrust_x = _clamp(vx_err * 0.3 - cw_x_accel * 0.005, -0.3, 0.3)

        desired_vz = -pz * 0.005
        vz_err = desired_vz - vz
        thrust_z = _clamp(vz_err * 0.3, -0.3, 0.3)

    elif rng > 10:
        # Close range: linear regime, direct corrections
        max_close_speed = 0.3 if not low_fuel else 0.15

        for axis, p, v in [('x', px, vx), ('y', py, vy), ('z', pz, vz)]:
            desired_v = _clamp(-p * 0.01, -max_close_speed, max_close_speed)
            v_err = desired_v - v
            t = _clamp(v_err * 0.5, -0.2, 0.2)
            if axis == 'x':
                thrust_x = t
            elif axis == 'y':
                thrust_y = t
            else:
                thrust_z = t

    else:
        # Final approach: very gentle
        max_close_speed = 0.08

        for axis, p, v in [('x', px, vx), ('y', py, vy), ('z', pz, vz)]:
            desired_v = _clamp(-p * 0.02, -max_close_speed, max_close_speed)
            v_err = desired_v - v
            t = _clamp(v_err * 0.8, -0.15, 0.15)
            if axis == 'x':
                thrust_x = t
            elif axis == 'y':
                thrust_y = t
            else:
                thrust_z = t

    # Safety: if closing too fast, brake
    if rng < 50 and speed > 1.0:
        brake_factor = min(1.0, speed / 2.0)
        if rng > 0.01:
            thrust_x = _clamp(vx / speed * (-brake_factor), -1.0, 1.0)
            thrust_y = _clamp(vy / speed * (-brake_factor), -1.0, 1.0)
            thrust_z = _clamp(vz / speed * (-brake_factor), -1.0, 1.0)

    if rng < 20 and speed > 0.5:
        brake_factor = min(1.0, speed)
        if rng > 0.01:
            thrust_x = _clamp(vx / speed * (-brake_factor), -1.0, 1.0)
            thrust_y = _clamp(vy / speed * (-brake_factor), -1.0, 1.0)
            thrust_z = _clamp(vz / speed * (-brake_factor), -1.0, 1.0)

    # Low fuel: scale down all thrusts
    if low_fuel:
        scale = 0.5
        thrust_x *= scale
        thrust_y *= scale
        thrust_z *= scale

    thrust_x = _clamp(thrust_x, -1.0, 1.0)
    thrust_y = _clamp(thrust_y, -1.0, 1.0)
    thrust_z = _clamp(thrust_z, -1.0, 1.0)

    # Attitude: damp rotation rates
    gyro_vals = {}
    for axis in ['pitch', 'roll', 'yaw']:
        readings = []
        for i in [1, 2, 3]:
            key_v = f'gyro_{i}_{axis}_dps'
            key_ok = f'gyro_{i}_{axis}_valid'
            if state.get(key_ok):
                readings.append(state[key_v])
        if readings:
            readings.sort()
            gyro_vals[axis] = readings[len(readings)//2]
        else:
            gyro_vals[axis] = 0.0

    max_att_rate = 0.035
    att_roll = _clamp(-gyro_vals['roll'] * math.pi / 180.0 * 0.5, -max_att_rate, max_att_rate)
    att_pitch = _clamp(-gyro_vals['pitch'] * math.pi / 180.0 * 0.5, -max_att_rate, max_att_rate)
    att_yaw = _clamp(-gyro_vals['yaw'] * math.pi / 180.0 * 0.5, -max_att_rate, max_att_rate)

    return {
        'thrust_x': thrust_x,
        'thrust_y': thrust_y,
        'thrust_z': thrust_z,
        'att_rate_roll': att_roll,
        'att_rate_pitch': att_pitch,
        'att_rate_yaw': att_yaw,
    }
