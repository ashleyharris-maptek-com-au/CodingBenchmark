#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <cstring>

struct Rect {
    int w, h, idx;
};

struct Placement {
    int board, x, y, rotated;
};

struct Node {
    int x, y, w, h;
};

struct Board {
    std::vector<Node> freeRects;
    int W, H;

    Board(int W, int H) : W(W), H(H) {
        freeRects.push_back({0, 0, W, H});
    }

    int scoreRect(int w, int h, int& bestX, int& bestY) const {
        int bestShort = INT_MAX;
        bestX = bestY = -1;
        for (auto& f : freeRects) {
            if (w <= f.w && h <= f.h) {
                int leftover = std::min(f.w - w, f.h - h);
                if (leftover < bestShort || (leftover == bestShort && f.y < bestY)) {
                    bestShort = leftover;
                    bestX = f.x; bestY = f.y;
                }
            }
        }
        return bestShort;
    }

    void place(int x, int y, int w, int h) {
        Node used = {x, y, w, h};
        size_t n = freeRects.size();
        for (size_t i = 0; i < n; ) {
            if (splitFreeNode(freeRects[i], used)) {
                freeRects[i] = freeRects.back();
                freeRects.pop_back();
                n--;
            } else {
                i++;
            }
        }
        pruneFreeList();
    }

    bool splitFreeNode(const Node& f, const Node& used) {
        if (used.x >= f.x + f.w || used.x + used.w <= f.x ||
            used.y >= f.y + f.h || used.y + used.h <= f.y)
            return false;

        if (used.x > f.x && used.x < f.x + f.w) {
            Node n = f; n.w = used.x - f.x;
            freeRects.push_back(n);
        }
        if (used.x + used.w < f.x + f.w) {
            Node n = f; n.w = f.x + f.w - (used.x + used.w); n.x = used.x + used.w;
            freeRects.push_back(n);
        }
        if (used.y > f.y && used.y < f.y + f.h) {
            Node n = f; n.h = used.y - f.y;
            freeRects.push_back(n);
        }
        if (used.y + used.h < f.y + f.h) {
            Node n = f; n.h = f.y + f.h - (used.y + used.h); n.y = used.y + used.h;
            freeRects.push_back(n);
        }
        return true;
    }

    void pruneFreeList() {
        for (size_t i = 0; i < freeRects.size(); i++) {
            for (size_t j = i + 1; j < freeRects.size(); ) {
                if (contains(freeRects[j], freeRects[i])) {
                    freeRects[i] = freeRects.back();
                    freeRects.pop_back();
                    if (i < freeRects.size()) {
                        j = i + 1;
                    } else break;
                    continue;
                }
                if (contains(freeRects[i], freeRects[j])) {
                    freeRects[j] = freeRects.back();
                    freeRects.pop_back();
                    continue;
                }
                j++;
            }
        }
    }

    static bool contains(const Node& a, const Node& b) {
        return b.x >= a.x && b.y >= a.y &&
               b.x + b.w <= a.x + a.w && b.y + b.h <= a.y + a.h;
    }
};

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int N, BW, BH;
    std::cin >> N >> BW >> BH;

    std::vector<Rect> rects(N);
    for (int i = 0; i < N; i++) {
        std::cin >> rects[i].w >> rects[i].h;
        rects[i].idx = i;
    }

    // Sort by max(w,h) desc, then area desc
    std::vector<int> order(N);
    std::iota(order.begin(), order.end(), 0);
    std::sort(order.begin(), order.end(), [&](int a, int b) {
        int ma = std::max(rects[a].w, rects[a].h);
        int mb = std::max(rects[b].w, rects[b].h);
        if (ma != mb) return ma > mb;
        return (long long)rects[a].w * rects[a].h > (long long)rects[b].w * rects[b].h;
    });

    std::vector<Placement> placements(N);
    std::vector<Board> boards;

    for (int oi = 0; oi < N; oi++) {
        int i = order[oi];
        int w = rects[i].w, h = rects[i].h;

        int bestBoard = -1, bestScore = INT_MAX, bestX = -1, bestY = -1, bestRot = 0;

        for (int b = 0; b < (int)boards.size(); b++) {
            int x1, y1, x2, y2;
            int s1 = boards[b].scoreRect(w, h, x1, y1);
            int s2 = INT_MAX;
            if (w != h) s2 = boards[b].scoreRect(h, w, x2, y2);

            if (s1 < bestScore) { bestScore = s1; bestBoard = b; bestX = x1; bestY = y1; bestRot = 0; }
            if (s2 < bestScore) { bestScore = s2; bestBoard = b; bestX = x2; bestY = y2; bestRot = 1; }
        }

        if (bestBoard == -1) {
            boards.emplace_back(BW, BH);
            bestBoard = (int)boards.size() - 1;
            int x1, y1, x2, y2;
            int s1 = boards[bestBoard].scoreRect(w, h, x1, y1);
            int s2 = INT_MAX;
            if (w != h) s2 = boards[bestBoard].scoreRect(h, w, x2, y2);
            if (s1 <= s2) { bestX = x1; bestY = y1; bestRot = 0; }
            else { bestX = x2; bestY = y2; bestRot = 1; }
        }

        int pw = bestRot ? h : w;
        int ph = bestRot ? w : h;
        boards[bestBoard].place(bestX, bestY, pw, ph);
        placements[i] = {bestBoard, bestX, bestY, bestRot};
    }

    std::cout << boards.size() << '\n';
    for (int i = 0; i < N; i++) {
        auto& p = placements[i];
        std::cout << p.board << ' ' << p.x << ' ' << p.y << ' ' << p.rotated << '\n';
    }
    return 0;
}
