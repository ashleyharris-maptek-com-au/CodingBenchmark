#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <cstring>

struct Rect {
    int w, h, idx;
};

struct Placement {
    int board, x, y, rotated;
};

// Maximal rectangles bin packing
struct MaxRectsBin {
    int W, H;
    struct FreeRect { int x, y, w, h; };
    std::vector<FreeRect> free_rects;

    MaxRectsBin(int W, int H) : W(W), H(H) {
        free_rects.push_back({0, 0, W, H});
    }

    // Best short side fit
    bool place(int rw, int rh, int &ox, int &oy) {
        int bestSSF = INT_MAX, bestLSF = INT_MAX;
        int bestIdx = -1;
        for (int i = 0; i < (int)free_rects.size(); i++) {
            auto &f = free_rects[i];
            if (rw <= f.w && rh <= f.h) {
                int ssf = std::min(f.w - rw, f.h - rh);
                int lsf = std::max(f.w - rw, f.h - rh);
                if (ssf < bestSSF || (ssf == bestSSF && lsf < bestLSF)) {
                    bestSSF = ssf; bestLSF = lsf;
                    bestIdx = i; ox = f.x; oy = f.y;
                }
            }
        }
        if (bestIdx == -1) return false;
        splitFreeRect(bestIdx, ox, oy, rw, rh);
        return true;
    }

    void splitFreeRect(int /*bestIdx*/, int px, int py, int rw, int rh) {
        // Split all overlapping free rects
        int n = (int)free_rects.size();
        for (int i = 0; i < n; i++) {
            auto &f = free_rects[i];
            // Check overlap with placed rect
            if (px >= f.x + f.w || px + rw <= f.x || py >= f.y + f.h || py + rh <= f.y)
                continue;
            // Split into up to 4 new rects
            if (px > f.x)
                free_rects.push_back({f.x, f.y, px - f.x, f.h});
            if (px + rw < f.x + f.w)
                free_rects.push_back({px + rw, f.y, f.x + f.w - px - rw, f.h});
            if (py > f.y)
                free_rects.push_back({f.x, f.y, f.w, py - f.y});
            if (py + rh < f.y + f.h)
                free_rects.push_back({f.x, py + rh, f.w, f.y + f.h - py - rh});
            // Remove this free rect
            free_rects[i] = free_rects[n - 1];
            free_rects.resize(n - 1);
            n--; i--;
        }
        // Remove contained rects
        for (int i = 0; i < (int)free_rects.size(); i++) {
            for (int j = i + 1; j < (int)free_rects.size(); ) {
                if (free_rects[i].x <= free_rects[j].x &&
                    free_rects[i].y <= free_rects[j].y &&
                    free_rects[i].x + free_rects[i].w >= free_rects[j].x + free_rects[j].w &&
                    free_rects[i].y + free_rects[i].h >= free_rects[j].y + free_rects[j].h) {
                    free_rects[j] = free_rects.back();
                    free_rects.pop_back();
                } else if (free_rects[j].x <= free_rects[i].x &&
                           free_rects[j].y <= free_rects[i].y &&
                           free_rects[j].x + free_rects[j].w >= free_rects[i].x + free_rects[i].w &&
                           free_rects[j].y + free_rects[j].h >= free_rects[i].y + free_rects[i].h) {
                    free_rects[i] = free_rects[j];
                    free_rects[j] = free_rects.back();
                    free_rects.pop_back();
                    j = i + 1;
                } else {
                    j++;
                }
            }
        }
    }
};

// Faster shelf-based packing for large instances
struct ShelfBin {
    int W, H;
    struct Shelf { int y, h, x_used; };
    std::vector<Shelf> shelves;

    ShelfBin(int W, int H) : W(W), H(H) {}

    bool place(int rw, int rh, int &ox, int &oy) {
        // Try existing shelves - best height fit
        int bestIdx = -1, bestWaste = INT_MAX;
        for (int i = 0; i < (int)shelves.size(); i++) {
            if (rw <= W - shelves[i].x_used && rh <= shelves[i].h) {
                int waste = shelves[i].h - rh;
                if (waste < bestWaste) { bestWaste = waste; bestIdx = i; }
            }
        }
        if (bestIdx >= 0) {
            ox = shelves[bestIdx].x_used;
            oy = shelves[bestIdx].y;
            shelves[bestIdx].x_used += rw;
            return true;
        }
        // New shelf
        int topY = 0;
        for (auto &s : shelves) topY = std::max(topY, s.y + s.h);
        if (topY + rh > H) return false;
        shelves.push_back({topY, rh, rw});
        ox = 0; oy = topY;
        return true;
    }
};

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int N, BW, BH;
    std::cin >> N >> BW >> BH;

    std::vector<Rect> rects(N);
    for (int i = 0; i < N; i++) {
        std::cin >> rects[i].w >> rects[i].h;
        rects[i].idx = i;
    }

    std::vector<Placement> placements(N);
    bool useShelf = (N > 2000);

    // Sort by max(w,h) descending then area descending
    std::vector<int> order(N);
    std::iota(order.begin(), order.end(), 0);
    std::sort(order.begin(), order.end(), [&](int a, int b) {
        int ma = std::max(rects[a].w, rects[a].h), mb = std::max(rects[b].w, rects[b].h);
        if (ma != mb) return ma > mb;
        return (long long)rects[a].w * rects[a].h > (long long)rects[b].w * rects[b].h;
    });

    if (useShelf) {
        std::vector<ShelfBin> bins;
        for (int oi : order) {
            int w = rects[oi].w, h = rects[oi].h;
            bool placed = false;
            for (int b = 0; b < (int)bins.size(); b++) {
                int ox, oy;
                // Try both orientations
                if (w <= BW && h <= BH && bins[b].place(w, h, ox, oy)) {
                    placements[oi] = {b, ox, oy, 0}; placed = true; break;
                }
                if (h <= BW && w <= BH && bins[b].place(h, w, ox, oy)) {
                    placements[oi] = {b, ox, oy, 1}; placed = true; break;
                }
            }
            if (!placed) {
                bins.emplace_back(BW, BH);
                int b = (int)bins.size() - 1;
                int ox, oy;
                if (w <= BW && h <= BH && bins[b].place(w, h, ox, oy)) {
                    placements[oi] = {b, ox, oy, 0};
                } else if (h <= BW && w <= BH && bins[b].place(h, w, ox, oy)) {
                    placements[oi] = {b, ox, oy, 1};
                }
            }
        }
        int numBoards = (int)bins.size();
        std::cout << numBoards << '\n';
        for (int i = 0; i < N; i++) {
            auto &p = placements[i];
            std::cout << p.board << ' ' << p.x << ' ' << p.y << ' ' << p.rotated << '\n';
        }
    } else {
        // Use maxrects for better quality on smaller instances
        // Try multiple sort orders and pick best
        auto tryPack = [&](std::vector<int> &ord) -> std::pair<int, std::vector<Placement>> {
            std::vector<MaxRectsBin> bins;
            std::vector<Placement> pl(N);
            for (int oi : ord) {
                int w = rects[oi].w, h = rects[oi].h;
                bool placed = false;
                for (int b = 0; b < (int)bins.size(); b++) {
                    int ox, oy;
                    // Try both orientations, pick better fit
                    bool ok1 = false, ok2 = false;
                    int ox1, oy1, ox2, oy2;
                    // We need to test without modifying - make copies for scoring
                    // Just try normal first, then rotated
                    if (w <= BW && h <= BH) {
                        MaxRectsBin tmp = bins[b];
                        if (tmp.place(w, h, ox1, oy1)) ok1 = true;
                    }
                    if (h <= BW && w <= BH && h != w) {
                        MaxRectsBin tmp = bins[b];
                        if (tmp.place(h, w, ox2, oy2)) ok2 = true;
                    }
                    if (ok1 || ok2) {
                        if (ok1 && ok2) {
                            // Pick the one with better position (lower y, then lower x)
                            if (oy1 < oy2 || (oy1 == oy2 && ox1 <= ox2)) {
                                bins[b].place(w, h, ox, oy);
                                pl[oi] = {b, ox, oy, 0};
                            } else {
                                bins[b].place(h, w, ox, oy);
                                pl[oi] = {b, ox, oy, 1};
                            }
                        } else if (ok1) {
                            bins[b].place(w, h, ox, oy);
                            pl[oi] = {b, ox, oy, 0};
                        } else {
                            bins[b].place(h, w, ox, oy);
                            pl[oi] = {b, ox, oy, 1};
                        }
                        placed = true; break;
                    }
                }
                if (!placed) {
                    bins.emplace_back(BW, BH);
                    int b = (int)bins.size() - 1;
                    int ox, oy;
                    if (w <= BW && h <= BH && bins[b].place(w, h, ox, oy)) {
                        pl[oi] = {b, ox, oy, 0};
                    } else if (h <= BW && w <= BH && bins[b].place(h, w, ox, oy)) {
                        pl[oi] = {b, ox, oy, 1};
                    }
                }
            }
            return {(int)bins.size(), pl};
        };

        // Sort 1: max dim desc
        auto best = tryPack(order);

        // Sort 2: area desc
        std::vector<int> order2(N);
        std::iota(order2.begin(), order2.end(), 0);
        std::sort(order2.begin(), order2.end(), [&](int a, int b) {
            return (long long)rects[a].w * rects[a].h > (long long)rects[b].w * rects[b].h;
        });
        auto r2 = tryPack(order2);
        if (r2.first < best.first) best = r2;

        // Sort 3: perimeter desc
        std::vector<int> order3(N);
        std::iota(order3.begin(), order3.end(), 0);
        std::sort(order3.begin(), order3.end(), [&](int a, int b) {
            return rects[a].w + rects[a].h > rects[b].w + rects[b].h;
        });
        auto r3 = tryPack(order3);
        if (r3.first < best.first) best = r3;

        std::cout << best.first << '\n';
        for (int i = 0; i < N; i++) {
            auto &p = best.second[i];
            std::cout << p.board << ' ' << p.x << ' ' << p.y << ' ' << p.rotated << '\n';
        }
    }
    return 0;
}
