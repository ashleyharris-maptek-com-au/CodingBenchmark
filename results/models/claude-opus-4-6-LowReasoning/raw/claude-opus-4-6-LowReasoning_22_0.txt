using System;
using System.Collections.Generic;
using System.IO;

class TetrisPlayer
{
    static int Width, Height;
    static bool[,] Board;

    static readonly int[][,] Pieces = new int[][,]
    {
        // I
        new int[,]{{1,1,1,1}},
        // O
        new int[,]{{1,1},{1,1}},
        // T
        new int[,]{{1,1,1},{0,1,0}},
        // S
        new int[,]{{0,1,1},{1,1,0}},
        // Z
        new int[,]{{1,1,0},{0,1,1}},
        // J
        new int[,]{{1,1,1},{0,0,1}},
        // L
        new int[,]{{1,1,1},{1,0,0}},
    };

    static int PieceIndex(string type)
    {
        switch(type){ case"I":return 0;case"O":return 1;case"T":return 2;case"S":return 3;case"Z":return 4;case"J":return 5;case"L":return 6;}
        return 0;
    }

    static int[,] Rotate(int[,] p)
    {
        int r=p.GetLength(0),c=p.GetLength(1);
        var n=new int[c,r];
        for(int i=0;i<r;i++)for(int j=0;j<c;j++)n[j,r-1-i]=p[i,j];
        return n;
    }

    static int[,] GetRotated(int pieceIdx, int rot)
    {
        var p=Pieces[pieceIdx];
        for(int i=0;i<rot;i++)p=Rotate(p);
        return p;
    }

    static int Drop(bool[,] board, int[,] piece, int col)
    {
        int pr=piece.GetLength(0),pc=piece.GetLength(1);
        if(col<0||col+pc>Width)return -1;
        for(int row=0;row<=Height-pr;row++)
        {
            bool collide=false;
            for(int i=0;i<pr&&!collide;i++)
                for(int j=0;j<pc&&!collide;j++)
                    if(piece[i,j]==1&&board[row+i,col+j])collide=true;
            if(collide)
            {
                int land=row-1;
                if(land<0)return -1;
                return land;
            }
        }
        return Height-pr;
    }

    static bool Place(bool[,] board, int[,] piece, int row, int col)
    {
        int pr=piece.GetLength(0),pc=piece.GetLength(1);
        for(int i=0;i<pr;i++)
            for(int j=0;j<pc;j++)
                if(piece[i,j]==1)
                {
                    if(row+i<0)return false;
                    board[row+i,col+j]=true;
                }
        return true;
    }

    static int ClearLines(bool[,] board)
    {
        int cleared=0;
        for(int r=Height-1;r>=0;r--)
        {
            bool full=true;
            for(int c=0;c<Width;c++)if(!board[r,c]){full=false;break;}
            if(full)
            {
                cleared++;
                for(int rr=r;rr>0;rr--)
                    for(int c=0;c<Width;c++)board[rr,c]=board[rr-1,c];
                for(int c=0;c<Width;c++)board[0,c]=false;
                r++;
            }
        }
        return cleared;
    }

    static bool[,] CloneBoard(bool[,] b)
    {
        var n=new bool[Height,Width];
        Buffer.BlockCopy(b,0,n,0,Height*Width);
        return n;
    }

    static double Evaluate(bool[,] board, int linesCleared)
    {
        double score=0;
        score += linesCleared * 400;

        int[] heights=new int[Width];
        for(int c=0;c<Width;c++)
        {
            heights[c]=0;
            for(int r=0;r<Height;r++)
                if(board[r,c]){heights[c]=Height-r;break;}
        }

        int maxH=0, sumH=0;
        for(int c=0;c<Width;c++){sumH+=heights[c];if(heights[c]>maxH)maxH=heights[c];}
        score -= sumH * 3.5;
        score -= maxH * 5.0;

        int bumpiness=0;
        for(int c=0;c<Width-1;c++)bumpiness+=Math.Abs(heights[c]-heights[c+1]);
        score -= bumpiness * 3.2;

        int holes=0;
        for(int c=0;c<Width;c++)
        {
            bool found=false;
            for(int r=0;r<Height;r++)
            {
                if(board[r,c])found=true;
                else if(found)holes++;
            }
        }
        score -= holes * 35.0;

        int coveredHoles=0;
        for(int c=0;c<Width;c++)
        {
            int cover=0;
            for(int r=0;r<Height;r++)
            {
                if(board[r,c])cover++;
                else if(cover>0)coveredHoles+=cover;
            }
        }
        score -= coveredHoles * 5.0;

        int wells=0;
        for(int c=0;c<Width;c++)
        {
            int lh=c>0?heights[c-1]:Height;
            int rh=c<Width-1?heights[c+1]:Height;
            int depth=Math.Min(lh,rh)-heights[c];
            if(depth>0)wells+=depth*depth;
        }
        score -= wells * 1.5;

        int rowTrans=0;
        for(int r=0;r<Height;r++)
            for(int c=0;c<Width-1;c++)
                if(board[r,c]!=board[r,c+1])rowTrans++;
        score -= rowTrans * 1.5;

        int colTrans=0;
        for(int r=0;r<Height-1;r++)
            for(int c=0;c<Width;c++)
                if(board[r,c]!=board[r+1,c])colTrans++;
        score -= colTrans * 1.0;

        if(maxH>Height-4)score -= (maxH-(Height-4))*50;

        return score;
    }

    static void Main()
    {
        var sr=new StreamReader(Console.OpenStandardInput());
        string[] header=sr.ReadLine().Trim().Split();
        Width=int.Parse(header[0]);
        Height=int.Parse(header[1]);
        int numPieces=int.Parse(header[2]);
        Board=new bool[Height,Width];

        var sw=new StreamWriter(Console.OpenStandardOutput());
        sw.AutoFlush=true;

        for(int p=0;p<numPieces;p++)
        {
            string line=sr.ReadLine();
            if(line==null)break;
            string type=line.Trim();
            int pi=PieceIndex(type);

            double bestScore=double.MinValue;
            int bestRot=0,bestCol=0;

            for(int rot=0;rot<4;rot++)
            {
                var piece=GetRotated(pi,rot);
                int pc=piece.GetLength(1);
                for(int col=0;col<=Width-pc;col++)
                {
                    int row=Drop(Board,piece,col);
                    if(row<0)continue;
                    var sim=CloneBoard(Board);
                    Place(sim,piece,row,col);
                    int cl=ClearLines(sim);
                    double sc=Evaluate(sim,cl);
                    if(sc>bestScore){bestScore=sc;bestRot=rot;bestCol=col;}
                }
            }

            sw.WriteLine(bestRot+" "+bestCol);

            string result=sr.ReadLine();
            if(result==null)break;
            result=result.Trim();
            if(result=="gameover")break;

            var piece2=GetRotated(pi,bestRot);
            int row2=Drop(Board,piece2,bestCol);
            if(row2>=0)
            {
                Place(Board,piece2,row2,bestCol);
                ClearLines(Board);
            }
        }
    }
}
