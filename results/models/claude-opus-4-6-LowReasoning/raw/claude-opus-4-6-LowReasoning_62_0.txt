declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const { width, height, grid, ignitions, towns, cutBudget } = data;

// Build a flat grid: true = burnable
const burnable: boolean[] = new Array(width * height);
for (let y = 0; y < height; y++) {
  const row = grid[y];
  for (let x = 0; x < width; x++) {
    burnable[y * width + x] = row[x] === '.';
  }
}

const idx = (x: number, y: number) => y * width + x;
const dirs = [[1,0],[-1,0],[0,1],[0,-1]];

// BFS fire spread from ignitions, respecting cuts
function bfs(cuts: Set<number>): Set<number> {
  const burned = new Set<number>();
  const queue: number[] = [];
  for (const [x, y] of ignitions) {
    const i = idx(x, y);
    if (!cuts.has(i)) {
      burned.add(i);
      queue.push(i);
    }
  }
  let head = 0;
  while (head < queue.length) {
    const ci = queue[head++];
    const cx = ci % width, cy = (ci - cx) / width;
    for (const [dx, dy] of dirs) {
      const nx = cx + dx, ny = cy + dy;
      if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;
      const ni = ny * width + nx;
      if (!burned.has(ni) && burnable[ni] && !cuts.has(ni)) {
        burned.add(ni);
        queue.push(ni);
      }
    }
  }
  return burned;
}

// Find which towns burn with no cuts
const noCutBurned = bfs(new Set());
const threatenedTowns = towns.filter((t: any) => noCutBurned.has(idx(t.x, t.y)));

if (threatenedTowns.length === 0 || cutBudget === 0) {
  console.log(JSON.stringify({ firebreaks: [] }));
  process.exit(0);
}

// For each threatened town, BFS backwards to find cells that connect fire to it.
// Then find min-cut cells on paths from ignitions to towns.

// Strategy: For each threatened town, find the shortest path from any ignition to it.
// The cells on these paths (excluding town itself) are candidates for cutting.
// Prioritize cells that protect the most town value.

// Better strategy: iterative greedy. Pick the cell whose removal saves the most value.
// But full BFS per candidate is expensive. We'll do a limited version.

// Approach: BFS from ignitions to get distance. BFS from each town backwards.
// Cells on shortest paths are the bottleneck candidates.
// Score each candidate by total value of towns it helps protect.

// BFS from ignitions (no cuts)
const distFromFire = new Int32Array(width * height).fill(-1);
{
  const queue: number[] = [];
  for (const [x, y] of ignitions) {
    const i = idx(x, y);
    if (burnable[i]) {
      distFromFire[i] = 0;
      queue.push(i);
    }
  }
  let head = 0;
  while (head < queue.length) {
    const ci = queue[head++];
    const cx = ci % width, cy = (ci - cx) / width;
    const nd = distFromFire[ci] + 1;
    for (const [dx, dy] of dirs) {
      const nx = cx + dx, ny = cy + dy;
      if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;
      const ni = ny * width + nx;
      if (distFromFire[ni] === -1 && burnable[ni]) {
        distFromFire[ni] = nd;
        queue.push(ni);
      }
    }
  }
}

// For each threatened town, BFS backward from town to find cells reachable
// Score cells by sum of town values they lie on shortest-path to
const cellScore = new Float64Array(width * height);

for (const town of threatenedTowns) {
  const ti = idx(town.x, town.y);
  const distToTown = distFromFire[ti];
  if (distToTown <= 0) continue;

  // BFS backwards from town, only through cells where distFromFire decreases toward ignition
  const visited = new Set<number>();
  const queue: number[] = [ti];
  visited.add(ti);
  let head = 0;
  while (head < queue.length) {
    const ci = queue[head++];
    const cx = ci % width, cy = (ci - cx) / width;
    if (ci !== ti) {
      cellScore[ci] += town.value;
    }
    for (const [dx, dy] of dirs) {
      const nx = cx + dx, ny = cy + dy;
      if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;
      const ni = ny * width + nx;
      if (!visited.has(ni) && burnable[ni] && distFromFire[ni] >= 0 && distFromFire[ni] < distFromFire[ci]) {
        visited.add(ni);
        queue.push(ni);
      }
    }
  }
}

// Greedy: pick top-scoring cells, then verify with BFS
// Collect candidates
const candidates: [number, number][] = [];
for (let i = 0; i < width * height; i++) {
  if (cellScore[i] > 0) {
    candidates.push([i, cellScore[i]]);
  }
}
candidates.sort((a, b) => b[1] - a[1]);

// Start with top candidates up to budget
let cuts = new Set<number>();
const budget = cutBudget as number;

// Take initial greedy set
for (let i = 0; i < Math.min(candidates.length, budget); i++) {
  cuts.add(candidates[i][0]);
}

// Iterative improvement: try greedy removal/addition
// First check which towns are saved
function savedValue(cutSet: Set<number>): number {
  const burned = bfs(cutSet);
  let saved = 0;
  for (const t of threatenedTowns) {
    if (!burned.has(idx(t.x, t.y))) saved += t.value;
  }
  return saved;
}

let bestVal = savedValue(cuts);
let bestCuts = new Set(cuts);

// If we haven't saved everything, try iterative greedy from scratch
if (bestVal < threatenedTowns.reduce((s: number, t: any) => s + t.value, 0)) {
  // True greedy: one cell at a time, pick the one that maximizes saved value
  // Limit iterations for scalability
  const maxGreedyIters = Math.min(budget, 200);
  const greedyCuts = new Set<number>();
  const topCandidates = candidates.slice(0, Math.min(candidates.length, budget * 10));

  for (let iter = 0; iter < maxGreedyIters; iter++) {
    let bestCell = -1;
    let bestCellVal = -1;

    // Sample candidates to keep it fast
    const sample = topCandidates.filter(c => !greedyCuts.has(c[0])).slice(0, 500);
    if (sample.length === 0) break;

    for (const [cell] of sample) {
      greedyCuts.add(cell);
      const v = savedValue(greedyCuts);
      if (v > bestCellVal) {
        bestCellVal = v;
        bestCell = cell;
      }
      greedyCuts.delete(cell);
    }

    if (bestCell === -1) break;
    greedyCuts.add(bestCell);

    if (bestCellVal > bestVal) {
      bestVal = bestCellVal;
      bestCuts = new Set(greedyCuts);
    }
  }
}

const firebreaks: [number, number][] = [];
for (const ci of bestCuts) {
  const x = ci % width;
  const y = (ci - x) / width;
  firebreaks.push([x, y]);
}

console.log(JSON.stringify({ firebreaks }));
