using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

class Program
{
    static int XCount, YCount, ZCount;
    static double XBlock, YBlock, ZBlock;
    static byte[,,] voxelData;
    static int[,] surfaceZ; // top of ground at each (x,y)

    struct VoxelInfo : IComparable<VoxelInfo>
    {
        public int X, Y, Z;
        public int Mass, Value;
        public double Profit;
        public int CompareTo(VoxelInfo o) => o.Profit.CompareTo(Profit);
    }

    struct Move
    {
        public int DigX, DigY, DigZ;
        public List<int> Steps; // pairs of x,y
        public int DestX, DestY;
    }

    static void Main()
    {
        using var stdin = Console.OpenStandardInput();
        var reader = new StreamReader(stdin);

        var header = reader.ReadLine().Split(' ');
        XCount = int.Parse(header[0]);
        YCount = int.Parse(header[1]);
        ZCount = int.Parse(header[2]);
        XBlock = double.Parse(header[3]);
        YBlock = double.Parse(header[4]);
        ZBlock = double.Parse(header[5]);

        voxelData = new byte[XCount, YCount, ZCount];
        surfaceZ = new int[XCount, YCount];

        for (int x = 0; x < XCount; x++)
            for (int y = 0; y < YCount; y++)
                surfaceZ[x, y] = -1;

        var rawStream = reader.BaseStream;
        for (int x = 0; x < XCount; x++)
        {
            for (int y = 0; y < YCount; y++)
            {
                for (int z = 0; z < ZCount; z++)
                {
                    int b = rawStream.ReadByte();
                    if (b < 0) b = 0;
                    voxelData[x, y, z] = (byte)b;
                    if (b != 0 && z > surfaceZ[x, y])
                        surfaceZ[x, y] = z;
                }
            }
        }

        var moves = new List<Move>();
        bool largeWorld = (long)XCount * YCount * ZCount > 500_000_000L;

        // Find all valuable voxels and rank them
        var candidates = new List<VoxelInfo>();
        for (int x = 0; x < XCount; x++)
        {
            for (int y = 0; y < YCount; y++)
            {
                for (int z = 0; z <= surfaceZ[x, y]; z++)
                {
                    byte d = voxelData[x, y, z];
                    if (d == 0) continue;
                    int mass = d & 15;
                    int val = d >> 4;
                    if (val == 0) continue;
                    double revenue = mass * val * 1000.0;
                    // Estimate cost: manhattan distance * mass for flat, plus digging overhead
                    int dist = x + y;
                    int depthAbove = surfaceZ[x, y] - z;
                    // rough cost: move ore dist steps, plus move depthAbove overburden somewhere
                    double moveCost = mass * dist; // flat cost
                    // overburden cost estimate: each layer needs to be moved ~2 steps avg
                    double overburdenCost = 0;
                    for (int oz = z + 1; oz <= surfaceZ[x, y]; oz++)
                    {
                        byte od = voxelData[x, y, oz];
                        int om = od & 15;
                        overburdenCost += om * 4; // rough
                    }
                    double profit = revenue - moveCost - overburdenCost;
                    if (profit > 0)
                    {
                        candidates.Add(new VoxelInfo { X = x, Y = y, Z = z, Mass = mass, Value = val, Profit = profit });
                    }
                }
            }
        }

        candidates.Sort();

        // Limit candidates for large worlds
        int maxOps = largeWorld ? 50000 : 500000;
        if (candidates.Count > maxOps) candidates = candidates.GetRange(0, maxOps);

        // Find a dump zone - pick edges far from plant
        // We'll dump at (XCount-1, YCount-1) area
        int dumpBaseX = XCount - 1;
        int dumpBaseY = YCount - 1;

        foreach (var c in candidates)
        {
            // Check if still exposed from above
            if (surfaceZ[c.X, c.Y] < c.Z) continue;

            // Dig everything above this voxel first (overburden)
            while (surfaceZ[c.X, c.Y] > c.Z)
            {
                int sz = surfaceZ[c.X, c.Y];
                byte od = voxelData[c.X, c.Y, sz];
                if (od == 0) { surfaceZ[c.X, c.Y]--; continue; }
                int om = od & 15;
                int ov = od >> 4;

                // If overburden is valuable and profitable to deliver to plant, do so
                bool deliverToPlant = ov > 0 && (ov * 1000 - (c.X + c.Y)) > 0;
                int destX, destY;
                if (deliverToPlant)
                {
                    destX = 0; destY = 0;
                }
                else
                {
                    // Find dump location
                    destX = dumpBaseX;
                    destY = dumpBaseY;
                }

                var path = FindPath(c.X, c.Y, destX, destY);
                if (path != null)
                {
                    var move = new Move { DigX = c.X, DigY = c.Y, DigZ = sz, Steps = path, DestX = destX, DestY = destY };
                    moves.Add(move);
                    voxelData[c.X, c.Y, sz] = 0;
                    surfaceZ[c.X, c.Y] = sz - 1;
                    // Update surface at destination if not plant delivery of valuable
                    if (!(destX == 0 && destY == 0 && ov > 0))
                    {
                        if (destX == 0 && destY == 0)
                        {
                            // junk at plant - costs money, avoid this
                            // Actually let's just dump elsewhere
                            // redo - skip this path, dump elsewhere instead
                        }
                        surfaceZ[destX, destY]++;
                        if (surfaceZ[destX, destY] < ZCount)
                            voxelData[destX, destY, surfaceZ[destX, destY]] = od;
                    }
                }
                else
                {
                    // Can't path, skip
                    break;
                }
            }

            // Now dig the target voxel
            if (surfaceZ[c.X, c.Y] != c.Z) continue;
            byte td = voxelData[c.X, c.Y, c.Z];
            if (td == 0) continue;
            int tv = td >> 4;
            if (tv == 0) continue;

            var orePath = FindPath(c.X, c.Y, 0, 0);
            if (orePath != null)
            {
                moves.Add(new Move { DigX = c.X, DigY = c.Y, DigZ = c.Z, Steps = orePath, DestX = 0, DestY = 0 });
                voxelData[c.X, c.Y, c.Z] = 0;
                surfaceZ[c.X, c.Y] = c.Z - 1;
                // valuable ore at plant ceases to exist
            }
        }

        // Output
        var output = new System.Text.StringBuilder();
        output.AppendLine(moves.Count.ToString());
        foreach (var m in moves)
        {
            int nSteps = m.Steps.Count / 2;
            output.Append(nSteps);
            output.Append(' '); output.Append(m.DigX);
            output.Append(' '); output.Append(m.DigY);
            output.Append(' '); output.Append(m.DigZ);
            for (int i = 0; i < m.Steps.Count; i++)
            {
                output.Append(' '); output.Append(m.Steps[i]);
            }
            output.Append(' '); output.Append(m.DestX);
            output.Append(' '); output.AppendLine(m.DestY.ToString());
        }
        Console.Write(output);
    }

    static List<int> FindPath(int sx, int sy, int dx, int dy)
    {
        // Simple greedy manhattan path following surface, respecting slope constraint
        // Slope constraint: can go up at most 1 Z per 1 XY step (10% slope since Z precision is 10x)
        var steps = new List<int>();
        int cx = sx, cy = sy;

        while (cx != dx || cy != dy)
        {
            int bestX = cx, bestY = cy;
            int bestScore = int.MaxValue;

            int[][] dirs = { new[]{1,0}, new[]{-1,0}, new[]{0,1}, new[]{0,-1} };
            foreach (var d in dirs)
            {
                int nx = cx + d[0], ny = cy + d[1];
                if (nx < 0 || nx >= XCount || ny < 0 || ny >= YCount) continue;

                int curZ = surfaceZ[cx, cy];
                int nextZ = surfaceZ[nx, ny];

                // Slope constraint: can climb at most 1 Z per step
                if (nextZ - curZ > 1) continue;
                // Downhill constraint: can descend at most 1 Z per step
                if (curZ - nextZ > 1) continue;

                int score = Math.Abs(nx - dx) + Math.Abs(ny - dy);
                // Prefer flat or downhill
                if (nextZ > curZ) score += 5;

                if (score < bestScore)
                {
                    bestScore = score;
                    bestX = nx;
                    bestY = ny;
                }
            }

            if (bestX == cx && bestY == cy)
                return null; // stuck

            steps.Add(bestX);
            steps.Add(bestY);
            cx = bestX;
            cy = bestY;
        }

        return steps;
    }
}
