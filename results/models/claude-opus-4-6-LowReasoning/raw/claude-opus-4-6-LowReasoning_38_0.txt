import sys

def solve():
    data = sys.stdin.read().split('\n')
    U, S = map(int, data[0].split())
    sets = []
    for i in range(1, S + 1):
        if data[i].strip():
            sets.append(set(map(int, data[i].split())))
        else:
            sets.append(set())

    # DLX (Dancing Links) for exact cover
    # Build column objects for each element 0..U-1

    # Node: L, R, U, D, C, row
    L = []
    R = []
    Up = []
    D = []
    C = []
    Row = []
    Size = []

    def new_node(l=0, r=0, u=0, d=0, c=0, row=-1):
        idx = len(L)
        L.append(l); R.append(r); Up.append(u); D.append(d); C.append(c); Row.append(row)
        return idx

    # Header node
    header = new_node()
    L[header] = R[header] = Up[header] = D[header] = header
    Size.append(0)

    # Column headers
    col_headers = []
    for j in range(U):
        ch = new_node()
        Size.append(0)
        C[ch] = ch
        Row[ch] = -1
        # insert left of header
        L[ch] = L[header]
        R[ch] = header
        R[L[header]] = ch
        L[header] = ch
        Up[ch] = ch
        D[ch] = ch
        col_headers.append(ch)

    # Add rows
    for i, s in enumerate(sets):
        first = -1
        for elem in sorted(s):
            ch = col_headers[elem]
            node = new_node(c=ch, row=i)
            Size.append(0)
            # insert above ch (at bottom of column)
            Up[node] = Up[ch]
            D[node] = ch
            D[Up[ch]] = node
            Up[ch] = node
            Size[ch] += 1
            if first == -1:
                first = node
                L[node] = node
                R[node] = node
            else:
                L[node] = L[first]
                R[node] = first
                R[L[first]] = node
                L[first] = node

    solution = []

    def cover(c):
        R[L[c]] = R[c]
        L[R[c]] = L[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                D[Up[j]] = D[j]
                Up[D[j]] = Up[j]
                Size[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = Up[c]
        while i != c:
            j = L[i]
            while j != i:
                Size[C[j]] += 1
                D[Up[j]] = j
                Up[D[j]] = j
                j = L[j]
            i = Up[i]
        R[L[c]] = c
        L[R[c]] = c

    def search():
        if R[header] == header:
            return True
        # Choose column with min size
        min_size = float('inf')
        col = -1
        j = R[header]
        while j != header:
            if Size[j] < min_size:
                min_size = Size[j]
                col = j
            j = R[j]
        if min_size == 0:
            return False
        cover(col)
        r = D[col]
        while r != col:
            solution.append(Row[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
            if search():
                return True
            solution.pop()
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            r = D[r]
        uncover(col)
        return False

    if search():
        print("SOLUTION")
        print(' '.join(map(str, sorted(solution))))
    else:
        print("NO SOLUTION")

solve()
