const double EPS = 1e-9;

struct V3 {
    double x,y,z;
    V3(double x=0,double y=0,double z=0):x(x),y(y),z(z){}
    V3 operator+(V3 b)const{return{x+b.x,y+b.y,z+b.z};}
    V3 operator-(V3 b)const{return{x-b.x,y-b.y,z-b.z};}
    V3 operator*(double s)const{return{x*s,y*s,z*s};}
    V3 operator/(double s)const{return{x/s,y/s,z/s};}
    double dot(V3 b)const{return x*b.x+y*b.y+z*b.z;}
    V3 cross(V3 b)const{return{y*b.z-z*b.y,z*b.x-x*b.z,x*b.y-y*b.x};}
    double len2()const{return x*x+y*y+z*z;}
    double len()const{return sqrt(len2());}
};

struct AABB {
    V3 lo{1e18,1e18,1e18},hi{-1e18,-1e18,-1e18};
    void add(V3 p){
        lo.x=min(lo.x,p.x);lo.y=min(lo.y,p.y);lo.z=min(lo.z,p.z);
        hi.x=max(hi.x,p.x);hi.y=max(hi.y,p.y);hi.z=max(hi.z,p.z);
    }
    void merge(const AABB&o){add(o.lo);add(o.hi);}
    bool overlaps(const AABB&o)const{
        return lo.x<=o.hi.x&&hi.x>=o.lo.x&&
               lo.y<=o.hi.y&&hi.y>=o.lo.y&&
               lo.z<=o.hi.z&&hi.z>=o.lo.z;
    }
};

struct BVHNode{AABB box;int left=-1,right=-1,start=0,count=0;};

struct Mesh {
    vector<V3> vt;
    vector<array<int,3>> fc;
    vector<BVHNode> bvh;
    vector<int> ord;
    vector<AABB> tb;

    void buildBoxes(){
        tb.resize(fc.size());
        for(int i=0;i<(int)fc.size();i++){
            tb[i]={};
            for(int j=0;j<3;j++) tb[i].add(vt[fc[i][j]]);
        }
    }
    int buildBVH(int lo,int hi){
        BVHNode nd;nd.box={};
        for(int i=lo;i<hi;i++) nd.box.merge(tb[ord[i]]);
        if(hi-lo<=4){
            nd.start=lo;nd.count=hi-lo;
            bvh.push_back(nd);return(int)bvh.size()-1;
        }
        V3 ext=nd.box.hi-nd.box.lo;
        int ax=0;
        if(ext.y>ext.x)ax=1;
        if(ext.z>(ax?ext.y:ext.x))ax=2;
        int mid=(lo+hi)/2;
        nth_element(ord.begin()+lo,ord.begin()+mid,ord.begin()+hi,
            [&](int a,int b){
                V3 ca=(tb[a].lo+tb[a].hi)*0.5,cb=(tb[b].lo+tb[b].hi)*0.5;
                return (ax==0?ca.x:ax==1?ca.y:ca.z)<(ax==0?cb.x:ax==1?cb.y:cb.z);
            });
        nd.start=-1;nd.count=0;
        int idx=(int)bvh.size();
        bvh.push_back(nd);
        int l=buildBVH(lo,mid);
        int r=buildBVH(mid,hi);
        bvh[idx].left=l;bvh[idx].right=r;
        bvh[idx].box={};
        bvh[idx].box.merge(bvh[l].box);
        bvh[idx].box.merge(bvh[r].box);
        return idx;
    }
    void build(){
        buildBoxes();
        ord.resize(fc.size());
        for(int i=0;i<(int)fc.size();i++)ord[i]=i;
        if(!fc.empty()){
            bvh.reserve(fc.size()*2);
            buildBVH(0,(int)fc.size());
        }
    }
    void queryOvlp(int nd,const AABB&box,vector<int>&res)const{
        if(nd<0)return;
        auto&n=bvh[nd];
        if(!n.box.overlaps(box))return;
        if(n.left==-1){
            for(int i=n.start;i<n.start+n.count;i++)res.push_back(ord[i]);
        }else{queryOvlp(n.left,box,res);queryOvlp(n.right,box,res);}
    }
    int rayC(V3 o,V3 d,V3 id,int nd)const{
        if(nd<0)return 0;
        auto&n=bvh[nd];
        double tmin=-1e18,tmax=1e18;
        for(int a=0;a<3;a++){
            double la=a==0?n.box.lo.x:a==1?n.box.lo.y:n.box.lo.z;
            double ha=a==0?n.box.hi.x:a==1?n.box.hi.y:n.box.hi.z;
            double oa=a==0?o.x:a==1?o.y:o.z;
            double ia=a==0?id.x:a==1?id.y:id.z;
            double t1=(la-oa)*ia,t2=(ha-oa)*ia;
            if(t1>t2)swap(t1,t2);
            tmin=max(tmin,t1);tmax=min(tmax,t2);
        }
        if(tmax<max(tmin,0.0)-1e-12)return 0;
        if(n.left==-1){
            int cnt=0;
            for(int i=n.start;i<n.start+n.count;i++){
                int ti=ord[i];
                V3 v0=vt[fc[ti][0]],v1=vt[fc[ti][1]],v2=vt[fc[ti][2]];
                V3 e1=v1-v0,e2=v2-v0,h=d.cross(e2);
                double a=e1.dot(h);
                if(fabs(a)<1e-12)continue;
                double f=1.0/a;
                V3 s=o-v0;
                double u=f*s.dot(h);
                if(u<-1e-8||u>1+1e-8)continue;
                V3 q=s.cross(e1);
                double v=f*d.dot(q);
                if(v<-1e-8||u+v>1+1e-8)continue;
                double t=f*e2.dot(q);
                if(t>1e-8)cnt++;
            }
            return cnt;
        }
        return rayC(o,d,id,n.left)+rayC(o,d,id,n.right);
    }
    bool isInside(V3 p)const{
        if(bvh.empty())return false;
        static const V3 dirs[]={
            {0.57735026918962576451,0.57735026918962576451,0.57735026918962576451},
            {0.8017837257372732,0.2672612419124244,0.5345224838248488},
            {0.1690308509457033,0.8451542547285166,0.5070925528371100}
        };
        int votes=0;
        for(int i=0;i<3;i++){
            V3 d=dirs[i],id{1.0/d.x,1.0/d.y,1.0/d.z};
            if(rayC(p,d,id,0)%2==1)votes++;
        }
        return votes>=2;
    }
};

bool triTriIsect(V3 a0,V3 a1,V3 a2,V3 b0,V3 b1,V3 b2){
    V3 nb=(b1-b0).cross(b2-b0);
    double db=nb.dot(b0);
    double da[3]={nb.dot(a0)-db,nb.dot(a1)-db,nb.dot(a2)-db};
    for(int i=0;i<3;i++)if(fabs(da[i])<EPS)da[i]=0;
    if(da[0]>0&&da[1]>0&&da[2]>0)return false;
    if(da[0]<0&&da[1]<0&&da[2]<0)return false;

    V3 na=(a1-a0).cross(a2-a0);
    double daA=na.dot(a0);
    double dbb[3]={na.dot(b0)-daA,na.dot(b1)-daA,na.dot(b2)-daA};
    for(int i=0;i<3;i++)if(fabs(dbb[i])<EPS)dbb[i]=0;
    if(dbb[0]>0&&dbb[1]>0&&dbb[2]>0)return false;
    if(dbb[0]<0&&dbb[1]<0&&dbb[2]<0)return false;

    if(da[0]==0&&da[1]==0&&da[2]==0)return false;

    V3 D=na.cross(nb);
    int ax=0;double mx=fabs(D.x);
    if(fabs(D.y)>mx){ax=1;mx=fabs(D.y);}
    if(fabs(D.z)>mx)ax=2;
    auto proj=[&](V3 v)->double{return ax==0?v.x:ax==1?v.y:v.z;};

    auto interval=[&](double d0,double d1,double d2,double p0,double p1,double p2,double&t0,double&t1){
        double ts[6];int tn=0;
        double dd[3]={d0,d1,d2},pp[3]={p0,p1,p2};
        for(int i=0;i<3;i++){
            int j=(i+1)%3;
            if(dd[i]==0)ts[tn++]=pp[i];
            if((dd[i]>0&&dd[j]<0)||(dd[i]<0&&dd[j]>0)){
                double t=dd[i]/(dd[i]-dd[j]);
                ts[tn++]=pp[i]+t*(pp[j]-pp[i]);
            }
        }
        if(tn==0){t0=t1=0;return;}
        t0=ts[0];t1=ts[0];
        for(int i=1;i<tn;i++){t0=min(t0,ts[i]);t1=max(t1,ts[i]);}
    };

    double ta0,ta1,tb0,tb1;
    interval(da[0],da[1],da[2],proj(a0),proj(a1),proj(a2),ta0,ta1);
    interval(dbb[0],dbb[1],dbb[2],proj(b0),proj(b1),proj(b2),tb0,tb1);
    if(ta1-ta0<EPS||tb1-tb0<EPS)return false;
    return ta0<tb1-EPS&&tb0<ta1-EPS;
}

using Poly=vector<V3>;

void splitPoly(const Poly&poly,V3 pn,double pd,Poly&fr,Poly&bk){
    fr.clear();bk.clear();
    int n=(int)poly.size();
    if(n<3)return;
    vector<double>dist(n);
    bool af=true,ab=true;
    for(int i=0;i<n;i++){
        dist[i]=pn.dot(poly[i])-pd;
        if(dist[i]>EPS)ab=false;
        if(dist[i]<-EPS)af=false;
    }
    if(af){fr=poly;return;}
    if(ab){bk=poly;return;}

    for(int i=0;i<n;i++){
        int j=(i+1)%n;
        if(dist[i]>EPS)fr.push_back(poly[i]);
        else if(dist[i]<-EPS)bk.push_back(poly[i]);
        else{fr.push_back(poly[i]);bk.push_back(poly[i]);}
        if((dist[i]>EPS&&dist[j]<-EPS)||(dist[i]<-EPS&&dist[j]>EPS)){
            double t=dist[i]/(dist[i]-dist[j]);
            V3 p=poly[i]+(poly[j]-poly[i])*t;
            fr.push_back(p);bk.push_back(p);
        }
    }
}

int main(){
    setvbuf(stdin,nullptr,_IOFBF,1<<22);
    setvbuf(stdout,nullptr,_IOFBF,1<<22);

    int va,fa,vb,fb;
    scanf("%d%d%d%d",&va,&fa,&vb,&fb);
    Mesh A,B;
    A.vt.resize(va);for(int i=0;i<va;i++)scanf("%lf%lf%lf",&A.vt[i].x,&A.vt[i].y,&A.vt[i].z);
    A.fc.resize(fa);for(int i=0;i<fa;i++)scanf("%d%d%d",&A.fc[i][0],&A.fc[i][1],&A.fc[i][2]);
    B.vt.resize(vb);for(int i=0;i<vb;i++)scanf("%lf%lf%lf",&B.vt[i].x,&B.vt[i].y,&B.vt[i].z);
    B.fc.resize(fb);for(int i=0;i<fb;i++)scanf("%d%d%d",&B.fc[i][0],&B.fc[i][1],&B.fc[i][2]);

    A.build();B.build();

    vector<vector<pair<V3,double>>>cutA(fa),cutB(fb);
    vector<bool>isA(fa,false),isB(fb,false);

    for(int i=0;i<fa;i++){
        vector<int>cands;
        if(!B.bvh.empty())B.queryOvlp(0,A.tb[i],cands);
        for(int j:cands){
            V3 a0=A.vt[A.fc[i][0]],a1=A.vt[A.fc[i][1]],a2=A.vt[A.fc[i][2]];
            V3 b0=B.vt[B.fc[j][0]],b1=B.vt[B.fc[j][1]],b2=B.vt[B.fc[j][2]];
            if(triTriIsect(a0,a1,a2,b0,b1,b2)){
                isA[i]=true;isB[j]=true;
                V3 nb=(b1-b0).cross(b2-b0);double l=nb.len();
                if(l>EPS){nb=nb/l;cutA[i].push_back({nb,nb.dot(b0)});}
                V3 na=(a1-a0).cross(a2-a0);l=na.len();
                if(l>EPS){na=na/l;cutB[j].push_back({na,na.dot(a0)});}
            }
        }
    }

    // Deduplicate cutting planes
    auto dedup=[](vector<pair<V3,double>>&pl){
        if(pl.size()<=1)return;
        sort(pl.begin(),pl.end(),[](const pair<V3,double>&a,const pair<V3,double>&b){
            long long ax=llround(a.first.x*1e7),bx=llround(b.first.x*1e7);
            if(ax!=bx)return ax<bx;
            long long ay=llround(a.first.y*1e7),by=llround(b.first.y*1e7);
            if(ay!=by)return ay<by;
            long long az=llround(a.first.z*1e7),bz=llround(b.first.z*1e7);
            if(az!=bz)return az<bz;
            return llround(a.second*1e7)<llround(b.second*1e7);
        });
        auto it=unique(pl.begin(),pl.end(),[](const pair<V3,double>&a,const pair<V3,double>&b){
            return llround(a.first.x*1e7)==llround(b.first.x*1e7)&&
                   llround(a.first.y*1e7)==llround(b.first.y*1e7)&&
                   llround(a.first.z*1e7)==llround(b.first.z*1e7)&&
                   llround(a.second*1e7)==llround(b.second*1e7);
        });
        pl.erase(it,pl.end());
    };
    for(int i=0;i<fa;i++)dedup(cutA[i]);
    for(int i=0;i<fb;i++)dedup(cutB[i]);

    vector<array<V3,3>>out;

    auto process=[&](Mesh&self,Mesh&other,vector<bool>&isect,vector<vector<pair<V3,double>>>&cuts,int nf){
        for(int i=0;i<nf;i++){
            V3 v0=self.vt[self.fc[i][0]],v1=self.vt[self.fc[i][1]],v2=self.vt[self.fc[i][2]];
            if(!isect[i]){
                V3 c=(v0+v1+v2)/3.0;
                if(!other.isInside(c))
                    out.push_back({v0,v1,v2});
            }else{
                vector<Poly>polys={{v0,v1,v2}};
                for(auto&[pn,pd]:cuts[i]){
                    vector<Poly>np;
                    for(auto&poly:polys){
                        Poly fr,bk;
                        splitPoly(poly,pn,pd,fr,bk);
                        if((int)fr.size()>=3)np.push_back(move(fr));
                        if((int)bk.size()>=3)np.push_back(move(bk));
                    }
                    polys=move(np);
                    if(polys.empty())break;
                    if(polys.size()>10000)break;
                }
                for(auto&poly:polys){
                    V3 c{0,0,0};
                    for(auto&p:poly)c=c+p;
                    c=c/(double)poly.size();
                    if(!other.isInside(c)){
                        for(int k=1;k+1<(int)poly.size();k++){
                            V3 e1=poly[k]-poly[0],e2=poly[k+1]-poly[0];
                            if(e1.cross(e2).len()>1e-12)
                                out.push_back({poly[0],poly[k],poly[k+1]});
                        }
                    }
                }
            }
        }
    };

    process(A,B,isA,cutA,fa);
    process(B,A,isB,cutB,fb);

    struct VKey{long long x,y,z;bool operator==(const VKey&o)const{return x==o.x&&y==o.y&&z==o.z;}};
    struct VHash{size_t operator()(const VKey&k)const{
        size_t h=17;h=h*2654435761ULL^(size_t)(k.x+4000000000LL);
        h=h*2654435761ULL^(size_t)(k.y+4000000000LL);
        h=h*2654435761ULL^(size_t)(k.z+4000000000LL);return h;
    }};

    unordered_map<VKey,int,VHash>vmap;
    vector<V3>ov;
    vector<array<int,3>>of;

    auto vid=[&](V3 v)->int{
        VKey k{llround(v.x*1e8),llround(v.y*1e8),llround(v.z*1e8)};
        auto it=vmap.find(k);
        if(it!=vmap.end())return it->second;
        int id=(int)ov.size();
        ov.push_back(v);vmap[k]=id;return id;
    };

    for(auto&t:out){
        int a=vid(t[0]),b=vid(t[1]),c=vid(t[2]);
        if(a!=b&&b!=c&&a!=c)of.push_back({a,b,c});
    }

    printf("%d %d\n",(int)ov.size(),(int)of.size());
    for(auto&v:ov)printf("%.17g %.17g %.17g\n",v.x,v.y,v.z);
    for(auto&f:of)printf("%d %d %d\n",f[0],f[1],f[2]);
}
