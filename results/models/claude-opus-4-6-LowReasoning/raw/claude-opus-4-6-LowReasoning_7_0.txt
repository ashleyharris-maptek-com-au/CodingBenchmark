using namespace std;

const double EPS = 1e-10;

struct Vec3 {
    double x,y,z;
    Vec3(double x=0,double y=0,double z=0):x(x),y(y),z(z){}
    Vec3 operator+(const Vec3&v)const{return{x+v.x,y+v.y,z+v.z};}
    Vec3 operator-(const Vec3&v)const{return{x-v.x,y-v.y,z-v.z};}
    Vec3 operator*(double t)const{return{x*t,y*t,z*t};}
    double dot(const Vec3&v)const{return x*v.x+y*v.y+z*v.z;}
    Vec3 cross(const Vec3&v)const{return{y*v.z-z*v.y,z*v.x-x*v.z,x*v.y-y*v.x};}
    double len2()const{return dot(*this);}
    double len()const{return sqrt(len2());}
};
Vec3 operator*(double t,const Vec3&v){return v*t;}

struct Vec2 {
    double x,y;
    Vec2(double x=0,double y=0):x(x),y(y){}
    Vec2 operator+(const Vec2&v)const{return{x+v.x,y+v.y};}
    Vec2 operator-(const Vec2&v)const{return{x-v.x,y-v.y};}
    Vec2 operator*(double t)const{return{x*t,y*t};}
    double dot(const Vec2&v)const{return x*v.x+y*v.y;}
    double cross(const Vec2&v)const{return x*v.y-y*v.x;}
    double len2()const{return dot(*this);}
};

struct AABB {
    Vec3 mn{1e18,1e18,1e18},mx{-1e18,-1e18,-1e18};
    void expand(const Vec3&p){
        mn.x=min(mn.x,p.x);mn.y=min(mn.y,p.y);mn.z=min(mn.z,p.z);
        mx.x=max(mx.x,p.x);mx.y=max(mx.y,p.y);mx.z=max(mx.z,p.z);
    }
    void expand(const AABB&b){expand(b.mn);expand(b.mx);}
    bool intersects(const AABB&b)const{
        return mn.x<=b.mx.x&&mx.x>=b.mn.x&&
               mn.y<=b.mx.y&&mx.y>=b.mn.y&&
               mn.z<=b.mx.z&&mx.z>=b.mn.z;
    }
    Vec3 center()const{return(mn+mx)*0.5;}
};

struct BVHNode { AABB box; int left=-1,right=-1,start=0,count=0; };

struct BVH {
    vector<BVHNode> nodes;
    vector<int> idx;

    void build(const vector<Vec3>&verts,const vector<array<int,3>>&faces){
        int n=(int)faces.size();
        idx.resize(n);iota(idx.begin(),idx.end(),0);
        vector<AABB> boxes(n);
        vector<Vec3> ctrs(n);
        for(int i=0;i<n;i++){
            for(int j=0;j<3;j++) boxes[i].expand(verts[faces[i][j]]);
            ctrs[i]=boxes[i].center();
        }
        nodes.reserve(2*n+1);
        buildR(0,n,boxes,ctrs);
    }

    int buildR(int lo,int hi,const vector<AABB>&boxes,const vector<Vec3>&ctrs){
        int id=(int)nodes.size();nodes.push_back({});
        for(int i=lo;i<hi;i++) nodes[id].box.expand(boxes[idx[i]]);
        if(hi-lo<=4){nodes[id].start=lo;nodes[id].count=hi-lo;return id;}
        Vec3 ext=nodes[id].box.mx-nodes[id].box.mn;
        int ax=(ext.x>=ext.y&&ext.x>=ext.z)?0:(ext.y>=ext.z)?1:2;
        int mid=(lo+hi)/2;
        nth_element(idx.begin()+lo,idx.begin()+mid,idx.begin()+hi,
            [&](int a,int b){return(&ctrs[a].x)[ax]<(&ctrs[b].x)[ax];});
        nodes[id].left=buildR(lo,mid,boxes,ctrs);
        nodes[id].right=buildR(mid,hi,boxes,ctrs);
        return id;
    }

    template<typename F>
    void query(const AABB&qb,F&&cb)const{qR(0,qb,cb);}
    template<typename F>
    void qR(int id,const AABB&qb,F&&cb)const{
        auto&nd=nodes[id];
        if(!nd.box.intersects(qb))return;
        if(nd.count>0){for(int i=nd.start;i<nd.start+nd.count;i++)cb(idx[i]);return;}
        qR(nd.left,qb,cb);qR(nd.right,qb,cb);
    }

    bool rayAABB(const Vec3&o,const Vec3&id,const AABB&b)const{
        double t1=(b.mn.x-o.x)*id.x,t2=(b.mx.x-o.x)*id.x;
        double t3=(b.mn.y-o.y)*id.y,t4=(b.mx.y-o.y)*id.y;
        double t5=(b.mn.z-o.z)*id.z,t6=(b.mx.z-o.z)*id.z;
        double tmin=max({min(t1,t2),min(t3,t4),min(t5,t6)});
        double tmax=min({max(t1,t2),max(t3,t4),max(t5,t6)});
        return tmax>=max(tmin,0.0);
    }

    template<typename F>
    void rayQuery(const Vec3&o,const Vec3&invd,F&&cb)const{rayQR(0,o,invd,cb);}
    template<typename F>
    void rayQR(int id,const Vec3&o,const Vec3&invd,F&&cb)const{
        auto&nd=nodes[id];
        if(!rayAABB(o,invd,nd.box))return;
        if(nd.count>0){for(int i=nd.start;i<nd.start+nd.count;i++)cb(idx[i]);return;}
        rayQR(nd.left,o,invd,cb);rayQR(nd.right,o,invd,cb);
    }
};

bool rayTri(const Vec3&o,const Vec3&d,const Vec3&v0,const Vec3&v1,const Vec3&v2,double&t){
    Vec3 e1=v1-v0,e2=v2-v0;
    Vec3 pv=d.cross(e2);
    double det=e1.dot(pv);
    if(fabs(det)<EPS)return false;
    double inv=1.0/det;
    Vec3 tv=o-v0;
    double u=tv.dot(pv)*inv;
    if(u<0||u>1)return false;
    Vec3 qv=tv.cross(e1);
    double v=d.dot(qv)*inv;
    if(v<0||u+v>1)return false;
    t=e2.dot(qv)*inv;
    return t>EPS;
}

bool isInside(const Vec3&pt,const vector<Vec3>&V,const vector<array<int,3>>&F,const BVH&bvh){
    Vec3 dir(1.0,0.123456789,0.0987654321);
    Vec3 invd(1.0/dir.x,1.0/dir.y,1.0/dir.z);
    int cnt=0;
    bvh.rayQuery(pt,invd,[&](int fi){
        double t;
        if(rayTri(pt,dir,V[F[fi][0]],V[F[fi][1]],V[F[fi][2]],t))cnt++;
    });
    return cnt%2==1;
}

bool edgeTriHit(const Vec3&p0,const Vec3&p1,const Vec3&v0,const Vec3&v1,const Vec3&v2,Vec3&hit){
    Vec3 n=(v1-v0).cross(v2-v0);
    double d0=n.dot(p0-v0),d1=n.dot(p1-v0);
    if(d0*d1>0)return false;
    double dn=d0-d1;
    if(fabs(dn)<EPS)return false;
    double t=d0/dn;
    t=max(0.0,min(1.0,t));
    hit=p0+t*(p1-p0);
    Vec3 e0=v1-v0,e1=v2-v0,e2=hit-v0;
    double d00=e0.dot(e0),d01=e0.dot(e1),d11=e1.dot(e1);
    double d20=e2.dot(e0),d21=e2.dot(e1);
    double det=d00*d11-d01*d01;
    if(fabs(det)<EPS)return false;
    double u=(d11*d20-d01*d21)/det;
    double v=(d00*d21-d01*d20)/det;
    double ep=-1e-6;
    return u>=ep&&v>=ep&&u+v<=1-ep;
}

bool triTriIsect(const Vec3&a0,const Vec3&a1,const Vec3&a2,
                 const Vec3&b0,const Vec3&b1,const Vec3&b2,
                 Vec3&s0,Vec3&s1){
    Vec3 pts[6];int np=0;
    Vec3 h;
    if(edgeTriHit(a0,a1,b0,b1,b2,h))pts[np++]=h;
    if(edgeTriHit(a1,a2,b0,b1,b2,h))pts[np++]=h;
    if(edgeTriHit(a2,a0,b0,b1,b2,h))pts[np++]=h;
    if(edgeTriHit(b0,b1,a0,a1,a2,h))pts[np++]=h;
    if(edgeTriHit(b1,b2,a0,a1,a2,h))pts[np++]=h;
    if(edgeTriHit(b2,b0,a0,a1,a2,h))pts[np++]=h;
    if(np<2)return false;
    double mx=0;s0=pts[0];s1=pts[1];
    for(int i=0;i<np;i++)for(int j=i+1;j<np;j++){
        double d=(pts[i]-pts[j]).len2();
        if(d>mx){mx=d;s0=pts[i];s1=pts[j];}
    }
    return mx>1e-16;
}

struct LocalFrame {
    Vec3 origin,u,v;
    void init(const Vec3&p0,const Vec3&p1,const Vec3&p2){
        origin=p0;
        u=p1-p0;double ul=u.len();
        if(ul>EPS)u=u*(1.0/ul);
        Vec3 n=u.cross(p2-p0);double nl=n.len();
        if(nl>EPS)n=n*(1.0/nl);
        v=n.cross(u);
    }
    Vec2 proj(const Vec3&p)const{Vec3 d=p-origin;return{d.dot(u),d.dot(v)};}
    Vec3 unproj(const Vec2&p)const{return origin+u*p.x+v*p.y;}
};

struct CDT {
    vector<Vec2> pts;
    vector<array<int,3>> tris;

    void init(const Vec2&a,const Vec2&b,const Vec2&c){
        pts={a,b,c};
        tris.push_back({0,1,2});
        if((pts[1]-pts[0]).cross(pts[2]-pts[0])<0)
            swap(tris[0][1],tris[0][2]);
    }

    int addPt(const Vec2&p){
        for(int i=0;i<(int)pts.size();i++)
            if((pts[i]-p).len2()<1e-14)return i;
        int id=(int)pts.size();
        pts.push_back(p);
        insertPt(id);
        return id;
    }

    void insertPt(int pid){
        Vec2 p=pts[pid];
        for(int i=0;i<(int)tris.size();i++){
            Vec2 a=pts[tris[i][0]],b=pts[tris[i][1]],c=pts[tris[i][2]];
            double d0=(p-a).cross(b-a);
            double d1=(p-b).cross(c-b);
            double d2=(p-c).cross(a-c);
            double area=fabs((b-a).cross(c-a));
            double ep=area*1e-8+1e-15;
            if(d0<-ep||d1<-ep||d2<-ep)continue;
            if(fabs(d0)<ep){splitEdge(i,2,pid);return;}
            if(fabs(d1)<ep){splitEdge(i,0,pid);return;}
            if(fabs(d2)<ep){splitEdge(i,1,pid);return;}
            auto old=tris[i];
            tris[i]={old[0],old[1],pid};
            tris.push_back({old[1],old[2],pid});
            tris.push_back({old[2],old[0],pid});
            return;
        }
    }

    void splitEdge(int ti,int ei,int pid){
        int a=tris[ti][ei],b=tris[ti][(ei+1)%3],c=tris[ti][(ei+2)%3];
        int tj=-1,ej=-1;
        for(int j=0;j<(int)tris.size();j++){
            if(j==ti)continue;
            for(int k=0;k<3;k++){
                if(tris[j][(k+1)%3]==c&&tris[j][(k+2)%3]==b){tj=j;ej=k;break;}
            }
            if(tj>=0)break;
        }
        tris[ti]={a,b,pid};
        tris.push_back({a,pid,c});
        if(tj>=0){
            int d=tris[tj][ej];
            tris[tj]={d,c,pid};
            tris.push_back({d,pid,b});
        }
    }

    bool hasEdge(int u,int v)const{
        for(auto&t:tris)for(int j=0;j<3;j++)
            if((t[j]==u&&t[(j+1)%3]==v)||(t[j]==v&&t[(j+1)%3]==u))return true;
        return false;
    }

    bool segCross(Vec2 a,Vec2 b,Vec2 c,Vec2 d){
        double d1=(b-a).cross(c-a),d2=(b-a).cross(d-a);
        double d3=(d-c).cross(a-c),d4=(d-c).cross(b-c);
        if(d1*d2<-1e-20&&d3*d4<-1e-20)return true;
        return false;
    }

    bool tryFlip(int ti,int ei){
        int a=tris[ti][ei],b=tris[ti][(ei+1)%3],c=tris[ti][(ei+2)%3];
        int tj=-1,ej=-1;
        for(int j=0;j<(int)tris.size();j++){
            if(j==ti)continue;
            for(int k=0;k<3;k++){
                if(tris[j][(k+1)%3]==c&&tris[j][(k+2)%3]==b){tj=j;ej=k;break;}
            }
            if(tj>=0)break;
        }
        if(tj<0)return false;
        int d=tris[tj][ej];
        double a1=(pts[b]-pts[a]).cross(pts[d]-pts[a]);
        double a2=(pts[d]-pts[a]).cross(pts[c]-pts[a]);
        if(a1<1e-15||a2<1e-15)return false;
        tris[ti]={a,b,d};
        tris[tj]={a,d,c};
        return true;
    }

    void enforceEdge(int u,int v){
        for(int iter=0;iter<200&&!hasEdge(u,v);iter++){
            bool flipped=false;
            for(int i=0;i<(int)tris.size()&&!flipped;i++){
                for(int j=0;j<3;j++){
                    int a=tris[i][(j+1)%3],b=tris[i][(j+2)%3];
                    if(a==u||a==v||b==u||b==v)continue;
                    if(segCross(pts[u],pts[v],pts[a],pts[b])){
                        if(tryFlip(i,j)){flipped=true;break;}
                    }
                }
            }
            if(!flipped)break;
        }
    }

    void removeDegenerate(){
        vector<array<int,3>> good;
        for(auto&t:tris){
            if(t[0]==t[1]||t[1]==t[2]||t[2]==t[0])continue;
            double a=(pts[t[1]]-pts[t[0]]).cross(pts[t[2]]-pts[t[0]]);
            if(fabs(a)>1e-20)good.push_back(t);
        }
        tris=good;
    }
};

int main(){
    // Fast I/O
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int VA,FA,VB,FB;
    scanf("%d%d%d%d",&VA,&FA,&VB,&FB);

    vector<Vec3> vertA(VA),vertB(VB);
    vector<array<int,3>> faceA(FA),faceB(FB);

    for(int i=0;i<VA;i++)scanf("%lf%lf%lf",&vertA[i].x,&vertA[i].y,&vertA[i].z);
    for(int i=0;i<FA;i++)scanf("%d%d%d",&faceA[i][0],&faceA[i][1],&faceA[i][2]);
    for(int i=0;i<VB;i++)scanf("%lf%lf%lf",&vertB[i].x,&vertB[i].y,&vertB[i].z);
    for(int i=0;i<FB;i++)scanf("%d%d%d",&faceB[i][0],&faceB[i][1],&faceB[i][2]);

    // Build BVHs
    BVH bvhA,bvhB;
    bvhA.build(vertA,faceA);
    bvhB.build(vertB,faceB);

    // Find all intersecting pairs and compute intersection segments
    struct ISeg { Vec3 p0,p1; };
    vector<vector<ISeg>> segsOnA(FA),segsOnB(FB);
    bool anyIntersection=false;

    for(int ai=0;ai<FA;ai++){
        AABB boxA;
        for(int j=0;j<3;j++) boxA.expand(vertA[faceA[ai][j]]);
        bvhB.query(boxA,[&](int bi){
            Vec3 s0,s1;
            if(triTriIsect(vertA[faceA[ai][0]],vertA[faceA[ai][1]],vertA[faceA[ai][2]],
                           vertB[faceB[bi][0]],vertB[faceB[bi][1]],vertB[faceB[bi][2]],s0,s1)){
                segsOnA[ai].push_back({s0,s1});
                segsOnB[bi].push_back({s0,s1});
                anyIntersection=true;
            }
        });
    }

    // Result vertices and faces
    vector<Vec3> RV;
    vector<array<int,3>> RF;

    if(!anyIntersection){
        // No intersection: either disjoint or one contains the other
        bool AinB=isInside(vertA[0],vertB,faceB,bvhB);
        bool BinA=isInside(vertB[0],vertA,faceA,bvhA);
        if(AinB){
            // A inside B → output B
            RV=vertB; RF=faceB;
        } else if(BinA){
            // B inside A → output A
            RV=vertA; RF=faceA;
        } else {
            // Disjoint → output both
            RV=vertA;
            RV.insert(RV.end(),vertB.begin(),vertB.end());
            RF=faceA;
            for(auto f:faceB){
                f[0]+=VA;f[1]+=VA;f[2]+=VA;
                RF.push_back(f);
            }
        }
    } else {
        // Full CSG pipeline
        // Global vertex pool: starts with A's and B's vertices
        RV=vertA;
        RV.insert(RV.end(),vertB.begin(),vertB.end());

        // Helper: find or add a vertex in RV (with merging)
        auto findOrAdd=[&](const Vec3&p)->int{
            // Only search among intersection points (after VA+VB)
            int base=VA+VB;
            for(int i=base;i<(int)RV.size();i++){
                if((RV[i]-p).len2()<1e-14)return i;
            }
            RV.push_back(p);
            return (int)RV.size()-1;
        };

        // Process mesh A triangles
        for(int ai=0;ai<FA;ai++){
            if(segsOnA[ai].empty()){
                // Classify whole triangle
                Vec3 cen=(vertA[faceA[ai][0]]+vertA[faceA[ai][1]]+vertA[faceA[ai][2]])*(1.0/3.0);
                if(!isInside(cen,vertB,faceB,bvhB)){
                    RF.push_back(faceA[ai]);
                }
            } else {
                // Split triangle and classify sub-triangles
                int vi0=faceA[ai][0],vi1=faceA[ai][1],vi2=faceA[ai][2];
                Vec3 p0=vertA[vi0],p1=vertA[vi1],p2=vertA[vi2];

                LocalFrame lf;
                lf.init(p0,p1,p2);
                Vec2 q0=lf.proj(p0),q1=lf.proj(p1),q2=lf.proj(p2);

                CDT cdt;
                cdt.init(q0,q1,q2);

                // Map CDT point index → global vertex index
                vector<int> cdtToGlobal={vi0,vi1,vi2};

                for(auto&seg:segsOnA[ai]){
                    Vec2 sp0=lf.proj(seg.p0),sp1=lf.proj(seg.p1);
                    int gi0=findOrAdd(seg.p0);
                    int gi1=findOrAdd(seg.p1);

                    int ci0=cdt.addPt(sp0);
                    while((int)cdtToGlobal.size()<=ci0) cdtToGlobal.push_back(-1);
                    cdtToGlobal[ci0]=gi0;

                    int ci1=cdt.addPt(sp1);
                    while((int)cdtToGlobal.size()<=ci1) cdtToGlobal.push_back(-1);
                    cdtToGlobal[ci1]=gi1;

                    cdt.enforceEdge(ci0,ci1);
                }

                cdt.removeDegenerate();

                // Classify each sub-triangle
                for(auto&t:cdt.tris){
                    // Compute 3D centroid
                    Vec2 c2=(cdt.pts[t[0]]+cdt.pts[t[1]]+cdt.pts[t[2]])*(1.0/3.0);
                    Vec3 c3=lf.unproj(c2);
                    if(!isInside(c3,vertB,faceB,bvhB)){
                        int g0=(t[0]<(int)cdtToGlobal.size())?cdtToGlobal[t[0]]:-1;
                        int g1=(t[1]<(int)cdtToGlobal.size())?cdtToGlobal[t[1]]:-1;
                        int g2=(t[2]<(int)cdtToGlobal.size())?cdtToGlobal[t[2]]:-1;
                        if(g0>=0&&g1>=0&&g2>=0){
                            RF.push_back({g0,g1,g2});
                        }
                    }
                }
            }
        }

        // Process mesh B triangles
        for(int bi=0;bi<FB;bi++){
            if(segsOnB[bi].empty()){
                Vec3 cen=(vertB[faceB[bi][0]]+vertB[faceB[bi][1]]+vertB[faceB[bi][2]])*(1.0/3.0);
                if(!isInside(cen,vertA,faceA,bvhA)){
                    RF.push_back({faceB[bi][0]+VA,faceB[bi][1]+VA,faceB[bi][2]+VA});
                }
            } else {
                int vi0=faceB[bi][0]+VA,vi1=faceB[bi][1]+VA,vi2=faceB[bi][2]+VA;
                Vec3 p0=vertB[faceB[bi][0]],p1=vertB[faceB[bi][1]],p2=vertB[faceB[bi][2]];

                LocalFrame lf;
                lf.init(p0,p1,p2);
                Vec2 q0=lf.proj(p0),q1=lf.proj(p1),q2=lf.proj(p2);

                CDT cdt;
                cdt.init(q0,q1,q2);

                vector<int> cdtToGlobal={vi0,vi1,vi2};

                for(auto&seg:segsOnB[bi]){
                    Vec2 sp0=lf.proj(seg.p0),sp1=lf.proj(seg.p1);
                    int gi0=findOrAdd(seg.p0);
                    int gi1=findOrAdd(seg.p1);

                    int ci0=cdt.addPt(sp0);
                    while((int)cdtToGlobal.size()<=ci0) cdtToGlobal.push_back(-1);
                    cdtToGlobal[ci0]=gi0;

                    int ci1=cdt.addPt(sp1);
                    while((int)cdtToGlobal.size()<=ci1) cdtToGlobal.push_back(-1);
                    cdtToGlobal[ci1]=gi1;

                    cdt.enforceEdge(ci0,ci1);
                }

                cdt.removeDegenerate();

                for(auto&t:cdt.tris){
                    Vec2 c2=(cdt.pts[t[0]]+cdt.pts[t[1]]+cdt.pts[t[2]])*(1.0/3.0);
                    Vec3 c3=lf.unproj(c2);
                    if(!isInside(c3,vertA,faceA,bvhA)){
                        int g0=(t[0]<(int)cdtToGlobal.size())?cdtToGlobal[t[0]]:-1;
                        int g1=(t[1]<(int)cdtToGlobal.size())?cdtToGlobal[t[1]]:-1;
                        int g2=(t[2]<(int)cdtToGlobal.size())?cdtToGlobal[t[2]]:-1;
                        if(g0>=0&&g1>=0&&g2>=0){
                            RF.push_back({g0,g1,g2});
                        }
                    }
                }
            }
        }
    }

    // Remove unused vertices and compact
    int NV=(int)RV.size();
    vector<int> used(NV,-1);
    for(auto&f:RF)for(int j=0;j<3;j++) used[f[j]]=1;
    int cnt=0;
    for(int i=0;i<NV;i++) if(used[i]>=0) used[i]=cnt++;
    vector<Vec3> FV(cnt);
    for(int i=0;i<NV;i++) if(used[i]>=0) FV[used[i]]=RV[i];
    for(auto&f:RF)for(int j=0;j<3;j++) f[j]=used[f[j]];

    printf("%d %d\n",cnt,(int)RF.size());
    for(int i=0;i<cnt;i++) printf("%.17g %.17g %.17g\n",FV[i].x,FV[i].y,FV[i].z);
    for(auto&f:RF) printf("%d %d %d\n",f[0],f[1],f[2]);
