; SPIR-V Geometry Shader: Shrink triangles toward centroid
               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %inColor %outColor
               OpExecutionMode %main Triangles
               OpExecutionMode %main Invocations 1
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 3

               ; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %inColor Location 2
               OpDecorate %outColor Location 0

               ; Types
      %void  = OpTypeVoid
  %voidfunc  = OpTypeFunction %void
     %float  = OpTypeFloat 32
   %v3float  = OpTypeVector %float 3
   %v4float  = OpTypeVector %float 4
      %uint  = OpTypeInt 32 0
       %int  = OpTypeInt 32 1

     %int_0  = OpConstant %int 0
     %int_1  = OpConstant %int 1
     %int_2  = OpConstant %int 2
    %uint_3  = OpConstant %uint 3
   %float_1  = OpConstant %float 1
 %float_0_7  = OpConstant %float 0.7
 %float_third = OpConstant %float 0.333333343

; Input gl_PerVertex block (array of 3)
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_3
%ptr_in_arr = OpTypePointer Input %arr_pv_in
%ptr_in_v4  = OpTypePointer Input %v4float
     %gl_in = OpVariable %ptr_in_arr Input

; Input color (array of 3 vec3)
%arr_v3_3   = OpTypeArray %v3float %uint_3
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_3
%ptr_in_v3  = OpTypePointer Input %v3float
   %inColor = OpVariable %ptr_in_arr_v3 Input

; Output gl_PerVertex block
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
    %gl_out = OpVariable %ptr_out_pv Output

; Output color
%ptr_out_v3 = OpTypePointer Output %v3float
  %outColor = OpVariable %ptr_out_v3 Output

      %main = OpFunction %void None %voidfunc
     %entry = OpLabel

               ; Load 3 positions
     %pp0    = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
      %p0    = OpLoad %v4float %pp0
     %pp1    = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
      %p1    = OpLoad %v4float %pp1
     %pp2    = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
      %p2    = OpLoad %v4float %pp2

               ; Load 3 colors
    %cptr0   = OpAccessChain %ptr_in_v3 %inColor %int_0
      %c0    = OpLoad %v3float %cptr0
    %cptr1   = OpAccessChain %ptr_in_v3 %inColor %int_1
      %c1    = OpLoad %v3float %cptr1
    %cptr2   = OpAccessChain %ptr_in_v3 %inColor %int_2
      %c2    = OpLoad %v3float %cptr2

               ; Centroid = (p0 + p1 + p2) / 3
    %sum01   = OpFAdd %v4float %p0 %p1
   %sum012   = OpFAdd %v4float %sum01 %p2
  %third_v   = OpCompositeConstruct %v4float %float_third %float_third %float_third %float_third
  %centroid  = OpFMul %v4float %sum012 %third_v

               ; Shrink factor vector
   %s_vec    = OpCompositeConstruct %v4float %float_0_7 %float_0_7 %float_0_7 %float_0_7

               ; Vertex 0: C + s * (p0 - C)
    %d0      = OpFSub %v4float %p0 %centroid
   %sd0      = OpFMul %v4float %s_vec %d0
  %new0      = OpFAdd %v4float %centroid %sd0
  %outpos0   = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %outpos0 %new0
               OpStore %outColor %c0
               OpEmitVertex

               ; Vertex 1: C + s * (p1 - C)
    %d1      = OpFSub %v4float %p1 %centroid
   %sd1      = OpFMul %v4float %s_vec %d1
  %new1      = OpFAdd %v4float %centroid %sd1
  %outpos1   = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %outpos1 %new1
               OpStore %outColor %c1
               OpEmitVertex

               ; Vertex 2: C + s * (p2 - C)
    %d2      = OpFSub %v4float %p2 %centroid
   %sd2      = OpFMul %v4float %s_vec %d2
  %new2      = OpFAdd %v4float %centroid %sd2
  %outpos2   = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %outpos2 %new2
               OpStore %outColor %c2
               OpEmitVertex

               OpEndPrimitive
               OpReturn
               OpFunctionEnd
