use std::io::{self, Read, Write, BufWriter};
use std::collections::BinaryHeap;

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let stdout = io::stdout();
    let mut out = BufWriter::new(stdout.lock());

    let mut iter = input.split_ascii_whitespace();
    let n: usize = iter.next().unwrap().parse().unwrap();
    let m: usize = iter.next().unwrap().parse().unwrap();

    let mut adj: Vec<Vec<usize>> = vec![Vec::new(); n];
    let mut edges: Vec<(usize, usize)> = Vec::with_capacity(m);

    for _ in 0..m {
        let u: usize = iter.next().unwrap().parse().unwrap();
        let v: usize = iter.next().unwrap().parse().unwrap();
        adj[u].push(v);
        adj[v].push(u);
        edges.push((u, v));
    }

    let mut in_cover = vec![false; n];
    let mut removed = vec![false; n];
    let mut degree: Vec<usize> = (0..n).map(|i| adj[i].len()).collect();

    // Kernelization: repeatedly apply reduction rules
    let mut changed = true;
    while changed {
        changed = false;
        for v in 0..n {
            if removed[v] || in_cover[v] {
                continue;
            }
            if degree[v] == 0 {
                removed[v] = true;
                changed = true;
            } else if degree[v] == 1 {
                // Take the neighbor
                let mut neighbor = usize::MAX;
                for &u in &adj[v] {
                    if !removed[u] && !in_cover[u] {
                        neighbor = u;
                        break;
                    }
                }
                if neighbor == usize::MAX {
                    removed[v] = true;
                    changed = true;
                    continue;
                }
                in_cover[neighbor] = true;
                // Remove neighbor and update degrees
                for i in 0..adj[neighbor].len() {
                    let w = adj[neighbor][i];
                    if !removed[w] && !in_cover[w] && degree[w] > 0 {
                        degree[w] -= 1;
                    }
                }
                removed[v] = true;
                changed = true;
            }
        }
    }

    // Greedy: pick highest degree vertex repeatedly
    // Use a max-heap of (degree, vertex)
    let mut heap: BinaryHeap<(usize, usize)> = BinaryHeap::new();
    for v in 0..n {
        if !removed[v] && !in_cover[v] && degree[v] > 0 {
            heap.push((degree[v], v));
        }
    }

    while let Some((d, v)) = heap.pop() {
        if removed[v] || in_cover[v] || degree[v] == 0 {
            continue;
        }
        if degree[v] != d {
            if degree[v] > 0 {
                heap.push((degree[v], v));
            }
            continue;
        }
        in_cover[v] = true;
        for i in 0..adj[v].len() {
            let w = adj[v][i];
            if !removed[w] && !in_cover[w] && degree[w] > 0 {
                degree[w] -= 1;
                if degree[w] > 0 {
                    heap.push((degree[w], w));
                }
            }
        }
    }

    // Verify all edges covered and collect result
    // (greedy + kernelization should cover all, but let's be safe)
    for &(u, v) in &edges {
        if !in_cover[u] && !in_cover[v] {
            in_cover[u] = true;
        }
    }

    // Local search: try removing vertices from cover if all their edges are still covered
    // Sort cover vertices by degree (ascending) to try removing low-degree first
    let mut cover_verts: Vec<usize> = (0..n).filter(|&v| in_cover[v]).collect();
    // Recompute actual neighbor-in-cover counts
    let mut cover_neighbors: Vec<usize> = vec![0; n];
    for v in 0..n {
        for &u in &adj[v] {
            if in_cover[u] {
                cover_neighbors[v] += 1;
            }
        }
    }

    cover_verts.sort_by_key(|&v| adj[v].len());

    for &v in &cover_verts {
        if !in_cover[v] {
            continue;
        }
        // Can remove v if every neighbor of v has at least one OTHER neighbor in cover
        // i.e., for every neighbor w of v: if (u,w) is an edge implied by w being neighbor of v,
        // then w must be in cover, OR w has another cover neighbor besides v
        let mut can_remove = true;
        for &w in &adj[v] {
            if in_cover[w] {
                continue; // edge (v,w) covered by w
            }
            // w not in cover, so edge (v,w) is only covered by v
            // if we remove v, edge uncovered
            can_remove = false;
            break;
        }
        if can_remove {
            in_cover[v] = false;
            for &w in &adj[v] {
                cover_neighbors[w] -= 1;
            }
        }
    }

    let result: Vec<usize> = (0..n).filter(|&v| in_cover[v]).collect();
    writeln!(out, "{}", result.len()).unwrap();
    let mut first = true;
    for v in &result {
        if !first {
            write!(out, " ").unwrap();
        }
        write!(out, "{}", v).unwrap();
        first = false;
    }
    writeln!(out).unwrap();
}
