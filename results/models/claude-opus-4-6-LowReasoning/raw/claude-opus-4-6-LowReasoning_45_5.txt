import struct

words = []

def w(val):
    words.append(val & 0xFFFFFFFF)

def op(wordcount, opcode, *operands):
    w((wordcount << 16) | opcode)
    for o in operands:
        w(o)

def op_str(opcode, *pre_operands, string="", post_operands=()):
    # Encode string as null-terminated, padded to 4-byte boundary
    encoded = string.encode('utf-8') + b'\x00'
    while len(encoded) % 4 != 0:
        encoded += b'\x00'
    str_words = len(encoded) // 4
    total = 1 + len(pre_operands) + str_words + len(post_operands)
    w((total << 16) | opcode)
    for o in pre_operands:
        w(o)
    for i in range(str_words):
        w(struct.unpack('<I', encoded[i*4:i*4+4])[0])
    for o in post_operands:
        w(o)

def float_to_uint(f):
    return struct.unpack('<I', struct.pack('<f', f))[0]

ID_CONST_0_8 = 23

ID_INT = 25

ID_ICONST_4 = 27  # cameraPos member index

# Temporaries start at 30
T = 30

def next_t():
    global T
    r = T
    T += 1
    return r

op(4, 43, ID_INT, ID_ICONST_3, 3)

op(4, 43, ID_INT, ID_ICONST_4, 4)

# Variables
# OpVariable %10 %13 Input
op(4, 59, ID_PTR_INPUT_VEC3, ID_VAR_WORLDPOS, 1)
# OpVariable %10 %14 Input
op(4, 59, ID_PTR_INPUT_VEC3, ID_VAR_NORMAL, 1)

# Function
# OpFunction %2 %17 None %3

op(5, 54, ID_VOID, ID_MAIN, 0, ID_FUNC_TYPE)

# OpLabel %18
op(2, 248, ID_LABEL)

# Load worldPos
t_worldpos = next_t()  # 30
op(4, 61, ID_VEC3, t_worldpos, ID_VAR_WORLDPOS)

# Load normal
t_normal_raw = next_t()  # 31
op(4, 61, ID_VEC3, t_normal_raw, ID_VAR_NORMAL)

# Normalize normal -> N
t_N = next_t()  # 32
# OpExtInst %5 %32 %1 Normalize(69) %31
op(6, 12, ID_VEC3, t_N, ID_GLSL, 69, t_normal_raw)

# Access chain for lightPos (member 3)
t_lightpos_ptr = next_t()  # 33
# OpAccessChain %12 %33 %16 %26
op(5, 65, ID_PTR_UNIFORM_VEC4, t_lightpos_ptr, ID_VAR_UBO, ID_ICONST_3)

# Load lightPos vec4
t_lightpos4 = next_t()  # 34
op(4, 61, ID_VEC4, t_lightpos4, t_lightpos_ptr)

# Extract xyz from lightPos -> vec3
t_lx = next_t()  # 35
op(5, 81, ID_FLOAT, t_lx, t_lightpos4, 0)

op(5, 81, ID_FLOAT, t_ly, t_lightpos4, 1)

t_lightpos3 = next_t()  # 38
op(6, 80, ID_VEC3, t_lightpos3, t_lx, t_ly, t_lz)

# Access chain for cameraPos (member 4)
t_campos_ptr = next_t()  # 39
op(5, 65, ID_PTR_UNIFORM_VEC4, t_campos_ptr, ID_VAR_UBO, ID_ICONST_4)

# Load cameraPos vec4
t_campos4 = next_t()  # 40
op(4, 61, ID_VEC4, t_campos4, t_campos_ptr)

# Extract xyz from cameraPos -> vec3
t_cx = next_t()  # 41

op(6, 80, ID_VEC3, t_campos3, t_cx, t_cy, t_cz)

# L = normalize(lightPos - worldPos)
t_l_unnorm = next_t()  # 45
op(5, 131, ID_VEC3, t_l_unnorm, t_lightpos3, t_worldpos)  # OpFSub

op(6, 12, ID_VEC3, t_L, ID_GLSL, 69, t_l_unnorm)  # Normalize

# V = normalize(cameraPos - worldPos)
t_v_unnorm = next_t()  # 47
op(5, 131, ID_VEC3, t_v_unnorm, t_campos3, t_worldpos)

op(6, 12, ID_VEC3, t_V, ID_GLSL, 69, t_v_unnorm)

# negL = -L (OpFNegate)
t_negL = next_t()  # 49
op(4, 127, ID_VEC3, t_negL, t_L)

# R = reflect(-L, N)  -> GLSL Reflect = 71
t_R = next_t()  # 50
op(7, 12, ID_VEC3, t_R, ID_GLSL, 71, t_negL, t_N)

# diffuse = max(dot(N, L), 0)
t_ndotl = next_t()  # 51
op(5, 148, ID_FLOAT, t_ndotl, t_N, t_L)  # OpDot
t_diffuse = next_t()  # 52
op(7, 12, ID_FLOAT, t_diffuse, ID_GLSL, 40, t_ndotl, ID_CONST_0)  # FMax

# specular = pow(max(dot(R, V), 0), 32)
t_rdotv = next_t()  # 53
op(5, 148, ID_FLOAT, t_rdotv, t_R, t_V)

op(7, 12, ID_FLOAT, t_rdotv_clamped, ID_GLSL, 40, t_rdotv, ID_CONST_0)
t_specular = next_t()  # 55
op(7, 12, ID_FLOAT, t_specular, ID_GLSL, 26, t_rdotv_clamped, ID_CONST_32)  # Pow

# intensity = 0.2 + 0.5*diffuse + 0.8*specular
t_diff_term = next_t()  # 56
op(5, 133, ID_FLOAT, t_diff_term, ID_CONST_0_5, t_diffuse)  # OpFMul
t_spec_term = next_t()  # 57
op(5, 133, ID_FLOAT, t_spec_term, ID_CONST_0_8, t_specular)
t_sum1 = next_t()  # 58
op(5, 129, ID_FLOAT, t_sum1, ID_CONST_0_2, t_diff_term)  # OpFAdd
t_intensity = next_t()  # 59
op(5, 129, ID_FLOAT, t_intensity, t_sum1, t_spec_term)

# outColor = vec4(intensity, intensity, intensity, 1.0)
t_out = next_t()  # 60
op(7, 80, ID_VEC4, t_out, t_intensity, t_intensity, t_intensity, ID_CONST_1)

# OpStore outColor
op(3, 62, ID_VAR_OUTCOLOR, t_out)

op(1, 253)

op(1, 56)
