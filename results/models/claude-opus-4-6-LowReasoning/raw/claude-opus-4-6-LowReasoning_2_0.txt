using System;
using System.Collections.Generic;
using System.IO;
using System.Text;

class ChinesePostman
{
    static int N, M;
    static List<int[]>[] adj;
    static int[] eu, ev, ew;

    // Binary min-heap for Dijkstra
    static int[] heapNode, heapPos, heapSize;
    static long[] heapKey;

    static void HeapInit()
    {
        heapSize = new int[]{0};
        if (heapNode == null || heapNode.Length < N)
        {
            heapNode = new int[N];
            heapPos = new int[N];
            heapKey = new long[N];
        }
        for (int i = 0; i < N; i++) heapPos[i] = -1;
    }

    static void HeapPush(int node, long key)
    {
        int i = heapSize[0]++;
        heapNode[i] = node; heapKey[i] = key; heapPos[node] = i;
        while (i > 0)
        {
            int p = (i - 1) / 2;
            if (heapKey[p] <= heapKey[i]) break;
            Swap(i, p); i = p;
        }
    }

    static void HeapDecrease(int node, long key)
    {
        int i = heapPos[node];
        heapKey[i] = key;
        while (i > 0)
        {
            int p = (i - 1) / 2;
            if (heapKey[p] <= heapKey[i]) break;
            Swap(i, p); i = p;
        }
    }

    static int HeapPop()
    {
        int node = heapNode[0];
        heapPos[node] = -1;
        int last = --heapSize[0];
        if (last > 0)
        {
            heapNode[0] = heapNode[last]; heapKey[0] = heapKey[last]; heapPos[heapNode[0]] = 0;
            int i = 0;
            while (true)
            {
                int l = 2*i+1, r = 2*i+2, s = i;
                if (l < heapSize[0] && heapKey[l] < heapKey[s]) s = l;
                if (r < heapSize[0] && heapKey[r] < heapKey[s]) s = r;
                if (s == i) break;
                Swap(i, s); i = s;
            }
        }
        return node;
    }

    static void Swap(int a, int b)
    {
        int tn = heapNode[a]; heapNode[a] = heapNode[b]; heapNode[b] = tn;
        long tk = heapKey[a]; heapKey[a] = heapKey[b]; heapKey[b] = tk;
        heapPos[heapNode[a]] = a; heapPos[heapNode[b]] = b;
    }

    static long[] Dijkstra(int src, out int[] predNode, out int[] predEdge)
    {
        var dist = new long[N];
        predNode = new int[N];
        predEdge = new int[N];
        for (int i = 0; i < N; i++) { dist[i] = long.MaxValue; predNode[i] = -1; predEdge[i] = -1; }
        dist[src] = 0;
        HeapInit();
        HeapPush(src, 0);
        while (heapSize[0] > 0)
        {
            int u = HeapPop();
            long d = dist[u];
            foreach (var e in adj[u])
            {
                int v = e[0], w = e[1], ei = e[2];
                long nd = d + w;
                if (nd < dist[v])
                {
                    dist[v] = nd;
                    predNode[v] = u;
                    predEdge[v] = ei;
                    if (heapPos[v] == -1) HeapPush(v, nd);
                    else HeapDecrease(v, nd);
                }
            }
        }
        return dist;
    }

    static void Main()
    {
        var input = Console.In.ReadToEnd();
        int pos = 0;

        int ReadInt()
        {
            while (pos < input.Length && (input[pos] == ' ' || input[pos] == '\n' || input[pos] == '\r')) pos++;
            int val = 0;
            while (pos < input.Length && input[pos] >= '0' && input[pos] <= '9')
                val = val * 10 + (input[pos++] - '0');
            return val;
        }

        N = ReadInt(); M = ReadInt();
        adj = new List<int[]>[N];
        for (int i = 0; i < N; i++) adj[i] = new List<int[]>();
        eu = new int[M]; ev = new int[M]; ew = new int[M];

        for (int i = 0; i < M; i++)
        {
            int u = ReadInt(), v = ReadInt(), w = ReadInt();
            eu[i] = u; ev[i] = v; ew[i] = w;
            adj[u].Add(new int[]{v, w, i});
            adj[v].Add(new int[]{u, w, i});
        }

        var degree = new int[N];
        for (int i = 0; i < M; i++) { degree[eu[i]]++; degree[ev[i]]++; }
        var odds = new List<int>();
        for (int i = 0; i < N; i++) if (degree[i] % 2 == 1) odds.Add(i);

        var edgeCount = new int[M];
        for (int i = 0; i < M; i++) edgeCount[i] = 1;

        if (odds.Count > 0)
        {
            int numOdd = odds.Count;

            if (numOdd <= 20)
            {
                var dist = new long[numOdd][];
                var predN = new int[numOdd][];
                var predE = new int[numOdd][];
                for (int oi = 0; oi < numOdd; oi++)
                    dist[oi] = Dijkstra(odds[oi], out predN[oi], out predE[oi]);

                int full = (1 << numOdd) - 1;
                var dp = new long[1 << numOdd];
                var dpPair = new int[1 << numOdd];
                for (int s = 0; s <= full; s++) dp[s] = long.MaxValue;
                dp[0] = 0;
                for (int s = 0; s <= full; s++)
                {
                    if (dp[s] == long.MaxValue) continue;
                    int first = -1;
                    for (int b = 0; b < numOdd; b++) if ((s & (1 << b)) == 0) { first = b; break; }
                    if (first == -1) continue;
                    for (int b = first + 1; b < numOdd; b++)
                    {
                        if ((s & (1 << b)) != 0) continue;
                        int ns = s | (1 << first) | (1 << b);
                        long cost = dist[first][odds[b]];
                        if (dp[s] + cost < dp[ns])
                        {
                            dp[ns] = dp[s] + cost;
                            dpPair[ns] = first * numOdd + b;
                        }
                    }
                }
                int mask = full;
                while (mask != 0)
                {
                    int pair = dpPair[mask];
                    int a = pair / numOdd, b = pair % numOdd;
                    int cur = odds[b];
                    while (cur != odds[a])
                    {
                        edgeCount[predE[a][cur]]++;
                        cur = predN[a][cur];
                    }
                    mask ^= (1 << a) | (1 << b);
                }
            }
            else
            {
                // Greedy nearest-neighbor matching
                var matched = new bool[N];
                int remaining = numOdd;

                while (remaining > 0)
                {
                    int src = -1;
                    foreach (var o in odds) if (!matched[o]) { src = o; break; }
                    if (src == -1) break;

                    int[] pN, pE;
                    var dist = Dijkstra(src, out pN, out pE);

                    long bestDist = long.MaxValue;
                    int bestNode = -1;
                    foreach (var o in odds)
                    {
                        if (o == src || matched[o]) continue;
                        if (dist[o] < bestDist) { bestDist = dist[o]; bestNode = o; }
                    }

                    if (bestNode == -1) break;
                    matched[src] = true; matched[bestNode] = true;
                    remaining -= 2;

                    int cur = bestNode;
                    while (cur != src)
                    {
                        edgeCount[pE[cur]]++;
                        cur = pN[cur];
                    }
                }
            }
        }

        // Build Euler multigraph
        int totalSlots = 0;
        for (int i = 0; i < M; i++) totalSlots += edgeCount[i];
        var eulerAdj = new List<int[]>[N];
        for (int i = 0; i < N; i++) eulerAdj[i] = new List<int[]>();
        int sid = 0;
        var used = new bool[totalSlots * 2];
        for (int i = 0; i < M; i++)
        {
            for (int c = 0; c < edgeCount[i]; c++)
            {
                int id1 = sid++, id2 = sid++;
                eulerAdj[eu[i]].Add(new int[]{ev[i], id1, id2});
                eulerAdj[ev[i]].Add(new int[]{eu[i], id2, id1});
            }
        }

        // Hierholzer
        var circuit = new List<int>(totalSlots + 1);
        var stack = new Stack<int>();
        var ptr = new int[N];
        stack.Push(0);
        while (stack.Count > 0)
        {
            int v = stack.Peek();
            bool found = false;
            while (ptr[v] < eulerAdj[v].Count)
            {
                var e = eulerAdj[v][ptr[v]];
                ptr[v]++;
                if (!used[e[1]])
                {
                    used[e[1]] = true;
                    used[e[2]] = true;
                    stack.Push(e[0]);
                    found = true;
                    break;
                }
            }
            if (!found)
            {
                stack.Pop();
                circuit.Add(v);
            }
        }

        var sb = new StringBuilder(circuit.Count * 4);
        for (int i = 0; i < circuit.Count; i++)
        {
            if (i > 0) sb.Append(' ');
            sb.Append(circuit[i]);
        }
        using (var bw = new BufferedStream(Console.OpenStandardOutput(), 1 << 16))
        using (var sw = new StreamWriter(bw))
            sw.WriteLine(sb);
    }
}
