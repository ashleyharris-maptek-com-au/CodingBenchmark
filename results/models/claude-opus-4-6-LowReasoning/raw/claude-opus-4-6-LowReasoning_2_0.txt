using System;
using System.Collections.Generic;
using System.IO;
using System.Text;

class Solution
{
    static int N, M;
    static List<(int to, long w, int idx)>[] adj;

    static void Main()
    {
        var br = new BufferedStream(Console.OpenStandardInput(), 1 << 16);
        var reader = new StreamReader(br);

        var nm = reader.ReadLine().Split(' ', StringSplitOptions.RemoveEmptyEntries);
        N = int.Parse(nm[0]);
        M = int.Parse(nm[1]);

        adj = new List<(int, long, int)>[N];
        for (int i = 0; i < N; i++) adj[i] = new List<(int, long, int)>();

        var edgeU = new int[M];
        var edgeV = new int[M];
        var edgeW = new long[M];

        for (int i = 0; i < M; i++)
        {
            var parts = reader.ReadLine().Split(' ', StringSplitOptions.RemoveEmptyEntries);
            int u = int.Parse(parts[0]), v = int.Parse(parts[1]);
            long w = long.Parse(parts[2]);
            edgeU[i] = u; edgeV[i] = v; edgeW[i] = w;
            adj[u].Add((v, w, i));
            adj[v].Add((u, w, i));
        }

        var degree = new int[N];
        for (int i = 0; i < M; i++)
        {
            degree[edgeU[i]]++;
            degree[edgeV[i]]++;
        }

        var oddVerts = new List<int>();
        for (int i = 0; i < N; i++)
            if (degree[i] % 2 == 1) oddVerts.Add(i);

        var edgeCount = new int[M];
        for (int i = 0; i < M; i++) edgeCount[i] = 1;

        int k = oddVerts.Count;

        if (k > 0)
        {
            if (k <= 20)
            {
                // Exact: Dijkstra from each odd vertex, then bitmask DP matching
                var distArr = new long[k][];
                var prevNode = new int[k][];
                var prevEdgeArr = new int[k][];

                for (int i = 0; i < k; i++)
                {
                    var (d, p, pe) = Dijkstra(oddVerts[i]);
                    distArr[i] = d;
                    prevNode[i] = p;
                    prevEdgeArr[i] = pe;
                }

                // Bitmask DP
                int full = (1 << k) - 1;
                var dp = new long[1 << k];
                var dpPair = new int[1 << k];
                for (int i = 0; i < dp.Length; i++) dp[i] = long.MaxValue;
                dp[0] = 0;

                for (int mask = 0; mask < (1 << k); mask++)
                {
                    if (dp[mask] == long.MaxValue) continue;
                    int first = -1;
                    for (int b = 0; b < k; b++)
                        if ((mask & (1 << b)) == 0) { first = b; break; }
                    if (first < 0) continue;

                    for (int j = first + 1; j < k; j++)
                    {
                        if ((mask & (1 << j)) != 0) continue;
                        int nmask = mask | (1 << first) | (1 << j);
                        long cost = dp[mask] + distArr[first][oddVerts[j]];
                        if (cost < dp[nmask])
                        {
                            dp[nmask] = cost;
                            dpPair[nmask] = (first << 16) | j;
                        }
                    }
                }

                int cur = full;
                while (cur != 0)
                {
                    int packed = dpPair[cur];
                    int a = packed >> 16, b2 = packed & 0xFFFF;
                    // Trace shortest path from oddVerts[a] to oddVerts[b2]
                    int target = oddVerts[a];
                    int c = oddVerts[b2];
                    while (c != target)
                    {
                        edgeCount[prevEdgeArr[a][c]]++;
                        c = prevNode[a][c];
                    }
                    cur ^= (1 << a) | (1 << b2);
                }
            }
            else
            {
                // Greedy nearest-neighbor matching with Dijkstra on demand
                var oddSet = new HashSet<int>();
                for (int i = 0; i < k; i++) oddSet.Add(oddVerts[i]);
                var matched = new bool[N];

                // Sort odd vertices for determinism
                oddVerts.Sort();

                foreach (int src in oddVerts)
                {
                    if (matched[src]) continue;
                    // Dijkstra from src, stop when we find nearest unmatched odd vertex
                    var dist = new long[N];
                    var prev = new int[N];
                    var prevEdge = new int[N];
                    for (int i = 0; i < N; i++) { dist[i] = long.MaxValue; prev[i] = -1; prevEdge[i] = -1; }
                    dist[src] = 0;

                    var pq = new PriorityQueue<int, long>();
                    pq.Enqueue(src, 0);
                    int found = -1;

                    while (pq.Count > 0)
                    {
                        pq.TryDequeue(out int u, out long d);
                        if (d > dist[u]) continue;
                        if (u != src && oddSet.Contains(u) && !matched[u])
                        {
                            found = u;
                            break;
                        }
                        foreach (var (v, w, idx) in adj[u])
                        {
                            long nd = d + w;
                            if (nd < dist[v])
                            {
                                dist[v] = nd;
                                prev[v] = u;
                                prevEdge[v] = idx;
                                pq.Enqueue(v, nd);
                            }
                        }
                    }

                    if (found >= 0)
                    {
                        matched[src] = true;
                        matched[found] = true;
                        int c = found;
                        while (c != src)
                        {
                            edgeCount[prevEdge[c]]++;
                            c = prev[c];
                        }
                    }
                }
            }
        }

        // Build multigraph for Euler circuit using Hierholzer's
        // Instead of expanding edges, track remaining traversals per directed half-edge
        // We'll use edge index + direction. For each original edge i with count c,
        // we need c traversals total (in either direction).

        // Build adjacency with edge-instance IDs
        int totalE = 0;
        for (int i = 0; i < M; i++) totalE += edgeCount[i];

        // For Hierholzer, we need to track which edge-instances are used.
        // Use a flat adjacency with pointers.
        var adjStart = new int[N + 1];
        for (int i = 0; i < M; i++)
        {
            adjStart[edgeU[i]] += edgeCount[i];
            adjStart[edgeV[i]] += edgeCount[i];
        }
        // prefix sum
        {
            int s = 0;
            for (int i = 0; i < N; i++) { int t = adjStart[i]; adjStart[i] = s; s += t; }
            adjStart[N] = s;
        }
        int totalHalf = adjStart[N]; // = 2 * totalE
        var adjTo = new int[totalHalf];
        var adjEid = new int[totalHalf]; // edge-instance id
        var adjPair = new int[totalHalf]; // index of reverse half-edge

        var pos = new int[N];
        Array.Copy(adjStart, pos, N);

        int eid = 0;
        for (int i = 0; i < M; i++)
        {
            int u = edgeU[i], v = edgeV[i];
            for (int c = 0; c < edgeCount[i]; c++)
            {
                int pu = pos[u], pv = pos[v];
                adjTo[pu] = v; adjEid[pu] = eid; adjPair[pu] = pv;
                adjTo[pv] = u; adjEid[pv] = eid; adjPair[pv] = pu;
                pos[u]++;
                pos[v]++;
                eid++;
            }
        }

        // Hierholzer's algorithm
        var used = new bool[totalHalf];
        var ptr = new int[N];
        Array.Copy(adjStart, ptr, N);

        var stack = new Stack<int>();
        var circuit = new List<int>();
        stack.Push(0);

        while (stack.Count > 0)
        {
            int v = stack.Peek();
            bool found = false;
            while (ptr[v] < adjStart[v + 1])
            {
                int hi = ptr[v];
                ptr[v]++;
                if (!used[hi])
                {
                    used[hi] = true;
                    used[adjPair[hi]] = true;
                    stack.Push(adjTo[hi]);
                    found = true;
                    break;
                }
            }
            if (!found)
            {
                stack.Pop();
                circuit.Add(v);
            }
        }

        var sb = new StringBuilder();
        for (int i = 0; i < circuit.Count; i++)
        {
            if (i > 0) sb.Append(' ');
            sb.Append(circuit[i]);
        }

        using var bw = new BufferedStream(Console.OpenStandardOutput(), 1 << 16);
        using var sw = new StreamWriter(bw);
        sw.Write(sb);
        sw.WriteLine();
    }

    static (long[] dist, int[] prev, int[] prevEdge) Dijkstra(int src)
    {
        var dist = new long[N];
        var prev = new int[N];
        var prevEdge = new int[N];
        for (int i = 0; i < N; i++) { dist[i] = long.MaxValue; prev[i] = -1; prevEdge[i] = -1; }
        dist[src] = 0;

        var pq = new PriorityQueue<int, long>();
        pq.Enqueue(src, 0);

        while (pq.Count > 0)
        {
            pq.TryDequeue(out int u, out long d);
            if (d > dist[u]) continue;
            foreach (var (v, w, idx) in adj[u])
            {
                long nd = d + w;
                if (nd < dist[v])
                {
                    dist[v] = nd;
                    prev[v] = u;
                    prevEdge[v] = idx;
                    pq.Enqueue(v, nd);
                }
            }
        }
        return (dist, prev, prevEdge);
    }
}
