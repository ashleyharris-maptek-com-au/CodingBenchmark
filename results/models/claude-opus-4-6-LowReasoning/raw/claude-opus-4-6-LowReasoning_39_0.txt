use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();
    let n: usize = it.next().unwrap().parse().unwrap();
    let m: usize = it.next().unwrap().parse().unwrap();
    let half = n / 2;

    let mut adj = vec![vec![]; n];
    let mut adj_matrix = vec![vec![false; n]; n];
    for _ in 0..m {
        let u: usize = it.next().unwrap().parse().unwrap();
        let v: usize = it.next().unwrap().parse().unwrap();
        if !adj_matrix[u][v] {
            adj[u].push(v);
            adj[v].push(u);
            adj_matrix[u][v] = true;
            adj_matrix[v][u] = true;
        }
    }

    if n == 0 {
        println!("0");
        println!();
        return;
    }

    // For small n, use bitmask enumeration if feasible
    // C(n, n/2) for n<=20 is at most ~184k, for n<=24 ~2.7M, for n<=26 ~10M
    if n <= 26 {
        let mut best_cut = usize::MAX;
        let mut best_set: Vec<usize> = Vec::new();

        // Enumerate all subsets of size half using Gosper's hack
        if half == 0 {
            println!("0");
            println!();
            return;
        }

        let mut set: u64 = (1u64 << half) - 1;
        let limit: u64 = 1u64 << n;

        while set < limit {
            let mut cut = 0usize;
            // Count cut edges
            for u in 0..n {
                if (set >> u) & 1 == 1 {
                    for &v in &adj[u] {
                        if (set >> v) & 1 == 0 {
                            cut += 1;
                        }
                    }
                }
            }
            // Each edge counted once (only from A side)
            if cut < best_cut {
                best_cut = cut;
                best_set.clear();
                for i in 0..n {
                    if (set >> i) & 1 == 1 {
                        best_set.push(i);
                    }
                }
            }

            // Gosper's hack: next subset of same popcount
            let c = set & (!set).wrapping_add(1);
            let r = set + c;
            set = (((r ^ set) >> 2) / c) | r;
        }

        println!("{}", best_cut);
        let parts: Vec<String> = best_set.iter().map(|x| x.to_string()).collect();
        println!("{}", parts.join(" "));
    } else {
        // Branch and bound for larger n
        let mut best_cut = usize::MAX;
        let mut best_partition = vec![0u8; n];
        let mut partition = vec![2u8; n]; // 0=A, 1=B, 2=unassigned
        let mut count_a = 0usize;
        let mut count_b = 0usize;
        let mut current_cut = 0usize;

        // Degree-based ordering: assign high-degree vertices first
        let mut order: Vec<usize> = (0..n).collect();
        order.sort_by(|&a, &b| adj[b].len().cmp(&adj[a].len()));

        // Get initial upper bound via greedy
        {
            let mut ga = 0usize;
            let mut gb = 0usize;
            let mut side = vec![2u8; n];
            for &v in &order {
                let mut ca = 0i64;
                let mut cb = 0i64;
                for &u in &adj[v] {
                    if side[u] == 0 { ca += 1; }
                    else if side[u] == 1 { cb += 1; }
                }
                // Put in side that has fewer cuts, respecting balance
                if ga < half && (gb >= half || ca >= cb) {
                    side[v] = 0;
                    ga += 1;
                } else {
                    side[v] = 1;
                    gb += 1;
                }
            }
            let mut cut = 0;
            for u in 0..n {
                if side[u] == 0 {
                    for &v in &adj[u] {
                        if side[v] == 1 { cut += 1; }
                    }
                }
            }
            best_cut = cut;
            best_partition = side;
        }

        fn branch(
            idx: usize, order: &[usize], adj: &Vec<Vec<usize>>, partition: &mut Vec<u8>,
            count_a: &mut usize, count_b: &mut usize, current_cut: &mut usize,
            best_cut: &mut usize, best_partition: &mut Vec<u8>, half: usize, n: usize,
        ) {
            if idx == n {
                if *current_cut < *best_cut {
                    *best_cut = *current_cut;
                    *best_partition = partition.clone();
                }
                return;
            }
            let v = order[idx];
            let remaining = n - idx;

            for side in 0u8..=1 {
                let count = if side == 0 { count_a } else { count_b };
                if *count >= half { continue; }
                let other_count = if side == 0 { *count_b } else { *count_a };
                // Check if remaining vertices can fill both sides
                if half - *count > remaining - 1 { continue; }
                // The other side also needs to be fillable
                // other_count + (remaining - 1 - (we take one more for our side)) must reach half
                // remaining-1 vertices left after this one; our side needs (half - count - 1) more
                // other side needs (half - other_count) more; sum = remaining - 1

                let mut added_cut = 0usize;
                for &u in &adj[v] {
                    if partition[u] != 2 && partition[u] != side {
                        added_cut += 1;
                    }
                }

                if *current_cut + added_cut >= *best_cut { continue; }

                partition[v] = side;
                *count += 1;
                *current_cut += added_cut;

                branch(idx + 1, order, adj, partition, count_a, count_b, current_cut,
                       best_cut, best_partition, half, n);

                *current_cut -= added_cut;
                *count -= 1;
                partition[v] = 2;
            }
        }

        branch(0, &order, &adj, &mut partition, &mut count_a, &mut count_b,
               &mut current_cut, &mut best_cut, &mut best_partition, half, n);

        println!("{}", best_cut);
        let parts: Vec<String> = (0..n).filter(|&i| best_partition[i] == 0)
            .map(|x| x.to_string()).collect();
        println!("{}", parts.join(" "));
    }
}
