const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const prob = data.probability;
const drones = data.drones;
const maxSteps = data.maxStepsPerDrone;
const rows = prob.length;
const cols = prob[0].length;

const dirs = [[0,0],[0,1],[0,-1],[1,0],[-1,0]];

// Track which cells have been visited by any drone
const visited = Array.from({length: rows}, () => new Float64Array(cols));

// For each drone, greedily plan a path maximizing collected probability
// Use a lookahead greedy approach
const paths = [];

// Sort drones by index to maintain order
const droneOrder = drones.map((d, i) => ({...d, idx: i}));

// Greedy assignment: each drone greedily picks highest-value path considering already-visited cells
for (const drone of droneOrder) {
  const path = [[drone.x, drone.y]];
  let cx = drone.x, cy = drone.y;

  // Mark starting cell
  if (prob[cy][cx] > 0) visited[cy][cx] = 1;

  for (let step = 0; step < maxSteps; step++) {
    let bestVal = -1;
    let bestX = cx, bestY = cy;

    // Lookahead: evaluate each immediate move by doing a short greedy rollout
    for (const [dy, dx] of dirs) {
      const nx = cx + dx, ny = cy + dy;
      if (nx < 0 || nx >= cols || ny < 0 || ny >= rows) continue;
      if (prob[ny][nx] === -1) continue;

      // Score: immediate value + greedy lookahead
      let val = (visited[ny][nx] ? 0 : prob[ny][nx]);

      // Short lookahead from (nx, ny)
      const depth = Math.min(6, maxSteps - step - 1);
      if (depth > 0) {
        let lx = nx, ly = ny;
        const tempVisited = new Set();
        if (!visited[ny][nx]) tempVisited.add(ny * cols + nx);

        for (let d = 0; d < depth; d++) {
          let bv = -1, blx = lx, bly = ly;
          for (const [ddy, ddx] of dirs) {
            const nnx = lx + ddx, nny = ly + ddy;
            if (nnx < 0 || nnx >= cols || nny < 0 || nny >= rows) continue;
            if (prob[nny][nnx] === -1) continue;
            const key = nny * cols + nnx;
            const v = (visited[nny][nnx] || tempVisited.has(key)) ? 0 : prob[nny][nnx];
            if (v > bv) { bv = v; blx = nnx; bly = nny; }
          }
          const key = bly * cols + blx;
          if (!visited[bly][blx] && !tempVisited.has(key)) {
            val += bv > 0 ? bv : 0;
            tempVisited.add(key);
          }
          lx = blx; ly = bly;
        }
      }

      if (val > bestVal) {
        bestVal = val;
        bestX = nx;
        bestY = ny;
      }
    }

    cx = bestX; cy = bestY;
    path.push([cx, cy]);
    if (prob[cy][cx] > 0) visited[cy][cx] = 1;
  }

  paths[drone.idx] = path;
}
