import struct, math

words = []

def w(val):
    words.append(val & 0xFFFFFFFF)

def op(wordcount, opcode):
    w((wordcount << 16) | opcode)

def string_words(s):
    b = s.encode('utf-8') + b'\x00'
    while len(b) % 4 != 0:
        b += b'\x00'
    for i in range(0, len(b), 4):
        w(struct.unpack('<I', b[i:i+4])[0])
    return len(b) // 4

def float_word(f):
    return struct.unpack('<I', struct.pack('<f', f))[0]

# IDs
EXT = 1
VOID = 2
FUNC_TYPE = 3
FLOAT = 4
VEC2 = 5
VEC3 = 6
VEC4 = 7
BOOL = 8
PTR_IN_VEC3 = 9
PTR_IN_VEC2 = 10
PTR_OUT_VEC4 = 11
VAR_NORMAL = 12   # location 1
VAR_UV = 13       # location 2
VAR_OUT = 14      # location 0
C_0 = 15
C_1 = 16
C_09 = 17
C_01 = 18
C_16 = 19
C_PI = 20
C_LIGHT = 21      # 0.57735 = 1/sqrt(3)
MAIN = 22
LABEL = 23
# function body results start at 24
BOUND = 50

# Header
w(0x07230203)  # magic
w(0x00010000)  # version 1.0
w(0)           # generator
w(BOUND)       # bound
w(0)           # reserved

# OpCapability Shader
op(2, 17); w(1)

# OpExtInstImport %EXT "GLSL.std.450"
slen = 4  # "GLSL.std.450\0" = 13 bytes -> 4 words
op(2 + slen, 11); w(EXT); string_words("GLSL.std.450")

# OpMemoryModel Logical GLSL450
op(3, 14); w(0); w(1)

# OpEntryPoint Fragment %MAIN "main" %VAR_NORMAL %VAR_UV %VAR_OUT
# "main\0" = 5 bytes -> 2 words
# wordcount = 1 + 1 + 1 + 2(name) + 3(interfaces) = 8
op(8, 15); w(4); w(MAIN); string_words("main"); w(VAR_NORMAL); w(VAR_UV); w(VAR_OUT)

# OpExecutionMode %MAIN OriginUpperLeft
op(3, 16); w(MAIN); w(7)

# Decorations
# OpDecorate target decoration [operands]
def decorate(target, dec, *operands):
    op(3 + len(operands), 71); w(target); w(dec)
    for o in operands: w(o)

decorate(VAR_NORMAL, 30, 1)  # Location 1
decorate(VAR_UV, 30, 2)      # Location 2
decorate(VAR_OUT, 30, 0)     # Location 0

# Types
# OpTypeVoid
op(2, 19); w(VOID)
# OpTypeFunction %FUNC_TYPE %VOID
op(3, 33); w(FUNC_TYPE); w(VOID)
# OpTypeFloat %FLOAT 32
op(3, 22); w(FLOAT); w(32)
# OpTypeBool
op(2, 20); w(BOOL)
# OpTypeVector %VEC2 %FLOAT 2
op(4, 23); w(VEC2); w(FLOAT); w(2)
# OpTypeVector %VEC3 %FLOAT 3
op(4, 23); w(VEC3); w(FLOAT); w(3)
# OpTypeVector %VEC4 %FLOAT 4
op(4, 23); w(VEC4); w(FLOAT); w(4)
# OpTypePointer %PTR_IN_VEC3 Input %VEC3
op(4, 32); w(PTR_IN_VEC3); w(1); w(VEC3)  # 1=Input
# OpTypePointer %PTR_IN_VEC2 Input %VEC2
op(4, 32); w(PTR_IN_VEC2); w(1); w(VEC2)
# OpTypePointer %PTR_OUT_VEC4 Output %VEC4
op(4, 32); w(PTR_OUT_VEC4); w(3); w(VEC4)  # 3=Output

# Variables
# OpVariable %PTR_IN_VEC3 %VAR_NORMAL Input
op(4, 59); w(PTR_IN_VEC3); w(VAR_NORMAL); w(1)
# OpVariable %PTR_IN_VEC2 %VAR_UV Input
op(4, 59); w(PTR_IN_VEC2); w(VAR_UV); w(1)
# OpVariable %PTR_OUT_VEC4 %VAR_OUT Output
op(4, 59); w(PTR_OUT_VEC4); w(VAR_OUT); w(3)

# Constants
def const_float(id, val):
    op(4, 43); w(FLOAT); w(id); w(float_word(val))

const_float(C_0, 0.0)
const_float(C_1, 1.0)
const_float(C_09, 0.9)
const_float(C_01, 0.1)
const_float(C_16, 16.0)
const_float(C_PI, math.pi)
const_float(C_LIGHT, 1.0 / math.sqrt(3.0))

# Function
# OpFunction %VOID %MAIN None %FUNC_TYPE
op(5, 54); w(VOID); w(MAIN); w(0); w(FUNC_TYPE)  # 0=None function control
# OpLabel
op(2, 248); w(LABEL)

nid = 24
def next_id():
    global nid
    r = nid
    nid += 1
    return r

# %24 = OpLoad %VEC2 %VAR_UV
load_uv = next_id()
op(4, 61); w(VEC2); w(load_uv); w(VAR_UV)

# %25 = OpCompositeExtract %FLOAT %load_uv 1  (uv.y)
uv_y = next_id()
op(5, 81); w(FLOAT); w(uv_y); w(load_uv); w(1)

# %26 = OpFMul %FLOAT %uv_y %C_PI
tmp1 = next_id()
op(5, 133); w(FLOAT); w(tmp1); w(uv_y); w(C_PI)

# %27 = OpFMul %FLOAT %tmp1 %C_16
tmp2 = next_id()
op(5, 133); w(FLOAT); w(tmp2); w(tmp1); w(C_16)

# %28 = OpExtInst %FLOAT %EXT Sin %tmp2  (Sin=13)
sin_val = next_id()
op(6, 12); w(FLOAT); w(sin_val); w(EXT); w(13); w(tmp2)

# %29 = OpFOrdGreaterThan %BOOL %sin_val %C_0  (opcode=186)
cmp = next_id()
op(5, 186); w(BOOL); w(cmp); w(sin_val); w(C_0)

# red = vec3(0.9, 0.1, 0.1)
red = next_id()
op(6, 80); w(VEC3); w(red); w(C_09); w(C_01); w(C_01)

# blue = vec3(0.1, 0.1, 0.9)
blue = next_id()
op(6, 80); w(VEC3); w(blue); w(C_01); w(C_01); w(C_09)

# OpSelect %VEC3 %cmp %red %blue (opcode=169)
color = next_id()
op(6, 169); w(VEC3); w(color); w(cmp); w(red); w(blue)

# Load normal
load_n = next_id()
op(4, 61); w(VEC3); w(load_n); w(VAR_NORMAL)

# Normalize normal: OpExtInst %VEC3 %EXT Normalize %load_n (Normalize=69)
norm_n = next_id()
op(6, 12); w(VEC3); w(norm_n); w(EXT); w(69); w(load_n)

# Light dir = vec3(0.577, 0.577, 0.577)
light = next_id()
op(6, 80); w(VEC3); w(light); w(C_LIGHT); w(C_LIGHT); w(C_LIGHT)

# dot(N, L)
dot_val = next_id()
op(5, 148); w(FLOAT); w(dot_val); w(norm_n); w(light)

# max(dot, 0.0): OpExtInst %FLOAT %EXT FMax %dot_val %C_0 (FMax=40)
max_val = next_id()
op(7, 12); w(FLOAT); w(max_val); w(EXT); w(40); w(dot_val); w(C_0)

# VectorTimesScalar: OpVectorTimesScalar %VEC3 %color %max_val (opcode=142)
shaded = next_id()
op(5, 142); w(VEC3); w(shaded); w(color); w(max_val)

# Extract components
r = next_id()
op(5, 81); w(FLOAT); w(r); w(shaded); w(0)
g = next_id()
op(5, 81); w(FLOAT); w(g); w(shaded); w(1)
b = next_id()
op(5, 81); w(FLOAT); w(b); w(shaded); w(2)

# Construct vec4(r, g, b, 1.0)
final = next_id()
op(7, 80); w(VEC4); w(final); w(r); w(g); w(b); w(C_1)

# OpStore %VAR_OUT %final
op(3, 62); w(VAR_OUT); w(final)

# OpReturn
op(1, 253)

# OpFunctionEnd
op(1, 56)

# Update bound
words[3] = nid

# Convert to hex
result = b''
for word in words:
    result += struct.pack('<I', word)

hex_str = result.hex()
print(hex_str)
