use std::io::{self, BufRead, BufReader, BufWriter, Read, Write};

const MAX_MEMORY_POINTS: usize = 10_000_000;
const BATCH_SIZE: usize = 50_000;

fn main() {
    let stdin = io::stdin();
    let stdout = io::stdout();
    let mut reader = BufReader::with_capacity(1 << 20, stdin.lock());
    let mut writer = BufWriter::with_capacity(1 << 20, stdout.lock());

    let mut header = String::new();
    reader.read_line(&mut header).unwrap();
    let parts: Vec<u64> = header.trim().split_whitespace()
        .map(|s| s.parse().unwrap()).collect();
    let n = parts[0] as usize;
    let k = parts[1] as usize;

    if n <= MAX_MEMORY_POINTS {
        in_memory_kmeans(&mut reader, &mut writer, n, k);
    } else {
        streaming_kmeans(&mut reader, &mut writer, n, k);
    }
}

fn read_points(reader: &mut impl Read, count: usize) -> Vec<[f32; 3]> {
    let mut buf = vec![0u8; count * 12];
    reader.read_exact(&mut buf).unwrap();
    let mut points = Vec::with_capacity(count);
    for i in 0..count {
        let off = i * 12;
        let x = f32::from_le_bytes(buf[off..off+4].try_into().unwrap());
        let y = f32::from_le_bytes(buf[off+4..off+8].try_into().unwrap());
        let z = f32::from_le_bytes(buf[off+8..off+12].try_into().unwrap());
        points.push([x, y, z]);
    }
    points
}

fn nearest(p: &[f32; 3], centroids: &[[f64; 3]]) -> u32 {
    let (px, py, pz) = (p[0] as f64, p[1] as f64, p[2] as f64);
    let mut best = 0u32;
    let mut best_d = f64::MAX;
    for (i, c) in centroids.iter().enumerate() {
        let dx = px - c[0];
        let dy = py - c[1];
        let dz = pz - c[2];
        let d = dx * dx + dy * dy + dz * dz;
        if d < best_d {
            best_d = d;
            best = i as u32;
        }
    }
    best
}

fn init_centroids_pp(points: &[[f32; 3]], k: usize) -> Vec<[f64; 3]> {
    let n = points.len();
    let mut centroids = Vec::with_capacity(k);
    let first = n / 2;
    centroids.push([points[first][0] as f64, points[first][1] as f64, points[first][2] as f64]);
    let mut dists = vec![f64::MAX; n];
    for _ in 1..k {
        let last = centroids.last().unwrap();
        let mut total = 0.0f64;
        for (i, p) in points.iter().enumerate() {
            let dx = p[0] as f64 - last[0];
            let dy = p[1] as f64 - last[1];
            let dz = p[2] as f64 - last[2];
            let d = dx * dx + dy * dy + dz * dz;
            if d < dists[i] { dists[i] = d; }
            total += dists[i];
        }
        let target = total * 0.5;
        let mut cum = 0.0;
        let mut chosen = 0;
        for (i, &d) in dists.iter().enumerate() {
            cum += d;
            if cum >= target { chosen = i; break; }
        }
        centroids.push([points[chosen][0] as f64, points[chosen][1] as f64, points[chosen][2] as f64]);
    }
    centroids
}

fn fix_empty(centroids: &mut [[f64; 3]], counts: &[u64], k: usize) {
    for j in 0..k {
        if counts[j] == 0 {
            let largest = counts.iter().enumerate().max_by_key(|&(_, &c)| c).unwrap().0;
            centroids[j] = centroids[largest];
            centroids[j][0] += 0.001;
        }
    }
}

fn in_memory_kmeans(reader: &mut impl Read, writer: &mut impl Write, n: usize, k: usize) {
    let points = read_points(reader, n);
    let max_iters = if n <= 1_000_000 { 20 } else { 10 };

    let sample: Vec<[f32; 3]> = if n > 100_000 {
        let step = n / 100_000;
        points.iter().step_by(step).copied().collect()
    } else {
        points.clone()
    };
    let mut centroids = init_centroids_pp(&sample, k);
    drop(sample);

    let mut assignments = vec![0u32; n];
    for _ in 0..max_iters {
        let mut changed = false;
        for (i, p) in points.iter().enumerate() {
            let c = nearest(p, &centroids);
            if c != assignments[i] { changed = true; assignments[i] = c; }
        }
        if !changed { break; }

        let mut sums = vec![[0.0f64; 3]; k];
        let mut counts = vec![0u64; k];
        for (i, p) in points.iter().enumerate() {
            let c = assignments[i] as usize;
            sums[c][0] += p[0] as f64;
            sums[c][1] += p[1] as f64;
            sums[c][2] += p[2] as f64;
            counts[c] += 1;
        }
        for j in 0..k {
            if counts[j] > 0 {
                centroids[j][0] = sums[j][0] / counts[j] as f64;
                centroids[j][1] = sums[j][1] / counts[j] as f64;
                centroids[j][2] = sums[j][2] / counts[j] as f64;
            }
        }
        fix_empty(&mut centroids, &counts, k);
    }

    let mut buf = vec![0u8; n * 4];
    for (i, &a) in assignments.iter().enumerate() {
        buf[i * 4..i * 4 + 4].copy_from_slice(&a.to_le_bytes());
    }
    writer.write_all(&buf).unwrap();
    writer.flush().unwrap();
}

fn streaming_kmeans(reader: &mut impl Read, writer: &mut impl Write, n: usize, k: usize) {
    let tmp_path = std::env::temp_dir().join("kmeans_tmp.bin");

    // Pass 1: Read all stdin to tmpfile, init centroids from first batch
    let init_size = 100_000.min(n);
    let init_points = read_points(reader, init_size);

    let mut tmpfile = std::fs::File::create(&tmp_path).unwrap();
    // Write init points
    for p in &init_points {
        tmpfile.write_all(&p[0].to_le_bytes()).unwrap();
        tmpfile.write_all(&p[1].to_le_bytes()).unwrap();
        tmpfile.write_all(&p[2].to_le_bytes()).unwrap();
    }
    // Copy rest of stdin
    let remaining = n - init_size;
    let mut done = 0;
    while done < remaining {
        let batch = BATCH_SIZE.min(remaining - done);
        let pts = read_points(reader, batch);
        for p in &pts {
            tmpfile.write_all(&p[0].to_le_bytes()).unwrap();
            tmpfile.write_all(&p[1].to_le_bytes()).unwrap();
            tmpfile.write_all(&p[2].to_le_bytes()).unwrap();
        }
        done += batch;
    }
    tmpfile.flush().unwrap();
    drop(tmpfile);

    let mut centroids = init_centroids_pp(&init_points, k);
    drop(init_points);

    let num_iters = if n > 100_000_000 { 3 } else { 5 };

    for _ in 0..num_iters {
        let mut sums = vec![[0.0f64; 3]; k];
        let mut counts = vec![0u64; k];

        let f = std::fs::File::open(&tmp_path).unwrap();
        let mut fr = BufReader::with_capacity(1 << 20, f);
        let mut done = 0;
        while done < n {
            let batch = BATCH_SIZE.min(n - done);
            let pts = read_points(&mut fr, batch);
            for p in &pts {
                let c = nearest(p, &centroids) as usize;
                sums[c][0] += p[0] as f64;
                sums[c][1] += p[1] as f64;
                sums[c][2] += p[2] as f64;
                counts[c] += 1;
            }
            done += batch;
        }

        for j in 0..k {
            if counts[j] > 0 {
                centroids[j][0] = sums[j][0] / counts[j] as f64;
                centroids[j][1] = sums[j][1] / counts[j] as f64;
                centroids[j][2] = sums[j][2] / counts[j] as f64;
            }
        }
        fix_empty(&mut centroids, &counts, k);
    }

    // Final assignment pass
    let f = std::fs::File::open(&tmp_path).unwrap();
    let mut fr = BufReader::with_capacity(1 << 20, f);
    let mut done = 0;
    while done < n {
        let batch = BATCH_SIZE.min(n - done);
        let pts = read_points(&mut fr, batch);
        let mut out = vec![0u8; batch * 4];
        for (i, p) in pts.iter().enumerate() {
            let c = nearest(p, &centroids);
            out[i * 4..i * 4 + 4].copy_from_slice(&c.to_le_bytes());
        }
        writer.write_all(&out).unwrap();
        done += batch;
    }
    writer.flush().unwrap();
    let _ = std::fs::remove_file(&tmp_path);
}
