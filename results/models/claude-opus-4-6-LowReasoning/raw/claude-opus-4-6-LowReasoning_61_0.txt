declare const require: any; const fs = require('fs'); const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const depot: [number, number] = data.depot;
const truckCount: number = data.truckCount;
const packages: {id: string, x: number, y: number}[] = data.packages;

function dist(ax: number, ay: number, bx: number, by: number): number {
  const dx = ax - bx, dy = ay - by;
  return Math.sqrt(dx * dx + dy * dy);
}

if (packages.length === 0) {
  console.log(JSON.stringify({routes: []}));
  process.exit(0);
}

// K-means style clustering then TSP per cluster via nearest-neighbor + 2-opt

const n = packages.length;
const k = Math.min(truckCount, n);

// Assign packages to clusters using angular sweep from depot
const angles: {idx: number, angle: number}[] = packages.map((p, i) => ({
  idx: i,
  angle: Math.atan2(p.y - depot[1], p.x - depot[0])
}));
angles.sort((a, b) => a.angle - b.angle);

const clusterSize = Math.ceil(n / k);
const clusters: number[][] = [];
for (let i = 0; i < k; i++) {
  clusters.push([]);
}
for (let i = 0; i < angles.length; i++) {
  clusters[Math.min(Math.floor(i / clusterSize), k - 1)].push(angles[i].idx);
}

// Nearest neighbor TSP for a cluster
function nearestNeighborRoute(indices: number[]): number[] {
  if (indices.length === 0) return [];
  if (indices.length === 1) return [indices[0]];

  const used = new Uint8Array(indices.length);
  const route: number[] = [];

  // Start from nearest to depot
  let bestFirst = 0, bestDist = Infinity;
  for (let i = 0; i < indices.length; i++) {
    const p = packages[indices[i]];
    const d = dist(depot[0], depot[1], p.x, p.y);
    if (d < bestDist) { bestDist = d; bestFirst = i; }
  }

  used[bestFirst] = 1;
  route.push(bestFirst);

  for (let step = 1; step < indices.length; step++) {
    const lastLocal = route[route.length - 1];
    const last = packages[indices[lastLocal]];
    let bestNext = -1, bestD = Infinity;
    for (let i = 0; i < indices.length; i++) {
      if (used[i]) continue;
      const p = packages[indices[i]];
      const d = dist(last.x, last.y, p.x, p.y);
      if (d < bestD) { bestD = d; bestNext = i; }
    }
    used[bestNext] = 1;
    route.push(bestNext);
  }

  return route.map(i => indices[i]);
}

// 2-opt improvement
function twoOpt(route: number[]): number[] {
  if (route.length < 3) return route;
  const m = route.length;

  // Precompute x,y arrays for speed
  const xs = new Float64Array(m);
  const ys = new Float64Array(m);
  for (let i = 0; i < m; i++) {
    xs[i] = packages[route[i]].x;
    ys[i] = packages[route[i]].y;
  }
  const dx = depot[0], dy = depot[1];

  let improved = true;
  let iterations = 0;
  const maxIter = Math.min(50, Math.max(5, Math.floor(5000 / m)));

  while (improved && iterations < maxIter) {
    improved = false;
    iterations++;
    for (let i = 0; i < m - 1; i++) {
      const ax = i === 0 ? dx : xs[i - 1];
      const ay = i === 0 ? dy : ys[i - 1];
      for (let j = i + 1; j < m; j++) {
        const bx = j === m - 1 ? dx : xs[j + 1];
        const by = j === m - 1 ? dy : ys[j + 1];

        const oldDist = dist(ax, ay, xs[i], ys[i]) + dist(xs[j], ys[j], bx, by);
        const newDist = dist(ax, ay, xs[j], ys[j]) + dist(xs[i], ys[i], bx, by);

        if (newDist < oldDist - 1e-10) {
          // Reverse segment [i..j]
          let lo = i, hi = j;
          while (lo < hi) {
            const tmp = route[lo]; route[lo] = route[hi]; route[hi] = tmp;
            const tx = xs[lo]; xs[lo] = xs[hi]; xs[hi] = tx;
            const ty = ys[lo]; ys[lo] = ys[hi]; ys[hi] = ty;
            lo++; hi--;
          }
          improved = true;
        }
      }
    }
  }
  return route;
}

// Or-opt: move single nodes between positions
function orOpt(route: number[]): number[] {
  if (route.length < 3) return route;
  const m = route.length;
  let improved = true;
  while (improved) {
    improved = false;
    for (let i = 0; i < m; i++) {
      const prev = i === 0 ? depot : packages[route[i - 1]];
      const cur = packages[route[i]];
      const next = i === m - 1 ? depot : packages[route[i + 1]];
      const removeCost = dist(prev.x, prev.y, cur.x, cur.y) + dist(cur.x, cur.y, next.x, next.y) - dist(prev.x, prev.y, next.x, next.y);

      for (let j = 0; j < m; j++) {
        if (j === i || j === i - 1) continue;
        const a = j === 0 && i !== 0 ? depot : (j <= i ? packages[route[j - 1]] : packages[route[j]]);
        const b = j <= i ? packages[route[j]] : (j === m - 1 && i !== m - 1 ? depot : packages[route[j + 1]]);
        // Actually let me skip or-opt for simplicity, 2-opt is good enough
        break;
      }
      if (improved) break;
    }
    break; // skip or-opt
  }
  return route;
}

const routes: string[][] = [];
for (const cluster of clusters) {
  if (cluster.length === 0) continue;
  let route = nearestNeighborRoute(cluster);
  route = twoOpt(route);
  routes.push(route.map(i => packages[i].id));
}

// Inter-route improvement: try moving packages between routes
function routeDist(route: string[]): number {
  if (route.length === 0) return 0;
  let d = 0;
  const pMap = new Map<string, {x:number,y:number}>();
  for (const p of packages) pMap.set(p.id, p);
  let cx = depot[0], cy = depot[1];
  for (const id of route) {
    const p = pMap.get(id)!;
    d += dist(cx, cy, p.x, p.y);
    cx = p.x; cy = p.y;
  }
  d += dist(cx, cy, depot[0], depot[1]);
  return d;
}
