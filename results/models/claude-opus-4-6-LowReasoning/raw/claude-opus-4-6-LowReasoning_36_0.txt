// For bitset approach (n <= 64*WORDS)
static const int MAXWORDS = 32; // supports up to 2048 vertices in bitset mode
static int nwords;

struct Bitset {
    uint64_t w[MAXWORDS];
    void clear() { memset(w, 0, sizeof(uint64_t)*nwords); }
    void set(int i) { w[i>>6] |= 1ULL << (i&63); }
    void reset(int i) { w[i>>6] &= ~(1ULL << (i&63)); }
    bool test(int i) const { return (w[i>>6] >> (i&63)) & 1; }
    int count() const {
        int c = 0;
        for (int i = 0; i < nwords; i++) c += __popcnt64(w[i]);
        return c;
    }
    Bitset operator&(const Bitset& o) const {
        Bitset r; for (int i=0;i<nwords;i++) r.w[i]=w[i]&o.w[i]; return r;
    }
    Bitset operator|(const Bitset& o) const {
        Bitset r; for (int i=0;i<nwords;i++) r.w[i]=w[i]|o.w[i]; return r;
    }
    Bitset operator~() const {
        Bitset r; for (int i=0;i<nwords;i++) r.w[i]=~w[i]; return r;
    }
    Bitset& operator&=(const Bitset& o) { for(int i=0;i<nwords;i++) w[i]&=o.w[i]; return *this; }
    Bitset& operator|=(const Bitset& o) { for(int i=0;i<nwords;i++) w[i]|=o.w[i]; return *this; }
    bool empty() const { for(int i=0;i<nwords;i++) if(w[i]) return false; return true; }
    int first() const {
        for (int i=0;i<nwords;i++) if(w[i]) {
            unsigned long idx;
            _BitScanForward64(&idx, w[i]);
            return (i<<6)|idx;
        }
        return -1;
    }
};

static vector<Bitset> adjBS;

// Greedy coloring for upper bound
static int colorBound(Bitset& P) {
    int colors = 0;
    Bitset uncolored = P;
    while (!uncolored.empty()) {
        colors++;
        Bitset avail = uncolored;
        while (!avail.empty()) {
            int v = avail.first();
            uncolored.reset(v);
            avail.reset(v);
            avail &= ~adjBS[v]; // remove neighbors from available
        }
    }
    return colors;
}

static void branchAndBound(Bitset& P, Bitset& X) {
    if (P.empty()) {
        if ((int)currentSet.size() > bestSize) {
            bestSize = (int)currentSet.size();
            bestSet = currentSet;
        }
        return;
    }

    // Upper bound: current + coloring of P
    int ub = (int)currentSet.size() + colorBound(P);
    if (ub <= bestSize) return;

    // Pick vertex with max connections in P (to branch on)
    // Try vertices in P
    // Build ordered list of vertices in P by degree in P (ascending -> branch on low degree first? or descending?)
    // Actually for MIS, branching on max-degree vertex in P is good for pruning

    vector<int> vertices;
    for (int i = 0; i < nwords; i++) {
        uint64_t bits = P.w[i];
        while (bits) {
            unsigned long idx;
            _BitScanForward64(&idx, bits);
            vertices.push_back((i<<6)|idx);
            bits &= bits-1;
        }
    }

    // Sort by degree in P descending (branch on highest degree first)
    sort(vertices.begin(), vertices.end(), [&](int a, int b) {
        return (adjBS[a] & P).count() > (adjBS[b] & P).count();
    });

    for (int v : vertices) {
        if (!P.test(v)) continue;

        // Include v
        currentSet.push_back(v);
        Bitset newP = P & ~adjBS[v];
        newP.reset(v);
        Bitset newX = X & ~adjBS[v];

        branchAndBound(newP, newX);

        currentSet.pop_back();
        P.reset(v);
        X.set(v);

        // Re-check bound
        ub = (int)currentSet.size() + colorBound(P);
        if (ub <= bestSize) return;
    }
}

        vector<int> order(n);
        iota(order.begin(), order.end(), 0);
        sort(order.begin(), order.end(), [](int a, int b){ return a < b; });

        removed.assign(n, false);
        inSet.assign(n, false);
        deg.resize(n);
        for (int i = 0; i < n; i++) deg[i] = (int)adj[i].size();

        // Iteratively remove degree-0 and degree-1 vertices
        queue<int> q;
        for (int i = 0; i < n; i++) if (deg[i] <= 1) q.push(i);

        vector<int> result;
        while (!q.empty()) {
            int v = q.front(); q.pop();
            if (removed[v]) continue;
            if (deg[v] == 0) {
                result.push_back(v);
                removed[v] = true;
            } else if (deg[v] == 1) {
                // Find neighbor
                result.push_back(v);
                removed[v] = true;
                for (int u : adj[v]) {
                    if (!removed[u]) {
                        removed[u] = true;
                        for (int w : adj[u]) {
                            if (!removed[w]) {
                                deg[w]--;
                                if (deg[w] <= 1) q.push(w);
                            }
                        }
                        break;
                    }
                }
            }
        }

        // Collect remaining vertices
        vector<int> remaining;
        map<int,int> remap;
        for (int i = 0; i < n; i++) {
            if (!removed[i]) {
                remap[i] = (int)remaining.size();
                remaining.push_back(i);
            }
        }

        int rn = (int)remaining.size();
        if (rn == 0) {
            cout << result.size() << "\n";
            for (int i = 0; i < (int)result.size(); i++) {
                cout << result[i];
                if (i+1 < (int)result.size()) cout << " ";
            }
            cout << "\n";
            return 0;
        }

        if (rn <= MAXWORDS * 64) {
            // Solve reduced graph with bitset B&B
            nwords = (rn + 63) / 64;
            adjBS.resize(rn);
            for (int i = 0; i < rn; i++) adjBS[i].clear();
            for (int i = 0; i < rn; i++) {
                int orig = remaining[i];
                for (int j : adj[orig]) {
                    if (!removed[j]) {
                        adjBS[i].set(remap[j]);
                    }
                }
            }

            Bitset P, X;
            P.clear(); X.clear();
            for (int i = 0; i < rn; i++) P.set(i);

            bestSize = 0;
            branchAndBound(P, X);

            for (int i = 0; i < bestSize; i++) {
                result.push_back(remaining[bestSet[i]]);
            }
        }

        cout << result.size() << "\n";
        for (int i = 0; i < (int)result.size(); i++) {
            cout << result[i];
            if (i+1 < (int)result.size()) cout << " ";
        }
        cout << "\n";
    }

    return 0;
