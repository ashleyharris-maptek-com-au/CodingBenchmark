; SPIR-V Geometry Shader - Wireframe quad edges from triangles with adjacency
               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %inColor %outColor
               OpExecutionMode %main InputTrianglesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 12
               OpExecutionMode %main Invocations 1

               ; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %inColor Location 2
               OpDecorate %outColor Location 0

               ; Types
      %void  = OpTypeVoid
   %voidfn   = OpTypeFunction %void
     %float  = OpTypeFloat 32
   %v3float  = OpTypeVector %float 3
   %v4float  = OpTypeVector %float 4
      %int   = OpTypeInt 32 1
     %uint   = OpTypeInt 32 0

               ; Constants
   %int_0    = OpConstant %int 0
   %int_1    = OpConstant %int 1
   %int_2    = OpConstant %int 2
   %int_4    = OpConstant %int 4
   %uint_6   = OpConstant %uint 6
  %float_0   = OpConstant %float 0.0
 %float_half = OpConstant %float 0.5
%float_thick = OpConstant %float 0.008
%float_nthick= OpConstant %float -0.008

               ; Input gl_PerVertex block
%gl_PerVertex_in = OpTypeStruct %v4float
 %arr_pv_in  = OpTypeArray %gl_PerVertex_in %uint_6
%ptr_in_arr  = OpTypePointer Input %arr_pv_in
 %ptr_in_v4  = OpTypePointer Input %v4float
    %gl_in   = OpVariable %ptr_in_arr Input

               ; Input color array
%arr_v3_6    = OpTypeArray %v3float %uint_6
%ptr_in_arr3 = OpTypePointer Input %arr_v3_6
 %ptr_in_v3  = OpTypePointer Input %v3float
   %inColor  = OpVariable %ptr_in_arr3 Input

               ; Output gl_PerVertex block
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv  = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4  = OpTypePointer Output %v4float
   %gl_out   = OpVariable %ptr_out_pv Output

               ; Output color
%ptr_out_v3  = OpTypePointer Output %v3float
  %outColor  = OpVariable %ptr_out_v3 Output

               ; Main function
     %main   = OpFunction %void None %voidfn
    %entry   = OpLabel

               ; Load positions for main triangle vertices (indices 0, 2, 4)
   %p0_ptr   = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
      %p0    = OpLoad %v4float %p0_ptr
   %p2_ptr   = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
      %p2    = OpLoad %v4float %p2_ptr
   %p4_ptr   = OpAccessChain %ptr_in_v4 %gl_in %int_4 %int_0
      %p4    = OpLoad %v4float %p4_ptr

               ; Load colors for main triangle vertices
  %c0_ptr    = OpAccessChain %ptr_in_v3 %inColor %int_0
     %c0     = OpLoad %v3float %c0_ptr
  %c2_ptr    = OpAccessChain %ptr_in_v3 %inColor %int_2
     %c2     = OpLoad %v3float %c2_ptr
  %c4_ptr    = OpAccessChain %ptr_in_v3 %inColor %int_4
     %c4     = OpLoad %v3float %c4_ptr

               ; ===== Edge 0→2 =====
               ; Direction vector in clip space (xy)
  %d02_4     = OpFSub %v4float %p2 %p0
  %d02_x     = OpCompositeExtract %float %d02_4 0
  %d02_y     = OpCompositeExtract %float %d02_4 1
               ; Perpendicular: (-dy, dx) normalized to thickness
  %nd02_x    = OpFNegate %float %d02_y
  %nd02_y    = OpCopyObject %float %d02_x
  %len02_sq  = OpFMul %float %nd02_x %nd02_x
  %tmp02     = OpFMul %float %nd02_y %nd02_y
  %len02sq   = OpFAdd %float %len02_sq %tmp02
  %len02     = OpExtInst %float %1 Sqrt %len02sq
  %inv_len02 = OpFDiv %float %float_thick %len02
  %ninv02    = OpFNegate %float %inv_len02
  %off02_x   = OpFMul %float %nd02_x %inv_len02
  %off02_y   = OpFMul %float %nd02_y %inv_len02
  %noff02_x  = OpFMul %float %nd02_x %ninv02
  %noff02_y  = OpFMul %float %nd02_y %ninv02
               ; Average color
  %ac02      = OpFAdd %v3float %c0 %c2
  %col02     = OpVectorTimesScalar %v3float %ac02 %float_half

               ; Vertex 1: p0 + offset
  %p0_x      = OpCompositeExtract %float %p0 0
  %p0_y      = OpCompositeExtract %float %p0 1
  %p0_z      = OpCompositeExtract %float %p0 2
  %p0_w      = OpCompositeExtract %float %p0 3
  %v1_x      = OpFAdd %float %p0_x %off02_x
  %v1_y      = OpFAdd %float %p0_y %off02_y
  %v1_pos    = OpCompositeConstruct %v4float %v1_x %v1_y %p0_z %p0_w
  %out_ptr1  = OpAccessChain %ptr_out_v4 %gl_out %int_0
               OpStore %out_ptr1 %v1_pos
               OpStore %outColor %col02
               OpEmitVertex

               ; Vertex 2: p0 - offset
  %v2_x      = OpFAdd %float %p0_x %noff02_x
  %v2_y      = OpFAdd %float %p0_y %noff02_y
  %v2_pos    = OpCompositeConstruct %v4float %v2_x %v2_y %p0_z %p0_w
               OpStore %out_ptr1 %v2_pos
               OpStore %outColor %col02
               OpEmitVertex

               ; Vertex 3: p2 + offset
  %p2_x      = OpCompositeExtract %float %p2 0
  %p2_y      = OpCompositeExtract %float %p2 1
  %p2_z      = OpCompositeExtract %float %p2 2
  %p2_w      = OpCompositeExtract %float %p2 3
  %v3_x      = OpFAdd %float %p2_x %off02_x
  %v3_y      = OpFAdd %float %p2_y %off02_y
  %v3_pos    = OpCompositeConstruct %v4float %v3_x %v3_y %p2_z %p2_w
               OpStore %out_ptr1 %v3_pos
               OpStore %outColor %col02
               OpEmitVertex

               ; Vertex 4: p2 - offset
  %v4_x      = OpFAdd %float %p2_x %noff02_x
  %v4_y      = OpFAdd %float %p2_y %noff02_y
  %v4_pos    = OpCompositeConstruct %v4float %v4_x %v4_y %p2_z %p2_w
               OpStore %out_ptr1 %v4_pos
               OpStore %outColor %col02
               OpEmitVertex
               OpEndPrimitive

               ; ===== Edge 2→4 =====
  %d24_4     = OpFSub %v4float %p4 %p2
  %d24_x     = OpCompositeExtract %float %d24_4 0
  %d24_y     = OpCompositeExtract %float %d24_4 1
  %nd24_x    = OpFNegate %float %d24_y
  %nd24_y    = OpCopyObject %float %d24_x
  %len24_sq  = OpFMul %float %nd24_x %nd24_x
  %tmp24     = OpFMul %float %nd24_y %nd24_y
  %len24sq   = OpFAdd %float %len24_sq %tmp24
  %len24     = OpExtInst %float %1 Sqrt %len24sq
  %inv_len24 = OpFDiv %float %float_thick %len24
  %ninv24    = OpFNegate %float %inv_len24
  %off24_x   = OpFMul %float %nd24_x %inv_len24
  %off24_y   = OpFMul %float %nd24_y %inv_len24
  %noff24_x  = OpFMul %float %nd24_x %ninv24
  %noff24_y  = OpFMul %float %nd24_y %ninv24
  %ac24      = OpFAdd %v3float %c2 %c4
  %col24     = OpVectorTimesScalar %v3float %ac24 %float_half

  %e2v1_x    = OpFAdd %float %p2_x %off24_x
  %e2v1_y    = OpFAdd %float %p2_y %off24_y
  %e2v1_pos  = OpCompositeConstruct %v4float %e2v1_x %e2v1_y %p2_z %p2_w
               OpStore %out_ptr1 %e2v1_pos
               OpStore %outColor %col24
               OpEmitVertex

  %e2v2_x    = OpFAdd %float %p2_x %noff24_x
  %e2v2_y    = OpFAdd %float %p2_y %noff24_y
  %e2v2_pos  = OpCompositeConstruct %v4float %e2v2_x %e2v2_y %p2_z %p2_w
               OpStore %out_ptr1 %e2v2_pos
               OpStore %outColor %col24
               OpEmitVertex

  %p4_x      = OpCompositeExtract %float %p4 0
  %p4_y      = OpCompositeExtract %float %p4 1
  %p4_z      = OpCompositeExtract %float %p4 2
  %p4_w      = OpCompositeExtract %float %p4 3
  %e2v3_x    = OpFAdd %float %p4_x %off24_x
  %e2v3_y    = OpFAdd %float %p4_y %off24_y
  %e2v3_pos  = OpCompositeConstruct %v4float %e2v3_x %e2v3_y %p4_z %p4_w
               OpStore %out_ptr1 %e2v3_pos
               OpStore %outColor %col24
               OpEmitVertex

  %e2v4_x    = OpFAdd %float %p4_x %noff24_x
  %e2v4_y    = OpFAdd %float %p4_y %noff24_y
  %e2v4_pos  = OpCompositeConstruct %v4float %e2v4_x %e2v4_y %p4_z %p4_w
               OpStore %out_ptr1 %e2v4_pos
               OpStore %outColor %col24
               OpEmitVertex
               OpEndPrimitive

               ; ===== Edge 4→0 =====
  %d40_4     = OpFSub %v4float %p0 %p4
  %d40_x     = OpCompositeExtract %float %d40_4 0
  %d40_y     = OpCompositeExtract %float %d40_4 1
  %nd40_x    = OpFNegate %float %d40_y
  %nd40_y    = OpCopyObject %float %d40_x
  %len40_sq  = OpFMul %float %nd40_x %nd40_x
  %tmp40     = OpFMul %float %nd40_y %nd40_y
  %len40sq   = OpFAdd %float %len40_sq %tmp40
  %len40     = OpExtInst %float %1 Sqrt %len40sq
  %inv_len40 = OpFDiv %float %float_thick %len40
  %ninv40    = OpFNegate %float %inv_len40
  %off40_x   = OpFMul %float %nd40_x %inv_len40
  %off40_y   = OpFMul %float %nd40_y %inv_len40
  %noff40_x  = OpFMul %float %nd40_x %ninv40
  %noff40_y  = OpFMul %float %nd40_y %ninv40
  %ac40      = OpFAdd %v3float %c4 %c0
  %col40     = OpVectorTimesScalar %v3float %ac40 %float_half

  %e3v1_x    = OpFAdd %float %p4_x %off40_x
  %e3v1_y    = OpFAdd %float %p4_y %off40_y
  %e3v1_pos  = OpCompositeConstruct %v4float %e3v1_x %e3v1_y %p4_z %p4_w
               OpStore %out_ptr1 %e3v1_pos
               OpStore %outColor %col40
               OpEmitVertex

  %e3v2_x    = OpFAdd %float %p4_x %noff40_x
  %e3v2_y    = OpFAdd %float %p4_y %noff40_y
  %e3v2_pos  = OpCompositeConstruct %v4float %e3v2_x %e3v2_y %p4_z %p4_w
               OpStore %out_ptr1 %e3v2_pos
               OpStore %outColor %col40
               OpEmitVertex

  %e3v3_x    = OpFAdd %float %p0_x %off40_x
  %e3v3_y    = OpFAdd %float %p0_y %off40_y
  %e3v3_pos  = OpCompositeConstruct %v4float %e3v3_x %e3v3_y %p0_z %p0_w
               OpStore %out_ptr1 %e3v3_pos
               OpStore %outColor %col40
               OpEmitVertex

  %e3v4_x    = OpFAdd %float %p0_x %noff40_x
  %e3v4_y    = OpFAdd %float %p0_y %noff40_y
  %e3v4_pos  = OpCompositeConstruct %v4float %e3v4_x %e3v4_y %p0_z %p0_w
               OpStore %out_ptr1 %e3v4_pos
               OpStore %outColor %col40
               OpEmitVertex
               OpEndPrimitive

               OpReturn
               OpFunctionEnd
