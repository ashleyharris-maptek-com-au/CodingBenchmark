use std::io::{self, Read};
use std::time::Instant;

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();
    let n: usize = it.next().unwrap().parse().unwrap();
    let m: usize = it.next().unwrap().parse().unwrap();

    let mut adj = vec![vec![]; n];
    for _ in 0..m {
        let u: usize = it.next().unwrap().parse().unwrap();
        let v: usize = it.next().unwrap().parse().unwrap();
        adj[u].push(v);
        adj[v].push(u);
    }

    let start = Instant::now();

    // Greedy: pick vertex that dominates the most uncovered vertices
    let mut in_set = vec![false; n];
    let mut dominated = vec![false; n];
    let mut remaining = n;

    while remaining > 0 {
        let mut best = 0usize;
        let mut best_score = 0usize;
        for v in 0..n {
            if in_set[v] { continue; }
            let mut score = if !dominated[v] { 1 } else { 0 };
            for &u in &adj[v] {
                if !dominated[u] { score += 1; }
            }
            if score > best_score {
                best_score = score;
                best = v;
            }
        }
        in_set[best] = true;
        if !dominated[best] {
            dominated[best] = true;
            remaining -= 1;
        }
        for &u in &adj[best] {
            if !dominated[u] {
                dominated[u] = true;
                remaining -= 1;
            }
        }
    }

    // Local search: try removing each vertex from the set
    // and check if the set is still dominating
    let time_limit = 240_000; // 4 minutes in ms

    // Iterative improvement: try swapping out vertices
    let mut improved = true;
    while improved && start.elapsed().as_millis() < time_limit {
        improved = false;

        // Try removing redundant vertices
        let set_vertices: Vec<usize> = (0..n).filter(|&v| in_set[v]).collect();
        for &v in &set_vertices {
            if start.elapsed().as_millis() >= time_limit { break; }
            in_set[v] = false;
            // Check if still dominating
            let mut valid = true;
            // v itself must be dominated by a neighbor in set
            if !adj[v].iter().any(|&u| in_set[u]) {
                valid = false;
            }
            if valid {
                // Each neighbor of v that was only dominated by v must have another dominator
                for &u in &adj[v] {
                    if in_set[u] { continue; }
                    // u needs to be dominated by someone in set
                    if !adj[u].iter().any(|&w| in_set[w]) {
                        valid = false;
                        break;
                    }
                }
            }
            if valid {
                improved = true;
            } else {
                in_set[v] = true;
            }
        }
    }

    // Try 1-2 swaps: remove one, add another that's not in set
    while start.elapsed().as_millis() < time_limit {
        let mut any_swap = false;
        let set_vertices: Vec<usize> = (0..n).filter(|&v| in_set[v]).collect();
        'outer: for &rem in &set_vertices {
            if start.elapsed().as_millis() >= time_limit { break; }
            in_set[rem] = false;
            // Find uncovered vertices
            let mut uncovered: Vec<usize> = Vec::new();
            // Check rem itself
            if !adj[rem].iter().any(|&u| in_set[u]) {
                uncovered.push(rem);
            }
            for &u in &adj[rem] {
                if in_set[u] { continue; }
                if !adj[u].iter().any(|&w| in_set[w]) {
                    uncovered.push(u);
                }
            }
            if uncovered.is_empty() {
                any_swap = true;
                continue;
            }
            // Try to find a single vertex (not in set, not rem) that covers all uncovered
            for &candidate in &adj[uncovered[0]] {
                if in_set[candidate] || candidate == rem { continue; }
                // Check if candidate covers all uncovered
                let mut covers_all = true;
                for &uc in &uncovered {
                    if uc != candidate && !adj[uc].contains(&candidate) {
                        covers_all = false;
                        break;
                    }
                }
                if covers_all {
                    in_set[candidate] = true;
                    // Verify it's actually net beneficial (same size but might enable future removals)
                    // Actually this is a swap (remove rem, add candidate) - same size
                    // Only useful if candidate enables further removals, skip for now
                    // But let's check if candidate was already providing more coverage
                    // Actually just do the swap and continue trying removals
                    any_swap = true;
                    continue 'outer;
                }
            }
            // Also check if uncovered[0] itself covers all
            if !in_set[uncovered[0]] {
                let candidate = uncovered[0];
                let mut covers_all = true;
                for &uc in &uncovered {
                    if uc != candidate && !adj[uc].contains(&candidate) {
                        covers_all = false;
                        break;
                    }
                }
                if covers_all {
                    in_set[candidate] = true;
                    any_swap = true;
                    continue 'outer;
                }
            }
            in_set[rem] = true;
        }
        if !any_swap { break; }
        // After swaps, try removals again
        let set_vertices2: Vec<usize> = (0..n).filter(|&v| in_set[v]).collect();
        let mut removed_any = false;
        for &v in &set_vertices2 {
            in_set[v] = false;
            let mut valid = true;
            if !adj[v].iter().any(|&u| in_set[u]) {
                valid = false;
            }
            if valid {
                for &u in &adj[v] {
                    if in_set[u] { continue; }
                    if !adj[u].iter().any(|&w| in_set[w]) {
                        valid = false;
                        break;
                    }
                }
            }
            if valid {
                removed_any = true;
            } else {
                in_set[v] = true;
            }
        }
        if !removed_any { break; }
    }

    let result: Vec<usize> = (0..n).filter(|&v| in_set[v]).collect();
    println!("{}", result.len());
    let s: Vec<String> = result.iter().map(|v| v.to_string()).collect();
    println!("{}", s.join(" "));
}
