using namespace std;

static float *gx, *gy;
static int N;

static inline float dist(int a, int b) {
    float dx = gx[a] - gx[b], dy = gy[a] - gy[b];
    return sqrtf(dx*dx + dy*dy);
}

static double tourLength(const vector<int>& tour) {
    double len = 0;
    for (int i = 0; i < N; i++) {
        int j = (i + 1) % N;
        len += dist(tour[i], tour[j]);
    }
    return len;
}

// Bitmask DP for small N (up to ~20)
static vector<int> solveTinyDP() {
    int n = N;
    int full = (1 << n) - 1;
    vector<vector<float>> dp(1 << n, vector<float>(n, 1e18f));
    vector<vector<int>> parent(1 << n, vector<int>(n, -1));
    dp[1][0] = 0;
    for (int mask = 1; mask <= full; mask++) {
        if (!(mask & 1)) continue;
        for (int u = 0; u < n; u++) {
            if (!(mask & (1 << u))) continue;
            if (dp[mask][u] >= 1e18f) continue;
            for (int v = 0; v < n; v++) {
                if (mask & (1 << v)) continue;
                int nm = mask | (1 << v);
                float nd = dp[mask][u] + dist(u, v);
                if (nd < dp[nm][v]) {
                    dp[nm][v] = nd;
                    parent[nm][v] = u;
                }
            }
        }
    }
    // find best end (return to 0)
    float best = 1e18f;
    int last = -1;
    for (int u = 1; u < n; u++) {
        float c = dp[full][u] + dist(u, 0);
        if (c < best) { best = c; last = u; }
    }
    vector<int> tour(n);
    int mask = full, cur = last;
    for (int i = n - 1; i >= 1; i--) {
        tour[i] = cur;
        int p = parent[mask][cur];
        mask ^= (1 << cur);
        cur = p;
    }
    tour[0] = 0;
    return tour;
}

// Grid-based spatial index for nearest neighbor
struct Grid {
    float minx, miny, maxx, maxy;
    float cellw, cellh;
    int gw, gh;
    vector<vector<int>> cells;

    void build(int maxCells) {
        minx = miny = 1e18f;
        maxx = maxy = -1e18f;
        for (int i = 0; i < N; i++) {
            if (gx[i] < minx) minx = gx[i];
            if (gx[i] > maxx) maxx = gx[i];
            if (gy[i] < miny) miny = gy[i];
            if (gy[i] > maxy) maxy = gy[i];
        }
        float dx = maxx - minx, dy = maxy - miny;
        if (dx < 1e-9f) dx = 1.0f;
        if (dy < 1e-9f) dy = 1.0f;
        int totalCells = min(maxCells, max(1, N / 4));
        gw = max(1, (int)sqrtf((float)totalCells * dx / dy));
        gh = max(1, totalCells / gw);
        cellw = dx / gw + 1e-9f;
        cellh = dy / gh + 1e-9f;
        cells.resize((size_t)gw * gh);
        for (int i = 0; i < N; i++) {
            int cx = min((int)((gx[i] - minx) / cellw), gw - 1);
            int cy = min((int)((gy[i] - miny) / cellh), gh - 1);
            cells[(size_t)cy * gw + cx].push_back(i);
        }
    }

    int nearest(int src, const vector<bool>& used) {
        float sx = gx[src], sy = gy[src];
        int cx = min((int)((sx - minx) / cellw), gw - 1);
        int cy = min((int)((sy - miny) / cellh), gh - 1);
        float bestd = 1e18f;
        int bestid = -1;
        for (int r = 0; ; r++) {
            // can we possibly find anything closer in ring r?
            if (r > 0) {
                float ringDist = min({
                    (r - 1) * cellw,
                    (r - 1) * cellh
                }) * 0.9f; // conservative
                if (ringDist > bestd && bestid >= 0) break;
            }
            if (cx - r < 0 && cx + r >= gw && cy - r < 0 && cy + r >= gh) break;
            int x0 = max(0, cx - r), x1 = min(gw - 1, cx + r);
            int y0 = max(0, cy - r), y1 = min(gh - 1, cy + r);
            for (int yy = y0; yy <= y1; yy++) {
                for (int xx = x0; xx <= x1; xx++) {
                    if (r > 0 && xx > x0 && xx < x1 && yy > y0 && yy < y1) continue;
                    for (int id : cells[(size_t)yy * gw + xx]) {
                        if (used[id]) continue;
                        float dx = gx[id] - sx, dy = gy[id] - sy;
                        float d = dx*dx + dy*dy;
                        if (d < bestd) { bestd = d; bestid = id; }
                    }
                }
            }
            if (bestid >= 0 && r >= 1) {
                // check if we can prune
                float minEdgeDist = 1e18f;
                if (cx - r - 1 >= 0) minEdgeDist = min(minEdgeDist, ((cx - r) * cellw + minx - sx));
                if (cx + r + 1 < gw) minEdgeDist = min(minEdgeDist, (sx - (cx + r + 1) * cellw - minx));
                // simplified: if best found, check next ring minimum possible distance
                float nextRingMin = 0;
                if (r >= 1) {
                    float dxMin = max(0.0f, (float)(r) * cellw - cellw);
                    float dyMin = max(0.0f, (float)(r) * cellh - cellh);
                    nextRingMin = dxMin * dxMin + dyMin * dyMin;
                    // not reliable, just break after finding in ring r and checking r+1
                }
                // We found something. Do one more ring to be safe, handled by loop continuing
                // Actually let's use a simpler approach: once found, check ring r+1 then break
            }
        }
        return bestid;
    }

    void remove(int id) {
        int cx = min((int)((gx[id] - minx) / cellw), gw - 1);
        int cy = min((int)((gy[id] - miny) / cellh), gh - 1);
        auto& cell = cells[(size_t)cy * gw + cx];
        for (int i = 0; i < (int)cell.size(); i++) {
            if (cell[i] == id) {
                cell[i] = cell.back();
                cell.pop_back();
                break;
            }
        }
    }
};

static vector<int> nearestNeighborGrid() {
    Grid grid;
    int maxCells = min(N * 2, 4000000);
    grid.build(maxCells);

    vector<int> tour;
    tour.reserve(N);
    vector<bool> used(N, false);
    tour.push_back(0);
    used[0] = true;
    grid.remove(0);

    for (int i = 1; i < N; i++) {
        int last = tour.back();
        int next = grid.nearest(last, used);
        tour.push_back(next);
        used[next] = true;
        grid.remove(next);
    }
    return tour;
}

static vector<int> nearestNeighborSimple() {
    vector<int> tour;
    tour.reserve(N);
    vector<bool> used(N, false);
    tour.push_back(0);
    used[0] = true;
    for (int i = 1; i < N; i++) {
        int last = tour.back();
        float bestd = 1e18f;
        int bestid = -1;
        for (int j = 0; j < N; j++) {
            if (used[j]) continue;
            float dx = gx[j] - gx[last], dy = gy[j] - gy[last];
            float d = dx*dx + dy*dy;
            if (d < bestd) { bestd = d; bestid = j; }
        }
        tour.push_back(bestid);
        used[bestid] = true;
    }
    return tour;
}

// 2-opt improvement
static void twoOpt(vector<int>& tour, double timeLimitSec) {
    auto start = chrono::steady_clock::now();
    int n = (int)tour.size();
    bool improved = true;
    int iterations = 0;
    while (improved) {
        improved = false;
        for (int i = 0; i < n - 1; i++) {
            for (int j = i + 2; j < n; j++) {
                if (i == 0 && j == n - 1) continue;
                int a = tour[i], b = tour[i+1];
                int c = tour[j], d = tour[(j+1) % n];
                float gain = dist(a, b) + dist(c, d) - dist(a, c) - dist(b, d);
                if (gain > 1e-6f) {
                    reverse(tour.begin() + i + 1, tour.begin() + j + 1);
                    improved = true;
                }
            }
            iterations++;
            if (iterations % 100 == 0) {
                auto now = chrono::steady_clock::now();
                double elapsed = chrono::duration<double>(now - start).count();
                if (elapsed > timeLimitSec) return;
            }
        }
    }
}

// Or-opt: move segments of 1, 2, or 3 cities
static void orOpt(vector<int>& tour, double timeLimitSec) {
    auto start = chrono::steady_clock::now();
    int n = (int)tour.size();
    bool improved = true;
    while (improved) {
        improved = false;
        for (int segLen = 1; segLen <= 3; segLen++) {
            for (int i = 1; i + segLen < n; i++) {
                int prev = tour[i - 1];
                int segStart = tour[i];
                int segEnd = tour[i + segLen - 1];
                int next = tour[(i + segLen) % n];

                float removeCost = dist(prev, segStart) + dist(segEnd, next) - dist(prev, next);

                for (int j = 0; j < n; j++) {
                    if (j >= i - 1 && j <= i + segLen) continue;
                    int a = tour[j], b = tour[(j + 1) % n];
                    float insertCost = dist(a, segStart) + dist(segEnd, b) - dist(a, b);
                    if (removeCost - insertCost > 1e-6f) {
                        // perform move
                        vector<int> seg(tour.begin() + i, tour.begin() + i + segLen);
                        tour.erase(tour.begin() + i, tour.begin() + i + segLen);
                        int insertPos = j < i ? j + 1 : j + 1 - segLen;
                        tour.insert(tour.begin() + insertPos, seg.begin(), seg.end());
                        improved = true;
                        goto next_iter;
                    }
                }
            }
        }
        next_iter:;
        auto now = chrono::steady_clock::now();
        if (chrono::duration<double>(now - start).count() > timeLimitSec) return;
    }
}

int main() {
    auto globalStart = chrono::steady_clock::now();

    // Fast I/O
    if (scanf("%d", &N) != 1) return 1;

    gx = (float*)malloc(sizeof(float) * N);
    gy = (float*)malloc(sizeof(float) * N);

    for (int i = 0; i < N; i++) {
        double x, y;
        if (scanf("%lf %lf", &x, &y) != 2) return 1;
        gx[i] = (float)x;
        gy[i] = (float)y;
    }

    vector<int> tour;

    if (N <= 1) {
        for (int i = 0; i < N; i++) printf("%d ", i);
        printf("\n");
        free(gx); free(gy);
        return 0;
    }

    if (N <= 20) {
        tour = solveTinyDP();
    } else if (N <= 5000) {
        tour = nearestNeighborSimple();
        double elapsed = chrono::duration<double>(chrono::steady_clock::now() - globalStart).count();
        double remain = 280.0 - elapsed;
        if (remain > 1.0) twoOpt(tour, remain * 0.7);
        elapsed = chrono::duration<double>(chrono::steady_clock::now() - globalStart).count();
        remain = 280.0 - elapsed;
        if (remain > 1.0) orOpt(tour, remain * 0.5);
        elapsed = chrono::duration<double>(chrono::steady_clock::now() - globalStart).count();
        remain = 280.0 - elapsed;
        if (remain > 1.0) twoOpt(tour, remain * 0.8);
    } else {
        tour = nearestNeighborGrid();
        double elapsed = chrono::duration<double>(chrono::steady_clock::now() - globalStart).count();
        double remain = 280.0 - elapsed;
        if (N <= 50000 && remain > 5.0) {
            twoOpt(tour, remain * 0.6);
            elapsed = chrono::duration<double>(chrono::steady_clock::now() - globalStart).count();
            remain = 280.0 - elapsed;
            if (remain > 2.0) orOpt(tour, remain * 0.5);
        }
    }

    // Output
    // Use large buffer for output
    vector<char> buf;
    buf.reserve((size_t)N * 9);
    for (int i = 0; i < N; i++) {
        char tmp[16];
        int len = sprintf(tmp, "%d", tour[i]);
        buf.insert(buf.end(), tmp, tmp + len);
        if (i < N - 1) buf.push_back(' ');
    }
    buf.push_back('\n');
    fwrite(buf.data(), 1, buf.size(), stdout);

    free(gx); free(gy);
    return 0;
}
