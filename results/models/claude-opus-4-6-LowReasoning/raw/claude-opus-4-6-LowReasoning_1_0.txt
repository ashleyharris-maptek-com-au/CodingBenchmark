#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <cmath>
#include <vector>
#include <algorithm>
#include <numeric>
#include <random>
#include <chrono>

using namespace std;

static float *gx, *gy;
static int N;

static inline float dist(int a, int b) {
    float dx = gx[a] - gx[b], dy = gy[a] - gy[b];
    return sqrtf(dx*dx + dy*dy);
}

// Nearest neighbor heuristic using spatial grid
static void nn_grid(vector<int>& tour) {
    tour.resize(N);
    tour[0] = 0;

    float minx = gx[0], maxx = gx[0], miny = gy[0], maxy = gy[0];
    for (int i = 1; i < N; i++) {
        if (gx[i] < minx) minx = gx[i];
        if (gx[i] > maxx) maxx = gx[i];
        if (gy[i] < miny) miny = gy[i];
        if (gy[i] > maxy) maxy = gy[i];
    }
    float rx = maxx - minx, ry = maxy - miny;
    if (rx < 1e-9f) rx = 1.0f;
    if (ry < 1e-9f) ry = 1.0f;

    int gridW = max(1, min((int)sqrtf((float)N * 0.5f), 10000));
    int gridH = gridW;
    float cellW = rx / gridW + 1e-12f;
    float cellH = ry / gridH + 1e-12f;

    // Build grid: each cell has a list of city indices
    vector<vector<int>> grid((size_t)gridW * gridH);
    for (int i = 0; i < N; i++) {
        int cx = min((int)((gx[i] - minx) / cellW), gridW - 1);
        int cy = min((int)((gy[i] - miny) / cellH), gridH - 1);
        grid[(size_t)cy * gridW + cx].push_back(i);
    }

    vector<bool> visited(N, false);
    visited[0] = true;
    int cur = 0;

    for (int step = 1; step < N; step++) {
        int cx = min((int)((gx[cur] - minx) / cellW), gridW - 1);
        int cy = min((int)((gy[cur] - miny) / cellH), gridH - 1);

        int best = -1;
        float bestD = 1e30f;

        for (int radius = 0; radius <= gridW + gridH; radius++) {
            int x0 = max(0, cx - radius), x1 = min(gridW - 1, cx + radius);
            int y0 = max(0, cy - radius), y1 = min(gridH - 1, cy + radius);

            for (int gy2 = y0; gy2 <= y1; gy2++) {
                for (int gx2 = x0; gx2 <= x1; gx2++) {
                    if (radius > 0 && gx2 > x0 && gx2 < x1 && gy2 > y0 && gy2 < y1) continue;
                    auto& cell = grid[(size_t)gy2 * gridW + gx2];
                    // Remove visited from cell while scanning
                    int sz = (int)cell.size();
                    for (int k = 0; k < sz; ) {
                        int c = cell[k];
                        if (visited[c]) {
                            cell[k] = cell[--sz];
                            cell.resize(sz);
                            continue;
                        }
                        float d = dist(cur, c);
                        if (d < bestD) { bestD = d; best = c; }
                        k++;
                    }
                }
            }
            if (best >= 0) {
                // Check if further rings can't improve
                float minPossibleDist = max(0.0f, (float)(radius - 1)) * min(cellW, cellH);
                if (minPossibleDist > bestD) break;
            }
        }

        visited[best] = true;
        tour[step] = best;
        cur = best;
    }
}

// Simple nearest neighbor (no grid, for small N)
static void nn_simple(vector<int>& tour) {
    tour.resize(N);
    tour[0] = 0;
    vector<bool> visited(N, false);
    visited[0] = true;
    int cur = 0;
    for (int step = 1; step < N; step++) {
        int best = -1;
        float bestD = 1e30f;
        for (int i = 0; i < N; i++) {
            if (!visited[i]) {
                float d = dist(cur, i);
                if (d < bestD) { bestD = d; best = i; }
            }
        }
        visited[best] = true;
        tour[step] = best;
        cur = best;
    }
}

// 2-opt improvement
static void two_opt(vector<int>& tour, double timeLimitSec) {
    auto start = chrono::steady_clock::now();
    int n = (int)tour.size();
    if (n < 4) return;

    // Build position lookup
    vector<int> pos(N);
    for (int i = 0; i < n; i++) pos[tour[i]] = i;

    bool improved = true;
    int passes = 0;
    int maxPasses = (n <= 5000) ? 100 : (n <= 50000) ? 5 : 2;

    while (improved && passes < maxPasses) {
        improved = false;
        passes++;
        for (int i = 0; i < n - 2; i++) {
            if (i % 1000 == 0) {
                auto now = chrono::steady_clock::now();
                double elapsed = chrono::duration<double>(now - start).count();
                if (elapsed > timeLimitSec) return;
            }
            int a = tour[i], b = tour[i + 1];
            float ab = dist(a, b);
            int jlim = (i == 0) ? n - 1 : n;
            for (int j = i + 2; j < jlim; j++) {
                int c = tour[j], d = tour[(j + 1) % n];
                float cd = dist(c, d);
                float ac = dist(a, c);
                float bd = dist(b, d);
                if (ac + bd < ab + cd - 1e-6f) {
                    // Reverse tour[i+1..j]
                    int lo = i + 1, hi = j;
                    while (lo < hi) {
                        swap(tour[lo], tour[hi]);
                        lo++; hi--;
                    }
                    improved = true;
                    break;
                }
            }
        }
    }
}

// Or-opt: move segments of 1,2,3 cities
static void or_opt(vector<int>& tour, double timeLimitSec) {
    auto start = chrono::steady_clock::now();
    int n = (int)tour.size();
    if (n < 5) return;

    bool improved = true;
    while (improved) {
        improved = false;
        auto now = chrono::steady_clock::now();
        if (chrono::duration<double>(now - start).count() > timeLimitSec) return;

        for (int segLen = 1; segLen <= 3; segLen++) {
            for (int i = 1; i + segLen < n; i++) {
                if (i % 2000 == 0) {
                    now = chrono::steady_clock::now();
                    if (chrono::duration<double>(now - start).count() > timeLimitSec) return;
                }
                int prev = tour[i - 1];
                int first = tour[i];
                int last = tour[i + segLen - 1];
                int next = tour[(i + segLen) % n];

                float removeCost = dist(prev, first) + dist(last, next);
                float bridgeCost = dist(prev, next);
                float gain0 = removeCost - bridgeCost;

                for (int j = 0; j < n; j++) {
                    if (j >= i - 1 && j <= i + segLen) continue;
                    int a = tour[j], b = tour[(j + 1) % n];
                    float insertCost = dist(a, first) + dist(last, b) - dist(a, b);
                    if (gain0 - insertCost > 1e-6f) {
                        // Perform the move
                        vector<int> seg(tour.begin() + i, tour.begin() + i + segLen);
                        tour.erase(tour.begin() + i, tour.begin() + i + segLen);
                        int insertPos = (j < i) ? j + 1 : j + 1 - segLen;
                        tour.insert(tour.begin() + insertPos, seg.begin(), seg.end());
                        improved = true;
                        goto next_pass;
                    }
                }
            }
        }
        next_pass:;
    }
}

// Exact DP for small N (bitmask)
static void solve_exact(vector<int>& tour) {
    // Held-Karp DP
    int n = N;
    vector<vector<float>> d(n, vector<float>(n));
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            d[i][j] = dist(i, j);

    int states = 1 << n;
    vector<vector<float>> dp(states, vector<float>(n, 1e30f));
    vector<vector<int>> parent(states, vector<int>(n, -1));
    dp[1][0] = 0;

    for (int mask = 1; mask < states; mask++) {
        for (int u = 0; u < n; u++) {
            if (!(mask & (1 << u))) continue;
            if (dp[mask][u] >= 1e29f) continue;
            for (int v = 0; v < n; v++) {
                if (mask & (1 << v)) continue;
                int nmask = mask | (1 << v);
                float nd = dp[mask][u] + d[u][v];
                if (nd < dp[nmask][v]) {
                    dp[nmask][v] = nd;
                    parent[nmask][v] = u;
                }
            }
        }
    }

    int full = states - 1;
    // Find best end (including return to 0)
    float best = 1e30f;
    int last = -1;
    for (int u = 1; u < n; u++) {
        float cost = dp[full][u] + d[u][0];
        if (cost < best) { best = cost; last = u; }
    }

    tour.resize(n);
    int mask = full;
    for (int i = n - 1; i >= 1; i--) {
        tour[i] = last;
        int prev = parent[mask][last];
        mask ^= (1 << last);
        last = prev;
    }
    tour[0] = 0;
}

int main() {
    // Fast I/O
    auto tStart = chrono::steady_clock::now();

    if (scanf("%d", &N) != 1) return 1;

    gx = (float*)malloc((size_t)N * sizeof(float));
    gy = (float*)malloc((size_t)N * sizeof(float));

    for (int i = 0; i < N; i++) {
        double x, y;
        if (scanf("%lf %lf", &x, &y) != 2) return 1;
        gx[i] = (float)x;
        gy[i] = (float)y;
    }

    vector<int> tour;

    if (N <= 1) {
        tour = {0};
        if (N == 0) tour.clear();
    } else if (N <= 20) {
        solve_exact(tour);
    } else if (N <= 5000) {
        nn_simple(tour);
        auto now = chrono::steady_clock::now();
        double elapsed = chrono::duration<double>(now - tStart).count();
        double remain = 280.0 - elapsed;
        if (remain > 1.0) {
            two_opt(tour, remain * 0.7);
            now = chrono::steady_clock::now();
            elapsed = chrono::duration<double>(now - tStart).count();
            remain = 280.0 - elapsed;
            if (remain > 1.0 && N <= 2000) or_opt(tour, remain * 0.8);
        }
    } else {
        nn_grid(tour);
        auto now = chrono::steady_clock::now();
        double elapsed = chrono::duration<double>(now - tStart).count();
        double remain = 280.0 - elapsed;
        if (remain > 5.0 && N <= 100000) {
            two_opt(tour, remain * 0.8);
        }
    }

    // Output
    for (int i = 0; i < N; i++) {
        if (i > 0) putchar(' ');
        printf("%d", tour[i]);
    }
    putchar('\n');

    free(gx);
    free(gy);
    return 0;
}
