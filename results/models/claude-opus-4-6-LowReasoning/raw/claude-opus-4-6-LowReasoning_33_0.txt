using System;
using System.Diagnostics;

class QAP
{
    static int N;
    static int[][] flow, dist;
    static Random rng = new Random(42);

    static long ComputeCost(int[] p)
    {
        long cost = 0;
        for (int i = 0; i < N; i++)
            for (int j = 0; j < N; j++)
                cost += (long)flow[i][j] * dist[p[i]][p[j]];
        return cost;
    }

    static long SwapDelta(int[] p, int a, int b)
    {
        long delta = 0;
        for (int k = 0; k < N; k++)
        {
            if (k == a || k == b) continue;
            delta += (long)flow[a][k] * (dist[p[b]][p[k]] - dist[p[a]][p[k]]);
            delta += (long)flow[b][k] * (dist[p[a]][p[k]] - dist[p[b]][p[k]]);
            delta += (long)flow[k][a] * (dist[p[k]][p[b]] - dist[p[k]][p[a]]);
            delta += (long)flow[k][b] * (dist[p[k]][p[a]] - dist[p[k]][p[b]]);
        }
        delta += (long)flow[a][a] * (dist[p[b]][p[b]] - dist[p[a]][p[a]]);
        delta += (long)flow[b][b] * (dist[p[a]][p[a]] - dist[p[b]][p[b]]);
        delta += (long)flow[a][b] * (dist[p[b]][p[a]] - dist[p[a]][p[b]]);
        delta += (long)flow[b][a] * (dist[p[a]][p[b]] - dist[p[b]][p[a]]);
        return delta;
    }

    static int[] GreedyConstruction()
    {
        // Compute row sums for flow and dist to guide greedy assignment
        long[] flowSum = new long[N];
        long[] distSum = new long[N];
        for (int i = 0; i < N; i++)
            for (int j = 0; j < N; j++)
            {
                flowSum[i] += flow[i][j] + flow[j][i];
                distSum[i] += dist[i][j] + dist[j][i];
            }

        int[] flowOrder = new int[N];
        int[] distOrder = new int[N];
        for (int i = 0; i < N; i++) { flowOrder[i] = i; distOrder[i] = i; }
        Array.Sort(flowOrder, (a, b) => flowSum[b].CompareTo(flowSum[a]));
        Array.Sort(distOrder, (a, b) => distSum[a].CompareTo(distSum[b]));

        int[] p = new int[N];
        for (int i = 0; i < N; i++)
            p[flowOrder[i]] = distOrder[i];
        return p;
    }

    static void LocalSearch(int[] p, ref long cost)
    {
        bool improved = true;
        while (improved)
        {
            improved = false;
            for (int i = 0; i < N - 1; i++)
            {
                for (int j = i + 1; j < N; j++)
                {
                    long d = SwapDelta(p, i, j);
                    if (d < 0)
                    {
                        int tmp = p[i]; p[i] = p[j]; p[j] = tmp;
                        cost += d;
                        improved = true;
                    }
                }
            }
        }
    }

    static void Main()
    {
        var sw = Stopwatch.StartNew();
        var tokens = Console.In.ReadToEnd().Split((char[])null, StringSplitOptions.RemoveEmptyEntries);
        int idx = 0;
        N = int.Parse(tokens[idx++]);

        flow = new int[N][];
        dist = new int[N][];
        for (int i = 0; i < N; i++)
        {
            flow[i] = new int[N];
            for (int j = 0; j < N; j++)
                flow[i][j] = int.Parse(tokens[idx++]);
        }
        for (int i = 0; i < N; i++)
        {
            dist[i] = new int[N];
            for (int j = 0; j < N; j++)
                dist[i][j] = int.Parse(tokens[idx++]);
        }

        int[] bestP = GreedyConstruction();
        long bestCost = ComputeCost(bestP);
        LocalSearch(bestP, ref bestCost);

        // Simulated annealing if time permits
        double timeLimit = N <= 100 ? 30.0 : (N <= 500 ? 120.0 : 240.0);
        int[] curP = (int[])bestP.Clone();
        long curCost = bestCost;

        double t0 = Math.Max(bestCost * 0.02, 1.0);
        double tEnd = 1.0;
        long iter = 0;

        while (true)
        {
            double elapsed = sw.Elapsed.TotalSeconds;
            if (elapsed >= timeLimit) break;

            double frac = elapsed / timeLimit;
            double temp = t0 * Math.Pow(tEnd / t0, frac);

            int a = rng.Next(N);
            int b = rng.Next(N - 1);
            if (b >= a) b++;

            long delta = SwapDelta(curP, a, b);
            if (delta < 0 || rng.NextDouble() < Math.Exp(-delta / temp))
            {
                int tmp = curP[a]; curP[a] = curP[b]; curP[b] = tmp;
                curCost += delta;
                if (curCost < bestCost)
                {
                    bestCost = curCost;
                    Array.Copy(curP, bestP, N);
                }
            }

            iter++;
            if ((iter & 0xFFFF) == 0 && sw.Elapsed.TotalSeconds >= timeLimit) break;
        }

        // Final local search on best
        LocalSearch(bestP, ref bestCost);

        Console.WriteLine(bestCost);
        Console.WriteLine(string.Join(" ", bestP));
    }
}
