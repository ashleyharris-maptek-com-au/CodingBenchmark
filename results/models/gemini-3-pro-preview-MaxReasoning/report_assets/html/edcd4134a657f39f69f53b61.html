#ifndef M_PI

#endif

#pragma pack(push, 1)
struct Cell {
    uint16_t a;
    uint16_t b;
    uint16_t c;
};
#pragma pack(pop)

#define NOMINMAX
#include <windows.h>

class Terrain {
private:
    std::string filepath;
    int64_t width = 0;
    int64_t depth = 0;
    int64_t height = 0;

    // Memory mapping handles
    HANDLE file_handle = INVALID_HANDLE_VALUE;
    HANDLE mapping_handle = NULL;
    void* mapped_data = nullptr;

    // Fallback stream
    mutable std::ifstream fallback_file;
    bool use_fallback = false;

public:
    Terrain() = default;
    Terrain(const std::string& path, int64_t w, int64_t d, int64_t h)
        : filepath(path), width(w), depth(d), height(h) {
        
        // Try memory mapping first
        file_handle = CreateFileA(
            filepath.c_str(),
            GENERIC_READ,
            FILE_SHARE_READ,
            NULL,
            OPEN_EXISTING,
            FILE_ATTRIBUTE_NORMAL,
            NULL
        );
        if (file_handle != INVALID_HANDLE_VALUE) {
            mapping_handle = CreateFileMappingA(
                file_handle,
                NULL,
                PAGE_READONLY,
                0,
                0,
                NULL
            );
            if (mapping_handle != NULL) {
                mapped_data = MapViewOfFile(
                    mapping_handle,
                    FILE_MAP_READ,
                    0,
                    0,
                    0
                );
            }
        }

        if (mapped_data == nullptr) {
            // Clean up handles if mapping failed
            if (mapping_handle != NULL) CloseHandle(mapping_handle);
            if (file_handle != INVALID_HANDLE_VALUE) CloseHandle(file_handle);
            file_handle = INVALID_HANDLE_VALUE;
            mapping_handle = NULL;

            // Open fallback stream
            fallback_file.open(filepath, std::ios::binary);
            use_fallback = true;
        }
    }

    ~Terrain() {
        if (mapped_data) {
            UnmapViewOfFile(mapped_data);
        }
        if (mapping_handle) {
            CloseHandle(mapping_handle);
        }
        if (file_handle != INVALID_HANDLE_VALUE) {
            CloseHandle(file_handle);
        }
    }

    bool isOpen() const {
        return (mapped_data != nullptr) || fallback_file.is_open();
    }

    Cell getCell(int64_t x, int64_t y) const {
        if (x < 0 || x >= width || y < 0 || y >= depth) {
            return {0, 0, 0};
        }
        if (!use_fallback && mapped_data != nullptr) {
            uint64_t offset = (x * depth + y) * 6;
            const uint16_t* ptr = reinterpret_cast<const uint16_t*>(static_cast<const char*>(mapped_data) + offset);
            return {ptr[0], ptr[1], ptr[2]};
        } else {
            if (!fallback_file.is_open()) return {0, 0, 0};
            uint64_t offset = (x * depth + y) * 6;
            fallback_file.seekg(offset);
            uint16_t data[3] = {0, 0, 0};
            fallback_file.read(reinterpret_cast<char*>(data), 6);
            return {data[0], data[1], data[2]};
        }
    }

    double getHMax(double x, double y) const {
        int ix = std::clamp(static_cast<int>(std::floor(x)), 0, static_cast<int>(width - 1));
        int iy = std::clamp(static_cast<int>(std::floor(y)), 0, static_cast<int>(depth - 1));
        Cell cell = getCell(ix, iy);
        double h = cell.a;
        if (cell.b != cell.c) {
            h = std::max(h, static_cast<double>(cell.c));
        }
        return h;
    }
};

enum State {
    CLIMB,
    CRUISE,
    LANDING,
    FINISHED
};

int main() {
    // Set up standard I/O optimization
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    int64_t width, depth, height;
    double gravity, max_thrust, fuel;
    double start_x, start_y, start_z;
    double target_x, target_y, target_z;

    if (!(std::cin >> width >> depth >> height >> gravity >> max_thrust >> fuel)) {
        return 1;
    }
    if (!(std::cin >> start_x >> start_y >> start_z >> target_x >> target_y >> target_z)) {
        return 1;
    }

    // Consume remainder of the line
    std::string dummy;
    std::getline(std::cin, dummy);

    // Read terrain file path
    std::string terrain_file_path;
    if (!std::getline(std::cin, terrain_file_path)) {
        return 1;
    }
    while (!terrain_file_path.empty() && (terrain_file_path.back() == '\r' || terrain_file_path.back() == '\n' || terrain_file_path.back() == ' ')) {
        terrain_file_path.pop_back();
    }
    while (!terrain_file_path.empty() && terrain_file_path.front() == ' ') {
        terrain_file_path.erase(terrain_file_path.begin());
    }

    // Open terrain file
    Terrain terrain(terrain_file_path, width, depth, height);

    // Determine safety margin based on height
    double H_margin = std::min(15.0, height * 0.15);
    if (H_margin < 5.0) H_margin = 5.0;

    // Path Generation (Up-Over-Down style with terrain smoothing)
    double D = std::hypot(target_x - start_x, target_y - start_y);
    uint64_t N = std::max(10ULL, std::min(2000ULL, static_cast<uint64_t>(D / 5.0)));
    std::vector<double> path_x(N), path_y(N), Z_min(N);
    for (uint64_t i = 0; i < N; ++i) {
        double u = static_cast<double>(i) / (N - 1);
        path_x[i] = (1.0 - u) * start_x + u * target_x;
        path_y[i] = (1.0 - u) * start_y + u * target_y;
        Z_min[i] = terrain.getHMax(path_x[i], path_y[i]) + H_margin;
    }

    std::vector<double> Z_safe = Z_min;
    Z_safe[0] = std::max(Z_safe[0], start_z);
    Z_safe[N - 1] = std::max(Z_safe[N - 1], target_z);

    // Limit climb/descent slope (e.g. max 0.35 slope)
    double dx = (N > 1) ? D / (N - 1) : 1.0;
    double max_slope = 0.35;
    double max_dz = max_slope * dx;

    // Propagate slope constraints
    for (int pass = 0; pass < 3; ++pass) {
        for (int i = static_cast<int>(N) - 2; i >= 0; --i) {
            Z_safe[i] = std::max(Z_safe[i], Z_safe[i + 1] - max_dz);
        }
        for (int i = 1; i < N; ++i) {
            Z_safe[i] = std::max(Z_safe[i], Z_safe[i - 1] - max_dz);
        }
    }

    // Clamp Z_safe below height
    for (uint64_t i = 0; i < N; ++i) {
        Z_safe[i] = std::min(Z_safe[i], height - 3.0);
    }

    // Determine cruise speed
    double a_h_max = std::sqrt(std::max(0.1, max_thrust * max_thrust - gravity * gravity));
    double V_cruise = std::min(200.0, std::sqrt(D * 0.3 * a_h_max));
    if (V_cruise < 5.0) V_cruise = 5.0;

    State state = CLIMB;
    uint64_t last_closest_idx = 0;
    double path_yaw = std::atan2(target_y - start_y, target_x - start_x);
    double z_target_moving = -1.0;

    // Start reading states and outputting commands
    std::string line;
    while (std::getline(std::cin, line)) {
        while (!line.empty() && (line.back() == '\r' || line.back() == '\n')) {
            line.pop_back();
        }
        if (line.empty()) continue;
        if (line.find("STATE") != std::string::npos) continue;

        std::stringstream ss(line);
        double x, y, z, vx, vy, vz, pitch, yaw, roll, pitch_rate, yaw_rate, roll_rate, fuel_rem;
        if (!(ss >> x >> y >> z >> vx >> vy >> vz >> pitch >> yaw >> roll >> pitch_rate >> yaw_rate >> roll_rate >> fuel_rem)) {
            continue;
        }

        double D_rem = std::hypot(target_x - x, target_y - y);

        // State Transitions
        if (state == CLIMB) {
            if (z >= Z_safe[0] - 2.0) {
                state = CRUISE;
            }
        } else if (state == CRUISE) {
            if (D_rem < 5.0) {
                state = LANDING;
            }
        }

        double x_target = start_x;
        double y_target = start_y;
        double z_target = Z_safe[0];
        double vx_des = 0.0;
        double vy_des = 0.0;
        double vz_des = 0.0;

        if (state == CLIMB) {
            // Vertical climb only
            x_target = start_x;
            y_target = start_y;
            z_target = Z_safe[0];
            
            vx_des = 1.0 * (x_target - x);
            vy_des = 1.0 * (y_target - y);
            // Limit horizontal drift corrections during climb
            vx_des = std::clamp(vx_des, -2.0, 2.0);
            vy_des = std::clamp(vy_des, -2.0, 2.0);

            vz_des = 2.0; // Steady climb speed
        } else if (state == CRUISE) {
            // Find closest index on path (must be non-decreasing)
            double min_dist_sq = 1e18;
            uint64_t closest_idx = last_closest_idx;
            for (uint64_t i = last_closest_idx; i < N; ++i) {
                double dx_p = x - path_x[i];
                double dy_p = y - path_y[i];
                double dist_sq = dx_p * dx_p + dy_p * dy_p;
                if (dist_sq < min_dist_sq) {
                    min_dist_sq = dist_sq;
                    closest_idx = i;
                }
            }
            last_closest_idx = closest_idx;

            // Lookahead distance scales with speed
            double speed_h = std::hypot(vx, vy);
            double L = 15.0 + 1.5 * speed_h;

            uint64_t lookahead_idx = closest_idx;
            double acc_dist = 0.0;
            for (uint64_t i = closest_idx + 1; i < N; ++i) {
                double dx_p = path_x[i] - path_x[i - 1];
                double dy_p = path_y[i] - path_y[i - 1];
                acc_dist += std::hypot(dx_p, dy_p);
                lookahead_idx = i;
                if (acc_dist >= L) {
                    break;
                }
            }

            x_target = path_x[lookahead_idx];
            y_target = path_y[lookahead_idx];
            z_target = Z_safe[lookahead_idx];

            // Target speed scales down as we approach the end target
            double V_des_h = std::min(V_cruise, std::sqrt(2.0 * 0.5 * a_h_max * D_rem));
            
            double tx_dx = x_target - x;
            double tx_dy = y_target - y;
            double tx_dist = std::hypot(tx_dx, tx_dy);
            if (tx_dist > 1e-3) {
                vx_des = V_des_h * (tx_dx / tx_dist);
                vy_des = V_des_h * (tx_dy / tx_dist);
            }

            vz_des = 0.5 * (z_target - z);
            vz_des = std::clamp(vz_des, -15.0, 15.0);

            // Ensure we do not fly below the local terrain safety limit
            double local_margin = H_margin;
            if (D_rem < 15.0) {
                local_margin = H_margin * (D_rem / 15.0);
            }
            double z_safe_current = terrain.getHMax(x, y) + local_margin;
            if (z_target < z_safe_current) {
                z_target = z_safe_current;
            }
        } else if (state == LANDING) {
            x_target = target_x;
            y_target = target_y;

            vx_des = 1.2 * (x_target - x);
            vy_des = 1.2 * (y_target - y);
            vx_des = std::clamp(vx_des, -2.0, 2.0);
            vy_des = std::clamp(vy_des, -2.0, 2.0);

            // Moving-target vertical descent logic
            if (z_target_moving < 0.0) {
                z_target_moving = z;
            }
            z_target_moving = std::max(target_z, z_target_moving - 1.2 * 0.1); // descend 1.2m/s
            z_target = z_target_moving;
            vz_des = (z_target_moving > target_z) ? -1.2 : 0.0;
        }

        // 3D Force PID / Feedback Controller
        double fx = 1.5 * (vx_des - vx);
        double fy = 1.5 * (vy_des - vy);
        double fz = 0.8 * (z_target - z) + 1.5 * (vz_des - vz) + gravity;

        if (fz < 0.1) fz = 0.1;

        // Thrust Prioritization
        double f_mag = std::sqrt(fx * fx + fy * fy + fz * fz);
        if (f_mag > max_thrust) {
            if (fz > max_thrust) {
                fz = max_thrust;
                fx = 0.0;
                fy = 0.0;
            } else {
                double allowed_h = std::sqrt(max_thrust * max_thrust - fz * fz);
                double current_h = std::hypot(fx, fy);
                if (current_h > 1e-3) {
                    fx = (fx / current_h) * allowed_h;
                    fy = (fy / current_h) * allowed_h;
                }
            }
            f_mag = max_thrust;
        }

        double thrust_cmd = f_mag / max_thrust;
        thrust_cmd = std::max(0.0, std::min(1.0, thrust_cmd));

        // Convert force directions to pitch and yaw
        double tx = fx / f_mag;
        double ty = fy / f_mag;
        double tz = fz / f_mag;

        double p_des = std::acos(std::max(0.0, std::min(1.0, tz)));
        double y_des = path_yaw;
        if (p_des > 0.02) {
            y_des = std::atan2(-ty, -tx);
        }

        double pitch_cmd = 0.0;
        double yaw_cmd = 0.0;
        double roll_cmd = 0.0;

        // Check Touchdown
        if (state == LANDING && std::abs(x - target_x) < 3.0 && std::abs(y - target_y) < 3.0 &&
            (z - target_z < 0.3 || (z - target_z < 1.0 && std::abs(vz) < 0.1))) {
            thrust_cmd = 0.0;
            pitch_cmd = 0.0;
            yaw_cmd = 0.0;
            roll_cmd = 0.0;
            state = FINISHED;
        }

        if (state != FINISHED) {
            // Attitude PD controller
            double e_p = p_des - pitch;
            
            // Branchless and fast angle normalization using fmod
            double e_y = std::fmod(y_des - yaw + M_PI, 2.0 * M_PI);
            if (e_y < 0) e_y += 2.0 * M_PI;
            e_y -= M_PI;

            double e_r = std::fmod(0.0 - roll + M_PI, 2.0 * M_PI);
            if (e_r < 0) e_r += 2.0 * M_PI;
            e_r -= M_PI;

            double u_p = 8.0 * e_p - 4.0 * pitch_rate;
            double u_y = 8.0 * e_y - 4.0 * yaw_rate;
            double u_r = 8.0 * e_r - 4.0 * roll_rate;

            pitch_cmd = std::clamp(u_p, -1.0, 1.0);
            yaw_cmd = std::clamp(u_y, -1.0, 1.0);
            roll_cmd = std::clamp(u_r, -1.0, 1.0);
        }

        // Print command
        std::cout << std::fixed << std::setprecision(6) 
                  << thrust_cmd << " " 
                  << pitch_cmd << " " 
                  << yaw_cmd << " " 
                  << roll_cmd << "\n";
        std::cout.flush();
    }

    return 0;
}
