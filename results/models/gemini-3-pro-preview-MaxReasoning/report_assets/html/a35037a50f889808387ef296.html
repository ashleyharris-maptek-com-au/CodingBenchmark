use std::io::{self, BufRead};
use std::time::Instant;

struct SimpleRng {
    state: u64,
}

impl SimpleRng {
    fn new(seed: u64) -> Self {
        Self { state: seed }
    }
    
    fn next_u64(&mut self) -> u64 {
        let mut x = self.state;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.state = x;
        x
    }
    
    fn next_usize(&mut self, limit: usize) -> usize {
        (self.next_u64() as usize) % limit
    }
    
    fn shuffle<T>(&mut self, slice: &mut [T]) {
        for i in (1..slice.len()).rev() {
            let j = self.next_usize(i + 1);
            slice.swap(i, j);
        }
    }
}

const PART_U: u8 = 0;
const PART_A: u8 = 1;
const PART_B: u8 = 2;

#[allow(clippy::too_many_arguments)]
fn dfs(
    depth: usize,
    current_cut: i32,
    rem_a: i32,
    rem_b: i32,
    m_u: i32,
    best_cut: &mut i32,
    best_partition: &mut Vec<usize>,
    partition: &mut Vec<u8>,
    d_a: &mut Vec<i32>,
    d_b: &mut Vec<i32>,
    d_u: &mut Vec<i32>,
    adj: &Vec<Vec<usize>>,
    adj_matrix: &Vec<Vec<bool>>,
    order: &Vec<usize>,
    n: usize,
    buffer: &mut [i32],
    start_time: &Instant,
    time_limit_ms: u128,
    timeout: &mut bool,
) {
    if *timeout {
        return;
    }
    if start_time.elapsed().as_millis() > time_limit_ms {
        *timeout = true;
        return;
    }

    if current_cut >= *best_cut {
        return;
    }

    // Optimization: if rem_a == 0, only B is possible
    if rem_a == 0 {
        let mut final_cut = current_cut;
        for i in depth..n {
            final_cut += d_a[order[i]];
        }
        if final_cut < *best_cut {
            *best_cut = final_cut;
            for i in 0..depth {
                best_partition[order[i]] = partition[order[i]] as usize;
            }
            for i in depth..n {
                best_partition[order[i]] = PART_B as usize;
            }
        }
        return;
    }

    // Optimization: if rem_b == 0, only A is possible
    if rem_b == 0 {
        let mut final_cut = current_cut;
        for i in depth..n {
            final_cut += d_b[order[i]];
        }
        if final_cut < *best_cut {
            *best_cut = final_cut;
            for i in 0..depth {
                best_partition[order[i]] = partition[order[i]] as usize;
            }
            for i in depth..n {
                best_partition[order[i]] = PART_A as usize;
            }
        }
        return;
    }

    // Simple lower bound check
    let mut lb_simple = current_cut;
    for i in depth..n {
        let u = order[i];
        lb_simple += std::cmp::min(d_a[u], d_b[u]);
    }
    if lb_simple >= *best_cut {
        return;
    }

    // Full lower bound calculation
    let mut lb_sum = 0;
    let limit_a_val = rem_a - 1;
    let limit_b_val = rem_b - 1;
    let has_limit_a = rem_a > 1;
    let has_limit_b = rem_b > 1;

    for i in depth..n {
        let u = order[i];
        let du = d_u[u];
        let limit_a = if has_limit_a { std::cmp::min(du, limit_a_val) } else { 0 };
        let limit_b = if has_limit_b { std::cmp::min(du, limit_b_val) } else { 0 };
        
        let two_fa = 2 * d_b[u] - limit_a;
        let two_fb = 2 * d_a[u] - limit_b;
        
        lb_sum += two_fb;
        buffer[i] = two_fa - two_fb;
    }

    let rem_a_idx = rem_a as usize;
    let slice = &mut buffer[depth..n];
    if rem_a_idx > 0 && rem_a_idx < slice.len() {
        let (left, _, _) = slice.select_nth_unstable(rem_a_idx);
        for &val in left.iter() {
            lb_sum += val;
        }
    } else if rem_a_idx == slice.len() {
        for &val in slice.iter() {
            lb_sum += val;
        }
    }

    let lower_bound = (2 * current_cut + 2 * m_u + lb_sum + 1) / 2;
    if lower_bound >= *best_cut {
        return;
    }

    // Branching on P[depth]
    let u = order[depth];

    // Heuristic: try the partition that increases cut less first
    let try_a_first = d_b[u] < d_a[u];

    let branches = if try_a_first {
        [(PART_A, rem_a - 1, d_b[u]), (PART_B, rem_b - 1, d_a[u])]
    } else {
        [(PART_B, rem_b - 1, d_a[u]), (PART_A, rem_a - 1, d_b[u])]
    };

    for &(part, new_rem, cut_inc) in &branches {
        // Apply transition
        partition[u] = part;
        let prev_m_u = m_u;
        let mut edges_removed_from_u = 0;
        
        for &neighbor in &adj[u] {
            if partition[neighbor] == PART_U {
                d_u[neighbor] -= 1;
                edges_removed_from_u += 1;
                if part == PART_A {
                    d_a[neighbor] += 1;
                } else {
                    d_b[neighbor] += 1;
                }
            }
        }

        let next_rem_a = if part == PART_A { new_rem } else { rem_a };
        let next_rem_b = if part == PART_B { new_rem } else { rem_b };

        dfs(
            depth + 1,
            current_cut + cut_inc,
            next_rem_a,
            next_rem_b,
            prev_m_u - edges_removed_from_u,
            best_cut,
            best_partition,
            partition,
            d_a,
            d_b,
            d_u,
            adj,
            adj_matrix,
            order,
            n,
            buffer,
            start_time,
            time_limit_ms,
            timeout,
        );

        // Backtrack transition
        for &neighbor in &adj[u] {
            if partition[neighbor] == PART_U {
                d_u[neighbor] += 1;
                if part == PART_A {
                    d_a[neighbor] -= 1;
                } else {
                    d_b[neighbor] -= 1;
                }
            }
        }
        partition[u] = PART_U;
    }
}

fn kernighan_lin(
    n: usize,
    adj: &Vec<Vec<usize>>,
    adj_matrix: &Vec<Vec<bool>>,
    rng: &mut SimpleRng,
    best_cut: &mut i32,
    best_partition: &mut Vec<usize>,
) {
    let mut vertices: Vec<usize> = (0..n).collect();
    rng.shuffle(&mut vertices);
    
    let mut partition = vec![0; n];
    for i in 0..n/2 {
        partition[vertices[i]] = 1;
    }
    for i in n/2..n {
        partition[vertices[i]] = 2;
    }
    
    let mut improved = true;
    while improved {
        improved = false;
        
        let mut d = vec![0; n];
        for u in 0..n {
            let mut ext = 0;
            let mut r#in = 0;
            let p_u = partition[u];
            for &v in &adj[u] {
                if partition[v] == p_u {
                    r#in += 1;
                } else {
                    ext += 1;
                }
            }
            d[u] = ext - r#in;
        }
        
        let mut swapped = vec![false; n];
        let mut swap_pairs = Vec::with_capacity(n / 2);
        let mut gains = Vec::with_capacity(n / 2);
        
        let mut temp_partition = partition.clone();
        let mut temp_d = d.clone();
        
        for _ in 0..n/2 {
            let mut best_a = None;
            let mut best_b = None;
            let mut max_gain = i32::MIN;
            
            for u in 0..n {
                if temp_partition[u] == 1 && !swapped[u] {
                    for v in 0..n {
                        if temp_partition[v] == 2 && !swapped[v] {
                            let gain = temp_d[u] + temp_d[v] - if adj_matrix[u][v] { 2 } else { 0 };
                            if gain > max_gain {
                                max_gain = gain;
                                best_a = Some(u);
                                best_b = Some(v);
                            }
                        }
                    }
                }
            }
            
            if let (Some(a), Some(b)) = (best_a, best_b) {
                swapped[a] = true;
                swapped[b] = true;
                swap_pairs.push((a, b));
                gains.push(max_gain);
                
                temp_partition[a] = 2;
                temp_partition[b] = 1;
                
                for x in 0..n {
                    if !swapped[x] {
                        let c_xa = adj_matrix[x][a];
                        let c_xb = adj_matrix[x][b];
                        if partition[x] == 1 {
                            temp_d[x] += if c_xa { 2 } else { 0 } - if c_xb { 2 } else { 0 };
                        } else {
                            temp_d[x] += if c_xb { 2 } else { 0 } - if c_xa { 2 } else { 0 };
                        }
                    }
                }
            } else {
                break;
            }
        }
        
        let mut cumulative_gain = 0;
        let mut max_cum_gain = 0;
        let mut best_k = 0;
        for i in 0..swap_pairs.len() {
            cumulative_gain += gains[i];
            if cumulative_gain > max_cum_gain {
                max_cum_gain = cumulative_gain;
                best_k = i + 1;
            }
        }
        
        if max_cum_gain > 0 {
            for i in 0..best_k {
                let (a, b) = swap_pairs[i];
                partition.swap(a, b);
            }
            improved = true;
        }
    }
    
    let mut cut = 0;
    for u in 0..n {
        for &v in &adj[u] {
            if u < v && partition[u] != partition[v] {
                cut += 1;
            }
        }
    }
    
    if cut < *best_cut {
        *best_cut = cut;
        *best_partition = partition;
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();
    
    let first_line = match lines.next() {
        Some(Ok(line)) => line,
        _ => return,
    };
    let parts: Vec<&str> = first_line.split_whitespace().collect();
    if parts.len() < 2 {
        return;
    }
    let n: usize = parts[0].parse().unwrap();
    let m: usize = parts[1].parse().unwrap();
    
    let mut adj = vec![Vec::new(); n];
    let mut adj_matrix = vec![vec![false; n]; n];
    
    for _ in 0..m {
        if let Some(Ok(line)) = lines.next() {
            let edge: Vec<&str> = line.split_whitespace().collect();
            if edge.len() >= 2 {
                let u: usize = edge[0].parse().unwrap();
                let v: usize = edge[1].parse().unwrap();
                adj[u].push(v);
                adj[v].push(u);
                adj_matrix[u][v] = true;
                adj_matrix[v][u] = true;
            }
        }
    }
    
    let start_time = Instant::now();
    
    // 1. Initial heuristic: Kernighan-Lin
    let mut best_cut = i32::MAX;
    let mut best_partition = vec![0; n];
    let mut rng = SimpleRng::new(42);
    
    // Run KL with multiple random restarts
    let kl_restarts = if n < 50 { 30 } else { 10 };
    for _ in 0..kl_restarts {
        kernighan_lin(n, &adj, &adj_matrix, &mut rng, &mut best_cut, &mut best_partition);
    }
    
    // 2. Vertex ordering for branch and bound
    let mut order = Vec::with_capacity(n);
    let mut added = vec![false; n];
    let mut conn = vec![0; n];

    let mut best_v = 0;
    let mut max_deg = -1;
    for i in 0..n {
        let deg = adj[i].len() as i32;
        if deg > max_deg {
            max_deg = deg;
            best_v = i;
        }
    }

    order.push(best_v);
    added[best_v] = true;
    for &neighbor in &adj[best_v] {
        conn[neighbor] += 1;
    }

    for _ in 1..n {
        let mut best_v = -1;
        let mut max_conn = -1;
        let mut max_deg = -1;
        for i in 0..n {
            if !added[i] {
                if conn[i] > max_conn || (conn[i] == max_conn && (adj[i].len() as i32) > max_deg) {
                    max_conn = conn[i];
                    max_deg = adj[i].len() as i32;
                    best_v = i as i32;
                }
            }
        }
        let u = best_v as usize;
        order.push(u);
        added[u] = true;
        for &neighbor in &adj[u] {
            conn[neighbor] += 1;
        }
    }
    
    // 3. Exact search (Branch and Bound)
    // Symmetries broken: force order[0] to be in partition A (PART_A)
    let mut partition = vec![PART_U; n];
    let mut d_a = vec![0; n];
    let mut d_b = vec![0; n];
    let mut d_u = vec![0; n];
    for u in 0..n {
        d_u[u] = adj[u].len() as i32;
    }
    
    // Assign order[0] to PART_A
    let u0 = order[0];
    partition[u0] = PART_A;
    let mut m_u = m as i32;
    for &neighbor in &adj[u0] {
        d_u[neighbor] -= 1;
        m_u -= 1;
        d_a[neighbor] += 1;
    }
    
    let mut buffer = vec![0; n];
    let mut timeout = false;
    
    // Depth-first search starting from depth 1
    // Remaining counts: rem_a = n/2 - 1, rem_b = n/2
    dfs(
        1,
        0,
        (n / 2) as i32 - 1,
        (n / 2) as i32,
        m_u,
        &mut best_cut,
        &mut best_partition,
        &mut partition,
        &mut d_a,
        &mut d_b,
        &mut d_u,
        &adj,
        &adj_matrix,
        &order,
        n,
        &mut buffer,
        &start_time,
        1800, // 1.8 seconds time limit
        &mut timeout,
    );
    
    // Print the results
    println!("{}", best_cut);
    let mut part_a_vertices = Vec::new();
    for i in 0..n {
        if best_partition[i] == PART_A as usize {
            part_a_vertices.push(i);
        }
    }
    
    let output_str = part_a_vertices
        .iter()
        .map(|v| v.to_string())
        .collect::<Vec<String>>()
        .join(" ");
    println!("{}", output_str);
}
