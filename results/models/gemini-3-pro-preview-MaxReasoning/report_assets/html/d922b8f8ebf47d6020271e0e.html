public struct ValueList {
    public int[] Array;
    public int Count;
    
    public ValueList(int capacity) {
        Array = new int[capacity];
        Count = 0;
    }
    
    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    public void Add(int val) {
        if (Count == Array.Length) {
            System.Array.Resize(ref Array, Array.Length * 2);
        }
        Array[Count++] = val;
    }
    
    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    public void Clear() {
        Count = 0;
    }
    
    public int[] ToArray() {
        int[] res = new int[Count];
        System.Array.Copy(Array, 0, res, 0, Count);
        return res;
    }
}

public class BitMatrix {
    public readonly ulong[] Data;
    public readonly int RowWords;
    
    public BitMatrix(int n) {
        RowWords = (n + 63) / 64;
        Data = new ulong[n * RowWords];
    }
    
    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    public void AddEdge(int u, int v) {
        int idx1 = u * RowWords + (v >> 6);
        Data[idx1] |= (1UL << (v & 63));
        int idx2 = v * RowWords + (u >> 6);
        Data[idx2] |= (1UL << (u & 63));
    }
    
    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    public bool IsAdjacent(int u, int v) {
        int idx = u * RowWords + (v >> 6);
        return (Data[idx] & (1UL << (v & 63))) != 0;
    }
}

public class FastScanner {
    private readonly Stream stream;
    private readonly byte[] buffer = new byte[65536];
    private int head = 0;
    private int tail = 0;
    
    public FastScanner() {
        stream = Console.OpenStandardInput();
    }
    
    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    private byte Read() {
        if (head >= tail) {
            head = 0;
            tail = stream.Read(buffer, 0, buffer.Length);
            if (tail <= 0) return 0;
        }
        return buffer[head++];
    }
    
    public int NextInt() {
        byte b = Read();
        while (b <= 32 && b != 0) b = Read();
        if (b == 0) return -1;
        int val = 0;
        while (b > 32) {
            if (b >= '0' && b <= '9') {
                val = val * 10 + (b - '0');
            }
            b = Read();
        }
        return val;
    }
}

public class Solver {
    private static int globalMaxCliqueSize = 0;
    private static int[] globalBestClique = new int[0];
    private static readonly object lockObject = new object();
    
    private static Graph graph;
    private static BitMatrix bitMatrix;
    private static int[] degrees;
    
    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    private static bool IsAdjacent(int u, int v) {
        if (bitMatrix != null) {
            return bitMatrix.IsAdjacent(u, v);
        }
        
        int uStart = graph.Head[u];
        int uLen = graph.Head[u + 1] - uStart;
        int vStart = graph.Head[v];
        int vLen = graph.Head[v + 1] - vStart;
        
        int start = uLen <= vLen ? uStart : vStart;
        int len = uLen <= vLen ? uLen : vLen;
        int target = uLen <= vLen ? v : u;
        
        int low = start;
        int high = start + len - 1;
        while (low <= high) {
            int mid = (low + high) >> 1;
            int val = graph.AdjData[mid];
            if (val < target) {
                low = mid + 1;
            } else if (val > target) {
                high = mid - 1;
            } else {
                return true;
            }
        }
        return false;
    }
    
    public static void Main(string[] args) {
        FastScanner scanner = new FastScanner();
        int numVertices = scanner.NextInt();
        if (numVertices <= 0) {
            Console.WriteLine("0");
            return;
        }
        int numEdges = scanner.NextInt();
        if (numEdges < 0) numEdges = 0;
        
        int[] from = new int[numEdges];
        int[] to = new int[numEdges];
        degrees = new int[numVertices];
        
        int actualEdges = 0;
        for (int i = 0; i < numEdges; i++) {
            int u = scanner.NextInt();
            if (u == -1) break;
            int v = scanner.NextInt();
            if (v == -1) break;
            if (u >= 0 && u < numVertices && v >= 0 && v < numVertices && u != v) {
                from[actualEdges] = u;
                to[actualEdges] = v;
                degrees[u]++;
                degrees[v]++;
                actualEdges++;
            }
        }
        
        int[] head = new int[numVertices + 1];
        for (int i = 0; i < numVertices; i++) {
            head[i + 1] = head[i] + degrees[i];
        }
        
        int[] adjData = new int[2 * actualEdges];
        int[] currentOffset = new int[numVertices];
        Array.Copy(head, currentOffset, numVertices);
        for (int i = 0; i < actualEdges; i++) {
            int u = from[i];
            int v = to[i];
            adjData[currentOffset[u]++] = v;
            adjData[currentOffset[v]++] = u;
        }
        
        // Sort each adjacency list to allow binary search
        for (int i = 0; i < numVertices; i++) {
            Array.Sort(adjData, head[i], head[i + 1] - head[i]);
        }
        
        // Remove duplicates (self-loops already excluded during reading)
        int writePtr = 0;
        int[] newHead = new int[numVertices + 1];
        for (int i = 0; i < numVertices; i++) {
            newHead[i] = writePtr;
            int start = head[i];
            int end = head[i + 1];
            int last = -1;
            for (int j = start; j < end; j++) {
                int v = adjData[j];
                if (v == last) continue; // duplicate
                adjData[writePtr++] = v;
                last = v;
            }
        }
        newHead[numVertices] = writePtr;
        head = newHead;
        
        // Update degrees
        for (int i = 0; i < numVertices; i++) {
            degrees[i] = head[i + 1] - head[i];
        }
        
        graph = new Graph(numVertices, writePtr / 2, head, adjData);
        
        // Initialize BitMatrix if V <= 40000
        if (numVertices <= 40000) {
            bitMatrix = new BitMatrix(numVertices);
            for (int u = 0; u < numVertices; u++) {
                int start = head[u];
                int end = head[u + 1];
                for (int i = start; i < end; i++) {
                    bitMatrix.AddEdge(u, adjData[i]);
                }
            }
        }
        
        // CancellationToken for timeout (4.5 minutes = 270 seconds)
        using var cts = new CancellationTokenSource(TimeSpan.FromSeconds(270));
        
        // --- PHASE 1: Run Heuristic Search ---
        // Sort vertices by degree descending to select starting points
        int[] sortedVertices = new int[numVertices];
        for (int i = 0; i < numVertices; i++) sortedVertices[i] = i;
        Array.Sort(sortedVertices, (x, y) => degrees[y].CompareTo(degrees[x]));
        
        // Choose start vertices: top 15000 by degree, and some random ones if V is large
        List<int> startPoints = new List<int>();
        if (numVertices <= 20000) {
            for (int i = 0; i < numVertices; i++) {
                startPoints.Add(i);
            }
        } else {
            HashSet<int> selected = new HashSet<int>();
            for (int i = 0; i < 15000; i++) {
                selected.Add(sortedVertices[i]);
            }
            Random rand = new Random(42);
            while (selected.Count < 20000) {
                int v = rand.Next(numVertices);
                selected.Add(v);
            }
            startPoints = new List<int>(selected);
        }
        
        var threadLocalHeuristic = new ThreadLocal<ThreadLocalState>(
            () => new ThreadLocalState(numVertices, cts.Token), 
            trackAllValues: false
        );
        
        Parallel.ForEach(startPoints, new ParallelOptions { MaxDegreeOfParallelism = Environment.ProcessorCount }, (startV) => {
            var state = threadLocalHeuristic.Value;
            var clique = FastGreedy(graph, startV, state.IsNeighbor, state.Clique, state.C, state.NextC, degrees);
            var improved = ImproveClique(graph, clique, state.IsNeighbor, state.K_neighbors, state.W, state, state.InClique);
            
            if (improved.Count > globalMaxCliqueSize) {
                lock (lockObject) {
                    if (improved.Count > globalMaxCliqueSize) {
                        globalMaxCliqueSize = improved.Count;
                        globalBestClique = improved.ToArray();
                    }
                }
            }
        });
        
        // --- PHASE 2: Exact Solver ---
        // Compute degeneracy ordering
        int[] order = GetDegeneracyOrdering(graph);
        int[] rank = new int[numVertices];
        for (int i = 0; i < numVertices; i++) {
            rank[order[i]] = i;
        }
        
        // Run exact branch-and-bound search
        try {
            Parallel.For(0, numVertices, new ParallelOptions { 
                CancellationToken = cts.Token,
                MaxDegreeOfParallelism = Environment.ProcessorCount 
            }, (step) => {
                int i = numVertices - 1 - step;
                int u = order[i];
                
                var state = threadLocalHeuristic.Value;
                if (state.Token.IsCancellationRequested) return;
                
                // Collect candidates: neighbors of u with rank > i
                ref ValueList candidates = ref state.NextCandidates[0];
                candidates.Clear();
                
                int start = graph.Head[u];
                int end = graph.Head[u + 1];
                for (int j = start; j < end; j++) {
                    int v = graph.AdjData[j];
                    if (rank[v] > i) {
                        candidates.Add(v);
                    }
                }
                
                if (1 + candidates.Count <= globalMaxCliqueSize) {
                    return; // Prune
                }
                
                // Sort candidates by vertex ID to allow fast IntersectSorted
                Array.Sort(candidates.Array, 0, candidates.Count);
                
                ref ValueList clique = ref state.CliqueExact;
                clique.Clear();
                clique.Add(u);
                
                Search(ref clique, ref candidates, 1, state);
            });
        } catch (OperationCanceledException) {
            // Timeout reached, we use the best clique found so far
        }
        
        // Print the result
        Console.WriteLine(globalMaxCliqueSize);
        Console.WriteLine(string.Join(" ", globalBestClique));
    }
    
    public static List<int> FastGreedy(Graph graph, int startVertex, bool[] isNeighbor, List<int> clique, List<int> C, List<int> nextC, int[] degrees) {
        clique.Clear();
        clique.Add(startVertex);
        
        C.Clear();
        int start = graph.Head[startVertex];
        int end = graph.Head[startVertex + 1];
        for (int i = start; i < end; i++) {
            C.Add(graph.AdjData[i]);
        }
        
        while (C.Count > 0) {
            int bestIdx = -1;
            int maxDeg = -1;
            for (int i = 0; i < C.Count; i++) {
                int deg = degrees[C[i]];
                if (deg > maxDeg) {
                    maxDeg = deg;
                    bestIdx = i;
                }
            }
            
            int u = C[bestIdx];
            clique.Add(u);
            
            int uStart = graph.Head[u];
            int uEnd = graph.Head[u + 1];
            for (int i = uStart; i < uEnd; i++) {
                isNeighbor[graph.AdjData[i]] = true;
            }
            
            nextC.Clear();
            for (int i = 0; i < C.Count; i++) {
                int w = C[i];
                if (w != u && isNeighbor[w]) {
                    nextC.Add(w);
                }
            }
            
            for (int i = uStart; i < uEnd; i++) {
                isNeighbor[graph.AdjData[i]] = false;
            }
            
            var tmp = C;
            C = nextC;
            nextC = tmp;
        }
        
        return clique;
    }
    
    public static List<int> ImproveClique(Graph graph, List<int> clique, bool[] isNeighbor, int[] K_neighbors, List<int> W, ThreadLocalState state, bool[] inClique) {
        int V = graph.N;
        List<int> K = new List<int>(clique);
        
        for (int i = 0; i < K.Count; i++) inClique[K[i]] = true;
        
        // Compute initial K_neighbors
        foreach (int v in K) {
            int start = graph.Head[v];
            int end = graph.Head[v + 1];
            for (int i = start; i < end; i++) {
                K_neighbors[graph.AdjData[i]]++;
            }
        }
        
        bool improved = true;
        while (improved) {
            improved = false;
            
            W.Clear();
            int kCount = K.Count;
            if (kCount > 1) {
                for (int j = 0; j < kCount; j++) {
                    int v = K[j];
                    int start = graph.Head[v];
                    int end = graph.Head[v + 1];
                    for (int idx = start; idx < end; idx++) {
                        int w = graph.AdjData[idx];
                        if (!inClique[w] && K_neighbors[w] == kCount - 1) {
                            W.Add(w);
                            K_neighbors[w] = -1; // mark as added to W to avoid duplicates
                        }
                    }
                }
                // Restore K_neighbors for elements in W
                for (int j = 0; j < W.Count; j++) {
                    K_neighbors[W[j]] = kCount - 1;
                }
            } else {
                for (int i = 0; i < V; i++) {
                    if (!inClique[i]) W.Add(i);
                }
            }
            
            if (W.Count < 2) break;
            
            for (int i = 0; i < kCount; i++) {
                int v = K[i];
                state.VertexToCliqueIndex[v] = i;
                state.CandidateLists[i].Clear();
            }
            
            foreach (int w in W) {
                int uniqueNonNeighbor = -1;
                foreach (int v in K) {
                    if (!IsAdjacent(w, v)) {
                        uniqueNonNeighbor = v;
                        break;
                    }
                }
                if (uniqueNonNeighbor != -1) {
                    int idx = state.VertexToCliqueIndex[uniqueNonNeighbor];
                    state.CandidateLists[idx].Add(w);
                }
            }
            
            int targetY = -1;
            int targetX = -1;
            int targetZ = -1;
            for (int i = 0; i < kCount; i++) {
                int y = K[i];
                List<int> cand = state.CandidateLists[i];
                if (cand.Count < 2) continue;
                
                bool foundPair = false;
                for (int j = 0; j < cand.Count; j++) {
                    int x = cand[j];
                    for (int l = j + 1; l < cand.Count; l++) {
                        int z = cand[l];
                        if (IsAdjacent(x, z)) {
                            targetY = y;
                            targetX = x;
                            targetZ = z;
                            foundPair = true;
                            break;
                        }
                    }
                    if (foundPair) break;
                }
                if (foundPair) break;
            }
            
            for (int i = 0; i < kCount; i++) {
                state.VertexToCliqueIndex[K[i]] = -1;
            }
            
            if (targetY != -1) {
                K.Remove(targetY);
                inClique[targetY] = false;
                
                K.Add(targetX);
                inClique[targetX] = true;
                
                K.Add(targetZ);
                inClique[targetZ] = true;
                
                int startY = graph.Head[targetY];
                int endY = graph.Head[targetY + 1];
                for (int i = startY; i < endY; i++) {
                    K_neighbors[graph.AdjData[i]]--;
                }
                
                int startX = graph.Head[targetX];
                int endX = graph.Head[targetX + 1];
                for (int i = startX; i < endX; i++) {
                    K_neighbors[graph.AdjData[i]]++;
                }
                
                int startZ = graph.Head[targetZ];
                int endZ = graph.Head[targetZ + 1];
                for (int i = startZ; i < endZ; i++) {
                    K_neighbors[graph.AdjData[i]]++;
                }
                
                improved = true;
            }
        }
        
        // Cleanup K_neighbors and inClique before returning
        foreach (int v in K) {
            inClique[v] = false;
            int start = graph.Head[v];
            int end = graph.Head[v + 1];
            for (int i = start; i < end; i++) {
                K_neighbors[graph.AdjData[i]] = 0;
            }
        }
        
        return K;
    }
    
    public static int[] GetDegeneracyOrdering(Graph graph) {
        int n = graph.N;
        int[] degrees = new int[n];
        for (int i = 0; i < n; i++) degrees[i] = graph.Head[i + 1] - graph.Head[i];

        int[] head = new int[n];
        for (int i = 0; i < n; i++) head[i] = -1;
        int[] next = new int[n];
        int[] prev = new int[n];
        
        for (int i = 0; i < n; i++) {
            int d = degrees[i];
            next[i] = head[d];
            if (head[d] != -1) prev[head[d]] = i;
            prev[i] = -1;
            head[d] = i;
        }
        
        int[] order = new int[n];
        int[] rank = new int[n];
        bool[] removed = new bool[n];
        
        int minDeg = 0;
        for (int step = 0; step < n; step++) {
            while (minDeg < n && head[minDeg] == -1) {
                minDeg++;
            }
            if (minDeg == n) break;
            
            int u = head[minDeg];
            head[minDeg] = next[u];
            if (next[u] != -1) prev[next[u]] = -1;
            
            order[step] = u;
            rank[u] = step;
            removed[u] = true;
            
            int start = graph.Head[u];
            int end = graph.Head[u + 1];
            for (int i = start; i < end; i++) {
                int v = graph.AdjData[i];
                if (!removed[v]) {
                    int d = degrees[v];
                    int p = prev[v];
                    int nx = next[v];
                    if (p != -1) next[p] = nx;
                    else head[d] = nx;
                    if (nx != -1) prev[nx] = p;
                    
                    degrees[v]--;
                    int newD = degrees[v];
                    
                    next[v] = head[newD];
                    if (head[newD] != -1) prev[head[newD]] = v;
                    prev[v] = -1;
                    head[newD] = v;
                    
                    if (newD < minDeg) {
                        minDeg = newD;
                    }
                }
            }
        }
        return order;
    }
    
    public static void IntersectSorted(ref ValueList a, int[] b, int bStart, int bLen, ref ValueList result) {
        result.Clear();
        int aCount = a.Count;
        int[] aArr = a.Array;
        int i = 0, j = 0;
        while (i < aCount && j < bLen) {
            int valA = aArr[i];
            int valB = b[bStart + j];
            if (valA < valB) {
                i++;
            } else if (valA > valB) {
                j++;
            } else {
                result.Add(valA);
                i++;
                j++;
            }
        }
    }
    
    private static void SortIndicesByColor(int[] indices, int[] colors, int left, int right) {
        if (left >= right) return;
        if (right - left <= 12) {
            for (int k = left + 1; k <= right; k++) {
                int key = indices[k];
                int keyColor = colors[key];
                int m = k - 1;
                while (m >= left && colors[indices[m]] > keyColor) {
                    indices[m + 1] = indices[m];
                    m--;
                }
                indices[m + 1] = key;
            }
            return;
        }
        
        int pivot = colors[indices[left + (right - left) / 2]];
        int i = left;
        int j = right;
        while (i <= j) {
            while (colors[indices[i]] < pivot) i++;
            while (colors[indices[j]] > pivot) j--;
            if (i <= j) {
                int tmp = indices[i];
                indices[i] = indices[j];
                indices[j] = tmp;
                i++;
                j--;
            }
        }
        SortIndicesByColor(indices, colors, left, j);
        SortIndicesByColor(indices, colors, i, right);
    }
    
    private static void Search(ref ValueList clique, ref ValueList candidates, int depth, ThreadLocalState state) {
        if (state.Token.IsCancellationRequested) return;
        
        if (clique.Count + candidates.Count <= globalMaxCliqueSize) return;
        if (candidates.Count == 0) {
            if (clique.Count > globalMaxCliqueSize) {
                lock (lockObject) {
                    if (clique.Count > globalMaxCliqueSize) {
                        globalMaxCliqueSize = clique.Count;
                        globalBestClique = clique.ToArray();
                    }
                }
            }
            return;
        }
        
        int candCount = candidates.Count;
        int[] colors = state.Colors;
        if (colors.Length < candCount) {
            state.Colors = new int[candCount * 2];
            colors = state.Colors;
        }
        
        int[] candArr = candidates.Array;
        bool[] usedColors = state.UsedColors;
        int maxColor = 0;
        
        if (bitMatrix != null) {
            ulong[] data = bitMatrix.Data;
            int rowWords = bitMatrix.RowWords;
            
            for (int i = 0; i < candCount; i++) {
                int u = candArr[i];
                int uOffset = u * rowWords;
                for (int j = 0; j < i; j++) {
                    int v = candArr[j];
                    if ((data[uOffset + (v >> 6)] & (1UL << (v & 63))) != 0) {
                        int c = colors[j];
                        usedColors[c] = true;
                    }
                }
                int color = 1;
                while (usedColors[color]) {
                    color++;
                }
                colors[i] = color;
                if (color > maxColor) maxColor = color;
                
                for (int j = 0; j < i; j++) {
                    int v = candArr[j];
                    if ((data[uOffset + (v >> 6)] & (1UL << (v & 63))) != 0) {
                        int c = colors[j];
                        usedColors[c] = false;
                    }
                }
            }
        } else {
            for (int i = 0; i < candCount; i++) {
                int u = candArr[i];
                for (int j = 0; j < i; j++) {
                    if (IsAdjacent(u, candArr[j])) {
                        int c = colors[j];
                        usedColors[c] = true;
                    }
                }
                int color = 1;
                while (usedColors[color]) {
                    color++;
                }
                colors[i] = color;
                if (color > maxColor) maxColor = color;
                
                for (int j = 0; j < i; j++) {
                    if (IsAdjacent(u, candArr[j])) {
                        int c = colors[j];
                        usedColors[c] = false;
                    }
                }
            }
        }
        
        if (clique.Count + maxColor <= globalMaxCliqueSize) return;
        
        if (maxColor == 1) {
            if (clique.Count + 1 > globalMaxCliqueSize) {
                lock (lockObject) {
                    if (clique.Count + 1 > globalMaxCliqueSize) {
                        globalMaxCliqueSize = clique.Count + 1;
                        int[] res = new int[clique.Count + 1];
                        Array.Copy(clique.Array, 0, res, 0, clique.Count);
                        res[clique.Count] = candArr[0];
                        globalBestClique = res;
                    }
                }
            }
            return;
        }
        
        int[] indices = state.Indices;
        if (indices.Length < candCount) {
            state.Indices = new int[candCount * 2];
            indices = state.Indices;
        }
        for (int i = 0; i < candCount; i++) indices[i] = i;
        
        if (candCount > 1) {
            SortIndicesByColor(indices, colors, 0, candCount - 1);
        }
        
        for (int i = candCount - 1; i >= 0; i--) {
            int idx = indices[i];
            int v = candArr[idx];
            int color = colors[idx];
            
            if (clique.Count + color <= globalMaxCliqueSize) {
                break;
            }
            
            clique.Add(v);
            if (depth >= state.NextCandidates.Length) {
                var newCandidates = new ValueList[state.NextCandidates.Length * 2];
                Array.Copy(state.NextCandidates, newCandidates, state.NextCandidates.Length);
                for (int j = state.NextCandidates.Length; j < newCandidates.Length; j++) {
                    newCandidates[j] = new ValueList(1000);
                }
                state.NextCandidates = newCandidates;
            }
            IntersectSorted(ref candidates, graph.AdjData, graph.Head[v], graph.Head[v + 1] - graph.Head[v], ref state.NextCandidates[depth]);
            Search(ref clique, ref state.NextCandidates[depth], depth + 1, state);
            clique.Count--;
        }
    }
