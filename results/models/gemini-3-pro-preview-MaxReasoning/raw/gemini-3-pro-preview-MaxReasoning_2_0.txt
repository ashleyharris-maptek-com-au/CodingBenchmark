using System;
using System.IO;
using System.Collections.Generic;

public class Solver
{
    public struct Edge
    {
        public int To;
        public int Weight;
        public int Id;
    }

    public struct OriginalEdge
    {
        public int U;
        public int V;
        public int Weight;
    }

    public struct CellEdge : IComparable<CellEdge>
    {
        public int U;
        public int V;
        public long Weight;
        public int MeetU;
        public int MeetV;
        public int MeetEdgeId;

        public int CompareTo(CellEdge other)
        {
            return Weight.CompareTo(other.Weight);
        }
    }

    public struct CellMstEdge
    {
        public int To;
        public int EdgeIndex;
    }

    public struct MultigraphEdge
    {
        public int To;
        public int Id;
    }

    public class FastScanner
    {
        private readonly Stream stream;
        private readonly byte[] buffer = new byte[1024 * 1024];
        private int head = 0;
        private int tail = 0;

        public FastScanner()
        {
            stream = Console.OpenStandardInput();
        }

        private byte Read()
        {
            if (head >= tail)
            {
                head = 0;
                tail = stream.Read(buffer, 0, buffer.Length);
                if (tail <= 0) return 0;
            }
            return buffer[head++];
        }

        public int NextInt()
        {
            byte c = Read();
            while (c <= 32)
            {
                if (c == 0) return -1;
                c = Read();
            }
            bool neg = false;
            if (c == '-')
            {
                neg = true;
                c = Read();
            }
            int res = 0;
            while (c > 32)
            {
                res = res * 10 + c - '0';
                c = Read();
            }
            return neg ? -res : res;
        }
    }

    public class MinHeap
    {
        private struct HeapNode
        {
            public long Dist;
            public int Vertex;
        }

        private HeapNode[] heap;
        private int size;

        public MinHeap(int capacity)
        {
            heap = new HeapNode[capacity];
            size = 0;
        }

        public void Push(long dist, int vertex)
        {
            if (size == heap.Length)
            {
                Array.Resize(ref heap, heap.Length * 2);
            }
            heap[size] = new HeapNode { Dist = dist, Vertex = vertex };
            Up(size++);
        }

        public (long dist, int vertex) Pop()
        {
            var res = heap[0];
            heap[0] = heap[--size];
            Down(0);
            return (res.Dist, res.Vertex);
        }

        public int Count => size;

        private void Up(int i)
        {
            while (i > 0)
            {
                int p = (i - 1) / 2;
                if (heap[i].Dist >= heap[p].Dist) break;
                var tmp = heap[i]; heap[i] = heap[p]; heap[p] = tmp;
                i = p;
            }
        }

        private void Down(int i)
        {
            while (i * 2 + 1 < size)
            {
                int l = i * 2 + 1;
                int r = l + 1;
                int best = l;
                if (r < size && heap[r].Dist < heap[l].Dist) best = r;
                if (heap[i].Dist <= heap[best].Dist) break;
                var tmp = heap[i]; heap[i] = heap[best]; heap[best] = tmp;
                i = best;
            }
        }
    }

    public class Dsu
    {
        private int[] parent;

        public Dsu(int size)
        {
            parent = new int[size];
            for (int i = 0; i < size; i++) parent[i] = i;
        }

        public int Find(int i)
        {
            int root = i;
            while (root != parent[root])
            {
                root = parent[root];
            }
            int curr = i;
            while (curr != root)
            {
                int nxt = parent[curr];
                parent[curr] = root;
                curr = nxt;
            }
            return root;
        }

        public bool Union(int i, int j)
        {
            int rootI = Find(i);
            int rootJ = Find(j);
            if (rootI != rootJ)
            {
                parent[rootI] = rootJ;
                return true;
            }
            return false;
        }
    }

    public static void Main(string[] args)
    {
        FastScanner sc = new FastScanner();
        int N = sc.NextInt();
        if (N <= 0) return;
        int M = sc.NextInt();

        List<Edge>[] adj = new List<Edge>[N];
        for (int i = 0; i < N; i++)
        {
            adj[i] = new List<Edge>();
        }

        OriginalEdge[] originalEdges = new OriginalEdge[M];
        int[] degree = new int[N];

        for (int i = 0; i < M; i++)
        {
            int u = sc.NextInt();
            int v = sc.NextInt();
            int w = sc.NextInt();

            adj[u].Add(new Edge { To = v, Weight = w, Id = i });
            adj[v].Add(new Edge { To = u, Weight = w, Id = i });

            originalEdges[i] = new OriginalEdge { U = u, V = v, Weight = w };
            degree[u]++;
            degree[v]++;
        }

        List<int> oddVertices = new List<int>();
        for (int i = 0; i < N; i++)
        {
            if (degree[i] % 2 != 0)
            {
                oddVertices.Add(i);
            }
        }

        List<int> extraEdges = new List<int>();

        if (oddVertices.Count > 0)
        {
            // Run multi-source Dijkstra from all odd vertices
            long[] dist = new long[N];
            int[] owner = new int[N];
            int[] parentNode = new int[N];
            int[] parentEdgeId = new int[N];

            for (int i = 0; i < N; i++)
            {
                dist[i] = long.MaxValue;
                owner[i] = -1;
                parentNode[i] = -1;
                parentEdgeId[i] = -1;
            }

            MinHeap pq = new MinHeap(N);
            foreach (int u in oddVertices)
            {
                dist[u] = 0;
                owner[u] = u;
                pq.Push(0, u);
            }

            List<CellEdge> cellEdges = new List<CellEdge>();

            while (pq.Count > 0)
            {
                var node = pq.Pop();
                long d = node.dist;
                int u = node.vertex;
                if (d > dist[u]) continue;

                foreach (var edge in adj[u])
                {
                    int v = edge.To;
                    long newDist = d + edge.Weight;
                    if (owner[v] == -1)
                    {
                        owner[v] = owner[u];
                        dist[v] = newDist;
                        parentNode[v] = u;
                        parentEdgeId[v] = edge.Id;
                        pq.Push(newDist, v);
                    }
                    else if (owner[v] != owner[u])
                    {
                        // To avoid duplication, only add edge from smaller owner to larger owner
                        if (owner[u] < owner[v])
                        {
                            cellEdges.Add(new CellEdge
                            {
                                U = owner[u],
                                V = owner[v],
                                Weight = dist[u] + dist[v] + edge.Weight,
                                MeetU = u,
                                MeetV = v,
                                MeetEdgeId = edge.Id
                            });
                        }
                    }
                }
            }

            // Find MST on the cell graph using Kruskal's
            cellEdges.Sort();
            Dsu dsu = new Dsu(N);
            List<CellEdge> cellMst = new List<CellEdge>();

            foreach (var ce in cellEdges)
            {
                if (dsu.Union(ce.U, ce.V))
                {
                    cellMst.Add(ce);
                }
            }

            // Build cell MST adjacency list
            List<CellMstEdge>[] cellAdj = new List<CellMstEdge>[N];
            foreach (int u in oddVertices)
            {
                cellAdj[u] = new List<CellMstEdge>();
            }

            for (int i = 0; i < cellMst.Count; i++)
            {
                var ce = cellMst[i];
                cellAdj[ce.U].Add(new CellMstEdge { To = ce.V, EdgeIndex = i });
                cellAdj[ce.V].Add(new CellMstEdge { To = ce.U, EdgeIndex = i });
            }

            // Run bottom-up parity correction on cell MST using BFS order
            int[] bfsOrder = new int[oddVertices.Count];
            int[] cellParent = new int[N];
            int[] cellParentEdgeIndex = new int[N];
            bool[] visited = new bool[N];

            int head = 0, tail = 0;
            int startNode = oddVertices[0];
            bfsOrder[tail++] = startNode;
            visited[startNode] = true;
            cellParent[startNode] = -1;
            cellParentEdgeIndex[startNode] = -1;

            while (head < tail)
            {
                int u = bfsOrder[head++];
                foreach (var edge in cellAdj[u])
                {
                    int v = edge.To;
                    if (!visited[v])
                    {
                        visited[v] = true;
                        cellParent[v] = u;
                        cellParentEdgeIndex[v] = edge.EdgeIndex;
                        bfsOrder[tail++] = v;
                    }
                }
            }

            bool[] needFlip = new bool[N];
            foreach (int u in oddVertices)
            {
                needFlip[u] = true;
            }

            List<int> selectedCellMstEdgeIndices = new List<int>();

            for (int i = tail - 1; i >= 0; i--)
            {
                int u = bfsOrder[i];
                if (needFlip[u])
                {
                    int p = cellParent[u];
                    if (p != -1)
                    {
                        selectedCellMstEdgeIndices.Add(cellParentEdgeIndex[u]);
                        needFlip[u] = false;
                        needFlip[p] = !needFlip[p];
                    }
                }
            }

            // Reconstruct paths for selected cell MST edges and add to extraEdges
            foreach (int idx in selectedCellMstEdgeIndices)
            {
                var ce = cellMst[idx];
                
                // Trace from MeetU to U
                int curr = ce.MeetU;
                while (curr != ce.U)
                {
                    extraEdges.Add(parentEdgeId[curr]);
                    curr = parentNode[curr];
                }
                
                // Add the meeting edge
                extraEdges.Add(ce.MeetEdgeId);
                
                // Trace from MeetV to V
                curr = ce.MeetV;
                while (curr != ce.V)
                {
                    extraEdges.Add(parentEdgeId[curr]);
                    curr = parentNode[curr];
                }
            }
        }

        // Build multigraph
        int totalEdges = M + extraEdges.Count;
        List<MultigraphEdge>[] multiAdj = new List<MultigraphEdge>[N];
        for (int i = 0; i < N; i++)
        {
            multiAdj[i] = new List<MultigraphEdge>();
        }

        for (int u = 0; u < N; u++)
        {
            foreach (var edge in adj[u])
            {
                multiAdj[u].Add(new MultigraphEdge { To = edge.To, Id = edge.Id });
            }
        }

        for (int i = 0; i < extraEdges.Count; i++)
        {
            int edgeId = extraEdges[i];
            int newEdgeId = M + i;
            var orig = originalEdges[edgeId];
            multiAdj[orig.U].Add(new MultigraphEdge { To = orig.V, Id = newEdgeId });
            multiAdj[orig.V].Add(new MultigraphEdge { To = orig.U, Id = newEdgeId });
        }

        // Run Hierholzer's algorithm
        int[] nextEdge = new int[N];
        bool[] used = new bool[totalEdges];
        Stack<int> currPath = new Stack<int>();
        List<int> circuit = new List<int>();

        currPath.Push(0);

        while (currPath.Count > 0)
        {
            int u = currPath.Peek();
            bool found = false;
            while (nextEdge[u] < multiAdj[u].Count)
            {
                var edge = multiAdj[u][nextEdge[u]];
                nextEdge[u]++;
                if (!used[edge.Id])
                {
                    used[edge.Id] = true;
                    currPath.Push(edge.To);
                    found = true;
                    break;
                }
            }
            if (!found)
            {
                circuit.Add(currPath.Pop());
            }
        }

        // Output results
        using (var sw = new StreamWriter(Console.OpenStandardOutput(), System.Text.Encoding.ASCII, 1024 * 1024))
        {
            sw.WriteLine(circuit.Count);
            for (int i = circuit.Count - 1; i >= 0; i--)
            {
                sw.Write(circuit[i]);
                if (i > 0) sw.Write(' ');
            }
            sw.WriteLine();
        }
    }
}
