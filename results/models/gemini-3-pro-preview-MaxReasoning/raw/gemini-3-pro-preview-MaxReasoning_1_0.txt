// Fast Scanner with heap-allocated buffer to prevent stack overflow
class FastScanner {
    static const int SZ = 65536 * 16;
    std::vector<char> buf;
    int len = 0;
    int pos = 0;
    FILE* fp;

    inline char next_char() {
        if (pos == len) {
            pos = 0;
            len = (int)fread(buf.data(), 1, SZ, fp);
            if (len == 0) return EOF;
        }
        return buf[pos++];
    }

public:
    FastScanner(FILE* f = stdin) : fp(f), buf(SZ) {}

    inline bool next_int(int& val) {
        char c = next_char();
        while (c != EOF && c <= ' ') c = next_char();
        if (c == EOF) return false;
        bool neg = false;
        if (c == '-') {
            neg = true;
            c = next_char();
        }
        val = 0;
        while (c >= '0' && c <= '9') {
            val = val * 10 + (c - '0');
            c = next_char();
        }
        if (neg) val = -val;
        return true;
    }

    inline bool next_float(float& val) {
        char c = next_char();
        while (c != EOF && c <= ' ') c = next_char();
        if (c == EOF) return false;
        bool neg = false;
        if (c == '-') {
            neg = true;
            c = next_char();
        }
        double res = 0.0;
        while (c >= '0' && c <= '9') {
            res = res * 10.0 + (c - '0');
            c = next_char();
        }
        if (c == '.') {
            c = next_char();
            double dec = 0.1;
            while (c >= '0' && c <= '9') {
                res += (c - '0') * dec;
                dec *= 0.1;
                c = next_char();
            }
        }
        if (c == 'e' || c == 'E') {
            c = next_char();
            bool exp_neg = false;
            if (c == '-') {
                exp_neg = true;
                c = next_char();
            } else if (c == '+') {
                c = next_char();
            }
            int exp_val = 0;
            while (c >= '0' && c <= '9') {
                exp_val = exp_val * 10 + (c - '0');
                c = next_char();
            }
            if (exp_neg) {
                res /= std::pow(10.0, exp_val);
            } else {
                res *= std::pow(10.0, exp_val);
            }
        }
        val = (float)(neg ? -res : res);
        return true;
    }
};

// Fast Printer with heap-allocated buffer to prevent stack overflow
class FastPrinter {
    static const int SZ = 65536 * 16;
    std::vector<char> buf;
    int pos = 0;
    FILE* fp;

    inline void write_char(char c) {
        if (pos == SZ) {
            fwrite(buf.data(), 1, SZ, fp);
            pos = 0;
        }
        buf[pos++] = c;
    }

public:
    FastPrinter(FILE* f = stdout) : fp(f), buf(SZ) {}
    ~FastPrinter() {
        if (pos > 0) {
            fwrite(buf.data(), 1, pos, fp);
        }
    }

    inline void write_int(int val) {
        if (val < 0) {
            write_char('-');
            val = -val;
        }
        char temp[12];
        int idx = 0;
        do {
            temp[idx++] = (char)('0' + (val % 10));
            val /= 10;
        } while (val > 0);
        while (idx > 0) {
            write_char(temp[--idx]);
        }
    }
    
    inline void write_space() {
        write_char(' ');
    }
    
    inline void write_newline() {
        write_char('\n');
    }
};

};

// Parallel loop implementation utilizing std::thread
template<typename F>
void parallel_for(size_t start, size_t end, F&& f) {
    size_t num_threads = std::thread::hardware_concurrency();
    if (num_threads == 0) num_threads = 4;
    size_t total = end - start;
    if (total < num_threads * 1000) {
        f(start, end);
        return;
    }
    std::vector<std::thread> threads;
    threads.reserve(num_threads);
    size_t chunk_size = total / num_threads;
    for (size_t i = 0; i < num_threads; ++i) {
        size_t s = start + i * chunk_size;
        size_t e = (i == num_threads - 1) ? end : s + chunk_size;
        threads.emplace_back([s, e, &f]() { f(s, e); });
    }
    for (auto& t : threads) {
        t.join();
    }
}

// Hilbert Curve Helper Functions
void rot(uint32_t n, uint32_t *x, uint32_t *y, uint32_t rx, uint32_t ry) {
    if (ry == 0) {
        if (rx == 1) {
            *x = n - 1 - *x;
            *y = n - 1 - *y;
        }
        uint32_t t = *x;
        *x = *y;
        *y = t;
    }
}

uint64_t xy2d(uint32_t n, uint32_t x, uint32_t y) {
    uint64_t d = 0;
    for (uint32_t s = n / 2; s > 0; s /= 2) {
        uint32_t rx = (x & s) > 0;
        uint32_t ry = (y & s) > 0;
        d += uint64_t(s) * s * ((3 * rx) ^ ry);
        rot(s, &x, &y, rx, ry);
    }
    return d;
}

// Small TSP strategy (N <= 2000) with full distance matrix
bool two_opt_small(std::vector<int>& tour, int N, const float* dist_matrix) {
    bool overall_improved = false;
    bool improved = true;
    int max_iters = 1000;
    while (improved && max_iters--) {
        improved = false;
        for (int i = 0; i < N - 1; ++i) {
            int u1 = tour[i];
            int u2 = tour[i + 1];
            int u1_offset = u1 * N;
            
            for (int j = i + 2; j < N; ++j) {
                int v1 = tour[j];
                int v2 = tour[(j + 1) % N];
                
                float d_curr = dist_matrix[u1_offset + u2] + dist_matrix[v1 * N + v2];
                float d_new = dist_matrix[u1_offset + v1] + dist_matrix[u2 * N + v2];
                
                if (d_new < d_curr - 1e-5f) {
                    std::reverse(tour.begin() + (i + 1), tour.begin() + (j + 1));
                    improved = true;
                    overall_improved = true;
                    u2 = tour[i + 1];
                }
            }
        }
    }
    return overall_improved;
}

std::vector<int> nearest_neighbor_tour(int start_city, int N, const float* dist_matrix) {
    std::vector<int> tour;
    tour.reserve(N);
    std::vector<bool> visited(N, false);
    
    int curr = start_city;
    tour.push_back(curr);
    visited[curr] = true;
    
    for (int step = 1; step < N; ++step) {
        int next_city = -1;
        float min_d = 1e20f;
        int curr_offset = curr * N;
        for (int i = 0; i < N; ++i) {
            if (!visited[i]) {
                float d = dist_matrix[curr_offset + i];
                if (d < min_d) {
                    min_d = d;
                    next_city = i;
                }
            }
        }
        curr = next_city;
        tour.push_back(curr);
        visited[curr] = true;
    }
    return tour;
}

// Medium TSP strategy (2000 < N <= 150000) using grid neighbor list 2-opt
std::vector<std::vector<int>> get_neighbors_grid(const std::vector<Point>& points, int K) {
    int n = points.size();
    std::vector<std::vector<int>> neighbors(n);
    
    float min_x = points[0].x, max_x = points[0].x;
    float min_y = points[0].y, max_y = points[0].y;
    for (const auto& p : points) {
        min_x = std::min(min_x, p.x);
        max_x = std::max(max_x, p.x);
        min_y = std::min(min_y, p.y);
        max_y = std::max(max_y, p.y);
    }
    
    float range_x = max_x - min_x;
    float range_y = max_y - min_y;
    if (range_x == 0) range_x = 1.0f;
    if (range_y == 0) range_y = 1.0f;
    
    int target_points_per_cell = 16;
    int num_cells = n / target_points_per_cell;
    if (num_cells < 1) num_cells = 1;
    
    float aspect = range_x / range_y;
    int cols = std::max(1, (int)std::round(std::sqrt(num_cells * aspect)));
    int rows = std::max(1, (int)std::round(std::sqrt(num_cells / aspect)));
    
    std::vector<std::vector<int>> grid(rows * cols);
    auto get_cell_idx = [&](float x, float y) {
        int col = (int)((x - min_x) / range_x * cols);
        int row = (int)((y - min_y) / range_y * rows);
        if (col < 0) col = 0; if (col >= cols) col = cols - 1;
        if (row < 0) row = 0; if (row >= rows) row = rows - 1;
        return row * cols + col;
    };
    
    for (int i = 0; i < n; ++i) {
        grid[get_cell_idx(points[i].x, points[i].y)].push_back(i);
    }
    
    parallel_for(0, n, [&](size_t start, size_t end) {
        std::vector<std::pair<float, int>> dists;
        dists.reserve(200);
        
        for (size_t i = start; i < end; ++i) {
            dists.clear();
            float px = points[i].x;
            float py = points[i].y;
            
            int col = (int)((px - min_x) / range_x * cols);
            int row = (int)((py - min_y) / range_y * rows);
            if (col < 0) col = 0; if (col >= cols) col = cols - 1;
            if (row < 0) row = 0; if (row >= rows) row = rows - 1;
            
            int r_limit = std::max(rows, cols);
            for (int r = 0; r <= r_limit; ++r) {
                for (int dr = -r; dr <= r; ++dr) {
                    for (int dc = -r; dc <= r; ++dc) {
                        if (std::abs(dr) != r && std::abs(dc) != r) continue;
                        int nr = row + dr;
                        int nc = col + dc;
                        if (nr >= 0 && nr < rows && nc >= 0 && nc < cols) {
                            int cell = nr * cols + nc;
                            for (int idx : grid[cell]) {
                                if (idx == (int)i) continue;
                                float dx = px - points[idx].x;
                                float dy = py - points[idx].y;
                                float d2 = dx * dx + dy * dy;
                                dists.push_back({d2, idx});
                            }
                        }
                    }
                }
                if ((int)dists.size() >= K + 10 || r >= 2) {
                    break;
                }
            }
            
            int actual_k = std::min(K, (int)dists.size());
            std::nth_element(dists.begin(), dists.begin() + actual_k, dists.end());
            neighbors[i].resize(actual_k);
            for (int k = 0; k < actual_k; ++k) {
                neighbors[i][k] = dists[k].second;
            }
        }
    });
    
    return neighbors;
}

inline void reverse_segment(std::vector<int>& tour, std::vector<int>& pos, int L, int R) {
    while (L < R) {
        int u = tour[L];
        int v = tour[R];
        tour[L] = v;
        tour[R] = u;
        pos[u] = R;
        pos[v] = L;
        L++;
        R--;
    }
}

void neighbor_2opt(std::vector<int>& tour, const std::vector<Point>& points, const std::vector<std::vector<int>>& neighbors, double time_limit_sec) {
    int n = tour.size();
    std::vector<int> pos(n);
    for (int i = 0; i < n; ++i) {
        pos[tour[i]] = i;
    }
    
    auto get_dist = [&](int u, int v) {
        float dx = points[u].x - points[v].x;
        float dy = points[u].y - points[v].y;
        return std::sqrt(dx * dx + dy * dy);
    };
    
    auto start_time = std::chrono::high_resolution_clock::now();
    
    bool improved = true;
    int iter = 0;
    while (improved) {
        improved = false;
        iter++;
        
        if (iter % 16 == 0) {
            auto curr_time = std::chrono::high_resolution_clock::now();
            double elapsed = std::chrono::duration<double>(curr_time - start_time).count();
            if (elapsed > time_limit_sec) {
                break;
            }
        }
        
        for (int u = 0; u < n; ++u) {
            int idx_u = pos[u];
            int idx_u2 = (idx_u + 1) % n;
            int u2 = tour[idx_u2];
            float dist_u_u2 = get_dist(u, u2);
            
            for (int v : neighbors[u]) {
                int idx_v = pos[v];
                if (idx_v == idx_u || idx_v == idx_u2) continue;
                
                int idx_v2 = (idx_v + 1) % n;
                int v2 = tour[idx_v2];
                if (v2 == u) continue;
                
                float dist_v_v2 = get_dist(v, v2);
                float dist_u_v = get_dist(u, v);
                float dist_u2_v2 = get_dist(u2, v2);
                
                float delta = (dist_u_v + dist_u2_v2) - (dist_u_u2 + dist_v_v2);
                if (delta < -1e-5f) {
                    int l = std::min(idx_u, idx_v) + 1;
                    int r = std::max(idx_u, idx_v);
                    reverse_segment(tour, pos, l, r);
                    improved = true;
                    break;
                }
            }
        }
    }
}

// Large/Extreme/Epic TSP strategy: Range-limited 2-opt in-place
void range_limited_2opt(std::vector<int>& tour, const std::vector<Point>& points, int W, double time_limit_sec) {
    int n = tour.size();
    auto get_dist = [&](int u, int v) {
        float dx = points[u].x - points[v].x;
        float dy = points[u].y - points[v].y;
        return std::sqrt(dx * dx + dy * dy);
    };
    
    auto start_time = std::chrono::high_resolution_clock::now();
    bool improved = true;
    int iter = 0;
    
    while (improved) {
        improved = false;
        iter++;
        
        if (iter % 4 == 0) {
            auto curr_time = std::chrono::high_resolution_clock::now();
            double elapsed = std::chrono::duration<double>(curr_time - start_time).count();
            if (elapsed > time_limit_sec) {
                break;
            }
        }
        
        for (int i = 0; i < n - 1; ++i) {
            int u = tour[i];
            int u2 = tour[i + 1];
            float dist_u_u2 = get_dist(u, u2);
            
            int limit = std::min(n, i + W);
            for (int j = i + 2; j < limit; ++j) {
                int v = tour[j];
                int v2 = tour[(j + 1) % n];
                
                float dist_v_v2 = get_dist(v, v2);
                float dist_u_v = get_dist(u, v);
                float dist_u2_v2 = get_dist(u2, v2);
                
                float delta = (dist_u_v + dist_u2_v2) - (dist_u_u2 + dist_v_v2);
                if (delta < -1e-5f) {
                    std::reverse(tour.begin() + (i + 1), tour.begin() + (j + 1));
                    improved = true;
                    u2 = tour[i + 1];
                    dist_u_u2 = get_dist(u, u2);
                }
            }
        }
    }
}

int main() {
    // Set standard fast I/O
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    FastScanner scanner(stdin);
    FastPrinter printer(stdout);
    
    int N;
    if (!scanner.next_int(N)) {
        return 0;
    }
    
    if (N <= 0) return 0;
    
    // Read the points
    std::vector<Point> points(N);
    for (int i = 0; i < N; ++i) {
        scanner.next_float(points[i].x);
        scanner.next_float(points[i].y);
    }
    
    std::vector<int> final_tour;
    
    if (N <= 100) {
        // Trivial Case: NN + 2-opt from all starting cities
        std::vector<float> dist_matrix(N * N);
        for (int i = 0; i < N; ++i) {
            for (int j = 0; j < N; ++j) {
                float dx = points[i].x - points[j].x;
                float dy = points[i].y - points[j].y;
                dist_matrix[i * N + j] = std::sqrt(dx * dx + dy * dy);
            }
        }
        
        float best_d = 1e20f;
        for (int start_city = 0; start_city < N; ++start_city) {
            auto tour = nearest_neighbor_tour(start_city, N, dist_matrix.data());
            two_opt_small(tour, N, dist_matrix.data());
            
            float d = 0;
            for (int i = 0; i < N; ++i) {
                d += dist_matrix[tour[i] * N + tour[(i + 1) % N]];
            }
            if (d < best_d) {
                best_d = d;
                final_tour = tour;
            }
        }
    } 
    else if (N <= 2000) {
        // Medium Case: NN + 2-opt from 40 distributed starts
        std::vector<float> dist_matrix(N * N);
        for (int i = 0; i < N; ++i) {
            for (int j = 0; j < N; ++j) {
                float dx = points[i].x - points[j].x;
                float dy = points[i].y - points[j].y;
                dist_matrix[i * N + j] = std::sqrt(dx * dx + dy * dy);
            }
        }
        
        float best_d = 1e20f;
        int num_starts = std::min(N, 40);
        for (int s = 0; s < num_starts; ++s) {
            int start_city = s * (N / num_starts);
            auto tour = nearest_neighbor_tour(start_city, N, dist_matrix.data());
            two_opt_small(tour, N, dist_matrix.data());
            
            float d = 0;
            for (int i = 0; i < N; ++i) {
                d += dist_matrix[tour[i] * N + tour[(i + 1) % N]];
            }
            if (d < best_d) {
                best_d = d;
                final_tour = tour;
            }
        }
    } 
    else {
        // Large, Extreme, Epic, and Ludicrous Cases: Hilbert curve + local optimization
        double min_x = points[0].x, max_x = points[0].x;
        double min_y = points[0].y, max_y = points[0].y;
        for (int i = 0; i < N; ++i) {
            if (points[i].x < min_x) min_x = points[i].x;
            if (points[i].x > max_x) max_x = points[i].x;
            if (points[i].y < min_y) min_y = points[i].y;
            if (points[i].y > max_y) max_y = points[i].y;
        }
        
        double max_range = std::max(max_x - min_x, max_y - min_y);
        if (max_range == 0.0) max_range = 1.0;
        
        uint32_t n_grid = 1 << 30;
        double scale = (n_grid - 1) / max_range;
        
        std::vector<City> cities(N);
        parallel_for(0, N, [&](size_t start, size_t end) {
            for (size_t i = start; i < end; ++i) {
                uint32_t xi = (uint32_t)((points[i].x - min_x) * scale);
                uint32_t yi = (uint32_t)((points[i].y - min_y) * scale);
                xi = std::min(xi, n_grid - 1);
                yi = std::min(yi, n_grid - 1);
                cities[i] = { xy2d(n_grid, xi, yi), (int)i };
            }
        });
        
        // Parallel sort
        std::sort(std::execution::par, cities.begin(), cities.end(), [](const City& a, const City& b) {
            return a.code < b.code;
        });
        
        final_tour.resize(N);
        for (int i = 0; i < N; ++i) {
            final_tour[i] = cities[i].id;
        }
        
        // Free cities memory to keep overhead low
        std::vector<City>().swap(cities);
        
        // Run local searches if N is within reasonable bounds
        if (N <= 150000) {
            // Large Case: Grid-based Neighbor Search 2-opt
            auto neighbors = get_neighbors_grid(points, 40);
            neighbor_2opt(final_tour, points, neighbors, 8.0);
        } 
        else if (N <= 3000000) {
            // Extreme Case: Range-limited 2-opt (W = 40, limit 15s)
            range_limited_2opt(final_tour, points, 40, 15.0);
        } 
        else if (N <= 10000000) {
            // Epic Case: Range-limited 2-opt (W = 10, limit 5s)
            range_limited_2opt(final_tour, points, 10, 5.0);
        }
    }
    
    // Rotate the tour so it starts at city 0
    int start_idx = -1;
    for (int i = 0; i < N; ++i) {
        if (final_tour[i] == 0) {
            start_idx = i;
            break;
        }
    }
    
    for (int i = 0; i < N; ++i) {
        printer.write_int(final_tour[(start_idx + i) % N]);
        printer.write_space();
    }
    printer.write_newline();
    
    return 0;
}
