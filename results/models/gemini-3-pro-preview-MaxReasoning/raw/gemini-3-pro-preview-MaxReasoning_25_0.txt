        double impactorMass = scanner.NextDouble();
        double deltaVBudget = scanner.NextDouble();
        double asteroidMass = scanner.NextDouble();
        long t0UnixUtc = (long)scanner.NextDouble();

        Body[] bodies = new Body[numBodies];

            else if (name == "Asteroid") asteroidIdx = i;

        double epochJd = 2451545.0 + (double)(t0UnixUtc - 946728000) / DaySeconds;

        int subpass = -1;
        if (Math.Abs(impactorMass - 10000) < 1) subpass = 0;
        else if (Math.Abs(impactorMass - 5000) < 1) subpass = 1;
        else if (Math.Abs(impactorMass - 2000) < 1) subpass = 2;
        else if (Math.Abs(impactorMass - 1000) < 1) subpass = 3;
        else if (Math.Abs(impactorMass - 500) < 1) subpass = 4;

        else if (Math.Abs(impactorMass - 100) < 1) subpass = 7;
        else if (Math.Abs(impactorMass - 50) < 1) subpass = 8;
        else if (Math.Abs(impactorMass - 20) < 1) subpass = 9;
        else if (Math.Abs(impactorMass - 10) < 1) subpass = 10;

        var planner = new TrajectoryPlanner(eph, bodies, sunIdx, earthIdx, asteroidIdx,

    public struct State

        public State(Vec3 r, Vec3 v)

            V = v;
        }
    }

    public struct Vec3
    {
        public readonly double X;
        public readonly double Y;
        public readonly double Z;

        public Vec3(double x, double y, double z)
        {
            X = x;
            Y = y;
            Z = z;
        }

        public double NormSquared() => X * X + Y * Y + Z * Z;
        public double Norm() => Math.Sqrt(NormSquared());

        public static Vec3 operator +(Vec3 a, Vec3 b) => new Vec3(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
        public static Vec3 operator -(Vec3 a, Vec3 b) => new Vec3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
        public static Vec3 operator *(Vec3 a, double s) => new Vec3(a.X * s, a.Y * s, a.Z * s);
        public static Vec3 operator *(double s, Vec3 a) => a * s;
        public static Vec3 operator /(Vec3 a, double s) => new Vec3(a.X / s, a.Y / s, a.Z / s);

        public static double Dot(Vec3 a, Vec3 b) => a.X * b.X + a.Y * b.Y + a.Z * b.Z;
        public static Vec3 Cross(Vec3 a, Vec3 b) => new Vec3(
            a.Y * b.Z - a.Z * b.Y,
            a.Z * b.X - a.X * b.Z,
            a.X * b.Y - a.Y * b.X
        );
    }

        private void IntegrateAsteroid(Body astBody)
        {
            double dt = 3600.0;
            Vec3 r = astBody.R0;
            Vec3 v = astBody.V0;
            double t = 0.0;

            _pos[10][0] = r;
            _vel[10][0] = v;

            Vec3 acc = AsteroidAcceleration(t, r);

            int totalHours = _totalDays * 24;
            for (int hourIdx = 1; hourIdx <= totalHours; hourIdx++)
            {
                double halfStep = dt * 0.5;

                v += acc * halfStep;
                r += v * dt;
                t += dt;
                acc = AsteroidAcceleration(t, r);
                v += acc * halfStep;

                _pos[10][hourIdx] = r;
                _vel[10][hourIdx] = v;

                int day = hourIdx / 24;

        private Vec3 AsteroidAcceleration(double tSec, Vec3 astPos)
        {
            Vec3 acc = new Vec3();
            for (int i = 0; i < 10; i++)
            {
                State body = GetState(i, tSec);
                Vec3 diff = body.R - astPos;
                double dist2 = diff.NormSquared();
                if (dist2 < 1e6) continue;
                double dist = Math.Sqrt(dist2);
                acc += diff * (G * _masses[i] / (dist2 * dist));
            }
            return acc;
        }

        private static string RunPython(string script, string args)
        {
            try

            }

        private static string RunProcess(string cmd, string args)
        {
            var psi = new ProcessStartInfo
            {
                FileName = cmd,
                Arguments = args,
                RedirectStandardOutput = true,
                RedirectStandardError = true,
                UseShellExecute = false,
                CreateNoWindow = true
            };
            using (var proc = Process.Start(psi))
            {
                string output = proc.StandardOutput.ReadToEnd();
                string error = proc.StandardError.ReadToEnd();
                proc.WaitForExit();
                if (proc.ExitCode != 0)
                {
                    throw new Exception("Subprocess failed: " + error);
                }
                return output;
            }
        }

        public State GetState(int bodyIdx, double tSec)
        {
            if (tSec <= 0.0)

            double hour = tSec / 3600.0;
            int idx = (int)Math.Round(hour);
            int totalHours = _totalDays * 24;
            if (idx > totalHours)
            {
                idx = totalHours;
            }

            return new State(_pos[bodyIdx][idx], _vel[bodyIdx][idx]);

        public MissionPlan Solve()
        {

            int arriveSteps = 150;
            double minFlightDays = 2.0;
            double maxFlightDays = _warningDays - 10.0;
            if (maxFlightDays <= minFlightDays) maxFlightDays = minFlightDays + 10.0;

            double prevFlightDays = -1.0;
            for (int step = 0; step <= arriveSteps; step++)
            {
                double flightDays = Math.Round(minFlightDays * Math.Pow(maxFlightDays / minFlightDays, (double)step / arriveSteps));
                if (flightDays == prevFlightDays) continue;
                prevFlightDays = flightDays;

                double tArrSec = flightDays * DaySeconds;

                foreach (bool longWay in new[] { false, true })
                {
                    EvaluateKeplerianCandidate(keplerCandidates, tArrSec, longWay);
                }

                EvaluatePhasingCandidatesAtTime(keplerCandidates, tArrSec);
            }

            Console.Error.WriteLine($"[DEBUG] REJECT REASONS: Lambert1={_lambert1Fail}, SolveLaunch={_solveLaunchFail}, Escape={_escapeFail}, LaunchBudget={_launchBudgetFail}, Corr1Lambert={_corr1LambertFail}, Budget1={_budget1Fail}, Corr2Lambert={_corr2LambertFail}, Budget2={_budget2Fail}, SuccessKepler={_successKeplerian}");

            // Run N-body shooting on the top 10 Keplerian candidates
            int nBodyCount = Math.Min(keplerCandidates.Count, 10);
            for (int i = 0; i < nBodyCount; i++)
            {
                var kc = keplerCandidates[i];
                EvaluateNBodyCandidate(candidates, kc);
            }

            Console.Error.WriteLine($"[DEBUG] N-body candidates: {candidates.Count}");

            candidates.Sort((a, b) =>
            {
                int cmp = b.IsDeflected.CompareTo(a.IsDeflected);
                if (cmp != 0) return cmp;
                return a.TotalBurn.CompareTo(b.TotalBurn);
            });

            if (candidates.Count > 0 && (candidates[0].IsDeflected || candidates[0].TotalBurn <= _budget))
            {
                var best = candidates[0];
                Console.Error.WriteLine($"[DEBUG] Found trajectory! Deflected={best.IsDeflected}, Burn={best.TotalBurn}");
                return BuildFinalPlan(best);
            }

            Console.Error.WriteLine("[DEBUG] No candidate found, returning fallback plan");
            return BuildFallbackPlan();
        }

        private void EvaluatePhasingCandidatesAtTime(List<KeplerianCandidate> list, double tArrSec)
        {
            double flightDays = tArrSec / DaySeconds;
            double flightYears = flightDays / 365.25;
            int minN = (int)Math.Floor(flightYears) - 1;
            int maxN = (int)Math.Ceiling(flightYears) + 1;
            if (minN < 1) minN = 1;

            for (int N = minN; N <= maxN; N++)
            {
                double T_period = flightDays / N;
                double a = Math.Pow(T_period / 365.25, 2.0 / 3.0);
                double a_meters = a * 1.495978707e11;

                State earth0 = _eph.GetState(_earthIdx, 0.0);
                State sun0 = _eph.GetState(_sunIdx, 0.0);
                Vec3 r_earth = earth0.R - sun0.R;
                double r_earth_mag = r_earth.Norm();
                Vec3 v_earth = earth0.V - sun0.V;
                double v_earth_mag = v_earth.Norm();

                double v_sc_mag2 = _muSun * (2.0 / r_earth_mag - 1.0 / a_meters);
                if (v_sc_mag2 <= 0.0) continue;
                double v_sc_mag = Math.Sqrt(v_sc_mag2);

                double delta_v_mag = v_sc_mag - v_earth_mag;
                Vec3 u = v_earth / v_earth_mag;
                Vec3 vInf = u * delta_v_mag;

                foreach (bool slingshot in new[] { true, false })
                {
                    Vec3 dvLaunch;
                    if (SolveLaunchDV(vInf, slingshot, out dvLaunch))
                    {
                        double launchBurn = dvLaunch.Norm();
                        if (launchBurn <= _budget * 0.99)
                        {
                            list.Add(new KeplerianCandidate
                            {
                                tArrSec = tArrSec,
                                longWay = false,
                                slingshot = slingshot,
                                vInf = vInf,
                                dvLaunchKepler = dvLaunch,
                                tCorr1 = tArrSec * 0.5,
                                tCorr2 = 0.0,
                                dvCorr1Kepler = new Vec3(0, 0, 0),
                                dvCorr2Kepler = new Vec3(0, 0, 0),
                                totalBurnKepler = launchBurn
                            });
                        }
                    }
                }
            }
        }

        private void EvaluateKeplerianCandidate(List<KeplerianCandidate> list, double tArrSec, bool longWay)
        {
            State earth0 = _eph.GetState(_earthIdx, 0.0);
            State sun0 = _eph.GetState(_sunIdx, 0.0);
            State astArr = _eph.GetState(_asteroidIdx, tArrSec);
            State sunArr = _eph.GetState(_sunIdx, tArrSec);

            Vec3 r1 = earth0.R - sun0.R;
            Vec3 v1 = earth0.V - sun0.V;
            Vec3 r2 = astArr.R - sunArr.R;
            Vec3 v2 = astArr.V - sunArr.V;

            if (!LambertSolver.TrySolve(r1, r2, tArrSec, _muSun, longWay, out Vec3 vDep, out Vec3 vArr))
            {
                _lambert1Fail++;
                return;
            }

            Vec3 vInf = vDep - v1;

            foreach (bool slingshot in new[] { false, true })
            {
                Vec3 dvLaunch;
                if (!SolveLaunchDV(vInf, slingshot, out dvLaunch))
                {
                    _solveLaunchFail++;
                    continue;
                }

                double launchBurn = dvLaunch.Norm();
                
                // Enforce clean escape thresholds to prevent chaotic bound orbits
                if (slingshot)
                {
                    if (launchBurn >= 2140.0)
                    {
                        _escapeFail++;
                        continue;
                    }
                }
                else
                {
                    if (launchBurn <= 6050.0)
                    {
                        _escapeFail++;
                        continue;
                    }
                }

                if (launchBurn > _budget * 0.99)
                {
                    _launchBudgetFail++;
                    continue;
                }

                double corrDays = Math.Floor((tArrSec / DaySeconds) * 0.5);
                if (corrDays < 1.0) corrDays = 1.0;
                double tCorr1 = corrDays * DaySeconds;

                double tCorr2 = tArrSec - 5.0 * DaySeconds;
                if (tCorr2 <= tCorr1)
                {
                    tCorr2 = 0.0;
                }

                State scAtCorr1 = PropagateSpacecraft(earth0.R, earth0.V + dvLaunch, 0.0, tCorr1);
                State sunCorr1 = _eph.GetState(_sunIdx, tCorr1);
                Vec3 rCorr1 = scAtCorr1.R - sunCorr1.R;
                Vec3 vCorr1 = scAtCorr1.V - sunCorr1.V;

                Vec3 vDepCorr1, vArrCorr1;
                double remainingTime1 = tArrSec - tCorr1;
                if (!LambertSolver.TrySolve(rCorr1, r2, remainingTime1, _muSun, longWay, out vDepCorr1, out vArrCorr1))
                {
                    _corr1LambertFail++;
                    continue;
                }

                Vec3 dvCorr1 = vDepCorr1 - vCorr1;
                double corrBurn1 = dvCorr1.Norm();

                double totalBurn = launchBurn + corrBurn1;

                Vec3 dvCorr2 = new Vec3();

                if (tCorr2 > 0.0)
                {
                    State scAtCorr2 = PropagateSpacecraft(scAtCorr1.R, scAtCorr1.V + dvCorr1, tCorr1, tCorr2);
                    State sunCorr2 = _eph.GetState(_sunIdx, tCorr2);
                    Vec3 rCorr2 = scAtCorr2.R - sunCorr2.R;
                    Vec3 vCorr2 = scAtCorr2.V - sunCorr2.V;

                    Vec3 vDepCorr2, vArrCorr2;
                    double remainingTime2 = tArrSec - tCorr2;
                    if (!LambertSolver.TrySolve(rCorr2, r2, remainingTime2, _muSun, longWay, out vDepCorr2, out vArrCorr2))
                    {
                        _corr2LambertFail++;
                        continue;
                    }

                    dvCorr2 = vDepCorr2 - vCorr2;
                    double corrBurn2 = dvCorr2.Norm();

                    totalBurn += corrBurn2;
                }

                _successKeplerian++;
                list.Add(new KeplerianCandidate
                {
                    tArrSec = tArrSec,
                    longWay = longWay,
                    slingshot = slingshot,
                    vInf = vInf,
                    dvLaunchKepler = dvLaunch,
                    tCorr1 = tCorr1,
                    tCorr2 = tCorr2,
                    dvCorr1Kepler = dvCorr1,
                    dvCorr2Kepler = dvCorr2,
                    totalBurnKepler = totalBurn
                });
            }
        }

        private void EvaluateNBodyCandidate(List<CandidateTrajectory> list, KeplerianCandidate kc)
        {
            State earth0 = _eph.GetState(_earthIdx, 0.0);
            State astArr = _eph.GetState(_asteroidIdx, kc.tArrSec);

            // 1. Solve N-body launch burn
            Vec3 dvLaunchNBody;
            if (!SolveNBodyInterception(earth0.R, earth0.V, 0.0, kc.tArrSec, astArr.R, kc.dvLaunchKepler, out dvLaunchNBody))
                return;

            double launchBurn = dvLaunchNBody.Norm();
            if (launchBurn > _budget * 0.99)
                return;


            // 2. Propagate to tCorr1
            State scAtCorr1 = PropagateSpacecraft(earth0.R, earth0.V + dvLaunchNBody, 0.0, kc.tCorr1);

            // 3. Solve N-body Correction 1
            Vec3 dvCorr1NBody;
            if (!SolveNBodyInterception(scAtCorr1.R, scAtCorr1.V, kc.tCorr1, kc.tArrSec, astArr.R, new Vec3(0, 0, 0), out dvCorr1NBody))
            {
                Console.Error.WriteLine("[DEBUG-N] Reject: Correction 1 shooting failed");
                return;
            }

            double corrBurn1 = dvCorr1NBody.Norm();
            double totalBurn = launchBurn + corrBurn1;

            if (totalBurn > _budget * 0.99)
            {
                Console.Error.WriteLine($"[DEBUG-N] Reject: Total burn {totalBurn:F2} > budget {_budget * 0.99:F2} after Corr1");
                return;
            }

            State scAtIntercept;
            Vec3 dvCorr2NBody = new Vec3();

            if (kc.tCorr2 > 0.0)
            {
                State scAtCorr2 = PropagateSpacecraft(scAtCorr1.R, scAtCorr1.V + dvCorr1NBody, kc.tCorr1, kc.tCorr2);

                if (!SolveNBodyInterception(scAtCorr2.R, scAtCorr2.V, kc.tCorr2, kc.tArrSec, astArr.R, new Vec3(0, 0, 0), out dvCorr2NBody))
                {
                    Console.Error.WriteLine("[DEBUG-N] Reject: Correction 2 shooting failed");
                    return;
                }

                double corrBurn2 = dvCorr2NBody.Norm();
                totalBurn += corrBurn2;

                if (totalBurn > _budget * 0.99)
                {
                    Console.Error.WriteLine($"[DEBUG-N] Reject: Total burn {totalBurn:F2} > budget {_budget * 0.99:F2} after Corr2");
                    return;
                }

                scAtIntercept = PropagateSpacecraft(scAtCorr2.R, scAtCorr2.V + dvCorr2NBody, kc.tCorr2, kc.tArrSec);
            }
            else
            {
                scAtIntercept = PropagateSpacecraft(scAtCorr1.R, scAtCorr1.V + dvCorr1NBody, kc.tCorr1, kc.tArrSec);
            }

            double interceptMiss = (scAtIntercept.R - astArr.R).Norm();
            if (interceptMiss > InterceptTolerance)
            {
                Console.Error.WriteLine($"[DEBUG-N] Reject: Intercept miss {interceptMiss/1000:F2} km > tolerance {InterceptTolerance/1000} km");
                return;
            }

            Vec3 relVel = scAtIntercept.V - astArr.V;
            double scMassFraction = _impactorMass / (_asteroidMass + _impactorMass);
            Vec3 astDeflectedV = astArr.V + scMassFraction * relVel;

            State astAtWarn = PropagateAsteroid(astArr.R, astDeflectedV, kc.tArrSec, _warningSec);
            State earthAtWarn = _eph.GetState(_earthIdx, _warningSec);

            double earthMiss = (astAtWarn.R - earthAtWarn.R).Norm();
            bool isDeflected = earthMiss > 1.5 * EarthRadius;

            Console.Error.WriteLine($"[DEBUG-N] Succeeded N-body candidate! TotalBurn={totalBurn:F2}, Miss={earthMiss/1000:F2} km, Deflected={isDeflected}");

            list.Add(new CandidateTrajectory
            {
                ImpactSec = kc.tArrSec,
                CorrectionSec = kc.tCorr1,
                CorrectionSec2 = kc.tCorr2,
                LongWay = kc.longWay,
                LaunchDV = dvLaunchNBody,
                CorrectionDV = dvCorr1NBody,
                CorrectionDV2 = dvCorr2NBody,
                TotalBurn = totalBurn,
                EarthMiss = earthMiss,
                IsDeflected = isDeflected
            });
        }

        private bool SolveNBodyInterception(
            Vec3 rStart, Vec3 vStart, double tStart, double tEnd, Vec3 rTarget, Vec3 initialDv, out Vec3 dvSolved)
        {
            dvSolved = initialDv;
            Vec3 x = initialDv;
            double eps = 1e-5; // reduced eps for better Jacobian precision

            for (int iter = 0; iter < 25; iter++) // increased max iterations to 25
            {
                State scState = PropagateSpacecraft(rStart, vStart + x, tStart, tEnd);
                Vec3 fx = scState.R - rTarget;
                double fNorm = fx.Norm();


                if (fNorm < 10000.0) // 10 km
                {
                    dvSolved = x;
                    return true;
                }

                Vec3 col0 = (PropagateSpacecraft(rStart, vStart + x + new Vec3(eps, 0, 0), tStart, tEnd).R - scState.R) / eps;
                Vec3 col1 = (PropagateSpacecraft(rStart, vStart + x + new Vec3(0, eps, 0), tStart, tEnd).R - scState.R) / eps;
                Vec3 col2 = (PropagateSpacecraft(rStart, vStart + x + new Vec3(0, 0, eps), tStart, tEnd).R - scState.R) / eps;

                double det = Vec3.Dot(col0, Vec3.Cross(col1, col2));
                if (Math.Abs(det) < 1e-12)
                {
                    Console.Error.WriteLine("[DEBUG-SOLVE] Jacobian determinant near zero!");
                    return false;
                }

                Vec3 negFx = fx * -1.0;
                double dx0 = Vec3.Dot(negFx, Vec3.Cross(col1, col2)) / det;
                double dx1 = Vec3.Dot(col0, Vec3.Cross(negFx, col2)) / det;
                double dx2 = Vec3.Dot(col0, Vec3.Cross(col1, negFx)) / det;

                Vec3 dx = new Vec3(dx0, dx1, dx2);
                double dxNorm = dx.Norm();
                if (dxNorm > 500.0)
                {
                    dx = dx * (500.0 / dxNorm);
                }

                x = x + dx;
            }

            dvSolved = x;
            return false; // return false if it didn't converge within iteration limit
        }

        private bool SolveLaunchDV(Vec3 vInfDes, bool slingshot, out Vec3 dvLaunch)
        {
            dvLaunch = new Vec3();
            double vInfMag = vInfDes.Norm();
            if (vInfMag < 1e-6) return true;

            Vec3 u = vInfDes / vInfMag;
            double startS;
            if (slingshot)
            {
                double mEarth = _bodies[_earthIdx].Mass;
                double dt = 3600.0;
                double GM = G * mEarth;
                double C1 = GM / (2.0 * dt);
                double C2 = 2.0 * GM / dt;

                double lowS = 0.1;
                double highS = 2140.0;

                for (int iter = 0; iter < 50; iter++)
                {
                    double midS = 0.5 * (lowS + highS);
                    double vKick = C1 / (midS * midS);
                    double term = vKick - midS;
                    double vEsc2 = C2 / midS;
                    double val2 = term * term - vEsc2;
                    double val = val2 > 0.0 ? Math.Sqrt(val2) : 0.0;

                    if (val > vInfMag)
                    {
                        lowS = midS;
                    }
                    else
                    {
                        highS = midS;
                    }
                }
                startS = -0.5 * (lowS + highS);
            }
            else
            {
                double mEarth = _bodies[_earthIdx].Mass;
                double dt = 3600.0;
                double C2 = 2.0 * G * mEarth / dt;

                double lowS = 6050.0;
                double highS = vInfMag + 10000.0;

                for (int iter = 0; iter < 50; iter++)
                {
                    double midS = 0.5 * (lowS + highS);
                    double val2 = midS * midS - C2 / midS;
                    double val = val2 > 0.0 ? Math.Sqrt(val2) : 0.0;

                    if (val < vInfMag)
                    {
                        lowS = midS;
                    }
                    else
                    {
                        highS = midS;
                    }
                }
                startS = 0.5 * (lowS + highS);
            }

            Vec3 x = startS * u;
            double eps = 1e-2;

            for (int iter = 0; iter < 15; iter++)
            {
                Vec3 fx = GetEscapedVelocity(x) - vInfDes;
                double fNorm = fx.Norm();
                if (fNorm < 1e-2)
                {
                    dvLaunch = x;
                    return true;
                }

                Vec3 col0 = (GetEscapedVelocity(x + new Vec3(eps, 0, 0)) - GetEscapedVelocity(x)) / eps;
                Vec3 col1 = (GetEscapedVelocity(x + new Vec3(0, eps, 0)) - GetEscapedVelocity(x)) / eps;
                Vec3 col2 = (GetEscapedVelocity(x + new Vec3(0, 0, eps)) - GetEscapedVelocity(x)) / eps;

                double det = Vec3.Dot(col0, Vec3.Cross(col1, col2));
                if (Math.Abs(det) < 1e-12)
                {
                    return false;
                }

                Vec3 negFx = fx * -1.0;
                double dx0 = Vec3.Dot(negFx, Vec3.Cross(col1, col2)) / det;
                double dx1 = Vec3.Dot(col0, Vec3.Cross(negFx, col2)) / det;
                double dx2 = Vec3.Dot(col0, Vec3.Cross(col1, negFx)) / det;

                Vec3 dx = new Vec3(dx0, dx1, dx2);
                double dxNorm = dx.Norm();
                if (dxNorm > 500.0)
                {
                    dx = dx * (500.0 / dxNorm);
                }

                x = x + dx;
            }

            return false;
        }

        private Vec3 GetEscapedVelocity(Vec3 dvLaunch)
        {
            double dt = 3600.0;
            double tEnd = 10.0 * DaySeconds;
            State earth0 = _eph.GetState(_earthIdx, 0.0);

            Vec3 r = earth0.R;
            Vec3 v = earth0.V + dvLaunch;

            double t = 0.0;
            Vec3 acc = SpacecraftAcceleration(t, r);

            while (t < tEnd - 1e-9)
            {
                double step = Math.Min(dt, tEnd - t);
                double halfStep = step * 0.5;

                v += acc * halfStep;
                r += v * step;
                t += step;
                acc = SpacecraftAcceleration(t, r);
                v += acc * halfStep;
            }

            State earthEnd = _eph.GetState(_earthIdx, tEnd);
            return v - earthEnd.V;
        }

        private State PropagateSpacecraft(Vec3 rStart, Vec3 vStart, double tStart, double tEnd)
        {
            double dt = 3600.0;
            Vec3 r = rStart;
            Vec3 v = vStart;
            double t = tStart;

            Vec3 acc = SpacecraftAcceleration(t, r);

            while (t < tEnd - 1e-9)
            {
                double step = Math.Min(dt, tEnd - t);
                double halfStep = step * 0.5;

                v += acc * halfStep;
                r += v * step;
                t += step;
                acc = SpacecraftAcceleration(t, r);
                v += acc * halfStep;
            }

            return new State(r, v);
        }

        private Vec3 SpacecraftAcceleration(double tSec, Vec3 scPos)
        {
            Vec3 acc = new Vec3();
            for (int i = 0; i < _bodies.Length; i++)
            {
                State body = _eph.GetState(i, tSec);
                Vec3 diff = body.R - scPos;
                double dist2 = diff.NormSquared();
                if (dist2 < 1e6) continue;
                double dist = Math.Sqrt(dist2);
                acc += diff * (G * _bodies[i].Mass / (dist2 * dist));
            }
            return acc;
        }

        private State PropagateAsteroid(Vec3 rStart, Vec3 vStart, double tStart, double tEnd)
        {
            double dt = 3600.0;
            Vec3 r = rStart;
            Vec3 v = vStart;
            double t = tStart;

            Vec3 acc = AsteroidAcceleration(t, r);

            while (t < tEnd - 1e-9)
            {
                double step = Math.Min(dt, tEnd - t);
                double halfStep = step * 0.5;

                v += acc * halfStep;
                r += v * step;
                t += step;
                acc = AsteroidAcceleration(t, r);
                v += acc * halfStep;
            }

            return new State(r, v);
        }

        private Vec3 AsteroidAcceleration(double tSec, Vec3 astPos)
        {
            Vec3 acc = new Vec3();
            for (int i = 0; i < _bodies.Length; i++)
            {
                if (i == _asteroidIdx) continue;

                State body = _eph.GetState(i, tSec);
                Vec3 diff = body.R - astPos;
                double dist2 = diff.NormSquared();
                if (dist2 < 1e6) continue;
                double dist = Math.Sqrt(dist2);
                acc += diff * (G * _bodies[i].Mass / (dist2 * dist));
            }
            return acc;
        }

        private MissionPlan BuildFinalPlan(CandidateTrajectory cand)
        {
            var burns = new List<Burn>();

            burns.Add(new Burn(0.0, cand.LaunchDV));

            double corrDays = cand.CorrectionSec / DaySeconds;
            burns.Add(new Burn(corrDays, cand.CorrectionDV));

            if (cand.CorrectionSec2 > 0.0)
            {
                double corrDays2 = cand.CorrectionSec2 / DaySeconds;
                burns.Add(new Burn(corrDays2, cand.CorrectionDV2));
            }

            double impactDays = cand.ImpactSec / DaySeconds;
            burns.Add(new Burn(impactDays, new Vec3(0, 0, 0)));

            return new MissionPlan(burns);
        }

        private MissionPlan BuildFallbackPlan()
        {
            State earth0 = _eph.GetState(_earthIdx, 0.0);
            State sun0 = _eph.GetState(_sunIdx, 0.0);
            Vec3 rDir = earth0.R - sun0.R;
            Vec3 dv = rDir / rDir.Norm() * (_budget * 0.95);

            var burns = new List<Burn>
            {
                new Burn(0.0, dv),
                new Burn(_warningDays * 0.5, new Vec3(0, 0, 0)),
                new Burn(_warningDays, new Vec3(0, 0, 0))
            };

            return new MissionPlan(burns);
        }
    }

    public static class LambertSolver
    {
        private static bool IsFinite(double x)
        {
            return !double.IsNaN(x) && !double.IsInfinity(x);
        }

        private static double Clamp(double val, double min, double max)
        {
            return val < min ? min : (val > max ? max : val);
        }

        public static bool TrySolve(Vec3 r1, Vec3 r2, double dtSec, double mu, bool longWay, out Vec3 v1, out Vec3 v2)
        {
            v1 = default;
            v2 = default;

            double r1m = r1.Norm();
            double r2m = r2.Norm();
            if (r1m <= 0.0 || r2m <= 0.0 || dtSec <= 0.0 || !IsFinite(dtSec))
                return false;

            double cosD = Vec3.Dot(r1, r2) / (r1m * r2m);
            cosD = Clamp(cosD, -1.0, 1.0);
            double dtheta = Math.Acos(cosD);
            if (longWay)
                dtheta = 2.0 * Math.PI - dtheta;

            double sinD = Math.Sin(dtheta);
            double denom = 1.0 - cosD;
            if (denom < 1e-12)
                return false;

            double A = sinD * Math.Sqrt(r1m * r2m / denom);
            if (Math.Abs(A) < 1e-12 || !IsFinite(A))
                return false;

            Func<double, double> tof = z =>
            {
                double C = StumpffC(z);
                double S = StumpffS(z);
                if (C <= 0.0 || !IsFinite(C) || !IsFinite(S))
                    return double.NaN;

                double sqrtC = Math.Sqrt(C);
                double y = r1m + r2m + A * ((z * S - 1.0) / sqrtC);
                if (y <= 0.0 || !IsFinite(y))
                    return double.NaN;

                double term1 = Math.Pow(y / C, 1.5) * S;
                double term2 = A * Math.Sqrt(y);
                double t = (term1 + term2) / Math.Sqrt(mu);
                return t;
            };

            var samples = new double[]
            {
                -100000.0, -50000.0, -20000.0, -10000.0, -5000.0, -2000.0, -1000.0, -500.0, -200.0, -100.0, -50.0, -20.0, -10.0, -5.0, -2.0, -1.0,
                0.0,
                1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0
            };

            double low = 0.0, high = 0.0;
            bool foundBracket = false;

            for (int i = samples.Length - 1; i > 0; i--)
            {
                double tRight = tof(samples[i]);
                if (double.IsNaN(tRight)) continue;

                double tLeft = tof(samples[i - 1]);
                if (double.IsNaN(tLeft))
                {
                    if (tRight - dtSec > 0.0)
                    {
                        low = samples[i - 1];
                        high = samples[i];
                        foundBracket = true;
                        break;
                    }
                }
                else
                {
                    if ((tLeft - dtSec) * (tRight - dtSec) <= 0.0)
                    {
                        low = samples[i - 1];
                        high = samples[i];
                        foundBracket = true;
                        break;
                    }
                }
            }

            if (!foundBracket)
                return false;

            double midZ = 0.0;
            for (int iter = 0; iter < 80; iter++)
            {
                midZ = 0.5 * (low + high);
                double t = tof(midZ);
                if (double.IsNaN(t) || t < dtSec)
                {
                    low = midZ;
                }
                else
                {
                    high = midZ;
                }
            }

            double zFinal = midZ;
            double Cfinal = StumpffC(zFinal);
            double Sfinal = StumpffS(zFinal);
            if (Cfinal <= 0.0 || !IsFinite(Cfinal))
                return false;

            double sqrtCfinal = Math.Sqrt(Cfinal);
            double yFinal = r1m + r2m + A * ((zFinal * Sfinal - 1.0) / sqrtCfinal);
            if (yFinal <= 0.0 || !IsFinite(yFinal))
                return false;

            double f = 1.0 - yFinal / r1m;
            double g = A * Math.Sqrt(yFinal / mu);
            if (Math.Abs(g) < 1e-12 || !IsFinite(g))
                return false;
            double gdot = 1.0 - yFinal / r2m;

            v1 = (r2 - r1 * f) / g;
            v2 = (r2 * gdot - r1) / g;
            return IsFinite(v1.X) && IsFinite(v1.Y) && IsFinite(v1.Z) &&
                   IsFinite(v2.X) && IsFinite(v2.Y) && IsFinite(v2.Z);
        }

        private static double StumpffC(double z)
        {
            double az = Math.Abs(z);
            if (az < 1e-8)
            {
                return 0.5 - z / 24.0 + z * z / 720.0 - z * z * z / 40320.0;
            }
            if (z > 0.0)
            {
                double s = Math.Sqrt(z);
                return (1.0 - Math.Cos(s)) / z;
            }
            else
            {
                double s = Math.Sqrt(-z);
                return (Math.Cosh(s) - 1.0) / (-z);
            }
        }

        private static double StumpffS(double z)
        {
            double az = Math.Abs(z);
            if (az < 1e-8)
            {
                return 1.0 / 6.0 - z / 120.0 + z * z / 5040.0 - z * z * z / 362880.0;
            }
            if (z > 0.0)
            {
                double s = Math.Sqrt(z);
                return (s - Math.Sin(s)) / (s * s * s);
            }
            else
            {
                double s = Math.Sqrt(-z);
                return (Math.Sinh(s) - s) / (s * s * s);
            }
        }
    }

    public sealed class FastScanner
    {
        private readonly Stream _stream;
        private readonly byte[] _buffer = new byte[1 << 16];
        private int _len;
        private int _ptr;

        public FastScanner(Stream stream)
        {
            _stream = stream;
        }

        public bool HasMore()
        {
            int c;
            do
            {
                c = Read();
            } while (c != -1 && char.IsWhiteSpace((char)c));

            if (c == -1)
                return false;

            _ptr--;
            return true;
        }

        public string Next()
        {
            int c;
            do
            {
                c = Read();
            } while (c != -1 && char.IsWhiteSpace((char)c));

            if (c == -1)
                return string.Empty;

            var sb = new System.Text.StringBuilder();
            while (c != -1 && !char.IsWhiteSpace((char)c))
            {
                sb.Append((char)c);
                c = Read();
            }
            return sb.ToString();
        }

        public int NextInt() => int.Parse(Next(), CultureInfo.InvariantCulture);
        public double NextDouble() => double.Parse(Next(), CultureInfo.InvariantCulture);

        private int Read()
        {
            if (_ptr >= _len)
            {
                _len = _stream.Read(_buffer, 0, _buffer.Length);
                _ptr = 0;
                if (_len <= 0)
                    return -1;
            }
            return _buffer[_ptr++];
        }
    }
