#version 450

layout(local_size_x = 256) in;

// Each boid stored as two vec4s: [pos.xyz, 0], [vel.xyz, 0]
// Boid i is at index i*2 and i*2+1
layout(set = 0, binding = 0) readonly buffer InputBoids {
    vec4 inData[];   // length = N * 2
};

layout(set = 0, binding = 1) buffer OutputBoids {
    vec4 outData[];  // length = N * 2
};

layout(set = 0, binding = 2) uniform Params {
    uvec4 counts;      // counts.x = N
    vec4 radii;        // radii.x=dt, radii.y=sepRadius, radii.z=alignRadius, radii.w=cohRadius
    vec4 weights;      // weights.x=sepWeight, weights.y=alignWeight, weights.z=cohWeight, weights.w=maxSpeed
    vec4 bounds;       // bounds.x=boundSize, bounds.y=boundForce
};

void main() {
    uint i = gl_GlobalInvocationID.x;
    uint N = counts.x;
    if (i >= N) {
        return;
    }

    vec3 pos_i = inData[i * 2].xyz;
    vec3 vel_i = inData[i * 2 + 1].xyz;

    vec3 sep_sum = vec3(0.0);
    vec3 align_sum = vec3(0.0);
    vec3 coh_sum = vec3(0.0);

    int sep_count = 0;
    int align_count = 0;
    int coh_count = 0;

    for (uint j = 0; j < N; j++) {
        if (i == j) {
            continue;
        }

        vec3 pos_j = inData[j * 2].xyz;
        vec3 vel_j = inData[j * 2 + 1].xyz;

        vec3 diff = pos_i - pos_j;
        float dist = length(diff);

        if (dist < 1e-6) {
            continue;
        }

        // Separation
        if (dist < radii.y) {
            sep_sum += diff / dist;
            sep_count++;
        }

        // Alignment
        if (dist < radii.z) {
            align_sum += vel_j;
            align_count++;
        }

        // Cohesion
        if (dist < radii.w) {
            coh_sum += pos_j;
            coh_count++;
        }
    }

    vec3 accel = vec3(0.0);

    if (sep_count > 0) {
        accel += weights.x * (sep_sum / float(sep_count));
    }

    if (align_count > 0) {
        accel += weights.y * ((align_sum / float(align_count)) - vel_i);
    }

    if (coh_count > 0) {
        accel += weights.z * ((coh_sum / float(coh_count)) - pos_i);
    }

    // Boundary force (push back toward center)
    vec3 bound_force = vec3(0.0);
    float half_bound = bounds.x * 0.5;
    float threshold = bounds.x * 0.4;

    if (abs(pos_i.x) > threshold) {
        bound_force.x = -bounds.y * (pos_i.x / half_bound);
    }
    if (abs(pos_i.y) > threshold) {
        bound_force.y = -bounds.y * (pos_i.y / half_bound);
    }
    if (abs(pos_i.z) > threshold) {
        bound_force.z = -bounds.y * (pos_i.z / half_bound);
    }

    accel += bound_force;

    // Update velocity
    float dt = radii.x;
    vec3 vel_next = vel_i + accel * dt;

    // Clamp speed to maxSpeed
    float max_speed = weights.w;
    float speed = length(vel_next);
    if (speed > max_speed && speed > 0.0) {
        vel_next = vel_next * (max_speed / speed);
    }

    // Update position
    vec3 pos_next = pos_i + vel_next * dt;

    // Write output
    outData[i * 2] = vec4(pos_next, 0.0);
    outData[i * 2 + 1] = vec4(vel_next, 0.0);
}
