; SPIR-V
; Version: 1.0
; Generator: hand-written
               OpCapability Geometry
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %worldPos %normal %color %outColor
               OpExecutionMode %main InputTrianglesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 3
               OpExecutionMode %main Invocations 1

               ; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %worldPos Location 0
               OpDecorate %normal Location 1
               OpDecorate %color Location 2
               OpDecorate %outColor Location 0

               ; Types
       %void = OpTypeVoid
       %func = OpTypeFunction %void
      %float = OpTypeFloat 32
       %uint = OpTypeInt 32 0
        %int = OpTypeInt 32 1

    %v3float = OpTypeVector %float 3
    %v4float = OpTypeVector %float 4

     %uint_6 = OpConstant %uint 6
      %int_0 = OpConstant %int 0
      %int_2 = OpConstant %int 2
      %int_4 = OpConstant %int 4

    %float_0 = OpConstant %float 0.0
    %float_0_2 = OpConstant %float 0.2
    %float_0_8 = OpConstant %float 0.8
    %float_3 = OpConstant %float 3.0
    %float_1 = OpConstant %float 1.0

    %v3_1_1_1 = OpConstantComposite %v3float %float_1 %float_1 %float_1

; Arrays from Vertex Shader
%gl_PerVertex_in = OpTypeStruct %v4float
 %arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_6
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
      %gl_in = OpVariable %ptr_in_arr_pv Input

%arr_v3float_6 = OpTypeArray %v3float %uint_6
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3float_6
   %worldPos = OpVariable %ptr_in_arr_v3 Input
     %normal = OpVariable %ptr_in_arr_v3 Input
      %color = OpVariable %ptr_in_arr_v3 Input

; Outputs
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
     %gl_out = OpVariable %ptr_out_pv Output

%ptr_out_v3 = OpTypePointer Output %v3float
   %outColor = OpVariable %ptr_out_v3 Output

; Pointers for inputs
 %ptr_in_v4 = OpTypePointer Input %v4float
 %ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_v4 = OpTypePointer Output %v4float

       %main = OpFunction %void None %func
      %entry = OpLabel

               ; 1. Load worldPos at 0, 2, 4
  %wp_ptr_0 = OpAccessChain %ptr_in_v3 %worldPos %int_0
       %p0  = OpLoad %v3float %wp_ptr_0
  %wp_ptr_2 = OpAccessChain %ptr_in_v3 %worldPos %int_2
       %p2  = OpLoad %v3float %wp_ptr_2
  %wp_ptr_4 = OpAccessChain %ptr_in_v3 %worldPos %int_4
       %p4  = OpLoad %v3float %wp_ptr_4

               ; Compute vectors: p2 - p0, p4 - p0
     %p2_p0 = OpFSub %v3float %p2 %p0
     %p4_p0 = OpFSub %v3float %p4 %p0

               ; Compute cross product manually:
     %a_x = OpCompositeExtract %float %p2_p0 0
     %a_y = OpCompositeExtract %float %p2_p0 1
     %a_z = OpCompositeExtract %float %p2_p0 2

     %b_x = OpCompositeExtract %float %p4_p0 0
     %b_y = OpCompositeExtract %float %p4_p0 1
     %b_z = OpCompositeExtract %float %p4_p0 2

               ; Component X: a.y * b.z - a.z * b.y
     %aybz = OpFMul %float %a_y %b_z
     %azby = OpFMul %float %a_z %b_y
     %cross_x = OpFSub %float %aybz %azby

               ; Component Y: a.z * b.x - a.x * b.z
     %azbx = OpFMul %float %a_z %b_x
     %axbz = OpFMul %float %a_x %b_z
     %cross_y = OpFSub %float %azbx %axbz

               ; Component Z: a.x * b.y - a.y * b.x
     %axby = OpFMul %float %a_x %b_y
     %aybx = OpFMul %float %a_y %b_x
     %cross_z = OpFSub %float %axby %aybx

               ; Construct cross product vector
     %cross = OpCompositeConstruct %v3float %cross_x %cross_y %cross_z

               ; Normalize cross product to get face normal N
         %N = OpExtInst %v3float %1 Normalize %cross

               ; Light direction: L = normalize(vec3(1, 1, 1))
         %L = OpExtInst %v3float %1 Normalize %v3_1_1_1

               ; Diffuse = max(dot(N, L), 0.0)
     %dot_NL = OpDot %float %N %L
   %diffuse = OpExtInst %float %1 FMax %dot_NL %float_0

               ; Base color = average of color[0], color[2], color[4]
   %col_ptr_0 = OpAccessChain %ptr_in_v3 %color %int_0
        %c0 = OpLoad %v3float %col_ptr_0
   %col_ptr_2 = OpAccessChain %ptr_in_v3 %color %int_2
        %c2 = OpLoad %v3float %col_ptr_2
   %col_ptr_4 = OpAccessChain %ptr_in_v3 %color %int_4
        %c4 = OpLoad %v3float %col_ptr_4

     %c_sum1 = OpFAdd %v3float %c0 %c2
     %c_sum2 = OpFAdd %v3float %c_sum1 %c4
  %float_1_3 = OpFDiv %float %float_1 %float_3
  %c_avg_factor = OpCompositeConstruct %v3float %float_1_3 %float_1_3 %float_1_3
     %base_c = OpFMul %v3float %c_sum2 %c_avg_factor

               ; Final color = base * (0.2 + 0.8 * diffuse)
    %factor1 = OpFMul %float %float_0_8 %diffuse
    %factor2 = OpFAdd %float %float_0_2 %factor1
%factor_vec = OpCompositeConstruct %v3float %factor2 %factor2 %factor2
    %final_c = OpFMul %v3float %base_c %factor_vec

               ; Emit the three vertices: 0, 2, 4
   %gp_out_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0

               ; Vertex 0:
    %gl_in_ptr_0 = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
        %pos_in_0 = OpLoad %v4float %gl_in_ptr_0
                   OpStore %gp_out_ptr %pos_in_0
                   OpStore %outColor %final_c
                   OpEmitVertex

               ; Vertex 2:
    %gl_in_ptr_2 = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
        %pos_in_2 = OpLoad %v4float %gl_in_ptr_2
                   OpStore %gp_out_ptr %pos_in_2
                   OpStore %outColor %final_c
                   OpEmitVertex

               ; Vertex 4:
    %gl_in_ptr_4 = OpAccessChain %ptr_in_v4 %gl_in %int_4 %int_0
        %pos_in_4 = OpLoad %v4float %gl_in_ptr_4
                   OpStore %gp_out_ptr %pos_in_4
                   OpStore %outColor %final_c
                   OpEmitVertex

                   OpEndPrimitive
                   OpReturn
                   OpFunctionEnd
