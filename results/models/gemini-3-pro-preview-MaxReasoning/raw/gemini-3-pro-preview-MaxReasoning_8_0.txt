use std::io::{self, BufReader, BufWriter, BufRead, Write};
use std::time::Instant;

#[derive(Copy, Clone, PartialEq, Eq)]
struct Point {
    x: u32,
    y: u32,
}

struct NybbleArray {
    data: Vec<u8>,
}

impl NybbleArray {
    fn new(size: usize) -> Self {
        let num_bytes = (size + 1) / 2;
        Self {
            data: vec![0; num_bytes],
        }
    }

    #[inline]
    unsafe fn get_unchecked(&self, index: usize) -> u8 {
        let byte_idx = index >> 1;
        let shift = (index & 1) << 2;
        (*self.data.get_unchecked(byte_idx) >> shift) & 0x0F
    }

    #[inline]
    unsafe fn set_unchecked(&mut self, index: usize, val: u8) {
        let byte_idx = index >> 1;
        let shift = (index & 1) << 2;
        let mask = !(0x0F << shift);
        let ptr = self.data.get_unchecked_mut(byte_idx);
        *ptr = (*ptr & mask) | ((val & 0x0F) << shift);
    }
}

fn reconstruct_path(
    meet_a: Point,
    meet_b: Point,
    start: Point,
    goal: Point,
    w: usize,
    grid: &NybbleArray,
) -> Vec<Point> {
    let mut path_a = Vec::new();
    let mut curr = meet_a;
    path_a.push(curr);
    while curr != start {
        let idx = (curr.y as usize) * w + (curr.x as usize);
        let val = unsafe { grid.get_unchecked(idx) };
        let dir = val - 2;
        let next = match dir {
            0 => Point { x: curr.x, y: curr.y - 1 }, // Up
            1 => Point { x: curr.x, y: curr.y + 1 }, // Down
            2 => Point { x: curr.x - 1, y: curr.y }, // Left
            3 => Point { x: curr.x + 1, y: curr.y }, // Right
            _ => unreachable!(),
        };
        curr = next;
        path_a.push(curr);
    }
    path_a.reverse();

    let mut path_b = Vec::new();
    let mut curr = meet_b;
    path_b.push(curr);
    while curr != goal {
        let idx = (curr.y as usize) * w + (curr.x as usize);
        let val = unsafe { grid.get_unchecked(idx) };
        let dir = val - 6;
        let next = match dir {
            0 => Point { x: curr.x, y: curr.y - 1 }, // Up
            1 => Point { x: curr.x, y: curr.y + 1 }, // Down
            2 => Point { x: curr.x - 1, y: curr.y }, // Left
            3 => Point { x: curr.x + 1, y: curr.y }, // Right
            _ => unreachable!(),
        };
        curr = next;
        path_b.push(curr);
    }

    path_a.extend(path_b);
    path_a
}

fn solve_maze() -> Result<(), Box<dyn std::error::Error>> {
    let stdin = io::stdin();
    let mut reader = BufReader::with_capacity(16 * 1024 * 1024, stdin.lock());
    let mut line_buf = Vec::new();

    // Read first line (H W)
    reader.read_until(b'\n', &mut line_buf)?;
    
    // Strip BOM if present
    let mut start_idx = 0;
    if line_buf.starts_with(&[0xEF, 0xBB, 0xBF]) {
        start_idx = 3;
    }
    
    let first_line = std::str::from_utf8(&line_buf[start_idx..])?;
    let mut parts = first_line.split_whitespace();
    let h: usize = parts.next().ok_or("Invalid format")?.parse()?;
    let w: usize = parts.next().ok_or("Invalid format")?.parse()?;
    line_buf.clear();

    eprintln!("H: {}, W: {}", h, w);
    let start_time = Instant::now();

    let total_cells = h * w;
    let mut grid = NybbleArray::new(total_cells);

    let mut start = Point { x: 0, y: 0 };
    let mut goal = Point { x: 0, y: 0 };

    for r in 0..h {
        reader.read_until(b'\n', &mut line_buf)?;
        let mut len = line_buf.len();
        while len > 0 && (line_buf[len - 1] == b'\n' || line_buf[len - 1] == b'\r') {
            len -= 1;
        }
        let row_bytes = &line_buf[..len];
        let limit = row_bytes.len().min(w);
        for c in 0..limit {
            let ch = row_bytes[c];
            let idx = r * w + c;
            match ch {
                b'#' => unsafe { grid.set_unchecked(idx, 1) },
                b'A' => {
                    start = Point { x: c as u32, y: r as u32 };
                    unsafe { grid.set_unchecked(idx, 2) };
                }
                b'B' => {
                    goal = Point { x: c as u32, y: r as u32 };
                    unsafe { grid.set_unchecked(idx, 6) };
                }
                _ => {}
            }
        }
        line_buf.clear();
    }

    eprintln!("Loaded in {:?}", start_time.elapsed());
    let solve_start = Instant::now();

    let stdout = io::stdout();
    let mut writer = BufWriter::with_capacity(16 * 1024 * 1024, stdout.lock());

    if start == goal {
        writeln!(writer, "1")?;
        writeln!(writer, "{} {}", start.x, start.y)?;
        writer.flush()?;
        return Ok(());
    }

    let mut front_a = Vec::new();
    let mut next_a = Vec::new();
    let mut front_b = Vec::new();
    let mut next_b = Vec::new();

    front_a.push(start);
    front_b.push(goal);

    let mut path = None;

    'outer: while !front_a.is_empty() && !front_b.is_empty() {
        if front_a.len() <= front_b.len() {
            for &curr in &front_a {
                let cx = curr.x as usize;
                let cy = curr.y as usize;

                // Up
                if cy > 0 {
                    let ny = cy - 1;
                    let n_idx = ny * w + cx;
                    let st = unsafe { grid.get_unchecked(n_idx) };
                    if st >= 6 && st <= 9 {
                        path = Some(reconstruct_path(curr, Point { x: cx as u32, y: ny as u32 }, start, goal, w, &grid));
                        break 'outer;
                    } else if st == 0 {
                        unsafe { grid.set_unchecked(n_idx, 2 + 1); }
                        next_a.push(Point { x: cx as u32, y: ny as u32 });
                    }
                }
                // Down
                if cy + 1 < h {
                    let ny = cy + 1;
                    let n_idx = ny * w + cx;
                    let st = unsafe { grid.get_unchecked(n_idx) };
                    if st >= 6 && st <= 9 {
                        path = Some(reconstruct_path(curr, Point { x: cx as u32, y: ny as u32 }, start, goal, w, &grid));
                        break 'outer;
                    } else if st == 0 {
                        unsafe { grid.set_unchecked(n_idx, 2 + 0); }
                        next_a.push(Point { x: cx as u32, y: ny as u32 });
                    }
                }
                // Left
                if cx > 0 {
                    let nx = cx - 1;
                    let n_idx = cy * w + nx;
                    let st = unsafe { grid.get_unchecked(n_idx) };
                    if st >= 6 && st <= 9 {
                        path = Some(reconstruct_path(curr, Point { x: nx as u32, y: cy as u32 }, start, goal, w, &grid));
                        break 'outer;
                    } else if st == 0 {
                        unsafe { grid.set_unchecked(n_idx, 2 + 3); }
                        next_a.push(Point { x: nx as u32, y: cy as u32 });
                    }
                }
                // Right
                if cx + 1 < w {
                    let nx = cx + 1;
                    let n_idx = cy * w + nx;
                    let st = unsafe { grid.get_unchecked(n_idx) };
                    if st >= 6 && st <= 9 {
                        path = Some(reconstruct_path(curr, Point { x: nx as u32, y: cy as u32 }, start, goal, w, &grid));
                        break 'outer;
                    } else if st == 0 {
                        unsafe { grid.set_unchecked(n_idx, 2 + 2); }
                        next_a.push(Point { x: nx as u32, y: cy as u32 });
                    }
                }
            }
            front_a.clear();
            std::mem::swap(&mut front_a, &mut next_a);
        } else {
            for &curr in &front_b {
                let cx = curr.x as usize;
                let cy = curr.y as usize;

                // Up
                if cy > 0 {
                    let ny = cy - 1;
                    let n_idx = ny * w + cx;
                    let st = unsafe { grid.get_unchecked(n_idx) };
                    if st >= 2 && st <= 5 {
                        path = Some(reconstruct_path(Point { x: cx as u32, y: ny as u32 }, curr, start, goal, w, &grid));
                        break 'outer;
                    } else if st == 0 {
                        unsafe { grid.set_unchecked(n_idx, 6 + 1); }
                        next_b.push(Point { x: cx as u32, y: ny as u32 });
                    }
                }
                // Down
                if cy + 1 < h {
                    let ny = cy + 1;
                    let n_idx = ny * w + cx;
                    let st = unsafe { grid.get_unchecked(n_idx) };
                    if st >= 2 && st <= 5 {
                        path = Some(reconstruct_path(Point { x: cx as u32, y: ny as u32 }, curr, start, goal, w, &grid));
                        break 'outer;
                    } else if st == 0 {
                        unsafe { grid.set_unchecked(n_idx, 6 + 0); }
                        next_b.push(Point { x: cx as u32, y: ny as u32 });
                    }
                }
                // Left
                if cx > 0 {
                    let nx = cx - 1;
                    let n_idx = cy * w + nx;
                    let st = unsafe { grid.get_unchecked(n_idx) };
                    if st >= 2 && st <= 5 {
                        path = Some(reconstruct_path(Point { x: nx as u32, y: cy as u32 }, curr, start, goal, w, &grid));
                        break 'outer;
                    } else if st == 0 {
                        unsafe { grid.set_unchecked(n_idx, 6 + 3); }
                        next_b.push(Point { x: nx as u32, y: cy as u32 });
                    }
                }
                // Right
                if cx + 1 < w {
                    let nx = cx + 1;
                    let n_idx = cy * w + nx;
                    let st = unsafe { grid.get_unchecked(n_idx) };
                    if st >= 2 && st <= 5 {
                        path = Some(reconstruct_path(Point { x: nx as u32, y: cy as u32 }, curr, start, goal, w, &grid));
                        break 'outer;
                    } else if st == 0 {
                        unsafe { grid.set_unchecked(n_idx, 6 + 2); }
                        next_b.push(Point { x: nx as u32, y: cy as u32 });
                    }
                }
            }
            front_b.clear();
            std::mem::swap(&mut front_b, &mut next_b);
        }
    }

    eprintln!("Solved in {:?}", solve_start.elapsed());

    if let Some(p) = path {
        writeln!(writer, "{}", p.len())?;
        for pt in p {
            writeln!(writer, "{} {}", pt.x, pt.y)?;
        }
    } else {
        // Output nothing if no path exists
    }

    writer.flush()?;
    Ok(())
}

fn main() {
    if let Err(e) = solve_maze() {
        eprintln!("Error: {}", e);
    }
}
