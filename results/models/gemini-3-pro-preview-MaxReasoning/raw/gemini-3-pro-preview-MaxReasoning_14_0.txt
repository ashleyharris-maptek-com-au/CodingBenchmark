using System;
using System.Collections.Generic;
using System.IO;
using System.Threading;
using System.Threading.Tasks;

class Program {
    static void Main() {
        try {
            // Read dimensions from standard input
            string line = Console.ReadLine();
            if (string.IsNullOrWhiteSpace(line)) {
                Console.WriteLine(0);
                return;
            }
            string[] parts = line.Split(new char[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
            int H = int.Parse(parts[0]);
            int W = int.Parse(parts[1]);
            
            // Read the board configuration
            string[] board = new string[H];
            for (int i = 0; i < H; i++) {
                string row = Console.ReadLine();
                if (row == null) {
                    board[i] = new string(' ', W);
                } else {
                    board[i] = row;
                }
            }
            
            // Initialize and run the solver
            Solver solver = new Solver(H, W, board);
            solver.Solve();
            solver.PrintResults();
        } catch (Exception) {
            // Safety fallback: if anything unexpected happens, output 0 safe cells
            Console.WriteLine(0);
        }
    }
}

/// <summary>
/// A high-performance solver for localized constraint satisfaction in Minesweeper.
/// Uses a depth-first search with pruning and node limits to solve complex local patterns.
/// Instances are designed to be completely allocation-free during search execution.
/// </summary>
class LocalCspSolver {
    public int[] Variables = new int[24];
    public int VarCount = 0;
    
    public int[] CandidateConstraints = new int[24];
    public int CandCount = 0;
    
    public int[] Constraints = new int[24];
    public int ConstCount = 0;
    
    public int[] ConstraintTargets = new int[24];
    public int[] ConstraintCurrentSums = new int[24];
    public int[] ConstraintUnassigned = new int[24];
    
    // Flat adjacency representation: VarConstraintIndices maps var index to constraint index.
    // Each variable belongs to at most 8 constraints.
    public int[] VarConstraintIndices = new int[24 * 8];
    public int[] VarConstraintCount = new int[24];
    
    public int[] Assignment = new int[24];
    public bool FoundAny = false;
    public long SafeMask = 0;
    public long MineMask = 0;
    public int NodeCount = 0;
    public bool Aborted = false;
    
    public Queue<int> Queue = new Queue<int>();
    
    public void Reset() {
        VarCount = 0;
        CandCount = 0;
        ConstCount = 0;
        FoundAny = false;
        SafeMask = 0;
        MineMask = 0;
        NodeCount = 0;
        Aborted = false;
        Queue.Clear();
    }
    
    /// <summary>
    /// Builds a local neighborhood of unknown cells and constraints starting from a seed revealed cell.
    /// Bounds the size of variables and constraints to keep the search space small and fast.
    /// </summary>
    public void BuildNeighborhood(int startRevealed, byte[] state, int H, int W) {
        Reset();
        
        CandidateConstraints[CandCount++] = startRevealed;
        Queue.Enqueue(startRevealed);
        
        // BFS to gather variables (unknown cells) up to a hard limit
        while (Queue.Count > 0 && VarCount < 20) {
            int currConst = Queue.Dequeue();
            int cr = currConst / W;
            int cc = currConst % W;
            
            for (int dr = -1; dr <= 1; dr++) {
                int nr = cr + dr;
                if (nr < 0 || nr >= H) continue;
                for (int dc = -1; dc <= 1; dc++) {
                    if (dr == 0 && dc == 0) continue;
                    int nc = cc + dc;
                    if (nc < 0 || nc >= W) continue;
                    int nb = nr * W + nc;
                    
                    if (state[nb] == ' ') {
                        int varIdx = -1;
                        for (int i = 0; i < VarCount; i++) {
                            if (Variables[i] == nb) {
                                varIdx = i;
                                break;
                            }
                        }
                        if (varIdx == -1) {
                            if (VarCount >= 20) break;
                            varIdx = VarCount;
                            Variables[VarCount++] = nb;
                            
                            // Traverse this new variable's neighbors to expand the local constraint frontier
                            int vr = nb / W;
                            int vc = nb % W;
                            for (int vdr = -1; vdr <= 1; vdr++) {
                                int vnr = vr + vdr;
                                if (vnr < 0 || vnr >= H) continue;
                                for (int vdc = -1; vdc <= 1; vdc++) {
                                    if (vdr == 0 && vdc == 0) continue;
                                    int vnc = vc + vdc;
                                    if (vnc < 0 || vnc >= W) continue;
                                    int vnb = vnr * W + vnc;
                                    char s = (char)state[vnb];
                                    if (s >= '1' && s <= '8') {
                                        bool foundCand = false;
                                        for (int k = 0; k < CandCount; k++) {
                                            if (CandidateConstraints[k] == vnb) {
                                                foundCand = true;
                                                break;
                                            }
                                        }
                                        if (!foundCand && CandCount < 24) {
                                            CandidateConstraints[CandCount++] = vnb;
                                            Queue.Enqueue(vnb);
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
                if (VarCount >= 20) break;
            }
        }
        
        // Filter candidate constraints: we only include constraints whose unknown neighbors
        // are fully contained within the selected local variables.
        for (int i = 0; i < CandCount; i++) {
            int cIdx = CandidateConstraints[i];
            bool allIn = true;
            int cr = cIdx / W;
            int cc = cIdx % W;
            for (int dr = -1; dr <= 1; dr++) {
                int nr = cr + dr;
                if (nr < 0 || nr >= H) continue;
                for (int dc = -1; dc <= 1; dc++) {
                    if (dr == 0 && dc == 0) continue;
                    int nc = cc + dc;
                    if (nc < 0 || nc >= W) continue;
                    int nb = nr * W + nc;
                    if (state[nb] == ' ') {
                        bool inVars = false;
                        for (int v = 0; v < VarCount; v++) {
                            if (Variables[v] == nb) {
                                inVars = true;
                                break;
                            }
                        }
                        if (!inVars) {
                            allIn = false;
                            break;
                        }
                    }
                }
                if (!allIn) break;
            }
            if (allIn && ConstCount < 24) {
                Constraints[ConstCount++] = cIdx;
            }
        }
    }
    
    /// <summary>
    /// Prepares the local constraint graph for the DFS.
    /// Sets targets and maps variables to constraints.
    /// </summary>
    public void PrepareDfs(int[] remainingMines, int H, int W) {
        for (int i = 0; i < VarCount; i++) {
            VarConstraintCount[i] = 0;
        }
        
        for (int j = 0; j < ConstCount; j++) {
            int cIdx = Constraints[j];
            ConstraintTargets[j] = remainingMines[cIdx];
            ConstraintCurrentSums[j] = 0;
            
            int cr = cIdx / W;
            int cc = cIdx % W;
            int unassigned = 0;
            
            for (int dr = -1; dr <= 1; dr++) {
                int nr = cr + dr;
                if (nr < 0 || nr >= H) continue;
                for (int dc = -1; dc <= 1; dc++) {
                    if (dr == 0 && dc == 0) continue;
                    int nc = cc + dc;
                    if (nc < 0 || nc >= W) continue;
                    int nb = nr * W + nc;
                    
                    for (int v = 0; v < VarCount; v++) {
                        if (Variables[v] == nb) {
                            unassigned++;
                            int idx = v * 8 + VarConstraintCount[v];
                            VarConstraintIndices[idx] = j;
                            VarConstraintCount[v]++;
                            break;
                        }
                    }
                }
            }
            ConstraintUnassigned[j] = unassigned;
        }
    }
    
    public void Solve() {
        Dfs(0);
    }
    
    /// <summary>
    /// Depth-First Search with backtracking, pruning, and node limits.
    /// </summary>
    private void Dfs(int varIdx) {
        NodeCount++;
        if (NodeCount > 2000) {
            Aborted = true;
            return;
        }
        
        if (varIdx == VarCount) {
            FoundAny = true;
            for (int i = 0; i < VarCount; i++) {
                if (Assignment[i] == 0) {
                    SafeMask |= (1L << i);
                } else {
                    MineMask |= (1L << i);
                }
            }
            return;
        }
        
        // Try value 0 (Safe)
        if (IsValid(varIdx, 0)) {
            Assign(varIdx, 0);
            Dfs(varIdx + 1);
            if (Aborted) return;
            Unassign(varIdx, 0);
        }
        
        // Try value 1 (Mine)
        if (IsValid(varIdx, 1)) {
            Assign(varIdx, 1);
            Dfs(varIdx + 1);
            if (Aborted) return;
            Unassign(varIdx, 1);
        }
    }
    
    /// <summary>
    /// Checks if assigning a value to a variable violates any constraints.
    /// </summary>
    private bool IsValid(int varIdx, int val) {
        int count = VarConstraintCount[varIdx];
        int start = varIdx * 8;
        for (int k = 0; k < count; k++) {
            int cIdx = VarConstraintIndices[start + k];
            int nextSum = ConstraintCurrentSums[cIdx] + val;
            int nextUnassigned = ConstraintUnassigned[cIdx] - 1;
            
            // Prune if current mines exceed target OR remaining spaces cannot satisfy target
            if (nextSum > ConstraintTargets[cIdx] || nextSum + nextUnassigned < ConstraintTargets[cIdx]) {
                return false;
            }
        }
        return true;
    }
    
    private void Assign(int varIdx, int val) {
        Assignment[varIdx] = val;
        int count = VarConstraintCount[varIdx];
        int start = varIdx * 8;
        for (int k = 0; k < count; k++) {
            int cIdx = VarConstraintIndices[start + k];
            ConstraintCurrentSums[cIdx] += val;
            ConstraintUnassigned[cIdx]--;
        }
    }
    
    private void Unassign(int varIdx, int val) {
        int count = VarConstraintCount[varIdx];
        int start = varIdx * 8;
        for (int k = 0; k < count; k++) {
            int cIdx = VarConstraintIndices[start + k];
            ConstraintCurrentSums[cIdx] -= val;
            ConstraintUnassigned[cIdx]++;
        }
    }
}

/// <summary>
/// Main solver orchestrator combining basic propagation, pairwise checks, and parallel Local CSP.
/// </summary>
class Solver {
    private int H;
    private int W;
    private byte[] state;
    private int[] remainingMines;
    private int[] remainingUnknowns;
    private Queue<int> activeQueue;
    private bool[] inQueue;
    private List<int> frontierUnknowns;
    
    // Pre-allocated arrays for pairwise checking (zero allocations)
    private int[] U_A = new int[8];
    private int[] U_B = new int[8];
    private int[] S = new int[8];
    private int[] A_minus_S = new int[8];
    private int[] B_minus_S = new int[8];
    private int[] rev = new int[8];
    
    private ThreadLocal<LocalCspSolver> solverPool;
    
    public Solver(int h, int w, string[] board) {
        H = h;
        W = w;
        state = new byte[H * W];
        remainingMines = new int[H * W];
        remainingUnknowns = new int[H * W];
        activeQueue = new Queue<int>();
        inQueue = new bool[H * W];
        frontierUnknowns = new List<int>();
        
        solverPool = new ThreadLocal<LocalCspSolver>(() => new LocalCspSolver());
        
        // Copy board characters to the state byte array
        for (int r = 0; r < H; r++) {
            string row = board[r];
            for (int c = 0; c < W; c++) {
                state[r * W + c] = (byte)row[c];
            }
        }
        
        // Initialize counts for remaining mines and unknowns
        for (int idx = 0; idx < H * W; idx++) {
            char s = (char)state[idx];
            if (s >= '1' && s <= '8') {
                int unknownCount = 0;
                int mineCount = 0;
                
                int r = idx / W;
                int c = idx % W;
                for (int dr = -1; dr <= 1; dr++) {
                    int nr = r + dr;
                    if (nr < 0 || nr >= H) continue;
                    for (int dc = -1; dc <= 1; dc++) {
                        if (dr == 0 && dc == 0) continue;
                        int nc = c + dc;
                        if (nc < 0 || nc >= W) continue;
                        int nb = nr * W + nc;
                        char nbs = (char)state[nb];
                        if (nbs == ' ') {
                            unknownCount++;
                        } else if (nbs == '*') {
                            mineCount++;
                        }
                    }
                }
                
                remainingMines[idx] = (s - '0') - mineCount;
                remainingUnknowns[idx] = unknownCount;
                
                // If a revealed number is immediately satisfied or trivial, queue it
                if (unknownCount > 0 && (remainingMines[idx] == 0 || remainingMines[idx] == unknownCount)) {
                    activeQueue.Enqueue(idx);
                    inQueue[idx] = true;
                }
            }
        }
        
        // Collect all frontier unknown cells to focus searches
        for (int i = 0; i < H * W; i++) {
            if (state[i] == ' ' && HasRevealedNeighbor(i)) {
                frontierUnknowns.Add(i);
            }
        }
    }
    
    private bool HasRevealedNeighbor(int idx) {
        int r = idx / W;
        int c = idx % W;
        for (int dr = -1; dr <= 1; dr++) {
            int nr = r + dr;
            if (nr < 0 || nr >= H) continue;
            for (int dc = -1; dc <= 1; dc++) {
                if (dr == 0 && dc == 0) continue;
                int nc = c + dc;
                if (nc < 0 || nc >= W) continue;
                int nb = nr * W + nc;
                char s = (char)state[nb];
                if (s >= '1' && s <= '8') return true;
            }
        }
        return false;
    }
    
    private void UpdateRevealedNeighborsOfSafe(int safeIdx) {
        int r = safeIdx / W;
        int c = safeIdx % W;
        for (int dr = -1; dr <= 1; dr++) {
            int nr = r + dr;
            if (nr < 0 || nr >= H) continue;
            for (int dc = -1; dc <= 1; dc++) {
                if (dr == 0 && dc == 0) continue;
                int nc = c + dc;
                if (nc < 0 || nc >= W) continue;
                int nb = nr * W + nc;
                char s = (char)state[nb];
                if (s >= '1' && s <= '8') {
                    remainingUnknowns[nb]--;
                    if (remainingUnknowns[nb] > 0 && (remainingMines[nb] == 0 || remainingMines[nb] == remainingUnknowns[nb])) {
                        if (!inQueue[nb]) {
                            activeQueue.Enqueue(nb);
                            inQueue[nb] = true;
                        }
                    }
                }
            }
        }
    }
    
    private void UpdateRevealedNeighborsOfMine(int mineIdx) {
        int r = mineIdx / W;
        int c = mineIdx % W;
        for (int dr = -1; dr <= 1; dr++) {
            int nr = r + dr;
            if (nr < 0 || nr >= H) continue;
            for (int dc = -1; dc <= 1; dc++) {
                if (dr == 0 && dc == 0) continue;
                int nc = c + dc;
                if (nc < 0 || nc >= W) continue;
                int nb = nr * W + nc;
                char s = (char)state[nb];
                if (s >= '1' && s <= '8') {
                    remainingUnknowns[nb]--;
                    remainingMines[nb]--;
                    if (remainingUnknowns[nb] > 0 && (remainingMines[nb] == 0 || remainingMines[nb] == remainingUnknowns[nb])) {
                        if (!inQueue[nb]) {
                            activeQueue.Enqueue(nb);
                            inQueue[nb] = true;
                        }
                    }
                }
            }
        }
    }
    
    /// <summary>
    /// Implements standard Minesweeper logic (all-safe or all-mine rules per cell).
    /// </summary>
    private bool RunBasicPropagation() {
        bool changed = false;
        while (activeQueue.Count > 0) {
            int idx = activeQueue.Dequeue();
            inQueue[idx] = false;
            
            int remMines = remainingMines[idx];
            int remUnknown = remainingUnknowns[idx];
            if (remUnknown == 0) continue;
            
            if (remMines == 0) {
                int r = idx / W;
                int c = idx % W;
                for (int dr = -1; dr <= 1; dr++) {
                    int nr = r + dr;
                    if (nr < 0 || nr >= H) continue;
                    for (int dc = -1; dc <= 1; dc++) {
                        if (dr == 0 && dc == 0) continue;
                        int nc = c + dc;
                        if (nc < 0 || nc >= W) continue;
                        int nb = nr * W + nc;
                        if (state[nb] == ' ') {
                            state[nb] = (byte)'s';
                            changed = true;
                            UpdateRevealedNeighborsOfSafe(nb);
                        }
                    }
                }
            } else if (remMines == remUnknown) {
                int r = idx / W;
                int c = idx % W;
                for (int dr = -1; dr <= 1; dr++) {
                    int nr = r + dr;
                    if (nr < 0 || nr >= H) continue;
                    for (int dc = -1; dc <= 1; dc++) {
                        if (dr == 0 && dc == 0) continue;
                        int nc = c + dc;
                        if (nc < 0 || nc >= W) continue;
                        int nb = nr * W + nc;
                        if (state[nb] == ' ') {
                            state[nb] = (byte)'m';
                            changed = true;
                            UpdateRevealedNeighborsOfMine(nb);
                        }
                    }
                }
            }
        }
        return changed;
    }
    
    private void GetUnknownNeighborsList(int idx, int[] list, out int count) {
        count = 0;
        int r = idx / W;
        int c = idx % W;
        for (int dr = -1; dr <= 1; dr++) {
            int nr = r + dr;
            if (nr < 0 || nr >= H) continue;
            for (int dc = -1; dc <= 1; dc++) {
                if (dr == 0 && dc == 0) continue;
                int nc = c + dc;
                if (nc < 0 || nc >= W) continue;
                int nb = nr * W + nc;
                if (state[nb] == ' ') {
                    list[count++] = nb;
                }
            }
        }
    }
    
    /// <summary>
    /// Performs generalized pairwise constraint reductions between overlapping constraints.
    /// </summary>
    private bool ProcessPair(int A, int B) {
        int sizeA = 0;
        GetUnknownNeighborsList(A, U_A, out sizeA);
        
        int sizeB = 0;
        GetUnknownNeighborsList(B, U_B, out sizeB);
        
        if (sizeA == 0 || sizeB == 0) return false;
        
        int sizeS = 0;
        int sizeA_minus_S = 0;
        int sizeB_minus_S = 0;
        
        for (int i = 0; i < sizeA; i++) {
            bool inB = false;
            for (int j = 0; j < sizeB; j++) {
                if (U_A[i] == U_B[j]) {
                    inB = true;
                    break;
                }
            }
            if (inB) {
                S[sizeS++] = U_A[i];
            } else {
                A_minus_S[sizeA_minus_S++] = U_A[i];
            }
        }
        
        for (int j = 0; j < sizeB; j++) {
            bool inA = false;
            for (int i = 0; i < sizeS; i++) {
                if (U_B[j] == S[i]) {
                    inA = true;
                    break;
                }
            }
            if (!inA) {
                B_minus_S[sizeB_minus_S++] = U_B[j];
            }
        }
        
        if (sizeS == 0) return false;
        
        int C_prime_A = remainingMines[A];
        int C_prime_B = remainingMines[B];
        
        bool changed = false;
        
        // Rule 1: L(A, S) == U(B, S)
        int L_A_S = Math.Max(0, C_prime_A - sizeA_minus_S);
        int U_B_S = Math.Min(C_prime_B, sizeS);
        if (L_A_S == U_B_S) {
            for (int i = 0; i < sizeA_minus_S; i++) {
                int x = A_minus_S[i];
                if (state[x] == ' ') {
                    state[x] = (byte)'m';
                    UpdateRevealedNeighborsOfMine(x);
                    changed = true;
                }
            }
            for (int i = 0; i < sizeB_minus_S; i++) {
                int x = B_minus_S[i];
                if (state[x] == ' ') {
                    state[x] = (byte)'s';
                    UpdateRevealedNeighborsOfSafe(x);
                    changed = true;
                }
            }
        }
        
        // Rule 2: L(B, S) == U(A, S)
        int L_B_S = Math.Max(0, C_prime_B - sizeB_minus_S);
        int U_A_S = Math.Min(C_prime_A, sizeS);
        if (L_B_S == U_A_S) {
            for (int i = 0; i < sizeB_minus_S; i++) {
                int x = B_minus_S[i];
                if (state[x] == ' ') {
                    state[x] = (byte)'m';
                    UpdateRevealedNeighborsOfMine(x);
                    changed = true;
                }
            }
            for (int i = 0; i < sizeA_minus_S; i++) {
                int x = A_minus_S[i];
                if (state[x] == ' ') {
                    state[x] = (byte)'s';
                    UpdateRevealedNeighborsOfSafe(x);
                    changed = true;
                }
            }
        }
        
        return changed;
    }
    
    /// <summary>
    /// Scans the frontier of unknown cells, running pairwise constraint checks.
    /// </summary>
    private bool RunPairwisePropagation() {
        bool changed = false;
        for (int k = 0; k < frontierUnknowns.Count; k++) {
            int idx = frontierUnknowns[k];
            if (state[idx] == ' ') {
                int r = idx / W;
                int c = idx % W;
                int revCount = 0;
                for (int dr = -1; dr <= 1; dr++) {
                    int nr = r + dr;
                    if (nr < 0 || nr >= H) continue;
                    for (int dc = -1; dc <= 1; dc++) {
                        if (dr == 0 && dc == 0) continue;
                        int nc = c + dc;
                        if (nc < 0 || nc >= W) continue;
                        int nb = nr * W + nc;
                        char s = (char)state[nb];
                        if (s >= '1' && s <= '8') {
                            rev[revCount++] = nb;
                        }
                    }
                }
                if (revCount >= 2) {
                    for (int i = 0; i < revCount; i++) {
                        for (int j = i + 1; j < revCount; j++) {
                            if (ProcessPair(rev[i], rev[j])) {
                                changed = true;
                            }
                        }
                    }
                }
            }
        }
        return changed;
    }
    
    private List<int> GetUnresolvedRevealedCells() {
        List<int> list = new List<int>();
        for (int i = 0; i < H * W; i++) {
            char s = (char)state[i];
            if (s >= '1' && s <= '8' && remainingUnknowns[i] > 0) {
                list.Add(i);
            }
        }
        return list;
    }
    
    /// <summary>
    /// Executes the full solver workflow, adapting to grid size and complexity.
    /// </summary>
    public void Solve() {
        // Step 1: Run fast initial basic propagation
        RunBasicPropagation();
        
        // Step 2: Loop basic + pairwise checks to solve simple patterns
        bool changed = true;
        while (changed) {
            changed = false;
            if (RunBasicPropagation()) changed = true;
            if (RunPairwisePropagation()) changed = true;
        }
        
        // Step 3: Run Local CSP solver on remaining unresolved cells
        bool globalChanged = true;
        int loopCount = 0;
        while (globalChanged && loopCount < 5) {
            globalChanged = false;
            loopCount++;
            
            List<int> unresolved = GetUnresolvedRevealedCells();
            if (unresolved.Count == 0) break;
            
            List<(int idx, bool isMine)> globalDeductions = new List<(int, bool)>();
            
            // Choose adaptive path: parallel execution on massive boards, sequential on small boards
            bool parallelCsp = (H * W > 256) && (unresolved.Count >= 20);
            
            if (parallelCsp) {
                object globalLock = new object();
                Parallel.ForEach(
                    unresolved,
                    () => new List<(int idx, bool isMine)>(),
                    (cell, loopState, localList) => {
                        var solver = solverPool.Value;
                        solver.BuildNeighborhood(cell, state, H, W);
                        solver.PrepareDfs(remainingMines, H, W);
                        solver.Solve();
                        if (solver.FoundAny && !solver.Aborted) {
                            for (int i = 0; i < solver.VarCount; i++) {
                                bool isSafe = (solver.MineMask & (1L << i)) == 0;
                                bool isMine = (solver.SafeMask & (1L << i)) == 0;
                                if (isSafe) {
                                    localList.Add((solver.Variables[i], false));
                                } else if (isMine) {
                                    localList.Add((solver.Variables[i], true));
                                }
                            }
                        }
                        return localList;
                    },
                    (localList) => {
                        lock (globalLock) {
                            globalDeductions.AddRange(localList);
                        }
                    }
                );
            } else {
                var solver = solverPool.Value;
                foreach (var cell in unresolved) {
                    solver.BuildNeighborhood(cell, state, H, W);
                    solver.PrepareDfs(remainingMines, H, W);
                    solver.Solve();
                    if (solver.FoundAny && !solver.Aborted) {
                        for (int i = 0; i < solver.VarCount; i++) {
                            bool isSafe = (solver.MineMask & (1L << i)) == 0;
                            bool isMine = (solver.SafeMask & (1L << i)) == 0;
                            if (isSafe) {
                                globalDeductions.Add((solver.Variables[i], false));
                            } else if (isMine) {
                                globalDeductions.Add((solver.Variables[i], true));
                            }
                        }
                    }
                }
            }
            
            // Apply all deductions found in this pass
            if (globalDeductions.Count > 0) {
                foreach (var deduction in globalDeductions) {
                    int cellIdx = deduction.idx;
                    bool isMine = deduction.isMine;
                    if (state[cellIdx] == ' ') {
                        if (isMine) {
                            state[cellIdx] = (byte)'m';
                            UpdateRevealedNeighborsOfMine(cellIdx);
                        } else {
                            state[cellIdx] = (byte)'s';
                            UpdateRevealedNeighborsOfSafe(cellIdx);
                        }
                        globalChanged = true;
                    }
                }
            }
            
            // If Local CSP updated any cells, re-run basic + pairwise propagation to propagate effects
            if (globalChanged) {
                bool innerChanged = true;
                while (innerChanged) {
                    innerChanged = false;
                    if (RunBasicPropagation()) innerChanged = true;
                    if (RunPairwisePropagation()) innerChanged = true;
                }
            }
        }
    }
    
    /// <summary>
    /// Outputs results using high-performance buffered streams.
    /// </summary>
    public void PrintResults() {
        List<int> safeCells = new List<int>();
        for (int i = 0; i < H * W; i++) {
            if (state[i] == 's') {
                safeCells.Add(i);
            }
        }
        
        Console.WriteLine(safeCells.Count);
        using (StreamWriter writer = new StreamWriter(Console.OpenStandardOutput(), System.Text.Encoding.ASCII, 65536)) {
            for (int i = 0; i < safeCells.Count; i++) {
                int idx = safeCells[i];
                int r = idx / W;
                int c = idx % W;
                writer.Write(c);
                writer.Write(' ');
                writer.Write(r);
                writer.WriteLine();
            }
        }
    }
