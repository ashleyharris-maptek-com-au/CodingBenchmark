using System;
using System.Collections.Generic;
using System.IO;
using System.Diagnostics;
using System.Threading;
using System.Threading.Tasks;
using System.Linq;

public struct Cuboid
{
    public double X, Y, Z;
    public double W, H, D;

    public Cuboid(double x, double y, double z, double w, double h, double d)
    {
        X = x; Y = y; Z = z;
        W = w; H = h; D = d;
    }
}

public struct Box
{
    public int Index;
    public double W, H, D;

    public Box(int index, double w, double h, double d)
    {
        Index = index;
        W = w; H = h; D = d;
    }

    public double Volume => W * H * D;
}

public struct Placement
{
    public int BoxIndex;
    public double X, Y, Z;
    public double W, H, D;

    public Placement(int boxIndex, double x, double y, double z, double w, double h, double d)
    {
        BoxIndex = boxIndex;
        X = x; Y = y; Z = z;
        W = w; H = h; D = d;
    }
}

public struct PlacementScore : IComparable<PlacementScore>
{
    public double Primary;
    public double Secondary;
    public double Tertiary;

    public PlacementScore(double p, double s, double t)
    {
        Primary = p;
        Secondary = s;
        Tertiary = t;
    }

    public int CompareTo(PlacementScore other)
    {
        int c = Primary.CompareTo(other.Primary);
        if (c != 0) return c;
        c = Secondary.CompareTo(other.Secondary);
        if (c != 0) return c;
        return Tertiary.CompareTo(other.Tertiary);
    }
}

public enum SortingRule
{
    VolumeDesc,
    VolumeAsc,
    MaxDimDesc,
    MaxDimAsc,
    MinDimDesc,
    MinDimAsc,
    SurfaceAreaDesc,
    SurfaceAreaAsc,
    MaxFaceAreaDesc,
    MaxFaceAreaAsc,
    SumDimDesc,
    SumDimAsc,
    SideRatioDesc,
    SideRatioAsc
}

public enum ScoringRule
{
    BSSF,
    BLSF,
    BAF,
    BL,
    CD
}

public class SolverConfig
{
    public SortingRule Sorting;
    public ScoringRule Scoring;
    public int MaxFreeSpaces;

    public SolverConfig(SortingRule sorting, ScoringRule scoring, int maxFreeSpaces)
    {
        Sorting = sorting;
        Scoring = scoring;
        MaxFreeSpaces = maxFreeSpaces;
    }
}

class FastScanner
{
    private readonly TextReader _reader;
    private readonly char[] _buffer = new char[65536];
    private int _bufferLen = 0;
    private int _bufferPtr = 0;

    public FastScanner(TextReader reader)
    {
        _reader = reader;
    }

    private char ReadChar()
    {
        if (_bufferPtr >= _bufferLen)
        {
            _bufferLen = _reader.Read(_buffer, 0, _buffer.Length);
            _bufferPtr = 0;
            if (_bufferLen == 0) return '\0';
        }
        return _buffer[_bufferPtr++];
    }

    public string NextToken()
    {
        var sb = new System.Text.StringBuilder();
        char ch;
        while ((ch = ReadChar()) != '\0' && char.IsWhiteSpace(ch)) { }
        if (ch == '\0') return null;
        sb.Append(ch);
        while ((ch = ReadChar()) != '\0' && !char.IsWhiteSpace(ch))
        {
            sb.Append(ch);
        }
        return sb.ToString();
    }

    public double NextDouble()
    {
        string token = NextToken();
        return token == null ? 0 : double.Parse(token, System.Globalization.CultureInfo.InvariantCulture);
    }

    public int NextInt()
    {
        string token = NextToken();
        return token == null ? 0 : int.Parse(token);
    }
}

class Program
{
    const double EPS = 1e-9;

    static List<(double W, double H, double D)> GetOrientations(double w, double h, double d)
    {
        var list = new List<(double, double, double)>(6);
        list.Add((w, h, d));
        list.Add((w, d, h));
        list.Add((h, w, d));
        list.Add((h, d, w));
        list.Add((d, w, h));
        list.Add((d, h, w));

        var unique = new List<(double, double, double)>();
        foreach (var item in list)
        {
            bool dup = false;
            foreach (var u in unique)
            {
                if (Math.Abs(u.Item1 - item.Item1) < 1e-9 &&
                    Math.Abs(u.Item2 - item.Item2) < 1e-9 &&
                    Math.Abs(u.Item3 - item.Item3) < 1e-9)
                {
                    dup = true;
                    break;
                }
            }
            if (!dup) unique.Add(item);
        }
        return unique;
    }

    static bool Overlaps(Cuboid a, Cuboid b)
    {
        return a.X < b.X + b.W - EPS && a.X + a.W > b.X + EPS &&
               a.Y < b.Y + b.H - EPS && a.Y + a.H > b.Y + EPS &&
               a.Z < b.Z + b.D - EPS && a.Z + a.D > b.Z + EPS;
    }

    static bool IsContained(Cuboid a, Cuboid b)
    {
        return a.X >= b.X - EPS && a.X + a.W <= b.X + b.W + EPS &&
               a.Y >= b.Y - EPS && a.Y + a.H <= b.Y + b.H + EPS &&
               a.Z >= b.Z - EPS && a.Z + a.D <= b.Z + b.D + EPS;
    }

    static List<Cuboid> SplitFreeSpace(Cuboid f, Cuboid b)
    {
        var list = new List<Cuboid>(6);

        if (b.X > f.X + EPS)
            list.Add(new Cuboid(f.X, f.Y, f.Z, b.X - f.X, f.H, f.D));

        if (b.X + b.W < f.X + f.W - EPS)
            list.Add(new Cuboid(b.X + b.W, f.Y, f.Z, (f.X + f.W) - (b.X + b.W), f.H, f.D));

        if (b.Y > f.Y + EPS)
            list.Add(new Cuboid(f.X, f.Y, f.Z, f.W, b.Y - f.Y, f.D));

        if (b.Y + b.H < f.Y + f.H - EPS)
            list.Add(new Cuboid(f.X, b.Y + b.H, f.Z, f.W, (f.Y + f.H) - (b.Y + b.H), f.D));

        if (b.Z > f.Z + EPS)
            list.Add(new Cuboid(f.X, f.Y, f.Z, f.W, f.H, b.Z - f.Z));

        if (b.Z + b.D < f.Z + f.D - EPS)
            list.Add(new Cuboid(f.X, f.Y, b.Z + b.D, f.W, f.H, (f.Z + f.D) - (b.Z + b.D)));

        return list;
    }

    static PlacementScore GetScore(Cuboid fs, double ow, double oh, double od, ScoringRule rule)
    {
        double leftoverW = fs.W - ow;
        double leftoverH = fs.H - oh;
        double leftoverD = fs.D - od;

        switch (rule)
        {
            case ScoringRule.BSSF:
                {
                    double primary = Math.Min(leftoverW, Math.Min(leftoverH, leftoverD));
                    double secondary = Math.Max(leftoverW, Math.Max(leftoverH, leftoverD));
                    return new PlacementScore(primary, secondary, 0);
                }
            case ScoringRule.BLSF:
                {
                    double primary = Math.Max(leftoverW, Math.Max(leftoverH, leftoverD));
                    double secondary = Math.Min(leftoverW, Math.Min(leftoverH, leftoverD));
                    return new PlacementScore(primary, secondary, 0);
                }
            case ScoringRule.BAF:
                {
                    double primary = fs.W * fs.H * fs.D - ow * oh * od;
                    double secondary = Math.Min(leftoverW, Math.Min(leftoverH, leftoverD));
                    return new PlacementScore(primary, secondary, 0);
                }
            case ScoringRule.BL:
                {
                    return new PlacementScore(fs.Z, fs.Y, fs.X);
                }
            case ScoringRule.CD:
                {
                    double primary = fs.X + ow + fs.Y + oh + fs.Z + od;
                    double secondary = fs.Z;
                    return new PlacementScore(primary, secondary, 0);
                }
            default:
                return new PlacementScore(double.MaxValue, double.MaxValue, double.MaxValue);
        }
    }

    static List<Box> SortBoxes(List<Box> boxes, SortingRule rule)
    {
        var sorted = new List<Box>(boxes);
        switch (rule)
        {
            case SortingRule.VolumeDesc:
                sorted.Sort((a, b) => b.Volume.CompareTo(a.Volume));
                break;
            case SortingRule.VolumeAsc:
                sorted.Sort((a, b) => a.Volume.CompareTo(b.Volume));
                break;
            case SortingRule.MaxDimDesc:
                sorted.Sort((a, b) => Math.Max(b.W, Math.Max(b.H, b.D)).CompareTo(Math.Max(a.W, Math.Max(a.H, a.D))));
                break;
            case SortingRule.MaxDimAsc:
                sorted.Sort((a, b) => Math.Max(a.W, Math.Max(a.H, a.D)).CompareTo(Math.Max(b.W, Math.Max(b.H, b.D))));
                break;
            case SortingRule.MinDimDesc:
                sorted.Sort((a, b) => Math.Min(b.W, Math.Min(b.H, b.D)).CompareTo(Math.Min(a.W, Math.Min(a.H, a.D))));
                break;
            case SortingRule.MinDimAsc:
                sorted.Sort((a, b) => Math.Min(a.W, Math.Min(a.H, a.D)).CompareTo(Math.Min(b.W, Math.Min(b.H, b.D))));
                break;
            case SortingRule.SurfaceAreaDesc:
                sorted.Sort((a, b) => (b.W * b.H + b.H * b.D + b.W * b.D).CompareTo(a.W * a.H + a.H * a.D + a.W * a.D));
                break;
            case SortingRule.SurfaceAreaAsc:
                sorted.Sort((a, b) => (a.W * a.H + a.H * a.D + a.W * a.D).CompareTo(b.W * b.H + b.H * b.D + b.W * b.D));
                break;
            case SortingRule.MaxFaceAreaDesc:
                sorted.Sort((a, b) =>
                {
                    double areaA = Math.Max(a.W * a.H, Math.Max(a.H * a.D, a.W * a.D));
                    double areaB = Math.Max(b.W * b.H, Math.Max(b.H * b.D, b.W * b.D));
                    return areaB.CompareTo(areaA);
                });
                break;
            case SortingRule.MaxFaceAreaAsc:
                sorted.Sort((a, b) =>
                {
                    double areaA = Math.Max(a.W * a.H, Math.Max(a.H * a.D, a.W * a.D));
                    double areaB = Math.Max(b.W * b.H, Math.Max(b.H * b.D, b.W * b.D));
                    return areaA.CompareTo(areaB);
                });
                break;
            case SortingRule.SumDimDesc:
                sorted.Sort((a, b) => (b.W + b.H + b.D).CompareTo(a.W + a.H + a.D));
                break;
            case SortingRule.SumDimAsc:
                sorted.Sort((a, b) => (a.W + a.H + a.D).CompareTo(b.W + b.H + b.D));
                break;
            case SortingRule.SideRatioDesc:
                sorted.Sort((a, b) =>
                {
                    double rA = Math.Max(a.W, Math.Max(a.H, a.D)) / Math.Max(EPS, Math.Min(a.W, Math.Min(a.H, a.D)));
                    double rB = Math.Max(b.W, Math.Max(b.H, b.D)) / Math.Max(EPS, Math.Min(b.W, Math.Min(b.H, b.D)));
                    return rB.CompareTo(rA);
                });
                break;
            case SortingRule.SideRatioAsc:
                sorted.Sort((a, b) =>
                {
                    double rA = Math.Max(a.W, Math.Max(a.H, a.D)) / Math.Max(EPS, Math.Min(a.W, Math.Min(a.H, a.D)));
                    double rB = Math.Max(b.W, Math.Max(b.H, b.D)) / Math.Max(EPS, Math.Min(b.W, Math.Min(b.H, b.D)));
                    return rA.CompareTo(rB);
                });
                break;
        }
        return sorted;
    }

    static List<Placement> Pack(List<Box> boxes, double W, double H, double D, SolverConfig config)
    {
        var sortedBoxes = SortBoxes(boxes, config.Sorting);
        var freeSpaces = new List<Cuboid> { new Cuboid(0, 0, 0, W, H, D) };
        var placements = new List<Placement>();

        foreach (var box in sortedBoxes)
        {
            var orientations = GetOrientations(box.W, box.H, box.D);
            int bestFsIndex = -1;
            double bestW = 0, bestH = 0, bestD = 0;
            PlacementScore bestScore = new PlacementScore(double.MaxValue, double.MaxValue, double.MaxValue);

            for (int i = 0; i < freeSpaces.Count; i++)
            {
                var fs = freeSpaces[i];
                foreach (var opt in orientations)
                {
                    if (opt.W <= fs.W + EPS && opt.H <= fs.H + EPS && opt.D <= fs.D + EPS)
                    {
                        var score = GetScore(fs, opt.W, opt.H, opt.D, config.Scoring);
                        if (score.CompareTo(bestScore) < 0)
                        {
                            bestScore = score;
                            bestFsIndex = i;
                            bestW = opt.W;
                            bestH = opt.H;
                            bestD = opt.D;
                        }
                    }
                }
            }

            if (bestFsIndex != -1)
            {
                var fs = freeSpaces[bestFsIndex];
                var placedBox = new Cuboid(fs.X, fs.Y, fs.Z, bestW, bestH, bestD);
                placements.Add(new Placement(box.Index, fs.X, fs.Y, fs.Z, bestW, bestH, bestD));

                // Split overlapping
                var candidates = new List<Cuboid>();
                for (int i = freeSpaces.Count - 1; i >= 0; i--)
                {
                    if (Overlaps(freeSpaces[i], placedBox))
                    {
                        var splits = SplitFreeSpace(freeSpaces[i], placedBox);
                        candidates.AddRange(splits);
                        freeSpaces.RemoveAt(i);
                    }
                }

                // Containment filter
                List<Cuboid> filteredCandidates = new List<Cuboid>();
                foreach (var c in candidates)
                {
                    bool contained = false;
                    for (int i = 0; i < freeSpaces.Count; i++)
                    {
                        if (IsContained(c, freeSpaces[i]))
                        {
                            contained = true;
                            break;
                        }
                    }
                    if (contained) continue;

                    for (int i = 0; i < filteredCandidates.Count; i++)
                    {
                        if (IsContained(c, filteredCandidates[i]))
                        {
                            contained = true;
                            break;
                        }
                    }
                    if (contained) continue;

                    for (int i = freeSpaces.Count - 1; i >= 0; i--)
                    {
                        if (IsContained(freeSpaces[i], c))
                        {
                            freeSpaces.RemoveAt(i);
                        }
                    }
                    for (int i = filteredCandidates.Count - 1; i >= 0; i--)
                    {
                        if (IsContained(filteredCandidates[i], c))
                        {
                            filteredCandidates.RemoveAt(i);
                        }
                    }

                    filteredCandidates.Add(c);
                }
                freeSpaces.AddRange(filteredCandidates);

                // Pruning
                if (freeSpaces.Count > config.MaxFreeSpaces)
                {
                    freeSpaces.Sort((a, b) => (b.W * b.H * b.D).CompareTo(a.W * a.H * a.D));
                    freeSpaces.RemoveRange(config.MaxFreeSpaces, freeSpaces.Count - config.MaxFreeSpaces);
                }
            }
        }

        return placements;
    }

    static bool ValidateSolution(List<Box> originalBoxes, double W, double H, double D, List<Placement> placements)
    {
        foreach (var p in placements)
        {
            if (p.X < -EPS || p.Y < -EPS || p.Z < -EPS ||
                p.X + p.W > W + EPS || p.Y + p.H > H + EPS || p.Z + p.D > D + EPS)
            {
                return false;
            }

            var orig = originalBoxes.First(b => b.Index == p.BoxIndex);
            var origDims = new[] { orig.W, orig.H, orig.D };
            var placedDims = new[] { p.W, p.H, p.D };
            Array.Sort(origDims);
            Array.Sort(placedDims);
            if (Math.Abs(origDims[0] - placedDims[0]) > EPS ||
                Math.Abs(origDims[1] - placedDims[1]) > EPS ||
                Math.Abs(origDims[2] - placedDims[2]) > EPS)
            {
                return false;
            }
        }

        // Only run O(N^2) overlap verification if the solution is small to avoid TLE
        if (placements.Count <= 10000)
        {
            for (int i = 0; i < placements.Count; i++)
            {
                var pi = placements[i];
                var cubA = new Cuboid(pi.X, pi.Y, pi.Z, pi.W, pi.H, pi.D);
                for (int j = i + 1; j < placements.Count; j++)
                {
                    var pj = placements[j];
                    var cubB = new Cuboid(pj.X, pj.Y, pj.Z, pj.W, pj.H, pj.D);
                    if (Overlaps(cubA, cubB))
                    {
                        return false;
                    }
                }
            }
        }

        return true;
    }

    public static void Main(string[] args)
    {
        var scanner = new FastScanner(Console.In);
        string tokenN = scanner.NextToken();
        if (tokenN == null) return;
        int N = int.Parse(tokenN);
        double W = scanner.NextDouble();
        double H = scanner.NextDouble();
        double D = scanner.NextDouble();

        var boxes = new List<Box>(N);
        for (int i = 0; i < N; i++)
        {
            double w = scanner.NextDouble();
            double h = scanner.NextDouble();
            double d = scanner.NextDouble();
            boxes.Add(new Box(i, w, h, d));
        }

        // Determine max free spaces based on N to balance quality and speed
        int maxFreeSpaces;
        if (N <= 200) maxFreeSpaces = 2500;
        else if (N <= 1000) maxFreeSpaces = 1500;
        else if (N <= 5000) maxFreeSpaces = 1000;
        else if (N <= 10000) maxFreeSpaces = 600;
        else maxFreeSpaces = 300;

        // Generate configurations (14 sorting strategies * 5 scoring strategies = 70 combinations)
        var configs = new List<SolverConfig>();
        foreach (SortingRule sorting in Enum.GetValues(typeof(SortingRule)))
        {
            foreach (ScoringRule scoring in Enum.GetValues(typeof(ScoringRule)))
            {
                configs.Add(new SolverConfig(sorting, scoring, maxFreeSpaces));
            }
        }

        var results = new List<Placement>[configs.Count];
        Parallel.For(0, configs.Count, i =>
        {
            results[i] = Pack(boxes, W, H, D, configs[i]);
        });

        int bestIndex = -1;
        int maxPacked = -1;
        for (int i = 0; i < results.Length; i++)
        {
            if (results[i].Count > maxPacked)
            {
                maxPacked = results[i].Count;
                bestIndex = i;
            }
        }

        var bestPlacements = results[bestIndex];

        // Validate final solution (safety check)
        bool isValid = ValidateSolution(boxes, W, H, D, bestPlacements);
        if (!isValid)
        {
            Console.WriteLine(0);
            return;
        }

        Console.WriteLine(bestPlacements.Count);
        foreach (var p in bestPlacements)
        {
            Console.WriteLine($"{p.BoxIndex} {p.X.ToString(System.Globalization.CultureInfo.InvariantCulture)} {p.Y.ToString(System.Globalization.CultureInfo.InvariantCulture)} {p.Z.ToString(System.Globalization.CultureInfo.InvariantCulture)}");
        }
    }
}
