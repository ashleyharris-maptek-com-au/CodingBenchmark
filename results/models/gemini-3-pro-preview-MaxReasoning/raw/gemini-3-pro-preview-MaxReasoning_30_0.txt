import sys
import time
import array
import random

def solve():
    t_start = time.time()
    
    # --- 1. Chunked Input Parsing ---
    clauses = array.array('i')
    remainder = b''
    chunk_size = 16 * 1024 * 1024  # 16 MB chunks
    num_vars = 0
    num_clauses = 0
    header_parsed = False
    
    # Read from standard input stream
    stdin_buffer = sys.stdin.buffer
    
    while True:
        chunk = stdin_buffer.read(chunk_size)
        if not chunk:
            break
        
        data = remainder + chunk
        tokens = data.split()
        
        # Determine remainder to handle boundary splits
        if not chunk.endswith((b' ', b'\n', b'\r', b'\t')):
            if tokens:
                remainder = tokens.pop()
            else:
                remainder = b''
        else:
            remainder = b''
            
        if not tokens:
            continue
            
        start_idx = 0
        if not header_parsed:
            if len(tokens) >= 2:
                num_vars = int(tokens[0])
                num_clauses = int(tokens[1])
                header_parsed = True
                start_idx = 2
            else:
                remainder = b' '.join(tokens) + b' ' + remainder
                continue
        
        if start_idx < len(tokens):
            clauses.extend(map(int, tokens[start_idx:]))
            
    if remainder:
        tokens = remainder.split()
        if tokens:
            if not header_parsed:
                if len(tokens) >= 2:
                    num_vars = int(tokens[0])
                    num_clauses = int(tokens[1])
                    header_parsed = True
                    if len(tokens) > 2:
                        clauses.extend(map(int, tokens[2:]))
            else:
                clauses.extend(map(int, tokens))
                
    if num_clauses == 0 or num_vars == 0:
        return
        
    # --- 2. Lit Counts & Majority Vote Initialization ---
    # We count positive and negative occurrences of all literals in one list
    # by using negative list indexing for negative literals.
    counts = [0] * (2 * num_vars + 2)
    for lit in clauses:
        counts[lit] += 1
        
    assignment = bytearray(num_vars + 1)
    for i in range(1, num_vars + 1):
        if counts[i] >= counts[-i]:
            assignment[i] = 1
        else:
            assignment[i] = 0
            
    # --- 3. Build CSR Occurrence Lists ---
    # offsets list of size 2*N + 3.
    # Positive literals range: offsets[i] to offsets[i+1].
    # Negative literals range: offsets[-i] to offsets[-i+1].
    offsets = [0] * (2 * num_vars + 3)
    s = 0
    for i in range(1, num_vars + 1):
        offsets[i] = s
        s += counts[i]
    offsets[num_vars + 1] = s
    for i in range(num_vars, 0, -1):
        offsets[-i] = s
        s += counts[-i]
    offsets[0] = s
    
    # Free counts to release memory
    counts = None
    
    clauses_of_lit = array.array('i', [0]) * s
    curr = list(offsets)
    for c_idx in range(num_clauses):
        idx = 3 * c_idx
        lit1 = clauses[idx]
        lit2 = clauses[idx+1]
        lit3 = clauses[idx+2]
        
        clauses_of_lit[curr[lit1]] = c_idx
        curr[lit1] += 1
        clauses_of_lit[curr[lit2]] = c_idx
        curr[lit2] += 1
        clauses_of_lit[curr[lit3]] = c_idx
        curr[lit3] += 1
        
    curr = None
    
    # --- 4. Evaluate Initial Clause Satisfaction ---
    num_true = bytearray(num_clauses)
    unsat_clauses = []
    unsat_idx_in_list = array.array('i', [-1]) * num_clauses
    
    for c_idx in range(num_clauses):
        idx = 3 * c_idx
        l1 = clauses[idx]
        l2 = clauses[idx+1]
        l3 = clauses[idx+2]
        
        val1 = assignment[l1] if l1 > 0 else (1 - assignment[-l1])
        val2 = assignment[l2] if l2 > 0 else (1 - assignment[-l2])
        val3 = assignment[l3] if l3 > 0 else (1 - assignment[-l3])
        
        nt = val1 + val2 + val3
        num_true[c_idx] = nt
        if nt == 0:
            unsat_idx_in_list[c_idx] = len(unsat_clauses)
            unsat_clauses.append(c_idx)
            
    # --- 5. WalkSAT Solver ---
    flips = 0
    max_flips = 20 * num_vars
    p = 0.5
    
    unsat_clauses_append = unsat_clauses.append
    unsat_clauses_pop = unsat_clauses.pop
    
    while unsat_clauses and flips < max_flips:
        flips += 1
        
        # Periodic timeout check (safe limit of 280 seconds)
        if flips % 500000 == 0:
            if time.time() - t_start > 280.0:
                break
                
        c_idx = random.choice(unsat_clauses)
        idx = 3 * c_idx
        l1 = clauses[idx]
        l2 = clauses[idx+1]
        l3 = clauses[idx+2]
        
        v1 = abs(l1)
        v2 = abs(l2)
        v3 = abs(l3)
        
        # Cyclic permutation of checking order to eliminate bias
        shift = flips % 3
        if shift == 0:
            first, second, third = v1, v2, v3
        elif shift == 1:
            first, second, third = v2, v3, v1
        else:
            first, second, third = v3, v1, v2
            
        # Compute breaks for first (no conditional wraps needed)
        b1 = 0
        if assignment[first]:
            for o_idx in range(offsets[first], offsets[first+1]):
                if num_true[clauses_of_lit[o_idx]] == 1:
                    b1 += 1
        else:
            for o_idx in range(offsets[-first], offsets[-first+1]):
                if num_true[clauses_of_lit[o_idx]] == 1:
                    b1 += 1
                    
        if b1 == 0:
            v_flip = first
        else:
            # Compute breaks for second
            b2 = 0
            if assignment[second]:
                for o_idx in range(offsets[second], offsets[second+1]):
                    if num_true[clauses_of_lit[o_idx]] == 1:
                        b2 += 1
            else:
                for o_idx in range(offsets[-second], offsets[-second+1]):
                    if num_true[clauses_of_lit[o_idx]] == 1:
                        b2 += 1
                        
            if b2 == 0:
                v_flip = second
            else:
                # Compute breaks for third
                b3 = 0
                if assignment[third]:
                    for o_idx in range(offsets[third], offsets[third+1]):
                        if num_true[clauses_of_lit[o_idx]] == 1:
                            b3 += 1
                else:
                    for o_idx in range(offsets[-third], offsets[-third+1]):
                        if num_true[clauses_of_lit[o_idx]] == 1:
                            b3 += 1
                            
                if b3 == 0:
                    v_flip = third
                else:
                    if random.random() < p:
                        v_flip = random.choice((v1, v2, v3))
                    else:
                        min_b = min(b1, b2, b3)
                        if min_b == b1:
                            v_flip = first
                        elif min_b == b2:
                            v_flip = second
                        else:
                            v_flip = third
                            
        # Perform flip of v_flip
        old_val = assignment[v_flip]
        new_val = 1 - old_val
        assignment[v_flip] = new_val
        
        # Update satisfied state arrays
        if old_val == 1:
            for o_idx in range(offsets[v_flip], offsets[v_flip+1]):
                c = clauses_of_lit[o_idx]
                nt = num_true[c]
                num_true[c] = nt - 1
                if nt == 1:
                    unsat_idx_in_list[c] = len(unsat_clauses)
                    unsat_clauses_append(c)
                    
            for o_idx in range(offsets[-v_flip], offsets[-v_flip+1]):
                c = clauses_of_lit[o_idx]
                nt = num_true[c]
                num_true[c] = nt + 1
                if nt == 0:
                    last_c = unsat_clauses[-1]
                    pos_in_list = unsat_idx_in_list[c]
                    unsat_clauses[pos_in_list] = last_c
                    unsat_idx_in_list[last_c] = pos_in_list
                    unsat_clauses_pop()
                    unsat_idx_in_list[c] = -1
        else:
            for o_idx in range(offsets[-v_flip], offsets[-v_flip+1]):
                c = clauses_of_lit[o_idx]
                nt = num_true[c]
                num_true[c] = nt - 1
                if nt == 1:
                    unsat_idx_in_list[c] = len(unsat_clauses)
                    unsat_clauses_append(c)
                    
            for o_idx in range(offsets[v_flip], offsets[v_flip+1]):
                c = clauses_of_lit[o_idx]
                nt = num_true[c]
                num_true[c] = nt + 1
                if nt == 0:
                    last_c = unsat_clauses[-1]
                    pos_in_list = unsat_idx_in_list[c]
                    unsat_clauses[pos_in_list] = last_c
                    unsat_idx_in_list[last_c] = pos_in_list
                    unsat_clauses_pop()
                    unsat_idx_in_list[c] = -1
                    
    # --- 6. Output Assignment ---
    if not unsat_clauses:
        # Loop-free C-level ASCII translation & slice assignment
        table = bytes.maketrans(b'\x00\x01', b'01')
        translated = assignment[1:].translate(table)
        
        out = bytearray(2 * num_vars - 1)
        out[::2] = translated
        out[1::2] = b' ' * (num_vars - 1)
        
        sys.stdout.buffer.write(b"SAT\n")
        sys.stdout.buffer.write(out + b"\n")
    else:
        sys.stdout.write("UNSAT\n")

if __name__ == "__main__":
    solve()
