#version 450

layout(location = 0) in vec3 worldPos;
layout(location = 1) in vec3 normal;
layout(location = 2) in vec2 uv;
layout(location = 3) in vec3 tangent;
layout(location = 4) in vec3 color;

layout(set = 0, binding = 0) uniform UBO {
    mat4 model;
    mat4 view;
    mat4 projection;
    vec4 lightPos;
    vec4 cameraPos;
    vec4 params;
} ubo;

layout(location = 0) out vec4 outColor;

void main() {
    // Re-normalize the interpolated normal
    vec3 N = normalize(normal);
    
    // Calculate light direction in world space
    vec3 L = normalize(ubo.lightPos.xyz - worldPos);
    
    // Lambertian diffuse
    float diffuse = max(dot(N, L), 0.0);
    
    // Shading factor
    float factor = 0.3 + 0.7 * diffuse;
    
    // 6x6 UV polka-dot pattern
    vec2 cellUV = fract(uv * 6.0);
    float dist = distance(cellUV, vec2(0.5));
    
    // Radius of the dots is 0.3 in cell space
    vec3 baseColor = (dist < 0.3) ? vec3(1.0, 0.8, 0.0) : vec3(0.1, 0.1, 0.3);
    
    // Final output color with alpha 1.0
    outColor = vec4(baseColor * factor, 1.0);
}
