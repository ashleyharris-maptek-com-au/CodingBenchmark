using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Text;
using System.Threading;
using System.Threading.Tasks;

class Program
{
    public struct Cut
    {
        public int Index;
        public int Length;
        public double SortKey;
    }

    public struct Placement
    {
        public int TargetBin;
        public int CutSortedIndex;
    }

    static int N;
    static int L;
    static int M;
    static Cut[] initialCuts;
    static int[] bestGlobalCutBin;
    static int bestGlobalBinsCount = int.MaxValue;
    static object globalLock = new object();

    public class SolverThreadState
    {
        public Cut[] LocalCuts;
        public int[] LocalTree;
        public int[] LocalBinHead;
        public int[] LocalCutNext;
        public int[] LocalBinCapacityRemaining;
        public int[] LocalBinTotalWeight;
        public int[] LocalActiveBins;
        public int[] LocalCutBin;
        public int LocalBestBinsCount;
        public int[] LocalBestCutBin;

        public SolverThreadState(Cut[] initialCuts)
        {
            LocalCuts = new Cut[N];
            Array.Copy(initialCuts, LocalCuts, N);

            LocalTree = new int[2 * M];
            LocalBinHead = new int[N];
            LocalCutNext = new int[N];
            LocalBinCapacityRemaining = new int[N];
            LocalBinTotalWeight = new int[N];
            LocalActiveBins = new int[N];
            LocalCutBin = new int[N];
            LocalBestCutBin = new int[N];
            LocalBestBinsCount = int.MaxValue;
        }

        public void SolveIteration(Random rand, double alpha, int lowerBound)
        {
            for (int i = 0; i < N; i++)
            {
                LocalCuts[i].SortKey = -LocalCuts[i].Length + rand.NextDouble() * alpha;
            }

            Array.Sort(LocalCuts, (a, b) => a.SortKey.CompareTo(b.SortKey));

            // Reset tree
            for (int i = 0; i < M; i++) LocalTree[M + i] = L;
            for (int i = M - 1; i >= 1; i--) LocalTree[i] = L;

            // Reset bins
            for (int i = 0; i < N; i++)
            {
                LocalBinHead[i] = -1;
                LocalCutNext[i] = -1;
                LocalBinCapacityRemaining[i] = L;
                LocalBinTotalWeight[i] = 0;
            }

            // FFD
            for (int i = 0; i < N; i++)
            {
                int w = LocalCuts[i].Length;
                int binIdx = FindFirstBin(w);
                if (binIdx == -1) continue;

                LocalCutNext[i] = LocalBinHead[binIdx];
                LocalBinHead[binIdx] = i;
                LocalBinCapacityRemaining[binIdx] -= w;
                LocalBinTotalWeight[binIdx] += w;
                UpdateTree(binIdx, LocalBinCapacityRemaining[binIdx]);
            }

            // Inactive bin zeroing
            for (int i = 0; i < M; i++)
            {
                if (i >= N || LocalBinHead[i] == -1)
                {
                    UpdateTree(i, 0);
                }
            }

            // Run local search
            while (true)
            {
                bool merged = RunBinMergingPass();
                if (!merged) break;
            }

            int binsUsed = CountActiveBins();
            if (binsUsed < LocalBestBinsCount)
            {
                LocalBestBinsCount = binsUsed;
                for (int b = 0; b < N; b++)
                {
                    int curr = LocalBinHead[b];
                    while (curr != -1)
                    {
                        LocalCutBin[LocalCuts[curr].Index] = b;
                        curr = LocalCutNext[curr];
                    }
                }
                Array.Copy(LocalCutBin, LocalBestCutBin, N);
            }
        }

        private int CountActiveBins()
        {
            int count = 0;
            for (int i = 0; i < N; i++)
            {
                if (LocalBinHead[i] != -1) count++;
            }
            return count;
        }

        private void SortBinsDescending(int[] arr, int left, int right)
        {
            if (left >= right) return;
            int pivot = LocalBinCapacityRemaining[arr[left + (right - left) / 2]];
            int i = left;
            int j = right;
            while (i <= j)
            {
                while (LocalBinCapacityRemaining[arr[i]] > pivot) i++;
                while (LocalBinCapacityRemaining[arr[j]] < pivot) j--;
                if (i <= j)
                {
                    int temp = arr[i];
                    arr[i] = arr[j];
                    arr[j] = temp;
                    i++;
                    j--;
                }
            }
            SortBinsDescending(arr, left, j);
            SortBinsDescending(arr, i, right);
        }

        private bool RunBinMergingPass()
        {
            int numActive = 0;
            for (int i = 0; i < N; i++)
            {
                if (LocalBinHead[i] != -1)
                {
                    LocalActiveBins[numActive++] = i;
                }
            }

            if (numActive <= 1) return false;

            SortBinsDescending(LocalActiveBins, 0, numActive - 1);

            bool anyMerged = false;
            List<int> tempCuts = new List<int>();
            List<Placement> placements = new List<Placement>();

            for (int i = 0; i < numActive; i++)
            {
                int b = LocalActiveBins[i];
                if (LocalBinHead[b] == -1) continue;

                int origCap = LocalBinCapacityRemaining[b];

                tempCuts.Clear();
                int curr = LocalBinHead[b];
                while (curr != -1)
                {
                    tempCuts.Add(curr);
                    curr = LocalCutNext[curr];
                }

                UpdateTree(b, 0);

                placements.Clear();
                bool success = true;

                for (int k = 0; k < tempCuts.Count; k++)
                {
                    int cIdx = tempCuts[k];
                    int length = LocalCuts[cIdx].Length;
                    int targetBin = FindFirstBin(length);
                    if (targetBin == -1)
                    {
                        success = false;
                        break;
                    }

                    LocalBinCapacityRemaining[targetBin] -= length;
                    UpdateTree(targetBin, LocalBinCapacityRemaining[targetBin]);
                    placements.Add(new Placement { TargetBin = targetBin, CutSortedIndex = cIdx });
                }

                if (success)
                {
                    for (int k = 0; k < placements.Count; k++)
                    {
                        int targetBin = placements[k].TargetBin;
                        int cIdx = placements[k].CutSortedIndex;

                        LocalCutNext[cIdx] = LocalBinHead[targetBin];
                        LocalBinHead[targetBin] = cIdx;
                        LocalBinTotalWeight[targetBin] += LocalCuts[cIdx].Length;
                    }

                    LocalBinHead[b] = -1;
                    LocalBinCapacityRemaining[b] = 0;
                    LocalBinTotalWeight[b] = 0;
                    UpdateTree(b, 0);

                    anyMerged = true;
                }
                else
                {
                    for (int k = 0; k < placements.Count; k++)
                    {
                        int targetBin = placements[k].TargetBin;
                        int cIdx = placements[k].CutSortedIndex;

                        LocalBinCapacityRemaining[targetBin] += LocalCuts[cIdx].Length;
                        UpdateTree(targetBin, LocalBinCapacityRemaining[targetBin]);
                    }

                    LocalBinCapacityRemaining[b] = origCap;
                    UpdateTree(b, origCap);
                }
            }

            return anyMerged;
        }

        private int FindFirstBin(int w)
        {
            if (LocalTree[1] < w) return -1;
            int node = 1;
            while (node < M)
            {
                if (LocalTree[node * 2] >= w)
                {
                    node = node * 2;
                }
                else
                {
                    node = node * 2 + 1;
                }
            }
            return node - M;
        }

        private void UpdateTree(int binIdx, int newCap)
        {
            int node = M + binIdx;
            LocalTree[node] = newCap;
            node >>= 1;
            while (node >= 1)
            {
                LocalTree[node] = Math.Max(LocalTree[node * 2], LocalTree[node * 2 + 1]);
                node >>= 1;
            }
        }
    }

    public static class FastParser
    {
        private static readonly byte[] buffer = new byte[1024 * 1024];
        private static int bufferCount = 0;
        private static int bufferPointer = 0;
        private static Stream stream = Console.OpenStandardInput();

        public static int ReadInt()
        {
            while (true)
            {
                if (bufferPointer >= bufferCount)
                {
                    bufferCount = stream.Read(buffer, 0, buffer.Length);
                    bufferPointer = 0;
                    if (bufferCount == 0) return -1; // EOF
                }
                byte b = buffer[bufferPointer++];
                if (b >= '0' && b <= '9')
                {
                    int val = b - '0';
                    while (true)
                    {
                        if (bufferPointer >= bufferCount)
                        {
                            bufferCount = stream.Read(buffer, 0, buffer.Length);
                            bufferPointer = 0;
                            if (bufferCount == 0) return val;
                        }
                        byte next = buffer[bufferPointer];
                        if (next >= '0' && next <= '9')
                        {
                            val = val * 10 + (next - '0');
                            bufferPointer++;
                        }
                        else
                        {
                            bufferPointer++; // Consume non-digit
                            return val;
                        }
                    }
                }
            }
        }
    }

    static void Main(string[] args)
    {
        N = FastParser.ReadInt();
        if (N <= 0)
        {
            Console.WriteLine(0);
            return;
        }
        L = FastParser.ReadInt();

        initialCuts = new Cut[N];
        long sumLengths = 0;
        for (int i = 0; i < N; i++)
        {
            int len = FastParser.ReadInt();
            initialCuts[i] = new Cut { Index = i, Length = len };
            sumLengths += len;
        }

        int lowerBound = (int)((sumLengths + L - 1) / L);

        M = 1;
        while (M < N) M <<= 1;
        
        bestGlobalCutBin = new int[N];

        Stopwatch sw = Stopwatch.StartNew();

        int maxThreads = Environment.ProcessorCount;
        int numIterations = 1;
        if (N < 1000) numIterations = 1000;
        else if (N < 10000) numIterations = 200;
        else if (N < 100000) numIterations = 40;
        else numIterations = 12;

        bool solvedOptimal = false;

        Parallel.For(0, numIterations, new ParallelOptions { MaxDegreeOfParallelism = maxThreads }, (iter, state) =>
        {
            if (Volatile.Read(ref solvedOptimal))
            {
                state.Stop();
                return;
            }

            if (sw.ElapsedMilliseconds > 240000)
            {
                state.Stop();
                return;
            }

            Random threadRand = new Random(42 + iter);
            double alpha = (iter == 0) ? 0.0 : (iter * 1.5);

            SolverThreadState localSolver = new SolverThreadState(initialCuts);
            localSolver.SolveIteration(threadRand, alpha, lowerBound);

            lock (globalLock)
            {
                if (localSolver.LocalBestBinsCount < bestGlobalBinsCount)
                {
                    bestGlobalBinsCount = localSolver.LocalBestBinsCount;
                    Array.Copy(localSolver.LocalBestCutBin, bestGlobalCutBin, N);

                    if (bestGlobalBinsCount <= lowerBound)
                    {
                        Volatile.Write(ref solvedOptimal, true);
                        state.Stop();
                    }
                }
            }
        });

        PrintOutput();
    }

    static void PrintOutput()
    {
        int[] binMap = new int[N];
        for (int i = 0; i < N; i++) binMap[i] = -1;

        int uniqueBins = 0;
        for (int i = 0; i < N; i++)
        {
            int b = bestGlobalCutBin[i];
            if (binMap[b] == -1)
            {
                binMap[b] = uniqueBins++;
            }
        }

        int[] head = new int[uniqueBins];
        int[] next = new int[N];
        for (int i = 0; i < uniqueBins; i++) head[i] = -1;

        for (int i = 0; i < N; i++)
        {
            int remappedBin = binMap[bestGlobalCutBin[i]];
            next[i] = head[remappedBin];
            head[remappedBin] = i;
        }

        StreamWriter writer = new StreamWriter(Console.OpenStandardOutput(), Encoding.ASCII, 65536);
        writer.WriteLine(uniqueBins);
        for (int b = 0; b < uniqueBins; b++)
        {
            int curr = head[b];
            bool first = true;
            while (curr != -1)
            {
                if (!first) writer.Write(' ');
                writer.Write(curr);
                first = false;
                curr = next[curr];
            }
            writer.WriteLine();
        }
        writer.Flush();
    }
}
