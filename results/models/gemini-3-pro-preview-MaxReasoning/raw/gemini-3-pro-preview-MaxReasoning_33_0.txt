using System;
using System.IO;
using System.Diagnostics;
using System.Collections.Generic;

class Solver {
    // Fast PRNG
    class FastRandom {
        private uint x = 123456789;
        private uint y = 362436069;
        private uint z = 521288629;
        private uint w = 88675123;

        public FastRandom(int seed) {
            x = (uint)seed;
            y = x ^ 362436069;
            z = y ^ 521288629;
            w = z ^ 88675123;
        }

        public uint NextUInt() {
            uint t = x ^ (x << 11);
            x = y; y = z; z = w;
            return w = w ^ (w >> 19) ^ t ^ (t >> 8);
        }

        public int Next(int max) {
            return (int)(((ulong)NextUInt() * (ulong)max) >> 32);
        }

        public double NextDouble() {
            return (double)NextUInt() / (double)uint.MaxValue;
        }
    }

    // Fast I/O Reader
    class FastReader {
        private Stream stream;
        private byte[] buffer = new byte[65536];
        private int bufferPointer = 0;
        private int bytesRead = 0;

        public FastReader() {
            stream = Console.OpenStandardInput();
        }

        private byte Read() {
            if (bufferPointer == bytesRead) {
                FillBuffer();
            }
            return bytesRead == 0 ? (byte)0 : buffer[bufferPointer++];
        }

        private void FillBuffer() {
            bufferPointer = 0;
            bytesRead = stream.Read(buffer, 0, buffer.Length);
        }

        public int NextInt() {
            int val = 0;
            byte c = Read();
            while (c <= 32) {
                if (c == 0) return -1; // EOF
                c = Read();
            }
            bool neg = false;
            if (c == '-') {
                neg = true;
                c = Read();
            }
            while (c >= '0' && c <= '9') {
                val = val * 10 + (c - '0');
                c = Read();
            }
            return neg ? -val : val;
        }
    }

    static long Cost(int[] p, int[][] F, int[][] D, int N) {
        long cost = 0;
        for (int i = 0; i < N; i++) {
            int pi = p[i];
            int[] Fi = F[i];
            for (int j = 0; j < N; j++) {
                cost += (long)Fi[j] * D[pi][p[j]];
            }
        }
        return cost;
    }

    static long Delta(int r, int s, int[] p, int[][] F, int[][] D, int[][] F_col, int[][] D_col, int N) {
        int u = p[r];
        int v = p[s];

        int[] Fr = F[r];
        int[] Fs = F[s];
        int[] F_col_r = F_col[r];
        int[] F_col_s = F_col[s];

        int[] Du = D[u];
        int[] Dv = D[v];
        int[] D_col_u = D_col[u];
        int[] D_col_v = D_col[v];

        long delta = 0;
        for (int k = 0; k < N; k++) {
            if (k == r || k == s) continue;
            int pk = p[k];

            long diff1 = ((long)Fr[k] - Fs[k]) * ((long)Dv[pk] - Du[pk]);
            long diff2 = ((long)F_col_r[k] - F_col_s[k]) * ((long)D_col_v[pk] - D_col_u[pk]);

            delta += diff1 + diff2;
        }

        delta += ((long)F[r][r] - F[s][s]) * ((long)D[v][v] - D[u][u]);
        delta += ((long)F[r][s] - F[s][r]) * ((long)D[v][u] - D[u][v]);

        return delta;
    }

    static double GetInitialTemperature(int[] p, int[][] F, int[][] D, int[][] F_col, int[][] D_col, int N, FastRandom rand) {
        long sum = 0;
        int count = 0;
        int[] tempP = (int[])p.Clone();
        for (int i = 0; i < 100; i++) {
            int r = rand.Next(N);
            int s = rand.Next(N);
            if (r == s) continue;
            long d = Delta(r, s, tempP, F, D, F_col, D_col, N);
            sum += Math.Abs(d);
            count++;
            
            int tmp = tempP[r];
            tempP[r] = tempP[s];
            tempP[s] = tmp;
        }
        double avg = count > 0 ? (double)sum / count : 1.0;
        if (avg < 1e-5) avg = 1.0;
        return avg;
    }

    static int[] GetSortHeuristicPermutation(int[][] F, int[][] D, int N) {
        long[] f_sum = new long[N];
        long[] d_sum = new long[N];
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < N; j++) {
                f_sum[i] += F[i][j] + F[j][i];
                d_sum[i] += D[i][j] + D[j][i];
            }
        }

        int[] fac = new int[N];
        int[] loc = new int[N];
        for (int i = 0; i < N; i++) {
            fac[i] = i;
            loc[i] = i;
        }

        Array.Sort(fac, (a, b) => f_sum[b].CompareTo(f_sum[a]));
        Array.Sort(loc, (a, b) => d_sum[a].CompareTo(d_sum[b]));

        int[] p = new int[N];
        for (int i = 0; i < N; i++) {
            p[fac[i]] = loc[i];
        }
        return p;
    }

    static int[] GetRandomPermutation(int N, FastRandom rand) {
        int[] p = new int[N];
        for (int i = 0; i < N; i++) {
            p[i] = i;
        }
        for (int i = N - 1; i > 0; i--) {
            int j = rand.Next(i + 1);
            int tmp = p[i];
            p[i] = p[j];
            p[j] = tmp;
        }
        return p;
    }

    static int[] Perturb(int[] p, int K, FastRandom rand) {
        int[] next = (int[])p.Clone();
        int N = p.Length;
        for (int i = 0; i < K; i++) {
            int r = rand.Next(N);
            int s = rand.Next(N);
            if (r != s) {
                int tmp = next[r];
                next[r] = next[s];
                next[s] = tmp;
            }
        }
        return next;
    }

    static bool HillClimb(int[] p, ref long currentCost, int[][] F, int[][] D, int[][] F_col, int[][] D_col, int N) {
        bool improved = false;
        bool localOptimal = false;
        while (!localOptimal) {
            localOptimal = true;
            for (int r = 0; r < N - 1; r++) {
                for (int s = r + 1; s < N; s++) {
                    long delta = Delta(r, s, p, F, D, F_col, D_col, N);
                    if (delta < 0) {
                        int tmp = p[r];
                        p[r] = p[s];
                        p[s] = tmp;
                        currentCost += delta;
                        localOptimal = false;
                        improved = true;
                    }
                }
            }
        }
        return improved;
    }

    static void RandomizedHillClimb(int[] p, ref long currentCost, int[][] F, int[][] D, int[][] F_col, int[][] D_col, int N, int maxFails, FastRandom rand) {
        int fails = 0;
        while (fails < maxFails) {
            int r = rand.Next(N);
            int s = rand.Next(N);
            if (r == s) continue;
            long delta = Delta(r, s, p, F, D, F_col, D_col, N);
            if (delta < 0) {
                int tmp = p[r];
                p[r] = p[s];
                p[s] = tmp;
                currentCost += delta;
                fails = 0;
            } else {
                fails++;
            }
        }
    }

    static long SolveSA(int[] p, int[] localBestP, ref long localBestCost, int[][] F, int[][] D, int[][] F_col, int[][] D_col, int N, double T0, double alpha, int maxSteps, FastRandom rand, Stopwatch stopwatch, long timeLimitMs) {
        long currentCost = Cost(p, F, D, N);
        localBestCost = currentCost;
        Array.Copy(p, localBestP, N);

        double T = T0;
        int blockSize = 10000;

        for (int step = 0; step < maxSteps; ) {
            int limit = Math.Min(maxSteps - step, blockSize);
            for (int i = 0; i < limit; i++) {
                int r = rand.Next(N);
                int s = rand.Next(N);
                if (r == s) continue;

                long delta = Delta(r, s, p, F, D, F_col, D_col, N);
                if (delta < 0) {
                    int tmp = p[r];
                    p[r] = p[s];
                    p[s] = tmp;
                    currentCost += delta;

                    if (currentCost < localBestCost) {
                        localBestCost = currentCost;
                        Array.Copy(p, localBestP, N);
                    }
                } else if (T > 1e-9) {
                    double arg = -delta / T;
                    if (arg > -20.0 && rand.NextDouble() < Math.Exp(arg)) {
                        int tmp = p[r];
                        p[r] = p[s];
                        p[s] = tmp;
                        currentCost += delta;
                    }
                }
                T *= alpha;
            }
            step += limit;

            if (stopwatch.ElapsedMilliseconds > timeLimitMs) {
                break;
            }
        }
        return currentCost;
    }

    static void Main() {
        FastReader reader = new FastReader();
        int N = reader.NextInt();
        if (N <= 0) return;

        int[][] F = new int[N][];
        for (int i = 0; i < N; i++) {
            F[i] = new int[N];
            for (int j = 0; j < N; j++) {
                F[i][j] = reader.NextInt();
            }
        }

        int[][] D = new int[N][];
        for (int i = 0; i < N; i++) {
            D[i] = new int[N];
            for (int j = 0; j < N; j++) {
                D[i][j] = reader.NextInt();
            }
        }

        // Precompute columns
        int[][] F_col = new int[N][];
        int[][] D_col = new int[N][];
        for (int i = 0; i < N; i++) {
            F_col[i] = new int[N];
            D_col[i] = new int[N];
        }
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < N; j++) {
                F_col[j][i] = F[i][j];
                D_col[j][i] = D[i][j];
            }
        }

        Stopwatch stopwatch = Stopwatch.StartNew();
        FastRandom rand = new FastRandom(42);

        // Parameters based on N
        long timeLimitMs;
        int maxNoImprovementRestarts;
        int maxRestarts;
        int saSteps;

        if (N <= 40) {
            timeLimitMs = 5000;
            maxNoImprovementRestarts = 200;
            maxRestarts = 1000;
            saSteps = N * 5000;
        } else if (N <= 100) {
            timeLimitMs = 15000;
            maxNoImprovementRestarts = 150;
            maxRestarts = 1000;
            saSteps = N * 5000;
        } else if (N <= 300) {
            timeLimitMs = 60000;
            maxNoImprovementRestarts = 80;
            maxRestarts = 500;
            saSteps = N * 5000;
        } else if (N <= 800) {
            timeLimitMs = 180000;
            maxNoImprovementRestarts = 30;
            maxRestarts = 200;
            saSteps = N * 4000;
        } else {
            timeLimitMs = 280000;
            maxNoImprovementRestarts = 10;
            maxRestarts = 100;
            saSteps = N * 2000;
        }

        // Initial Permutation: Sort Heuristic
        int[] bestP = GetSortHeuristicPermutation(F, D, N);
        long bestCost = Cost(bestP, F, D, N);

        // Initial Hill Climbing
        long currentCost = bestCost;
        int[] currentP = (int[])bestP.Clone();
        if (N <= 400) {
            HillClimb(currentP, ref currentCost, F, D, F_col, D_col, N);
        } else {
            RandomizedHillClimb(currentP, ref currentCost, F, D, F_col, D_col, N, N * 20, rand);
        }

        if (currentCost < bestCost) {
            bestCost = currentCost;
            Array.Copy(currentP, bestP, N);
        }

        int numRestarts = 0;
        int noImprovementCount = 0;

        int[] startP = new int[N];
        int[] localBestP = new int[N];

        while (stopwatch.ElapsedMilliseconds < timeLimitMs && numRestarts < maxRestarts && noImprovementCount < maxNoImprovementRestarts) {
            // Decide starting permutation for this ILS restart
            if (rand.NextDouble() < 0.7) {
                // Perturb the best permutation found so far
                int K = Math.Max(3, N / 10);
                startP = Perturb(bestP, K, rand);
            } else if (rand.NextDouble() < 0.85) {
                // Perturb the sort heuristic
                int[] sortP = GetSortHeuristicPermutation(F, D, N);
                int K = Math.Max(3, N / 10);
                startP = Perturb(sortP, K, rand);
            } else {
                // Completely random
                startP = GetRandomPermutation(N, rand);
            }

            double T0 = GetInitialTemperature(startP, F, D, F_col, D_col, N, rand);
            double alpha = Math.Pow(1e-4, 1.0 / saSteps);

            long localBestCost = 0;
            SolveSA(startP, localBestP, ref localBestCost, F, D, F_col, D_col, N, T0, alpha, saSteps, rand, stopwatch, timeLimitMs);

            // Polish the local best solution
            if (N <= 400) {
                HillClimb(localBestP, ref localBestCost, F, D, F_col, D_col, N);
            } else {
                RandomizedHillClimb(localBestP, ref localBestCost, F, D, F_col, D_col, N, N * 20, rand);
            }

            if (localBestCost < bestCost) {
                bestCost = localBestCost;
                Array.Copy(localBestP, bestP, N);
                noImprovementCount = 0;
            } else {
                noImprovementCount++;
            }

            numRestarts++;
        }

        // Print final result
        Console.WriteLine(bestCost);
        for (int i = 0; i < N; i++) {
            Console.Write(bestP[i]);
            if (i < N - 1) Console.Write(" ");
        }
        Console.WriteLine();
    }
}
