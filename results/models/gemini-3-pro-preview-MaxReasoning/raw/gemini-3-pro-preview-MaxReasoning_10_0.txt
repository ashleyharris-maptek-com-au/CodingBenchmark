#include <iostream>
#include <vector>
#include <algorithm>
#include <chrono>
#include <functional>
#include <thread>
#include <mutex>
#include <cmath>

struct Rectangle {
    int id;
    int w;
    int h;
    int area;
};

struct Placement {
    int id;
    int board_idx;
    int x;
    int y;
    int rotated;
};

struct SkylineSegment {
    int x;
    int y;
    int width;
};

struct FreeRect {
    int x;
    int y;
    int w;
    int h;
};

struct PlacementOption {
    bool valid = false;
    bool from_free_list = false;
    int free_list_idx = -1;
    int x = 0;
    int y = 0;
    int rotated = 0;
    double score = 1e18;
};

struct Solution {
    int num_boards;
    std::vector<Placement> placements;
};

void add_free_rect(std::vector<FreeRect>& free_rects, FreeRect new_fr, int free_list_cap) {
    if (new_fr.w <= 0 || new_fr.h <= 0) return;
    for (const auto& fr : free_rects) {
        if (new_fr.x >= fr.x && new_fr.y >= fr.y &&
            new_fr.x + new_fr.w <= fr.x + fr.w &&
            new_fr.y + new_fr.h <= fr.y + fr.h) {
            return;
        }
    }
    free_rects.erase(
        std::remove_if(free_rects.begin(), free_rects.end(), [&](const FreeRect& fr) {
            return fr.x >= new_fr.x && fr.y >= new_fr.y &&
                   fr.x + fr.w <= new_fr.x + new_fr.w &&
                   fr.y + fr.h <= new_fr.y + new_fr.h;
        }),
        free_rects.end()
    );
    if ((int)free_rects.size() < free_list_cap) {
        free_rects.push_back(new_fr);
    } else {
        int min_idx = -1;
        int min_area = new_fr.w * new_fr.h;
        for (size_t i = 0; i < free_rects.size(); ++i) {
            int area = free_rects[i].w * free_rects[i].h;
            if (area < min_area) {
                min_area = area;
                min_idx = i;
            }
        }
        if (min_idx != -1) {
            free_rects[min_idx] = new_fr;
        }
    }
}

void register_placement_in_free_list(std::vector<FreeRect>& free_rects, const FreeRect& R, int free_list_cap) {
    std::vector<FreeRect> candidates;
    bool any_intersection = false;
    for (const auto& fr : free_rects) {
        if (R.x < fr.x + fr.w && R.x + R.w > fr.x &&
            R.y < fr.y + fr.h && R.y + R.h > fr.y) {
            any_intersection = true;
            if (R.x > fr.x) candidates.push_back({fr.x, fr.y, R.x - fr.x, fr.h});
            if (R.x + R.w < fr.x + fr.w) candidates.push_back({R.x + R.w, fr.y, fr.x + fr.w - (R.x + R.w), fr.h});
            if (R.y > fr.y) candidates.push_back({fr.x, fr.y, fr.w, R.y - fr.y});
            if (R.y + R.h < fr.y + fr.h) candidates.push_back({fr.x, R.y + R.h, fr.w, fr.y + fr.h - (R.y + R.h)});
        } else {
            candidates.push_back(fr);
        }
    }
    if (!any_intersection) return;
    free_rects.clear();
    for (const auto& cand : candidates) {
        add_free_rect(free_rects, cand, free_list_cap);
    }
}

PlacementOption find_best_placement(
    int rw, int rh,
    const std::vector<SkylineSegment>& segments,
    const std::vector<FreeRect>& free_rects,
    int board_width, int board_height,
    double alpha
) {
    PlacementOption best_opt;
    int num_rotations = (rw == rh) ? 1 : 2;
    for (int rot = 0; rot < num_rotations; ++rot) {
        int w = (rot == 0) ? rw : rh;
        int h = (rot == 0) ? rh : rw;
        if (w > board_width || h > board_height) continue;
        
        for (size_t f = 0; f < free_rects.size(); ++f) {
            const auto& fr = free_rects[f];
            if (fr.w >= w && fr.h >= h) {
                double score = (fr.y + h) + 0.1 * std::min(fr.w - w, fr.h - h);
                if (score < best_opt.score) {
                    best_opt.valid = true;
                    best_opt.from_free_list = true;
                    best_opt.free_list_idx = f;
                    best_opt.x = fr.x;
                    best_opt.y = fr.y;
                    best_opt.rotated = rot;
                    best_opt.score = score;
                }
            }
        }
        
        for (size_t i = 0; i < segments.size(); ++i) {
            int x = segments[i].x;
            if (x + w > board_width) continue;
            
            int y_max = 0;
            size_t curr_idx = i;
            while (curr_idx < segments.size() && segments[curr_idx].x < x + w) {
                y_max = std::max(y_max, segments[curr_idx].y);
                curr_idx++;
            }
            if (y_max + h > board_height) continue;
            
            double score;
            if (alpha == 0.0) {
                score = (y_max + h);
            } else {
                int waste_area = 0;
                curr_idx = i;
                while (curr_idx < segments.size() && segments[curr_idx].x < x + w) {
                    int seg_left = segments[curr_idx].x;
                    int seg_right = seg_left + segments[curr_idx].width;
                    int overlap_w = std::min(seg_right, x + w) - std::max(seg_left, x);
                    waste_area += (y_max - segments[curr_idx].y) * overlap_w;
                    curr_idx++;
                }
                score = (y_max + h) + alpha * waste_area;
            }
            
            if (score < best_opt.score) {
                best_opt.valid = true;
                best_opt.from_free_list = false;
                best_opt.free_list_idx = -1;
                best_opt.x = x;
                best_opt.y = y_max;
                best_opt.rotated = rot;
                best_opt.score = score;
            }
        }
    }
    return best_opt;
}

void place_on_skyline(
    std::vector<SkylineSegment>& segments,
    std::vector<FreeRect>& free_rects,
    int x, int y_max, int w, int h,
    int free_list_cap
) {
    int y_new = y_max + h;
    for (const auto& seg : segments) {
        if (seg.x + seg.width <= x || seg.x >= x + w) continue;
        if (seg.y < y_max) {
            int hole_x = std::max(seg.x, x);
            int hole_w = std::min(seg.x + seg.width, x + w) - hole_x;
            int hole_h = y_max - seg.y;
            add_free_rect(free_rects, {hole_x, seg.y, hole_w, hole_h}, free_list_cap);
        }
    }
    std::vector<SkylineSegment> new_segments;
    new_segments.reserve(segments.size() + 2);
    bool inserted = false;
    for (const auto& seg : segments) {
        if (seg.x + seg.width <= x) {
            new_segments.push_back(seg);
        } else if (seg.x >= x + w) {
            if (!inserted) {
                new_segments.push_back({x, y_new, w});
                inserted = true;
            }
            new_segments.push_back(seg);
        } else {
            if (seg.x < x) new_segments.push_back({seg.x, seg.y, x - seg.x});
            if (seg.x + seg.width > x + w) {
                if (!inserted) {
                    new_segments.push_back({x, y_new, w});
                    inserted = true;
                }
                new_segments.push_back({x + w, seg.y, seg.x + seg.width - (x + w)});
            }
        }
    }
    if (!inserted) new_segments.push_back({x, y_new, w});
    segments.clear();
    for (const auto& seg : new_segments) {
        if (segments.empty()) {
            segments.push_back(seg);
        } else {
            auto& prev = segments.back();
            if (prev.y == seg.y) prev.width += seg.width;
            else segments.push_back(seg);
        }
    }
}

Solution solve_single_config(
    const std::vector<Rectangle>& rects,
    int board_width, int board_height,
    double alpha,
    int free_list_cap
) {
    int N = rects.size();
    std::vector<bool> packed(N, false);
    std::vector<Placement> placements(N);
    int num_boards = 0;
    int unpacked_count = N;
    
    while (unpacked_count > 0) {
        int board_idx = num_boards++;
        std::vector<SkylineSegment> segments = {{0, 0, board_width}};
        std::vector<FreeRect> free_rects;
        int remaining_area = board_width * board_height;
        
        bool board_progress = true;
        bool packed_any_on_board = false;
        
        while (board_progress) {
            board_progress = false;
            for (int i = 0; i < N; ++i) {
                if (packed[i]) continue;
                if (remaining_area < rects[i].area) continue;
                
                PlacementOption opt = find_best_placement(
                    rects[i].w, rects[i].h,
                    segments, free_rects,
                    board_width, board_height,
                    alpha
                );
                if (opt.valid) {
                    int w = (opt.rotated == 0) ? rects[i].w : rects[i].h;
                    int h = (opt.rotated == 0) ? rects[i].h : rects[i].w;
                    if (opt.from_free_list) {
                        register_placement_in_free_list(free_rects, {opt.x, opt.y, w, h}, free_list_cap);
                    } else {
                        place_on_skyline(segments, free_rects, opt.x, opt.y, w, h, free_list_cap);
                    }
                    packed[i] = true;
                    unpacked_count--;
                    remaining_area -= rects[i].area;
                    board_progress = true;
                    packed_any_on_board = true;
                    placements[rects[i].id] = {rects[i].id, board_idx, opt.x, opt.y, opt.rotated};
                }
            }
        }
        
        if (!packed_any_on_board) {
            for (int i = 0; i < N; ++i) {
                if (!packed[i]) {
                    packed[i] = true;
                    unpacked_count--;
                    placements[rects[i].id] = {rects[i].id, board_idx, 0, 0, 0};
                    break;
                }
            }
        }
    }
    return {num_boards, placements};
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    int N, bw, bh;
    if (!(std::cin >> N >> bw >> bh)) return 0;
    
    std::vector<Rectangle> rects(N);
    for (int i = 0; i < N; ++i) {
        rects[i].id = i;
        std::cin >> rects[i].w >> rects[i].h;
        rects[i].area = rects[i].w * rects[i].h;
    }
    
    if (N == 0) {
        std::cout << "0\n";
        return 0;
    }
    
    auto comp_area = [](const Rectangle& a, const Rectangle& b) {
        if (a.area != b.area) return a.area > b.area;
        return a.id < b.id;
    };
    
    auto comp_max_dim = [](const Rectangle& a, const Rectangle& b) {
        int ma = std::max(a.w, a.h);
        int mb = std::max(b.w, b.h);
        if (ma != mb) return ma > mb;
        int mia = std::min(a.w, a.h);
        int mib = std::min(b.w, b.h);
        if (mia != mib) return mia > mib;
        return a.id < b.id;
    };
    
    auto comp_perimeter = [](const Rectangle& a, const Rectangle& b) {
        int pa = a.w + a.h;
        int pb = b.w + b.h;
        if (pa != pb) return pa > pb;
        return a.id < b.id;
    };

    auto comp_height = [](const Rectangle& a, const Rectangle& b) {
        if (a.h != b.h) return a.h > b.h;
        if (a.w != b.w) return a.w > b.w;
        return a.id < b.id;
    };

    auto comp_width = [](const Rectangle& a, const Rectangle& b) {
        if (a.w != b.w) return a.w > b.w;
        if (a.h != b.h) return a.h > b.h;
        return a.id < b.id;
    };

    auto comp_min_dim = [](const Rectangle& a, const Rectangle& b) {
        int mia = std::min(a.w, a.h);
        int mib = std::min(b.w, b.h);
        if (mia != mib) return mia > mib;
        int ma = std::max(a.w, a.h);
        int mb = std::max(b.w, b.h);
        if (ma != mb) return ma > mb;
        return a.id < b.id;
    };

    auto comp_ratio = [](const Rectangle& a, const Rectangle& b) {
        double ra = (double)std::max(a.w, a.h) / std::min(a.w, a.h);
        double rb = (double)std::max(b.w, b.h) / std::min(b.w, b.h);
        if (std::abs(ra - rb) > 1e-5) return ra > rb;
        if (a.area != b.area) return a.area > b.area;
        return a.id < b.id;
    };

    struct ThreadConfig {
        std::vector<Rectangle> sorted_rects;
        double alpha;
        int free_list_cap;
    };

    std::vector<ThreadConfig> configs;
    auto add_config = [&](auto comp, double alpha, int cap) {
        std::vector<Rectangle> temp = rects;
        std::sort(temp.begin(), temp.end(), comp);
        configs.push_back({std::move(temp), alpha, cap});
    };

    add_config(comp_max_dim, 0.0, 150);
    add_config(comp_max_dim, 0.5, 150);
    add_config(comp_max_dim, 0.0, 300);
    add_config(comp_max_dim, 0.5, 300);
    
    add_config(comp_area, 0.0, 150);
    add_config(comp_area, 0.5, 150);
    add_config(comp_area, 0.0, 300);
    add_config(comp_area, 0.5, 300);
    
    add_config(comp_perimeter, 0.0, 150);
    add_config(comp_perimeter, 0.5, 150);
    add_config(comp_perimeter, 0.0, 300);
    add_config(comp_perimeter, 0.5, 300);
    
    add_config(comp_height, 0.0, 150);
    add_config(comp_height, 0.5, 150);
    
    add_config(comp_width, 0.0, 150);
    add_config(comp_width, 0.5, 150);
    
    add_config(comp_min_dim, 0.0, 150);
    add_config(comp_min_dim, 0.5, 150);
    
    add_config(comp_ratio, 0.0, 150);
    add_config(comp_ratio, 0.5, 150);

    std::vector<std::thread> threads;
    std::mutex result_mutex;
    Solution best_sol;
    best_sol.num_boards = 1e9;

    for (const auto& config : configs) {
        threads.emplace_back([&, config]() {
            Solution sol = solve_single_config(config.sorted_rects, bw, bh, config.alpha, config.free_list_cap);
            std::lock_guard<std::mutex> lock(result_mutex);
            if (sol.num_boards < best_sol.num_boards) {
                best_sol = std::move(sol);
            }
        });
    }

    for (auto& t : threads) {
        t.join();
    }

    std::cout << best_sol.num_boards << "\n";
    for (int i = 0; i < N; ++i) {
        std::cout << best_sol.placements[i].board_idx << " "
                  << best_sol.placements[i].x << " "
                  << best_sol.placements[i].y << " "
                  << best_sol.placements[i].rotated << "\n";
    }

    return 0;
}
