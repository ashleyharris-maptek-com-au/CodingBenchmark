// Fast I/O reader
struct FastInput {
    static const int BUF_SIZE = 65536; // 64KB buffer
    std::vector<char> buf;
    int head = 0;
    int tail = 0;
    
    FastInput() : buf(BUF_SIZE) {}
    
    inline char next_char() {
        if (head >= tail) {
            head = 0;
            tail = fread(buf.data(), 1, BUF_SIZE, stdin);
            if (tail == 0) return EOF;
        }
        return buf[head++];
    }
    
    inline bool read_uint(uint32_t& val) {
        char c = next_char();
        while (c != EOF && c <= ' ') {
            c = next_char();
        }
        if (c == EOF) return false;
        val = 0;
        while (c >= '0' && c <= '9') {
            val = val * 10 + (c - '0');
            c = next_char();
        }
        return true;
    }
};

struct EdgePair {
    uint32_t u, v;
};

// Edge streaming to handle very large graphs without memory issues
struct EdgeStream {
    bool on_disk = false;
    std::string filename = "temp_edges.bin";
    FILE* file = nullptr;
    std::vector<EdgePair> mem_edges;
    
    void init(uint64_t M) {
        if (M > 10000000) { // > 10M edges
            on_disk = true;
            file = fopen(filename.c_str(), "wb");
            if (!file) {
                on_disk = false;
            }
        }
        if (!on_disk) {
            mem_edges.reserve(M);
        }
    }
    
    void write_edge(uint32_t u, uint32_t v) {
        if (on_disk) {
            mem_edges.push_back({u, v});
            if (mem_edges.size() >= 1000000) {
                flush();
            }
        } else {
            mem_edges.push_back({u, v});
        }
    }
    
    void flush() {
        if (on_disk && !mem_edges.empty()) {
            fwrite(mem_edges.data(), sizeof(EdgePair), mem_edges.size(), file);
            mem_edges.clear();
        }
    }
    
    void close_write() {
        if (on_disk) {
            flush();
            fclose(file);
            file = nullptr;
        }
    }
    
    struct Reader {
        EdgeStream& stream;
        FILE* file = nullptr;
        std::vector<EdgePair> buf;
        size_t buf_idx = 0;
        size_t buf_size = 0;
        size_t mem_idx = 0;
        
        Reader(EdgeStream& stream) : stream(stream) {
            if (stream.on_disk) {
                file = fopen(stream.filename.c_str(), "rb");
                buf.resize(1000000);
            }
        }
        
        ~Reader() {
            if (file) {
                fclose(file);
            }
        }
        
        bool next_edge(uint32_t& u, uint32_t& v) {
            if (stream.on_disk) {
                if (buf_idx >= buf_size) {
                    buf_idx = 0;
                    buf_size = fread(buf.data(), sizeof(EdgePair), buf.size(), file);
                    if (buf_size == 0) return false;
                }
                u = buf[buf_idx].u;
                v = buf[buf_idx].v;
                buf_idx++;
                return true;
            } else {
                if (mem_idx >= stream.mem_edges.size()) return false;
                u = stream.mem_edges[mem_idx].u;
                v = stream.mem_edges[mem_idx].v;
                mem_idx++;
                return true;
            }
        }
    };
    
    void cleanup() {
        mem_edges.clear();
        mem_edges.shrink_to_fit();
        if (on_disk) {
            std::remove(filename.c_str());
        }
    }
};

// Disjoint Set Union
struct DSU {
    std::vector<int> parent;
    
    void init(int n) {
        parent.resize(n);
        for (int i = 0; i < n; ++i) parent[i] = i;
    }
    
    int find(int i) {
        int root = i;
        while (root != parent[root]) {
            root = parent[root];
        }
        int curr = i;
        while (curr != root) {
            int nxt = parent[curr];
            parent[curr] = root;
            curr = nxt;
        }
        return root;
    }
    
    bool unite(int i, int j) {
        int root_i = find(i);
        int root_j = find(j);
        if (root_i != root_j) {
            parent[root_i] = root_j;
            return true;
        }
        return false;
    }
};

// Iterative Tarjan's Bridge Finder to avoid recursion stack overflow
struct BridgeFinder {
    uint32_t N;
    const uint64_t* offsets;
    const uint32_t* edges;
    const uint32_t* edge_ids;
    
    std::vector<uint32_t> tin;
    std::vector<uint32_t> low;
    std::vector<bool> visited;
    
    std::pair<uint32_t, uint32_t> find_bridge() {
        tin.assign(N, 0);
        low.assign(N, 0);
        visited.assign(N, false);
        uint32_t timer = 0;
        
        struct StackFrame {
            uint32_t u;
            uint64_t edge_idx;
            uint32_t p_edge_id;
        };
        std::vector<StackFrame> stack;
        stack.reserve(N);
        
        for (uint32_t i = 0; i < N; ++i) {
            if (!visited[i]) {
                visited[i] = true;
                tin[i] = low[i] = ++timer;
                stack.push_back({i, offsets[i], (uint32_t)-1});
                
                while (!stack.empty()) {
                    uint32_t u = stack.back().u;
                    uint64_t idx = stack.back().edge_idx;
                    uint32_t p_edge_id = stack.back().p_edge_id;
                    
                    if (idx < offsets[u + 1]) {
                        uint32_t v = edges[idx];
                        uint32_t eid = edge_ids[idx];
                        stack.back().edge_idx++;
                        
                        if (eid == p_edge_id) {
                            continue;
                        }
                        
                        if (visited[v]) {
                            low[u] = std::min(low[u], tin[v]);
                        } else {
                            visited[v] = true;
                            tin[v] = low[v] = ++timer;
                            stack.push_back({v, offsets[v], eid});
                        }
                    } else {
                        stack.pop_back();
                        if (!stack.empty()) {
                            uint32_t parent = stack.back().u;
                            low[parent] = std::min(low[parent], low[u]);
                            
                            if (low[u] > tin[parent]) {
                                return {parent, u};
                            }
                        }
                    }
                }
            }
        }
        return {(uint32_t)-1, (uint32_t)-1};
    }
};

struct KeptEdge {
    uint32_t u;
    uint32_t v;
};

struct ContractionEdge {
    uint32_t to;
    uint32_t weight;
};

// Rebuild contracted graph based on DSU
std::pair<uint32_t, std::vector<std::vector<ContractionEdge>>> rebuild_graph(
    uint32_t n,
    const std::vector<std::vector<ContractionEdge>>& adj,
    DSU& dsu,
    std::vector<uint32_t>& comp_map
) {
    std::vector<uint32_t> reps;
    for (uint32_t i = 0; i < n; ++i) {
        if (dsu.parent[i] == i) {
            reps.push_back(i);
        }
    }
    uint32_t n_new = reps.size();
    std::vector<uint32_t> old_to_new(n);
    for (uint32_t i = 0; i < n_new; ++i) {
        old_to_new[reps[i]] = i;
    }
    
    for (uint32_t i = 0; i < comp_map.size(); ++i) {
        comp_map[i] = old_to_new[dsu.find(comp_map[i])];
    }
    
    std::vector<std::vector<ContractionEdge>> adj_new(n_new);
    for (uint32_t i = 0; i < n; ++i) {
        uint32_t u_new = old_to_new[dsu.find(i)];
        for (const auto& edge : adj[i]) {
            uint32_t v_new = old_to_new[dsu.find(edge.to)];
            if (u_new != v_new) {
                adj_new[u_new].push_back({v_new, edge.weight});
            }
        }
    }
    
    for (uint32_t i = 0; i < n_new; ++i) {
        auto& list = adj_new[i];
        if (list.empty()) continue;
        std::sort(list.begin(), list.end(), [](const ContractionEdge& a, const ContractionEdge& b) {
            return a.to < b.to;
        });
        std::vector<ContractionEdge> merged;
        merged.push_back(list[0]);
        for (size_t j = 1; j < list.size(); ++j) {
            if (list[j].to == merged.back().to) {
                merged.back().weight += list[j].weight;
            } else {
                merged.push_back(list[j]);
            }
        }
        list = std::move(merged);
    }
    
    return {n_new, adj_new};
}

// PR Contractions on weighted graph
bool run_pr_contractions(
    uint32_t n,
    const std::vector<std::vector<ContractionEdge>>& adj,
    uint32_t lambda,
    DSU& dsu
) {
    bool contracted = false;
    dsu.init(n);
    
    for (uint32_t u = 0; u < n; ++u) {
        for (const auto& edge : adj[u]) {
            uint32_t v = edge.to;
            if (u >= v) continue;
            
            uint32_t strength = edge.weight;
            size_t ptr_u = 0, ptr_v = 0;
            const auto& list_u = adj[u];
            const auto& list_v = adj[v];
            
            while (ptr_u < list_u.size() && ptr_v < list_v.size()) {
                if (list_u[ptr_u].to == list_v[ptr_v].to) {
                    strength += std::min(list_u[ptr_u].weight, list_v[ptr_v].weight);
                    ptr_u++;
                    ptr_v++;
                } else if (list_u[ptr_u].to < list_v[ptr_v].to) {
                    ptr_u++;
                } else {
                    ptr_v++;
                }
            }
            
            if (strength >= lambda) {
                if (dsu.unite(u, v)) {
                    contracted = true;
                }
            }
        }
    }
    return contracted;
}

// Stoer-Wagner algorithm for small weighted graphs
std::pair<int, std::vector<uint32_t>> stoer_wagner(uint32_t n, std::vector<std::vector<int>> w) {
    int min_cut = 1e9;
    std::vector<uint32_t> best_partition;
    
    std::vector<std::vector<uint32_t>> components(n);
    for (uint32_t i = 0; i < n; ++i) {
        components[i] = {i};
    }
    
    std::vector<uint32_t> v(n);
    std::iota(v.begin(), v.end(), 0U);
    
    for (uint32_t phase = 0; phase < n - 1; ++phase) {
        uint32_t sz = n - phase;
        std::vector<int> dist(sz, 0);
        std::vector<bool> visited(sz, false);
        uint32_t prev = (uint32_t)-1, curr = (uint32_t)-1;
        
        for (uint32_t step = 0; step < sz; ++step) {
            uint32_t next_v = (uint32_t)-1;
            int max_d = -1;
            for (uint32_t i = 0; i < sz; ++i) {
                if (!visited[i] && dist[i] > max_d) {
                    max_d = dist[i];
                    next_v = i;
                }
            }
            if (next_v == (uint32_t)-1) {
                for (uint32_t i = 0; i < sz; ++i) {
                    if (!visited[i]) {
                        next_v = i;
                        break;
                    }
                }
            }
            visited[next_v] = true;
            prev = curr;
            curr = next_v;
            
            for (uint32_t i = 0; i < sz; ++i) {
                if (!visited[i]) {
                    dist[i] += w[v[curr]][v[i]];
                }
            }
        }
        
        int cut_val = dist[curr];
        if (cut_val < min_cut) {
            min_cut = cut_val;
            best_partition = components[v[curr]];
        }
        
        for (uint32_t i = 0; i < sz; ++i) {
            if (i != curr && i != prev) {
                w[v[prev]][v[i]] += w[v[curr]][v[i]];
                w[v[i]][v[prev]] = w[v[prev]][v[i]];
            }
        }
        
        components[v[prev]].insert(components[v[prev]].end(), components[v[curr]].begin(), components[v[curr]].end());
        v.erase(v.begin() + curr);
    }
    
    return {min_cut, best_partition};
}

// Dinic's Algorithm for max-flow min-cut on weighted graphs
struct Dinic {
    uint32_t n;
    struct FlowEdge {
        uint32_t to;
        int cap;
        int flow;
        int rev;
    };
    std::vector<std::vector<FlowEdge>> adj;
    std::vector<int> level;
    std::vector<uint32_t> ptr;
    
    Dinic(uint32_t n) : n(n), adj(n), level(n), ptr(n) {}
    
    void add_edge(uint32_t from, uint32_t to, int cap) {
        adj[from].push_back({to, cap, 0, (int)adj[to].size()});
        adj[to].push_back({from, cap, 0, (int)adj[from].size() - 1});
    }
    
    bool bfs(uint32_t s, uint32_t t) {
        std::fill(level.begin(), level.end(), -1);
        level[s] = 0;
        std::vector<uint32_t> q;
        q.push_back(s);
        uint32_t head = 0;
        while (head < q.size()) {
            uint32_t v = q[head++];
            for (auto& edge : adj[v]) {
                if (edge.cap - edge.flow > 0 && level[edge.to] == -1) {
                    level[edge.to] = level[v] + 1;
                    q.push_back(edge.to);
                }
            }
        }
        return level[t] != -1;
    }
    
    int dfs(uint32_t v, uint32_t t, int pushed) {
        if (pushed == 0) return 0;
        if (v == t) return pushed;
        for (uint32_t& cid = ptr[v]; cid < adj[v].size(); ++cid) {
            auto& edge = adj[v][cid];
            uint32_t tr = edge.to;
            if (level[v] + 1 != level[tr] || edge.cap - edge.flow == 0) continue;
            int push = dfs(tr, t, std::min(pushed, edge.cap - edge.flow));
            if (push == 0) continue;
            edge.flow += push;
            adj[tr][edge.rev].flow -= push;
            return push;
        }
        return 0;
    }
    
    int max_flow(uint32_t s, uint32_t t, int limit) {
        int flow = 0;
        while (flow < limit && bfs(s, t)) {
            std::fill(ptr.begin(), ptr.end(), 0);
            while (int pushed = dfs(s, t, limit - flow)) {
                flow += pushed;
                if (flow >= limit) break;
            }
        }
        return flow;
    }
    
    std::vector<bool> get_cut(uint32_t s) {
        std::vector<bool> visited(n, false);
        std::vector<uint32_t> q;
        q.push_back(s);
        visited[s] = true;
        uint32_t head = 0;
        while (head < q.size()) {
            uint32_t v = q[head++];
            for (auto& edge : adj[v]) {
                if (edge.cap - edge.flow > 0 && !visited[edge.to]) {
                    visited[edge.to] = true;
                    q.push_back(edge.to);
                }
            }
        }
        return visited;
    }
};

int main() {
    // Fast I/O setup
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    FastInput reader;
    uint32_t N, M;
    if (!reader.read_uint(N) || !reader.read_uint(M)) {
        return 0;
    }
    
    EdgeStream stream;
    stream.init(M);
    
    std::vector<uint32_t> degrees(N, 0);
    for (uint32_t i = 0; i < M; ++i) {
        uint32_t u, v;
        if (!reader.read_uint(u) || !reader.read_uint(v)) {
            return 0;
        }
        stream.write_edge(u, v);
        degrees[u]++;
        degrees[v]++;
    }
    stream.close_write();
    
    // Find minimum degree and a candidate vertex
    uint32_t d_min = 1e9;
    uint32_t s = 0;
    for (uint32_t i = 0; i < N; ++i) {
        if (degrees[i] < d_min) {
            d_min = degrees[i];
            s = i;
        }
    }
    
    // Quick handle for trivially small cuts or single nodes
    if (N < 2) {
        std::cout << 0 << "\n";
        stream.cleanup();
        return 0;
    }
    
    if (d_min == 1) {
        // Cut is 1, and it's the edge incident to s
        std::cout << 1 << "\n";
        EdgeStream::Reader st_reader(stream);
        uint32_t u, v;
        while (st_reader.next_edge(u, v)) {
            if (u == s || v == s) {
                std::cout << u << " " << v << "\n";
                break;
            }
        }
        stream.cleanup();
        return 0;
    }
    
    // Build CSR representation of the original graph to find bridges and run NI reduction
    std::vector<uint64_t> offsets(N + 1, 0);
    for (uint32_t i = 0; i < N; ++i) {
        offsets[i + 1] = offsets[i] + degrees[i];
    }
    
    std::vector<uint32_t> edges(2 * M);
    std::vector<uint32_t> edge_ids(2 * M);
    std::vector<uint64_t> current_positions = offsets;
    
    {
        EdgeStream::Reader st_reader(stream);
        uint32_t u, v;
        uint32_t edge_id = 0;
        while (st_reader.next_edge(u, v)) {
            uint64_t pos_u = current_positions[u]++;
            uint64_t pos_v = current_positions[v]++;
            edges[pos_u] = v;
            edge_ids[pos_u] = edge_id;
            edges[pos_v] = u;
            edge_ids[pos_v] = edge_id;
            edge_id++;
        }
    }
    
    // Run bridge finding
    BridgeFinder bridge_finder{N, offsets.data(), edges.data(), edge_ids.data()};
    auto bridge = bridge_finder.find_bridge();
    if (bridge.first != (uint32_t)-1) {
        std::cout << 1 << "\n";
        std::cout << bridge.first << " " << bridge.second << "\n";
        stream.cleanup();
        return 0;
    }
    
    if (d_min == 2) {
        // Min cut is 2 (since no bridges exist and minimum degree is 2)
        std::cout << 2 << "\n";
        uint64_t start_idx = offsets[s];
        uint64_t end_idx = offsets[s + 1];
        int count = 0;
        for (uint64_t i = start_idx; i < end_idx; ++i) {
            std::cout << s << " " << edges[i] << "\n";
            count++;
            if (count == 2) break;
        }
        stream.cleanup();
        return 0;
    }
    
    // Now run Nagamochi-Ibaraki Forest Decomposition
    // We want to keep up to k = d_min forests
    uint32_t k = std::min(d_min, 15U); // Cap at 15 to keep DSU array small and fast
    std::vector<DSU> dsus(k);
    for (uint32_t i = 0; i < k; ++i) {
        dsus[i].init(N);
    }
    
    std::vector<KeptEdge> kept_edges;
    kept_edges.reserve(k * N);
    
    {
        EdgeStream::Reader st_reader(stream);
        uint32_t u, v;
        while (st_reader.next_edge(u, v)) {
            for (uint32_t i = 0; i < k; ++i) {
                if (dsus[i].unite(u, v)) {
                    kept_edges.push_back({u, v});
                    break;
                }
            }
        }
    }
    
    // Clean up edge stream since we have the kept edges
    stream.cleanup();
    
    // Build initial adjacency list for contraction
    std::vector<std::vector<ContractionEdge>> adj_curr(N);
    for (const auto& ke : kept_edges) {
        adj_curr[ke.u].push_back({ke.v, 1});
        adj_curr[ke.v].push_back({ke.u, 1});
    }
    
    for (uint32_t i = 0; i < N; ++i) {
        std::sort(adj_curr[i].begin(), adj_curr[i].end(), [](const ContractionEdge& a, const ContractionEdge& b) {
            return a.to < b.to;
        });
    }
    
    uint32_t n_curr = N;
    std::vector<uint32_t> comp_map(N);
    std::iota(comp_map.begin(), comp_map.end(), 0);
    
    uint32_t lambda = d_min;
    
    // Iteratively run PR Contractions
    while (true) {
        DSU dsu;
        bool contracted = run_pr_contractions(n_curr, adj_curr, lambda, dsu);
        if (!contracted) break;
        auto [n_new, adj_new] = rebuild_graph(n_curr, adj_curr, dsu, comp_map);
        n_curr = n_new;
        adj_curr = std::move(adj_new);
    }
    
    // Best partition representation
    std::vector<bool> best_in_A(N, false);
    // Initially, let s be on one side, others on the other side
    best_in_A[s] = true;
    
    if (n_curr > 1 && n_curr <= 2000) {
        // Run Stoer-Wagner on the contracted graph
        std::vector<std::vector<int>> w(n_curr, std::vector<int>(n_curr, 0));
        for (uint32_t u = 0; u < n_curr; ++u) {
            for (const auto& edge : adj_curr[u]) {
                w[u][edge.to] = edge.weight;
            }
        }
        
        auto [sw_cut, sw_partition] = stoer_wagner(n_curr, w);
        if (sw_cut < (int)lambda) {
            lambda = sw_cut;
            std::vector<bool> partition_set(n_curr, false);
            for (uint32_t x : sw_partition) {
                partition_set[x] = true;
            }
            for (uint32_t i = 0; i < N; ++i) {
                best_in_A[i] = partition_set[comp_map[i]];
            }
        }
    } else if (n_curr > 2000) {
        // Run randomized Dinic's max-flow min-cut on the contracted graph
        uint32_t s_red = comp_map[s];
        
        // Find candidates: 100 with smallest degrees, and 100 random original vertices
        std::vector<std::pair<uint32_t, uint32_t>> deg_v(n_curr);
        for (uint32_t i = 0; i < n_curr; ++i) {
            uint32_t d = 0;
            for (const auto& edge : adj_curr[i]) d += edge.weight;
            deg_v[i] = {d, i};
        }
        
        // Check if any single vertex in the contracted graph has degree < lambda
        for (uint32_t i = 0; i < n_curr; ++i) {
            if (deg_v[i].first < lambda) {
                lambda = deg_v[i].first;
                for (uint32_t u = 0; u < N; ++u) {
                    best_in_A[u] = (comp_map[u] == i);
                }
            }
        }
        
        std::sort(deg_v.begin(), deg_v.end());
        
        std::vector<uint32_t> candidates;
        for (uint32_t i = 0; i < std::min(n_curr, 100U); ++i) {
            if (deg_v[i].second != s_red) {
                candidates.push_back(deg_v[i].second);
            }
        }
        
        // Add 100 random original mapped to contracted
        srand(1337);
        for (int i = 0; i < 100; ++i) {
            uint32_t r = rand() % N;
            uint32_t t_red = comp_map[r];
            if (t_red != s_red) {
                candidates.push_back(t_red);
            }
        }
        
        // De-duplicate candidates
        std::sort(candidates.begin(), candidates.end());
        candidates.erase(std::unique(candidates.begin(), candidates.end()), candidates.end());
        
        for (uint32_t t_red : candidates) {
            if (lambda <= 2) break; // Lower bound reached (since no bridges exist)
            
            Dinic dinic(n_curr);
            for (uint32_t u = 0; u < n_curr; ++u) {
                for (const auto& edge : adj_curr[u]) {
                    if (u < edge.to) {
                        dinic.add_edge(u, edge.to, edge.weight);
                    }
                }
            }
            
            int flow = dinic.max_flow(s_red, t_red, lambda - 1);
            if (flow < (int)lambda) {
                lambda = flow;
                auto cut = dinic.get_cut(s_red);
                for (uint32_t i = 0; i < N; ++i) {
                    best_in_A[i] = cut[comp_map[i]];
                }
            }
        }
    }
    
    // Print the min-cut edges using the best partition
    // We scan the original edges from CSR
    std::vector<std::pair<uint32_t, uint32_t>> cut_edges;
    cut_edges.reserve(lambda);
    
    for (uint32_t u = 0; u < N; ++u) {
        uint64_t start_idx = offsets[u];
        uint64_t end_idx = offsets[u + 1];
        for (uint64_t i = start_idx; i < end_idx; ++i) {
            uint32_t v = edges[i];
            if (u < v && best_in_A[u] != best_in_A[v]) {
                cut_edges.push_back({u, v});
            }
        }
    }
    
    std::cout << cut_edges.size() << "\n";
    for (const auto& edge : cut_edges) {
        std::cout << edge.first << " " << edge.second << "\n";
    }
    
    return 0;
}
