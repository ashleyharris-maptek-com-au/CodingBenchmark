@group(0) @binding(0) var<storage, read>       baseData : array<u32, 16>;
@group(0) @binding(1) var<storage, read_write>  result   : array<u32, 4>;
@group(0) @binding(2) var<uniform>              params   : vec4<u32>;
// params.x = difficulty, params.y = range_start, params.z = range_size

fn rotl(x: u32, shift: u32) -> u32 {
    return (x << shift) | (x >> (32u - shift));
}

@compute @workgroup_size(256)
fn main(@builtin(global_invocation_id) gid : vec3<u32>) {
    if (gid.x >= params.z) {
        return;
    }

    let nonce = params.y + gid.x;

    var state : array<u32, 16>;
    for (var i = 0u; i < 16u; i = i + 1u) {
        state[i] = baseData[i] ^ (nonce + i);
    }

    for (var round = 0u; round < 8u; round = round + 1u) {
        // Group 1: 0, 1, 2, 3
        state[0u] = state[0u] + state[1u];
        state[3u] = rotl(state[3u] ^ state[0u], 16u);
        state[2u] = state[2u] + state[3u];
        state[1u] = rotl(state[1u] ^ state[2u], 12u);
        state[0u] = state[0u] + state[1u];
        state[3u] = rotl(state[3u] ^ state[0u], 8u);
        state[2u] = state[2u] + state[3u];
        state[1u] = rotl(state[1u] ^ state[2u], 7u);

        // Group 2: 4, 5, 6, 7
        state[4u] = state[4u] + state[5u];
        state[7u] = rotl(state[7u] ^ state[4u], 16u);
        state[6u] = state[6u] + state[7u];
        state[5u] = rotl(state[5u] ^ state[6u], 12u);
        state[4u] = state[4u] + state[5u];
        state[7u] = rotl(state[7u] ^ state[4u], 8u);
        state[6u] = state[6u] + state[7u];
        state[5u] = rotl(state[5u] ^ state[6u], 7u);

        // Group 3: 8, 9, 10, 11
        state[8u] = state[8u] + state[9u];
        state[11u] = rotl(state[11u] ^ state[8u], 16u);
        state[10u] = state[10u] + state[11u];
        state[9u] = rotl(state[9u] ^ state[10u], 12u);
        state[8u] = state[8u] + state[9u];
        state[11u] = rotl(state[11u] ^ state[8u], 8u);
        state[10u] = state[10u] + state[11u];
        state[9u] = rotl(state[9u] ^ state[10u], 7u);

        // Group 4: 12, 13, 14, 15
        state[12u] = state[12u] + state[13u];
        state[15u] = rotl(state[15u] ^ state[12u], 16u);
        state[14u] = state[14u] + state[15u];
        state[13u] = rotl(state[13u] ^ state[14u], 12u);
        state[12u] = state[12u] + state[13u];
        state[15u] = rotl(state[15u] ^ state[12u], 8u);
        state[14u] = state[14u] + state[15u];
        state[13u] = rotl(state[13u] ^ state[14u], 7u);
    }

    let h0 = state[0u] ^ state[2u] ^ state[4u] ^ state[6u] ^ state[8u] ^ state[10u] ^ state[12u] ^ state[14u];
    let h1 = state[1u] ^ state[3u] ^ state[5u] ^ state[7u] ^ state[9u] ^ state[11u] ^ state[13u] ^ state[15u];

    var trailing_zeros = 0u;
    if (h0 == 0u) {
        trailing_zeros = 32u + countTrailingZeros(h1);
    } else {
        trailing_zeros = countTrailingZeros(h0);
    }

    if (trailing_zeros >= params.x) {
        result[0u] = 1u;
        result[1u] = nonce;
        result[2u] = h0;
        result[3u] = h1;
    }
}
