public enum HeuristicType
{
    DBL,
    CornerDistance,
    MinHeight
}

public struct Vector3D
{
    public double X;
    public double Y;
    public double Z;

    public Vector3D(double x, double y, double z)
    {
        X = x; Y = y; Z = z;
    }

    public static Vector3D operator +(Vector3D a, Vector3D b) => new Vector3D(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
    public static Vector3D operator -(Vector3D a, Vector3D b) => new Vector3D(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
    public static Vector3D operator *(Vector3D a, double b) => new Vector3D(a.X * b, a.Y * b, a.Z * b);

    public static double Dot(Vector3D a, Vector3D b) => a.X * b.X + a.Y * b.Y + a.Z * b.Z;
    public static Vector3D Cross(Vector3D a, Vector3D b) => new Vector3D(
        a.Y * b.Z - a.Z * b.Y,
        a.Z * b.X - a.X * b.Z,
        a.X * b.Y - a.Y * b.X
    );

    public double Length() => Math.Sqrt(X * X + Y * Y + Z * Z);
    public Vector3D Normalized()
    {
        double len = Length();
        if (len < 1e-9) return new Vector3D(0, 0, 0);
        return new Vector3D(X / len, Y / len, Z / len);
    }
}

public struct Quaternion
{
    public double W;
    public double X;
    public double Y;
    public double Z;

    public Quaternion(double w, double x, double y, double z)
    {
        W = w; X = x; Y = y; Z = z;
    }

    public static Quaternion Multiply(Quaternion a, Quaternion b)
    {
        return new Quaternion(
            a.W * b.W - a.X * b.X - a.Y * b.Y - a.Z * b.Z,
            a.W * b.X + a.X * b.W + a.Y * b.Z - a.Z * b.Y,
            a.W * b.Y - a.X * b.Z + a.Y * b.W + a.Z * b.X,
            a.W * b.Z + a.X * b.Y - a.Y * b.X + a.Z * b.W
        );
    }

    public Vector3D Rotate(Vector3D p)
    {
        Vector3D v = new Vector3D(X, Y, Z);
        Vector3D cross1 = Vector3D.Cross(v, p) + p * W;
        Vector3D cross2 = Vector3D.Cross(v, cross1);
        return p + cross2 * 2.0;
    }

    public Quaternion Normalized()
    {
        double len = Math.Sqrt(W * W + X * X + Y * Y + Z * Z);
        if (len < 1e-9) return new Quaternion(1, 0, 0, 0);
        return new Quaternion(W / len, X / len, Y / len, Z / len);
    }
}

public struct Vector3Int
{
    public int X;
    public int Y;
    public int Z;
    public Vector3Int(int x, int y, int z) { X = x; Y = y; Z = z; }
}

public struct ExtremePoint : IEquatable<ExtremePoint>
{
    public int X;
    public int Y;
    public int Z;

    public ExtremePoint(int x, int y, int z) { X = x; Y = y; Z = z; }

    public bool Equals(ExtremePoint other) => X == other.X && Y == other.Y && Z == other.Z;
    public override int GetHashCode() => X ^ (Y << 10) ^ (Z << 20);
    public override bool Equals(object obj) => obj is ExtremePoint other && Equals(other);
}

public class VoxelShape
{
    public readonly int Width;
    public readonly int Height;
    public readonly int Depth;
    public readonly int MinCx;
    public readonly int MinCy;
    public readonly int MinCz;
    public readonly Quaternion Rotation;
    public readonly int[] VoxelOffsets;
    public readonly Vector3Int[] Voxels;

    public VoxelShape(int w, int h, int d, int minCx, int minCy, int minCz, Quaternion rot, Vector3Int[] voxels, int gridWidth, int gridHeight)
    {
        Width = w;
        Height = h;
        Depth = d;
        MinCx = minCx;
        MinCy = minCy;
        MinCz = minCz;
        Rotation = rot;
        Voxels = voxels;
        VoxelOffsets = new int[voxels.Length];
        for (int i = 0; i < voxels.Length; i++)
        {
            VoxelOffsets[i] = voxels[i].X + voxels[i].Y * gridWidth + voxels[i].Z * gridWidth * gridHeight;
        }
    }
}

public class VoxelGrid
{
    public readonly int Width;
    public readonly int Height;
    public readonly int Depth;
    public readonly byte[] Data;

    public VoxelGrid(int w, int h, int d)
    {
        Width = w;
        Height = h;
        Depth = d;
        Data = new byte[w * h * d];
    }

    public bool IsOccupied(int x, int y, int z)
    {
        return Data[x + y * Width + z * Width * Height] != 0;
    }

    public bool CanPlace(VoxelShape shape, int gx, int gy, int gz)
    {
        int baseIdx = gx + gy * Width + gz * Width * Height;
        var offsets = shape.VoxelOffsets;
        for (int i = 0; i < offsets.Length; i++)
        {
            if (Data[baseIdx + offsets[i]] != 0)
                return false;
        }
        return true;
    }

    public void Place(VoxelShape shape, int gx, int gy, int gz)
    {
        int baseIdx = gx + gy * Width + gz * Width * Height;
        var offsets = shape.VoxelOffsets;
        for (int i = 0; i < offsets.Length; i++)
        {
            Data[baseIdx + offsets[i]] = 1;
        }
    }
}

public class Program
{
    public static void Main(string[] args)
    {
        // Set invariant culture
        Thread.CurrentThread.CurrentCulture = CultureInfo.InvariantCulture;
        Thread.CurrentThread.CurrentUICulture = CultureInfo.InvariantCulture;

        var stopwatch = Stopwatch.StartNew();
        double timeLimitSeconds = 285.0; // 4 minutes 45 seconds

        // Parse input
        string line;
        int V = 0, F = 0;
        while ((line = Console.ReadLine()) != null)
        {
            line = line.Trim();
            if (string.IsNullOrEmpty(line)) continue;
            var parts = line.Split((char[])null, StringSplitOptions.RemoveEmptyEntries);
            if (parts.Length >= 2)
            {
                V = int.Parse(parts[0]);
                F = int.Parse(parts[1]);
                break;
            }
        }

        var vertices = new Vector3D[V];
        for (int i = 0; i < V; i++)
        {
            while ((line = Console.ReadLine()) != null)
            {
                line = line.Trim();
                if (string.IsNullOrEmpty(line)) continue;
                var parts = line.Split((char[])null, StringSplitOptions.RemoveEmptyEntries);
                vertices[i] = new Vector3D(
                    double.Parse(parts[0]),
                    double.Parse(parts[1]),
                    double.Parse(parts[2])
                );
                break;
            }
        }

        var faces = new int[F][];
        for (int i = 0; i < F; i++)
        {
            while ((line = Console.ReadLine()) != null)
            {
                line = line.Trim();
                if (string.IsNullOrEmpty(line)) continue;
                var parts = line.Split((char[])null, StringSplitOptions.RemoveEmptyEntries);
                faces[i] = new int[] {
                    int.Parse(parts[0]),
                    int.Parse(parts[1]),
                    int.Parse(parts[2])
                };
                break;
            }
        }

        // Check if 1-indexed and convert if needed
        int minIndex = int.MaxValue;
        int maxIndex = int.MinValue;
        for (int i = 0; i < F; i++)
        {
            minIndex = Math.Min(minIndex, Math.Min(faces[i][0], Math.Min(faces[i][1], faces[i][2])));
            maxIndex = Math.Max(maxIndex, Math.Max(faces[i][0], Math.Max(faces[i][1], faces[i][2])));
        }
        if (minIndex == 1 || maxIndex == V)
        {
            for (int i = 0; i < F; i++)
            {
                faces[i][0]--;
                faces[i][1]--;
                faces[i][2]--;
            }
        }

        double W = 0, H = 0, D = 0;
        while ((line = Console.ReadLine()) != null)
        {
            line = line.Trim();
            if (string.IsNullOrEmpty(line)) continue;
            var parts = line.Split((char[])null, StringSplitOptions.RemoveEmptyEntries);
            W = double.Parse(parts[0]);
            H = double.Parse(parts[1]);
            D = double.Parse(parts[2]);
            break;
        }

        // Find initial bounding box of polyhedron
        double minX = vertices[0].X, maxX = vertices[0].X;
        double minY = vertices[0].Y, maxY = vertices[0].Y;
        double minZ = vertices[0].Z, maxZ = vertices[0].Z;
        for (int i = 1; i < V; i++)
        {
            minX = Math.Min(minX, vertices[i].X); maxX = Math.Max(maxX, vertices[i].X);
            minY = Math.Min(minY, vertices[i].Y); maxY = Math.Max(maxY, vertices[i].Y);
            minZ = Math.Min(minZ, vertices[i].Z); maxZ = Math.Max(maxZ, vertices[i].Z);
        }

        double pw = maxX - minX;
        double ph = maxY - minY;
        double pd = maxZ - minZ;
        double min_dim_p = Math.Min(pw, Math.Min(ph, pd));
        if (min_dim_p < 1e-5) min_dim_p = 1e-5;

        double V_p = pw * ph * pd;
        if (V_p < 1e-9) V_p = 1e-9;
        double V_c = W * H * D;

        // Generate rotations
        var rotations = GenerateRotations(vertices, faces);

        object lockObj = new object();
        int bestPackedCount = -1;
        List<PackedItem> bestPackedItems = null;
        double bestS = 0;

        int numThreads = Environment.ProcessorCount;
        if (numThreads < 1) numThreads = 1;

        Parallel.For(0, numThreads, threadId =>
        {
            var random = new Random(threadId ^ 1337);

            double baseS;
            int targetMaxItems = 30000;
            if (V_c / V_p > targetMaxItems)
            {
                baseS = Math.Pow(V_c / targetMaxItems, 1.0 / 3.0);
                baseS = Math.Max(baseS, min_dim_p);
            }
            else
            {
                baseS = min_dim_p;
                double max_container_dim = Math.Max(W, Math.Max(H, D));
                double max_grid_dim = 150.0;
                if (max_container_dim / baseS > max_grid_dim)
                {
                    baseS = max_container_dim / max_grid_dim;
                }
            }

            while (stopwatch.Elapsed.TotalSeconds < timeLimitSeconds)
            {
                double factor = 1.0;
                if (threadId > 0)
                {
                    factor = 0.8 + random.NextDouble() * 0.5; // [0.8, 1.3]
                }

                double S = baseS * factor;
                int G_W = (int)Math.Floor(W / S);
                int G_H = (int)Math.Floor(H / S);
                int G_D = (int)Math.Floor(D / S);

                if (G_W <= 0) G_W = 1;
                if (G_H <= 0) G_H = 1;
                if (G_D <= 0) G_D = 1;

                long gridCells = (long)G_W * G_H * G_D;
                if (gridCells > 12000000)
                {
                    continue;
                }

                // Voxelize for all unique rotations at this S
                var shapes = new List<VoxelShape>();
                foreach (var rot in rotations)
                {
                    var shape = Voxelize(vertices, faces, rot, S, G_W, G_H);
                    bool duplicate = false;
                    foreach (var existing in shapes)
                    {
                        if (ShapesAreIdentical(shape, existing))
                        {
                            duplicate = true;
                            break;
                        }
                    }
                    if (!duplicate)
                    {
                        shapes.Add(shape);
                    }
                }

                var heuristic = (HeuristicType)(random.Next(3));
                int k = random.Next(1, 4);

                var packed = RunPacking(G_W, G_H, G_D, shapes, heuristic, k, random, stopwatch, timeLimitSeconds);
                if (packed != null)
                {
                    lock (lockObj)
                    {
                        if (packed.Count > bestPackedCount)
                        {
                            bestPackedCount = packed.Count;
                            bestPackedItems = packed;
                            bestS = S;
                        }
                    }
                }

                if (V_c / V_p > 20000)
                {
                    break;
                }
            }
        });

        // Print output
        if (bestPackedItems != null)
        {
            Console.WriteLine(bestPackedItems.Count);
            foreach (var item in bestPackedItems)
            {
                double tx = (item.Gx - item.Shape.MinCx) * bestS;
                double ty = (item.Gy - item.Shape.MinCy) * bestS;
                double tz = (item.Gz - item.Shape.MinCz) * bestS;
                var q = item.Shape.Rotation;
                Console.WriteLine($"{tx:F6} {ty:F6} {tz:F6} {q.W:F6} {q.X:F6} {q.Y:F6} {q.Z:F6}");
            }
        }
        else
        {
            Console.WriteLine(0);
        }
    }

    private static List<Quaternion> GenerateRotations(Vector3D[] vertices, int[][] faces)
    {
        var list = new List<Quaternion>();
        var primaryAxes = new Vector3D[]
        {
            new Vector3D(1, 0, 0),
            new Vector3D(-1, 0, 0),
            new Vector3D(0, 1, 0),
            new Vector3D(0, -1, 0),
            new Vector3D(0, 0, 1),
            new Vector3D(0, 0, -1)
        };

        // 1. Add 24 standard orthogonal rotations
        for (int i = 0; i < 6; i++)
        {
            var vx = primaryAxes[i];
            for (int j = 0; j < 6; j++)
            {
                var vy = primaryAxes[j];
                if (Math.Abs(Vector3D.Dot(vx, vy)) < 1e-9)
                {
                    var vz = Vector3D.Cross(vx, vy);
                    var q = MatrixToQuaternion(vx, vy, vz);
                    list.Add(q);
                }
            }
        }

        // 2. Add face-aligned rotations
        var uniqueNormals = new List<Vector3D>();
        for (int f = 0; f < faces.Length; f++)
        {
            var face = faces[f];
            Vector3D a = vertices[face[0]];
            Vector3D b = vertices[face[1]];
            Vector3D c = vertices[face[2]];
            Vector3D normal = Vector3D.Cross(b - a, c - a).Normalized();

            if (normal.Length() > 0.1)
            {
                bool duplicate = false;
                foreach (var un in uniqueNormals)
                {
                    if (Math.Abs(Vector3D.Dot(normal, un)) > 0.99)
                    {
                        duplicate = true;
                        break;
                    }
                }
                if (!duplicate)
                {
                    uniqueNormals.Add(normal);
                }
            }
        }

        foreach (var n in uniqueNormals)
        {
            foreach (var u in primaryAxes)
            {
                var qAlign = GetRotationToAlign(n, u);
                double[] angles = { 0, Math.PI / 2, Math.PI, 3 * Math.PI / 2 };
                foreach (var angle in angles)
                {
                    var qRot = new Quaternion(
                        Math.Cos(angle / 2.0),
                        u.X * Math.Sin(angle / 2.0),
                        u.Y * Math.Sin(angle / 2.0),
                        u.Z * Math.Sin(angle / 2.0)
                    );
                    var qCombined = Quaternion.Multiply(qRot, qAlign).Normalized();
                    list.Add(qCombined);
                }
            }
        }

        // Filter duplicates
        var uniqueRotations = new List<Quaternion>();
        foreach (var q in list)
        {
            bool duplicate = false;
            foreach (var uq in uniqueRotations)
            {
                double dot = q.W * uq.W + q.X * uq.X + q.Y * uq.Y + q.Z * uq.Z;
                if (Math.Abs(dot) > 0.9999)
                {
                    duplicate = true;
                    break;
                }
            }
            if (!duplicate)
            {
                uniqueRotations.Add(q);
            }
        }

        return uniqueRotations;
    }

    private static Quaternion MatrixToQuaternion(Vector3D vx, Vector3D vy, Vector3D vz)
    {
        double m00 = vx.X, m01 = vy.X, m02 = vz.X;
        double m10 = vx.Y, m11 = vy.Y, m12 = vz.Y;
        double m20 = vx.Z, m21 = vy.Z, m22 = vz.Z;

        double tr = m00 + m11 + m22;
        double qw, qx, qy, qz;

        if (tr > 0)
        {
            double S = Math.Sqrt(tr + 1.0) * 2.0;
            qw = 0.25 * S;
            qx = (m21 - m12) / S;
            qy = (m02 - m20) / S;
            qz = (m10 - m01) / S;
        }
        else if ((m00 > m11) && (m00 > m22))
        {
            double S = Math.Sqrt(1.0 + m00 - m11 - m22) * 2.0;
            qw = (m21 - m12) / S;
            qx = 0.25 * S;
            qy = (m01 + m10) / S;
            qz = (m02 + m20) / S;
        }
        else if (m11 > m22)
        {
            double S = Math.Sqrt(1.0 + m11 - m00 - m22) * 2.0;
            qw = (m02 - m20) / S;
            qx = (m01 + m10) / S;
            qy = 0.25 * S;
            qz = (m12 + m21) / S;
        }
        else
        {
            double S = Math.Sqrt(1.0 + m22 - m00 - m11) * 2.0;
            qw = (m10 - m01) / S;
            qx = (m02 + m20) / S;
            qy = (m12 + m21) / S;
            qz = 0.25 * S;
        }

        return new Quaternion(qw, qx, qy, qz).Normalized();
    }

    private static Quaternion GetRotationToAlign(Vector3D n, Vector3D u)
    {
        double d = Vector3D.Dot(n, u);
        if (d > 0.99999)
        {
            return new Quaternion(1, 0, 0, 0);
        }
        else if (d < -0.99999)
        {
            Vector3D p;
            if (Math.Abs(n.X) > 0.5) p = new Vector3D(-n.Y, n.X, 0).Normalized();
            else p = new Vector3D(0, -n.Z, n.Y).Normalized();
            return new Quaternion(0, p.X, p.Y, p.Z);
        }
        else
        {
            Vector3D v = Vector3D.Cross(n, u);
            return new Quaternion(d + 1, v.X, v.Y, v.Z).Normalized();
        }
    }

    private static VoxelShape Voxelize(Vector3D[] vertices, int[][] faces, Quaternion rot, double S, int gridWidth, int gridHeight)
    {
        var rotVertices = new Vector3D[vertices.Length];
        for (int i = 0; i < vertices.Length; i++)
        {
            rotVertices[i] = rot.Rotate(vertices[i]);
        }

        double minX = double.MaxValue, maxX = double.MinValue;
        double minY = double.MaxValue, maxY = double.MinValue;
        double minZ = double.MaxValue, maxZ = double.MinValue;
        foreach (var v in rotVertices)
        {
            minX = Math.Min(minX, v.X); maxX = Math.Max(maxX, v.X);
            minY = Math.Min(minY, v.Y); maxY = Math.Max(maxY, v.Y);
            minZ = Math.Min(minZ, v.Z); maxZ = Math.Max(maxZ, v.Z);
        }

        // Shrink slightly towards center to avoid boundary touching problems
        double cX = (minX + maxX) / 2.0;
        double cY = (minY + maxY) / 2.0;
        double cZ = (minZ + maxZ) / 2.0;
        for (int i = 0; i < rotVertices.Length; i++)
        {
            rotVertices[i] = new Vector3D(
                cX + (rotVertices[i].X - cX) * 0.9999,
                cY + (rotVertices[i].Y - cY) * 0.9999,
                cZ + (rotVertices[i].Z - cZ) * 0.9999
            );
        }

        // Recalculate bounding box on shrunken vertices
        minX = double.MaxValue; maxX = double.MinValue;
        minY = double.MaxValue; maxY = double.MinValue;
        minZ = double.MaxValue; maxZ = double.MinValue;
        foreach (var v in rotVertices)
        {
            minX = Math.Min(minX, v.X); maxX = Math.Max(maxX, v.X);
            minY = Math.Min(minY, v.Y); maxY = Math.Max(maxY, v.Y);
            minZ = Math.Min(minZ, v.Z); maxZ = Math.Max(maxZ, v.Z);
        }

        int minCx = (int)Math.Floor(minX / S);
        int maxCx = (int)Math.Floor(maxX / S);
        int minCy = (int)Math.Floor(minY / S);
        int maxCy = (int)Math.Floor(maxY / S);
        int minCz = (int)Math.Floor(minZ / S);
        int maxCz = (int)Math.Floor(maxZ / S);

        int width = maxCx - minCx + 1;
        int height = maxCy - minCy + 1;
        int depth = maxCz - minCz + 1;

        if (width <= 0) width = 1;
        if (height <= 0) height = 1;
        if (depth <= 0) depth = 1;

        bool[,,] occupied = new bool[width, height, depth];
        Vector3D boxHalfSize = new Vector3D(S / 2.0 + 1e-9, S / 2.0 + 1e-9, S / 2.0 + 1e-9);

        for (int f = 0; f < faces.Length; f++)
        {
            var face = faces[f];
            Vector3D a = rotVertices[face[0]];
            Vector3D b = rotVertices[face[1]];
            Vector3D c = rotVertices[face[2]];

            double fMinX = Math.Min(a.X, Math.Min(b.X, c.X));
            double fMaxX = Math.Max(a.X, Math.Max(b.X, c.X));
            double fMinY = Math.Min(a.Y, Math.Min(b.Y, c.Y));
            double fMaxY = Math.Max(a.Y, Math.Max(b.Y, c.Y));
            double fMinZ = Math.Min(a.Z, Math.Min(b.Z, c.Z));
            double fMaxZ = Math.Max(a.Z, Math.Max(b.Z, c.Z));

            int fMinCx = (int)Math.Floor(fMinX / S);
            int fMaxCx = (int)Math.Floor(fMaxX / S);
            int fMinCy = (int)Math.Floor(fMinY / S);
            int fMaxCy = (int)Math.Floor(fMaxY / S);
            int fMinCz = (int)Math.Floor(fMinZ / S);
            int fMaxCz = (int)Math.Floor(fMaxZ / S);

            fMinCx = Math.Max(fMinCx, minCx); fMaxCx = Math.Min(fMaxCx, maxCx);
            fMinCy = Math.Max(fMinCy, minCy); fMaxCy = Math.Min(fMaxCy, maxCy);
            fMinCz = Math.Max(fMinCz, minCz); fMaxCz = Math.Min(fMaxCz, maxCz);

            for (int cx = fMinCx; cx <= fMaxCx; cx++)
            {
                for (int cy = fMinCy; cy <= fMaxCy; cy++)
                {
                    for (int cz = fMinCz; cz <= fMaxCz; cz++)
                    {
                        int ix = cx - minCx;
                        int iy = cy - minCy;
                        int iz = cz - minCz;

                        if (occupied[ix, iy, iz]) continue;

                        Vector3D boxCenter = new Vector3D((cx + 0.5) * S, (cy + 0.5) * S, (cz + 0.5) * S);
                        if (TestTriangleBox(a, b, c, boxCenter, boxHalfSize))
                        {
                            occupied[ix, iy, iz] = true;
                        }
                    }
                }
            }
        }

        var intersections = new List<double>();
        for (int ix = 0; ix < width; ix++)
        {
            int cx = minCx + ix;
            double rx = (cx + 0.5) * S + 1e-9;

            for (int iy = 0; iy < height; iy++)
            {
                int cy = minCy + iy;
                double ry = (cy + 0.5) * S + 1e-9;

                intersections.Clear();

                for (int f = 0; f < faces.Length; f++)
                {
                    var face = faces[f];
                    Vector3D a = rotVertices[face[0]];
                    Vector3D b = rotVertices[face[1]];
                    Vector3D c = rotVertices[face[2]];

                    if (PointInTriangle2D(rx, ry, a, b, c))
                    {
                        Vector3D ab = b - a;
                        Vector3D ac = c - a;
                        Vector3D normal = Vector3D.Cross(ab, ac);
                        if (Math.Abs(normal.Z) > 1e-9)
                        {
                            double z = a.Z - (normal.X * (rx - a.X) + normal.Y * (ry - a.Y)) / normal.Z;
                            intersections.Add(z);
                        }
                    }
                }

                if (intersections.Count >= 2)
                {
                    intersections.Sort();
                    for (int k = 0; k < intersections.Count - 1; k += 2)
                    {
                        double zStart = intersections[k];
                        double zEnd = intersections[k + 1];

                        int czStart = (int)Math.Ceiling(zStart / S - 0.5);
                        int czEnd = (int)Math.Floor(zEnd / S - 0.5);

                        czStart = Math.Max(czStart, minCz);
                        czEnd = Math.Min(czEnd, maxCz);

                        for (int cz = czStart; cz <= czEnd; cz++)
                        {
                            int iz = cz - minCz;
                            occupied[ix, iy, iz] = true;
                        }
                    }
                }
            }
        }

        var voxels = new List<Vector3Int>();
        for (int ix = 0; ix < width; ix++)
        {
            for (int iy = 0; iy < height; iy++)
            {
                for (int iz = 0; iz < depth; iz++)
                {
                    if (occupied[ix, iy, iz])
                    {
                        voxels.Add(new Vector3Int(ix, iy, iz));
                    }
                }
            }
        }

        if (voxels.Count == 0)
        {
            voxels.Add(new Vector3Int(0, 0, 0));
        }

        return new VoxelShape(width, height, depth, minCx, minCy, minCz, rot, voxels.ToArray(), gridWidth, gridHeight);
    }

    private static bool PointInTriangle2D(double px, double py, Vector3D a, Vector3D b, Vector3D c)
    {
        double sign1 = (b.X - a.X) * (py - a.Y) - (b.Y - a.Y) * (px - a.X);
        double sign2 = (c.X - b.X) * (py - b.Y) - (c.Y - b.Y) * (px - b.X);
        double sign3 = (a.X - c.X) * (py - c.Y) - (a.Y - c.Y) * (px - c.X);

        bool has_neg = (sign1 < 0) || (sign2 < 0) || (sign3 < 0);
        bool has_pos = (sign1 > 0) || (sign2 > 0) || (sign3 > 0);

        return !(has_neg && has_pos);
    }

    private static bool TestTriangleBox(Vector3D a, Vector3D b, Vector3D c, Vector3D boxCenter, Vector3D boxHalfSize)
    {
        Vector3D v0 = a - boxCenter;
        Vector3D v1 = b - boxCenter;
        Vector3D v2 = c - boxCenter;

        Vector3D e0 = v1 - v0;
        Vector3D e1 = v2 - v1;
        Vector3D e2 = v0 - v2;

        double r, p0, p1, p2, minP, maxP;

        // Test 1: Box normals
        minP = Math.Min(v0.X, Math.Min(v1.X, v2.X));
        maxP = Math.Max(v0.X, Math.Max(v1.X, v2.X));
        if (minP > boxHalfSize.X || maxP < -boxHalfSize.X) return false;

        minP = Math.Min(v0.Y, Math.Min(v1.Y, v2.Y));
        maxP = Math.Max(v0.Y, Math.Max(v1.Y, v2.Y));
        if (minP > boxHalfSize.Y || maxP < -boxHalfSize.Y) return false;

        minP = Math.Min(v0.Z, Math.Min(v1.Z, v2.Z));
        maxP = Math.Max(v0.Z, Math.Max(v1.Z, v2.Z));
        if (minP > boxHalfSize.Z || maxP < -boxHalfSize.Z) return false;

        // Test 2: Triangle normal
        Vector3D normal = Vector3D.Cross(e0, e1);
        r = boxHalfSize.X * Math.Abs(normal.X) + boxHalfSize.Y * Math.Abs(normal.Y) + boxHalfSize.Z * Math.Abs(normal.Z);
        double d = Vector3D.Dot(normal, v0);
        if (Math.Abs(d) > r) return false;

        // Test 3: 9 cross product axes
        // Axis: X x e0
        r = boxHalfSize.Y * Math.Abs(e0.Z) + boxHalfSize.Z * Math.Abs(e0.Y);
        p0 = -v0.Y * e0.Z + v0.Z * e0.Y;
        p2 = -v2.Y * e0.Z + v2.Z * e0.Y;
        if (Math.Min(p0, p2) > r || Math.Max(p0, p2) < -r) return false;

        // Axis: X x e1
        r = boxHalfSize.Y * Math.Abs(e1.Z) + boxHalfSize.Z * Math.Abs(e1.Y);
        p0 = -v0.Y * e1.Z + v0.Z * e1.Y;
        p1 = -v1.Y * e1.Z + v1.Z * e1.Y;
        if (Math.Min(p0, p1) > r || Math.Max(p0, p1) < -r) return false;

        // Axis: X x e2
        r = boxHalfSize.Y * Math.Abs(e2.Z) + boxHalfSize.Z * Math.Abs(e2.Y);
        p0 = -v0.Y * e2.Z + v0.Z * e2.Y;
        p1 = -v1.Y * e2.Z + v1.Z * e2.Y;
        if (Math.Min(p0, p1) > r || Math.Max(p0, p1) < -r) return false;

        // Axis: Y x e0
        r = boxHalfSize.X * Math.Abs(e0.Z) + boxHalfSize.Z * Math.Abs(e0.X);
        p0 = v0.X * e0.Z - v0.Z * e0.X;
        p2 = v2.X * e0.Z - v2.Z * e0.X;
        if (Math.Min(p0, p2) > r || Math.Max(p0, p2) < -r) return false;

        // Axis: Y x e1
        r = boxHalfSize.X * Math.Abs(e1.Z) + boxHalfSize.Z * Math.Abs(e1.X);
        p0 = v0.X * e1.Z - v0.Z * e1.X;
        p1 = v1.X * e1.Z - v1.Z * e1.X;
        if (Math.Min(p0, p1) > r || Math.Max(p0, p1) < -r) return false;

        // Axis: Y x e2
        r = boxHalfSize.X * Math.Abs(e2.Z) + boxHalfSize.Z * Math.Abs(e2.X);
        p0 = v0.X * e2.Z - v0.Z * e2.X;
        p1 = v1.X * e2.Z - v1.Z * e2.X;
        if (Math.Min(p0, p1) > r || Math.Max(p0, p1) < -r) return false;

        // Axis: Z x e0
        r = boxHalfSize.X * Math.Abs(e0.Y) + boxHalfSize.Y * Math.Abs(e0.X);
        p0 = -v0.X * e0.Y + v0.Y * e0.X;
        p2 = -v2.X * e0.Y + v2.Y * e0.X;
        if (Math.Min(p0, p2) > r || Math.Max(p0, p2) < -r) return false;

        // Axis: Z x e1
        r = boxHalfSize.X * Math.Abs(e1.Y) + boxHalfSize.Y * Math.Abs(e1.X);
        p0 = -v0.X * e1.Y + v0.Y * e1.X;
        p1 = -v1.X * e1.Y + v1.Y * e1.X;
        if (Math.Min(p0, p1) > r || Math.Max(p0, p1) < -r) return false;

        // Axis: Z x e2
        r = boxHalfSize.X * Math.Abs(e2.Y) + boxHalfSize.Y * Math.Abs(e2.X);
        p0 = -v0.X * e2.Y + v0.Y * e2.X;
        p1 = -v1.X * e2.Y + v1.Y * e2.X;
        if (Math.Min(p0, p1) > r || Math.Max(p0, p1) < -r) return false;

        return true;
    }

    private static bool ShapesAreIdentical(VoxelShape s1, VoxelShape s2)
    {
        if (s1.Width != s2.Width || s1.Height != s2.Height || s1.Depth != s2.Depth) return false;
        if (s1.Voxels.Length != s2.Voxels.Length) return false;
        for (int i = 0; i < s1.Voxels.Length; i++)
        {
            if (s1.Voxels[i].X != s2.Voxels[i].X ||
                s1.Voxels[i].Y != s2.Voxels[i].Y ||
                s1.Voxels[i].Z != s2.Voxels[i].Z)
            {
                return false;
            }
        }
        return true;
    }

    private static List<PackedItem> RunPacking(
        int G_W, int G_H, int G_D,
        List<VoxelShape> shapes,
        HeuristicType heuristic,
        int k,
        Random random,
        Stopwatch stopwatch,
        double timeLimitSeconds)
    {
        var grid = new VoxelGrid(G_W, G_H, G_D);
        var eps = new List<ExtremePoint> { new ExtremePoint(0, 0, 0) };
        var packedItems = new List<PackedItem>();

        int currentMaxX = 0, currentMaxY = 0, currentMaxZ = 0;

        var candidates = new List<Candidate>();
        var validCandidates = new List<Candidate>();

        while (eps.Count > 0)
        {
            if (stopwatch.Elapsed.TotalSeconds >= timeLimitSeconds)
            {
                return null;
            }

            candidates.Clear();

            for (int i = 0; i < eps.Count; i++)
            {
                var ep = eps[i];
                for (int j = 0; j < shapes.Count; j++)
                {
                    var shape = shapes[j];
                    if (ep.X + shape.Width <= G_W &&
                        ep.Y + shape.Height <= G_H &&
                        ep.Z + shape.Depth <= G_D)
                    {
                        double score = ScorePlacement(ep.X, ep.Y, ep.Z, shape, heuristic, currentMaxX, currentMaxY, currentMaxZ);
                        candidates.Add(new Candidate(ep, shape, score));
                    }
                }
            }

            if (candidates.Count == 0) break;

            candidates.Sort((c1, c2) => c1.Score.CompareTo(c2.Score));

            Candidate chosen = null;
            if (k == 1)
            {
                for (int i = 0; i < candidates.Count; i++)
                {
                    var c = candidates[i];
                    if (grid.CanPlace(c.Shape, c.EP.X, c.EP.Y, c.EP.Z))
                    {
                        chosen = c;
                        break;
                    }
                }
            }
            else
            {
                validCandidates.Clear();
                for (int i = 0; i < candidates.Count; i++)
                {
                    var c = candidates[i];
                    if (grid.CanPlace(c.Shape, c.EP.X, c.EP.Y, c.EP.Z))
                    {
                        validCandidates.Add(c);
                        if (validCandidates.Count >= k) break;
                    }
                }

                if (validCandidates.Count > 0)
                {
                    int idx = random.Next(validCandidates.Count);
                    chosen = validCandidates[idx];
                }
            }

            if (chosen == null) break;

            grid.Place(chosen.Shape, chosen.EP.X, chosen.EP.Y, chosen.EP.Z);
            packedItems.Add(new PackedItem(chosen.EP.X, chosen.EP.Y, chosen.EP.Z, chosen.Shape));

            currentMaxX = Math.Max(currentMaxX, chosen.EP.X + chosen.Shape.Width);
            currentMaxY = Math.Max(currentMaxY, chosen.EP.Y + chosen.Shape.Height);
            currentMaxZ = Math.Max(currentMaxZ, chosen.EP.Z + chosen.Shape.Depth);

            int x1 = chosen.EP.X;
            int x2 = x1 + chosen.Shape.Width - 1;
            int y1 = chosen.EP.Y;
            int y2 = y1 + chosen.Shape.Height - 1;
            int z1 = chosen.EP.Z;
            int z2 = z1 + chosen.Shape.Depth - 1;

            var nextEps = new List<ExtremePoint>(eps.Count + 3);
            nextEps.Add(new ExtremePoint(x2 + 1, chosen.EP.Y, chosen.EP.Z));
            nextEps.Add(new ExtremePoint(chosen.EP.X, y2 + 1, chosen.EP.Z));
            nextEps.Add(new ExtremePoint(chosen.EP.X, chosen.EP.Y, z2 + 1));

            for (int i = 0; i < eps.Count; i++)
            {
                var ep = eps[i];
                if (ep.X >= x1 && ep.X <= x2 && ep.Y >= y1 && ep.Y <= y2 && ep.Z >= z1 && ep.Z <= z2)
                {
                    nextEps.Add(new ExtremePoint(x2 + 1, ep.Y, ep.Z));
                    nextEps.Add(new ExtremePoint(ep.X, y2 + 1, ep.Z));
                    nextEps.Add(new ExtremePoint(ep.X, ep.Y, z2 + 1));
                }
                else
                {
                    nextEps.Add(ep);
                }
            }

            eps.Clear();
            var visited = new HashSet<ExtremePoint>();
            for (int i = 0; i < nextEps.Count; i++)
            {
                var ep = nextEps[i];
                if (ep.X < G_W && ep.Y < G_H && ep.Z < G_D)
                {
                    if (!grid.IsOccupied(ep.X, ep.Y, ep.Z))
                    {
                        if (visited.Add(ep))
                        {
                            eps.Add(ep);
                        }
                    }
                }
            }
        }

        return packedItems;
    }

    private static double ScorePlacement(int gx, int gy, int gz, VoxelShape shape, HeuristicType heuristic, int currentMaxX, int currentMaxY, int currentMaxZ)
    {
        switch (heuristic)
        {
            case HeuristicType.DBL:
                return gz * 1000000.0 + gy * 1000.0 + gx;
            case HeuristicType.CornerDistance:
                double cx = gx + shape.Width;
                double cy = gy + shape.Height;
                double cz = gz + shape.Depth;
                return cx * cx + cy * cy + cz * cz;
            case HeuristicType.MinHeight:
                int newMaxZ = Math.Max(currentMaxZ, gz + shape.Depth);
                int newMaxY = Math.Max(currentMaxY, gy + shape.Height);
                int newMaxX = Math.Max(currentMaxX, gx + shape.Width);
                return newMaxZ * 1000000.0 + newMaxY * 1000.0 + newMaxX;
            default:
                return gz * 1000000.0 + gy * 1000.0 + gx;
        }
    }
