import sys

# Increase recursion depth for deep search trees
sys.setrecursionlimit(20000)

def solve():
    # Read all input from standard input once
    content = sys.stdin.read()
    lines = content.splitlines()
    
    # Find the header line U S
    header_idx = -1
    for i, line in enumerate(lines):
        if line.strip():
            header_idx = i
            break
            
    if header_idx == -1:
        print("NO SOLUTION")
        return
        
    try:
        parts = lines[header_idx].split()
        U = int(parts[0])
        S = int(parts[1])
    except (ValueError, IndexError):
        print("NO SOLUTION")
        return
        
    sets = []
    curr_line = header_idx + 1
    for _ in range(S):
        if curr_line < len(lines):
            line = lines[curr_line]
            curr_line += 1
            # Parse elements, filter to only keep valid elements 0..U-1
            elements = []
            for x in line.split():
                try:
                    val = int(x)
                    if 0 <= val < U:
                        elements.append(val)
                except ValueError:
                    pass
            sets.append(elements)
        else:
            sets.append([])
            
    # Quick check: if U is 0, empty solution is the unique exact cover
    if U == 0:
        print("SOLUTION")
        print("")
        return
        
    # Build bitmasks for each set
    masks = []
    for s in sets:
        m = 0
        for x in s:
            m |= (1 << x)
        masks.append(m)
        
    # Track which set indices cover each element
    sets_covering = [[] for _ in range(U)]
    for j, s in enumerate(sets):
        for x in s:
            sets_covering[x].append(j)
            
    # If any element cannot be covered by any set, there's no solution
    for e in range(U):
        if not sets_covering[e]:
            print("NO SOLUTION")
            return
            
    # Power of 2 lookup for bitwise operations
    pow2 = [1 << e for e in range(U)]
    
    target_mask = (1 << U) - 1
    chosen = []
    
    def solve_dfs(uncovered, covered_mask):
        if not uncovered:
            return True
            
        # MRV: Choose the uncovered element with the fewest compatible sets
        best_e = -1
        min_count = S + 1
        best_sets = []
        
        for e in uncovered:
            curr_sets = [j for j in sets_covering[e] if not (masks[j] & covered_mask)]
            count = len(curr_sets)
            if count < min_count:
                min_count = count
                best_e = e
                best_sets = curr_sets
                if min_count == 0:
                    break
                    
        if min_count == 0:
            return False
            
        # Try all compatible sets for best_e
        for j in best_sets:
            chosen.append(j)
            next_uncovered = [e for e in uncovered if not (masks[j] & pow2[e])]
            if solve_dfs(next_uncovered, covered_mask | masks[j]):
                return True
            chosen.pop()
            
        return False

    initial_uncovered = list(range(U))
    if solve_dfs(initial_uncovered, 0):
        print("SOLUTION")
        # Print sorted list of 0-based chosen set indices
        print(" ".join(map(str, sorted(chosen))))
    else:
        print("NO SOLUTION")

if __name__ == "__main__":
    solve()
