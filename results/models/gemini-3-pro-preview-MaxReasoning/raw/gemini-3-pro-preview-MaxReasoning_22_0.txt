using System;
using System.Collections.Generic;

public class Program {
    public struct PieceRotation {
        public int Width;
        public int Height;
        public uint[] RowMasks;
        public int[][] Coords;
    }

    public class Piece {
        public char Type;
        public PieceRotation[] Rotations;
    }

    private static Dictionary<char, Piece> pieces;

    public static void Main(string[] args) {
        // Read header: width height num_pieces
        string headerLine = Console.ReadLine();
        if (string.IsNullOrEmpty(headerLine)) return;

        string[] headerParts = headerLine.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
        if (headerParts.Length < 3) return;

        int width = int.Parse(headerParts[0]);
        int height = int.Parse(headerParts[1]);
        int numPieces = int.Parse(headerParts[2]);

        pieces = InitPieces();

        // Keep track of the board state
        // Board is represented as an array of uints, where row 0 is bottom
        uint[] board = new uint[height];
        uint[] tempBoard = new uint[height];

        // Adaptive weights based on game complexity (board size and total pieces)
        double wEroded = 3.418;
        double wHoles = 7.899;
        double wWells = 3.385;
        double wRowTrans = 3.217;
        double wColTrans = 9.348;
        double wLandingHeight = 4.500;

        // Marathon mode adjustments
        if (numPieces >= 50000) {
            wHoles = 13.5;
            wWells = 4.5;
            wRowTrans = 3.5;
            wColTrans = 10.5;
        }

        // Wide board adjustments
        if (width >= 20) {
            wRowTrans *= 0.6;
            wColTrans *= 0.6;
        }

        for (int p = 0; p < numPieces; p++) {
            string pieceTypeLine = Console.ReadLine();
            if (string.IsNullOrEmpty(pieceTypeLine)) break;
            
            pieceTypeLine = pieceTypeLine.Trim();
            if (pieceTypeLine.Length == 0) continue;
            char pieceType = pieceTypeLine[0];

            if (!pieces.ContainsKey(pieceType)) {
                // Invalid piece type, default response
                Console.WriteLine("0 0");
                Console.Out.Flush();
                continue;
            }

            // Find optimal move
            int bestRotation = 0;
            int bestColumn = 0;
            double bestScore = double.MinValue;

            Piece piece = pieces[pieceType];

            // Compute current fill ratio of the board
            int currentMaxHeight = 0;
            for (int r = height - 1; r >= 0; r--) {
                if (board[r] != 0) {
                    currentMaxHeight = r + 1;
                    break;
                }
            }
            double fillRatio = (double)currentMaxHeight / height;

            // Danger zone adjustments
            double currentWEroded = wEroded;
            double currentWHoles = wHoles;
            double currentWWells = wWells;
            double currentWRowTrans = wRowTrans;
            double currentWColTrans = wColTrans;
            double currentWLandingHeight = wLandingHeight;

            if (fillRatio > 0.6) {
                // Very high stack, play defensively
                currentWLandingHeight = 9.5;
                currentWEroded = 6.0;
                currentWWells = 1.0;
            } else if (fillRatio > 0.4) {
                // Moderately high stack
                currentWLandingHeight = 6.5;
                currentWEroded = 4.5;
                currentWWells = 2.0;
            }

            for (int rot = 0; rot < 4; rot++) {
                PieceRotation rInfo = piece.Rotations[rot];
                int maxCol = width - rInfo.Width;
                for (int col = 0; col <= maxCol; col++) {
                    Array.Copy(board, tempBoard, height);

                    int linesCleared;
                    double landingHeight;
                    int erodedCells;

                    bool success = PlacePiece(tempBoard, height, width, rInfo, col, out linesCleared, out landingHeight, out erodedCells);
                    if (!success) {
                        continue; // Immediate game over
                    }

                    int rowTrans = GetRowTransitions(tempBoard, height, width);
                    int colTrans = GetColTransitions(tempBoard, height, width);
                    int holes = GetHoles(tempBoard, height, width);
                    int wellSum = GetWellSum(tempBoard, height, width);

                    double score = (erodedCells * currentWEroded)
                                 - (holes * currentWHoles)
                                 - (wellSum * currentWWells)
                                 - (rowTrans * currentWRowTrans)
                                 - (colTrans * currentWColTrans)
                                 - (landingHeight * currentWLandingHeight);

                    if (score > bestScore) {
                        bestScore = score;
                        bestRotation = rot;
                        bestColumn = col;
                    }
                }
            }

            if (bestScore == double.MinValue) {
                // No moves found that avoid immediate game over
                bestRotation = 0;
                bestColumn = Math.Max(0, (width - piece.Rotations[0].Width) / 2);
            }

            // Output rotation and column
            Console.WriteLine($"{bestRotation} {bestColumn}");
            Console.Out.Flush();

            // Read result from simulator
            string result = Console.ReadLine();
            if (string.IsNullOrEmpty(result)) break;
            result = result.Trim();

            if (result == "gameover") {
                break;
            }

            // Update our internal board state with the chosen move
            int dummy1, dummy3;
            double dummy2;
            PlacePiece(board, height, width, piece.Rotations[bestRotation], bestColumn, out dummy1, out dummy2, out dummy3);
        }
    }

    private static int FindLandingRow(uint[] board, int height, int width, PieceRotation rotation, int col) {
        for (int startRow = height - 1; startRow >= 0; startRow--) {
            bool canPlaceHere = true;
            for (int r = 0; r < rotation.Height; r++) {
                int boardRow = startRow + r;
                if (boardRow < 0) {
                    canPlaceHere = false;
                    break;
                }
                if (boardRow < height) {
                    uint pieceRowMask = rotation.RowMasks[r] << col;
                    if ((board[boardRow] & pieceRowMask) != 0) {
                        canPlaceHere = false;
                        break;
                    }
                }
            }
            if (!canPlaceHere) {
                return startRow + 1;
            }
        }
        return 0;
    }

    private static bool PlacePiece(uint[] board, int height, int width, PieceRotation rotation, int col, out int linesCleared, out double landingHeight, out int erodedCells) {
        linesCleared = 0;
        landingHeight = 0;
        erodedCells = 0;

        int landingRow = FindLandingRow(board, height, width, rotation, col);

        // Place the piece
        for (int r = 0; r < rotation.Height; r++) {
            int boardRow = landingRow + r;
            if (boardRow >= height) {
                return false; // GameOver
            }
            uint pieceRowMask = rotation.RowMasks[r] << col;
            board[boardRow] |= pieceRowMask;
        }

        // Calculate landing height
        double sumRow = 0;
        for (int i = 0; i < rotation.Coords.Length; i++) {
            sumRow += landingRow + rotation.Coords[i][0];
        }
        landingHeight = sumRow / rotation.Coords.Length;

        // Check for line clears and eroded cells
        uint fullRowMask = (1u << width) - 1;
        int clearedCount = 0;
        int pieceBlocksInCleared = 0;

        for (int r = 0; r < height; r++) {
            if (board[r] == fullRowMask) {
                clearedCount++;
                if (r >= landingRow && r < landingRow + rotation.Height) {
                    pieceBlocksInCleared += PopCount(rotation.RowMasks[r - landingRow]);
                }
            }
        }

        erodedCells = clearedCount * pieceBlocksInCleared;
        linesCleared = clearedCount;

        // Perform row deletion
        if (clearedCount > 0) {
            int writeIdx = 0;
            for (int readIdx = 0; readIdx < height; readIdx++) {
                if (board[readIdx] != fullRowMask) {
                    board[writeIdx] = board[readIdx];
                    writeIdx++;
                }
            }
            while (writeIdx < height) {
                board[writeIdx] = 0;
                writeIdx++;
            }
        }

        return true;
    }

    private static int PopCount(uint x) {
        x = x - ((x >> 1) & 0x55555555);
        x = (x & 0x33333333) + ((x >> 2) & 0x33333333);
        return (int)((((x + (x >> 4)) & 0x0F0F0F0F) * 0x01010101) >> 24);
    }

    private static int GetRowTransitions(uint[] board, int height, int width) {
        int transitions = 0;
        for (int r = 0; r < height; r++) {
            uint row = board[r];
            uint prevBit = 1; // Left wall is occupied
            for (int c = 0; c < width; c++) {
                uint bit = (row >> c) & 1u;
                if (bit != prevBit) {
                    transitions++;
                }
                prevBit = bit;
            }
            if (prevBit != 1) {
                transitions++; // Transition to right wall (occupied)
            }
        }
        return transitions;
    }

    private static int GetColTransitions(uint[] board, int height, int width) {
        int transitions = 0;
        for (int c = 0; c < width; c++) {
            uint prevBit = 1; // Bottom wall is occupied
            for (int r = 0; r < height; r++) {
                uint bit = (board[r] >> c) & 1u;
                if (bit != prevBit) {
                    transitions++;
                }
                prevBit = bit;
            }
            if (prevBit != 0) {
                transitions++; // Transition to empty space above
            }
        }
        return transitions;
    }

    private static int GetHoles(uint[] board, int height, int width) {
        int holes = 0;
        for (int c = 0; c < width; c++) {
            bool foundBlock = false;
            for (int r = height - 1; r >= 0; r--) {
                bool occupied = ((board[r] >> c) & 1u) != 0;
                if (occupied) {
                    foundBlock = true;
                } else if (foundBlock) {
                    holes++;
                }
            }
        }
        return holes;
    }

    private static int GetWellSum(uint[] board, int height, int width) {
        int wellSum = 0;
        for (int c = 0; c < width; c++) {
            int depth = 0;
            for (int r = height - 1; r >= 0; r--) {
                bool occupied = ((board[r] >> c) & 1u) != 0;
                if (!occupied) {
                    bool leftOccupied = (c == 0) || (((board[r] >> (c - 1)) & 1u) != 0);
                    bool rightOccupied = (c == width - 1) || (((board[r] >> (c + 1)) & 1u) != 0);
                    if (leftOccupied && rightOccupied) {
                        depth++;
                        wellSum += depth;
                    } else {
                        depth = 0;
                    }
                } else {
                    depth = 0;
                }
            }
        }
        return wellSum;
    }

    private static Dictionary<char, Piece> InitPieces() {
        var pieces = new Dictionary<char, Piece>();
        
        var rawCoords = new Dictionary<char, int[][][]> {
            { 'I', new int[][][] {
                new int[][] { new int[] {0,0}, new int[] {0,1}, new int[] {0,2}, new int[] {0,3} },
                new int[][] { new int[] {0,0}, new int[] {1,0}, new int[] {2,0}, new int[] {3,0} },
                new int[][] { new int[] {0,0}, new int[] {0,1}, new int[] {0,2}, new int[] {0,3} },
                new int[][] { new int[] {0,0}, new int[] {1,0}, new int[] {2,0}, new int[] {3,0} }
            }},
            { 'O', new int[][][] {
                new int[][] { new int[] {0,0}, new int[] {0,1}, new int[] {1,0}, new int[] {1,1} },
                new int[][] { new int[] {0,0}, new int[] {0,1}, new int[] {1,0}, new int[] {1,1} },
                new int[][] { new int[] {0,0}, new int[] {0,1}, new int[] {1,0}, new int[] {1,1} },
                new int[][] { new int[] {0,0}, new int[] {0,1}, new int[] {1,0}, new int[] {1,1} }
            }},
            { 'T', new int[][][] {
                new int[][] { new int[] {0,0}, new int[] {0,1}, new int[] {0,2}, new int[] {1,1} },
                new int[][] { new int[] {0,0}, new int[] {1,0}, new int[] {2,0}, new int[] {1,1} },
                new int[][] { new int[] {1,0}, new int[] {1,1}, new int[] {1,2}, new int[] {0,1} },
                new int[][] { new int[] {0,1}, new int[] {1,0}, new int[] {1,1}, new int[] {2,1} }
            }},
            { 'S', new int[][][] {
                new int[][] { new int[] {0,0}, new int[] {0,1}, new int[] {1,1}, new int[] {1,2} },
                new int[][] { new int[] {0,1}, new int[] {1,0}, new int[] {1,1}, new int[] {2,0} },
                new int[][] { new int[] {0,0}, new int[] {0,1}, new int[] {1,1}, new int[] {1,2} },
                new int[][] { new int[] {0,1}, new int[] {1,0}, new int[] {1,1}, new int[] {2,0} }
            }},
            { 'Z', new int[][][] {
                new int[][] { new int[] {0,1}, new int[] {0,2}, new int[] {1,0}, new int[] {1,1} },
                new int[][] { new int[] {0,0}, new int[] {1,0}, new int[] {1,1}, new int[] {2,1} },
                new int[][] { new int[] {0,1}, new int[] {0,2}, new int[] {1,0}, new int[] {1,1} },
                new int[][] { new int[] {0,0}, new int[] {1,0}, new int[] {1,1}, new int[] {2,1} }
            }},
            { 'J', new int[][][] {
                new int[][] { new int[] {0,0}, new int[] {1,0}, new int[] {1,1}, new int[] {1,2} },
                new int[][] { new int[] {0,0}, new int[] {0,1}, new int[] {1,0}, new int[] {2,0} },
                new int[][] { new int[] {0,0}, new int[] {0,1}, new int[] {0,2}, new int[] {1,2} },
                new int[][] { new int[] {0,1}, new int[] {1,1}, new int[] {2,0}, new int[] {2,1} }
            }},
            { 'L', new int[][][] {
                new int[][] { new int[] {0,2}, new int[] {1,0}, new int[] {1,1}, new int[] {1,2} },
                new int[][] { new int[] {0,0}, new int[] {1,0}, new int[] {2,0}, new int[] {2,1} },
                new int[][] { new int[] {0,0}, new int[] {0,1}, new int[] {0,2}, new int[] {1,0} },
                new int[][] { new int[] {0,0}, new int[] {0,1}, new int[] {1,1}, new int[] {2,1} }
            }}
        };

        foreach (var kvp in rawCoords) {
            var piece = new Piece {
                Type = kvp.Key,
                Rotations = new PieceRotation[4]
            };
            for (int rot = 0; rot < 4; rot++) {
                var coords = kvp.Value[rot];
                int maxRow = 0;
                int maxCol = 0;
                foreach (var c in coords) {
                    if (c[0] > maxRow) maxRow = c[0];
                    if (c[1] > maxCol) maxCol = c[1];
                }
                int height = maxRow + 1;
                int width = maxCol + 1;
                var rowMasks = new uint[height];
                foreach (var c in coords) {
                    rowMasks[c[0]] |= (1u << c[1]);
                }
                piece.Rotations[rot] = new PieceRotation {
                    Width = width,
                    Height = height,
                    RowMasks = rowMasks,
                    Coords = coords
                };
            }
            pieces[kvp.Key] = piece;
        }
        return pieces;
    }
}
