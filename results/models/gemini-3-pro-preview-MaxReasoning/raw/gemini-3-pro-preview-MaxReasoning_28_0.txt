use std::io::{self, Read, Write, BufWriter};

struct FastReader {
    buffer: Vec<u8>,
    pos: usize,
}

impl FastReader {
    fn new() -> Self {
        let mut io_buffer = Vec::new();
        io::stdin().read_to_end(&mut io_buffer).expect("Failed to read stdin");
        Self {
            buffer: io_buffer,
            pos: 0,
        }
    }

    #[inline(always)]
    fn next_u32(&mut self) -> Option<u32> {
        while self.pos < self.buffer.len() && self.buffer[self.pos] <= b' ' {
            self.pos += 1;
        }
        if self.pos >= self.buffer.len() {
            return None;
        }
        let mut val = 0u32;
        while self.pos < self.buffer.len() && self.buffer[self.pos] > b' ' {
            val = val * 10 + (self.buffer[self.pos] - b'0') as u32;
            self.pos += 1;
        }
        Some(val)
    }
}

#[inline(always)]
fn are_connected(u1: usize, u2: usize, offsets: &[usize], edges: &[u32]) -> bool {
    let start = offsets[u1];
    let end = offsets[u1 + 1];
    let slice = &edges[start..end];
    slice.binary_search(&(u2 as u32)).is_ok()
}

fn main() {
    let mut reader = FastReader::new();
    let num_vertices = match reader.next_u32() {
        Some(val) => val as usize,
        None => return,
    };
    let num_edges = match reader.next_u32() {
        Some(val) => val as usize,
        None => return,
    };

    if num_vertices == 0 {
        let stdout = io::stdout();
        let mut writer = BufWriter::new(stdout.lock());
        writeln!(writer, "0").unwrap();
        writeln!(writer).unwrap();
        return;
    }

    let mut edge_list = Vec::with_capacity(num_edges);
    let mut degrees = vec![0u32; num_vertices];

    for _ in 0..num_edges {
        let u = match reader.next_u32() {
            Some(val) => val as usize,
            None => break,
        };
        let v = match reader.next_u32() {
            Some(val) => val as usize,
            None => break,
        };
        if u < num_vertices && v < num_vertices {
            edge_list.push((u, v));
            degrees[u] += 1;
            degrees[v] += 1;
        }
    }

    // Build CSR offsets
    let mut offsets = vec![0usize; num_vertices + 1];
    for i in 0..num_vertices {
        offsets[i + 1] = offsets[i] + degrees[i] as usize;
    }

    // Populate CSR edges
    let mut current_offsets = offsets.clone();
    let mut edges = vec![0u32; edge_list.len() * 2];

    for &(u, v) in &edge_list {
        edges[current_offsets[u]] = v as u32;
        current_offsets[u] += 1;
        edges[current_offsets[v]] = u as u32;
        current_offsets[v] += 1;
    }

    // Sort neighbors for binary search in are_connected
    for i in 0..num_vertices {
        let start = offsets[i];
        let end = offsets[i + 1];
        edges[start..end].sort_unstable();
    }

    // Free edge_list memory
    drop(edge_list);
    drop(current_offsets);

    let mut degree = degrees;
    let mut deleted = vec![false; num_vertices];
    let mut in_cover = vec![false; num_vertices];
    let mut deg1_queue = Vec::new();
    let mut deg2_queue = Vec::new();

    let mut max_degree = 0usize;
    for &d in &degree {
        if d as usize > max_degree {
            max_degree = d as usize;
        }
    }

    let mut buckets = vec![Vec::new(); max_degree + 1];

    for u in 0..num_vertices {
        let d = degree[u] as usize;
        if d <= 1 {
            deg1_queue.push(u as u32);
        } else if d == 2 {
            deg2_queue.push(u as u32);
        } else {
            buckets[d].push(u as u32);
        }
    }

    while !deg1_queue.is_empty() || !deg2_queue.is_empty() || max_degree > 1 {
        // 1. Process all degree 0 and 1 reductions
        if !deg1_queue.is_empty() {
            while let Some(u) = deg1_queue.pop() {
                let u = u as usize;
                if deleted[u] {
                    continue;
                }
                let deg = degree[u];
                if deg == 0 {
                    deleted[u] = true;
                } else if deg == 1 {
                    // Find unique active neighbor of u
                    let mut active_neighbor = None;
                    let start = offsets[u];
                    let end = offsets[u + 1];
                    for idx in start..end {
                        let v = edges[idx] as usize;
                        if !deleted[v] {
                            active_neighbor = Some(v);
                            break;
                        }
                    }
                    if let Some(w) = active_neighbor {
                        in_cover[w] = true;
                        deleted[w] = true;

                        // Decrement neighbor degrees
                        let w_start = offsets[w];
                        let w_end = offsets[w + 1];
                        for idx in w_start..w_end {
                            let v = edges[idx] as usize;
                            if !deleted[v] {
                                degree[v] -= 1;
                                let new_d = degree[v];
                                if new_d <= 1 {
                                    deg1_queue.push(v as u32);
                                } else if new_d == 2 {
                                    deg2_queue.push(v as u32);
                                } else {
                                    buckets[new_d as usize].push(v as u32);
                                }
                            }
                        }
                    }
                }
            }
            continue;
        }

        // 2. Process one degree 2 triangle reduction
        if !deg2_queue.is_empty() {
            if let Some(u) = deg2_queue.pop() {
                let u = u as usize;
                if deleted[u] || degree[u] != 2 {
                    continue;
                }
                // Find the two active neighbors
                let mut neighbors = [0usize; 2];
                let mut count = 0;
                let start = offsets[u];
                let end = offsets[u + 1];
                for idx in start..end {
                    let v = edges[idx] as usize;
                    if !deleted[v] {
                        if count < 2 {
                            neighbors[count] = v;
                        }
                        count += 1;
                    }
                }
                if count == 2 {
                    let u1 = neighbors[0];
                    let u2 = neighbors[1];
                    if are_connected(u1, u2, &offsets, &edges) {
                        in_cover[u1] = true;
                        in_cover[u2] = true;
                        deleted[u1] = true;
                        deleted[u2] = true;
                        deleted[u] = true;

                        // Decrement degrees of neighbors of u1 and u2
                        for &w in &[u1, u2] {
                            let w_start = offsets[w];
                            let w_end = offsets[w + 1];
                            for idx in w_start..w_end {
                                let v = edges[idx] as usize;
                                if !deleted[v] {
                                    degree[v] -= 1;
                                    let new_d = degree[v];
                                    if new_d <= 1 {
                                        deg1_queue.push(v as u32);
                                    } else if new_d == 2 {
                                        deg2_queue.push(v as u32);
                                    } else {
                                        buckets[new_d as usize].push(v as u32);
                                    }
                                }
                            }
                        }
                    }
                }
            }
            continue;
        }

        // 3. Greedy choice
        if max_degree > 1 {
            while max_degree > 1 && buckets[max_degree].is_empty() {
                max_degree -= 1;
            }
            if max_degree <= 1 {
                continue;
            }
            if let Some(u) = buckets[max_degree].pop() {
                let u = u as usize;
                if deleted[u] || degree[u] as usize != max_degree {
                    continue;
                }

                // Add u to cover
                in_cover[u] = true;
                deleted[u] = true;

                // Decrement neighbor degrees
                let start = offsets[u];
                let end = offsets[u + 1];
                for idx in start..end {
                    let v = edges[idx] as usize;
                    if !deleted[v] {
                        degree[v] -= 1;
                        let new_d = degree[v];
                        if new_d <= 1 {
                            deg1_queue.push(v as u32);
                        } else if new_d == 2 {
                            deg2_queue.push(v as u32);
                        } else {
                            buckets[new_d as usize].push(v as u32);
                        }
                    }
                }
            }
        }
    }

    // Local Search Refinement
    let start_time = std::time::Instant::now();
    let mut num_uncovered_neighbors = vec![0u32; num_vertices];
    for u in 0..num_vertices {
        let start = offsets[u];
        let end = offsets[u + 1];
        let mut count = 0;
        for idx in start..end {
            let v = edges[idx] as usize;
            if !in_cover[v] {
                count += 1;
            }
        }
        num_uncovered_neighbors[u] = count;
    }

    // Remove initially redundant vertices
    for u in 0..num_vertices {
        if in_cover[u] && num_uncovered_neighbors[u] == 0 {
            in_cover[u] = false;
            let start = offsets[u];
            let end = offsets[u + 1];
            for idx in start..end {
                let v = edges[idx] as usize;
                num_uncovered_neighbors[v] += 1;
            }
        }
    }

    let mut improved = true;
    let mut passes = 0;
    while improved && passes < 15 {
        if start_time.elapsed().as_secs_f64() > 4.0 {
            break;
        }
        improved = false;
        for u in 0..num_vertices {
            if in_cover[u] && num_uncovered_neighbors[u] == 1 {
                // Find the unique neighbor w not in cover
                let mut w_opt = None;
                let start = offsets[u];
                let end = offsets[u + 1];
                for idx in start..end {
                    let v = edges[idx] as usize;
                    if !in_cover[v] {
                        w_opt = Some(v);
                        break;
                    }
                }

                if let Some(w) = w_opt {
                    // Try exchanging u and w
                    in_cover[u] = false;
                    let u_start = offsets[u];
                    let u_end = offsets[u + 1];
                    for idx in u_start..u_end {
                        let v = edges[idx] as usize;
                        num_uncovered_neighbors[v] += 1;
                    }

                    in_cover[w] = true;
                    let w_start = offsets[w];
                    let w_end = offsets[w + 1];
                    for idx in w_start..w_end {
                        let v = edges[idx] as usize;
                        num_uncovered_neighbors[v] -= 1;
                    }

                    // Search for redundant x in neighbors of w (excluding u)
                    let mut x_opt = None;
                    for idx in w_start..w_end {
                        let x = edges[idx] as usize;
                        if x != u && in_cover[x] && num_uncovered_neighbors[x] == 0 {
                            x_opt = Some(x);
                            break;
                        }
                    }

                    if let Some(x) = x_opt {
                        in_cover[x] = false;
                        let x_start = offsets[x];
                        let x_end = offsets[x + 1];
                        for idx in x_start..x_end {
                            let v = edges[idx] as usize;
                            num_uncovered_neighbors[v] += 1;
                        }
                        improved = true;
                    } else {
                        // Undo exchange
                        in_cover[w] = false;
                        for idx in w_start..w_end {
                            let v = edges[idx] as usize;
                            num_uncovered_neighbors[v] += 1;
                        }
                        in_cover[u] = true;
                        for idx in u_start..u_end {
                            let v = edges[idx] as usize;
                            num_uncovered_neighbors[v] -= 1;
                        }
                    }
                }
            }
        }
        passes += 1;
    }

    // Safety validation step: ensure all edges are covered
    for u in 0..num_vertices {
        let start = offsets[u];
        let end = offsets[u + 1];
        for idx in start..end {
            let v = edges[idx] as usize;
            if !in_cover[u] && !in_cover[v] {
                in_cover[u] = true;
            }
        }
    }

    // Final cleanup of redundant vertices
    for u in 0..num_vertices {
        if in_cover[u] {
            let mut redundant = true;
            let start = offsets[u];
            let end = offsets[u + 1];
            for idx in start..end {
                let v = edges[idx] as usize;
                if !in_cover[v] {
                    redundant = false;
                    break;
                }
            }
            if redundant {
                in_cover[u] = false;
            }
        }
    }

    let cover: Vec<usize> = (0..num_vertices)
        .filter(|&u| in_cover[u])
        .collect();

    let stdout = io::stdout();
    let mut writer = BufWriter::new(stdout.lock());
    writeln!(writer, "{}", cover.len()).unwrap();
    for (i, &u) in cover.iter().enumerate() {
        if i > 0 {
            write!(writer, " ").unwrap();
        }
        write!(writer, "{}", u).unwrap();
    }
    writeln!(writer).unwrap();
}
