// ============================================================================
// Suffix Automaton (SAM) Solver for Small, Medium, and Large Scales
// Time Complexity: O(Sum of |S_i| * log(Sigma)), Space Complexity: O(Min(|S_i|))
// ============================================================================

struct SAM {
    struct State {
        int len, link;
        std::map<char, int> next;
        int first_pos;
    };
    std::vector<State> st;
    int sz, last;
    
    void init(int maxlen) {
        st.clear();
        st.resize(maxlen * 2 + 10);
        st[0].len = 0;
        st[0].link = -1;
        st[0].first_pos = -1;
        sz = 1;
        last = 0;
    }
    
    void extend(char c, int pos) {
        int cur = sz++;
        st[cur].len = st[last].len + 1;
        st[cur].first_pos = pos;
        int p = last;
        while (p != -1 && !st[p].next.count(c)) {
            st[p].next[c] = cur;
            p = st[p].link;
        }
        if (p == -1) {
            st[cur].link = 0;
        } else {
            int q = st[p].next[c];
            if (st[p].len + 1 == st[q].len) {
                st[cur].link = q;
            } else {
                int clone = sz++;
                st[clone].len = st[p].len + 1;
                st[clone].next = st[q].next;
                st[clone].link = st[q].link;
                st[clone].first_pos = st[q].first_pos;
                while (p != -1 && st[p].next[c] == q) {
                    st[p].next[c] = clone;
                    p = st[p].link;
                }
                st[q].link = st[cur].link = clone;
            }
        }
        last = cur;
    }
};

std::string solve_sam(const std::vector<std::string>& S) {
    int N = S.size();
    if (N == 0) return "";
    if (N == 1) return S[0];
    
    int s0_idx = 0;
    for (int i = 1; i < N; ++i) {
        if (S[i].size() < S[s0_idx].size()) {
            s0_idx = i;
        }
    }
    const std::string& S0 = S[s0_idx];
    
    SAM sam;
    sam.init(S0.size());
    for (int i = 0; i < S0.size(); ++i) {
        sam.extend(S0[i], i);
    }
    
    int num_states = sam.sz;
    std::vector<int> min_match(num_states);
    for (int i = 0; i < num_states; ++i) {
        min_match[i] = sam.st[i].len;
    }
    
    std::vector<int> order(num_states);
    for (int i = 0; i < num_states; ++i) order[i] = i;
    std::sort(order.begin(), order.end(), [&](int a, int b) {
        return sam.st[a].len > sam.st[b].len;
    });
    
    for (int i = 0; i < N; ++i) {
        if (i == s0_idx) continue;
        const std::string& Si = S[i];
        std::vector<int> curr_match(num_states, 0);
        int curr = 0, curr_len = 0;
        for (char c : Si) {
            while (curr != -1 && !sam.st[curr].next.count(c)) {
                curr = sam.st[curr].link;
                if (curr != -1) curr_len = sam.st[curr].len;
            }
            if (curr == -1) {
                curr = 0;
                curr_len = 0;
            } else {
                curr = sam.st[curr].next[c];
                curr_len++;
            }
            curr_match[curr] = std::max(curr_match[curr], curr_len);
        }
        
        for (int u : order) {
            if (u == 0) continue;
            int link = sam.st[u].link;
            curr_match[link] = std::max(curr_match[link], std::min(sam.st[link].len, curr_match[u]));
        }
        
        for (int j = 0; j < num_states; ++j) {
            min_match[j] = std::min(min_match[j], curr_match[j]);
        }
    }
    
    int max_len = 0;
    int best_u = 0;
    for (int i = 1; i < num_states; ++i) {
        if (min_match[i] > max_len) {
            max_len = min_match[i];
            best_u = i;
        }
    }
    
    if (max_len == 0) return "";
    int end_pos = sam.st[best_u].first_pos;
    return S0.substr(end_pos - max_len + 1, max_len);
}

// ============================================================================
// Parallel Rolling Hash Solver for Extreme Scale (10MB+ strings, 100+ strings)
// Time Complexity: O(Sum of |S_i| log |S_i| / Cores), Space Complexity: O(Peak Memory)
// ============================================================================

struct RollingHash {
    std::unique_ptr<uint64_t[]> P;
    size_t n = 0;
    static std::vector<uint64_t> Pow;
    static uint64_t B;
    
    static void init_powers(size_t max_len) {
        if (Pow.size() >= max_len + 1) return;
        size_t old_size = Pow.size();
        Pow.resize(max_len + 1);
        if (old_size == 0) {
            Pow[0] = 1;
            old_size = 1;
        }
        for (size_t i = old_size; i <= max_len; ++i) {
            Pow[i] = Pow[i-1] * B;
        }
    }
    
    void init(const std::string& s) {
        n = s.size();
        P = std::unique_ptr<uint64_t[]>(new uint64_t[n + 1]);
        P[0] = 0;
        const uint8_t* data = reinterpret_cast<const uint8_t*>(s.data());
        uint64_t cur = 0;
        for (size_t i = 0; i < n; ++i) {
            cur = cur * B + data[i];
            P[i+1] = cur;
        }
    }
    
    uint64_t get_hash(size_t i, size_t L) const {
        return P[i+L] - P[i] * Pow[L];
    }
};

uint64_t RollingHash::B = 0;
std::vector<uint64_t> RollingHash::Pow;

struct Candidate {
    uint64_t hash;
    int start_idx;
    bool operator<(const Candidate& o) const {
        return hash < o.hash;
    }
    bool operator==(const Candidate& o) const {
        return hash == o.hash;
    }
};

void radix_sort_uint64(uint64_t* data, uint64_t* temp, size_t n) {
    const int BITS = 16;
    const int BUCKETS = 1 << BITS;
    const uint64_t MASK = BUCKETS - 1;
    
    size_t count[BUCKETS];
    uint64_t* src = data;
    uint64_t* dst = temp;
    
    for (int shift = 0; shift < 64; shift += BITS) {
        std::fill(count, count + BUCKETS, 0);
        for (size_t i = 0; i < n; ++i) {
            count[(src[i] >> shift) & MASK]++;
        }
        size_t pref = 0;
        for (int i = 0; i < BUCKETS; ++i) {
            size_t c = count[i];
            count[i] = pref;
            pref += c;
        }
        for (size_t i = 0; i < n; ++i) {
            size_t digit = (src[i] >> shift) & MASK;
            dst[count[digit]++] = src[i];
        }
        std::swap(src, dst);
    }
}

void radix_sort_candidate(Candidate* data, Candidate* temp, size_t n) {
    const int BITS = 16;
    const int BUCKETS = 1 << BITS;
    const uint64_t MASK = BUCKETS - 1;
    
    size_t count[BUCKETS];
    Candidate* src = data;
    Candidate* dst = temp;
    
    for (int shift = 0; shift < 64; shift += BITS) {
        std::fill(count, count + BUCKETS, 0);
        for (size_t i = 0; i < n; ++i) {
            count[(src[i].hash >> shift) & MASK]++;
        }
        size_t pref = 0;
        for (int i = 0; i < BUCKETS; ++i) {
            size_t c = count[i];
            count[i] = pref;
            pref += c;
        }
        for (size_t i = 0; i < n; ++i) {
            size_t digit = (src[i].hash >> shift) & MASK;
            dst[count[digit]++] = src[i];
        }
        std::swap(src, dst);
    }
}

template<typename F>
void parallel_for(int start, int end, int num_threads, F&& f) {
    int actual_threads = std::min(num_threads, end - start);
    if (actual_threads <= 1) {
        for (int i = start; i < end; ++i) {
            f(i);
        }
        return;
    }
    std::atomic<int> current_idx(start);
    std::vector<std::thread> threads;
    for (int t = 0; t < actual_threads; ++t) {
        threads.emplace_back([&]() {
            while (true) {
                int idx = current_idx.fetch_add(1);
                if (idx >= end) break;
                f(idx);
            }
        });
    }
    for (auto& t : threads) {
        t.join();
    }
}

bool check_length(int L, const std::vector<std::string>& S, const RollingHash& rh0, int num_threads, std::string& out_str) {
    int N = S.size();
    const std::string& S0 = S[0];
    size_t count0 = S0.size() - L + 1;
    
    std::vector<Candidate> current_candidates(count0);
    for (size_t j = 0; j < count0; ++j) {
        current_candidates[j] = { rh0.get_hash(j, L), static_cast<int>(j) };
    }
    
    std::vector<Candidate> temp_cand(count0);
    radix_sort_candidate(current_candidates.data(), temp_cand.data(), count0);
    
    auto last = std::unique(current_candidates.begin(), current_candidates.end());
    current_candidates.erase(last, current_candidates.end());
    
    int next_str_idx = 1;
    
    // Process sequentially until candidate size < 1000 to avoid memory/CPU overhead
    while (next_str_idx < N && current_candidates.size() >= 1000) {
        int i = next_str_idx++;
        
        RollingHash rhi;
        rhi.init(S[i]);
        size_t count_i = S[i].size() - L + 1;
        
        std::unique_ptr<uint64_t[]> hashes_i(new uint64_t[count_i]);
        for (size_t j = 0; j < count_i; ++j) {
            hashes_i[j] = rhi.get_hash(j, L);
        }
        
        std::unique_ptr<uint64_t[]> temp_i(new uint64_t[count_i]);
        radix_sort_uint64(hashes_i.get(), temp_i.get(), count_i);
        
        size_t unique_count = std::unique(hashes_i.get(), hashes_i.get() + count_i) - hashes_i.get();
        
        std::vector<Candidate> next_res;
        auto it1 = current_candidates.begin();
        uint64_t* it2 = hashes_i.get();
        uint64_t* end2 = hashes_i.get() + unique_count;
        
        while (it1 != current_candidates.end() && it2 != end2) {
            if (it1->hash == *it2) {
                next_res.push_back(*it1);
                it1++;
                it2++;
            } else if (it1->hash < *it2) {
                it1++;
            } else {
                it2++;
            }
        }
        current_candidates = std::move(next_res);
        if (current_candidates.empty()) {
            return false;
        }
    }
    
    if (next_str_idx >= N) {
        // We checked all strings
        for (const auto& cand : current_candidates) {
            std::string_view sub(S0.data() + cand.start_idx, L);
            bool ok = true;
            for (int i = 1; i < N; ++i) {
                if (S[i].find(sub) == std::string::npos) {
                    ok = false;
                    break;
                }
            }
            if (ok) {
                out_str = std::string(sub);
                return true;
            }
        }
        return false;
    }
    
    std::atomic<bool> abort_search(false);
    
    // Check remaining strings in parallel using find (highly optimized parallel search)
    parallel_for(next_str_idx, N, num_threads, [&](int i) {
        if (abort_search.load()) return;
        
        bool has_any = false;
        for (const auto& cand : current_candidates) {
            std::string_view sub(S0.data() + cand.start_idx, L);
            if (S[i].find(sub) != std::string::npos) {
                has_any = true;
                break;
            }
        }
        if (!has_any) {
            abort_search.store(true);
        }
    });
    
    if (abort_search.load()) {
        return false;
    }
    
    for (const auto& cand : current_candidates) {
        std::string_view sub(S0.data() + cand.start_idx, L);
        bool ok = true;
        for (int i = next_str_idx; i < N; ++i) {
            if (S[i].find(sub) == std::string::npos) {
                ok = false;
                break;
            }
        }
        if (ok) {
            out_str = std::string(sub);
            return true;
        }
    }
    
    return false;
}

std::string solve_parallel_hash(const std::vector<std::string>& S) {
    int N = S.size();
    if (N == 0) return "";
    if (N == 1) return S[0];
    
    int s0_idx = 0;
    for (int i = 1; i < N; ++i) {
        if (S[i].size() < S[s0_idx].size()) {
            s0_idx = i;
        }
    }
    
    std::vector<std::string> strings = S;
    std::swap(strings[0], strings[s0_idx]);
    
    const std::string& S0 = strings[0];
    size_t M = S0.size();
    
    std::random_device rd;
    std::mt19937_64 rng(rd());
    RollingHash::B = rng() | 1;
    
    size_t max_len = 0;
    for (const auto& s : strings) {
        max_len = std::max(max_len, s.size());
    }
    RollingHash::init_powers(max_len);
    
    RollingHash rh0;
    rh0.init(S0);
    
    int low = 1, high = M;
    std::string best_lcs = "";
    
    size_t mem_per_thread = max_len * 16;
    int num_threads = std::thread::hardware_concurrency();
    if (num_threads < 1) num_threads = 1;
    if (mem_per_thread > 0) {
        // Safe limit of 32 GB memory allocation for threads
        int safe_threads = static_cast<int>((32ULL * 1024 * 1024 * 1024ULL) / mem_per_thread);
        if (safe_threads < 1) safe_threads = 1;
        num_threads = std::min(num_threads, safe_threads);
    }
    
    while (low <= high) {
        int mid = low + (high - low) / 2;
        std::string current_match = "";
        if (check_length(mid, strings, rh0, num_threads, current_match)) {
            best_lcs = std::move(current_match);
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }
    
    return best_lcs;
}

// Adaptive Orchestration
// ============================================================================

std::string solve(std::vector<std::string>& S) {
    // 1. Remove duplicate strings to optimize search space
    std::sort(S.begin(), S.end());
    auto last = std::unique(S.begin(), S.end());
    S.erase(last, S.end());
    
    int N = S.size();
    if (N == 0) return "";
    if (N == 1) return S[0];
    
    size_t total_len = 0;
    for (const auto& s : S) {
        total_len += s.size();
    }
    
    // Threshold of 1MB total length to switch from deterministic SAM to Parallel Rolling Hash
    if (total_len <= 1000000ULL) {
        return solve_sam(S);
    } else {
        return solve_parallel_hash(S);
    }
}

int main() {
    // Optimize standard input/output streams for speed
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    
    int N;
    if (!(std::cin >> N)) {
        std::cout << "\n";
        return 0;
    }
    
    std::vector<std::string> S;
    S.reserve(N);
    std::string line;
    std::getline(std::cin, line); // consume newline after N
    
    for (int i = 0; i < N; ++i) {
        if (std::getline(std::cin, line)) {
            if (!line.empty() && line.back() == '\r') {
                line.pop_back();
            }
            S.push_back(std::move(line));
        }
    }
    
    std::string lcs = solve(S);
    std::cout << lcs << "\n";
    
    return 0;
}
