use std::f64::consts::PI;
use std::io::{self, Read};

#[derive(Clone, Copy, Debug)]
struct Point {
    x: f64,
    y: f64,
}

impl std::ops::Sub for Point {
    type Output = Point;
    fn sub(self, other: Point) -> Point {
        Point { x: self.x - other.x, y: self.y - other.y }
    }
}

impl std::ops::Add for Point {
    type Output = Point;
    fn add(self, other: Point) -> Point {
        Point { x: self.x + other.x, y: self.y + other.y }
    }
}

fn ccw(a: Point, b: Point, c: Point) -> f64 {
    (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x)
}

fn segments_strictly_intersect(a: Point, b: Point, c: Point, d: Point) -> bool {
    let cp1 = ccw(a, b, c);
    let cp2 = ccw(a, b, d);
    let cp3 = ccw(c, d, a);
    let cp4 = ccw(c, d, b);
    
    let epsilon = 1e-9;
    let cross1 = (cp1 > epsilon && cp2 < -epsilon) || (cp1 < -epsilon && cp2 > epsilon);
    let cross2 = (cp3 > epsilon && cp4 < -epsilon) || (cp3 < -epsilon && cp4 > epsilon);
    cross1 && cross2
}

fn point_on_segment(p: Point, a: Point, b: Point) -> bool {
    let epsilon = 1e-9;
    if ccw(a, b, p).abs() > epsilon {
        return false;
    }
    let min_x = a.x.min(b.x) - epsilon;
    let max_x = a.x.max(b.x) + epsilon;
    let min_y = a.y.min(b.y) - epsilon;
    let max_y = a.y.max(b.y) + epsilon;
    p.x >= min_x && p.x <= max_x && p.y >= min_y && p.y <= max_y
}

fn point_strictly_in_polygon(p: Point, poly: &[Point]) -> bool {
    let n = poly.len();
    for i in 0..n {
        let a = poly[i];
        let b = poly[(i + 1) % n];
        if point_on_segment(p, a, b) {
            return false;
        }
    }
    
    let mut inside = false;
    for i in 0..n {
        let j = (i + 1) % n;
        let pi = poly[i];
        let pj = poly[j];
        if ((pi.y > p.y) != (pj.y > p.y)) &&
           (p.x < (pj.x - pi.x) * (p.y - pi.y) / (pj.y - pi.y) + pi.x) {
            inside = !inside;
        }
    }
    inside
}

fn point_in_or_on_polygon(p: Point, poly: &[Point]) -> bool {
    let n = poly.len();
    for i in 0..n {
        let a = poly[i];
        let b = poly[(i + 1) % n];
        if point_on_segment(p, a, b) {
            return true;
        }
    }
    
    let mut inside = false;
    for i in 0..n {
        let j = (i + 1) % n;
        let pi = poly[i];
        let pj = poly[j];
        if ((pi.y > p.y) != (pj.y > p.y)) &&
           (p.x < (pj.x - pi.x) * (p.y - pi.y) / (pj.y - pi.y) + pi.x) {
            inside = !inside;
        }
    }
    inside
}

fn compute_area(poly: &[Point]) -> f64 {
    let mut area = 0.0;
    let n = poly.len();
    for i in 0..n {
        let p1 = poly[i];
        let p2 = poly[(i + 1) % n];
        area += p1.x * p2.y - p2.x * p1.y;
    }
    (area * 0.5).abs()
}

fn compute_centroid(poly: &[Point]) -> Point {
    let mut cx = 0.0;
    let mut cy = 0.0;
    let mut area = 0.0;
    let n = poly.len();
    for i in 0..n {
        let p1 = poly[i];
        let p2 = poly[(i + 1) % n];
        let factor = p1.x * p2.y - p2.x * p1.y;
        cx += (p1.x + p2.x) * factor;
        cy += (p1.y + p2.y) * factor;
        area += factor;
    }
    area *= 0.5;
    if area.abs() > 1e-9 {
        Point {
            x: cx / (6.0 * area),
            y: cy / (6.0 * area),
        }
    } else {
        let mut sx = 0.0;
        let mut sy = 0.0;
        for p in poly {
            sx += p.x;
            sy += p.y;
        }
        Point { x: sx / n as f64, y: sy / n as f64 }
    }
}

fn is_ccw(poly: &[Point]) -> bool {
    let mut area = 0.0;
    let n = poly.len();
    for i in 0..n {
        let p1 = poly[i];
        let p2 = poly[(i + 1) % n];
        area += p1.x * p2.y - p2.x * p1.y;
    }
    area > 0.0
}

fn point_in_triangle(p: Point, a: Point, b: Point, c: Point) -> bool {
    let s1 = ccw(a, b, p);
    let s2 = ccw(b, c, p);
    let s3 = ccw(c, a, p);
    let has_neg = (s1 < -1e-9) || (s2 < -1e-9) || (s3 < -1e-9);
    let has_pos = (s1 > 1e-9) || (s2 > 1e-9) || (s3 > 1e-9);
    !(has_neg && has_pos)
}

fn find_interior_point(poly: &[Point]) -> Point {
    let n = poly.len();
    if n == 3 {
        return Point {
            x: (poly[0].x + poly[1].x + poly[2].x) / 3.0,
            y: (poly[0].y + poly[1].y + poly[2].y) / 3.0,
        };
    }
    let ccw_oriented = is_ccw(poly);
    for i in 0..n {
        let prev = (i + n - 1) % n;
        let next = (i + 1) % n;
        let a = poly[prev];
        let b = poly[i];
        let c = poly[next];
        
        let val = ccw(a, b, c);
        let is_convex = if ccw_oriented { val > 1e-9 } else { val < -1e-9 };
        
        if is_convex {
            let mut is_ear = true;
            for j in 0..n {
                if j == prev || j == i || j == next {
                    continue;
                }
                if point_in_triangle(poly[j], a, b, c) {
                    is_ear = false;
                    break;
                }
            }
            if is_ear {
                return Point {
                    x: (a.x + b.x + c.x) / 3.0,
                    y: (a.y + b.y + c.y) / 3.0,
                };
            }
        }
    }
    let mut sx = 0.0;
    let mut sy = 0.0;
    for p in poly {
        sx += p.x;
        sy += p.y;
    }
    Point { x: sx / n as f64, y: sy / n as f64 }
}

fn get_bbox(poly: &[Point]) -> (f64, f64, f64, f64) {
    let mut xmin = poly[0].x;
    let mut xmax = poly[0].x;
    let mut ymin = poly[0].y;
    let mut ymax = poly[0].y;
    for &p in poly.iter().skip(1) {
        if p.x < xmin { xmin = p.x; }
        if p.x > xmax { xmax = p.x; }
        if p.y < ymin { ymin = p.y; }
        if p.y > ymax { ymax = p.y; }
    }
    (xmin, xmax, ymin, ymax)
}

fn rotate_point(p: Point, angle: f64) -> Point {
    let cos_a = angle.cos();
    let sin_a = angle.sin();
    Point {
        x: p.x * cos_a - p.y * sin_a,
        y: p.x * sin_a + p.y * cos_a,
    }
}

fn rotate_polygon(poly: &[Point], angle: f64) -> Vec<Point> {
    let cos_a = angle.cos();
    let sin_a = angle.sin();
    poly.iter()
        .map(|p| Point {
            x: p.x * cos_a - p.y * sin_a,
            y: p.x * sin_a + p.y * cos_a,
        })
        .collect()
}

fn is_axis_aligned_rect(poly: &[Point]) -> bool {
    if poly.len() != 4 {
        return false;
    }
    for i in 0..4 {
        let p1 = poly[i];
        let p2 = poly[(i + 1) % 4];
        let dx = (p1.x - p2.x).abs();
        let dy = (p1.y - p2.y).abs();
        if dx > 1e-9 && dy > 1e-9 {
            return false;
        }
    }
    true
}

fn segments_intersect_any(p1: Point, p2: Point, poly: &[Point]) -> bool {
    let n = poly.len();
    for i in 0..n {
        let s1 = poly[i];
        let s2 = poly[(i + 1) % n];
        if segments_strictly_intersect(p1, p2, s1, s2) {
            return true;
        }
    }
    false
}

fn is_contained_in_stock(
    p: &[Point],
    ip: Point,
    stock: &[Point],
    stock_is_rect: bool,
    stock_bbox: (f64, f64, f64, f64),
) -> bool {
    if stock_is_rect {
        let (xmin, xmax, ymin, ymax) = stock_bbox;
        let mut px_min = p[0].x;
        let mut px_max = p[0].x;
        let mut py_min = p[0].y;
        let mut py_max = p[0].y;
        for &pt in p.iter().skip(1) {
            if pt.x < px_min { px_min = pt.x; }
            if pt.x > px_max { px_max = pt.x; }
            if pt.y < py_min { py_min = pt.y; }
            if pt.y > py_max { py_max = pt.y; }
        }
        return px_min >= xmin - 1e-9 && px_max <= xmax + 1e-9 &&
               py_min >= ymin - 1e-9 && py_max <= ymax + 1e-9;
    }
    
    if !point_strictly_in_polygon(ip, stock) {
        return false;
    }
    
    for &pt in p {
        if !point_in_or_on_polygon(pt, stock) {
            return false;
        }
    }
    
    let n_p = p.len();
    for i in 0..n_p {
        let p1 = p[i];
        let p2 = p[(i + 1) % n_p];
        if segments_intersect_any(p1, p2, stock) {
            return false;
        }
    }
    true
}

fn polygons_overlap(
    a: &[Point],
    b: &[Point],
    ia: Point,
    ib: Point,
) -> bool {
    let n_a = a.len();
    for i in 0..n_a {
        let a1 = a[i];
        let a2 = a[(i + 1) % n_a];
        if segments_intersect_any(a1, a2, b) {
            return true;
        }
    }
    
    for &p in a {
        if point_strictly_in_polygon(p, b) {
            return true;
        }
    }
    
    for &p in b {
        if point_strictly_in_polygon(p, a) {
            return true;
        }
    }
    
    if point_strictly_in_polygon(ia, b) {
        return true;
    }
    
    if point_strictly_in_polygon(ib, a) {
        return true;
    }
    
    false
}

#[derive(Clone)]
struct SpatialGrid {
    cell_size: f64,
    min_x: f64,
    min_y: f64,
    cols: usize,
    rows: usize,
    cells: Vec<Vec<usize>>,
}

impl SpatialGrid {
    fn new(min_x: f64, max_x: f64, min_y: f64, max_y: f64, num_cells_per_dim: usize) -> Self {
        let w = max_x - min_x;
        let h = max_y - min_y;
        let cell_size = (w.max(h) / num_cells_per_dim as f64).max(1e-5);
        let cols = ((w / cell_size).ceil() as usize).max(1);
        let rows = ((h / cell_size).ceil() as usize).max(1);
        Self {
            cell_size,
            min_x,
            min_y,
            cols,
            rows,
            cells: vec![Vec::new(); cols * rows],
        }
    }

    fn get_cell_range(&self, xmin: f64, xmax: f64, ymin: f64, ymax: f64) -> (usize, usize, usize, usize) {
        let col_min = (((xmin - self.min_x) / self.cell_size).floor() as isize).max(0) as usize;
        let col_max = (((xmax - self.min_x) / self.cell_size).floor() as isize).max(0) as usize;
        let col_max = col_max.min(self.cols - 1);
        let row_min = (((ymin - self.min_y) / self.cell_size).floor() as isize).max(0) as usize;
        let row_max = (((ymax - self.min_y) / self.cell_size).floor() as isize).max(0) as usize;
        let row_max = row_max.min(self.rows - 1);
        (col_min, col_max, row_min, row_max)
    }

    fn insert(&mut self, idx: usize, xmin: f64, xmax: f64, ymin: f64, ymax: f64) {
        let (col_min, col_max, row_min, row_max) = self.get_cell_range(xmin, xmax, ymin, ymax);
        for r in row_min..=row_max {
            for c in col_min..=col_max {
                let cell_idx = r * self.cols + c;
                self.cells[cell_idx].push(idx);
            }
        }
    }

    fn query(&self, xmin: f64, xmax: f64, ymin: f64, ymax: f64, out: &mut Vec<usize>) {
        out.clear();
        let (col_min, col_max, row_min, row_max) = self.get_cell_range(xmin, xmax, ymin, ymax);
        for r in row_min..=row_max {
            for c in col_min..=col_max {
                let cell_idx = r * self.cols + c;
                for &idx in &self.cells[cell_idx] {
                    out.push(idx);
                }
            }
        }
        out.sort_unstable();
        out.dedup();
    }
}

#[allow(dead_code)]
struct InputPiece {
    id: usize,
    vertices: Vec<Point>,
    area: f64,
    radius: f64,
    centroid: Point,
    interior_point: Point,
}

#[allow(dead_code)]
struct PlacedPiece {
    rotation: f64,
    translation: Point,
    vertices: Vec<Point>,
    centroid: Point,
    radius: f64,
    interior_point: Point,
}

struct Stock {
    placed_pieces: Vec<PlacedPiece>,
    grid: SpatialGrid,
}

fn get_anchors(stock_vertices: &[Point], placed: &[PlacedPiece], l: usize) -> Vec<Point> {
    let mut pool = Vec::new();
    for &p in stock_vertices {
        pool.push(p);
    }
    let start = if placed.len() > l { placed.len() - l } else { 0 };
    for p in &placed[start..] {
        for &pt in &p.vertices {
            pool.push(pt);
        }
    }
    pool.sort_unstable_by(|a, b| {
        a.y.partial_cmp(&b.y).unwrap_or(std::cmp::Ordering::Equal)
            .then_with(|| a.x.partial_cmp(&b.x).unwrap_or(std::cmp::Ordering::Equal))
    });
    pool.dedup_by(|a, b| (a.x - b.x).abs() < 1e-5 && (a.y - b.y).abs() < 1e-5);
    pool
}

fn try_place_piece(
    piece: &InputPiece,
    stock: &Stock,
    stock_vertices: &[Point],
    stock_bbox: (f64, f64, f64, f64),
    stock_is_rect: bool,
    num_rotations: usize,
    max_anchors: usize,
    query_buf: &mut Vec<usize>,
) -> Option<(Point, f64)> {
    let anchors = get_anchors(stock_vertices, &stock.placed_pieces, max_anchors);
    
    let mut candidates = Vec::new();
    
    for r_idx in 0..num_rotations {
        let rotation = 2.0 * PI * (r_idx as f64) / (num_rotations as f64);
        let rotated_vertices = rotate_polygon(&piece.vertices, rotation);
        let (dx_min, dx_max, dy_min, dy_max) = get_bbox(&rotated_vertices);
        let interior_point = rotate_point(piece.interior_point, rotation);
        let rotated_centroid = rotate_point(piece.centroid, rotation);
        
        // 1. Anchor-based candidates
        for &anchor in &anchors {
            for &v in &rotated_vertices {
                let translation = anchor - v;
                let px_min = translation.x + dx_min;
                let px_max = translation.x + dx_max;
                let py_min = translation.y + dy_min;
                let py_max = translation.y + dy_max;
                
                if px_min < stock_bbox.0 - 1e-9 || px_max > stock_bbox.1 + 1e-9 ||
                   py_min < stock_bbox.2 - 1e-9 || py_max > stock_bbox.3 + 1e-9 {
                    continue;
                }
                
                let score = py_max;
                candidates.push((translation, rotation, score, rotated_vertices.clone(), interior_point, rotated_centroid));
            }
        }
        
        // 2. Fallbacks for empty/packed stocks (centroid and bbox alignments)
        if stock.placed_pieces.is_empty() {
            let stock_centroid = compute_centroid(stock_vertices);
            let translation = stock_centroid - rotated_centroid;
            let px_min = translation.x + dx_min;
            let px_max = translation.x + dx_max;
            let py_min = translation.y + dy_min;
            let py_max = translation.y + dy_max;
            if px_min >= stock_bbox.0 - 1e-9 && px_max <= stock_bbox.1 + 1e-9 &&
               py_min >= stock_bbox.2 - 1e-9 && py_max <= stock_bbox.3 + 1e-9 {
                candidates.push((translation, rotation, py_max, rotated_vertices.clone(), interior_point, rotated_centroid));
            }
        }
    }
    
    candidates.sort_unstable_by(|a, b| a.2.partial_cmp(&b.2).unwrap_or(std::cmp::Ordering::Equal));
    
    for (translation, rotation, _, rotated_vertices, interior_point, rotated_centroid) in candidates {
        let placed_vertices: Vec<Point> = rotated_vertices.iter().map(|&v| v + translation).collect();
        let placed_interior = interior_point + translation;
        let placed_centroid = rotated_centroid + translation;
        
        if !is_contained_in_stock(&placed_vertices, placed_interior, stock_vertices, stock_is_rect, stock_bbox) {
            continue;
        }
        
        let (px_min, px_max, py_min, py_max) = get_bbox(&placed_vertices);
        stock.grid.query(px_min, px_max, py_min, py_max, query_buf);
        
        let mut overlap = false;
        for &idx in query_buf.iter() {
            let other = &stock.placed_pieces[idx];
            
            let dist_sq = (placed_centroid.x - other.centroid.x).powi(2) +
                          (placed_centroid.y - other.centroid.y).powi(2);
            let r_sum = piece.radius + other.radius;
            if dist_sq > r_sum.powi(2) + 1e-9 {
                continue;
            }
            
            if polygons_overlap(&placed_vertices, &other.vertices, placed_interior, other.interior_point) {
                overlap = true;
                break;
            }
        }
        
        if !overlap {
            return Some((translation, rotation));
        }
    }
    
    None
}

fn main() -> io::Result<()> {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input)?;
    let mut tokens = input.split_whitespace();
    
    let sv_str = match tokens.next() {
        Some(s) => s,
        None => return Ok(()),
    };
    let sv: usize = sv_str.parse().unwrap();
    let np: usize = tokens.next().unwrap().parse().unwrap();
    
    let mut stock_vertices = Vec::with_capacity(sv);
    for _ in 0..sv {
        let x: f64 = tokens.next().unwrap().parse().unwrap();
        let y: f64 = tokens.next().unwrap().parse().unwrap();
        stock_vertices.push(Point { x, y });
    }
    
    let stock_bbox = get_bbox(&stock_vertices);
    let stock_is_rect = is_axis_aligned_rect(&stock_vertices);
    
    let mut pieces = Vec::with_capacity(np);
    for i in 0..np {
        let pv: usize = tokens.next().unwrap().parse().unwrap();
        let mut vertices = Vec::with_capacity(pv);
        for _ in 0..pv {
            let x: f64 = tokens.next().unwrap().parse().unwrap();
            let y: f64 = tokens.next().unwrap().parse().unwrap();
            vertices.push(Point { x, y });
        }
        
        let area = compute_area(&vertices);
        let centroid = compute_centroid(&vertices);
        
        let mut max_dist_sq = 0.0;
        for &p in &vertices {
            let dist_sq = (p.x - centroid.x).powi(2) + (p.y - centroid.y).powi(2);
            if dist_sq > max_dist_sq {
                max_dist_sq = dist_sq;
            }
        }
        let radius = max_dist_sq.sqrt();
        let interior_point = find_interior_point(&vertices);
        
        pieces.push(InputPiece {
            id: i,
            vertices,
            area,
            radius,
            centroid,
            interior_point,
        });
    }
    
    let mut sorted_pieces: Vec<usize> = (0..np).collect();
    sorted_pieces.sort_by(|&a, &b| {
        pieces[b].area.partial_cmp(&pieces[a].area).unwrap_or(std::cmp::Ordering::Equal)
    });
    
    // Multi-tier configurations
    let num_rotations;
    let max_anchors;
    let grid_size;
    let max_stock_search;
    
    if np <= 50 {
        num_rotations = 16;
        max_anchors = 100;
        grid_size = 50;
        max_stock_search = 50;
    } else if np <= 500 {
        num_rotations = 8;
        max_anchors = 50;
        grid_size = 50;
        max_stock_search = 50;
    } else if np <= 2000 {
        num_rotations = 4;
        max_anchors = 20;
        grid_size = 80;
        max_stock_search = 15;
    } else {
        num_rotations = 2;
        max_anchors = 10;
        grid_size = 100;
        max_stock_search = 5;
    }
    
    let mut stocks: Vec<Stock> = Vec::new();
    let mut results = vec![(0, Point { x: 0.0, y: 0.0 }, 0.0); np];
    let mut query_buf = Vec::new();
    
    for &piece_id in &sorted_pieces {
        let piece = &pieces[piece_id];
        let mut placed = false;
        
        let stock_search_start = if stocks.len() > max_stock_search {
            stocks.len() - max_stock_search
        } else {
            0
        };
        
        for stock_idx in (stock_search_start..stocks.len()).rev() {
            let stock = &mut stocks[stock_idx];
            if let Some((translation, rotation)) = try_place_piece(
                piece,
                stock,
                &stock_vertices,
                stock_bbox,
                stock_is_rect,
                num_rotations,
                max_anchors,
                &mut query_buf,
            ) {
                let rotated_vertices = rotate_polygon(&piece.vertices, rotation);
                let placed_vertices: Vec<Point> = rotated_vertices.iter().map(|&v| v + translation).collect();
                let centroid = rotate_point(piece.centroid, rotation) + translation;
                let interior_point = rotate_point(piece.interior_point, rotation) + translation;
                
                let placed_piece = PlacedPiece {
                    rotation,
                    translation,
                    vertices: placed_vertices.clone(),
                    centroid,
                    radius: piece.radius,
                    interior_point,
                };
                
                let (px_min, px_max, py_min, py_max) = get_bbox(&placed_vertices);
                stock.grid.insert(stock.placed_pieces.len(), px_min, px_max, py_min, py_max);
                stock.placed_pieces.push(placed_piece);
                
                results[piece_id] = (stock_idx, translation, rotation);
                placed = true;
                break;
            }
        }
        
        if !placed {
            let stock_idx = stocks.len();
            let mut grid = SpatialGrid::new(stock_bbox.0, stock_bbox.1, stock_bbox.2, stock_bbox.3, grid_size);
            
            let mut empty_stock = Stock {
                placed_pieces: Vec::new(),
                grid: grid.clone(),
            };
            
            let (translation, rotation) = try_place_piece(
                piece,
                &empty_stock,
                &stock_vertices,
                stock_bbox,
                stock_is_rect,
                num_rotations.max(8), // give empty stock slightly more rotations if needed
                max_anchors.max(20),
                &mut query_buf,
            ).expect("Piece must fit in empty stock!");
            
            let rotated_vertices = rotate_polygon(&piece.vertices, rotation);
            let placed_vertices: Vec<Point> = rotated_vertices.iter().map(|&v| v + translation).collect();
            let centroid = rotate_point(piece.centroid, rotation) + translation;
            let interior_point = rotate_point(piece.interior_point, rotation) + translation;
            
            let placed_piece = PlacedPiece {
                rotation,
                translation,
                vertices: placed_vertices.clone(),
                centroid,
                radius: piece.radius,
                interior_point,
            };
            
            let (px_min, px_max, py_min, py_max) = get_bbox(&placed_vertices);
            grid.insert(0, px_min, px_max, py_min, py_max);
            
            empty_stock.placed_pieces.push(placed_piece);
            empty_stock.grid = grid;
            
            stocks.push(empty_stock);
            results[piece_id] = (stock_idx, translation, rotation);
        }
    }
    
    println!("{}", stocks.len());
    for r in results {
        println!("{} {} {} {}", r.0, r.1.x, r.1.y, r.2);
    }
    
    Ok(())
}
