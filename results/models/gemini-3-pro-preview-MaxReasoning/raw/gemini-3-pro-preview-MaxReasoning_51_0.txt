import math

# Persistent memory for PIDs and states
MEM = {
    'initialized': False,
    'prev_time': 0.0,
    'pitch_integral': 0.0,
    'roll_integral': 0.0,
    'speed_integral': 0.0,
    'alt_integral': 0.0,
    'yaw_integral': 0.0,
    'elevator_trim': 0.0,
}

def get_median(lst):
    if not lst:
        return 0.0
    s = sorted(lst)
    n = len(s)
    if n % 2 == 1:
        return s[n // 2]
    return (s[n // 2 - 1] + s[n // 2]) / 2.0

def get_isa_rho(alt_ft):
    alt_m = alt_ft * 0.3048
    alt_m = max(0, min(alt_m, 20000))
    if alt_m <= 11000:
        T = 288.15 - 0.0065 * alt_m
        rho = 1.225 * (T / 288.15)**(9.80665 / (0.0065 * 287.05) - 1.0)
    else:
        T = 216.65
        rho_11 = 1.225 * (216.65 / 288.15)**(9.80665 / (0.0065 * 287.05) - 1.0)
        rho = rho_11 * math.exp(-9.80665 * (alt_m - 11000) / (287.05 * 216.65))
    return max(rho, 0.01)

def autopilot_step(state: dict, dt: float) -> dict:
    global MEM
    
    # 1. Initialization / Reset on new scenario
    current_time = state.get('sim_time_s', 0.0)
    if not MEM['initialized'] or current_time < MEM['prev_time'] or current_time <= 2 * dt:
        MEM['initialized'] = True
        MEM['prev_time'] = current_time
        MEM['pitch_integral'] = 0.0
        MEM['roll_integral'] = 0.0
        MEM['speed_integral'] = 0.0
        MEM['alt_integral'] = 0.0
        MEM['yaw_integral'] = 0.0
        MEM['elevator_trim'] = state.get('elevator_trim_deg', 0.0) / 4.0 # trim_max is 4.0
        
    MEM['prev_time'] = current_time

    # 2. Sensor Fusion & Calibration
    # Reliable Altitude (ADC pressure altitude is fail-safe)
    adc_alts = [state[f'adc_{i}_pressure_alt_ft'] for i in range(1, 4) if f'adc_{i}_pressure_alt_ft' in state]
    alt_est = get_median(adc_alts) if adc_alts else state.get('baro_alt_1_ft', 0.0)

    # Reliable Airspeed (derive analytically from ground speed and wind to bypass pitot failure)
    hdg_rad = math.radians(state['true_heading_deg'])
    wind_dir_rad = math.radians(state['wind_dir_deg'])
    gs = state['ground_speed_kt']
    ws = state['wind_speed_kt']
    v_air_n = gs * math.cos(hdg_rad) - ws * math.cos(wind_dir_rad)
    v_air_e = gs * math.sin(hdg_rad) - ws * math.sin(wind_dir_rad)
    tas_est = math.sqrt(v_air_n**2 + v_air_e**2)
    rho = get_isa_rho(alt_est)
    ias_est = tas_est * math.sqrt(rho / 1.225)
    
    # Attitude (IRS median)
    irs_pitches = []
    irs_rolls = []
    for i in range(1, 4):
        if state.get(f'irs_{i}_valid', False):
            irs_pitches.append(state[f'irs_{i}_pitch_deg'])
            irs_rolls.append(state[f'irs_{i}_roll_deg'])
    if not irs_pitches:
        irs_pitches = [state[f'irs_{i}_pitch_deg'] for i in range(1, 4)]
    if not irs_rolls:
        irs_rolls = [state[f'irs_{i}_roll_deg'] for i in range(1, 4)]
        
    pitch_est = get_median(irs_pitches)
    roll_est = get_median(irs_rolls)

    # SWAP GYROS TO CORRECT SIMULATOR NAMING BUG
    gyro_ps = [] # roll rates (stored in gyro_i_pitch_rate_dps)
    gyro_qs = [] # pitch rates (stored in gyro_i_roll_rate_dps)
    gyro_ys = [] # yaw rates (stored in gyro_i_yaw_rate_dps)
    for i in range(1, 4):
        if state.get(f'gyro_{i}_valid', False):
            gyro_ps.append(state[f'gyro_{i}_pitch_rate_dps'])
            gyro_qs.append(state[f'gyro_{i}_roll_rate_dps'])
            gyro_ys.append(state[f'gyro_{i}_yaw_rate_dps'])
    if not gyro_ps:
        gyro_ps = [state[f'gyro_{i}_pitch_rate_dps'] for i in range(1, 4)]
        gyro_qs = [state[f'gyro_{i}_roll_rate_dps'] for i in range(1, 4)]
        gyro_ys = [state[f'gyro_{i}_yaw_rate_dps'] for i in range(1, 4)]
        
    pitch_rate = get_median(gyro_qs)
    roll_rate = get_median(gyro_ps)
    yaw_rate = get_median(gyro_ys)

    # AoA (cross-check sensors with pitch - flight_path_angle)
    aoa_phys_est = pitch_est - state['flight_path_angle_deg']
    aoa_vals = []
    for i in range(1, 4):
        if state.get(f'aoa_{i}_valid', False):
            aoa_vals.append(state[f'aoa_{i}_deg'])
    valid_aoa = [v for v in aoa_vals if abs(v - aoa_phys_est) < 5.0]
    aoa_est = get_median(valid_aoa) if valid_aoa else aoa_phys_est

    # 3. Targets
    target_alt = state.get('target_alt_ft', alt_est)
    target_hdg = state.get('target_hdg_deg', state['true_heading_deg'])
    target_ias = state.get('target_ias_kt', ias_est)

    # Override target airspeed in case of steep descent
    if target_alt < alt_est - 5000.0:
        target_ias = 340.0

    # 4. Flight Controllers
    # --- Unusual Attitude Recovery ---
    unusual_attitude = False
    if abs(roll_est) > 45.0:
        unusual_attitude = True
        phi_cmd = 0.0 # Wings level first!
    else:
        # --- Lateral Controller ---
        hdg_err = (target_hdg - state['true_heading_deg'] + 180) % 360 - 180
        # Command roll proportional to heading error, clamp to ±30 degrees
        phi_cmd = max(-30.0, min(30.0, 1.5 * hdg_err))
    
    roll_err = phi_cmd - roll_est
    MEM['roll_integral'] = max(-30.0, min(30.0, MEM['roll_integral'] + roll_err * dt))
    
    # Aileron PID (Ki_roll increased from 0.005 to 0.01 to counteract steady roll biases)
    aileron = 0.04 * roll_err + 0.01 * MEM['roll_integral'] - 0.015 * roll_rate
    aileron = max(-1.0, min(1.0, aileron))

    # --- Vertical Controller ---
    # Altitude to VS with PI controller
    alt_err = target_alt - alt_est
    MEM['alt_integral'] = max(-15000.0, min(15000.0, MEM['alt_integral'] + alt_err * dt))
    
    # In case of emergency descent, allow very fast descent
    min_vs = -25000.0 if alt_err < -5000.0 else -6000.0
    vs_cmd = 0.8 * alt_err + 0.1 * MEM['alt_integral']
    vs_cmd = max(min_vs, min(3000.0, vs_cmd))
    
    tas_fps = max(100.0, tas_est * 1.68781)
    gamma_cmd_deg = math.degrees(math.asin(max(-0.5, min(0.5, (vs_cmd / 60.0) / tas_fps))))
    
    # Analytical trim AoA for level flight (compensated for bank angle)
    tas_ms = tas_est * 0.514444
    qbar = 0.5 * rho * tas_ms**2
    mass = state.get('gross_weight_kg', 65000)
    CL_needed = (mass * 9.80665) / (max(qbar * 122.6, 1000.0) * max(0.5, math.cos(math.radians(roll_est))))
    aoa_trim_deg = math.degrees((CL_needed - 0.28) / 5.5)
    aoa_trim_deg = max(-2.0, min(12.0, aoa_trim_deg))
    
    pitch_cmd = aoa_trim_deg + gamma_cmd_deg
    
    # Speed-on-Pitch: if engines cannot maintain speed, adjust pitch to prioritize speed
    speed_diff = target_ias - ias_est
    if speed_diff > 5.0:
        # Too slow: pitch down to regain speed (0.5 deg pitch down per knot deficiency)
        pitch_cmd -= max(-15.0, min(15.0, 0.5 * (speed_diff - 5.0)))
    elif speed_diff < -15.0:
        # Too fast: pitch up to slow down
        pitch_cmd += max(-15.0, min(15.0, 0.5 * (-15.0 - speed_diff)))
        
    # Stall protection: limit pitch_cmd so AoA does not exceed safety margin
    aoa_limit = 13.0 if state.get('flap_pos_left_deg', 0.0) > 5.0 else 10.0
    pitch_cmd = min(pitch_cmd, state['flight_path_angle_deg'] + aoa_limit)
    
    # Overspeed protection: pitch up to slow down
    if ias_est > 340.0:
        overspeed_pitch = (ias_est - 340.0) * 1.0
        pitch_cmd = max(pitch_cmd, aoa_trim_deg + overspeed_pitch)
        
    if unusual_attitude:
        # Keep pitch near neutral while recovering roll
        pitch_cmd = max(-5.0, min(5.0, pitch_cmd))
        
    pitch_cmd = max(-20.0, min(25.0, pitch_cmd))
    
    pitch_err = pitch_cmd - pitch_est
    MEM['pitch_integral'] = max(-30.0, min(30.0, MEM['pitch_integral'] + pitch_err * dt))
    
    # Elevator PID
    elevator = 0.06 * pitch_err + 0.008 * MEM['pitch_integral'] - 0.02 * pitch_rate
    elevator = max(-1.0, min(1.0, elevator))
    
    # Auto-trim
    MEM['elevator_trim'] = max(-1.0, min(1.0, MEM['elevator_trim'] + 0.05 * elevator * dt))

    # --- Speed / Thrust Controller with Snappy PI + Feedforward ---
    speed_err = target_ias - ias_est
    MEM['speed_integral'] = max(-25.0, min(25.0, MEM['speed_integral'] + speed_err * dt))
    
    drag_ff = 0.0
    if state.get('flap_pos_left_deg', 0.0) > 5.0:
        drag_ff += 0.1
    if state.get('gear_handle_down', False):
        drag_ff += 0.1
        
    climb_ff = 0.02 * gamma_cmd_deg
    
    throttle_base = 0.58 + drag_ff + climb_ff
    throttle_cmd = throttle_base + 0.03 * speed_err + 0.01 * MEM['speed_integral']
    throttle_cmd = max(0.0, min(1.0, throttle_cmd))
    
    # Split throttle for left/right engine, shutdown if on fire
    throttle_1 = 0.0 if state.get('engine_1_fire', False) else throttle_cmd
    throttle_2 = 0.0 if state.get('engine_2_fire', False) else throttle_cmd

    # --- Yaw Controller ---
    # Asymmetric thrust compensation
    n1_1 = state.get('engine_1_n1_pct', 0.0)
    n1_2 = state.get('engine_2_n1_pct', 0.0)
    rudder_asym = - 60.0 * (n1_1 - n1_2) / max(ias_est, 80.0)**2
    
    # Yaw rate damping (Dutch roll damper) only. Do NOT use accel_y to avoid unstable kinematic feedback loops.
    rudder = rudder_asym - 0.05 * yaw_rate
    rudder = max(-1.0, min(1.0, rudder))

    # --- Speedbrake handling ---
    speedbrake = 0.0
    if ias_est > 330.0:
        speedbrake = min(1.0, (ias_est - 330.0) / 20.0)

    # Maintain existing gear and flaps settings if not overridden
    # We do not override flaps/gear, which keeps their manual handle settings.
    
    return {
        'elevator': elevator,
        'aileron': aileron,
        'rudder': rudder,
        'throttle_1': throttle_1,
        'throttle_2': throttle_2,
        'elevator_trim': MEM['elevator_trim'],
        'speedbrake': speedbrake,
    }
