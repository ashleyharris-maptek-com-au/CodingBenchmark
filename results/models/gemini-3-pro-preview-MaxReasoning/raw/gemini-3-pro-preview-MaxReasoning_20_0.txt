#![allow(non_snake_case, dead_code)]
use std::collections::BinaryHeap;
use std::io::{self, BufRead};
use std::time::Instant;

#[derive(Copy, Clone)]
struct HeapElem {
    dist_sq: f32,
    idx: u32,
}

impl PartialEq for HeapElem {
    fn eq(&self, other: &Self) -> bool {
        self.dist_sq == other.dist_sq
    }
}

impl Eq for HeapElem {}

impl PartialOrd for HeapElem {
    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
        self.dist_sq.partial_cmp(&other.dist_sq)
    }
}

impl Ord for HeapElem {
    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
        self.partial_cmp(other).unwrap_or(std::cmp::Ordering::Equal)
    }
}

struct KdNode {
    point: [f32; 3],
    idx: u32,
    left: u32,
    right: u32,
}

fn build_kd_tree(
    points: &mut [([f32; 3], u32)],
    nodes: &mut Vec<KdNode>,
    depth: usize,
) -> u32 {
    if points.is_empty() {
        return u32::MAX;
    }
    let axis = depth % 3;
    let mid = points.len() / 2;
    points.select_nth_unstable_by(mid, |a, b| {
        a.0[axis].partial_cmp(&b.0[axis]).unwrap_or(std::cmp::Ordering::Equal)
    });
    
    let node_idx = nodes.len() as u32;
    nodes.push(KdNode {
        point: points[mid].0,
        idx: points[mid].1,
        left: u32::MAX,
        right: u32::MAX,
    });
    
    let left_idx = build_kd_tree(&mut points[..mid], nodes, depth + 1);
    let right_idx = build_kd_tree(&mut points[mid + 1..], nodes, depth + 1);
    
    nodes[node_idx as usize].left = left_idx;
    nodes[node_idx as usize].right = right_idx;
    
    node_idx
}

fn knn_search(
    nodes: &[KdNode],
    root: u32,
    query: [f32; 3],
    k: usize,
    heap: &mut BinaryHeap<HeapElem>,
    depth: usize,
) {
    if root == u32::MAX {
        return;
    }
    let node = &nodes[root as usize];
    let dist_sq = (node.point[0] - query[0]).powi(2)
        + (node.point[1] - query[1]).powi(2)
        + (node.point[2] - query[2]).powi(2);
        
    if heap.len() < k {
        heap.push(HeapElem { dist_sq, idx: node.idx });
    } else if let Some(top) = heap.peek() {
        if dist_sq < top.dist_sq {
            heap.pop();
            heap.push(HeapElem { dist_sq, idx: node.idx });
        }
    }
    
    let axis = depth % 3;
    let diff = query[axis] - node.point[axis];
    let (near_child, far_child) = if diff <= 0.0 {
        (node.left, node.right)
    } else {
        (node.right, node.left)
    };
    
    knn_search(nodes, near_child, query, k, heap, depth + 1);
    
    let check_far = if heap.len() < k {
        true
    } else if let Some(top) = heap.peek() {
        diff.powi(2) < top.dist_sq
    } else {
        false
    };
    
    if check_far {
        knn_search(nodes, far_child, query, k, heap, depth + 1);
    }
}

struct HoleKdNode {
    point: [f32; 3],
    hole_idx: u32,
    left: u32,
    right: u32,
}

fn build_hole_kd_tree(
    points: &mut [([f32; 3], u32)],
    nodes: &mut Vec<HoleKdNode>,
    depth: usize,
) -> u32 {
    if points.is_empty() {
        return u32::MAX;
    }
    let axis = depth % 3;
    let mid = points.len() / 2;
    points.select_nth_unstable_by(mid, |a, b| {
        a.0[axis].partial_cmp(&b.0[axis]).unwrap_or(std::cmp::Ordering::Equal)
    });
    
    let node_idx = nodes.len() as u32;
    nodes.push(HoleKdNode {
        point: points[mid].0,
        hole_idx: points[mid].1,
        left: u32::MAX,
        right: u32::MAX,
    });
    
    let left_idx = build_hole_kd_tree(&mut points[..mid], nodes, depth + 1);
    let right_idx = build_hole_kd_tree(&mut points[mid + 1..], nodes, depth + 1);
    
    nodes[node_idx as usize].left = left_idx;
    nodes[node_idx as usize].right = right_idx;
    
    node_idx
}

fn hole_knn_search(
    nodes: &[HoleKdNode],
    root: u32,
    query: [f32; 3],
    k: usize,
    exclude_hole_idx: u32,
    heap: &mut BinaryHeap<HeapElem>,
    depth: usize,
) {
    if root == u32::MAX {
        return;
    }
    let node = &nodes[root as usize];
    let dist_sq = (node.point[0] - query[0]).powi(2)
        + (node.point[1] - query[1]).powi(2)
        + (node.point[2] - query[2]).powi(2);
        
    if node.hole_idx != exclude_hole_idx {
        if heap.len() < k {
            heap.push(HeapElem { dist_sq, idx: node.hole_idx });
        } else if let Some(top) = heap.peek() {
            if dist_sq < top.dist_sq {
                heap.pop();
                heap.push(HeapElem { dist_sq, idx: node.hole_idx });
            }
        }
    }
    
    let axis = depth % 3;
    let diff = query[axis] - node.point[axis];
    let (near_child, far_child) = if diff <= 0.0 {
        (node.left, node.right)
    } else {
        (node.right, node.left)
    };
    
    hole_knn_search(nodes, near_child, query, k, exclude_hole_idx, heap, depth + 1);
    
    let check_far = if heap.len() < k {
        true
    } else if let Some(top) = heap.peek() {
        diff.powi(2) < top.dist_sq
    } else {
        false
    };
    
    if check_far {
        hole_knn_search(nodes, far_child, query, k, exclude_hole_idx, heap, depth + 1);
    }
}

fn get_neighbor_holes(
    nodes: &[HoleKdNode],
    root: u32,
    h_idx: u32,
    start: [f32; 3],
    mid: [f32; 3],
    end: [f32; 3],
) -> Vec<u32> {
    let mut heap = BinaryHeap::new();
    hole_knn_search(nodes, root, start, 5, h_idx, &mut heap, 0);
    hole_knn_search(nodes, root, mid, 5, h_idx, &mut heap, 0);
    hole_knn_search(nodes, root, end, 5, h_idx, &mut heap, 0);
    
    let mut neighbors = Vec::new();
    while let Some(elem) = heap.pop() {
        neighbors.push(elem.idx);
    }
    neighbors.sort_unstable();
    neighbors.dedup();
    neighbors
}

struct HoleInfo {
    id: String,
    start_x: f32,
    start_y: f32,
    start_z: f32,
    dir_x: f32,
    dir_y: f32,
    dir_z: f32,
    length: f32,
    num_samples: usize,
    start_sample_idx: u32,
    end_sample_idx: u32,
}

struct Sample {
    hole_idx: u32,
    sample_idx_in_hole: u32,
    depth: f32,
    x: f32,
    y: f32,
    z: f32,
}

struct PropertyStats {
    name: String,
    median: f32,
    mad: f32,
    min_val: f32,
    max_val: f32,
}

struct Suspect {
    hole_id: String,
    sample_idx: u32,
    property_name: String,
    confidence: f32,
}

fn parse_float(s: &str) -> Option<f32> {
    let s_lower = s.to_lowercase();
    if s_lower == "nan" || s_lower == "na" || s_lower == "null" || s_lower == "none" || s == "-" || s == "*" {
        None
    } else {
        s.parse::<f32>().ok()
    }
}

fn get_same_hole_neighbor_indices(s: &Sample, h: &HoleInfo) -> Vec<u32> {
    let mut same_indices = Vec::new();
    let cur = s.sample_idx_in_hole as i32;
    let max_idx = (h.end_sample_idx - h.start_sample_idx) as i32;
    for offset in &[-2, -1, 1, 2] {
        let idx = cur + offset;
        if idx >= 0 && idx < max_idx {
            same_indices.push(h.start_sample_idx + idx as u32);
        }
    }
    same_indices
}

fn evaluate_sample_outliers(
    s_idx: u32,
    same_indices: &[u32],
    cross_indices: &[u32],
    all_samples: &[Sample],
    global_values: &[Option<f32>],
    holes: &[HoleInfo],
    property_stats: &[PropertyStats],
    out_suspects: &mut Vec<Suspect>,
) {
    let s = &all_samples[s_idx as usize];
    let num_properties = property_stats.len();
    
    for prop_idx in 0..num_properties {
        let val_opt = global_values[s_idx as usize * num_properties + prop_idx];
        let v = match val_opt {
            Some(x) => x,
            None => continue,
        };
        
        let stats = &property_stats[prop_idx];
        let mut neighbors = Vec::new();
        
        for &idx in same_indices {
            if let Some(nv) = global_values[idx as usize * num_properties + prop_idx] {
                neighbors.push(nv);
            }
        }
        
        for &idx in cross_indices {
            if let Some(nv) = global_values[idx as usize * num_properties + prop_idx] {
                neighbors.push(nv);
            }
        }
        
        if neighbors.len() < 3 {
            continue;
        }
        
        neighbors.sort_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
        let n_len = neighbors.len();
        let median = if n_len % 2 == 1 {
            neighbors[n_len / 2]
        } else {
            (neighbors[n_len / 2 - 1] + neighbors[n_len / 2]) * 0.5
        };
        
        let mut diffs: Vec<f32> = neighbors.iter().map(|&x| (x - median).abs()).collect();
        diffs.sort_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
        let mad = if n_len % 2 == 1 {
            diffs[n_len / 2]
        } else {
            (diffs[n_len / 2 - 1] + diffs[n_len / 2]) * 0.5
        };
        
        let global_range = stats.max_val - stats.min_val;
        let denom = mad + 0.05 * stats.mad + 1e-6 * (global_range + 1.0);
        let score = (v - median).abs() / denom;
        
        if score > 7.0 {
            out_suspects.push(Suspect {
                hole_id: holes[s.hole_idx as usize].id.clone(),
                sample_idx: s.sample_idx_in_hole,
                property_name: stats.name.clone(),
                confidence: score,
            });
        }
    }
}

fn compute_global_stats(name: &str, values: &[Option<f32>]) -> PropertyStats {
    let mut valid_vals: Vec<f32> = values.iter().filter_map(|&x| x).collect();
    if valid_vals.is_empty() {
        return PropertyStats {
            name: name.to_string(),
            median: 0.0,
            mad: 0.0,
            min_val: 0.0,
            max_val: 0.0,
        };
    }
    
    let n = valid_vals.len();
    let mid = n / 2;
    valid_vals.select_nth_unstable_by(mid, |a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
    let median = valid_vals[mid];
    
    let mut diffs: Vec<f32> = valid_vals.iter().map(|&x| (x - median).abs()).collect();
    diffs.select_nth_unstable_by(mid, |a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
    let mad = diffs[mid];
    
    let mut min_val = valid_vals[0];
    let mut max_val = valid_vals[0];
    for &val in &valid_vals {
        if val < min_val { min_val = val; }
        if val > max_val { max_val = val; }
    }
    
    PropertyStats {
        name: name.to_string(),
        median,
        mad,
        min_val,
        max_val,
    }
}

fn main() -> io::Result<()> {
    let start_time = Instant::now();
    let stdin = io::stdin();
    let mut reader = io::BufReader::new(stdin.lock());
    let mut line = String::new();
    
    if reader.read_line(&mut line)? == 0 {
        return Ok(());
    }
    let mut parts = line.split_whitespace();
    let num_holes: usize = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0);
    let num_properties: usize = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0);
    
    line.clear();
    if reader.read_line(&mut line)? == 0 {
        return Ok(());
    }
    let property_names: Vec<String> = line.split_whitespace().map(|s| s.to_string()).collect();
    
    let mut holes = Vec::with_capacity(num_holes);
    let mut all_samples = Vec::new();
    let mut global_values = Vec::new();
    
    for _ in 0..num_holes {
        line.clear();
        if reader.read_line(&mut line)? == 0 {
            break;
        }
        let mut parts = line.split_whitespace();
        let hole_id = match parts.next() {
            Some(id) => id.to_string(),
            None => continue,
        };
        let start_x: f32 = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);
        let start_y: f32 = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);
        let start_z: f32 = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);
        let dir_x: f32 = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);
        let dir_y: f32 = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);
        let dir_z: f32 = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);
        let length: f32 = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);
        let num_samples: usize = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0);
        
        let start_sample_idx = all_samples.len() as u32;
        
        let norm = (dir_x * dir_x + dir_y * dir_y + dir_z * dir_z).sqrt();
        let (ux, uy, uz) = if norm > 1e-6 {
            (dir_x / norm, dir_y / norm, dir_z / norm)
        } else {
            (0.0, 0.0, 0.0)
        };
        
        let hole_idx = holes.len() as u32;
        
        for sample_idx_in_hole in 0..num_samples {
            line.clear();
            if reader.read_line(&mut line)? == 0 {
                break;
            }
            let mut s_parts = line.split_whitespace();
            let depth: f32 = match s_parts.next() {
                Some(d) => d.parse().unwrap_or(0.0),
                None => 0.0,
            };
            
            let x = start_x + depth * ux;
            let y = start_y + depth * uy;
            let z = start_z + depth * uz;
            
            all_samples.push(Sample {
                hole_idx,
                sample_idx_in_hole: sample_idx_in_hole as u32,
                depth,
                x,
                y,
                z,
            });
            
            for _ in 0..num_properties {
                let val_str = s_parts.next().unwrap_or("NaN");
                global_values.push(parse_float(val_str));
            }
        }
        
        let end_sample_idx = all_samples.len() as u32;
        
        holes.push(HoleInfo {
            id: hole_id,
            start_x, start_y, start_z,
            dir_x, dir_y, dir_z,
            length,
            num_samples,
            start_sample_idx,
            end_sample_idx,
        });
    }
    
    // Compute property stats
    let mut property_stats = Vec::with_capacity(num_properties);
    for prop_idx in 0..num_properties {
        let mut prop_vals = Vec::with_capacity(all_samples.len());
        for s_idx in 0..all_samples.len() {
            prop_vals.push(global_values[s_idx * num_properties + prop_idx]);
        }
        let stats = compute_global_stats(&property_names[prop_idx], &prop_vals);
        property_stats.push(stats);
    }
    
    let num_threads = std::thread::available_parallelism()
        .map(|x| x.get())
        .unwrap_or(1);
    
    let total_samples = all_samples.len();
    
    let mut all_suspects = if total_samples < 30000 {
        // Global search strategy
        let mut global_points: Vec<([f32; 3], u32)> = all_samples
            .iter()
            .enumerate()
            .map(|(idx, s)| ([s.x, s.y, s.z], idx as u32))
            .collect();
        let mut global_nodes = Vec::with_capacity(global_points.len());
        let global_root = build_kd_tree(&mut global_points, &mut global_nodes, 0);
        
        std::thread::scope(|s| {
            let mut handles = Vec::new();
            let chunk_size = (total_samples + num_threads - 1) / num_threads;
            for chunk_idx in 0..num_threads {
                let start = chunk_idx * chunk_size;
                let end = (start + chunk_size).min(total_samples);
                if start >= end { continue; }
                
                let global_nodes_ref = &global_nodes;
                let all_samples_ref = &all_samples;
                let global_values_ref = &global_values;
                let holes_ref = &holes;
                let stats_ref = &property_stats;
                
                let handle = s.spawn(move || {
                    let mut local_suspects = Vec::new();
                    for s_idx in start..end {
                        let S = &all_samples_ref[s_idx];
                        let same_indices = get_same_hole_neighbor_indices(S, &holes_ref[S.hole_idx as usize]);
                        
                        let mut heap = BinaryHeap::new();
                        knn_search(global_nodes_ref, global_root, [S.x, S.y, S.z], 20, &mut heap, 0);
                        
                        let mut candidates = Vec::new();
                        while let Some(elem) = heap.pop() {
                            candidates.push(elem);
                        }
                        candidates.reverse();
                        
                        let mut cross_indices = Vec::new();
                        for elem in candidates {
                            let ns = &all_samples_ref[elem.idx as usize];
                            if ns.hole_idx != S.hole_idx {
                                cross_indices.push(elem.idx);
                                if cross_indices.len() == 6 {
                                    break;
                                }
                            }
                        }
                        
                        evaluate_sample_outliers(
                            s_idx as u32,
                            &same_indices,
                            &cross_indices,
                            all_samples_ref,
                            global_values_ref,
                            holes_ref,
                            stats_ref,
                            &mut local_suspects,
                        );
                    }
                    local_suspects
                });
                handles.push(handle);
            }
            
            let mut result = Vec::new();
            for h in handles {
                result.extend(h.join().unwrap());
            }
            result
        })
    } else {
        // Hole-partitioned search strategy
        let mut hole_kd_points = Vec::with_capacity(holes.len() * 3);
        for (h_idx, h) in holes.iter().enumerate() {
            let norm = (h.dir_x.powi(2) + h.dir_y.powi(2) + h.dir_z.powi(2)).sqrt();
            let (ux, uy, uz) = if norm > 1e-6 {
                (h.dir_x / norm, h.dir_y / norm, h.dir_z / norm)
            } else {
                (0.0, 0.0, 0.0)
            };
            let start = [h.start_x, h.start_y, h.start_z];
            let mid = [
                h.start_x + 0.5 * h.length * ux,
                h.start_y + 0.5 * h.length * uy,
                h.start_z + 0.5 * h.length * uz,
            ];
            let end = [
                h.start_x + h.length * ux,
                h.start_y + h.length * uy,
                h.start_z + h.length * uz,
            ];
            hole_kd_points.push((start, h_idx as u32));
            hole_kd_points.push((mid, h_idx as u32));
            hole_kd_points.push((end, h_idx as u32));
        }
        
        let mut hole_kd_nodes = Vec::with_capacity(hole_kd_points.len());
        let hole_kd_root = build_hole_kd_tree(&mut hole_kd_points, &mut hole_kd_nodes, 0);
        
        std::thread::scope(|s| {
            let mut handles = Vec::new();
            let chunk_size = (holes.len() + num_threads - 1) / num_threads;
            
            for chunk_idx in 0..num_threads {
                let start = chunk_idx * chunk_size;
                let end = (start + chunk_size).min(holes.len());
                if start >= end { continue; }
                
                let hole_kd_nodes_ref = &hole_kd_nodes;
                let all_samples_ref = &all_samples;
                let global_values_ref = &global_values;
                let holes_ref = &holes;
                let stats_ref = &property_stats;
                
                let handle = s.spawn(move || {
                    let mut local_suspects = Vec::new();
                    for h_idx in start..end {
                        let H = &holes_ref[h_idx];
                        let norm = (H.dir_x.powi(2) + H.dir_y.powi(2) + H.dir_z.powi(2)).sqrt();
                        let (ux, uy, uz) = if norm > 1e-6 {
                            (H.dir_x / norm, H.dir_y / norm, H.dir_z / norm)
                        } else {
                            (0.0, 0.0, 0.0)
                        };
                        let h_start = [H.start_x, H.start_y, H.start_z];
                        let h_mid = [
                            H.start_x + 0.5 * H.length * ux,
                            H.start_y + 0.5 * H.length * uy,
                            H.start_z + 0.5 * H.length * uz,
                        ];
                        let h_end = [
                            H.start_x + H.length * ux,
                            H.start_y + H.length * uy,
                            H.start_z + H.length * uz,
                        ];
                        
                        let neighbor_holes = get_neighbor_holes(
                            hole_kd_nodes_ref,
                            hole_kd_root,
                            h_idx as u32,
                            h_start,
                            h_mid,
                            h_end,
                        );
                        
                        let mut neighbor_samples = Vec::new();
                        for &nh_idx in &neighbor_holes {
                            let nh = &holes_ref[nh_idx as usize];
                            for s_idx in nh.start_sample_idx..nh.end_sample_idx {
                                let ns = &all_samples_ref[s_idx as usize];
                                neighbor_samples.push((ns.x, ns.y, ns.z, s_idx));
                            }
                        }
                        
                        let has_neighbors = !neighbor_samples.is_empty();
                        
                        let mut local_nodes = Vec::new();
                        let mut local_root = u32::MAX;
                        if neighbor_samples.len() > 128 {
                            let mut local_points: Vec<([f32; 3], u32)> = neighbor_samples
                                .iter()
                                .map(|&(x, y, z, idx)| ([x, y, z], idx))
                                .collect();
                            local_nodes = Vec::with_capacity(local_points.len());
                            local_root = build_kd_tree(&mut local_points, &mut local_nodes, 0);
                        }
                        
                        for s_idx in H.start_sample_idx..H.end_sample_idx {
                            let S = &all_samples_ref[s_idx as usize];
                            let same_indices = get_same_hole_neighbor_indices(S, H);
                            
                            let mut cross_indices = Vec::new();
                            if has_neighbors {
                                let mut heap = BinaryHeap::new();
                                if neighbor_samples.len() <= 128 {
                                    for &(nx, ny, nz, ns_idx) in &neighbor_samples {
                                        let dist_sq = (nx - S.x).powi(2) + (ny - S.y).powi(2) + (nz - S.z).powi(2);
                                        if heap.len() < 6 {
                                            heap.push(HeapElem { dist_sq, idx: ns_idx });
                                        } else if let Some(top) = heap.peek() {
                                            if dist_sq < top.dist_sq {
                                                heap.pop();
                                                heap.push(HeapElem { dist_sq, idx: ns_idx });
                                            }
                                        }
                                    }
                                } else {
                                    knn_search(&local_nodes, local_root, [S.x, S.y, S.z], 6, &mut heap, 0);
                                }
                                
                                while let Some(elem) = heap.pop() {
                                    cross_indices.push(elem.idx);
                                }
                            }
                            
                            evaluate_sample_outliers(
                                s_idx,
                                &same_indices,
                                &cross_indices,
                                all_samples_ref,
                                global_values_ref,
                                holes_ref,
                                stats_ref,
                                &mut local_suspects,
                            );
                        }
                    }
                    local_suspects
                });
                handles.push(handle);
            }
            
            let mut result = Vec::new();
            for h in handles {
                result.extend(h.join().unwrap());
            }
            result
        })
    };
    
    // Sort suspects by confidence desc
    all_suspects.sort_by(|a, b| {
        b.confidence.partial_cmp(&a.confidence)
            .unwrap_or(std::cmp::Ordering::Equal)
            .then_with(|| a.hole_id.cmp(&b.hole_id))
            .then_with(|| a.sample_idx.cmp(&b.sample_idx))
            .then_with(|| a.property_name.cmp(&b.property_name))
    });
    
    // Output results
    println!("{}", all_suspects.len());
    for s in &all_suspects {
        println!("{} {} {} {:.4}", s.hole_id, s.sample_idx, s.property_name, s.confidence);
    }
    
    eprintln!("Processed {} samples in {:?}", total_samples, start_time.elapsed());
    Ok(())
}
