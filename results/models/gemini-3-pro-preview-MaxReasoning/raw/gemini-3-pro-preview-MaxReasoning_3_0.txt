use std::io::{self, Read};

struct FastReader {
    buffer: Vec<u8>,
    pos: usize,
}

impl FastReader {
    fn new() -> Self {
        let mut buffer = Vec::new();
        io::stdin().read_to_end(&mut buffer).unwrap();
        Self { buffer, pos: 0 }
    }

    fn next_u32(&mut self) -> Option<u32> {
        while self.pos < self.buffer.len() && self.buffer[self.pos] <= b' ' {
            self.pos += 1;
        }
        if self.pos >= self.buffer.len() {
            return None;
        }
        let mut val = 0u32;
        while self.pos < self.buffer.len() && self.buffer[self.pos] > b' ' {
            val = val * 10 + (self.buffer[self.pos] - b'0') as u32;
            self.pos += 1;
        }
        Some(val)
    }
}

fn layout_component(
    vc: usize,
    edges: &[(u32, u32)],
    initial_x: &[f64],
    initial_y: &[f64],
) -> (Vec<f64>, Vec<f64>) {
    let mut px = initial_x.to_vec();
    let mut py = initial_y.to_vec();

    if vc <= 1 {
        return (px, py);
    }

    let max_iters = if vc <= 20 {
        500
    } else if vc <= 100 {
        300
    } else if vc <= 1000 {
        200
    } else if vc <= 10000 {
        100
    } else {
        50
    };

    let mut disp_x = vec![0.0f64; vc];
    let mut disp_y = vec![0.0f64; vc];

    let use_grid = vc > 500;
    let cell_size = 3.0f64;
    let table_size = (vc * 2).next_power_of_two();
    let mut head = vec![-1_i32; table_size];
    let mut next = vec![-1_i32; vc];

    let hash = |cx: i32, cy: i32| -> usize {
        let h = (cx.wrapping_mul(73856093) ^ cy.wrapping_mul(19349663)) & (table_size as i32 - 1);
        h as usize
    };

    let temp_0 = 1.5f64;

    for iter in 0..max_iters {
        disp_x.fill(0.0);
        disp_y.fill(0.0);

        // 1. Repulsive forces
        if use_grid {
            head.fill(-1);
            for i in 0..vc {
                let cx = (px[i] / cell_size).floor() as i32;
                let cy = (py[i] / cell_size).floor() as i32;
                let h = hash(cx, cy);
                next[i] = head[h];
                head[h] = i as i32;
            }

            for i in 0..vc {
                let xi = px[i];
                let yi = py[i];
                let cx_i = (xi / cell_size).floor() as i32;
                let cy_i = (yi / cell_size).floor() as i32;

                for dx_cell in -1..=1 {
                    for dy_cell in -1..=1 {
                        let cx = cx_i + dx_cell;
                        let cy = cy_i + dy_cell;
                        let h = hash(cx, cy);
                        let mut curr = head[h];
                        while curr != -1 {
                            let j = curr as usize;
                            if j > i {
                                let cx_j = (px[j] / cell_size).floor() as i32;
                                let cy_j = (py[j] / cell_size).floor() as i32;
                                if cx_j == cx && cy_j == cy {
                                    let mut dx = xi - px[j];
                                    let mut dy = yi - py[j];
                                    let mut d2 = dx * dx + dy * dy;
                                    if d2 < 9.0 {
                                        if d2 < 1e-9 {
                                            dx = (i as f64 * 0.13 + 0.17).cos() * 1e-3;
                                            dy = (i as f64 * 0.13 + 0.17).sin() * 1e-3;
                                            d2 = dx * dx + dy * dy;
                                        }
                                        let d2_clamped = d2.max(0.01);
                                        let factor = 1.44 / d2_clamped;
                                        let fx = dx * factor;
                                        let fy = dy * factor;
                                        disp_x[i] += fx;
                                        disp_y[i] += fy;
                                        disp_x[j] -= fx;
                                        disp_y[j] -= fy;
                                    }
                                }
                            }
                            curr = next[curr as usize];
                        }
                    }
                }
            }
        } else {
            for i in 0..vc {
                let xi = px[i];
                let yi = py[i];
                for j in (i + 1)..vc {
                    let mut dx = xi - px[j];
                    let mut dy = yi - py[j];
                    let mut d2 = dx * dx + dy * dy;
                    if d2 < 1e-9 {
                        dx = (i as f64 * 0.13 + 0.17).cos() * 1e-3;
                        dy = (i as f64 * 0.13 + 0.17).sin() * 1e-3;
                        d2 = dx * dx + dy * dy;
                    }
                    let d2_clamped = d2.max(0.01);
                    let factor = 1.44 / d2_clamped;
                    let fx = dx * factor;
                    let fy = dy * factor;
                    disp_x[i] += fx;
                    disp_y[i] += fy;
                    disp_x[j] -= fx;
                    disp_y[j] -= fy;
                }
            }
        }

        // 2. Attractive forces
        for &(u, v) in edges {
            let u = u as usize;
            let v = v as usize;
            let dx = px[u] - px[v];
            let dy = py[u] - py[v];
            let d2 = dx * dx + dy * dy;
            if d2 > 1e-9 {
                let d = d2.sqrt();
                let factor = d / 1.2;
                let fx = dx * factor;
                let fy = dy * factor;
                disp_x[u] -= fx;
                disp_y[u] -= fy;
                disp_x[v] += fx;
                disp_y[v] += fy;
            }
        }

        // 3. Gravity
        let mut sum_x = 0.0;
        let mut sum_y = 0.0;
        for i in 0..vc {
            sum_x += px[i];
            sum_y += py[i];
        }
        let cx = sum_x / vc as f64;
        let cy = sum_y / vc as f64;
        for i in 0..vc {
            disp_x[i] -= 0.01 * (px[i] - cx);
            disp_y[i] -= 0.01 * (py[i] - cy);
        }

        // 4. Update positions with temperature limiting
        let temp = temp_0 * (1.0 - (iter as f64 / max_iters as f64));
        let temp2 = temp * temp;
        for i in 0..vc {
            let dx = disp_x[i];
            let dy = disp_y[i];
            let d2 = dx * dx + dy * dy;
            if d2 > 1e-9 {
                if d2 <= temp2 {
                    px[i] += dx;
                    py[i] += dy;
                } else {
                    let d = d2.sqrt();
                    px[i] += (dx / d) * temp;
                    py[i] += (dy / d) * temp;
                }
            }
        }
    }

    (px, py)
}

fn main() {
    let mut reader = FastReader::new();
    let n = match reader.next_u32() {
        Some(val) => val as usize,
        None => return,
    };
    let m = match reader.next_u32() {
        Some(val) => val as usize,
        None => 0,
    };

    let mut edges = Vec::with_capacity(m);
    for _ in 0..m {
        let u = match reader.next_u32() {
            Some(val) => val,
            None => break,
        };
        let v = match reader.next_u32() {
            Some(val) => val,
            None => break,
        };
        if u < n as u32 && v < n as u32 && u != v {
            let (u_min, v_max) = if u < v { (u, v) } else { (v, u) };
            edges.push((u_min, v_max));
        }
    }

    // Remove duplicates
    edges.sort_unstable();
    edges.dedup();

    // Adjacency list
    let mut degrees = vec![0; n];
    for &(u, v) in &edges {
        degrees[u as usize] += 1;
        degrees[v as usize] += 1;
    }
    let mut adj = vec![Vec::new(); n];
    for i in 0..n {
        adj[i].reserve(degrees[i]);
    }
    for &(u, v) in &edges {
        adj[u as usize].push(v);
        adj[v as usize].push(u);
    }

    // Find connected components
    let mut visited = vec![false; n];
    let mut components = Vec::new();
    for i in 0..n {
        if !visited[i] {
            let mut comp = Vec::new();
            let mut queue = Vec::new();
            queue.push(i);
            visited[i] = true;
            let mut head = 0;
            while head < queue.len() {
                let u = queue[head];
                head += 1;
                comp.push(u);
                for &v in &adj[u] {
                    let v_usz = v as usize;
                    if !visited[v_usz] {
                        visited[v_usz] = true;
                        queue.push(v_usz);
                    }
                }
            }
            components.push(comp);
        }
    }

    // For each component, run BFS initialization and lay out
    let mut final_x = vec![0.0f64; n];
    let mut final_y = vec![0.0f64; n];
    let mut comp_w = vec![0.0f64; components.len()];
    let mut comp_h = vec![0.0f64; components.len()];

    let mut global_to_local = vec![usize::MAX; n];

    for (comp_idx, component) in components.iter().enumerate() {
        let vc = component.len();
        if vc == 1 {
            comp_w[comp_idx] = 0.0;
            comp_h[comp_idx] = 0.0;
            continue;
        }

        // Find start node for BFS (highest degree node in the component)
        let mut start_node = component[0];
        let mut max_deg = adj[start_node].len();
        for &u in component {
            if adj[u].len() > max_deg {
                max_deg = adj[u].len();
                start_node = u;
            }
        }

        // BFS traversal for depth calculation
        let mut local_visited = vec![false; vc];
        for (local_idx, &u) in component.iter().enumerate() {
            global_to_local[u] = local_idx;
        }

        let mut queue = Vec::with_capacity(vc);
        let mut depths = vec![0u32; vc];

        let start_local = global_to_local[start_node];
        queue.push(start_local);
        local_visited[start_local] = true;
        depths[start_local] = 0;

        let mut head = 0;
        while head < queue.len() {
            let u_local = queue[head];
            let u_global = component[u_local];
            let du = depths[u_local];
            head += 1;
            for &v_global in &adj[u_global] {
                let v_local = global_to_local[v_global as usize];
                if v_local != usize::MAX && !local_visited[v_local] {
                    local_visited[v_local] = true;
                    depths[v_local] = du + 1;
                    queue.push(v_local);
                }
            }
        }

        let mut max_d = 0;
        for &d in &depths {
            if d > max_d {
                max_d = d;
            }
        }

        let mut count_at_depth = vec![0; (max_d + 1) as usize];
        for &d in &depths {
            count_at_depth[d as usize] += 1;
        }

        let mut offset_at_depth = vec![0; (max_d + 2) as usize];
        for d in 0..=max_d {
            offset_at_depth[(d + 1) as usize] = offset_at_depth[d as usize] + count_at_depth[d as usize];
        }

        let mut sorted_by_depth = vec![0; vc];
        let mut curr_offset = offset_at_depth.clone();
        for &u_local in &queue {
            let d = depths[u_local] as usize;
            let idx = curr_offset[d];
            sorted_by_depth[idx] = u_local;
            curr_offset[d] += 1;
        }

        let mut radius = vec![0.0f64; (max_d + 1) as usize];
        radius[0] = 0.0;
        for d in 1..=max_d {
            let count = count_at_depth[d as usize] as f64;
            let needed_circumference = count * 1.5;
            let needed_radius = needed_circumference / (2.0 * std::f64::consts::PI);
            radius[d as usize] = radius[(d - 1) as usize] + needed_radius.max(2.5);
        }

        let mut init_x = vec![0.0f64; vc];
        let mut init_y = vec![0.0f64; vc];

        for d in 0..=max_d {
            let start_idx = offset_at_depth[d as usize];
            let end_idx = offset_at_depth[(d + 1) as usize];
            let count = end_idx - start_idx;
            let r = radius[d as usize];
            for j in 0..count {
                let u_local = sorted_by_depth[start_idx + j];
                if count == 1 {
                    if d == 0 {
                        init_x[u_local] = 0.0;
                        init_y[u_local] = 0.0;
                    } else {
                        let theta = (u_local as f64) * 2.39996;
                        init_x[u_local] = r * theta.cos();
                        init_y[u_local] = r * theta.sin();
                    }
                } else {
                    let theta = (j as f64 / count as f64) * 2.0 * std::f64::consts::PI;
                    init_x[u_local] = r * theta.cos();
                    init_y[u_local] = r * theta.sin();
                }
            }
        }

        // Filter edges to only include those in this component
        let mut local_edges = Vec::new();
        for &(u, v) in &edges {
            let lu = global_to_local[u as usize];
            let lv = global_to_local[v as usize];
            if lu != usize::MAX && lv != usize::MAX {
                local_edges.push((lu as u32, lv as u32));
            }
        }

        // Run force-directed layout
        let (lx, ly) = layout_component(vc, &local_edges, &init_x, &init_y);

        // Center coordinates of component at (0, 0)
        let mut sum_lx = 0.0;
        let mut sum_ly = 0.0;
        for i in 0..vc {
            sum_lx += lx[i];
            sum_ly += ly[i];
        }
        let clx = sum_lx / vc as f64;
        let cly = sum_ly / vc as f64;

        let mut xmin = f64::MAX;
        let mut xmax = f64::MIN;
        let mut ymin = f64::MAX;
        let mut ymax = f64::MIN;

        for i in 0..vc {
            let u_global = component[i];
            let cx_val = lx[i] - clx;
            let cy_val = ly[i] - cly;
            final_x[u_global] = cx_val;
            final_y[u_global] = cy_val;

            if cx_val < xmin { xmin = cx_val; }
            if cx_val > xmax { xmax = cx_val; }
            if cy_val < ymin { ymin = cy_val; }
            if cy_val > ymax { ymax = cy_val; }
        }

        comp_w[comp_idx] = xmax - xmin;
        comp_h[comp_idx] = ymax - ymin;

        // Reset global_to_local mapping for these nodes
        for &u in component {
            global_to_local[u] = usize::MAX;
        }
    }

    // Pack components
    let mut comp_indices: Vec<usize> = (0..components.len()).collect();
    comp_indices.sort_by(|&a, &b| {
        let ha = comp_h[a];
        let hb = comp_h[b];
        hb.partial_cmp(&ha).unwrap_or(std::cmp::Ordering::Equal)
    });

    let margin = 4.0f64;
    let mut total_area = 0.0f64;
    for i in 0..components.len() {
        total_area += (comp_w[i] + margin) * (comp_h[i] + margin);
    }
    let target_width = total_area.sqrt();

    let mut current_x = 0.0f64;
    let mut current_y = 0.0f64;
    let mut row_height = 0.0f64;

    let mut packed_x = vec![0.0f64; components.len()];
    let mut packed_y = vec![0.0f64; components.len()];

    for &comp_idx in &comp_indices {
        let cw = comp_w[comp_idx];
        let ch = comp_h[comp_idx];
        if current_x + cw > target_width && current_x > 0.0 {
            current_x = 0.0;
            current_y += row_height + margin;
            row_height = 0.0;
        }
        packed_x[comp_idx] = current_x + cw / 2.0;
        packed_y[comp_idx] = current_y + ch / 2.0;

        current_x += cw + margin;
        if ch > row_height {
            row_height = ch;
        }
    }

    for (comp_idx, component) in components.iter().enumerate() {
        let px = packed_x[comp_idx];
        let py = packed_y[comp_idx];
        for &u in component {
            final_x[u] += px;
            final_y[u] += py;
        }
    }

    // Output results
    use std::io::Write;
    let stdout = io::stdout();
    let mut out = io::BufWriter::new(stdout.lock());

    for i in 0..n {
        let _ = writeln!(out, "{:.6} {:.6}", final_x[i], final_y[i]);
    }
}
