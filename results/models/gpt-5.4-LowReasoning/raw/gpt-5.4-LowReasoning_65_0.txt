declare const require: any;
declare const process: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type TaskInput = {
  id: string | number;
  duration: number;
  deps?: Array<string | number>;
  deadline?: number;
  weight?: number;
};

type Task = {
  id: string | number;
  duration: number;
  deps: number[];
  deadline: number;
  weight: number;
  succs: number[];
  indegree: number;
  release: number;
  tail: number;
  descWeight: number;
  avgSuccDeadline: number;
};

type ScheduleEntry = {
  id: string | number;
  start: number;
  machine: number;
};

class MinHeap<T> {
  private readonly data: T[] = [];
  private readonly less: (a: T, b: T) => boolean;

  constructor(less: (a: T, b: T) => boolean) {
    this.less = less;
  }

  size(): number {
    return this.data.length;
  }

  peek(): T | undefined {
    return this.data[0];
  }

  push(value: T): void {
    const arr = this.data;
    arr.push(value);
    let i = arr.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (!this.less(arr[i], arr[p])) break;
      [arr[i], arr[p]] = [arr[p], arr[i]];
      i = p;
    }
  }

  pop(): T | undefined {
    const arr = this.data;
    if (arr.length === 0) return undefined;
    const top = arr[0];
    const last = arr.pop()!;
    if (arr.length > 0) {
      arr[0] = last;
      let i = 0;
      while (true) {
        const l = i * 2 + 1;
        const r = l + 1;
        let best = i;
        if (l < arr.length && this.less(arr[l], arr[best])) best = l;
        if (r < arr.length && this.less(arr[r], arr[best])) best = r;
        if (best === i) break;
        [arr[i], arr[best]] = [arr[best], arr[i]];
        i = best;
      }
    }
    return top;
  }
}

const machineCount: number = Math.max(1, (data.machineCount | 0) || 1);
const inputTasks: TaskInput[] = Array.isArray(data.tasks) ? data.tasks : [];
const n = inputTasks.length;

if (n === 0) {
  process.stdout.write(JSON.stringify({ schedule: [] }));
  process.exit(0);
}

const idToIndex = new Map<string | number, number>();
for (let i = 0; i < n; i++) {
  idToIndex.set(inputTasks[i].id, i);
}

const tasks: Task[] = new Array(n);
for (let i = 0; i < n; i++) {
  const src = inputTasks[i];
  tasks[i] = {
    id: src.id,
    duration: Math.max(0, src.duration | 0),
    deps: [],
    deadline: Number.isFinite(src.deadline) ? Number(src.deadline) : Number.MAX_SAFE_INTEGER / 4,
    weight: Number.isFinite(src.weight) ? Math.max(0, Number(src.weight)) : 1,
    succs: [],
    indegree: 0,
    release: 0,
    tail: 0,
    descWeight: 0,
    avgSuccDeadline: 0,
  };
}

for (let i = 0; i < n; i++) {
  const depIds = Array.isArray(inputTasks[i].deps) ? inputTasks[i].deps! : [];
  const seen = new Set<number>();
  for (const depId of depIds) {
    const depIndex = idToIndex.get(depId);
    if (depIndex === undefined || depIndex === i || seen.has(depIndex)) continue;
    seen.add(depIndex);
    tasks[i].deps.push(depIndex);
    tasks[i].indegree++;
    tasks[depIndex].succs.push(i);
  }
}

const topo: number[] = [];
const queue: number[] = [];
for (let i = 0; i < n; i++) {
  if (tasks[i].indegree === 0) queue.push(i);
}

for (let head = 0; head < queue.length; head++) {
  const u = queue[head];
  topo.push(u);
  const finish = tasks[u].release + tasks[u].duration;
  for (const v of tasks[u].succs) {
    if (finish > tasks[v].release) tasks[v].release = finish;
    tasks[v].indegree--;
    if (tasks[v].indegree === 0) queue.push(v);
  }
}

if (topo.length !== n) {
  const fallbackOrder = [];
  const remaining = new Set<number>();
  for (let i = 0; i < n; i++) remaining.add(i);
  for (const idx of topo) remaining.delete(idx);
  fallbackOrder.push(...topo.map((idx) => tasks[idx].id));
  const rest = Array.from(remaining).sort((a, b) => {
    const ta = tasks[a];
    const tb = tasks[b];
    if (ta.deadline !== tb.deadline) return ta.deadline - tb.deadline;
    if (tb.weight !== ta.weight) return tb.weight - ta.weight;
    if (tb.duration !== ta.duration) return tb.duration - ta.duration;
    return a - b;
  });
  for (const idx of rest) fallbackOrder.push(tasks[idx].id);
  process.stdout.write(JSON.stringify({ order: fallbackOrder }));
  process.exit(0);
}

for (let p = n - 1; p >= 0; p--) {
  const u = topo[p];
  let bestTail = tasks[u].duration;
  let totalWeight = tasks[u].weight;
  let succDeadlineSum = 0;
  let succCount = 0;
  for (const v of tasks[u].succs) {
    const candidate = tasks[u].duration + tasks[v].tail;
    if (candidate > bestTail) bestTail = candidate;
    totalWeight += tasks[v].descWeight;
    succDeadlineSum += tasks[v].deadline;
    succCount++;
  }
  tasks[u].tail = bestTail;
  tasks[u].descWeight = totalWeight;
  tasks[u].avgSuccDeadline = succCount > 0 ? succDeadlineSum / succCount : tasks[u].deadline;
}

const remainingDeps: number[] = new Array(n);
const depFinishMax: number[] = new Array(n).fill(0);
for (let i = 0; i < n; i++) {
  remainingDeps[i] = tasks[i].deps.length;
}

const machineHeap = new MinHeap<{ freeAt: number; machine: number }>((a, b) => {
  if (a.freeAt !== b.freeAt) return a.freeAt < b.freeAt;
  return a.machine < b.machine;
});
for (let machine = 0; machine < machineCount; machine++) {
  machineHeap.push({ freeAt: 0, machine });
}

const readyHeap = new MinHeap<{ idx: number }>((a, b) => {
  const ta = tasks[a.idx];
  const tb = tasks[b.idx];
  const urgentA = ta.deadline - depFinishMax[a.idx] - ta.tail;
  const urgentB = tb.deadline - depFinishMax[b.idx] - tb.tail;
  if (urgentA !== urgentB) return urgentA < urgentB;

  const weightedTailA = ta.tail * (1 + ta.weight);
  const weightedTailB = tb.tail * (1 + tb.weight);
  if (weightedTailA !== weightedTailB) return weightedTailA > weightedTailB;

  const ratioA = ta.weight / Math.max(1, ta.duration);
  const ratioB = tb.weight / Math.max(1, tb.duration);
  if (ratioA !== ratioB) return ratioA > ratioB;

  if (ta.descWeight !== tb.descWeight) return ta.descWeight > tb.descWeight;
  if (ta.avgSuccDeadline !== tb.avgSuccDeadline) return ta.avgSuccDeadline < tb.avgSuccDeadline;
  if (ta.duration !== tb.duration) return ta.duration > tb.duration;
  return a.idx < b.idx;
});

for (let i = 0; i < n; i++) {
  if (remainingDeps[i] === 0) readyHeap.push({ idx: i });
}

const schedule: ScheduleEntry[] = [];
let scheduled = 0;

while (scheduled < n && readyHeap.size() > 0) {
  const machineState = machineHeap.pop()!;
  const choice = readyHeap.pop()!;
  const idx = choice.idx;
  const start = Math.max(machineState.freeAt, depFinishMax[idx]);
  const finish = start + tasks[idx].duration;

  schedule.push({
    id: tasks[idx].id,
    start,
    machine: machineState.machine,
  });
  scheduled++;

  machineHeap.push({ freeAt: finish, machine: machineState.machine });

  for (const succ of tasks[idx].succs) {
    if (finish > depFinishMax[succ]) depFinishMax[succ] = finish;
    remainingDeps[succ]--;
    if (remainingDeps[succ] === 0) readyHeap.push({ idx: succ });
  }
}

if (scheduled !== n) {
  const order = topo.map((idx) => tasks[idx].id);
  process.stdout.write(JSON.stringify({ order }));
} else {
  process.stdout.write(JSON.stringify({ schedule }));
}
