declare const require: any;
declare const process: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type Grid = string[];

interface Robot {
  id: number;
  x: number;
  y: number;
}

interface Job {
  id: number;
  x: number;
  y: number;
  priority: number;
}

interface Point {
  x: number;
  y: number;
}

const grid: Grid = data.grid;
const robots: Robot[] = data.robots;
const jobs: Job[] = data.jobs;

const height = grid.length;
const width = height > 0 ? grid[0].length : 0;
const cellCount = width * height;

function inBounds(x: number, y: number): boolean {
  return x >= 0 && x < width && y >= 0 && y < height;
}

function cellIndex(x: number, y: number): number {
  return y * width + x;
}

function isOpen(x: number, y: number): boolean {
  return inBounds(x, y) && grid[y].charCodeAt(x) !== 35;
}

const uniquePoints: Point[] = [];
const pointKeyToIndex = new Map<string, number>();

function getPointIndex(x: number, y: number): number {
  const key = `${x},${y}`;
  const existing = pointKeyToIndex.get(key);
  if (existing !== undefined) {
    return existing;
  }
  const index = uniquePoints.length;
  uniquePoints.push({ x, y });
  pointKeyToIndex.set(key, index);
  return index;
}

const jobPointIndices: number[] = new Array(jobs.length);
for (let i = 0; i < jobs.length; i++) {
  jobPointIndices[i] = getPointIndex(jobs[i].x, jobs[i].y);
}

for (let i = 0; i < robots.length; i++) {
  getPointIndex(robots[i].x, robots[i].y);
}

const jobsAtPoint = new Map<number, number[]>();
for (let i = 0; i < jobs.length; i++) {
  const pointIndex = jobPointIndices[i];
  let bucket = jobsAtPoint.get(pointIndex);
  if (bucket === undefined) {
    bucket = [];
    jobsAtPoint.set(pointIndex, bucket);
  }
  bucket.push(i);
}

const sourceToJobDistances = new Map<number, Int32Array>();
const queue = new Int32Array(cellCount > 0 ? cellCount : 1);
const distGrid = new Int32Array(cellCount > 0 ? cellCount : 1);

function computeDistancesToJobs(sourcePointIndex: number): Int32Array {
  const cached = sourceToJobDistances.get(sourcePointIndex);
  if (cached !== undefined) {
    return cached;
  }

  const result = new Int32Array(jobs.length);
  result.fill(-1);

  if (jobs.length === 0) {
    sourceToJobDistances.set(sourcePointIndex, result);
    return result;
  }

  const source = uniquePoints[sourcePointIndex];
  if (!isOpen(source.x, source.y)) {
    sourceToJobDistances.set(sourcePointIndex, result);
    return result;
  }

  distGrid.fill(-1);
  let head = 0;
  let tail = 0;
  const startCell = cellIndex(source.x, source.y);
  distGrid[startCell] = 0;
  queue[tail++] = startCell;

  let remainingTargets = jobs.length;

  while (head < tail && remainingTargets > 0) {
    const currentCell = queue[head++];
    const currentDistance = distGrid[currentCell];
    const x = currentCell % width;
    const y = (currentCell / width) | 0;

    const pointIndex = pointKeyToIndex.get(`${x},${y}`);
    if (pointIndex !== undefined) {
      const bucket = jobsAtPoint.get(pointIndex);
      if (bucket !== undefined) {
        for (let i = 0; i < bucket.length; i++) {
          const jobIndex = bucket[i];
          if (result[jobIndex] === -1) {
            result[jobIndex] = currentDistance;
            remainingTargets--;
          }
        }
      }
    }

    const nextDistance = currentDistance + 1;

    if (x > 0) {
      const nextCell = currentCell - 1;
      if (distGrid[nextCell] === -1 && grid[y].charCodeAt(x - 1) !== 35) {
        distGrid[nextCell] = nextDistance;
        queue[tail++] = nextCell;
      }
    }
    if (x + 1 < width) {
      const nextCell = currentCell + 1;
      if (distGrid[nextCell] === -1 && grid[y].charCodeAt(x + 1) !== 35) {
        distGrid[nextCell] = nextDistance;
        queue[tail++] = nextCell;
      }
    }
    if (y > 0) {
      const nextCell = currentCell - width;
      if (distGrid[nextCell] === -1 && grid[y - 1].charCodeAt(x) !== 35) {
        distGrid[nextCell] = nextDistance;
        queue[tail++] = nextCell;
      }
    }
    if (y + 1 < height) {
      const nextCell = currentCell + width;
      if (distGrid[nextCell] === -1 && grid[y + 1].charCodeAt(x) !== 35) {
        distGrid[nextCell] = nextDistance;
        queue[tail++] = nextCell;
      }
    }
  }

  sourceToJobDistances.set(sourcePointIndex, result);
  return result;
}

const assignments: number[][] = [];
let maxRobotId = -1;
for (let i = 0; i < robots.length; i++) {
  if (robots[i].id > maxRobotId) {
    maxRobotId = robots[i].id;
  }
}
for (let i = 0; i <= maxRobotId; i++) {
  assignments.push([]);
}

robots.sort((a, b) => a.id - b.id);

const robotPositions = new Array<number>(robots.length);
for (let i = 0; i < robots.length; i++) {
  robotPositions[i] = getPointIndex(robots[i].x, robots[i].y);
}

const remainingJobIndices = new Array<number>(jobs.length);
for (let i = 0; i < jobs.length; i++) {
  remainingJobIndices[i] = i;
}

remainingJobIndices.sort((a, b) => {
  const pa = jobs[a].priority;
  const pb = jobs[b].priority;
  if (pa !== pb) {
    return pb - pa;
  }
  const ya = jobs[a].y;
  const yb = jobs[b].y;
  if (ya !== yb) {
    return ya - yb;
  }
  const xa = jobs[a].x;
  const xb = jobs[b].x;
  if (xa !== xb) {
    return xa - xb;
  }
  return jobs[a].id - jobs[b].id;
});

for (let i = 0; i < remainingJobIndices.length; i++) {
  const jobIndex = remainingJobIndices[i];
  let bestRobotListIndex = 0;
  let bestDistance = Number.POSITIVE_INFINITY;
  let bestAssignedCount = Number.POSITIVE_INFINITY;
  let bestRobotId = Number.POSITIVE_INFINITY;

  for (let r = 0; r < robots.length; r++) {
    const distances = computeDistancesToJobs(robotPositions[r]);
    const d = distances[jobIndex];
    const distance = d >= 0 ? d : 1_000_000_000;
    const assignedCount = assignments[robots[r].id]?.length ?? 0;
    const robotId = robots[r].id;

    if (
      distance < bestDistance ||
      (distance === bestDistance && assignedCount < bestAssignedCount) ||
      (distance === bestDistance && assignedCount === bestAssignedCount && robotId < bestRobotId)
    ) {
      bestRobotListIndex = r;
      bestDistance = distance;
      bestAssignedCount = assignedCount;
      bestRobotId = robotId;
    }
  }

  const chosenRobot = robots[bestRobotListIndex];
  assignments[chosenRobot.id].push(jobs[jobIndex].id);
  robotPositions[bestRobotListIndex] = jobPointIndices[jobIndex];
}

process.stdout.write(JSON.stringify({ assignments }));
