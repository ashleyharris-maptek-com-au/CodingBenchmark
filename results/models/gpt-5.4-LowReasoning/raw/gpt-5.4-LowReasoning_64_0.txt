type Cell = [number, number];

declare const require: any;
declare const process: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type DroneInput = {
  id: string | number;
  x: number;
  y: number;
};

const probability: number[][] = Array.isArray(data.probability) ? data.probability : [];
const drones: DroneInput[] = Array.isArray(data.drones) ? data.drones : [];
const maxStepsPerDrone: number = Math.max(0, (data.maxStepsPerDrone | 0) || 0);

const height = probability.length;
const width = height > 0 && Array.isArray(probability[0]) ? probability[0].length : 0;
const cellCount = width * height;

function indexOf(x: number, y: number): number {
  return y * width + x;
}

function inBounds(x: number, y: number): boolean {
  return x >= 0 && y >= 0 && x < width && y < height;
}

function isPassable(x: number, y: number): boolean {
  return inBounds(x, y) && probability[y][x] !== -1;
}

if (height === 0 || width === 0 || cellCount === 0) {
  const emptyPaths = drones.map(() => [[0, 0]]);
  process.stdout.write(JSON.stringify({ paths: emptyPaths }));
  process.exit(0);
}

const positiveValue = new Float64Array(cellCount);
const passable = new Uint8Array(cellCount);

for (let y = 0; y < height; y++) {
  const row = probability[y];
  for (let x = 0; x < width; x++) {
    const idx = indexOf(x, y);
    const v = row[x];
    if (v !== -1) {
      passable[idx] = 1;
      if (v > 0) positiveValue[idx] = v;
    }
  }
}

const potential = new Float64Array(cellCount);
const kernelRadius = 2;
for (let y = 0; y < height; y++) {
  for (let x = 0; x < width; x++) {
    const idx = indexOf(x, y);
    if (!passable[idx]) continue;
    let sum = positiveValue[idx] * 1.5;
    for (let dy = -kernelRadius; dy <= kernelRadius; dy++) {
      const ny = y + dy;
      if (ny < 0 || ny >= height) continue;
      for (let dx = -kernelRadius; dx <= kernelRadius; dx++) {
        const nx = x + dx;
        if (nx < 0 || nx >= width) continue;
        const dist = Math.abs(dx) + Math.abs(dy);
        if (dist === 0 || dist > kernelRadius) continue;
        const nidx = indexOf(nx, ny);
        const pv = positiveValue[nidx];
        if (pv > 0) {
          sum += pv / (dist + 1);
        }
      }
    }
    potential[idx] = sum;
  }
}

const covered = new Uint8Array(cellCount);
const plannedTargetCount = new Int16Array(cellCount);

function markCoveredAt(x: number, y: number): void {
  if (!inBounds(x, y)) return;
  const idx = indexOf(x, y);
  if (passable[idx] && positiveValue[idx] > 0) covered[idx] = 1;
}

for (const d of drones) {
  if (isPassable(d.x, d.y)) {
    markCoveredAt(d.x, d.y);
  }
}

const queue = new Int32Array(cellCount);
const dist = new Int32Array(cellCount);
const prev = new Int32Array(cellCount);
const seenStamp = new Int32Array(cellCount);
let bfsStamp = 0;

function evaluateCell(idx: number, distance: number, remaining: number): number {
  const baseValue = covered[idx] ? 0 : positiveValue[idx];
  const area = potential[idx];
  const slack = remaining - distance;
  const congestion = plannedTargetCount[idx] * 0.35;
  return baseValue * 5.0 + area * 1.2 + Math.min(slack, 8) * 0.03 - distance * 0.55 - congestion;
}

function reconstructPath(endIdx: number, startIdx: number): number[] {
  const rev: number[] = [];
  let cur = endIdx;
  rev.push(cur);
  while (cur !== startIdx) {
    cur = prev[cur];
    if (cur < 0) break;
    rev.push(cur);
  }
  rev.reverse();
  return rev;
}

function findBestReachableTarget(startX: number, startY: number, remaining: number): number[] {
  const startIdx = indexOf(startX, startY);
  bfsStamp++;
  let head = 0;
  let tail = 0;
  queue[tail++] = startIdx;
  seenStamp[startIdx] = bfsStamp;
  dist[startIdx] = 0;
  prev[startIdx] = -1;

  let bestIdx = startIdx;
  let bestScore = evaluateCell(startIdx, 0, remaining);

  while (head < tail) {
    const idx = queue[head++];
    const d = dist[idx];
    if (d > remaining) continue;

    const score = evaluateCell(idx, d, remaining);
    if (score > bestScore + 1e-9 || (Math.abs(score - bestScore) <= 1e-9 && d < dist[bestIdx])) {
      bestScore = score;
      bestIdx = idx;
    }

    if (d === remaining) continue;

    const x = idx % width;
    const y = (idx / width) | 0;
    const neighbors = [
      [x + 1, y],
      [x - 1, y],
      [x, y + 1],
      [x, y - 1],
    ];
    for (let i = 0; i < 4; i++) {
      const nx = neighbors[i][0];
      const ny = neighbors[i][1];
      if (!inBounds(nx, ny)) continue;
      const nidx = indexOf(nx, ny);
      if (!passable[nidx] || seenStamp[nidx] === bfsStamp) continue;
      seenStamp[nidx] = bfsStamp;
      dist[nidx] = d + 1;
      prev[nidx] = idx;
      queue[tail++] = nidx;
    }
  }

  return reconstructPath(bestIdx, startIdx);
}

function appendStayMoves(path: Cell[], totalMoves: number): void {
  while (path.length < totalMoves + 1) {
    const last = path[path.length - 1];
    path.push([last[0], last[1]]);
  }
}

const paths: Cell[][] = [];

for (let droneIndex = 0; droneIndex < drones.length; droneIndex++) {
  const drone = drones[droneIndex];
  let x = isPassable(drone.x, drone.y) ? drone.x : 0;
  let y = isPassable(drone.x, drone.y) ? drone.y : 0;

  if (!isPassable(x, y)) {
    outer:
    for (let yy = 0; yy < height; yy++) {
      for (let xx = 0; xx < width; xx++) {
        if (isPassable(xx, yy)) {
          x = xx;
          y = yy;
          break outer;
        }
      }
    }
  }

  const path: Cell[] = [[x, y]];
  markCoveredAt(x, y);

  while (path.length - 1 < maxStepsPerDrone) {
    const remaining = maxStepsPerDrone - (path.length - 1);
    const targetPath = findBestReachableTarget(x, y, remaining);

    if (targetPath.length <= 1) break;

    const targetIdx = targetPath[targetPath.length - 1];
    plannedTargetCount[targetIdx]++;

    for (let i = 1; i < targetPath.length && path.length - 1 < maxStepsPerDrone; i++) {
      const idx = targetPath[i];
      x = idx % width;
      y = (idx / width) | 0;
      path.push([x, y]);
      markCoveredAt(x, y);
    }

    plannedTargetCount[targetIdx]--;

    if (targetPath.length === 1) break;
  }

  appendStayMoves(path, maxStepsPerDrone);
  paths.push(path);
}

process.stdout.write(JSON.stringify({ paths }));
