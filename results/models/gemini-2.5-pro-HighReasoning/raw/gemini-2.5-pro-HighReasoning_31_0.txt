fn subset_sum(set: &[i32], sum: i32) -> bool {
    let n = set.len();
    let mut dp = vec![vec![false; (sum + 1) as usize]; n + 1];

    for i in 0..=n {
        dp[i][0] = true;
    }

    for i in 1..=n {
        for j in 1..=sum {
            if set[i - 1] > j {
                dp[i][j as usize] = dp[i - 1][j as usize];
            } else {
                dp[i][j as usize] = dp[i - 1][j as usize] || dp[i - 1][(j - set[i - 1]) as usize];
            }
        }
    }

    dp[n][sum as usize]
}

fn main() {
    let set = [3, 34, 4, 12, 5, 2];
    let sum = 9;
    if subset_sum(&set, sum) {
        println!("Found a subset with given sum");
    } else {
        println!("No subset with given sum");
    }

    let set2 = [3, 34, 4, 12, 5, 2];
    let sum2 = 30;
    if subset_sum(&set2, sum2) {
        println!("Found a subset with given sum");
    } else {
        println!("No subset with given sum");
    }
}
