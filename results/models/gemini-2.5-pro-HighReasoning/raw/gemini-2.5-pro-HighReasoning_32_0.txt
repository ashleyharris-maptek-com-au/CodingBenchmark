#include "SteinerTree.h"
#include <iostream>
#include <vector>
#include <queue>
#include <map>
#include <algorithm>
#include <climits>

SteinerTree::SteinerTree(const Graph& g, const std::set<int>& terminals) : g(g), terminals(terminals), steiner_tree(g.get_n()) {}

void SteinerTree::find_steiner_tree() {
    std::cout << "Finding Steiner Tree..." << std::endl;

    // 1. Create a complete graph of terminals
    std::vector<int> terminal_vec(terminals.begin(), terminals.end());
    int num_terminals = terminal_vec.size();
    if (num_terminals < 2) {
        // Not a valid Steiner problem
        return;
    }

    Graph terminal_graph(num_terminals);
    std::vector<std::vector<int>> paths(num_terminals * num_terminals);

    for (int i = 0; i < num_terminals; ++i) {
        for (int j = i + 1; j < num_terminals; ++j) {
            int u = terminal_vec[i];
            int v = terminal_vec[j];
            auto shortest = g.shortest_path(u, v);
            terminal_graph.add_edge(i, j, shortest.first);
            paths[i * num_terminals + j] = shortest.second;
        }
    }

    // 2. Find MST of the terminal graph (Prim's algorithm)
    std::vector<int> parent(num_terminals, -1);
    std::vector<int> key(num_terminals, INT_MAX);
    std::vector<bool> mst_set(num_terminals, false);

    key[0] = 0;
    parent[0] = -1;

    for (int count = 0; count < num_terminals - 1; ++count) {
        int min_key = INT_MAX;
        int u = -1;

        for (int v = 0; v < num_terminals; ++v) {
            if (!mst_set[v] && key[v] < min_key) {
                min_key = key[v];
                u = v;
            }
        }

        if (u == -1) {
            continue;
        }

        mst_set[u] = true;

        for (auto& edge : terminal_graph.get_adj()[u]) {
            int v = edge.first;
            int weight = edge.second;
            if (!mst_set[v] && weight < key[v]) {
                parent[v] = u;
                key[v] = weight;
            }
        }
    }

    // 3. Reconstruct the Steiner tree
    for (int i = 1; i < num_terminals; ++i) {
        int u_idx = parent[i];
        int v_idx = i;

        if (u_idx > v_idx) std::swap(u_idx, v_idx);

        const auto& path = paths[u_idx * num_terminals + v_idx];
        for (size_t j = 0; j < path.size() - 1; ++j) {
            int u_path = path[j];
            int v_path = path[j+1];

            int weight = -1;
            for(auto& edge : g.get_adj()[u_path]) {
                if (edge.first == v_path) {
                    weight = edge.second;
                    break;
                }
            }
            if (weight != -1) {
                steiner_tree.add_edge(u_path, v_path, weight);
            }
        }
    }
}

void SteinerTree::print_steiner_tree() {
    std::cout << "Steiner Tree:" << std::endl;
    steiner_tree.print_graph();
}
