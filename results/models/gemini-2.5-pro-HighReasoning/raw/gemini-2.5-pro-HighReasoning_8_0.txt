use std::collections::VecDeque;
use std::io::{self, BufRead, BufWriter, Write};

fn main() {
    let stdin = io::stdin();
    let mut reader = stdin.lock();

    let mut dimensions = String::new();
    reader.read_line(&mut dimensions).unwrap();
    let dimensions: Vec<usize> = dimensions
        .trim()
        .split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();
    let (h, w) = (dimensions[0], dimensions[1]);

    let mut maze = vec![0; h * w];
    let mut start = 0;
    let mut end = 0;

    for (r, line) in reader.lines().enumerate().take(h) {
        let line = line.unwrap();
        for (c, char) in line.chars().enumerate() {
            let index = r * w + c;
            maze[index] = match char {
                '#' => 1,
                ' ' => 0,
                'A' => {
                    start = index;
                    0
                }
                'B' => {
                    end = index;
                    0
                }
                _ => 0,
            };
        }
    }

    let mut parents = vec![0; h * w];
    let mut queue = VecDeque::new();

    queue.push_back(start);
    parents[start] = 5; // Special marker for the start node

    let mut found = false;
    while let Some(curr) = queue.pop_front() {
        if curr == end {
            found = true;
            break;
        }

        let r = curr / w;
        let c = curr % w;

        // Neighbors: Up, Down, Left, Right
        let neighbors = [
            (r.wrapping_sub(1), c, 1u8), // Up, parent = 1 (N)
            (r + 1, c, 2u8),             // Down, parent = 2 (S)
            (r, c.wrapping_sub(1), 3u8), // Left, parent = 3 (W)
            (r, c + 1, 4u8),             // Right, parent = 4 (E)
        ];

        for &(nr, nc, dir) in &neighbors {
            if nr < h && nc < w {
                let next_idx = nr * w + nc;
                if maze[next_idx] == 0 && parents[next_idx] == 0 {
                    parents[next_idx] = dir;
                    queue.push_back(next_idx);
                }
            }
        }
    }

    if found {
        let mut path = Vec::new();
        let mut curr = end;
        while curr != start {
            path.push(curr);
            let r = curr / w;
            let c = curr % w;
            match parents[curr] {
                1 => curr = (r + 1) * w + c, // Parent was South, so we came from South
                2 => curr = (r - 1) * w + c, // Parent was North
                3 => curr = r * w + c + 1, // Parent was East
                4 => curr = r * w + c - 1, // Parent was West
                _ => break,
            }
        }
        path.push(start);
        path.reverse();

        let stdout = io::stdout();
        let mut writer = BufWriter::new(stdout.lock());
        writeln!(writer, "{}", path.len() -1).unwrap();
        for &index in &path {
            let r = index / w;
            let c = index % w;
            writeln!(writer, "{} {}", c, r).unwrap();
        }
    } else {
        println!("-1");
    }
}
