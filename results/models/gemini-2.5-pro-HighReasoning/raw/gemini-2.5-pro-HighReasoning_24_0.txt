#include "include/Game.h"
#include "include/Lander.h"
#include <iostream>
#include <cstdlib> // For rand()

Game::Game() : isRunning(false) {
    std::cout << "Game created" << std::endl;
}

void Game::run() {
    isRunning = true;
    while (isRunning) {
        processEvents();
        update(0.016f); // Assuming 60 FPS
        render();

        if (lander.getAltitude() <= 0) {
            isRunning = false;
            std::cout << "Lander has landed!" << std::endl;
        }
    }
}

void Game::processEvents() {
    // A simple way to simulate thruster input
    // In a real game, this would come from keyboard or controller input
    if (rand() % 100 < 20) { // 20% chance of thruster firing
        lander.applyThruster();
    }
}

void Game::update(float dt) {
    lander.update(dt);
}

void Game::render() {
    // Clear the screen (optional, but good practice)
    // system("cls"); // For Windows
    // system("clear"); // For Linux/macOS

    std::cout << "Altitude: " << lander.getAltitude() << " m" << std::endl;
    std::cout << "Velocity: " << lander.getVelocity() << " m/s" << std::endl;
    std::cout << "Fuel: " << lander.getFuel() << std::endl;
    std::cout << "--------------------" << std::endl;
}
