use std::io::{self, BufRead};
use std::sync::mpsc::{self, Receiver, Sender};
use std::thread;
use std::time::Duration;

struct PIDController {
    kp: f64,
    ki: f64,
    kd: f64,
    integral: f64,
    previous_error: f64,
}

impl PIDController {
    fn new(kp: f64, ki: f64, kd: f64) -> Self {
        PIDController {
            kp,
            ki,
            kd,
            integral: 0.0,
            previous_error: 0.0,
        }
    }

    fn update(&mut self, error: f64, dt: f64) -> f64 {
        self.integral += error * dt;
        let derivative = (error - self.previous_error) / dt;
        let output = self.kp * error + self.ki * self.integral + self.kd * derivative;
        self.previous_error = error;
        output
    }
}

fn main() {
    // Create channels for communication between threads
    let (stdin_sender, stdin_receiver) = mpsc::channel();
    let (controller_sender, controller_receiver) = mpsc::channel();

    // Spawn stdin reader thread
    let stdin_thread = thread::spawn(move || {
        let stdin = io::stdin();
        let mut reader = stdin.lock();
        let mut line = String::new();

        loop {
            match reader.read_line(&mut line) {
                Ok(bytes_read) if bytes_read > 0 => {
                    if let Err(e) = stdin_sender.send(line.clone()) {
                        eprintln!("Failed to send line: {}", e);
                        break;
                    }
                    line.clear();
                }
                Ok(_) => break, // EOF
                Err(e) => {
                    eprintln!("Error reading from stdin: {}", e);
                    break;
                }
            }
        }
    });

    // Spawn controller thread
    let controller_thread = thread::spawn(move || {
        // Map and game data
        let mut width = 0;
        let mut height = 0;
        let mut gravity = 0.0;
        let mut max_thrust = 0.0;
        let mut fuel = 0.0;
        let mut start_x = 0.0;
        let mut start_y = 0.0;
        let mut target_x = 0.0;
        let mut target_y = 0.0;
        let mut map: Vec<String> = Vec::new();

        let mut lines = stdin_receiver.iter();

        // Parse map data
        if let Some(line) = lines.next() {
            let parts: Vec<&str> = line.split_whitespace().collect();
            if parts.len() >= 5 {
                width = parts[0].parse().unwrap_or_default();
                height = parts[1].parse().unwrap_or_default();
                gravity = parts[2].parse().unwrap_or_default();
                max_thrust = parts[3].parse().unwrap_or_default();
                fuel = parts[4].parse().unwrap_or_default();
            }
        }

        if let Some(line) = lines.next() {
            let parts: Vec<f64> = line.split_whitespace().map(|s| s.parse().unwrap_or_default()).collect();
            if parts.len() >= 4 {
                start_x = parts[0];
                start_y = parts[1];
                target_x = parts[2];
                target_y = parts[3];
            }
        }

        for _ in 0..height {
            if let Some(line) = lines.next() {
                map.push(line.trim().to_string());
            }
        }

        // PID controllers
        let mut altitude_pid = PIDController::new(0.5, 0.1, 0.2);
        let mut angle_pid = PIDController::new(0.8, 0.2, 0.3);

        let dt = 0.1;

        // Initial state
        let mut x = start_x;
        let mut y = start_y;
        let mut vx = 0.0;
        let mut vy = 0.0;
        let mut angle = 0.0;
        let mut angular_vel = 0.0;
        let mut fuel_remaining = fuel;

        for line in lines {
            let parts: Vec<f64> = line
                .split_whitespace()
                .map(|s| s.parse().unwrap_or_default())
                .collect();
            
            if parts.len() >= 7 {
                x = parts[0];
                y = parts[1];
                vx = parts[2];
                vy = parts[3];
                angle = parts[4];
                angular_vel = parts[5];
                fuel_remaining = parts[6];
            }
            
            let dist_to_target = ((target_x - x).powi(2) + (target_y - y).powi(2)).sqrt();
            
            let (thrust, turn) = if dist_to_target < 10.0 && y > target_y - 5.0 {
                // Final landing sequence
                let error_y = (target_y - 5.0) - y;
                let altitude_correction = altitude_pid.update(error_y, dt);

                let error_vx = 0.0 - vx;
                let error_vy = 1.0 - vy; // Target a slow descent
                
                let target_angle = (error_vx * 0.5).max(-0.2).min(0.2);
                let error_angle = target_angle - angle;
                let angle_correction = angle_pid.update(error_angle - angular_vel * 0.5, dt);
                
                let thrust = (gravity / max_thrust + altitude_correction + error_vy * 0.1).max(0.0).min(1.0);
                let turn = angle_correction.max(-1.0).min(1.0);
                (thrust, turn)
            } else {
                // Navigate towards target
                let error_x = target_x - x;
                
                let target_vy = if y > target_y - 20.0 { -1.0 } else { 5.0 };
                let error_y = target_y - y;

                let error_vy = target_vy - vy;
                let altitude_correction = altitude_pid.update(error_vy, dt);

                let target_angle = (error_x * 0.01).max(-0.7).min(0.7);
                let error_angle = target_angle - angle;
                let angle_correction = angle_pid.update(error_angle - angular_vel * 0.3, dt);

                let thrust = (altitude_correction).max(0.0).min(1.0);
                let turn = angle_correction.max(-1.0).min(1.0);
                (thrust, turn)
            };

            if let Err(e) = controller_sender.send((thrust, turn)) {
                eprintln!("Failed to send command: {}", e);
                break;
            }
            thread::sleep(Duration::from_millis(20));
        }
    });

    // Main thread for writing to stdout
    for (thrust, turn) in controller_receiver {
        println!("{} {}", thrust, turn);
    }

    // Wait for threads to finish
    if let Err(e) = stdin_thread.join() {
        eprintln!("Error joining stdin thread: {:?}", e);
    }
    if let Err(e) = controller_thread.join() {
        eprintln!("Error joining controller thread: {:?}", e);
    }
}
