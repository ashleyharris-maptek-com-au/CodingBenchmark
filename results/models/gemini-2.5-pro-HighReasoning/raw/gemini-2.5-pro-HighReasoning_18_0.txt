#include <iostream>
#include <vector>
#include <random>
#include <algorithm>
#include <map>

struct Edge {
    int src, dest;
};

class Graph {
public:
    int V, E;
    std::vector<Edge> edges;

    Graph(int v, int e) : V(v), E(e) {}

    void addEdge(int src, int dest) {
        Edge new_edge;
        new_edge.src = src;
        new_edge.dest = dest;
        edges.push_back(new_edge);
    }
};

int kargerMinCut(Graph& graph) {
    int V = graph.V;
    std::vector<int> parent(V);
    for (int i = 0; i < V; ++i) {
        parent[i] = i;
    }

    int vertices = V;
    while (vertices > 2) {
        int i = rand() % graph.E;
        
        int subset1 = parent[graph.edges[i].src];
        int subset2 = parent[graph.edges[i].dest];

        if (subset1 != subset2) {
            vertices--;
            for (int j = 0; j < V; ++j) {
                if (parent[j] == subset2) {
                    parent[j] = subset1;
                }
            }
        }
    }

    int cut_edges = 0;
    for (int i = 0; i < graph.E; ++i) {
        if (parent[graph.edges[i].src] != parent[graph.edges[i].dest]) {
            cut_edges++;
        }
    }

    return cut_edges;
}

int main() {
    srand(time(NULL));

    int V = 4, E = 5;
    Graph graph(V, E);

    graph.addEdge(0, 1);
    graph.addEdge(0, 2);
    graph.addEdge(0, 3);
    graph.addEdge(1, 3);
    graph.addEdge(2, 3);

    int min_cut = E;

    for (int i = 0; i < V * V; ++i) {
        Graph current_graph = graph;
        min_cut = std::min(min_cut, kargerMinCut(current_graph));
    }

    std::cout << "Minimum cut is " << min_cut << std::endl;

    return 0;
}
