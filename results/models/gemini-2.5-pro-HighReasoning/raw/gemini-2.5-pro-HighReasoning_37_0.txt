using System;
using System.Collections.Generic;
using System.Linq;

public class DirectedFeedbackVertexSet
{
    private readonly int V;
    private readonly List<int>[] adj;
    private readonly Dictionary<string, List<int>> memo = new Dictionary<string, List<int>>();

    public DirectedFeedbackVertexSet(int v)
    {
        V = v;
        adj = new List<int>[v];
        for (int i = 0; i < v; i++)
        {
            adj[i] = new List<int>();
        }
    }

    public void AddEdge(int u, int v)
    {
        if (u >= 0 && u < V && v >= 0 && v < V)
        {
            adj[u].Add(v);
        }
    }

    public List<int> FindMinFVS()
    {
        var allVertices = new HashSet<int>(Enumerable.Range(0, V));
        return Solve(allVertices);
    }

    private List<int> Solve(HashSet<int> currentVertices)
    {
        var key = string.Join(",", currentVertices.OrderBy(x => x));
        if (memo.ContainsKey(key))
        {
            return new List<int>(memo[key]);
        }

        var cycle = FindCycle(currentVertices);

        if (cycle == null || cycle.Count == 0)
        {
            return new List<int>();
        }

        List<int>? minFvs = null;

        foreach (var vertex in cycle)
        {
            var nextVertices = new HashSet<int>(currentVertices);
            nextVertices.Remove(vertex);

            var fvs = Solve(nextVertices);
            fvs.Add(vertex);

            if (minFvs == null || fvs.Count < minFvs.Count)
            {
                minFvs = fvs;
            }
        }

        var result = minFvs ?? new List<int>();
        memo[key] = new List<int>(result);
        return result;
    }

    private List<int>? FindCycle(HashSet<int> vertices)
    {
        var visiting = new HashSet<int>();
        var visited = new HashSet<int>();
        var parent = new Dictionary<int, int>();

        foreach (var v in vertices)
        {
            if (!visited.Contains(v))
            {
                var cycle = DFSCycleCheck(v, vertices, visiting, visited, parent);
                if (cycle != null)
                {
                    return cycle;
                }
            }
        }
        return null;
    }

    private List<int>? DFSCycleCheck(int u, HashSet<int> vertices, HashSet<int> visiting, HashSet<int> visited, Dictionary<int, int> parent)
    {
        visiting.Add(u);

        foreach (var v in adj[u])
        {
            if (!vertices.Contains(v)) continue;

            if (visiting.Contains(v))
            {
                var cycle = new List<int> { v };
                int curr = u;
                while (curr != v && parent.ContainsKey(curr))
                {
                    cycle.Add(curr);
                    curr = parent[curr];
                }
                cycle.Reverse();
                return cycle;
            }

            if (!visited.Contains(v))
            {
                parent[v] = u;
                var cycle = DFSCycleCheck(v, vertices, visiting, visited, parent);
                if (cycle != null)
                {
                    return cycle;
                }
            }
        }

        visiting.Remove(u);
        visited.Add(u);
        return null;
    }
}

public class Program
{
    public static void Main(string[] args)
    {
        Console.WriteLine("Solving Directed Feedback Vertex Set problem.");

        // Example 1: A simple cycle
        var dfvs1 = new DirectedFeedbackVertexSet(3);
        dfvs1.AddEdge(0, 1);
        dfvs1.AddEdge(1, 2);
        dfvs1.AddEdge(2, 0);

        var fvs1 = dfvs1.FindMinFVS();
        Console.WriteLine($"Example 1: Minimum FVS size: {fvs1.Count}");
        Console.WriteLine($"FVS: [{string.Join(", ", fvs1)}]");

        // Example 2: A more complex graph
        var dfvs2 = new DirectedFeedbackVertexSet(4);
        dfvs2.AddEdge(0, 1);
        dfvs2.AddEdge(1, 2);
        dfvs2.AddEdge(2, 0); // Cycle 0-1-2
        dfvs2.AddEdge(0, 3);
        dfvs2.AddEdge(3, 2);

        var fvs2 = dfvs2.FindMinFVS();
        Console.WriteLine($"Example 2: Minimum FVS size: {fvs2.Count}");
        Console.WriteLine($"FVS: [{string.Join(", ", fvs2)}]");
    }
}
