using System;
using System.Collections.Generic;
using System.Globalization;
using System.Linq;

public struct Vector
{
    public double X, Y, Z;
    public Vector(double x, double y, double z) { X = x; Y = y; Z = z; }
    public static Vector operator +(Vector a, Vector b) => new Vector(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
    public static Vector operator -(Vector a, Vector b) => new Vector(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
    public static Vector operator *(Vector a, double s) => new Vector(a.X * s, a.Y * s, a.Z * s);
    public static Vector operator /(Vector a, double s) => new Vector(a.X / s, a.Y / s, a.Z / s);
    public double MagnitudeSquared() => X * X + Y * Y + Z * Z;
    public double Magnitude() => Math.Sqrt(MagnitudeSquared());
    public Vector Normalize() => this / Magnitude();
    public override string ToString() => $"{X.ToString(CultureInfo.InvariantCulture)} {Y.ToString(CultureInfo.InvariantCulture)} {Z.ToString(CultureInfo.InvariantCulture)}";
}

public class Body
{
    public string Name { get; set; }
    public double Mass { get; set; }
    public Vector Position { get; set; }
    public Vector Velocity { get; set; }
    public Body(string n, double m, Vector p, Vector v) { Name = n; Mass = m; Position = p; Velocity = v; }
    public Body Clone() => new Body(Name, Mass, Position, Velocity);
}

public class NBody
{
    private const double G = 6.67430e-11;
    public List<Body> Bodies { get; }
    public NBody(List<Body> bodies) { Bodies = bodies.Select(b => b.Clone()).ToList(); }

    public void Step(double dt)
    {
        var forces = Bodies.ToDictionary(b => b, b => new Vector(0, 0, 0));
        for (int i = 0; i < Bodies.Count; i++)
        {
            for (int j = i + 1; j < Bodies.Count; j++)
            {
                var r = Bodies[j].Position - Bodies[i].Position;
                var d2 = r.MagnitudeSquared();
                if (d2 == 0) continue;
                var f = G * Bodies[i].Mass * Bodies[j].Mass / d2;
                var force = r.Normalize() * f;
                forces[Bodies[i]] += force;
                forces[Bodies[j]] -= force;
            }
        }
        foreach (var body in Bodies)
        {
            var a = forces[body] / body.Mass;
            body.Velocity += a * dt;
            body.Position += body.Velocity * dt;
        }
    }
}

public class Program
{
    const double DaySec = 86400;

    public static void Main()
    {
        var parts = Console.ReadLine().Split();
        var numBodies = int.Parse(parts[0]);
        var warningDays = double.Parse(parts[1], CultureInfo.InvariantCulture);
        var impactorMass = double.Parse(parts[2], CultureInfo.InvariantCulture);
        var dVBudget = double.Parse(parts[3], CultureInfo.InvariantCulture);

        var initialBodies = new List<Body>();
        for (int i = 0; i < numBodies; i++)
        {
            parts = Console.ReadLine().Split();
            initialBodies.Add(new Body(parts[0], double.Parse(parts[1], CultureInfo.InvariantCulture),
                new Vector(double.Parse(parts[2], CultureInfo.InvariantCulture), double.Parse(parts[3], CultureInfo.InvariantCulture), double.Parse(parts[4], CultureInfo.InvariantCulture)),
                new Vector(double.Parse(parts[5], CultureInfo.InvariantCulture), double.Parse(parts[6], CultureInfo.InvariantCulture), double.Parse(parts[7], CultureInfo.InvariantCulture))));
        }

        double launchTimeDays = 0;
        double interceptTimeDays = warningDays * 0.9;

        Vector burnDv = FindBurn(initialBodies, impactorMass, dVBudget, launchTimeDays, interceptTimeDays);

        Console.WriteLine("2");
        Console.WriteLine($"{launchTimeDays.ToString(CultureInfo.InvariantCulture)} {burnDv.ToString()}");
        Console.WriteLine($"{interceptTimeDays.ToString(CultureInfo.InvariantCulture)} 0 0 0");
    }

    static Vector FindBurn(List<Body> initialBodies, double impactorMass, double dVBudget, double launchTime, double interceptTime)
    {
        var sim = new NBody(initialBodies);
        for(double t=0; t<launchTime*DaySec; t+=3600) sim.Step(3600);
        var earth = sim.Bodies.First(b => b.Name == "Earth");

        var targetSim = new NBody(initialBodies);
        for(double t=0; t<interceptTime*DaySec; t+=3600) targetSim.Step(3600);
        var asteroidTarget = targetSim.Bodies.First(b => b.Name == "Asteroid").Position;

        Vector vGuess = (asteroidTarget - earth.Position) / ((interceptTime - launchTime) * DaySec) - earth.Velocity;
        vGuess = vGuess.Magnitude() > dVBudget ? vGuess.Normalize() * dVBudget : vGuess;

        for(int iter=0; iter<5; iter++) // Iteratively refine guess
        {
            var testSim = new NBody(sim.Bodies);
            var spacecraft = new Body("Spacecraft", impactorMass, earth.Position, earth.Velocity + vGuess);
            testSim.Bodies.Add(spacecraft);

            for(double t=launchTime*DaySec; t<interceptTime*DaySec; t+=3600) testSim.Step(3600);

            var finalPos = testSim.Bodies.First(b=>b.Name=="Spacecraft").Position;
            var error = asteroidTarget - finalPos;
            
            var correctionVel = error / ((interceptTime - launchTime) * DaySec);
            vGuess += correctionVel;

            if (vGuess.Magnitude() > dVBudget) vGuess = vGuess.Normalize() * dVBudget;
        }
        return vGuess;
    }
}
