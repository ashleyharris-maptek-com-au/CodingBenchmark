use std::io::{self, Read};
use geo::{Polygon as GeoPolygon, Point as GeoPoint, Area, Contains, Intersects, BoundingRect};
use nalgebra::{Rotation2, Vector2};

/// Represents a piece to be placed on the stock.
#[derive(Debug, Clone)]
struct Piece {
    polygon: GeoPolygon<f64>,
    id: usize,
}

/// Represents a piece that has already been placed on the stock.
#[derive(Debug, Clone)]
struct PlacedPiece {
    polygon: GeoPolygon<f64>,
    id: usize,
    stock_index: usize,
}

/// Rotates a polygon around the origin (0, 0).
fn rotate(polygon: &GeoPolygon<f64>, angle: f64) -> GeoPolygon<f64> {
    let rot = Rotation2::new(angle);
    let rotated_vertices = polygon.exterior().points().map(|p| {
        let point_vec = Vector2::new(p.x(), p.y());
        let rotated_point = rot * point_vec;
        GeoPoint::new(rotated_point.x, rotated_point.y)
    }).collect::<Vec<_>>();
    GeoPolygon::new(rotated_vertices.into(), vec![])
}

/// Translates a polygon by a given offset.
fn translate(polygon: &GeoPolygon<f64>, dx: f64, dy: f64) -> GeoPolygon<f64> {
    let translated_vertices = polygon.exterior().points().map(|p| {
        GeoPoint::new(p.x() + dx, p.y() + dy)
    }).collect::<Vec<_>>();
    GeoPolygon::new(translated_vertices.into(), vec![])
}

/// Finds a valid placement for a piece on a given stock.
fn find_placement(
    piece: &Piece,
    stock: &GeoPolygon<f64>,
    placed_pieces: &[PlacedPiece],
    rotation_steps: usize,
) -> Option<(f64, f64, f64)> {
    let stock_bounding_box = match stock.bounding_rect() {
        Some(rect) => rect,
        None => return None,
    };
    let (min, max) = (stock_bounding_box.min(), stock_bounding_box.max());

    // Use a step size proportional to the stock size
    let step = (stock_bounding_box.width() + stock_bounding_box.height()) / 100.0;

    let mut x = min.x;
    while x < max.x {
        let mut y = min.y;
        while y < max.y {
            for i in 0..rotation_steps {
                let angle = (i as f64) * (2.0 * std::f64::consts::PI / rotation_steps as f64);
                let rotated_piece = rotate(&piece.polygon, angle);
                let placed_piece = translate(&rotated_piece, x, y);

                if !stock.contains(&placed_piece) {
                    continue;
                }

                let mut intersects = false;
                for other_piece in placed_pieces {
                    if placed_piece.intersects(&other_piece.polygon) {
                        intersects = true;
                        break;
                    }
                }

                if !intersects {
                    return Some((x, y, angle));
                }
            }
            y += step;
        }
        x += step;
    }

    None
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut lines = input.lines();

    let first_line = lines.next().unwrap();
    let mut parts = first_line.split_whitespace();
    let sv: usize = parts.next().unwrap().parse().unwrap();
    let np: usize = parts.next().unwrap().parse().unwrap();

    let stock_vertices: Vec<GeoPoint<f64>> = (0..sv).map(|_| {
        let line = lines.next().unwrap();
        let mut parts = line.split_whitespace();
        let x: f64 = parts.next().unwrap().parse().unwrap();
        let y: f64 = parts.next().unwrap().parse().unwrap();
        GeoPoint::new(x, y)
    }).collect();
    let stock_polygon = GeoPolygon::new(stock_vertices.into(), vec![]);

    let mut pieces: Vec<Piece> = (0..np).map(|i| {
        let line = lines.next().unwrap();
        let pv: usize = line.parse().unwrap();
        let piece_vertices: Vec<GeoPoint<f64>> = (0..pv).map(|_| {
            let line = lines.next().unwrap();
            let mut parts = line.split_whitespace();
            let x: f64 = parts.next().unwrap().parse().unwrap();
            let y: f64 = parts.next().unwrap().parse().unwrap();
            GeoPoint::new(x, y)
        }).collect();
        Piece {
            polygon: GeoPolygon::new(piece_vertices.into(), vec![]),
            id: i,
        }
    }).collect();

    // Sort pieces by area in descending order
    pieces.sort_by(|a, b| b.polygon.unsigned_area().partial_cmp(&a.polygon.unsigned_area()).unwrap());

    let mut placements = vec![(0, 0.0, 0.0, 0.0); np];
    let mut placed_pieces: Vec<PlacedPiece> = Vec::new();
    let mut stocks = vec![stock_polygon];

    let rotation_steps = if np > 1000 { 4 } else if np > 100 { 8 } else { 16 };

    for piece in &pieces {
        let mut placed = false;
        for (stock_index, stock) in stocks.iter().enumerate() {
            let relevant_placed_pieces: Vec<PlacedPiece> = placed_pieces.iter().filter(|p| p.stock_index == stock_index).cloned().collect();
            if let Some((x, y, angle)) = find_placement(piece, stock, &relevant_placed_pieces, rotation_steps) {
                placements[piece.id] = (stock_index, x, y, angle);
                let rotated_piece = rotate(&piece.polygon, angle);
                let final_piece = translate(&rotated_piece, x, y);
                placed_pieces.push(PlacedPiece { polygon: final_piece, id: piece.id, stock_index });
                placed = true;
                break;
            }
        }

        if !placed {
            let new_stock_index = stocks.len();
            let new_stock = stocks[0].clone(); // All new stocks are same as the first one
            stocks.push(new_stock);

            if let Some((x, y, angle)) = find_placement(piece, &stocks[new_stock_index], &[], rotation_steps) {
                 placements[piece.id] = (new_stock_index, x, y, angle);
                let rotated_piece = rotate(&piece.polygon, angle);
                let final_piece = translate(&rotated_piece, x, y);
                placed_pieces.push(PlacedPiece { polygon: final_piece, id: piece.id, stock_index: new_stock_index });
            } else {
                // If the piece cannot be placed in a new empty stock, it's impossible to place.
                // Place it at (0,0,0) in stock 0 as a fallback.
                placements[piece.id] = (0, 0.0, 0.0, 0.0);
            }
        }
    }

    println!("{}", stocks.len());
    for i in 0..np {
        let p = placements[i];
        println!("{} {} {} {}", p.0, p.1, p.2, p.3);
    }
}
