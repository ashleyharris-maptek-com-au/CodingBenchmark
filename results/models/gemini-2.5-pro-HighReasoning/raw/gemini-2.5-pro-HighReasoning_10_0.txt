#include <iostream>
#include <vector>
#include <algorithm>
#include <limits>

struct Rect {
    int id;
    int w, h;
};

struct PlacedRect {
    int id;
    int x, y;
    bool rotated;
};

struct FreeRect {
    int x, y, w, h;
};

struct Board {
    std::vector<PlacedRect> placed_rects;
    std::vector<FreeRect> free_rects;
    Board(int width, int height) {
        free_rects.push_back({0, 0, width, height});
    }
};

bool compareRects(const Rect& a, const Rect& b) {
    return std::max(a.w, a.h) > std::max(b.w, b.h);
}

void split_free_rect(std::vector<FreeRect>& free_rects, const PlacedRect& placed, int rect_w, int rect_h) {
    std::vector<FreeRect> new_free_rects;
    for (const auto& free_rect : free_rects) {
        int fr_x1 = free_rect.x;
        int fr_y1 = free_rect.y;
        int fr_x2 = free_rect.x + free_rect.w;
        int fr_y2 = free_rect.y + free_rect.h;

        int pr_x1 = placed.x;
        int pr_y1 = placed.y;
        int pr_x2 = placed.x + rect_w;
        int pr_y2 = placed.y + rect_h;

        if (pr_x1 >= fr_x2 || pr_x2 <= fr_x1 || pr_y1 >= fr_y2 || pr_y2 <= fr_y1) {
            new_free_rects.push_back(free_rect);
            continue;
        }

        if (pr_x1 > fr_x1)
            new_free_rects.push_back({fr_x1, fr_y1, pr_x1 - fr_x1, fr_y2 - fr_y1});
        if (pr_x2 < fr_x2)
            new_free_rects.push_back({pr_x2, fr_y1, fr_x2 - pr_x2, fr_y2 - fr_y1});
        if (pr_y1 > fr_y1)
            new_free_rects.push_back({fr_x1, fr_y1, fr_x2 - fr_x1, pr_y1 - fr_y1});
        if (pr_y2 < fr_y2)
            new_free_rects.push_back({fr_x1, pr_y2, fr_x2 - fr_x1, fr_y2 - fr_y1});
    }
    free_rects = new_free_rects;
}


bool try_place(Board& board, const Rect& rect, PlacedRect& placed_rect) {
    int best_short_fit = std::numeric_limits<int>::max();
    int best_long_fit = std::numeric_limits<int>::max();
    int best_free_rect_idx = -1;
    bool rotated = false;

    for (int r = 0; r < 2; ++r) {
        int w = r == 0 ? rect.w : rect.h;
        int h = r == 0 ? rect.h : rect.w;

        for (size_t i = 0; i < board.free_rects.size(); ++i) {
            const auto& free_rect = board.free_rects[i];
            if (free_rect.w >= w && free_rect.h >= h) {
                int short_fit = std::min(free_rect.w - w, free_rect.h - h);
                int long_fit = std::max(free_rect.w - w, free_rect.h - h);

                if (short_fit < best_short_fit || (short_fit == best_short_fit && long_fit < best_long_fit)) {
                    best_short_fit = short_fit;
                    best_long_fit = long_fit;
                    best_free_rect_idx = i;
                    rotated = (r == 1);
                }
            }
        }
    }

    if (best_free_rect_idx != -1) {
        const auto& best_free_rect = board.free_rects[best_free_rect_idx];
        placed_rect = {rect.id, best_free_rect.x, best_free_rect.y, rotated};
        
        int w = rotated ? rect.h : rect.w;
        int h = rotated ? rect.w : rect.h;
        
        board.placed_rects.push_back(placed_rect);

        std::vector<FreeRect> new_free_rects;
        for(size_t i=0; i< board.free_rects.size(); ++i){
            if(i != best_free_rect_idx){
                new_free_rects.push_back(board.free_rects[i]);
            }
        }

        split_free_rect(new_free_rects, placed_rect, w, h);
        board.free_rects = new_free_rects;
        
        return true;
    }

    return false;
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);

    int n, board_w, board_h;
    std::cin >> n >> board_w >> board_h;

    std::vector<Rect> rects(n);
    for (int i = 0; i < n; ++i) {
        rects[i].id = i;
        std::cin >> rects[i].w >> rects[i].h;
    }

    std::sort(rects.begin(), rects.end(), compareRects);

    std::vector<Board> boards;
    std::vector<PlacedRect> result_placements(n);

    for (const auto& rect : rects) {
        bool placed = false;
        int best_board_idx = -1;
        PlacedRect pr;
        
        int best_short_fit = std::numeric_limits<int>::max();
        int best_long_fit = std::numeric_limits<int>::max();


        for (size_t i = 0; i < boards.size(); ++i) {
             PlacedRect current_pr;
             if (try_place(boards[i], rect, current_pr)) {
                // This is a simplification, in a real scenario we'd need to properly
                // evaluate the quality of the placement beyond just fitting.
                // For now, first fit is fine.
                best_board_idx = i;
                pr = current_pr;
                placed = true;
                break;
             }
        }

        if (placed) {
             result_placements[rect.id] = pr;
             result_placements[rect.id].id = best_board_idx;
        } else {
            boards.emplace_back(board_w, board_h);
            if (try_place(boards.back(), rect, pr)) {
                result_placements[rect.id] = pr;
                result_placements[rect.id].id = boards.size() - 1;
            }
        }
    }

    std::cout << boards.size() << std::endl;
    for (int i = 0; i < n; ++i) {
        // Find the original rect to get the w/h for the placed rect
        const auto& original_rect = rects[i];
        for(const auto& p : result_placements){
            if(p.id == original_rect.id){
                 std::cout << p.id << " " << p.x << " " << p.y << " " << (p.rotated ? 1 : 0) << std::endl;
                 break;
            }
        }
    }

    return 0;
}
