using System;
using System.Collections.Generic;
using System.Linq;
using System.Runtime.Intrinsics.X86;
using System.Runtime.Intrinsics;
using System.Threading.Tasks;
using System.Numerics;

class Graph
{
    private readonly ulong[] _adjacencyMatrix;
    public int Order { get; }
    private readonly int _rowLength;

    public Graph(int order)
    {
        Order = order;
        _rowLength = (order + 63) / 64;
        _adjacencyMatrix = new ulong[order * _rowLength];
    }

    public void AddEdge(int u, int v)
    {
        var index1 = u * _rowLength + v / 64;
        var mask1 = 1UL << (v % 64);
        _adjacencyMatrix[index1] |= mask1;

        var index2 = v * _rowLength + u / 64;
        var mask2 = 1UL << (u % 64);
        _adjacencyMatrix[index2] |= mask2;
    }

    public bool HasEdge(int u, int v)
    {
        var index = u * _rowLength + v / 64;
        var mask = 1UL << (v % 64);
        return (_adjacencyMatrix[index] & mask) != 0;
    }

    public int GetDegree(int u)
    {
        int degree = 0;
        for (int i = 0; i < _rowLength; i++)
        {
            degree += BitOperations.PopCount(_adjacencyMatrix[u * _rowLength + i]);
        }
        return degree;
    }

    public ulong[] GetNeighborsBitmask(int u)
    {
        var bitmask = new ulong[_rowLength];
        var startIndex = u * _rowLength;
        for (int i = 0; i < _rowLength; i++)
        {
            bitmask[i] = _adjacencyMatrix[startIndex + i];
        }
        return bitmask;
    }
}

class BronKerbosch
{
    private readonly Graph _graph;
    private List<int> _maxClique = new List<int>();
    private readonly object _lock = new object();
    private readonly int _rowLength;

    public BronKerbosch(Graph graph)
    {
        _graph = graph;
        _rowLength = (graph.Order + 63) / 64;
    }

    public List<int> FindMaxClique()
    {
        // Greedy heuristic for a good lower bound
        var greedyClique = GreedyClique();
        lock (_lock)
        {
            if (greedyClique.Count > _maxClique.Count)
            {
                _maxClique = greedyClique;
            }
        }

        var candidates = new ulong[_rowLength];
        var excluded = new ulong[_rowLength];
        for (int i = 0; i < _graph.Order; i++)
        {
            var index = i / 64;
            var mask = 1UL << (i % 64);
            candidates[index] |= mask;
        }

        FindCliques(new List<int>(), candidates, excluded);

        return _maxClique;
    }

    private List<int> GreedyClique()
    {
        var bestClique = new List<int>();
        for (int i = 0; i < _graph.Order; i++)
        {
            var clique = new List<int> { i };
            var neighbors = _graph.GetNeighborsBitmask(i);
            for (int j = i + 1; j < _graph.Order; j++)
            {
                var index = j / 64;
                var mask = 1UL << (j % 64);
                if ((neighbors[index] & mask) != 0)
                {
                    var isNeighborToAll = true;
                    foreach (var member in clique)
                    {
                        if (!_graph.HasEdge(j, member))
                        {
                            isNeighborToAll = false;
                            break;
                        }
                    }
                    if (isNeighborToAll)
                    {
                        clique.Add(j);
                    }
                }
            }
            if (clique.Count > bestClique.Count)
            {
                bestClique = clique;
            }
        }
        return bestClique;
    }

    private void FindCliques(List<int> clique, ulong[] candidates, ulong[] excluded)
    {
        if (IsAllZeros(candidates) && IsAllZeros(excluded))
        {
            lock (_lock)
            {
                if (clique.Count > _maxClique.Count)
                {
                    _maxClique = new List<int>(clique);
                }
            }
            return;
        }
        
        lock (_lock)
        {
            if (clique.Count + PopCount(candidates) <= _maxClique.Count)
            {
                return;
            }
        }


        var pivot = FindPivot(candidates, excluded);
        var pivotNeighbors = _graph.GetNeighborsBitmask(pivot);
        var candidatesWithoutPivotNeighbors = AndNot(candidates, pivotNeighbors);

        var tempCandidates = new ulong[_rowLength];
        Array.Copy(candidatesWithoutPivotNeighbors, tempCandidates, _rowLength);

        var verticesToProcess = new List<int>();
        for (int i = 0; i < _graph.Order; i++)
        {
            var index = i / 64;
            var mask = 1UL << (i % 64);
            if ((tempCandidates[index] & mask) != 0)
            {
                verticesToProcess.Add(i);
            }
        }
        
        foreach(var vertex in verticesToProcess)
        {
            var newClique = new List<int>(clique) { vertex };
            var neighbors = _graph.GetNeighborsBitmask(vertex);
            var newCandidates = And(candidates, neighbors);
            var newExcluded = And(excluded, neighbors);

            FindCliques(newClique, newCandidates, newExcluded);

            var index = vertex / 64;
            var mask = 1UL << (vertex % 64);
            candidates[index] &= ~mask;
            excluded[index] |= mask;
        }
    }

    private bool IsAllZeros(ulong[] array)
    {
        for (int i = 0; i < _rowLength; i++)
        {
            if (array[i] != 0) return false;
        }
        return true;
    }

    private int PopCount(ulong[] array)
    {
        int count = 0;
        for (int i = 0; i < _rowLength; i++)
        {
            count += BitOperations.PopCount(array[i]);
        }
        return count;
    }

    private int FindPivot(ulong[] candidates, ulong[] excluded)
    {
        var combined = Or(candidates, excluded);
        int maxDegree = -1;
        int pivot = -1;
        for (int i = 0; i < _graph.Order; i++)
        {
            var index = i / 64;
            var mask = 1UL << (i % 64);
            if ((combined[index] & mask) != 0)
            {
                var neighbors = _graph.GetNeighborsBitmask(i);
                var intersection = And(candidates, neighbors);
                var degree = PopCount(intersection);

                if (degree > maxDegree)
                {
                    maxDegree = degree;
                    pivot = i;
                }
            }
        }
        return pivot;
    }
    
    private ulong[] Or(ulong[] a, ulong[] b)
    {
        var result = new ulong[_rowLength];
        var i = 0;
        if (Avx2.IsSupported)
        {
            unsafe
            {
                fixed (ulong* pA = a, pB = b, pResult = result)
                {
                    for (; i <= _rowLength - 4; i += 4)
                    {
                        var vecA = Avx2.LoadVector256(pA + i);
                        var vecB = Avx2.LoadVector256(pB + i);
                        var vecResult = Avx2.Or(vecA, vecB);
                        Avx2.Store(pResult + i, vecResult);
                    }
                }
            }
        }
        for (; i < _rowLength; i++)
        {
            result[i] = a[i] | b[i];
        }
        return result;
    }

    private ulong[] And(ulong[] a, ulong[] b)
    {
        var result = new ulong[_rowLength];
        var i = 0;
        if (Avx2.IsSupported)
        {
            unsafe
            {
                fixed (ulong* pA = a, pB = b, pResult = result)
                {
                    for (; i <= _rowLength - 4; i += 4)
                    {
                        var vecA = Avx2.LoadVector256(pA + i);
                        var vecB = Avx2.LoadVector256(pB + i);
                        var vecResult = Avx2.And(vecA, vecB);
                        Avx2.Store(pResult + i, vecResult);
                    }
                }
            }
        }
        for (; i < _rowLength; i++)
        {
            result[i] = a[i] & b[i];
        }
        return result;
    }

    private ulong[] AndNot(ulong[] a, ulong[] b)
    {
        var result = new ulong[_rowLength];
        var i = 0;
        if (Avx2.IsSupported)
        {
            unsafe
            {
                fixed (ulong* pA = a, pB = b, pResult = result)
                {
                    for (; i <= _rowLength - 4; i += 4)
                    {
                        var vecA = Avx2.LoadVector256(pA + i);
                        var vecB = Avx2.LoadVector256(pB + i);
                        var vecResult = Avx2.AndNot(vecB, vecA);
                        Avx2.Store(pResult + i, vecResult);
                    }
                }
            }
        }
        for (; i < _rowLength; i++)
        {
            result[i] = a[i] & ~b[i];
        }
        return result;
    }
}

class Program
{
    static void Main(string[] args)
    {
        var graph = new Graph(5);
        graph.AddEdge(0, 1);
        graph.AddEdge(0, 2);
        graph.AddEdge(0, 3);
        graph.AddEdge(1, 2);
        graph.AddEdge(1, 3);
        graph.AddEdge(2, 3);
        graph.AddEdge(3, 4);

        var bronKerbosch = new BronKerbosch(graph);
        var maxClique = bronKerbosch.FindMaxClique();

        Console.WriteLine("Maximum clique size: " + maxClique.Count);
        Console.WriteLine("Maximum clique: " + string.Join(", ", maxClique));
    }
}
