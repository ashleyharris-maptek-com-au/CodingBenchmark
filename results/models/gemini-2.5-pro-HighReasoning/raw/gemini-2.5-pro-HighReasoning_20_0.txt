use std::io::{self, BufRead};
use std::collections::HashMap;

#[derive(Debug, Clone, Copy, PartialEq)]
struct Point {
    x: f64,
    y: f64,
    z: f64,
}

#[derive(Debug, Clone)]
struct Sample {
    depth: f64,
    properties: Vec<f64>,
    point: Point,
    hole_id: String,
    sample_idx_in_hole: usize,
}

#[derive(Debug)]
struct Hole {
    id: String,
    start_x: f64,
    start_y: f64,
    start_z: f64,
    dir_x: f64,
    dir_y: f64,
    dir_z: f64,
    length: f64,
    num_samples: usize,
    samples: Vec<Sample>,
}

#[derive(Debug)]
struct Outlier {
    hole_id: String,
    sample_idx: usize,
    property_name: String,
    confidence: f64,
}

fn distance(p1: &Point, p2: &Point) -> f64 {
    ((p1.x - p2.x).powi(2) + (p1.y - p2.y).powi(2) + (p1.z - p2.z).powi(2)).sqrt()
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = lines.next().unwrap().unwrap();
    let mut parts = first_line.split_whitespace();
    let num_holes = parts.next().unwrap().parse::<usize>().unwrap();
    let num_properties = parts.next().unwrap().parse::<usize>().unwrap();

    let property_names_line = lines.next().unwrap().unwrap();
    let property_names: Vec<String> = property_names_line
        .split_whitespace()
        .map(|s| s.to_string())
        .collect();

    let mut holes: Vec<Hole> = Vec::with_capacity(num_holes);
    let mut all_samples: Vec<Sample> = Vec::new();

    for hole_idx in 0..num_holes {
        let hole_line = lines.next().unwrap().unwrap();
        let mut hole_parts = hole_line.split_whitespace();
        let id = hole_parts.next().unwrap().to_string();
        let start_x = hole_parts.next().unwrap().parse::<f64>().unwrap();
        let start_y = hole_parts.next().unwrap().parse::<f64>().unwrap();
        let start_z = hole_parts.next().unwrap().parse::<f64>().unwrap();
        let dir_x = hole_parts.next().unwrap().parse::<f64>().unwrap();
        let dir_y = hole_parts.next().unwrap().parse::<f64>().unwrap();
        let dir_z = hole_parts.next().unwrap().parse::<f64>().unwrap();
        let length = hole_parts.next().unwrap().parse::<f64>().unwrap();
        let num_samples = hole_parts.next().unwrap().parse::<usize>().unwrap();

        let mut samples: Vec<Sample> = Vec::with_capacity(num_samples);
        for sample_idx_in_hole in 0..num_samples {
            let sample_line = lines.next().unwrap().unwrap();
            let mut sample_parts = sample_line.split_whitespace();
            let depth = sample_parts.next().unwrap().parse::<f64>().unwrap();
            let properties: Vec<f64> = (0..num_properties)
                .map(|_| sample_parts.next().unwrap().parse::<f64>().unwrap())
                .collect();

            let point = Point {
                x: start_x + dir_x * depth,
                y: start_y + dir_y * depth,
                z: start_z + dir_z * depth,
            };

            samples.push(Sample {
                depth,
                properties,
                point,
                hole_id: id.clone(),
                sample_idx_in_hole,
            });
        }
        all_samples.extend_from_slice(&samples);

        holes.push(Hole {
            id,
            start_x,
            start_y,
            start_z,
            dir_x,
            dir_y,
            dir_z,
            length,
            num_samples,
            samples,
        });
    }
    
    let dataset_size = all_samples.len();
    let mut outliers: Vec<Outlier> = Vec::new();

    if dataset_size < 200_000 {
        let radius = 100.0;
        for (sample_idx, sample) in all_samples.iter().enumerate() {
            let mut neighbors = Vec::new();
            for (i, other_sample) in all_samples.iter().enumerate() {
                if i == sample_idx {
                    continue;
                }
                if distance(&sample.point, &other_sample.point) <= radius {
                    neighbors.push(other_sample);
                }
            }

            if neighbors.is_empty() {
                continue;
            }

            for prop_idx in 0..num_properties {
                let mut neighbor_prop_sum = 0.0;
                for neighbor in &neighbors {
                    neighbor_prop_sum += neighbor.properties[prop_idx];
                }
                let neighbor_prop_avg = neighbor_prop_sum / neighbors.len() as f64;

                let diff = (sample.properties[prop_idx] - neighbor_prop_avg).abs();
                let confidence = if neighbor_prop_avg != 0.0 {
                    diff / neighbor_prop_avg
                } else {
                    0.0
                };

                if confidence > 0.5 {
                    outliers.push(Outlier {
                        hole_id: sample.hole_id.clone(),
                        sample_idx: sample.sample_idx_in_hole,
                        property_name: property_names[prop_idx].clone(),
                        confidence,
                    });
                }
            }
        }
    } else {
        let cell_size = 100.0;
        let mut grid: HashMap<(isize, isize, isize), Vec<usize>> = HashMap::new();

        for (i, sample) in all_samples.iter().enumerate() {
            let cell_x = (sample.point.x / cell_size).floor() as isize;
            let cell_y = (sample.point.y / cell_size).floor() as isize;
            let cell_z = (sample.point.z / cell_size).floor() as isize;
            grid.entry((cell_x, cell_y, cell_z)).or_default().push(i);
        }
        
        for (sample_idx, sample) in all_samples.iter().enumerate() {
            let cell_x = (sample.point.x / cell_size).floor() as isize;
            let cell_y = (sample.point.y / cell_size).floor() as isize;
            let cell_z = (sample.point.z / cell_size).floor() as isize;

            let mut neighbors = Vec::new();
            for dx in -1..=1 {
                for dy in -1..=1 {
                    for dz in -1..=1 {
                        if let Some(cell_samples) = grid.get(&(cell_x + dx, cell_y + dy, cell_z + dz)) {
                            for &other_sample_idx in cell_samples {
                                if sample_idx == other_sample_idx {
                                    continue;
                                }
                                let other_sample = &all_samples[other_sample_idx];
                                if distance(&sample.point, &other_sample.point) <= cell_size {
                                    neighbors.push(other_sample);
                                }
                            }
                        }
                    }
                }
            }

            if neighbors.is_empty() {
                continue;
            }

            for prop_idx in 0..num_properties {
                let mut neighbor_prop_sum = 0.0;
                for neighbor in &neighbors {
                    neighbor_prop_sum += neighbor.properties[prop_idx];
                }
                let neighbor_prop_avg = neighbor_prop_sum / neighbors.len() as f64;

                let diff = (sample.properties[prop_idx] - neighbor_prop_avg).abs();
                let confidence = if neighbor_prop_avg != 0.0 {
                    diff / neighbor_prop_avg
                } else {
                    0.0
                };

                if confidence > 0.5 {
                    outliers.push(Outlier {
                        hole_id: sample.hole_id.clone(),
                        sample_idx: sample.sample_idx_in_hole,
                        property_name: property_names[prop_idx].clone(),
                        confidence,
                    });
                }
            }
        }
    }

    outliers.sort_by(|a, b| b.confidence.partial_cmp(&a.confidence).unwrap());

    let mut final_outliers = Vec::new();
    for outlier in outliers {
        let mut found = false;
        for final_outlier in &mut final_outliers {
            if final_outlier.hole_id == outlier.hole_id && final_outlier.sample_idx == outlier.sample_idx && final_outlier.property_name == outlier.property_name {
                found = true;
                if outlier.confidence > final_outlier.confidence {
                    final_outlier.confidence = outlier.confidence;
                }
                break;
            }
        }
        if !found {
            final_outliers.push(outlier);
        }
    }

    println!("{}", final_outliers.len());
    for outlier in final_outliers {
        println!(
            "{} {} {} {}",
            outlier.hole_id, outlier.sample_idx, outlier.property_name, outlier.confidence
        );
    }
}
