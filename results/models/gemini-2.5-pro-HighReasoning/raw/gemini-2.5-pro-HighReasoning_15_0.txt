use std::io::{self, BufRead};
use std::ops::{Add, Sub, Mul, Div};

//==================================================================================================
//
// 3D Vector and Point
//
//==================================================================================================

#[derive(Clone, Copy, Debug, PartialEq)]
struct Vec3 {
    x: f64,
    y: f64,
    z: f64,
}

impl Vec3 {
    fn new(x: f64, y: f64, z: f64) -> Self {
        Self { x, y, z }
    }

    fn dot(&self, other: Self) -> f64 {
        self.x * other.x + self.y * other.y + self.z * other.z
    }
}

impl Add for Vec3 {
    type Output = Self;
    fn add(self, other: Self) -> Self {
        Self {
            x: self.x + other.x,
            y: self.y + other.y,
            z: self.z + other.z,
        }
    }
}

impl Sub for Vec3 {
    type Output = Self;
    fn sub(self, other: Self) -> Self {
        Self {
            x: self.x - other.x,
            y: self.y - other.y,
            z: self.z - other.z,
        }
    }
}

impl Mul<f64> for Vec3 {
    type Output = Self;
    fn mul(self, rhs: f64) -> Self {
        Self {
            x: self.x * rhs,
            y: self.y * rhs,
            z: self.z * rhs,
        }
    }
}

#[derive(Clone, Copy, Debug, PartialEq)]
struct Point2D {
    x: f64,
    y: f64,
}

impl Point2D {
    fn new(x: f64, y: f64) -> Self {
        Self { x, y }
    }
}

//==================================================================================================
//
// Quaternion
//
//==================================================================================================

#[derive(Clone, Copy, Debug)]
struct Quaternion {
    w: f64,
    x: f64,
    y: f64,
    z: f64,
}

impl Quaternion {
    fn new(w: f64, x: f64, y: f64, z: f64) -> Self {
        Self { w, x, y, z }
    }

    fn normalize(&self) -> Self {
        let mag = (self.w * self.w + self.x * self.x + self.y * self.y + self.z * self.z).sqrt();
        if mag == 0.0 {
            return Self { w: 1.0, x: 0.0, y: 0.0, z: 0.0 };
        }
        Self {
            w: self.w / mag,
            x: self.x / mag,
            y: self.y / mag,
            z: self.z / mag,
        }
    }
}

//==================================================================================================
//
// 3x3 Matrix
//
//==================================================================================================

#[derive(Clone, Copy, Debug)]
struct Mat3 {
    m: [[f64; 3]; 3],
}

impl From<Quaternion> for Mat3 {
    fn from(q: Quaternion) -> Self {
        let q = q.normalize();
        let w = q.w;
        let x = q.x;
        let y = q.y;
        let z = q.z;

        let m11 = 1.0 - 2.0 * y * y - 2.0 * z * z;
        let m12 = 2.0 * x * y - 2.0 * w * z;
        let m13 = 2.0 * x * z + 2.0 * w * y;

        let m21 = 2.0 * x * y + 2.0 * w * z;
        let m22 = 1.0 - 2.0 * x * x - 2.0 * z * z;
        let m23 = 2.0 * y * z - 2.0 * w * x;

        let m31 = 2.0 * x * z - 2.0 * w * y;
        let m32 = 2.0 * y * z + 2.0 * w * x;
        let m33 = 1.0 - 2.0 * x * x - 2.0 * y * y;

        Self {
            m: [
                [m11, m12, m13],
                [m21, m22, m23],
                [m31, m32, m33],
            ],
        }
    }
}

impl Mul<Vec3> for Mat3 {
    type Output = Vec3;
    fn mul(self, v: Vec3) -> Vec3 {
        Vec3 {
            x: self.m[0][0] * v.x + self.m[0][1] * v.y + self.m[0][2] * v.z,
            y: self.m[1][0] * v.x + self.m[1][1] * v.y + self.m[1][2] * v.z,
            z: self.m[2][0] * v.x + self.m[2][1] * v.y + self.m[2][2] * v.z,
        }
    }
}

//==================================================================================================
//
// Tetrahedron Shadow
//
//==================================================================================================

struct TetrahedronShadow {
    vertices: Vec<Point2D>,
}

impl TetrahedronShadow {
    fn new(rotation: Quaternion) -> Self {
        let base_vertices = [
            Vec3::new(1.0, 1.0, 1.0),
            Vec3::new(1.0, -1.0, -1.0),
            Vec3::new(-1.0, 1.0, -1.0),
            Vec3::new(-1.0, -1.0, 1.0),
        ];

        let rotation_matrix = Mat3::from(rotation);
        let rotated_vertices: Vec<Vec3> = base_vertices
            .iter()
            .map(|&v| rotation_matrix * v)
            .collect();

        let projected_points: Vec<Point2D> = rotated_vertices
            .iter()
            .map(|&v| Point2D::new(v.x, v.y))
            .collect();
        
        Self {
            vertices: Self::convex_hull(projected_points),
        }
    }

    fn convex_hull(mut points: Vec<Point2D>) -> Vec<Point2D> {
        if points.len() <= 3 {
            return points;
        }

        // Find the point with the lowest y-coordinate
        let mut start_idx = 0;
        for i in 1..points.len() {
            if points[i].y < points[start_idx].y || (points[i].y == points[start_idx].y && points[i].x < points[start_idx].x) {
                start_idx = i;
            }
        }
        points.swap(0, start_idx);

        let start_point = points[0];
        points[1..].sort_unstable_by(|a, b| {
            let cross_product = (a.x - start_point.x) * (b.y - start_point.y) - (a.y - start_point.y) * (b.x - start_point.x);
            if cross_product == 0.0 {
                let dist_a = (a.x - start_point.x).powi(2) + (a.y - start_point.y).powi(2);
                let dist_b = (b.x - start_point.x).powi(2) + (b.y - start_point.y).powi(2);
                dist_a.partial_cmp(&dist_b).unwrap()
            } else if cross_product > 0.0 {
                std::cmp::Ordering::Less
            } else {
                std::cmp::Ordering::Greater
            }
        });

        let mut hull = vec![points[0], points[1]];
        for i in 2..points.len() {
            while hull.len() >= 2 {
                let p1 = hull[hull.len() - 2];
                let p2 = hull[hull.len() - 1];
                let p3 = points[i];
                let cross_product = (p2.x - p1.x) * (p3.y - p1.y) - (p2.y - p1.y) * (p3.x - p1.x);
                if cross_product >= 0.0 {
                    hull.pop();
                } else {
                    break;
                }
            }
            hull.push(points[i]);
        }
        hull
    }

    fn contains(&self, point: Point2D) -> bool {
        let n = self.vertices.len();
        if n < 3 {
            return false;
        }
        let mut sign = 0.0;
        for i in 0..n {
            let p1 = self.vertices[i];
            let p2 = self.vertices[(i + 1) % n];
            let cross_product = (point.x - p1.x) * (p2.y - p1.y) - (point.y - p1.y) * (p2.x - p1.x);
            if i == 0 {
                sign = cross_product.signum();
            } else if cross_product.signum() != 0.0 && cross_product.signum() != sign {
                return false;
            }
        }
        true
    }
}


//==================================================================================================
//
// Main Logic
//
//==================================================================================================

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let n: usize = lines.next().unwrap().unwrap().trim().parse().unwrap();

    let mut shadows = Vec::with_capacity(n);
    for _ in 0..n {
        let line = lines.next().unwrap().unwrap();
        let parts: Vec<f64> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
        let q = Quaternion::new(parts[0], parts[1], parts[2], parts[3]);
        shadows.push(TetrahedronShadow::new(q));
    }

    let min_coord = -2.0;
    let max_coord = 2.0;
    let resolution = 2000;
    let step = (max_coord - min_coord) / resolution as f64;
    let mut inside_count = 0;

    for i in 0..resolution {
        let y = min_coord + (i as f64 + 0.5) * step;
        for j in 0..resolution {
            let x = min_coord + (j as f64 + 0.5) * step;
            let point = Point2D::new(x, y);
            for shadow in &shadows {
                if shadow.contains(point) {
                    inside_count += 1;
                    break;
                }
            }
        }
    }

    let total_points = resolution * resolution;
    let bounding_box_area = (max_coord - min_coord).powi(2);
    let area = (inside_count as f64 / total_points as f64) * bounding_box_area;

    println!("{}", area);
    std::fs::write("answer.txt", area.to_string()).expect("Unable to write answer to file");
}
