using System;
using System.Collections.Generic;
using System.Linq;

namespace CuttingStockProblem
{
    public class Item
    {
        public int Length { get; set; }
        public int Demand { get; set; }
    }

    public class Pattern
    {
        public List<int> Cuts { get; set; } = new List<int>();
        public int Waste { get; set; }

        public override string ToString()
        {
            return string.Join(", ", Cuts) + $", Waste: {Waste}";
        }
    }

    public class CuttingStock
    {
        private readonly int _stockLength;
        private readonly List<Item> _items;
        private List<Pattern> _allPatterns;

        public CuttingStock(int stockLength, List<Item> items)
        {
            _stockLength = stockLength;
            _items = items.OrderByDescending(i => i.Length).ToList();
            _allPatterns = new List<Pattern>();
        }

        public void Solve()
        {
            Console.WriteLine("Solving Cutting Stock Problem...");
            Console.WriteLine($"Stock Length: {_stockLength}");
            Console.WriteLine("Items:");
            var remainingDemand = new Dictionary<int, int>();
            foreach (var item in _items)
            {
                Console.WriteLine($"- Length: {item.Length}, Demand: {item.Demand}");
                remainingDemand[item.Length] = item.Demand;
            }
            Console.WriteLine();

            GenerateAllPatterns();

            var usedStockPieces = new List<Pattern>();

            while (remainingDemand.Values.Sum() > 0)
            {
                Pattern bestPattern = null;
                double bestPatternScore = -1;

                foreach (var pattern in _allPatterns)
                {
                    double currentScore = 0;
                    bool usefulPattern = false;
                    foreach (var cut in pattern.Cuts)
                    {
                        if (remainingDemand.ContainsKey(cut) && remainingDemand[cut] > 0)
                        {
                            currentScore += cut; // Add the length of the useful cut
                            usefulPattern = true;
                        }
                    }

                    if (usefulPattern)
                    {
                        // Penalize waste, the higher the waste the lower the score
                        currentScore -= pattern.Waste;
                    }


                    if (currentScore > bestPatternScore)
                    {
                        bestPatternScore = currentScore;
                        bestPattern = pattern;
                    }
                }

                if (bestPattern == null)
                {
                    Console.WriteLine("Could not find a pattern to satisfy remaining demand. This can happen if an item's length is greater than the stock length.");
                    break;
                }

                usedStockPieces.Add(bestPattern);
                foreach (var cut in bestPattern.Cuts)
                {
                    if (remainingDemand.ContainsKey(cut) && remainingDemand[cut] > 0)
                    {
                        remainingDemand[cut]--;
                    }
                }
            }

            Console.WriteLine("\nSolution:");
            Console.WriteLine($"Total stock pieces used: {usedStockPieces.Count}");

            var patternGroups = usedStockPieces
                .Select(p => string.Join(", ", p.Cuts.OrderBy(c => c)))
                .GroupBy(s => s)
                .Select(g => new { PatternString = g.Key, Count = g.Count() });

            if (patternGroups.Any())
            {
                Console.WriteLine("Patterns used:");
                foreach (var group in patternGroups)
                {
                    var cuts = group.PatternString.Split(new[] { ", " }, StringSplitOptions.None).Select(int.Parse);
                    var waste = _stockLength - cuts.Sum();
                    Console.WriteLine($"- {group.Count} x (Cuts: {group.PatternString}, Waste: {waste})");
                }
            }

            var totalWaste = usedStockPieces.Sum(p => p.Waste);
            Console.WriteLine($"Total waste: {totalWaste}");
        }

        private void GenerateAllPatterns()
        {
            var currentCuts = new List<int>();
            GeneratePatternsRecursive(_stockLength, 0, currentCuts);
            // Remove duplicates and patterns with no cuts
            _allPatterns = _allPatterns
                .Where(p => p.Cuts.Any())
                .GroupBy(p => string.Join(",", p.Cuts.OrderBy(c => c)))
                .Select(g => g.First())
                .ToList();
        }

        private void GeneratePatternsRecursive(int remainingLength, int startItemIndex, List<int> currentCuts)
        {
            _allPatterns.Add(new Pattern { Cuts = new List<int>(currentCuts), Waste = remainingLength });

            for (int i = startItemIndex; i < _items.Count; i++)
            {
                var item = _items[i];
                if (item.Length <= remainingLength)
                {
                    currentCuts.Add(item.Length);
                    GeneratePatternsRecursive(remainingLength - item.Length, i, currentCuts);
                    currentCuts.RemoveAt(currentCuts.Count - 1); // backtrack
                }
            }
        }
    }

    public class Program
    {
        public static void Main(string[] args)
        {
            var stockLength = 100;
            var items = new List<Item>
            {
                new Item { Length = 30, Demand = 10 },
                new Item { Length = 40, Demand = 12 },
                new Item { Length = 55, Demand = 8 },
            };

            var cuttingStock = new CuttingStock(stockLength, items);
            cuttingStock.Solve();
        }
    }
}
