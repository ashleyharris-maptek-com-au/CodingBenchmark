use std::io::{self, BufRead};
use std::collections::HashSet;

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = lines.next().unwrap().unwrap();
    let mut parts = first_line.split_whitespace();
    let num_vertices: usize = parts.next().unwrap().parse().unwrap();
    let _num_edges: usize = parts.next().unwrap().parse().unwrap();

    let mut adj = vec![Vec::new(); num_vertices];
    let mut edges = Vec::new();
    for line in lines {
        let line = line.unwrap();
        let mut parts = line.split_whitespace();
        let u: usize = parts.next().unwrap().parse().unwrap();
        let v: usize = parts.next().unwrap().parse().unwrap();
        adj[u].push(edges.len());
        adj[v].push(edges.len());
        edges.push((u, v));
    }
    
    let num_edges = edges.len();
    let mut cover = HashSet::new();
    let mut edge_covered = vec![false; num_edges];
    
    for i in 0..num_edges {
        if !edge_covered[i] {
            let (u, v) = edges[i];
            cover.insert(u);
            cover.insert(v);
            
            for &edge_idx in &adj[u] {
                edge_covered[edge_idx] = true;
            }
            for &edge_idx in &adj[v] {
                edge_covered[edge_idx] = true;
            }
        }
    }

    println!("{}", cover.len());
    let mut cover_vec: Vec<_> = cover.into_iter().collect();
    cover_vec.sort();
    let mut output = String::new();
    for (i, v) in cover_vec.iter().enumerate() {
        if i > 0 {
            output.push(' ');
        }
        output.push_str(&v.to_string());
    }
    println!("{}", output);
}
