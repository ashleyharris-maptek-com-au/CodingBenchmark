using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

public class Program
{
    public static void Main(string[] args)
    {
        var solver = new VoxelMiner();
        solver.Run();
    }
}

public class VoxelMiner
{
    private int countX, countY, countZ;
    private Dictionary<(int, int), SortedList<int, byte>> worldColumns;
    private int[,] surfaceZ;
    private List<string> operations = new List<string>();

    public void Run()
    {
        ReadInput();
        Solve();
        WriteOutput();
    }

    private void ReadInput()
    {
        var header = Console.ReadLine().Split(' ');
        countX = int.Parse(header[0]);
        countY = int.Parse(header[1]);
        countZ = int.Parse(header[2]);

        worldColumns = new Dictionary<(int, int), SortedList<int, byte>>();
        surfaceZ = new int[countX, countY];

        using (var stream = Console.OpenStandardInput())
        {
            for (int x = 0; x < countX; x++)
            {
                for (int y = 0; y < countY; y++)
                {
                    var column = new SortedList<int, byte>();
                    for (int z = 0; z < countZ; z++)
                    {
                        int d = stream.ReadByte();
                        if (d == -1) break;
                        if (d > 0)
                        {
                            column.Add(z, (byte)d);
                        }
                    }
                    if (column.Count > 0)
                    {
                        worldColumns[(x, y)] = column;
                        surfaceZ[x, y] = column.Keys.Last();
                    }
                    else
                    {
                        surfaceZ[x, y] = -1;
                    }
                }
                if (stream.Position >= stream.Length && (x < countX -1 || y < countY - 1) ) break;
            }
        }
    }

    private void Solve()
    {
        var pathfinder = new Pathfinder(countX, countY, surfaceZ);

        while (true)
        {
            (int x, int y, int z) bestDigSite = (-1, -1, -1);
            (int x, int y) bestDumpSite = (-1, -1);
            decimal bestProfit = 0;
            List<(int, int)> bestPath = null;

            for (int x = 0; x < countX; x++)
            {
                for (int y = 0; y < countY; y++)
                {
                    if (!worldColumns.ContainsKey((x, y)) || worldColumns[(x, y)].Count == 0) continue;

                    int z = surfaceZ[x, y];
                    byte voxelData = worldColumns[(x, y)][z];
                    int mass = GetMass(voxelData);
                    int value = GetValue(voxelData);

                    if (mass == 0) continue;

                    if (value > 0) // Profitable ore
                    {
                        var (path, cost) = pathfinder.FindPath(x, y, 0, 0, mass);
                        if (path != null)
                        {
                            decimal revenue = mass * value * 1000;
                            decimal profit = revenue - cost;

                            if (profit > bestProfit)
                            {
                                bestProfit = profit;
                                bestDigSite = (x, y, z);
                                bestDumpSite = (0, 0);
                                bestPath = path;
                            }
                        }
                    }
                    else // Waste rock - only consider moving it if it's a net gain (which is never, so this logic is for future expansion)
                    {
                        // A simple greedy algorithm won't move waste unless there are no profitable moves.
                        // For now, we can imagine a scenario where removing waste is a negative cost action
                        // that enables a future profitable move. We can simulate this with a small negative profit.
                        (int dumpX, int dumpY) = FindBetterDumpSite(x, y);
                        var (path, cost) = pathfinder.FindPath(x, y, dumpX, dumpY, mass);
                        if (path != null)
                        {
                            // Assign a small "profit" to clearing waste, to be chosen only when no better options exist.
                            // This cost should be slightly better than 0, but worse than any real profit.
                            decimal conceptualProfit = -cost - 1; // Always negative, but less negative is better.
                            
                            if (bestProfit <= 0 && conceptualProfit > bestProfit)
                            {
                                bestProfit = conceptualProfit;
                                bestDigSite = (x, y, z);
                                bestDumpSite = (dumpX, dumpY);
                                bestPath = path;
                            }
                        }
                    }
                }
            }

            if (bestDigSite.x == -1 || bestPath == null)
            {
                break; // No more moves found
            }
             if (bestProfit <= 0 && !(GetMass(worldColumns[(bestDigSite.x,bestDigSite.y)][bestDigSite.z])>0 && GetValue(worldColumns[(bestDigSite.x,bestDigSite.y)][bestDigSite.z])==0) )
            {
                break; // No more profitable moves
            }


            // Execute the best move
            int digX = bestDigSite.x;
            int digY = bestDigSite.y;
            int digZ = bestDigSite.z;
            byte dugVoxelData = worldColumns[(digX, digY)][digZ];

            // 1. Remove voxel
            var columnToDig = worldColumns[(digX, digY)];
            columnToDig.Remove(digZ);
            if (columnToDig.Count == 0)
            {
                surfaceZ[digX, digY] = -1;
            }
            else
            {
                surfaceZ[digX, digY] = columnToDig.Keys.Last();
            }
            pathfinder.UpdateSurface(digX, digY, surfaceZ[digX, digY]);

            // 2. Add voxel to new location (if not plant)
            int dumpX = bestDumpSite.x;
            int dumpY = bestDumpSite.y;
            bool isOre = GetValue(dugVoxelData) > 0;

            if (!(dumpX == 0 && dumpY == 0 && isOre))
            {
                if (!worldColumns.ContainsKey((dumpX, dumpY)))
                {
                    worldColumns[(dumpX, dumpY)] = new SortedList<int, byte>();
                }
                var columnToDump = worldColumns[(dumpX, dumpY)];
                int newZ = surfaceZ[dumpX, dumpY] + 1;
                columnToDump.Add(newZ, dugVoxelData); 
                surfaceZ[dumpX, dumpY] = newZ;
                pathfinder.UpdateSurface(dumpX, dumpY, newZ);
            }
            
            // 3. Record operation
            int N = bestPath.Count -1;
            var steps = new List<string>();
            for(int i=0; i< N; i++)
            {
                steps.Add(bestPath[i+1].Item1.ToString());
                steps.Add(bestPath[i+1].Item2.ToString());
            }
            string pathStr = string.Join(" ", steps);

            operations.Add($"{N} {digX} {digY} {digZ} {pathStr} {dumpX} {dumpY}");
        }
    }

    private (int, int) FindBetterDumpSite(int currentX, int currentY)
    {
        int bestX = 0;
        int bestY = 0;
        int minZ = int.MaxValue;
        int radius = 10;

        for (int x = Math.Max(0, currentX - radius); x < Math.Min(countX, currentX + radius); x++)
        {
            for (int y = Math.Max(0, currentY - radius); y < Math.Min(countY, currentY + radius); y++)
            {
                if (x == 0 && y == 0) continue; // Don't dump at the plant
                int z = surfaceZ[x, y];
                if (z < minZ)
                {
                    minZ = z;
                    bestX = x;
                    bestY = y;
                }
            }
        }
        return (bestX, bestY);
    }
    
    private void WriteOutput()
    {
        Console.WriteLine(operations.Count);
        foreach (var op in operations)
        {
            Console.WriteLine(op);
        }
    }

    private int GetMass(byte d) => d % 16;
    private int GetValue(byte d) => d / 16;
}

public class Pathfinder
{
    private int countX, countY;
    private int[,] surfaceZ;

    public Pathfinder(int countX, int countY, int[,] surfaceZ)
    {
        this.countX = countX;
        this.countY = countY;
        this.surfaceZ = surfaceZ;
    }

    public void UpdateSurface(int x, int y, int z)
    {
        surfaceZ[x, y] = z;
    }

    private class PathNode : IComparable<PathNode>
    {
        public int X { get; }
        public int Y { get; }
        public decimal G_Cost { get; set; } = decimal.MaxValue;
        public decimal H_Cost { get; set; }
        public decimal F_Cost => G_Cost + H_Cost;
        public PathNode Parent { get; set; }

        public PathNode(int x, int y) { X = x; Y = y; }

        public int CompareTo(PathNode other)
        {
            int f_compare = F_Cost.CompareTo(other.F_Cost);
            if (f_compare == 0) return H_Cost.CompareTo(other.H_Cost);
            return f_compare;
        }
        public override bool Equals(object obj) => obj is PathNode other && X == other.X && Y == other.Y;
        public override int GetHashCode() => (X, Y).GetHashCode();
    }

    public (List<(int, int)>, decimal) FindPath(int startX, int startY, int endX, int endY, int mass)
    {
        var openSet = new PriorityQueue<PathNode>();
        var allNodes = new Dictionary<(int, int), PathNode>();

        var startNode = new PathNode(startX, startY) { G_Cost = 0, H_Cost = ManhattanDistance(startX, startY, endX, endY) };
        openSet.Enqueue(startNode);
        allNodes[(startX, startY)] = startNode;

        while (openSet.Count > 0)
        {
            var currentNode = openSet.Dequeue();

            if (currentNode.X == endX && currentNode.Y == endY)
            {
                return (ReconstructPath(currentNode), currentNode.G_Cost);
            }

            foreach (var neighborPos in GetNeighbors(currentNode.X, currentNode.Y))
            {
                int nX = neighborPos.Item1;
                int nY = neighborPos.Item2;

                decimal moveCost = CalculateMoveCost(currentNode.X, currentNode.Y, nX, nY, mass);
                if (moveCost == decimal.MaxValue) continue;

                decimal newGCost = currentNode.G_Cost + moveCost;

                if (!allNodes.TryGetValue((nX, nY), out var neighborNode))
                {
                    neighborNode = new PathNode(nX, nY)
                    {
                        H_Cost = ManhattanDistance(nX, nY, endX, endY)
                    };
                    allNodes[(nX, nY)] = neighborNode;
                }
                
                if(newGCost < neighborNode.G_Cost)
                {
                    neighborNode.Parent = currentNode;
                    neighborNode.G_Cost = newGCost;
                    if(!openSet.Contains(neighborNode))
                    {
                       openSet.Enqueue(neighborNode);
                    }
                }
            }
        }

        return (null, -1);
    }
    
    private decimal CalculateMoveCost(int fromX, int fromY, int toX, int toY, int mass)
    {
        int fromZ = surfaceZ[fromX, fromY];
        int toZ = surfaceZ[toX, toY];
        if (fromZ == -1) fromZ = 0;
        if (toZ == -1) toZ = 0;

        int deltaZ = toZ - fromZ;
        if (Math.Abs(deltaZ) > 1) return decimal.MaxValue;

        if (deltaZ > 0) return 5 * mass; // Uphill
        if (deltaZ < 0) return 0; // Downhill
        return 1 * mass; // Flat
    }

    private List<(int, int)> ReconstructPath(PathNode endNode)
    {
        var path = new List<(int, int)>();
        var current = endNode;
        while (current != null)
        {
            path.Add((current.X, current.Y));
            current = current.Parent;
        }
        path.Reverse();
        return path;
    }

    private int ManhattanDistance(int x1, int y1, int x2, int y2) => Math.Abs(x1 - x2) + Math.Abs(y1 - y2);

    private IEnumerable<(int, int)> GetNeighbors(int x, int y)
    {
        if (x > 0) yield return (x - 1, y);
        if (x < countX - 1) yield return (x + 1, y);
        if (y > 0) yield return (x, y - 1);
        if (y < countY - 1) yield return (x, y + 1);
    }
}

public class PriorityQueue<T> where T : IComparable<T>
{
    private List<T> data;
    private HashSet<T> set;

    public int Count => data.Count;

    public PriorityQueue()
    {
        this.data = new List<T>();
        this.set = new HashSet<T>();
    }

    public void Enqueue(T item)
    {
        if(set.Contains(item)) return;
        data.Add(item);
        set.Add(item);
        int ci = data.Count - 1;
        while (ci > 0)
        {
            int pi = (ci - 1) / 2;
            if (data[ci].CompareTo(data[pi]) >= 0) break;
            (data[ci], data[pi]) = (data[pi], data[ci]);
            ci = pi;
        }
    }

    public T Dequeue()
    {
        int li = data.Count - 1;
        T frontItem = data[0];
        data[0] = data[li];
        data.RemoveAt(li);
        set.Remove(frontItem);


        --li;
        int pi = 0;
        while (true)
        {
            int ci = pi * 2 + 1;
            if (ci > li) break;
            int sc = ci + 1;
            if (sc <= li && data[sc].CompareTo(data[ci]) < 0)
                ci = sc;
            if (data[pi].CompareTo(data[ci]) <= 0) break;
            (data[pi], data[ci]) = (data[ci], data[pi]);
            pi = ci;
        }
        return frontItem;
    }

    public bool Contains(T item)
    {
        return set.Contains(item);
    }
}
