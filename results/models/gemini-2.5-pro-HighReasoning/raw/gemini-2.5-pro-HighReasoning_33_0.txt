using System;
using System.Collections.Generic;
using System.Linq;

public class QapSolver
{
    private readonly int[][] _flow;
    private readonly int[][] _distance;
    private readonly int _size;
    private readonly Random _random;

    public QapSolver(int[][] flow, int[][] distance)
    {
        // Add validation for flow and distance matrices
        if (flow == null || distance == null || flow.Length != distance.Length)
        {
            throw new ArgumentException("Invalid flow or distance matrices.");
        }
        _flow = flow;
        _distance = distance;
        _size = flow.Length;
        _random = new Random();
    }

    public int[] Solve(double initialTemperature = 10000, double coolingRate = 0.99, int maxIterations = 100000)
    {
        var currentSolution = GenerateRandomSolution();
        var currentCost = CalculateCost(currentSolution);

        var bestSolution = (int[])currentSolution.Clone();
        var bestCost = currentCost;

        var temperature = initialTemperature;

        for (int i = 0; i < maxIterations; i++)
        {
            if (temperature <= 0)
            {
                break;
            }

            var newSolution = GenerateNeighbor(currentSolution);
            var newCost = CalculateCost(newSolution);

            var deltaCost = newCost - currentCost;

            if (deltaCost < 0 || _random.NextDouble() < Math.Exp(-deltaCost / temperature))
            {
                currentSolution = newSolution;
                currentCost = newCost;

                if (currentCost < bestCost)
                {
                    bestSolution = (int[])currentSolution.Clone();
                    bestCost = currentCost;
                }
            }

            temperature *= coolingRate;
        }

        return bestSolution;
    }

    private int CalculateCost(int[] solution)
    {
        int cost = 0;
        for (int i = 0; i < _size; i++)
        {
            for (int j = 0; j < _size; j++)
            {
                if (i == j) continue;
                cost += _flow[i][j] * _distance[solution[i]][solution[j]];
            }
        }
        return cost;
    }

    private int[] GenerateRandomSolution()
    {
        var solution = new int[_size];
        for (int i = 0; i < _size; i++)
        {
            solution[i] = i;
        }

        for (int i = 0; i < _size - 1; i++)
        {
            var j = _random.Next(i, _size);
            var temp = solution[i];
            solution[i] = solution[j];
            solution[j] = temp;
        }
        return solution;
    }

    private int[] GenerateNeighbor(int[] solution)
    {
        var newSolution = (int[])solution.Clone();
        var i = _random.Next(0, _size);
        var j = _random.Next(0, _size);

        var temp = newSolution[i];
        newSolution[i] = newSolution[j];
        newSolution[j] = temp;

        return newSolution;
    }

}
