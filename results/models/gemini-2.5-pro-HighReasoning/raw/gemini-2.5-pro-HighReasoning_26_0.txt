use ndarray::{Array, Array2, Axis};
use rand::Rng;
use linfa::traits::{Fit, Predict};
use linfa::DatasetBase;
use linfa_clustering::KMeans;

fn main() {
    // 0. Define the number of clusters
    let n_clusters = 3;

    // 1. Generate random 3D points
    let n_points = 100;
    let mut rng = rand::thread_rng();
    let points: Array2<f64> = Array::from_shape_fn((n_points, 3), |_| rng.gen_range(-10.0..10.0));

    // 2. Create a linfa dataset
    // No targets are needed for clustering, so we use a dummy target array
    let targets = Array::from_shape_vec(vec![(); n_points], (n_points,)).unwrap();
    let dataset = DatasetBase::new(points.clone(), targets);


    // 3. Configure and run the k-means algorithm
    let model = KMeans::params(n_clusters)
        .fit(&dataset)
        .expect("KMeans fitting failed");

    // 4. Get the cluster assignments for each point
    let cluster_assignments = model.predict(&dataset);

    // 5. Print the results
    println!("Cluster assignments:");
    for (i, &cluster) in cluster_assignments.iter().enumerate() {
        let point = points.index_axis(Axis(0), i);
        println!("Point {}: {:?} -> Cluster {}", i, point.to_vec(), cluster);
    }
}
