using System;
using System.Collections.Generic;
using System.Drawing;

namespace Tetris
{
    public enum TetrominoShape
    {
        I, O, T, S, Z, J, L
    }

    public class Tetromino
    {
        public TetrominoShape Shape { get; }
        public Color Color { get; }
        public int[,] Matrix { get; private set; }
        public int X { get; set; }
        public int Y { get; set; }

        public Tetromino(TetrominoShape shape)
        {
            Shape = shape;
            X = 0;
            Y = 0;
            (Matrix, Color) = GetTetrominoData(shape);
        }

        private static (int[,], Color) GetTetrominoData(TetrominoShape shape)
        {
            switch (shape)
            {
                case TetrominoShape.I:
                    return (new int[,] { { 1, 1, 1, 1 } }, Color.Cyan);
                case TetrominoShape.O:
                    return (new int[,] { { 1, 1 }, { 1, 1 } }, Color.Yellow);
                case TetrominoShape.T:
                    return (new int[,] { { 0, 1, 0 }, { 1, 1, 1 } }, Color.Purple);
                case TetrominoShape.S:
                    return (new int[,] { { 0, 1, 1 }, { 1, 1, 0 } }, Color.Green);
                case TetrominoShape.Z:
                    return (new int[,] { { 1, 1, 0 }, { 0, 1, 1 } }, Color.Red);
                case TetrominoShape.J:
                    return (new int[,] { { 1, 0, 0 }, { 1, 1, 1 } }, Color.Blue);
                case TetrominoShape.L:
                    return (new int[,] { { 0, 0, 1 }, { 1, 1, 1 } }, Color.Orange);
                default:
                    throw new ArgumentOutOfRangeException(nameof(shape), shape, null);
            }
        }

        public void Rotate()
        {
            int width = Matrix.GetLength(1);
            int height = Matrix.GetLength(0);
            int[,] newMatrix = new int[width, height];
            for (int i = 0; i < height; i++)
            {
                for (int j = 0; j < width; j++)
                {
                    newMatrix[j, height - 1 - i] = Matrix[i, j];
                }
            }
            Matrix = newMatrix;
        }
    }

    public class Board
    {
        public int Width { get; }
        public int Height { get; }
        public int[,] Grid { get; }

        public Board(int width, int height)
        {
            Width = width;
            Height = height;
            Grid = new int[width, height];
        }

        public bool IsCollision(Tetromino tetromino)
        {
            for (int i = 0; i < tetromino.Matrix.GetLength(0); i++)
            {
                for (int j = 0; j < tetromino.Matrix.GetLength(1); j++)
                {
                    if (tetromino.Matrix[i, j] == 1)
                    {
                        int x = tetromino.X + j;
                        int y = tetromino.Y + i;
                        if (x < 0 || x >= Width || y < 0 || y >= Height || Grid[x, y] == 1)
                        {
                            return true;
                        }
                    }
                }
            }
            return false;
        }

        public void PlacePiece(Tetromino tetromino)
        {
            for (int i = 0; i < tetromino.Matrix.GetLength(0); i++)
            {
                for (int j = 0; j < tetromino.Matrix.GetLength(1); j++)
                {
                    if (tetromino.Matrix[i, j] == 1)
                    {
                        Grid[tetromino.X + j, tetromino.Y + i] = 1;
                    }
                }
            }
        }

        public int ClearLines()
        {
            int linesCleared = 0;
            for (int y = Height - 1; y >= 0; y--)
            {
                bool isLineFull = true;
                for (int x = 0; x < Width; x++)
                {
                    if (Grid[x, y] == 0)
                    {
                        isLineFull = false;
                        break;
                    }
                }

                if (isLineFull)
                {
                    linesCleared++;
                    for (int y2 = y; y2 > 0; y2--)
                    {
                        for (int x = 0; x < Width; x++)
                        {
                            Grid[x, y2] = Grid[x, y2 - 1];
                        }
                    }
                    y++;
                }
            }
            return linesCleared;
        }
    }

    public class Game
    {
        public Board Board { get; }
        public Tetromino CurrentPiece { get; private set; }
        public Tetromino NextPiece { get; private set; }
        public int Score { get; private set; }
        public bool IsGameOver { get; private set; }

        private readonly Random _random = new Random();

        public Game(int width, int height)
        {
            Board = new Board(width, height);
            NextPiece = GetRandomTetromino();
            NewPiece();
        }

        private Tetromino GetRandomTetromino()
        {
            var values = Enum.GetValues(typeof(TetrominoShape));
            var randomShape = (TetrominoShape)values.GetValue(_random.Next(values.Length));
            return new Tetromino(randomShape);
        }

        private void NewPiece()
        {
            CurrentPiece = NextPiece;
            NextPiece = GetRandomTetromino();
            CurrentPiece.X = Board.Width / 2 - CurrentPiece.Matrix.GetLength(1) / 2;
            CurrentPiece.Y = 0;

            if (Board.IsCollision(CurrentPiece))
            {
                IsGameOver = true;
            }
        }

        public void Update()
        {
            if (IsGameOver)
            {
                return;
            }

            CurrentPiece.Y++;
            if (Board.IsCollision(CurrentPiece))
            {
                CurrentPiece.Y--;
                Board.PlacePiece(CurrentPiece);
                int linesCleared = Board.ClearLines();
                Score += linesCleared * linesCleared;
                NewPiece();
            }
        }

        public void MoveLeft()
        {
            CurrentPiece.X--;
            if (Board.IsCollision(CurrentPiece))
            {
                CurrentPiece.X++;
            }
        }

        public void MoveRight()
        {
            CurrentPiece.X++;
            if (Board.IsCollision(CurrentPiece))
            {
                CurrentPiece.X--;
            }
        }

        public void Rotate()
        {
            CurrentPiece.Rotate();
            if (Board.IsCollision(CurrentPiece))
            {
                // try to un-rotate
                for(int i = 0; i < 3; i++)
                {
                    CurrentPiece.Rotate();
                }
            }
        }
    }
}
