using System.Collections.Generic;
using BinPacking3D.Models;

namespace BinPacking3D
{
    public class GuillotineBinPack
    {
        private class Node
        {
            public int X, Y, Z;
            public int Width, Height, Depth;
            public bool IsOccupied;
        }

        private readonly int _binWidth;
        private readonly int _binHeight;
        private readonly int _binDepth;

        private readonly List<Node> _nodes = new List<Node>();

        public GuillotineBinPack(int width, int height, int depth)
        {
            _binWidth = width;
            _binHeight = height;
            _binDepth = depth;
            var root = new Node { Width = width, Height = height, Depth = depth };
            _nodes.Add(root);
        }

        public Box? Insert(int width, int height, int depth)
        {
            Node? bestNode = null;
            var bestScore = int.MaxValue;

            foreach (var node in _nodes)
            {
                if (node.IsOccupied) continue;

                if (node.Width >= width && node.Height >= height && node.Depth >= depth)
                {
                    var score = node.Width * node.Height * node.Depth - width * height * depth;
                    if (score < bestScore)
                    {
                        bestScore = score;
                        bestNode = node;
                    }
                }
            }

            if (bestNode == null) return null;

            SplitNode(bestNode, width, height, depth);

            return new Box(bestNode.Width, bestNode.Height, bestNode.Depth)
            {
                Position = new Vector3(bestNode.X, bestNode.Y, bestNode.Z)
            };
        }

        private void SplitNode(Node node, int width, int height, int depth)
        {
            node.IsOccupied = true;

            var right = new Node
            {
                X = node.X + width,
                Y = node.Y,
                Z = node.Z,
                Width = node.Width - width,
                Height = height,
                Depth = depth
            };

            var top = new Node
            {
                X = node.X,
                Y = node.Y + height,
                Z = node.Z,
                Width = node.Width,
                Height = node.Height - height,
                Depth = depth
            };
            
            var back = new Node
            {
                X = node.X,
                Y = node.Y,
                Z = node.Z + depth,
                Width = node.Width,
                Height = node.Height,
                Depth = node.Depth - depth
            };

            if (right.Width > 0 && right.Height > 0 && right.Depth > 0) _nodes.Add(right);
            if (top.Width > 0 && top.Height > 0 && top.Depth > 0) _nodes.Add(top);
            if (back.Width > 0 && back.Height > 0 && back.Depth > 0) _nodes.Add(back);
            
            node.Width = width;
            node.Height = height;
            node.Depth = depth;
        }
    }
}
