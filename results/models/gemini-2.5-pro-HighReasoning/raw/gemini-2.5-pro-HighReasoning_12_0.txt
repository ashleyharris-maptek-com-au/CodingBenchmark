using System.Collections.Generic;
using System.Linq;

namespace BinPacking3D
{
    public class BinPacker
    {
        public List<Bin> Bins { get; } = new List<Bin>();
        private readonly Bin _binTemplate;

        public BinPacker(int binWidth, int binHeight, int binDepth)
        {
            _binTemplate = new Bin(binWidth, binHeight, binDepth);
        }

        public void Pack(List<Item> items)
        {
            // A simple packing algorithm will be implemented here.
            // For now, it just puts items into bins without checking for space.
            var sortedItems = items.OrderByDescending(x => x.Volume).ToList();

            foreach (var item in sortedItems)
            {
                var bin = FindBinForItem(item);
                bin.AddItem(item);
            }
        }

        private Bin FindBinForItem(Item item)
        {
            foreach (var bin in Bins)
            {
                if (item.Width <= bin.Width && item.Height <= bin.Height && item.Depth <= bin.Depth && item.Volume <= bin.RemainingVolume)
                {
                    return bin;
                }
            }

            var newBin = new Bin(_binTemplate.Width, _binTemplate.Height, _binTemplate.Depth);
            Bins.Add(newBin);
            return newBin;
        }
    }
}
