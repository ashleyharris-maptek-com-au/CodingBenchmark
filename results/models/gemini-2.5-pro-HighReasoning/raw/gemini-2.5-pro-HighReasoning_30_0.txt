import itertools

def solve_3sat(clauses):
    """
    Solves a 3-SAT problem using a simple backtracking algorithm.

    Args:
        clauses: A list of clauses, where each clause is a list of 3 literals.
                 A literal is a non-zero integer. A positive integer represents a
                 variable, and a negative integer represents its negation.
                 For example, [[1, 2, -3], [-1, -2, 4]] represents
                 (x1 V x2 V !x3) AND (!x1 V !x2 V x4).

    Returns:
        A tuple (is_satisfiable, assignment), where is_satisfiable is a boolean
        and assignment is a dictionary mapping variable numbers to boolean values
        (e.g., {1: True, 2: False, ...}) if a solution is found, otherwise None.
    """
    variables = sorted(list(set(abs(l) for clause in clauses for l in clause)))
    num_vars = len(variables)

    for values in itertools.product([True, False], repeat=num_vars):
        assignment = dict(zip(variables, values))
        is_satisfied = True
        for clause in clauses:
            clause_satisfied = False
            for literal in clause:
                var = abs(literal)
                is_negated = literal < 0
                if (not is_negated and assignment.get(var)) or 
                   (is_negated and not assignment.get(var)):
                    clause_satisfied = True
                    break
            if not clause_satisfied:
                is_satisfied = False
                break
        if is_satisfied:
            return True, assignment

    return False, None

if __name__ == '__main__':
    # Example 3-SAT problem:
    # (x1 V x2 V !x3) AND (!x1 V !x2 V x3) AND (x1 V !x2 V x3)
    clauses = [[1, 2, -3], [-1, -2, 3], [1, -2, 3]]

    is_sat, solution = solve_3sat(clauses)

    with open('answer.txt', 'w') as f:
        if is_sat:
            f.write("The 3-SAT problem is satisfiable.
")
            f.write("Satisfying assignment:
")
            # Sort variables for consistent output
            sorted_vars = sorted(solution.keys())
            for var in sorted_vars:
                f.write(f"x{var}: {solution[var]}
")
        else:
            f.write("The 3-SAT problem is not satisfiable.
")
    
    # Store the structured answer in answer.json
    import json
    with open('answer.json', 'w') as f:
        json.dump({
            "is_satisfiable": is_sat,
            "assignment": solution
        }, f, indent=4)
