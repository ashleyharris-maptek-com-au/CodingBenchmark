using System;
using System.Collections.Generic;
using System.Linq;

public class Solver
{
    public static void Main(string[] args)
    {
        var line = Console.ReadLine();
        var dimensions = line.Split(' ');
        var height = int.Parse(dimensions[0]);
        var width = int.Parse(dimensions[1]);

        var board = new char[height][];
        for (var i = 0; i < height; i++)
        {
            board[i] = Console.ReadLine().ToCharArray();
        }

        var solver = new MinesweeperSolver();
        var safeCells = solver.Solve(board, height, width);

        Console.WriteLine(safeCells.Count);
        foreach (var cell in safeCells)
        {
            Console.WriteLine($"{cell.X} {cell.Y}");
        }
    }
}

/// <summary>
/// A Minesweeper solver that uses deterministic logic to find provably safe cells.
/// </summary>
public class MinesweeperSolver
{
    /// <summary>
    /// Solves the given Minesweeper board.
    /// </summary>
    /// <param name="board">The Minesweeper board.</param>
    /// <param name="height">The height of the board.</param>
    /// <param name="width">The width of the board.</param>
    /// <returns>A list of points representing the safe cells.</returns>
    public List<Point> Solve(char[][] board, int height, int width)
    {
        var safeCells = new HashSet<Point>();
        var worklist = new Queue<Point>();

        // Initialize the worklist with all numbered cells.
        for (var y = 0; y < height; y++)
        {
            for (var x = 0; x < width; x++)
            {
                if (board[y][x] >= '0' && board[y][x] <= '8')
                {
                    worklist.Enqueue(new Point(x, y));
                }
            }
        }

        // Process the worklist until it's empty.
        while (worklist.Count > 0)
        {
            var point = worklist.Dequeue();
            var x = point.X;
            var y = point.Y;

            // Skip cells that are not numbered.
            if (!(board[y][x] >= '0' && board[y][x] <= '8')) continue;

            var revealedValue = board[y][x] - '0';
            var neighbors = GetNeighbors(x, y, height, width);
            var flaggedNeighbors = 0;
            var unknownNeighbors = new List<Point>();

            // Analyze the neighbors of the current cell.
            foreach (var neighbor in neighbors)
            {
                if (board[neighbor.Y][neighbor.X] == '*')
                {
                    flaggedNeighbors++;
                }
                else if (board[neighbor.Y][neighbor.X] == ' ')
                {
                    unknownNeighbors.Add(neighbor);
                }
            }

            // Rule 1: If the number on a cell is equal to the number of its flagged neighbors,
            // then all of its other unrevealed neighbors are safe.
            if (revealedValue == flaggedNeighbors && unknownNeighbors.Count > 0)
            {
                foreach (var unknown in unknownNeighbors)
                {
                    if (safeCells.Add(unknown))
                    {
                        // A new safe cell was found. We don't know its value yet,
                        // so we don't add it to the worklist.
                    }
                }
            }

            // Rule 2: If the number on a cell is equal to the sum of its flagged and unknown neighbors,
            // then all of its unknown neighbors must be mines.
            if (revealedValue == flaggedNeighbors + unknownNeighbors.Count && unknownNeighbors.Count > 0)
            {
                foreach (var unknown in unknownNeighbors)
                {
                    if (board[unknown.Y][unknown.X] != '*')
                    {
                        board[unknown.Y][unknown.X] = '*';
                        // A new mine was found. Its neighbors' states have changed,
                        // so we add them to the worklist to be re-evaluated.
                        var newMineNeighbors = GetNeighbors(unknown.X, unknown.Y, height, width);
                        foreach (var neighborOfNewMine in newMineNeighbors)
                        {
                            worklist.Enqueue(neighborOfNewMine);
                        }
                    }
                }
            }
        }

        return safeCells.ToList();
    }

    private List<Point> GetNeighbors(int x, int y, int height, int width)
    {
        var neighbors = new List<Point>();
        for (var yOffset = -1; yOffset <= 1; yOffset++)
        {
            for (var xOffset = -1; xOffset <= 1; xOffset++)
            {
                if (yOffset == 0 && xOffset == 0) continue;

                var newY = y + yOffset;
                var newX = x + xOffset;

                if (newX >= 0 && newX < width && newY >= 0 && newY < height)
                {
                    neighbors.Add(new Point(newX, newY));
                }
            }
        }
        return neighbors;
    }
}

public struct Point
{
    public int X;
    public int Y;

    public Point(int x, int y)
    {
        X = x;
        Y = y;
    }
}
