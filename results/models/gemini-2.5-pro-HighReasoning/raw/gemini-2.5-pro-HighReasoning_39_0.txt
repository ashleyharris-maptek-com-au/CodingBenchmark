use std::collections::{HashMap, HashSet};

struct Graph {
    adj: Vec<Vec<usize>>,
}

impl Graph {
    fn new(num_vertices: usize) -> Self {
        Graph {
            adj: vec![Vec::new(); num_vertices],
        }
    }

    fn add_edge(&mut self, u: usize, v: usize) {
        self.adj[u].push(v);
        self.adj[v].push(u);
    }

    fn num_vertices(&self) -> usize {
        self.adj.len()
    }
}

fn main() {
    // Example Usage
    let mut graph = Graph::new(8);
    graph.add_edge(0, 1);
    graph.add_edge(0, 2);
    graph.add_edge(1, 3);
    graph.add_edge(2, 3);
    graph.add_edge(4, 5);
    graph.add_edge(4, 6);
    graph.add_edge(5, 7);
    graph.add_edge(6, 7);
    graph.add_edge(3, 4); // The edge connecting the two clusters

    let (partition_a, partition_b, cut_size) = kernighan_lin(&graph);

    println!("Partition A: {:?}", partition_a);
    println!("Partition B: {:?}", partition_b);
    println!("Cut Size: {}", cut_size);
}

fn kernighan_lin(graph: &Graph) -> (HashSet<usize>, HashSet<usize>, usize) {
    // 1. Initial Partition
    let num_vertices = graph.num_vertices();
    let mut partition_a: HashSet<usize> = (0..num_vertices / 2).collect();
    let mut partition_b: HashSet<usize> = (num_vertices / 2..num_vertices).collect();

    loop {
        let mut gains = Vec::new();
        let mut swapped_pairs = Vec::new();
        let mut temp_a = partition_a.clone();
        let mut temp_b = partition_b.clone();
        let mut locked_vertices = HashSet::new();

        for _ in 0..num_vertices / 2 {
            let d_values = compute_d_values(graph, &temp_a, &temp_b);
            let mut max_gain = -1.0;
            let mut best_pair = (0, 0);

            for &u in temp_a.iter() {
                if locked_vertices.contains(&u) {
                    continue;
                }
                for &v in temp_b.iter() {
                    if locked_vertices.contains(&v) {
                        continue;
                    }
                    let cost = if graph.adj[u].contains(&v) { 2 } else { 0 };
                    let gain = d_values[&u] + d_values[&v] - cost;
                    if gain as f64 > max_gain {
                        max_gain = gain as f64;
                        best_pair = (u, v);
                    }
                }
            }

            if max_gain > -1.0 {
                let (u, v) = best_pair;
                swapped_pairs.push(best_pair);
                gains.push(max_gain);

                locked_vertices.insert(u);
                locked_vertices.insert(v);

                // Move nodes between partitions for the next iteration of the inner loop
                temp_a.remove(&u);
                temp_b.remove(&v);
                temp_a.insert(v);
                temp_b.insert(u);
            }
        }

        let mut max_cumulative_gain = 0.0;
        let mut best_k = 0;
        let mut cumulative_gain = 0.0;
        for (k, &gain) in gains.iter().enumerate() {
            cumulative_gain += gain;
            if cumulative_gain > max_cumulative_gain {
                max_cumulative_gain = cumulative_gain;
                best_k = k + 1;
            }
        }

        if best_k > 0 {
            for i in 0..best_k {
                let (u, v) = swapped_pairs[i];
                partition_a.remove(&u);
                partition_b.remove(&v);
                partition_a.insert(v);
                partition_b.insert(u);
            }
        } else {
            break;
        }
    }

    let cut_size = calculate_cut_size(graph, &partition_a, &partition_b);
    (partition_a, partition_b, cut_size)
}


fn compute_d_values(
    graph: &Graph,
    partition_a: &HashSet<usize>,
    partition_b: &HashSet<usize>,
) -> HashMap<usize, i32> {
    let mut d_values = HashMap::new();
    let all_vertices: HashSet<usize> = partition_a.union(partition_b).cloned().collect();

    for &u in all_vertices.iter() {
        let mut internal_cost = 0;
        let mut external_cost = 0;
        let current_partition = if partition_a.contains(&u) {
            partition_a
        } else {
            partition_b
        };

        for &v in &graph.adj[u] {
            if current_partition.contains(&v) {
                internal_cost += 1;
            } else {
                external_cost += 1;
            }
        }
        d_values.insert(u, external_cost - internal_cost);
    }
    d_values
}

fn calculate_cut_size(graph: &Graph, partition_a: &HashSet<usize>, partition_b: &HashSet<usize>) -> usize {
    let mut cut_size = 0;
    for &u in partition_a.iter() {
        for &v in &graph.adj[u] {
            if partition_b.contains(&v) {
                cut_size += 1;
            }
        }
    }
    cut_size
}
