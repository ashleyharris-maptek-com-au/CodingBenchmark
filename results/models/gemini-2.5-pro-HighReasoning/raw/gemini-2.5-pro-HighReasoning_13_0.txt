#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

// Function to find the longest common substring of two strings
string findLCS(const string& s1, const string& s2) {
    int m = s1.length();
    int n = s2.length();

    if (m == 0 || n == 0) {
        return "";
    }

    vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
    int maxLength = 0;
    int endIndex = 0;

    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (s1[i - 1] == s2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
                if (dp[i][j] > maxLength) {
                    maxLength = dp[i][j];
                    endIndex = i - 1;
                }
            } else {
                dp[i][j] = 0;
            }
        }
    }

    if (maxLength == 0) {
        return "";
    }

    return s1.substr(endIndex - maxLength + 1, maxLength);
}

// Function to find all occurrences of a substring in a string
vector<size_t> findAllOccurrences(const string& text, const string& pattern) {
    vector<size_t> occurrences;
    if (pattern.empty()) return occurrences;
    for (size_t pos = text.find(pattern); pos != string::npos; pos = text.find(pattern, pos + 1)) {
        occurrences.push_back(pos);
    }
    return occurrences;
}


// Function to check if a substring is common to all strings
bool isCommon(const string& sub, const vector<string>& strings) {
    for (size_t i = 1; i < strings.size(); ++i) {
        if (strings[i].find(sub) == string::npos) {
            return false;
        }
    }
    return true;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;
    cin >> n;

    if (n <= 0) {
        cout << endl;
        return 0;
    }

    vector<string> strings(n);
    for (int i = 0; i < n; ++i) {
        cin >> strings[i];
    }

    if (n == 1) {
        cout << strings[0] << endl;
        return 0;
    }

    // Sort strings by length to start with the shortest
    sort(strings.begin(), strings.end(), [](const string& a, const string& b) {
        return a.length() < b.length();
    });

    string shortest = strings[0];
    string longestCommon = "";

    for (int len = shortest.length(); len > 0; --len) {
        for (int i = 0; i <= (int)shortest.length() - len; ++i) {
            string sub = shortest.substr(i, len);
            if (isCommon(sub, strings)) {
                cout << sub << endl;
                return 0;
            }
        }
    }

    cout << longestCommon << endl;

    return 0;
}
