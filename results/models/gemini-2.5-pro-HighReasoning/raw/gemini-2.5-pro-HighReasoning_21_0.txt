#include "Snake.h"
#include <iostream>
#include <random>

Snake::Snake(int dimensions, int board_size) : dimensions(dimensions), board_size(board_size) {
    // Initialize snake in the center of the board
    std::vector<int> head(dimensions, board_size / 2);
    body.push_front(head);

    // Initial direction (e.g., +1 in the first dimension)
    direction.resize(dimensions, 0);
    direction[0] = 1;

    place_food();
}

void Snake::move() {
    std::vector<int> new_head = body.front();
    for (int i = 0; i < dimensions; ++i) {
        new_head[i] += direction[i];
    }

    if (new_head == food) {
        body.push_front(new_head);
        place_food();
    } else {
        body.push_front(new_head);
        body.pop_back();
    }
}

void Snake::change_direction(const std::vector<int>& new_direction) {
    // The new direction can't be the opposite of the current one
    bool opposite = true;
    for (int i = 0; i < dimensions; ++i) {
        if (new_direction[i] != -direction[i]) {
            opposite = false;
            break;
        }
    }

    if (!opposite) {
        direction = new_direction;
    }
}

bool Snake::check_collision() const {
    const auto& head = body.front();

    // Check for boundary collision
    for (int i = 0; i < dimensions; ++i) {
        if (head[i] < 0 || head[i] >= board_size) {
            return true;
        }
    }

    // Check for self-collision
    for (auto it = std::next(body.begin()); it != body.end(); ++it) {
        if (*it == head) {
            return true;
        }
    }

    return false;
}

void Snake::place_food() {
    std::random_device rd;
    std::mt19937 gen(rd());
    std::uniform_int_distribution<> distrib(0, board_size - 1);

    bool food_on_snake;
    do {
        food_on_snake = false;
        food.clear();
        for (int i = 0; i < dimensions; ++i) {
            food.push_back(distrib(gen));
        }

        for (const auto& segment : body) {
            if (segment == food) {
                food_on_snake = true;
                break;
            }
        }
    } while (food_on_snake);
}

const std::deque<std::vector<int>>& Snake::get_body() const {
    return body;
}

const std::vector<int>& Snake::get_food() const {
    return food;
}

void Snake::print_board() const {
    if (dimensions != 2) {
        std::cout << "Board printing is only supported for 2D games." << std::endl;
        return;
    }

    std::vector<std::vector<char>> board(board_size, std::vector<char>(board_size, '.'));

    // Place food
    if (!food.empty()) {
        board[food[1]][food[0]] = 'F';
    }

    // Place snake
    for (const auto& segment : body) {
        board[segment[1]][segment[0]] = 'S';
    }

    // Print board
    for (int i = 0; i < board_size; ++i) {
        for (int j = 0; j < board_size; ++j) {
            std::cout << board[i][j] << ' ';
        }
        std::cout << std::endl;
    }
    std::cout << std::endl;
}
