use std::io::{self, BufRead};

#[derive(Clone, Copy, PartialEq)]
enum Cell {
    Empty,
    Obstacle,
}

struct Grid {
    width: usize,
    height: usize,
    cells: Vec<Vec<Cell>>,
    total_non_obstacles: usize,
}

impl Grid {
    fn new(width: usize, height: usize) -> Self {
        Grid {
            width,
            height,
            cells: vec![vec![Cell::Empty; height]; width],
            total_non_obstacles: width * height,
        }
    }

    fn add_obstacle(&mut self, x: usize, y: usize) {
        if self.cells[x][y] == Cell::Empty {
            self.cells[x][y] = Cell::Obstacle;
            self.total_non_obstacles -= 1;
        }
    }

    fn is_valid(&self, x: isize, y: isize) -> bool {
        x >= 0 && x < self.width as isize && y >= 0 && y < self.height as isize
    }

    fn is_connected(&self) -> bool {
        if self.total_non_obstacles == 0 {
            return true;
        }

        let mut visited = vec![vec![false; self.height]; self.width];
        let mut queue = std::collections::VecDeque::new();
        let mut count = 0;

        if self.cells[0][0] == Cell::Obstacle {
            return false;
        }

        queue.push_back((0, 0));
        visited[0][0] = true;

        while let Some((x, y)) = queue.pop_front() {
            count += 1;
            let moves = [(0, 1), (0, -1), (1, 0), (-1, 0)];
            for (dx, dy) in &moves {
                let next_x = x as isize + dx;
                let next_y = y as isize + dy;

                if self.is_valid(next_x, next_y) {
                    let next_x = next_x as usize;
                    let next_y = next_y as usize;
                    if !visited[next_x][next_y] && self.cells[next_x][next_y] != Cell::Obstacle {
                        visited[next_x][next_y] = true;
                        queue.push_back((next_x, next_y));
                    }
                }
            }
        }
        count == self.total_non_obstacles
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = lines.next().unwrap().unwrap();
    let mut parts = first_line.split_whitespace();
    let width: usize = parts.next().unwrap().parse().unwrap();
    let height: usize = parts.next().unwrap().parse().unwrap();
    let obstacle_count: usize = parts.next().unwrap().parse().unwrap();
    let require_cycle: bool = parts.next().unwrap().parse::<u8>().unwrap() == 1;

    let mut grid = Grid::new(width, height);

    for _ in 0..obstacle_count {
        let line = lines.next().unwrap().unwrap();
        let mut parts = line.split_whitespace();
        let x: usize = parts.next().unwrap().parse().unwrap();
        let y: usize = parts.next().unwrap().parse().unwrap();
        grid.add_obstacle(x, y);
    }
    
    if grid.cells[0][0] == Cell::Obstacle {
        println!("0");
        return;
    }

    if !grid.is_connected() {
        println!("0");
        return;
    }

    let mut path = Vec::with_capacity(grid.total_non_obstacles);
    let mut visited = vec![vec![false; height]; width];
    
    if solve(&grid, (0, 0), &mut path, &mut visited, require_cycle) {
        println!("{}", path.len());
        for (x, y) in path {
            println!("{} {}", x, y);
        }
    } else {
        println!("0");
    }
}

fn solve(
    grid: &Grid,
    current_pos: (usize, usize),
    path: &mut Vec<(usize, usize)>,
    visited: &mut Vec<Vec<bool>>,
    require_cycle: bool,
) -> bool {
    let (x, y) = current_pos;

    if visited[x][y] {
        return false;
    }
    
    visited[x][y] = true;
    path.push(current_pos);

    if path.len() == grid.total_non_obstacles {
        if require_cycle {
            let (last_x, last_y) = path[path.len() - 1];
            let (start_x, start_y) = (0,0);
            if (last_x as isize - start_x as isize).abs() + (last_y as isize - start_y as isize).abs() == 1 {
                return true;
            } else {
                path.pop();
                visited[x][y] = false;
                return false;
            }
        }
        return true;
    }

    let moves = [(0, 1), (0, -1), (1, 0), (-1, 0)];
    let mut neighbors = Vec::new();
    for (dx, dy) in &moves {
        let next_x = x as isize + dx;
        let next_y = y as isize + dy;

        if grid.is_valid(next_x, next_y) {
            let next_x = next_x as usize;
            let next_y = next_y as usize;
            if grid.cells[next_x][next_y] != Cell::Obstacle && !visited[next_x][next_y] {
                neighbors.push((next_x, next_y));
            }
        }
    }

    neighbors.sort_by_key(|&neighbor| count_unvisited_neighbors(grid, neighbor, visited));

    for (next_x, next_y) in neighbors {
        if solve(grid, (next_x, next_y), path, visited, require_cycle) {
            return true;
        }
    }

    path.pop();
    visited[x][y] = false;
    false
}

fn count_unvisited_neighbors(grid: &Grid, pos: (usize, usize), visited: &Vec<Vec<bool>>) -> usize {
    let (x, y) = pos;
    let mut count = 0;
    let moves = [(0, 1), (0, -1), (1, 0), (-1, 0)];
    for (dx, dy) in &moves {
        let next_x = x as isize + dx;
        let next_y = y as isize + dy;

        if grid.is_valid(next_x, next_y) {
            let next_x = next_x as usize;
            let next_y = next_y as usize;
            if grid.cells[next_x][next_y] != Cell::Obstacle && !visited[next_x][next_y] {
                count += 1;
            }
        }
    }
    count
}
