#include "csg.h"
#include <cmath>
#include <algorithm>

namespace csg {

// Vector implementation
Vector::Vector(double x, double y, double z) : x(x), y(y), z(z) {}

Vector Vector::operator+(const Vector& other) const {
    return Vector(x + other.x, y + other.y, z + other.z);
}

Vector Vector::operator-(const Vector& other) const {
    return Vector(x - other.x, y - other.y, z - other.z);
}

Vector Vector::operator*(double scalar) const {
    return Vector(x * scalar, y * scalar, z * scalar);
}

Vector Vector::operator/(double scalar) const {
    return Vector(x / scalar, y / scalar, z / scalar);
}

double Vector::dot(const Vector& other) const {
    return x * other.x + y * other.y + z * other.z;
}

Vector Vector::cross(const Vector& other) const {
    return Vector(
        y * other.z - z * other.y,
        z * other.x - x * other.z,
        x * other.y - y * other.x
    );
}

double Vector::length() const {
    return std::sqrt(dot(*this));
}

Vector Vector::normalized() const {
    return *this / length();
}

Vector Vector::negated() const {
    return Vector(-x, -y, -z);
}

void Vector::flip() {
    x = -x;
    y = -y;
    z = -z;
}

// Vertex implementation
Vertex::Vertex(const Vector& pos, const Vector& normal) : pos(pos), normal(normal) {}

void Vertex::flip() {
    normal.flip();
}

Vertex Vertex::interpolate(const Vertex& other, double t) const {
    return Vertex(
        pos.operator+(other.pos.operator-(pos).operator*(t)),
        normal.operator+(other.normal.operator-(normal).operator*(t)).normalized()
    );
}

// Polygon implementation
Polygon::Polygon(const std::vector<Vertex>& vertices, void* shared) 
    : vertices(vertices), shared(shared) {}

void Polygon::flip() {
    std::reverse(vertices.begin(), vertices.end());
    for (auto& vertex : vertices) {
        vertex.flip();
    }
}

// Solid implementation
Solid Solid::fromPolygons(const std::vector<Polygon>& polygons) {
    Solid solid;
    solid.polygons = polygons;
    return solid;
}

Solid* Solid::clone() const {
    Solid* newSolid = new Solid();
    newSolid->polygons = this->polygons;
    return newSolid;
}

// Plane implementation
Plane::Plane(const Vector& normal, double w) : normal(normal), w(w) {}

Plane Plane::fromPoints(const Vector& a, const Vector& b, const Vector& c) {
    Vector n = (b - a).cross(c - a).normalized();
    return Plane(n, n.dot(a));
}

void Plane::flip() {
    normal.flip();
    w = -w;
}

void Plane::splitPolygon(const Polygon& polygon, std::vector<Polygon>& coplanarFront,
                         std::vector<Polygon>& coplanarBack, std::vector<Polygon>& front,
                         std::vector<Polygon>& back) const {
    const double EPSILON = 1e-5;
    enum PointLocation { COPLANAR = 0, FRONT = 1, BACK = 2, SPANNING = 3 };

    auto getPointLocation = [&](const Vector& point) {
        double t = this->normal.dot(point) - this->w;
        if (t < -EPSILON) return BACK;
        if (t > EPSILON) return FRONT;
        return COPLANAR;
    };

    PointLocation polygonLocation = COPLANAR;
    std::vector<PointLocation> pointLocations;
    for (const auto& vertex : polygon.vertices) {
        PointLocation loc = getPointLocation(vertex.pos);
        polygonLocation = (PointLocation)(polygonLocation | loc);
        pointLocations.push_back(loc);
    }

    if (polygonLocation == COPLANAR) {
        if (this->normal.dot(polygon.vertices[0].normal) > 0) {
            coplanarFront.push_back(polygon);
        } else {
            coplanarBack.push_back(polygon);
        }
    } else if (polygonLocation == FRONT) {
        front.push_back(polygon);
    } else if (polygonLocation == BACK) {
        back.push_back(polygon);
    } else { // SPANNING
        std::vector<Vertex> f, b;
        for (size_t i = 0; i < polygon.vertices.size(); ++i) {
            size_t j = (i + 1) % polygon.vertices.size();
            const Vertex& vi = polygon.vertices[i];
            const Vertex& vj = polygon.vertices[j];
            PointLocation ti = pointLocations[i];
            PointLocation tj = pointLocations[j];

            if (ti != BACK) f.push_back(vi);
            if (ti != FRONT) b.push_back(vi);
            if ((ti | tj) == SPANNING) {
                double t = (this->w - this->normal.dot(vi.pos)) / this->normal.dot(vj.pos - vi.pos);
                Vertex v = vi.interpolate(vj, t);
                f.push_back(v);
                b.push_back(v);
            }
        }
        if (f.size() >= 3) front.push_back(Polygon(f, polygon.shared));
        if (b.size() >= 3) back.push_back(Polygon(b, polygon.shared));
    }
}

// Node implementation
Node::Node(const std::vector<Polygon>& polygons) : front(nullptr), back(nullptr) {
    if (!polygons.empty()) {
        build(polygons);
    }
}

Node::~Node() {
    delete front;
    delete back;
}

Node* Node::clone() const {
    Node* newNode = new Node();
    newNode->plane = this->plane;
    newNode->polygons = this->polygons;
    if (this->front) newNode->front = this->front->clone();
    if (this->back) newNode->back = this->back->clone();
    return newNode;
}

void Node::invert() {
    for (auto& poly : polygons) {
        poly.flip();
    }
    plane.flip();
    if (front) front->invert();
    if (back) back->invert();
    std::swap(front, back);
}

std::vector<Polygon> Node::clipPolygons(const std::vector<Polygon>& polys) const {
    if (!this->plane.normal.length()) {
        return polys;
    }
    std::vector<Polygon> frontPolys, backPolys;
    for (const auto& poly : polys) {
        this->plane.splitPolygon(poly, frontPolys, backPolys, frontPolys, backPolys);
    }
    if (this->front) frontPolys = this->front->clipPolygons(frontPolys);
    if (this->back) backPolys = this->back->clipPolygons(backPolys);
    else backPolys.clear();
    
    frontPolys.insert(frontPolys.end(), backPolys.begin(), backPolys.end());
    return frontPolys;
}

void Node::clipTo(const Node* other) {
    this->polygons = other->clipPolygons(this->polygons);
    if (this->front) this->front->clipTo(other);
    if (this->back) this->back->clipTo(other);
}

void Node::build(const std::vector<Polygon>& polys) {
    if (polys.empty()) return;
    if (!this->plane.normal.length()) {
        this->plane = Plane::fromPoints(polys[0].vertices[0].pos, polys[0].vertices[1].pos, polys[0].vertices[2].pos);
    }
    std::vector<Polygon> frontPolys, backPolys;
    for (const auto& poly : polys) {
        this->plane.splitPolygon(poly, this->polygons, this->polygons, frontPolys, backPolys);
    }
    if (frontPolys.size() > 0) {
        if (!this->front) this->front = new Node();
        this->front->build(frontPolys);
    }
    if (backPolys.size() > 0) {
        if (!this->back) this->back = new Node();
        this->back->build(backPolys);
    }
}

std::vector<Polygon> Node::allPolygons() const {
    std::vector<Polygon> allPolys = this->polygons;
    if (front) {
        std::vector<Polygon> frontPolys = front->allPolygons();
        allPolys.insert(allPolys.end(), frontPolys.begin(), frontPolys.end());
    }
    if (back) {
        std::vector<Polygon> backPolys = back->allPolygons();
        allPolys.insert(allPolys.end(), backPolys.begin(), backPolys.end());
    }
    return allPolys;
}

// CSG operations
Solid* csg_union(const Solid* a, const Solid* b) {
    Node* a_tree = new Node(a->polygons);
    Node* b_tree = new Node(b->polygons);
    a_tree->clipTo(b_tree);
    b_tree->clipTo(a_tree);
    b_tree->invert();
    b_tree->clipTo(a_tree);
    b_tree->invert();
    a_tree->build(b_tree->allPolygons());
    Solid* result = new Solid();
    result->polygons = a_tree->allPolygons();
    delete a_tree;
    delete b_tree;
    return result;
}

Solid* csg_subtract(const Solid* a, const Solid* b) {
    Node* a_tree = new Node(a->polygons);
    Node* b_tree = new Node(b->polygons);
    a_tree->invert();
    a_tree->clipTo(b_tree);
    b_tree->clipTo(a_tree);
    b_tree->invert();
    b_tree->clipTo(a_tree);
    b_tree->invert();
    a_tree->build(b_tree->allPolygons());
    a_tree->invert();
    Solid* result = new Solid();
    result->polygons = a_tree->allPolygons();
    delete a_tree;
    delete b_tree;
    return result;
}

Solid* csg_intersect(const Solid* a, const Solid* b) {
    Node* a_tree = new Node(a->polygons);
    Node* b_tree = new Node(b->polygons);
    a_tree->invert();
    b_tree->clipTo(a_tree);
    b_tree->invert();
    a_tree->clipTo(b_tree);
    b_tree->clipTo(a_tree);
    a_tree->build(b_tree->allPolygons());
    a_tree->invert();
    Solid* result = new Solid();
    result->polygons = a_tree->allPolygons();
    delete a_tree;
    delete b_tree;
    return result;
}
