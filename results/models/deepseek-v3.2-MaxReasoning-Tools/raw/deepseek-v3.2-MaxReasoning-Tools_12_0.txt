    public class Program
    {
        static void Main()
        {
            // Read input
            var input = Console.In.ReadToEnd().Trim().Split('\n', StringSplitOptions.RemoveEmptyEntries);
            var firstLine = input[0].Split(' ', StringSplitOptions.RemoveEmptyEntries);
            int V = int.Parse(firstLine[0]);
            int F = int.Parse(firstLine[1]);
            
            // Read vertices
            Vector3[] vertices = new Vector3[V];
            for (int i = 0; i < V; i++)
            {
                var coords = input[i + 1].Split(' ', StringSplitOptions.RemoveEmptyEntries);
                vertices[i] = new Vector3(
                    float.Parse(coords[0]),
                    float.Parse(coords[1]),
                    float.Parse(coords[2])
                );
            }
            
            // Read faces
            int[][] faces = new int[F][];
            for (int i = 0; i < F; i++)
            {
                var indices = input[i + V + 1].Split(' ', StringSplitOptions.RemoveEmptyEntries);
                faces[i] = new int[] {
                    int.Parse(indices[0]),
                    int.Parse(indices[1]),
                    int.Parse(indices[2])
                };
            }
            
            // Read container dimensions
            var dims = input[V + F + 1].Split(' ', StringSplitOptions.RemoveEmptyEntries);
            float W = float.Parse(dims[0]);
            float H = float.Parse(dims[1]);
            float D = float.Parse(dims[2]);
            
            // Create polyhedron and container
            Polyhedron poly = new Polyhedron(vertices, faces);
            Container container = new Container(W, H, D);
            
            // Analyze problem scale
            ProblemScale scale = AnalyzeScale(poly, container);
            
            // Choose and execute packing strategy
            List<PackedItem> packedItems = PackAdaptively(poly, container, scale);
            
            // Output results
            Console.WriteLine(packedItems.Count);
            foreach (var item in packedItems)
            {
                Console.WriteLine($"{item.Position.X:F6} {item.Position.Y:F6} {item.Position.Z:F6} " +
                                 $"{item.Rotation.W:F6} {item.Rotation.X:F6} {item.Rotation.Y:F6} {item.Rotation.Z:F6}");
            }
        }
        
        static ProblemScale AnalyzeScale(Polyhedron poly, Container container)
        {
            // Calculate bounding box of polyhedron
            var bounds = poly.GetBoundingBox();
            float polyVolume = bounds.Width * bounds.Height * bounds.Depth;
            float containerVolume = container.Width * container.Height * container.Depth;
            
            // Determine scale based on container size and complexity
            float maxDim = Math.Max(container.Width, Math.Max(container.Height, container.Depth));
            
            if (maxDim <= 5)
                return ProblemScale.Trivial;
            else if (maxDim <= 20)
                return ProblemScale.Medium;
            else if (maxDim <= 100)
                return ProblemScale.Large;
            else
                return ProblemScale.Extreme;
        }
        
        static List<PackedItem> PackAdaptively(Polyhedron poly, Container container, ProblemScale scale)
        {
            IPackingStrategy strategy;
            
            switch (scale)
            {
                case ProblemScale.Trivial:
                    strategy = new ExhaustivePackingStrategy();
                    break;
                case ProblemScale.Medium:
                    strategy = new GridBasedPackingStrategy();
                    break;
                case ProblemScale.Large:
                    strategy = new OctreePackingStrategy();
                    break;
                case ProblemScale.Extreme:
                    strategy = new SpatialHashingPackingStrategy();
                    break;
                default:
                    strategy = new GridBasedPackingStrategy();
                    break;
            }
            
            return strategy.Pack(poly, container);
        }
    }
    
    public enum ProblemScale

        Trivial,

    public struct Vector3
    {
        public float X, Y, Z;
        
        public Vector3(float x, float y, float z)
        {
            X = x; Y = y; Z = z;
        }
        
        public static Vector3 operator +(Vector3 a, Vector3 b) => 
            new Vector3(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
        
        public static Vector3 operator -(Vector3 a, Vector3 b) => 
            new Vector3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
        
        public float Length() => (float)Math.Sqrt(X * X + Y * Y + Z * Z);
        
        public Vector3 Normalize()
        {
            float len = Length();
            return len > 0 ? new Vector3(X / len, Y / len, Z / len) : this;
        }
    }
    
    public struct Quaternion
    {
        public float W, X, Y, Z;
        
        public Quaternion(float w, float x, float y, float z)
        {
            W = w; X = x; Y = y; Z = z;
        }
        
        public static Quaternion Identity => new Quaternion(1, 0, 0, 0);
        
        public static Quaternion FromAxisAngle(Vector3 axis, float angle)
        {
            float halfAngle = angle * 0.5f;
            float sin = (float)Math.Sin(halfAngle);
            float cos = (float)Math.Cos(halfAngle);
            return new Quaternion(cos, axis.X * sin, axis.Y * sin, axis.Z * sin);
        }
        
        public static Quaternion operator *(Quaternion a, Quaternion b) => new Quaternion(
            a.W * b.W - a.X * b.X - a.Y * b.Y - a.Z * b.Z,
            a.W * b.X + a.X * b.W + a.Y * b.Z - a.Z * b.Y,
            a.W * b.Y - a.X * b.Z + a.Y * b.W + a.Z * b.X,
            a.W * b.Z + a.X * b.Y - a.Y * b.X + a.Z * b.W
        );
        
        public Vector3 Rotate(Vector3 v)
        {
            Quaternion qv = new Quaternion(0, v.X, v.Y, v.Z);
            Quaternion qConj = new Quaternion(W, -X, -Y, -Z);
            Quaternion result = this * qv * qConj;
            return new Vector3(result.X, result.Y, result.Z);
        }
    }
    
    public class Polyhedron
    {
        public Vector3[] Vertices { get; }
        public int[][] Faces { get; }
        private BoundingBox? _bounds;
        
        public Polyhedron(Vector3[] vertices, int[][] faces)
        {
            Vertices = vertices;
            Faces = faces;
        }
        
        public BoundingBox GetBoundingBox()
        {
            if (_bounds.HasValue) return _bounds.Value;
            
            float minX = float.MaxValue, minY = float.MaxValue, minZ = float.MaxValue;
            float maxX = float.MinValue, maxY = float.MinValue, maxZ = float.MinValue;
            
            foreach (var v in Vertices)
            {
                minX = Math.Min(minX, v.X);
                minY = Math.Min(minY, v.Y);
                minZ = Math.Min(minZ, v.Z);
                maxX = Math.Max(maxX, v.X);
                maxY = Math.Max(maxY, v.Y);
                maxZ = Math.Max(maxZ, v.Z);
            }
            
            _bounds = new BoundingBox(minX, minY, minZ, maxX - minX, maxY - minY, maxZ - minZ);
            return _bounds.Value;
        }
        
        public Polyhedron Transform(Vector3 translation, Quaternion rotation)
        {
            Vector3[] transformed = new Vector3[Vertices.Length];
            for (int i = 0; i < Vertices.Length; i++)
            {
                transformed[i] = rotation.Rotate(Vertices[i]) + translation;
            }
            return new Polyhedron(transformed, Faces);
        }
    }
    
    public struct BoundingBox
    {
        public float X, Y, Z;
        public float Width, Height, Depth;
        
        public BoundingBox(float x, float y, float z, float width, float height, float depth)
        {
            X = x; Y = y; Z = z;
            Width = width; Height = height; Depth = depth;
        }
        
        public bool Intersects(BoundingBox other)
        {
            return X < other.X + other.Width && X + Width > other.X &&
                   Y < other.Y + other.Height && Y + Height > other.Y &&
                   Z < other.Z + other.Depth && Z + Depth > other.Z;
        }
        
        public bool Contains(BoundingBox other)
        {
            return X <= other.X && Y <= other.Y && Z <= other.Z &&
                   X + Width >= other.X + other.Width &&
                   Y + Height >= other.Y + other.Height &&
                   Z + Depth >= other.Z + other.Depth;
        }
    }
    
    public class Container
    {
        public float Width { get; }
        public float Height { get; }
        public float Depth { get; }
        
        public Container(float width, float height, float depth)
        {
            Width = width;
            Height = height;
            Depth = depth;
        }
        
        public bool CanPlace(Polyhedron poly, Vector3 position)
        {
            var bounds = poly.GetBoundingBox();
            var placedBounds = new BoundingBox(
                position.X + bounds.X,
                position.Y + bounds.Y,
                position.Z + bounds.Z,
                bounds.Width,
                bounds.Height,
                bounds.Depth
            );
            
            return placedBounds.X >= 0 && placedBounds.Y >= 0 && placedBounds.Z >= 0 &&
                   placedBounds.X + placedBounds.Width <= Width &&
                   placedBounds.Y + placedBounds.Height <= Height &&
                   placedBounds.Z + placedBounds.Depth <= Depth;
        }
    }
    
    public struct PackedItem

        public Vector3 Position { get; }
        public Quaternion Rotation { get; }
        
        public PackedItem(Vector3 position, Quaternion rotation)
        {
            Position = position;
            Rotation = rotation;

    public interface IPackingStrategy

        List<PackedItem> Pack(Polyhedron poly, Container container);
    }
    
    // Strategy for trivial problems - exhaustive search with limited rotations
    public class ExhaustivePackingStrategy : IPackingStrategy
    {
        public List<PackedItem> Pack(Polyhedron poly, Container container)
        {
            var result = new List<PackedItem>();
            var bounds = poly.GetBoundingBox();
            
            // Limited set of rotations for trivial problems
            Quaternion[] rotations = GenerateLimitedRotations();
            
            float stepX = bounds.Width;
            float stepY = bounds.Height;
            float stepZ = bounds.Depth;
            
            for (float x = 0; x <= container.Width - bounds.Width; x += stepX)
            {
                for (float y = 0; y <= container.Height - bounds.Height; y += stepY)
                {
                    for (float z = 0; z <= container.Depth - bounds.Depth; z += stepZ)
                    {
                        foreach (var rot in rotations)
                        {
                            var transformed = poly.Transform(new Vector3(x, y, z), rot);
                            if (container.CanPlace(transformed, new Vector3(x, y, z)))
                            {
                                result.Add(new PackedItem(new Vector3(x, y, z), rot));
                                // Move to next position
                                break;
                            }
                        }
                    }
                }
            }
            
            return result;
        }
        
        private Quaternion[] GenerateLimitedRotations()
        {
            // Generate 24 rotations of a cube (symmetry group)
            var rotations = new List<Quaternion>();
            
            // Identity
            rotations.Add(Quaternion.Identity);
            
            // 90 degree rotations around each axis
            Vector3[] axes = { 
                new Vector3(1, 0, 0), 
                new Vector3(0, 1, 0), 
                new Vector3(0, 0, 1) 
            };
            
            foreach (var axis in axes)
            {
                for (int i = 1; i <= 3; i++)
                {
                    float angle = (float)(i * Math.PI / 2);
                    rotations.Add(Quaternion.FromAxisAngle(axis, angle));
                }
            }
            
            return rotations.ToArray();
        }
    }
    
    // Strategy for medium problems - grid-based placement
    public class GridBasedPackingStrategy : IPackingStrategy
    {
        public List<PackedItem> Pack(Polyhedron poly, Container container)
        {
            var result = new List<PackedItem>();
            var bounds = poly.GetBoundingBox();
            
            // Create occupancy grid
            int gridX = (int)Math.Ceiling(container.Width / bounds.Width);
            int gridY = (int)Math.Ceiling(container.Height / bounds.Height);
            int gridZ = (int)Math.Ceiling(container.Depth / bounds.Depth);
            
            bool[,,] occupied = new bool[gridX, gridY, gridZ];
            
            // Try to place in grid cells
            for (int x = 0; x < gridX; x++)
            {
                for (int y = 0; y < gridY; y++)
                {
                    for (int z = 0; z < gridZ; z++)
                    {
                        if (!occupied[x, y, z])
                        {
                            float posX = x * bounds.Width;
                            float posY = y * bounds.Height;
                            float posZ = z * bounds.Depth;
                            
                            // Try different orientations
                            Quaternion[] rotations = GenerateRotationsForGrid();
                            
                            foreach (var rot in rotations)
                            {
                                var transformed = poly.Transform(new Vector3(posX, posY, posZ), rot);
                                if (container.CanPlace(transformed, new Vector3(posX, posY, posZ)))
                                {
                                    // Mark occupied space
                                    int cellsX = (int)Math.Ceiling(bounds.Width / bounds.Width);
                                    int cellsY = (int)Math.Ceiling(bounds.Height / bounds.Height);
                                    int cellsZ = (int)Math.Ceiling(bounds.Depth / bounds.Depth);
                                    
                                    for (int dx = 0; dx < cellsX && x + dx < gridX; dx++)
                                        for (int dy = 0; dy < cellsY && y + dy < gridY; dy++)
                                            for (int dz = 0; dz < cellsZ && z + dz < gridZ; dz++)
                                                occupied[x + dx, y + dy, z + dz] = true;
                                    
                                    result.Add(new PackedItem(new Vector3(posX, posY, posZ), rot));
                                    break;
                                }
                            }
                        }
                    }
                }
            }
            
            return result;
        }
        
        private Quaternion[] GenerateRotationsForGrid()
        {
            // Fewer rotations for grid-based strategy
            return new Quaternion[]
            {
                Quaternion.Identity,
                Quaternion.FromAxisAngle(new Vector3(0, 1, 0), (float)Math.PI / 2),
                Quaternion.FromAxisAngle(new Vector3(0, 1, 0), (float)Math.PI),
                Quaternion.FromAxisAngle(new Vector3(1, 0, 0), (float)Math.PI / 2)
            };
        }
    }
    
    // Strategy for large problems - octree-based spatial partitioning
    public class OctreePackingStrategy : IPackingStrategy
    {
        public List<PackedItem> Pack(Polyhedron poly, Container container)
        {
            var result = new List<PackedItem>();
            var bounds = poly.GetBoundingBox();
            
            // Create octree
            OctreeNode root = new OctreeNode(0, 0, 0, container.Width, container.Height, container.Depth);
            
            // Generate candidate positions using octree subdivision
            var candidatePositions = GenerateCandidatePositions(root, bounds);
            
            // Sort positions by heuristic (e.g., start from corners)
            candidatePositions.Sort((a, b) => 
                (a.X + a.Y + a.Z).CompareTo(b.X + b.Y + b.Z));
            
            foreach (var pos in candidatePositions)
            {
                // Try a few key rotations for large problems
                Quaternion[] rotations = GetKeyRotations();
                
                foreach (var rot in rotations)
                {
                    var transformed = poly.Transform(pos, rot);
                    if (container.CanPlace(transformed, pos) && !IntersectsAny(result, poly, pos, rot, container))
                    {
                        result.Add(new PackedItem(pos, rot));
                        break;
                    }
                }
            }
            
            return result;
        }
        
        private List<Vector3> GenerateCandidatePositions(OctreeNode node, BoundingBox itemBounds)
        {
            var positions = new List<Vector3>();
            
            // Subdivide space and generate positions at subdivision boundaries
            float stepX = Math.Max(itemBounds.Width, node.Width / 4);
            float stepY = Math.Max(itemBounds.Height, node.Height / 4);
            float stepZ = Math.Max(itemBounds.Depth, node.Depth / 4);
            
            for (float x = node.X; x <= node.X + node.Width - itemBounds.Width; x += stepX)
            {
                for (float y = node.Y; y <= node.Y + node.Height - itemBounds.Height; y += stepY)
                {
                    for (float z = node.Z; z <= node.Z + node.Depth - itemBounds.Depth; z += stepZ)
                    {
                        positions.Add(new Vector3(x, y, z));
                    }
                }
            }
            
            return positions;
        }
        
        private Quaternion[] GetKeyRotations()
        {
            // Minimal set of rotations for large problems
            return new Quaternion[]
            {
                Quaternion.Identity,
                Quaternion.FromAxisAngle(new Vector3(0, 0, 1), (float)Math.PI / 2),
                Quaternion.FromAxisAngle(new Vector3(0, 1, 0), (float)Math.PI / 2)
            };
        }
        
        private bool IntersectsAny(List<PackedItem> placed, Polyhedron poly, Vector3 position, Quaternion rotation, Container container)
        {
            // Simplified intersection check for performance
            var newBounds = poly.GetBoundingBox();
            newBounds = new BoundingBox(
                position.X + newBounds.X,
                position.Y + newBounds.Y,
                position.Z + newBounds.Z,
                newBounds.Width,
                newBounds.Height,
                newBounds.Depth
            );
            
            foreach (var item in placed)
            {
                var placedPoly = poly.Transform(item.Position, item.Rotation);
                var placedBounds = placedPoly.GetBoundingBox();
                
                if (newBounds.Intersects(placedBounds))
                    return true;
            }
            
            return false;
        }
    }
    
    // Strategy for extreme problems - spatial hashing for maximum performance
    public class SpatialHashingPackingStrategy : IPackingStrategy
    {
        public List<PackedItem> Pack(Polyhedron poly, Container container)
        {
            var result = new List<PackedItem>();
            var bounds = poly.GetBoundingBox();
            
            // Use spatial hashing for fast occupancy checks
            float cellSize = Math.Max(bounds.Width, Math.Max(bounds.Height, bounds.Depth));
            SpatialHashGrid grid = new SpatialHashGrid(container.Width, container.Height, container.Depth, cellSize);
            
            // Generate positions along a space-filling curve (Z-order/Morton)
            var positions = GenerateSpaceFillingPositions(container, bounds);
            
            // Use only identity rotation for extreme scale (maximize count)
            Quaternion rotation = Quaternion.Identity;
            
            foreach (var pos in positions)
            {
                if (grid.CanPlace(pos, bounds) && container.CanPlace(poly, pos))
                {
                    grid.MarkOccupied(pos, bounds);
                    result.Add(new PackedItem(pos, rotation));
                }
            }
            
            return result;
        }
        
        private List<Vector3> GenerateSpaceFillingPositions(Container container, BoundingBox bounds)
        {
            var positions = new List<Vector3>();
            
            // Simple grid pattern for extreme performance
            float stepX = bounds.Width;
            float stepY = bounds.Height;
            float stepZ = bounds.Depth;
            
            for (float z = 0; z <= container.Depth - bounds.Depth; z += stepZ)
            {
                for (float y = 0; y <= container.Height - bounds.Height; y += stepY)
                {
                    for (float x = 0; x <= container.Width - bounds.Width; x += stepX)
                    {
                        positions.Add(new Vector3(x, y, z));
                    }
                }
            }
            
            return positions;
        }
    }
    
    public class OctreeNode
    {
        public float X, Y, Z;
        public float Width, Height, Depth;
        
        public OctreeNode(float x, float y, float z, float width, float height, float depth)
        {
            X = x; Y = y; Z = z;
            Width = width; Height = height; Depth = depth;
        }
    }
    
    public class SpatialHashGrid
    {
        private Dictionary<long, bool> occupiedCells = new Dictionary<long, bool>();
        private float cellSize;
        private float containerWidth, containerHeight, containerDepth;
        
        public SpatialHashGrid(float width, float height, float depth, float cellSize)
        {
            this.cellSize = cellSize;
            containerWidth = width;
            containerHeight = height;
            containerDepth = depth;
        }
        
        public long GetHashKey(float x, float y, float z)
        {
            int ix = (int)(x / cellSize);
            int iy = (int)(y / cellSize);
            int iz = (int)(z / cellSize);
            
            // Simple hash combining coordinates
            return ((long)ix << 42) | ((long)iy << 21) | (long)iz;
        }
        
        public bool CanPlace(Vector3 position, BoundingBox bounds)
        {
            // Check all cells that would be occupied by this item
            int cellsX = (int)Math.Ceiling(bounds.Width / cellSize);
            int cellsY = (int)Math.Ceiling(bounds.Height / cellSize);
            int cellsZ = (int)Math.Ceiling(bounds.Depth / cellSize);
            
            for (int dx = 0; dx <= cellsX; dx++)
            {
                for (int dy = 0; dy <= cellsY; dy++)
                {
                    for (int dz = 0; dz <= cellsZ; dz++)
                    {
                        float checkX = position.X + dx * cellSize;
                        float checkY = position.Y + dy * cellSize;
                        float checkZ = position.Z + dz * cellSize;
                        
                        if (checkX >= containerWidth || checkY >= containerHeight || checkZ >= containerDepth)
                            return false;
                            
                        long key = GetHashKey(checkX, checkY, checkZ);
                        if (occupiedCells.ContainsKey(key))
                            return false;
                    }
                }
            }
            
            return true;
        }
        
        public void MarkOccupied(Vector3 position, BoundingBox bounds)
        {
            int cellsX = (int)Math.Ceiling(bounds.Width / cellSize);
            int cellsY = (int)Math.Ceiling(bounds.Height / cellSize);
            int cellsZ = (int)Math.Ceiling(bounds.Depth / cellSize);
            
            for (int dx = 0; dx <= cellsX; dx++)
            {
                for (int dy = 0; dy <= cellsY; dy++)
                {
                    for (int dz = 0; dz <= cellsZ; dz++)
                    {
                        float markX = position.X + dx * cellSize;
                        float markY = position.Y + dy * cellSize;
                        float markZ = position.Z + dz * cellSize;
                        
                        if (markX < containerWidth && markY < containerHeight && markZ < containerDepth)
                        {
                            long key = GetHashKey(markX, markY, markZ);
                            occupiedCells[key] = true;
                        }
                    }
                }
            }
        }
    }
