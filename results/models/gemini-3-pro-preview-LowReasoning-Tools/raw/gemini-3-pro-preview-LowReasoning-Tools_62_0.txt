declare const require: any;
const fs = require('fs');

function solve() {
  const input: string = fs.readFileSync(0, 'utf8');
  if (!input.trim()) {
    console.log(JSON.stringify({ firebreaks: [] }));
    return;
  }
  const data = JSON.parse(input);

  const width: number = data.width;
  const height: number = data.height;
  const grid: string[] = data.grid;
  const ignitions: [number, number][] = data.ignitions;
  const towns: { id: number; x: number; y: number; value: number }[] = data.towns;
  const cutBudget: number = data.cutBudget;

  // 1. Mark burnable cells
  const isBurnable = new Uint8Array(width * height);
  for (let y = 0; y < height; y++) {
    for (let x = 0; x < width; x++) {
      const c = y * width + x;
      if (grid[y][x] === '.') {
        isBurnable[c] = 1;
      }
    }
  }
  // Ignitions must be burnable (source of fire)
  for (const ign of ignitions) {
    const c = ign[1] * width + ign[0];
    isBurnable[c] = 1;
  }

  // 2. Find reachable cells from ignitions
  const reachable = new Uint8Array(width * height);
  const queue: number[] = [];
  for (const ign of ignitions) {
    const cell = ign[1] * width + ign[0];
    reachable[cell] = 1;
    queue.push(cell);
  }
  let head = 0;
  while (head < queue.length) {
    const cell = queue[head++];
    const x = cell % width;
    const y = Math.floor(cell / width);

    const tryNb = (nx: number, ny: number) => {
      const nb = ny * width + nx;
      if (grid[ny][nx] === '.' && !reachable[nb]) {
        reachable[nb] = 1;
        queue.push(nb);
      }
    };

    if (x > 0) tryNb(x - 1, y);
    if (x < width - 1) tryNb(x + 1, y);
    if (y > 0) tryNb(x, y - 1);
    if (y < height - 1) tryNb(x, y + 1);
  }

  const isIgnition = new Uint8Array(width * height);
  for (const ign of ignitions) {
    isIgnition[ign[1] * width + ign[0]] = 1;
  }

  // 3. Collect and sum town values by cell
  const townValue = new Float64Array(width * height);
  for (const t of towns) {
    const cell = t.y * width + t.x;
    if (reachable[cell] === 1 && !isIgnition[cell]) {
      townValue[cell] += t.value;
    }
  }

  const townCells: number[] = [];
  for (let c = 0; c < width * height; c++) {
    if (townValue[c] > 0) {
      townCells.push(c);
    }
  }

  if (cutBudget === 0 || townCells.length === 0) {
    console.log(JSON.stringify({ firebreaks: [] }));
    return;
  }

  townCells.sort((a, b) => townValue[b] - townValue[a]);

  // Set node capacities
  const capacity = new Float64Array(width * height);
  for (let c = 0; c < width * height; c++) {
    if (isIgnition[c]) {
      capacity[c] = Infinity;
    } else if (townValue[c] > 0) {
      capacity[c] = Infinity;
    } else {
      capacity[c] = 1;
    }
  }

  // Allocate visited and parentState arrays once
  const parentState = new Int32Array(2 * width * height);
  const visited = new Int32Array(2 * width * height);
  let visitedToken = 0;

  function runFlowWithState(
    targetSinks: Uint8Array,
    initialState: any
  ): any {
    const flow = initialState.flow.slice();
    const pred = initialState.pred.slice();
    const succ = initialState.succ.slice();
    let totalFlow = initialState.totalFlow;

    while (true) {
      if (totalFlow > cutBudget) {
        return null;
      }

      visitedToken++;

      const queue: number[] = [];
      for (const ign of ignitions) {
        const cell = ign[1] * width + ign[0];
        const state = (cell << 1) | 1;
        visited[state] = visitedToken;
        parentState[state] = -1;
        queue.push(state);
      }

      let foundSinkState = -1;
      let head = 0;

      while (head < queue.length) {
        const state = queue[head++];
        const cell = state >> 1;
        const isOut = (state & 1) === 1;

        if (!isOut && targetSinks[cell]) {
          foundSinkState = state;
          break;
        }

        if (!isOut) {
          if (capacity[cell] === Infinity || flow[cell] === 0) {
            const nextState = (cell << 1) | 1;
            if (visited[nextState] !== visitedToken) {
              visited[nextState] = visitedToken;
              parentState[nextState] = state;
              queue.push(nextState);
            }
          }
          if (capacity[cell] === 1 && flow[cell] === 1) {
            const p = pred[cell];
            if (p !== -1) {
              const nextState = (p << 1) | 1;
              if (visited[nextState] !== visitedToken) {
                visited[nextState] = visitedToken;
                parentState[nextState] = state;
                queue.push(nextState);
              }
            }
          }
        } else {
          if (capacity[cell] === 1 && flow[cell] === 1) {
            const nextState = cell << 1;
            if (visited[nextState] !== visitedToken) {
              visited[nextState] = visitedToken;
              parentState[nextState] = state;
              queue.push(nextState);
            }
          }

          const x = cell % width;
          const y = Math.floor(cell / width);
          const succCell = succ[cell];

          const tryNeighbor = (nx: number, ny: number) => {
            const nb = ny * width + nx;
            if (isBurnable[nb]) {
              if (capacity[cell] === Infinity || succCell !== nb) {
                const nextState = nb << 1;
                if (visited[nextState] !== visitedToken) {
                  visited[nextState] = visitedToken;
                  parentState[nextState] = state;
                  queue.push(nextState);
                }
              }
            }
          };

          if (x > 0) tryNeighbor(x - 1, y);
          if (x < width - 1) tryNeighbor(x + 1, y);
          if (y > 0) tryNeighbor(x, y - 1);
          if (y < height - 1) tryNeighbor(x, y + 1);
        }
      }

      if (foundSinkState === -1) {
        break;
      }

      const path: number[] = [];
      let curr = foundSinkState;
      while (curr !== -1) {
        path.push(curr);
        curr = parentState[curr];
      }
      path.reverse();

      for (let i = 0; i < path.length; i++) {
        const state = path[i];
        const cell = state >> 1;
        const isOut = (state & 1) === 1;

        if (!isOut && i + 1 < path.length && path[i + 1] === ((cell << 1) | 1)) {
          if (capacity[cell] === 1) {
            const p = path[i - 1] >> 1;
            const s = path[i + 2] >> 1;
            flow[cell] = 1;
            pred[cell] = p;
            succ[cell] = s;
          }
          i++;
        } else if (isOut && i + 1 < path.length && path[i + 1] === (cell << 1)) {
          if (capacity[cell] === 1) {
            flow[cell] = 0;
            pred[cell] = -1;
            succ[cell] = -1;
          }
          i++;
        }
      }

      totalFlow++;
    }

    return { flow, pred, succ, totalFlow };
  }

  function runFlow(targetSinks: Uint8Array): any {
    const initialState = {
      flow: new Int32Array(width * height),
      pred: new Int32Array(width * height),
      succ: new Int32Array(width * height),
      totalFlow: 0
    };
    initialState.pred.fill(-1);
    initialState.succ.fill(-1);
    return runFlowWithState(targetSinks, initialState);
  }

  function findReachableStates(
    flow: any,
    pred: any,
    succ: any
  ): Uint8Array {
    visitedToken++;
    const queue: number[] = [];

    for (const ign of ignitions) {
      const cell = ign[1] * width + ign[0];
      const state = (cell << 1) | 1;
      visited[state] = visitedToken;
      queue.push(state);
    }

    let head = 0;
    while (head < queue.length) {
      const state = queue[head++];
      const cell = state >> 1;
      const isOut = (state & 1) === 1;

      if (!isOut) {
        if (capacity[cell] === Infinity || flow[cell] === 0) {
          const nextState = (cell << 1) | 1;
          if (visited[nextState] !== visitedToken) {
            visited[nextState] = visitedToken;
            queue.push(nextState);
          }
        }
        if (capacity[cell] === 1 && flow[cell] === 1) {
          const p = pred[cell];
          if (p !== -1) {
            const nextState = (p << 1) | 1;
            if (visited[nextState] !== visitedToken) {
              visited[nextState] = visitedToken;
              queue.push(nextState);
            }
          }
        }
      } else {
        if (capacity[cell] === 1 && flow[cell] === 1) {
          const nextState = cell << 1;
          if (visited[nextState] !== visitedToken) {
            visited[nextState] = visitedToken;
            queue.push(nextState);
          }
        }

        const x = cell % width;
        const y = Math.floor(cell / width);
        const succCell = succ[cell];

        const tryNeighbor = (nx: number, ny: number) => {
          const nb = ny * width + nx;
          if (isBurnable[nb]) {
            if (capacity[cell] === Infinity || succCell !== nb) {
              const nextState = nb << 1;
              if (visited[nextState] !== visitedToken) {
                visited[nextState] = visitedToken;
                queue.push(nextState);
              }
            }
          }
        };

        if (x > 0) tryNeighbor(x - 1, y);
        if (x < width - 1) tryNeighbor(x + 1, y);
        if (y > 0) tryNeighbor(x, y - 1);
        if (y < height - 1) tryNeighbor(x, y + 1);
      }
    }

    const S_reach = new Uint8Array(2 * width * height);
    for (let i = 0; i < 2 * width * height; i++) {
      if (visited[i] === visitedToken) {
        S_reach[i] = 1;
      }
    }
    return S_reach;
  }

  // First try to save all reachable towns
  const allSinks = new Uint8Array(width * height);
  for (const c of townCells) {
    allSinks[c] = 1;
  }

  const allState = runFlow(allSinks);

  let finalFlowState = allState;

  if (allState === null) {
    // Cannot save all towns within budget, run greedy algorithm
    const candidates = townCells.slice(0, 100);

    let bestFlowState = {
      flow: new Int32Array(width * height),
      pred: new Int32Array(width * height),
      succ: new Int32Array(width * height),
      totalFlow: 0
    };
    bestFlowState.pred.fill(-1);
    bestFlowState.succ.fill(-1);

    const activeSinks = new Uint8Array(width * height);

    while (true) {
      let bestCandidate = -1;
      let bestRatio = -1;
      let bestCandidateState = null;

      for (const c of candidates) {
        if (activeSinks[c]) continue;

        activeSinks[c] = 1;
        const testState = runFlowWithState(activeSinks, bestFlowState);
        activeSinks[c] = 0;

        if (testState !== null) {
          const delta = testState.totalFlow - bestFlowState.totalFlow;
          const cost = delta === 0 ? 1e-6 : delta;
          const ratio = townValue[c] / cost;
          if (ratio > bestRatio) {
            bestRatio = ratio;
            bestCandidate = c;
            bestCandidateState = testState;
          }
        }
      }

      if (bestCandidate === -1) {
        break;
      }

      activeSinks[bestCandidate] = 1;
      bestFlowState = bestCandidateState!;
    }

    finalFlowState = bestFlowState;
  }

  // Reconstruct the cut cells
  const S_reach = findReachableStates(
    finalFlowState!.flow,
    finalFlowState!.pred,
    finalFlowState!.succ
  );

  const cutCells: number[] = [];
  for (let c = 0; c < width * height; c++) {
    if (capacity[c] === 1 && S_reach[c << 1] === 1 && S_reach[(c << 1) | 1] === 0) {
      cutCells.push(c);
    }
  }

  const firebreaks = cutCells.map(c => {
    const x = c % width;
    const y = Math.floor(c / width);
    return [x, y];
  });

  console.log(JSON.stringify({ firebreaks }));
}

solve();
