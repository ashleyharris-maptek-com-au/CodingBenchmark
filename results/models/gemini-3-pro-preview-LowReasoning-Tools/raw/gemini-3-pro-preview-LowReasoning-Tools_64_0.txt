declare const require: any;
const fs = require('fs');

interface Drone {
  id: number;
  x: number;
  y: number;
}

interface InputData {
  probability: number[][];
  drones: Drone[];
  maxStepsPerDrone: number;
}

function solve() {
  const input: string = fs.readFileSync(0, 'utf8');
  if (!input.trim()) {
    console.log(JSON.stringify({ paths: [] }));
    return;
  }
  const data: InputData = JSON.parse(input);

  const probability = data.probability;
  const drones = data.drones;
  const maxStepsPerDrone = data.maxStepsPerDrone;

  const H = probability.length;
  if (H === 0) {
    console.log(JSON.stringify({ paths: [] }));
    return;
  }
  const W = probability[0].length;
  if (W === 0) {
    console.log(JSON.stringify({ paths: Array(drones.length).fill([]) }));
    return;
  }

  // Pre-collect starting cells' probabilities and set them to 0
  for (const drone of drones) {
    if (drone.y >= 0 && drone.y < H && drone.x >= 0 && drone.x < W) {
      if (probability[drone.y][drone.x] > 0) {
        probability[drone.y][drone.x] = 0;
      }
    }
  }

  const decay = 0.9;
  const K = 30; // Diffusion iterations
  
  // Decide lookahead depth L based on scale of problem
  const totalSteps = drones.length * maxStepsPerDrone;
  let L = 5;
  if (totalSteps > 10000) {
    L = 4;
  }
  if (totalSteps > 50000) {
    L = 3;
  }
  L = Math.min(L, maxStepsPerDrone);

  // We will plan paths sequentially for each drone
  const paths: [number, number][][] = [];

  // Setup diffusion function
  function computeDiffusion(prob: number[][]): Float64Array[] {
    let current = Array.from({ length: H }, (_, y) => {
      const arr = new Float64Array(W);
      for (let x = 0; x < W; x++) {
        arr[x] = prob[y][x] < 0 ? 0 : prob[y][x];
      }
      return arr;
    });
    let next = Array.from({ length: H }, () => new Float64Array(W));
    
    const dx = [0, 0, -1, 1];
    const dy = [-1, 1, 0, 0];
    
    for (let iter = 0; iter < K; iter++) {
      let changed = false;
      for (let y = 0; y < H; y++) {
        for (let x = 0; x < W; x++) {
          if (prob[y][x] < 0) {
            next[y][x] = 0;
            continue;
          }
          let maxNeighbor = 0;
          for (let i = 0; i < 4; i++) {
            const nx = x + dx[i];
            const ny = y + dy[i];
            if (nx >= 0 && nx < W && ny >= 0 && ny < H && prob[ny][nx] >= 0) {
              if (current[ny][nx] > maxNeighbor) {
                maxNeighbor = current[ny][nx];
              }
            }
          }
          const val = Math.max(current[y][x], decay * maxNeighbor);
          next[y][x] = val;
          if (Math.abs(val - current[y][x]) > 1e-6) {
            changed = true;
          }
        }
      }
      const temp = current;
      current = next;
      next = temp;
      if (!changed) break;
    }
    return current;
  }

  // Determine recompute interval for diffusion
  const N = W * H;
  const maxOps = 5000000;
  const totalDiffOps = totalSteps * K * N;
  let recomputeInterval = Math.max(1, Math.floor(totalDiffOps / maxOps));
  // Recompute at least 10 times per drone
  recomputeInterval = Math.min(recomputeInterval, Math.max(1, Math.floor(maxStepsPerDrone / 10)));

  let stepsSinceRecompute = recomputeInterval;
  let diffused = Array.from({ length: H }, () => new Float64Array(W));

  const dx = [0, 0, 0, -1, 1];
  const dy = [0, -1, 1, 0, 0];

  const dfsVisited = new Uint8Array(H * W);

  for (let d = 0; d < drones.length; d++) {
    const drone = drones[d];
    let cx = drone.x;
    let cy = drone.y;

    const path: [number, number][] = [[cx, cy]];
    const visitedCount = new Uint32Array(H * W);
    visitedCount[cy * W + cx] = 1;

    for (let step = 0; step < maxStepsPerDrone; step++) {
      // Recompute diffusion if interval reached
      if (stepsSinceRecompute >= recomputeInterval || step === 0) {
        diffused = computeDiffusion(probability);
        stepsSinceRecompute = 0;
      }

      const currentL = Math.min(L, maxStepsPerDrone - step);
      if (currentL <= 0) break;

      let bestMoveX = cx;
      let bestMoveY = cy;
      let bestScore = -Infinity;

      // DFS helper
      function dfs(x: number, y: number, depth: number, currentSum: number, pathLength: number): number {
        if (depth === 0) {
          const idx = y * W + x;
          const count = visitedCount[idx] + (dfsVisited[idx] ? 1 : 0);
          const effDiffused = diffused[y][x] / (1 + count);
          return currentSum + Math.pow(decay, pathLength) * effDiffused;
        }

        let maxScore = -Infinity;
        for (let i = 0; i < 5; i++) {
          const nx = x + dx[i];
          const ny = y + dy[i];
          if (nx >= 0 && nx < W && ny >= 0 && ny < H && probability[ny][nx] !== -1) {
            const nidx = ny * W + nx;
            const wasVisited = dfsVisited[nidx] !== 0;

            let addedVal = 0;
            if (!wasVisited) {
              addedVal = probability[ny][nx] * Math.pow(decay, pathLength + 1);
              dfsVisited[nidx] = 1;
            }

            const score = dfs(nx, ny, depth - 1, currentSum + addedVal, pathLength + 1);
            if (score > maxScore) {
              maxScore = score;
            }

            if (!wasVisited) {
              dfsVisited[nidx] = 0;
            }
          }
        }
        return maxScore;
      }

      // Evaluate the first step moves
      for (let i = 0; i < 5; i++) {
        const nx = cx + dx[i];
        const ny = cy + dy[i];
        if (nx >= 0 && nx < W && ny >= 0 && ny < H && probability[ny][nx] !== -1) {
          const nidx = ny * W + nx;
          const wasVisited = dfsVisited[nidx] !== 0;

          let addedVal = 0;
          if (!wasVisited) {
            addedVal = probability[ny][nx] * decay;
            dfsVisited[nidx] = 1;
          }

          const score = dfs(nx, ny, currentL - 1, addedVal, 1);
          if (score > bestScore) {
            bestScore = score;
            bestMoveX = nx;
            bestMoveY = ny;
          }

          if (!wasVisited) {
            dfsVisited[nidx] = 0;
          }
        }
      }

      // Make the best move
      cx = bestMoveX;
      cy = bestMoveY;
      path.push([cx, cy]);
      visitedCount[cy * W + cx]++;
      probability[cy][cx] = 0; // mark as collected
      stepsSinceRecompute++;
    }

    paths.push(path);
  }

  console.log(JSON.stringify({ paths }));
}

solve();
