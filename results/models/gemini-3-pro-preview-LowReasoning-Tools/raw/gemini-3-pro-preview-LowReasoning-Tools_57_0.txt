const fs = require('fs');

function solve() {
  const input = fs.readFileSync(0, 'utf8');
  if (!input.trim()) return;
  const data = JSON.parse(input);

  const width = data.width;
  const height = data.height;
  const grid = data.grid;
  const ignitions = data.ignitions;
  const towns = data.towns;
  const cutBudget = data.cutBudget;

  const numCells = width * height;

  // 1. Run BFS from ignitions to find reachable cells (baseline fire spread)
  const reachable = new Uint8Array(numCells);
  const q = [];
  for (const [x, y] of ignitions) {
    const id = y * width + x;
    if (grid[y][x] !== '~') {
      q.push(id);
      reachable[id] = 1;
    }
  }
  let qHead = 0;
  while (qHead < q.length) {
    const curr = q[qHead++];
    const cx = curr % width;
    const cy = Math.floor(curr / width);
    const neighbors = [
      [cx + 1, cy],
      [cx - 1, cy],
      [cx, cy + 1],
      [cx, cy - 1]
    ];
    for (const [nx, ny] of neighbors) {
      if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
        if (grid[ny][nx] !== '~') {
          const nid = ny * width + nx;
          if (!reachable[nid]) {
            reachable[nid] = 1;
            q.push(nid);
          }
        }
      }
    }
  }

  // Identify vulnerable towns (towns that would burn in the baseline)
  const vulnerableTowns = [];
  const townIdx = new Int32Array(numCells).fill(-1);
  for (let i = 0; i < towns.length; i++) {
    const t = towns[i];
    const id = t.y * width + t.x;
    townIdx[id] = i;
    if (reachable[id] === 1) {
      vulnerableTowns.push(t);
    }
  }

  // If no towns would burn anyway, or budget is 0, we need no cuts
  if (vulnerableTowns.length === 0 || cutBudget <= 0) {
    console.log(JSON.stringify({ firebreaks: [] }));
    return;
  }

  const isIgnition = new Uint8Array(numCells);
  for (const [x, y] of ignitions) {
    isIgnition[y * width + x] = 1;
  }

  // 2. Build the flow network
  const S = 2 * numCells;
  const T = 2 * numCells + 1;
  const numNodes = T + 1;
  const maxEdges = 16 * numCells;

  const head = new Int32Array(numNodes).fill(-1);
  const to = new Int32Array(maxEdges);
  const cap = new Int32Array(maxEdges);
  const flow = new Int32Array(maxEdges);
  const next = new Int32Array(maxEdges);
  let edgeCount = 0;

  function addEdge(u, v, c) {
    // Forward edge
    to[edgeCount] = v;
    cap[edgeCount] = c;
    flow[edgeCount] = 0;
    next[edgeCount] = head[u];
    head[u] = edgeCount++;

    // Reverse edge
    to[edgeCount] = u;
    cap[edgeCount] = 0;
    flow[edgeCount] = 0;
    next[edgeCount] = head[v];
    head[v] = edgeCount++;
  }

  // Add internal and neighbor transition edges for reachable cells
  for (let u = 0; u < numCells; u++) {
    if (reachable[u] === 1) {
      const u_in = 2 * u;
      const u_out = 2 * u + 1;
      let internalCap = 1;
      if (isIgnition[u] || townIdx[u] !== -1) {
        internalCap = 1000000; // Large capacity for source/sink nodes
      }
      addEdge(u_in, u_out, internalCap);

      const cx = u % width;
      const cy = Math.floor(u / width);
      const neighbors = [
        [cx + 1, cy],
        [cx - 1, cy],
        [cx, cy + 1],
        [cx, cy - 1]
      ];
      for (const [nx, ny] of neighbors) {
        if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
          const v = ny * width + nx;
          if (grid[ny][nx] !== '~' && reachable[v] === 1) {
            const v_in = 2 * v;
            addEdge(u_out, v_in, 1000000);
          }
        }
      }
    }
  }

  // Connect S to ignitions
  for (const [x, y] of ignitions) {
    const u = y * width + x;
    addEdge(S, 2 * u, 1000000);
  }

  // Connect towns to T (initially with capacity 0)
  const townEdgeIdx = new Int32Array(towns.length).fill(-1);
  for (let i = 0; i < towns.length; i++) {
    const t = towns[i];
    const u = t.y * width + t.x;
    if (reachable[u] === 1) {
      addEdge(2 * u + 1, T, 0);
      townEdgeIdx[i] = edgeCount - 2;
    }
  }

  // 3. Optimized Dinic's Algorithm
  const level = new Int32Array(numNodes).fill(-1);
  const queue = new Int32Array(numNodes);
  const ptr = new Int32Array(numNodes);
  const ptrToken = new Int32Array(numNodes);
  let queueLength = 0;
  let currentPtrToken = 0;

  function bfs(src, dest) {
    for (let i = 0; i < queueLength; i++) {
      level[queue[i]] = -1;
    }
    queueLength = 0;

    level[src] = 0;
    queue[queueLength++] = src;
    let headQ = 0;
    while (headQ < queueLength) {
      const u = queue[headQ++];
      for (let e = head[u]; e !== -1; e = next[e]) {
        const v = to[e];
        if (cap[e] - flow[e] > 0 && level[v] === -1) {
          level[v] = level[u] + 1;
          queue[queueLength++] = v;
        }
      }
    }
    return level[dest] !== -1;
  }

  function dfs(u, dest, pushed) {
    if (pushed === 0) return 0;
    if (u === dest) return pushed;

    if (ptrToken[u] !== currentPtrToken) {
      ptrToken[u] = currentPtrToken;
      ptr[u] = head[u];
    }

    for (let cid = ptr[u]; cid !== -1; cid = next[cid]) {
      ptr[u] = cid;
      const v = to[cid];
      const tr = cap[cid] - flow[cid];
      if (level[u] + 1 !== level[v] || tr === 0) continue;
      const push = dfs(v, dest, Math.min(pushed, tr));
      if (push === 0) continue;
      flow[cid] += push;
      flow[cid ^ 1] -= push;
      return push;
    }
    return 0;
  }

  function cleanupBfs() {
    for (let i = 0; i < queueLength; i++) {
      level[queue[i]] = -1;
    }
    queueLength = 0;
  }

  function dinic(src, dest, limit) {
    let totalFlow = 0;
    queueLength = 0;
    while (bfs(src, dest)) {
      currentPtrToken++;
      while (true) {
        const pushed = dfs(src, dest, limit - totalFlow);
        if (pushed === 0) break;
        totalFlow += pushed;
        if (totalFlow >= limit) {
          cleanupBfs();
          return totalFlow;
        }
      }
    }
    cleanupBfs();
    return totalFlow;
  }

  function resetFlow() {
    flow.fill(0);
  }

  // 4. Calculate individual min-cut cost for each vulnerable town
  const individualCosts = new Int32Array(towns.length);
  for (let i = 0; i < towns.length; i++) {
    const eIdx = townEdgeIdx[i];
    if (eIdx === -1) {
      individualCosts[i] = 0;
      continue;
    }

    resetFlow();
    for (let j = 0; j < towns.length; j++) {
      const idx = townEdgeIdx[j];
      if (idx !== -1) {
        cap[idx] = (j === i) ? 1000000 : 0;
      }
    }

    individualCosts[i] = dinic(S, T, cutBudget + 1);
  }

  // Filter out vulnerable towns that cannot be saved individually within budget
  const candidateTowns = [];
  for (const t of vulnerableTowns) {
    const cost = individualCosts[t.id];
    if (cost <= cutBudget) {
      candidateTowns.push({
        id: t.id,
        x: t.x,
        y: t.y,
        value: t.value,
        individualCost: cost
      });
    }
  }

  if (candidateTowns.length === 0) {
    console.log(JSON.stringify({ firebreaks: [] }));
    return;
  }

  // 5. Function to run the greedy selection for a given order of towns
  const savedFlow = new Int32Array(maxEdges);

  function getMinCutCuts() {
    const visited = new Uint8Array(numNodes);
    const qC = new Int32Array(numNodes);
    let qHead = 0;
    let qTail = 0;

    qC[qTail++] = S;
    visited[S] = 1;

    while (qHead < qTail) {
      const u = qC[qHead++];
      for (let e = head[u]; e !== -1; e = next[e]) {
        const v = to[e];
        if (cap[e] - flow[e] > 0 && !visited[v]) {
          visited[v] = 1;
          qC[qTail++] = v;
        }
      }
    }

    const cuts = [];
    for (let u = 0; u < numCells; u++) {
      if (reachable[u] === 1 && !isIgnition[u] && townIdx[u] === -1) {
        const u_in = 2 * u;
        const u_out = 2 * u + 1;
        if (visited[u_in] && !visited[u_out]) {
          const cx = u % width;
          const cy = Math.floor(u / width);
          cuts.push([cx, cy]);
        }
      }
    }
    return cuts;
  }

  function runGreedyForOrder(sortedTowns) {
    resetFlow();
    for (let i = 0; i < towns.length; i++) {
      const eIdx = townEdgeIdx[i];
      if (eIdx !== -1) {
        cap[eIdx] = 0;
      }
    }

    let currentValue = 0;
    let currentFlow = 0;

    for (const t of sortedTowns) {
      const eIdx = townEdgeIdx[t.id];
      if (eIdx === -1) continue;

      savedFlow.set(flow);
      cap[eIdx] = 1000000;

      const additionalFlow = dinic(S, T, cutBudget + 1 - currentFlow);
      const newFlow = currentFlow + additionalFlow;

      if (newFlow <= cutBudget) {
        currentFlow = newFlow;
        currentValue += t.value;
      } else {
        flow.set(savedFlow);
        cap[eIdx] = 0;
      }
    }

    const cuts = getMinCutCuts();
    return { value: currentValue, cuts: cuts };
  }

  // 6. Run trials to find the best selection of towns to save
  const startTime = Date.now();
  const timeLimit = 400; // Limit search to 400ms to stay safe

  let bestValue = -1;
  let bestCuts = [];

  let trial = 0;
  while (true) {
    // Deterministic strategies for the first few trials, then randomized
    let sorted = [...candidateTowns];
    if (trial === 0) {
      // Sort by value/cost descending
      sorted.sort((a, b) => b.value / b.individualCost - a.value / a.individualCost);
    } else if (trial === 1) {
      // Sort by value descending
      sorted.sort((a, b) => b.value - a.value);
    } else if (trial === 2) {
      // Sort by value^2 / cost descending
      sorted.sort((a, b) => (b.value * b.value) / b.individualCost - (a.value * a.value) / a.individualCost);
    } else if (trial === 3) {
      // Sort by cost ascending
      sorted.sort((a, b) => a.individualCost - b.individualCost);
    } else {
      // Randomized priority search
      if (Date.now() - startTime >= timeLimit) {
        break;
      }
      sorted.sort((a, b) => {
        const noiseA = 1.0 + (Math.random() - 0.5) * 0.4;
        const noiseB = 1.0 + (Math.random() - 0.5) * 0.4;
        const scoreA = (a.value / a.individualCost) * noiseA;
        const scoreB = (b.value / b.individualCost) * noiseB;
        return scoreB - scoreA;
      });
    }

    const result = runGreedyForOrder(sorted);
    if (result.value > bestValue) {
      bestValue = result.value;
      bestCuts = result.cuts;
    }

    trial++;
    if (trial >= 100) {
      // If we did 100 trials, we are already very likely to have found the optimal or near-optimal
      break;
    }
  }

  // Output the firebreaks
  console.log(JSON.stringify({ firebreaks: bestCuts }));
}

solve();
