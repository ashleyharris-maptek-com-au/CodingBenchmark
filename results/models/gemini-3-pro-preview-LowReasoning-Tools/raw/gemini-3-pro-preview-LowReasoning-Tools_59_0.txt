const fs = require('fs');

function main() {
  const input = fs.readFileSync(0, 'utf8');
  if (!input) return;
  const data = JSON.parse(input);

  const probability = data.probability;
  const drones = data.drones;
  const maxStepsPerDrone = data.maxStepsPerDrone;

  const height = probability.length;
  const width = probability[0].length;

  const startTime = Date.now();

  // Multi-source BFS to precompute distance to nearest reward
  const minDistToReward = Array.from({length: height}, () => new Int32Array(width).fill(-1));
  const queue = [];
  let head = 0;

  for (let r = 0; r < height; r++) {
    for (let c = 0; c < width; c++) {
      if (probability[r][c] > 0) {
        minDistToReward[r][c] = 0;
        queue.push(c, r);
      }
    }
  }

  const dirs = [[0, 1], [0, -1], [1, 0], [-1, 0]];

  while (head < queue.length) {
    const cx = queue[head++];
    const cy = queue[head++];
    const cd = minDistToReward[cy][cx];

    for (const [dx, dy] of dirs) {
      const nx = cx + dx;
      const ny = cy + dy;
      if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
        if (probability[ny][nx] !== -1 && minDistToReward[ny][nx] === -1) {
          minDistToReward[ny][nx] = cd + 1;
          queue.push(nx, ny);
        }
      }
    }
  }

  const globalVisited = new Int32Array(width * height);
  const tempVisited = new Uint8Array(width * height);
  const beamVisited = new Uint8Array(width * height);
  const currentPathVisited = new Uint8Array(width * height);

  let B = 10;
  let maxDepth = 10;
  const totalSteps = drones.length * maxStepsPerDrone;
  if (totalSteps > 1000) {
    B = 5;
    maxDepth = 6;
  } else if (totalSteps > 500) {
    B = 8;
    maxDepth = 8;
  } else {
    B = 15;
    maxDepth = 12;
  }

  const qSize = (2 * (maxDepth + 1) * (maxDepth + 1) + 10) * 3;
  const q = new Int32Array(qSize);

  let totalCoverageScore = 0;

  function addPathToGlobal(path) {
    for (const [x, y] of path) {
      const key = y * width + x;
      if (globalVisited[key] === 0 && probability[y][x] > 0) {
        totalCoverageScore += probability[y][x];
      }
      globalVisited[key]++;
    }
  }

  function removePathFromGlobal(path) {
    for (const [x, y] of path) {
      const key = y * width + x;
      globalVisited[key]--;
      if (globalVisited[key] === 0 && probability[y][x] > 0) {
        totalCoverageScore -= probability[y][x];
      }
    }
  }

  function getHeuristic(x, y, pathKeys, maxDepth) {
    let score = 0;
    let foundReward = false;

    for (let i = 0; i < pathKeys.length; i++) {
      currentPathVisited[pathKeys[i]] = 1;
    }

    let qHead = 0;
    let qTail = 0;

    const startKey = y * width + x;
    tempVisited[startKey] = 1;
    const localVisited = [startKey];

    q[qTail++] = x;
    q[qTail++] = y;
    q[qTail++] = 0;

    while (qHead < qTail) {
      const cx = q[qHead++];
      const cy = q[qHead++];
      const cd = q[qHead++];

      const cellKey = cy * width + cx;
      if (probability[cy][cx] > 0 && globalVisited[cellKey] === 0 && currentPathVisited[cellKey] === 0) {
        score += probability[cy][cx] / (cd + 1);
        foundReward = true;
      }

      if (cd < maxDepth) {
        for (const [dx, dy] of dirs) {
          const nx = cx + dx;
          const ny = cy + dy;
          if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
            if (probability[ny][nx] !== -1) {
              const nkey = ny * width + nx;
              if (tempVisited[nkey] === 0) {
                tempVisited[nkey] = 1;
                localVisited.push(nkey);
                q[qTail++] = nx;
                q[qTail++] = ny;
                q[qTail++] = cd + 1;
              }
            }
          }
        }
      }
    }

    for (const k of localVisited) {
      tempVisited[k] = 0;
    }
    for (let i = 0; i < pathKeys.length; i++) {
      currentPathVisited[pathKeys[i]] = 0;
    }

    if (foundReward) {
      return score;
    } else {
      const d = minDistToReward[y][x];
      if (d === -1) {
        return -1000000;
      }
      return -d * 0.01;
    }
  }

  const dirsWithStay = [[0, 0], [0, 1], [0, -1], [1, 0], [-1, 0]];

  function planDronePath(droneIndex) {
    const drone = drones[droneIndex];
    const startX = drone.x;
    const startY = drone.y;

    const startKey = startY * width + startX;
    const startReward = (globalVisited[startKey] === 0 && probability[startY][startX] > 0) ? probability[startY][startX] : 0;

    let beam = [{
      x: startX,
      y: startY,
      path: [[startX, startY]],
      pathKeys: [startKey],
      score: startReward,
      hScore: startReward + getHeuristic(startX, startY, [startKey], maxDepth)
    }];

    for (let step = 1; step <= maxStepsPerDrone; step++) {
      const candidates = [];

      for (const state of beam) {
        for (let i = 0; i < state.pathKeys.length; i++) {
          beamVisited[state.pathKeys[i]] = 1;
        }

        for (const [dx, dy] of dirsWithStay) {
          const nx = state.x + dx;
          const ny = state.y + dy;

          if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
            if (probability[ny][nx] !== -1) {
              const nkey = ny * width + nx;
              const isNewVisit = (globalVisited[nkey] === 0 && beamVisited[nkey] === 0 && probability[ny][nx] > 0);
              const newScore = state.score + (isNewVisit ? probability[ny][nx] : 0);

              const newPathKeys = state.pathKeys.concat([nkey]);
              const newPath = state.path.concat([[nx, ny]]);

              const h = getHeuristic(nx, ny, newPathKeys, maxDepth);
              const newHScore = newScore + h;

              candidates.push({
                x: nx,
                y: ny,
                path: newPath,
                pathKeys: newPathKeys,
                score: newScore,
                hScore: newHScore
              });
            }
          }
        }

        for (let i = 0; i < state.pathKeys.length; i++) {
          beamVisited[state.pathKeys[i]] = 0;
        }
      }

      if (candidates.length === 0) {
        break;
      }

      const seen = new Map();
      const nextCandidates = [];
      for (const cand of candidates) {
        const key = cand.y * width + cand.x;
        if (!seen.has(key)) {
          seen.set(key, []);
        }
        seen.get(key).push(cand);
      }

      for (const [key, list] of seen) {
        list.sort((a, b) => b.hScore - a.hScore);
        nextCandidates.push(...list.slice(0, 2));
      }

      nextCandidates.sort((a, b) => b.hScore - a.hScore);
      beam = nextCandidates.slice(0, B);
    }

    return beam[0].path;
  }

  const paths = [];
  for (let i = 0; i < drones.length; i++) {
    paths.push([]);
  }

  for (let i = 0; i < drones.length; i++) {
    const path = planDronePath(i);
    paths[i] = path;
    addPathToGlobal(path);
  }

  while (Date.now() - startTime < 1000) {
    let improved = false;
    for (let i = 0; i < drones.length; i++) {
      if (Date.now() - startTime >= 1200) break;

      const oldPath = paths[i];
      const oldScore = totalCoverageScore;

      removePathFromGlobal(oldPath);
      const newPath = planDronePath(i);
      addPathToGlobal(newPath);
      paths[i] = newPath;

      if (totalCoverageScore > oldScore) {
        improved = true;
      } else {
        removePathFromGlobal(newPath);
        addPathToGlobal(oldPath);
        paths[i] = oldPath;
      }
    }
    if (!improved) {
      break;
    }
  }

  console.log(JSON.stringify({ paths }));
}

main();
