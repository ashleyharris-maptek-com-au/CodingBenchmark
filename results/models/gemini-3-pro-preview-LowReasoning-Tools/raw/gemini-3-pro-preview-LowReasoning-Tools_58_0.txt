const fs = require('fs');

function main() {
  const input = fs.readFileSync(0, 'utf8');
  const data = JSON.parse(input);

  const grid = data.grid;
  const H = grid.length;
  const W = grid[0].length;

  const robots = data.robots;
  const jobs = data.jobs;

  // Environment variable limit or default
  const limitEnv = process.env.WAREHOUSE_EXACT_JOB_LIMIT;
  const WAREHOUSE_EXACT_JOB_LIMIT = limitEnv ? parseInt(limitEnv, 10) : 150;
  const largeCase = jobs.length > WAREHOUSE_EXACT_JOB_LIMIT;

  // Create jobs map
  const jobsMap = {};
  for (const job of jobs) {
    jobsMap[job.id] = job;
  }

  // remaining holds job IDs, sorted ascending
  const remaining = jobs.map(j => j.id).sort((a, b) => {
    const na = Number(a);
    const nb = Number(b);
    return na - nb;
  });

  // Initialize assignments
  const assignments = robots.map(() => []);

  // Current positions of robots
  const current = robots.map(r => ({ x: r.x, y: r.y }));

  // Cache for grid distances
  const cache = new Map();

  function getDistance(pos, target) {
    if (largeCase) {
      return Math.abs(pos.x - target.x) + Math.abs(pos.y - target.y);
    } else {
      const startKey = pos.y * W + pos.x;
      let dists = cache.get(startKey);
      if (!dists) {
        dists = new Int32Array(W * H);
        dists.fill(1e9);
        dists[startKey] = 0;

        const q = new Int32Array(W * H);
        let head = 0;
        let tail = 0;
        q[tail++] = startKey;

        while (head < tail) {
          const key = q[head++];
          const x = key % W;
          const y = Math.floor(key / W);
          const nd = dists[key] + 1;

          // Neighbors: (x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)
          if (x + 1 < W && grid[y][x + 1] !== '#') {
            const nkey = key + 1;
            if (dists[nkey] === 1e9) {
              dists[nkey] = nd;
              q[tail++] = nkey;
            }
          }
          if (x - 1 >= 0 && grid[y][x - 1] !== '#') {
            const nkey = key - 1;
            if (dists[nkey] === 1e9) {
              dists[nkey] = nd;
              q[tail++] = nkey;
            }
          }
          if (y + 1 < H && grid[y + 1][x] !== '#') {
            const nkey = key + W;
            if (dists[nkey] === 1e9) {
              dists[nkey] = nd;
              q[tail++] = nkey;
            }
          }
          if (y - 1 >= 0 && grid[y - 1][x] !== '#') {
            const nkey = key - W;
            if (dists[nkey] === 1e9) {
              dists[nkey] = nd;
              q[tail++] = nkey;
            }
          }
        }
        cache.set(startKey, dists);
      }
      return dists[target.y * W + target.x];
    }
  }

  while (remaining.length > 0) {
    let best = null;
    for (let ri = 0; ri < current.length; ri++) {
      const pos = current[ri];
      for (let i = 0; i < remaining.length; i++) {
        const jid = remaining[i];
        const d = getDistance(pos, jobsMap[jid]);
        if (best === null || d < best.d) {
          best = { d, ri, jid, index: i };
        }
      }
    }
    assignments[best.ri].push(best.jid);
    current[best.ri] = jobsMap[best.jid];
    remaining.splice(best.index, 1);
  }

  console.log(JSON.stringify({ assignments }));
}

main();
