const fs = require('fs');

function dist(p1, p2) {
  const dx = p1.x - p2.x;
  const dy = p1.y - p2.y;
  return Math.sqrt(dx * dx + dy * dy);
}

function reverseSubarray(arr, i, j) {
  while (i < j) {
    const temp = arr[i];
    arr[i] = arr[j];
    arr[j] = temp;
    i++;
    j--;
  }
}

function getTourDist(depot, tour) {
  if (tour.length === 0) return 0;
  let total = dist(depot, tour[0]);
  for (let i = 0; i < tour.length - 1; i++) {
    total += dist(tour[i], tour[i + 1]);
  }
  total += dist(tour[tour.length - 1], depot);
  return total;
}

function optimize2Opt(depot, tour) {
  const L = tour.length;
  let improved = true;
  let iterations = 0;
  const maxIterations = L > 1000 ? 5 : (L > 500 ? 10 : 100);
  const W = L > 500 ? 50 : L;
  
  const refined = tour.slice();
  
  while (improved && iterations < maxIterations) {
    improved = false;
    iterations++;
    for (let i = 0; i < L; i++) {
      const prev = i === 0 ? depot : refined[i - 1];
      const end = Math.min(L - 1, i + W);
      for (let j = i + 1; j <= end; j++) {
        const next = j === L - 1 ? depot : refined[j + 1];
        
        const oldCost = dist(prev, refined[i]) + dist(refined[j], next);
        const newCost = dist(prev, refined[j]) + dist(refined[i], next);
        
        if (newCost < oldCost - 1e-9) {
          reverseSubarray(refined, i, j);
          improved = true;
        }
      }
    }
  }
  return refined;
}

function solveTSP(depot, packages, run2Opt = true) {
  if (packages.length === 0) return [];
  if (packages.length === 1) return [packages[0].id];
  
  // Try Candidate 1: Nearest Neighbor
  let tour1 = [];
  {
    const visited = new Set();
    let current = depot;
    for (let i = 0; i < packages.length; i++) {
      let bestDist = Infinity;
      let bestIdx = -1;
      for (let j = 0; j < packages.length; j++) {
        if (visited.has(j)) continue;
        const d = dist(current, packages[j]);
        if (d < bestDist) {
          bestDist = d;
          bestIdx = j;
        }
      }
      tour1.push(packages[bestIdx]);
      visited.add(bestIdx);
      current = packages[bestIdx];
    }
  }
  
  // Try Candidate 2: Angular Sort
  let tour2 = packages.slice().sort((a, b) => {
    const angleA = Math.atan2(a.y - depot.y, a.x - depot.x);
    const angleB = Math.atan2(b.y - depot.y, b.x - depot.x);
    return angleA - angleB;
  });
  
  if (run2Opt) {
    tour1 = optimize2Opt(depot, tour1);
    tour2 = optimize2Opt(depot, tour2);
  }
  
  const dist1 = getTourDist(depot, tour1);
  const dist2 = getTourDist(depot, tour2);
  
  const bestTour = dist1 < dist2 ? tour1 : tour2;
  return bestTour.map(p => p.id);
}

function getAdjustedAngle(angle, offset) {
  let diff = angle - offset;
  diff = diff % (2 * Math.PI);
  if (diff < 0) diff += 2 * Math.PI;
  return diff;
}

// 1. Angular Sweep VRP Solver
function runSweep(depot, packages, K, offsetCount = 16) {
  const N = packages.length;
  if (N === 0) return [];
  
  const packagesWithAngles = packages.map(p => {
    const angle = Math.atan2(p.y - depot.y, p.x - depot.x);
    return { p, angle };
  });
  
  let bestOffset = 0;
  let bestDist = Infinity;
  
  for (let step = 0; step < offsetCount; step++) {
    const offset = (2 * Math.PI * step) / offsetCount;
    
    packagesWithAngles.sort((a, b) => {
      return getAdjustedAngle(a.angle, offset) - getAdjustedAngle(b.angle, offset);
    });
    
    const routes = [];
    let startIdx = 0;
    for (let k = 0; k < K; k++) {
      const size = Math.floor(N / K) + (k < (N % K) ? 1 : 0);
      const clusterPackages = [];
      for (let i = 0; i < size; i++) {
        clusterPackages.push(packagesWithAngles[startIdx + i].p);
      }
      startIdx += size;
      
      const route = solveTSP(depot, clusterPackages, false);
      routes.push(route);
    }
    
    const packagesMap = new Map(packages.map(p => [p.id, p]));
    const totalDist = computeTotalDistance(routes, depot, packagesMap);
    if (totalDist < bestDist) {
      bestDist = totalDist;
      bestOffset = offset;
    }
  }
  
  packagesWithAngles.sort((a, b) => {
    return getAdjustedAngle(a.angle, bestOffset) - getAdjustedAngle(b.angle, bestOffset);
  });
  
  const finalRoutes = [];
  let startIdx = 0;
  for (let k = 0; k < K; k++) {
    const size = Math.floor(N / K) + (k < (N % K) ? 1 : 0);
    const clusterPackages = [];
    for (let i = 0; i < size; i++) {
      clusterPackages.push(packagesWithAngles[startIdx + i].p);
    }
    startIdx += size;
    
    const route = solveTSP(depot, clusterPackages, true);
    finalRoutes.push(route);
  }
  
  return finalRoutes;
}

// 2. K-Means VRP Solver
function runKMeans(depot, packages, K, maxKMeansIterations = 15) {
  const N = packages.length;
  if (N === 0) return [];
  if (K >= N) {
    return packages.map(p => [p.id]);
  }
  
  const centroids = [];
  centroids.push({ x: packages[0].x, y: packages[0].y });
  const chosen = new Set([0]);
  
  const minDistSq = new Array(N).fill(Infinity);
  for (let k = 1; k < K; k++) {
    const prevCentroid = centroids[k - 1];
    let bestIdx = -1;
    let maxDist = -1;
    for (let i = 0; i < N; i++) {
      if (chosen.has(i)) continue;
      const dSq = Math.pow(packages[i].x - prevCentroid.x, 2) + Math.pow(packages[i].y - prevCentroid.y, 2);
      if (dSq < minDistSq[i]) {
        minDistSq[i] = dSq;
      }
      if (minDistSq[i] > maxDist) {
        maxDist = minDistSq[i];
        bestIdx = i;
      }
    }
    if (bestIdx === -1) {
      for (let i = 0; i < N; i++) {
        if (!chosen.has(i)) {
          bestIdx = i;
          break;
        }
      }
    }
    centroids.push({ x: packages[bestIdx].x, y: packages[bestIdx].y });
    chosen.add(bestIdx);
  }
  
  let assignments = new Array(N).fill(-1);
  for (let iter = 0; iter < maxKMeansIterations; iter++) {
    let changed = false;
    const clusterSizes = new Array(K).fill(0);
    const sumX = new Array(K).fill(0);
    const sumY = new Array(K).fill(0);
    
    for (let i = 0; i < N; i++) {
      let bestCentroidIdx = -1;
      let minD = Infinity;
      for (let k = 0; k < K; k++) {
        const d = Math.pow(packages[i].x - centroids[k].x, 2) + Math.pow(packages[i].y - centroids[k].y, 2);
        if (d < minD) {
          minD = d;
          bestCentroidIdx = k;
        }
      }
      if (assignments[i] !== bestCentroidIdx) {
        assignments[i] = bestCentroidIdx;
        changed = true;
      }
      clusterSizes[bestCentroidIdx]++;
      sumX[bestCentroidIdx] += packages[i].x;
      sumY[bestCentroidIdx] += packages[i].y;
    }
    
    if (!changed) break;
    
    for (let k = 0; k < K; k++) {
      if (clusterSizes[k] > 0) {
        centroids[k].x = sumX[k] / clusterSizes[k];
        centroids[k].y = sumY[k] / clusterSizes[k];
      } else {
        let bestIdx = -1;
        let maxD = -1;
        for (let i = 0; i < N; i++) {
          const d = Math.pow(packages[i].x - centroids[assignments[i]].x, 2) + Math.pow(packages[i].y - centroids[assignments[i]].y, 2);
          if (d > maxD) {
            maxD = d;
            bestIdx = i;
          }
        }
        if (bestIdx !== -1) {
          centroids[k].x = packages[bestIdx].x;
          centroids[k].y = packages[bestIdx].y;
          assignments[bestIdx] = k;
        }
      }
    }
  }
  
  const clusters = Array.from({ length: K }, () => []);
  for (let i = 0; i < N; i++) {
    clusters[assignments[i]].push(packages[i]);
  }
  
  const routes = [];
  for (let k = 0; k < K; k++) {
    routes.push(solveTSP(depot, clusters[k]));
  }
  return routes;
}

// 3. Clarke-Wright Savings VRP Solver
function runClarkeWright(depot, packages, K) {
  const N = packages.length;
  if (N === 0) return [];
  
  const routes = [];
  for (let i = 0; i < N; i++) {
    routes.push([i]);
  }
  const routeOf = new Array(N);
  for (let i = 0; i < N; i++) {
    routeOf[i] = i;
  }
  
  const dists = Array.from({ length: N }, () => new Float64Array(N));
  for (let i = 0; i < N; i++) {
    for (let j = i + 1; j < N; j++) {
      const d = dist(packages[i], packages[j]);
      dists[i][j] = d;
      dists[j][i] = d;
    }
  }
  
  const savings = [];
  const pairAdded = new Uint8Array(N * N);
  
  for (let i = 0; i < N; i++) {
    const dDepotI = dist(depot, packages[i]);
    const candidates = [];
    for (let j = 0; j < N; j++) {
      if (i === j) continue;
      candidates.push({ j, d: dists[i][j] });
    }
    candidates.sort((a, b) => a.d - b.d);
    
    const limit = Math.min(candidates.length, 50);
    for (let k = 0; k < limit; k++) {
      const j = candidates[k].j;
      const minVal = Math.min(i, j);
      const maxVal = Math.max(i, j);
      const idx = minVal * N + maxVal;
      if (pairAdded[idx] === 1) continue;
      pairAdded[idx] = 1;
      
      const dDepotJ = dist(depot, packages[j]);
      const s = dDepotI + dDepotJ - dists[i][j];
      savings.push({ i, j, s });
    }
  }
  
  savings.sort((a, b) => b.s - a.s);
  
  let activeRouteCount = N;
  
  for (let idx = 0; idx < savings.length; idx++) {
    if (activeRouteCount <= K) break;
    
    const { i, j } = savings[idx];
    const r1 = routeOf[i];
    const r2 = routeOf[j];
    
    if (r1 === r2) continue;
    
    const R1 = routes[r1];
    const R2 = routes[r2];
    
    const first1 = R1[0];
    const last1 = R1[R1.length - 1];
    const first2 = R2[0];
    const last2 = R2[R2.length - 1];
    
    const iIsEnd = (i === first1 || i === last1);
    const jIsEnd = (j === first2 || j === last2);
    
    if (iIsEnd && jIsEnd) {
      let merged;
      if (i === last1 && j === first2) {
        merged = R1.concat(R2);
      } else if (i === first1 && j === first2) {
        merged = R1.slice().reverse().concat(R2);
      } else if (i === last1 && j === last2) {
        merged = R1.concat(R2.slice().reverse());
      } else {
        merged = R2.concat(R1);
      }
      
      routes[r1] = merged;
      routes[r2] = null;
      
      for (let m = 0; m < merged.length; m++) {
        routeOf[merged[m]] = r1;
      }
      
      activeRouteCount--;
    }
  }
  
  const activeRoutes = routes.filter(r => r !== null);
  const finalRoutes = activeRoutes.map(r => {
    const pkgs = r.map(idx => packages[idx]);
    return solveTSP(depot, pkgs, true);
  });
  
  while (finalRoutes.length < K) {
    finalRoutes.push([]);
  }
  
  return finalRoutes;
}

function computeTotalDistance(routes, depot, packagesMap) {
  let total = 0;
  for (const route of routes) {
    if (route.length === 0) continue;
    total += dist(depot, packagesMap.get(route[0]));
    for (let i = 0; i < route.length - 1; i++) {
      total += dist(packagesMap.get(route[i]), packagesMap.get(route[i + 1]));
    }
    total += dist(packagesMap.get(route[route.length - 1]), depot);
  }
  return total;
}

function main() {
  const input = fs.readFileSync(0, 'utf8');
  if (!input || input.trim() === '') {
    console.log(JSON.stringify({ routes: [] }));
    return;
  }
  
  const data = JSON.parse(input);
  const depot = { x: data.depot[0], y: data.depot[1] };
  const truckCount = data.truckCount;
  const packages = data.packages;
  
  if (!packages || packages.length === 0) {
    console.log(JSON.stringify({ routes: [] }));
    return;
  }
  
  const K = Math.min(truckCount, packages.length);
  const packagesMap = new Map(packages.map(p => [p.id, p]));
  
  let bestRoutes = null;
  let bestDist = Infinity;
  
  // Strategy 1: Angular Sweep
  const routesSweep = runSweep(depot, packages, K);
  const distSweep = computeTotalDistance(routesSweep, depot, packagesMap);
  if (distSweep < bestDist) {
    bestDist = distSweep;
    bestRoutes = routesSweep;
  }
  
  // Strategy 2: KMeans
  const routesKMeans = runKMeans(depot, packages, K);
  const distKMeans = computeTotalDistance(routesKMeans, depot, packagesMap);
  if (distKMeans < bestDist) {
    bestDist = distKMeans;
    bestRoutes = routesKMeans;
  }
  
  // Strategy 3: Clarke-Wright (for smaller scenarios to save time)
  if (packages.length <= 2000) {
    const routesCW = runClarkeWright(depot, packages, K);
    const distCW = computeTotalDistance(routesCW, depot, packagesMap);
    if (distCW < bestDist) {
      bestDist = distCW;
      bestRoutes = routesCW;
    }
  }
  
  // Filter out any empty routes
  const activeRoutes = bestRoutes.filter(r => r && r.length > 0);
  
  console.log(JSON.stringify({ routes: activeRoutes }));
}

main();
