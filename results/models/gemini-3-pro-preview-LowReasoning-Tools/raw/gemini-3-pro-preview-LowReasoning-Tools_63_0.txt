declare const require: any;
const fs = require('fs');

interface Robot {
  id: number;
  x: number;
  y: number;
}

interface Job {
  id: number;
  x: number;
  y: number;
  priority: number;
}

interface InputData {
  grid: string[];
  robots: Robot[];
  jobs: Job[];
}

function solve() {
  const programStartTime = Date.now();

  let input: string;
  try {
    input = fs.readFileSync(0, 'utf8');
  } catch (e) {
    return;
  }
  if (!input) return;
  const data: InputData = JSON.parse(input);

  const grid = data.grid;
  const robots = data.robots;
  const jobs = data.jobs;

  const H = grid.length;
  if (H === 0) {
    console.log(JSON.stringify({ assignments: [] }));
    return;
  }
  const W = grid[0].length;
  if (W === 0) {
    console.log(JSON.stringify({ assignments: [] }));
    return;
  }

  const R = robots.length;
  const J = jobs.length;
  const N = R + J;

  // Flattened blocked representation
  const blocked = new Uint8Array(W * H);
  for (let y = 0; y < H; y++) {
    const row = grid[y];
    for (let x = 0; x < W; x++) {
      if (row[x] === '#') {
        blocked[y * W + x] = 1;
      }
    }
  }

  // Node coordinates and indices
  const nodeX = new Int32Array(N);
  const nodeY = new Int32Array(N);
  const nodeIdx = new Int32Array(N);

  for (let r = 0; r < R; r++) {
    nodeX[r] = robots[r].x;
    nodeY[r] = robots[r].y;
    nodeIdx[r] = robots[r].y * W + robots[r].x;
  }
  for (let j = 0; j < J; j++) {
    nodeX[R + j] = jobs[j].x;
    nodeY[R + j] = jobs[j].y;
    nodeIdx[R + j] = jobs[j].y * W + jobs[j].x;
  }

  // Distance matrix cache: N arrays of size N, allocated lazily
  const distCache: (Int32Array | null)[] = Array.from({ length: N }, () => null);

  // Pre-allocated BFS structures
  let runId = 0;
  const visited = new Int32Array(W * H);
  const dist = new Int32Array(W * H);
  const queue = new Int32Array(W * H);

  function runBFS(u: number) {
    if (distCache[u] === null) {
      distCache[u] = new Int32Array(N);
      distCache[u]!.fill(-2);
    }
    const uArr = distCache[u]!;

    runId++;
    const startIdx = nodeIdx[u];
    
    visited[startIdx] = runId;
    dist[startIdx] = 0;
    
    let head = 0;
    let tail = 0;
    queue[tail++] = startIdx;
    
    while (head < tail) {
      const curr = queue[head++];
      const d = dist[curr];
      const currX = curr % W;
      const currY = (curr / W) | 0;
      
      // Left
      if (currX > 0) {
        const next = curr - 1;
        if (!blocked[next] && visited[next] !== runId) {
          visited[next] = runId;
          dist[next] = d + 1;
          queue[tail++] = next;
        }
      }
      // Right
      if (currX < W - 1) {
        const next = curr + 1;
        if (!blocked[next] && visited[next] !== runId) {
          visited[next] = runId;
          dist[next] = d + 1;
          queue[tail++] = next;
        }
      }
      // Up
      if (currY > 0) {
        const next = curr - W;
        if (!blocked[next] && visited[next] !== runId) {
          visited[next] = runId;
          dist[next] = d + 1;
          queue[tail++] = next;
        }
      }
      // Down
      if (currY < H - 1) {
        const next = curr + W;
        if (!blocked[next] && visited[next] !== runId) {
          visited[next] = runId;
          dist[next] = d + 1;
          queue[tail++] = next;
        }
      }
    }
    
    // Populate distMatrix for u to all v
    for (let v = 0; v < N; v++) {
      const vIdx = nodeIdx[v];
      const d = (visited[vIdx] === runId) ? dist[vIdx] : -1;
      uArr[v] = d;
      if (distCache[v] !== null) {
        distCache[v]![u] = d;
      }
    }
  }

  function getDist(u: number, v: number): number {
    if (distCache[u] !== null && distCache[u]![v] !== -2) {
      return distCache[u]![v];
    }
    if (distCache[v] !== null && distCache[v]![u] !== -2) {
      return distCache[v]![u];
    }
    runBFS(u);
    return distCache[u]![v];
  }

  function pathDist(r: number, path: number[]): number {
    if (path.length === 0) return 0;
    let d = getDist(r, path[0]);
    if (d === -1) return Infinity;
    for (let i = 0; i < path.length - 1; i++) {
      const step = getDist(path[i], path[i + 1]);
      if (step === -1) return Infinity;
      d += step;
    }
    return d;
  }

  function calculateTotalDistance(assignments: number[][]): number {
    let total = 0;
    for (let r = 0; r < R; r++) {
      const d = pathDist(r, assignments[r]);
      if (d === Infinity) return Infinity;
      total += d;
    }
    return total;
  }

  // Greedy Strategy 1: Global Nearest Neighbor
  function greedyGlobal(priorityOrder: 'none' | 'desc' | 'asc'): number[][] {
    const assignments: number[][] = Array.from({ length: R }, () => []);
    const currentPos = new Int32Array(R);
    for (let r = 0; r < R; r++) currentPos[r] = r;
    
    const assigned = new Uint8Array(N);
    let assignedCount = 0;
    
    if (priorityOrder === 'none') {
      while (assignedCount < J) {
        let bestDist = Infinity;
        let bestR = -1;
        let bestJ = -1;
        
        for (let r = 0; r < R; r++) {
          const u = currentPos[r];
          for (let j = R; j < N; j++) {
            if (assigned[j]) continue;
            const d = getDist(u, j);
            if (d === -1) continue;
            if (d < bestDist) {
              bestDist = d;
              bestR = r;
              bestJ = j;
            } else if (d === bestDist) {
              if (bestJ === -1 || jobs[j - R].priority > jobs[bestJ - R].priority) {
                bestR = r;
                bestJ = j;
              }
            }
          }
        }
        
        if (bestJ === -1) {
          for (let j = R; j < N; j++) {
            if (!assigned[j]) {
              assignments[0].push(j);
              assigned[j] = 1;
              assignedCount++;
            }
          }
          break;
        }
        
        assignments[bestR].push(bestJ);
        currentPos[bestR] = bestJ;
        assigned[bestJ] = 1;
        assignedCount++;
      }
    } else {
      const sortedJobs = jobs.map((job, idx) => ({ job, nodeIdx: R + idx }));
      if (priorityOrder === 'desc') {
        sortedJobs.sort((a, b) => b.job.priority - a.job.priority || a.nodeIdx - b.nodeIdx);
      } else {
        sortedJobs.sort((a, b) => a.job.priority - b.job.priority || a.nodeIdx - b.nodeIdx);
      }
      
      for (const item of sortedJobs) {
        const j = item.nodeIdx;
        let bestDist = Infinity;
        let bestR = 0;
        for (let r = 0; r < R; r++) {
          const u = currentPos[r];
          const d = getDist(u, j);
          if (d !== -1 && d < bestDist) {
            bestDist = d;
            bestR = r;
          }
        }
        assignments[bestR].push(j);
        currentPos[bestR] = j;
        assigned[j] = 1;
        assignedCount++;
      }
    }
    
    return assignments;
  }

  // Greedy Strategy 2: Round Robin Nearest Neighbor
  function greedyRoundRobin(): number[][] {
    const assignments: number[][] = Array.from({ length: R }, () => []);
    const currentPos = new Int32Array(R);
    for (let r = 0; r < R; r++) currentPos[r] = r;
    
    const assigned = new Uint8Array(N);
    let assignedCount = 0;
    
    while (assignedCount < J) {
      let progress = false;
      for (let r = 0; r < R; r++) {
        const u = currentPos[r];
        let bestDist = Infinity;
        let bestJ = -1;
        for (let j = R; j < N; j++) {
          if (assigned[j]) continue;
          const d = getDist(u, j);
          if (d !== -1 && d < bestDist) {
            bestDist = d;
            bestJ = j;
          }
        }
        if (bestJ !== -1) {
          assignments[r].push(bestJ);
          currentPos[r] = bestJ;
          assigned[bestJ] = 1;
          assignedCount++;
          progress = true;
        }
      }
      if (!progress) {
        for (let j = R; j < N; j++) {
          if (!assigned[j]) {
            assignments[0].push(j);
            assigned[j] = 1;
            assignedCount++;
          }
        }
        break;
      }
    }
    return assignments;
  }

  // Greedy Strategy 3: Static Closest
  function greedyStaticClosest(): number[][] {
    const assignments: number[][] = Array.from({ length: R }, () => []);
    
    for (let j = R; j < N; j++) {
      let bestDist = Infinity;
      let bestR = 0;
      for (let r = 0; r < R; r++) {
        const d = getDist(r, j);
        if (d !== -1 && d < bestDist) {
          bestDist = d;
          bestR = r;
        }
      }
      assignments[bestR].push(j);
    }
    
    for (let r = 0; r < R; r++) {
      const path = assignments[r];
      if (path.length <= 1) continue;
      const sortedPath: number[] = [];
      let curr = r;
      const pathSet = new Set(path);
      while (pathSet.size > 0) {
        let bestDist = Infinity;
        let bestNext = -1;
        for (const j of pathSet) {
          const d = getDist(curr, j);
          if (d !== -1 && d < bestDist) {
            bestDist = d;
            bestNext = j;
          }
        }
        if (bestNext === -1) {
          for (const j of pathSet) {
            sortedPath.push(j);
          }
          break;
        }
        sortedPath.push(bestNext);
        pathSet.delete(bestNext);
        curr = bestNext;
      }
      assignments[r] = sortedPath;
    }
    return assignments;
  }

  // Evaluate candidate initializations
  const candidates = [
    greedyGlobal('none'),
    greedyGlobal('desc'),
    greedyGlobal('asc'),
    greedyRoundRobin(),
    greedyStaticClosest()
  ];

  let bestInitialDist = Infinity;
  let bestAssignments = candidates[0];

  for (const cand of candidates) {
    const d = calculateTotalDistance(cand);
    if (d < bestInitialDist) {
      bestInitialDist = d;
      bestAssignments = cand;
    }
  }

  // Local Search (2-opt, Relocate, Swap)
  function shuffle(array: number[]) {
    for (let i = array.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      const temp = array[i];
      array[i] = array[j];
      array[j] = temp;
    }
  }

  function runLocalSearch(initAssignments: number[][], maxTimeMs: number): number[][] {
    const startTime = Date.now();
    let current = initAssignments.map(path => [...path]);
    
    const pathDistances = new Int32Array(R);
    for (let r = 0; r < R; r++) {
      pathDistances[r] = pathDist(r, current[r]);
    }
    
    const robotIndices = Array.from({ length: R }, (_, i) => i);
    let improved = true;
    let timedOut = false;
    
    while (improved && !timedOut) {
      improved = false;
      shuffle(robotIndices);
      
      // 1. Intra-route 2-opt
      for (const r of robotIndices) {
        if (Date.now() - startTime > maxTimeMs) {
          timedOut = true;
          break;
        }
        const path = current[r];
        if (path.length < 3) continue;
        
        let bestRDist = pathDistances[r];
        let bestI = -1;
        let bestJ = -1;
        
        for (let i = 0; i < path.length - 1; i++) {
          for (let j = i + 1; j < path.length; j++) {
            const oldD = (i > 0 ? getDist(path[i - 1], path[i]) : getDist(r, path[0]))
                       + (j < path.length - 1 ? getDist(path[j], path[j + 1]) : 0);
            
            const newD = (i > 0 ? getDist(path[i - 1], path[j]) : getDist(r, path[j]))
                       + (j < path.length - 1 ? getDist(path[i], path[j + 1]) : 0);
            
            const newRDist = pathDistances[r] - oldD + newD;
            if (newRDist < bestRDist) {
              bestRDist = newRDist;
              bestI = i;
              bestJ = j;
            }
          }
        }
        
        if (bestI !== -1) {
          let left = bestI;
          let right = bestJ;
          while (left < right) {
            const temp = path[left];
            path[left] = path[right];
            path[right] = temp;
            left++;
            right--;
          }
          pathDistances[r] = bestRDist;
          improved = true;
        }
      }
      if (timedOut) break;
      if (improved) continue;
      
      // 2. Relocate a job
      for (const r1 of robotIndices) {
        if (Date.now() - startTime > maxTimeMs) {
          timedOut = true;
          break;
        }
        const p1 = current[r1];
        if (p1.length === 0) continue;
        
        const idxs1 = Array.from({ length: p1.length }, (_, i) => i);
        shuffle(idxs1);
        
        for (const idx1 of idxs1) {
          if (Date.now() - startTime > maxTimeMs) {
            timedOut = true;
            break;
          }
          const job = p1[idx1];
          
          const r2s = [...robotIndices];
          for (const r2 of r2s) {
            if (Date.now() - startTime > maxTimeMs) {
              timedOut = true;
              break;
            }
            const p2 = current[r2];
            
            if (r1 === r2) {
              if (p1.length < 2) continue;
              for (let idx2 = 0; idx2 <= p1.length; idx2++) {
                if (idx2 === idx1 || idx2 === idx1 + 1) continue;
                
                const newP = [...p1];
                newP.splice(idx1, 1);
                const insertIdx = idx2 > idx1 ? idx2 - 1 : idx2;
                newP.splice(insertIdx, 0, job);
                
                const newD = pathDist(r1, newP);
                if (newD < pathDistances[r1]) {
                  current[r1] = newP;
                  pathDistances[r1] = newD;
                  improved = true;
                  break;
                }
              }
            } else {
              for (let idx2 = 0; idx2 <= p2.length; idx2++) {
                const newP1 = [...p1];
                newP1.splice(idx1, 1);
                
                const newP2 = [...p2];
                newP2.splice(idx2, 0, job);
                
                const newD1 = pathDist(r1, newP1);
                const newD2 = pathDist(r2, newP2);
                
                if (newD1 !== Infinity && newD2 !== Infinity && (newD1 + newD2) < (pathDistances[r1] + pathDistances[r2])) {
                  current[r1] = newP1;
                  current[r2] = newP2;
                  pathDistances[r1] = newD1;
                  pathDistances[r2] = newD2;
                  improved = true;
                  break;
                }
              }
            }
            if (improved || timedOut) break;
          }
          if (improved || timedOut) break;
        }
        if (improved || timedOut) break;
      }
      if (timedOut) break;
      if (improved) continue;
      
      // 3. Swap two jobs
      for (const r1 of robotIndices) {
        if (Date.now() - startTime > maxTimeMs) {
          timedOut = true;
          break;
        }
        const p1 = current[r1];
        if (p1.length === 0) continue;
        
        for (let idx1 = 0; idx1 < p1.length; idx1++) {
          if (Date.now() - startTime > maxTimeMs) {
            timedOut = true;
            break;
          }
          
          for (const r2 of robotIndices) {
            if (r2 <= r1) continue;
            if (Date.now() - startTime > maxTimeMs) {
              timedOut = true;
              break;
            }
            const p2 = current[r2];
            if (p2.length === 0) continue;
            
            for (let idx2 = 0; idx2 < p2.length; idx2++) {
              const job1 = p1[idx1];
              const job2 = p2[idx2];
              
              const newP1 = [...p1];
              newP1[idx1] = job2;
              
              const newP2 = [...p2];
              newP2[idx2] = job1;
              
              const newD1 = pathDist(r1, newP1);
              const newD2 = pathDist(r2, newP2);
              
              if (newD1 !== Infinity && newD2 !== Infinity && (newD1 + newD2) < (pathDistances[r1] + pathDistances[r2])) {
                current[r1] = newP1;
                current[r2] = newP2;
                pathDistances[r1] = newD1;
                pathDistances[r2] = newD2;
                improved = true;
                break;
              }
            }
            if (improved || timedOut) break;
          }
          if (improved || timedOut) break;
        }
        if (improved || timedOut) break;
      }
    }
    return current;
  }

  // Budget remaining time for local search safely
  const maxLocalSearchTime = Math.min(2000, Math.max(100, 4500 - (Date.now() - programStartTime)));
  const optimizedAssignments = runLocalSearch(bestAssignments, maxLocalSearchTime);

  // Map to final output
  const maxRobotId = Math.max(...robots.map(r => r.id));
  const outputAssignments: number[][] = Array.from({ length: maxRobotId + 1 }, () => []);
  
  for (let r = 0; r < R; r++) {
    const robotId = robots[r].id;
    outputAssignments[robotId] = optimizedAssignments[r].map(nodeIdx => jobs[nodeIdx - R].id);
  }

  console.log(JSON.stringify({ assignments: outputAssignments }));
}

solve();
