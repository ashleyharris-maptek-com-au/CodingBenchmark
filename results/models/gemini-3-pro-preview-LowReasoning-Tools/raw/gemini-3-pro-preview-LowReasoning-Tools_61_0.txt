declare const require: any;
const fs = require('fs');

interface Package {
  id: number;
  x: number;
  y: number;
}

interface InputData {
  depot: [number, number];
  truckCount: number;
  packages: Package[];
}

function solve() {
  const input: string = fs.readFileSync(0, 'utf8');
  const data: InputData = JSON.parse(input);
  const depot = data.depot;
  const truckCount = data.truckCount;
  const packages = data.packages;
  const N = packages.length;

  if (N === 0) {
    console.log(JSON.stringify({ routes: [] }));
    return;
  }

  // 1. Sort packages by polar angle relative to the depot
  const packagesWithAngle = packages.map(p => {
    const dx = p.x - depot[0];
    const dy = p.y - depot[1];
    return {
      id: p.id,
      x: p.x,
      y: p.y,
      angle: Math.atan2(dy, dx)
    };
  });
  packagesWithAngle.sort((a, b) => a.angle - b.angle);

  // 2. Partition packages into K routes
  const K = Math.min(truckCount, N);
  const routes: number[][] = [];

  let maxIter = 5;
  if (N > 100000) {
    maxIter = 1;
  } else if (N > 20000) {
    maxIter = 2;
  }

  for (let t = 0; t < K; t++) {
    const startIdx = Math.floor((N * t) / K);
    const endIdx = Math.floor((N * (t + 1)) / K);
    const chunk = packagesWithAngle.slice(startIdx, endIdx);
    if (chunk.length === 0) continue;

    const route = solveTSPForChunk(chunk, depot, maxIter);
    routes.push(route);
  }

  console.log(JSON.stringify({ routes }));
}

function solveTSPForChunk(
  chunk: { id: number; x: number; y: number }[],
  depot: [number, number],
  maxIter: number
): number[] {
  const M = chunk.length;
  if (M === 1) {
    return [chunk[0].id];
  }

  if (M > 1000) {
    // Return the angular order directly for very large route sizes to prevent timeout
    return chunk.map(p => p.id);
  }

  // Flat arrays for fast V8 access
  const xs = new Float64Array(M);
  const ys = new Float64Array(M);
  for (let i = 0; i < M; i++) {
    xs[i] = chunk[i].x;
    ys[i] = chunk[i].y;
  }

  // Nearest Neighbor heuristic
  const visited = new Uint8Array(M);
  const routeIdx = new Int32Array(M);
  let curX = depot[0];
  let curY = depot[1];

  for (let step = 0; step < M; step++) {
    let bestIdx = -1;
    let bestDistSq = Infinity;
    for (let i = 0; i < M; i++) {
      if (visited[i] === 0) {
        const dx = xs[i] - curX;
        const dy = ys[i] - curY;
        const distSq = dx * dx + dy * dy;
        if (distSq < bestDistSq) {
          bestDistSq = distSq;
          bestIdx = i;
        }
      }
    }
    visited[bestIdx] = 1;
    routeIdx[step] = bestIdx;
    curX = xs[bestIdx];
    curY = ys[bestIdx];
  }

  // 2-opt local search
  const tour = new Int32Array(M + 2);
  tour[0] = -1; // -1 represents depot
  for (let i = 0; i < M; i++) {
    tour[i + 1] = routeIdx[i];
  }
  tour[M + 1] = -1;

  const txs = new Float64Array(M + 2);
  const tys = new Float64Array(M + 2);
  txs[0] = depot[0]; tys[0] = depot[1];
  txs[M + 1] = depot[0]; tys[M + 1] = depot[1];
  for (let i = 0; i < M; i++) {
    txs[i + 1] = xs[routeIdx[i]];
    tys[i + 1] = ys[routeIdx[i]];
  }

  let improved = true;
  let iter = 0;
  while (improved && iter < maxIter) {
    improved = false;
    for (let i = 0; i < M; i++) {
      const x_i = txs[i];
      const y_i = tys[i];
      const x_i1 = txs[i + 1];
      const y_i1 = tys[i + 1];
      const dx_i_i1 = x_i - x_i1;
      const dy_i_i1 = y_i - y_i1;
      const curDist1 = Math.sqrt(dx_i_i1 * dx_i_i1 + dy_i_i1 * dy_i_i1);

      for (let j = i + 2; j <= M; j++) {
        const x_j = txs[j];
        const y_j = tys[j];
        const x_j1 = txs[j + 1];
        const y_j1 = tys[j + 1];

        const dx_j_j1 = x_j - x_j1;
        const dy_j_j1 = y_j - y_j1;
        const curDist2 = Math.sqrt(dx_j_j1 * dx_j_j1 + dy_j_j1 * dy_j_j1);

        const dx_i_j = x_i - x_j;
        const dy_i_j = y_i - y_j;
        const newDist1 = Math.sqrt(dx_i_j * dx_i_j + dy_i_j * dy_i_j);

        const dx_i1_j1 = x_i1 - x_j1;
        const dy_i1_j1 = y_i1 - y_j1;
        const newDist2 = Math.sqrt(dx_i1_j1 * dx_i1_j1 + dy_i1_j1 * dy_i1_j1);

        const curDist = curDist1 + curDist2;
        const newDist = newDist1 + newDist2;

        if (newDist < curDist - 1e-9) {
          // Reverse segment from i+1 to j
          let l = i + 1;
          let r = j;
          while (l < r) {
            const tempIdx = tour[l]; tour[l] = tour[r]; tour[r] = tempIdx;
            const tempX = txs[l]; txs[l] = txs[r]; txs[r] = tempX;
            const tempY = tys[l]; tys[l] = tys[r]; tys[r] = tempY;
            l++;
            r--;
          }
          improved = true;
          break;
        }
      }
      if (improved) {
        break;
      }
    }
    iter++;
  }

  const finalRoute: number[] = [];
  for (let i = 1; i <= M; i++) {
    finalRoute.push(chunk[tour[i]].id);
  }
  return finalRoute;
}

solve();
