const fs = require('fs');

function solve() {
    let input;
    try {
        input = fs.readFileSync(0, 'utf8');
    } catch (e) {
        return;
    }
    if (!input || input.trim() === "") return;
    const data = JSON.parse(input);

    const machineCount = data.machineCount || 1;
    const tasks = data.tasks || [];
    const n = tasks.length;

    if (n === 0) {
        console.log(JSON.stringify({ schedule: [] }));
        return;
    }

    const idToIndex = new Map();
    const taskIds = [];
    for (let i = 0; i < n; i++) {
        const id = tasks[i].id;
        idToIndex.set(id, i);
        taskIds.push(id);
    }

    const duration = new Float64Array(n);
    const deadline = new Float64Array(n);
    const weight = new Float64Array(n);
    const adj = Array.from({ length: n }, () => []);
    const parents = Array.from({ length: n }, () => []);
    const inDegree = new Int32Array(n);

    for (let i = 0; i < n; i++) {
        const task = tasks[i];
        const u = idToIndex.get(task.id);
        duration[u] = task.duration || 0;
        deadline[u] = task.deadline !== undefined ? task.deadline : 1e9;
        weight[u] = task.weight !== undefined ? task.weight : 1;
    }

    for (let i = 0; i < n; i++) {
        const task = tasks[i];
        const u = idToIndex.get(task.id);
        const deps = task.deps || [];
        for (let j = 0; j < deps.length; j++) {
            const p = idToIndex.get(deps[j]);
            if (p !== undefined) {
                adj[p].push(u);
                parents[u].push(p);
                inDegree[u]++;
            }
        }
    }

    class MinHeap {
        constructor(compareFn) {
            this.data = [];
            this.compare = compareFn;
        }
        push(val) {
            this.data.push(val);
            this.up(this.data.length - 1);
        }
        pop() {
            if (this.data.length === 0) return null;
            const top = this.data[0];
            const bottom = this.data.pop();
            if (this.data.length > 0) {
                this.data[0] = bottom;
                this.down(0);
            }
            return top;
        }
        size() {
            return this.data.length;
        }
        up(i) {
            while (i > 0) {
                const p = (i - 1) >> 1;
                if (this.compare(this.data[i], this.data[p]) < 0) {
                    const tmp = this.data[i];
                    this.data[i] = this.data[p];
                    this.data[p] = tmp;
                    i = p;
                } else {
                    break;
                }
            }
        }
        down(i) {
            const len = this.data.length;
            while ((i << 1) + 1 < len) {
                let child = (i << 1) + 1;
                if (child + 1 < len && this.compare(this.data[child + 1], this.data[child]) < 0) {
                    child++;
                }
                if (this.compare(this.data[child], this.data[i]) < 0) {
                    const tmp = this.data[i];
                    this.data[i] = this.data[child];
                    this.data[child] = tmp;
                    i = child;
                } else {
                    break;
                }
            }
        }
    }

    function getPriorityOrder(priorityVal) {
        const heap = new MinHeap((a, b) => {
            if (priorityVal[a] !== priorityVal[b]) {
                return priorityVal[a] - priorityVal[b];
            }
            return a - b;
        });

        const tempInDegree = inDegree.slice();
        for (let i = 0; i < n; i++) {
            if (tempInDegree[i] === 0) {
                heap.push(i);
            }
        }

        const order = [];
        while (heap.size() > 0) {
            const u = heap.pop();
            order.push(u);
            const children = adj[u];
            for (let i = 0; i < children.length; i++) {
                const v = children[i];
                tempInDegree[v]--;
                if (tempInDegree[v] === 0) {
                    heap.push(v);
                }
            }
        }
        
        if (order.length < n) {
            const visited = new Uint8Array(n);
            for (let i = 0; i < order.length; i++) {
                visited[order[i]] = 1;
            }
            for (let i = 0; i < n; i++) {
                if (!visited[i]) {
                    order.push(i);
                }
            }
        }

        return order;
    }

    const compTime = new Float64Array(n);
    const startTime = new Float64Array(n);
    const assignedMachine = new Int32Array(n);
    const est = new Float64Array(n);
    const machineFreeTime = new Float64Array(machineCount);
    const machineId = new Int32Array(machineCount);

    function simulateBinaryReuse(order) {
        for (let m = 0; m < machineCount; m++) {
            machineFreeTime[m] = 0;
            machineId[m] = m;
        }
        est.fill(0);

        let makespan = 0;
        let totalWeightedLateness = 0;

        for (let i = 0; i < n; i++) {
            const u = order[i];
            const Ei = est[u];

            let chosenIdx = 0;
            if (machineFreeTime[0] < Ei) {
                let low = 0;
                let high = machineCount - 1;
                let ans = 0;
                while (low <= high) {
                    const mid = (low + high) >> 1;
                    if (machineFreeTime[mid] <= Ei) {
                        ans = mid;
                        low = mid + 1;
                    } else {
                        high = mid - 1;
                    }
                }
                chosenIdx = ans;
            }

            const mFreeTime = machineFreeTime[chosenIdx];
            const mId = machineId[chosenIdx];

            const S_u = Math.max(Ei, mFreeTime);
            const C_u = S_u + duration[u];

            startTime[u] = S_u;
            compTime[u] = C_u;
            assignedMachine[u] = mId;

            const newFreeTime = C_u;

            let low = 0;
            let high = machineCount - 1;
            let insertIdx = machineCount;
            while (low <= high) {
                const mid = (low + high) >> 1;
                if (machineFreeTime[mid] >= newFreeTime) {
                    insertIdx = mid;
                    high = mid - 1;
                } else {
                    low = mid + 1;
                }
            }

            if (insertIdx > chosenIdx) {
                const targetInsert = insertIdx - 1;
                for (let k = chosenIdx; k < targetInsert; k++) {
                    machineFreeTime[k] = machineFreeTime[k + 1];
                    machineId[k] = machineId[k + 1];
                }
                machineFreeTime[targetInsert] = newFreeTime;
                machineId[targetInsert] = mId;
            } else {
                for (let k = chosenIdx; k > insertIdx; k--) {
                    machineFreeTime[k] = machineFreeTime[k - 1];
                    machineId[k] = machineId[k - 1];
                }
                machineFreeTime[insertIdx] = newFreeTime;
                machineId[insertIdx] = mId;
            }

            if (C_u > makespan) {
                makespan = C_u;
            }

            const lateness = Math.max(0, C_u - deadline[u]);
            totalWeightedLateness += weight[u] * lateness;

            const children = adj[u];
            for (let j = 0; j < children.length; j++) {
                const v = children[j];
                if (C_u > est[v]) {
                    est[v] = C_u;
                }
            }
        }

        return makespan + totalWeightedLateness;
    }

    const baseOrder = getPriorityOrder(new Float64Array(n));
    
    const tempEst = new Float64Array(n);
    for (let i = 0; i < n; i++) {
        const u = baseOrder[i];
        const children = adj[u];
        for (let j = 0; j < children.length; j++) {
            const v = children[j];
            tempEst[v] = Math.max(tempEst[v], tempEst[u] + duration[u]);
        }
    }

    const lct = new Float64Array(n);
    for (let i = 0; i < n; i++) {
        lct[i] = deadline[i];
    }
    for (let i = n - 1; i >= 0; i--) {
        const u = baseOrder[i];
        const pList = parents[u];
        for (let j = 0; j < pList.length; j++) {
            const p = pList[j];
            lct[p] = Math.min(lct[p], lct[u] - duration[u]);
        }
    }

    const lst = new Float64Array(n);
    const slack = new Float64Array(n);
    for (let i = 0; i < n; i++) {
        lst[i] = lct[i] - duration[i];
        slack[i] = lst[i] - tempEst[i];
    }

    const wspt = new Float64Array(n);
    for (let i = 0; i < n; i++) {
        wspt[i] = -weight[i] / (duration[i] || 1);
    }

    const initialCandidates = [];
    initialCandidates.push(deadline);
    initialCandidates.push(lst);
    initialCandidates.push(slack);
    initialCandidates.push(wspt);

    const C_values = [1, 10, 100, 1000, 10000, 100000, 1000000];
    for (let i = 0; i < C_values.length; i++) {
        const C = C_values[i];
        const combined = new Float64Array(n);
        for (let j = 0; j < n; j++) {
            combined[j] = lst[j] + C * wspt[j];
        }
        initialCandidates.push(combined);
    }

    let bestCost = Infinity;
    let bestOrder = null;

    for (let i = 0; i < initialCandidates.length; i++) {
        const ord = getPriorityOrder(initialCandidates[i]);
        const cost = simulateBinaryReuse(ord);
        if (cost < bestCost) {
            bestCost = cost;
            bestOrder = ord;
        }
    }

    const currentOrder = bestOrder.slice();
    const pos = new Int32Array(n);
    for (let i = 0; i < n; i++) {
        pos[currentOrder[i]] = i;
    }

    let currentCost = bestCost;
    let bestGlobalCost = bestCost;
    const bestGlobalOrder = bestOrder.slice();

    let timeLimit = 1000;
    if (n <= 100) {
        timeLimit = 200;
    } else if (n > 1000) {
        timeLimit = 2000;
    }

    const saStartTime = Date.now();
    const T0 = Math.max(1.0, bestCost * 0.0005);
    const Tmin = 0.01;

    function shiftMove(u, fromIdx, toIdx) {
        if (fromIdx < toIdx) {
            for (let k = fromIdx; k < toIdx; k++) {
                currentOrder[k] = currentOrder[k + 1];
                pos[currentOrder[k]] = k;
            }
        } else {
            for (let k = fromIdx; k > toIdx; k--) {
                currentOrder[k] = currentOrder[k - 1];
                pos[currentOrder[k]] = k;
            }
        }
        currentOrder[toIdx] = u;
        pos[u] = toIdx;
    }

    while (true) {
        const elapsed = Date.now() - saStartTime;
        if (elapsed >= timeLimit) break;

        const progress = elapsed / timeLimit;
        const T = T0 * Math.pow(Tmin / T0, progress);

        const randMove = Math.random();
        if (randMove < 0.8) {
            let u;
            let targetLeft = false;
            
            const lateTasks = [];
            for (let i = 0; i < n; i++) {
                if (compTime[i] > deadline[i]) {
                    lateTasks.push(i);
                }
            }

            if (lateTasks.length > 0 && Math.random() < 0.7) {
                u = lateTasks[Math.floor(Math.random() * lateTasks.length)];
                targetLeft = true;
            } else {
                u = Math.floor(Math.random() * n);
            }

            const fromIdx = pos[u];

            let maxParentIdx = -1;
            const pList = parents[u];
            for (let j = 0; j < pList.length; j++) {
                const pIdx = pos[pList[j]];
                if (pIdx > maxParentIdx) maxParentIdx = pIdx;
            }
            let minChildIdx = n;
            const cList = adj[u];
            for (let j = 0; j < cList.length; j++) {
                const cIdx = pos[cList[j]];
                if (cIdx < minChildIdx) minChildIdx = cIdx;
            }

            let toIdx = fromIdx;
            if (targetLeft) {
                const leftBound = maxParentIdx + 1;
                const rightBound = fromIdx - 1;
                if (leftBound <= rightBound) {
                    toIdx = Math.floor(Math.random() * (rightBound - leftBound + 1)) + leftBound;
                } else {
                    if (maxParentIdx + 1 < minChildIdx - 1) {
                        while (toIdx === fromIdx) {
                            toIdx = Math.floor(Math.random() * (minChildIdx - maxParentIdx - 1)) + maxParentIdx + 1;
                        }
                    }
                }
            } else {
                if (maxParentIdx + 1 < minChildIdx - 1) {
                    while (toIdx === fromIdx) {
                        toIdx = Math.floor(Math.random() * (minChildIdx - maxParentIdx - 1)) + maxParentIdx + 1;
                    }
                }
            }

            if (toIdx === fromIdx) continue;

            shiftMove(u, fromIdx, toIdx);

            const nextCost = simulateBinaryReuse(currentOrder);
            const delta = nextCost - currentCost;

            let accept = false;
            if (delta <= 0) {
                accept = true;
            } else {
                if (Math.random() < Math.exp(-delta / T)) {
                    accept = true;
                }
            }

            if (accept) {
                currentCost = nextCost;
                if (currentCost < bestGlobalCost) {
                    bestGlobalCost = currentCost;
                    for (let j = 0; j < n; j++) bestGlobalOrder[j] = currentOrder[j];
                }
            } else {
                shiftMove(u, toIdx, fromIdx);
                simulateBinaryReuse(currentOrder);
            }
        } else {
            if (n < 2) continue;
            const i = Math.floor(Math.random() * (n - 1));
            const u = currentOrder[i];
            const v = currentOrder[i + 1];

            let isParent = false;
            const uAdj = adj[u];
            for (let j = 0; j < uAdj.length; j++) {
                if (uAdj[j] === v) {
                    isParent = true;
                    break;
                }
            }

            if (!isParent) {
                currentOrder[i] = v;
                currentOrder[i + 1] = u;
                pos[v] = i;
                pos[u] = i + 1;

                const nextCost = simulateBinaryReuse(currentOrder);
                const delta = nextCost - currentCost;

                let accept = false;
                if (delta <= 0) {
                    accept = true;
                } else {
                    if (Math.random() < Math.exp(-delta / T)) {
                        accept = true;
                    }
                }

                if (accept) {
                    currentCost = nextCost;
                    if (currentCost < bestGlobalCost) {
                        bestGlobalCost = currentCost;
                        for (let j = 0; j < n; j++) bestGlobalOrder[j] = currentOrder[j];
                    }
                } else {
                    currentOrder[i] = u;
                    currentOrder[i + 1] = v;
                    pos[u] = i;
                    pos[v] = i + 1;
                    simulateBinaryReuse(currentOrder);
                }
            }
        }
    }

    simulateBinaryReuse(bestGlobalOrder);

    const output = {
        schedule: []
    };
    for (let i = 0; i < n; i++) {
        const taskId = taskIds[i];
        output.schedule.push({
            id: taskId,
            start: startTime[i],
            machine: assignedMachine[i]
        });
    }

    console.log(JSON.stringify(output));
}

solve();
