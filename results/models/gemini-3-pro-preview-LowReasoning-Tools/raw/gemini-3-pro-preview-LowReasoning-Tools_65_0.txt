declare const require: any;
const fs = require('fs');

interface Task {
    id: string | number;
    duration: number;
    deps: (string | number)[];
    deadline: number;
    weight: number;
}

interface ScheduleItem {
    id: string | number;
    start: number;
    machine: number;
}

class MinHeap<T> {
    private data: T[] = [];
    private compare: (a: T, b: T) => number;
    constructor(compare: (a: T, b: T) => number) {
        this.compare = compare;
    }
    push(val: T) {
        this.data.push(val);
        this.up(this.data.length - 1);
    }
    pop(): T | undefined {
        if (this.data.length === 0) return undefined;
        const top = this.data[0];
        const bottom = this.data.pop()!;
        if (this.data.length > 0) {
            this.data[0] = bottom;
            this.down(0);
        }
        return top;
    }
    peek(): T | undefined {
        return this.data[0];
    }
    size(): number {
        return this.data.length;
    }
    private up(i: number) {
        while (i > 0) {
            const p = (i - 1) >> 1;
            if (this.compare(this.data[i], this.data[p]) < 0) {
                const tmp = this.data[i];
                this.data[i] = this.data[p];
                this.data[p] = tmp;
                i = p;
            } else {
                break;
            }
        }
    }
    private down(i: number) {
        const len = this.data.length;
        while ((i << 1) + 1 < len) {
            let left = (i << 1) + 1;
            let right = left + 1;
            let best = i;
            if (this.compare(this.data[left], this.data[best]) < 0) best = left;
            if (right < len && this.compare(this.data[right], this.data[best]) < 0) best = right;
            if (best !== i) {
                const tmp = this.data[i];
                this.data[i] = this.data[best];
                this.data[best] = tmp;
                i = best;
            } else {
                break;
            }
        }
    }
}

function solve(inputJson: string): string {
    const data = JSON.parse(inputJson);
    const machineCount: number = data.machineCount;
    const taskList: Task[] = data.tasks;
    const N = taskList.length;

    // Map task ID to 0..N-1 index
    const idToIndex = new Map<string | number, number>();
    const indexToId: (string | number)[] = new Array(N);
    for (let i = 0; i < N; i++) {
        const id = taskList[i].id;
        idToIndex.set(id, i);
        indexToId[i] = id;
    }

    // Flat arrays for task properties
    const durations = new Int32Array(N);
    const deadlines = new Float64Array(N);
    const weights = new Float64Array(N);
    for (let i = 0; i < N; i++) {
        durations[i] = taskList[i].duration;
        deadlines[i] = taskList[i].deadline;
        weights[i] = taskList[i].weight;
    }

    // Build adjacency lists using indices
    const predecessors: number[][] = new Array(N);
    const successors: number[][] = new Array(N);
    for (let i = 0; i < N; i++) {
        predecessors[i] = [];
        successors[i] = [];
    }

    const inDegreeBase = new Int32Array(N);
    for (let i = 0; i < N; i++) {
        let validDepsCount = 0;
        for (const depId of taskList[i].deps) {
            const depIdx = idToIndex.get(depId);
            if (depIdx !== undefined) {
                predecessors[i].push(depIdx);
                successors[depIdx].push(i);
                validDepsCount++;
            }
        }
        inDegreeBase[i] = validDepsCount;
    }

    // Precalculate static metrics using iterative topological sort (to avoid stack overflow)
    const cpMap = new Float64Array(N);
    const outDegree = new Int32Array(N);
    const maxSuccCP = new Float64Array(N);

    for (let i = 0; i < N; i++) {
        outDegree[i] = successors[i].length;
    }

    const queueCP = new Int32Array(N);
    let qCPHead = 0;
    let qCPTail = 0;
    for (let i = 0; i < N; i++) {
        if (outDegree[i] === 0) {
            cpMap[i] = durations[i];
            queueCP[qCPTail++] = i;
        }
    }

    while (qCPHead < qCPTail) {
        const curr = queueCP[qCPHead++];
        const currCP = cpMap[curr];
        const preds = predecessors[curr];
        for (let j = 0; j < preds.length; j++) {
            const P = preds[j];
            outDegree[P]--;
            if (currCP > maxSuccCP[P]) {
                maxSuccCP[P] = currCP;
            }
            if (outDegree[P] === 0) {
                cpMap[P] = durations[P] + maxSuccCP[P];
                queueCP[qCPTail++] = P;
            }
        }
    }

    // Handle unreachable/cyclic nodes
    for (let i = 0; i < N; i++) {
        if (cpMap[i] === 0 && durations[i] > 0) {
            cpMap[i] = durations[i];
        }
    }

    // Compute descendant metrics (exact for small N, fallback to immediate successors for large N)
    const descDurMap = new Float64Array(N);
    const descCountMap = new Int32Array(N);

    if (N <= 2000) {
        const visited = new Uint8Array(N);
        const q = new Int32Array(N);
        for (let i = 0; i < N; i++) {
            visited.fill(0);
            visited[i] = 1;
            q[0] = i;
            let qHead = 0;
            let qTail = 1;
            let totalDur = 0;
            while (qHead < qTail) {
                const curr = q[qHead++];
                totalDur += durations[curr];
                const succs = successors[curr];
                for (let j = 0; j < succs.length; j++) {
                    const S = succs[j];
                    if (visited[S] === 0) {
                        visited[S] = 1;
                        q[qTail++] = S;
                    }
                }
            }
            descDurMap[i] = totalDur;
            descCountMap[i] = qTail - 1;
        }
    } else {
        for (let i = 0; i < N; i++) {
            let totalDur = 0;
            const succs = successors[i];
            for (let j = 0; j < succs.length; j++) {
                totalDur += durations[succs[j]];
            }
            descDurMap[i] = totalDur;
            descCountMap[i] = succs.length;
        }
    }

    // Preallocate reusable simulation structures
    const inDegree = new Int32Array(N);
    const completed = new Float64Array(N);

    function runSimulation(priorityArray: Float64Array): { schedule: ScheduleItem[]; cost: number } {
        inDegree.set(inDegreeBase);
        completed.fill(-1);

        const RArray = new Float64Array(N);
        const taskMachine = new Int32Array(N);

        const availableReady = new MinHeap<number>((a, b) => priorityArray[b] - priorityArray[a]);
        const pendingReady = new MinHeap<number>((a, b) => RArray[a] - RArray[b]);
        const runningTasks = new MinHeap<number>((a, b) => completed[a] - completed[b]);

        const idleMachines: number[] = [];
        for (let m = 0; m < machineCount; m++) {
            idleMachines.push(m);
        }

        for (let i = 0; i < N; i++) {
            if (inDegree[i] === 0) {
                RArray[i] = 0;
                availableReady.push(i);
            }
        }

        let time = 0;
        const schedule: ScheduleItem[] = [];

        while (true) {
            while (pendingReady.size() > 0 && RArray[pendingReady.peek()!] <= time) {
                availableReady.push(pendingReady.pop()!);
            }

            while (idleMachines.length > 0 && availableReady.size() > 0) {
                const idx = availableReady.pop()!;
                const machine = idleMachines.pop()!;
                taskMachine[idx] = machine;
                completed[idx] = time + durations[idx];
                schedule.push({ id: indexToId[idx], start: time, machine });
                runningTasks.push(idx);
            }

            let nextTime = Infinity;
            if (runningTasks.size() > 0) {
                nextTime = Math.min(nextTime, completed[runningTasks.peek()!]);
            }
            if (idleMachines.length > 0 && pendingReady.size() > 0) {
                nextTime = Math.min(nextTime, RArray[pendingReady.peek()!]);
            }

            if (nextTime === Infinity) {
                break;
            }

            time = nextTime;

            while (runningTasks.size() > 0 && completed[runningTasks.peek()!] === time) {
                const idx = runningTasks.pop()!;
                idleMachines.push(taskMachine[idx]);

                const succs = successors[idx];
                for (let j = 0; j < succs.length; j++) {
                    const S = succs[j];
                    inDegree[S]--;
                    if (inDegree[S] === 0) {
                        let maxDepCompleted = 0;
                        const preds = predecessors[S];
                        for (let k = 0; k < preds.length; k++) {
                            const dep = preds[k];
                            if (completed[dep] > maxDepCompleted) {
                                maxDepCompleted = completed[dep];
                            }
                        }
                        RArray[S] = maxDepCompleted;
                        if (maxDepCompleted <= time) {
                            availableReady.push(S);
                        } else {
                            pendingReady.push(S);
                        }
                    }
                }
            }
        }

        let makespan = 0;
        let weightedLateness = 0;
        for (let i = 0; i < N; i++) {
            const comp = completed[i];
            if (comp > 0) {
                if (comp > makespan) makespan = comp;
                const lateness = Math.max(0, comp - deadlines[i]);
                weightedLateness += weights[i] * lateness;
            }
        }

        return { schedule, cost: makespan + weightedLateness };
    }

    // Build the heuristic arrays
    const heuristicGetters: { name: string; getVal: (i: number) => number; fast: boolean }[] = [
        { name: 'CP', getVal: (i) => cpMap[i], fast: true },
        { name: 'EDF', getVal: (i) => -deadlines[i], fast: true },
        { name: 'Weight', getVal: (i) => weights[i], fast: false },
        { name: 'WSPT', getVal: (i) => weights[i] / (durations[i] || 1), fast: true },
        { name: 'SPT', getVal: (i) => -durations[i], fast: false },
        { name: 'LPT', getVal: (i) => durations[i], fast: false },
        { name: 'DescDur', getVal: (i) => descDurMap[i], fast: false },
        { name: 'DescCount', getVal: (i) => descCountMap[i], fast: false },
        { name: 'Slack', getVal: (i) => -(deadlines[i] - durations[i]), fast: true },
        { name: 'Composite1', getVal: (i) => 10 * cpMap[i] - deadlines[i], fast: false },
        { name: 'Composite2', getVal: (i) => 10 * cpMap[i] + 10 * weights[i] - deadlines[i], fast: true },
        { name: 'Composite3', getVal: (i) => weights[i] * cpMap[i] - deadlines[i], fast: true },
        { name: 'Composite4', getVal: (i) => weights[i] / (deadlines[i] - durations[i] + 1), fast: false },
        { name: 'Composite5', getVal: (i) => -deadlines[i] / (weights[i] + 1), fast: false },
        { name: 'Composite6', getVal: (i) => cpMap[i] / (deadlines[i] + 1), fast: false },
        { name: 'Composite7', getVal: (i) => descDurMap[i] / (deadlines[i] + 1), fast: false },
        { name: 'Composite8', getVal: (i) => weights[i] * descDurMap[i] - deadlines[i], fast: false },
        { name: 'Composite9', getVal: (i) => weights[i] * descCountMap[i] - deadlines[i], fast: false },
        { name: 'Composite10', getVal: (i) => cpMap[i] - deadlines[i] / (weights[i] + 1), fast: true },
        { name: 'Composite11', getVal: (i) => (descDurMap[i] + durations[i]) / (deadlines[i] + 1), fast: false },
        { name: 'Composite12', getVal: (i) => -deadlines[i] - durations[i], fast: false },
        { name: 'Composite13', getVal: (i) => -deadlines[i] + durations[i], fast: false },
        { name: 'Composite14', getVal: (i) => weights[i] * (cpMap[i] + descDurMap[i]) - deadlines[i], fast: true },
        { name: 'Composite15', getVal: (i) => cpMap[i] - 2 * deadlines[i] + 5 * weights[i], fast: false },
    ];

    const activeHeuristics = N <= 20000 ? heuristicGetters : heuristicGetters.filter(h => h.fast);

    const startTime = Date.now();
    let bestSchedule: ScheduleItem[] = [];
    let bestCost = Infinity;

    const heuristicResults: { priorityArray: Float64Array; cost: number }[] = [];

    for (const h of activeHeuristics) {
        const priorityArray = new Float64Array(N);
        for (let i = 0; i < N; i++) {
            priorityArray[i] = h.getVal(i);
        }
        const res = runSimulation(priorityArray);
        heuristicResults.push({ priorityArray, cost: res.cost });

        if (res.cost < bestCost) {
            bestCost = res.cost;
            bestSchedule = res.schedule;
        }
    }

    const deterministicDuration = Date.now() - startTime;
    const TIME_LIMIT = Math.min(1200, Math.max(deterministicDuration * 2.5, 400));

    if (TIME_LIMIT > 0 && heuristicResults.length > 0) {
        heuristicResults.sort((a, b) => a.cost - b.cost);
        const topCandidates = heuristicResults.slice(0, Math.min(5, heuristicResults.length));
        const perturbedPriority = new Float64Array(N);

        while (Date.now() - startTime < TIME_LIMIT) {
            const cand = topCandidates[Math.floor(Math.random() * topCandidates.length)];
            const noiseScale = 0.2;

            for (let i = 0; i < N; i++) {
                const base = cand.priorityArray[i];
                const mult = 1 + (Math.random() - 0.5) * noiseScale;
                const add = (Math.random() - 0.5) * 0.1;
                perturbedPriority[i] = base * mult + add;
            }

            const res = runSimulation(perturbedPriority);
            if (res.cost < bestCost) {
                bestCost = res.cost;
                bestSchedule = res.schedule;
            }
        }
    }

    // Safety fallback: schedule any tasks that did not run (e.g., due to cycles)
    if (bestSchedule.length < N) {
        const scheduledIds = new Set(bestSchedule.map(x => x.id));
        let maxEndTime = 0;
        for (const item of bestSchedule) {
            const t = taskList[idToIndex.get(item.id)!];
            maxEndTime = Math.max(maxEndTime, item.start + t.duration);
        }
        let machine = 0;
        for (let i = 0; i < N; i++) {
            const t = taskList[i];
            if (!scheduledIds.has(t.id)) {
                bestSchedule.push({ id: t.id, start: maxEndTime, machine });
                maxEndTime += t.duration;
            }
        }
    }

    return JSON.stringify({ schedule: bestSchedule });
}

// Read input from stdin and solve
const input = fs.readFileSync(0, 'utf8');
const output = solve(input);
process.stdout.write(output);
