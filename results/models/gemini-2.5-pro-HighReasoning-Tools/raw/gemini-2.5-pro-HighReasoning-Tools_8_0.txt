use std::collections::VecDeque;
use std::io::{self, BufRead};

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
struct Point {
    x: usize,
    y: usize,
}

fn main() {
    // Read the maze from stdin
    let (maze, start, end) = read_input();

    // Solve the maze
    if let Some(path) = solve_maze(&maze, start, end) {
        // Print the path
        println!("{}", path.len());
        for point in path {
            println!("{} {}", point.x, point.y);
        }
    } else {
        // No path found
        println!("0");
    }
}

/// Reads the maze from stdin and returns the maze grid, start point, and end point.
fn read_input() -> (Vec<Vec<char>>, Point, Point) {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Read maze dimensions
    let first_line = lines.next().unwrap().unwrap();
    let mut parts = first_line.split_whitespace();
    let height: usize = parts.next().unwrap().parse().unwrap();
    let width: usize = parts.next().unwrap().parse().unwrap();

    let mut maze = Vec::with_capacity(height);
    let mut start = Point { x: 0, y: 0 };
    let mut end = Point { x: 0, y: 0 };

    // Read maze grid
    for y in 0..height {
        let row_str = lines.next().unwrap().unwrap();
        let row: Vec<char> = row_str.chars().collect();
        for x in 0..width {
            match row[x] {
                'A' => start = Point { x, y },
                'B' => end = Point { x, y },
                _ => {}
            }
        }
        maze.push(row);
    }

    (maze, start, end)
}

/// Solves the maze using Breadth-First Search (BFS).
fn solve_maze(maze: &Vec<Vec<char>>, start: Point, end: Point) -> Option<Vec<Point>> {
    let height = maze.len();
    let width = maze[0].len();

    // A 2D vector to keep track of visited cells.
    let mut visited = vec![vec![false; width]; height];

    // A queue for the BFS, storing the points to visit.
    let mut queue: VecDeque<Point> = VecDeque::new();

    // A 2D vector to store the predecessor of each cell.
    let mut predecessor = vec![vec![None; width]; height];

    // Start BFS from the start point.
    queue.push_back(start);
    visited[start.y][start.x] = true;

    while let Some(current) = queue.pop_front() {
        // If we reached the end, reconstruct and return the path.
        if current == end {
            return Some(reconstruct_path(start, end, &predecessor));
        }

        // Explore neighbors.
        for_each_neighbor(current, height, width, |neighbor| {
            if !visited[neighbor.y][neighbor.x] && maze[neighbor.y][neighbor.x] != '#' {
                visited[neighbor.y][neighbor.x] = true;
                predecessor[neighbor.y][neighbor.x] = Some(current);
                queue.push_back(neighbor);
            }
        });
    }

    // No path found.
    None
}

/// Reconstructs the path from the end to the start using the predecessor map.
fn reconstruct_path(start: Point, end: Point, predecessor: &Vec<Vec<Option<Point>>>) -> Vec<Point> {
    let mut path = Vec::new();
    let mut current = end;

    while current != start {
        path.push(current);
        current = predecessor[current.y][current.x].unwrap();
    }
    path.push(start);
    path.reverse();
    path
}

/// Calls a closure for each valid neighbor of a given point.
fn for_each_neighbor<F>(point: Point, height: usize, width: usize, mut f: F)
where
    F: FnMut(Point),
{
    let (x, y) = (point.x, point.y);

    // Up
    if y > 0 {
        f(Point { x, y: y - 1 });
    }
    // Down
    if y < height - 1 {
        f(Point { x, y: y + 1 });
    }
    // Left
    if x > 0 {
        f(Point { x: x - 1, y });
    }
    // Right
    if x < width - 1 {
        f(Point { x: x + 1, y });
    }
}
