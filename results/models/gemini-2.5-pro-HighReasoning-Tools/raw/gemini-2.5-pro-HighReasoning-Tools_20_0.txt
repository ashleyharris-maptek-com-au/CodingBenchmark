use std::io::{self, BufRead};
use std::collections::HashMap;

#[derive(Debug, Clone, Copy)]
struct Point {
    x: f64,
    y: f64,
    z: f64,
}

#[derive(Debug, Clone)]
struct Sample {
    depth: f64,
    properties: Vec<f64>,
    position: Point,
    hole_id: i32,
    sample_idx: usize,
}

#[derive(Debug)]
struct SuspectEntry {
    hole_id: i32,
    sample_idx: usize,
    property_name: String,
    confidence: f64,
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines().map(|l| l.unwrap());

    let first_line = lines.next().unwrap();
    let mut parts = first_line.split_whitespace();
    let num_holes: i32 = parts.next().unwrap().parse().unwrap();
    let num_properties: usize = parts.next().unwrap().parse().unwrap();

    let second_line = lines.next().unwrap();
    let property_names: Vec<String> = second_line.split_whitespace().map(|s| s.to_string()).collect();

    let mut all_samples: Vec<Sample> = Vec::new();
    
    for _ in 0..num_holes {
        let hole_line = lines.next().unwrap();
        let mut hole_parts = hole_line.split_whitespace();
        let id: i32 = hole_parts.next().unwrap().parse().unwrap();
        let start_x: f64 = hole_parts.next().unwrap().parse().unwrap();
        let start_y: f64 = hole_parts.next().unwrap().parse().unwrap();
        let start_z: f64 = hole_parts.next().unwrap().parse().unwrap();
        let dir_x: f64 = hole_parts.next().unwrap().parse().unwrap();
        let dir_y: f64 = hole_parts.next().unwrap().parse().unwrap();
        let dir_z: f64 = hole_parts.next().unwrap().parse().unwrap();
        let _length: f64 = hole_parts.next().unwrap().parse().unwrap();
        let num_samples: i32 = hole_parts.next().unwrap().parse().unwrap();

        let start = Point { x: start_x, y: start_y, z: start_z };
        let dir = Point { x: dir_x, y: dir_y, z: dir_z };

        for i in 0..num_samples {
            let sample_line = lines.next().unwrap();
            let mut sample_parts = sample_line.split_whitespace();
            let depth: f64 = sample_parts.next().unwrap().parse().unwrap();
            let properties: Vec<f64> = (0..num_properties)
                .map(|_| sample_parts.next().unwrap().parse().unwrap())
                .collect();
            let position = Point {
                x: start.x + depth * dir.x,
                y: start.y + depth * dir.y,
                z: start.z + depth * dir.z,
            };

            let sample = Sample {
                depth,
                properties,
                position,
                hole_id: id,
                sample_idx: i as usize,
            };
            all_samples.push(sample);
        }
    }

    let total_samples = all_samples.len();
    
    let grid_size = if total_samples > 1_000_000 {
        200.0
    } else if total_samples > 100_000 {
        100.0
    } else {
        50.0
    };

    let mut grid: HashMap<(i32, i32, i32), Vec<usize>> = HashMap::new();
    for (i, sample) in all_samples.iter().enumerate() {
        let grid_x = (sample.position.x / grid_size) as i32;
        let grid_y = (sample.position.y / grid_size) as i32;
        let grid_z = (sample.position.z / grid_size) as i32;
        grid.entry((grid_x, grid_y, grid_z)).or_default().push(i);
    }
    
    let mut suspect_entries: Vec<SuspectEntry> = Vec::new();

    for (sample_idx, sample) in all_samples.iter().enumerate() {
        let grid_x = (sample.position.x / grid_size) as i32;
        let grid_y = (sample.position.y / grid_size) as i32;
        let grid_z = (sample.position.z / grid_size) as i32;

        let mut neighbors = Vec::new();
        for dx in -1..=1 {
            for dy in -1..=1 {
                for dz in -1..=1 {
                    if let Some(cell_samples) = grid.get(&(grid_x + dx, grid_y + dy, grid_z + dz)) {
                        for &neighbor_idx in cell_samples {
                            if sample_idx != neighbor_idx {
                                let neighbor = &all_samples[neighbor_idx];
                                let dist_sq = (sample.position.x - neighbor.position.x).powi(2) +
                                              (sample.position.y - neighbor.position.y).powi(2) +
                                              (sample.position.z - neighbor.position.z).powi(2);
                                if dist_sq < (grid_size * 2.0).powi(2) {
                                    neighbors.push(neighbor);
                                }
                            }
                        }
                    }
                }
            }
        }

        if neighbors.len() < 5 { continue; }

        for prop_idx in 0..num_properties {
            let neighbor_values: Vec<f64> = neighbors.iter().map(|s| s.properties[prop_idx]).collect();
            
            let mean = neighbor_values.iter().sum::<f64>() / neighbor_values.len() as f64;
            let std_dev = (neighbor_values.iter().map(|v| (v - mean).powi(2)).sum::<f64>() / neighbor_values.len() as f64).sqrt();

            if std_dev > 1e-6 {
                let diff = (sample.properties[prop_idx] - mean).abs();
                if diff > 3.0 * std_dev {
                    suspect_entries.push(SuspectEntry {
                        hole_id: sample.hole_id,
                        sample_idx: sample.sample_idx,
                        property_name: property_names[prop_idx].clone(),
                        confidence: diff / std_dev,
                    });
                }
            }
        }
    }
    
    suspect_entries.sort_by(|a, b| b.confidence.partial_cmp(&a.confidence).unwrap());
    
    println!("{}", suspect_entries.len());
    for entry in suspect_entries {
        println!("{} {} {} {:.2}", entry.hole_id, entry.sample_idx, entry.property_name, entry.confidence);
    }
}
