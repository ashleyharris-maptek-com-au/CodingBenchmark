using System;
using System.Collections.Generic;
using System.Linq;

public class BinPackingSolver
{
    public static void Main(string[] args)
    {
        var (polyhedron, container) = ParseInput();
        var packedPolyhedra = new List<PackedPolyhedron>();

        double step = 1;
        var volume = container.Width * container.Height * container.Depth;
        if (volume < 1000)
        {
            step = 0.5;
        } else if (volume > 100000)
        {
            step = 5;
        }


        var rotations = new List<Quaternion>
        {
            new Quaternion(1, 0, 0, 0),
            new Quaternion(0.707, 0.707, 0, 0),
            new Quaternion(0.707, 0, 0.707, 0),
            new Quaternion(0.707, 0, 0, 0.707)
        };

        bool added = true;
        while(added)
        {
            added = false;
            for (double z = 0; z < container.Depth; z += step)
            {
                for (double y = 0; y < container.Height; y += step)
                {
                    for (double x = 0; x < container.Width; x += step)
                    {
                        foreach (var rotation in rotations)
                        {
                            var placement = new PackedPolyhedron
                            {
                                Polyhedron = polyhedron,
                                Position = new Vector3(x, y, z),
                                Rotation = rotation
                            };

                            if (CanPlace(placement, container, packedPolyhedra))
                            {
                                packedPolyhedra.Add(placement);
                                added = true;
                                goto next_item;
                            }
                        }
                    }
                }
            }
            next_item:;
        }


        Console.WriteLine(packedPolyhedra.Count);
        foreach (var packed in packedPolyhedra)
        {
            Console.WriteLine($"{packed.Position.X} {packed.Position.Y} {packed.Position.Z} {packed.Rotation.W} {packed.Rotation.X} {packed.Rotation.Y} {packed.Rotation.Z}");
        }
    }

    public static bool CanPlace(PackedPolyhedron placement, Container container, List<PackedPolyhedron> existingPolyhedra)
    {
        var aabb = placement.Polyhedron.GetAABB(placement.Rotation);
        if (placement.Position.X + aabb.Min.X < 0 || placement.Position.X + aabb.Max.X > container.Width ||
            placement.Position.Y + aabb.Min.Y < 0 || placement.Position.Y + aabb.Max.Y > container.Height ||
            placement.Position.Z + aabb.Min.Z < 0 || placement.Position.Z + aabb.Max.Z > container.Depth)
        {
            return false;
        }

        foreach (var existing in existingPolyhedra)
        {
            if (Intersects(placement, existing))
            {
                return false;
            }
        }
        return true;
    }

    public static bool Intersects(PackedPolyhedron p1, PackedPolyhedron p2)
    {
        var aabb1 = p1.Polyhedron.GetAABB(p1.Rotation);
        var aabb2 = p2.Polyhedron.GetAABB(p2.Rotation);
        return (p1.Position.X + aabb1.Min.X <= p2.Position.X + aabb2.Max.X && p1.Position.X + aabb1.Max.X >= p2.Position.X + aabb2.Min.X) &&
               (p1.Position.Y + aabb1.Min.Y <= p2.Position.Y + aabb2.Max.Y && p1.Position.Y + aabb1.Max.Y >= p2.Position.Y + aabb2.Min.Y) &&
               (p1.Position.Z + aabb1.Min.Z <= p2.Position.Z + aabb2.Max.Z && p1.Position.Z + aabb1.Max.Z >= p2.Position.Z + aabb2.Min.Z);
    }

    public static (Polyhedron, Container) ParseInput()
    {
        var polyhedron = new Polyhedron();
        var line = Console.ReadLine().Split(' ');
        var vertexCount = int.Parse(line[0]);
        var faceCount = int.Parse(line[1]);

        for (int i = 0; i < vertexCount; i++)
        {
            line = Console.ReadLine().Split(' ');
            polyhedron.Vertices.Add(new Vector3(double.Parse(line[0]), double.Parse(line[1]), double.Parse(line[2])));
        }

        for (int i = 0; i < faceCount; i++)
        {
            line = Console.ReadLine().Split(' ');
            polyhedron.Faces.Add(line.Select(int.Parse).ToArray());
        }

        line = Console.ReadLine().Split(' ');
        var container = new Container(double.Parse(line[0]), double.Parse(line[1]), double.Parse(line[2]));
        return (polyhedron, container);
    }

    public class Vector3
    {
        public double X, Y, Z;
        public Vector3(double x, double y, double z) { X = x; Y = y; Z = z; }
    }

    public class Quaternion
    {
        public double W, X, Y, Z;
        public Quaternion(double w, double x, double y, double z) { W = w; X = x; Y = y; Z = z; }
    }

    public class Polyhedron
    {
        public List<Vector3> Vertices = new List<Vector3>();
        public List<int[]> Faces = new List<int[]>();

        public AABB GetAABB(Quaternion rotation)
        {
            var min = new Vector3(double.MaxValue, double.MaxValue, double.MaxValue);
            var max = new Vector3(double.MinValue, double.MinValue, double.MinValue);

            foreach (var vertex in Vertices)
            {
                var rotatedVertex = Rotate(vertex, rotation);
                min.X = Math.Min(min.X, rotatedVertex.X);
                min.Y = Math.Min(min.Y, rotatedVertex.Y);
                min.Z = Math.Min(min.Z, rotatedVertex.Z);
                max.X = Math.Max(max.X, rotatedVertex.X);
                max.Y = Math.Max(max.Y, rotatedVertex.Y);
                max.Z = Math.Max(max.Z, rotatedVertex.Z);
            }
            return new AABB { Min = min, Max = max };
        }

        private Vector3 Rotate(Vector3 point, Quaternion rot)
        {
            var p = new Quaternion(0, point.X, point.Y, point.Z);
            var q_inv = new Quaternion(rot.W, -rot.X, -rot.Y, -rot.Z);
            var p_rotated = Multiply(Multiply(rot, p), q_inv);
            return new Vector3(p_rotated.X, p_rotated.Y, p_rotated.Z);
        }

        private Quaternion Multiply(Quaternion q1, Quaternion q2)
        {
            return new Quaternion(
                q1.W * q2.W - q1.X * q2.X - q1.Y * q2.Y - q1.Z * q2.Z,
                q1.W * q2.X + q1.X * q2.W + q1.Y * q2.Z - q1.Z * q2.Y,
                q1.W * q2.Y - q1.X * q2.Z + q1.Y * q2.W + q1.Z * q2.X,
                q1.W * q2.Z + q1.X * q2.Y - q1.Y * q2.X + q1.Z * q2.W
            );
        }
    }

    public class Container
    {
        public double Width, Height, Depth;
        public Container(double w, double h, double d) { Width = w; Height = h; Depth = d; }
    }

    public class PackedPolyhedron
    {
        public Polyhedron Polyhedron;
        public Vector3 Position;
        public Quaternion Rotation;
    }

    public class AABB
    {
        public Vector3 Min;
        public Vector3 Max;
    }
}
