#include <iostream>
#include <fstream>
#include <vector>
#include <algorithm>
#include <ctime>
#include <cstdlib>

// A structure to represent a graph edge
struct Edge {
    int src, dest;
};

// A structure to represent a subset for union-find.
// Used to track the subsets of vertices in the graph.
struct subset {
    int parent;
    int rank;
};

// A utility function to find the subset of an element i (uses path compression)
int find(std::vector<subset>& subsets, int i) {
    if (subsets[i].parent != i)
        subsets[i].parent = find(subsets, subsets[i].parent);
    return subsets[i].parent;
}

// A utility function to do union of two subsets of x and y (uses union by rank)
void Union(std::vector<subset>& subsets, int x, int y) {
    int xroot = find(subsets, x);
    int yroot = find(subsets, y);

    if (subsets[xroot].rank < subsets[yroot].rank)
        subsets[xroot].parent = yroot;
    else if (subsets[xroot].rank > subsets[yroot].rank)
        subsets[yroot].parent = xroot;
    else {
        subsets[yroot].parent = xroot;
        subsets[xroot].rank++;
    }
}

// Karger's algorithm to find the minimum cut in a connected, undirected graph
int kargerMinCut(int V, int E, std::vector<Edge>& edge) {
    // Create V subsets with single elements
    std::vector<subset> subsets(V);
    for (int v = 0; v < V; ++v) {
        subsets[v].parent = v;
        subsets[v].rank = 0;
    }

    // Initially, there are V vertices in the graph
    int vertices = V;
    while (vertices > 2) {
        // Pick a random edge from the edge list
        int i = rand() % E;

        // Find the subsets of the two vertices of the randomly picked edge
        int subset1 = find(subsets, edge[i].src);
        int subset2 = find(subsets, edge[i].dest);

        // If the two vertices are already in the same subset, it means
        // this edge is a self-loop. We can ignore it.
        if (subset1 == subset2)
            continue;
        else {
            // Otherwise, contract the edge by merging the two subsets.
            // The number of vertices is reduced by one.
            vertices--;
            Union(subsets, subset1, subset2);
        }
    }

    // After the while loop, there are only 2 vertices left.
    // The cut is the number of edges that cross between the two remaining subsets.
    int cut_edges = 0;
    for (int i = 0; i < E; i++) {
        int subset1 = find(subsets, edge[i].src);
        int subset2 = find(subsets, edge[i].dest);
        if (subset1 != subset2)
            cut_edges++;
    }

    return cut_edges;
}

int main() {
    // ---- Graph Creation ----
    // V = Number of vertices, E = Number of edges
    int V = 4;
    int E = 5;
    std::vector<Edge> edge = {
        {0, 1},
        {0, 2},
        {0, 3},
        {1, 3},
        {2, 3}
    };

    // Seed the random number generator
    srand(time(NULL));

    // --- Karger's Algorithm ---
    // Karger's algorithm is a randomized algorithm. The probability of finding the
    // minimum cut in a single run is low. Therefore, we run the algorithm multiple
    // times to increase the probability of finding the correct minimum cut.
    // The number of iterations is chosen to be high enough for a good probability.
    // A common choice is V^2 or V^2 * log(V).
    int min_cut = E; // The minimum cut cannot be larger than the number of edges
    int iterations = V * V; 
    for (int i = 0; i < iterations; ++i) {
        // We need to create a copy of the edges for each iteration
        // because the algorithm modifies the graph representation (implicitly).
        // However, in this DSU implementation, the 'subsets' are local to each call,
        // so we can reuse the same edge list.

        int cut = kargerMinCut(V, E, edge);
        if (cut < min_cut) {
            min_cut = cut;
        }
    }

    // --- Output ---
    std::cout << "The minimum cut found is: " << min_cut << std::endl;
    
    // Store the result in the 'answer.txt' file
    std::ofstream outfile("answer.txt");
    if (outfile.is_open()) {
        outfile << min_cut;
        outfile.close();
    } else {
        std::cerr << "Error: Unable to open answer.txt for writing." << std::endl;
    }

    return 0;
}
