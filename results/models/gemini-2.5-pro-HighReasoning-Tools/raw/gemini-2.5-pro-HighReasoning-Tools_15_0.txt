use std::io::{self, Read};
use nalgebra_glm as glm;
use earcutr;
use rand::Rng;

#[derive(Clone, Copy, Debug)]
struct Tetrahedron {
    position: glm::Vec3,
    rotation: glm::Quat,
    scale: f64,
}

// ... (rest of the code is the same as the previous version)
// Standard tetrahedron vertices
const STANDARD_TETRAHEDRON_VERTICES: [glm::Vec3; 4] = [
    glm::Vec3::new(1.0, 0.0, -0.707),
    glm::Vec3::new(-1.0, 0.0, -0.707),
    glm::Vec3::new(0.0, 1.0, 0.707),
    glm::Vec3::new(0.0, -1.0, 0.707),
];
// Standard tetrahedron edges
const STANDARD_TETRAHEDRON_EDGES: [(usize, usize); 6] = [
    (0, 1), (0, 2), (0, 3), (1, 2), (1, 3), (2, 3)
];

// Function to get the transformed vertices of a tetrahedron
fn get_transformed_vertices(tetrahedron: &Tetrahedron) -> [glm::Vec3; 4] {
    let mut transformed_vertices = [glm::Vec3::zeros(); 4];
    let transform = glm::translation(&tetrahedron.position)
        * glm::quat_to_mat4(&tetrahedron.rotation)
        * glm::scaling(&glm::vec3(tetrahedron.scale, tetrahedron.scale, tetrahedron.scale));

    for (i, vertex) in STANDARD_TETRAHEDRON_VERTICES.iter().enumerate() {
        let transformed_vertex = transform * vertex.to_homogeneous();
        transformed_vertices[i] = glm::vec3(transformed_vertex.x, transformed_vertex.y, transformed_vertex.z);
    }
    transformed_vertices
}

// Function to get the face normals of a tetrahedron
fn get_face_normals(vertices: &[glm::Vec3; 4]) -> [glm::Vec3; 4] {
    [
        glm::normalize(&glm::cross(&(vertices[1] - vertices[0]), &(vertices[2] - vertices[0]))),
        glm::normalize(&glm::cross(&(vertices[2] - vertices[1]), &(vertices[3] - vertices[1]))),
        glm::normalize(&glm::cross(&(vertices[3] - vertices[2]), &(vertices[0] - vertices[2]))),
        glm::normalize(&glm::cross(&(vertices[0] - vertices[3]), &(vertices[1] - vertices[3]))),
    ]
}

// Function to get the axes to test for SAT
fn get_axes(vertices1: &[glm::Vec3; 4], vertices2: &[glm::Vec3; 4]) -> Vec<glm::Vec3> {
    let mut axes = Vec::new();
    // Face normals of both tetrahedrons
    axes.extend_from_slice(&get_face_normals(vertices1));
    axes.extend_from_slice(&get_face_normals(vertices2));

    // Edge cross products
    for (i1, j1) in STANDARD_TETRAHEDRON_EDGES.iter() {
        for (i2, j2) in STANDARD_TETRAHEDRON_EDGES.iter() {
            let edge1 = vertices1[*j1] - vertices1[*i1];
            let edge2 = vertices2[*j2] - vertices2[*i2];
            let cross_product = glm::cross(&edge1, &edge2);
            if cross_product.magnitude() > 1e-6 {
                axes.push(glm::normalize(&cross_product));
            }
        }
    }
    axes
}

// Function to project vertices onto an axis
fn project(vertices: &[glm::Vec3; 4], axis: &glm::Vec3) -> (f64, f64) {
    let mut min = f64::INFINITY;
    let mut max = f64::NEG_INFINITY;
    for vertex in vertices {
        let projection = glm::dot(vertex, axis);
        if projection < min {
            min = projection;
        }
        if projection > max {
            max = projection;
        }
    }
    (min, max)
}

// SAT check for two tetrahedrons
fn check_collision(tetra1: &Tetrahedron, tetra2: &Tetrahedron) -> bool {
    let vertices1 = get_transformed_vertices(tetra1);
    let vertices2 = get_transformed_vertices(tetra2);

    let axes = get_axes(&vertices1, &vertices2);

    for axis in axes {
        let (min1, max1) = project(&vertices1, &axis);
        let (min2, max2) = project(&vertices2, &axis);

        if max1 < min2 || max2 < min1 {
            return false; // Found a separating axis
        }
    }
    true // No separating axis found
}

// Function to project 3D points to a 2D plane
fn project_to_2d(points: &[glm::Vec3], sun_direction: &glm::Vec3) -> Vec<glm::Vec2> {
    let view_matrix = glm::look_at(&glm::vec3(0.0,0.0,0.0), &sun_direction, &glm::vec3(0.0, 1.0, 0.0));
    points.iter().map(|p| {
        let projected_point = view_matrix * p.to_homogeneous();
        glm::vec2(projected_point.x, projected_point.y)
    }).collect()
}

// Point in polygon test
fn point_in_polygon(point: &glm::Vec2, polygon: &[glm::Vec2]) -> bool {
    let mut inside = false;
    let mut j = polygon.len() - 1;
    for i in 0..polygon.len() {
        let pi = polygon[i];
        let pj = polygon[j];
        if (pi.y > point.y) != (pj.y > point.y)
            && (point.x < (pj.x - pi.x) * (point.y - pi.y) / (pj.y - pi.y) + pi.x)
        {
            inside = !inside;
        }
        j = i;
    }
    inside
}

// Function to check if a triangle is covered by a shadow
fn is_triangle_covered(triangle: &(glm::Vec2, glm::Vec2, glm::Vec2), shadow_polygon: &[glm::Vec2]) -> bool {
    point_in_polygon(&triangle.0, shadow_polygon) &&
    point_in_polygon(&triangle.1, shadow_polygon) &&
    point_in_polygon(&triangle.2, shadow_polygon)
}

// Convex hull using Graham scan
fn convex_hull(points: &mut [glm::Vec2]) -> Vec<glm::Vec2> {
    if points.len() < 3 {
        return points.to_vec();
    }

    // Find the point with the lowest y-coordinate
    let mut start_idx = 0;
    for i in 1..points.len() {
        if points[i].y < points[start_idx].y {
            start_idx = i;
        } else if points[i].y == points[start_idx].y && points[i].x < points[start_idx].x {
            start_idx = i;
        }
    }
    points.swap(0, start_idx);
    let start_point = points[0];

    // Sort the points by polar angle with respect to the start point
    points[1..].sort_unstable_by(|a, b| {
        let cross_product = (a.x - start_point.x) * (b.y - start_point.y) - (b.x - start_point.x) * (a.y - start_point.y);
        if cross_product > 0.0 {
            std::cmp::Ordering::Less
        } else if cross_product < 0.0 {
            std::cmp::Ordering::Greater
        } else {
            let dist_a = (a.x - start_point.x).powi(2) + (a.y - start_point.y).powi(2);
            let dist_b = (b.x - start_point.x).powi(2) + (b.y - start_point.y).powi(2);
            dist_a.partial_cmp(&dist_b).unwrap()
        }
    });

    let mut hull = vec![points[0]];
    if points.len() > 1 {
        hull.push(points[1]);
    }
    for i in 2..points.len() {
        while hull.len() >= 2 {
            let p1 = hull[hull.len() - 2];
            let p2 = hull[hull.len() - 1];
            let p3 = points[i];
            let cross_product = (p2.x - p1.x) * (p3.y - p1.y) - (p2.y - p1.y) * (p3.x - p1.x);
            if cross_product <= 0.0 { // Use <= to handle collinear points
                hull.pop();
            } else {
                break;
            }
        }
        hull.push(points[i]);
    }
    hull
}


fn main() {
    let mut rng = rand::thread_rng();
    // ... (input reading and triangulation remains the same)
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut lines = input.lines();
    let nv: usize = lines.next().unwrap().parse().unwrap();
    let mut vertices = Vec::new();
    let mut flat_vertices = Vec::new();
    for _ in 0..nv {
        let line = lines.next().unwrap();
        let mut parts = line.split_whitespace();
        let x: f64 = parts.next().unwrap().parse().unwrap();
        let y: f64 = parts.next().unwrap().parse().unwrap();
        vertices.push(glm::vec2(x, y));
        flat_vertices.push(x);
        flat_vertices.push(y);
    }
    let line = lines.next().unwrap();
    let mut parts = line.split_whitespace();
    let sx: f64 = parts.next().unwrap().parse().unwrap();
    let sy: f64 = parts.next().unwrap().parse().unwrap();
    let sz: f64 = parts.next().unwrap().parse().unwrap();
    let sun_direction = glm::normalize(&glm::vec3(sx, sy, sz));
    let indices = earcutr::earcut(&flat_vertices, &[], 2).unwrap();
    let mut triangles: Vec<((glm::Vec2, glm::Vec2, glm::Vec2), f64)> = Vec::new();
    for i in (0..indices.len()).step_by(3) {
        let i1 = indices[i];
        let i2 = indices[i+1];
        let i3 = indices[i+2];
        let v1 = vertices[i1];
        let v2 = vertices[i2];
        let v3 = vertices[i3];
        let area = 0.5 * ((v2.x - v1.x) * (v3.y - v1.y) - (v3.x - v1.x) * (v2.y - v1.y)).abs();
        triangles.push(((v1, v2, v3), area));
    }
    triangles.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap());


    let mut tetrahedrons: Vec<Tetrahedron> = Vec::new();
    
    let mut uncovered_triangles = triangles;
    let max_tetrahedrons = uncovered_triangles.len(); // Heuristic limit
    
    while !uncovered_triangles.is_empty() && tetrahedrons.len() < max_tetrahedrons {
        let (triangle_to_cover, _) = uncovered_triangles[0];
        let centroid = (triangle_to_cover.0 + triangle_to_cover.1 + triangle_to_cover.2) / 3.0;

        let mut best_tetra = None;
        let mut best_score = f64::NEG_INFINITY;
        let num_rotations = if nv > 100 { 5 } else { 10 };

        // Try different rotations and scales to find the best fit
        for _ in 0..num_rotations { 
            let axis = glm::normalize(&glm::vec3(rng.gen(), rng.gen(), rng.gen()));
            let angle = rng.gen::<f64>() * 2.0 * std::f64::consts::PI;
            let rotation = glm::quat_angle_axis(angle, &axis);
            for scale_factor in (1..=10).rev() {
                let scale = scale_factor as f64;
                let mut tetra = Tetrahedron {
                    position: glm::vec3(centroid.x, centroid.y, 5.0),
                    rotation,
                    scale,
                };

                // Collision resolution
                let mut collision_tries = 0;
                let mut collision = true;
                while collision && collision_tries < 5 {
                    collision = false;
                    for existing_tetra in &tetrahedrons {
                        if check_collision(&tetra, existing_tetra) {
                            tetra.position.z += 1.0;
                            collision = true;
                            collision_tries += 1;
                            break;
                        }
                    }
                }
                if collision { continue; }

                let transformed_vertices = get_transformed_vertices(&tetra);
                let mut shadow_points = project_to_2d(&transformed_vertices.to_vec(), &sun_direction);
                let shadow_hull = convex_hull(&mut shadow_points);
                
                if shadow_hull.len() < 3 { continue; }

                let covered_area: f64 = uncovered_triangles.iter().filter(|(t, _)| is_triangle_covered(t, &shadow_hull)).map(|(_, area)| area).sum();
                let score = covered_area / scale;
                if score > best_score {
                    best_score = score;
                    best_tetra = Some(tetra);
                }
            }
        }

        if let Some(tetra) = best_tetra {
            tetrahedrons.push(tetra);
            let transformed_vertices = get_transformed_vertices(&tetra);
            let mut shadow_points = project_to_2d(&transformed_vertices.to_vec(), &sun_direction);
            let shadow_hull = convex_hull(&mut shadow_points);
            if shadow_hull.len() >= 3 {
                uncovered_triangles.retain(|(t, _)| !is_triangle_covered(t, &shadow_hull));
            }
        } else {
             // If we can't place a tetrahedron for the largest uncovered triangle, 
             // we are likely stuck. Remove it and try the next one.
             uncovered_triangles.remove(0);
        }
    }


    println!("{}", tetrahedrons.len());
    for tetrahedron in &tetrahedrons {
        println!(
            "{} {} {} {} {} {} {} {}",
            tetrahedron.position.x,
            tetrahedron.position.y,
            tetrahedron.position.z,
            tetrahedron.rotation.w,
            tetrahedron.rotation.i,
            tetrahedron.rotation.j,
            tetrahedron.rotation.k,
            tetrahedron.scale
        );
    }
}
