using System;
using System.Collections.Generic;
using System.Linq;

public class CuttingStock
{
    public static void Main(string[] args)
    {
        // Sample data
        var stockLengths = new List<double> { 3000, 3000, 3000, 3000 };
        var requiredPieces = new Dictionary<double, int>
        {
            { 1200, 5 },
            { 750, 5 },
            { 600, 5 }
        };

        var solution = Solve(stockLengths, requiredPieces);

        Console.WriteLine("Cutting Stock Solution:");
        for (int i = 0; i < solution.Count; i++)
        {
            Console.WriteLine($"Stock piece {i + 1} (length {stockLengths[i]}):");
            double totalCut = 0;
            foreach (var piece in solution[i])
            {
                Console.WriteLine($"  - Cut: {piece}");
                totalCut += piece;
            }
            Console.WriteLine($"  - Waste: {stockLengths[i] - totalCut}");
        }
    }

    public static List<List<double>> Solve(List<double> stockLengths, Dictionary<double, int> requiredPieces)
    {
        var solution = new List<List<double>>();
        for (int i = 0; i < stockLengths.Count; i++)
        {
            solution.Add(new List<double>());
        }

        var remainingLengths = new List<double>(stockLengths);
        var required = requiredPieces.SelectMany(p => Enumerable.Repeat(p.Key, p.Value)).OrderByDescending(l => l).ToList();

        foreach (var piece in required)
        {
            bool placed = false;
            for (int i = 0; i < remainingLengths.Count; i++)
            {
                if (remainingLengths[i] >= piece)
                {
                    solution[i].Add(piece);
                    remainingLengths[i] -= piece;
                    placed = true;
                    break;
                }
            }
            if (!placed)
            {
                // This piece could not be placed in any of the existing stock pieces.
                // For a more advanced solution, we could add a new stock piece.
                // For this greedy algorithm, we'll just ignore it, though this is not ideal.
                Console.WriteLine($"Warning: Could not place piece of length {piece}");
            }
        }

        return solution;
    }
}
