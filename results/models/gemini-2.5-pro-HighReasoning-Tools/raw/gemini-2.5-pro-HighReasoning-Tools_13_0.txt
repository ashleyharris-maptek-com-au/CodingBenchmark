#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

// Function to find the length of the Longest Common Substring
std::string longestCommonSubstring(const std::string& str1, const std::string& str2) {
    int m = str1.length();
    int n = str2.length();

    if (m == 0 || n == 0) {
        return "";
    }

    // Create a DP table to store lengths of longest common suffixes.
    // dp[i][j] contains length of longest common suffix of str1[0..i-1] and str2[0..j-1].
    std::vector<std::vector<int>> dp(m + 1, std::vector<int>(n + 1, 0));

    // To store the length of the longest common substring.
    int maxLength = 0;

    // To store the ending position of the longest common substring in str1.
    int endIndex = m;

    // Fill the dp table in a bottom-up manner.
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (str1[i - 1] == str2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
                if (dp[i][j] > maxLength) {
                    maxLength = dp[i][j];
                    endIndex = i;
                }
            } else {
                dp[i][j] = 0;
            }
        }
    }

    if (maxLength == 0) {
        return ""; // No common substring
    }

    // The longest common substring is from index (endIndex - maxLength) to (endIndex - 1) in str1
    return str1.substr(endIndex - maxLength, maxLength);
}

int main() {
    std::string str1 = "ABAB";
    std::string str2 = "BABA";
    std::cout << "String 1: " << str1 << std::endl;
    std::cout << "String 2: " << str2 << std::endl;
    std::cout << "Longest Common Substring is: " << longestCommonSubstring(str1, str2) << std::endl;

    std::string str3 = "GeeksforGeeks";
    std::string str4 = "GeeksQuiz";
    std::cout << "
String 1: " << str3 << std::endl;
    std::cout << "String 2: " << str4 << std::endl;
    std::cout << "Longest Common Substring is: " << longestCommonSubstring(str3, str4) << std::endl;

    return 0;
}
