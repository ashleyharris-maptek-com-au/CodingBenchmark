#include <iostream>
#include <vector>
#include <algorithm>
#include <set>

// Represents a single rectangle to be cut.
struct Rectangle {
    int id;
    int width;
    int height;
};

// Represents a placed rectangle in the bin.
struct PlacedRectangle {
    int id;
    int x;
    int y;
    int width;
    int height;
};

// The bin where rectangles are placed.
class Bin {
public:
    Bin(int width, int height) : width_(width), height_(height) {}

    bool canPlace(const Rectangle& rect, int x, int y) const {
        if (x + rect.width > width_ || y + rect.height > height_) {
            return false;
        }

        for (const auto& placed_rect : placed_rectangles_) {
            if (x < placed_rect.x + placed_rect.width &&
                x + rect.width > placed_rect.x &&
                y < placed_rect.y + placed_rect.height &&
                y + rect.height > placed_rect.y) {
                return false; // Collision detected.
            }
        }
        return true;
    }

    void place(const Rectangle& rect, int x, int y) {
        placed_rectangles_.push_back({rect.id, x, y, rect.width, rect.height});
    }

    const std::vector<PlacedRectangle>& getPlacedRectangles() const {
        return placed_rectangles_;
    }

    int getWidth() const { return width_; }
    int getHeight() const { return height_; }

private:
    int width_;
    int height_;
    std::vector<PlacedRectangle> placed_rectangles_;
};

// More sophisticated greedy algorithm using candidate positions.
// This algorithm sorts the rectangles by area in descending order.
// It maintains a set of candidate positions where a new rectangle can be placed.
// Initially, this set contains only (0,0). When a rectangle is placed,
// new candidate positions are added based on the corners of the placed rectangle.
void solve(Bin& bin, std::vector<Rectangle>& rectangles) {
    // Sort rectangles by area in descending order.
    std::sort(rectangles.begin(), rectangles.end(), [](const Rectangle& a, const Rectangle& b) {
        return (a.width * a.height) > (b.width * b.height);
    });

    // Candidate positions for placing new rectangles.
    std::set<std::pair<int, int>> candidate_positions;
    candidate_positions.insert({0, 0});

    for (const auto& rect : rectangles) {
        int best_x = -1;
        int best_y = -1;
        bool placed = false;

        std::pair<int, int> best_pos;

        for (const auto& pos : candidate_positions) {
            // Try placing with original orientation
            if (bin.canPlace(rect, pos.first, pos.second)) {
                best_x = pos.first;
                best_y = pos.second;
                best_pos = pos;
                break; 
            }
        }

        if (best_x != -1) {
            bin.place(rect, best_x, best_y);
            candidate_positions.erase(best_pos);
            candidate_positions.insert({best_x + rect.width, best_y});
            candidate_positions.insert({best_x, best_y + rect.height});
            placed = true;
        }


        if (!placed) {
            std::cout << "Could not place rectangle " << rect.id << std::endl;
        }
    }
}


int main() {
    // Example Usage
    Bin bin(100, 100);
    std::vector<Rectangle> rectangles = {
        {1, 10, 20}, {2, 25, 15}, {3, 30, 30}, {4, 12, 40}, {5, 50, 50}
    };

    solve(bin, rectangles);

    std::cout << "Placed Rectangles:" << std::endl;
    for (const auto& p_rect : bin.getPlacedRectangles()) {
        std::cout << "ID: " << p_rect.id << " at (" << p_rect.x << ", " << p_rect.y << ") "
                  << "size " << p_rect.width << "x" << p_rect.height << std::endl;
    }

    return 0;
}
