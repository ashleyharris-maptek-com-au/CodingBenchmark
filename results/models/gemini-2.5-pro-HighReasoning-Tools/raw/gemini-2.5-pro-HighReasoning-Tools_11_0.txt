use geo::{BoundingRect, Centroid, Coord, Intersects, LineString, Point, Polygon, Rect, Rotate, Translate};

struct StockSheet {
    width: f64,
    height: f64,
}

fn can_place(stock_sheet: &StockSheet, polygon: &Polygon<f64>, position: &Point<f64>) -> bool {
    let (width, height) = (stock_sheet.width, stock_sheet.height);
    let translated_polygon = polygon.translate(position.x(), position.y());
    translated_polygon.exterior().points().all(|p| {
        p.x() >= 0.0 && p.x() <= width && p.y() >= 0.0 && p.y() <= height
    })
}

fn bounding_box(polygons: &Vec<Polygon<f64>>) -> Option<Rect<f64>> {
    if polygons.is_empty() {
        return None;
    }

    let mut min_x = std::f64::MAX;
    let mut min_y = std::f64::MAX;
    let mut max_x = std::f64::MIN;
    let mut max_y = std::f64::MIN;

    for polygon in polygons {
        let bbox = polygon.bounding_rect().unwrap();
        min_x = min_x.min(bbox.min().x);
        min_y = min_y.min(bbox.min().y);
        max_x = max_x.max(bbox.max().x);
        max_y = max_y.max(bbox.max().y);
    }

    Some(Rect::new(
        Coord { x: min_x, y: min_y },
        Coord { x: max_x, y: max_y },
    ))
}

fn main() {
    let stock_sheet = StockSheet {
        width: 100.0,
        height: 100.0,
    };

    let polygons = vec![
        Polygon::new(
            LineString::from(vec![(0., 0.), (10., 0.), (10., 10.), (0., 10.)]),
            vec![],
        ),
        Polygon::new(
            LineString::from(vec![(0., 0.), (5., 0.), (5., 20.), (0., 20.)]),
            vec![],
        ),
    ];

    let mut placed_polygons = Vec::new();

    for polygon in polygons {
        let mut best_position = None;
        let mut best_rotation = 0.0;
        let mut best_score = std::f64::MAX;

        for &angle in &[0.0, 90.0, 180.0, 270.0] {
            let centroid = polygon.centroid().unwrap();
            let rotated_polygon = polygon.rotate_around_point(angle, centroid);
            let step = 1.0;

            for i in 0..=(stock_sheet.width as i32) {
                for j in 0..=(stock_sheet.height as i32) {
                    let position = Point::new(i as f64 * step, j as f64 * step);
                    if can_place(&stock_sheet, &rotated_polygon, &position) {
                        let translated_polygon = rotated_polygon.translate(position.x(), position.y());
                        if placed_polygons.iter().all(|p: &Polygon<f64>| !p.intersects(&translated_polygon)) {
                            let mut temp_placed_polygons = placed_polygons.clone();
                            temp_placed_polygons.push(translated_polygon);
                            let score = match bounding_box(&temp_placed_polygons) {
                                Some(bbox) => bbox.width() * bbox.height(),
                                None => 0.0,
                            };

                            if score < best_score {
                                best_score = score;
                                best_position = Some(position);
                                best_rotation = angle;
                            }
                        }
                    }
                }
            }
        }

        if let Some(position) = best_position {
            let rotated_polygon = polygon.rotate_around_centroid(best_rotation);
            let translated_polygon = rotated_polygon.translate(position.x(), position.y());
            placed_polygons.push(translated_polygon);
            println!("Placed a polygon at ({}, {}) with rotation {} degrees", position.x(), position.y(), best_rotation);
        } else {
            println!("Could not place a polygon.");
        }
    }
}
