using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Globalization;
using System.Linq;

public class Program
{
    public static void Main(string[] args)
    {
        var watch = Stopwatch.StartNew();
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;

        string[] firstLine = Console.ReadLine().Split();
        int n = int.Parse(firstLine[0]);
        double mu = double.Parse(firstLine[1]);

        string[] startOrbitLine = Console.ReadLine().Split();
        var startOrbit = new Orbit(
            new Vector3(double.Parse(startOrbitLine[2]), double.Parse(startOrbitLine[3]), double.Parse(startOrbitLine[4])),
            new Vector3(double.Parse(startOrbitLine[5]), double.Parse(startOrbitLine[6]), double.Parse(startOrbitLine[7]))
        );

        var stations = new List<Orbit>();
        for (int i = 0; i < n; i++)
        {
            string[] stationLine = Console.ReadLine().Split();
            stations.Add(new Orbit(
                new Vector3(double.Parse(stationLine[2]), double.Parse(stationLine[3]), double.Parse(stationLine[4])),
                new Vector3(double.Parse(stationLine[5]), double.Parse(stationLine[6]), double.Parse(stationLine[7]))
            ));
        }

        double[,] costMatrix = new double[n, n];
        double dt = 3600; // 1 hour

        Parallel.For(0, n, i =>
        {
            for (int j = 0; j < n; j++)
            {
                if (i == j)
                {
                    costMatrix[i, j] = 0;
                    continue;
                }

                try
                {
                    var (v1, v2) = LambertSolver.Solve(mu, stations[i].R, stations[j].R, dt, true);
                    costMatrix[i, j] = (v1 - stations[i].V).Magnitude + (stations[j].V - v2).Magnitude;
                }
                catch (Exception)
                {
                    costMatrix[i, j] = double.MaxValue;
                }
            }
        });

        var solver = new SimulatedAnnealing();
        var tour = solver.Solve(costMatrix, n, startOrbit, stations, mu, dt);

        Console.WriteLine(string.Join(" ", tour));
    }
}


public class SimulatedAnnealing
{
    private Random _random = new Random();

    public List<int> Solve(double[,] costMatrix, int n, Orbit startOrbit, List<Orbit> stations, double mu, double dt)
    {
        double temp = 10000;
        double coolingRate = 0.003;

        // Find the closest station to the starting point
        double minCost = double.MaxValue;
        int firstStation = 0;
        for (int i = 0; i < n; i++)
        {
            try
            {
                var (v1, v2) = LambertSolver.Solve(mu, startOrbit.R, stations[i].R, dt, true);
                double cost = (v1 - startOrbit.V).Magnitude;
                if (cost < minCost)
                {
                    minCost = cost;
                    firstStation = i;
                }
            }
            catch (Exception)
            {
                // Ignore stations that are not reachable
            }
        }

        var currentSolution = new List<int>();
        currentSolution.Add(firstStation);
        var remainingStations = Enumerable.Range(0, n).Where(i => i != firstStation).ToList();
        
        while(remainingStations.Any())
        {
            int lastStation = currentSolution.Last();
            int nextStation = remainingStations.OrderBy(s => costMatrix[lastStation, s]).First();
            currentSolution.Add(nextStation);
            remainingStations.Remove(nextStation);
        }


        var bestSolution = new List<int>(currentSolution);

        while (temp > 1)
        {
            var newSolution = new List<int>(currentSolution);

            int pos1 = _random.Next(1, n);
            int pos2 = _random.Next(1, n);

            (newSolution[pos1], newSolution[pos2]) = (newSolution[pos2], newSolution[pos1]);

            double currentEnergy = CalculateEnergy(currentSolution, costMatrix, n);
            double newEnergy = CalculateEnergy(newSolution, costMatrix, n);

            if (AcceptanceProbability(currentEnergy, newEnergy, temp) > _random.NextDouble())
            {
                currentSolution = new List<int>(newSolution);
            }

            if (CalculateEnergy(currentSolution, costMatrix, n) < CalculateEnergy(bestSolution, costMatrix, n))
            {
                bestSolution = new List<int>(currentSolution);
            }

            temp *= 1 - coolingRate;
        }

        return bestSolution;
    }

    private double CalculateEnergy(List<int> solution, double[,] costMatrix, int n)
    {
        double energy = 0;
        for (int i = 0; i < n - 1; i++)
        {
            energy += costMatrix[solution[i], solution[i + 1]];
        }
        return energy;
    }

    private double AcceptanceProbability(double currentEnergy, double newEnergy, double temp)
    {
        if (newEnergy < currentEnergy)
        {
            return 1.0;
        }

        return Math.Exp((currentEnergy - newEnergy) / temp);
    }
}


public class Vector3
{
    public double X, Y, Z;

    public Vector3(double x, double y, double z)
    {
        X = x;
        Y = y;
        Z = z;
    }

    public double Magnitude => Math.Sqrt(X * X + Y * Y + Z * Z);

    public static Vector3 operator +(Vector3 a, Vector3 b) => new Vector3(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
    public static Vector3 operator -(Vector3 a, Vector3 b) => new Vector3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
    public static Vector3 operator *(Vector3 a, double s) => new Vector3(a.X * s, a.Y * s, a.Z * s);
    public static Vector3 operator *(double s, Vector3 a) => new Vector3(a.X * s, a.Y * s, a.Z * s);
    public static double Dot(Vector3 a, Vector3 b) => a.X * b.X + a.Y * b.Y + a.Z * b.Z;
    public static Vector3 Cross(Vector3 a, Vector3 b) => new Vector3(a.Y * b.Z - a.Z * b.Y, a.Z * b.X - a.X * b.Z, a.X * b.Y - a.Y * b.X);
}

public class Orbit
{
    public Vector3 R, V; // Position and Velocity

    public Orbit(Vector3 r, Vector3 v)
    {
        R = r;
        V = v;
    }
}


public class LambertSolver
{
    private const int MaxIterations = 100;
    private const double Tolerance = 1e-8;

    public static (Vector3, Vector3) Solve(double mu, Vector3 r1, Vector3 r2, double dt, bool prograde)
    {
        double r1_mag = r1.Magnitude;
        double r2_mag = r2.Magnitude;

        double cos_dnu = Vector3.Dot(r1, r2) / (r1_mag * r2_mag);
        double dnu = prograde ? Math.Acos(cos_dnu) : 2 * Math.PI - Math.Acos(cos_dnu);

        double a = Math.Sin(dnu) * Math.Sqrt(r1_mag * r2_mag / (1 - cos_dnu));

        if (Math.Abs(a) < 1e-6)
        {
            // Parabolic case, not handled
            throw new Exception("Parabolic trajectory not handled");
        }

        double psi = 0;
        double psi_low = -4 * Math.PI;
        double psi_up = 4 * Math.PI;

        double y = 0;
        for (int i = 0; i < MaxIterations; i++)
        {
            y = r1_mag + r2_mag + a * (psi * S(psi) - 1) / Math.Sqrt(C(psi));

            if (Math.Abs(a) > 1e-6 && y < 0)
            {
                // Refine psi_low
                psi_low = psi;
            }

            double x = Math.Sqrt(y / C(psi));
            double t = (x * x * x * S(psi) + a * Math.Sqrt(y)) / Math.Sqrt(mu);

            if (Math.Abs(t - dt) < Tolerance)
            {
                break;
            }

            if (t < dt)
            {
                psi_low = psi;
            }
            else
            {
                psi_up = psi;
            }

            psi = (psi_up + psi_low) / 2;
        }

        double f = 1 - y / r1_mag;
        double g = a * Math.Sqrt(y / mu);
        double g_dot = 1 - y / r2_mag;

        Vector3 v1 = (r2 - r1 * f) * (1 / g);
        Vector3 v2 = (r2 * g_dot - r1) * (1 / g);

        return (v1, v2);
    }

    private static double S(double z)
    {
        if (z > 0)
        {
            return (Math.Sqrt(z) - Math.Sin(Math.Sqrt(z))) / (Math.Sqrt(z) * Math.Sqrt(z) * Math.Sqrt(z));
        }
        else if (z < 0)
        {
            return (Math.Sinh(Math.Sqrt(-z)) - Math.Sqrt(-z)) / (Math.Sqrt(-z) * Math.Sqrt(-z) * Math.Sqrt(-z));
        }
        else
        {
            return 1.0 / 6.0;
        }
    }

    private static double C(double z)
    {
        if (z > 0)
        {
            return (1 - Math.Cos(Math.Sqrt(z))) / z;
        }
        else if (z < 0)
        {
            return (Math.Cosh(Math.Sqrt(-z)) - 1) / (-z);
        }
        else
        {
            return 1.0 / 2.0;
        }
    }
}
