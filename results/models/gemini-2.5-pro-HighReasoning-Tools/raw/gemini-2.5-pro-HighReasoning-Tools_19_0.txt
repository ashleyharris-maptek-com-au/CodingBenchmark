using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

public class Program
{
    public static void Main(string[] args)
    {
        var world = ReadWorld();
        var operations = new List<MoveOperation>();
        var profitableVoxels = GetProfitableVoxels(world);

        while (profitableVoxels.Count > 0)
        {
            var voxel = profitableVoxels.First();
            profitableVoxels.RemoveAt(0);

            if (world.Voxels[voxel.X, voxel.Y, voxel.Z] == null) continue;

            if (!IsDiggable(voxel, world))
            {
                ClearPathToVoxel(voxel, world, operations, profitableVoxels);
            }
            
            var path = FindPath(voxel, 0, 0, world, true); // Allow digging
            if (path != null)
            {
                var profit = CalculateProfit(voxel, path);
                if (profit > 0)
                {
                    // Dig out the path
                    for (int i = 0; i < path.Steps.Count - 1; i++)
                    {
                        var step = path.Steps[i];
                        for(int z = 0; z< world.ZVoxelCount; z++)
                        {
                            var existingVoxel = world.Voxels[step.Item1, step.Item2, z];
                            if (existingVoxel != null)
                            {
                                var dumpSite = FindDumpSite(existingVoxel, world);
                                if (dumpSite != null)
                                {
                                    var dumpPath = FindPath(existingVoxel, dumpSite.X, dumpSite.Y, world, false); // No digging for dump
                                    if (dumpPath != null)
                                    {
                                        operations.Add(new MoveOperation
                                        {
                                            DigX = existingVoxel.X,
                                            DigY = existingVoxel.Y,
                                            DigZ = existingVoxel.Z,
                                            Steps = dumpPath.Steps,
                                            DestX = dumpSite.X,
                                            DestY = dumpSite.Y
                                        });
                                        world.Voxels[existingVoxel.X, existingVoxel.Y, existingVoxel.Z] = null;
                                    }
                                }
                            }
                        }
                    }


                    operations.Add(new MoveOperation
                    {
                        DigX = voxel.X,
                        DigY = voxel.Y,
                        DigZ = voxel.Z,
                        Steps = path.Steps,
                        DestX = 0,
                        DestY = 0
                    });

                    world.Voxels[voxel.X, voxel.Y, voxel.Z] = null;
                    HandleVoxelCollapse(new Voxel{X = voxel.X, Y = voxel.Y, Z = voxel.Z}, world);
                    UpdateDiggableVoxels(new Voxel{X = voxel.X, Y = voxel.Y, Z = voxel.Z}, world, profitableVoxels);
                }
            }
        }


        // Output
        PrintOperations(operations);
    }
    
    public static void ClearPathToVoxel(Voxel voxel, World world, List<MoveOperation> operations, List<Voxel> profitableVoxels)
    {
        if (IsDiggable(voxel, world)) return;

        if (voxel.Z < world.ZVoxelCount - 1)
        {
            var above = world.Voxels[voxel.X, voxel.Y, voxel.Z + 1];
            if (above != null)
            {
                ClearPathToVoxel(above, world, operations, profitableVoxels);
                
                if (world.Voxels[above.X, above.Y, above.Z] == null) return;

                var dumpSite = FindDumpSite(above, world);
                if (dumpSite != null)
                {
                    var path = FindPath(above, dumpSite.X, dumpSite.Y, world, false);
                    if (path != null)
                    {
                        operations.Add(new MoveOperation
                        {
                            DigX = above.X,
                            DigY = above.Y,
                            DigZ = above.Z,
                            Steps = path.Steps,
                            DestX = dumpSite.X,
                            DestY = dumpSite.Y
                        });
                        world.Voxels[above.X, above.Y, above.Z] = null;
                        var surfaceZ = GetSurfaceZ(dumpSite.X, dumpSite.Y, world);
                        if(surfaceZ < world.ZVoxelCount -1)
                        {
                            world.Voxels[dumpSite.X, dumpSite.Y, surfaceZ + 1] = new Voxel { Mass = above.Mass, Value = 0 };
                        }
                        
                        HandleVoxelCollapse(above, world);
                        UpdateDiggableVoxels(above, world, profitableVoxels);
                    }
                }
            }
        }
    }

    public static Voxel FindDumpSite(Voxel voxel, World world)
    {
        // Find a low point away from valuable ore to dump the voxel
        for (int x = 0; x < world.XVoxelCount; x++)
        {
            for (int y = 0; y < world.YVoxelCount; y++)
            {
                if (GetSurfaceZ(x, y, world) < 5) // Arbitrary low Z value
                {
                    // Check if there is valuable ore nearby
                    bool valuableNearby = false;
                    for (int dx = -5; dx <= 5; dx++)
                    {
                        for (int dy = -5; dy <= 5; dy++)
                        {
                            int newX = x + dx;
                            int newY = y + dy;
                            if (newX >= 0 && newX < world.XVoxelCount && newY >= 0 && newY < world.YVoxelCount)
                            {
                                for(int z = 0; z < world.ZVoxelCount; z++)
                                {
                                    if (world.Voxels[newX, newY, z]?.Value > 0)
                                    {
                                        valuableNearby = true;
                                        break;
                                    }
                                }
                            }
                        }
                        if (valuableNearby) break;
                    }

                    if (!valuableNearby)
                    {
                        return new Voxel { X = x, Y = y };
                    }
                }
            }
        }
        return new Voxel { X = world.XVoxelCount -1, Y = world.YVoxelCount - 1}; // Default dump site
    }

    public static void HandleVoxelCollapse(Voxel removedVoxel, World world)
    {
       // Simplified collapse logic for performance
        for(int x = 0; x < world.XVoxelCount; x++)
        {
            for(int y = 0; y < world.YVoxelCount; y++)
            {
                for(int z = 1; z < world.ZVoxelCount; z++)
                {
                    var voxel = world.Voxels[x, y, z];
                    if (voxel != null && world.Voxels[x, y, z - 1] == null && !IsSupported(voxel, world))
                    {
                        world.Voxels[x, y, z] = null;
                        world.Voxels[x, y, z - 1] = voxel;
                    }
                }
            }
        }
    }

    public static bool IsSupported(Voxel voxel, World world)
    {
        if (voxel.Z == 0) return true;
        
        var q = new Queue<Voxel>();
        q.Enqueue(voxel);

        var visited = new HashSet<string>();
        visited.Add($"{voxel.X},{voxel.Y},{voxel.Z}");
        
        while(q.Count > 0)
        {
            var current = q.Dequeue();
            if (current.Z == 0) return true;

            int[] dx = { 0, 0, 1, -1};
            int[] dy = { 1, -1, 0, 0};

            for(int i = 0; i < 4; i++)
            {
                int newX = current.X + dx[i];
                int newY = current.Y + dy[i];
                int newZ = current.Z -1;
                
                if (newX >= 0 && newX < world.XVoxelCount && newY >= 0 && newY < world.YVoxelCount && newZ >= 0)
                {
                    var neighbor = world.Voxels[newX, newY, current.Z];
                    if (neighbor != null && !visited.Contains($"{newX},{newY},{current.Z}"))
                    {
                        visited.Add($"{newX},{newY},{current.Z}");
                        q.Enqueue(neighbor);
                    }

                    var below = world.Voxels[current.X, current.Y, newZ];
                     if (below != null && !visited.Contains($"{current.X},{current.Y},{newZ}"))
                    {
                        visited.Add($"{current.X},{current.Y},{newZ}");
                        q.Enqueue(below);
                    }
                }
            }
        }
        return false;
    }


    public static World ReadWorld()
    {
        string header = Console.ReadLine();
        string[] headerParts = header.Split(' ');
        int xVoxelCount = int.Parse(headerParts[0]);
        int yVoxelCount = int.Parse(headerParts[1]);
        int zVoxelCount = int.Parse(headerParts[2]);
        double xBlockSize = double.Parse(headerParts[3]);
        double yBlockSize = double.Parse(headerParts[4]);
        double zBlockSize = double.Parse(headerParts[5]);

        var world = new World
        {
            XVoxelCount = xVoxelCount,
            YVoxelCount = yVoxelCount,
            ZVoxelCount = zVoxelCount,
            XBlockSize = xBlockSize,
            YBlockSize = yBlockSize,
            ZBlockSize = zBlockSize,
            Voxels = new Voxel[xVoxelCount, yVoxelCount, zVoxelCount]
        };

        using (var stdin = Console.OpenStandardInput())
        {
            for (int x = 0; x < xVoxelCount; x++)
            {
                for (int y = 0; y < yVoxelCount; y++)
                {
                    for (int z = 0; z < zVoxelCount; z++)
                    {
                        int d = stdin.ReadByte();
                        if (d == -1) break;
                        if (d == 0) continue;

                        int massInTonnes = d % 16;
                        int valuePerTonne = d / 16;

                        world.Voxels[x, y, z] = new Voxel
                        {
                            Mass = massInTonnes,
                            Value = valuePerTonne,
                            X = x,
                            Y = y,
                            Z = z
                        };
                    }
                }
            }
        }

        return world;
    }

    public static List<Voxel> GetProfitableVoxels(World world)
    {
        var profitableVoxels = new List<Voxel>();
        for (int x = 0; x < world.XVoxelCount; x++)
        {
            for (int y = 0; y < world.YVoxelCount; y++)
            {
                for (int z = 0; z < world.ZVoxelCount; z++)
                {
                    var voxel = world.Voxels[x, y, z];
                    if (voxel != null && voxel.Value > 0)
                    {
                        profitableVoxels.Add(voxel);
                    }
                }
            }
        }
        return profitableVoxels.OrderByDescending(v => v.Value + v.Mass).ToList();
    }

    public static void UpdateDiggableVoxels(Voxel removedVoxel, World world, List<Voxel> profitableVoxels)
    {
        int x = removedVoxel.X;
        int y = removedVoxel.Y;
        int z = removedVoxel.Z;

        int[] dx = { 0, 0, 1, -1, 0, 0 };
        int[] dy = { 1, -1, 0, 0, 0, 0 };
        int[] dz = { 0, 0, 0, 0, 1, -1 };

        for (int i = 0; i < 6; i++)
        {
            int newX = x + dx[i];
            int newY = y + dy[i];
            int newZ = z + dz[i];

            if (newX >= 0 && newX < world.XVoxelCount && newY >= 0 && newY < world.YVoxelCount && newZ >= 0 && newZ < world.ZVoxelCount)
            {
                var voxel = world.Voxels[newX, newY, newZ];
                if (voxel != null && voxel.Value > 0 && IsDiggable(voxel, world) && !profitableVoxels.Any(v => v.X == voxel.X && v.Y == voxel.Y && v.Z == voxel.Z))
                {
                    profitableVoxels.Add(voxel);
                }
            }
        }
    }


    public static bool IsDiggable(Voxel voxel, World world)
    {
        if (voxel.Z == world.ZVoxelCount - 1) return true;
        if (world.Voxels[voxel.X, voxel.Y, voxel.Z + 1] == null) return true;

        if (voxel.X > 0 && world.Voxels[voxel.X - 1, voxel.Y, voxel.Z] == null) return true;
        if (voxel.X < world.XVoxelCount - 1 && world.Voxels[voxel.X + 1, voxel.Y, voxel.Z] == null) return true;
        if (voxel.Y > 0 && world.Voxels[voxel.X, voxel.Y - 1, voxel.Z] == null) return true;
        if (voxel.Y < world.YVoxelCount - 1 && world.Voxels[voxel.X, voxel.Y + 1, voxel.Z] == null) return true;

        return false;
    }

    public static Path FindPath(Voxel start, int destX, int destY, World world, bool allowDigging)
    {
        var openSet = new List<Node>();
        var closedSet = new HashSet<string>();

        var startNode = new Node { X = start.X, Y = start.Y, Z = start.Z, G = 0 };
        startNode.H = Math.Abs(start.X - destX) + Math.Abs(start.Y - destY);
        openSet.Add(startNode);

        while (openSet.Count > 0)
        {
            var currentNode = openSet.OrderBy(n => n.F).First();

            if (currentNode.X == destX && currentNode.Y == destY)
            {
                var path = new Path { Steps = new List<Tuple<int, int>>(), Cost = currentNode.G };
                var temp = currentNode;
                while (temp != null)
                {
                    path.Steps.Insert(0, new Tuple<int, int>(temp.X, temp.Y));
                    temp = temp.Parent;
                }
                return path;
            }

            openSet.Remove(currentNode);
            closedSet.Add($"{currentNode.X},{currentNode.Y},{currentNode.Z}");

            foreach (var neighbor in GetNeighbors(currentNode, world, destX, destY, allowDigging))
            {
                var key = $"{neighbor.X},{neighbor.Y},{neighbor.Z}";
                if (closedSet.Contains(key)) continue;

                var existingNode = openSet.FirstOrDefault(n => n.X == neighbor.X && n.Y == neighbor.Y && n.Z == neighbor.Z);
                if (existingNode == null)
                {
                    openSet.Add(neighbor);
                }
                else if (neighbor.G < existingNode.G)
                {
                    existingNode.G = neighbor.G;
                    existingNode.Parent = neighbor.Parent;
                }
            }
        }

        return null;
    }

    public static List<Node> GetNeighbors(Node node, World world, int destX, int destY, bool allowDigging)
    {
        var neighbors = new List<Node>();
        int[] dx = { 0, 0, 1, -1 };
        int[] dy = { 1, -1, 0, 0 };

        for (int i = 0; i < 4; i++)
        {
            int newX = node.X + dx[i];
            int newY = node.Y + dy[i];

            if (newX >= 0 && newX < world.XVoxelCount && newY >= 0 && newY < world.YVoxelCount)
            {
                int currentZ = node.Z;
                for (int zOffset = -1; zOffset <= 1; zOffset++)
                {
                    int newZ = currentZ + zOffset;
                    if (newZ >= 0 && newZ < world.ZVoxelCount)
                    {
                        var neighborVoxel = world.Voxels[newX, newY, newZ];
                        if (neighborVoxel == null || allowDigging)
                        {
                            var neighbor = new Node { X = newX, Y = newY, Z = newZ, Parent = node };
                            double cost = Math.Abs(newX - node.X) + Math.Abs(newY - node.Y) + Math.Abs(newZ - node.Z);

                            if (newZ > currentZ)
                            {
                                cost *= 5;
                            }

                            if (neighborVoxel != null)
                            {
                                cost += neighborVoxel.Mass * 1; // Cost to dig
                            }

                            neighbor.G = node.G + cost;
                            neighbor.H = Math.Abs(neighbor.X - destX) + Math.Abs(neighbor.Y - destY);
                            neighbors.Add(neighbor);
                        }
                    }
                }
            }
        }
        return neighbors;
    }

    public static int GetSurfaceZ(int x, int y, World world)
    {
        for (int z = world.ZVoxelCount - 1; z >= 0; z--)
        {
            if (world.Voxels[x, y, z] != null)
            {
                return z;
            }
        }
        return 0;
    }

    public static double CalculateProfit(Voxel voxel, Path path)
    {
        double revenue = voxel.Mass * voxel.Value * 1000;
        if (voxel.Value == 0)
        {
            revenue = -1000 * voxel.Mass;
        }

        double cost = path.Cost * voxel.Mass;
        return revenue - cost;
    }

    public static void PrintOperations(List<MoveOperation> operations)
    {
        Console.WriteLine(operations.Count);
        foreach (var op in operations)
        {
            var stepsStr = string.Join(" ", op.Steps.Select(s => $"{s.Item1} {s.Item2}"));
            Console.WriteLine($"{op.Steps.Count} {op.DigX} {op.DigY} {op.DigZ} {stepsStr} {op.DestX} {op.DestY}");
        }
    }
}

public class Node
{
    public int X { get; set; }
    public int Y { get; set; }
    public int Z { get; set; }
    public double G { get; set; } // Cost from start
    public double H { get; set; } // Heuristic
    public double F => G + H;
    public Node Parent { get; set; }
}

public class Voxel
{
    public int Mass { get; set; }
    public int Value { get; set; }
    public int X { get; set; }
    public int Y { get; set; }
    public int Z { get; set; }
}

public class World
{
    public int XVoxelCount { get; set; }
    public int YVoxelCount { get; set; }
    public int ZVoxelCount { get; set; }
    public double XBlockSize { get; set; }
    public double YBlockSize { get; set; }
    public double ZBlockSize { get; set; }
    public Voxel[,,] Voxels { get; set; }
}

public class MoveOperation
{
    public int DigX { get; set; }
    public int DigY { get; set; }
    public int DigZ { get; set; }
    public List<Tuple<int, int>> Steps { get; set; }
    public int DestX { get; set; }
    public int DestY { get; set; }
}

public class Path
{
    public List<Tuple<int, int>> Steps { get; set; }
    public double Cost { get; set; }
}
