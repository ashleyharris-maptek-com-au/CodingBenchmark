using System;
using System.Collections.Generic;
using System.Linq;

namespace BinPacking3D
{
    public class Algorithm
    {
        public List<Bin> Pack(List<Box> boxes, int binWidth, int binHeight, int binDepth)
        {
            var bins = new List<Bin>();
            var sortedBoxes = boxes.OrderByDescending(b => b.Volume).ToList();

            foreach (var box in sortedBoxes)
            {
                var packed = false;
                foreach (var bin in bins)
                {
                    foreach (var rotation in box.GetRotations())
                    {
                        if (TryPack(rotation, bin))
                        {
                            packed = true;
                            break;
                        }
                    }
                    if (packed)
                    {
                        break;
                    }
                }

                if (!packed)
                {
                    var newBin = new Bin(binWidth, binHeight, binDepth);
                    foreach (var rotation in box.GetRotations())
                    {
                        if (TryPack(rotation, newBin))
                        {
                            bins.Add(newBin);
                            packed = true;
                            break;
                        }
                    }
                    if (!packed)
                    {
                        // Box is too large for a new bin in any rotation
                    }
                }
            }

            return bins;
        }

        private bool TryPack(Box box, Bin bin)
        {
            if (bin.Items.Count == 0)
            {
                if (box.Width <= bin.Width && box.Height <= bin.Height && box.Depth <= bin.Depth)
                {
                    bin.Items.Add(new Item(box, new Point3D(0, 0, 0)));
                    return true;
                }
                return false;
            }

            foreach (var pivot in GetPivotPoints(bin))
            {
                if (CanFit(box, pivot, bin))
                {
                    bin.Items.Add(new Item(box, pivot));
                    return true;
                }
            }

            return false;
        }

        private bool CanFit(Box box, Point3D pivot, Bin bin)
        {
            if (pivot.X + box.Width > bin.Width || pivot.Y + box.Height > bin.Height || pivot.Z + box.Depth > bin.Depth)
            {
                return false;
            }

            foreach (var item in bin.Items)
            {
                if (IsOverlapping(item, box, pivot))
                {
                    return false;
                }
            }
            return true;
        }

        private bool IsOverlapping(Item item1, Box box2, Point3D pos2)
        {
            var pos1 = item1.Position;
            var box1 = item1.Box;

            return (pos1.X < pos2.X + box2.Width && pos1.X + box1.Width > pos2.X &&
                    pos1.Y < pos2.Y + box2.Height && pos1.Y + box1.Height > pos2.Y &&
                    pos1.Z < pos2.Z + box2.Depth && pos1.Z + box1.Depth > pos2.Z);
        }


        private IEnumerable<Point3D> GetPivotPoints(Bin bin)
        {
            var pivots = new HashSet<Point3D>();
            pivots.Add(new Point3D(0, 0, 0));

            foreach (var item in bin.Items)
            {
                pivots.Add(new Point3D(item.Position.X + item.Box.Width, item.Position.Y, item.Position.Z));
                pivots.Add(new Point3D(item.Position.X, item.Position.Y + item.Box.Height, item.Position.Z));
                pivots.Add(new Point3D(item.Position.X, item.Position.Y, item.Position.Z + item.Box.Depth));
            }

            return pivots;
        }
    }
}
