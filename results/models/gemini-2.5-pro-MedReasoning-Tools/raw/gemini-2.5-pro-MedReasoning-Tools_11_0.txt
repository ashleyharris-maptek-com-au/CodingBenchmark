use std::io::{self, BufRead};

#[derive(Debug, Clone)]
struct Point {
    x: f64,
    y: f64,
}

#[derive(Debug, Clone)]
struct Polygon {
    vertices: Vec<Point>,
}

#[derive(Debug)]
struct Placement {
    stock_index: usize,
    x: f64,
    y: f64,
    rotation: f64,
}

impl Polygon {
    fn rotate(&self, angle: f64) -> Polygon {
        let cos_a = angle.cos();
        let sin_a = angle.sin();
        let mut new_vertices = Vec::new();
        for p in &self.vertices {
            let new_x = p.x * cos_a - p.y * sin_a;
            let new_y = p.x * sin_a + p.y * cos_a;
            new_vertices.push(Point { x: new_x, y: new_y });
        }
        Polygon { vertices: new_vertices }
    }

    fn translate(&self, dx: f64, dy: f64) -> Polygon {
        let mut new_vertices = Vec::new();
        for p in &self.vertices {
            new_vertices.push(Point { x: p.x + dx, y: p.y + dy });
        }
        Polygon { vertices: new_vertices }
    }
}

fn parse_input() -> (Polygon, Vec<Polygon>) {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = lines.next().unwrap().unwrap();
    let mut parts = first_line.split_whitespace();
    let sv: usize = parts.next().unwrap().parse().unwrap();
    let np: usize = parts.next().unwrap().parse().unwrap();

    let mut stock_vertices = Vec::new();
    for _ in 0..sv {
        let line = lines.next().unwrap().unwrap();
        let mut parts = line.split_whitespace();
        let x: f64 = parts.next().unwrap().parse().unwrap();
        let y: f64 = parts.next().unwrap().parse().unwrap();
        stock_vertices.push(Point { x, y });
    }
    let stock_polygon = Polygon { vertices: stock_vertices };

    let mut pieces = Vec::new();
    for _ in 0..np {
        let line = lines.next().unwrap().unwrap();
        let pv: usize = line.parse().unwrap();
        let mut piece_vertices = Vec::new();
        for _ in 0..pv {
            let line = lines.next().unwrap().unwrap();
            let mut parts = line.split_whitespace();
            let x: f64 = parts.next().unwrap().parse().unwrap();
            let y: f64 = parts.next().unwrap().parse().unwrap();
            piece_vertices.push(Point { x, y });
        }
        pieces.push(Polygon { vertices: piece_vertices });
    }

    (stock_polygon, pieces)
}

fn point_in_polygon(point: &Point, polygon: &Polygon) -> bool {
    let mut intersections = 0;
    let mut j = polygon.vertices.len() - 1;
    for i in 0..polygon.vertices.len() {
        let pi = &polygon.vertices[i];
        let pj = &polygon.vertices[j];
        if (pi.y > point.y) != (pj.y > point.y)
            && (point.x < (pj.x - pi.x) * (point.y - pi.y) / (pj.y - pi.y) + pi.x)
        {
            intersections += 1;
        }
        j = i;
    }
    intersections % 2 == 1
}

fn polygons_intersect(p1: &Polygon, p2: &Polygon) -> bool {
    let mut axes = get_axes(p1);
    axes.append(&mut get_axes(p2));

    for axis in axes {
        let p1_proj = project(p1, &axis);
        let p2_proj = project(p2, &axis);

        if !overlap(p1_proj, p2_proj) {
            return false;
        }
    }

    true
}

fn get_axes(polygon: &Polygon) -> Vec<Point> {
    let mut axes = Vec::new();
    for i in 0..polygon.vertices.len() {
        let p1 = &polygon.vertices[i];
        let p2 = &polygon.vertices[(i + 1) % polygon.vertices.len()];
        let edge = Point { x: p1.x - p2.x, y: p1.y - p2.y };
        let normal = Point { x: -edge.y, y: edge.x };
        axes.push(normal);
    }
    axes
}

fn project(polygon: &Polygon, axis: &Point) -> (f64, f64) {
    let mut min = f64::MAX;
    let mut max = f64::MIN;
    for p in &polygon.vertices {
        let dot_product = p.x * axis.x + p.y * axis.y;
        min = min.min(dot_product);
        max = max.max(dot_product);
    }
    (min, max)
}

fn overlap(p1: (f64, f64), p2: (f64, f64)) -> bool {
    p1.0 < p2.1 && p2.0 < p1.1
}

fn place_pieces(stock_polygon: &Polygon, pieces: &Vec<Polygon>) -> (usize, Vec<Placement>) {
    let mut placements = Vec::new();
    let mut placed_polygons: Vec<Vec<Polygon>> = vec![Vec::new()];
    let rotation_steps = 8;

    for (i, piece) in pieces.iter().enumerate() {
        let mut placed = false;
        'stock_loop: for stock_index in 0..placed_polygons.len() {
            let mut candidate_points = stock_polygon.vertices.clone();
            for poly in &placed_polygons[stock_index] {
                candidate_points.extend(poly.vertices.clone());
            }

            for i in 0..rotation_steps {
                let angle = (i as f64) * std::f64::consts::PI * 2.0 / (rotation_steps as f64);
                let rotated_piece = piece.rotate(angle);

                for p_candidate in &candidate_points {
                    let x = p_candidate.x;
                    let y = p_candidate.y;

                    let translated_piece = rotated_piece.translate(x, y);

                    let mut fits = true;
                    // Check if it's inside the stock polygon
                    for p in &translated_piece.vertices {
                        if !point_in_polygon(p, stock_polygon) {
                            fits = false;
                            break;
                        }
                    }

                    if !fits {
                        continue;
                    }

                    // Check for overlap with other pieces
                    for placed_piece in &placed_polygons[stock_index] {
                        if polygons_intersect(&translated_piece, placed_piece) {
                            fits = false;
                            break;
                        }
                    }

                    if fits {
                        placements.push(Placement {
                            stock_index,
                            x,
                            y,
                            rotation: angle,
                        });
                        placed_polygons[stock_index].push(translated_piece);
                        placed = true;
                        break 'stock_loop;
                    }
                }
            }
        }

        if !placed {
            // Did not fit in any existing stock, create a new one
            placed_polygons.push(Vec::new());
            let stock_index = placed_polygons.len() - 1;
            
            // For simplicity, just place it at the origin of the new stock
            // A better approach would be to find the best position in the new stock sheet.
            placements.push(Placement {
                stock_index,
                x: 0.0,
                y: 0.0,
                rotation: 0.0,
            });
            placed_polygons[stock_index].push(piece.clone());
        }
    }

    (placed_polygons.len(), placements)
}

fn main() {
    let (stock_polygon, pieces) = parse_input();
    let (num_stocks, placements) = place_pieces(&stock_polygon, &pieces);

    println!("{}", num_stocks);
    for p in placements {
        println!("{} {} {} {}", p.stock_index, p.x, p.y, p.rotation);
    }
}
