#pragma once

#include "Shape.h"
#include <memory>
#include <algorithm>
#include <iostream>

enum class CSGOperation {
    Union,
    Intersection,
    Difference
};

class CSG : public Shape {
public:
    std::shared_ptr<Shape> left;
    std::shared_ptr<Shape> right;
    CSGOperation op;

    CSG(std::shared_ptr<Shape> left, std::shared_ptr<Shape> right, CSGOperation op)
        : left(left), right(right), op(op) {}

    bool is_inside(const Vector& p) const {
        // This is a more robust inside test, but still has limitations.
        // It assumes shapes are closed and watertight.
        auto test_ray = Ray(p, Vector(1, 0, 0));
        return (left->intersect(test_ray).size() % 2 == 1) || (right->intersect(test_ray).size() % 2 == 1);
    }

    std::vector<Hit> intersect(const Ray& ray) const override {
        std::vector<Hit> left_hits = left->intersect(ray);
        std::vector<Hit> right_hits = right->intersect(ray);

        std::vector<Hit> all_hits;
        all_hits.insert(all_hits.end(), left_hits.begin(), left_hits.end());
        all_hits.insert(all_hits.end(), right_hits.begin(), right_hits.end());

        std::sort(all_hits.begin(), all_hits.end(), [](const Hit& a, const Hit& b) {
            return a.t < b.t;
        });

        if (op == CSGOperation::Union) {
            return filter_union(all_hits);
        }
        // Other operations would be handled here
        return {};
    }

private:
    bool is_inside_shape(const Shape* shape, const Vector& point) const {
        Ray test_ray(point, Vector(0,0,1));
        std::vector<Hit> hits = shape->intersect(test_ray);
        return hits.size() % 2 == 1;
    }

    std::vector<Hit> filter_union(const std::vector<Hit>& all_hits) const {
        std::vector<Hit> result_hits;
        bool in_left = false;
        bool in_right = false;

        for (const auto& hit : all_hits) {
            bool hit_is_from_left = (hit.shape == left.get());

            if (hit_is_from_left) {
                if (!in_right) {
                    result_hits.push_back(hit);
                }
                in_left = !in_left;
            } else { // Hit is from right
                if (!in_left) {
                    result_hits.push_back(hit);
                }
                in_right = !in_right;
            }
        }
        return result_hits;
    }
};
