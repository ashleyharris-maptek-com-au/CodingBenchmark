using System;
using System.Collections.Generic;
using System.Globalization;
using System.Linq;

public class OrbitalTspSolver
{
    public const double EarthMu = 398600.4418; // km^3/s^2

    public static void Main(string[] args)
    {
        var input = Console.ReadLine().Split(' ');
        int n = int.Parse(input[0]);
        double mu = double.Parse(input[1], CultureInfo.InvariantCulture);

        var startOrbitLine = Console.ReadLine().Split(' ');
        var startPosition = new Vector3d(
            double.Parse(startOrbitLine[2], CultureInfo.InvariantCulture),
            double.Parse(startOrbitLine[3], CultureInfo.InvariantCulture),
            double.Parse(startOrbitLine[4], CultureInfo.InvariantCulture)
        );
        var startVelocity = new Vector3d(
            double.Parse(startOrbitLine[5], CultureInfo.InvariantCulture),
            double.Parse(startOrbitLine[6], CultureInfo.InvariantCulture),
            double.Parse(startOrbitLine[7], CultureInfo.InvariantCulture)
        );

        var stations = new List<Station>();
        for (int i = 0; i < n; i++)
        {
            var stationOrbitLine = Console.ReadLine().Split(' ');
            stations.Add(new Station
            {
                Id = i,
                Position = new Vector3d(
                    double.Parse(stationOrbitLine[2], CultureInfo.InvariantCulture),
                    double.Parse(stationOrbitLine[3], CultureInfo.InvariantCulture),
                    double.Parse(stationOrbitLine[4], CultureInfo.InvariantCulture)
                ),
                Velocity = new Vector3d(
                    double.Parse(stationOrbitLine[5], CultureInfo.InvariantCulture),
                    double.Parse(stationOrbitLine[6], CultureInfo.InvariantCulture),
                    double.Parse(stationOrbitLine[7], CultureInfo.InvariantCulture)
                )
            });
        }

        var tour = SolveTsp(startPosition, startVelocity, stations, mu);
        Console.WriteLine(string.Join(" ", tour));
    }

    public static List<int> SolveTsp(Vector3d startPosition, Vector3d startVelocity, List<Station> stations, double mu)
    {
        var unvisited = new List<Station>(stations);
        var tour = new List<int>();
        var currentPosition = startPosition;
        var currentVelocity = startVelocity;

        while (unvisited.Any())
        {
            Station bestStation = null;
            double minDeltaV = double.MaxValue;

            foreach (var station in unvisited)
            {
                // Simplified time of flight: 1 hour. A real implementation would optimize this.
                double timeOfFlight = 3600; 
                try
                {
                    var (v1, _) = LambertSolver.Solve(currentPosition, station.Position, timeOfFlight, mu, true);
                    double deltaV = (v1 - currentVelocity).Magnitude;
                    if (deltaV < minDeltaV)
                    {
                        minDeltaV = deltaV;
                        bestStation = station;
                    }
                }
                catch (Exception)
                {
                    // Ignore stations that are unreachable
                }
            }

            if (bestStation != null)
            {
                tour.Add(bestStation.Id);
                unvisited.Remove(bestStation);
                currentPosition = bestStation.Position;
                currentVelocity = bestStation.Velocity;
            }
            else
            {
                // If no station is reachable, break the loop
                break;
            }
        }

        return tour;
    }
}

public class Station
{
    public int Id { get; set; }
    public Vector3d Position { get; set; }
    public Vector3d Velocity { get; set; }
}

public struct Vector3d
{
    public double X, Y, Z;

    public Vector3d(double x, double y, double z)
    {
        X = x;
        Y = y;
        Z = z;
    }

    public double Magnitude => Math.Sqrt(X * X + Y * Y + Z * Z);
    public Vector3d Normalized => this / Magnitude;

    public static Vector3d operator +(Vector3d a, Vector3d b) => new Vector3d(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
    public static Vector3d operator -(Vector3d a, Vector3d b) => new Vector3d(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
    public static Vector3d operator *(Vector3d a, double s) => new Vector3d(a.X * s, a.Y * s, a.Z * s);
    public static Vector3d operator /(Vector3d a, double s) => new Vector3d(a.X / s, a.Y / s, a.Z / s);

    public static double Dot(Vector3d a, Vector3d b) => a.X * b.X + a.Y * b.Y + a.Z * b.Z;
    public static Vector3d Cross(Vector3d a, Vector3d b) => new Vector3d(a.Y * b.Z - a.Z * b.Y, a.Z * b.X - a.X * b.Z, a.X * b.Y - a.Y * b.X);

    public override string ToString() => $"({X}, {Y}, {Z})";
}

public static class LambertSolver
{
    public static (Vector3d, Vector3d) Solve(Vector3d r1Vec, Vector3d r2Vec, double dt, double mu, bool shortWay)
    {
        double r1 = r1Vec.Magnitude;
        double r2 = r2Vec.Magnitude;

        double cosDv = Vector3d.Dot(r1Vec, r2Vec) / (r1 * r2);
        double sinDv = (shortWay ? 1 : -1) * Math.Sqrt(1 - cosDv * cosDv);
        
        // Handle the 180 degree transfer case
        if (Math.Abs(1.0 + cosDv) < 1e-9) // A is undefined
        {
            // This is a special case that requires a different approach.
            // For this TSP, we can consider it an invalid transfer.
            throw new InvalidOperationException("180 degree transfer not supported");
        }
        double A = sinDv * Math.Sqrt(r1 * r2 / (1 - cosDv));


        double z = 0;
        double lowerBound = -100;
        double upperBound = 10000;

        for (int i = 0; i < 100; i++) // Bisection method
        {
            z = (upperBound + lowerBound) / 2.0;
            double c_z = C(z);
            double s_z = S(z);

            double y = r1 + r2 + A * (z * s_z - 1) / Math.Sqrt(c_z);
            if (A > 0 && y < 0)
            {
                lowerBound = z;
                continue;
            }

            double dt_calc = (Math.Pow(y / c_z, 1.5) * s_z + A * Math.Sqrt(y)) / Math.Sqrt(mu);

            if (dt_calc <= dt)
            {
                lowerBound = z;
            }
            else
            {
                upperBound = z;
            }
            
            if (Math.Abs(dt_calc - dt) < 1e-6)
            {
                break;
            }
        }
        
        double y_final = r1 + r2 + A * (z * S(z) - 1) / Math.Sqrt(C(z));

        double f = 1 - y_final / r1;
        double g = A * Math.Sqrt(y_final / mu);
        double gDot = 1 - y_final / r2;

        Vector3d v1 = (r2Vec - r1Vec * f) / g;
        Vector3d v2 = (r2Vec * gDot - r1Vec) / g;

        return (v1, v2);
    }

    private static double C(double z)
    {
        if (z > 1e-6) return (1 - Math.Cos(Math.Sqrt(z))) / z;
        if (z < -1e-6) return (Math.Cosh(Math.Sqrt(-z)) - 1) / (-z);
        return 1.0 / 2.0;
    }

    private static double S(double z)
    {
        if (z > 1e-6)
        {
            double sqrtZ = Math.Sqrt(z);
            return (sqrtZ - Math.Sin(sqrtZ)) / (sqrtZ * sqrtZ * sqrtZ);
        }
        if (z < -1e-6)
        {
            double sqrtNegZ = Math.Sqrt(-z);
            return (Math.Sinh(sqrtNegZ) - sqrtNegZ) / (sqrtNegZ * sqrtNegZ * sqrtNegZ);
        }
        return 1.0 / 6.0;
    }
}
