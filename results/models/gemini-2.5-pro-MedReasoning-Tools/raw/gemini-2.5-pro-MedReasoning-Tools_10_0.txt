#include <iostream>
#include <fstream>
#include <vector>
#include <algorithm>
#include <nlohmann/json.hpp>

#include "Rectangle.h"
#include "Bin.h"

using json = nlohmann::json;

// Function to parse the input file
void parseInput(const std::string& filename, int& binWidth, int& binHeight, std::vector<Rectangle>& rectangles) {
    std::ifstream inputFile(filename);
    if (!inputFile.is_open()) {
        std::cerr << "Error opening input file." << std::endl;
        return;
    }

    inputFile >> binWidth >> binHeight;

    int id = 0;
    int width, height;
    while (inputFile >> width >> height) {
        rectangles.push_back({width, height, id++});
    }
}

// First-Fit algorithm
void firstFit(std::vector<Rectangle>& rectangles, std::vector<Bin>& bins, int binWidth, int binHeight) {
    // Sort rectangles by height in descending order
    std::sort(rectangles.begin(), rectangles.end(), [](const Rectangle& a, const Rectangle& b) {
        return a.height > b.height;
    });

    for (const auto& rect : rectangles) {
        bool placed = false;
        for (auto& bin : bins) {
            for (int y = 0; y <= bin.getHeight() - rect.height; ++y) {
                for (int x = 0; x <= bin.getWidth() - rect.width; ++x) {
                    if (bin.canPlace(rect, x, y)) {
                        bin.place(rect, x, y);
                        placed = true;
                        goto next_rect;
                    }
                }
            }
        }

        if (!placed) {
            Bin newBin(binWidth, binHeight);
            newBin.place(rect, 0, 0);
            bins.push_back(newBin);
        }
        next_rect:;
    }
}

// Function to write the output to a JSON file
void writeOutput(const std::string& filename, const std::vector<Bin>& bins) {
    json output;
    output["bins"] = json::array();

    for (const auto& bin : bins) {
        json binJson;
        binJson["width"] = bin.getWidth();
        binJson["height"] = bin.getHeight();
        binJson["rectangles"] = json::array();

        for (const auto& placedRect : bin.getPlacedRectangles()) {
            json rectJson;
            rectJson["id"] = placedRect.rect.id;
            rectJson["x"] = placedRect.x;
            rectJson["y"] = placedRect.y;
            rectJson["width"] = placedRect.rect.width;
            rectJson["height"] = placedRect.rect.height;
            binJson["rectangles"].push_back(rectJson);
        }
        output["bins"].push_back(binJson);
    }

    std::ofstream outputFile(filename);
    if (!outputFile.is_open()) {
        std::cerr << "Error opening output file." << std::endl;
        return;
    }

    outputFile << output.dump(4);
}

int main() {
    int binWidth, binHeight;
    std::vector<Rectangle> rectangles;
    parseInput("question.txt", binWidth, binHeight, rectangles);

    std::vector<Bin> bins;
    firstFit(rectangles, bins, binWidth, binHeight);

    writeOutput("answer.json", bins);

    return 0;
}
