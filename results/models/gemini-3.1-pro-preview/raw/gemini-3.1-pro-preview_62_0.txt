declare const require: any;
const fs = require('fs');

interface Town {
    id: string;
    x: number;
    y: number;
    value: number;
}

interface Input {
    width: number;
    height: number;
    grid: string[];
    ignitions: [number, number][];
    towns: Town[];
    cutBudget: number;
}

const inputStr: string = fs.readFileSync(0, 'utf8');
const data: Input = JSON.parse(inputStr);

const width = data.width;
const height = data.height;
const gridStr = data.grid;
const ignitions = data.ignitions;
const towns = data.towns;
let cutBudget = data.cutBudget;

const grid = new Uint8Array(width * height);
for (let y = 0; y < height; y++) {
    const row = gridStr[y];
    for (let x = 0; x < width; x++) {
        if (row[x] === '~') {
            grid[y * width + x] = 1;
        }
    }
}

const townMap = new Map<number, Town>();
const townIndices = new Set<number>();
for (const town of towns) {
    const idx = town.y * width + town.x;
    townMap.set(idx, town);
    townIndices.add(idx);
}

const ignitionIndices = new Set<number>();
for (const [x, y] of ignitions) {
    ignitionIndices.add(y * width + x);
}

const dx = [0, 1, 0, -1];
const dy = [-1, 0, 1, 0];

function simulate(cuts: number[]): { burnedValue: number, burnedCells: number[] } {
    const localGrid = new Uint8Array(grid);
    for (let i = 0; i < cuts.length; i++) {
        localGrid[cuts[i]] = 2;
    }

    const q: number[] = [];
    for (const [x, y] of ignitions) {
        const idx = y * width + x;
        if (localGrid[idx] === 0) {
            localGrid[idx] = 3;
            q.push(idx);
        }
    }

    let head = 0;
    let burnedValue = 0;
    const burnedCells: number[] = [];

    while (head < q.length) {
        const curr = q[head++];
        burnedCells.push(curr);
        
        if (townIndices.has(curr)) {
            burnedValue += townMap.get(curr)!.value;
        }

        const x = curr % width;
        const y = Math.floor(curr / width);

        for (let i = 0; i < 4; i++) {
            const nx = x + dx[i];
            const ny = y + dy[i];

            if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                const nIdx = ny * width + nx;
                if (localGrid[nIdx] === 0) {
                    localGrid[nIdx] = 3;
                    q.push(nIdx);
                }
            }
        }
    }

    return { burnedValue, burnedCells };
}

const currentCuts: number[] = [];
const initialSim = simulate([]);
let bestBurnedValue = initialSim.burnedValue;

const startTime = Date.now();
const timeLimit = 4000; // Time limit for heuristic search

while (currentCuts.length < cutBudget) {
    if (Date.now() - startTime > timeLimit) break;

    const currentSim = simulate(currentCuts);
    if (currentSim.burnedValue === 0) break;

    let currentCandidates = currentSim.burnedCells.filter(idx => !townIndices.has(idx) && !ignitionIndices.has(idx));
    
    const maxEvals = 100;
    if (currentCandidates.length > maxEvals) {
        const subset = new Set<number>();
        for (let i = 0; i < maxEvals / 2 && i < currentCandidates.length; i++) {
            subset.add(currentCandidates[i]);
        }
        while (subset.size < maxEvals && subset.size < currentCandidates.length) {
            subset.add(currentCandidates[Math.floor(Math.random() * currentCandidates.length)]);
        }
        currentCandidates = Array.from(subset);
    }

    let bestCut = -1;
    let minBurnedValue = bestBurnedValue;
    let minBurnedCellsCount = Infinity;

    for (let i = 0; i < currentCandidates.length; i++) {
        const candidate = currentCandidates[i];
        currentCuts.push(candidate);
        const sim = simulate(currentCuts);
        currentCuts.pop();

        if (sim.burnedValue < minBurnedValue || (sim.burnedValue === minBurnedValue && sim.burnedCells.length < minBurnedCellsCount)) {
            minBurnedValue = sim.burnedValue;
            minBurnedCellsCount = sim.burnedCells.length;
            bestCut = candidate;
        }
    }

    if (bestCut !== -1) {
        currentCuts.push(bestCut);
        bestBurnedValue = minBurnedValue;
    } else {
        break; 
    }
}

const firebreaks = currentCuts.map(idx => [idx % width, Math.floor(idx / width)]);
console.log(JSON.stringify({ firebreaks }));
