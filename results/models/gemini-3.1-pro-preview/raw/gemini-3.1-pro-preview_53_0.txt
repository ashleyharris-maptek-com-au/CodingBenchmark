import math

class AutopilotState:
    def __init__(self):
        self.last_x = 0.0
        self.last_y = 0.0
        self.last_z = 0.0
        self.last_vx = 0.0
        self.last_vy = 0.0
        self.last_vz = 0.0
        self.last_pitch = 0.0
        self.last_roll = 0.0
        self.last_yaw = 0.0
        
        self.int_vx = 0.0
        self.int_vy = 0.0
        self.int_vz = 0.0
        
        self.initialized = False

ap_state = AutopilotState()

def autopilot_step(state: dict, dt: float) -> dict:
    global ap_state
    
    # 1. Calculate orbital mean motion (n)
    sma = 6771000.0
    if state.get('ground_semi_major_axis_m_valid', False):
        sma = state.get('ground_semi_major_axis_m', 6771000.0)
    elif state.get('ground_altitude_km_valid', False):
        sma = state.get('ground_altitude_km', 400.0) * 1000.0 + 6371000.0
    mu = 3.986004418e14
    n = math.sqrt(mu / (sma**3))
    
    # 2. Extract Accelerations
    accels = []
    for i in range(1, 4):
        if state.get(f'accel_{i}_x_valid', False):
            accels.append((
                state.get(f'accel_{i}_x_ms2', 0.0),
                state.get(f'accel_{i}_y_ms2', 0.0),
                state.get(f'accel_{i}_z_ms2', 0.0)
            ))
    accel_x, accel_y, accel_z = 0.0, 0.0, 0.0
    if len(accels) == 3:
        accel_x = sorted([a[0] for a in accels])[1]
        accel_y = sorted([a[1] for a in accels])[1]
        accel_z = sorted([a[2] for a in accels])[1]
    elif len(accels) > 0:
        accel_x = sum([a[0] for a in accels]) / len(accels)
        accel_y = sum([a[1] for a in accels]) / len(accels)
        accel_z = sum([a[2] for a in accels]) / len(accels)

    # 3. Extract Gyro Rates
    gyros = []
    for i in range(1, 4):
        if state.get(f'gyro_{i}_pitch_valid', False):
            gyros.append((
                state.get(f'gyro_{i}_pitch_dps', 0.0),
                state.get(f'gyro_{i}_roll_dps', 0.0),
                state.get(f'gyro_{i}_yaw_dps', 0.0)
            ))
    g_pitch, g_roll, g_yaw = 0.0, 0.0, 0.0
    if len(gyros) == 3:
        g_pitch = sorted([g[0] for g in gyros])[1]
        g_roll = sorted([g[1] for g in gyros])[1]
        g_yaw = sorted([g[2] for g in gyros])[1]
    elif len(gyros) > 0:
        g_pitch = sum([g[0] for g in gyros]) / len(gyros)
        g_roll = sum([g[1] for g in gyros]) / len(gyros)
        g_yaw = sum([g[2] for g in gyros]) / len(gyros)

    # 4. Extract Attitude
    pitch, roll, yaw = None, None, None
    if state.get('star_tracker_1_pitch_valid', False):
        pitch = state.get('star_tracker_1_pitch_deg')
        roll = state.get('star_tracker_1_roll_deg')
        yaw = state.get('star_tracker_1_yaw_deg')
    elif state.get('star_tracker_2_pitch_valid', False):
        pitch = state.get('star_tracker_2_pitch_deg')
        roll = state.get('star_tracker_2_roll_deg')
        yaw = state.get('star_tracker_2_yaw_deg')

    if pitch is None:
        if ap_state.initialized:
            pitch = ap_state.last_pitch + g_pitch * dt
            roll = ap_state.last_roll + g_roll * dt
            yaw = ap_state.last_yaw + g_yaw * dt
        else:
            pitch, roll, yaw = 0.0, 0.0, 0.0

    # 5. Extract Relative Pos/Vel
    x, y, z, vx, vy, vz = None, None, None, None, None, None
    
    if state.get('radar_rel_x_valid', False):
        x = state.get('radar_rel_pos_x_m')
    elif state.get('lidar_rel_x_valid', False):
        x = state.get('lidar_rel_pos_x_m')

    if state.get('radar_rel_y_valid', False):
        y = state.get('radar_rel_pos_y_m')
    elif state.get('lidar_rel_y_valid', False):
        y = state.get('lidar_rel_pos_y_m')
        
    if state.get('radar_rel_z_valid', False):
        z = state.get('radar_rel_pos_z_m')
    elif state.get('lidar_rel_z_valid', False):
        z = state.get('lidar_rel_pos_z_m')
        
    if state.get('radar_relv_x_valid', False):
        vx = state.get('radar_rel_vel_x_ms')
    elif state.get('lidar_relv_x_valid', False):
        vx = state.get('lidar_rel_vel_x_ms')
        
    if state.get('radar_relv_y_valid', False):
        vy = state.get('radar_rel_vel_y_ms')
    elif state.get('lidar_relv_y_valid', False):
        vy = state.get('lidar_rel_vel_y_ms')
        
    if state.get('radar_relv_z_valid', False):
        vz = state.get('radar_rel_vel_z_ms')
    elif state.get('lidar_relv_z_valid', False):
        vz = state.get('lidar_rel_vel_z_ms')

    # 6. State Estimator Predictor (Fallback for missing/flickering sensors)
    if ap_state.initialized:
        px = ap_state.last_x
        py = ap_state.last_y
        pz = ap_state.last_z
        pvx = ap_state.last_vx
        pvy = ap_state.last_vy
        pvz = ap_state.last_vz
        
        # CW Dynamics + Measured Acceleration exactly models the physics
        ax = 2 * n * pvy + 3 * n**2 * px + accel_x
        ay = -2 * n * pvx + accel_y
        az = -n**2 * pz + accel_z
        
        pred_vx = pvx + ax * dt
        pred_vy = pvy + ay * dt
        pred_vz = pvz + az * dt
        pred_x = px + pvx * dt + 0.5 * ax * dt**2
        pred_y = py + pvy * dt + 0.5 * ay * dt**2
        pred_z = pz + pvz * dt + 0.5 * az * dt**2
        
        if x is None: x = pred_x
        if y is None: y = pred_y
        if z is None: z = pred_z
        if vx is None: vx = pred_vx
        if vy is None: vy = pred_vy
        if vz is None: vz = pred_vz
    else:
        if x is None: x = 0.0
        if y is None: y = 0.0
        if z is None: z = 0.0
        if vx is None: vx = 0.0
        if vy is None: vy = 0.0
        if vz is None: vz = 0.0
        ap_state.initialized = True

    # 7. Update State
    ap_state.last_x = x
    ap_state.last_y = y
    ap_state.last_z = z
    ap_state.last_vx = vx
    ap_state.last_vy = vy
    ap_state.last_vz = vz
    ap_state.last_pitch = pitch
    ap_state.last_roll = roll
    ap_state.last_yaw = yaw

    # 8. Guidance (Target Velocity)
    dist = math.sqrt(x**2 + y**2 + z**2)
    speed_limit = min(3.0, 0.02 + 0.01 * dist)

    kp_pos = 0.02
    v_des_x = -kp_pos * x
    v_des_y = -kp_pos * y
    v_des_z = -kp_pos * z
    
    v_des_mag = math.sqrt(v_des_x**2 + v_des_y**2 + v_des_z**2)
    if v_des_mag > speed_limit:
        scale = speed_limit / v_des_mag
        v_des_x *= scale
        v_des_y *= scale
        v_des_z *= scale

    # 9. Velocity Controller (PD + I)
    kd_vel = 0.05
    ki_vel = 0.002
    
    ap_state.int_vx += (v_des_x - vx) * dt
    ap_state.int_vy += (v_des_y - vy) * dt
    ap_state.int_vz += (v_des_z - vz) * dt
    
    max_int = 0.5 / ki_vel
    ap_state.int_vx = max(-max_int, min(max_int, ap_state.int_vx))
    ap_state.int_vy = max(-max_int, min(max_int, ap_state.int_vy))
    ap_state.int_vz = max(-max_int, min(max_int, ap_state.int_vz))

    ax_pd = kd_vel * (v_des_x - vx) + ki_vel * ap_state.int_vx
    ay_pd = kd_vel * (v_des_y - vy) + ki_vel * ap_state.int_vy
    az_pd = kd_vel * (v_des_z - vz) + ki_vel * ap_state.int_vz

    # 10. Feedback Linearization (CW Cancellation)
    cancel_ax = - (2 * n * vy + 3 * n**2 * x)
    cancel_ay = 2 * n * vx
    cancel_az = n**2 * z

    ax_cmd = ax_pd + cancel_ax
    ay_cmd = ay_pd + cancel_ay
    az_cmd = az_pd + cancel_az

    # 11. Command Generation
    A_MAX = 0.1 # Assumed max acceleration (m/s^2)
    thrust_x = max(-1.0, min(1.0, ax_cmd / A_MAX))
    thrust_y = max(-1.0, min(1.0, ay_cmd / A_MAX))
    thrust_z = max(-1.0, min(1.0, az_cmd / A_MAX))

    # Attitude commands to maintain 0 alignment
    cmd_roll_rate_rad = math.radians(-0.05 * roll)
    cmd_pitch_rate_rad = math.radians(-0.05 * pitch)
    cmd_yaw_rate_rad = math.radians(-0.05 * yaw)

    att_rate_roll = max(-0.035, min(0.035, cmd_roll_rate_rad))
    att_rate_pitch = max(-0.035, min(0.035, cmd_pitch_rate_rad))
    att_rate_yaw = max(-0.035, min(0.035, cmd_yaw_rate_rad))

    return {
        'thrust_x': float(thrust_x),
        'thrust_y': float(thrust_y),
        'thrust_z': float(thrust_z),
        'att_rate_roll': float(att_rate_roll),
        'att_rate_pitch': float(att_rate_pitch),
        'att_rate_yaw': float(att_rate_yaw)
    }
