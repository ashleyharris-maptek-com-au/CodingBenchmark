class Controller:
    def __init__(self):
        self.i_gs = 0.0
        self.i_loc = 0.0
        self.i_spd = 0.0
        self.prev_gs = 0.0
        self.prev_loc = 0.0
        self.prev_spd = 0.0
        self.runway_elev = None
        self.go_around = False

ctrl = Controller()

def get_median(state, prefix, count):
    vals = []
    for i in range(1, count + 1):
        if state.get(f"{prefix}_{i}_valid", False):
            v = state.get(f"{prefix}_{i}_dots")
            if v is None: v = state.get(f"{prefix}_{i}_kt")
            if v is None: v = state.get(f"{prefix}_{i}_deg")
            if v is not None:
                vals.append(v)
    if not vals:
        return 0.0
    vals.sort()
    return vals[len(vals)//2]

def autopilot_step(state: dict, dt: float) -> dict:
    global ctrl

    # Extract inputs using sensor voting
    gs_dots = get_median(state, "glideslope", 3)
    loc_dots = get_median(state, "localizer", 3)
    ias = get_median(state, "ias", 3)
    if ias == 0.0: ias = state.get("target_ias_kt", 137.0)
    
    vs_vals = []
    for i in range(1, 4):
        if state.get(f"irs_{i}_valid", False):
            v = state.get(f"irs_{i}_vs_fpm")
            if v is not None: vs_vals.append(v)
    vs = sorted(vs_vals)[len(vs_vals)//2] if vs_vals else -700.0

    track = state.get("gps_1_track_deg", 0.0)
    hdg = state.get("irs_1_heading_deg", 0.0)
    
    # Altitude and fallback
    radio_alt = state.get("radio_alt_ft")
    gps_alt = state.get("gps_1_alt_ft", 0.0)
    
    if radio_alt is not None and radio_alt > 0:
        if ctrl.runway_elev is None:
            ctrl.runway_elev = gps_alt - radio_alt
        else:
            ctrl.runway_elev = 0.99 * ctrl.runway_elev + 0.01 * (gps_alt - radio_alt)
    else:
        if ctrl.runway_elev is not None:
            radio_alt = gps_alt - ctrl.runway_elev
        else:
            radio_alt = 2000.0 

    vref = state.get("target_vref_kt", 137.0)
    wow = state.get("weight_on_wheels", False)
    on_ground = state.get("on_ground", False)
    if wow or on_ground or radio_alt < 1.0:
        wow = True

    eng1 = state.get("engine_1_running", True)
    eng2 = state.get("engine_2_running", True)
    dual_eng_fail = not eng1 and not eng2

    # Go-around logic
    if 50 < radio_alt < 200 and not dual_eng_fail:
        if abs(gs_dots) > 2.0 or abs(loc_dots) > 2.0 or abs(ias - vref) > 25:
            ctrl.go_around = True

    if ctrl.go_around:
        return {
            "elevator": 0.5,
            "aileron": 0.0,
            "rudder": 0.0,
            "throttle_1": 1.0 if eng1 else 0.0,
            "throttle_2": 1.0 if eng2 else 0.0,
            "flaps": 15.0,
            "gear": 0.0,
            "speedbrake": 0.0,
            "elevator_trim": 0.5
        }

    gear = 1.0
    flaps = 40.0
    speedbrake = 0.0
    spoiler = 0.0
    autobrake = 0.0
    rev1 = 0.0
    rev2 = 0.0
    elevator = 0.0
    aileron = 0.0
    rudder = 0.0
    throttle_1 = 0.0
    throttle_2 = 0.0

    if not wow:
        # Pitch Control
        if dual_eng_fail:
            err_spd = ias - vref
            p_p = 0.1
            d_p = 0.05
            elevator = (err_spd * p_p) + ((err_spd - ctrl.prev_spd)/dt * d_p)
        else:
            err_gs = gs_dots
            ctrl.i_gs += err_gs * dt
            p_gs = -0.15
            i_gs = -0.01
            d_gs = -0.1
            elevator = err_gs * p_gs + ctrl.i_gs * i_gs + (err_gs - ctrl.prev_gs)/dt * d_gs

            if radio_alt < 50.0: # Flare
                target_vs = -150.0
                err_vs = target_vs - vs
                elevator += err_vs * 0.001
                elevator += (50.0 - radio_alt) * 0.005

        # Roll Control
        err_loc = loc_dots
        ctrl.i_loc += err_loc * dt
        p_loc = -0.2
        i_loc = -0.005
        d_loc = -0.1
        aileron = err_loc * p_loc + ctrl.i_loc * i_loc + (err_loc - ctrl.prev_loc)/dt * d_loc

        # Yaw Control
        crab_angle = track - hdg
        crab_angle = (crab_angle + 180) % 360 - 180
        
        if radio_alt < 50.0:
            rudder = crab_angle * 0.05
            aileron -= rudder * 0.5

        # Speed Control
        if not dual_eng_fail:
            err_spd = vref - ias
            ctrl.i_spd += err_spd * dt
            p_spd = 0.15
            i_spd = 0.02
            d_spd = 0.1
            throttle = err_spd * p_spd + ctrl.i_spd * i_spd + (err_spd - ctrl.prev_spd)/dt * d_spd
            
            if radio_alt < 30.0:
                throttle = 0.0
            throttle = max(min(throttle, 1.0), 0.0)

            if not eng1 and eng2:
                throttle_1 = 0.0
                throttle_2 = min(throttle * 2.0, 1.0)
                rudder += 0.3 * throttle_2
            elif eng1 and not eng2:
                throttle_1 = min(throttle * 2.0, 1.0)
                throttle_2 = 0.0
                rudder -= 0.3 * throttle_1
            else:
                throttle_1 = throttle
                throttle_2 = throttle

        # Aileron stuck compensation
        if aileron > 0.9:
            rudder += 0.2
        elif aileron < -0.9:
            rudder -= 0.2

        ctrl.prev_gs = gs_dots
        ctrl.prev_loc = loc_dots
        ctrl.prev_spd = ias

    else:
        # Ground Rollout
        elevator = -0.5
        rudder = loc_dots * -0.3
        spoiler = 1.0
        autobrake = 1.0
        speedbrake = 1.0
        
        gs = state.get("groundspeed_kt", 0.0)
        if gs > 60.0:
            rev1 = 1.0 if eng1 else 0.0
            rev2 = 1.0 if eng2 else 0.0

    return {
        "elevator": max(min(elevator, 1.0), -1.0),
        "aileron": max(min(aileron, 1.0), -1.0),
        "rudder": max(min(rudder, 1.0), -1.0),
        "throttle_1": max(min(throttle_1, 1.0), 0.0),
        "throttle_2": max(min(throttle_2, 1.0), 0.0),
        "flaps": flaps,
        "gear": gear,
        "speedbrake": speedbrake,
        "spoiler": spoiler,
        "autobrake": autobrake,
        "reverse_1": rev1,
        "reverse_2": rev2,
        "elevator_trim": max(min(elevator, 1.0), -1.0)
    }
