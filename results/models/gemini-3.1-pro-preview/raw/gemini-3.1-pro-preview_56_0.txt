const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    if (!input || input.trim() === '') return;
    const data = JSON.parse(input);
    const depot = data.depot;
    const packages = data.packages;

    if (!packages || packages.length === 0) {
        console.log(JSON.stringify({ routes: [] }));
        return;
    }

    const N = packages.length;
    
    // 1. Grid-based spatial index to construct a very fast Nearest Neighbor initial tour
    let minX = depot[0], maxX = depot[0];
    let minY = depot[1], maxY = depot[1];
    for (let i = 0; i < N; i++) {
        let p = packages[i];
        if (p.x < minX) minX = p.x;
        if (p.x > maxX) maxX = p.x;
        if (p.y < minY) minY = p.y;
        if (p.y > maxY) maxY = p.y;
    }

    const C = Math.max(1, Math.floor(Math.sqrt(N)));
    const cellW = (maxX - minX) / C + 1e-9;
    const cellH = (maxY - minY) / C + 1e-9;

    const grid = new Array(C * C);
    for (let i = 0; i < C * C; i++) grid[i] = [];

    for (let i = 0; i < N; i++) {
        let p = packages[i];
        let c = Math.floor((p.x - minX) / cellW);
        let r = Math.floor((p.y - minY) / cellH);
        if (c < 0) c = 0; if (c >= C) c = C - 1;
        if (r < 0) r = 0; if (r >= C) r = C - 1;
        grid[r * C + c].push(p);
    }

    function getDistCoords(x1, y1, x2, y2) {
        let dx = x1 - x2;
        let dy = y1 - y2;
        return Math.sqrt(dx * dx + dy * dy);
    }

    function getDist(p1, p2) {
        let dx = p1.x - p2.x;
        let dy = p1.y - p2.y;
        return Math.sqrt(dx * dx + dy * dy);
    }

    let tour = [];
    let currX = depot[0];
    let currY = depot[1];

    let visitedCount = 0;
    while (visitedCount < N) {
        let bestDist = Infinity;
        let bestP = null;
        let bestCell = null;
        let bestIdx = -1;

        let startC = Math.floor((currX - minX) / cellW);
        let startR = Math.floor((currY - minY) / cellH);
        if (startC < 0) startC = 0; if (startC >= C) startC = C - 1;
        if (startR < 0) startR = 0; if (startR >= C) startR = C - 1;

        let maxR = Math.max(startC, C - 1 - startC, startR, C - 1 - startR);
        for (let radius = 0; radius <= maxR; radius++) {
            if (radius > 0) {
                let boxMinX = minX + Math.max(0, startC - radius + 1) * cellW;
                let boxMaxX = minX + Math.min(C, startC + radius) * cellW;
                let boxMinY = minY + Math.max(0, startR - radius + 1) * cellH;
                let boxMaxY = minY + Math.min(C, startR + radius) * cellH;

                let minDistToRing = 0;
                if (currX >= boxMinX && currX <= boxMaxX && currY >= boxMinY && currY <= boxMaxY) {
                    minDistToRing = Infinity;
                    if (startC - radius >= 0) minDistToRing = Math.min(minDistToRing, currX - boxMinX);
                    if (startC + radius < C)  minDistToRing = Math.min(minDistToRing, boxMaxX - currX);
                    if (startR - radius >= 0) minDistToRing = Math.min(minDistToRing, currY - boxMinY);
                    if (startR + radius < C)  minDistToRing = Math.min(minDistToRing, boxMaxY - currY);
                    if (minDistToRing === Infinity) minDistToRing = 0;
                } else {
                    let bdx = Math.max(0, boxMinX - currX, currX - boxMaxX);
                    let bdy = Math.max(0, boxMinY - currY, currY - boxMaxY);
                    minDistToRing = Math.sqrt(bdx * bdx + bdy * bdy);
                }

                if (minDistToRing >= bestDist) {
                    break;
                }
            }

            let rStartC = Math.max(0, startC - radius);
            let rEndC = Math.min(C - 1, startC + radius);
            let rStartR = Math.max(0, startR - radius);
            let rEndR = Math.min(C - 1, startR + radius);

            let innerMinC = startC - radius + 1;
            let innerMaxC = startC + radius - 1;
            let innerMinR = startR - radius + 1;
            let innerMaxR = startR + radius - 1;

            for (let r = rStartR; r <= rEndR; r++) {
                for (let c = rStartC; c <= rEndC; c++) {
                    if (radius > 0 && c >= innerMinC && c <= innerMaxC && r >= innerMinR && r <= innerMaxR) {
                        continue;
                    }

                    let cellMinX = minX + c * cellW;
                    let cellMaxX = cellMinX + cellW;
                    let cellMinY = minY + r * cellH;
                    let cellMaxY = cellMinY + cellH;
                    
                    let cdx = Math.max(0, cellMinX - currX, currX - cellMaxX);
                    let cdy = Math.max(0, cellMinY - currY, currY - cellMaxY);
                    let distToCell = Math.sqrt(cdx * cdx + cdy * cdy);
                    if (distToCell >= bestDist) continue;

                    let cell = grid[r * C + c];
                    for (let i = 0; i < cell.length; i++) {
                        let p = cell[i];
                        let d = getDistCoords(currX, currY, p.x, p.y);
                        if (d < bestDist) {
                            bestDist = d;
                            bestP = p;
                            bestCell = cell;
                            bestIdx = i;
                        }
                    }
                }
            }
        }

        if (bestP) {
            bestCell[bestIdx] = bestCell[bestCell.length - 1];
            bestCell.pop();
            tour.push(bestP);
            currX = bestP.x;
            currY = bestP.y;
            visitedCount++;
        } else {
            // Precision failsafe
            let anyP = null, anyCell = null, anyIdx = -1;
            for(let i = 0; i < grid.length; i++) {
                if(grid[i].length > 0) {
                    anyP = grid[i][0];
                    anyCell = grid[i];
                    anyIdx = 0;
                    break;
                }
            }
            if(anyP) {
                anyCell[anyIdx] = anyCell[anyCell.length - 1];
                anyCell.pop();
                tour.push(anyP);
                currX = anyP.x;
                currY = anyP.y;
                visitedCount++;
            } else {
                break;
            }
        }
    }

    // 2. Heavy Local Search: 2-Opt and Node Relocation to reach near-global optimum
    const maxTime = 4000; // Limit to 4.0s for robustness in tight environments
    const startTime = Date.now();
    let depotNode = { x: depot[0], y: depot[1] };
    let windowLimit = N < 10000 ? N : 600;

    let anyImprovement = true;
    while (anyImprovement && (Date.now() - startTime < maxTime)) {
        anyImprovement = false;
        
        let improved2Opt = true;
        while (improved2Opt && (Date.now() - startTime < maxTime)) {
            improved2Opt = false;
            for (let i = 0; i < N - 1; i++) {
                if ((i & 127) === 0 && Date.now() - startTime >= maxTime) break;
                let maxJ = Math.min(N - 1, i + windowLimit);
                for (let j = i + 1; j <= maxJ; j++) {
                    let p_i_minus_1 = i === 0 ? depotNode : tour[i - 1];
                    let p_i = tour[i];
                    let p_j = tour[j];
                    let p_j_plus_1 = j === N - 1 ? depotNode : tour[j + 1];

                    let d_old = getDist(p_i_minus_1, p_i) + getDist(p_j, p_j_plus_1);
                    let d_new = getDist(p_i_minus_1, p_j) + getDist(p_i, p_j_plus_1);

                    if (d_new < d_old - 1e-9) {
                        let left = i;
                        let right = j;
                        while (left < right) {
                            let temp = tour[left];
                            tour[left] = tour[right];
                            tour[right] = temp;
                            left++;
                            right--;
                        }
                        improved2Opt = true;
                        anyImprovement = true;
                    }
                }
            }
        }

        let improvedRelocate = true;
        while (improvedRelocate && (Date.now() - startTime < maxTime)) {
            improvedRelocate = false;
            for (let i = 0; i < N; i++) {
                if ((i & 127) === 0 && Date.now() - startTime >= maxTime) break;
                
                let p_i = tour[i];
                let p_prev = i === 0 ? depotNode : tour[i - 1];
                let p_next = i === N - 1 ? depotNode : tour[i + 1];
                let d_removed = getDist(p_prev, p_i) + getDist(p_i, p_next) - getDist(p_prev, p_next);
                
                let bestJ = -2;
                let bestSave = 1e-9;
                
                let minJ = Math.max(-1, i - windowLimit);
                let maxJ = Math.min(N - 1, i + windowLimit);
                
                for (let j = minJ; j <= maxJ; j++) {
                    if (j === i || j === i - 1) continue;
                    
                    let p_j = j === -1 ? depotNode : tour[j];
                    let p_j_next = j === N - 1 ? depotNode : tour[j + 1];
                    
                    let d_added = getDist(p_j, p_i) + getDist(p_i, p_j_next) - getDist(p_j, p_j_next);
                    let save = d_removed - d_added;
                    
                    if (save > bestSave) {
                        bestSave = save;
                        bestJ = j;
                    }
                }
                
                if (bestJ !== -2) {
                    let node = tour.splice(i, 1)[0];
                    let insertPos = bestJ > i ? bestJ : bestJ + 1;
                    tour.splice(insertPos, 0, node);
                    improvedRelocate = true;
                    anyImprovement = true;
                    break; 
                }
            }
        }
    }

    // 3. Return a single giant route for the absolute minimum total distance possible
    console.log(JSON.stringify({ routes: [tour.map(p => p.id)] }));
}

solve();
