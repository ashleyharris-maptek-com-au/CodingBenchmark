const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    if (!input.trim()) return;
    const data = JSON.parse(input);
    
    const { width, height, grid, ignitions, towns, cutBudget } = data;
    
    const map = grid.map(row => row.split(''));
    
    const isIgnition = Array.from({ length: height }, () => new Uint8Array(width));
    for (const [ix, iy] of ignitions) {
        if (ix >= 0 && ix < width && iy >= 0 && iy < height) {
            isIgnition[iy][ix] = 1;
        }
    }
    
    const isTown = Array.from({ length: height }, () => new Uint8Array(width));
    for (const t of towns) {
        if (t.x >= 0 && t.x < width && t.y >= 0 && t.y < height) {
            isTown[t.y][t.x] = 1;
        }
    }

    function getReachable() {
        const reachable = Array.from({ length: height }, () => new Uint8Array(width));
        const queue = new Int32Array(width * height * 2);
        let head = 0;
        let tail = 0;
        
        for (const [ix, iy] of ignitions) {
            if (ix >= 0 && ix < width && iy >= 0 && iy < height) {
                reachable[iy][ix] = 1;
                queue[tail++] = ix;
                queue[tail++] = iy;
            }
        }
        
        const dx = [1, -1, 0, 0];
        const dy = [0, 0, 1, -1];
        
        while (head < tail) {
            const cx = queue[head++];
            const cy = queue[head++];
            
            for (let i = 0; i < 4; i++) {
                const nx = cx + dx[i];
                const ny = cy + dy[i];
                if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                    if (!reachable[ny][nx] && map[ny][nx] === '.') {
                        reachable[ny][nx] = 1;
                        queue[tail++] = nx;
                        queue[tail++] = ny;
                    }
                }
            }
        }
        return reachable;
    }
    
    let reachable = getReachable();
    
    const townsInDanger = towns.filter(t => reachable[t.y][t.x]);
    
    const dx = [1, -1, 0, 0];
    const dy = [0, 0, 1, -1];
    
    const townStats = townsInDanger.map(t => {
        let cost = 0;
        const toCut = [];
        for (let i = 0; i < 4; i++) {
            const nx = t.x + dx[i];
            const ny = t.y + dy[i];
            if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
                if (map[ny][nx] === '.') {
                    if (isTown[ny][nx] || isIgnition[ny][nx]) {
                        cost += 1000000;
                    } else {
                        cost++;
                        toCut.push([nx, ny]);
                    }
                }
            }
        }
        return { town: t, cost, toCut, ratio: t.value / (cost || 0.001) };
    });
    
    townStats.sort((a, b) => b.ratio - a.ratio);
    
    let currentBudget = cutBudget;
    const firebreaks = [];
    const cutSet = new Set();
    
    for (const stat of townStats) {
        if (stat.cost <= currentBudget) {
            const newCuts = [];
            for (const [cx, cy] of stat.toCut) {
                const key = `${cx},${cy}`;
                if (!cutSet.has(key)) {
                    newCuts.push([cx, cy]);
                }
            }
            if (newCuts.length <= currentBudget) {
                for (const [cx, cy] of newCuts) {
                    cutSet.add(`${cx},${cy}`);
                    firebreaks.push([cx, cy]);
                    map[cy][cx] = '~';
                    currentBudget--;
                }
            }
        }
    }
    
    console.log(JSON.stringify({ firebreaks }));
}

solve();
