declare const require: any;
const fs = require('fs');

interface Task {
    id: string | number;
    duration: number;
    deps: (string | number)[];
    deadline: number;
    weight: number;
}

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);
    const tasks: Task[] = data.tasks;
    
    // Map tasks by id
    const taskMap = new Map<string | number, Task>();
    for (const t of tasks) {
        taskMap.set(t.id, t);
    }
    
    // Build adj and inDegree
    const adj = new Map<string | number, (string | number)[]>();
    const inDegree = new Map<string | number, number>();
    
    for (const t of tasks) {
        adj.set(t.id, []);
        inDegree.set(t.id, 0);
    }
    
    for (const t of tasks) {
        for (const dep of t.deps) {
            if (adj.has(dep)) {
                adj.get(dep)!.push(t.id);
            }
            if (inDegree.has(t.id)) {
                inDegree.set(t.id, inDegree.get(t.id)! + 1);
            }
        }
    }
    
    // Calculate b_level
    // We need reverse topological order to calculate b_level
    const bLevel = new Map<string | number, number>();
    const outDegree = new Map<string | number, number>();
    const revAdj = new Map<string | number, (string | number)[]>();
    
    for (const t of tasks) {
        outDegree.set(t.id, adj.get(t.id)!.length);
        revAdj.set(t.id, t.deps.filter(d => adj.has(d)));
    }
    
    const revReady: (string | number)[] = [];
    for (const t of tasks) {
        if (outDegree.get(t.id) === 0) {
            revReady.push(t.id);
            bLevel.set(t.id, t.duration);
        } else {
            bLevel.set(t.id, 0);
        }
    }
    
    let head = 0;
    while (head < revReady.length) {
        const curr = revReady[head++];
        const currBLevel = bLevel.get(curr)!;
        
        const deps = revAdj.get(curr)!;
        for (const dep of deps) {
            const depTask = taskMap.get(dep)!;
            const newBLevel = currBLevel + depTask.duration;
            if (newBLevel > bLevel.get(dep)!) {
                bLevel.set(dep, newBLevel);
            }
            
            outDegree.set(dep, outDegree.get(dep)! - 1);
            if (outDegree.get(dep) === 0) {
                revReady.push(dep);
            }
        }
    }
    
    // In case of cycles or disconnected components, ensure all tasks have a baseline b_level
    for (const t of tasks) {
        if (!bLevel.has(t.id)) {
            bLevel.set(t.id, t.duration);
        }
    }
    
    // Now topological sort from start
    const ready: (string | number)[] = [];
    for (const t of tasks) {
        if (inDegree.get(t.id) === 0) {
            ready.push(t.id);
        }
    }
    
    const order: (string | number)[] = [];
    
    // Score function
    const getScore = (id: string | number) => {
        const t = taskMap.get(id)!;
        // High b_level is good, high weight is good, low deadline is good.
        return bLevel.get(id)! * 10000 + t.weight * 100 - t.deadline;
    };
    
    while (ready.length > 0) {
        // Find best in ready
        let bestIdx = 0;
        let bestScore = getScore(ready[0]);
        
        for (let i = 1; i < ready.length; i++) {
            const score = getScore(ready[i]);
            if (score > bestScore) {
                bestScore = score;
                bestIdx = i;
            }
        }
        
        const curr = ready[bestIdx];
        ready[bestIdx] = ready[ready.length - 1];
        ready.pop();
        
        order.push(curr);
        
        for (const next of adj.get(curr)!) {
            const ind = inDegree.get(next)! - 1;
            inDegree.set(next, ind);
            if (ind === 0) {
                ready.push(next);
            }
        }
    }
    
    // If there are cycles, some tasks might be left out, let's just append remaining tasks just in case
    const inOrder = new Set(order);
    if (order.length < tasks.length) {
        for (const t of tasks) {
            if (!inOrder.has(t.id)) {
                order.push(t.id);
            }
        }
    }
    
    console.log(JSON.stringify({ order }));
}

solve();
