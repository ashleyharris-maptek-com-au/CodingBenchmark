const fs = require('fs');

const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const grid = data.grid;
const R = grid.length;
const C = grid[0].length;
const robots = data.robots;
const jobs = data.jobs;

const maxNodes = R * C;
let q0_x = new Int32Array(maxNodes);
let q0_y = new Int32Array(maxNodes);
let q0_d = new Int32Array(maxNodes);

let q2_x = new Int32Array(maxNodes);
let q2_y = new Int32Array(maxNodes);
let q2_d = new Int32Array(maxNodes);

const visited = new Int32Array(maxNodes);
let visitCounter = 0;

const distCache = new Map();

function getDistId(x1, y1, x2, y2) {
    if (x1 < x2 || (x1 === x2 && y1 < y2)) {
        return `${x1},${y1},${x2},${y2}`;
    }
    return `${x2},${y2},${x1},${y1}`;
}

const dx = [0, 0, 1, -1];
const dy = [1, -1, 0, 0];

function getDistAStar(p1, p2) {
    if (p1.x === p2.x && p1.y === p2.y) return 0;
    const id = getDistId(p1.x, p1.y, p2.x, p2.y);
    let cached = distCache.get(id);
    if (cached !== undefined) return cached;

    visitCounter++;
    if (visitCounter > 2000000000) {
        visited.fill(0);
        visitCounter = 1;
    }
    
    let head0 = 0;
    let tail0 = 0;
    let tail2 = 0;

    q0_x[tail0] = p1.x;
    q0_y[tail0] = p1.y;
    q0_d[tail0++] = 0;
    visited[p1.y * C + p1.x] = visitCounter;

    while(head0 < tail0 || tail2 > 0) {
        if (head0 >= tail0) {
            let tmp_x = q0_x; q0_x = q2_x; q2_x = tmp_x;
            let tmp_y = q0_y; q0_y = q2_y; q2_y = tmp_y;
            let tmp_d = q0_d; q0_d = q2_d; q2_d = tmp_d;
            
            tail0 = tail2;
            head0 = 0;
            tail2 = 0;
        }
        
        const x = q0_x[head0];
        const y = q0_y[head0];
        const d = q0_d[head0++];
        
        if (x === p2.x && y === p2.y) {
            distCache.set(id, d);
            return d;
        }
        
        const h_u = Math.abs(x - p2.x) + Math.abs(y - p2.y);
        
        for(let i=0; i<4; i++) {
            const nx = x + dx[i];
            const ny = y + dy[i];
            if (nx >= 0 && nx < C && ny >= 0 && ny < R) {
                if (grid[ny].charCodeAt(nx) !== 35 && visited[ny * C + nx] !== visitCounter) {
                    visited[ny * C + nx] = visitCounter;
                    const h_v = Math.abs(nx - p2.x) + Math.abs(ny - p2.y);
                    if (h_v === h_u - 1) {
                        q0_x[tail0] = nx;
                        q0_y[tail0] = ny;
                        q0_d[tail0++] = d + 1;
                    } else {
                        q2_x[tail2] = nx;
                        q2_y[tail2] = ny;
                        q2_d[tail2++] = d + 1;
                    }
                }
            }
        }
    }
    distCache.set(id, Infinity);
    return Infinity;
}

const assignments = Array.from({length: robots.length}, () => []);
const robotPos = robots.map(r => ({x: r.x, y: r.y}));
const isAssigned = new Uint8Array(jobs.length);
let numAssigned = 0;

const closestJobIdx = new Int32Array(robots.length).fill(-1);
const closestJobDist = new Float64Array(robots.length).fill(Infinity);

function updateClosest(rIdx) {
    let bestDist = Infinity;
    let bestJIdx = -1;
    let rPos = robotPos[rIdx];
    for (let j = 0; j < jobs.length; j++) {
        if (isAssigned[j]) continue;
        let job = jobs[j];
        let h = Math.abs(rPos.x - job.x) + Math.abs(rPos.y - job.y);
        if (h > bestDist) continue; 
        
        let d = getDistAStar(rPos, job);
        if (d < bestDist) {
            bestDist = d;
            bestJIdx = j;
        } else if (d === bestDist) {
            if (bestJIdx !== -1) {
                if (job.priority > jobs[bestJIdx].priority) {
                    bestJIdx = j;
                } else if (job.priority === jobs[bestJIdx].priority && job.id < jobs[bestJIdx].id) {
                    bestJIdx = j;
                }
            }
        }
    }
    closestJobIdx[rIdx] = bestJIdx;
    closestJobDist[rIdx] = bestDist;
}

for (let i = 0; i < robots.length; i++) {
    updateClosest(i);
}

while (numAssigned < jobs.length) {
    let bestDist = Infinity;
    let bestR = -1;
    
    for (let i = 0; i < robots.length; i++) {
        if (closestJobDist[i] < bestDist) {
            bestDist = closestJobDist[i];
            bestR = i;
        } else if (closestJobDist[i] === bestDist && bestDist !== Infinity) {
            let j1 = jobs[closestJobIdx[i]];
            let j2 = jobs[closestJobIdx[bestR]];
            if (j1 && j2) {
                if (j1.priority > j2.priority) {
                    bestR = i;
                } else if (j1.priority === j2.priority && j1.id < j2.id) {
                    bestR = i;
                }
            }
        }
    }
    
    if (bestR === -1 || bestDist === Infinity) {
        for(let j=0; j<jobs.length; j++) {
            if(!isAssigned[j]) {
                assignments[0].push(jobs[j].id);
                isAssigned[j] = 1;
                numAssigned++;
            }
        }
        break;
    }
    
    let jIdx = closestJobIdx[bestR];
    isAssigned[jIdx] = 1;
    numAssigned++;
    
    assignments[bestR].push(jobs[jIdx].id);
    robotPos[bestR] = {x: jobs[jIdx].x, y: jobs[jIdx].y};
    
    if (numAssigned < jobs.length) {
        updateClosest(bestR);
        for (let i = 0; i < robots.length; i++) {
            if (i !== bestR && closestJobIdx[i] === jIdx) {
                updateClosest(i);
            }
        }
    }
}

const jobById = new Map();
for(let job of jobs) jobById.set(job.id, job);

function optimizeRoute(rIdx) {
    let route = assignments[rIdx]; 
    if (route.length < 2) return false;
    
    let anyImproved = false;
    let improved = true;
    let localIterCount = 0;
    while (improved && localIterCount < 100) {
        localIterCount++;
        improved = false;
        for (let i = 0; i < route.length - 1; i++) {
            for (let j = i + 1; j < route.length; j++) {
                let p_prev = i === 0 ? robots[rIdx] : jobById.get(route[i-1]);
                let p_i = jobById.get(route[i]);
                let p_j = jobById.get(route[j]);
                let p_next = j === route.length - 1 ? null : jobById.get(route[j+1]);
                
                let d_current = getDistAStar(p_prev, p_i);
                if (p_next) d_current += getDistAStar(p_j, p_next);
                
                let d_new = getDistAStar(p_prev, p_j);
                if (p_next) d_new += getDistAStar(p_i, p_next);
                
                if (d_new < d_current) {
                    let sub = route.slice(i, j + 1);
                    sub.reverse();
                    for (let k = 0; k < sub.length; k++) {
                        route[i + k] = sub[k];
                    }
                    improved = true;
                    anyImproved = true;
                }
            }
        }
    }
    return anyImproved;
}

function interRouteOptimization() {
    let improved = true;
    let anyImproved = false;
    let localIterCount = 0;
    while(improved && localIterCount < 100) {
        localIterCount++;
        improved = false;
        for(let r1=0; r1<robots.length; r1++) {
            for(let r2=0; r2<robots.length; r2++) {
                if(r1 === r2) continue;
                
                let route1 = assignments[r1];
                let route2 = assignments[r2];
                
                for(let i=0; i<route1.length; i++) {
                    let job_id = route1[i];
                    let job = jobById.get(job_id);
                    let prev1 = i === 0 ? robots[r1] : jobById.get(route1[i-1]);
                    let next1 = i === route1.length-1 ? null : jobById.get(route1[i+1]);
                    
                    let cost_removed = getDistAStar(prev1, job);
                    if(next1) cost_removed += getDistAStar(job, next1) - getDistAStar(prev1, next1);
                    
                    let best_j = -1;
                    let max_savings = 0;
                    
                    for(let j=0; j<=route2.length; j++) {
                        let prev2 = j === 0 ? robots[r2] : jobById.get(route2[j-1]);
                        let next2 = j === route2.length ? null : jobById.get(route2[j]);
                        
                        let cost_added = getDistAStar(prev2, job);
                        if(next2) cost_added += getDistAStar(job, next2) - getDistAStar(prev2, next2);
                        
                        if(cost_added < cost_removed) {
                            let savings = cost_removed - cost_added;
                            if(savings > max_savings) {
                                max_savings = savings;
                                best_j = j;
                            }
                        }
                    }
                    
                    if(best_j !== -1) {
                        route1.splice(i, 1);
                        route2.splice(best_j, 0, job_id);
                        improved = true;
                        anyImproved = true;
                        break;
                    }
                }
                if(improved) break;
            }
            if(improved) break;
        }
    }
    return anyImproved;
}

let globalImproved = true;
let iterCount = 0;
while (globalImproved && iterCount < 20) {
    globalImproved = false;
    for (let r = 0; r < robots.length; r++) {
        if (optimizeRoute(r)) {
            globalImproved = true;
        }
    }
    if (interRouteOptimization()) {
        globalImproved = true;
    }
    iterCount++;
}

let maxId = 0;
for(let r of robots) if(r.id > maxId) maxId = r.id;

const outputAssignments = Array.from({length: maxId + 1}, () => []);
for(let r=0; r<robots.length; r++) {
    outputAssignments[robots[r].id] = assignments[r];
}

console.log(JSON.stringify({assignments: outputAssignments}));
