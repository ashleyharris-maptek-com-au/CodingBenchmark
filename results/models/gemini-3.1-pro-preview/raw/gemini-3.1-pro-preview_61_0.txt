declare const require: any;
const fs = require('fs');

function solve() {
    const input: string = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);

    const depot = data.depot;
    const packages = data.packages;

    if (!packages || packages.length === 0) {
        console.log(JSON.stringify({ routes: [] }));
        return;
    }

    const n = packages.length;
    const unvisitedX = new Float64Array(n);
    const unvisitedY = new Float64Array(n);
    const unvisitedIds = new Array(n);

    for (let i = 0; i < n; i++) {
        unvisitedX[i] = packages[i].x;
        unvisitedY[i] = packages[i].y;
        unvisitedIds[i] = packages[i].id;
    }

    let numUnvisited = n;
    let currX = depot[0];
    let currY = depot[1];

    const routeX = new Float64Array(n + 2);
    const routeY = new Float64Array(n + 2);
    const routeIdsStr = new Array(n + 2);

    routeX[0] = depot[0];
    routeY[0] = depot[1];
    routeIdsStr[0] = "depot";

    for (let step = 0; step < n; step++) {
        let bestDistSq = Infinity;
        let bestIdx = -1;
        for (let i = 0; i < numUnvisited; i++) {
            const dx = unvisitedX[i] - currX;
            const dy = unvisitedY[i] - currY;
            const dSq = dx * dx + dy * dy;
            if (dSq < bestDistSq) {
                bestDistSq = dSq;
                bestIdx = i;
            }
        }

        currX = unvisitedX[bestIdx];
        currY = unvisitedY[bestIdx];
        routeX[step + 1] = currX;
        routeY[step + 1] = currY;
        routeIdsStr[step + 1] = unvisitedIds[bestIdx];

        numUnvisited--;
        unvisitedX[bestIdx] = unvisitedX[numUnvisited];
        unvisitedY[bestIdx] = unvisitedY[numUnvisited];
        unvisitedIds[bestIdx] = unvisitedIds[numUnvisited];
    }

    routeX[n + 1] = depot[0];
    routeY[n + 1] = depot[1];
    routeIdsStr[n + 1] = "depot";

    const startTime = Date.now();
    let improved = true;
    const N_pts = n + 2;

    while (improved) {
        improved = false;
        for (let i = 0; i < N_pts - 2; i++) {
            const p1x = routeX[i], p1y = routeY[i];
            let p2x = routeX[i + 1], p2y = routeY[i + 1];
            let d12 = Math.sqrt((p1x - p2x) * (p1x - p2x) + (p1y - p2y) * (p1y - p2y));

            for (let j = i + 2; j < N_pts - 1; j++) {
                const p3x = routeX[j], p3y = routeY[j];
                const p4x = routeX[j + 1], p4y = routeY[j + 1];
                const d34 = Math.sqrt((p3x - p4x) * (p3x - p4x) + (p3y - p4y) * (p3y - p4y));

                const max_d13 = d12 + d34;
                const d13_sq = (p1x - p3x) * (p1x - p3x) + (p1y - p3y) * (p1y - p3y);

                if (d13_sq >= max_d13 * max_d13) continue;

                const d13 = Math.sqrt(d13_sq);
                const d24 = Math.sqrt((p2x - p4x) * (p2x - p4x) + (p2y - p4y) * (p2y - p4y));

                if (d13 + d24 < d12 + d34 - 1e-9) {
                    let left = i + 1;
                    let right = j;
                    while (left < right) {
                        const tx = routeX[left]; routeX[left] = routeX[right]; routeX[right] = tx;
                        const ty = routeY[left]; routeY[left] = routeY[right]; routeY[right] = ty;
                        const tid = routeIdsStr[left]; routeIdsStr[left] = routeIdsStr[right]; routeIdsStr[right] = tid;
                        left++;
                        right--;
                    }
                    improved = true;
                    p2x = routeX[i + 1];
                    p2y = routeY[i + 1];
                    d12 = Math.sqrt((p1x - p2x) * (p1x - p2x) + (p1y - p2y) * (p1y - p2y));
                }
            }
            if ((i & 1023) === 0 && Date.now() - startTime > 8000) {
                improved = false;
                break;
            }
        }
        if (Date.now() - startTime > 8000) break;
    }

    const finalRoute = [];
    for (let i = 1; i <= n; i++) {
        finalRoute.push(routeIdsStr[i]);
    }

    console.log(JSON.stringify({ routes: [finalRoute] }));
}

solve();
