const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    if (!input.trim()) return;
    const data = JSON.parse(input);
    
    const prob = data.probability;
    const height = prob.length;
    const width = prob[0].length;
    const drones = data.drones;
    const maxSteps = data.maxStepsPerDrone;
    
    const visited = Array.from({ length: height }, () => new Array(width).fill(false));
    
    const paths = drones.map(d => [[d.x, d.y]]);
    for (const d of drones) {
        visited[d.y][d.x] = true;
    }
    
    const dx = [0, 1, 0, -1, 0];
    const dy = [-1, 0, 1, 0, 0]; // Up, Right, Down, Left, Stay
    
    for (let step = 0; step < maxSteps; step++) {
        for (let i = 0; i < drones.length; i++) {
            const currentPath = paths[i];
            const [cx, cy] = currentPath[currentPath.length - 1];
            
            let bestScore = -1;
            let bestMove = [cx, cy];
            
            const localVis = new Set();
            
            function dfs(x, y, depth, currentScore, firstMove) {
                if (currentScore > bestScore) {
                    bestScore = currentScore;
                    bestMove = firstMove;
                }
                if (depth === 0) return;
                
                for (let dir = 0; dir < 5; dir++) {
                    const nx = x + dx[dir];
                    const ny = y + dy[dir];
                    if (nx >= 0 && nx < width && ny >= 0 && ny < height && prob[ny][nx] !== -1) {
                        const key = ny * width + nx;
                        const wasVisited = visited[ny][nx] || localVis.has(key);
                        let nextScore = currentScore;
                        if (!wasVisited) {
                            nextScore += prob[ny][nx];
                            localVis.add(key);
                        }
                        
                        dfs(nx, ny, depth - 1, nextScore, firstMove || [nx, ny]);
                        
                        if (!wasVisited) {
                            localVis.delete(key);
                        }
                    }
                }
            }
            
            dfs(cx, cy, 6, 0, null);
            
            if (bestScore === 0) {
                // BFS to find nearest unvisited with positive prob
                let queue = [[cx, cy]];
                let qHead = 0;
                let bfsVis = new Uint8Array(width * height);
                bfsVis[cy * width + cx] = 1;
                let parent = new Map();
                let target = null;
                
                while (qHead < queue.length && qHead < 2000) { // limit search
                    let [qx, qy] = queue[qHead++];
                    if (!visited[qy][qx] && prob[qy][qx] > 0) {
                        target = [qx, qy];
                        break;
                    }
                    for (let dir = 0; dir < 4; dir++) { // no stay
                        const nx = qx + dx[dir];
                        const ny = qy + dy[dir];
                        if (nx >= 0 && nx < width && ny >= 0 && ny < height && prob[ny][nx] !== -1) {
                            const key = ny * width + nx;
                            if (!bfsVis[key]) {
                                bfsVis[key] = 1;
                                parent.set(key, [qx, qy, nx, ny]); 
                                queue.push([nx, ny]);
                            }
                        }
                    }
                }
                
                if (target) {
                    let currKey = target[1] * width + target[0];
                    while (true) {
                        let p = parent.get(currKey);
                        if (!p) break;
                        if (p[0] === cx && p[1] === cy) {
                            bestMove = [p[2], p[3]];
                            break;
                        }
                        currKey = p[1] * width + p[0];
                    }
                }
            }
            
            const [nx, ny] = bestMove;
            paths[i].push([nx, ny]);
            visited[ny][nx] = true;
        }
    }
    
    console.log(JSON.stringify({ paths }));
}

solve();
