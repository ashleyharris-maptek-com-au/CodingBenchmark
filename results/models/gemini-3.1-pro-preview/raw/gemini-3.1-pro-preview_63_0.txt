declare const require: any;
const fs = require('fs');

interface Robot {
    id: number;
    x: number;
    y: number;
}

interface Job {
    id: number;
    x: number;
    y: number;
    priority: number;
}

function runAll() {
    const input: string = fs.readFileSync(0, 'utf8');
    const data = JSON.parse(input);
    const grid: string[] = data.grid;
    const robots: Robot[] = data.robots;
    const jobs: Job[] = data.jobs;

    if (robots.length === 0) {
        console.log(JSON.stringify({ assignments: [] }));
        return;
    }

    const R = grid.length;
    const C = grid[0].length;

    const visited = new Int32Array(R * C);
    visited.fill(0);
    let globalVisitId = 0;

    const queue = new Int32Array(R * C);
    const dist = new Int32Array(R * C);
    
    function solveGreedy(jobSortFn: (a: Job, b: Job) => number) {
        const robotPositions = new Int32Array(robots.length);
        const robotIds = new Int32Array(robots.length);

        const headRobotAt = new Int32Array(R * C);
        headRobotAt.fill(-1);
        const nextRobot = new Int32Array(robots.length);

        for (let i = 0; i < robots.length; i++) {
            const r = robots[i];
            const cIdx = r.y * C + r.x;
            robotPositions[i] = cIdx;
            robotIds[i] = r.id;
            
            nextRobot[i] = headRobotAt[cIdx];
            headRobotAt[cIdx] = i;
        }

        function moveRobot(rIdx: number, fromIdx: number, toIdx: number) {
            let prev = -1;
            let curr = headRobotAt[fromIdx];
            while (curr !== -1) {
                if (curr === rIdx) {
                    if (prev === -1) {
                        headRobotAt[fromIdx] = nextRobot[curr];
                    } else {
                        nextRobot[prev] = nextRobot[curr];
                    }
                    break;
                }
                prev = curr;
                curr = nextRobot[curr];
            }
            
            nextRobot[rIdx] = headRobotAt[toIdx];
            headRobotAt[toIdx] = rIdx;
            robotPositions[rIdx] = toIdx;
        }

        const sortedJobs = [...jobs].sort(jobSortFn);

        const assignments = new Map<number, number[]>();
        for (const r of robots) assignments.set(r.id, []);

        let totalTravel = 0;

        for (const job of sortedJobs) {
            globalVisitId++;
            const visitId = globalVisitId;
            const startIdx = job.y * C + job.x;
            
            let head = 0;
            let tail = 0;
            
            queue[tail++] = startIdx;
            visited[startIdx] = visitId;
            dist[startIdx] = 0;
            
            let foundDist = -1;
            let bestRId = Infinity;
            let bestRIdx = -1;
            
            while (head < tail) {
                const curr = queue[head++];
                const d = dist[curr];
                
                if (foundDist !== -1 && d > foundDist) {
                    break;
                }
                
                let rCurr = headRobotAt[curr];
                while (rCurr !== -1) {
                    foundDist = d;
                    const rId = robotIds[rCurr];
                    if (rId < bestRId) {
                        bestRId = rId;
                        bestRIdx = rCurr;
                    }
                    rCurr = nextRobot[rCurr];
                }
                
                if (foundDist === -1) {
                    const cy = Math.floor(curr / C);
                    const cx = curr % C;
                    
                    if (cy > 0) {
                        const next = curr - C;
                        if (grid[cy - 1][cx] !== '#' && visited[next] !== visitId) {
                            visited[next] = visitId;
                            dist[next] = d + 1;
                            queue[tail++] = next;
                        }
                    }
                    if (cy < R - 1) {
                        const next = curr + C;
                        if (grid[cy + 1][cx] !== '#' && visited[next] !== visitId) {
                            visited[next] = visitId;
                            dist[next] = d + 1;
                            queue[tail++] = next;
                        }
                    }
                    if (cx > 0) {
                        const next = curr - 1;
                        if (grid[cy][cx - 1] !== '#' && visited[next] !== visitId) {
                            visited[next] = visitId;
                            dist[next] = d + 1;
                            queue[tail++] = next;
                        }
                    }
                    if (cx < C - 1) {
                        const next = curr + 1;
                        if (grid[cy][cx + 1] !== '#' && visited[next] !== visitId) {
                            visited[next] = visitId;
                            dist[next] = d + 1;
                            queue[tail++] = next;
                        }
                    }
                }
            }
            
            if (bestRIdx !== -1) {
                assignments.get(bestRId)!.push(job.id);
                moveRobot(bestRIdx, robotPositions[bestRIdx], startIdx);
                totalTravel += foundDist;
            } else {
                let fallbackRIdx = 0;
                let fallbackRId = robotIds[0];
                for (let i = 1; i < robots.length; i++) {
                    if (robotIds[i] < fallbackRId) {
                        fallbackRId = robotIds[i];
                        fallbackRIdx = i;
                    }
                }
                assignments.get(fallbackRId)!.push(job.id);
                moveRobot(fallbackRIdx, robotPositions[fallbackRIdx], startIdx);
                totalTravel += 1e9;
            }
        }
        
        return { assignments, totalTravel };
    }

    function solveGreedyRoundRobin() {
        const robotPositions = new Int32Array(robots.length);
        const robotIds = new Int32Array(robots.length);
        const robotSorted = [...robots].sort((a, b) => a.id - b.id);
        
        for (let i = 0; i < robotSorted.length; i++) {
            const r = robotSorted[i];
            robotPositions[i] = r.y * C + r.x;
            robotIds[i] = r.id;
        }

        const assignments = new Map<number, number[]>();
        for (const r of robots) assignments.set(r.id, []);

        let totalTravel = 0;

        const jobAssigned = new Uint8Array(jobs.length);
        let unassignedCount = jobs.length;

        const headJobAt = new Int32Array(R * C);
        headJobAt.fill(-1);
        const nextJob = new Int32Array(jobs.length);
        for (let i = 0; i < jobs.length; i++) {
            const j = jobs[i];
            const cIdx = j.y * C + j.x;
            nextJob[i] = headJobAt[cIdx];
            headJobAt[cIdx] = i;
        }

        function removeJob(jIdx: number, cIdx: number) {
            let prev = -1;
            let curr = headJobAt[cIdx];
            while (curr !== -1) {
                if (curr === jIdx) {
                    if (prev === -1) {
                        headJobAt[cIdx] = nextJob[curr];
                    } else {
                        nextJob[prev] = nextJob[curr];
                    }
                    break;
                }
                prev = curr;
                curr = nextJob[curr];
            }
        }

        while (unassignedCount > 0) {
            let assignedInRound = 0;
            for (let i = 0; i < robotSorted.length; i++) {
                if (unassignedCount === 0) break;

                globalVisitId++;
                const visitId = globalVisitId;
                const startIdx = robotPositions[i];
                
                let head = 0;
                let tail = 0;
                
                queue[tail++] = startIdx;
                visited[startIdx] = visitId;
                dist[startIdx] = 0;
                
                let foundDist = -1;
                let bestJId = Infinity;
                let bestJIdx = -1;
                let bestJPriority = -Infinity;
                
                while (head < tail) {
                    const curr = queue[head++];
                    const d = dist[curr];
                    
                    if (foundDist !== -1 && d > foundDist) {
                        break;
                    }
                    
                    let jCurr = headJobAt[curr];
                    while (jCurr !== -1) {
                        if (jobAssigned[jCurr] === 0) {
                            foundDist = d;
                            const job = jobs[jCurr];
                            
                            let better = false;
                            if (bestJIdx === -1) {
                                better = true;
                            } else {
                                if (job.priority > bestJPriority) {
                                    better = true;
                                } else if (job.priority === bestJPriority) {
                                    if (job.id < bestJId) better = true;
                                }
                            }
                            
                            if (better) {
                                bestJId = job.id;
                                bestJIdx = jCurr;
                                bestJPriority = job.priority;
                            }
                        }
                        jCurr = nextJob[jCurr];
                    }
                    
                    if (foundDist === -1) {
                        const cy = Math.floor(curr / C);
                        const cx = curr % C;
                        
                        if (cy > 0) {
                            const next = curr - C;
                            if (grid[cy - 1][cx] !== '#' && visited[next] !== visitId) {
                                visited[next] = visitId;
                                dist[next] = d + 1;
                                queue[tail++] = next;
                            }
                        }
                        if (cy < R - 1) {
                            const next = curr + C;
                            if (grid[cy + 1][cx] !== '#' && visited[next] !== visitId) {
                                visited[next] = visitId;
                                dist[next] = d + 1;
                                queue[tail++] = next;
                            }
                        }
                        if (cx > 0) {
                            const next = curr - 1;
                            if (grid[cy][cx - 1] !== '#' && visited[next] !== visitId) {
                                visited[next] = visitId;
                                dist[next] = d + 1;
                                queue[tail++] = next;
                            }
                        }
                        if (cx < C - 1) {
                            const next = curr + 1;
                            if (grid[cy][cx + 1] !== '#' && visited[next] !== visitId) {
                                visited[next] = visitId;
                                dist[next] = d + 1;
                                queue[tail++] = next;
                            }
                        }
                    }
                }
                
                if (bestJIdx !== -1) {
                    assignments.get(robotIds[i])!.push(bestJId);
                    jobAssigned[bestJIdx] = 1;
                    const job = jobs[bestJIdx];
                    removeJob(bestJIdx, job.y * C + job.x);
                    robotPositions[i] = job.y * C + job.x;
                    totalTravel += foundDist;
                    unassignedCount--;
                    assignedInRound++;
                }
            }
            if (assignedInRound === 0) break;
        }
        
        if (unassignedCount > 0) {
            for (let i = 0; i < jobs.length; i++) {
                if (jobAssigned[i] === 0) {
                    assignments.get(robotIds[0])!.push(jobs[i].id);
                    totalTravel += 1e9;
                }
            }
        }
        
        return { assignments, totalTravel };
    }

    const variations = [
        (a: Job, b: Job) => { if (a.priority !== b.priority) return b.priority - a.priority; return a.id - b.id; },
        (a: Job, b: Job) => { if (a.priority !== b.priority) return a.priority - b.priority; return a.id - b.id; },
        (a: Job, b: Job) => a.id - b.id,
        (a: Job, b: Job) => b.id - a.id
    ];

    let bestTravel = Infinity;
    let bestAssignments = null;

    for (const fn of variations) {
        const res = solveGreedy(fn);
        if (res.totalTravel < bestTravel) {
            bestTravel = res.totalTravel;
            bestAssignments = res.assignments;
        }
    }

    const resRoundRobin = solveGreedyRoundRobin();
    if (resRoundRobin.totalTravel < bestTravel) {
        bestTravel = resRoundRobin.totalTravel;
        bestAssignments = resRoundRobin.assignments;
    }

    let maxId = -1;
    for (const r of robots) {
        if (r.id > maxId) maxId = r.id;
    }
    const outputArr: number[][] = [];
    for (let i = 0; i <= maxId; i++) {
        outputArr.push(bestAssignments!.get(i) || []);
    }
    
    console.log(JSON.stringify({ assignments: outputArr }));
}

runAll();
