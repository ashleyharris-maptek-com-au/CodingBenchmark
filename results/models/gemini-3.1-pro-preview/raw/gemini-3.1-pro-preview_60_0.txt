const fs = require('fs');

function solve() {
    const input = fs.readFileSync(0, 'utf8');
    if (!input.trim()) return;
    const data = JSON.parse(input);
    const tasks = data.tasks;

    const taskMap = new Map();
    const inDegree = new Map();
    const adj = new Map();

    for (const t of tasks) {
        taskMap.set(t.id, t);
        inDegree.set(t.id, 0);
        adj.set(t.id, []);
    }

    for (const t of tasks) {
        if (t.deps) {
            for (const dep of t.deps) {
                if (inDegree.has(t.id)) {
                    inDegree.set(t.id, inDegree.get(t.id) + 1);
                }
                if (adj.has(dep)) {
                    adj.get(dep).push(t.id);
                }
            }
        }
    }

    const memo = new Map();
    function getCP(id) {
        if (memo.has(id)) return memo.get(id);
        let max = 0;
        for (const next of adj.get(id)) {
            max = Math.max(max, getCP(next));
        }
        const res = taskMap.get(id).duration + max;
        memo.set(id, res);
        return res;
    }

    for (const t of tasks) {
        getCP(t.id);
    }

    const ready = [];
    for (const t of tasks) {
        if (inDegree.get(t.id) === 0) {
            ready.push(t.id);
        }
    }

    const order = [];
    while (ready.length > 0) {
        ready.sort((a, b) => {
            const ta = taskMap.get(a);
            const tb = taskMap.get(b);
            const sa = ta.deadline - ta.weight * 10 - getCP(a) * 2;
            const sb = tb.deadline - tb.weight * 10 - getCP(b) * 2;
            return sa - sb;
        });
        
        const curr = ready.shift();
        order.push(curr);
        
        for (const next of adj.get(curr)) {
            inDegree.set(next, inDegree.get(next) - 1);
            if (inDegree.get(next) === 0) {
                ready.push(next);
            }
        }
    }

    console.log(JSON.stringify({ order }));
}

solve();
