import math

class PIDController:
    def __init__(self, kp, ki, kd, out_min, out_max):
        self.kp = kp
        self.ki = ki
        self.kd = kd
        self.out_min = out_min
        self.out_max = out_max
        self.integral = 0.0
        self.prev_error = 0.0

    def compute(self, error, dt):
        if dt <= 0:
            return 0.0
        self.integral += error * dt
        # Anti-windup
        if self.ki != 0:
            i_max = self.out_max / self.ki
            i_min = self.out_min / self.ki
            self.integral = max(min(self.integral, i_max), i_min)
        
        derivative = (error - self.prev_error) / dt
        self.prev_error = error
        
        output = self.kp * error + self.ki * self.integral + self.kd * derivative
        return max(min(output, self.out_max), self.out_min)

pid_roll = PIDController(0.03, 0.001, 0.02, -1.0, 1.0)
pid_hdg = PIDController(1.5, 0.0, 0.0, -30.0, 30.0) 
pid_vs = PIDController(0.001, 0.0001, 0.0005, -1.0, 1.0)
pid_alt = PIDController(8.0, 0.0, 0.0, -2500.0, 2500.0) 
pid_ias = PIDController(0.05, 0.005, 0.01, -0.5, 0.5) 

def get_consensus(v1, v2, v3, threshold):
    d12 = abs(v1 - v2)
    d23 = abs(v2 - v3)
    d13 = abs(v1 - v3)
    
    if d12 < threshold and d13 < threshold and d23 < threshold:
        return sorted([v1, v2, v3])[1]
    elif d12 < threshold:
        return (v1 + v2) / 2.0
    elif d23 < threshold:
        return (v2 + v3) / 2.0
    elif d13 < threshold:
        return (v1 + v3) / 2.0
    else:
        return sorted([v1, v2, v3])[1]

def get_sensor(state, prefix, suffix, threshold=None):
    v1 = state.get(f"{prefix}_1_{suffix}", 0.0)
    v2 = state.get(f"{prefix}_2_{suffix}", 0.0)
    v3 = state.get(f"{prefix}_3_{suffix}", 0.0)
    
    if threshold is not None:
        return get_consensus(v1, v2, v3, threshold)
    return sorted([v1, v2, v3])[1]

def angle_diff(target, current):
    diff = (target - current) % 360
    if diff > 180:
        diff -= 360
    return diff

def autopilot_step(state: dict, dt: float) -> dict:
    if not state.get('ap_engaged', False):
        pid_roll.integral = 0.0
        pid_hdg.integral = 0.0
        pid_vs.integral = 0.0
        pid_alt.integral = 0.0
        pid_ias.integral = 0.0
        return {}

    # Sensor Fusion
    alt_baro = get_sensor(state, 'baro_alt', 'ft', threshold=200)
    gps_alt = get_sensor(state, 'gps', 'alt_ft', threshold=200)
    radar_alt = get_sensor(state, 'radar_alt', 'ft', threshold=50)
    
    # Cross-check altitude sources
    if abs(alt_baro - gps_alt) > 1000:
        # If baro is wildly different from GPS, rely on GPS for safety
        alt = gps_alt
    else:
        alt = alt_baro

    irs_heading = get_sensor(state, 'irs', 'heading_deg', threshold=5)
    irs_roll = get_sensor(state, 'irs', 'roll_deg', threshold=5)
    irs_pitch = get_sensor(state, 'irs', 'pitch_deg', threshold=5)
    
    ias = get_sensor(state, 'ias', 'kt', threshold=20)
    gps_gs = get_sensor(state, 'gps', 'gs_kt', threshold=20)
    
    # Pitot tube icing / failure logic
    if ias < 80 and gps_gs > 150:
        ias = gps_gs
        
    vs_fpm = get_sensor(state, 'irs', 'vs_fpm', threshold=300)
    pitch_rate = get_sensor(state, 'gyro', 'pitch_rate_dps', threshold=2)
    roll_rate = get_sensor(state, 'gyro', 'roll_rate_dps', threshold=2)
    yaw_rate = get_sensor(state, 'gyro', 'yaw_rate_dps', threshold=2)

    target_alt = state.get('target_alt_ft', alt)
    target_hdg = state.get('target_hdg_deg', irs_heading)
    target_ias = state.get('target_ias_kt', ias)

    controls = {}

    # Lateral Control
    hdg_error = angle_diff(target_hdg, irs_heading)
    target_roll = pid_hdg.compute(hdg_error, dt)
    roll_error = target_roll - irs_roll
    
    # Combine PID with roll rate damping
    aileron_cmd = pid_roll.compute(roll_error, dt) - roll_rate * 0.02
    controls['aileron'] = max(-1.0, min(aileron_cmd, 1.0))

    # Longitudinal Control
    alt_error = target_alt - alt
    target_vs = pid_alt.compute(alt_error, dt)
    
    # Pitch dampening based on pitch rate
    vs_error = target_vs - vs_fpm
    elevator_cmd = pid_vs.compute(vs_error, dt) - pitch_rate * 0.05
    controls['elevator'] = max(-1.0, min(elevator_cmd, 1.0))
    
    # Auto-trim integrates elevator command slowly
    controls['elevator_trim'] = max(-1.0, min(controls['elevator'] * 0.05, 1.0))

    # Thrust Control
    if state.get('at_engaged', False):
        ias_error = target_ias - ias
        throttle_delta = pid_ias.compute(ias_error, dt)
        
        cur_t1 = state.get('throttle_1_pos', 0.5)
        cur_t2 = state.get('throttle_2_pos', 0.5)
        
        t1 = max(0.0, min(cur_t1 + throttle_delta * dt, 1.0))
        t2 = max(0.0, min(cur_t2 + throttle_delta * dt, 1.0))
        
        controls['throttle_1'] = t1
        controls['throttle_2'] = t2

    # Yaw Damper (Coordinated Turn)
    V_fps = max(ias, 80) * 1.68781
    target_yaw_rate = math.degrees((32.2 * math.tan(math.radians(irs_roll))) / V_fps)
    yaw_error = target_yaw_rate - yaw_rate
    controls['rudder'] = max(-1.0, min(yaw_error * 0.05, 1.0))

    return controls
