declare const require: any;
declare const process: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type Robot = { id: number; x: number; y: number };
type Job = { id: number; x: number; y: number; priority: number };

const grid: string[] = data.grid || [];
const robots: Robot[] = (data.robots || []).slice().sort((a: Robot, b: Robot) => a.id - b.id);
const jobs: Job[] = data.jobs || [];

const h = grid.length;
const w = h > 0 ? grid[0].length : 0;
const pointCount = robots.length + jobs.length;
const pointsX = new Int32Array(pointCount);
const pointsY = new Int32Array(pointCount);
for (let i = 0; i < robots.length; i++) {
  pointsX[i] = robots[i].x;
  pointsY[i] = robots[i].y;
}
for (let i = 0; i < jobs.length; i++) {
  pointsX[robots.length + i] = jobs[i].x;
  pointsY[robots.length + i] = jobs[i].y;
}

const cellCount = w * h;
const passable = new Uint8Array(cellCount);
for (let y = 0; y < h; y++) {
  const row = grid[y] || "";
  for (let x = 0; x < w; x++) {
    passable[y * w + x] = row.charCodeAt(x) === 35 ? 0 : 1;
  }
}

const byCell = new Map<number, number[]>();
for (let i = 0; i < pointCount; i++) {
  const key = pointsY[i] * w + pointsX[i];
  let list = byCell.get(key);
  if (list === undefined) {
    list = [];
    byCell.set(key, list);
  }
  list.push(i);
}

const distRows: Array<Int32Array | undefined> = new Array(pointCount);
const bfsDist = new Int32Array(cellCount);
const seen = new Int32Array(cellCount);
const queue = new Int32Array(cellCount);
let stamp = 0;

function buildRow(source: number): Int32Array {
  const existing = distRows[source];
  if (existing !== undefined) return existing;

  const row = new Int32Array(pointCount);
  row.fill(1 << 28);
  if (source < 0 || source >= pointCount || w === 0 || h === 0) {
    distRows[source] = row;
    return row;
  }

  stamp++;
  if (stamp === 2147483000) {
    seen.fill(0);
    stamp = 1;
  }

  const sx = pointsX[source];
  const sy = pointsY[source];
  const start = sy * w + sx;
  if (sx < 0 || sx >= w || sy < 0 || sy >= h || passable[start] === 0) {
    distRows[source] = row;
    return row;
  }

  let head = 0;
  let tail = 0;
  queue[tail++] = start;
  seen[start] = stamp;
  bfsDist[start] = 0;
  let remaining = pointCount;

  while (head < tail && remaining > 0) {
    const c = queue[head++];
    const d = bfsDist[c];
    const here = byCell.get(c);
    if (here !== undefined) {
      for (let i = 0; i < here.length; i++) {
        const p = here[i];
        if (row[p] !== d) {
          row[p] = d;
          remaining--;
        }
      }
    }

    const x = c % w;
    let n = c - w;
    if (c >= w && passable[n] && seen[n] !== stamp) {
      seen[n] = stamp;
      bfsDist[n] = d + 1;
      queue[tail++] = n;
    }
    n = c + w;
    if (n < cellCount && passable[n] && seen[n] !== stamp) {
      seen[n] = stamp;
      bfsDist[n] = d + 1;
      queue[tail++] = n;
    }
    n = c - 1;
    if (x > 0 && passable[n] && seen[n] !== stamp) {
      seen[n] = stamp;
      bfsDist[n] = d + 1;
      queue[tail++] = n;
    }
    n = c + 1;
    if (x + 1 < w && passable[n] && seen[n] !== stamp) {
      seen[n] = stamp;
      bfsDist[n] = d + 1;
      queue[tail++] = n;
    }
  }

  distRows[source] = row;
  return row;
}

function dist(a: number, b: number): number {
  if (a === b) return 0;
  return buildRow(a)[b];
}

const robotCount = robots.length;
const jobCount = jobs.length;
const assignments: number[][] = Array.from({ length: robotCount }, () => []);
const routeCost: number[] = new Array(robotCount).fill(0);

if (robotCount > 0 && jobCount > 0) {
  const order = Array.from({ length: jobCount }, (_, i) => i);
  order.sort((a, b) => {
    const pr = (jobs[b].priority || 0) - (jobs[a].priority || 0);
    if (pr !== 0) return pr;
    return jobs[a].id - jobs[b].id;
  });

  const endPoint = new Int32Array(robotCount);
  for (let r = 0; r < robotCount; r++) endPoint[r] = r;

  for (const ji of order) {
    const p = robotCount + ji;
    let bestR = 0;
    let bestScore = Number.POSITIVE_INFINITY;
    for (let r = 0; r < robotCount; r++) {
      const d = dist(endPoint[r], p);
      const score = routeCost[r] + d * 1.35 + assignments[r].length * 0.05;
      if (score < bestScore || (score === bestScore && robots[r].id < robots[bestR].id)) {
        bestScore = score;
        bestR = r;
      }
    }
    assignments[bestR].push(ji);
    routeCost[bestR] += dist(endPoint[bestR], p);
    endPoint[bestR] = p;
  }

  function recomputeRoute(r: number): number {
    let cost = 0;
    let prev = r;
    const route = assignments[r];
    for (let i = 0; i < route.length; i++) {
      const p = robotCount + route[i];
      cost += dist(prev, p);
      prev = p;
    }
    routeCost[r] = cost;
    return cost;
  }

  function routeInsertionDelta(r: number, jobIndex: number, pos: number): number {
    const route = assignments[r];
    const p = robotCount + jobIndex;
    const before = pos === 0 ? r : robotCount + route[pos - 1];
    if (pos === route.length) return dist(before, p);
    const after = robotCount + route[pos];
    return dist(before, p) + dist(p, after) - dist(before, after);
  }

  const improveUntil = Date.now() + 1850;
  let changed = true;
  while (changed && Date.now() < improveUntil) {
    changed = false;

    for (let r = 0; r < robotCount && Date.now() < improveUntil; r++) {
      const route = assignments[r];
      for (let i = 0; i + 2 < route.length && Date.now() < improveUntil; i++) {
        const a = i === 0 ? r : robotCount + route[i - 1];
        const b = robotCount + route[i];
        for (let k = i + 1; k + 1 < route.length; k++) {
          const c = robotCount + route[k];
          const d = robotCount + route[k + 1];
          const oldCost = dist(a, b) + dist(c, d);
          const newCost = dist(a, c) + dist(b, d);
          if (newCost + 1e-9 < oldCost) {
            let lo = i;
            let hi = k;
            while (lo < hi) {
              const tmp = route[lo];
              route[lo] = route[hi];
              route[hi] = tmp;
              lo++;
              hi--;
            }
            recomputeRoute(r);
            changed = true;
          }
        }
      }
    }

    for (let from = 0; from < robotCount && Date.now() < improveUntil; from++) {
      for (let idx = 0; idx < assignments[from].length && Date.now() < improveUntil; idx++) {
        const jobIndex = assignments[from][idx];
        const prev = idx === 0 ? from : robotCount + assignments[from][idx - 1];
        const cur = robotCount + jobIndex;
        const removeDelta = idx + 1 < assignments[from].length
          ? dist(prev, robotCount + assignments[from][idx + 1]) - dist(prev, cur) - dist(cur, robotCount + assignments[from][idx + 1])
          : -dist(prev, cur);

        let bestTo = from;
        let bestPos = idx;
        let bestDelta = 0;
        for (let to = 0; to < robotCount; to++) {
          const maxPos = assignments[to].length + (to === from ? 0 : 1);
          for (let pos = 0; pos <= maxPos; pos++) {
            if (to === from && (pos === idx || pos === idx + 1)) continue;
            let delta: number;
            if (to === from) {
              const saved = assignments[from].splice(idx, 1)[0];
              const adjPos = pos > idx ? pos - 1 : pos;
              delta = -routeCost[from];
              assignments[from].splice(adjPos, 0, saved);
              delta += recomputeRoute(from);
              assignments[from].splice(adjPos, 1);
              assignments[from].splice(idx, 0, saved);
              recomputeRoute(from);
            } else {
              delta = removeDelta + routeInsertionDelta(to, jobIndex, pos);
            }
            if (delta < bestDelta) {
              bestDelta = delta;
              bestTo = to;
              bestPos = pos;
            }
          }
        }

        if (bestDelta < -1e-9) {
          const moved = assignments[from].splice(idx, 1)[0];
          if (bestTo === from) {
            const pos = bestPos > idx ? bestPos - 1 : bestPos;
            assignments[from].splice(pos, 0, moved);
            recomputeRoute(from);
          } else {
            assignments[bestTo].splice(bestPos, 0, moved);
            recomputeRoute(from);
            recomputeRoute(bestTo);
          }
          changed = true;
          idx = Math.max(-1, idx - 2);
        }
      }
    }
  }
}

const byRobotId: number[][] = [];
for (let i = 0; i < robotCount; i++) {
  byRobotId[robots[i].id] = assignments[i].map((ji) => jobs[ji].id);
}
for (let i = 0; i < byRobotId.length; i++) {
  if (byRobotId[i] === undefined) byRobotId[i] = [];
}

process.stdout.write(JSON.stringify({ assignments: byRobotId }));
