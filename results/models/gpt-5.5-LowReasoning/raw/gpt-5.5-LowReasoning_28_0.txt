use std::collections::VecDeque;
use std::io::{self, Read, Write};

struct Scanner {
    data: Vec<u8>,
    pos: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut data = Vec::new();
        io::stdin().read_to_end(&mut data).unwrap();
        Self { data, pos: 0 }
    }

    fn next_usize(&mut self) -> Option<usize> {
        while self.pos < self.data.len() && self.data[self.pos].is_ascii_whitespace() {
            self.pos += 1;
        }
        if self.pos == self.data.len() {
            return None;
        }
        let mut value = 0usize;
        while self.pos < self.data.len() && !self.data[self.pos].is_ascii_whitespace() {
            value = value * 10 + usize::from(self.data[self.pos] - b'0');
            self.pos += 1;
        }
        Some(value)
    }
}

fn main() {
    let mut input = Scanner::new();
    let Some(n) = input.next_usize() else {
        return;
    };
    let m = input.next_usize().unwrap_or(0);

    let mut edges_u: Vec<usize> = Vec::with_capacity(m);
    let mut edges_v: Vec<usize> = Vec::with_capacity(m);
    let mut degree = vec![0usize; n];

    for _ in 0..m {
        let u = input.next_usize().unwrap_or(0);
        let v = input.next_usize().unwrap_or(0);
        if u >= n || v >= n {
            continue;
        }
        let id = edges_u.len();
        edges_u.push(u);
        edges_v.push(v);
        degree[u] += 1;
        if u != v {
            degree[v] += 1;
        }
        if id + 1 == m {
            break;
        }
    }

    let edge_count = edges_u.len();
    let mut offset = vec![0usize; n + 1];
    for i in 0..n {
        offset[i + 1] = offset[i] + degree[i];
    }

    let mut cursor = offset[..n].to_vec();
    let mut adjacency = vec![0usize; offset[n]];
    for id in 0..edge_count {
        let u = edges_u[id];
        let v = edges_v[id];
        let pu = cursor[u];
        adjacency[pu] = id;
        cursor[u] += 1;
        if u != v {
            let pv = cursor[v];
            adjacency[pv] = id;
            cursor[v] += 1;
        }
    }

    let mut matched = vec![false; n];
    let mut selected = vec![false; n];

    for id in 0..edge_count {
        let u = edges_u[id];
        let v = edges_v[id];
        if u == v {
            if !selected[u] {
                selected[u] = true;
            }
            matched[u] = true;
        } else if !matched[u] && !matched[v] {
            matched[u] = true;
            matched[v] = true;
            selected[u] = true;
            selected[v] = true;
        }
    }

    let mut private_edges = vec![0usize; n];
    for id in 0..edge_count {
        let u = edges_u[id];
        let v = edges_v[id];
        if u == v {
            if selected[u] {
                private_edges[u] += 1;
            }
            continue;
        }
        match (selected[u], selected[v]) {
            (true, false) => private_edges[u] += 1,
            (false, true) => private_edges[v] += 1,
            _ => {}
        }
    }

    let mut queue = VecDeque::new();
    let mut in_queue = vec![false; n];
    for v in 0..n {
        if selected[v] && private_edges[v] == 0 {
            queue.push_back(v);
            in_queue[v] = true;
        }
    }

    while let Some(x) = queue.pop_front() {
        in_queue[x] = false;
        if !selected[x] || private_edges[x] != 0 {
            continue;
        }

        selected[x] = false;
        for &edge_id in &adjacency[offset[x]..offset[x + 1]] {
            let a = edges_u[edge_id];
            let b = edges_v[edge_id];
            if a == b {
                continue;
            }
            let y = if a == x { b } else { a };
            if selected[y] {
                private_edges[y] += 1;
            }
        }
    }

    let count = selected.iter().filter(|&&is_selected| is_selected).count();
    let mut out = String::with_capacity(count.saturating_mul(12).saturating_add(32));
    out.push_str(&count.to_string());
    out.push('\n');

    let mut first = true;
    for (vertex, &is_selected) in selected.iter().enumerate() {
        if is_selected {
            if !first {
                out.push(' ');
            }
            first = false;
            out.push_str(&vertex.to_string());
        }
    }
    out.push('\n');

    let mut stdout = io::BufWriter::new(io::stdout().lock());
    stdout.write_all(out.as_bytes()).unwrap();
}
