import sys
import math
from itertools import combinations


def read_input():
    data = sys.stdin.read().strip().split()
    if not data:
        return 0, 0, 0, (0.0, 0.0), [], []
    it = iter(data)
    n = int(next(it))
    k = int(next(it))
    cap = float(next(it))
    depot = (float(next(it)), float(next(it)))
    pts, demands = [], []
    for _ in range(n):
        x = float(next(it))
        y = float(next(it))
        d = float(next(it))
        pts.append((x, y))
        demands.append(d)
    return n, k, cap, depot, pts, demands


def build_dist(depot, pts):
    all_pts = [depot] + pts
    m = len(all_pts)
    dist = [[0.0] * m for _ in range(m)]
    for i in range(m):
        xi, yi = all_pts[i]
        for j in range(i + 1, m):
            xj, yj = all_pts[j]
            v = math.hypot(xi - xj, yi - yj)
            dist[i][j] = dist[j][i] = v
    return dist


def route_load(route, demands):
    return sum(demands[i] for i in route)


def route_distance(route, dist):
    if not route:
        return 0.0
    total = dist[0][route[0] + 1]
    for a, b in zip(route, route[1:]):
        total += dist[a + 1][b + 1]
    total += dist[route[-1] + 1][0]
    return total


def total_distance(routes, dist):
    return sum(route_distance(r, dist) for r in routes)


def two_opt(route, dist):
    n = len(route)
    if n < 4:
        return route
    route = route[:]
    improved = True
    passes = 0
    while improved and passes < 60:
        improved = False
        passes += 1
        for i in range(n - 2):
            a = 0 if i == 0 else route[i - 1] + 1
            b = route[i] + 1
            for j in range(i + 1, n):
                c = route[j] + 1
                d = 0 if j == n - 1 else route[j + 1] + 1
                if dist[a][c] + dist[b][d] + 1e-12 < dist[a][b] + dist[c][d]:
                    route[i:j + 1] = reversed(route[i:j + 1])
                    improved = True
        if not improved:
            break
    return route


def improve_routes(routes, dist):
    return [two_opt(r, dist) for r in routes if r]


def clarke_wright(n, k, cap, demands, dist):
    routes = {i: [i] for i in range(n)}
    loads = {i: demands[i] for i in range(n)}
    owner = list(range(n))
    savings = []
    for i, j in combinations(range(n), 2):
        savings.append((dist[0][i + 1] + dist[0][j + 1] - dist[i + 1][j + 1], i, j))
    savings.sort(reverse=True)

    next_id = n
    for _, i, j in savings:
        ri, rj = owner[i], owner[j]
        if ri == rj or ri not in routes or rj not in routes:
            continue
        a, b = routes[ri], routes[rj]
        if loads[ri] + loads[rj] > cap + 1e-9:
            continue

        merged = None
        if a[-1] == i and b[0] == j:
            merged = a + b
        elif a[0] == i and b[-1] == j:
            merged = b + a
        elif a[0] == i and b[0] == j:
            merged = list(reversed(a)) + b
        elif a[-1] == i and b[-1] == j:
            merged = a + list(reversed(b))
        if merged is None:
            continue

        new_load = loads[ri] + loads[rj]
        del routes[ri], routes[rj], loads[ri], loads[rj]
        routes[next_id] = merged
        loads[next_id] = new_load
        for c in merged:
            owner[c] = next_id
        next_id += 1

    return list(routes.values())


def sweep_routes(n, k, cap, depot, pts, demands, shift=0.0):
    cx, cy = depot
    order = list(range(n))
    order.sort(key=lambda i: (math.atan2(pts[i][1] - cy, pts[i][0] - cx) + shift) % (2 * math.pi))
    routes, cur, load = [], [], 0.0
    for i in order:
        if cur and load + demands[i] > cap + 1e-9:
            routes.append(cur)
            cur, load = [], 0.0
        cur.append(i)
        load += demands[i]
    if cur:
        routes.append(cur)
    return routes


def first_fit_decreasing(n, k, cap, demands):
    routes, loads = [], []
    for i in sorted(range(n), key=lambda x: (-demands[x], x)):
        best = -1
        best_left = float("inf")
        for r, load in enumerate(loads):
            left = cap - load - demands[i]
            if left >= -1e-9 and left < best_left:
                best, best_left = r, left
        if best == -1:
            routes.append([i])
            loads.append(demands[i])
        else:
            routes[best].append(i)
            loads[best] += demands[i]
    return routes


def nearest_order(route, dist):
    if len(route) <= 2:
        return route[:]
    unused = set(route)
    cur = min(unused, key=lambda i: dist[0][i + 1])
    ordered = [cur]
    unused.remove(cur)
    while unused:
        cur = min(unused, key=lambda i: dist[ordered[-1] + 1][i + 1])
        ordered.append(cur)
        unused.remove(cur)
    return ordered


def valid(routes, n, k, cap, demands):
    if len(routes) > k:
        return False
    seen = []
    for r in routes:
        if sum(demands[i] for i in r) > cap + 1e-9:
            return False
        seen.extend(r)
    return sorted(seen) == list(range(n))


def solve():
    n, k, cap, depot, pts, demands = read_input()
    if n == 0:
        print("0.000000")
        print(0)
        return

    dist = build_dist(depot, pts)
    candidates = []

    cw = improve_routes(clarke_wright(n, k, cap, demands, dist), dist)
    candidates.append(cw)

    shifts = max(1, min(n, 72))
    for s in range(shifts):
        routes = sweep_routes(n, k, cap, depot, pts, demands, 2 * math.pi * s / shifts)
        routes = [nearest_order(r, dist) for r in routes]
        candidates.append(improve_routes(routes, dist))

    ffd = [nearest_order(r, dist) for r in first_fit_decreasing(n, k, cap, demands)]
    candidates.append(improve_routes(ffd, dist))

    feasible = [r for r in candidates if valid(r, n, k, cap, demands)]
    if feasible:
        best = min(feasible, key=lambda r: total_distance(r, dist))
    else:
        # If the instance itself is infeasible for K vehicles, still output a
        # complete capacity-respecting solution when possible.
        capacity_ok = [r for r in candidates if valid(r, n, max(k, n), cap, demands)]
        best = min(capacity_ok or candidates, key=lambda r: (len(r), total_distance(r, dist)))

    best = improve_routes(best, dist)
    print(f"{total_distance(best, dist):.6f}")
    print(len(best))
    for r in best:
        print(" ".join(map(str, r)))


if __name__ == "__main__":
    solve()
