; SPIR-V
; Version: 1.0
OpCapability Shader
%glsl = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %inWorldPos %inNormal %inUV %outColor
OpExecutionMode %main OriginUpperLeft
OpSource GLSL 450
OpName %main "main"
OpName %inWorldPos "worldPos"
OpName %inNormal "normal"
OpName %inUV "uv"
OpName %outColor "outColor"
OpName %UBO "UBO"
OpMemberName %UBO 0 "model"
OpMemberName %UBO 1 "view"
OpMemberName %UBO 2 "projection"
OpMemberName %UBO 3 "lightPos"
OpMemberName %UBO 4 "cameraPos"
OpMemberName %UBO 5 "params"
OpName %ubo "ubo"
OpDecorate %inWorldPos Location 0
OpDecorate %inNormal Location 1
OpDecorate %inUV Location 2
OpDecorate %outColor Location 0
OpDecorate %UBO Block
OpMemberDecorate %UBO 0 ColMajor
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 0 MatrixStride 16
OpMemberDecorate %UBO 1 ColMajor
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 1 MatrixStride 16
OpMemberDecorate %UBO 2 ColMajor
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 2 MatrixStride 16
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0
%void = OpTypeVoid
%fn = OpTypeFunction %void
%float = OpTypeFloat 32
%bool = OpTypeBool
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%mat4 = OpTypeMatrix %v4float 4
%UBO = OpTypeStruct %mat4 %mat4 %mat4 %v4float %v4float %v4float
%ptr_Input_v3float = OpTypePointer Input %v3float
%ptr_Input_v2float = OpTypePointer Input %v2float
%ptr_Output_v4float = OpTypePointer Output %v4float
%ptr_Uniform_UBO = OpTypePointer Uniform %UBO
%ptr_Uniform_v4float = OpTypePointer Uniform %v4float
%int = OpTypeInt 32 1
%c0 = OpConstant %int 0
%c3 = OpConstant %int 3
%f0 = OpConstant %float 0.0
%f005 = OpConstant %float 0.05
%f01 = OpConstant %float 0.1
%f02 = OpConstant %float 0.2
%f04 = OpConstant %float 0.4
%f05 = OpConstant %float 0.5
%f06 = OpConstant %float 0.6
%f07 = OpConstant %float 0.7
%f08 = OpConstant %float 8.0
%f095 = OpConstant %float 0.95
%f1 = OpConstant %float 1.0
%f16 = OpConstant %float 16.0
%f2 = OpConstant %float 2.0
%v2scale = OpConstantComposite %v2float %f08 %f16
%brickColor = OpConstantComposite %v3float %f07 %f02 %f01
%mortarColor = OpConstantComposite %v3float %f07 %f07 %f07
%inWorldPos = OpVariable %ptr_Input_v3float Input
%inNormal = OpVariable %ptr_Input_v3float Input
%inUV = OpVariable %ptr_Input_v2float Input
%outColor = OpVariable %ptr_Output_v4float Output
%ubo = OpVariable %ptr_Uniform_UBO Uniform
%main = OpFunction %void None %fn
%entry = OpLabel
%uv = OpLoad %v2float %inUV
%scaledUV = OpFMul %v2float %uv %v2scale
%scaledX = OpCompositeExtract %float %scaledUV 0
%scaledY = OpCompositeExtract %float %scaledUV 1
%row = OpExtInst %float %glsl Floor %scaledY
%rowHalf = OpFMul %float %row %f05
%rowHalfFloor = OpExtInst %float %glsl Floor %rowHalf
%evenPart = OpFMul %float %rowHalfFloor %f2
%rowParity = OpFSub %float %row %evenPart
%rowOffset = OpFMul %float %rowParity %f05
%offsetX = OpFAdd %float %scaledX %rowOffset
%floorX = OpExtInst %float %glsl Floor %offsetX
%cellX = OpFSub %float %offsetX %floorX
%floorY = OpExtInst %float %glsl Floor %scaledY
%cellY = OpFSub %float %scaledY %floorY
%leftMortar = OpFOrdLessThan %bool %cellX %f005
%rightMortar = OpFOrdGreaterThan %bool %cellX %f095
%xMortar = OpLogicalOr %bool %leftMortar %rightMortar
%yMortar = OpFOrdLessThan %bool %cellY %f01
%isMortar = OpLogicalOr %bool %xMortar %yMortar
%baseColor = OpSelect %v3float %isMortar %mortarColor %brickColor
%normal = OpLoad %v3float %inNormal
%n = OpExtInst %v3float %glsl Normalize %normal
%worldPos = OpLoad %v3float %inWorldPos
%lightPtr = OpAccessChain %ptr_Uniform_v4float %ubo %c3
%light4 = OpLoad %v4float %lightPtr
%lightX = OpCompositeExtract %float %light4 0
%lightY = OpCompositeExtract %float %light4 1
%lightZ = OpCompositeExtract %float %light4 2
%light3 = OpCompositeConstruct %v3float %lightX %lightY %lightZ
%toLight = OpFSub %v3float %light3 %worldPos
%l = OpExtInst %v3float %glsl Normalize %toLight
%dotNL = OpDot %float %n %l
%diffuse = OpExtInst %float %glsl FMax %dotNL %f0
%diffusePart = OpFMul %float %diffuse %f06
%shade = OpFAdd %float %f04 %diffusePart
%shaded = OpVectorTimesScalar %v3float %baseColor %shade
%outR = OpCompositeExtract %float %shaded 0
%outG = OpCompositeExtract %float %shaded 1
%outB = OpCompositeExtract %float %shaded 2
%finalColor = OpCompositeConstruct %v4float %outR %outG %outB %f1
OpStore %outColor %finalColor
OpReturn
OpFunctionEnd
