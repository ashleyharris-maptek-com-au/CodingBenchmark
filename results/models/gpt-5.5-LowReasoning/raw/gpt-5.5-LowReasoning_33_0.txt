using System;
using System.Diagnostics;
using System.IO;
using System.Text;

public class Program
{
    static int N;
    static int[,] F;
    static int[,] D;
    static Stopwatch Watch = Stopwatch.StartNew();
    const int TimeLimitMs = 285000;

    public static void Main()
    {
        var fs = new FastScanner();
        N = fs.NextInt();
        F = new int[N, N];
        D = new int[N, N];

        for (int i = 0; i < N; i++)
            for (int j = 0; j < N; j++)
                F[i, j] = fs.NextInt();
        for (int i = 0; i < N; i++)
            for (int j = 0; j < N; j++)
                D[i, j] = fs.NextInt();

        int[] p;
        long bestCost;

        if (N <= 10)
        {
            p = ExactSolve(out bestCost);
        }
        else
        {
            p = InitialPermutation();
            bestCost = Cost(p);
            Improve(p, ref bestCost);
        }

        var sb = new StringBuilder();
        sb.AppendLine(bestCost.ToString());
        for (int i = 0; i < N; i++)
        {
            if (i > 0) sb.Append(' ');
            sb.Append(p[i]);
        }
        Console.WriteLine(sb.ToString());
    }

    static int[] InitialPermutation()
    {
        long[] flowWeight = new long[N];
        long[] distWeight = new long[N];

        for (int i = 0; i < N; i++)
        {
            long fs = 0, ds = 0;
            for (int j = 0; j < N; j++)
            {
                fs += Math.Abs((long)F[i, j]) + Math.Abs((long)F[j, i]);
                ds += Math.Abs((long)D[i, j]) + Math.Abs((long)D[j, i]);
            }
            flowWeight[i] = fs;
            distWeight[i] = ds;
        }

        int[] fac = Range(N);
        int[] loc = Range(N);
        Array.Sort(fac, (a, b) => flowWeight[b].CompareTo(flowWeight[a]));
        Array.Sort(loc, (a, b) => distWeight[a].CompareTo(distWeight[b]));

        int[] p = new int[N];
        for (int i = 0; i < N; i++) p[fac[i]] = loc[i];
        return p;
    }

    static void Improve(int[] p, ref long cost)
    {
        if (N <= 450)
        {
            bool changed = true;
            while (changed && Watch.ElapsedMilliseconds < TimeLimitMs)
            {
                changed = false;
                for (int a = 0; a < N - 1 && Watch.ElapsedMilliseconds < TimeLimitMs; a++)
                {
                    for (int b = a + 1; b < N; b++)
                    {
                        long d = SwapDelta(p, a, b);
                        if (d < 0)
                        {
                            Swap(p, a, b);
                            cost += d;
                            changed = true;
                        }
                    }
                }
            }
        }
        else
        {
            var rnd = new Random(1234567);
            long tries = 0, maxTries = Math.Max(200000L, 8000000L / N);
            int stale = 0, staleLimit = Math.Max(5000, N * 20);
            while (Watch.ElapsedMilliseconds < TimeLimitMs && tries < maxTries && stale < staleLimit)
            {
                int a = rnd.Next(N);
                int b = rnd.Next(N - 1);
                if (b >= a) b++;
                if (a > b) { int t = a; a = b; b = t; }

                long d = SwapDelta(p, a, b);
                tries++;
                if (d < 0)
                {
                    Swap(p, a, b);
                    cost += d;
                    stale = 0;
                }
                else
                {
                    stale++;
                }
            }
        }
    }

    static long SwapDelta(int[] p, int a, int b)
    {
        int pa = p[a], pb = p[b];
        long oldCost = 0, newCost = 0;

        for (int k = 0; k < N; k++)
        {
            if (k == a || k == b) continue;
            int pk = p[k];

            oldCost += (long)F[a, k] * D[pa, pk] + (long)F[k, a] * D[pk, pa];
            oldCost += (long)F[b, k] * D[pb, pk] + (long)F[k, b] * D[pk, pb];

            newCost += (long)F[a, k] * D[pb, pk] + (long)F[k, a] * D[pk, pb];
            newCost += (long)F[b, k] * D[pa, pk] + (long)F[k, b] * D[pk, pa];
        }

        oldCost += (long)F[a, a] * D[pa, pa] + (long)F[b, b] * D[pb, pb];
        oldCost += (long)F[a, b] * D[pa, pb] + (long)F[b, a] * D[pb, pa];

        newCost += (long)F[a, a] * D[pb, pb] + (long)F[b, b] * D[pa, pa];
        newCost += (long)F[a, b] * D[pb, pa] + (long)F[b, a] * D[pa, pb];

        return newCost - oldCost;
    }

    static long Cost(int[] p)
    {
        long c = 0;
        for (int i = 0; i < N; i++)
        {
            int pi = p[i];
            for (int j = 0; j < N; j++)
                c += (long)F[i, j] * D[pi, p[j]];
        }
        return c;
    }

    static int[] ExactSolve(out long bestCost)
    {
        int[] cur = new int[N];
        int[] best = new int[N];
        bool[] used = new bool[N];
        bestCost = long.MaxValue;
        ExactDfs(0, 0, cur, best, used, ref bestCost);
        return best;
    }

    static void ExactDfs(int depth, long partial, int[] cur, int[] best, bool[] used, ref long bestCost)
    {
        if (depth == N)
        {
            if (partial < bestCost)
            {
                bestCost = partial;
                Array.Copy(cur, best, N);
            }
            return;
        }

        for (int loc = 0; loc < N; loc++)
        {
            if (used[loc]) continue;
            long add = (long)F[depth, depth] * D[loc, loc];
            for (int i = 0; i < depth; i++)
            {
                add += (long)F[depth, i] * D[loc, cur[i]];
                add += (long)F[i, depth] * D[cur[i], loc];
            }
            used[loc] = true;
            cur[depth] = loc;
            ExactDfs(depth + 1, partial + add, cur, best, used, ref bestCost);
            used[loc] = false;
        }
    }

    static int[] Range(int n)
    {
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = i;
        return a;
    }

    static void Swap(int[] a, int i, int j)
    {
        int t = a[i];
        a[i] = a[j];
        a[j] = t;
    }

    sealed class FastScanner
    {
        readonly Stream input = Console.OpenStandardInput();
        readonly byte[] buffer = new byte[1 << 16];
        int len;
        int ptr;

        int Read()
        {
            if (ptr >= len)
            {
                len = input.Read(buffer, 0, buffer.Length);
                ptr = 0;
                if (len == 0) return -1;
            }
            return buffer[ptr++];
        }

        public int NextInt()
        {
            int c;
            do { c = Read(); } while (c <= 32 && c >= 0);

            int sign = 1;
            if (c == '-')
            {
                sign = -1;
                c = Read();
            }

            int v = 0;
            while (c > 32)
            {
                v = v * 10 + c - '0';
                c = Read();
            }
            return v * sign;
        }
    }
}
