OpCapability Geometry
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main InputPoints
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 18
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2
OpDecorate %outColor Location 0

%void = OpTypeVoid
%func = OpTypeFunction %void
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_1 = OpConstant %uint 1
%int_0 = OpConstant %int 0
%float_0 = OpConstant %float 0
%float_004 = OpConstant %float 0.04
%float_n004 = OpConstant %float -0.04
%float_002 = OpConstant %float 0.02
%float_n002 = OpConstant %float -0.02
%float_0034641 = OpConstant %float 0.034641016
%float_n0034641 = OpConstant %float -0.034641016
%float_075 = OpConstant %float 0.75

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_1
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%ptr_in_v4 = OpTypePointer Input %v4float
%arr_v3_in = OpTypeArray %v3float %uint_1
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%ptr_in_v3 = OpTypePointer Input %v3float
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float

%gl_in = OpVariable %ptr_in_arr_pv Input
%worldPos = OpVariable %ptr_in_arr_v3 Input
%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input
%gl_out = OpVariable %ptr_out_pv Output
%outColor = OpVariable %ptr_out_v3 Output

%main = OpFunction %void None %func
%entry = OpLabel
%in_pos_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%in_pos = OpLoad %v4float %in_pos_ptr
%in_color_ptr = OpAccessChain %ptr_in_v3 %color %int_0
%in_color = OpLoad %v3float %in_color_ptr
%edge_color = OpVectorTimesScalar %v3float %in_color %float_075
%x = OpCompositeExtract %float %in_pos 0
%y = OpCompositeExtract %float %in_pos 1
%z = OpCompositeExtract %float %in_pos 2
%w = OpCompositeExtract %float %in_pos 3
%x_e0 = OpFAdd %float %x %float_004
%y_e0 = OpFAdd %float %y %float_0
%e0 = OpCompositeConstruct %v4float %x_e0 %y_e0 %z %w
%x_e1 = OpFAdd %float %x %float_002
%y_e1 = OpFAdd %float %y %float_0034641
%e1 = OpCompositeConstruct %v4float %x_e1 %y_e1 %z %w
%x_e2 = OpFAdd %float %x %float_n002
%y_e2 = OpFAdd %float %y %float_0034641
%e2 = OpCompositeConstruct %v4float %x_e2 %y_e2 %z %w
%x_e3 = OpFAdd %float %x %float_n004
%y_e3 = OpFAdd %float %y %float_0
%e3 = OpCompositeConstruct %v4float %x_e3 %y_e3 %z %w
%x_e4 = OpFAdd %float %x %float_n002
%y_e4 = OpFAdd %float %y %float_n0034641
%e4 = OpCompositeConstruct %v4float %x_e4 %y_e4 %z %w
%x_e5 = OpFAdd %float %x %float_002
%y_e5 = OpFAdd %float %y %float_n0034641
%e5 = OpCompositeConstruct %v4float %x_e5 %y_e5 %z %w
%out_pos_ptr = OpAccessChain %ptr_out_v4 %gl_out %int_0

OpStore %out_pos_ptr %in_pos
OpStore %outColor %in_color
OpEmitVertex
OpStore %out_pos_ptr %e0
OpStore %outColor %edge_color
OpEmitVertex
OpStore %out_pos_ptr %e1
OpStore %outColor %edge_color
OpEmitVertex
OpEndPrimitive

OpStore %out_pos_ptr %in_pos
OpStore %outColor %in_color
OpEmitVertex
OpStore %out_pos_ptr %e1
OpStore %outColor %edge_color
OpEmitVertex
OpStore %out_pos_ptr %e2
OpStore %outColor %edge_color
OpEmitVertex
OpEndPrimitive

OpStore %out_pos_ptr %in_pos
OpStore %outColor %in_color
OpEmitVertex
OpStore %out_pos_ptr %e2
OpStore %outColor %edge_color
OpEmitVertex
OpStore %out_pos_ptr %e3
OpStore %outColor %edge_color
OpEmitVertex
OpEndPrimitive

OpStore %out_pos_ptr %in_pos
OpStore %outColor %in_color
OpEmitVertex
OpStore %out_pos_ptr %e3
OpStore %outColor %edge_color
OpEmitVertex
OpStore %out_pos_ptr %e4
OpStore %outColor %edge_color
OpEmitVertex
OpEndPrimitive

OpStore %out_pos_ptr %in_pos
OpStore %outColor %in_color
OpEmitVertex
OpStore %out_pos_ptr %e4
OpStore %outColor %edge_color
OpEmitVertex
OpStore %out_pos_ptr %e5
OpStore %outColor %edge_color
OpEmitVertex
OpEndPrimitive

OpStore %out_pos_ptr %in_pos
OpStore %outColor %in_color
OpEmitVertex
OpStore %out_pos_ptr %e5
OpStore %outColor %edge_color
OpEmitVertex
OpStore %out_pos_ptr %e0
OpStore %outColor %edge_color
OpEmitVertex
OpEndPrimitive

OpReturn
OpFunctionEnd
