#include <algorithm>
#include <cstdint>
#include <functional>
#include <iostream>
#include <limits>
#include <numeric>
#include <queue>
#include <unordered_set>
#include <utility>
#include <vector>

using namespace std;

struct Edge {
    int u, v;
    long long w;
};

struct AdjEdge {
    int to;
    int id;
    long long w;
};

struct DSU {
    vector<int> p, r;
    explicit DSU(int n = 0) { init(n); }
    void init(int n) {
        p.resize(n);
        r.assign(n, 0);
        iota(p.begin(), p.end(), 0);
    }
    int find(int x) {
        while (p[x] != x) {
            p[x] = p[p[x]];
            x = p[x];
        }
        return x;
    }
    bool unite(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b) return false;
        if (r[a] < r[b]) swap(a, b);
        p[b] = a;
        if (r[a] == r[b]) ++r[a];
        return true;
    }
};

static const long long INF = numeric_limits<long long>::max() / 4;

static int bit_index(int bit) {
    int idx = 0;
    while ((1 << idx) != bit) ++idx;
    return idx;
}

static void dijkstra_from_sources(
    int n,
    const vector<vector<AdjEdge>>& adj,
    vector<long long>& dist,
    vector<int>* parent_edge,
    vector<int>* changed_vertices = nullptr
) {
    priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;
    for (int i = 0; i < n; ++i) {
        if (dist[i] < INF) pq.push({dist[i], i});
    }

    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();
        if (d != dist[u]) continue;
        for (const AdjEdge& e : adj[u]) {
            long long nd = d + e.w;
            if (nd < dist[e.to]) {
                dist[e.to] = nd;
                if (parent_edge) (*parent_edge)[e.to] = e.id;
                if (changed_vertices) changed_vertices->push_back(e.to);
                pq.push({nd, e.to});
            }
        }
    }
}

static vector<int> tree_cleanup(
    int n,
    const vector<Edge>& edges,
    const vector<int>& candidate_edges,
    const vector<char>& is_terminal
) {
    vector<int> unique_edges = candidate_edges;
    sort(unique_edges.begin(), unique_edges.end());
    unique_edges.erase(unique(unique_edges.begin(), unique_edges.end()), unique_edges.end());

    sort(unique_edges.begin(), unique_edges.end(), [&](int a, int b) {
        if (edges[a].w != edges[b].w) return edges[a].w < edges[b].w;
        return a < b;
    });

    DSU dsu(n);
    vector<int> forest;
    for (int id : unique_edges) {
        if (dsu.unite(edges[id].u, edges[id].v)) forest.push_back(id);
    }

    vector<vector<pair<int, int>>> tadj(n);
    vector<char> alive(forest.size(), 1);
    vector<int> degree(n, 0);
    for (int i = 0; i < (int)forest.size(); ++i) {
        const Edge& e = edges[forest[i]];
        tadj[e.u].push_back({e.v, i});
        tadj[e.v].push_back({e.u, i});
        ++degree[e.u];
        ++degree[e.v];
    }

    queue<int> q;
    for (int v = 0; v < n; ++v) {
        if (!is_terminal[v] && degree[v] == 1) q.push(v);
    }

    while (!q.empty()) {
        int v = q.front();
        q.pop();
        if (is_terminal[v] || degree[v] != 1) continue;
        for (auto [to, idx] : tadj[v]) {
            if (!alive[idx]) continue;
            alive[idx] = 0;
            --degree[v];
            --degree[to];
            if (!is_terminal[to] && degree[to] == 1) q.push(to);
            break;
        }
    }

    vector<int> result;
    for (int i = 0; i < (int)forest.size(); ++i) {
        if (alive[i]) result.push_back(forest[i]);
    }
    return result;
}

static vector<int> heuristic_solution(
    int n,
    const vector<Edge>& edges,
    const vector<vector<AdjEdge>>& adj,
    const vector<int>& terminals,
    const vector<char>& is_terminal
) {
    int k = (int)terminals.size();
    vector<vector<long long>> dist(k, vector<long long>(n, INF));
    vector<vector<int>> parent(k, vector<int>(n, -1));

    for (int i = 0; i < k; ++i) {
        dist[i][terminals[i]] = 0;
        dijkstra_from_sources(n, adj, dist[i], &parent[i]);
    }

    struct KEdge { int a, b; long long w; };
    vector<KEdge> complete;
    for (int i = 0; i < k; ++i) {
        for (int j = i + 1; j < k; ++j) {
            if (dist[i][terminals[j]] < INF) complete.push_back({i, j, dist[i][terminals[j]]});
        }
    }
    sort(complete.begin(), complete.end(), [](const KEdge& a, const KEdge& b) {
        return a.w < b.w;
    });

    DSU kd(k);
    vector<int> chosen;
    for (const KEdge& e : complete) {
        if (!kd.unite(e.a, e.b)) continue;
        int cur = terminals[e.b];
        while (cur != terminals[e.a]) {
            int id = parent[e.a][cur];
            if (id < 0) break;
            chosen.push_back(id);
            cur = edges[id].u ^ edges[id].v ^ cur;
        }
    }
    return tree_cleanup(n, edges, chosen, is_terminal);
}

static vector<int> exact_solution(
    int n,
    const vector<Edge>& edges,
    const vector<vector<AdjEdge>>& adj,
    const vector<int>& terminals,
    const vector<char>& is_terminal
) {
    int k = (int)terminals.size();
    int states = 1 << k;
    size_t total = (size_t)states * (size_t)n;

    vector<long long> dp(total, INF);
    vector<int> parent_edge(total, -1);
    vector<int> split_mask(total, 0);

    auto at = [n](int mask, int v) -> size_t {
        return (size_t)mask * (size_t)n + (size_t)v;
    };

    for (int i = 0; i < k; ++i) {
        dp[at(1 << i, terminals[i])] = 0;
    }

    for (int mask = 1; mask < states; ++mask) {
        for (int sub = (mask - 1) & mask; sub; sub = (sub - 1) & mask) {
            int other = mask ^ sub;
            if (sub > other) continue;
            for (int v = 0; v < n; ++v) {
                long long a = dp[at(sub, v)];
                long long b = dp[at(other, v)];
                if (a == INF || b == INF) continue;
                long long cand = a + b;
                size_t idx = at(mask, v);
                if (cand < dp[idx]) {
                    dp[idx] = cand;
                    split_mask[idx] = sub;
                    parent_edge[idx] = -1;
                }
            }
        }

        vector<long long> dist(n);
        vector<int> pedge(n, -1);
        for (int v = 0; v < n; ++v) dist[v] = dp[at(mask, v)];

        priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;
        for (int v = 0; v < n; ++v) {
            if (dist[v] < INF) pq.push({dist[v], v});
        }
        while (!pq.empty()) {
            auto [d, u] = pq.top();
            pq.pop();
            if (d != dist[u]) continue;
            for (const AdjEdge& e : adj[u]) {
                long long nd = d + e.w;
                if (nd < dist[e.to]) {
                    dist[e.to] = nd;
                    pedge[e.to] = e.id;
                    pq.push({nd, e.to});
                }
            }
        }
        for (int v = 0; v < n; ++v) {
            size_t idx = at(mask, v);
            if (dist[v] < dp[idx]) {
                dp[idx] = dist[v];
                parent_edge[idx] = pedge[v];
                split_mask[idx] = 0;
            }
        }
    }

    int full = states - 1;
    int root = 0;
    for (int v = 1; v < n; ++v) {
        if (dp[at(full, v)] < dp[at(full, root)]) root = v;
    }

    vector<int> chosen;
    function<void(int, int)> build = [&](int mask, int v) {
        size_t idx = at(mask, v);
        int sm = split_mask[idx];
        if (sm != 0) {
            build(sm, v);
            build(mask ^ sm, v);
            return;
        }

        int bit = mask & -mask;
        if (mask == bit && v == terminals[bit_index(bit)]) return;

        int id = parent_edge[idx];
        if (id < 0) return;
        chosen.push_back(id);
        int prev = edges[id].u ^ edges[id].v ^ v;
        build(mask, prev);
    };

    build(full, root);
    return tree_cleanup(n, edges, chosen, is_terminal);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m, k;
    if (!(cin >> n >> m >> k)) return 0;

    vector<Edge> edges;
    edges.reserve(m);
    vector<vector<AdjEdge>> adj(n);
    for (int i = 0; i < m; ++i) {
        int u, v;
        long long w;
        cin >> u >> v >> w;
        edges.push_back({u, v, w});
        adj[u].push_back({v, i, w});
        adj[v].push_back({u, i, w});
    }

    vector<int> terminals(k);
    vector<char> is_terminal(n, 0);
    for (int i = 0; i < k; ++i) {
        cin >> terminals[i];
        is_terminal[terminals[i]] = 1;
    }

    vector<int> answer;
    if (k == 0) {
        cout << "0 0\n";
        return 0;
    }
    if (k == 1) {
        cout << "0 0\n";
        return 0;
    }

    const size_t exact_limit = 30000000ULL;
    if (k <= 20 && ((size_t)1 << k) * (size_t)n <= exact_limit) {
        answer = exact_solution(n, edges, adj, terminals, is_terminal);
    } else {
        answer = heuristic_solution(n, edges, adj, terminals, is_terminal);
    }

    long long total = 0;
    for (int id : answer) total += edges[id].w;

    cout << total << ' ' << answer.size() << '\n';
    for (int id : answer) {
        cout << edges[id].u << ' ' << edges[id].v << '\n';
    }
    return 0;
}
