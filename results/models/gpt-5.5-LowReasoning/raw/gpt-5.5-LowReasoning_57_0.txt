const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const W = data.width | 0, H = data.height | 0, N = W * H;
const grid = data.grid || [];
const budget = Math.max(0, data.cutBudget | 0);
const ignitions = Array.isArray(data.ignitions) ? data.ignitions : [];
const towns = Array.isArray(data.towns) ? data.towns : [];
const INF = 1 << 28;

function id(x, y) { return y * W + x; }
function xy(i) { return [i % W, (i / W) | 0]; }
function inside(x, y) { return x >= 0 && y >= 0 && x < W && y < H; }
function burnableId(i) { const y = (i / W) | 0, x = i - y * W; return grid[y] && grid[y][x] === '.'; }
function burnableXY(x, y) { return inside(x, y) && grid[y] && grid[y][x] === '.'; }

const ignitionSet = new Uint8Array(N);
for (const p of ignitions) {
  if (p && burnableXY(p[0] | 0, p[1] | 0)) ignitionSet[id(p[0] | 0, p[1] | 0)] = 1;
}
const townSet = new Uint8Array(N);
for (const t of towns) {
  const x = t.x | 0, y = t.y | 0;
  if (inside(x, y)) townSet[id(x, y)] = 1;
}

function simulate(cutSet) {
  const burned = new Uint8Array(N);
  const q = new Int32Array(N);
  let head = 0, tail = 0;
  for (const p of ignitions) {
    if (!p) continue;
    const x = p[0] | 0, y = p[1] | 0, s = id(x, y);
    if (burnableXY(x, y) && !cutSet.has(s) && !burned[s]) {
      burned[s] = 1; q[tail++] = s;
    }
  }
  while (head < tail) {
    const v = q[head++], x = v % W, y = (v / W) | 0;
    let n;
    if (x > 0) { n = v - 1; if (!burned[n] && burnableId(n) && !cutSet.has(n)) burned[n] = 1, q[tail++] = n; }
    if (x + 1 < W) { n = v + 1; if (!burned[n] && burnableId(n) && !cutSet.has(n)) burned[n] = 1, q[tail++] = n; }
    if (y > 0) { n = v - W; if (!burned[n] && burnableId(n) && !cutSet.has(n)) burned[n] = 1, q[tail++] = n; }
    if (y + 1 < H) { n = v + W; if (!burned[n] && burnableId(n) && !cutSet.has(n)) burned[n] = 1, q[tail++] = n; }
  }
  return burned;
}

const emptyCut = new Set();
const baselineBurned = simulate(emptyCut);
let baselineValue = 0;
const riskyTowns = [];
for (const t of towns) {
  const x = t.x | 0, y = t.y | 0, v = Number(t.value) || 0, cell = id(x, y);
  if (inside(x, y) && baselineBurned[cell]) {
    baselineValue += v;
    riskyTowns.push({ x, y, value: v, cell });
  }
}

if (budget <= 0 || baselineValue <= 0 || riskyTowns.length === 0) {
  console.log(JSON.stringify({ firebreaks: [] }));
  process.exit(0);
}

class Dinic {
  constructor(n) {
    this.n = n;
    this.g = Array.from({ length: n }, () => []);
    this.level = new Int32Array(n);
    this.it = new Int32Array(n);
  }
  addEdge(fr, to, cap) {
    const f = { to, rev: this.g[to].length, cap, orig: cap };
    const r = { to: fr, rev: this.g[fr].length, cap: 0, orig: 0 };
    this.g[fr].push(f); this.g[to].push(r);
  }
  bfs(s, t) {
    this.level.fill(-1);
    const q = new Int32Array(this.n);
    let h = 0, r = 0;
    this.level[s] = 0; q[r++] = s;
    while (h < r) {
      const v = q[h++];
      for (const e of this.g[v]) if (e.cap > 0 && this.level[e.to] < 0) {
        this.level[e.to] = this.level[v] + 1;
        if (e.to === t) return true;
        q[r++] = e.to;
      }
    }
    return this.level[t] >= 0;
  }
  dfs(v, t, f) {
    if (v === t) return f;
    for (let i = this.it[v]; i < this.g[v].length; i++, this.it[v] = i) {
      const e = this.g[v][i];
      if (e.cap > 0 && this.level[v] + 1 === this.level[e.to]) {
        const ret = this.dfs(e.to, t, Math.min(f, e.cap));
        if (ret > 0) {
          e.cap -= ret;
          this.g[e.to][e.rev].cap += ret;
          return ret;
        }
      }
    }
    return 0;
  }
  maxflow(s, t, limit) {
    let flow = 0;
    while (flow <= limit && this.bfs(s, t)) {
      this.it.fill(0);
      let f;
      while (flow <= limit && (f = this.dfs(s, t, INF)) > 0) flow += f;
    }
    return flow;
  }
  reachable(s) {
    const seen = new Uint8Array(this.n), q = new Int32Array(this.n);
    let h = 0, r = 0; seen[s] = 1; q[r++] = s;
    while (h < r) {
      const v = q[h++];
      for (const e of this.g[v]) if (e.cap > 0 && !seen[e.to]) seen[e.to] = 1, q[r++] = e.to;
    }
    return seen;
  }
}

function minCutForSinks(sinkCells, currentCuts) {
  if (!sinkCells.length) return [];
  let minX = W, minY = H, maxX = -1, maxY = -1;
  const points = [];
  for (const p of ignitions) if (p && burnableXY(p[0] | 0, p[1] | 0)) points.push([p[0] | 0, p[1] | 0]);
  for (const c of sinkCells) points.push(xy(c));
  for (const [x, y] of points) { if (x < minX) minX = x; if (x > maxX) maxX = x; if (y < minY) minY = y; if (y > maxY) maxY = y; }
  const area = W * H;
  let pad = area <= 90000 ? Math.max(W, H) : Math.min(Math.max(W, H), 90 + budget * 4);
  minX = Math.max(0, minX - pad); maxX = Math.min(W - 1, maxX + pad);
  minY = Math.max(0, minY - pad); maxY = Math.min(H - 1, maxY + pad);
  const bw = maxX - minX + 1, bh = maxY - minY + 1, boxN = bw * bh;
  if (boxN > 180000) return [];

  const local = new Int32Array(boxN);
  local.fill(-1);
  const cells = [];
  for (let y = minY; y <= maxY; y++) for (let x = minX; x <= maxX; x++) {
    const c = id(x, y);
    if (burnableId(c) && baselineBurned[c] && !currentCuts.has(c)) {
      local[(y - minY) * bw + (x - minX)] = cells.length;
      cells.push(c);
    }
  }
  const source = cells.length * 2, sink = source + 1;
  const din = new Dinic(sink + 1);
  const sinkMark = new Uint8Array(N);
  for (const c of sinkCells) sinkMark[c] = 1;
  for (let li = 0; li < cells.length; li++) {
    const c = cells[li], x = c % W, y = (c / W) | 0;
    const inn = li * 2, out = inn + 1;
    const protectedCell = ignitionSet[c] || sinkMark[c] || townSet[c];
    din.addEdge(inn, out, protectedCell ? INF : 1);
    if (ignitionSet[c]) din.addEdge(source, inn, INF);
    if (sinkMark[c]) din.addEdge(out, sink, INF);
    const tryN = (nx, ny) => {
      if (nx < minX || nx > maxX || ny < minY || ny > maxY) return;
      const ni = local[(ny - minY) * bw + (nx - minX)];
      if (ni >= 0) din.addEdge(out, ni * 2, INF);
    };
    tryN(x - 1, y); tryN(x + 1, y); tryN(x, y - 1); tryN(x, y + 1);
  }
  const flow = din.maxflow(source, sink, budget + 1);
  if (flow <= 0 || flow > budget) return [];
  const reach = din.reachable(source), cut = [];
  for (let li = 0; li < cells.length; li++) {
    const inn = li * 2, out = inn + 1;
    if (reach[inn] && !reach[out] && !ignitionSet[cells[li]] && !sinkMark[cells[li]] && !townSet[cells[li]]) cut.push(cells[li]);
  }
  return cut.length <= budget ? cut : [];
}

function savedValue(cutSet) {
  const burned = simulate(cutSet);
  let saved = 0;
  for (const t of riskyTowns) if (!burned[t.cell]) saved += t.value;
  return saved;
}

const candidates = [];
const top = riskyTowns.slice().sort((a, b) => b.value - a.value).slice(0, Math.min(36, riskyTowns.length));
for (const t of top) candidates.push([t.cell]);
for (const k of [2, 3, 5, 8, 13, 21, 34]) {
  if (k <= top.length) candidates.push(top.slice(0, k).map(t => t.cell));
}

let chosen = new Set(), bestSaved = 0;
const hardLimit = Date.now() + 8500;
while (chosen.size < budget && Date.now() < hardLimit) {
  let bestCut = null, bestGain = 0, bestTotal = bestSaved;
  for (const sinks of candidates) {
    if (Date.now() >= hardLimit) break;
    const cut = minCutForSinks(sinks, chosen);
    if (!cut.length || cut.length + chosen.size > budget) continue;
    const trial = new Set(chosen);
    for (const c of cut) trial.add(c);
    if (trial.size > budget) continue;
    const total = savedValue(trial);
    const gain = total - bestSaved;
    if (gain > bestGain || (gain === bestGain && total > bestTotal && cut.length < (bestCut ? bestCut.length : 1e9))) {
      bestGain = gain; bestTotal = total; bestCut = cut;
    }
  }
  if (!bestCut || bestGain <= 0) break;
  for (const c of bestCut) chosen.add(c);
  bestSaved = bestTotal;
}

// Spend leftover cells on cheap one-cell choke attempts around threatened towns.
if (chosen.size < budget) {
  const burnedNow = simulate(chosen);
  const singles = [];
  for (let c = 0; c < N; c++) {
    if (!burnedNow[c] || ignitionSet[c] || townSet[c] || chosen.has(c)) continue;
    const x = c % W, y = (c / W) | 0;
    let nearTown = false;
    for (const t of top.slice(0, 12)) {
      if (Math.abs(t.x - x) + Math.abs(t.y - y) <= 8 + budget) { nearTown = true; break; }
    }
    if (nearTown) singles.push(c);
  }
  let tries = 0;
  while (chosen.size < budget && tries < singles.length && Date.now() < hardLimit) {
    let best = -1, bestTotal = bestSaved;
    for (let i = 0; i < singles.length && i < 2000; i++) {
      const c = singles[(tries + i) % singles.length];
      if (chosen.has(c)) continue;
      const trial = new Set(chosen); trial.add(c);
      const total = savedValue(trial);
      if (total > bestTotal) { bestTotal = total; best = c; }
    }
    tries += 2000;
    if (best < 0) break;
    chosen.add(best); bestSaved = bestTotal;
  }
}

const firebreaks = [];
for (const c of chosen) {
  if (firebreaks.length >= budget) break;
  if (burnableId(c) && !ignitionSet[c] && !townSet[c]) firebreaks.push(xy(c));
}
console.log(JSON.stringify({ firebreaks }));
