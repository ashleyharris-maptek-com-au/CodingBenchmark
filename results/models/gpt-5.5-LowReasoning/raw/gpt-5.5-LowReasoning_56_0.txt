const fs = require('fs'); const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const packages = Array.isArray(data.packages) ? data.packages : [];
const n = packages.length;

if (n === 0) {
  process.stdout.write(JSON.stringify({ routes: [] }));
  process.exit(0);
}

const depot = Array.isArray(data.depot) ? data.depot : [0, 0];
const dx0 = Number(depot[0]) || 0;
const dy0 = Number(depot[1]) || 0;

const xs = new Float64Array(n);
const ys = new Float64Array(n);
const ids = new Array(n);
let cx = 0, cy = 0;
for (let i = 0; i < n; i++) {
  const p = packages[i];
  xs[i] = Number(p.x) || 0;
  ys[i] = Number(p.y) || 0;
  ids[i] = p.id;
  cx += xs[i];
  cy += ys[i];
}
cx /= n;
cy /= n;

function distIdx(a, b) {
  const x = xs[a] - xs[b];
  const y = ys[a] - ys[b];
  return Math.hypot(x, y);
}

function distDepot(a) {
  return Math.hypot(xs[a] - dx0, ys[a] - dy0);
}

function routeLength(route) {
  const m = route.length;
  if (m === 0) return 0;
  let total = distDepot(route[0]) + distDepot(route[m - 1]);
  for (let i = 1; i < m; i++) total += distIdx(route[i - 1], route[i]);
  return total;
}

function reversedCopy(route) {
  const out = route.slice();
  out.reverse();
  return out;
}

function rotateCopy(route, shift) {
  const m = route.length;
  if (m <= 1 || shift === 0) return route.slice();
  const out = new Array(m);
  for (let i = 0; i < m; i++) out[i] = route[(i + shift) % m];
  return out;
}

function sortedBy(compare) {
  const a = Array.from({ length: n }, (_, i) => i);
  a.sort(compare);
  return a;
}

function polarRoute(ox, oy) {
  return sortedBy((a, b) => {
    const aa = Math.atan2(ys[a] - oy, xs[a] - ox);
    const bb = Math.atan2(ys[b] - oy, xs[b] - ox);
    if (aa !== bb) return aa - bb;
    const ra = (xs[a] - ox) ** 2 + (ys[a] - oy) ** 2;
    const rb = (xs[b] - ox) ** 2 + (ys[b] - oy) ** 2;
    return ra - rb;
  });
}

function nearestNeighborRoute() {
  const unused = new Set();
  for (let i = 0; i < n; i++) unused.add(i);
  let bestStart = 0, bestD = Infinity;
  for (let i = 0; i < n; i++) {
    const d = distDepot(i);
    if (d < bestD) {
      bestD = d;
      bestStart = i;
    }
  }
  const route = [bestStart];
  unused.delete(bestStart);
  let cur = bestStart;
  while (unused.size) {
    let best = -1, bd = Infinity;
    for (const j of unused) {
      const d = distIdx(cur, j);
      if (d < bd) {
        bd = d;
        best = j;
      }
    }
    route.push(best);
    unused.delete(best);
    cur = best;
  }
  return route;
}

function improve2Opt(route, timeLimitMs) {
  const m = route.length;
  if (m < 4) return route;
  const start = Date.now();
  const maxSpan = m <= 900 ? m - 1 : (m <= 5000 ? 140 : 70);
  const passes = m <= 900 ? 20 : 6;
  const eps = 1e-10;

  for (let pass = 0; pass < passes; pass++) {
    let changed = false;
    for (let i = 0; i < m - 1; i++) {
      if ((i & 1023) === 0 && Date.now() - start > timeLimitMs) return route;
      const a = i === 0 ? -1 : route[i - 1];
      const b = route[i];
      const oldLeft = a < 0 ? distDepot(b) : distIdx(a, b);
      const maxJ = Math.min(m - 1, i + maxSpan);
      for (let j = i + 1; j <= maxJ; j++) {
        const c = route[j];
        const d = j === m - 1 ? -1 : route[j + 1];
        const oldRight = d < 0 ? distDepot(c) : distIdx(c, d);
        const newLeft = a < 0 ? distDepot(c) : distIdx(a, c);
        const newRight = d < 0 ? distDepot(b) : distIdx(b, d);
        if (newLeft + newRight + eps < oldLeft + oldRight) {
          let l = i, r = j;
          while (l < r) {
            const tmp = route[l];
            route[l++] = route[r];
            route[r--] = tmp;
          }
          changed = true;
          break;
        }
      }
    }
    if (!changed) break;
  }
  return route;
}

let candidates = [];
candidates.push(polarRoute(dx0, dy0));
candidates.push(polarRoute(cx, cy));
candidates.push(sortedBy((a, b) => xs[a] - xs[b] || ys[a] - ys[b]));
candidates.push(sortedBy((a, b) => ys[a] - ys[b] || xs[a] - xs[b]));

if (n <= 3500) candidates.push(nearestNeighborRoute());

let bestRoute = candidates[0];
let bestLen = Infinity;
for (const route of candidates) {
  const variants = [route, reversedCopy(route)];
  if (route.length > 12) {
    variants.push(rotateCopy(route, Math.floor(route.length / 4)));
    variants.push(rotateCopy(route, Math.floor(route.length / 2)));
  }
  for (const v of variants) {
    const len = routeLength(v);
    if (len < bestLen) {
      bestLen = len;
      bestRoute = v;
    }
  }
}

const budgetMs = n <= 1000 ? 3500 : (n <= 5000 ? 5500 : 7500);
bestRoute = improve2Opt(bestRoute.slice(), budgetMs);

const routeIds = bestRoute.map(i => ids[i]);
process.stdout.write(JSON.stringify({ routes: [routeIds] }));
