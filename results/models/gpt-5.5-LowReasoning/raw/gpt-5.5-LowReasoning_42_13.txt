; SPIR-V
; Version: 1.0
OpCapability Geometry
%1 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %inWorldPos %inNormal %inColor %gl_out %outColor
OpExecutionMode %main InputTrianglesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 12
OpExecutionMode %main Invocations 1
OpSource GLSL 450
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %inWorldPos Location 0
OpDecorate %inNormal Location 1
OpDecorate %inColor Location 2
OpDecorate %outColor Location 0
%void = OpTypeVoid
%float = OpTypeFloat 32
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint_6 = OpConstant %uint 6
%int_0 = OpConstant %int 0
%int_2 = OpConstant %int 2
%int_4 = OpConstant %int 4
%float_0 = OpConstant %float 0
%float_0_5 = OpConstant %float 0.5
%float_0_008 = OpConstant %float 0.008
%gl_PerVertex_in = OpTypeStruct %v4float
%gl_PerVertex_out = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_6
%arr_v3_in = OpTypeArray %v3float %uint_6
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%fn_main = OpTypeFunction %void
%fn_emit_edge = OpTypeFunction %void %v4float %v4float %v3float %v3float
%gl_in = OpVariable %ptr_in_arr_pv Input
%inWorldPos = OpVariable %ptr_in_arr_v3 Input
%inNormal = OpVariable %ptr_in_arr_v3 Input
%inColor = OpVariable %ptr_in_arr_v3 Input
%gl_out = OpVariable %ptr_out_pv Output
%outColor = OpVariable %ptr_out_v3 Output
%emitEdge = OpFunction %void None %fn_emit_edge
%p0 = OpFunctionParameter %v4float
%p1 = OpFunctionParameter %v4float
%c0 = OpFunctionParameter %v3float
%c1 = OpFunctionParameter %v3float
%emit_entry = OpLabel
%p0xy = OpVectorShuffle %v2float %p0 %p0 0 1
%p1xy = OpVectorShuffle %v2float %p1 %p1 0 1
%dir = OpFSub %v2float %p1xy %p0xy
%dx = OpCompositeExtract %float %dir 0
%dy = OpCompositeExtract %float %dir 1
%dx2 = OpFMul %float %dx %dx
%dy2 = OpFMul %float %dy %dy
%len2 = OpFAdd %float %dx2 %dy2
%invLen = OpExtInst %float %1 InverseSqrt %len2
%negDy = OpFNegate %float %dy
%perpX0 = OpFMul %float %negDy %invLen
%perpY0 = OpFMul %float %dx %invLen
%perpX = OpFMul %float %perpX0 %float_0_008
%perpY = OpFMul %float %perpY0 %float_0_008
%off = OpCompositeConstruct %v4float %perpX %perpY %float_0 %float_0
%p0Plus = OpFAdd %v4float %p0 %off
%p0Minus = OpFSub %v4float %p0 %off
%p1Plus = OpFAdd %v4float %p1 %off
%p1Minus = OpFSub %v4float %p1 %off
%sumColor = OpFAdd %v3float %c0 %c1
%edgeColor = OpVectorTimesScalar %v3float %sumColor %float_0_5
%outPos0 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %outPos0 %p0Plus
OpStore %outColor %edgeColor
OpEmitVertex
%outPos1 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %outPos1 %p0Minus
OpStore %outColor %edgeColor
OpEmitVertex
%outPos2 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %outPos2 %p1Plus
OpStore %outColor %edgeColor
OpEmitVertex
%outPos3 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %outPos3 %p1Minus
OpStore %outColor %edgeColor
OpEmitVertex
OpEndPrimitive
OpReturn
OpFunctionEnd
%main = OpFunction %void None %fn_main
%main_entry = OpLabel
%p0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%p2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%p4_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_4 %int_0
%pos0 = OpLoad %v4float %p0_ptr
%pos2 = OpLoad %v4float %p2_ptr
%pos4 = OpLoad %v4float %p4_ptr
%c0_ptr = OpAccessChain %ptr_in_v3 %inColor %int_0
%c2_ptr = OpAccessChain %ptr_in_v3 %inColor %int_2
%c4_ptr = OpAccessChain %ptr_in_v3 %inColor %int_4
%color0 = OpLoad %v3float %c0_ptr
%color2 = OpLoad %v3float %c2_ptr
%color4 = OpLoad %v3float %c4_ptr
%call_edge_0 = OpFunctionCall %void %emitEdge %pos0 %pos2 %color0 %color2
%call_edge_1 = OpFunctionCall %void %emitEdge %pos2 %pos4 %color2 %color4
%call_edge_2 = OpFunctionCall %void %emitEdge %pos4 %pos0 %color4 %color0
OpReturn
OpFunctionEnd
