OpCapability Shader
OpCapability Geometry
OpMemoryModel Logical GLSL450
%glsl = OpExtInstImport "GLSL.std.450"
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main InputLinesAdjacency
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2
OpDecorate %outColor Location 0
%void = OpTypeVoid
%fn = OpTypeFunction %void
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%uint_4 = OpConstant %uint 4
%int = OpTypeInt 32 1
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%int_3 = OpConstant %int 3
%float_0 = OpConstant %float 0
%float_0_025 = OpConstant %float 0.025
%gl_PerVertex_in = OpTypeStruct %v4float
%gl_PerVertex_out = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_4
%arr_v3float_4 = OpTypeArray %v3float %uint_4
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3float_4
%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%gl_in = OpVariable %ptr_in_arr_pv Input
%worldPos = OpVariable %ptr_in_arr_v3 Input
%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input
%gl_out = OpVariable %ptr_out_pv Output
%outColor = OpVariable %ptr_out_v3 Output
%main = OpFunction %void None %fn
%entry = OpLabel
%p0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%p1_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%p2_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%p3_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_3 %int_0
%p0 = OpLoad %v4float %p0_ptr
%p1 = OpLoad %v4float %p1_ptr
%p2 = OpLoad %v4float %p2_ptr
%p3 = OpLoad %v4float %p3_ptr
%p0xy = OpVectorShuffle %v2float %p0 %p0 0 1
%p1xy = OpVectorShuffle %v2float %p1 %p1 0 1
%p2xy = OpVectorShuffle %v2float %p2 %p2 0 1
%p3xy = OpVectorShuffle %v2float %p3 %p3 0 1
%d0 = OpFSub %v2float %p2xy %p0xy
%d1 = OpFSub %v2float %p3xy %p1xy
%t0 = OpExtInst %v2float %glsl Normalize %d0
%t1 = OpExtInst %v2float %glsl Normalize %d1
%t0x = OpCompositeExtract %float %t0 0
%t0y = OpCompositeExtract %float %t0 1
%t1x = OpCompositeExtract %float %t1 0
%t1y = OpCompositeExtract %float %t1 1
%neg_t0y = OpFNegate %float %t0y
%neg_t1y = OpFNegate %float %t1y
%perp0 = OpCompositeConstruct %v2float %neg_t0y %t0x
%perp1 = OpCompositeConstruct %v2float %neg_t1y %t1x
%offset0 = OpVectorTimesScalar %v2float %perp0 %float_0_025
%offset1 = OpVectorTimesScalar %v2float %perp1 %float_0_025
%start_plus_xy = OpFAdd %v2float %p1xy %offset0
%start_minus_xy = OpFSub %v2float %p1xy %offset0
%end_plus_xy = OpFAdd %v2float %p2xy %offset1
%end_minus_xy = OpFSub %v2float %p2xy %offset1
%p1z = OpCompositeExtract %float %p1 2
%p1w = OpCompositeExtract %float %p1 3
%p2z = OpCompositeExtract %float %p2 2
%p2w = OpCompositeExtract %float %p2 3
%start_plus_x = OpCompositeExtract %float %start_plus_xy 0
%start_plus_y = OpCompositeExtract %float %start_plus_xy 1
%start_minus_x = OpCompositeExtract %float %start_minus_xy 0
%start_minus_y = OpCompositeExtract %float %start_minus_xy 1
%end_plus_x = OpCompositeExtract %float %end_plus_xy 0
%end_plus_y = OpCompositeExtract %float %end_plus_xy 1
%end_minus_x = OpCompositeExtract %float %end_minus_xy 0
%end_minus_y = OpCompositeExtract %float %end_minus_xy 1
%start_plus = OpCompositeConstruct %v4float %start_plus_x %start_plus_y %p1z %p1w
%start_minus = OpCompositeConstruct %v4float %start_minus_x %start_minus_y %p1z %p1w
%end_plus = OpCompositeConstruct %v4float %end_plus_x %end_plus_y %p2z %p2w
%end_minus = OpCompositeConstruct %v4float %end_minus_x %end_minus_y %p2z %p2w
%color1_ptr = OpAccessChain %ptr_in_v3 %color %int_1
%color2_ptr = OpAccessChain %ptr_in_v3 %color %int_2
%color1 = OpLoad %v3float %color1_ptr
%color2 = OpLoad %v3float %color2_ptr
%out_pos_ptr0 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_ptr0 %start_plus
OpStore %outColor %color1
OpEmitVertex
%out_pos_ptr1 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_ptr1 %start_minus
OpStore %outColor %color1
OpEmitVertex
%out_pos_ptr2 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_ptr2 %end_plus
OpStore %outColor %color2
OpEmitVertex
%out_pos_ptr3 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_ptr3 %end_minus
OpStore %outColor %color2
OpEmitVertex
OpEndPrimitive
OpReturn
OpFunctionEnd
