const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const grid = data.grid || [];
const robots = data.robots || [];
const jobs = data.jobs || [];

const height = grid.length;
const width = height ? grid[0].length : 0;
const cellCount = width * height;

function indexOf(x, y) {
  return y * width + x;
}

function isOpen(x, y) {
  return y >= 0 && y < height && x >= 0 && x < width && grid[y][x] !== '#';
}

const maxRobotId = robots.reduce((m, r) => Math.max(m, r.id), -1);
const assignments = Array.from({ length: maxRobotId + 1 }, () => []);

if (jobs.length === 0 || robots.length === 0 || height === 0 || width === 0) {
  console.log(JSON.stringify({ assignments }));
  process.exit(0);
}

const bfsCache = new Map();
const queue = new Int32Array(cellCount);

function bfsFrom(sourceIndex) {
  let cached = bfsCache.get(sourceIndex);
  if (cached) return cached;

  const dist = new Int32Array(cellCount);
  dist.fill(-1);

  let head = 0;
  let tail = 0;
  dist[sourceIndex] = 0;
  queue[tail++] = sourceIndex;

  while (head < tail) {
    const current = queue[head++];
    const x = current % width;
    const y = (current / width) | 0;
    const nextDistance = dist[current] + 1;

    if (x > 0) {
      const n = current - 1;
      if (dist[n] < 0 && grid[y][x - 1] !== '#') {
        dist[n] = nextDistance;
        queue[tail++] = n;
      }
    }
    if (x + 1 < width) {
      const n = current + 1;
      if (dist[n] < 0 && grid[y][x + 1] !== '#') {
        dist[n] = nextDistance;
        queue[tail++] = n;
      }
    }
    if (y > 0) {
      const n = current - width;
      if (dist[n] < 0 && grid[y - 1][x] !== '#') {
        dist[n] = nextDistance;
        queue[tail++] = n;
      }
    }
    if (y + 1 < height) {
      const n = current + width;
      if (dist[n] < 0 && grid[y + 1][x] !== '#') {
        dist[n] = nextDistance;
        queue[tail++] = n;
      }
    }
  }

  bfsCache.set(sourceIndex, dist);
  return dist;
}

function distanceBetween(aIndex, bIndex) {
  const d = bfsFrom(aIndex)[bIndex];
  if (d >= 0) return d;

  const ax = aIndex % width;
  const ay = (aIndex / width) | 0;
  const bx = bIndex % width;
  const by = (bIndex / width) | 0;
  return 1_000_000_000 + Math.abs(ax - bx) + Math.abs(ay - by);
}

const robotState = robots
  .slice()
  .sort((a, b) => a.id - b.id)
  .map((robot) => ({
    id: robot.id,
    pos: indexOf(robot.x, robot.y),
    travel: 0
  }));

const orderedJobs = jobs
  .map((job, inputOrder) => ({
    id: job.id,
    priority: job.priority || 0,
    pos: indexOf(job.x, job.y),
    inputOrder
  }))
  .sort((a, b) => {
    if (b.priority !== a.priority) return b.priority - a.priority;
    if (a.id !== b.id) return a.id < b.id ? -1 : 1;
    return a.inputOrder - b.inputOrder;
  });

for (const job of orderedJobs) {
  let bestRobot = robotState[0];
  let bestAdded = distanceBetween(bestRobot.pos, job.pos);
  let bestTotal = bestRobot.travel + bestAdded;

  for (let i = 1; i < robotState.length; i++) {
    const robot = robotState[i];
    const added = distanceBetween(robot.pos, job.pos);
    const total = robot.travel + added;

    if (
      total < bestTotal ||
      (total === bestTotal && added < bestAdded) ||
      (total === bestTotal && added === bestAdded && robot.id < bestRobot.id)
    ) {
      bestRobot = robot;
      bestAdded = added;
      bestTotal = total;
    }
  }

  assignments[bestRobot.id].push(job.id);
  bestRobot.travel += bestAdded;
  bestRobot.pos = job.pos;
}

console.log(JSON.stringify({ assignments }));
