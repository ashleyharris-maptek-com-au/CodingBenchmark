OpCapability Shader
OpCapability Geometry
%1 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %worldPos %normal %color %gl_out %outColor
OpExecutionMode %main InputLines
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 4
OpExecutionMode %main Invocations 1
OpSource GLSL 450
OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %color Location 2
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %outColor Location 0
%void = OpTypeVoid
%fn_void = OpTypeFunction %void
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_2 = OpConstant %uint 2
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%float_1 = OpConstant %float 1
%float_0_04 = OpConstant %float 0.04
%float_0_005 = OpConstant %float 0.005
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_2
%arr_v3_in = OpTypeArray %v3float %uint_2
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_in
%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float
%gl_in = OpVariable %ptr_in_arr_pv Input
%worldPos = OpVariable %ptr_in_arr_v3 Input
%normal = OpVariable %ptr_in_arr_v3 Input
%color = OpVariable %ptr_in_arr_v3 Input
%gl_out = OpVariable %ptr_out_pv Output
%outColor = OpVariable %ptr_out_v3 Output
%main = OpFunction %void None %fn_void
%entry = OpLabel
%p0_ptr = OpAccessChain %ptr_in_v3 %worldPos %int_0
%p1_ptr = OpAccessChain %ptr_in_v3 %worldPos %int_1
%p0 = OpLoad %v3float %p0_ptr
%p1 = OpLoad %v3float %p1_ptr
%c0_ptr = OpAccessChain %ptr_in_v3 %color %int_0
%c1_ptr = OpAccessChain %ptr_in_v3 %color %int_1
%c0 = OpLoad %v3float %c0_ptr
%c1 = OpLoad %v3float %c1_ptr
%p0xy = OpVectorShuffle %v2float %p0 %p0 0 1
%p1xy = OpVectorShuffle %v2float %p1 %p1 0 1
%line_xy = OpFSub %v2float %p1xy %p0xy
%dx = OpCompositeExtract %float %line_xy 0
%dy = OpCompositeExtract %float %line_xy 1
%neg_dy = OpFNegate %float %dy
%perp_raw = OpCompositeConstruct %v2float %neg_dy %dx
%perp = OpExtInst %v2float %1 Normalize %perp_raw
%perp_start = OpVectorTimesScalar %v2float %perp %float_0_04
%perp_end = OpVectorTimesScalar %v2float %perp %float_0_005
%p0_plus_xy = OpFAdd %v2float %p0xy %perp_start
%p0_minus_xy = OpFSub %v2float %p0xy %perp_start
%p1_plus_xy = OpFAdd %v2float %p1xy %perp_end
%p1_minus_xy = OpFSub %v2float %p1xy %perp_end
%p0_z = OpCompositeExtract %float %p0 2
%p1_z = OpCompositeExtract %float %p1 2
%p0_plus_x = OpCompositeExtract %float %p0_plus_xy 0
%p0_plus_y = OpCompositeExtract %float %p0_plus_xy 1
%pos0 = OpCompositeConstruct %v4float %p0_plus_x %p0_plus_y %p0_z %float_1
%out_pos_ptr0 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_ptr0 %pos0
OpStore %outColor %c0
OpEmitVertex
%p0_minus_x = OpCompositeExtract %float %p0_minus_xy 0
%p0_minus_y = OpCompositeExtract %float %p0_minus_xy 1
%pos1 = OpCompositeConstruct %v4float %p0_minus_x %p0_minus_y %p0_z %float_1
%out_pos_ptr1 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_ptr1 %pos1
OpStore %outColor %c0
OpEmitVertex
%p1_plus_x = OpCompositeExtract %float %p1_plus_xy 0
%p1_plus_y = OpCompositeExtract %float %p1_plus_xy 1
%pos2 = OpCompositeConstruct %v4float %p1_plus_x %p1_plus_y %p1_z %float_1
%out_pos_ptr2 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_ptr2 %pos2
OpStore %outColor %c1
OpEmitVertex
%p1_minus_x = OpCompositeExtract %float %p1_minus_xy 0
%p1_minus_y = OpCompositeExtract %float %p1_minus_xy 1
%pos3 = OpCompositeConstruct %v4float %p1_minus_x %p1_minus_y %p1_z %float_1
%out_pos_ptr3 = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %out_pos_ptr3 %pos3
OpStore %outColor %c1
OpEmitVertex
OpEndPrimitive
OpReturn
OpFunctionEnd
