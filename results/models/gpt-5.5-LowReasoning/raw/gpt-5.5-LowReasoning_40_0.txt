#include <algorithm>
#include <cstdint>
#include <iostream>
#include <numeric>
#include <vector>

using namespace std;

struct Solver {
    int n;
    vector<long long> w;
    vector<vector<unsigned long long>> adj;
    vector<int> best, cur;
    long long bestVal = 0, curVal = 0;

    Solver(int n_, vector<long long> w_) : n(n_), w(std::move(w_)) {
        int blocks = (n + 63) >> 6;
        adj.assign(n, vector<unsigned long long>(blocks, 0));
        best.assign(n, 0);
        cur.assign(n, 0);
    }

    void addEdge(int u, int v) {
        if (u == v) {
            if (u >= 0 && u < n) w[u] = min(w[u], 0LL);
            return;
        }
        if (u < 0 || u >= n || v < 0 || v >= n) return;
        adj[u][v >> 6] |= 1ULL << (v & 63);
        adj[v][u >> 6] |= 1ULL << (u & 63);
    }

    static bool getBit(const vector<unsigned long long>& bits, int i) {
        return (bits[i >> 6] >> (i & 63)) & 1ULL;
    }

    static void clearBit(vector<unsigned long long>& bits, int i) {
        bits[i >> 6] &= ~(1ULL << (i & 63));
    }

    vector<int> vertices(const vector<unsigned long long>& p) const {
        vector<int> out;
        for (int b = 0; b < (int)p.size(); ++b) {
            unsigned long long x = p[b];
            while (x) {
                int t = __builtin_ctzll(x);
                int v = (b << 6) + t;
                if (v < n) out.push_back(v);
                x &= x - 1;
            }
        }
        return out;
    }

    long long cliqueCoverUpperBound(const vector<unsigned long long>& p) const {
        vector<int> vs = vertices(p);
        sort(vs.begin(), vs.end(), [&](int a, int b) {
            if (w[a] != w[b]) return w[a] > w[b];
            return a < b;
        });

        vector<vector<int>> cliques;
        vector<long long> mx;
        for (int v : vs) {
            bool placed = false;
            for (int c = 0; c < (int)cliques.size() && !placed; ++c) {
                bool ok = true;
                for (int u : cliques[c]) {
                    if (!getBit(adj[v], u)) {
                        ok = false;
                        break;
                    }
                }
                if (ok) {
                    cliques[c].push_back(v);
                    mx[c] = max(mx[c], w[v]);
                    placed = true;
                }
            }
            if (!placed) {
                cliques.push_back(vector<int>{v});
                mx.push_back(w[v]);
            }
        }

        long long ub = 0;
        for (long long x : mx) ub += max(0LL, x);
        return ub;
    }

    bool emptySet(const vector<unsigned long long>& p) const {
        for (auto x : p) if (x) return false;
        return true;
    }

    int chooseVertex(const vector<unsigned long long>& p) const {
        vector<int> vs = vertices(p);
        int bestV = vs[0];
        long long bestScore = -1;
        for (int v : vs) {
            long long deg = 0;
            for (int b = 0; b < (int)p.size(); ++b) {
                deg += __builtin_popcountll(p[b] & adj[v][b]);
            }
            long long score = deg * 1000003LL + max(0LL, w[v]);
            if (score > bestScore) {
                bestScore = score;
                bestV = v;
            }
        }
        return bestV;
    }

    void saveBest() {
        if (curVal > bestVal) {
            bestVal = curVal;
            best = cur;
        }
    }

    void dfs(vector<unsigned long long>& p) {
        if (emptySet(p)) {
            saveBest();
            return;
        }

        if (curVal + cliqueCoverUpperBound(p) <= bestVal) return;

        int v = chooseVertex(p);

        if (w[v] > 0) {
            vector<unsigned long long> with = p;
            clearBit(with, v);
            for (int b = 0; b < (int)with.size(); ++b) with[b] &= ~adj[v][b];
            cur[v] = 1;
            curVal += w[v];
            dfs(with);
            curVal -= w[v];
            cur[v] = 0;
        }

        clearBit(p, v);
        dfs(p);
    }

    void solve() {
        vector<unsigned long long> p((n + 63) >> 6, 0);
        for (int i = 0; i < n; ++i) {
            if (w[i] > 0) p[i >> 6] |= 1ULL << (i & 63);
        }
        dfs(p);
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    if (!(cin >> n >> m)) return 0;

    vector<long long> c(n);
    for (int i = 0; i < n; ++i) cin >> c[i];

    Solver solver(n, c);
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        solver.addEdge(u, v);
    }

    solver.solve();

    cout << solver.bestVal << '\n';
    for (int i = 0; i < n; ++i) {
        if (i) cout << ' ';
        cout << solver.best[i];
    }
    cout << '\n';
    return 0;
}
