; SPIR-V
; Version: 1.0
OpCapability Shader
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %inWorldPos %inNormal %inUV %inTangent %inColor %outColor
OpExecutionMode %main OriginUpperLeft
OpDecorate %inWorldPos Location 0
OpDecorate %inNormal Location 1
OpDecorate %inUV Location 2
OpDecorate %inTangent Location 3
OpDecorate %inColor Location 4
OpDecorate %outColor Location 0
OpDecorate %UBO Block
OpMemberDecorate %UBO 0 ColMajor
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 0 MatrixStride 16
OpMemberDecorate %UBO 1 ColMajor
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 1 MatrixStride 16
OpMemberDecorate %UBO 2 ColMajor
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 2 MatrixStride 16
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0
%void = OpTypeVoid
%fn_void = OpTypeFunction %void
%bool = OpTypeBool
%float = OpTypeFloat 32
%int = OpTypeInt 32 1
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%mat4 = OpTypeMatrix %v4float 4
%UBO = OpTypeStruct %mat4 %mat4 %mat4 %v4float %v4float %v4float
%ptr_Input_v2float = OpTypePointer Input %v2float
%ptr_Input_v3float = OpTypePointer Input %v3float
%ptr_Output_v4float = OpTypePointer Output %v4float
%ptr_Uniform_UBO = OpTypePointer Uniform %UBO
%ptr_Function_v2float = OpTypePointer Function %v2float
%ptr_Function_int = OpTypePointer Function %int
%inWorldPos = OpVariable %ptr_Input_v3float Input
%inNormal = OpVariable %ptr_Input_v3float Input
%inUV = OpVariable %ptr_Input_v2float Input
%inTangent = OpVariable %ptr_Input_v3float Input
%inColor = OpVariable %ptr_Input_v3float Input
%outColor = OpVariable %ptr_Output_v4float Output
%ubo = OpVariable %ptr_Uniform_UBO Uniform
%float_0 = OpConstant %float 0
%float_0_5 = OpConstant %float 0.5
%float_1 = OpConstant %float 1
%float_2 = OpConstant %float 2
%float_3 = OpConstant %float 3
%float_4 = OpConstant %float 4
%float_20 = OpConstant %float 20
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_20 = OpConstant %int 20
%v2float_0 = OpConstantComposite %v2float %float_0 %float_0
%main = OpFunction %void None %fn_void
%entry = OpLabel
%z = OpVariable %ptr_Function_v2float Function
%iterCount = OpVariable %ptr_Function_int Function
OpStore %z %v2float_0
OpStore %iterCount %int_0
%uv = OpLoad %v2float %inUV
%uvx = OpCompositeExtract %float %uv 0
%uvy = OpCompositeExtract %float %uv 1
%cx_mul = OpFMul %float %uvx %float_3
%cx = OpFSub %float %cx_mul %float_2
%cy_mul = OpFMul %float %uvy %float_2
%cy = OpFSub %float %cy_mul %float_1
%c = OpCompositeConstruct %v2float %cx %cy
OpBranch %loop
%loop = OpLabel
OpLoopMerge %loopMerge %loopContinue None
OpBranch %loopBody
%loopBody = OpLabel
%i = OpLoad %int %iterCount
%keepGoing = OpSLessThan %bool %i %int_20
OpBranchConditional %keepGoing %iterate %loopMerge
%iterate = OpLabel
%zv = OpLoad %v2float %z
%zx = OpCompositeExtract %float %zv 0
%zy = OpCompositeExtract %float %zv 1
%zx2 = OpFMul %float %zx %zx
%zy2 = OpFMul %float %zy %zy
%zxy = OpFMul %float %zx %zy
%two_zxy = OpFMul %float %zxy %float_2
%newx_base = OpFSub %float %zx2 %zy2
%newx = OpFAdd %float %newx_base %cx
%newy = OpFAdd %float %two_zxy %cy
%newz = OpCompositeConstruct %v2float %newx %newy
OpStore %z %newz
%nextI = OpIAdd %int %i %int_1
OpStore %iterCount %nextI
%newx2 = OpFMul %float %newx %newx
%newy2 = OpFMul %float %newy %newy
%mag2 = OpFAdd %float %newx2 %newy2
%escaped = OpFOrdGreaterThan %bool %mag2 %float_4
OpSelectionMerge %afterEscape None
OpBranchConditional %escaped %loopMerge %afterEscape
%afterEscape = OpLabel
OpBranch %loopContinue
%loopContinue = OpLabel
OpBranch %loop
%loopMerge = OpLabel
%finalI = OpLoad %int %iterCount
%finalF = OpConvertSToF %float %finalI
%t = OpFDiv %float %finalF %float_20
%g = OpFMul %float %t %float_0_5
%b = OpFSub %float %float_1 %t
%color = OpCompositeConstruct %v4float %t %g %b %float_1
OpStore %outColor %color
OpReturn
OpFunctionEnd
