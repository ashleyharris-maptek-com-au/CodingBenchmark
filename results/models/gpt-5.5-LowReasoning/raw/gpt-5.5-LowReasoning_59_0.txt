const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const grid = data.probability || [];
const drones = data.drones || [];
const maxSteps = Math.max(0, data.maxStepsPerDrone | 0);
const height = grid.length;
const width = height ? grid[0].length : 0;
const total = width * height;

function idx(x, y) {
  return y * width + x;
}

function xy(i) {
  return [i % width, Math.floor(i / width)];
}

function inBounds(x, y) {
  return x >= 0 && y >= 0 && x < width && y < height;
}

function isOpenIndex(i) {
  const y = Math.floor(i / width);
  const x = i - y * width;
  return grid[y][x] !== -1;
}

if (!height || !width || !Array.isArray(drones)) {
  process.stdout.write(JSON.stringify({ paths: [] }));
  process.exit(0);
}

const values = new Float64Array(total);
let openCount = 0;
for (let y = 0; y < height; y++) {
  for (let x = 0; x < width; x++) {
    const i = idx(x, y);
    const v = Number(grid[y][x]);
    if (v !== -1) {
      openCount++;
      values[i] = Math.max(0, v);
    } else {
      values[i] = -1;
    }
  }
}

const covered = new Uint8Array(total);
const seen = new Int32Array(total);
const dist = new Int32Array(total);
const parent = new Int32Array(total);
const queue = new Int32Array(Math.max(1, total));
let stamp = 1;

const dirs = [
  [1, 0],
  [-1, 0],
  [0, 1],
  [0, -1]
];

function localPotential(i) {
  let score = covered[i] ? 0 : Math.max(0, values[i]);
  const x = i % width;
  const y = Math.floor(i / width);
  if (x > 0) {
    const n = i - 1;
    if (!covered[n] && values[n] > 0) score += values[n] * 0.22;
  }
  if (x + 1 < width) {
    const n = i + 1;
    if (!covered[n] && values[n] > 0) score += values[n] * 0.22;
  }
  if (y > 0) {
    const n = i - width;
    if (!covered[n] && values[n] > 0) score += values[n] * 0.22;
  }
  if (y + 1 < height) {
    const n = i + width;
    if (!covered[n] && values[n] > 0) score += values[n] * 0.22;
  }
  return score;
}

function findTargetFast(start, remaining) {
  if (remaining <= 0 || start < 0 || start >= total || !isOpenIndex(start)) {
    return { target: start, distance: 0 };
  }

  stamp++;
  if (stamp === 2147483647) {
    seen.fill(0);
    stamp = 1;
  }

  let head = 0;
  let tail = 0;
  queue[tail++] = start;
  seen[start] = stamp;
  dist[start] = 0;
  parent[start] = -1;

  let best = -1;
  let bestScore = 0;
  let bestDist = 0;
  const exploreLimit = Math.min(openCount, Math.max(3000, Math.min(100000, remaining * 4000 + 3000)));

  while (head < tail && tail <= exploreLimit) {
    const cur = queue[head++];
    const d = dist[cur];
    const pot = localPotential(cur);
    if (pot > 0) {
      const score = pot / Math.pow(d + 1, 0.78);
      if (score > bestScore || (score === bestScore && d < bestDist)) {
        bestScore = score;
        best = cur;
        bestDist = d;
      }
    }
    if (d >= remaining) continue;

    const x = cur % width;
    const y = Math.floor(cur / width);
    let n;
    if (x > 0) {
      n = cur - 1;
      if (seen[n] !== stamp && values[n] !== -1) {
        seen[n] = stamp; dist[n] = d + 1; parent[n] = cur; queue[tail++] = n;
      }
    }
    if (x + 1 < width) {
      n = cur + 1;
      if (seen[n] !== stamp && values[n] !== -1) {
        seen[n] = stamp; dist[n] = d + 1; parent[n] = cur; queue[tail++] = n;
      }
    }
    if (y > 0) {
      n = cur - width;
      if (seen[n] !== stamp && values[n] !== -1) {
        seen[n] = stamp; dist[n] = d + 1; parent[n] = cur; queue[tail++] = n;
      }
    }
    if (y + 1 < height) {
      n = cur + width;
      if (seen[n] !== stamp && values[n] !== -1) {
        seen[n] = stamp; dist[n] = d + 1; parent[n] = cur; queue[tail++] = n;
      }
    }
  }

  if (best !== -1) return { target: best, distance: bestDist };
  return { target: start, distance: 0 };
}

function appendRoute(path, target, current) {
  const rev = [];
  let p = target;
  while (p !== -1 && p !== current) {
    rev.push(p);
    p = parent[p];
  }
  for (let i = rev.length - 1; i >= 0; i--) {
    const cell = rev[i];
    path.push(xy(cell));
    covered[cell] = 1;
  }
}

function fallbackStart(drone) {
  let x = Math.trunc(Number(drone.x));
  let y = Math.trunc(Number(drone.y));
  if (inBounds(x, y) && grid[y][x] !== -1) return idx(x, y);

  for (let yy = 0; yy < height; yy++) {
    for (let xx = 0; xx < width; xx++) {
      if (grid[yy][xx] !== -1) return idx(xx, yy);
    }
  }
  return 0;
}

const paths = [];

for (const drone of drones) {
  let current = fallbackStart(drone);
  const path = [xy(current)];
  if (values[current] !== -1) covered[current] = 1;

  let remaining = maxSteps;
  let idleGuard = 0;
  while (remaining > 0 && idleGuard < 3) {
    const choice = findTargetFast(current, remaining);
    if (choice.target === current || choice.distance === 0) {
      if (!covered[current] && values[current] > 0) {
        covered[current] = 1;
      } else {
        idleGuard++;
        path.push(xy(current));
        remaining--;
      }
      continue;
    }

    appendRoute(path, choice.target, current);
    remaining -= choice.distance;
    current = choice.target;
    idleGuard = 0;
  }

  while (remaining > 0 && path.length < maxSteps + 1) {
    path.push(xy(current));
    remaining--;
  }
  paths.push(path.slice(0, maxSteps + 1));
}

process.stdout.write(JSON.stringify({ paths }));
