; SPIR-V
; Version: 1.0
OpCapability Shader
%1 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %inWorldPos %inNormal %inUV %outColor
OpExecutionMode %main OriginUpperLeft
OpSource GLSL 450
OpName %main "main"
OpName %inWorldPos "worldPos"
OpName %inNormal "normal"
OpName %inUV "uv"
OpName %outColor "outColor"
OpName %UBO "UBO"
OpMemberName %UBO 0 "model"
OpMemberName %UBO 1 "view"
OpMemberName %UBO 2 "projection"
OpMemberName %UBO 3 "lightPos"
OpMemberName %UBO 4 "cameraPos"
OpMemberName %UBO 5 "params"
OpName %ubo "ubo"
OpDecorate %inWorldPos Location 0
OpDecorate %inNormal Location 1
OpDecorate %inUV Location 2
OpDecorate %outColor Location 0
OpMemberDecorate %UBO 0 ColMajor
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 0 MatrixStride 16
OpMemberDecorate %UBO 1 ColMajor
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 1 MatrixStride 16
OpMemberDecorate %UBO 2 ColMajor
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 2 MatrixStride 16
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %UBO Block
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0
%void = OpTypeVoid
%3 = OpTypeFunction %void
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%mat4v4float = OpTypeMatrix %v4float 4
%ptr_Input_v3float = OpTypePointer Input %v3float
%ptr_Input_v2float = OpTypePointer Input %v2float
%ptr_Output_v4float = OpTypePointer Output %v4float
%ptr_Uniform_v4float = OpTypePointer Uniform %v4float
%ptr_Uniform_UBO = OpTypePointer Uniform %UBO
%UBO = OpTypeStruct %mat4v4float %mat4v4float %mat4v4float %v4float %v4float %v4float
%inWorldPos = OpVariable %ptr_Input_v3float Input
%inNormal = OpVariable %ptr_Input_v3float Input
%inUV = OpVariable %ptr_Input_v2float Input
%outColor = OpVariable %ptr_Output_v4float Output
%ubo = OpVariable %ptr_Uniform_UBO Uniform
%int = OpTypeInt 32 1
%bool = OpTypeBool
%int_3 = OpConstant %int 3
%uint_0 = OpConstant %int 0
%uint_1 = OpConstant %int 1
%float_0 = OpConstant %float 0
%float_1 = OpConstant %float 1
%float_0_1 = OpConstant %float 0.1
%float_0_4 = OpConstant %float 0.4
%float_0_6 = OpConstant %float 0.6
%float_0_9 = OpConstant %float 0.9
%float_pi = OpConstant %float 3.14159
%float_16 = OpConstant %float 16
%red = OpConstantComposite %v3float %float_0_9 %float_0_1 %float_0_1
%blue = OpConstantComposite %v3float %float_0_1 %float_0_1 %float_0_9
%main = OpFunction %void None %3
%5 = OpLabel
%uv = OpLoad %v2float %inUV
%uv_y = OpCompositeExtract %float %uv 1
%angle_a = OpFMul %float %uv_y %float_pi
%angle = OpFMul %float %angle_a %float_16
%wave = OpExtInst %float %1 Sin %angle
%is_red = OpFOrdGreaterThan %bool %wave %float_0
%base_color = OpSelect %v3float %is_red %red %blue
%n0 = OpLoad %v3float %inNormal
%n = OpExtInst %v3float %1 Normalize %n0
%world_pos = OpLoad %v3float %inWorldPos
%light_ptr = OpAccessChain %ptr_Uniform_v4float %ubo %int_3
%light4 = OpLoad %v4float %light_ptr
%light_pos = OpVectorShuffle %v3float %light4 %light4 0 1 2
%to_light0 = OpFSub %v3float %light_pos %world_pos
%to_light = OpExtInst %v3float %1 Normalize %to_light0
%ndotl = OpDot %float %n %to_light
%diffuse = OpExtInst %float %1 FMax %ndotl %float_0
%shade_a = OpFMul %float %float_0_6 %diffuse
%shade = OpFAdd %float %float_0_4 %shade_a
%shaded_rgb = OpVectorTimesScalar %v3float %base_color %shade
%r = OpCompositeExtract %float %shaded_rgb 0
%g = OpCompositeExtract %float %shaded_rgb 1
%b = OpCompositeExtract %float %shaded_rgb 2
%color = OpCompositeConstruct %v4float %r %g %b %float_1
OpStore %outColor %color
OpReturn
OpFunctionEnd
