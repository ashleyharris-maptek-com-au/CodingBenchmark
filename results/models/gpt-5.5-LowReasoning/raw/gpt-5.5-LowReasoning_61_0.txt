declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type Pkg = { id: any; x: number; y: number; dx: number; dy: number; r: number; a: number; idx: number };

const depotX = Number(data.depot[0]);
const depotY = Number(data.depot[1]);
const truckCount = Math.max(1, Math.floor(Number(data.truckCount) || 1));
const raw = Array.isArray(data.packages) ? data.packages : [];

const pkgs: Pkg[] = raw.map((p: any, i: number) => {
  const x = Number(p.x);
  const y = Number(p.y);
  const dx = x - depotX;
  const dy = y - depotY;
  return { id: p.id, x, y, dx, dy, r: Math.hypot(dx, dy), a: Math.atan2(dy, dx), idx: i };
});

if (pkgs.length === 0) {
  console.log(JSON.stringify({ routes: [] }));
} else {
  const maxRoutes = Math.min(truckCount, pkgs.length);
  const startTime = Date.now();
  const TIME_LIMIT_MS = pkgs.length <= 400 ? 8500 : pkgs.length <= 1200 ? 7500 : 6200;

  function dist(a: Pkg, b: Pkg): number {
    return Math.hypot(a.x - b.x, a.y - b.y);
  }

  function routeLength(route: Pkg[]): number {
    if (route.length === 0) return 0;
    let s = route[0].r + route[route.length - 1].r;
    for (let i = 1; i < route.length; i++) s += dist(route[i - 1], route[i]);
    return s;
  }

  function totalLength(routes: Pkg[][]): number {
    let s = 0;
    for (const r of routes) s += routeLength(r);
    return s;
  }

  function nearestNeighbor(items: Pkg[]): Pkg[] {
    const n = items.length;
    if (n <= 2) return items.slice();
    const used = new Array<boolean>(n).fill(false);
    let bestStart = 0;
    for (let i = 1; i < n; i++) if (items[i].r > items[bestStart].r) bestStart = i;
    const out: Pkg[] = [items[bestStart]];
    used[bestStart] = true;
    let cur = items[bestStart];
    for (let step = 1; step < n; step++) {
      let best = -1;
      let bd = Infinity;
      for (let i = 0; i < n; i++) {
        if (!used[i]) {
          const d = dist(cur, items[i]);
          if (d < bd || (d === bd && items[i].idx < items[best]?.idx)) {
            bd = d;
            best = i;
          }
        }
      }
      used[best] = true;
      cur = items[best];
      out.push(cur);
    }
    return out;
  }

  function twoOpt(route: Pkg[], limitPasses: number): Pkg[] {
    const n = route.length;
    if (n < 4) return route;
    let passes = 0;
    let improved = true;
    while (improved && passes++ < limitPasses && Date.now() - startTime < TIME_LIMIT_MS) {
      improved = false;
      for (let i = 0; i < n - 2; i++) {
        const aDepot = i === 0;
        const a = route[i];
        const b = route[i + 1];
        const oldLeft = aDepot ? a.r : dist(route[i - 1], a);
        for (let j = i + 2; j < n; j++) {
          const c = route[j];
          const dNode = j + 1 < n ? route[j + 1] : undefined;
          const oldRight = dNode ? dist(c, dNode) : c.r;
          const neuLeft = aDepot ? c.r : dist(route[i - 1], c);
          const neuRight = dNode ? dist(a, dNode) : a.r;
          if (neuLeft + neuRight + 1e-9 < oldLeft + oldRight) {
            let l = i, rr = j;
            while (l < rr) {
              const t = route[l];
              route[l++] = route[rr];
              route[rr--] = t;
            }
            improved = true;
          }
        }
      }
    }
    return route;
  }

  function optimizeRoute(items: Pkg[]): Pkg[] {
    if (items.length <= 1) return items.slice();
    const byAngle = items.slice().sort((a, b) => a.a - b.a || b.r - a.r || a.idx - b.idx);
    const nn = nearestNeighbor(items);
    let best = routeLength(byAngle) <= routeLength(nn) ? byAngle : nn;
    const passes = items.length <= 160 ? 9 : items.length <= 500 ? 5 : 2;
    best = twoOpt(best.slice(), passes);
    const rev = best.slice().reverse();
    return routeLength(rev) < routeLength(best) ? rev : best;
  }

  function sweepRoutes(count: number, phase: number, radialMix: boolean): Pkg[][] {
    const sorted = pkgs.slice().sort((p, q) => {
      let aa = p.a + phase;
      let bb = q.a + phase;
      if (aa > Math.PI) aa -= Math.PI * 2;
      if (bb > Math.PI) bb -= Math.PI * 2;
      if (aa < -Math.PI) aa += Math.PI * 2;
      if (bb < -Math.PI) bb += Math.PI * 2;
      return aa - bb || (radialMix ? p.r - q.r : q.r - p.r) || p.idx - q.idx;
    });
    const routes: Pkg[][] = [];
    for (let i = 0; i < count; i++) {
      const lo = Math.floor(i * sorted.length / count);
      const hi = Math.floor((i + 1) * sorted.length / count);
      if (hi > lo) routes.push(optimizeRoute(sorted.slice(lo, hi)));
    }
    return routes;
  }

  function deterministicKMeans(count: number): Pkg[][] {
    const centers: { x: number; y: number }[] = [];
    let first = 0;
    for (let i = 1; i < pkgs.length; i++) if (pkgs[i].r > pkgs[first].r) first = i;
    centers.push({ x: pkgs[first].x, y: pkgs[first].y });
    while (centers.length < count) {
      let best = pkgs[0], bd = -1;
      for (const p of pkgs) {
        let md = Infinity;
        for (const c of centers) {
          const d = (p.x - c.x) * (p.x - c.x) + (p.y - c.y) * (p.y - c.y);
          if (d < md) md = d;
        }
        if (md > bd || (md === bd && p.idx < best.idx)) { bd = md; best = p; }
      }
      centers.push({ x: best.x, y: best.y });
    }
    let groups: Pkg[][] = [];
    for (let iter = 0; iter < 8; iter++) {
      groups = Array.from({ length: count }, () => []);
      for (const p of pkgs) {
        let bi = 0, bd = Infinity;
        for (let i = 0; i < centers.length; i++) {
          const c = centers[i];
          const d = (p.x - c.x) * (p.x - c.x) + (p.y - c.y) * (p.y - c.y) + 0.05 * p.r * p.r;
          if (d < bd) { bd = d; bi = i; }
        }
        groups[bi].push(p);
      }
      for (let i = 0; i < count; i++) {
        if (groups[i].length === 0) continue;
        let sx = 0, sy = 0;
        for (const p of groups[i]) { sx += p.x; sy += p.y; }
        centers[i] = { x: sx / groups[i].length, y: sy / groups[i].length };
      }
    }
    return groups.filter(g => g.length).map(g => optimizeRoute(g));
  }

  let bestRoutes: Pkg[][] = [optimizeRoute(pkgs)];
  let bestScore = totalLength(bestRoutes);

  function consider(routes: Pkg[][]) {
    const s = totalLength(routes);
    if (s < bestScore) { bestScore = s; bestRoutes = routes; }
  }

  const routeCounts = new Set<number>([1, maxRoutes]);
  for (let m = 2; m <= maxRoutes; m++) {
    if (m <= 12 || m === maxRoutes || m % Math.ceil(maxRoutes / 10) === 0) routeCounts.add(m);
  }

  const phases = [0, Math.PI / 6, Math.PI / 3, Math.PI / 2, -Math.PI / 4];
  for (const m of Array.from(routeCounts).sort((a, b) => a - b)) {
    if (Date.now() - startTime > TIME_LIMIT_MS) break;
    for (const ph of phases) {
      consider(sweepRoutes(m, ph, false));
      if (Date.now() - startTime > TIME_LIMIT_MS) break;
      consider(sweepRoutes(m, ph, true));
    }
    if (m <= Math.min(maxRoutes, 20) && pkgs.length <= 2500 && Date.now() - startTime < TIME_LIMIT_MS) {
      consider(deterministicKMeans(m));
    }
  }

  const routes = bestRoutes
    .filter(r => r.length > 0)
    .map(r => r.map(p => p.id));
  console.log(JSON.stringify({ routes }));
}
