; SPIR-V
; Version: 1.0

               OpCapability Shader
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Fragment %main "main" %inWorldPos %inNormal %outColor
               OpExecutionMode %main OriginUpperLeft

               OpName %main "main"
               OpName %inWorldPos "worldPos"
               OpName %inNormal "normal"
               OpName %outColor "outColor"
               OpName %UBO "UBO"
               OpMemberName %UBO 0 "model"
               OpMemberName %UBO 1 "view"
               OpMemberName %UBO 2 "projection"
               OpMemberName %UBO 3 "lightPos"
               OpMemberName %UBO 4 "cameraPos"
               OpMemberName %UBO 5 "params"
               OpName %ubo "ubo"

               OpDecorate %inWorldPos Location 0
               OpDecorate %inNormal Location 1
               OpDecorate %outColor Location 0
               OpDecorate %mat4v4float ColMajor
               OpDecorate %mat4v4float MatrixStride 16
               OpMemberDecorate %UBO 0 ColMajor
               OpMemberDecorate %UBO 0 Offset 0
               OpMemberDecorate %UBO 0 MatrixStride 16
               OpMemberDecorate %UBO 1 ColMajor
               OpMemberDecorate %UBO 1 Offset 64
               OpMemberDecorate %UBO 1 MatrixStride 16
               OpMemberDecorate %UBO 2 ColMajor
               OpMemberDecorate %UBO 2 Offset 128
               OpMemberDecorate %UBO 2 MatrixStride 16
               OpMemberDecorate %UBO 3 Offset 192
               OpMemberDecorate %UBO 4 Offset 208
               OpMemberDecorate %UBO 5 Offset 224
               OpDecorate %UBO Block
               OpDecorate %ubo DescriptorSet 0
               OpDecorate %ubo Binding 0

       %void = OpTypeVoid
          %3 = OpTypeFunction %void
      %float = OpTypeFloat 32
    %v3float = OpTypeVector %float 3
    %v4float = OpTypeVector %float 4
%_ptr_Input_v3float = OpTypePointer Input %v3float
%_ptr_Output_v4float = OpTypePointer Output %v4float
%mat4v4float = OpTypeMatrix %v4float 4
        %UBO = OpTypeStruct %mat4v4float %mat4v4float %mat4v4float %v4float %v4float %v4float
%_ptr_Uniform_UBO = OpTypePointer Uniform %UBO
        %int = OpTypeInt 32 1
%_ptr_Uniform_v4float = OpTypePointer Uniform %v4float

   %float_0 = OpConstant %float 0
 %float_0_2 = OpConstant %float 0.2
 %float_0_3 = OpConstant %float 0.3
 %float_0_5 = OpConstant %float 0.5
 %float_0_8 = OpConstant %float 0.8
   %float_1 = OpConstant %float 1
   %float_3 = OpConstant %float 3
      %int_3 = OpConstant %int 3
      %int_4 = OpConstant %int 4

%baseColor = OpConstantComposite %v3float %float_0_2 %float_0_3 %float_0_8
   %rimTint = OpConstantComposite %v3float %float_1 %float_0_8 %float_0_5

 %inWorldPos = OpVariable %_ptr_Input_v3float Input
   %inNormal = OpVariable %_ptr_Input_v3float Input
   %outColor = OpVariable %_ptr_Output_v4float Output
        %ubo = OpVariable %_ptr_Uniform_UBO Uniform

       %main = OpFunction %void None %3
          %5 = OpLabel
   %worldPos = OpLoad %v3float %inWorldPos
  %normalRaw = OpLoad %v3float %inNormal
          %N = OpExtInst %v3float %1 Normalize %normalRaw

%lightPosPtr = OpAccessChain %_ptr_Uniform_v4float %ubo %int_3
%lightPos4 = OpLoad %v4float %lightPosPtr
  %lightPos = OpVectorShuffle %v3float %lightPos4 %lightPos4 0 1 2
       %Lraw = OpFSub %v3float %lightPos %worldPos
          %L = OpExtInst %v3float %1 Normalize %Lraw
      %ndotl = OpDot %float %N %L
    %diffuse = OpExtInst %float %1 FMax %ndotl %float_0

%cameraPosPtr = OpAccessChain %_ptr_Uniform_v4float %ubo %int_4
%cameraPos4 = OpLoad %v4float %cameraPosPtr
  %cameraPos = OpVectorShuffle %v3float %cameraPos4 %cameraPos4 0 1 2
       %Vraw = OpFSub %v3float %cameraPos %worldPos
          %V = OpExtInst %v3float %1 Normalize %Vraw
      %ndotv = OpDot %float %N %V
   %viewTerm = OpExtInst %float %1 FMax %ndotv %float_0
    %oneMinus = OpFSub %float %float_1 %viewTerm
        %rim = OpExtInst %float %1 Pow %oneMinus %float_3

%diffuseHalf = OpFMul %float %float_0_5 %diffuse
      %scale = OpFAdd %float %float_0_3 %diffuseHalf
%baseScaled = OpVectorTimesScalar %v3float %baseColor %scale
  %rimColor = OpVectorTimesScalar %v3float %rimTint %rim
      %color = OpFAdd %v3float %baseScaled %rimColor
   %outValue = OpCompositeConstruct %v4float %color %float_1
               OpStore %outColor %outValue
               OpReturn
               OpFunctionEnd
