#include <algorithm>
#include <cstdint>
#include <iostream>
#include <numeric>
#include <queue>
#include <unordered_set>
#include <vector>

using namespace std;

class MaxClique {
    int n;
    vector<vector<unsigned long long>> adj;
    vector<int> best, cur;

    static int popcount64(unsigned long long x) {
        int c = 0;
        while (x) {
            x &= x - 1;
            ++c;
        }
        return c;
    }

    bool hasEdge(int a, int b) const {
        return (adj[a][b >> 6] >> (b & 63)) & 1ULL;
    }

    vector<int> intersectPrefix(const vector<int>& p, int upto, int v) const {
        vector<int> r;
        r.reserve(upto);
        for (int i = 0; i < upto; ++i) {
            if (hasEdge(p[i], v)) r.push_back(p[i]);
        }
        return r;
    }

    void colorSort(const vector<int>& p, vector<int>& out, vector<int>& colors) const {
        vector<int> rem = p;
        out.clear();
        colors.clear();
        int color = 0;
        while (!rem.empty()) {
            ++color;
            vector<int> nextRem;
            vector<int> thisColor;
            for (int v : rem) {
                bool ok = true;
                for (int u : thisColor) {
                    if (hasEdge(u, v)) {
                        ok = false;
                        break;
                    }
                }
                if (ok) {
                    thisColor.push_back(v);
                    out.push_back(v);
                    colors.push_back(color);
                } else {
                    nextRem.push_back(v);
                }
            }
            rem.swap(nextRem);
        }
    }

    void expand(vector<int> p) {
        vector<int> ordered, colors;
        colorSort(p, ordered, colors);
        for (int i = (int)ordered.size() - 1; i >= 0; --i) {
            if ((int)cur.size() + colors[i] <= (int)best.size()) return;
            int v = ordered[i];
            cur.push_back(v);
            vector<int> np = intersectPrefix(ordered, i, v);
            if (np.empty()) {
                if (cur.size() > best.size()) best = cur;
            } else {
                expand(np);
            }
            cur.pop_back();
        }
    }

public:
    explicit MaxClique(const vector<vector<unsigned long long>>& complementAdj)
        : n((int)complementAdj.size()), adj(complementAdj) {}

    vector<int> solve() {
        vector<int> p(n);
        iota(p.begin(), p.end(), 0);
        sort(p.begin(), p.end(), [&](int a, int b) {
            int da = 0, db = 0;
            for (auto x : adj[a]) da += popcount64(x);
            for (auto x : adj[b]) db += popcount64(x);
            return da > db;
        });
        expand(p);
        return best;
    }
};

static vector<int> solveBipartiteComponent(
    const vector<int>& comp,
    const vector<vector<int>>& g,
    const vector<int>& side
) {
    int n = (int)g.size();
    vector<int> left;
    for (int v : comp) if (side[v] == 0) left.push_back(v);

    vector<int> pairU(n, -1), pairV(n, -1), dist(n);
    auto bfs = [&]() {
        queue<int> q;
        bool found = false;
        for (int u : left) {
            if (pairU[u] == -1) {
                dist[u] = 0;
                q.push(u);
            } else {
                dist[u] = -1;
            }
        }
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (int v : g[u]) {
                int u2 = pairV[v];
                if (u2 == -1) {
                    found = true;
                } else if (dist[u2] == -1) {
                    dist[u2] = dist[u] + 1;
                    q.push(u2);
                }
            }
        }
        return found;
    };

    auto dfs = [&](auto&& self, int u) -> bool {
        for (int v : g[u]) {
            int u2 = pairV[v];
            if (u2 == -1 || (dist[u2] == dist[u] + 1 && self(self, u2))) {
                pairU[u] = v;
                pairV[v] = u;
                return true;
            }
        }
        dist[u] = -1;
        return false;
    };

    while (bfs()) {
        for (int u : left) {
            if (pairU[u] == -1) dfs(dfs, u);
        }
    }

    vector<char> visL(n, 0), visR(n, 0);
    queue<int> q;
    for (int u : left) {
        if (pairU[u] == -1) {
            visL[u] = 1;
            q.push(u);
        }
    }
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (int v : g[u]) {
            if (pairU[u] != v && !visR[v]) {
                visR[v] = 1;
                int u2 = pairV[v];
                if (u2 != -1 && !visL[u2]) {
                    visL[u2] = 1;
                    q.push(u2);
                }
            }
        }
    }

    vector<int> independent;
    for (int v : comp) {
        bool inMinVertexCover = (side[v] == 0) ? !visL[v] : visR[v];
        if (!inMinVertexCover) independent.push_back(v);
    }
    return independent;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    if (!(cin >> n >> m)) return 0;

    vector<vector<int>> g(n);
    vector<unordered_set<int>> edgeSet(n);
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        if (u == v || u < 0 || v < 0 || u >= n || v >= n) continue;
        if (edgeSet[u].insert(v).second) {
            edgeSet[v].insert(u);
            g[u].push_back(v);
            g[v].push_back(u);
        }
    }

    vector<int> answer;
    vector<char> seen(n, 0);

    for (int start = 0; start < n; ++start) {
        if (seen[start]) continue;

        vector<int> comp;
        queue<int> q;
        q.push(start);
        seen[start] = 1;
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            comp.push_back(v);
            for (int to : g[v]) {
                if (!seen[to]) {
                    seen[to] = 1;
                    q.push(to);
                }
            }
        }

        if (comp.size() == 1) {
            answer.push_back(comp[0]);
            continue;
        }

        long long internalEdges = 0;
        for (int v : comp) internalEdges += (long long)g[v].size();
        internalEdges /= 2;
        long long sll = (long long)comp.size();
        if (internalEdges == sll * (sll - 1) / 2) {
            answer.push_back(comp[0]);
            continue;
        }

        vector<int> side(n, -1);
        bool bip = true;
        side[comp[0]] = 0;
        q.push(comp[0]);
        while (!q.empty() && bip) {
            int v = q.front();
            q.pop();
            for (int to : g[v]) {
                if (side[to] == -1) {
                    side[to] = side[v] ^ 1;
                    q.push(to);
                } else if (side[to] == side[v]) {
                    bip = false;
                    break;
                }
            }
        }

        if (bip) {
            vector<int> part = solveBipartiteComponent(comp, g, side);
            answer.insert(answer.end(), part.begin(), part.end());
            continue;
        }

        int s = (int)comp.size();
        vector<int> pos(n, -1);
        for (int i = 0; i < s; ++i) pos[comp[i]] = i;

        int words = (s + 63) >> 6;
        vector<vector<unsigned long long>> cadj(s, vector<unsigned long long>(words, 0));
        for (int i = 0; i < s; ++i) {
            for (int j = i + 1; j < s; ++j) {
                int a = comp[i], b = comp[j];
                if (!edgeSet[a].count(b)) {
                    cadj[i][j >> 6] |= 1ULL << (j & 63);
                    cadj[j][i >> 6] |= 1ULL << (i & 63);
                }
            }
        }

        MaxClique solver(cadj);
        vector<int> local = solver.solve();
        for (int idx : local) answer.push_back(comp[idx]);
    }

    cout << answer.size() << '\n';
    for (size_t i = 0; i < answer.size(); ++i) {
        if (i) cout << ' ';
        cout << answer[i];
    }
    cout << '\n';
    return 0;
}
