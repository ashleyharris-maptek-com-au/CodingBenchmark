use std::cmp::Ordering;
use std::collections::BinaryHeap;
use std::io::{self, Read};

#[derive(Clone, Copy, Eq, PartialEq)]
struct Candidate {
    gain: usize,
    degree: usize,
    v: usize,
}

impl Ord for Candidate {
    fn cmp(&self, other: &Self) -> Ordering {
        self.gain
            .cmp(&other.gain)
            .then_with(|| self.degree.cmp(&other.degree))
            .then_with(|| other.v.cmp(&self.v))
    }
}

impl PartialOrd for Candidate {
    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        Some(self.cmp(other))
    }
}

fn coverage_gain(v: usize, adj: &[Vec<usize>], covered: &[bool], chosen: &[bool]) -> usize {
    if chosen[v] {
        return 0;
    }

    let mut gain = usize::from(!covered[v]);
    for &u in &adj[v] {
        if !covered[u] {
            gain += 1;
        }
    }
    gain
}

fn mark_covered(v: usize, adj: &[Vec<usize>], covered: &mut [bool]) -> usize {
    let mut newly = 0usize;
    if !covered[v] {
        covered[v] = true;
        newly += 1;
    }
    for &u in &adj[v] {
        if !covered[u] {
            covered[u] = true;
            newly += 1;
        }
    }
    newly
}

fn add_domination(v: usize, adj: &[Vec<usize>], dominated_count: &mut [u16]) {
    dominated_count[v] = dominated_count[v].saturating_add(1);
    for &u in &adj[v] {
        dominated_count[u] = dominated_count[u].saturating_add(1);
    }
}

fn can_remove(v: usize, adj: &[Vec<usize>], dominated_count: &[u16]) -> bool {
    if dominated_count[v] <= 1 {
        return false;
    }
    adj[v].iter().all(|&u| dominated_count[u] > 1)
}

fn remove_domination(v: usize, adj: &[Vec<usize>], dominated_count: &mut [u16]) {
    dominated_count[v] -= 1;
    for &u in &adj[v] {
        dominated_count[u] -= 1;
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n = match it.next().and_then(|s| s.parse::<usize>().ok()) {
        Some(x) => x,
        None => return,
    };
    let m = it
        .next()
        .and_then(|s| s.parse::<usize>().ok())
        .unwrap_or(0);

    let mut adj = vec![Vec::<usize>::new(); n];
    for _ in 0..m {
        let Some(a) = it.next().and_then(|s| s.parse::<usize>().ok()) else {
            break;
        };
        let Some(b) = it.next().and_then(|s| s.parse::<usize>().ok()) else {
            break;
        };
        if a < n && b < n && a != b {
            adj[a].push(b);
            adj[b].push(a);
        }
    }

    for nbrs in &mut adj {
        nbrs.sort_unstable();
        nbrs.dedup();
    }

    let mut heap = BinaryHeap::new();
    for v in 0..n {
        heap.push(Candidate {
            gain: adj[v].len() + 1,
            degree: adj[v].len(),
            v,
        });
    }

    let mut covered = vec![false; n];
    let mut chosen = vec![false; n];
    let mut solution = Vec::new();
    let mut covered_total = 0usize;

    while covered_total < n {
        let Some(mut cand) = heap.pop() else {
            break;
        };
        if chosen[cand.v] {
            continue;
        }

        let real_gain = coverage_gain(cand.v, &adj, &covered, &chosen);
        let next_gain = heap.peek().map_or(0, |c| c.gain);
        if real_gain == 0 {
            continue;
        }
        if real_gain < next_gain {
            cand.gain = real_gain;
            heap.push(cand);
            continue;
        }

        chosen[cand.v] = true;
        solution.push(cand.v);
        covered_total += mark_covered(cand.v, &adj, &mut covered);
    }

    // Safety net for malformed edge lists or any unexpected heap exhaustion.
    for v in 0..n {
        if !covered[v] {
            chosen[v] = true;
            solution.push(v);
            mark_covered(v, &adj, &mut covered);
        }
    }

    let mut dominated_count = vec![0u16; n];
    for &v in &solution {
        add_domination(v, &adj, &mut dominated_count);
    }

    // Remove redundant vertices. Low-degree vertices are tried first because
    // they tend to cover less unique territory and are more often disposable.
    solution.sort_by_key(|&v| (adj[v].len(), v));
    let mut keep = chosen;
    for &v in &solution {
        if keep[v] && can_remove(v, &adj, &dominated_count) {
            keep[v] = false;
            remove_domination(v, &adj, &mut dominated_count);
        }
    }

    let mut answer: Vec<usize> = (0..n).filter(|&v| keep[v]).collect();
    answer.sort_unstable();

    println!("{}", answer.len());
    for (i, v) in answer.iter().enumerate() {
        if i > 0 {
            print!(" ");
        }
        print!("{}", v);
    }
    println!();
}
