declare const require: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type Town = { id: string; x: number; y: number; value: number };

const W: number = data.width | 0;
const H: number = data.height | 0;
const N = W * H;
const grid: string[] = data.grid || [];
const budget: number = Math.max(0, data.cutBudget | 0);
const towns: Town[] = data.towns || [];
const ignitions: number[][] = data.ignitions || [];

const forest = new Uint8Array(N);
for (let y = 0; y < H; y++) {
  const row = grid[y] || "";
  for (let x = 0; x < W; x++) {
    if (row.charCodeAt(x) === 46) forest[y * W + x] = 1;
  }
}

const isTown = new Uint8Array(N);
const townValueAt = new Float64Array(N);
for (const t of towns) {
  if (t.x >= 0 && t.x < W && t.y >= 0 && t.y < H) {
    const p = t.y * W + t.x;
    isTown[p] = 1;
    townValueAt[p] += Number(t.value) || 0;
  }
}

const isIgnition = new Uint8Array(N);
const ignitionList: number[] = [];
for (const q of ignitions) {
  const x = q[0] | 0, y = q[1] | 0;
  if (x >= 0 && x < W && y >= 0 && y < H) {
    const p = y * W + x;
    if (forest[p] && !isIgnition[p]) {
      isIgnition[p] = 1;
      ignitionList.push(p);
    }
  }
}

function neighbors(p: number, out: number[]): number {
  let n = 0;
  const x = p % W;
  if (x > 0) out[n++] = p - 1;
  if (x + 1 < W) out[n++] = p + 1;
  if (p >= W) out[n++] = p - W;
  if (p + W < N) out[n++] = p + W;
  return n;
}

function burn(cuts: Uint8Array, wantDist: boolean): { burned: Uint8Array; dist: Int32Array; value: number } {
  const burned = new Uint8Array(N);
  const dist = wantDist ? new Int32Array(N) : new Int32Array(0);
  if (wantDist) dist.fill(-1);
  const q = new Int32Array(N);
  let head = 0, tail = 0;
  for (const p of ignitionList) {
    if (forest[p] && !cuts[p] && !burned[p]) {
      burned[p] = 1;
      if (wantDist) dist[p] = 0;
      q[tail++] = p;
    }
  }
  const nb = [0, 0, 0, 0];
  while (head < tail) {
    const p = q[head++];
    const d = wantDist ? dist[p] + 1 : 0;
    const c = neighbors(p, nb);
    for (let i = 0; i < c; i++) {
      const r = nb[i];
      if (forest[r] && !cuts[r] && !burned[r]) {
        burned[r] = 1;
        if (wantDist) dist[r] = d;
        q[tail++] = r;
      }
    }
  }
  let value = 0;
  for (const t of towns) {
    const p = t.y * W + t.x;
    if (p >= 0 && p < N && burned[p]) value += Number(t.value) || 0;
  }
  return { burned, dist, value };
}

function addCandidate(p: number, seen: Uint8Array, arr: number[]) {
  if (p >= 0 && p < N && forest[p] && !isIgnition[p] && !isTown[p] && !seen[p]) {
    seen[p] = 1;
    arr.push(p);
  }
}

const emptyCuts = new Uint8Array(N);
const base = burn(emptyCuts, true);
const baseValue = base.value;
if (budget <= 0 || baseValue <= 0 || ignitionList.length === 0) {
  process.stdout.write(JSON.stringify({ firebreaks: [] }));
} else {
  const selected = new Uint8Array(N);
  const selectedList: number[] = [];
  const candSeen = new Uint8Array(N);
  const candidates: number[] = [];
  const nb = [0, 0, 0, 0];

  for (const t of towns) {
    const start = t.y * W + t.x;
    if (start < 0 || start >= N || !base.burned[start]) continue;

    let p = start;
    for (let step = 0; step < 20000 && base.dist[p] > 0; step++) {
      addCandidate(p, candSeen, candidates);
      let best = p, bestD = base.dist[p];
      const c = neighbors(p, nb);
      for (let i = 0; i < c; i++) {
        const r = nb[i];
        if (base.dist[r] >= 0 && base.dist[r] < bestD) {
          best = r;
          bestD = base.dist[r];
        }
      }
      if (best === p) break;
      p = best;
    }

    const radius = 3;
    for (let dy = -radius; dy <= radius; dy++) {
      for (let dx = -radius; dx <= radius; dx++) {
        if (Math.abs(dx) + Math.abs(dy) <= radius) {
          const x = t.x + dx, y = t.y + dy;
          if (x >= 0 && x < W && y >= 0 && y < H) addCandidate(y * W + x, candSeen, candidates);
        }
      }
    }
  }

  for (const p of ignitionList) {
    const c = neighbors(p, nb);
    for (let i = 0; i < c; i++) addCandidate(nb[i], candSeen, candidates);
  }

  candidates.sort((a, b) => {
    const va = townValueAt[a] * 1000 - base.dist[a];
    const vb = townValueAt[b] * 1000 - base.dist[b];
    return vb - va;
  });
  if (candidates.length > 900) candidates.length = 900;

  type Plan = { cells: number[]; estimate: number };
  const plans: Plan[] = [];

  function makeRingPlan(t: Town) {
    const start = t.y * W + t.x;
    if (start < 0 || start >= N || !forest[start] || !base.burned[start] || isIgnition[start]) return;
    const inRegion = new Uint8Array(N);
    const q = new Int32Array(Math.min(N, 12000));
    let head = 0, tail = 0;
    inRegion[start] = 1;
    q[tail++] = start;
    let regionValue = townValueAt[start];
    let bestCells: number[] | null = null;
    let bestScore = 0;
    const maxRegion = Math.min(N, Math.max(250, budget * 80 + 1000));
    for (let processed = 0; head < tail && processed < maxRegion; processed++) {
      const p = q[head++];
      const c = neighbors(p, nb);
      for (let i = 0; i < c; i++) {
        const r = nb[i];
        if (forest[r] && base.burned[r] && !inRegion[r] && !isIgnition[r]) {
          inRegion[r] = 1;
          if (tail < q.length) q[tail++] = r;
          regionValue += townValueAt[r];
        }
      }
      if (processed % 16 !== 0) continue;
      const boundarySeen = new Uint8Array(N);
      const boundary: number[] = [];
      for (let j = 0; j < tail; j++) {
        const rp = q[j];
        const cc = neighbors(rp, nb);
        for (let k = 0; k < cc; k++) {
          const r = nb[k];
          if (forest[r] && base.burned[r] && !inRegion[r] && !isIgnition[r] && !isTown[r] && !boundarySeen[r]) {
            boundarySeen[r] = 1;
            boundary.push(r);
            if (boundary.length > budget) break;
          }
        }
        if (boundary.length > budget) break;
      }
      if (boundary.length > 0 && boundary.length <= budget) {
        const score = regionValue / Math.pow(boundary.length, 0.85);
        if (score > bestScore) {
          bestScore = score;
          bestCells = boundary.slice();
        }
      }
    }
    if (bestCells) plans.push({ cells: bestCells, estimate: bestScore });
  }

  const burnedTowns = towns
    .filter(t => {
      const p = t.y * W + t.x;
      return p >= 0 && p < N && base.burned[p];
    })
    .sort((a, b) => (Number(b.value) || 0) - (Number(a.value) || 0))
    .slice(0, 80);
  for (const t of burnedTowns) makeRingPlan(t);
  plans.sort((a, b) => b.estimate - a.estimate);
  if (plans.length > 120) plans.length = 120;

  let currentValue = baseValue;
  const startTime = Date.now();

  function tryCells(cells: number[]): number {
    let added = 0;
    for (const p of cells) if (!selected[p]) added++;
    if (added === 0 || selectedList.length + added > budget) return -1;
    for (const p of cells) selected[p] = 1;
    const v = burn(selected, false).value;
    for (const p of cells) {
      let keep = false;
      for (let i = 0; i < selectedList.length; i++) if (selectedList[i] === p) { keep = true; break; }
      if (!keep) selected[p] = 0;
    }
    return currentValue - v;
  }

  while (selectedList.length < budget && Date.now() - startTime < 8500) {
    let bestCells: number[] | null = null;
    let bestGain = 0;
    let bestRatio = 0;

    for (let i = 0; i < plans.length && Date.now() - startTime < 8500; i++) {
      const cells = plans[i].cells.filter(p => !selected[p]);
      const gain = tryCells(cells);
      if (gain > 0) {
        const ratio = gain / Math.pow(cells.length, 0.9);
        if (ratio > bestRatio) {
          bestRatio = ratio;
          bestGain = gain;
          bestCells = cells;
        }
      }
    }

    const singleLimit = selectedList.length < 12 ? candidates.length : Math.min(candidates.length, 240);
    for (let i = 0; i < singleLimit && Date.now() - startTime < 8500; i++) {
      const p = candidates[i];
      if (selected[p]) continue;
      selected[p] = 1;
      const v = burn(selected, false).value;
      selected[p] = 0;
      const gain = currentValue - v;
      if (gain > bestGain) {
        bestGain = gain;
        bestRatio = gain;
        bestCells = [p];
      }
    }

    if (!bestCells || bestGain <= 0) break;
    for (const p of bestCells) {
      if (!selected[p] && selectedList.length < budget) {
        selected[p] = 1;
        selectedList.push(p);
      }
    }
    currentValue = burn(selected, false).value;
    if (currentValue <= 0) break;
  }

  const firebreaks = selectedList.map(p => [p % W, Math.floor(p / W)]);
  process.stdout.write(JSON.stringify({ firebreaks }));
}
