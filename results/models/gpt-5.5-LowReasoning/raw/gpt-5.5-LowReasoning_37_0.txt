using System;
using System.Collections.Generic;
using System.Text;

public static class Program
{
    static int n, m;
    static List<int>[] g = Array.Empty<List<int>>();
    static List<int>[] rg = Array.Empty<List<int>>();
    static int[] best = Array.Empty<int>();

    public static void Main()
    {
        var fs = new FastScanner();
        n = fs.NextInt();
        m = fs.NextInt();
        g = new List<int>[n];
        rg = new List<int>[n];
        for (int i = 0; i < n; i++) { g[i] = new List<int>(); rg[i] = new List<int>(); }
        for (int i = 0; i < m; i++)
        {
            int u = fs.NextInt(), v = fs.NextInt();
            if ((uint)u < (uint)n && (uint)v < (uint)n)
            {
                g[u].Add(v);
                rg[v].Add(u);
            }
        }

        var all = new List<int>(n);
        for (int i = 0; i < n; i++) all.Add(i);

        var ans = SolveSet(all, new bool[n]);
        ans.Sort();
        Console.WriteLine(ans.Count);
        Console.WriteLine(string.Join(" ", ans));
    }

    static List<int> SolveSet(List<int> vertices, bool[] removedBase)
    {
        var active = new bool[n];
        foreach (int v in vertices) if (!removedBase[v]) active[v] = true;

        Reduce(active);
        var comps = SCCs(active);
        var result = new List<int>();

        foreach (var comp in comps)
        {
            if (comp.Count == 1)
            {
                int v = comp[0];
                bool self = false;
                foreach (int to in g[v]) if (to == v) { self = true; break; }
                if (self) result.Add(v);
                continue;
            }

            best = comp.ToArray();
            var rem = new bool[n];
            Search(comp, rem, 0);
            result.AddRange(best);
        }
        return result;
    }

    static void Search(List<int> comp, bool[] rem, int chosen)
    {
        if (chosen >= best.Length) return;

        var cycle = FindCycle(comp, rem);
        if (cycle.Count == 0)
        {
            var cur = new int[chosen];
            int k = 0;
            foreach (int v in comp) if (rem[v]) cur[k++] = v;
            best = cur;
            return;
        }

        // Any directed feedback vertex set must hit every directed cycle.
        // Branch on deleting one vertex from a concrete cycle.
        cycle.Sort((a, b) =>
        {
            int da = Score(a, rem), db = Score(b, rem);
            return db.CompareTo(da);
        });

        foreach (int v in cycle)
        {
            if (rem[v]) continue;
            rem[v] = true;
            Search(comp, rem, chosen + 1);
            rem[v] = false;
        }
    }

    static int Score(int v, bool[] rem)
    {
        int s = 0;
        foreach (int to in g[v]) if (!rem[to]) s++;
        foreach (int from in rg[v]) if (!rem[from]) s++;
        return s;
    }

    static List<int> FindCycle(List<int> comp, bool[] rem)
    {
        var inComp = new bool[n];
        foreach (int v in comp) if (!rem[v]) inComp[v] = true;

        var color = new byte[n];
        var parent = new int[n];
        Array.Fill(parent, -1);

        foreach (int start in comp)
        {
            if (!inComp[start] || color[start] != 0) continue;
            var stack = new Stack<(int v, int idx)>();
            stack.Push((start, 0));
            color[start] = 1;

            while (stack.Count > 0)
            {
                var top = stack.Pop();
                int v = top.v, idx = top.idx;
                if (idx >= g[v].Count)
                {
                    color[v] = 2;
                    continue;
                }

                stack.Push((v, idx + 1));
                int to = g[v][idx];
                if (!inComp[to]) continue;

                if (color[to] == 0)
                {
                    parent[to] = v;
                    color[to] = 1;
                    stack.Push((to, 0));
                }
                else if (color[to] == 1)
                {
                    var cyc = new List<int> { to };
                    for (int x = v; x != to && x != -1; x = parent[x]) cyc.Add(x);
                    return cyc;
                }
            }
        }
        return new List<int>();
    }

    static void Reduce(bool[] active)
    {
        var indeg = new int[n];
        var outdeg = new int[n];
        var q = new Queue<int>();
        for (int v = 0; v < n; v++)
        {
            if (!active[v]) continue;
            foreach (int to in g[v]) if (active[to]) outdeg[v]++;
            foreach (int from in rg[v]) if (active[from]) indeg[v]++;
        }
        for (int v = 0; v < n; v++)
            if (active[v] && (indeg[v] == 0 || outdeg[v] == 0)) q.Enqueue(v);

        while (q.Count > 0)
        {
            int v = q.Dequeue();
            if (!active[v] || (indeg[v] != 0 && outdeg[v] != 0)) continue;
            active[v] = false;
            foreach (int to in g[v])
                if (active[to] && --indeg[to] == 0) q.Enqueue(to);
            foreach (int from in rg[v])
                if (active[from] && --outdeg[from] == 0) q.Enqueue(from);
        }
    }

    static List<List<int>> SCCs(bool[] active)
    {
        var order = new List<int>();
        var seen = new bool[n];
        for (int s = 0; s < n; s++)
        {
            if (!active[s] || seen[s]) continue;
            var st = new Stack<(int v, int idx)>();
            st.Push((s, 0));
            seen[s] = true;
            while (st.Count > 0)
            {
                var top = st.Pop();
                int v = top.v, idx = top.idx;
                if (idx >= g[v].Count)
                {
                    order.Add(v);
                    continue;
                }
                st.Push((v, idx + 1));
                int to = g[v][idx];
                if (active[to] && !seen[to])
                {
                    seen[to] = true;
                    st.Push((to, 0));
                }
            }
        }

        var comps = new List<List<int>>();
        Array.Fill(seen, false);
        for (int i = order.Count - 1; i >= 0; i--)
        {
            int s = order[i];
            if (seen[s]) continue;
            var comp = new List<int>();
            var st = new Stack<int>();
            st.Push(s);
            seen[s] = true;
            while (st.Count > 0)
            {
                int v = st.Pop();
                comp.Add(v);
                foreach (int to in rg[v])
                    if (active[to] && !seen[to])
                    {
                        seen[to] = true;
                        st.Push(to);
                    }
            }
            comps.Add(comp);
        }
        return comps;
    }

    sealed class FastScanner
    {
        readonly byte[] data;
        int p;
        public FastScanner()
        {
            using var input = Console.OpenStandardInput();
            data = new byte[1 << 20];
            var chunks = new List<byte[]>();
            int total = 0, r;
            while ((r = input.Read(data, 0, data.Length)) > 0)
            {
                var c = new byte[r];
                Buffer.BlockCopy(data, 0, c, 0, r);
                chunks.Add(c);
                total += r;
            }
            data = new byte[total + 1];
            int off = 0;
            foreach (var c in chunks)
            {
                Buffer.BlockCopy(c, 0, data, off, c.Length);
                off += c.Length;
            }
        }
        public int NextInt()
        {
            while (data[p] <= 32) p++;
            int sign = 1;
            if (data[p] == '-') { sign = -1; p++; }
            int x = 0;
            while (data[p] > 32) x = x * 10 + data[p++] - '0';
            return x * sign;
        }
    }
}
