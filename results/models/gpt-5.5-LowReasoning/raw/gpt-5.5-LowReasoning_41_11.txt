; SPIR-V
; Version: 1.0
OpCapability Shader
%1 = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %in_worldPos %in_normal %in_uv %outColor
OpExecutionMode %main OriginUpperLeft
OpSource GLSL 450
OpName %main "main"
OpName %in_worldPos "worldPos"
OpName %in_normal "normal"
OpName %in_uv "uv"
OpName %outColor "outColor"
OpName %UBO "UBO"
OpMemberName %UBO 0 "model"
OpMemberName %UBO 1 "view"
OpMemberName %UBO 2 "projection"
OpMemberName %UBO 3 "lightPos"
OpMemberName %UBO 4 "cameraPos"
OpMemberName %UBO 5 "params"
OpName %ubo "ubo"
OpDecorate %in_worldPos Location 0
OpDecorate %in_normal Location 1
OpDecorate %in_uv Location 2
OpDecorate %outColor Location 0
OpDecorate %UBO Block
OpMemberDecorate %UBO 0 ColMajor
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 0 MatrixStride 16
OpMemberDecorate %UBO 1 ColMajor
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 1 MatrixStride 16
OpMemberDecorate %UBO 2 ColMajor
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 2 MatrixStride 16
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0
%void = OpTypeVoid
%func = OpTypeFunction %void
%float = OpTypeFloat 32
%v2float = OpTypeVector %float 2
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%mat4 = OpTypeMatrix %v4float 4
%UBO = OpTypeStruct %mat4 %mat4 %mat4 %v4float %v4float %v4float
%ptr_Input_v3float = OpTypePointer Input %v3float
%ptr_Input_v2float = OpTypePointer Input %v2float
%ptr_Output_v4float = OpTypePointer Output %v4float
%ptr_Uniform_UBO = OpTypePointer Uniform %UBO
%ptr_Uniform_v4float = OpTypePointer Uniform %v4float
%bool = OpTypeBool
%int = OpTypeInt 32 1
%int_3 = OpConstant %int 3
%float_0 = OpConstant %float 0
%float_0_1 = OpConstant %float 0.1
%float_0_3 = OpConstant %float 0.3
%float_0_5 = OpConstant %float 0.5
%float_0_7 = OpConstant %float 0.7
%float_0_8 = OpConstant %float 0.8
%float_1 = OpConstant %float 1
%float_6 = OpConstant %float 6
%v2float_6 = OpConstantComposite %v2float %float_6 %float_6
%v2float_0_5 = OpConstantComposite %v2float %float_0_5 %float_0_5
%yellow = OpConstantComposite %v3float %float_1 %float_0_8 %float_0
%darkBlue = OpConstantComposite %v3float %float_0_1 %float_0_1 %float_0_3
%in_worldPos = OpVariable %ptr_Input_v3float Input
%in_normal = OpVariable %ptr_Input_v3float Input
%in_uv = OpVariable %ptr_Input_v2float Input
%outColor = OpVariable %ptr_Output_v4float Output
%ubo = OpVariable %ptr_Uniform_UBO Uniform
%main = OpFunction %void None %func
%entry = OpLabel
%uv = OpLoad %v2float %in_uv
%uvScaled = OpFMul %v2float %uv %v2float_6
%uvFloor = OpExtInst %v2float %1 Floor %uvScaled
%cell = OpFSub %v2float %uvScaled %uvFloor
%centered = OpFSub %v2float %cell %v2float_0_5
%dist = OpExtInst %float %1 Length %centered
%inside = OpFOrdLessThanEqual %bool %dist %float_0_3
%baseColor = OpSelect %v3float %inside %yellow %darkBlue
%normalValue = OpLoad %v3float %in_normal
%n = OpExtInst %v3float %1 Normalize %normalValue
%world = OpLoad %v3float %in_worldPos
%lightPtr = OpAccessChain %ptr_Uniform_v4float %ubo %int_3
%light4 = OpLoad %v4float %lightPtr
%light3 = OpVectorShuffle %v3float %light4 %light4 0 1 2
%toLight = OpFSub %v3float %light3 %world
%lightDir = OpExtInst %v3float %1 Normalize %toLight
%ndotl = OpDot %float %n %lightDir
%diffuse = OpExtInst %float %1 FMax %ndotl %float_0
%diffuseScaled = OpFMul %float %diffuse %float_0_7
%factor = OpFAdd %float %float_0_3 %diffuseScaled
%shaded = OpVectorTimesScalar %v3float %baseColor %factor
%out = OpCompositeConstruct %v4float %shaded %float_1
OpStore %outColor %out
OpReturn
OpFunctionEnd
