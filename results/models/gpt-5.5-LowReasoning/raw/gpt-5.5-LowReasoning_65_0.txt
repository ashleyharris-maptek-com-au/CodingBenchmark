declare const require: any;
declare const process: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type TaskInput = {
  id: any;
  duration: number;
  deps?: any[];
  deadline?: number;
  weight?: number;
};

type HeapItem = { key: number[]; value: number };

class MinHeap {
  private a: HeapItem[] = [];

  get size(): number {
    return this.a.length;
  }

  push(item: HeapItem): void {
    const a = this.a;
    a.push(item);
    let i = a.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (cmpKey(a[p].key, item.key) <= 0) break;
      a[i] = a[p];
      i = p;
    }
    a[i] = item;
  }

  pop(): HeapItem | undefined {
    const a = this.a;
    if (a.length === 0) return undefined;
    const root = a[0];
    const x = a.pop()!;
    if (a.length > 0) {
      let i = 0;
      while (true) {
        let c = i * 2 + 1;
        if (c >= a.length) break;
        const r = c + 1;
        if (r < a.length && cmpKey(a[r].key, a[c].key) < 0) c = r;
        if (cmpKey(a[c].key, x.key) >= 0) break;
        a[i] = a[c];
        i = c;
      }
      a[i] = x;
    }
    return root;
  }
}

function cmpKey(a: number[], b: number[]): number {
  const n = Math.min(a.length, b.length);
  for (let i = 0; i < n; i++) {
    const d = a[i] - b[i];
    if (d !== 0) return d;
  }
  return a.length - b.length;
}

const tasks: TaskInput[] = Array.isArray(data.tasks) ? data.tasks : [];
const n = tasks.length;
const machineCount = Math.max(1, Math.floor(Number(data.machineCount) || 1));

const idToIndex = new Map<string, number>();
for (let i = 0; i < n; i++) idToIndex.set(JSON.stringify(tasks[i].id), i);

const duration = new Array<number>(n);
const deadline = new Array<number>(n);
const weight = new Array<number>(n);
const indeg0 = new Array<number>(n).fill(0);
const children: number[][] = Array.from({ length: n }, () => []);
const parents: number[][] = Array.from({ length: n }, () => []);

let totalDuration = 0;
for (let i = 0; i < n; i++) {
  const d = Math.max(0, Math.floor(Number(tasks[i].duration) || 0));
  duration[i] = d;
  totalDuration += d;
  const dl = Number(tasks[i].deadline);
  deadline[i] = Number.isFinite(dl) ? dl : totalDuration + 1000000000;
  const w = Number(tasks[i].weight);
  weight[i] = Number.isFinite(w) && w > 0 ? w : 1;
}

for (let i = 0; i < n; i++) {
  const deps = Array.isArray(tasks[i].deps) ? tasks[i].deps! : [];
  for (const depId of deps) {
    const p = idToIndex.get(JSON.stringify(depId));
    if (p === undefined || p === i) continue;
    parents[i].push(p);
    children[p].push(i);
    indeg0[i]++;
  }
}

const topo: number[] = [];
const q: number[] = [];
const indegTopo = indeg0.slice();
for (let i = 0; i < n; i++) if (indegTopo[i] === 0) q.push(i);
for (let head = 0; head < q.length; head++) {
  const u = q[head];
  topo.push(u);
  for (const v of children[u]) {
    indegTopo[v]--;
    if (indegTopo[v] === 0) q.push(v);
  }
}
if (topo.length < n) {
  const seen = new Array<boolean>(n).fill(false);
  for (const x of topo) seen[x] = true;
  for (let i = 0; i < n; i++) if (!seen[i]) topo.push(i);
}

const downLen = duration.slice();
const downWeight = weight.slice();
const minSuccDeadline = deadline.slice();
for (let ti = topo.length - 1; ti >= 0; ti--) {
  const u = topo[ti];
  for (const v of children[u]) {
    const candLen = duration[u] + downLen[v];
    if (candLen > downLen[u]) downLen[u] = candLen;
    downWeight[u] += downWeight[v] * 0.35;
    if (minSuccDeadline[v] < minSuccDeadline[u]) minSuccDeadline[u] = minSuccDeadline[v];
  }
}

const avgLoad = Math.ceil(totalDuration / machineCount);
const indeg = indeg0.slice();
const readyTime = new Array<number>(n).fill(0);
const done = new Array<boolean>(n).fill(false);
const machineHeap = new MinHeap();
const readyHeap = new MinHeap();

for (let m = 0; m < machineCount; m++) machineHeap.push({ key: [0, m], value: m });

function readyKey(i: number, now: number): number[] {
  const estFinish = now + duration[i];
  const effectiveDeadline = Math.min(deadline[i], minSuccDeadline[i]);
  const slack = effectiveDeadline - estFinish - Math.max(0, downLen[i] - duration[i]);
  const density = weight[i] / Math.max(1, duration[i]);
  return [
    slack,
    deadline[i] - estFinish,
    -density,
    -downWeight[i],
    -downLen[i],
    -duration[i],
    i
  ];
}

function enqueueReady(i: number, now: number): void {
  readyHeap.push({ key: readyKey(i, now), value: i });
}

for (let i = 0; i < n; i++) if (indeg[i] === 0) enqueueReady(i, 0);

const machineAvail = new Array<number>(machineCount).fill(0);
const schedule: { id: any; start: number; machine: number }[] = [];
let scheduled = 0;

while (scheduled < n && readyHeap.size > 0) {
  const mh = machineHeap.pop()!;
  const m = mh.value;
  let now = machineAvail[m];
  let chosen: number | undefined;
  const deferred: HeapItem[] = [];

  while (readyHeap.size > 0) {
    const item = readyHeap.pop()!;
    const i = item.value;
    if (done[i] || indeg[i] !== 0) continue;
    if (readyTime[i] <= now) {
      chosen = i;
      break;
    }
    deferred.push(item);
  }

  if (chosen === undefined) {
    for (const item of deferred) readyHeap.push(item);
    if (readyHeap.size === 0) break;
    const nextItem = readyHeap.pop()!;
    chosen = nextItem.value;
    now = Math.max(now, readyTime[chosen]);
  } else {
    for (const item of deferred) readyHeap.push(item);
  }

  const i: number = chosen as number;
  const start = Math.max(now, readyTime[i]);
  schedule.push({ id: tasks[i].id, start, machine: m });
  done[i] = true;
  scheduled++;
  const finish = start + duration[i];
  machineAvail[m] = finish;
  machineHeap.push({ key: [finish, m], value: m });

  for (const v of children[i]) {
    if (finish > readyTime[v]) readyTime[v] = finish;
    indeg[v]--;
    if (indeg[v] === 0) enqueueReady(v, Math.min(finish, avgLoad));
  }
}

if (scheduled < n) {
  for (const i of topo) {
    if (done[i]) continue;
    let bestM = 0;
    for (let m = 1; m < machineCount; m++) {
      if (machineAvail[m] < machineAvail[bestM]) bestM = m;
    }
    let start = machineAvail[bestM];
    for (const p of parents[i]) {
      const rec = schedule.find(x => JSON.stringify(x.id) === JSON.stringify(tasks[p].id));
      if (rec) start = Math.max(start, rec.start + duration[p]);
    }
    schedule.push({ id: tasks[i].id, start, machine: bestM });
    machineAvail[bestM] = start + duration[i];
  }
}

process.stdout.write(JSON.stringify({ schedule }));
