declare const require: any;
declare const process: any;
const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type Drone = { id?: string; x: number; y: number };
type Point = [number, number];

const grid: number[][] = Array.isArray(data.probability) ? data.probability : [];
const drones: Drone[] = Array.isArray(data.drones) ? data.drones : [];
const maxSteps: number = Math.max(0, Math.floor(Number(data.maxStepsPerDrone) || 0));

const height = grid.length;
const width = height > 0 && Array.isArray(grid[0]) ? grid[0].length : 0;
const totalCells = width * height;

function inside(x: number, y: number): boolean {
  return y >= 0 && y < height && x >= 0 && x < width;
}

function idx(x: number, y: number): number {
  return y * width + x;
}

function passable(x: number, y: number): boolean {
  return inside(x, y) && grid[y][x] !== -1;
}

const covered = new Uint8Array(Math.max(0, totalCells));
const positives: { i: number; x: number; y: number; v: number }[] = [];
let maxValue = 0;

for (let y = 0; y < height; y++) {
  const row = grid[y] || [];
  for (let x = 0; x < width; x++) {
    const v = Number(row[x]) || 0;
    if (v > 0) {
      positives.push({ i: idx(x, y), x, y, v });
      if (v > maxValue) maxValue = v;
    }
  }
}

positives.sort((a, b) => b.v - a.v);
const candidateLimit = totalCells <= 40000 ? positives.length : Math.min(positives.length, 7000);
const candidates = positives.slice(0, candidateLimit);

const dirs = [
  [1, 0],
  [-1, 0],
  [0, 1],
  [0, -1],
] as const;
const moves = [
  [0, 0],
  [1, 0],
  [-1, 0],
  [0, 1],
  [0, -1],
] as const;

const seen = new Int32Array(Math.max(0, totalCells));
const dist = new Int32Array(Math.max(0, totalCells));
const parent = new Int32Array(Math.max(0, totalCells));
let stamp = 1;

function bfsFrom(sx: number, sy: number, maxDist: number): number[] {
  const q: number[] = [];
  if (!passable(sx, sy) || totalCells === 0) return q;
  if (stamp === 2147483647) {
    seen.fill(0);
    stamp = 1;
  }
  const s = idx(sx, sy);
  seen[s] = stamp;
  dist[s] = 0;
  parent[s] = -1;
  q.push(s);
  let head = 0;
  while (head < q.length) {
    const cur = q[head++];
    const d = dist[cur];
    if (d >= maxDist) continue;
    const x = cur % width;
    const y = (cur / width) | 0;
    for (const [dx, dy] of dirs) {
      const nx = x + dx;
      const ny = y + dy;
      if (!passable(nx, ny)) continue;
      const ni = idx(nx, ny);
      if (seen[ni] === stamp) continue;
      seen[ni] = stamp;
      dist[ni] = d + 1;
      parent[ni] = cur;
      q.push(ni);
    }
  }
  return q;
}

function localPotential(x: number, y: number, coveredNow: Uint8Array): number {
  let s = 0;
  for (let dy = -3; dy <= 3; dy++) {
    for (let dx = -3; dx <= 3; dx++) {
      const md = Math.abs(dx) + Math.abs(dy);
      if (md > 3) continue;
      const nx = x + dx;
      const ny = y + dy;
      if (!passable(nx, ny)) continue;
      const ni = idx(nx, ny);
      if (coveredNow[ni]) continue;
      const v = Number(grid[ny][nx]) || 0;
      if (v > 0) s += v / (md + 1);
    }
  }
  return s;
}

function reconstructPath(target: number): Point[] {
  const rev: Point[] = [];
  let cur = target;
  while (cur >= 0) {
    rev.push([cur % width, (cur / width) | 0]);
    cur = parent[cur];
  }
  rev.reverse();
  return rev;
}

function bestReachableTarget(sx: number, sy: number, remaining: number): Point[] | null {
  stamp++;
  bfsFrom(sx, sy, remaining);
  let best = -1;
  let bestScore = 0;
  let checked = 0;
  const scan = candidates.length > 0 ? candidates : positives;
  for (const c of scan) {
    if (covered[c.i] || seen[c.i] !== stamp) continue;
    const d = dist[c.i];
    if (d < 0 || d > remaining) continue;
    const score = c.v / Math.pow(d + 1, 0.78) + localPotential(c.x, c.y, covered) * 0.20;
    if (score > bestScore) {
      bestScore = score;
      best = c.i;
    }
    checked++;
    if (checked > 2500 && bestScore > maxValue * 0.40) break;
  }
  if (best < 0) return null;
  return reconstructPath(best);
}

function chooseLocalStep(x: number, y: number): Point {
  let bx = x;
  let by = y;
  let bs = -1;
  for (const [dx, dy] of moves) {
    const nx = x + dx;
    const ny = y + dy;
    if (!passable(nx, ny)) continue;
    const ni = idx(nx, ny);
    const immediate = covered[ni] ? 0 : Math.max(0, Number(grid[ny][nx]) || 0);
    const score = immediate * 2.5 + localPotential(nx, ny, covered) - (dx === 0 && dy === 0 ? 0.01 : 0);
    if (score > bs) {
      bs = score;
      bx = nx;
      by = ny;
    }
  }
  return [bx, by];
}

const paths: Point[][] = [];

for (const drone of drones) {
  let x = Math.floor(Number(drone.x));
  let y = Math.floor(Number(drone.y));
  if (!passable(x, y)) {
    let found = false;
    for (let yy = 0; yy < height && !found; yy++) {
      for (let xx = 0; xx < width; xx++) {
        if (passable(xx, yy)) {
          x = xx;
          y = yy;
          found = true;
          break;
        }
      }
    }
  }

  const path: Point[] = passable(x, y) ? [[x, y]] : [];
  if (path.length) covered[idx(x, y)] = 1;

  while (path.length > 0 && path.length <= maxSteps) {
    const remaining = maxSteps + 1 - path.length;
    if (remaining <= 0) break;

    let segment: Point[] | null = null;
    if (remaining >= 2) segment = bestReachableTarget(x, y, remaining);

    if (segment && segment.length > 1) {
      for (let k = 1; k < segment.length && path.length <= maxSteps; k++) {
        const [nx, ny] = segment[k];
        path.push([nx, ny]);
        covered[idx(nx, ny)] = 1;
        x = nx;
        y = ny;
      }
    } else {
      const [nx, ny] = chooseLocalStep(x, y);
      path.push([nx, ny]);
      covered[idx(nx, ny)] = 1;
      x = nx;
      y = ny;
    }
  }

  while (path.length > maxSteps + 1) path.pop();
  paths.push(path);
}

process.stdout.write(JSON.stringify({ paths }));
