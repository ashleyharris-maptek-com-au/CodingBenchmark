const fs = require('fs');
const input = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

const tasks = Array.isArray(data.tasks) ? data.tasks : [];
const n = tasks.length;
const machineCount = Math.max(1, data.machineCount | 0);

class MinHeap {
  constructor(cmp) {
    this.a = [];
    this.cmp = cmp;
  }
  get size() {
    return this.a.length;
  }
  push(x) {
    const a = this.a;
    a.push(x);
    let i = a.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (this.cmp(a[p], x) <= 0) break;
      a[i] = a[p];
      i = p;
    }
    a[i] = x;
  }
  pop() {
    const a = this.a;
    if (a.length === 0) return undefined;
    const top = a[0];
    const x = a.pop();
    if (a.length > 0) {
      let i = 0;
      while (true) {
        let l = i * 2 + 1;
        if (l >= a.length) break;
        let r = l + 1;
        let c = r < a.length && this.cmp(a[r], a[l]) < 0 ? r : l;
        if (this.cmp(a[c], x) >= 0) break;
        a[i] = a[c];
        i = c;
      }
      a[i] = x;
    }
    return top;
  }
}

if (n === 0) {
  console.log(JSON.stringify({ schedule: [] }));
  process.exit(0);
}

const idToIndex = new Map();
for (let i = 0; i < n; i++) idToIndex.set(tasks[i].id, i);

const children = Array.from({ length: n }, () => []);
const indeg0 = new Int32Array(n);
const duration = new Float64Array(n);
const deadline = new Float64Array(n);
const weight = new Float64Array(n);
let totalDuration = 0;

for (let i = 0; i < n; i++) {
  const t = tasks[i];
  duration[i] = Math.max(0, Number(t.duration) || 0);
  deadline[i] = Number.isFinite(Number(t.deadline)) ? Number(t.deadline) : 1e15;
  weight[i] = Math.max(0.000001, Number(t.weight) || 1);
  totalDuration += duration[i];
}

for (let i = 0; i < n; i++) {
  const deps = Array.isArray(tasks[i].deps) ? tasks[i].deps : [];
  for (const depId of deps) {
    const j = idToIndex.get(depId);
    if (j === undefined) continue;
    children[j].push(i);
    indeg0[i]++;
  }
}

const indegTopo = new Int32Array(indeg0);
const queue = [];
let qh = 0;
for (let i = 0; i < n; i++) if (indegTopo[i] === 0) queue.push(i);
const topo = [];
while (qh < queue.length) {
  const u = queue[qh++];
  topo.push(u);
  for (const v of children[u]) {
    if (--indegTopo[v] === 0) queue.push(v);
  }
}

if (topo.length !== n) {
  const rest = [];
  const seen = new Uint8Array(n);
  for (const i of topo) seen[i] = 1;
  for (let i = 0; i < n; i++) if (!seen[i]) rest.push(tasks[i].id);
  console.log(JSON.stringify({ order: topo.map(i => tasks[i].id).concat(rest) }));
  process.exit(0);
}

const bottom = new Float64Array(n);
const downWeight = new Float64Array(n);
for (let ti = n - 1; ti >= 0; ti--) {
  const u = topo[ti];
  let bestChild = 0;
  let childW = 0;
  for (const v of children[u]) {
    if (bottom[v] > bestChild) bestChild = bottom[v];
    childW += downWeight[v] * 0.35;
  }
  bottom[u] = duration[u] + bestChild;
  downWeight[u] = weight[u] + childW;
}

const remainingDeps = new Int32Array(indeg0);
const readyTime = new Float64Array(n);
const done = new Uint8Array(n);
const ready = [];
for (let i = 0; i < n; i++) {
  if (remainingDeps[i] === 0) ready.push(i);
}

const machines = new MinHeap((a, b) => (a.time - b.time) || (a.machine - b.machine));
for (let m = 0; m < machineCount; m++) machines.push({ time: 0, machine: m });

const schedule = [];
const horizon = Math.max(1, totalDuration / Math.max(1, machineCount));
let scheduled = 0;

function betterTask(a, b, now) {
  const ca = now + duration[a];
  const cb = now + duration[b];
  const la = Math.max(0, ca - deadline[a]);
  const lb = Math.max(0, cb - deadline[b]);
  const slackA = deadline[a] - now - bottom[a];
  const slackB = deadline[b] - now - bottom[b];
  const scoreA =
    weight[a] * (la + duration[a] * 0.18) +
    downWeight[a] * Math.max(0, -slackA) * 0.09 +
    bottom[a] * 0.015 +
    weight[a] / Math.sqrt(duration[a] + 1);
  const scoreB =
    weight[b] * (lb + duration[b] * 0.18) +
    downWeight[b] * Math.max(0, -slackB) * 0.09 +
    bottom[b] * 0.015 +
    weight[b] / Math.sqrt(duration[b] + 1);
  if (scoreA !== scoreB) return scoreA > scoreB;
  if (slackA !== slackB) return slackA < slackB;
  if (deadline[a] !== deadline[b]) return deadline[a] < deadline[b];
  if (bottom[a] !== bottom[b]) return bottom[a] > bottom[b];
  return String(tasks[a].id) < String(tasks[b].id);
}

function pickReady(now) {
  let bestPos = -1;
  let best = -1;
  for (let i = 0; i < ready.length; i++) {
    const t = ready[i];
    if (readyTime[t] > now) continue;
    if (best < 0 || betterTask(t, best, now)) {
      best = t;
      bestPos = i;
    }
  }
  if (bestPos < 0) return -1;
  const last = ready.pop();
  if (bestPos < ready.length) ready[bestPos] = last;
  return best;
}

while (scheduled < n) {
  const slot = machines.pop();
  let now = slot.time;
  let t = pickReady(now);
  if (t < 0) {
    let soon = Infinity;
    for (let i = 0; i < ready.length; i++) if (readyTime[ready[i]] < soon) soon = readyTime[ready[i]];
    now = Math.max(now, soon);
    t = pickReady(now);
  }
  if (t < 0) break;

  const start = Math.ceil(Math.max(now, readyTime[t]));
  const finish = start + duration[t];
  done[t] = 1;
  scheduled++;
  schedule.push({ id: tasks[t].id, start, machine: slot.machine });

  for (const v of children[t]) {
    if (finish > readyTime[v]) readyTime[v] = finish;
    if (--remainingDeps[v] === 0) ready.push(v);
  }

  machines.push({ time: finish, machine: slot.machine });

  if (scheduled < n && ready.length === 0 && machines.size === 0) break;
}

if (scheduled !== n) {
  const order = schedule.map(x => x.id);
  for (let i = 0; i < n; i++) if (!done[i]) order.push(tasks[i].id);
  console.log(JSON.stringify({ order }));
} else {
  console.log(JSON.stringify({ schedule }));
}
