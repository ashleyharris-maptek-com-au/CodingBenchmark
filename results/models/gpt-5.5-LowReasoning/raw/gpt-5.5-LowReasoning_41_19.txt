; SPIR-V
; Version: 1.0
OpCapability Shader
%glsl = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %inWorldPos %inNormal %inColor %outColor
OpExecutionMode %main OriginUpperLeft
OpSource GLSL 450
OpName %main "main"
OpName %UBO "UBO"
OpMemberName %UBO 0 "model"
OpMemberName %UBO 1 "view"
OpMemberName %UBO 2 "projection"
OpMemberName %UBO 3 "lightPos"
OpMemberName %UBO 4 "cameraPos"
OpMemberName %UBO 5 "params"
OpName %ubo "ubo"
OpName %inWorldPos "worldPos"
OpName %inNormal "normal"
OpName %inColor "color"
OpName %outColor "outColor"
OpDecorate %inWorldPos Location 0
OpDecorate %inNormal Location 1
OpDecorate %inColor Location 4
OpDecorate %outColor Location 0
OpDecorate %UBO Block
OpMemberDecorate %UBO 0 ColMajor
OpMemberDecorate %UBO 0 Offset 0
OpMemberDecorate %UBO 0 MatrixStride 16
OpMemberDecorate %UBO 1 ColMajor
OpMemberDecorate %UBO 1 Offset 64
OpMemberDecorate %UBO 1 MatrixStride 16
OpMemberDecorate %UBO 2 ColMajor
OpMemberDecorate %UBO 2 Offset 128
OpMemberDecorate %UBO 2 MatrixStride 16
OpMemberDecorate %UBO 3 Offset 192
OpMemberDecorate %UBO 4 Offset 208
OpMemberDecorate %UBO 5 Offset 224
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0
%void = OpTypeVoid
%fnVoid = OpTypeFunction %void
%float = OpTypeFloat 32
%int = OpTypeInt 32 1
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%mat4v4float = OpTypeMatrix %v4float 4
%UBO = OpTypeStruct %mat4v4float %mat4v4float %mat4v4float %v4float %v4float %v4float
%ptrUniformUBO = OpTypePointer Uniform %UBO
%ptrUniformV4 = OpTypePointer Uniform %v4float
%ptrInputV3 = OpTypePointer Input %v3float
%ptrOutputV4 = OpTypePointer Output %v4float
%ubo = OpVariable %ptrUniformUBO Uniform
%inWorldPos = OpVariable %ptrInputV3 Input
%inNormal = OpVariable %ptrInputV3 Input
%inColor = OpVariable %ptrInputV3 Input
%outColor = OpVariable %ptrOutputV4 Output
%int3 = OpConstant %int 3
%int4 = OpConstant %int 4
%float0 = OpConstant %float 0
%float1 = OpConstant %float 1
%float005 = OpConstant %float 0.05
%float01 = OpConstant %float 0.1
%float32 = OpConstant %float 32
%ambient = OpConstantComposite %v3float %float01 %float005 %float005
%white = OpConstantComposite %v3float %float1 %float1 %float1
%zero3 = OpConstantComposite %v3float %float0 %float0 %float0
%one3 = OpConstantComposite %v3float %float1 %float1 %float1
%main = OpFunction %void None %fnVoid
%entry = OpLabel
%worldPos = OpLoad %v3float %inWorldPos
%normal = OpLoad %v3float %inNormal
%matColor = OpLoad %v3float %inColor
%lightPtr = OpAccessChain %ptrUniformV4 %ubo %int3
%light4 = OpLoad %v4float %lightPtr
%lightPos = OpVectorShuffle %v3float %light4 %light4 0 1 2
%cameraPtr = OpAccessChain %ptrUniformV4 %ubo %int4
%camera4 = OpLoad %v4float %cameraPtr
%cameraPos = OpVectorShuffle %v3float %camera4 %camera4 0 1 2
%N = OpExtInst %v3float %glsl Normalize %normal
%lightDelta = OpFSub %v3float %lightPos %worldPos
%L = OpExtInst %v3float %glsl Normalize %lightDelta
%viewDelta = OpFSub %v3float %cameraPos %worldPos
%V = OpExtInst %v3float %glsl Normalize %viewDelta
%ndotlRaw = OpDot %float %N %L
%ndotl = OpExtInst %float %glsl FMax %ndotlRaw %float0
%diffuse = OpVectorTimesScalar %v3float %matColor %ndotl
%negL = OpFNegate %v3float %L
%R = OpExtInst %v3float %glsl Reflect %negL %N
%rdotvRaw = OpDot %float %R %V
%rdotv = OpExtInst %float %glsl FMax %rdotvRaw %float0
%specPower = OpExtInst %float %glsl Pow %rdotv %float32
%specular = OpVectorTimesScalar %v3float %white %specPower
%ambientDiffuse = OpFAdd %v3float %ambient %diffuse
%sumColor = OpFAdd %v3float %ambientDiffuse %specular
%clamped = OpExtInst %v3float %glsl FClamp %sumColor %zero3 %one3
%rgba = OpCompositeConstruct %v4float %clamped %float1
OpStore %outColor %rgba
OpReturn
OpFunctionEnd
