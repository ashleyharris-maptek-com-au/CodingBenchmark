use std::io::{self, Read};

fn dfs(
    pos: usize,
    order: &[usize],
    adj: &[Vec<usize>],
    need_a: usize,
    need_b: usize,
    assign: &mut [i8],
    cur_cut: usize,
    best_cut: &mut usize,
    best_assign: &mut Vec<i8>,
) {
    if cur_cut >= *best_cut {
        return;
    }

    if pos == order.len() {
        if need_a == 0 && need_b == 0 {
            *best_cut = cur_cut;
            best_assign.clone_from_slice(assign);
        }
        return;
    }

    if need_a > order.len() - pos || need_b > order.len() - pos {
        return;
    }

    let v = order[pos];

    if need_a > 0 {
        let mut add = 0usize;
        for &u in &adj[v] {
            if assign[u] == 1 {
                add += 1;
            }
        }
        assign[v] = 0;
        dfs(
            pos + 1,
            order,
            adj,
            need_a - 1,
            need_b,
            assign,
            cur_cut + add,
            best_cut,
            best_assign,
        );
        assign[v] = -1;
    }

    if need_b > 0 {
        let mut add = 0usize;
        for &u in &adj[v] {
            if assign[u] == 0 {
                add += 1;
            }
        }
        assign[v] = 1;
        dfs(
            pos + 1,
            order,
            adj,
            need_a,
            need_b - 1,
            assign,
            cur_cut + add,
            best_cut,
            best_assign,
        );
        assign[v] = -1;
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();
    let m: usize = it.next().unwrap().parse().unwrap();

    let mut edges = Vec::with_capacity(m);
    let mut adj = vec![Vec::<usize>::new(); n];

    for _ in 0..m {
        let u: usize = it.next().unwrap().parse().unwrap();
        let v: usize = it.next().unwrap().parse().unwrap();
        if u == v {
            continue;
        }
        edges.push((u, v));
        adj[u].push(v);
        adj[v].push(u);
    }

    let half = n / 2;

    let mut order: Vec<usize> = (0..n).collect();
    order.sort_by_key(|&v| std::cmp::Reverse(adj[v].len()));

    let mut assign = vec![-1i8; n];
    let mut start_pos = 0usize;
    let mut need_a = half;
    let need_b = half;

    // A and B are interchangeable, so force vertex 0 into A.
    assign[0] = 0;
    need_a -= 1;

    if let Some(p) = order.iter().position(|&v| v == 0) {
        order.swap(0, p);
        start_pos = 1;
    }

    let mut best_cut = usize::MAX;
    let mut best_assign = assign.clone();

    dfs(
        start_pos,
        &order,
        &adj,
        need_a,
        need_b,
        &mut assign,
        0,
        &mut best_cut,
        &mut best_assign,
    );

    let mut part_a: Vec<usize> = (0..n).filter(|&v| best_assign[v] == 0).collect();
    part_a.sort_unstable();

    let mut actual_cut = 0usize;
    let mut in_a = vec![false; n];
    for &v in &part_a {
        in_a[v] = true;
    }
    for &(u, v) in &edges {
        if in_a[u] != in_a[v] {
            actual_cut += 1;
        }
    }

    println!("{}", actual_cut);
    for (i, v) in part_a.iter().enumerate() {
        if i > 0 {
            print!(" ");
        }
        print!("{}", v);
    }
    println!();
}
