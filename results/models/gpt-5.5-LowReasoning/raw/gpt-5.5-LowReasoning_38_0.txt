import sys


def main():
    lines = sys.stdin.read().splitlines()
    if not lines:
        return

    first = lines[0].split()
    if len(first) < 2:
        print("NO SOLUTION")
        return

    u = int(first[0])
    s = int(first[1])

    sets = []
    elem_to_sets = [set() for _ in range(u)]

    for i in range(s):
        line = lines[i + 1] if i + 1 < len(lines) else ""
        elems = set()
        for tok in line.split():
            x = int(tok)
            if 0 <= x < u:
                elems.add(x)
        fs = frozenset(elems)
        sets.append(fs)
        for x in fs:
            elem_to_sets[x].add(i)

    # Empty universe has the empty exact cover.
    if u == 0:
        print("SOLUTION")
        print()
        return

    uncovered = set(range(u))
    active_sets = set(range(s))
    chosen = []

    sys.setrecursionlimit(max(1000, u + s + 50))

    def search():
        if not uncovered:
            return True

        # Choose the uncovered element with the fewest currently usable sets.
        best_elem = None
        best_candidates = None
        best_count = None

        for e in uncovered:
            candidates = []
            for idx in elem_to_sets[e]:
                if idx in active_sets and sets[idx] <= uncovered:
                    candidates.append(idx)
            count = len(candidates)
            if count == 0:
                return False
            if best_count is None or count < best_count or (
                count == best_count and e < best_elem
            ):
                best_elem = e
                best_candidates = candidates
                best_count = count
                if count == 1:
                    break

        best_candidates.sort()

        for idx in best_candidates:
            elems = sets[idx]
            if not elems <= uncovered:
                continue

            chosen.append(idx)

            removed_uncovered = elems
            removed_active = []
            for j in list(active_sets):
                if j == idx or sets[j] & elems:
                    active_sets.remove(j)
                    removed_active.append(j)
            uncovered.difference_update(elems)

            if search():
                return True

            uncovered.update(removed_uncovered)
            for j in removed_active:
                active_sets.add(j)
            chosen.pop()

        return False

    if search():
        print("SOLUTION")
        print(" ".join(map(str, sorted(chosen))))
    else:
        print("NO SOLUTION")


if __name__ == "__main__":
    main()
