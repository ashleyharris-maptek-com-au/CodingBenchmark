using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Linq;

public static class Program
{
    const int ExactSmallN = 2200;
    const int ExactMaxCandidates = 1400;
    const int TimeLimitMs = 292000;

    static int N, M;
    static int[][] Adj = Array.Empty<int[]>();
    static int[] Degree = Array.Empty<int>();
    static int[] Best = Array.Empty<int>();
    static Stopwatch Timer = new Stopwatch();
    static long DeadlineTicks;

    public static void Main()
    {
        var fs = new FastScanner(Console.OpenStandardInput());
        N = fs.NextInt();
        M = fs.NextInt();
        if (N <= 0)
        {
            Console.WriteLine(0);
            Console.WriteLine();
            return;
        }

        var lists = new List<int>[N];
        for (int i = 0; i < N; i++) lists[i] = new List<int>();

        for (int i = 0; i < M; i++)
        {
            int u = fs.NextInt();
            int v = fs.NextInt();
            if ((uint)u >= (uint)N || (uint)v >= (uint)N || u == v) continue;
            lists[u].Add(v);
            lists[v].Add(u);
        }

        Adj = new int[N][];
        Degree = new int[N];
        for (int i = 0; i < N; i++)
        {
            if (lists[i].Count > 1)
            {
                lists[i].Sort();
                int w = 1;
                for (int r = 1; r < lists[i].Count; r++)
                    if (lists[i][r] != lists[i][w - 1]) lists[i][w++] = lists[i][r];
                if (w < lists[i].Count) lists[i].RemoveRange(w, lists[i].Count - w);
            }
            Adj[i] = lists[i].ToArray();
            Degree[i] = Adj[i].Length;
        }

        Timer.Start();
        DeadlineTicks = TimeSpan.FromMilliseconds(TimeLimitMs).Ticks;

        Best = GreedyClique();
        TryImproveByLocalSearch();

        if (N <= ExactSmallN)
        {
            ExactOnWholeGraph();
        }
        else
        {
            ExactAroundPromisingVertices();
        }

        Best = VerifyClique(Best) ? Best : RepairToClique(Best);
        Array.Sort(Best);
        Console.WriteLine(Best.Length);
        Console.WriteLine(string.Join(" ", Best));
    }

    static bool OutOfTime()
    {
        return Timer.ElapsedTicks >= DeadlineTicks;
    }

    static int[] GreedyClique()
    {
        int[] order = Enumerable.Range(0, N).ToArray();
        Array.Sort(order, (a, b) => Degree[b].CompareTo(Degree[a]));

        int[] best = N > 0 ? new[] { order[0] } : Array.Empty<int>();
        int starts = Math.Min(order.Length, N <= 50000 ? 256 : 64);
        for (int s = 0; s < starts && !OutOfTime(); s++)
        {
            var clique = new List<int>();
            var candidates = new List<int>();
            int first = order[s];
            clique.Add(first);
            foreach (int v in Adj[first]) if (Degree[v] >= best.Length - 1) candidates.Add(v);
            candidates.Sort((a, b) => Degree[b].CompareTo(Degree[a]));

            while (candidates.Count > 0)
            {
                int v = candidates[0];
                candidates.RemoveAt(0);
                bool ok = true;
                for (int i = 0; i < clique.Count; i++)
                    if (!HasEdge(v, clique[i])) { ok = false; break; }
                if (!ok) continue;

                clique.Add(v);
                for (int i = candidates.Count - 1; i >= 0; i--)
                    if (!HasEdge(v, candidates[i])) candidates.RemoveAt(i);
            }
            if (clique.Count > best.Length) best = clique.ToArray();
        }
        return best;
    }

    static void TryImproveByLocalSearch()
    {
        if (Best.Length == 0) return;
        var cand = new List<int>();
        for (int v = 0; v < N && !OutOfTime(); v++)
        {
            if (Degree[v] < Best.Length) continue;
            bool all = true;
            for (int i = 0; i < Best.Length; i++)
                if (!HasEdge(v, Best[i])) { all = false; break; }
            if (all)
            {
                cand.Add(v);
                if (cand.Count > 10000) break;
            }
        }
        if (cand.Count > 0)
        {
            var tmp = Best.Concat(cand).Distinct().ToArray();
            if (VerifyClique(tmp)) Best = tmp;
        }
    }

    static void ExactOnWholeGraph()
    {
        int[] vertices = Enumerable.Range(0, N).Where(v => Degree[v] >= Best.Length - 1).ToArray();
        if (vertices.Length <= ExactMaxCandidates)
            RunExact(vertices);
        else
            ExactAroundPromisingVertices();
    }

    static void ExactAroundPromisingVertices()
    {
        int[] order = Enumerable.Range(0, N).Where(v => Degree[v] >= Best.Length)
            .OrderByDescending(v => Degree[v]).Take(512).ToArray();
        foreach (int root in order)
        {
            if (OutOfTime()) break;
            var verts = new List<int> { root };
            foreach (int v in Adj[root])
                if (Degree[v] >= Best.Length) verts.Add(v);
            if (verts.Count <= Best.Length || verts.Count > ExactMaxCandidates) continue;
            RunExact(verts.ToArray());
        }
    }

    static void RunExact(int[] vertices)
    {
        if (OutOfTime() || vertices.Length <= Best.Length) return;

        Array.Sort(vertices, (a, b) => Degree[b].CompareTo(Degree[a]));
        var map = new Dictionary<int, int>(vertices.Length * 2);
        for (int i = 0; i < vertices.Length; i++) map[vertices[i]] = i;

        ulong[] masks = new ulong[vertices.Length];
        if (vertices.Length <= 64)
        {
            for (int i = 0; i < vertices.Length; i++)
            {
                ulong m = 0;
                foreach (int nb in Adj[vertices[i]])
                    if (map.TryGetValue(nb, out int j)) m |= 1UL << j;
                masks[i] = m;
            }
            Expand64(0, (vertices.Length == 64 ? ulong.MaxValue : ((1UL << vertices.Length) - 1)), masks, vertices, new List<int>());
            return;
        }

        var p = new List<int>(vertices);
        ExpandList(new List<int>(), p);
    }

    static void Expand64(int depth, ulong p, ulong[] masks, int[] verts, List<int> cur)
    {
        if (p == 0)
        {
            if (cur.Count > Best.Length) Best = cur.ToArray();
            return;
        }
        if (cur.Count + PopCount(p) <= Best.Length || OutOfTime()) return;

        while (p != 0)
        {
            if (cur.Count + PopCount(p) <= Best.Length || OutOfTime()) return;
            int i = BitOperations.TrailingZeroCount(p);
            p &= ~(1UL << i);
            cur.Add(verts[i]);
            Expand64(depth + 1, p & masks[i], masks, verts, cur);
            cur.RemoveAt(cur.Count - 1);
        }
    }

    static void ExpandList(List<int> cur, List<int> p)
    {
        if (p.Count == 0)
        {
            if (cur.Count > Best.Length) Best = cur.ToArray();
            return;
        }
        if (cur.Count + p.Count <= Best.Length || OutOfTime()) return;

        ColorSort(p, out int[] colors);
        for (int idx = p.Count - 1; idx >= 0; idx--)
        {
            if (cur.Count + colors[idx] <= Best.Length || OutOfTime()) return;
            int v = p[idx];
            cur.Add(v);
            var np = new List<int>();
            for (int j = 0; j < idx; j++)
                if (HasEdge(v, p[j])) np.Add(p[j]);
            ExpandList(cur, np);
            cur.RemoveAt(cur.Count - 1);
            p.RemoveAt(idx);
        }
    }

    static void ColorSort(List<int> p, out int[] colors)
    {
        p.Sort((a, b) => Degree[a].CompareTo(Degree[b]));
        colors = new int[p.Count];
        var colorClasses = new List<List<int>>();
        for (int i = 0; i < p.Count; i++)
        {
            int v = p[i];
            int c;
            for (c = 0; c < colorClasses.Count; c++)
            {
                bool conflict = false;
                foreach (int u in colorClasses[c])
                    if (HasEdge(u, v)) { conflict = true; break; }
                if (!conflict) break;
            }
            if (c == colorClasses.Count) colorClasses.Add(new List<int>());
            colorClasses[c].Add(v);
            colors[i] = c + 1;
        }
    }

    static bool HasEdge(int a, int b)
    {
        return Array.BinarySearch(Adj[a], b) >= 0;
    }

    static bool VerifyClique(int[] c)
    {
        for (int i = 0; i < c.Length; i++)
            for (int j = i + 1; j < c.Length; j++)
                if (!HasEdge(c[i], c[j])) return false;
        return true;
    }

    static int[] RepairToClique(int[] vertices)
    {
        var res = new List<int>();
        foreach (int v in vertices.OrderByDescending(x => Degree[x]))
        {
            bool ok = true;
            foreach (int u in res)
                if (!HasEdge(u, v)) { ok = false; break; }
            if (ok) res.Add(v);
        }
        if (res.Count == 0 && N > 0) res.Add(0);
        return res.ToArray();
    }

    static int PopCount(ulong x)
    {
        x -= (x >> 1) & 0x5555555555555555UL;
        x = (x & 0x3333333333333333UL) + ((x >> 2) & 0x3333333333333333UL);
        return (int)((((x + (x >> 4)) & 0x0F0F0F0F0F0F0F0FUL) * 0x0101010101010101UL) >> 56);
    }

    static class BitOperations
    {
        static readonly int[] DeBruijn = {
            0, 1, 2, 7, 3, 13, 8, 19, 4, 25, 14, 28, 9, 34, 20, 40,
            5, 17, 26, 38, 15, 46, 29, 48, 10, 31, 35, 54, 21, 50, 41, 57,
            63, 6, 12, 18, 24, 27, 33, 39, 16, 37, 45, 47, 30, 53, 49, 56,
            62, 11, 23, 32, 36, 44, 52, 55, 61, 22, 43, 51, 60, 42, 59, 58
        };

        public static int TrailingZeroCount(ulong value)
        {
            return DeBruijn[(int)(((value & (~value + 1)) * 0x0218A392CD3D5DBFUL) >> 58)];
        }
    }

    sealed class FastScanner
    {
        readonly Stream _stream;
        readonly byte[] _buffer = new byte[1 << 20];
        int _len, _ptr;

        public FastScanner(Stream stream) { _stream = stream; }

        int Read()
        {
            if (_ptr >= _len)
            {
                _len = _stream.Read(_buffer, 0, _buffer.Length);
                _ptr = 0;
                if (_len == 0) return -1;
            }
            return _buffer[_ptr++];
        }

        public int NextInt()
        {
            int c;
            do { c = Read(); } while (c <= 32 && c >= 0);
            int sign = 1;
            if (c == '-') { sign = -1; c = Read(); }
            int val = 0;
            while (c > 32)
            {
                val = val * 10 + c - '0';
                c = Read();
            }
            return val * sign;
        }
    }
