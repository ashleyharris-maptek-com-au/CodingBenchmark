declare const require: any;
const fs = require('fs');

interface Point {
    x: number;
    y: number;
}

interface Package {
    id: number;
    x: number;
    y: number;
}

interface InputData {
    depot: Point;
    truckCount: number;
    packages: Package[];
}

function distance(p1: Point, p2: Point): number {
    return Math.sqrt(Math.pow(p1.x - p2.x, 2) + Math.pow(p1.y - p2.y, 2));
}

function solve() {
    const input: string = fs.readFileSync(0, 'utf8');
    const data: InputData = JSON.parse(input);

    const depot = data.depot;
    const truckCount = data.truckCount;
    const unvisitedPackages = [...data.packages];
    const routes: number[][] = Array.from({ length: truckCount }, () => []);

    // Greedy heuristic: Assign packages to trucks based on nearest neighbor 
    // to the last package in their current route (or depot if empty).

    while (unvisitedPackages.length > 0) {
        for (let i = 0; i < truckCount && unvisitedPackages.length > 0; i++) {
            const route = routes[i];
            const lastLocation = route.length === 0 ? depot : data.packages.find(p => p.id === route[route.length - 1])!;
            
            let nearestIdx = 0;
            let minDistance = Infinity;

            for (let j = 0; j < unvisitedPackages.length; j++) {
                const dist = distance(lastLocation, unvisitedPackages[j]);
                if (dist < minDistance) {
                    minDistance = dist;
                    nearestIdx = j;
                }
            }

            const packageToVisit = unvisitedPackages.splice(nearestIdx, 1)[0];
            route.push(packageToVisit.id);
        }
    }

    process.stdout.write(JSON.stringify({ routes }));
}

solve();
