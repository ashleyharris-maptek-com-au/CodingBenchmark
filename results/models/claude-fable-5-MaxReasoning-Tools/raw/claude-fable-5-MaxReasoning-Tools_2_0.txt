// Chinese Postman Problem (Route Inspection) solver.
//
// Pipeline:
//   1. Read the graph (fast byte-level stdin parser), build a CSR adjacency.
//   2. Find odd-degree vertices (there are always an even number, T).
//        T == 0           : graph is already Eulerian.
//        T <= 20          : exact minimum-weight perfect matching by bitmask DP
//                           over pairwise shortest-path distances (optimal tour).
//        moderate T, T*M  : full pairwise shortest paths (early-exit Dijkstra),
//        small            : greedy matching + 2-swap local improvement.
//        huge T           : repeated multi-source ("Voronoi") Dijkstra rounds:
//                           every unmatched odd vertex is a source; each edge that
//                           crosses two Voronoi cells proposes a pairing of the two
//                           sources with cost dist(a,u)+w(u,v)+dist(v,b); pairings
//                           are taken greedily by cost. Each round matches a large
//                           fraction of the remaining odd vertices; the tail is
//                           finished with a small distance-matrix match or
//                           nearest-neighbour searches.
//   3. Duplicate the edges of each matched pair's shortest path -> every degree even.
//   4. Hierholzer's algorithm (iterative) for the Euler circuit starting at node 0.
//
// Output: the node sequence of the circuit (starts and ends at node 0, covers every
// edge at least once). No length prefix is printed (it is optional per the spec).

using System;
using System.Collections.Generic;
using System.IO;

public static class Program
{
    static int N, M;
    static int[] EU, EV;      // edge endpoints
    static long[] EW;         // edge weights (clamped to >= 0)

    // CSR of the original graph (2*M directed arcs)
    static int[] gStart, gTo, gEdge;
    static long[] gW;

    // Shared Dijkstra state (stamped so no O(N) clearing between runs)
    static long[] dist;
    static int[] vstamp, parArc, lab;
    static int stampCur;
    static long[] hKey; static int[] hVal; static int hCnt; // binary heap

    static int[] odd; static int T;      // odd-degree vertices
    static int[] oddIdxOf;               // vertex -> index in odd[], else -1
    static bool[] matchedNode;
    static int[] mapBuf;                 // scratch vertex->subset-index map (-1 default)

    // duplicated edges added by the matching phase
    static List<int> dupU = new List<int>();
    static List<int> dupV = new List<int>();

    // scratch for Voronoi rounds
    static long[] candKey; static int[] candEdge;

    const long INF = long.MaxValue / 4;

    public static void Main()
    {
        byte[] data = ReadAll(Console.OpenStandardInput());
        int pos = 0;
        N = (int)NextLong(data, ref pos);
        M = (int)NextLong(data, ref pos);
        if (N <= 0) { WriteBytes("0\n"); return; }
        EU = new int[M]; EV = new int[M]; EW = new long[M];
        for (int e = 0; e < M; e++)
        {
            EU[e] = (int)NextLong(data, ref pos);
            EV[e] = (int)NextLong(data, ref pos);
            long w = NextLong(data, ref pos);
            EW[e] = w < 0 ? 0 : w;
        }
        data = null;

        if (M == 0) { WriteBytes("0\n"); return; }

        BuildCsr();

        int t = 0;
        for (int v = 0; v < N; v++) if (((gStart[v + 1] - gStart[v]) & 1) != 0) t++;
        T = t;
        if (T > 0)
        {
            odd = new int[T];
            oddIdxOf = new int[N];
            for (int v = 0; v < N; v++) oddIdxOf[v] = -1;
            int k = 0;
            for (int v = 0; v < N; v++)
                if (((gStart[v + 1] - gStart[v]) & 1) != 0) { oddIdxOf[v] = k; odd[k++] = v; }
            matchedNode = new bool[N];
            InitDijkstra();

            if (T <= 20)
            {
                SolveExactDP();
            }
            else if (M < N)
            {
                // forest (possibly plus a few self loops / parallel edges):
                // the shortest-path-tree parity fix is exact on trees
                TreeParityFix();
            }
            else if (T <= 1400 && (long)T * M <= 30000000L)
            {
                long[] mat = BuildMatrix(odd, T, oddIdxOf);
                GreedyMatrixMatch(odd, T, mat);
                FinishLeftoversByNearest();
            }
            else
            {
                SolveVoronoi();
            }
        }

        EulerAndOutput();
    }

    // ---------------- input / output ----------------

    static int inputLen;

    static byte[] ReadAll(Stream s)
    {
        byte[] buf = new byte[1 << 20];
        int len = 0;
        while (true)
        {
            if (len == buf.Length)
            {
                byte[] nb = new byte[buf.Length * 2];
                Array.Copy(buf, nb, len);
                buf = nb;
            }
            int r = s.Read(buf, len, buf.Length - len);
            if (r <= 0) break;
            len += r;
        }
        inputLen = len;
        return buf;
    }

    static long NextLong(byte[] b, ref int p)
    {
        while (p < inputLen && (b[p] < (byte)'0' || b[p] > (byte)'9') && b[p] != (byte)'-') p++;
        bool neg = false;
        if (p < inputLen && b[p] == (byte)'-') { neg = true; p++; }
        long v = 0;
        while (p < inputLen && b[p] >= (byte)'0' && b[p] <= (byte)'9')
        {
            v = v * 10 + (b[p] - (byte)'0');
            p++;
        }
        return neg ? -v : v;
    }

    static void WriteBytes(string s)
    {
        Stream os = Console.OpenStandardOutput();
        byte[] b = new byte[s.Length];
        for (int i = 0; i < s.Length; i++) b[i] = (byte)s[i];
        os.Write(b, 0, b.Length);
        os.Flush();
    }

    static void WriteRoute(int[] route, int rc)
    {
        byte[] buf = new byte[(long)rc * 12 + 16 > int.MaxValue ? int.MaxValue : rc * 12 + 16];
        int o = 0;
        byte[] tmp = new byte[12];
        for (int i = 0; i < rc; i++)
        {
            int x = route[i];
            if (x == 0) buf[o++] = (byte)'0';
            else
            {
                int tl = 0;
                while (x > 0) { tmp[tl++] = (byte)('0' + (x % 10)); x /= 10; }
                while (tl > 0) buf[o++] = tmp[--tl];
            }
            buf[o++] = (byte)(((i & 31) == 31) ? '\n' : ' ');
        }
        if (o > 0) buf[o - 1] = (byte)'\n';
        Stream os = Console.OpenStandardOutput();
        os.Write(buf, 0, o);
        os.Flush();
    }

    // ---------------- graph ----------------

    static void BuildCsr()
    {
        gStart = new int[N + 1];
        for (int e = 0; e < M; e++) { gStart[EU[e] + 1]++; gStart[EV[e] + 1]++; }
        for (int v = 0; v < N; v++) gStart[v + 1] += gStart[v];
        int A = 2 * M;
        gTo = new int[A]; gEdge = new int[A]; gW = new long[A];
        int[] cur = new int[N];
        Array.Copy(gStart, cur, N);
        for (int e = 0; e < M; e++)
        {
            int u = EU[e], v = EV[e];
            int c = cur[u]++; gTo[c] = v; gEdge[c] = e; gW[c] = EW[e];
            c = cur[v]++; gTo[c] = u; gEdge[c] = e; gW[c] = EW[e];
        }
    }

    static void InitDijkstra()
    {
        dist = new long[N];
        vstamp = new int[N];
        parArc = new int[N];
        lab = new int[N];
        mapBuf = new int[N];
        for (int i = 0; i < N; i++) mapBuf[i] = -1;
        int hcap = 2 * M + N + 8;
        hKey = new long[hcap];
        hVal = new int[hcap];
    }

    // ---------------- binary heap ----------------

    static void HPush(long key, int val)
    {
        int i = hCnt++;
        while (i > 0)
        {
            int p = (i - 1) >> 1;
            if (hKey[p] <= key) break;
            hKey[i] = hKey[p]; hVal[i] = hVal[p];
            i = p;
        }
        hKey[i] = key; hVal[i] = val;
    }

    static void HPop(out long key, out int val)
    {
        key = hKey[0]; val = hVal[0];
        int n = --hCnt;
        if (n > 0)
        {
            long k = hKey[n]; int v = hVal[n];
            int i = 0;
            while (true)
            {
                int c = 2 * i + 1;
                if (c >= n) break;
                if (c + 1 < n && hKey[c + 1] < hKey[c]) c++;
                if (hKey[c] >= k) break;
                hKey[i] = hKey[c]; hVal[i] = hVal[c];
                i = c;
            }
            hKey[i] = k; hVal[i] = v;
        }
    }

    // ---------------- Dijkstra variants ----------------

    // Fills row[map[x]] = dist(src,x) for every x with map[x] >= 0.
    // Early-exits once `want` mapped vertices have been finalized.
    static void DijkstraRow(int src, long[] row, int[] map, int want)
    {
        stampCur++; hCnt = 0;
        dist[src] = 0; vstamp[src] = stampCur;
        HPush(0, src);
        int got = 0;
        while (hCnt > 0)
        {
            long d; int x;
            HPop(out d, out x);
            if (d != dist[x]) continue;
            int oi = map[x];
            if (oi >= 0)
            {
                row[oi] = d;
                if (++got == want) return;
            }
            int end = gStart[x + 1];
            for (int a = gStart[x]; a < end; a++)
            {
                int to = gTo[a];
                if (to == x) continue;
                long nd = d + gW[a];
                if (vstamp[to] != stampCur || nd < dist[to])
                {
                    vstamp[to] = stampCur; dist[to] = nd;
                    HPush(nd, to);
                }
            }
        }
    }

    // Shortest path src->dst; duplicates its edges.
    static void DijkstraPathDup(int src, int dst)
    {
        stampCur++; hCnt = 0;
        dist[src] = 0; vstamp[src] = stampCur; parArc[src] = -1;
        HPush(0, src);
        while (hCnt > 0)
        {
            long d; int x;
            HPop(out d, out x);
            if (d != dist[x]) continue;
            if (x == dst) break;
            int end = gStart[x + 1];
            for (int a = gStart[x]; a < end; a++)
            {
                int to = gTo[a];
                if (to == x) continue;
                long nd = d + gW[a];
                if (vstamp[to] != stampCur || nd < dist[to])
                {
                    vstamp[to] = stampCur; dist[to] = nd; parArc[to] = a;
                    HPush(nd, to);
                }
            }
        }
        if (vstamp[dst] == stampCur) DupWalk(dst);
    }

    // Grows from src until the nearest unmatched odd vertex is found; matches
    // the two and duplicates the connecting path.
    static void DijkstraNearestDup(int src)
    {
        stampCur++; hCnt = 0;
        dist[src] = 0; vstamp[src] = stampCur; parArc[src] = -1;
        HPush(0, src);
        while (hCnt > 0)
        {
            long d; int x;
            HPop(out d, out x);
            if (d != dist[x]) continue;
            if (x != src && oddIdxOf[x] >= 0 && !matchedNode[x])
            {
                matchedNode[x] = true; matchedNode[src] = true;
                DupWalk(x);
                return;
            }
            int end = gStart[x + 1];
            for (int a = gStart[x]; a < end; a++)
            {
                int to = gTo[a];
                if (to == x) continue;
                long nd = d + gW[a];
                if (vstamp[to] != stampCur || nd < dist[to])
                {
                    vstamp[to] = stampCur; dist[to] = nd; parArc[to] = a;
                    HPush(nd, to);
                }
            }
        }
        matchedNode[src] = true; // unreachable partner (disconnected input) - bail
    }

    // Duplicates edges along the parent chain from x back to its source.
    static void DupWalk(int x)
    {
        while (parArc[x] != -1)
        {
            int a = parArc[x];
            int e = gEdge[a];
            dupU.Add(EU[e]); dupV.Add(EV[e]);
            x = (EU[e] == x) ? EV[e] : EU[e];
        }
    }

    // ---------------- matching strategies ----------------

    static long[] BuildMatrix(int[] nodes, int cnt, int[] map)
    {
        long[] mat = new long[cnt * cnt];
        long[] row = new long[cnt];
        for (int i = 0; i < cnt; i++)
        {
            for (int j = 0; j < cnt; j++) row[j] = INF;
            DijkstraRow(nodes[i], row, map, cnt);
            Array.Copy(row, 0, mat, i * cnt, cnt);
        }
        return mat;
    }

    static void ApplyPair(int aNode, int bNode)
    {
        matchedNode[aNode] = true; matchedNode[bNode] = true;
        DijkstraPathDup(aNode, bNode);
    }

    static void FinishLeftoversByNearest()
    {
        for (int i = 0; i < T; i++)
        {
            int v = odd[i];
            if (!matchedNode[v]) DijkstraNearestDup(v);
        }
    }

    // Guaranteed-fast parity repair for all remaining unmatched odd vertices:
    // build a shortest-path forest (one Dijkstra tree per component), then
    // duplicate exactly those tree edges whose subtree contains an odd number
    // of unmatched odd vertices. One pass, O((V+E) log V), duplicates at most
    // one copy of each tree edge; optimal on trees/stars and immune to hub
    // structures that starve the Voronoi rounds.
    static void TreeParityFix()
    {
        stampCur++;
        int st = stampCur;
        int[] order = new int[N];
        int oc = 0;
        for (int i = 0; i < T; i++)
        {
            int root = odd[i];
            if (matchedNode[root] || vstamp[root] == st) continue;
            hCnt = 0;
            dist[root] = 0; vstamp[root] = st; parArc[root] = -1;
            HPush(0, root);
            while (hCnt > 0)
            {
                long d; int x;
                HPop(out d, out x);
                if (d != dist[x]) continue;
                order[oc++] = x;
                int end = gStart[x + 1];
                for (int a = gStart[x]; a < end; a++)
                {
                    int to = gTo[a];
                    if (to == x) continue;
                    long nd = d + gW[a];
                    if (vstamp[to] != st || nd < dist[to])
                    {
                        vstamp[to] = st; dist[to] = nd; parArc[to] = a;
                        HPush(nd, to);
                    }
                }
            }
        }
        int[] cntOdd = lab; // scratch reuse
        for (int i = 0; i < oc; i++)
        {
            int x = order[i];
            cntOdd[x] = (oddIdxOf[x] >= 0 && !matchedNode[x]) ? 1 : 0;
        }
        // pop order finalizes parents before children, so reverse = leaves first
        for (int i = oc - 1; i >= 0; i--)
        {
            int x = order[i];
            if ((cntOdd[x] & 1) != 0 && parArc[x] != -1)
            {
                int a = parArc[x];
                int e = gEdge[a];
                dupU.Add(EU[e]); dupV.Add(EV[e]);
                int p = (EU[e] == x) ? EV[e] : EU[e];
                cntOdd[p] ^= 1;
            }
        }
        for (int i = 0; i < T; i++) matchedNode[odd[i]] = true;
    }

    // Exact minimum-weight perfect matching for T <= 20 via bitmask DP.
    static void SolveExactDP()
    {
        long[] mat = BuildMatrix(odd, T, oddIdxOf);
        int full = (1 << T) - 1;
        long[] dp = new long[1 << T];
        int[] ch = new int[1 << T];
        dp[0] = 0;
        for (int mask = 1; mask <= full; mask++)
        {
            int pc;
            unchecked
            {
                pc = mask - ((mask >> 1) & 0x55555555);
                pc = (pc & 0x33333333) + ((pc >> 2) & 0x33333333);
                pc = (((pc + (pc >> 4)) & 0x0F0F0F0F) * 0x01010101) >> 24;
            }
            if ((pc & 1) != 0) { dp[mask] = INF; ch[mask] = -1; continue; }
            int i = 0;
            while (((mask >> i) & 1) == 0) i++;
            long best = INF; int bj = -1;
            int rem = mask & ~(1 << i);
            for (int j = i + 1; j < T; j++)
            {
                if (((rem >> j) & 1) == 0) continue;
                long dij = mat[i * T + j];
                if (dij >= INF) continue;
                long c = dp[rem & ~(1 << j)] + dij;
                if (c < best) { best = c; bj = j; }
            }
            dp[mask] = best; ch[mask] = bj;
        }
        int m2 = full;
        while (m2 != 0)
        {
            int i = 0;
            while (((m2 >> i) & 1) == 0) i++;
            int j = ch[m2];
            if (j < 0) break; // disconnected anomaly: finish below
            ApplyPair(odd[i], odd[j]);
            m2 &= ~(1 << i); m2 &= ~(1 << j);
        }
        FinishLeftoversByNearest();
    }

    // Greedy matching over a full pairwise distance matrix + 2-swap improvement.
    static void GreedyMatrixMatch(int[] nodes, int cnt, long[] mat)
    {
        int pcap = cnt * (cnt - 1) / 2;
        long[] pk = new long[pcap]; int[] pv = new int[pcap];
        int pcnt = 0;
        for (int i = 0; i < cnt; i++)
        {
            int baseI = i * cnt;
            for (int j = i + 1; j < cnt; j++)
            {
                long d = mat[baseI + j];
                if (d >= INF) continue;
                pk[pcnt] = d; pv[pcnt] = i * cnt + j; pcnt++;
            }
        }
        Array.Sort(pk, pv, 0, pcnt);
        int[] mate = new int[cnt];
        for (int i = 0; i < cnt; i++) mate[i] = -1;
        int made = 0;
        for (int k = 0; k < pcnt && made * 2 < cnt; k++)
        {
            int i = pv[k] / cnt, j = pv[k] % cnt;
            if (mate[i] >= 0 || mate[j] >= 0) continue;
            mate[i] = j; mate[j] = i; made++;
        }
        int[] pa = new int[made], pb = new int[made];
        int q0 = 0;
        for (int i = 0; i < cnt; i++)
            if (mate[i] > i) { pa[q0] = i; pb[q0] = mate[i]; q0++; }

        // 2-swap local improvement between pairs
        bool improved = true; int pass = 0;
        while (improved && pass++ < 8)
        {
            improved = false;
            for (int p = 0; p < made; p++)
            {
                for (int q = p + 1; q < made; q++)
                {
                    int a = pa[p], b = pb[p], c = pa[q], d = pb[q];
                    long cur = mat[a * cnt + b] + mat[c * cnt + d];
                    long x1 = mat[a * cnt + c], y1 = mat[b * cnt + d];
                    long x2 = mat[a * cnt + d], y2 = mat[b * cnt + c];
                    long o1 = (x1 >= INF || y1 >= INF) ? INF : x1 + y1;
                    long o2 = (x2 >= INF || y2 >= INF) ? INF : x2 + y2;
                    if (o1 < cur || o2 < cur)
                    {
                        if (o1 <= o2) { pb[p] = c; pa[q] = b; pb[q] = d; }
                        else { pb[p] = d; pa[q] = b; pb[q] = c; }
                        improved = true;
                    }
                }
            }
        }
        for (int p = 0; p < made; p++) ApplyPair(nodes[pa[p]], nodes[pb[p]]);
    }

    // One multi-source round: Voronoi partition around unmatched odd sources,
    // then greedy matching over cheapest cell-crossing edges. Returns pairs made.
    static int VoronoiRound(int[] srcs, int cnt)
    {
        stampCur++; hCnt = 0;
        for (int i = 0; i < cnt; i++)
        {
            int s = srcs[i];
            dist[s] = 0; vstamp[s] = stampCur; parArc[s] = -1; lab[s] = s;
            HPush(0, s);
        }
        while (hCnt > 0)
        {
            long d; int x;
            HPop(out d, out x);
            if (d != dist[x]) continue;
            int end = gStart[x + 1];
            for (int a = gStart[x]; a < end; a++)
            {
                int to = gTo[a];
                if (to == x) continue;
                long nd = d + gW[a];
                if (vstamp[to] != stampCur || nd < dist[to])
                {
                    vstamp[to] = stampCur; dist[to] = nd;
                    parArc[to] = a; lab[to] = lab[x];
                    HPush(nd, to);
                }
            }
        }
        int cc = 0;
        for (int e = 0; e < M; e++)
        {
            int u = EU[e], v = EV[e];
            if (vstamp[u] != stampCur || vstamp[v] != stampCur) continue;
            if (lab[u] == lab[v]) continue;
            candKey[cc] = dist[u] + dist[v] + EW[e];
            candEdge[cc] = e; cc++;
        }
        Array.Sort(candKey, candEdge, 0, cc);
        int m = 0;
        for (int i = 0; i < cc; i++)
        {
            int e = candEdge[i];
            int a = lab[EU[e]], b = lab[EV[e]];
            if (matchedNode[a] || matchedNode[b]) continue;
            matchedNode[a] = true; matchedNode[b] = true;
            DupWalk(EU[e]); DupWalk(EV[e]);
            dupU.Add(EU[e]); dupV.Add(EV[e]);
            m++;
        }
        return m;
    }

    static void SolveVoronoi()
    {
        candKey = new long[M > 0 ? M : 1];
        candEdge = new int[M > 0 ? M : 1];
        int[] srcs = new int[T];
        int round = 0;
        while (true)
        {
            int cnt = 0;
            for (int i = 0; i < T; i++)
            {
                int v = odd[i];
                if (!matchedNode[v]) srcs[cnt++] = v;
            }
            if (cnt == 0) return;
            // Small remainder: finish with a quality distance-matrix match when
            // the per-source Dijkstra cost is affordable.
            if (cnt <= 1400 && (long)cnt * M <= 30000000L && (M <= 250000 || cnt <= 8))
            {
                for (int i = 0; i < cnt; i++) mapBuf[srcs[i]] = i;
                long[] mat = BuildMatrix(srcs, cnt, mapBuf);
                GreedyMatrixMatch(srcs, cnt, mat);
                for (int i = 0; i < cnt; i++) mapBuf[srcs[i]] = -1;
                TreeParityFix();
                return;
            }
            if (round >= 25) { TreeParityFix(); return; }
            int m = VoronoiRound(srcs, cnt);
            round++;
            // Slow progress (e.g. hub-dominated topology): repair the rest in one pass.
            if (m * 8 < cnt) { TreeParityFix(); return; }
        }
    }

    // ---------------- Euler circuit ----------------

    static void EulerAndOutput()
    {
        int D = dupU.Count;
        int M2 = M + D;
        int[] s2 = new int[N + 1];
        for (int e = 0; e < M; e++) { s2[EU[e] + 1]++; s2[EV[e] + 1]++; }
        for (int k = 0; k < D; k++) { s2[dupU[k] + 1]++; s2[dupV[k] + 1]++; }
        for (int v = 0; v < N; v++) s2[v + 1] += s2[v];
        int A2 = 2 * M2;
        int[] to2 = new int[A2]; int[] eid2 = new int[A2];
        int[] cur = new int[N];
        Array.Copy(s2, cur, N);
        for (int e = 0; e < M; e++)
        {
            int u = EU[e], v = EV[e];
            int c = cur[u]++; to2[c] = v; eid2[c] = e;
            c = cur[v]++; to2[c] = u; eid2[c] = e;
        }
        for (int k = 0; k < D; k++)
        {
            int u = dupU[k], v = dupV[k]; int e = M + k;
            int c = cur[u]++; to2[c] = v; eid2[c] = e;
            c = cur[v]++; to2[c] = u; eid2[c] = e;
        }
        int[] it = cur;
        Array.Copy(s2, it, N);
        byte[] used = new byte[M2];
        int[] stk = new int[M2 + 2]; int sp = 0;
        stk[sp++] = 0;
        int[] route = new int[M2 + 2]; int rc = 0;
        while (sp > 0)
        {
            int v = stk[sp - 1];
            int i = it[v]; int end = s2[v + 1];
            while (i < end && used[eid2[i]] != 0) i++;
            if (i == end)
            {
                it[v] = i;
                route[rc++] = v;
                sp--;
            }
            else
            {
                used[eid2[i]] = 1;
                it[v] = i + 1;
                stk[sp++] = to2[i];
            }
        }
        WriteRoute(route, rc);
    }
