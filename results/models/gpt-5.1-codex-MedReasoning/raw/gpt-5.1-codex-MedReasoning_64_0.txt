```typescript
declare const require: any;

const fs = require('fs');

const input: string = fs.readFileSync(0, 'utf8');
let data: any = {};
try {
    data = JSON.parse(input);
} catch {
    data = {};
}

const rawGrid: any[] = Array.isArray(data?.probability) ? data.probability : [];
const rawDrones: any[] = Array.isArray(data?.drones) ? data.drones : [];
const parsedMaxSteps = Number(data?.maxStepsPerDrone);
const maxStepsPerDrone: number = Number.isFinite(parsedMaxSteps) ? Math.max(0, Math.floor(parsedMaxSteps)) : 0;

const height: number = rawGrid.length;
let width = 0;
for (const row of rawGrid) {
    if (Array.isArray(row)) {
        width = Math.max(width, row.length);
    }
}

let outputPaths: number[][][] = [];

if (height === 0 || width === 0) {
    outputPaths = rawDrones.map((drone: any) => {
        const xVal = Number(drone?.x);
        const yVal = Number(drone?.y);
        const x = Number.isFinite(xVal) ? Math.round(xVal) : 0;
        const y = Number.isFinite(yVal) ? Math.round(yVal) : 0;
        return [[x, y]];
    });
} else {
    const grid: number[][] = Array.from({ length: height }, (_, y) => {
        const sourceRow = Array.isArray(rawGrid[y]) ? rawGrid[y] : [];
        const rowArray = new Array<number>(width).fill(0);
        for (let x = 0; x < width; x++) {
            const numeric = Number(sourceRow[x]);
            rowArray[x] = Number.isFinite(numeric) ? numeric : 0;
        }
        return rowArray;
    });

    const positiveCells: { x: number; y: number; value: number }[] = [];
    for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
            const value = grid[y][x];
            if (value > 0) {
                positiveCells.push({ x, y, value });
            }
        }
    }

    const directions: [number, number][] = [
        [1, 0],
        [-1, 0],
        [0, 1],
        [0, -1],
    ];

    const globalCovered = new Set<string>();
    let coveredPositiveCount = 0;

    const clamp = (value: number, min: number, max: number): number => {
        if (value < min) return min;
        if (value > max) return max;
        return value;
    };

    const cellKey = (x: number, y: number): string => `${x},${y}`;

    const isWalkable = (x: number, y: number): boolean => {
        if (x < 0 || x >= width || y < 0 || y >= height) return false;
        return grid[y][x] >= 0;
    };

    const findNearestWalkable = (sx: number, sy: number): [number, number] | null => {
        let bestX = -1;
        let bestY = -1;
        let bestDist = Infinity;
        for (let y = 0; y < height; y++) {
            for (let x = 0; x < width; x++) {
                if (grid[y][x] >= 0) {
                    const dist = Math.abs(x - sx) + Math.abs(y - sy);
                    if (dist < bestDist) {
                        bestDist = dist;
                        bestX = x;
                        bestY = y;
                    }
                }
            }
        }
        if (!Number.isFinite(bestDist)) {
            return null;
        }
        return [bestX, bestY];
    };

    const ensureValidStart = (rawX: number, rawY: number): [number, number] => {
        let x = Number.isFinite(rawX) ? Math.round(rawX) : 0;
        let y = Number.isFinite(rawY) ? Math.round(rawY) : 0;
        x = clamp(x, 0, width - 1);
        y = clamp(y, 0, height - 1);
        if (isWalkable(x, y)) {
            return [x, y];
        }
        const fallback = findNearestWalkable(x, y);
        if (fallback) {
            return fallback;
        }
        return [x, y];
    };

    const markVisited = (x: number, y: number): void => {
        const key = cellKey(x, y);
        if (!globalCovered.has(key)) {
            globalCovered.add(key);
            if (y >= 0 && y < height && x >= 0 && x < width) {
                if (grid[y][x] > 0) {
                    coveredPositiveCount++;
                }
            }
        }
    };

    interface BFSResult {
        dist: number[][];
        parentX: number[][];
        parentY: number[][];
    }

    const computeBFSLimited = (startX: number, startY: number, stepsLimit: number): BFSResult => {
        const dist = Array.from({ length: height }, () => Array<number>(width).fill(-1));
        const parentX = Array.from({ length: height }, () => Array<number>(width).fill(-1));
        const parentY = Array.from({ length: height }, () => Array<number>(width).fill(-1));
        const queueX: number[] = [];
        const queueY: number[] = [];
        dist[startY][startX] = 0;
        parentX[startY][startX] = startX;
        parentY[startY][startX] = startY;
        queueX.push(startX);
        queueY.push(startY);
        let head = 0;
        while (head < queueX.length) {
            const x = queueX[head];
            const y = queueY[head];
            head++;
            const currentDist = dist[y][x];
            if (currentDist >= stepsLimit) {
                continue;
            }
            for (const [dx, dy] of directions) {
                const nx = x + dx;
                const ny = y + dy;
                if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;
                if (grid[ny][nx] < 0) continue;
                if (dist[ny][nx] !== -1) continue;
                dist[ny][nx] = currentDist + 1;
                parentX[ny][nx] = x;
                parentY[ny][nx] = y;
                queueX.push(nx);
                queueY.push(ny);
            }
        }
        return { dist, parentX, parentY };
    };

    const reconstructPath = (
        targetX: number,
        targetY: number,
        startX: number,
        startY: number,
        parentX: number[][],
        parentY: number[][]
    ): [number, number][] | null => {
        const reversed: [number, number][] = [[targetX, targetY]];
        let cx = targetX;
        let cy = targetY;
        while (!(cx === startX && cy === startY)) {
            const pxRow = parentX[cy];
            const pyRow = parentY[cy];
            if (!pxRow || !pyRow) {
                return null;
            }
            const px = pxRow[cx];
            const py = pyRow[cx];
            if (px === undefined || py === undefined || px === -1 || py === -1) {
                return null;
            }
            cx = px;
            cy = py;
            reversed.push([cx, cy]);
        }
        reversed.reverse();
        return reversed;
    };

    const planPathForDrone = (droneData: any): number[][] => {
        const rawX = Number(droneData?.x);
        const rawY = Number(droneData?.y);
        const [startX, startY] = ensureValidStart(rawX, rawY);
        const path: number[][] = [[startX, startY]];
        let currentX = startX;
        let currentY = startY;
        let stepsUsed = 0;
        markVisited(startX, startY);

        if (maxStepsPerDrone <= 0 || positiveCells.length === 0 || coveredPositiveCount >= positiveCells.length) {
            return path;
        }

        while (stepsUsed < maxStepsPerDrone) {
            if (coveredPositiveCount >= positiveCells.length) {
                break;
            }
            const stepsRemaining = maxStepsPerDrone - stepsUsed;
            if (stepsRemaining <= 0) {
                break;
            }
            const bfs = computeBFSLimited(currentX, currentY, stepsRemaining);
            let bestScore = -Infinity;
            let bestTarget: { x: number; y: number } | null = null;

            for (const cell of positiveCells) {
                const keyStr = cellKey(cell.x, cell.y);
                if (globalCovered.has(keyStr)) {
                    continue;
                }
                const distance = bfs.dist[cell.y][cell.x];
                if (distance === undefined || distance === -1 || distance > stepsRemaining) {
                    continue;
                }
                const score = cell.value / (1 + distance * 0.6);
                if (score > bestScore) {
                    bestScore = score;
                    bestTarget = { x: cell.x, y: cell.y };
                }
            }

            if (!bestTarget) {
                break;
            }

            const pathSegment = reconstructPath(
                bestTarget.x,
                bestTarget.y,
                currentX,
                currentY,
                bfs.parentX,
                bfs.parentY
            );
            if (!pathSegment) {
                break;
            }

            for (let i = 1; i < pathSegment.length && stepsUsed < maxStepsPerDrone; i++) {
                const [nextX, nextY] = pathSegment[i];
                path.push([nextX, nextY]);
                stepsUsed++;
                currentX = nextX;
                currentY = nextY;
                markVisited(nextX, nextY);
            }
        }

        return path;
    };

    outputPaths = rawDrones.map((drone: any) => planPathForDrone(drone));
}

console.log(JSON.stringify({ paths: outputPaths }));
```
