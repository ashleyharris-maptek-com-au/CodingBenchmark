declare const require: any;
declare const process: any;

const fs = require('fs');
const input: string = fs.readFileSync(0, 'utf8');

if (!input || input.trim().length === 0) {
  console.log('{"assignments":[]}');
  process.exit(0);
}

const data = JSON.parse(input);
const rawGrid: any[] = Array.isArray(data.grid) ? data.grid : [];
const grid: string[] = rawGrid.map((row) => (typeof row === 'string' ? row : ''));
const height = grid.length;
const width = grid.reduce((max, row) => Math.max(max, row.length), 0);

const assignments: number[][] =
  width === 0 || height === 0 ? simpleAssignment(data) : gridAssignment(data, grid, width, height);

console.log(JSON.stringify({ assignments }));

function simpleAssignment(data: any): number[][] {
  const robotsInput: any[] = Array.isArray(data.robots) ? data.robots : [];
  let maxRobotId = -1;
  for (const robot of robotsInput) {
    if (robot && typeof robot.id === 'number' && robot.id > maxRobotId) {
      maxRobotId = robot.id;
    }
  }
  const assignments: number[][] =
    maxRobotId >= 0 ? Array.from({ length: maxRobotId + 1 }, () => [] as number[]) : [];
  if (robotsInput.length === 0) {
    return assignments;
  }
  const robotIds = robotsInput
    .map((robot) => (robot && typeof robot.id === 'number' ? robot.id : 0))
    .sort((a, b) => a - b);

  const jobsInput: any[] = Array.isArray(data.jobs) ? data.jobs : [];
  const sortedJobs = jobsInput.slice().sort((a, b) => {
    const pa = typeof a?.priority === 'number' ? a.priority : 0;
    const pb = typeof b?.priority === 'number' ? b.priority : 0;
    if (pb !== pa) {
      return pb - pa;
    }
    const idA = typeof a?.id === 'number' ? a.id : 0;
    const idB = typeof b?.id === 'number' ? b.id : 0;
    return idA - idB;
  });

  for (let i = 0; i < sortedJobs.length; i++) {
    const job = sortedJobs[i];
    const jobId = typeof job?.id === 'number' ? job.id : i;
    const robotId = robotIds[i % robotIds.length];
    if (!assignments[robotId]) {
      assignments[robotId] = [];
    }
    assignments[robotId].push(jobId);
  }
  return assignments;
}

function gridAssignment(data: any, grid: string[], width: number, height: number): number[][] {
  const totalCells = width * height;
  const passable = new Uint8Array(totalCells);
  for (let y = 0; y < height; y++) {
    const row = typeof grid[y] === 'string' ? grid[y] : '';
    for (let x = 0; x < width; x++) {
      const ch = row.charAt(x);
      passable[y * width + x] = ch === '.' ? 1 : 0;
    }
  }

  const robotsInput: any[] = Array.isArray(data.robots) ? data.robots : [];
  let maxRobotId = -1;
  for (const robot of robotsInput) {
    if (robot && typeof robot.id === 'number' && robot.id > maxRobotId) {
      maxRobotId = robot.id;
    }
  }
  const assignments: number[][] =
    maxRobotId >= 0 ? Array.from({ length: maxRobotId + 1 }, () => [] as number[]) : [];

  interface RobotRecord {
    id: number;
    currentKey: number;
    totalDistance: number;
    assignments: number[];
  }

  const robotRecords: RobotRecord[] = [];
  for (const robot of robotsInput) {
    if (!robot || typeof robot.id !== 'number' || robot.id < 0) {
      continue;
    }
    const id = robot.id;
    const xRaw = typeof robot.x === 'number' ? robot.x : 0;
    const yRaw = typeof robot.y === 'number' ? robot.y : 0;
    const x = clamp(Math.floor(xRaw), 0, width - 1);
    const y = clamp(Math.floor(yRaw), 0, height - 1);
    const key = y * width + x;
    passable[key] = 1;
    ensureAssignmentSlot(assignments, id);
    robotRecords.push({
      id,
      currentKey: key,
      totalDistance: 0,
      assignments: assignments[id],
    });
  }

  if (robotRecords.length === 0) {
    return assignments;
  }

  const jobsInput: any[] = Array.isArray(data.jobs) ? data.jobs : [];
  const jobCount = jobsInput.length;
  interface JobRecord {
    id: number;
    priority: number;
    key: number;
    index: number;
  }
  const jobList: JobRecord[] = [];
  const jobKeys: number[] = new Array(jobCount);
  const targetMask = new Uint8Array(totalCells);
  let uniqueTargetCount = 0;

  for (let i = 0; i < jobCount; i++) {
    const job = jobsInput[i] || {};
    const id = typeof job.id === 'number' ? job.id : i;
    const xRaw = typeof job.x === 'number' ? job.x : 0;
    const yRaw = typeof job.y === 'number' ? job.y : 0;
    const x = clamp(Math.floor(xRaw), 0, width - 1);
    const y = clamp(Math.floor(yRaw), 0, height - 1);
    const key = y * width + x;
    passable[key] = 1;
    if (targetMask[key] === 0) {
      targetMask[key] = 1;
      uniqueTargetCount++;
    }
    const priority = typeof job.priority === 'number' ? job.priority : 0;
    jobKeys[i] = key;
    jobList.push({ id, priority, key, index: i });
  }

  if (jobList.length === 0) {
    return assignments;
  }

  robotRecords.sort((a, b) => a.id - b.id);

  const distCache = new Map<number, Int32Array>();
  const workingDist = new Int32Array(totalCells);
  const workingQueue = new Int32Array(totalCells);
  const fallbackDistance = Math.max(1, width * height * 4 + 10);

  function getDistancesFrom(sourceKey: number): Int32Array {
    const cached = distCache.get(sourceKey);
    if (cached) {
      return cached;
    }

    workingDist.fill(-1);
    let head = 0;
    let tail = 0;
    workingDist[sourceKey] = 0;
    workingQueue[tail++] = sourceKey;
    let remainingTargets = uniqueTargetCount;

    while (head < tail && remainingTargets > 0) {
      const idx = workingQueue[head++];
      const nextDist = workingDist[idx] + 1;

      if (targetMask[idx]) {
        remainingTargets--;
        if (remainingTargets === 0) {
          // Distances to all job cells determined.
          // No need to expand further neighbors.
          break;
        }
      }

      const x = idx % width;
      const y = (idx / width) | 0;

      if (x > 0) {
        const neighbor = idx - 1;
        if (passable[neighbor] && workingDist[neighbor] === -1) {
          workingDist[neighbor] = nextDist;
          workingQueue[tail++] = neighbor;
        }
      }
      if (x + 1 < width) {
        const neighbor = idx + 1;
        if (passable[neighbor] && workingDist[neighbor] === -1) {
          workingDist[neighbor] = nextDist;
          workingQueue[tail++] = neighbor;
        }
      }
      if (y > 0) {
        const neighbor = idx - width;
        if (passable[neighbor] && workingDist[neighbor] === -1) {
          workingDist[neighbor] = nextDist;
          workingQueue[tail++] = neighbor;
        }
      }
      if (y + 1 < height) {
        const neighbor = idx + width;
        if (passable[neighbor] && workingDist[neighbor] === -1) {
          workingDist[neighbor] = nextDist;
          workingQueue[tail++] = neighbor;
        }
      }
    }

    const distances = new Int32Array(jobCount);
    for (let i = 0; i < jobCount; i++) {
      distances[i] = workingDist[jobKeys[i]];
    }
    distCache.set(sourceKey, distances);
    return distances;
  }

  const sortedJobs = jobList.slice().sort((a, b) => {
    if (b.priority !== a.priority) {
      return b.priority - a.priority;
    }
    if (a.id !== b.id) {
      return a.id - b.id;
    }
    return a.index - b.index;
  });

  for (const job of sortedJobs) {
    let bestRobot: RobotRecord | null = null;
    let bestScore = Number.POSITIVE_INFINITY;
    let bestDistValue = Number.POSITIVE_INFINITY;

    for (const robot of robotRecords) {
      const distances = getDistancesFrom(robot.currentKey);
      const rawDist = distances[job.index];
      const distValue = rawDist >= 0 ? rawDist : fallbackDistance;
      const score = robot.totalDistance + distValue;

      if (
        !bestRobot ||
        score < bestScore ||
        (score === bestScore &&
          (distValue < bestDistValue ||
            (distValue === bestDistValue && robot.id < bestRobot.id)))
      ) {
        bestRobot = robot;
        bestScore = score;
        bestDistValue = distValue;
      }
    }

    if (bestRobot) {
      bestRobot.assignments.push(job.id);
      bestRobot.totalDistance += bestDistValue;
      bestRobot.currentKey = job.key;
    }
  }

  return assignments;
}

function clamp(value: number, min: number, max: number): number {
  if (max < min) {
    return min;
  }
  if (value < min) {
    return min;
  }
  if (value > max) {
    return max;
  }
  return value;
}

function ensureAssignmentSlot(assignments: number[][], id: number): void {
  if (id < 0) {
    return;
  }
  while (assignments.length <= id) {
    assignments.push([]);
  }
  if (!assignments[id]) {
    assignments[id] = [];
  }
}
