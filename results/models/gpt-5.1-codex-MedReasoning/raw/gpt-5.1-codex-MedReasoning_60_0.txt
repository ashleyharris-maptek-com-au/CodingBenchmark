class BinaryHeap {
  constructor(compare) {
    this.data = [];
    this.compare = compare;
  }
  size() {
    return this.data.length;
  }
  isEmpty() {
    return this.data.length === 0;
  2}
  peek() {
    return this.data.length ? this.data[0] : null;
  }
  push(value) {
    const heap = this.data;
    heap.push(value);
    this._siftUp(heap.length - 1);
  }
  pop() {
    const heap = this.data;
    if (heap.length === 0) return null;
    const top = heap[0];
    const last = heap.pop();
    if (heap.length > 0) {
      heap[0] = last;
      this._siftDown(0);
    }
    return top;
  }
  _siftUp(index) {
    const heap = this.data;
    const compare = this.compare;
    while (index > 0) {
      const parent = (index - 1) >> 1;
      if (!compare(heap[index], heap[parent])) break;
      [heap[index], heap[parent]] = [heap[parent], heap[index]];
      index = parent;
    }
  }
  _siftDown(index) {
    const heap = this.data;
    const compare = this.compare;
    const length = heap.length;
    while (true) {
      const left = index * 2 + 1;
      if (left >= length) break;
      let best = left;
      const right = left + 1;
      if (right < length && compare(heap[right], heap[left])) {
        best = right;
      }
      if (!compare(heap[best], heap[index])) break;
      [heap[index], heap[best]] = [heap[best], heap[index]];
      index = best;
    }
  }
}

(function main() {
  let input = '';
  try {
    input = fs.readFileSync(0, 'utf8');
  } catch (err) {
    input = '';
  }
  if (!input.trim()) {
    console.log(JSON.stringify({ schedule: [] }));
    return;
  }
  let data;
  try {
    data = JSON.parse(input);
  } catch (err) {
    console.log(JSON.stringify({ schedule: [] }));
    return;
  }

  let machineCount = Number(data.machineCount);
  if (!Number.isFinite(machineCount) || machineCount < 1) machineCount = 1;
  machineCount = Math.floor(machineCount);
  if (machineCount < 1) machineCount = 1;

  const tasks = Array.isArray(data.tasks) ? data.tasks : [];
  const n = tasks.length;
  if (n === 0) {
    console.log(JSON.stringify({ schedule: [] }));
    return;
  }

  const idToIndex = new Map();
  const taskIds = new Array(n);
  for (let i = 0; i < n; i++) {
    let tid = tasks[i].id;
    if (tid === undefined || tid === null) {
      tid = i;
    }
    taskIds[i] = tid;
    idToIndex.set(tid, i);
  }
  const naturalOrder = taskIds.slice();

  const successors = Array.from({ length: n }, () => []);
  const indegree = new Array(n).fill(0);

  for (let i = 0; i < n; i++) {
    const deps = Array.isArray(tasks[i].deps) ? tasks[i].deps : [];
    for (let j = 0; j < deps.length; j++) {
      const depIdx = idToIndex.get(deps[j]);
      if (depIdx === undefined) continue;
      successors[depIdx].push(i);
      indegree[i]++;
    }
  }

  const indegreeCopy = indegree.slice();
  const queue = new Array(n);
  let qHead = 0;
  let qTail = 0;
  for (let i = 0; i < n; i++) {
    if (indegreeCopy[i] === 0) queue[qTail++] = i;
  }
  const topoOrder = [];
  while (qHead < qTail) {
    const node = queue[qHead++];
    topoOrder.push(node);
    const succList = successors[node];
    for (let j = 0; j < succList.length; j++) {
      const succ = succList[j];
      indegreeCopy[succ]--;
      if (indegreeCopy[succ] === 0) queue[qTail++] = succ;
    }
  }
  if (topoOrder.length !== n) {
    console.log(JSON.stringify({ order: naturalOrder }));
    return;
  }
  const topoIds = topoOrder.map((idx) => taskIds[idx]);

  const durations = new Array(n);
  const deadlines = new Array(n);
  const weights = new Array(n);
  for (let i = 0; i < n; i++) {
    let dVal = Number(tasks[i].duration);
    if (!Number.isFinite(dVal)) {
      dVal = 1;
    } else if (dVal < 0) {
      dVal = 0;
    }
    durations[i] = dVal;

    const deadlineRaw = tasks[i].deadline;
    let deadlineNum;
    if (deadlineRaw === undefined || deadlineRaw === null) {
      deadlineNum = Infinity;
    } else {
      deadlineNum = Number(deadlineRaw);
      if (!Number.isFinite(deadlineNum)) deadlineNum = Infinity;
    }
    deadlines[i] = deadlineNum;

    let wVal = Number(tasks[i].weight);
    if (!Number.isFinite(wVal)) wVal = 1;
    if (wVal < 0) wVal = 0;
    weights[i] = wVal;
  }
  const idKeys = taskIds.map((id) => String(id));

  const upward = new Array(n).fill(0);
  for (let idx = topoOrder.length - 1; idx >= 0; idx--) {
    const node = topoOrder[idx];
    let best = 0;
    const succList = successors[node];
    for (let j = 0; j < succList.length; j++) {
      const succ = succList[j];
      if (upward[succ] > best) best = upward[succ];
    }
    upward[node] = durations[node] + best;
  }

  const readyHeap = new BinaryHeap((a, b) => {
    if (a.slack !== b.slack) return a.slack < b.slack;
    if (a.weightDensity !== b.weightDensity) return a.weightDensity > b.weightDensity;
    if (a.upward !== b.upward) return a.upward > b.upward;
    if (a.readyTime !== b.readyTime) return a.readyTime < b.readyTime;
    if (a.duration !== b.duration) return a.duration < b.duration;
    return a.idKey < b.idKey;
  });

  const idleMachines = new BinaryHeap((a, b) => {
    if (a.availableTime !== b.availableTime) return a.availableTime < b.availableTime;
    return a.machine < b.machine;
  });

  const eventHeap = new BinaryHeap((a, b) => {
    if (a.time !== b.time) return a.time < b.time;
    if (a.machine !== b.machine) return a.machine < b.machine;
    return a.taskIdx < b.taskIdx;
  });

  const remainingDeps = indegree.slice();
  const depReadyTime = new Array(n).fill(0);
  const schedule = [];

  function pushReady(idx, readyTime) {
    const duration = durations[idx];
    const deadline = deadlines[idx];
    let slack = Infinity;
    if (Number.isFinite(deadline)) {
      slack = deadline - (readyTime + duration);
      if (!Number.isFinite(slack)) slack = Infinity;
    }
    let density;
    if (duration > 0) {
      density = weights[idx] / duration;
    } else {
      density = weights[idx] > 0 ? Number.POSITIVE_INFINITY : 0;
    }
    readyHeap.push({
      idx,
      readyTime,
      slack,
      weightDensity: density,
      upward: upward[idx],
      duration,
      idKey: idKeys[idx],
    });
  }

  function trySchedule() {
    while (!readyHeap.isEmpty() && !idleMachines.isEmpty()) {
      const machine = idleMachines.pop();
      const taskEntry = readyHeap.pop();
      const startTime = Math.max(machine.availableTime, taskEntry.readyTime);
      const finishTime = startTime + taskEntry.duration;
      schedule.push({
        id: taskIds[taskEntry.idx],
        start: startTime,
        machine: machine.machine,
      });
      eventHeap.push({
        time: finishTime,
        machine: machine.machine,
        taskIdx: taskEntry.idx,
      });
    }
  }

  for (let m = 0; m < machineCount; m++) {
    idleMachines.push({ machine: m, availableTime: 0 });
  }
  for (let i = 0; i < n; i++) {
    if (remainingDeps[i] === 0) {
      pushReady(i, 0);
    }
  }

  trySchedule();

  let completed = 0;
  while (completed < n) {
    const event = eventHeap.pop();
    if (!event) {
      console.log(JSON.stringify({ order: topoIds }));
      return;
    }
    const finishTime = event.time;
    const finishedIdx = event.taskIdx;
    completed++;
    idleMachines.push({ machine: event.machine, availableTime: finishTime });

    const succList = successors[finishedIdx];
    for (let j = 0; j < succList.length; j++) {
      const succIdx = succList[j];
      if (depReadyTime[succIdx] < finishTime) {
        depReadyTime[succIdx] = finishTime;
      }
      remainingDeps[succIdx]--;
      if (remainingDeps[succIdx] === 0) {
        pushReady(succIdx, depReadyTime[succIdx]);
      }
    }

    trySchedule();
  }

  if (schedule.length !== n) {
    console.log(JSON.stringify({ order: topoIds }));
    return;
  }

  console.log(JSON.stringify({ schedule }));
})();
