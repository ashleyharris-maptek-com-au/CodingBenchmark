declare const require: any;
const fs = require('fs');

const input: string = fs.readFileSync(0, 'utf8');
const trimmed = input.trim();
const data = trimmed ? JSON.parse(trimmed) : {};

interface Task {
  id: string;
  duration: number;
  deps: string[];
  deadline?: number;
  weight: number;
}

interface ReadyEntry {
  taskId: string;
  priority: number;
  deadlineValue: number;
  weight: number;
  duration: number;
  readyTime: number;
  seq: number;
}

class BinaryHeap<T> {
  private data: T[] = [];
  constructor(private comparator: (a: T, b: T) => number) {}
  size(): number {
    return this.data.length;
  }
  push(item: T): void {
    this.data.push(item);
    this.bubbleUp(this.data.length - 1);
  }
  pop(): T | undefined {
    if (this.data.length === 0) {
      return undefined;
    }
    const top = this.data[0];
    const last = this.data.pop() as T;
    if (this.data.length > 0) {
      this.data[0] = last;
      this.bubbleDown(0);
    }
    return top;
  }
  private bubbleUp(index: number): void {
    while (index > 0) {
      const parent = (index - 1) >> 1;
      if (this.comparator(this.data[index], this.data[parent]) < 0) {
        [this.data[index], this.data[parent]] = [this.data[parent], this.data[index]];
        index = parent;
      } else {
        break;
      }
    }
  }
  private bubbleDown(index: number): void {
    const length = this.data.length;
    while (true) {
      let smallest = index;
      const left = (index << 1) + 1;
      const right = left + 1;
      if (left < length && this.comparator(this.data[left], this.data[smallest]) < 0) {
        smallest = left;
      }
      if (right < length && this.comparator(this.data[right], this.data[smallest]) < 0) {
        smallest = right;
      }
      if (smallest !== index) {
        [this.data[index], this.data[smallest]] = [this.data[smallest], this.data[index]];
        index = smallest;
      } else {
        break;
      }
    }
  }
}

function computeTopologicalOrder(
  tasksList: Task[],
  adjacency: Map<string, string[]>,
  indegreeTemplate: Map<string, number>
): string[] {
  const indegreeCopy = new Map<string, number>();
  indegreeTemplate.forEach((value, key) => indegreeCopy.set(key, value));
  const heap = new BinaryHeap<string>((a, b) => (a < b ? -1 : a > b ? 1 : 0));
  indegreeCopy.forEach((deg, id) => {
    if (deg === 0) {
      heap.push(id);
    }
  });
  const order: string[] = [];
  while (heap.size() > 0) {
    const current = heap.pop() as string;
    order.push(current);
    const neighbors = adjacency.get(current) || [];
    for (const next of neighbors) {
      const newDeg = (indegreeCopy.get(next) ?? 0) - 1;
      indegreeCopy.set(next, newDeg);
      if (newDeg === 0) {
        heap.push(next);
      }
    }
  }
  if (order.length < tasksList.length) {
    const seen = new Set(order);
    for (const task of tasksList) {
      if (!seen.has(task.id)) {
        order.push(task.id);
        seen.add(task.id);
      }
    }
  }
  return order;
}

const machineCount = Math.max(1, Math.floor(Number(data.machineCount) || 1));
const rawTasks: any[] = Array.isArray(data.tasks) ? data.tasks : [];

const tasksList: Task[] = [];
const tasksMap = new Map<string, Task>();

for (let i = 0; i < rawTasks.length; i++) {
  const raw = rawTasks[i] ?? {};
  let idValue = raw.id;
  if (idValue === undefined || idValue === null || idValue === '') {
    idValue = `task_${i}`;
  }
  const id = String(idValue);
  const durationRaw = Number(raw.duration);
  const duration = Math.max(0, Number.isFinite(durationRaw) ? Math.round(durationRaw) : 1);
  const depsRaw = Array.isArray(raw.deps) ? raw.deps : [];
  const deps = depsRaw.map((d: any) => String(d));
  const deadlineRaw = Number(raw.deadline);
  const deadline = Number.isFinite(deadlineRaw) ? deadlineRaw : undefined;
  const weightRaw = Number(raw.weight);
  const weight = Number.isFinite(weightRaw) ? Math.max(0, weightRaw) : 1;
  const task: Task = { id, duration, deps, deadline, weight };
  tasksList.push(task);
  tasksMap.set(id, task);
}

if (tasksList.length === 0) {
  console.log(JSON.stringify({ schedule: [] }));
  process.exit(0);
}

const adjacency = new Map<string, string[]>();
const readyTimeMap = new Map<string, number>();
for (const task of tasksList) {
  adjacency.set(task.id, []);
  readyTimeMap.set(task.id, 0);
}

let maxDeadline = -Infinity;
for (const task of tasksList) {
  if (typeof task.deadline === 'number' && isFinite(task.deadline)) {
    if (task.deadline > maxDeadline) {
      maxDeadline = task.deadline;
    }
  }
}
const deadlineFallbackBase = Number.isFinite(maxDeadline) ? maxDeadline : 0;

const indegree = new Map<string, number>();
const indegreeInitial = new Map<string, number>();

for (const task of tasksList) {
  const seenDeps = new Set<string>();
  const filtered: string[] = [];
  for (const dep of task.deps) {
    if (dep === task.id) {
      continue;
    }
    if (!tasksMap.has(dep)) {
      continue;
    }
    if (seenDeps.has(dep)) {
      continue;
       }
    seenDeps.add(dep);
    filtered.push(dep);
    adjacency.get(dep)?.push(task.id);
  }
  task.deps = filtered;
  const deg = filtered.length;
  indegree.set(task.id, deg);
  indegreeInitial.set(task.id, deg);
}

let readySeq = 0;
const readyHeap = new BinaryHeap<ReadyEntry>((a, b) => {
  if (a.priority !== b.priority) {
    return a.priority - b.priority;
  }
  if (a.deadlineValue !== b.deadlineValue) {
    return a.deadlineValue - b.deadlineValue;
  }
  if (a.weight !== b.weight) {
    return b.weight - a.weight;
  }
  if (a.duration !== b.duration) {
    return a.duration - b.duration;
  }
  if (a.readyTime !== b.readyTime) {
    return a.readyTime - b.readyTime;
  }
  return a.seq - b.seq;
});

const createReadyEntry = (taskId: string): ReadyEntry => {
  const task = tasksMap.get(taskId) as Task;
  const readyTime = readyTimeMap.get(taskId) ?? 0;
  const duration = task.duration;
  const weight = task.weight;
  const rawDeadline = task.deadline;
  const effectiveDeadline = Number.isFinite(rawDeadline)
    ? (rawDeadline as number)
    : deadlineFallbackBase + readyTime + duration;
  const slack = effectiveDeadline - readyTime - duration;
  const slackAdjustment = slack < 0 ? slack * 5 : slack * 0.1;
  const priority = effectiveDeadline + readyTime * 0.25 + duration * 0.1 + slackAdjustment - weight * 0.1;
  return {
    taskId,
    priority,
    deadlineValue: effectiveDeadline,
    weight,
    duration,
    readyTime,
    seq: readySeq++,
  };
};

for (const task of tasksList) {
  if ((indegree.get(task.id) ?? 0) === 0) {
    readyHeap.push(createReadyEntry(task.id));
  }
}

const machineAvail = new Array<number>(machineCount).fill(0);
const scheduleRecords: { id: string; start: number; machine: number }[] = [];
const totalTasks = tasksList.length;
let scheduledCount = 0;

while (scheduledCount < totalTasks && readyHeap.size() > 0) {
  const entry = readyHeap.pop() as ReadyEntry;
  const task = tasksMap.get(entry.taskId);
  if (!task) {
    continue;
  }
  const readyTime = readyTimeMap.get(task.id) ?? 0;
  let bestMachine = 0;
  let bestStart = Number.POSITIVE_INFINITY;
  for (let m = 0; m < machineCount; m++) {
    const start = Math.max(machineAvail[m], readyTime);
    if (
      start < bestStart ||
      (start === bestStart &&
        (machineAvail[m] < machineAvail[bestMachine] ||
          (machineAvail[m] === machineAvail[bestMachine] && m < bestMachine)))
    ) {
      bestStart = start;
      bestMachine = m;
    }
  }
  const startTime = Math.max(machineAvail[bestMachine], readyTime);
  const finishTime = startTime + task.duration;
  scheduleRecords.push({ id: task.id, start: startTime, machine: bestMachine });
  machineAvail[bestMachine] = finishTime;
  scheduledCount += 1;

  const dependents = adjacency.get(task.id) || [];
  for (const childId of dependents) {
    const prevReady = readyTimeMap.get(childId) ?? 0;
    if (finishTime > prevReady) {
      readyTimeMap.set(childId, finishTime);
    }
    const newIndegree = (indegree.get(childId) ?? 0) - 1;
    indegree.set(childId, newIndegree);
    if (newIndegree === 0) {
      readyHeap.push(createReadyEntry(childId));
    }
  }
}

if (scheduledCount === totalTasks) {
  scheduleRecords.sort((a, b) => {
    if (a.start !== b.start) {
      return a.start - b.start;
    }
    if (a.machine !== b.machine) {
      return a.machine - b.machine;
    }
    return a.id.localeCompare(b.id);
  });
  console.log(JSON.stringify({ schedule: scheduleRecords }));
} else {
  const fallbackOrder = computeTopologicalOrder(tasksList, adjacency, indegreeInitial);
  console.log(JSON.stringify({ order: fallbackOrder }));
}
