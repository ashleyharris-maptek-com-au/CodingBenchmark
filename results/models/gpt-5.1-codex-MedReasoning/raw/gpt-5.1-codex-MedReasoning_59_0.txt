```javascript
const fs = require('fs');

const input = fs.readFileSync(0, 'utf8');
const data = input.length ? JSON.parse(input) : {};

const rawGrid = Array.isArray(data.probability) ? data.probability : [];
const drones = Array.isArray(data.drones) ? data.drones : [];

let maxSteps = Number(data.maxStepsPerDrone);
if (!Number.isFinite(maxSteps) || maxSteps < 0) {
  maxSteps = 0;
} else {
  maxSteps = Math.floor(maxSteps);
}
const MAX_INT_STEPS = 2147483646;
if (maxSteps > MAX_INT_STEPS) {
  maxSteps = MAX_INT_STEPS;
}

const height = rawGrid.length;
const width = rawGrid.reduce((max, row) => {
  if (Array.isArray(row) && row.length > max) {
    return row.length;
  }
  return max;
}, 0);

let paths;

if (height === 0 || width === 0) {
  paths = drones.map((drone) => {
    const obj = typeof drone === 'object' && drone !== null ? drone : {};
    const sx = Number.isFinite(obj.x) ? Math.round(obj.x) : 0;
    const sy = Number.isFinite(obj.y) ? Math.round(obj.y) : 0;
    const path = [[sx, sy]];
    for (let step = 0; step < maxSteps; step++) {
      path.push([sx, sy]);
    }
    return path;
  });
} else {
  const totalCells = height * width;
  const gridValues = new Float64Array(totalCells);
  gridValues.fill(-1);

  for (let y = 0; y < height; y++) {
    const row = Array.isArray(rawGrid[y]) ? rawGrid[y] : [];
    const limit = row.length;
    for (let x = 0; x < limit; x++) {
      const rawVal = row[x];
      if (typeof rawVal === 'number' && Number.isFinite(rawVal)) {
        gridValues[y * width + x] = rawVal;
      }
    }
  }

  const assigned = new Uint8Array(totalCells);
  let remainingValuable = 0;
  let firstPassableIdx = -1;

  for (let idx = 0; idx < totalCells; idx++) {
    const value = gridValues[idx];
    if (value >= 0 && firstPassableIdx === -1) {
      firstPassableIdx = idx;
    }
    if (value > 0) {
      remainingValuable++;
    }
  }

  const distArr = new Int32Array(totalCells);
  const prevArr = new Int32Array(totalCells);
  const visitedMark = new Int32Array(totalCells);
  const queue = new Int32Array(totalCells);
  let visitToken = 0;

  function bfsLimit(startIdx, limit) {
    visitToken++;
    if (visitToken === 2147483647) {
      visitedMark.fill(0);
      visitToken = 1;
    }
    const mark = visitToken;
    visitedMark[startIdx] = mark;
    distArr[startIdx] = 0;
    prevArr[startIdx] = -1;

    let head = 0;
    let tail = 0;
    queue[tail++] = startIdx;

    while (head < tail) {
      const idx = queue[head++];
      const baseDist = distArr[idx];
      if (baseDist >= limit) {
        continue;
      }

      const y = (idx / width) | 0;
      const x = idx - y * width;

      let nIdx;
      if (x + 1 < width) {
        nIdx = idx + 1;
        if (gridValues[nIdx] >= 0 && visitedMark[nIdx] !== mark) {
          visitedMark[nIdx] = mark;
          distArr[nIdx] = baseDist + 1;
          prevArr[nIdx] = idx;
          queue[tail++] = nIdx;
        }
      }
      if (x - 1 >= 0) {
        nIdx = idx - 1;
        if (gridValues[nIdx] >= 0 && visitedMark[nIdx] !== mark) {
          visitedMark[nIdx] = mark;
          distArr[nIdx] = baseDist + 1;
          prevArr[nIdx] = idx;
          queue[tail++] = nIdx;
        }
      }
      if (y + 1 < height) {
        nIdx = idx + width;
        if (gridValues[nIdx] >= 0 && visitedMark[nIdx] !== mark) {
          visitedMark[nIdx] = mark;
          distArr[nIdx] = baseDist + 1;
          prevArr[nIdx] = idx;
          queue[tail++] = nIdx;
        }
      }
      if (y - 1 >= 0) {
        nIdx = idx - width;
        if (gridValues[nIdx] >= 0 && visitedMark[nIdx] !== mark) {
          visitedMark[nIdx] = mark;
          distArr[nIdx] = baseDist + 1;
          prevArr[nIdx] = idx;
          queue[tail++] = nIdx;
        }
      }
    }

    return tail;
  }

  function buildPathIndices(targetIdx) {
    const sequence = [];
    let current = targetIdx;
    while (true) {
      const prev = prevArr[current];
      if (prev === -1) {
        break;
      }
      sequence.push(current);
      current = prev;
    }
    sequence.reverse();
    return sequence;
  }

  function markIfValuableIdx(idx) {
    if (idx < 0) {
      return;
    }
    if (gridValues[idx] > 0 && assigned[idx] === 0) {
      assigned[idx] = 1;
      remainingValuable--;
    }
  }

  function normalizeStartCoords(drone) {
    const obj = typeof drone === 'object' && drone !== null ? drone : {};
    let nx = Number.isFinite(obj.x) ? Math.round(obj.x) : 0;
    let ny = Number.isFinite(obj.y) ? Math.round(obj.y) : 0;
    if (nx < 0) nx = 0;
    else if (nx >= width) nx = width - 1;
    if (ny < 0) ny = 0;
    else if (ny >= height) ny = height - 1;
    let idx = ny * width + nx;
    if (gridValues[idx] >= 0) {
      return { x: nx, y: ny, idx };
    }
    if (firstPassableIdx !== -1) {
      return {
        x: firstPassableIdx % width,
        y: (firstPassableIdx / width) | 0,
        idx: firstPassableIdx,
      };
    }
    return { x: nx, y: ny, idx };
  }

  const resultPaths = [];

  for (const drone of drones) {
    const start = normalizeStartCoords(drone);
    let currentIdx = start.idx;
    let currentX = start.x;
    let currentY = start.y;

    const path = [[currentX, currentY]];
    let stepsUsed = 0;

    markIfValuableIdx(currentIdx);

    while (stepsUsed < maxSteps && remainingValuable > 0) {
      const remaining = maxSteps - stepsUsed;
      const visitedCount = bfsLimit(currentIdx, remaining);

      let bestIdx = -1;
      let bestMetric = -Infinity;
      let bestValue = -Infinity;
      let bestDistance = Infinity;

      for (let v = 0; v < visitedCount; v++) {
        const cellIdx = queue[v];
        if (assigned[cellIdx]) continue;
        const cellValue = gridValues[cellIdx];
        if (cellValue <= 0) continue;
        const distance = distArr[cellIdx];
        if (distance > remaining) continue;
        const metric = cellValue / (distance + 1);
        if (
          metric > bestMetric ||
          (metric === bestMetric &&
            (cellValue > bestValue ||
              (cellValue === bestValue && distance < bestDistance)))
        ) {
          bestMetric = metric;
          bestValue = cellValue;
          bestDistance = distance;
          bestIdx = cellIdx;
        }
      }

      if (bestIdx === -1) {
        break;
      }

      const pathIndices = buildPathIndices(bestIdx);
      if (pathIndices.length === 0 && currentIdx === bestIdx) {
        markIfValuableIdx(bestIdx);
        break;
      }

      for (let s = 0; s < pathIndices.length && stepsUsed < maxSteps; s++) {
        const stepIdx = pathIndices[s];
        const x = stepIdx % width;
        const y = (stepIdx / width) | 0;
        path.push([x, y]);
        stepsUsed++;
        currentIdx = stepIdx;
        currentX = x;
        currentY = y;
        markIfValuableIdx(stepIdx);
      }
    }

    while (stepsUsed < maxSteps) {
      path.push([currentX, currentY]);
      stepsUsed++;
    }

    resultPaths.push(path);
  }

  paths = resultPaths;
}

console.log(JSON.stringify({ paths }));
```
