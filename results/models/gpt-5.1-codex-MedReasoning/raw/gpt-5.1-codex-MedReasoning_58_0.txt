"use strict";

const fs = require('fs');

const EPS = 1e-6;
const INF = 1e12;
const BIG_DISTANCE = 1e9;

const rawInput = fs.readFileSync(0, 'utf8');
const trimmed = rawInput.trim();
if (!trimmed) {
    console.log('{"assignments":[]}');
    process.exit(0);
}
const data = JSON.parse(trimmed);

const grid = Array.isArray(data.grid) ? data.grid : [];
const robotsInput = Array.isArray(data.robots) ? data.robots : [];
const jobsInput = Array.isArray(data.jobs) ? data.jobs : [];

const height = grid.length;
const width = height ? grid[0].length : 0;
const cellCount = width * height;

if (cellCount === 0) {
    const assignments = buildFallbackAssignments(robotsInput, jobsInput);
    console.log(JSON.stringify({ assignments }));
    process.exit(0);
}

const robotsNormalized = [];
for (const robot of robotsInput) {
    if (!robot) continue;
    const rid = normalizeRobotId(robot.id);
    if (rid < 0) continue;
    const rx = clampCoordinate(robot.x, width);
    const ry = clampCoordinate(robot.y, height);
    robotsNormalized.push({ id: rid, x: rx, y: ry });
}
if (!robotsNormalized.length) {
    console.log(JSON.stringify({ assignments: [] }));
    process.exit(0);
}

const jobsList = jobsInput;
const jobCount = jobsList.length;
const jobIds = jobsList.map(job => (job ? job.id : null));
const jobPriorities = jobsList.map(job => toPriority(job ? job.priority : 0));
const jobPointIndex = new Array(jobCount);

const walkable = new Uint8Array(cellCount);
for (let y = 0; y < height; y++) {
    const row = grid[y] || '';
    for (let x = 0; x < width; x++) {
        const char = row[x] || '.';
        walkable[y * width + x] = char === '#' ? 0 : 1;
    }
}

const points = [];
const robotPointIndexById = new Map();
for (const robot of robotsNormalized) {
    const idx = points.length;
    points.push({ x: robot.x, y: robot.y });
    robotPointIndexById.set(robot.id, idx);
}
for (let i = 0; i < jobCount; i++) {
    const job = jobsList[i] || { x: 0, y: 0 };
    const jx = clampCoordinate(job.x, width);
    const jy = clampCoordinate(job.y, height);
    const idx = points.length;
    points.push({ x: jx, y: jy });
    jobPointIndex[i] = idx;
}

const distMatrix = computePairDistances(points, width, height, walkable);

const robotsSorted = robotsNormalized.slice().sort((a, b) => a.id - b.id);
const robotStates = robotsSorted
    .map(robot => ({
        id: robot.id,
        startPoint: robotPointIndexById.get(robot.id),
        jobs: []
    }))
    .filter(state => state.startPoint !== undefined);

if (!robotStates.length) {
    console.log(JSON.stringify({ assignments: [] }));
    process.exit(0);
}

const jobOrder = jobCount ? Array.from({ length: jobCount }, (_, i) => i).sort(jobIndexComparator) : [];

const expectedLoad = robotStates.length ? jobCount / robotStates.length : 0;
const loadWeight = Math.min(2, Math.max(0.1, expectedLoad * 0.05));

for (const jobIdx of jobOrder) {
    let bestState = null;
    let bestScore = Infinity;
    for (const state of robotStates) {
        const dist = distMatrix[state.startPoint][jobPointIndex[jobIdx]];
        const baseCost = dist >= 0 ? dist : BIG_DISTANCE;
        const score = baseCost + loadWeight * state.jobs.length;
        if (
            !bestState ||
            score + EPS < bestScore ||
            (Math.abs(score - bestScore) <= EPS && state.id < bestState.id)
        ) {
            bestScore = score;
            bestState = state;
        }
    }
    if (bestState) {
        bestState.jobs.push(jobIdx);
    }
}

for (const state of robotStates) {
    state.jobs = buildNearestRoute(state.jobs, state.startPoint);
    optimizeRouteOrder(state, 3);
}

if (jobCount > 0 && robotStates.length > 1) {
    const maxMoves = Math.max(jobCount * 2, robotStates.length * 2);
    relocateJobs(robotStates, maxMoves);
}

ensureAllJobsCovered(robotStates);

for (const state of robotStates) {
    optimizeRouteOrder(state, 3);
}

let maxRobotId = -1;
for (const robot of robotsSorted) {
    if (robot.id > maxRobotId) maxRobotId = robot.id;
}
const assignments = Array.from({ length: maxRobotId + 1 }, () => []);
const stateById = new Map();
for (const state of robotStates) stateById.set(state.id, state);

for (let id = 0; id <= maxRobotId; id++) {
    const state = stateById.get(id);
    if (state) {
        assignments[id] = state.jobs.map(idx => jobIds[idx]);
    } else {
        assignments[id] = [];
    }
}

console.log(JSON.stringify({ assignments }));

function normalizeRobotId(value) {
    if (typeof value === 'number' && Number.isInteger(value) && value >= 0) return value;
    if (typeof value === 'string') {
        const trimmed = value.trim();
        if (trimmed === '') return -1;
        const num = Number(trimmed);
        if (Number.isInteger(num) && num >= 0) return num;
    }
    return -1;
}

function toPriority(value) {
    if (typeof value === 'number' && Number.isFinite(value)) return value;
    const num = Number(value);
    return Number.isFinite(num) ? num : 0;
}

function clampCoordinate(value, limit) {
    if (limit <= 0) return 0;
    const num = Number(value);
    if (!Number.isFinite(num)) return 0;
    const rounded = Math.round(num);
    if (rounded < 0) return 0;
    if (rounded >= limit) return limit - 1;
    return rounded;
}

function buildFallbackAssignments(robots, jobs) {
    const validRobots = [];
    let maxId = -1;
    for (const robot of robots) {
        if (!robot) continue;
        const rid = normalizeRobotId(robot.id);
        if (rid < 0) continue;
        validRobots.push({ id: rid });
        if (rid > maxId) maxId = rid;
    }
    if (maxId < 0) return [];
    const assignments = Array.from({ length: maxId + 1 }, () => []);
    if (!validRobots.length) return assignments;
    validRobots.sort((a, b) => a.id - b.id);
    for (let i = 0; i < jobs.length; i++) {
        const robot = validRobots[i % validRobots.length];
        assignments[robot.id].push(jobs[i] ? jobs[i].id : null);
    }
    return assignments;
}

function computePairDistances(pointsArr, w, h, walkableArr) {
    const count = pointsArr.length;
    const matrix = Array.from({ length: count }, () => new Int32Array(count).fill(-1));
    for (let i = 0; i < count; i++) matrix[i][i] = 0;
    const cellPoints = Array(w * h);
    for (let i = 0; i < count; i++) {
        const cellIdx = pointsArr[i].y * w + pointsArr[i].x;
        if (!cellPoints[cellIdx]) cellPoints[cellIdx] = [];
        cellPoints[cellIdx].push(i);
    }
    const queue = new Int32Array(w * h);
    const cellDist = new Int32Array(w * h);
    for (let src = 0; src < count; src++) {
        cellDist.fill(-1);
        let head = 0;
        let tail = 0;
        const startCell = pointsArr[src].y * w + pointsArr[src].x;
        cellDist[startCell] = 0;
        queue[tail++] = startCell;
        let remaining = count - 1;
        while (head < tail && remaining > 0) {
            const cell = queue[head++];
            const distHere = cellDist[cell];
            const list = cellPoints[cell];
            if (list) {
                for (let idx = 0; idx < list.length; idx++) {
                    const dest = list[idx];
                    if (matrix[src][dest] === -1) {
                        matrix[src][dest] = distHere;
                        matrix[dest][src] = distHere;
                        if (dest !== src) remaining--;
                    }
                }
            }
            const y = (cell / w) | 0;
            const x = cell - y * w;
            const nextDist = distHere + 1;

            if (y > 0) {
                const up = cell - w;
                if (walkableArr[up] && cellDist[up] === -1) {
                    cellDist[up] = nextDist;
                    queue[tail++] = up;
                }
            }
            if (y + 1 < h) {
                const down = cell + w;
                if (walkableArr[down] && cellDist[down] === -1) {
                    cellDist[down] = nextDist;
                    queue[tail++] = down;
                }
            }
            if (x > 0) {
                const left = cell - 1;
                if (walkableArr[left] && cellDist[left] === -1) {
                    cellDist[left] = nextDist;
                    queue[tail++] = left;
                }
            }
            if (x + 1 < w) {
                const right = cell + 1;
                if (walkableArr[right] && cellDist[right] === -1) {
                    cellDist[right] = nextDist;
                    queue[tail++] = right;
                }
            }
        }
    }
    return matrix;
}

function jobIndexComparator(a, b) {
    const pa = jobPriorities[a];
    const pb = jobPriorities[b];
    if (pa !== pb) return pb - pa;
    return compareJobIds(a, b);
}

function compareJobIds(indexA, indexB) {
    const sa = String(jobIds[indexA]);
    const sb = String(jobIds[indexB]);
    if (sa === sb) return 0;
    return sa < sb ? -1 : 1;
}

function buildNearestRoute(jobList, startPoint) {
    if (!jobList || !jobList.length) return [];
    const unique = [];
    const seen = new Set();
    for (const jobIdx of jobList) {
        if (!seen.has(jobIdx)) {
            seen.add(jobIdx);
            unique.push(jobIdx);
        }
    }
    const remaining = new Set(unique);
    const route = [];
    let currentPoint = startPoint;
    while (remaining.size) {
        let bestJob = null;
        let bestDist = INF;
        for (const jobIdx of remaining) {
            const point = jobPointIndex[jobIdx];
            const dist = distMatrix[currentPoint][point];
            const actual = dist >= 0 ? dist : INF;
            if (actual + EPS < bestDist) {
                bestDist = actual;
                bestJob = jobIdx;
            } else if (Math.abs(actual - bestDist) <= EPS && bestJob !== null) {
                const priA = jobPriorities[jobIdx];
                const priB = jobPriorities[bestJob];
                if (priA > priB) {
                    bestJob = jobIdx;
                } else if (priA === priB && compareJobIds(jobIdx, bestJob) < 0) {
                    bestJob = jobIdx;
                }
            }
        }
        if (bestJob === null || bestDist >= INF / 2) {
            const leftovers = Array.from(remaining).sort((a, b) => {
                const pa = jobPriorities[a];
                const pb = jobPriorities[b];
                if (pa !== pb) return pb - pa;
                return compareJobIds(a, b);
            });
            for (const jobIdx of leftovers) route.push(jobIdx);
            break;
        }
        route.push(bestJob);
        remaining.delete(bestJob);
        currentPoint = jobPointIndex[bestJob];
    }
    return route;
}

function optimizeRouteOrder(state, maxLoops) {
    twoOptOnRoute(state.jobs, state.startPoint, maxLoops);
}

function twoOptOnRoute(route, startPoint, maxLoops) {
    const n = route.length;
    if (n < 3) return;
    let loop = 0;
    while (loop < maxLoops) {
        let improved = false;
        for (let i = 0; i < n - 1; i++) {
            const prevPoint = i === 0 ? startPoint : jobPointIndex[route[i - 1]];
            const pointI = jobPointIndex[route[i]];
            const distPrevI = distMatrix[prevPoint][pointI];
            if (distPrevI < 0) continue;
            for (let k = i + 1; k < n; k++) {
                const pointK = jobPointIndex[route[k]];
                const nextPoint = k === n - 1 ? null : jobPointIndex[route[k + 1]];
                const distKNext = nextPoint !== null ? distMatrix[pointK][nextPoint] : 0;
                if (nextPoint !== null && distKNext < 0) continue;
                const distPrevK = distMatrix[prevPoint][pointK];
                if (distPrevK < 0) continue;
                const distINext = nextPoint !== null ? distMatrix[pointI][nextPoint] : 0;
                if (nextPoint !== null && distINext < 0) continue;
                const oldCost = distPrevI + (nextPoint !== null ? distKNext : 0);
                const newCost = distPrevK + (nextPoint !== null ? distINext : 0);
                if (newCost + EPS < oldCost) {
                    reverseSubarray(route, i, k);
                    improved = true;
                    break;
                }
            }
            if (improved) break;
        }
        if (!improved) break;
        loop++;
    }
}

function reverseSubarray(arr, start, end) {
    while (start < end) {
        const tmp = arr[start];
        arr[start] = arr[end];
        arr[end] = tmp;
        start++;
        end--;
    }
}

function bestInsertion(state, jobIdx) {
    const jobPoint = jobPointIndex[jobIdx];
    const route = state.jobs;
    if (route.length === 0) {
        const dist = distMatrix[state.startPoint][jobPoint];
        return { delta: dist >= 0 ? dist : INF, pos: 0 };
    }
    let bestDelta = INF;
    let bestPos = route.length;
    let prevPoint = state.startPoint;
    for (let pos = 0; pos <= route.length; pos++) {
        const nextPoint = pos < route.length ? jobPointIndex[route[pos]] : null;
        const prevCandidate = prevPoint;
        let valid = true;
        const distPrev = distMatrix[prevCandidate][jobPoint];
        if (distPrev < 0) valid = false;
        let distJobNext = 0;
        if (nextPoint !== null) {
            distJobNext = distMatrix[jobPoint][nextPoint];
            if (distJobNext < 0) valid = false;
        }
        let oldCost = 0;
        if (nextPoint !== null) {
            oldCost = distMatrix[prevCandidate][nextPoint];
            if (oldCost < 0) valid = false;
        }
        if (valid) {
            const newCost = distPrev + (nextPoint !== null ? distJobNext : 0);
            const delta = newCost - oldCost;
            if (delta + EPS < bestDelta || (Math.abs(delta - bestDelta) <= EPS && pos < bestPos)) {
                bestDelta = delta;
                bestPos = pos;
            }
        }
        if (nextPoint !== null) {
            prevPoint = nextPoint;
        }
    }
    return { delta: bestDelta, pos: bestPos };
}

function removalDelta(state, index) {
    const route = state.jobs;
    if (index < 0 || index >= route.length) return { delta: INF };
    const jobIdx = route[index];
    const jobPoint = jobPointIndex[jobIdx];
    const prevPoint = index === 0 ? state.startPoint : jobPointIndex[route[index - 1]];
    const nextPoint = index === route.length - 1 ? null : jobPointIndex[route[index + 1]];
    const distPrevCurr = distMatrix[prevPoint][jobPoint];
    if (distPrevCurr < 0) return { delta: INF };
    let distCurrNext = 0;
    if (nextPoint !== null) {
        distCurrNext = distMatrix[jobPoint][nextPoint];
        if (distCurrNext < 0) return { delta: INF };
    }
    let distPrevNext = 0;
    if (nextPoint !== null) {
        distPrevNext = distMatrix[prevPoint][nextPoint];
        if (distPrevNext < 0) return { delta: INF };
    }
    const oldCost = distPrevCurr + (nextPoint !== null ? distCurrNext : 0);
    const newCost = nextPoint !== null ? distPrevNext : 0;
    return { delta: newCost - oldCost };
}

function relocateJobs(states, maxMoves) {
    if (states.length < 2) return;
    let moves = 0;
    while (moves < maxMoves) {
        let moved = false;
        outer: for (let a = 0; a < states.length; a++) {
            const fromState = states[a];
            for (let idx = 0; idx < fromState.jobs.length; idx++) {
                const removal = removalDelta(fromState, idx);
                if (!Number.isFinite(removal.delta)) continue;
                const jobIdx = fromState.jobs[idx];
                for (let b = 0; b < states.length; b++) {
                    if (a === b) continue;
                    const toState = states[b];
                    const insertion = bestInsertion(toState, jobIdx);
                    if (!Number.isFinite(insertion.delta)) continue;
                    const gain = removal.delta + insertion.delta;
                    if (gain < -EPS) {
                        const [jobMoved] = fromState.jobs.splice(idx, 1);
                        toState.jobs.splice(insertion.pos, 0, jobMoved);
                        optimizeRouteOrder(fromState, 1);
                        optimizeRouteOrder(toState, 1);
                        moved = true;
                        moves++;
                        break outer;
                    }
                }
            }
        }
        if (!moved) break;
    }
}

function ensureAllJobsCovered(states) {
    if (!states.length || jobCount === 0) return;
    const seen = new Array(jobCount).fill(false);
    for (const state of states) {
        const filtered = [];
        for (const jobIdx of state.jobs) {
            if (!seen[jobIdx]) {
                seen[jobIdx] = true;
                filtered.push(jobIdx);
            }
        }
        state.jobs = filtered;
    }
    const missing = [];
    for (let i = 0; i < jobCount; i++) {
        if (!seen[i]) missing.push(i);
    }
    for (const jobIdx of missing) {
        let bestState = states[0];
        let bestPos = states[0].jobs.length;
        let bestDelta = INF;
        for (const state of states) {
            const insertion = bestInsertion(state, jobIdx);
            if (insertion.delta + EPS < bestDelta) {
                bestDelta = insertion.delta;
                bestState = state;
                bestPos = insertion.pos;
            }
        }
        if (bestState) {
            bestState.jobs.splice(bestPos, 0, jobIdx);
        }
    }
}
