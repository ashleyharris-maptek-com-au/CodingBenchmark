```typescript
declare const require: any;

const fs = require('fs');

const input: string = fs.readFileSync(0, 'utf8');
const data = JSON.parse(input);

type Package = { id: string | number; x: number; y: number };

const depotArray: number[] = Array.isArray(data.depot) ? data.depot : [0, 0];
const depotX = Number(depotArray[0]) || 0;
const depotY = Number(depotArray[1]) || 0;

const truckCountRaw = typeof data.truckCount === 'number' ? Math.floor(data.truckCount) : 0;
const truckCount = Math.max(0, truckCountRaw);

const packages: Package[] = Array.isArray(data.packages)
  ? data.packages.map((pkg: any, idx: number) => {
      const x = Number(pkg?.x);
      const y = Number(pkg?.y);
      return {
        id: pkg?.id !== undefined ? pkg.id : idx,
        x: Number.isFinite(x) ? x : 0,
        y: Number.isFinite(y) ? y : 0,
      };
    })
  : [];

const totalPackages = packages.length;

function distance(ax: number, ay: number, bx: number, by: number): number {
  return Math.hypot(ax - bx, ay - by);
}

function distanceSq(ax: number, ay: number, bx: number, by: number): number {
  const dx = ax - bx;
  const dy = ay - by;
  return dx * dx + dy * dy;
}

function initializeCenters(pkgList: Package[], k: number): { x: number; y: number }[] {
  const centers: { x: number; y: number }[] = [];
  const n = pkgList.length;
  if (n === 0) return centers;

  let sumX = 0;
  let sumY = 0;
  for (const pkg of pkgList) {
    sumX += pkg.x;
    sumY += pkg.y;
  }
  const centroidX = sumX / n;
  const centroidY = sumY / n;

  let firstIndex = 0;
  let bestDist = Infinity;
  for (let i = 0; i < n; i++) {
    const d = distanceSq(pkgList[i].x, pkgList[i].y, centroidX, centroidY);
    if (d < bestDist) {
      bestDist = d;
      firstIndex = i;
    }
  }

  centers.push({ x: pkgList[firstIndex].x, y: pkgList[firstIndex].y });
  const chosen = new Set<number>([firstIndex]);

  while (centers.length < k) {
    let bestCandidate = -1;
    let bestDistance = -1;

    for (let i = 0; i < n; i++) {
      if (chosen.has(i)) continue;
      let minDist = Infinity;
      for (const center of centers) {
        const d = distanceSq(pkgList[i].x, pkgList[i].y, center.x, center.y);
        if (d < minDist) {
          minDist = d;
        }
      }
      if (minDist > bestDistance) {
        bestDistance = minDist;
        bestCandidate = i;
      }
    }

    if (bestCandidate === -1) {
      break;
    }
    chosen.add(bestCandidate);
    centers.push({ x: pkgList[bestCandidate].x, y: pkgList[bestCandidate].y });
  }

  while (centers.length < k) {
    centers.push({ x: pkgList[0].x, y: pkgList[0].y });
  }

  return centers;
}

function kMeans(pkgList: Package[], k: number, maxIterations = 20): number[][] {
  const n = pkgList.length;
  if (k <= 0 || n === 0) return [];
  if (k >= n) {
    return pkgList.map((_, idx) => [idx]);
  }

  let centers = initializeCenters(pkgList, k);
  let clusters: number[][] = [];

  for (let iteration = 0; iteration < maxIterations; iteration++) {
    const newClusters: number[][] = Array.from({ length: k }, () => []);

    for (let i = 0; i < n; i++) {
      let bestCenter = 0;
      let bestDist = Infinity;
      for (let c = 0; c < k; c++) {
        const d = distanceSq(pkgList[i].x, pkgList[i].y, centers[c].x, centers[c].y);
        if (d < bestDist) {
          bestDist = d;
          bestCenter = c;
        }
      }
      newClusters[bestCenter].push(i);
    }

    for (let c = 0; c < k; c++) {
      if (newClusters[c].length === 0) {
        let donor = -1;
        let donorSize = 0;
        for (let d = 0; d < k; d++) {
          if (newClusters[d].length > donorSize) {
            donorSize = newClusters[d].length;
            donor = d;
          }
        }
        if (donor >= 0 && donor !== c && newClusters[donor].length > 1) {
          newClusters[c].push(newClusters[donor].pop() as number);
        }
      }
    }

    const newCenters: { x: number; y: number }[] = [];
    for (let c = 0; c < k; c++) {
      const cluster = newClusters[c];
      if (cluster.length === 0) {
        newCenters.push(centers[c]);
        continue;
      }
      let sumX = 0;
      let sumY = 0;
      for (const idx of cluster) {
        sumX += pkgList[idx].x;
        sumY += pkgList[idx].y;
      }
      newCenters.push({ x: sumX / cluster.length, y: sumY / cluster.length });
    }

    let shift = 0;
    for (let c = 0; c < k; c++) {
      shift += distanceSq(centers[c].x, centers[c].y, newCenters[c].x, newCenters[c].y);
    }

    centers = newCenters;
    clusters = newClusters;

    if (shift < 1e-6) {
      break;
    }
  }

  return clusters;
}

function buildRoute(indices: number[]): number[] {
  if (indices.length <= 1) return indices.slice();

  const unused = new Set<number>(indices);
  const sequence: number[] = [];

  while (unused.size > 0) {
    let choose = -1;
    let bestDist = Infinity;

    for (const idx of unused) {
      let currentDist = 0;
      if (sequence.length === 0) {
        currentDist = distance(pkgList[idx].x, pkgList[idx].y, depotX, depotY);
      } else {
        const lastIdx = sequence[sequence.length - 1];
        currentDist = distance(pkgList[idx].x, pkgList[idx].y, pkgList[lastIdx].x, pkgList[lastIdx].y);
      }
      if (currentDist < bestDist) {
        bestDist = currentDist;
        choose = idx;
      }
    }

    if (choose === -1) break;
    sequence.push(choose);
    unused.delete(choose);
  }

  return sequence;
}

function improveRoute(order: number[]): number[] {
  const len = order.length;
  if (len <= 2) return order;

  const maxIterations = len * len;
  let iterations = 0;

  while (iterations < maxIterations) {
    let improved = false;

    for (let i = 0; i < len - 1 && !improved; i++) {
      for (let k = i + 1; k < len && !improved; k++) {
        if (k - i <= 1) continue;

        const ax = i === 0 ? depotX : pkgList[order[i - 1]].x;
        const ay = i === 0 ? depotY : pkgList[order[i - 1]].y;
        const bx = pkgList[order[i]].x;
        const by = pkgList[order[i]].y;
        const cx = pkgList[order[k]].x;
        const cy = pkgList[order[k]].y;
        const dx = k === len - 1 ? depotX : pkgList[order[k + 1]].x;
        const dy = k === len - 1 ? depotY : pkgList[order[k + 1]].y;

        const delta =
          distance(ax, ay, cx, cy) +
          distance(bx, by, dx, dy) -
          distance(ax, ay, bx, by) -
          distance(cx, cy, dx, dy);

        if (delta < -1e-6) {
          const slice = order.slice(i, k + 1).reverse();
          for (let m = 0; m < slice.length; m++) {
            order[i + m] = slice[m];
          }
          improved = true;
        }
      }
    }

    if (!improved) break;
    iterations++;
  }

  return order;
}

const pkgList = packages;
let activeRoutes = 0;

if (totalPackages === 0) {
  activeRoutes = 0;
} else if (truckCount > 0) {
  activeRoutes = Math.min(truckCount, totalPackages);
} else {
  activeRoutes = Math.min(1, totalPackages);
}

function planRoutes(routeCount: number): number[][] {
  if (routeCount <= 0) return [];

  let clusters: number[][];
  if (routeCount >= totalPackages) {
    clusters = pkgList.map((_, idx) => [idx]);
  } else {
    clusters = kMeans(pkgList, routeCount);
  }

  const clusterData = clusters.map((cluster) => {
    let sumX = 0;
    let sumY = 0;
    for (const idx of cluster) {
      sumX += pkgList[idx].x;
      sumY += pkgList[idx].y;
    }
    const cx = cluster.length > 0 ? sumX / cluster.length : depotX;
    const cy = cluster.length > 0 ? sumY / cluster.length : depotY;
    const angle = Math.atan2(cy - depotY, cx - depotX);
    return { cluster, angle };
  });

  clusterData.sort((a, b) => a.angle - b.angle);

  const routes: number[][] = [];
  for (const entry of clusterData) {
    const route = improveRoute(buildRoute(entry.cluster));
    routes.push(route);
  }

  return routes;
}

const plannedRoutes = totalPackages > 0 ? planRoutes(activeRoutes) : [];
const finalRoutes: (string | number)[][] = plannedRoutes.map((route) =>
  route.map((idx) => pkgList[idx].id),
);

const totalOutputRoutes = truckCount > 0 ? Math.max(truckCount, finalRoutes.length) : finalRoutes.length;

while (finalRoutes.length < totalOutputRoutes) {
  finalRoutes.push([]);
}

const output = { routes: finalRoutes };
process.stdout.write(JSON.stringify(output));
```
