```typescript
declare const require: any;

const fs = require('fs');

interface InputTown {
    id?: string | number;
    x: number;
    y: number;
    value: number;
}

interface InputData {
    width: number;
    height: number;
    grid: string[];
    ignitions: number[][];
    towns: InputTown[];
    cutBudget: number;
}

interface CellEntry {
    townIndices: number[];
    currentSum: number;
}

interface HeapNode {
    cellIdx: number;
    score: number;
}

const MAX_PATH_CELLS = 4000;

const input: string = fs.readFileSync(0, 'utf8');
let data: InputData;
try {
    data = JSON.parse(input);
} catch {
    console.log(JSON.stringify({ firebreaks: [] }));
    process.exit(0);
}

const width = Math.max(0, Math.floor(Number(data?.width) || 0));
const height = Math.max(0, Math.floor(Number(data?.height) || 0));
const cutBudget = Math.max(0, Math.floor(Number(data?.cutBudget) || 0));

if (width === 0 || height === 0 || cutBudget === 0) {
    console.log(JSON.stringify({ firebreaks: [] }));
    process.exit(0);
}

const gridInput = Array.isArray(data?.grid) ? data.grid : [];
const gridRows: string[] = new Array<string>(height);
for (let y = 0; y < height; y++) {
    const row = gridInput[y];
    gridRows[y] = typeof row === 'string' ? row : '';
}

const isBurnable = (x: number, y: number): boolean => {
    if (x < 0 || x >= width || y < 0 || y >= height) return false;
    const row = gridRows[y];
    if (!row) return false;
    return row.charCodeAt(x) === 46; // '.'
};

const size = width * height;
const arrival = new Array<number>(size).fill(Number.POSITIVE_INFINITY);
const parent = new Int32Array(size);
parent.fill(-1);

const queue: number[] = [];
let qHead = 0;

const ignitions = Array.isArray(data?.ignitions) ? data.ignitions : [];
for (const ign of ignitions) {
    if (!Array.isArray(ign) || ign.length < 2) continue;
    const ixRaw = Number(ign[0]);
    const iyRaw = Number(ign[1]);
    if (!Number.isFinite(ixRaw) || !Number.isFinite(iyRaw)) continue;
    const ix = Math.round(ixRaw);
    const iy = Math.round(iyRaw);
    if (!isBurnable(ix, iy)) continue;
    const idx = iy * width + ix;
    if (arrival[idx] !== Number.POSITIVE_INFINITY) continue;
    arrival[idx] = 0;
    parent[idx] = -1;
    queue.push(idx);
}

const dirs = [
    [1, 0],
    [-1, 0],
    [0, 1],
    [0, -1],
];

while (qHead < queue.length) {
    const currentIdx = queue[qHead++];
    const baseTime = arrival[currentIdx];
    const cx = currentIdx % width;
    const cy = (currentIdx / width) | 0;
    for (const [dx, dy] of dirs) {
        const nx = cx + dx;
        const ny = cy + dy;
        if (!isBurnable(nx, ny)) continue;
        const nIdx = ny * width + nx;
        if (arrival[nIdx] !== Number.POSITIVE_INFINITY) continue;
        arrival[nIdx] = baseTime + 1;
        parent[nIdx] = currentIdx;
        queue.push(nIdx);
    }
}

const townsData: any[] = Array.isArray(data?.towns) ? data.towns : [];
const townCount = townsData.length;
const townValues = new Array<number>(townCount).fill(0);
const townCells = new Array<number>(townCount).fill(-1);
const townPathCells: number[][] = new Array(townCount);
const threatenedIndices: number[] = [];
const cellEntries = new Map<number, CellEntry>();

for (let i = 0; i < townCount; i++) {
    townPathCells[i] = [];
}

for (let i = 0; i < townCount; i++) {
    const town = townsData[i] || {};
    const rawValue = Number(town.value);
    const value = Number.isFinite(rawValue) ? Math.max(0, rawValue) : 0;
    townValues[i] = value;
    if (value <= 0) continue;

    const txNum = Number(town.x);
    const tyNum = Number(town.y);
    if (!Number.isFinite(txNum) || !Number.isFinite(tyNum)) continue;
    const tx = Math.round(txNum);
    const ty = Math.round(tyNum);
    if (tx < 0 || tx >= width || ty < 0 || ty >= height) continue;
    if (!isBurnable(tx, ty)) continue;

    const cellIdx = ty * width + tx;
    townCells[i] = cellIdx;
    if (!Number.isFinite(arrival[cellIdx])) continue;

    const path: number[] = [];
    let current = cellIdx;
    let steps = 0;
    while (current !== -1 && steps < MAX_PATH_CELLS) {
        path.push(current);
        current = parent[current];
        steps++;
    }
    if (path.length === 0) continue;
    townPathCells[i] = path;
    threatenedIndices.push(i);

    for (const pIdx of path) {
        let entry = cellEntries.get(pIdx);
        if (!entry) {
            entry = { townIndices: [], currentSum: 0 };
            cellEntries.set(pIdx, entry);
        }
        entry.townIndices.push(i);
        entry.currentSum += value;
    }
}

const heap: HeapNode[] = [];
const heapPush = (node: HeapNode): void => {
    heap.push(node);
    let idx = heap.length - 1;
    while (idx > 0) {
        const parentIdx = (idx - 1) >> 1;
        if (heap[parentIdx].score >= node.score) break;
        heap[idx] = heap[parentIdx];
        idx = parentIdx;
    }
    heap[idx] = node;
};

const heapPop = (): HeapNode | undefined => {
    if (heap.length === 0) return undefined;
    const top = heap[0];
    const last = heap.pop()!;
    if (heap.length > 0) {
        let idx = 0;
        while (true) {
            const left = idx * 2 + 1;
            if (left >= heap.length) break;
            const right = left + 1;
            let child = left;
            if (right < heap.length && heap[right].score > heap[left].score) {
                child = right;
            }
            if (heap[child].score <= last.score) break;
            heap[idx] = heap[child];
            idx = child;
        }
        heap[idx] = last;
    }
    return top;
};

for (const [cellIdx, entry] of cellEntries) {
    if (entry.currentSum > 0) {
        heapPush({ cellIdx, score: entry.currentSum });
    }
}

const selectedCells: number[] = [];
const selectedSet = new Set<number>();
const townSaved = new Array<boolean>(townCount).fill(false);

let budgetLeft = cutBudget;

while (budgetLeft > 0) {
    const node = heapPop();
    if (!node) break;
    const entry = cellEntries.get(node.cellIdx);
    if (!entry) continue;
    if (selectedSet.has(node.cellIdx)) continue;
    if (entry.currentSum <= 0) continue;
    if (node.score !== entry.currentSum) {
        if (entry.currentSum > 0) {
            heapPush({ cellIdx: node.cellIdx, score: entry.currentSum });
        }
        continue;
    }
    selectedSet.add(node.cellIdx);
    selectedCells.push(node.cellIdx);
    budgetLeft--;

    const affected = entry.townIndices;
    entry.currentSum = 0;
    for (const tIdx of affected) {
        if (townSaved[tIdx]) continue;
        townSaved[tIdx] = true;
        const pathCells = townPathCells[tIdx];
        if (!pathCells || pathCells.length === 0) continue;
        const value = townValues[tIdx];
        if (value <= 0) continue;
        for (const cIdx of pathCells) {
            const e2 = cellEntries.get(cIdx);
            if (e2 && e2.currentSum > 0) {
                e2.currentSum -= value;
                if (e2.currentSum < 0) e2.currentSum = 0;
            }
        }
    }
}

if (budgetLeft > 0 && threatenedIndices.length > 0) {
    const fallbackOrder = threatenedIndices.slice().sort((a, b) => townValues[b] - townValues[a]);
    for (const tIdx of fallbackOrder) {
        if (budgetLeft === 0) break;
        if (townSaved[tIdx]) continue;
        const cellIdx = townCells[tIdx];
        if (cellIdx < 0) continue;
        if (selectedSet.has(cellIdx)) continue;
        selectedSet.add(cellIdx);
        selectedCells.push(cellIdx);
        budgetLeft--;
        townSaved[tIdx] = true;
    }
}

const firebreaks = selectedCells.map(idx => [idx % width, Math.floor(idx / width)]);
console.log(JSON.stringify({ firebreaks }));
```
