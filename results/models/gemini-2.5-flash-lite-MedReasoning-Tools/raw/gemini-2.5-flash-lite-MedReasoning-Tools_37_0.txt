using System;
using System.Collections.Generic;
using System.Linq;

namespace DFVSSolver
{
    class Program
    {
        static void Main(string[] args)
        {
            string[] line1 = Console.ReadLine().Split();
            int n = int.Parse(line1[0]);
            int m = int.Parse(line1[1]);

            List<HashSet<int>> adj = new List<HashSet<int>>(n);
            List<HashSet<int>> revAdj = new List<HashSet<int>>(n);
            for (int i = 0; i < n; i++)
            {
                adj.Add(new HashSet<int>());
                revAdj.Add(new HashSet<int>());
            }

            for (int i = 0; i < m; i++)
            {
                string[] edge = Console.ReadLine().Split();
                int u = int.Parse(edge[0]);
                int v = int.Parse(edge[1]);
                if (u == v) continue; // Self-loops handled separately
                adj[u].Add(v);
                revAdj[v].Add(u);
            }

            List<int> result = new List<int>();
            
            // Basic preprocessing: Remove sources and sinks iteratively
            bool changed = true;
            while (changed)
            {
                changed = false;
                for (int i = 0; i < n; i++)
                {
                    if (adj[i] == null) continue;
                    
                    if (adj[i].Count == 0 || revAdj[i].Count == 0)
                    {
                        RemoveVertex(i, n, adj, revAdj);
                        changed = true;
                    }
                }
            }

            // Greedy heuristic for remaining graph
            while (true)
            {
                int maxScore = -1;
                int bestVertex = -1;
                bool found = false;

                for (int i = 0; i < n; i++)
                {
                    if (adj[i] == null) continue;
                    
                    int score = adj[i].Count * revAdj[i].Count;
                    if (score > maxScore)
                    {
                        maxScore = score;
                        bestVertex = i;
                        found = true;
                    }
                }

                if (!found || maxScore == 0) break;

                result.Add(bestVertex);
                RemoveVertex(bestVertex, n, adj, revAdj);
            }

            Console.WriteLine(result.Count);
            Console.WriteLine(string.Join(" ", result));
        }

        static void RemoveVertex(int v, int n, List<HashSet<int>> adj, List<HashSet<int>> revAdj)
        {
            if (adj[v] != null)
            {
                foreach (int neighbor in adj[v])
                {
                    revAdj[neighbor].Remove(v);
                }
                adj[v] = null;
            }

            if (revAdj[v] != null)
            {
                foreach (int prev in revAdj[v])
                {
                    adj[prev].Remove(v);
                }
                revAdj[v] = null;
            }
        }
    }
}
