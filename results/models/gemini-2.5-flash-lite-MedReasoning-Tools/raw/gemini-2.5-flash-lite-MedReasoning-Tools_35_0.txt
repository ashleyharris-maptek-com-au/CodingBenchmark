use std::collections::HashSet;

struct Graph {
    adj: Vec<Vec<usize>>,
    num_vertices: usize,
}

impl Graph {
    fn new(num_vertices: usize) -> Self {
        Graph {
            adj: vec![vec![]; num_vertices],
            num_vertices,
        }
    }

    fn add_edge(&mut self, u: usize, v: usize) {
        self.adj[u].push(v);
        self.adj[v].push(u);
    }
}

fn solve(graph: &Graph) -> HashSet<usize> {
    let mut best_set: Option<HashSet<usize>> = None;
    let mut current_set: HashSet<usize> = HashSet::new();
    let mut covered_count: Vec<usize> = vec![0; graph.num_vertices];
    let mut covered_nodes: usize = 0;

    backtrack(
        graph,
        &mut current_set,
        &mut covered_count,
        &mut covered_nodes,
        &mut best_set,
    );

    best_set.unwrap_or_default()
}

fn backtrack(
    graph: &Graph,
    current_set: &mut HashSet<usize>,
    covered_count: &mut Vec<usize>,
    covered_nodes: &mut usize,
    best_set: &mut Option<HashSet<usize>>,
) {
    if let Some(best) = best_set {
        if current_set.len() >= best.len() {
            return;
        }
    }

    if *covered_nodes == graph.num_vertices {
        *best_set = Some(current_set.clone());
        return;
    }

    // Pick an uncovered vertex (with smallest degree? Maybe heuristic)
    let u = (0..graph.num_vertices)
        .find(|&v| covered_count[v] == 0)
        .expect("Should have an uncovered vertex");

    // Options to cover u: add u or one of its neighbors to the set
    let mut candidates = vec![u];
    candidates.extend(graph.adj[u].iter().cloned());

    for &candidate in &candidates {
        if current_set.contains(&candidate) {
            continue;
        }

        // Apply candidate
        current_set.insert(candidate);

        let mut newly_covered = Vec::new();

        // Update covered
        let mut nodes_to_update = vec![candidate];
        nodes_to_update.extend(graph.adj[candidate].iter().cloned());

        for &node in &nodes_to_update {
            if covered_count[node] == 0 {
                newly_covered.push(node);
                *covered_nodes += 1;
            }
            covered_count[node] += 1;
        }

        backtrack(graph, current_set, covered_count, covered_nodes, best_set);

        // Backtrack
        for &node in &nodes_to_update {
            covered_count[node] -= 1;
            if covered_count[node] == 0 {
                *covered_nodes -= 1;
            }
        }
        current_set.remove(&candidate);
    }
}

fn main() {
    let mut graph = Graph::new(5);
    graph.add_edge(0, 1);
    graph.add_edge(1, 2);
    graph.add_edge(2, 3);
    graph.add_edge(3, 4);
    graph.add_edge(4, 0);

    let mds = solve(&graph);
    println!("MDS: {:?}", mds);
}
