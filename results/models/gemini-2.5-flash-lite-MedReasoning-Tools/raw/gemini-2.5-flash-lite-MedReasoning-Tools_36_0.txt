#include <iostream>
#include <vector>
#include <algorithm>
#include <bitset>

using namespace std;

// Using a simple backtracking with pruning for Maximum Independent Set
// For larger graphs, this might need more advanced techniques like bitmasking or Bron-Kerbosch.
// Given the constraints, let's start with a robust backtracking approach.

int n, m;
vector<vector<int>> adj;
vector<int> max_set;

void solve(vector<int>& candidates, vector<int>& current_set, vector<bitset<1000>>& adj_bitset) {
    if (candidates.empty()) {
        if (current_set.size() > max_set.size()) {
            max_set = current_set;
        }
        return;
    }

    // Pruning
    if (current_set.size() + candidates.size() <= max_set.size()) {
        return;
    }

    int v = candidates.back();
    candidates.pop_back();

    // Option 1: Include v in the independent set
    bool can_include = true;
    for (int u : current_set) {
        if (adj_bitset[v].test(u)) {
            can_include = false;
            break;
        }
    }

    if (can_include) {
        current_set.push_back(v);
        vector<int> next_candidates;
        for (int c : candidates) {
            if (!adj_bitset[v].test(c)) {
                next_candidates.push_back(c);
            }
        }
        solve(next_candidates, current_set, adj_bitset);
        current_set.pop_back();
    }

    // Option 2: Exclude v
    solve(candidates, current_set, adj_bitset);
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    if (!(cin >> n >> m)) return 0;

    adj.resize(n);
    vector<bitset<1000>> adj_bitset(n);

    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        if (u >= 0 && u < n && v >= 0 && v < n && u != v) {
            adj_bitset[u].set(v);
            adj_bitset[v].set(u);
        }
    }

    vector<int> candidates(n);
    for (int i = 0; i < n; ++i) candidates[i] = i;

    vector<int> current_set;
    solve(candidates, current_set, adj_bitset);

    cout << max_set.size() << "\n";
    for (int i = 0; i < max_set.size(); ++i) {
        cout << max_set[i] << (i == max_set.size() - 1 ? "" : " ");
    }
    cout << endl;

    return 0;
}
