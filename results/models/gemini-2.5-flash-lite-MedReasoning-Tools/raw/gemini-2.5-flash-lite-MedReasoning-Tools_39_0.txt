use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let first_line = lines.next().unwrap().unwrap();
    let parts: Vec<usize> = first_line
        .split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();
    let n = parts[0];
    let m = parts[1];

    let mut adj = vec![vec![false; n]; n];
    for _ in 0..m {
        let line = lines.next().unwrap().unwrap();
        let edge: Vec<usize> = line
            .split_whitespace()
            .map(|s| s.parse().unwrap())
            .collect();
        let u = edge[0];
        let v = edge[1];
        adj[u][v] = true;
        adj[v][u] = true;
    }

    // Brute force for small n
    let mut min_cut = usize::MAX;
    let mut best_partition = Vec::new();

    // We need to choose n/2 vertices out of n
    // Use combinations
    use itertools::Itertools;
    for partition in (0..n).combinations(n / 2) {
        let mut is_in_a = vec![false; n];
        for &v in &partition {
            is_in_a[v] = true;
        }

        let mut current_cut = 0;
        for u in 0..n {
            for v in (u + 1)..n {
                if adj[u][v] && is_in_a[u] != is_in_a[v] {
                    current_cut += 1;
                }
            }
        }

        if current_cut < min_cut {
            min_cut = current_cut;
            best_partition = partition;
        }
    }

    println!("{}", min_cut);
    println!("{}", best_partition.iter().map(|v| v.to_string()).collect::<Vec<_>>().join(" "));
}
