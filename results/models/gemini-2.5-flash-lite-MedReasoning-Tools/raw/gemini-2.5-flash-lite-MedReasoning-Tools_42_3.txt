; SPIR-V Assembly Geometry Shader: Line to Quad Ribbon
; Input: InputLines (array size 2)
; Output: OutputTriangleStrip, OutputVertices 4

               OpCapability Geometry
               OpCapability Shader
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %outColor %inColor
               OpExecutionMode %main InputLines
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 4
               OpExecutionMode %main Invocations 1

; Decorators
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_in Input
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %gl_out Output
               OpDecorate %outColor Location 0
               OpDecorate %inColor Location 2

; Types
       %void = OpTypeVoid
       %func = OpTypeFunction %void
      %float = OpTypeFloat 32
     %v4float = OpTypeVector %float 4
     %v3float = OpTypeVector %float 3
        %uint = OpTypeInt 32 0
         %int = OpTypeInt 32 1
      %uint_2 = OpConstant %uint 2
      %uint_0 = OpConstant %uint 0
      %int_0 = OpConstant %int 0
      %int_1 = OpConstant %int 1
    %float_0 = OpConstant %float 0.0

; Interfaces
%gl_PerVertex_in = OpTypeStruct %v4float
  %arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
      %gl_in = OpVariable %ptr_in_arr_pv Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
     %gl_out = OpVariable %ptr_out_pv Output

%arr_v3_2 = OpTypeArray %v3float %uint_2
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_2
    %inColor = OpVariable %ptr_in_arr_v3 Input

%ptr_out_v3 = OpTypePointer Output %v3float
   %outColor = OpVariable %ptr_out_v3 Output

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_v4_pos = OpTypePointer Output %v4float

; Thickness constant
  %f_thick = OpConstant %float 0.03
  %f_1 = OpConstant %float 1.0

       %main = OpFunction %void None %func
         %entry = OpLabel

; 1. Read Inputs
       %p0_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
           %p0 = OpLoad %v4float %p0_ptr
       %p1_ptr = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
           %p1 = OpLoad %v4float %p1_ptr
       %c0_ptr = OpAccessChain %ptr_in_v3 %inColor %int_0
           %c0 = OpLoad %v3float %c0_ptr
       %c1_ptr = OpAccessChain %ptr_in_v3 %inColor %int_1
           %c1 = OpLoad %v3float %c1_ptr

; 2. Compute Direction: p1 - p0 (in clip space, ignoring W for now)
       %p0_v3 = OpVectorShuffle %v3float %p0 %p0 0 1 2
       %p1_v3 = OpVectorShuffle %v3float %p1 %p1 0 1 2
       %vec_dir = OpFSub %v3float %p1_v3 %p0_v3

; 3. Perpendicular
       %perp_x = OpFNegate %float %vec_dir.y ; Needs index, assuming vec_dir is v3
       ; Corrected: vec_dir is v3. dir.y is index 1.
       %dir_y = OpCompositeExtract %float %vec_dir 1
       %dir_x = OpCompositeExtract %float %vec_dir 0
       %perp_x_val = OpFNegate %float %dir_y
       %perp_y_val = %dir_x
       
       %perp_v3 = OpCompositeConstruct %v3float %perp_x_val %perp_y_val %float_0
       %perp_n = OpExtInst %v3float %1 Normalize %perp_v3
       %perp = OpVectorTimesScalar %v3float %perp_n %f_thick

; 4. Emit 4 Vertices
; Vertex 0: p0 + perp, color c0
       %v0_pos_v3 = OpFAdd %v3float %p0_v3 %perp
       %v0_pos = OpVectorShuffle %v4float %v0_pos_v3 %v0_pos_v3 0 1 2 3 ; need to make W=1
       %v0_pos_v4 = OpCompositeInsert %v4float %f_1 %v0_pos_v3 3
       
       %gl_out_pos_ptr = OpAccessChain %ptr_out_v4_pos %gl_out %int_0
       OpStore %outColor %c0
       OpStore %gl_out_pos_ptr %v0_pos_v4
       OpEmitVertex

; Vertex 1: p0 - perp, color c0
       %v1_pos_v3 = OpFSub %v3float %p0_v3 %perp
       %v1_pos_v4 = OpCompositeInsert %v4float %f_1 %v1_pos_v3 3
       OpStore %outColor %c0
       OpStore %gl_out_pos_ptr %v1_pos_v4
       OpEmitVertex

; Vertex 2: p1 + perp, color c1
       %v2_pos_v3 = OpFAdd %v3float %p1_v3 %perp
       %v2_pos_v4 = OpCompositeInsert %v4float %f_1 %v2_pos_v3 3
       OpStore %outColor %c1
       OpStore %gl_out_pos_ptr %v2_pos_v4
       OpEmitVertex

; Vertex 3: p1 - perp, color c1
       %v3_pos_v3 = OpFSub %v3float %p1_v3 %perp
       %v3_pos_v4 = OpCompositeInsert %v4float %f_1 %v3_pos_v3 3
       OpStore %outColor %c1
       OpStore %gl_out_pos_ptr %v3_pos_v4
       OpEmitVertex

       OpEndPrimitive
       OpReturn
       OpFunctionEnd
