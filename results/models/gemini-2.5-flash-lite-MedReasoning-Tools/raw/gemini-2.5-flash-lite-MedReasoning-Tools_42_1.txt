; Geometry shader to expand points into diamond shapes
; InputPrimitive: InputPoints (size 1)
; OutputPrimitive: OutputTriangleStrip (max 4 vertices)

               OpCapability Geometry
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %outColor_out %inColor_in %inPos_in
               OpExecutionMode %main InputPoints
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 4
               OpExecutionMode %main Invocations 1

; Layout/Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_in Input
               
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %gl_out Output

               OpDecorate %inPos_in Location 0
               OpDecorate %inColor_in Location 2
               OpDecorate %outColor_out Location 0

; Types
%void = OpTypeVoid
%float = OpTypeFloat 32
%v3float = OpTypeVector %float 3
%v4float = OpTypeVector %float 4
%int = OpTypeInt 32 1
%uint = OpTypeInt 32 0
%ptr_float = OpTypePointer Output %float
%int_0 = OpConstant %int 0
%uint_1 = OpConstant %uint 1
%float_1 = OpConstant %float 1.0
%float_h = OpConstant %float 0.05
%float_neg_h = OpConstant %float -0.05

; Input/Output Structs
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_1
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

; User inputs
%arr_v3_1 = OpTypeArray %v3float %uint_1
%ptr_in_arr_v3 = OpTypePointer Input %arr_v3_1
%inPos_in = OpVariable %ptr_in_arr_v3 Input
%inColor_in = OpVariable %ptr_in_arr_v3 Input

; User outputs
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor_out = OpVariable %ptr_out_v3 Output

; Pointers for access chains
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_v4 = OpTypePointer Output %v4float

; Main function
%fn_main = OpTypeFunction %void
%main = OpFunction %void None %fn_main
%entry = OpLabel

; Get input color (shared for all vertices)
%ptr_in_color = OpAccessChain %ptr_in_v3 %inColor_in %int_0
%color = OpLoad %v3float %ptr_in_color
OpStore %outColor_out %color

; Get base position (clip space, assuming vec3 input)
%ptr_in_pos = OpAccessChain %ptr_in_v3 %inPos_in %int_0
%basePos = OpLoad %v3float %ptr_in_pos
%x = OpCompositeExtract %float %basePos 0
%y = OpCompositeExtract %float %basePos 1
%z = OpCompositeExtract %float %basePos 2

; 1. Top: (x, y+h)
%y_top = OpFAdd %float %y %float_h
%pos_top = OpCompositeConstruct %v4float %x %y_top %z %float_1
%ptr_out_pos = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out_pos %pos_top
OpEmitVertex

; 2. Right: (x+h, y)
%x_right = OpFAdd %float %x %float_h
%pos_right = OpCompositeConstruct %v4float %x_right %y %z %float_1
OpStore %ptr_out_pos %pos_right
OpEmitVertex

; 3. Left: (x-h, y)
%x_left = OpFAdd %float %x %float_neg_h
%pos_left = OpCompositeConstruct %v4float %x_left %y %z %float_1
OpStore %ptr_out_pos %pos_left
OpEmitVertex

; 4. Bottom: (x, y-h)
%y_bot = OpFAdd %float %y %float_neg_h
%pos_bot = OpCompositeConstruct %v4float %x %y_bot %z %float_1
OpStore %ptr_out_pos %pos_bot
OpEmitVertex

OpEndPrimitive
OpReturn
OpFunctionEnd
