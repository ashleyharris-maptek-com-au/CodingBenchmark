import sys

def solve_exact_cover():
    lines = sys.stdin.readlines()
    if not lines:
        return
        
    first_line = lines[0].split()
    if not first_line: return
    U = int(first_line[0])
    S = int(first_line[1])
    
    sets = []
    for i in range(1, S + 1):
        sets.append(list(map(int, lines[i].split())))
    
    # Algorithm X using simple backtracking
    # Universe elements: 0 to U-1
    # Sets: 0 to S-1
    
    def get_solution():
        covered = [False] * U
        chosen_indices = []
        
        def backtrack():
            if all(covered):
                return True
            
            # Find an uncovered element
            uncovered_element = -1
            for i in range(U):
                if not covered[i]:
                    uncovered_element = i
                    break
            
            if uncovered_element == -1:
                return True # All covered
                
            # Try all sets covering this element
            for s_idx, s in enumerate(sets):
                if s_idx in chosen_indices: continue
                
                if uncovered_element in s:
                    # Check if all elements in set are currently uncovered
                    if all(not covered[e] for e in s):
                        # Choose this set
                        for e in s:
                            covered[e] = True
                        chosen_indices.append(s_idx)
                        
                        if backtrack():
                            return True
                        
                        # Backtrack
                        chosen_indices.pop()
                        for e in s:
                            covered[e] = False
            
            return False

        if backtrack():
            print("SOLUTION")
            print(*(chosen_indices))
        else:
            print("NO SOLUTION")

    get_solution()
    
if __name__ == "__main__":
    solve_exact_cover()
