; SPIR-V Geometry Shader
; Input: InputTrianglesAdjacency (6 vertices)
; Output: OutputTriangleStrip, OutputVertices 3

               OpCapability Geometry
               OpExtension "SPV_KHR_shader_draw_parameters"
          %1 = OpExtInstImport "GLSL.std.450"
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %outColor %worldPos_in %color_in
               OpExecutionMode %main InputTrianglesAdjacency
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 3
               OpExecutionMode %main Invocations 1

; --- Decorations ---
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %worldPos_in Location 0
               OpDecorate %color_in Location 2
               OpDecorate %outColor Location 0

; --- Types ---
       %void = OpTypeVoid
       %func  = OpTypeFunction %void
      %float = OpTypeFloat 32
      %v3float = OpTypeVector %float 3
      %v4float = OpTypeVector %float 4
       %uint  = OpTypeInt 32 0
        %int  = OpTypeInt 32 1
      %uint_6 = OpConstant %uint 6
      %int_0  = OpConstant %int 0
      %int_1  = OpConstant %int 1
      %int_2  = OpConstant %int 2
      %int_4  = OpConstant %int 4

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_6
%ptr_input_arr = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_input_arr Input

%arr_v3_6 = OpTypeArray %v3float %uint_6
%ptr_input_v3_arr = OpTypePointer Input %arr_v3_6
%worldPos_in = OpVariable %ptr_input_v3_arr Input
%color_in = OpVariable %ptr_input_v3_arr Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_output_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_output_pv Output
%ptr_output_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_output_v3 Output

%ptr_input_v4 = OpTypePointer Input %v4float
%ptr_output_v4 = OpTypePointer Output %v4float

; --- Main ---
%main = OpFunction %void None %func
%entry = OpLabel

; 1. Get positions for face normal calculation (vertices 0, 2, 4)
%ptr_p0_in = OpAccessChain %ptr_input_v3_arr %worldPos_in %int_0
%p0 = OpLoad %v3float %ptr_p0_in
%ptr_p2_in = OpAccessChain %ptr_input_v3_arr %worldPos_in %int_2
%p2 = OpLoad %v3float %ptr_p2_in
%ptr_p4_in = OpAccessChain %ptr_input_v3_arr %worldPos_in %int_4
%p4 = OpLoad %v3float %ptr_p4_in

; 2. Compute face normal: N = normalize(cross(p2-p0, p4-p0))
%v20 = OpFSub %v3float %p2 %p0
%v40 = OpFSub %v3float %p4 %p0
%cross = OpExtInst %v3float %1 Cross %v20 %v40
%N = OpExtInst %v3float %1 Normalize %cross

; 3. Light direction: L = normalize(vec3(1, 1, 1))
%L_raw = OpConstantComposite %v3float %float 1.0 %float 1.0 %float 1.0
%L = OpExtInst %v3float %1 Normalize %L_raw

; 4. Diffuse = max(dot(N, L), 0.0)
%dot = OpDot %float %N %L
%zero = OpConstant %float 0.0
%diffuse = OpExtInst %float %1 FMax %dot %zero

; 5. Base color = average(color[0], color[2], color[4])
%ptr_c0_in = OpAccessChain %ptr_input_v3_arr %color_in %int_0
%c0 = OpLoad %v3float %ptr_c0_in
%ptr_c2_in = OpAccessChain %ptr_input_v3_arr %color_in %int_2
%c2 = OpLoad %v3float %ptr_c2_in
%ptr_c4_in = OpAccessChain %ptr_input_v3_arr %color_in %int_4
%c4 = OpLoad %v3float %ptr_c4_in
%sum_c = OpFAdd %v3float %c0 %c2
%sum_c2 = OpFAdd %v3float %sum_c %c4
%avg_c = OpFDiv %v3float %sum_c2 %float 3.0

; 6. Final color = base * (0.2 + 0.8 * diffuse)
%factor = OpFMul %float %diffuse %float 0.8
%factor2 = OpFAdd %float %factor %float 0.2
%finalColor = OpVectorTimesScalar %v3float %avg_c %factor2

; 7. Emit 3 vertices (indices 0, 2, 4)
; Helper to emit a vertex
%emit = OpLabel
%ptr_out_pos = OpAccessChain %ptr_output_v4 %gl_out %int_0

; Vertex 0
%ptr_pos0_in = OpAccessChain %ptr_input_v4 %gl_in %int_0 %int_0
%pos0 = OpLoad %v4float %ptr_pos0_in
OpStore %ptr_out_pos %pos0
OpStore %outColor %finalColor
OpEmitVertex

; Vertex 2
%ptr_pos2_in = OpAccessChain %ptr_input_v4 %gl_in %int_2 %int_0
%pos2 = OpLoad %v4float %ptr_pos2_in
OpStore %ptr_out_pos %pos2
OpStore %outColor %finalColor
OpEmitVertex

; Vertex 4
%ptr_pos4_in = OpAccessChain %ptr_input_v4 %gl_in %int_4 %int_0
%pos4 = OpLoad %v4float %ptr_pos4_in
OpStore %ptr_out_pos %pos4
OpStore %outColor %finalColor
OpEmitVertex

OpEndPrimitive

OpReturn
OpFunctionEnd
