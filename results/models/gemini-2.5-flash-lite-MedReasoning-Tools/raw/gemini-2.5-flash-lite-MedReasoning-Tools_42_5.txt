; SPIR-V Geometry Shader
OpCapability Geometry
OpCapability Shader
%glsl = OpExtInstImport "GLSL.std.450"
OpMemoryModel Logical GLSL450
OpEntryPoint Geometry %main "main" %gl_in %gl_out %outColor %inColor

OpExecutionMode %main InputLines
OpExecutionMode %main OutputTriangleStrip
OpExecutionMode %main OutputVertices 7
OpExecutionMode %main Invocations 1

OpDecorate %gl_PerVertex_in Block
OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
OpDecorate %gl_PerVertex_out Block
OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
OpDecorate %gl_in Input
OpDecorate %gl_out Output
OpDecorate %inColor Location 1
OpDecorate %outColor Location 0

%void = OpTypeVoid
%void_func = OpTypeFunction %void
%float = OpTypeFloat 32
%v4float = OpTypeVector %float 4
%v3float = OpTypeVector %float 3
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_2 = OpConstant %uint 2
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%float_0 = OpConstant %float 0.0
%float_0_015 = OpConstant %float 0.015
%float_0_05 = OpConstant %float 0.05
%float_0_75 = OpConstant %float 0.75

%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_2
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%gl_in = OpVariable %ptr_in_arr_pv Input

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%gl_out = OpVariable %ptr_out_pv Output

%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v4_pos = OpAccessChain %ptr_out_v4 %gl_out %int_0

%arr_v3_in = OpTypeArray %v3float %uint_2
%ptr_in_v3 = OpTypePointer Input %v3float
%ptr_out_v3 = OpTypePointer Output %v3float
%outColor = OpVariable %ptr_out_v3 Output
%inColor = OpVariable %arr_v3_in Input

%main = OpFunction %void None %void_func
%label = OpLabel
%p0_ptr = OpAccessChain %ptr_in_arr_pv %gl_in %int_0
%p0_v4_ptr = OpAccessChain %ptr_in_v4 %p0_ptr %int_0
%p0 = OpLoad %v4float %p0_v4_ptr
%p1_ptr = OpAccessChain %ptr_in_arr_pv %gl_in %int_1
%p1_v4_ptr = OpAccessChain %ptr_in_v4 %p1_ptr %int_0
%p1 = OpLoad %v4float %p1_v4_ptr
%c0_ptr = OpAccessChain %ptr_in_v3 %inColor %int_0
%c0 = OpLoad %v3float %c0_ptr
%c1_ptr = OpAccessChain %ptr_in_v3 %inColor %int_1
%c1 = OpLoad %v3float %c1_ptr

%dir = OpFSub %v4float %p1 %p0
%dir_xy = OpVectorShuffle %v3float %dir %dir 0 1 2
%dir_x = OpCompositeExtract %float %dir_xy 0
%dir_y = OpCompositeExtract %float %dir_xy 1
%perp_raw = OpCompositeConstruct %v3float %dir_y %dir_x %float_0 ; Simplified: perp = (-dir.y, dir.x)
%perp_len = OpExtInst %float %glsl Length %perp_raw
%perp = OpFDiv %v3float %perp_raw %perp_len
%perp_v4 = OpCompositeConstruct %v4float %perp %float_0
%delta = OpVectorTimesScalar %v4float %dir %float_0_75
%mid = OpFAdd %v4float %p0 %delta

%off_shaft = OpVectorTimesScalar %v4float %perp_v4 %float_0_015
%off_arrow = OpVectorTimesScalar %v4float %perp_v4 %float_0_05

%p0_plus = OpFAdd %v4float %p0 %off_shaft
%p0_minus = OpFSub %v4float %p0 %off_shaft
%mid_plus_shaft = OpFAdd %v4float %mid %off_shaft
%mid_minus_shaft = OpFSub %v4float %mid %off_shaft
%mid_plus_arrow = OpFAdd %v4float %mid %off_arrow
%mid_minus_arrow = OpFSub %v4float %mid %off_arrow

OpStore %outColor %c0
OpStore %ptr_out_v4_pos %p0_plus
OpEmitVertex
OpStore %ptr_out_v4_pos %p0_minus
OpEmitVertex
OpStore %ptr_out_v4_pos %mid_plus_shaft
OpEmitVertex
OpStore %ptr_out_v4_pos %mid_minus_shaft
OpEmitVertex
OpEndPrimitive

OpStore %outColor %c1
OpStore %ptr_out_v4_pos %mid_plus_arrow
OpEmitVertex
OpStore %ptr_out_v4_pos %mid_minus_arrow
OpEmitVertex
OpStore %ptr_out_v4_pos %p1
OpEmitVertex
OpEndPrimitive
OpReturn
OpFunctionEnd
