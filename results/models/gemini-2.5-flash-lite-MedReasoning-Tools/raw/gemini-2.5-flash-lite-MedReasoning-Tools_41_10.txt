; Get UV.y
%uv_val = OpLoad %vec2 %uv
%uv_y = OpCompositeExtract %float %uv_val 1

; stripes = sin(uv.y * PI * 16.0)
%stripes_arg = OpFMul %float %uv_y %float_PI
%stripes_arg2 = OpFMul %float %stripes_arg %float_16
%stripes = OpExtInst %float %1 Sin %stripes_arg2

; Color calculation
; If stripes > 0, red, else blue
%is_red = OpFOrdGreaterThan %float %stripes %float_0
%red_col = OpCompositeConstruct %vec3 %float_0_9 %float_0_1 %float_0_1
%blue_col = OpCompositeConstruct %vec3 %float_0_1 %float_0_1 %float_0_9

; Lambertian Shading
; lightPos = ubo.lightPos

%wpos = OpLoad %vec3 %worldPos
%lightDir = OpFSub %vec3 %lightPos3 %wpos
%norm = OpLoad %vec3 %normal
%norm_n = OpExtInst %vec3 %1 Normalize %norm
%lightDir_n = OpExtInst %vec3 %1 Normalize %lightDir

%diffuse = OpDot %float %norm_n %lightDir_n
%diffuse_clamped = OpExtInst %float %1 FClamp %diffuse %float_0 %float_1

; shading = 0.4 + 0.6 * diffuse
%shading_term = OpFMul %float %float_0_6 %diffuse_clamped
%shading = OpFAdd %float %float_0_4 %shading_term

%final_rgb = OpVectorTimesScalar %vec3 %base_col %shading
%final_rgba = OpCompositeConstruct %vec4 %final_rgb %float_1
