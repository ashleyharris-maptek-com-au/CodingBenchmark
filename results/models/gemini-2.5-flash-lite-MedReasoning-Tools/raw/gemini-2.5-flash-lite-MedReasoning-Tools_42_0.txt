; SPIR-V Geometry Shader
; Input: InputPoints (1 vertex)
; Output: TriangleStrip (4 vertices)
; Task: Expand each point into a quad (h=0.04)

               OpCapability Geometry
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %inColor %outColor
               OpExecutionMode %main InputPoints
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 4
               OpExecutionMode %main Invocations 1

; Decorations
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %inColor Location 2
               OpDecorate %outColor Location 0

; Types/Constants
%void         = OpTypeVoid
%func         = OpTypeFunction %void
%float        = OpTypeFloat 32
%v3float      = OpTypeVector %float 3
%v4float      = OpTypeVector %float 4
%int          = OpTypeInt 32 1
%uint         = OpTypeInt 32 0
%int_0        = OpConstant %int 0
%float_0      = OpConstant %float 0
%float_1      = OpConstant %float 1
%h            = OpConstant %float 0.04
%neg_h        = OpConstant %float -0.04

; Interface
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in    = OpTypeArray %gl_PerVertex_in 1
%ptr_arr_pv   = OpTypePointer Input %arr_pv_in
%gl_in        = OpVariable %ptr_arr_pv Input

%ptr_v4_in    = OpTypePointer Input %v4float
%inColor_arr  = OpTypeArray %v3float 1
%ptr_inColor  = OpTypePointer Input %inColor_arr
%inColor      = OpVariable %ptr_inColor Input
%ptr_v3_in    = OpTypePointer Input %v3float

%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_pv_out   = OpTypePointer Output %gl_PerVertex_out
%gl_out       = OpVariable %ptr_pv_out Output

%ptr_v3_out   = OpTypePointer Output %v3float
%outColor     = OpVariable %ptr_v3_out Output

%ptr_v4_out   = OpTypePointer Output %v4float

; Main function
%main         = OpFunction %void None %func
%label        = OpLabel

; Load input position (gl_in[0].gl_Position)
%ptr_pos_in   = OpAccessChain %ptr_v4_in %gl_in %int_0 %int_0
%pos          = OpLoad %v4float %ptr_pos_in
%x            = OpCompositeExtract %float %pos 0
%y            = OpCompositeExtract %float %pos 1
%z            = OpCompositeExtract %float %pos 2
%w            = OpCompositeExtract %float %pos 3

; Load color (inColor[0])
%ptr_c_in     = OpAccessChain %ptr_v3_in %inColor %int_0
%color        = OpLoad %v3float %ptr_c_in
OpStore %outColor %color

; Emit 4 vertices for the quad
; Vertex 0: (x-h, y-h)
%x0           = OpFAdd %float %x %neg_h
%y0           = OpFAdd %float %y %neg_h
%v0           = OpCompositeConstruct %v4float %x0 %y0 %z %w
%ptr_pv0      = OpAccessChain %ptr_v4_out %gl_out %int_0
OpStore %ptr_pv0 %v0
OpEmitVertex

; Vertex 1: (x+h, y-h)
%x1           = OpFAdd %float %x %h
%y1           = OpFAdd %float %y %neg_h
%v1           = OpCompositeConstruct %v4float %x1 %y1 %z %w
%ptr_pv1      = OpAccessChain %ptr_v4_out %gl_out %int_0
OpStore %ptr_pv1 %v1
OpEmitVertex

; Vertex 2: (x-h, y+h)
%x2           = OpFAdd %float %x %neg_h
%y2           = OpFAdd %float %y %h
%v2           = OpCompositeConstruct %v4float %x2 %y2 %z %w
%ptr_pv2      = OpAccessChain %ptr_v4_out %gl_out %int_0
OpStore %ptr_pv2 %v2
OpEmitVertex

; Vertex 3: (x+h, y+h)
%x3           = OpFAdd %float %x %h
%y3           = OpFAdd %float %y %h
%v3           = OpCompositeConstruct %v4float %x3 %y3 %z %w
%ptr_pv3      = OpAccessChain %ptr_v4_out %gl_out %int_0
OpStore %ptr_pv3 %v3
OpEmitVertex

OpEndPrimitive
OpReturn
OpFunctionEnd
