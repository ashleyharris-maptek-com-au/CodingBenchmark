; Geometry Shader: Shrink Triangles
; Input: Triangles (size 3)
; Output: TriangleStrip, max 3 vertices

               OpCapability Geometry
               OpCapability Shader
               OpMemoryModel Logical GLSL450
               OpEntryPoint Geometry %main "main" %gl_in %gl_out %inColor %outColor
               OpExecutionMode %main InputTriangles
               OpExecutionMode %main OutputTriangleStrip
               OpExecutionMode %main OutputVertices 3
               OpExecutionMode %main Invocations 1

; Decorate Interface
               OpDecorate %gl_PerVertex_in Block
               OpMemberDecorate %gl_PerVertex_in 0 BuiltIn Position
               OpDecorate %gl_PerVertex_out Block
               OpMemberDecorate %gl_PerVertex_out 0 BuiltIn Position
               OpDecorate %inColor Location 0
               OpDecorate %outColor Location 0

; Types
%void = OpTypeVoid
%float = OpTypeFloat 32
%v4float = OpTypeVector %float 4
%v3float = OpTypeVector %float 3
%uint = OpTypeInt 32 0
%int = OpTypeInt 32 1
%uint_3 = OpConstant %uint 3
%int_0 = OpConstant %int 0
%int_1 = OpConstant %int 1
%int_2 = OpConstant %int 2
%float_0_333333 = OpConstant %float 0.3333333
%float_0_7 = OpConstant %float 0.7

; Input Types
%gl_PerVertex_in = OpTypeStruct %v4float
%arr_pv_in = OpTypeArray %gl_PerVertex_in %uint_3
%ptr_in_arr_pv = OpTypePointer Input %arr_pv_in
%ptr_in_v4 = OpTypePointer Input %v4float
%ptr_in_v3 = OpTypePointer Input %v3float
%arr_inColor = OpTypeArray %v3float %uint_3
%ptr_in_arr_c = OpTypePointer Input %arr_inColor

; Output Types
%gl_PerVertex_out = OpTypeStruct %v4float
%ptr_out_pv = OpTypePointer Output %gl_PerVertex_out
%ptr_out_v4 = OpTypePointer Output %v4float
%ptr_out_v3 = OpTypePointer Output %v3float

; Variables
%gl_in = OpVariable %ptr_in_arr_pv Input
%gl_out = OpVariable %ptr_out_pv Output
%inColor = OpVariable %ptr_in_arr_c Input
%outColor = OpVariable %ptr_out_v3 Output

; Main
%main = OpFunction %void None None
%label = OpLabel

; 1. Get Positions
%ptr_p0 = OpAccessChain %ptr_in_v4 %gl_in %int_0 %int_0
%p0 = OpLoad %v4float %ptr_p0
%ptr_p1 = OpAccessChain %ptr_in_v4 %gl_in %int_1 %int_0
%p1 = OpLoad %v4float %ptr_p1
%ptr_p2 = OpAccessChain %ptr_in_v4 %gl_in %int_2 %int_0
%p2 = OpLoad %v4float %ptr_p2

; 2. Compute Centroid C = (p0 + p1 + p2) / 3
%sum01 = OpFAdd %v4float %p0 %p1
%sum012 = OpFAdd %v4float %sum01 %p2
%C = OpVectorTimesScalar %v4float %sum012 %float_0_333333
%C3 = OpVectorShuffle %v3float %C %C 0 1 2

; 3. Loop: Process and emit 3 vertices
OpBranch %loop_label

%loop_label = OpLabel
%idx = OpPhi %int %int_0 %label %next_idx %loop_label

; Access input
%ptr_in_p = OpAccessChain %ptr_in_v4 %gl_in %idx %int_0
%pos = OpLoad %v4float %ptr_in_p
%ptr_in_c = OpAccessChain %ptr_in_v3 %inColor %idx
%col = OpLoad %v3float %ptr_in_c

; Shrink: new_pos = C + s * (pos - C)
%pos3 = OpVectorShuffle %v3float %pos %pos 0 1 2
%diff = OpFSub %v3float %pos3 %C3
%scaled = OpVectorTimesScalar %v3float %diff %float_0_7
%new_pos3 = OpFAdd %v3float %C3 %scaled

; Convert back to v4
%new_pos_x = OpCompositeExtract %float %new_pos3 0
%new_pos_y = OpCompositeExtract %float %new_pos3 1
%new_pos_z = OpCompositeExtract %float %new_pos3 2
%orig_w = OpCompositeExtract %float %pos 3 ; Keep original w
%new_pos4 = OpCompositeConstruct %v4float %new_pos_x %new_pos_y %new_pos_z %orig_w

; Store Output
%ptr_out_pos = OpAccessChain %ptr_out_v4 %gl_out %int_0
OpStore %ptr_out_pos %new_pos4
OpStore %outColor %col

; Emit
OpEmitVertex

%next_idx = OpIAdd %int %idx %int_1
%cond = OpULessThan %int %next_idx %uint_3
OpBranchConditional %cond %loop_label %end_label

%end_label = OpLabel
OpEndPrimitive
OpReturn
OpFunctionEnd
