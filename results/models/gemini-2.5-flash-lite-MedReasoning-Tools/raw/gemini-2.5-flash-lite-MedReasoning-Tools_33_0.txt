using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

class QAPSolver
{
    static void Main()
    {
        // Read N
        string line = Console.ReadLine();
        if (string.IsNullOrWhiteSpace(line)) return;
        int N = int.Parse(line.Trim());

        int[] flow = new int[N * N];
        int[] dist = new int[N * N];

        // Read flow matrix
        for (int i = 0; i < N; i++)
        {
            string[] parts = Console.ReadLine().Split(new[] { ' ', '	' }, StringSplitOptions.RemoveEmptyEntries);
            for (int j = 0; j < N; j++)
            {
                flow[i * N + j] = int.Parse(parts[j]);
            }
        }

        // Read distance matrix
        for (int i = 0; i < N; i++)
        {
            string[] parts = Console.ReadLine().Split(new[] { ' ', '	' }, StringSplitOptions.RemoveEmptyEntries);
            for (int j = 0; j < N; j++)
            {
                dist[i * N + j] = int.Parse(parts[j]);
            }
        }

        // Initial solution: identity permutation
        int[] p = new int[N];
        for (int i = 0; i < N; i++) p[i] = i;

        // Simulated Annealing
        Random rand = new Random();
        long currentCost = CalculateCost(N, flow, dist, p);
        
        double temp = 1000.0;
        double coolingRate = 0.99995;
        
        DateTime startTime = DateTime.Now;
        while ((DateTime.Now - startTime).TotalSeconds < 280) // 4.6 minutes
        {
            int i = rand.Next(N);
            int j = rand.Next(N);
            if (i == j) continue;

            // Swap
            int tempP = p[i];
            p[i] = p[j];
            p[j] = tempP;

            long newCost = CalculateCost(N, flow, dist, p);
            
            if (newCost < currentCost || Math.Exp((currentCost - newCost) / temp) > rand.NextDouble())
            {
                currentCost = newCost;
            }
            else
            {
                // Backtrack
                tempP = p[i];
                p[i] = p[j];
                p[j] = tempP;
            }
            
            temp *= coolingRate;
        }

        // Output
        Console.WriteLine(currentCost);
        Console.WriteLine(string.Join(" ", p));
    }

    static long CalculateCost(int N, int[] flow, int[] dist, int[] p)
    {
        long cost = 0;
        for (int i = 0; i < N; i++)
        {
            for (int j = 0; j < N; j++)
            {
                cost += (long)flow[i * N + j] * dist[p[i] * N + p[j]];
            }
        }
        return cost;
    }
}
