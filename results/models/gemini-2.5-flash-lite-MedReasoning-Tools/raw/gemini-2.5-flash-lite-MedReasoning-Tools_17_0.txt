using System;
using System.Collections.Generic;
using System.Linq;

namespace BinPacking
{
    public struct Box
    {
        public int Index;
        public int W, H, D;
        public long GetVolume() { return (long)W * H * D; }
    }

    public struct PackedBox
    {
        public int Index;
        public int X, Y, Z;
    }

    public class Packer
    {
        private int _containerW, _containerH, _containerD;
        private List<PackedBox> _packedBoxes = new List<PackedBox>();
        private List<Space> _emptySpaces = new List<Space>();

        private struct Space
        {
            public int X, Y, Z, W, H, D;
        }

        public Packer(int w, int h, int d)
        {
            _containerW = w;
            _containerH = h;
            _containerD = d;
            _emptySpaces.Add(new Space { X = 0, Y = 0, Z = 0, W = w, H = h, D = d });
        }

        public bool TryPack(Box box, out PackedBox packedBox)
        {
            packedBox = default(PackedBox);

            for (int i = 0; i < _emptySpaces.Count; i++)
            {
                var space = _emptySpaces[i];
                if (box.W <= space.W && box.H <= space.H && box.D <= space.D)
                {
                    packedBox = new PackedBox { Index = box.Index, X = space.X, Y = space.Y, Z = space.Z };
                    _packedBoxes.Add(packedBox);

                    // Split the remaining space
                    _emptySpaces.RemoveAt(i);
                    SplitSpace(space, box);
                    return true;
                }
            }
            return false;
        }

        private void SplitSpace(Space space, Box box)
        {
            // Simple space splitting strategy: split into 3 remaining spaces
            if (space.W > box.W)
                _emptySpaces.Add(new Space { X = space.X + box.W, Y = space.Y, Z = space.Z, W = space.W - box.W, H = box.H, D = box.D });
            if (space.H > box.H)
                _emptySpaces.Add(new Space { X = space.X, Y = space.Y + box.H, Z = space.Z, W = space.W, H = space.H - box.H, D = box.D });
            if (space.D > box.D)
                _emptySpaces.Add(new Space { X = space.X, Y = space.Y, Z = space.Z + box.D, W = space.W, H = space.H, D = space.D - box.D });
        }

        public void PrintResult()
        {
            Console.WriteLine(_packedBoxes.Count);
            foreach (var b in _packedBoxes)
            {
                Console.WriteLine(b.Index + " " + b.X + " " + b.Y + " " + b.Z);
            }
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            string line = Console.ReadLine();
            if (string.IsNullOrEmpty(line)) return;
            string[] line1 = line.Split();
            int n = int.Parse(line1[0]);
            int cw = int.Parse(line1[1]);
            int ch = int.Parse(line1[2]);
            int cd = int.Parse(line1[3]);

            List<Box> boxes = new List<Box>();
            for (int i = 0; i < n; i++)
            {
                string bLine = Console.ReadLine();
                if (string.IsNullOrEmpty(bLine)) break;
                string[] b = bLine.Split();
                boxes.Add(new Box { Index = i, W = int.Parse(b[0]), H = int.Parse(b[1]), D = int.Parse(b[2]) });
            }

            // FFD: Sort by volume descending
            var sortedBoxes = boxes.OrderByDescending(b => b.GetVolume()).ToList();

            Packer packer = new Packer(cw, ch, cd);
            PackedBox pb;
            foreach (var box in sortedBoxes)
            {
                packer.TryPack(box, out pb);
            }

            packer.PrintResult();
        }
    }
}
