; SPIR-V
; Version: 1.0
OpCapability Shader
OpMemoryModel Logical GLSL450
OpEntryPoint Fragment %main "main" %worldPos %normal %uv %tangent %color %outColor %ubo
OpExecutionMode %main OriginUpperLeft
OpSource GLSL 450

%1 = OpExtInstImport "GLSL.std.450"

; --- Decorations ---
OpDecorate %outColor Location 0
OpDecorate %worldPos Location 0
OpDecorate %normal Location 1
OpDecorate %uv Location 2
OpDecorate %tangent Location 3
OpDecorate %color Location 4
OpDecorate %ubo DescriptorSet 0
OpDecorate %ubo Binding 0
OpDecorate %uboBlock Block
OpMemberDecorate %uboBlock 0 Offset 0
OpMemberDecorate %uboBlock 1 Offset 64
OpMemberDecorate %uboBlock 2 Offset 128
OpMemberDecorate %uboBlock 3 Offset 192
OpMemberDecorate %uboBlock 4 Offset 208
OpMemberDecorate %uboBlock 5 Offset 224

; --- Types and Constants ---
%void = OpTypeVoid
%func = OpTypeFunction %void
%float = OpTypeFloat 32
%int = OpTypeInt 32 0
%bool = OpTypeBool
%vec3 = OpTypeVector %float 3
%vec4 = OpTypeVector %float 4
%mat4 = OpTypeMatrix %vec4 4
%vec2 = OpTypeVector %float 2
%float_0 = OpConstant %float 0.0
%float_0_5 = OpConstant %float 0.5
%float_1 = OpConstant %float 1.0
%float_2 = OpConstant %float 2.0
%int_4 = OpConstant %int 4

%inputVec3 = OpTypePointer Input %vec3
%inputVec2 = OpTypePointer Input %vec2
%outputVec4 = OpTypePointer Output %vec4

%uboBlock = OpTypeStruct %mat4 %mat4 %mat4 %vec4 %vec4 %vec4
%uboPtr = OpTypePointer Uniform %uboBlock
%uniformVec4 = OpTypePointer Uniform %vec4

; --- Variables ---
%worldPos = OpVariable %inputVec3 Input
%normal = OpVariable %inputVec3 Input
%uv = OpVariable %inputVec2 Input
%tangent = OpVariable %inputVec3 Input
%color = OpVariable %inputVec3 Input
%outColor = OpVariable %outputVec4 Output
%ubo = OpVariable %uboPtr Uniform

; --- Main Function ---
%main = OpFunction %void None %func
%label = OpLabel

%n_raw = OpLoad %vec3 %normal
%n_norm = OpExtInst %vec3 %1 Normalize %n_raw

%wPos = OpLoad %vec3 %worldPos
%camPos_ptr = OpAccessChain %uniformVec4 %ubo %int_4
%camPos_vec4 = OpLoad %vec4 %camPos_ptr
%camPos = OpVectorShuffle %vec3 %camPos_vec4 %camPos_vec4 0 1 2

%v_raw = OpFSub %vec3 %camPos %wPos
%v = OpExtInst %vec3 %1 Normalize %v_raw

%dot = OpDot %float %n_norm %v
%ndotv = OpExtInst %float %1 FMax %dot %float_0

%is_less = OpFOrdLessThan %bool %ndotv %float_0_5

%ndotv_2 = OpFMul %float %ndotv %float_2
%color_less = OpCompositeConstruct %vec4 %float_1 %ndotv_2 %float_0 %float_1

%ndotv_minus_05 = OpFSub %float %ndotv %float_0_5
%factor = OpFMul %float %ndotv_minus_05 %float_2
%rgb_val = OpFSub %float %float_1 %factor
%color_greater = OpCompositeConstruct %vec4 %rgb_val %rgb_val %factor %float_1

%final_color = OpSelect %vec4 %is_less %color_less %color_greater

OpStore %outColor %final_color
OpReturn
OpFunctionEnd
