using System;
using System.Collections.Generic;
using System.Linq;

public class CuttingStockSolver
{
    struct Cut
    {
        public int Length;
        public int OriginalIndex;
    }

    class Stock
    {
        public int RemainingCapacity;
        public List<int> CutIndices = new List<int>();
    }

    public static void Main(string[] args)
    {
        // Read input
        string line1 = Console.ReadLine();
        if (line1 == null) return;
        string[] firstLine = line1.Trim().Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
        int n = int.Parse(firstLine[0]);
        int stockLength = int.Parse(firstLine[1]);

        string line2 = Console.ReadLine();
        if (line2 == null) return;
        string[] secondLine = line2.Trim().Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
        Cut[] cuts = new Cut[n];
        for (int i = 0; i < n; i++)
        {
            cuts[i] = new Cut { Length = int.Parse(secondLine[i]), OriginalIndex = i };
        }

        // Sort cuts in descending order
        Array.Sort(cuts, (a, b) => b.Length.CompareTo(a.Length));

        // First Fit Decreasing
        List<Stock> stocks = new List<Stock>();

        foreach (var cut in cuts)
        {
            bool placed = false;
            // Simple linear search for first fit. For millions of cuts,
            // this can be optimized with a data structure.
            foreach (var stock in stocks)
            {
                if (stock.RemainingCapacity >= cut.Length)
                {
                    stock.RemainingCapacity -= cut.Length;
                    stock.CutIndices.Add(cut.OriginalIndex);
                    placed = true;
                    break;
                }
            }

            if (!placed)
            {
                Stock newStock = new Stock { RemainingCapacity = stockLength - cut.Length };
                newStock.CutIndices.Add(cut.OriginalIndex);
                stocks.Add(newStock);
            }
        }

        // Output results
        Console.WriteLine(stocks.Count);
        foreach (var stock in stocks)
        {
            Console.WriteLine(string.Join(" ", stock.CutIndices));
        }
    }
}
