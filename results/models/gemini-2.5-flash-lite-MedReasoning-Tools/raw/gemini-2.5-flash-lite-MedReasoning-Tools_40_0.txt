#include <iostream>
#include <vector>
#include <numeric>
#include <algorithm>

using namespace std;

// Maximum Weight Independent Set solver
// Maximize sum(c_i * x_i) subject to x_u + x_v <= 1 for all edges (u, v)

int n, m;
vector<long long> c;
vector<vector<int>> adj;
long long best_obj = -1;
vector<int> best_x;
vector<int> current_x;

void solve(int idx, long long current_obj) {
    if (idx == n) {
        if (current_obj > best_obj) {
            best_obj = current_obj;
            best_x = current_x;
        }
        return;
    }

    // Pruning: if current + remaining < best_obj, skip
    long long remaining = 0;
    for (int i = idx; i < n; ++i) if (c[i] > 0) remaining += c[i];
    if (current_obj + remaining <= best_obj) return;

    // Try setting x_idx = 1 (if possible)
    bool can_take = true;
    for (int neighbor : adj[idx]) {
        if (neighbor < idx && current_x[neighbor] == 1) {
            can_take = false;
            break;
        }
    }

    if (can_take) {
        current_x[idx] = 1;
        solve(idx + 1, current_obj + c[idx]);
        current_x[idx] = 0;
    }

    // Try setting x_idx = 0
    solve(idx + 1, current_obj);
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    if (!(cin >> n >> m)) return 0;
    c.resize(n);
    for (int i = 0; i < n; ++i) cin >> c[i];
    
    adj.resize(n);
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    current_x.resize(n, 0);
    solve(0, 0);

    cout << best_obj << endl;
    for (int i = 0; i < n; ++i) {
        cout << best_x[i] << (i == n - 1 ? "" : " ");
    }
    cout << endl;

    return 0;
}
